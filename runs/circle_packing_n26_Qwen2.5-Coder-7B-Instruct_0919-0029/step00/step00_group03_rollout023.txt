import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    eps = 1e-9

    # Seed configuration
    grid = np.array([(i / 5 + 0.1, j / 5 + 0.1) for j in range(5) for i in range(5)])
    centers[:25] = grid
    centers[25] = (0.5, 0.5)

    for shake in range((max_iter // shake_rounds) + 1):
        for iter in range(shake_rounds):
            S_previous = np.sum(radii)
            converged = True
            for i in range(26):
                radii[i] += g
                for j in range(i + 1, 26):
                    d = np.linalg.norm(centers[i] - centers[j])
                    if radii[i] + radii[j] > d:
                        padding = 0.5 * g
                        false_move_radius_i = 0
                        if d - radii[i] - radii[j] > 1e-12:
                            false_move_radius_i = (d - radii[j]) / 2
                        if i != 25:
                            for k in range(25):
                                d_k = np.linalg.norm(centers[i] - centers[k])
                                false_move_radius_i = max(false_move_radius_i, (d_k - radii[k]) / 2)
                        for k in range(4):
                            if k == 0 or k == 1:
                                delta_k = np.array([1 + padding if x == k else 0 for x in [0, 1]])
                            elif k == 2 or k == 3:
                                delta_k = np.array([0 if y == k - 2 else 1 + padding for y in [0, 1]])
                            false_move_radius_i = max(false_move_radius_i, centers[i][k % 2] + padding)
                        false_move_radius_i = min(false_move_radius_i)
                        delta_move = np.array([delta_k for k in range(2)]) * delta
                        centers[i] += delta_move
                        radii[i] = false_move_radius_i
                if radii[i] < 0:
                    radii[i] = 0
                elif radii[i] * 2 > 1:
                    radii[i] = 0.5
                elif radii[i] > centers[i][0] or radii[i] > 1 - centers[i][0] or radii[i] > centers[i][1] or radii[i] > 1 - centers[i][1]:
                    radii[i] = centers[i][0] if centers[i][0] > centers[i][1] else centers[i][1]
                    centers[i] -= centers[i] * 2 * radii[i] / np.linalg.norm(centers[i])
                    radii[i] = centers[i][0] if centers[i][0] > centers[i][1] else centers[i][1]
            if np.sum(radii) - S_previous < 1e-8:
                converged = False
                break
        if converged:
            break

    for i in range(26):
        if radii[i] < 0:
            radii[i] = 0
        elif radii[i] * 2 > 1:
            radii[i] = 0.5
        elif radii[i] > centers[i][0] or radii[i] > 1 - centers[i][0] or radii[i] > centers[i][1] or radii[i] > 1 - centers[i][1]:
            radii[i] = centers[i][0] if centers[i][0] > centers[i][1] else centers[i][1]
            centers[i] -= centers[i] * 2 * radii[i] / np.linalg.norm(centers[i])
            radii[i] = centers[i][0] if centers[i][0] > centers[i][1] else centers[i][1]

    return centers, radii, np.sum(radii)
