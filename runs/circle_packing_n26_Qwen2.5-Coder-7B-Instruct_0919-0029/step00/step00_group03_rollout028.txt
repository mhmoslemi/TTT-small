import numpy as np
from scipy.optimize import linprog
import random

# Geometry preprocessing
def d_wall(p):
    return min(p[0], 1 - p[0], p[1], 1 - p[1])

def d_pair(p1, p2):
    return np.linalg.norm(np.array(p1) - np.array(p2))

# Inner LP problem
def solve_inner_lp(centers):
    n = centers.shape[0]
    c = np.ones(n)
    A1 = -np.eye(n)
    b1 = [d_wall(p) for p in centers]
    A2 = np.zeros((n * (n - 1) // 2, n))
    b2 = []
    for i in range(n):
        for j in range(i + 1, n):
            A2[i * (n - i) // 2 + j - i - 1, i] = 1
            A2[i * (n - i) // 2 + j - i - 1, j] = 1
            b2.append(d_pair(centers[i], centers[j]))
    A = np.vstack((A1, A2))
    b = np.hstack((b1, b2))
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x
