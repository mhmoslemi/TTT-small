import numpy as np
from scipy.optimize import minimize, linprog

def distance(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = distance(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    return -np.sum(r)

def constraint_wall(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    con = []
    for i in range(n):
        con.append(x[i] - r[i])
        con.append(1 - x[i] - r[i])
        con.append(y[i] - r[i])
        con.append(1 - y[i] - r[i])
    return con

def constraint_pairwise(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    con = []
    for i in range(n):
        for j in range(i + 1, n):
            con.append(distance([x[i], y[i]], [x[j], y[j]]) - (r[i] + r[j]))
    return con

def cobyla_phase(z):
    n = 26
    bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n
    con_wall = {'type': 'ineq', 'fun': constraint_wall}
    con_pairwise = {'type': 'ineq', 'fun': constraint_pairwise}
    con = [con_wall, con_pairwise]
    opt_result = minimize(objective, z, method='COBYLA', bounds=bounds, constraints=con, options={'rhobeg': 0.1, 'rhoend': 1e-8, 'maxfun': 20000, 'tol': 1e-7})
    return opt_result

def pso_phase(z_best):
    n = 26
    num_particles = 30
    num_iterations = 2000
    max_velocity = 0.1
    inertia_weight = 0.7
    cognitive_coefficient = 1.5
    social_coefficient = 1.5

    def evaluate_radius(u):
        x = u[:n]
        y = u[n:2*n]
        r = np.zeros(n)
        dist = np.zeros(n * n // 2)
        for i in range(n):
            for j in range(i + 1, n):
                dist[i * n + j] = distance([x[i], y[i]], [x[j], y[j]])
                r[i] = min(x[i], 1 - x[i], y[i], 1 - y[i])
                r[j] = min(x[j], 1 - x[j], y[j], 1 - y[j])
        linprog_result = linprog(-r, A_ineq=-dist.reshape(n, n // 2).T, b_ineq=np.zeros(n // 2), bounds=(0, None))
        if not linprog_result.success:
            return -np.inf
        return -linprog_result.fun

    velocities = np.zeros((num_particles, 2*n))
    positions = np.zeros((num_particles, 2*n))

    for i in range(num_particles):
        if i < 10:
            positions[i] = z_best[np.random.permutation(n)], z_best[np.random.permutation(n)]
            velocities[i] = np.random.uniform(-max_velocity, max_velocity, 2*n)
        elif i < 20:
            positions[i] = np.random.uniform(0, 1, 2*n)
            velocities[i] = np.random.uniform(-max_velocity, max_velocity, 2*n)
        else:
            positions[i] = (z_best[np.random.permutation(n)] + np.random.uniform(-0.05, 0.05, n)).reshape(2*n)
            velocities[i] = np.random.uniform(-max_velocity, max_velocity, 2*n)

    personal_best =.positions.copy()
    best_personal_value = np.array([evaluate_radius(pos) for pos in positions])

    global_best = positions[np.argmax(best_personal_value)]
    global_best_value = best_personal_value.max()

    for t in range(num_iterations):
        for i in range(num_particles):
            velocities[i] = inertia_weight * velocities[i] + cognitive_coefficient * np.random.rand(2*n) * (personal_best[i] - positions[i]) + social_coefficient * np.random.rand(2*n) * (global_best - positions[i])
            for j in range(2*n):
                if velocities[i][j] > max_velocity:
                    velocities[i][j] = max_velocity
                elif velocities[i][j] < -max_velocity:
                    velocities[i][j] = -max_velocity

            positions[i] += velocities[i]
            for j in range(2*n):
                if positions[i][j] < 0:
                    positions[i][j] = 0
                elif positions[i][j] > 1:
                    positions[i][j] = 1

            current_value = evaluate_radius(positions[i])
            if current_value > best_personal_value[i]:
                personal_best[i] = positions[i]
                best_personal_value[i] = current_value

            if current_value > global_best_value:
                global_best = positions[i]
                global_best_value = current_value

    return global_best

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_grid = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    initial_radii = np.array([np.min([0.1 + 0.2 * i, 1 - 0.1 - 0.2 * i, 0.1 + 0.2 * j, 1 - 0.1 - 0.2 * j]) for i, j in initial_grid, initial_grid])
    z_initial = np.hstack((initial_grid.flatten(), initial_radii))
    cobyla_result = cobyla_phase(z_initial)
    z_best = cobyla_result.x
    pso_result = pso_phase(z_best)
    z_final = pso_result.copy()
    x = z_final[:n]
    y = z_final[n:2*n]
    r = z_final[2*n:]
    linprog_result = linprog(-r, A_ineq=-np.array([distance([x[i], y[i]], [x[j], y[j]]) for i in range(n) for j in range(i + 1, n)]).reshape(n, n // 2).T, b_ineq=np.zeros(n // 2), bounds=(0, None))
    if not linprog_result.success:
        r *= 0.9999999
        r /= np.sum(r)
    centers = np.vstack((x, y)).T
    radii = r
    sum_radii = np.sum(radii)
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        radii[radii < 0] = 0
        radii *= 0.9999999
        sum_radii = np.sum(radii)
    return centers, radii, sum_radii
