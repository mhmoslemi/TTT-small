from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    hex_grid = np.array([[0.1 + 0.2 * col, 0.1 + 0.2 * row] for row in range(5) for col in range(5)] + [[0.5, 0.5]])
    radii_guess = 0.1 * np.ones(n)

    def objective(z):
        return -np.sum(z[n:])

    def constraint_boundary(z):
        return np.min(-z[:n*p])
    
    def pairwise_constraints(z):
        p = int(len(z) / (n * 2))
        A = np.zeros((p * n, p * n))
        b = np.zeros(p * n)
        for i in range(n):
            for j in range(i + 1, n):
                A[i * p + j, i * p] = 10
                A[i * p + j, j * p] = 10
                b[i * p + j] = 1
        return A, b

    constraints = []
    for i in range(n * 2):
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[i] - hex_grid[i // 2, i % 2]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[i] - hex_grid[i // 2, i % 2]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[i] - hex_grid[i // 2, i % 2]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[i] - hex_grid[i // 2, i % 2]})
    A, b = pairwise_constraints(radii_guess[:n*2])

    b_updated = np.hstack((b, np.zeros(n * 2)))
