import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = squareform(pdist(centers, 'euclidean'))
    
    c = -np.ones(n)
    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = borders[i]
        idx += 1
        
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = distances[i, j]
            idx += 1
            
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    radii = result.x.clip(0)
    sum_radii = np.sum(radii)
    
    return radii, sum_radii


def random_move(centers, step):
    n = centers.shape[0]
    i = np.random.randint(n)
    new_pos = centers[i] + np.random.normal(scale=step, size=2)
    new_pos = np.clip(new_pos, 0, 1)
    return (i, new_pos)


def simulated_annealing_centers(rng, max_iter=30000):
    n = 26
    centers = rng.random((n, 2))
    best = np.inf
    
    for _ in range(20):
        current = centers.copy()
        current_radii, current_sum = solve_lp(current)
        if current_sum < best:
            best = current_sum
            best_centers = current.copy()
        
        step = 0.05
        temperature = 0.02
        T_end = 1e-5
        
        for _ in range(max_iter):
            idx, new_pos = random_move(current, step)
            current[idx] = new_pos
            new_radii, new_sum = solve_lp(current)
            delta = new_sum - current_sum
            if delta > 0 or np.random.rand() < np.exp(delta / temperature):
                current_sum = new_sum
            temperature = T_end + (T_end - T_end) * (_ / max_iter)

    return best_centers, best, best + 0


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    centers, sum_radii = simulated_annealing_centers(rng)
    radii, _ = solve_lp(centers)
    
    return centers, radii, sum_radii
