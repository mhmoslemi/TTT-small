import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    centers = np.array(centers)
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)
    A_ub = np.zeros((n + n*(n-1)//2, n))
    A_eq = np.zeros((0, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    b_eq = np.zeros(0)
    count = 0
    for i in range(n):
        A_ub[count, i] = 1
        b_ub[count] = borders[i]
        count += 1
    triu_idx = np.triu_indices(n, k=1)
    for i, j in zip(triu_idx[0], triu_idx[1]):
        A_ub[count] = np.zeros(n)
        A_ub[count, i] = 1
        A_ub[count, j] = 1
        b_ub[count] = distances[i, j]
        count += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    radii = result.x
    radii[radii < 0] = 0
    return radii, np.sum(radii)

def simulated_annealing_centers(n, max_iter, rng):
    centers = rng.uniform(0, 1, size=(n, 2))
    step = 0.05
    best_frame = centers.copy()
    best_radii, best_sum = solve_lp(centers)
    temperature = 0.02
    temperature_final = 1e-5
    for _ in range(max_iter):
        i = rng.integers(n)
        new_pos = centers[i] + step * (np.random.randn(2) * 2 - 1)
        new_pos = np.clip(new_pos, 0, 1)
        new_frame = centers.copy()
        new_frame[i] = new_pos
        new_radii, new_sum = solve_lp(new_frame)
        delta = new_sum - best_sum
        accept = delta > 0 or np.exp(delta / temperature) > rng.rand()
        if accept:
            centers = new_frame.copy()
            best_frame = centers.copy()
            best_radii, best_sum = new_radii, new_sum
        if temperature > temperature_final:
            temperature *= 0.99995
    return best_frame, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    n = 26
    max_iter = 30000
    centers, radii, sum_radii = simulated_annealing_centers(n, max_iter, rng)
    return centers, radii, sum_radii
