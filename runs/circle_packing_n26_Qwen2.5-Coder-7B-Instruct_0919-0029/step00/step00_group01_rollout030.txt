import numpy as np
from scipy.optimize import linprog
from scipy.spatial import Voronoi, voronoi_plot_2d

def lp_optimize(radii, contacts, n):
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    A = np.zeros((n, n))
    b = np.zeros(n)

    for i in range(n):
        A[i, i] = 1
        b[i] = 1 - max(radii[i], 0)
        A_eq[i, i] = -1
        b_eq[i] = 0

    result = linprog(radii, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    return result.x

    vor = Voronoi(circles)
    incenters = np.array([vor.points[0], vor.points[1], vor.points[2], vor.points[3]])
    return incenters
