import numpy as np
from scipy.spatial import distance_matrix

def build_random_feasible(n_circles, r0):
    centers = []
    radii = [r0] * n_circles
    while len(centers) < n_circles:
        c = np.random.rand(2)
        if all(all((c - c0)**2 >= (r0 + r0) ** 2 for c0, r0 in zip(centers, radii))):
            centers.append(c)
    centers = np.array(centers)
    return centers, radii

def simulated_annealing(centers, radii, initial_sum):
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
   cool = 0.9995
    T0 = 1.0
    n_shift = 1000
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = initial_sum
    
    for k in range(n_iter):
        T = T0 * cool**k
        i = np.random.randint(len(centers))
        if np.random.rand() < move_prob:
            dx, dy = np.random.uniform(-shift_max, shift_max, 2)
            new_center = centers[i] + np.array([dx, dy])
            new_center = np.clip(new_center, a_min=radii[i], a_max=1-radii[i])
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_radius = np.clip(radii[i] + dr, 0, 1-min(new_center[0], 1-new_center[0], new_center[1], 1-new_center[1]))
            new_center = centers[i]
        
        distances = distance_matrix([new_center], centers)
        feasible = np.all(distances + radii >= (radii + new_radius) - 1e-12)
        if feasible:
            new_sum = new_radius + np.array(radii)
            if new_sum >= best_sum:
                best_sum = new_sum
                best_centers, best_radii = new_center, new_radius
            elif np.random.rand() < np.exp((new_sum - initial_sum) / T):
                centers[i], radii[i] = new_center, new_radius
                initial_sum = new_sum
    
    return best_centers, best_radii, best_sum

def deterministic_polish(centers, radii):
    n_shift = 1000
    shift_epsilon = 1e-4
    growth_epsilon = 1e-4
    
    for _ in range(n_shift):  # Growth passes
        for i in range(len(centers)):
            while True:
                r_try = radii[i] + growth_epsilon
                distances = np.linalg.norm(centers[i] - centers, axis=1)
                feasible = np.all(distances >= (radii + r_try) - growth_epsilon)
                if not feasible:
                    break
                radii[i] = r_try
    
    for _ in range(5):  # Shift passes
        for _ in range(len(centers)):
            i = np.random.randint(len(centers))
            d = np.random.choice([-shift_epsilon, shift_epsilon], 2)
            new_center = centers[i] + d
            new_center = np.clip(new_center, a_min=0, a_max=1)
            while True:
                distances = np.linalg.norm(new_center - centers, axis=1)
                feasible = np.all(distances >= (radii + radii[i]) - shift_epsilon)
                if not feasible:
                    break
                new_center -= d * (0.5 * (1 + np.random.rand()))  # Smaller random step
                new_center = np.clip(new_center, a_min=0, a_max=1)
            centers[i] = new_center
    
    return centers, radii
