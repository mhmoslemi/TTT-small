import numpy as np
from scipy.optimize import linprog, minimize
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(z, centers, penalty_weight, tolerance):
    n = len(centers)
    radii = z[:n]
    remainders = z[n:]
    r_i = radii.reshape(-1, 1)
    pair_penalties = np.clip(r_i + r_i.T - np.linalg.norm(centers - centers.T, axis=2), 0, None)**2
    boundary_penalties = np.maximum(0, r_i - centers)**2 + np.maximum(0, r_i - (1 - centers)**2)**2
    return -radii.sum() + penalty_weight * (pair_penalties.sum() + boundary_penalties.sum())

def penalty(z, centers, tolerance):
    n = len(centers)
    radii = z[:n]
    remainders = z[n:]
    r_i = radii.reshape(-1, 1)
    pair_penalties = np.clip(r_i + r_i.T - np.linalg.norm(centers - centers.T, axis=2), 0, None)**2
    boundary_penalties = np.maximum(0, r_i - centers)**2 + np.maximum(0, r_i - (1 - centers)**2)**2
    return pair_penalties.sum() + boundary_penalties.sum()

def validate_move(z, centers, move_index, delta_x, delta_y, delta_r, tolerance):
    n = len(centers)
    radii = z[:n]
    proposed_centers = np.array(centers)
    proposed_centers[move_index] += np.array((delta_x, delta_y))
    proposed_radii = np.array(radii)
    proposed_radii[move_index] += delta_r
    proposed_radii = np.clip(proposed_radii, 0, 1)
    if not (0 <= proposed_centers[:, 0] <= 1 and 0 <= proposed_centers[:, 1] <= 1).all():
        return False
    proposed_penalty = penalty(proposed_centers.flatten(), proposed_centers, tolerance)
    return proposed_penalty < penalty(z, centers, tolerance)

def simulated_annealing(centers, penalty_weight, tolerance, iterations, initial_temperature):
    n = len(centers)
    n_variables = n * 2
    initial_radii = np.zeros(n)
    initial_remainders = np.zeros(n)
    initial_state = np.concatenate((initial_radii, initial_remainders))
    best_state = initial_state
    best_energy = objective(initial_state, centers, penalty_weight, tolerance)
    current_state = initial_state
    current_energy = initial_energy
    temperature = initial_temperature
    temperature_drop_rate = 0.001
    for step in range(iterations):
        move_index = random.randint(0, n - 1)
        delta_x = locals()[f'delta_x_{move_index}' if move_index > 0 else 'delta_x']
        delta_y = locals()[f'delta_y_{move_index}' if move_index > 0 else 'delta_y']
        delta_r = locals()[f'delta_r_{move_index}' if move_index > 0 else 'delta_r']
        if validate_move(current_state, centers, move_index, delta_x, delta_y, delta_r, tolerance):
            new_state = current_state.copy()
            new_state[move_index * 2] += delta_x
            new_state[move_index * 2 + 1] += delta_y
            new_state[n + move_index] += delta_r
            new_state = np.clip(new_state, 0, 1)
            new_energy = objective(new_state, centers, penalty_weight, tolerance)
            energy_difference = new_energy - current_energy
            if energy_difference <= 0 or random.random() < np.exp(-energy_difference / temperature):
                current_state = new_state
                current_energy = new_energy
                if current_energy < best_energy:
                    best_state = current_state
                    best_energy = current_energy
        temperature *= 1 - temperature_drop_rate
    return best_state, best_energy

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_radii = np.zeros(n)
    initial_remainders = np.zeros(n)
    centers = np.random.rand(n, 2)  # Uniform random initialisation
    penalty_weight = 5e4
    tolerance = 1e-12
    iterations = 20000
    initial_temperature = 0.5
    best_state, best_energy = simulated_annealing(centers, penalty_weight, tolerance, iterations, initial_temperature)
    optimized_radii = best_state[:n]
    constrained_radii = np.clip(optimized_radii, 0, 1)
    lp_result = linprog(
        -constrained_radii,
        A_ub=np.concatenate((np.ones((n, n)), np.eye(n), -np.eye(n), -np.diag(2 * (1 - centers[:, 0])), -np.diag(2 * centers[:, 0]), -np.diag(2 * (1 - centers[:, 1])), -np.diag(2 * centers[:, 1])), axis=1),
        b_ub=np.concatenate((np.ones(n) * (1 - 1e-12), 
                             centers[:, 0], 
                             1 - centers[:, 0], 
                             centers[:, 1], 
                             1 - centers[:, 1])),
        bounds=[(0, 1) for _ in range(2 * n)],
        method='highs'
    )
    lp_optimal_radii = lp_result.x[:n]
    lp_optimal_radii = np.clip(lp_optimal_radii, 0, 1)
    return centers, lp_optimal_radii, -lp_result.fun

centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
