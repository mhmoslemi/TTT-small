import numpy as np
from scipy.optimize import linprog

def validate_packing_inner(centers, radii):
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(centers.shape[0]):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        if (x < 0 or x > 1 or y < 0 or y > 1):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def validate_packing(centers, radii):
   , message = validate_packing_inner(centers, radii)
    return message == "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dim = 2
    bounds = [(0, 1), (0, 1), (0, 1)]
    sigma = np.array([0.04, 0.04, 0.01])
    alpha = 0.9995
    T0 = 0.5
    N = 20000
    restarts = 18
    lambda_val = 50000
    p = 2
    epsilon = 1e-12

    def energy(z):
        r = z[2::3]
        x, y = z[:26:3], z[1::3]
        d = np.sqrt(np.sum((x[:, None] - x[None, :]) ** 2 + (y[:, None] - y[None, :]) ** 2, axis=2))
        V = np.nan_to_num((r + r[:, None] - d) ** p, nan=0)
        V[~np.isfinite(V)] = 0
        B = np.maximum(x - r, 0) + np.maximum(x - (1 - r), 0) + np.maximum(y - r, 0) + np.maximum(y - (1 - r), 0)
        return -np.sum(r) + lambda_val * (np.sum(V) + np.sum(B))

    def propose_move(z):
        i = np.random.randint(n)
        delta = np.random.normal(0, sigma[0] * T, dim)
        delta[2:] *= lambda_val * sigma[2] * T
        delta = np.clip(delta, -1.1, 1.1)
        new_z = z.copy()
        new_z[i * dim:(i + 1) * dim] += delta
        return new_z

    def acceptance_probability(new_energy, old_energy, T):
        return min(1, np.exp((old_energy - new_energy) / T))

    def restart():
        z = np.random.uniform(0, 1, dim * n + 3 * n)
        r = z[2::3]
        return z, r, np.sum(r)

    best_energy = float('-inf')
    best_centers = None
    best_radii = None

    for _ in range(restarts):
        z, _, _ = restart()
        T = T0
        for k in range(N):
            new_z = propose_move(z)
            new_energy = energy(new_z)
            accept = acceptance_probability(new_energy, energy(z), T)
            if new_energy <= energy(z) or np.random.rand() < accept:
                z = new_z
            T *= alpha

            if new_energy > best_energy:
                best_energy = new_energy
                best_centers = z[:dim * n].reshape(n, dim)
                best_radii = z[dim * n:].reshape(n,)

    if not validate_packing(best_centers, best_radii):
        raise Exception("Optimal packing not valid")

    x, y = best_centers[:, 0], best_centers[:, 1]
    r = np.maximum(best_radii.reshape(n,), 0)
    x, y, r = x.reshape(-1, 1), y.reshape(-1, 1), r.reshape(-1, 1)
    centers = np.hstack((x, y))
    XS = x.flatten()

    if np.any(XS <= epsilon) or np.any(XS >= 1 - epsilon):
        raise Exception("Optimal packing has center off bounds")

    A = np.zeros((n * n, n * 3))
    for i in range(n):
        for j in range(n):
            if i != j:
                A[i * n + j] = np.array([1.0, 1.0, -2 * (x[i] - x[j])])
                A[i * n + j, 3 * i + 2] = -1.0
                A[i * n + j, 3 * j] = 1.0

    b = np.zeros(n * n)

    G = np.zeros((2 * n, n * 3))
    for i in range(n):
        G[i * 2: (i + 1) * 2, 3 * i: (i + 1) * 3] = np.eye(3)
        G[i * 2: (i + 1) * 2, 3 * i + 2] = - np.ones((2, 1))

    h = np.zeros(2 * n)
    h[n:] = np.ones(n) - epsilon

    res = linprog(-np.ones(n), A_ub=A, b_ub=b, A_eq=None, b_eq=None, bounds=bounds, method='highs')

    if not res.success:
        raise Exception("Linear Programming failed")

    radii = np.maximum(res.x.reshape(n,), 0)
    return best_centers, radii, -res.fun

centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
print("Centers:", centers)
print("Radii:", radii)
