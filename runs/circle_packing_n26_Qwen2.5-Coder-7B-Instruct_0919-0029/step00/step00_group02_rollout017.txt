import numpy as np
import scipy.spatial
from scipy.optimize import minimize

def build_random_feasible(n, r0):
    centers = []
    while len(centers) < n:
        c = np.random.rand(2)
        if all(np.sqrt(np.sum((c - ci)**2)) >= r0 + ri for ci, ri in centers):
            centers.append((c, r0))
    return np.array([c for c, _ in centers]), np.array([r for _, r in centers])

def feasible(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        if not ((radii[i] >= 0) and 
                (0 <= centers[i][0] - radii[i]) and 
                (centers[i][0] + radii[i] <= 1) and 
                (0 <= centers[i][1] - radii[i]) and 
                (centers[i][1] + radii[i] <= 1)):
            return False
        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            rsum = radii[i] + radii[j]
            if dx * dx + dy * dy < rsum * rsum:
                return False
    return True

def objective(x):
    centers, radii = x.reshape(((26, 2), (26,)))
    return -radii.sum()

def simulated_annealing(centers, radii, sum_radii):
    n_iter = 200000
    T = 1.0
    cool = 0.9995
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    n_shift = 1000
    step_size_dec = 0.95

    kd_tree = scipy.spatial.KDTree(centers)

    best_centers = centers
    best_radii = radii
    best_sum = sum_radii

    for k in range(n_iter):
        if k % n_shift == 0:
            shift_max *= step_size_dec
            radius_max *= step_size_dec

        i = np.random.randint(26)
        if np.random.rand() < move_prob:
            dx, dy = np.random.uniform(-shift_max, shift_max, size=2)
            new_pos = np.clip(centers[i] + [dx, dy], radii[i], 1 - radii[i])
            new_r = radii[i]
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_r = np.clip(radii[i] + dr, 0, min(centers[i][0], 1 - centers[i][0],
                                                 centers[i][1], 1 - centers[i][1]))
            new_pos = centers[i]

        new_centers = np.where(best_centers == centers[i], new_pos, best_centers)
        new_radii = np.where(best_radii == radii[i], new_r, best_radii)
        new_sum = new_radii.sum()

        if feasible(new_centers, new_radii):
            if new_sum >= best_sum:
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centers
                    best_radii = new_radii
                T *= cool
            else:
                if np.random.rand() < np.exp((new_sum - best_sum) / T):
                    best_sum = new_sum
                    best_centers = new_centers
                    best_radii = new_radii
                    T *= cool

    return best_centers, best_radii, best_sum

def deterministic_polish(centers, radii):
    improve = True
    while improve:
        improve = False
        for i in range(26):
            r_max = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            while r_max > radii[i]:
                r_test = radii[i] + 1e-4
                if np.all(np.sqrt(np.sum((centers[i] - centers[:, :2])**2, axis=1)) >= np.maximum(radii, r_test)) and \
                   np.all(np.sqrt(np.sum((centers[i] - centers[j][:2])**2)) >= r_test + radii[j] for j in range(26) if j != i):
                    radii[i] = r_test
                    improve = True

    return centers, radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    for i in range(n):
        if radii[i] < 0:
            raise ValueError(f"Circle {i} has negative radius {radii[i]}")

        if not ((0 <= centers[i][0] - radii[i]) and 
                (centers[i][0] + radii[i] <= 1) and 
                (0 <= centers[i][1] - radii[i]) and 
                (centers[i][1] + radii[i] <= 1)):
            raise ValueError(f"Circle {i} at ({centers[i][0]},{centers[i][1]}) r={radii[i]} outside unit square")

        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            rsum = radii[i] + radii[j]
            if dx * dx + dy * dy < rsum * rsum:
                raise ValueError(f"Circles {i} and {j} overlap")

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii = build_random_feasible(26, r0=0.05)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()

    centers, radii, best_sum = simulated_annealing(
        centers, radii, best_sum
    )

    centers, radii = deterministic_polish(centers, radii)

    try:
        validate_packing(centers, radii)
    except ValueError as e:
        raise ValueError(f"Packing validation failed: {e}")

    return centers, radii, best_sum
