import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)
    
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append([-1 if j == i else 0 for j in range(n)])
        b_ub.append(borders[i])
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([-1 if k == i else 0 for k in range(n)] + [-1 if k == j else 0 for k in range(n)])
            b_ub.append(distances[i, j])
    
    A_ub, b_ub = np.array(A_ub), np.array(b_ub)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=bounds)
    radii = -result.x
    return radii, np.sum(radii)

def simulated_annealing_centers(n, max_iter, rng):
    step = 0.05
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for _ in range(max_iter):
        centers = np.clip(rng.normal(0.5, step, (n, 2)), 0, 1)
        radii, sum_radii = solve_lp(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = radii
    
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 30000
    rng = np.random.default_rng(42)
    
    best_centers, best_radii, best_sum = simulated_annealing_centers(n, max_iter, rng)
    
    return best_centers, best_radii, best_sum
