import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centres):
    n = centres.shape[0]
    A = np.zeros((325 + 4 * n, n))
    b = np.zeros(325 + 4 * n)
    c = -np.ones(n)
    
    for i in range(n):
        A[i, i] = 1
    
    row = n
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dist - 1e-12
            row += 1
    
    for i in range(n):
        for j in range(4):
            if j == 0:
                A[row, i] = 1
                b[row] = centres[i, 0] - 1e-12
            elif j == 1:
                A[row, i] = -1
                b[row] = 1 - centres[i, 0] - 1e-12
            elif j == 2:
                A[row, i] = 1
                b[row] = centres[i, 1] - 1e-12
            elif j == 3:
                A[row, i] = -1
                b[row] = 1 - centres[i, 1] - 1e-12
            row += 1
            
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return result.x, -result.fun

def run_packing():
    NUM_STARS = 30
    NUM_ITERS = 2000
    step = 0.05
    step_min = 0.01
    decay_factor = 0.995
    
    best_sum = 0
    best_centres = np.zeros((26, 2))
    
    for start in range(NUM_STARS):
        centres = np.random.rand(26, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(0, 26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            
            new_centres = centres.copy()
            new_centres[idx] = new_c
            
            new_radii, new_sum = lp_optimum(new_centres)
            
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
    
        step = max(step_min, step * decay_factor)
    
    final_radii, final_sum = lp_optimum(best_centres)
    return best_centres, final_radii, final_sum
