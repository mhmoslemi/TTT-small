import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import Delaunay, Voronoi, voronoi_plot_2d
import matplotlib.pyplot as plt

def lp_radii(centers):
    n = centers.shape[0]
    A = np.empty((n * (n - 1) // 2 + 2 * n, 2 * n))
    b = np.empty(n * (n - 1) // 2 + 2 * n)
    A[:n, range(0, 2 * n, 2)] = 1
    A[:n, range(1, 2 * n, 2)] = 0
    b[:n] = centers[:, 0]
    A[n:n + n, range(0, 2 * n, 2)] = 0
    A[n:n + n, range(1, 2 * n, 2)] = 1
    b[n:n + n] = 1 - centers[:, 1]
    for i in range(0, n * (n - 1) // 2):
        A[n + n + i, 2 * i] = 1
        A[n + n + i, 2 * i + 1] = 0
        A[n + n + i, 2 * (i // (n - 1) + 1) % n] = -1
        A[n + n + i, 2 * ((i // (n - 1) + 1) % n) + 1] = 0
        b[n + n + i] = centers[i // (n - 1), 1]
    for i in range(0, n * (n - 1) // 2):
        A[n + n + i + n * (n - 1) // 2, 2 * i] = 0
        A[n + n + i + n * (n - 1) // 2, 2 * i + 1] = 1
        A[n + n + i + n * (n - 1) // 2, 2 * ((i // (n - 1) + 1) % n)] = 0
        A[n + n + i + n * (n - 1) // 2, 2 * ((i // (n - 1) + 1) % n) + 1] = -1
        b[n + n + i + n * (n - 1) // 2] = 1 - centers[i // (n - 1), 0]
    result = linprog([1] * (2 * n), A_ub=A, b_ub=b, bounds=(0, 1))
    return result.x[:n]

def gradient_based_optimization():
    np.random.seed(0)
    centers = np.random.rand(26, 2)
    best_centers = centers.copy()
    best_sum = 0.0
    max_iter = 200
    tol = 1e-5
    n_points = 53
    ε = 1e-4
    α = 0.01
    
    for iter in range(max_iter):
        radii = lp_radii(centers)
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_centers = centers.copy()
            best_sum = current_sum
        
        for i in range(26):
            non_overlapping_positions = centers.copy()
            non_overlapping_positions[i] += ε * np.array([1, 0])
            non_overlapping_positions[centers[:, 0] < 0] += ε * np.array([1, 0])
            non_overlapping_positions[centers[:, 0] > 1] -= ε * np.array([1, 0])
            r_plus = lp_radii(non_overlapping_positions)
            non_overlapping_positions[i] -= 2 * ε * np.array([1, 0])
            r_minus = lp_radii(non_overlapping_positions)
            grad_x = (r_plus[0:len(r_plus)-2] - r_minus[0:len(r_minus)-2]) / (2 * ε)
            
            non_overlapping_positions[i] = centers[i] + ε * np.array([0, 1])
            non_overlapping_positions[centers[:, 1] < 0] += ε * np.array([0, 1])
            non_overlapping_positions[centers[:, 1] > 1] -= ε * np.array([0, 1])
            r_plus = lp_radii(non_overlapping_positions)
            non_overlapping_positions[i] -= 2 * ε * np.array([0, 1])
            r_minus = lp_radii(non_overlapping_positions)
            grad_y = (r_plus[0:len(r_plus)-2] - r_minus[0:len(r_minus)-2]) / (2 * ε)
            
            grad = np.array([grad_x, grad_y]).T
            centers[i] += α * grad[i]
            centers[i, centers[i] < 0] = 0
            centers[i, centers[i] > 1] = 1
        
        if iter > 100 and abs(current_sum - best_sum) < tol:
            break
    
    centers = best_centers
    while True:
        best_sum = np.sum(lp_radii(centers))
        for k in range(10):
            for i in range(26):
                new_radius = lp_radii(np.insert(centers[:i], i, centers[i] + 1e-3, axis=0, obj=-1)[np.insert(radii, i, 0, axis=0)])
                if new_radius > radii[i] and new_radius < 1 - centers[i][0] and new_radius < 1 - centers[i][1]:
                    centers[i, 0] += 1e-3
                    centers[i, 1] += 1e-3
            radii = lp_radii(centers)
            current_sum = np.sum(radii)
            if current_sum <= best_sum:
                break
    
    return centers, lp_radii(centers), np.sum(lp_radii(centers))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = gradient_based_optimization()
    return centers, radii, sum_radii
