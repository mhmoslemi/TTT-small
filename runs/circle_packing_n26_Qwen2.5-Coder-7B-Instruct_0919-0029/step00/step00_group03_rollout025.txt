import numpy as np
from scipy.optimize import minimize, linprog
from sklearn.decomposition import PCA

def expand_grid():
    xs = np.arange(0.1, 1, 0.2)
    ys = np.arange(0.1, 1, 0.2)
    centers = np.array([(x, y) for y in ys for x in xs])
    centers = np.vstack((centers, (0.5, 0.5)))
    return centers

def initial_radii(centers):
    wall_dist = np.min(np.concatenate((centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])))
    return np.full(centers.shape[0], wall_dist)

def objective(z):
    radii = z[52:]
    return -np.sum(radii)

def by_center_constraints(z, centers):
    radii = z[52:]
    constraints = []
    for i in range(len(radii)):
        constraints.extend([centers[i, 0] - radii[i],
                           1 - centers[i, 0] - radii[i],
                           centers[i, 1] - radii[i],
                           1 - centers[i, 1] - radii[i]])
    return constraints

def pairwise_constraints(z, centers):
    radii = z[52:]
    constraints = []
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist - (radii[i] + radii[j]))
    return constraints

def cobyla_refinement(centers):
    n = centers.shape[0]
    initial_guess = np.ravel(centers, order='F')
    initial_guess = np.concatenate((initial_guess, initial_radii(centers)))
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    constraints = list(by_center_constraints(initial_guess, centers))
    constraints.extend(pairwise_constraints(initial_guess, centers))
    result = minimize(objective, initial_guess, method='COBYLA', bounds=bounds, constraints=constraints)
    return result.x
