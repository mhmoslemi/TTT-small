import numpy as np
from scipy.optimize import cobyla, linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial grid placement
    for i in range(5):
        for j in range(5):
            centers[i*5 + j] = np.array([0.1 + 0.2*i, 0.1 + 0.2*j])
    centers[25] = np.array([0.5, 0.5])

    # Initial radius guess
    radii = np.minimum(centers[:,0], 1 - centers[:,0]).min()
    radii = np.minimum(radii, centers[:,1], 1 - centers[:,1]).min()
    
    # Copy radii to a single array
    z = np.hstack((centers.flatten(), radii))
    
    # Objective function (negative because COBYLA minimizes)
    def objective(z):
        return -np.sum(z[-n:])
    
    # Constraints
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2*i] - z[-n+i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[2*i] - z[-n+i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2*i+1] - z[-n+i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[2*i+1] - z[-n+i]})
    
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: 
                          np.linalg.norm(z[2*i:2*i+2] - z[2*j:2*j+2]) - (z[-n+i] + z[-n+j])})
    
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    
    # COBYLA optimization
    result = cobyla.objective_function_calls()
    result = cobyla.minimize(objective, z, cons=cons, bounds=bounds, rhobeg=0.1, rhoend=1e-8, maxfun=20000)
    
    if not result.success:
        return centers, radii, np.sum(radii)
    
    centers = result.x[:2*n].reshape((n, 2))
    radii = result.x[2*n:]
    
    # PSO optimization for center positions
    N = 30
    T = 2000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    velocity = np.zeros((N, 2*n))
    position = np.random.rand(N, 2*n)
    personal_best = np.copy(position)
    global_best = personal_best[0]
    
    def calculate_r_is(position):
        u = position.reshape(-1)
        dists = np.linalg.norm(u[2*i:2*i+2] - u[2*j:2*j+2] for i in range(n) for j in range(i+1, n))
        wall = np.minimum(u[2*i], 1 - u[2*i]).min()
        r_is = np.minimum(wall, dists / 2)
        return r_is
    
    for _ in range(T):
        for i in range(N):
            r_is = calculate_r_is(position)
            personal_best[i] = position[i] if r_is.sum() > calculate_r_is(personal_best[i]).sum() else personal_best[i]
            global_best = personal_best[i] if r_is.sum() > calculate_r_is(global_best).sum() else global_best
        
        for i in range(N):
            v = w * velocity[i] + c1 * np.random.rand(2*n) * (personal_best[i] - position[i]) + c2 * np.random.rand(2*n) * (global_best - position[i])
            v = np.clip(v, -v_max, v_max)
            position[i] = position[i] + v
            position[i] = np.clip(position[i], 0, 1)
    
    centers = position[np.argmax(calculate_r_is(position))].reshape((n, 2))
    radii = calculate_r_is(position[np.argmax(calculate_r_is(position))])
    
    # Validation and scaling
    is_valid, _ = validate_packing(centers, radii)
    if not is_valid:
        radii[radii < 0] = 0
        radii = radii * 0.9999999
        is_valid, _ = validate_packing(centers, radii)
    
    return centers, radii, np.sum(radii)
