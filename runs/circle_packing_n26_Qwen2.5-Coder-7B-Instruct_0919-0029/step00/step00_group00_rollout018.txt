import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> (np.ndarray, np.ndarray, float):
    NUM_STARTS = 25
    INIT_R = 0.01
    SIM_ITERS = 2500
    DAMPING = 0.99
    EPSILON = 1e-12

    best_centres = None
    best_radii = None
    best_sum = -np.inf

    def solve_lp_for_centres(centres):
        n = len(centres)
        c = -np.ones(n)
        A_eq = np.zeros((n + 2*n, n))
        b_eq = np.zeros(n + 2*n)
        
        # Pairwise distance constraints
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centres[i] - centres[j])
                A_eq[j+n][i] = 1
                A_eq[j+n][j] = 1
                b_eq[j+n] = dist - 2 * EPSILON
        
        # Boundary constraints
        for i in range(n):
            x, y = centres[i]
            A_eq[2*n + i][i] = 1
            b_eq[2*n + i] = x - EPSILON
            A_eq[2*n + n + i][i] = 1
            b_eq[2*n + n + i] = 1 - x - EPSILON
            A_eq[2*n + 2*n + i][i] = 1
            b_eq[2*n + 2*n + i] = y - EPSILON
            A_eq[2*n + 3*n + i][i] = 1
            b_eq[2*n + 3*n + i] = 1 - y - EPSILON
        
        r_bounds = (0, None)
        
        result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=r_bounds, method='highs')
        return result.x, -result.fun

    def physics_step(centers, radii):
        n = len(centers)
        overlaps = np.zeros((n, n))
        moves = np.zeros((n, 2))
        
        # Overlap detection
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                overlap = np.maximum(0, radii[i] + radii[j] - dist)
                overlaps[i][j] = overlap
                overlaps[j][i] = overlap
        
        # Resolve overlaps
        for i in range(n):
            for j in range(i+1, n):
                if overlaps[i][j] > 0:
                    diff = centers[i] - centers[j]
                    direction = diff / np.linalg.norm(diff)
                    move_i = -overlaps[i][j] / 2 * direction
                    move_j = overlaps[i][j] / 2 * direction
                    moves[i] -= move_i
                    moves[j] -= move_j

        # Apply boundary corrections
        for i in range(n):
            x, y = centers[i]
            radius = radii[i]
            if x - radius < 0:
                moves[i][0] += radius - x
            elif x + radius > 1:
                moves[i][0] -= x + radius - 1
            if y - radius < 0:
                moves[i][1] += radius - y
            elif y + radius > 1:
                moves[i][1] -= y + radius - 1
        
        # Simultaneous uniform growth
        delta_r = 0.001
        while True:
            new_radii = radii + delta_r
            n_overlaps = overlaps > 0
            if np.any(n_overlaps):
                delta_r *= 0.9
            else:
                break
        
        return centers + moves, new_radii
    
    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)
        radii = np.full(26, INIT_R)
        
        # Physics simulation
        for iter in range(SIM_ITERS):
            centres, radii = physics_step(centres, radii)
        
        # LP post-processing
        opt_radii, opt_sum = solve_lp_for_centres(centres)
        
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centres = centres.copy()
            best_radii = opt_radii.copy()
    
    # Final LP to lock in best configuration
    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
