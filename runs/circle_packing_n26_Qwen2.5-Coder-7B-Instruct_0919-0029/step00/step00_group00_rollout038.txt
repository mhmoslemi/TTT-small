import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog

INIT_R = 0.01
SIM_ITERS = 4000
NUM_STARTS = 30
EPS = 1e-12

def run_packing() -> (np.ndarray, np.ndarray, float):
    best_sum = np.inf
    best_centres = best_radii = None
    
    for start in range(NUM_STARTS):
        # Generate random centers
        centres = np.random.rand(26, 2)
        radii = np.full(26, INIT_R)
        
        for iteration in range(SIM_ITERS):
            # Compute pairwise distances
            dists = distance.cdist(centres, centres)
            
            # Resolve overlaps
            overlaps = (dists < (radii[:, np.newaxis] + radii[np.newaxis, :])) * (dists > EPS)
            overlaps &= np.triu(np.ones((26, 26), dtype=bool), k=1)
            for i, j in zip(*np.where(overlaps)):
                c_i = centres[i]
                c_j = centres[j]
                delta = (radii[i] + radii[j] - dists[i, j]) / 2
                move = delta * (c_j - c_i) / np.linalg.norm(c_j - c_i)
                centres[i] -= move / 2
                centres[j] += move / 2
            
            # Boundary correction
            for i in range(26):
                if centres[i, 0] - radii[i] < 0:
                    centres[i, 0] += abs(centres[i, 0] - radii[i] + EPS)
                elif centres[i, 0] + radii[i] > 1:
                    centres[i, 0] -= abs(centres[i, 0] + radii[i] - 1 - EPS)
                if centres[i, 1] - radii[i] < 0:
                    centres[i, 1] += abs(centres[i, 1] - radii[i] + EPS)
                elif centres[i, 1] + radii[i] > 1:
                    centres[i, 1] -= abs(centres[i, 1] + radii[i] - 1 - EPS)
            
            # Simultaneous radius growth with damping
            growth_step = 0.001
            for i in range(26):
                if overlaps[i].any():
                    growth_step /= 2
                    continue
                if radii[i] + growth_step > min(1, min(1-ct[0] for ct in centres), min(1-ct[1] for ct in centres)):
                    radii[i] += growth_step
                
                for j, o in enumerate(overlaps[i]):
                    if o:
                        growth_step /= 2
                        break
                else:
                    break
            
            if np.all(~overlaps):
                break
    
    # Solve the LP for the best centers found
    opt_radii, opt_sum = solve_lp_for_centres(centres)
    return centres, opt_radii, opt_sum

def solve_lp_for_centres(centres: np.ndarray) -> (np.ndarray, float):
    n = centres.shape[0]
    A = np.zeros((n*(n-1)//2, n))
    b = np.full(n*(n-1)//2, 0)
    c = -np.ones(n)
    
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            A[k, i] = 1
            A[k, j] = 1
            b[k] = distance.euclidean(centres[i], centres[j]) - EPS
            k += 1
    
    for i in range(n):
        A[k, i] = 1
        b[k] = centres[i, 0] - EPS
        k += 1
        A[k, i] = -1
        b[k] = -centres[i, 0] - EPS
        k += 1
        A[k, i] = 1
        b[k] = centres[i, 1] - EPS
        k += 1
        A[k, i] = -1
        b[k] = -centres[i, 1] - EPS
        k += 1
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)])
    
    if result.success:
        return result.x, -result.fun
    else:
        raise ValueError("LP failed to solve successfully")
