import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    Delta = 1 / k
    grid_points = np.array([(i * Delta, j * Delta) for i in range(k) for j in range(k)])
    Rmax = np.minimum(grid_points[:, 0], 1 - grid_points[:, 0])[:, None]
    Rmax = np.minimum(Rmax, np.minimum(grid_points[:, 1], 1 - grid_points[:, 1]))
    vars_by_grid = {tuple(p): (b_g, r_g) for (i, j), (p, R) in enumerate(zip(grid_points, Rmax)) for b_g, r_g in [(f'b_{i}_{j}', f'r_{i}_{j}'), (f'b_{i}_{j}_r', f'r_{i}_{j}_r')]}

    model = pywraplp.Solver.CreateSolver('SCIP')

    for (ix, jx), (b_g, r_g) in vars_by_grid.items():
        model.AddVariable(0, 1, name=b_g)
        model.AddVariable(0, Rmax[ix, 0], name=r_g)

    for g, (b_g, r_g) in vars_by_grid.items():
        model.Add(constraint=model.Sum([model.NumVar(0, float('inf'), name=f'd_{tuple(g)}_{tuple(h)}') for h, (b_h, r_h) in vars_by_grid.items() if np.linalg.norm(g - h) >= Rmax[ix, 0] + Rmax[ix, 1]]) <= np.linalg.norm(g - h) + 2 - (model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}') + model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}_r') * ((model.NumVar(0, float('inf'), name=f'b_{tuple(g)}') != model.NumVar(0, float('inf'), name=f'b_{tuple(h)}')) * model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}')) * model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}_r') * ((model.NumVar(0, float('inf'), name=f'b_{tuple(g)}_r') != model.NumVar(0, float('inf'), name=f'b_{tuple(h)}_r')) * model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}_r') * model.NumVar(0, float('inf'), name=f'M_{tuple(g)}_{tuple(h)}_r'))))) - (model.NumVar(0, float('inf'), name=f'd_{tuple(g)}_{tuple(h)}_r') for d_g_h_r in vars_by_grid.items() if np.linalg.norm(g - h) >= Rmax[ix, 0] + Rmax[ix, 1])) - (model.NumVar(0, float('inf'), name=f'd_{tuple(g)}_{tuple(h)}_r') for d_g_h_r in vars_by_grid.items() if np.linalg.norm(g - h) >= Rmax[ix, 0] + Rmax[ix, 1])) - (model.NumVar(0, float('inf'), name=f'd_{tuple(g)}_{tuple(h)}_r') for d_g_h_r in vars_by_grid.items() if np.linalg.norm(g - h) >= Rmax[ix, 0] + Rmax[ix, 1])) - (model.NumVar(0, float('inf'), name=f'd_{tuple(g)}_{tuple(h)}_r') for d_g_h_r in vars_by_grid.items() if np.linalg.norm(g - h) >= Rmax[ix, 0] + Rmax[ix, 1]))
        
    model.Maximize(model.Sum([r_g for _, r_g in vars_by_grid.values()]))

    return model, vars_by_grid

def solve_milp(model, time_limit=600):
    model.SetTimeLimit(time_limit * 1000)
    model.SetAbsoluteGapTolerance(1e-6)

    status = model.Solve()
    if status == pywraplp.Solver.OPTIMAL or status == pywraplp.Solver.FEASIBLE:
        best_solution = {key: (model.Value(var) for var in [val for (key, val) in model.variables() if model.Name(var) == str(key).replace('_', '')]) for key in [key for (key, val) in model.variables()]}
        return status, best_solution
    else:
        return status, None
