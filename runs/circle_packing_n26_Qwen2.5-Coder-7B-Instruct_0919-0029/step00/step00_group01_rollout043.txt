import numpy as np
from scipy.optimize import differential_evolution, linprog

def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None)**2
                          + np.clip(xs-1, 0, None)**2
                          + np.clip(-ys, 0, None)**2
                          + np.clip(ys-1, 0, None)**2)
    if np.any(rs < 0):
        penalty += 1e7 * np.sum(np.clip(-rs, 0, None)**2)
    for i in range(26):
        for j in range(i+1, 26):
            d = np.hypot(xs[i]-xs[j], ys[i]-ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += 1e7 * gap**2
    return penalty

def de_objective(v):
    return -(np.sum(v[52:]) + compute_penalty(v))

def run_de_search(seed):
    np.random.seed(seed)
    bounds = [(-0.25, 1.25)] * 52
    res = differential_evolution(de_objective, bounds, popsize=50, maxiter=2000, tol=1e-6, strategy='intermediate')
    return res.x

def run_packing():
    seeds = [np.random.randint(1, int(1e6)) for _ in range(2)]
    centres = [run_de_search(seed) for seed in seeds]
    radii, sum_radii = max(solve_lps_for_centres(centres[0]), solve_lps_for_centres(centres[1]))
    return np.array(centres[np.argmax(sum_radii)]).reshape(26, 2), radii, np.sum(radii)
