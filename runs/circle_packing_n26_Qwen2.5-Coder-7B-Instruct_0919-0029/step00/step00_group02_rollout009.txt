import numpy as np
from ortools.linear_solver import pywraplp Solver

def build_milp_grid(k):
    n_vars = k * k
    solver = Solver.CreateSolver('SCIP')

    b = [(i, j) for i in range(k) for j in range(k)]
    Rmax = np.zeros((k, k))
    for i in range(k):
        for j in range(k):
            Rmax[i, j] = min(i + 1, j + 1, k - i, k - j)

    x = {}
    for p in b:
        x[p] = solver.NumVar(0, Rmax[p], f'x_{p}')

    for p in b:
        for q in b:
            if p > q:
                dx = p[0] - q[0]
                dy = p[1] - q[1]
                d = np.sqrt(dx * dx + dy * dy)
                d_max = Rmax[p] + Rmax[q]
                if d >= d_max - 1e-6:
                    continue
                M = d - d_max + 1e-6
                solver.Add(x[q] + x[p] <= d + M * (2 - x[p] - x[q]))

    solver.Add(sum(x[p] for p in b) == 26)
    solver.Maximize(sum(x[p] for p in b))
    return solver, x

def solve_milp(solver, time_limit):
    solver.SetTimeLimit(time_limit * 1000)
    status = solver.Solve()
    best_sol = {p: solver.Value(x[p]) for p, x in x.items()}
    return status, best_sol

def extract_from_sol(sol, grid_vars):
    centers = []
    radii = []
    for p, x in grid_vars.items():
        if sol[p] > 1e-6:
            centers.append((p[0] / 20, p[1] / 20))
            radii.append(sol[p])
    return np.array(centers), np.array(radii)

def check_overlap(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            if np.sqrt(dx * dx + dy * dy) < radii[i] + radii[j] - 1e-12:
                return True
    return False

def polish(centers, radii):
    initial_sum = radii.sum()
    for _ in range(5):
        improved = False
        for i in range(radii.shape[0]):
            for dr in [1e-4]:
                radii[i] += dr
                if check_overlap(centers, radii):
                    radii[i] -= dr
                else:
                    improve = True
                    break
            if improve:
                break
        if not improved:
            break
    return centers, np.maximum(0, np.minimum(radii, 1 - radii))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
