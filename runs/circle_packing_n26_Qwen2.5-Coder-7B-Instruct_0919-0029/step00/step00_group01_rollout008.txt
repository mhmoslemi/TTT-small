import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, idx=i: borders[idx] - r[idx]})
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, idx1=i, idx2=j: distances[idx1, idx2] - (r[idx1] + r[idx2])})
    bounds = [(0, None)] * n
    c = -np.ones(n)
    result = linprog(c, A_ub=constraints, bounds=bounds, method='highs')
    radii = np.clip(result.x, 0, None)
    sum_radii = np.sum(radii)
    return radii, sum_radii

def random_move(centers, step):
    i = np.random.randint(centers.shape[0])
    new_pos = centers[i] + step * np.random.randn(2)
    new_pos = np.clip(new_pos, 0, 1)
    return new_pos

def simulated_annealing_centers(n, max_iter, rng):
    centers = rng.random((n, 2))
    step = 0.1
    T = 0.02
    T_end = 1e-5
    best_centers = centers
    best_radii, best_sum = solve_lp(centers)
    for _ in range(max_iter):
        i = np.random.randint(n)
        new_center = random_move(centers, step)
        centers[i] = new_center
        new_radii, new_sum = solve_lp(centers)
        if new_sum > best_sum or rng.rand() < np.exp((best_sum - new_sum) / T):
            best_centers = centers
            best_radii = new_radii
            best_sum = new_sum
        T *= T_end / T
    return best_centers, best_radii, best_sum

def run_packing() -> (np.ndarray, np.ndarray, float):
    n = 26
    max_iter = 30000
    rng = np.random.default_rng(42)
    centers, radii, sum_radii = simulated_annealing_centers(n, max_iter, rng)
    return centers, radii, sum_radii
