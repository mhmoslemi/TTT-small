import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    eps = 1e-9

    # Seed the configuration
    centers = np.array([
        [0.1, 0.1],
        [0.1, 0.3],
        [0.1, 0.5],
        [0.1, 0.7],
        [0.1, 0.9],
        [0.3, 0.1],
        [0.3, 0.3],
        [0.3, 0.5],
        [0.3, 0.7],
        [0.3, 0.9],
        [0.5, 0.1],
        [0.5, 0.3],
        [0.5, 0.5],
        [0.5, 0.7],
        [0.5, 0.9],
        [0.7, 0.1],
        [0.7, 0.3],
        [0.7, 0.5],
        [0.7, 0.7],
        [0.7, 0.9],
        [0.9, 0.1],
        [0.9, 0.3],
        [0.9, 0.5],
        [0.9, 0.7],
        [0.9, 0.9],
        [0.5, 0.5]
    ])

    for iteration in range(max_iter):
        done = True
        for i in range(n):
            radii[i] += g
            d2_center = np.dot(centers[i], centers[i]) - 1
            d2_neighbors = np.nansum((centers[i] - centers[:i**2]) ** 2, axis=1)
            max_d2 = np.max(np.vstack((d2_center, d2_neighbors)))
            radii[i] = min(np.sqrt(max_d2 / 2), 0.5, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-12:
                    distance_to_wall_x = 0.5 - abs(centers[i][0] - 0.5)
                    distance_to_wall_y = 0.5 - abs(centers[i][1] - 0.5)
                    distance_to_neighbor = (radii[i] + radii[j]) - dist
                    if distance_to_neighbor > 0:
                        if distance_to_wall_x <= distance_to_wall_y and distance_to_wall_x <= distance_to_neighbor:
                            centers[i][:] += np.sign(centers[i][0] - 0.5) * delta * np.array([1, 0])
                        elif distance_to_wall_x >= distance_to_wall_y and distance_to_wall_x <= distance_to_neighbor:
                            centers[i][:] += np.sign(centers[i][0] - 0.5) * delta * np.array([-1, 0])
                        elif distance_to_wall_y <= distance_to_wall_x and distance_to_wall_y <= distance_to_neighbor:
                            centers[i][:] += np.sign(centers[i][1] - 0.5) * delta * np.array([0, 1])
                        elif distance_to_wall_y >= distance_to_wall_x and distance_to_wall_y <= distance_to_neighbor:
                            centers[i][:] += np.sign(centers[i][1] - 0.5) * delta * np.array([0, -1])
                        for k in range(n):
                            radii[k] = min(np.sqrt(np.dot(centers[k], centers[k]) - 1), 0.5, centers[k, 0], 1 - centers[k, 0], centers[k, 1], 1 - centers[k, 1])
                    done = False

        if done:
            break

        if iteration % shake_rounds == 0:
            shake_indices = np.random.choice(n, 1)
            for shake_index in shake_indices:
                centers[shake_index][:] += np.random.uniform(-0.02, 0.02, 2)
                centers[shake_index] = np.clip(centers[shake_index], 0, 1)
                for k in range(n):
                    radii[k] = min(np.sqrt(np.dot(centers[k], centers[k]) - 1), 0.5, centers[k, 0], 1 - centers[k, 0], centers[k, 1], 1 - centers[k, 1])

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
