import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    N = 26
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    # Initialize seed configuration
    grid_size = 25
    center = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(grid_size) for j in range(grid_size)])
    center = np.insert(center, 25, np.array([0.5, 0.5]), axis=0)
    
    # Divide the circles into groups
    centers[1:26] = center[:25]
    
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    
    for _ in range(max_iter):
        new_radii = radii + g
        conflicts = np.zeros(N, dtype=bool)
        
        # Check wall and neighbor conflicts
        for i in range(N):
            if new_radii[i] > centers[i, 0] or new_radii[i] > 1 - centers[i, 0] or new_radii[i] > centers[i, 1] or new_radii[i] > 1 - centers[i, 1]:
                conflicts[i] = True
            for j in range(i + 1, N):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < new_radii[i] + new_radii[j]:
                    conflicts[i] = True
                    conflicts[j] = True
        
        # Resolve conflicts
        for i in np.where(conflicts)[0]:
            nearest_radius = np.inf
            nearest_index = -1
            for j in range(N):
                if j != i:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < nearest_radius:
                        nearest_radius = dist
                        nearest_index = j
            if nearest_radius < 2 * new_radii[i]:
                smallest_radius = np.inf
                if (i >= 1 and i < 26) and new_radii[i] < new_radii[i - 1] and new_radii[i] < radii[i - 1]:
                    dist = np.linalg.norm(centers[i] - centers[i - 1])
                    if dist < 2 * new_radii[i]:
                        smallest_radius = min(smallest_radius, dist)
                if (i >= 25 and i < 5**2) and new_radii[i] < new_radii[i + 25] and new_radii[i] < radii[i + 25]:
                    dist = np.linalg.norm(centers[i] - centers[i + 25])
                    if dist < 2 * new_radii[i]:
                        smallest_radius = min(smallest_radius, dist)
                if (i >= 1 and i % 25 != 0) and new_radii[i] < new_radii[i - 1] and new_radii[i] < radii[i - 1]:
                    dist = np.linalg.norm(centers[i] - centers[i - 1])
                    if dist < 2 * new_radii[i]:
                        smallest_radius = min(smallest_radius, dist)
                if (i >= 25 and (i + 1) % 25 != 0) and new_radii[i] < new_radii[i + 1] and new_radii[i] < radii[i + 1]:
                    dist = np.linalg.norm(centers[i] - centers[i + 1])
                    if dist < 2 * new_radii[i]:
                        smallest_radius = min(smallest_radius, dist)
                new_radii[i] = smallest_radius / 2
        
        # Check convergence
        if np.sum(new_radii) - np.sum(radii) < 1e-8:
            break
        
        # Update centers based on the closest tighter constraint
        for i in np.where(conflicts)[0]:
            if nearest_index != -1:
                dist = np.linalg.norm(centers[i] - centers[nearest_index])
                if dist < 2 * new_radii[i]:
                    delta_vector = (centers[i] - centers[nearest_index]) / dist * delta / 2
                    centers[i] += delta_vector
            else:
                if centers[i, 0] > 1 - centers[i, 0]:
                    centers[i, 0] -= delta
                elif centers[i, 0] < centers[i, 0]:
                    centers[i, 0] += delta
                if centers[i, 1] > 1 - centers[i, 1]:
                    centers[i, 1] -= delta
                elif centers[i, 1] < centers[i, 1]:
                    centers[i, 1] += delta
        
        # Recompute exact maximal allowed radius
        for i in range(N):
            new_radii[i] = min(new_radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        
        # Apply shake for every shake_rounds iterations
        if (_ + 1) % shake_rounds == 0:
            shake_index = np.random.randint(N)
            shake_vector = np.random.uniform(-0.02, 0.02, size=2)
            shake_vector = shake_vector / np.linalg.norm(shake_vector)
            centers[shake_index] += shake_vector
            centers[shake_index] = np.clip(centers[shake_index], 0, 1)
            new_radii[shake_index] = min(1, centers[shake_index, 0], 1 - centers[shake_index, 0], centers[shake_index, 1], 1 - centers[shake_index, 1])
        
        radii = new_radii
    
    return centers, radii, np.sum(radii)
