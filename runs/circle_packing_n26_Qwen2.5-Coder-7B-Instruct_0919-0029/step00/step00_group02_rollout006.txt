import numpy as np
from ortools.linear_solver import pywraplp Solver

def build_milp_grid(k=20):
    Δ = 1.0 / k
    Rmax = 0.5
    grid_points = [(i * Δ, j * Δ) for i in range(k) for j in range(k)]
    vars_by_grid = {}
    model = Solver.CreateSolver('GLOP')
    for i, (x, y) in enumerate(grid_points):
        b = model.NumVar(0, 1, f'b_{i}')
        r = model.NumVar(0, Rmax, f'r_{i}')
        model.Add(r <= b * Rmax)
        vars_by_grid[(x, y)] = (b, r)
    model.Add(sum(vars_by_grid[(x, y)][1] for (x, y) in grid_points) == 26)
    return model, vars_by_grid

def solve_milp(model, time_limit=600):
    model.SetNumThreads(1)
    model.SetTimeLimit(time_limit)
    model.SetObjectiveOffset(0)
    model.Solve()
    return model.Status(), {pos: (b.solution_value(), r.solution_value()) for pos, (b, r) in vars_by_grid.items()}

def extract_solution(best_solution, vars_by_grid):
    centers = []
    radii = []
    for (x, y), (b, r) in vars_by_grid.items():
        if b > 0.5:
            centers.append([x, y])
            radii.append(r)
    return np.array(centers), np.array(radii)

def polynomial(x, y, c):
    return c[0] + c[1] * x + c[2] * y + c[3] * x * x + c[4] * x * y + c[5] * y * y + c[6] * x * x * x + c[7] * x * x * y + c[8] * x * y * y + c[9] * y * y * y

def find_nearest_center(centers, radius, x, y):
    nearest = None
    min_dist = float('inf')
    for cx, cy in centers:
        dist = np.sqrt((cx - x) ** 2 + (cy - y) ** 2)
        if dist < min_dist:
            min_dist = dist
            nearest = (cx, cy)
    if min_dist < radius:
        px, py = nearest
        angle = np.arctan2(y - py, x - px)
        return px + radius * np.cos(angle - np.pi/2), py + radius * np.sin(angle - np.pi/2)
    return x, y

def polish_solution(centers, radii):
    centers = np.asarray(centers)
    radii = np.asarray(radii)
    for _ in range(10):
        new_centers = []
        for i in range(len(centers)):
            cx, cy = centers[i]
            rx, ry = radii[i]
            new_rx = radii[i]
            new_ry = radii[i]
            if i > 0:
                nearest_center = find_nearest_center(centers[:i], radii[:i], cx, cy)
                nearest_radius = radii[centers.tolist().index(nearest_center)]
                dist = np.sqrt((cx - nearest_center[0]) ** 2 + (cy - nearest_center[1]) ** 2)
                if dist < new_rx + nearest_radius:
                    new_rx = (new_rx + nearest_radius) / 2
            if i < len(centers) - 1:
                nearest_center = find_nearest_center(centers[i+1:], radii[i+1:], cx, cy)
                nearest_radius = radii[centers.tolist().index(nearest_center, i+1)]
                dist = np.sqrt((cx - nearest_center[0]) ** 2 + (cy - nearest_center[1]) ** 2)
                if dist < new_ry + nearest_radius:
                    new_ry = (new_ry + nearest_radius) / 2
            new_centers.append([find_nearest_center(centers, max(new_rx, new_ry), cx, cy), new_rx, new_ry])
        centers = np.array([center for center, rx, ry in new_centers])
        radii = np.array([radius for center, rx, ry in new_centers])
    return centers, radii

def validate_packing(centers, radii):
    n = len(centers)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            raise Exception(f"Circle {i} has negative radius {r}")

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise Exception(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise Exception(f"Circles {i} and {j} overlap")

    return "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    model, vars_by_grid = build_milp_grid(k=20)
    status, best_solution = solve_milp(model, time_limit=600)
    if status != Solver.OPTIMAL and status != Solver.FEASIBLE:
        raise Exception('MILP solver did not find a feasible packing')
    centers, radii = extract_solution(best_solution, vars_by_grid)
    centers, radii = polish_solution(centers, radii)
    validate_packing(centers, radii)
    return centers, radii, radii.sum()
