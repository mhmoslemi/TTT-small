import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    n = k * k
    delta = 1.0 / k
    M = 2.0
    
    model = pywraplp.Solver.CreateSolver('SCIP')
    
    vars_by_grid = np.empty((k, k), dtype=object)
    b_g = np.empty((k, k), dtype=object)
    r_g = np.empty((k, k), dtype=object)
    
    for i in range(k):
        for j in range(k):
            grid_point = (i * delta, j * delta)
            r_max_g = min(grid_point[0], 1 - grid_point[0], grid_point[1], 1 - grid_point[1])
            
            b_g[i, j] = model.NumVar(0, 1, 'b_{}_{}'.format(i, j), pywraplp.VariableType.BINARY)
            r_g[i, j] = model.NumVar(0, r_max_g, 'r_{}_{}'.format(i, j))
            
            model.Add(r_g[i, j] <= b_g[i, j] * r_max_g)
            vars_by_grid[i, j] = (b_g[i, j], r_g[i, j], grid_point)
    
    model.Add(sum(b_g.flatten()) == 26)
    
    for i in range(k):
        for j in range(k):
            if i > 0 or j > 0:
                for di in range(-1, 2):
                    for dj in range(-1, 2):
                        if 0 <= i + di < k and 0 <= j + dj < k:
                            g = (i, j)
                            h = (i + di, j + dj)
                            d_max_gh = min(max(grid_point[0], grid_point[1]), 1 - min(grid_point[0], grid_point[1]))
                            d_gh = np.sqrt((grid_point[0] - vars_by_grid[h][2][0]) ** 2 + (grid_point[1] - vars_by_grid[h][2][1]) ** 2)
                            if d_gh < d_max_gh + 1e-6:
                                model.Add(vars_by_grid[g][1] + vars_by_grid[h][1] <= d_gh + M * (2 - vars_by_grid[g][0] - vars_by_grid[h][0]))
                            elif d_gh < d_max_gh + 1e-3:
                                epsilon = 1e-6
                                M_gh = d_gh - d_max_gh + epsilon
                                model.Add(vars_by_grid[g][1] + vars_by_grid[h][1] <= d_gh + M_gh * (2 - vars_by_grid[g][0] - vars_by_grid[h][0]))
    
    model.Maximize(sum(r_g.flatten()))
    
    return model, vars_by_grid

def solve_milp(model, time_limit=600):
    solver = model.Solver()
    solver.parameters.num_threads = 1
    solver.parameters.time_limit_seconds = time_limit
    solver.parameters.mip_gap_abs = 1e-6
    
    status = solver.Solve()
    
    return status, model

def extract_solution(best_solution, vars_by_grid):
    k = len(vars_by_grid)
    centers = []
    radii = []
    
    for i in range(k):
        for j in range(k):
            if best_solution.Value(vars_by_grid[i, j][0]) > 0.5:
                centers.append(vars_by_grid[i, j][2])
                radii.append(best_solution.Value(vars_by_grid[i, j][1]))
    
    return np.array(centers), np.array(radii)

def polish_solution(centers, radii):
    eps = 1e-4
    delta_r = 1e-4
    
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        while True:
            overlaps = False
            for j in range(len(centers)):
                if j != i:
                    dx, dy = centers[j] - centers[i]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < r + radii[j] and dist > r + radii[j] - 1e-6:
                        overlaps = True
                        break
            if overlaps:
                r += delta_r
                centers[i] = [x + eps * np.sign(dx), y + eps * np.sign(dy)]
            else:
                break
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
