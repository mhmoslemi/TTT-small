import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

def compute_penalty(z, λ, p):
    n = z.size // 3
    x, y, r = z[:-1].reshape((-1, 3)).T
    d = np.sqrt(np.sum((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2, axis=2))
    overlap = np.maximum(0, (r[:, np.newaxis] + r[np.newaxis, :]) - d)
    boundary = np.maximum(0, r - np.minimum(x, 1 - x, y, 1 - y))
    return -np.sum(r) + λ * np.sum(overlap**p + boundary**p)

def compute_gradient(z, λ, p):
    n = z.size // 3
    dx, dy, dr = np.gradient(z)
    x, y, r = z[:-1].reshape((-1, 3)).T
    d = np.sqrt(np.sum((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2, axis=2))
    overlap = np.maximum(0, (r[:, np.newaxis] + r[np.newaxis, :]) - d)
    boundary = np.maximum(0, r - np.minimum(x, 1 - x, y, 1 - y))
    dr += λ * np.sum((2 * np.outer(r, r) / d**2) * overlap * (d - r[:, np.newaxis] - r[np.newaxis, :]) ** (p-2), axis=1)
    dr -= λ * np.sum(np.sign(boundary) * p * boundary**(p-1), axis=1)
    ddx = λ * np.sum((2 * x[:, np.newaxis] * np.outer(r, r) / d**4) * overlap * (d - r[:, np.newaxis] - r[np.newaxis, :]) ** (p-1) + 
                     np.outer(np.sign(boundary), p * boundary**(p-1)), axis=1)
    ddy = λ * np.sum((2 * y[:, np.newaxis] * np.outer(r, r) / d**4) * overlap * (d - r[:, np.newaxis] - r[np.newaxis, :]) ** (p-1) + 
                     np.outer(np.sign(boundary), p * boundary**(p-1)), axis=1)
    return np.concatenate((np.zeros(ndim=2, shape=(26, 1)), ddx, ddy, dr))

def lp_optimize_centres(centres):
    n = 26
    x, y = centres
    radius_constraints = []
    for i in range(n):
        radius_constraints.append({'type': 'ineq', 'fun': lambda z, i=i: min(x[i], 1 - x[i], y[i], 1 - y[i]) - z[i]})
    total_radius = LinearConstraint(np.ones(n), 0, 1) 
    x_bounds = [(0, 1) for _ in range(n)]
    y_bounds = [(0, 1) for _ in range(n)]
    r_bounds = [(0, 1) for _ in range(n)]
    bounds = Bounds(np.concatenate((x_bounds, y_bounds, r_bounds)))
    result = minimize(lambda r: -np.sum(r), np.ones(n), 
                     constraints=[total_radius] + radius_constraints, 
                     bounds=bounds,
                     method='highs')
    if not result.success:
        raise ValueError("LP optimization failed")
    return result.x, np.sum(result.x)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    λ, p = 5e4, 2
    n_starts = 18
    max_evals = 2000
    tolerance = 1e-6
    best_sum = -np.inf
    best_solution = None

    for _ in range(n_starts):
        initial_guess = np.hstack([np.random.rand(26) * 0.9 + 0.05,
                                  np.random.rand(26) * 0.9 + 0.05,
                                  np.ones(26) * 0.01])
        result = minimize(compute_penalty, initial_guess,
                         args=(λ, p),
                         jac=compute_gradient,
                         bounds=Bounds(low=0, high=1, shape=(initial_guess.size, 1)),
                         method='L-BFGS-B',
                         options={'maxfun': max_evals})
        if result.success and np.sum(result.fun) / n_starts > tolerance:
            centres, sum_radii = lp_optimize_centres(result.x[:-1].reshape(-1, 2))
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_solution = (centres, result.x[-1], sum_radii)
    
    return best_solution[0], best_solution[1], best_solution[2]
