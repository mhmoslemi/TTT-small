import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    m = 325 + 4 * n

    # Construct the coefficient matrix
    A = np.zeros((m, n))
    b = np.zeros(m)

    # Pairwise constraints r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = 2 * i + j - (i + 1)
            A[row, [i, j]] = 1.0
            b[row] = np.linalg.norm(centers[i] - centers[j]) - 1e-12

    # Boundary constraints r_i <= min(x_i, 1 - x_i, y_i, 1 - y_i)
    for i in range(n):
        A[m - 4 * i - 3:i - 1:-1] = 1.0
        b[m - 4 * i - 3:i - 1:-1] = np.minimum(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]) - 1e-12

    # Linear objective function to maximize sum of radii
    c = -np.ones(n)

    # Bounds on radii to ensure they are non-negative
    bounds = [(0, None) for _ in range(n)]

    # Solve the linear programming problem
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    # Return the radii and the maximum sum of radii
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # for reproducibility
    num_starts = 25
    num_iters = 2000
    step = 0.05
    step_min = 0.01
    decay_factor = 0.995

    best_centres = None
    best_sum = -np.inf

    for start in range(num_starts):
        centres = np.random.rand(26, 2)
        for it in range(num_iters):
            idx = np.random.randint(0, 26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c
            new_radii, new_sum = lp_optimum(new_centres)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centres)

    return best_centres, final_radii, final_sum

# Run the packing function and print the results
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of Radii: {sum_radii}")
