```python
import numpy as np
from scipy.optimize import linprog

NUM_STARTS = 30
SIM_ITERS = 2500
INIT_R = 0.01
DELTA_R = 0.001
SHRINK_FACTOR = 0.99
EPSILON = 1e-12

def solve_lp_for_centres(centers):
    n = centers.shape[0]
    radii = np.full(n, INIT_R)
    A_eq, b_eq = [], []
    for i in range(n):
        A_eq.append([1 if j == i else 0 for j in range(n)])
        b_eq.append(radii[i])
    A_ineq = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append([-1, 0, 0, 0, 0])
            b_ineq.append(-(radii[i] + radii[j]))
        for j in [0, 1, 2, 3]:
            A_ineq.append([0, -1 if j % 2 == 0 else 1, -1 if j % 2 == 0 else 1, 0, 0])
            b_ineq.append(-radii[j])
    res = linprog(-np.ones(n), A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    if not res.success:
        raise Exception("LP failed")
    optimal_radii = res.x
    return optimal_radii, np.sum(optimal_radii)

def run_packing() -> (np.ndarray, np.ndarray, float):
    best_centres = None
    best_radii   = None
    best_sum     = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26,2)        # uniform sample
        radii   = np.full(26, INIT_R)

        # --- physics simulation ---
        for iter in range(SIM_ITERS):
            distances = np.linalg.norm(centers[:, None] - centers, axis=2)
            overlaps = (radii[:, None] + radii >= distances - EPSILON)
            overlap_pairs = np.nonzero(overlaps & np.triu(overlaps, k=1))

            for i, j in zip(*overlap_pairs):
                u = (centers[i] - centers[j]) / np.linalg.norm(centers[i] - centers[j])
                delta = 0.5 * (radii[i] + radii[j] - distances[i, j])
                centers[i] -= delta * u / 2
                centers[j] += delta * u / 2

            boundary_shifts = np.min(np.stack([np.maximum(0, centers[:, i] - (radii[:] if j == i else 0)), 
                                                np.maximum(0, (1 - centers[:, i]) - (radii[:] if j == i else 0))], axis=1), axis=1)
            centers += np.diag(boundary_shifts)

            radii_change = np.minimum(np.ones(26) * DELTA_R, -1 * overlaps.sum(axis=1))
            for new_r, r in zip(np.add(radii, radii_change), radii):
                new_centres = centers + np.diag(np.clip(new_r - radii, 0, None) / 2)
                distances = np.linalg.norm(new_centres[:, None] - new_centres, axis=2)
                overlap_count = np.sum((np.maximum(0, new_r + radii >= distances - EPSILON) & np.triu(overlaps, k=1)).sum(axis=1) > 0)
                if overlap_count == 0:
                    radii = new_r
                    break
            DELTA_R *= SHRINK_FACTOR

        # --- LP post-processing ---
        opt_radii, opt_sum = solve_lp_for_centres(centres)

        if opt_sum > best_sum:
            best_sum   = opt_sum
            best_centres = centres.copy()
            best_radii   = opt_radii.copy()

    # final LP to lock in best configuration
    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
```
