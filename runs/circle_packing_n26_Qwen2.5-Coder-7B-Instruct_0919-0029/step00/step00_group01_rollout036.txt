import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = len(centers)
    borders = np.minimum.reduce([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]])
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(distances, np.inf)
    
    A_ub = []
    for i in range(n):
        A_ub.append([-1, 0] * n)
        A_ub.append([1, 0] * n)
        A_ub.append([0, -1] * n)
        A_ub.append([0, 1] * n)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([1] * i + [-1] + [1] * (j - i - 1) + [-1] + [1] * (n - j - 1))
    
    A_ub = np.array(A_ub)
    b_ub = np.concatenate((borders, distances)).flatten()
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
    radii = np.maximum(result.x, 0)
    sum_radii = np.sum(radii)
    return radii, sum_radii

def random_move(centers, step, rng):
    i = rng.integers(0, centers.shape[0])
    new_pos = centers[i] + rng.normal(scale=step, size=2)
    new_pos = np.clip(new_pos, 0, 1)
    centers[i] = new_pos
    return centers

def simulated_annealing_centers(n, max_iter, rng, inner_count=1000):
    centers = rng.random((n, 2))
    best_centers = centers.copy()
    best_rsum = 0.0
    
    for inner_iter in range(inner_count):
        best_sum = 0.0
        for iter in range(max_iter):
            centers = random_move(centers, 0.05, rng)
            radii, sum_radii = solve_lp(centers)
            if sum_radii > best_sum:
                best_sum = sum_radii
                if sum_radii > best_rsum:
                    best_centers = centers.copy()
                    best_rsum = sum_radii
        
        centers = best_centers.copy()  # Reuse best so far
    
    return best_centers, best_rsum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 30000
    inner_count = 1000
    rng = np.random.default_rng(42)
    
    best_centers, best_rsum = simulated_annealing_centers(n, max_iter, rng, inner_count)
    radii, _ = solve_lp(best_centers)
    
    return best_centers, radii, best_rsum
