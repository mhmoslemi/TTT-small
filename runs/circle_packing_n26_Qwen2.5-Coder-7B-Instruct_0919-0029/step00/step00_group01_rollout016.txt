import numpy as np
from scipy.optimize import differential_evolution, linprog

def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    # boundary
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None)**2
                          + np.clip(xs-1, 0, None)**2
                          + np.clip(-ys, 0, None)**2
                          + np.clip(ys-1, 0, None)**2)
    if np.any(rs < 0):
        penalty += 1e7 * np.sum(np.clip(-rs, 0, None)**2)
    # pairwise
    for i in range(26):
        for j in range(i+1, 26):
            d = np.hypot(xs[i]-xs[j], ys[i]-ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += 1e7 * gap**2
    return penalty

def de_objective(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = compute_penalty(v)
    return -(np.sum(rs) - penalty)

def solve_lps_for_centres(centres):
    def radii_objective(r):
        return -np.sum(r)
    A = []
    b = []
    for i in range(26):
        A.append(np.hstack([np.zeros(52), -np.eye(26)[i], np.array([1])]))
        b.append(0)
    for i in range(26):
        A.append(np.hstack([np.zeros(52), -np.eye(26)[i], np.array([-1])]))
        b.append(0)
    for i in range(26):
        A.append(np.hstack([np.zeros(52), np.eye(26)[i], np.array([1])]))
        b.append(min(centres[i, 0], 1-centres[i, 0], centres[i, 1], 1-centres[i, 1]))
    for i in range(26):
        for j in range(i+1, 26):
            A.append(np.hstack([np.zeros(52), -np.eye(26)[i], -np.eye(26)[j], np.array([1])]))
            b.append(0)
            A.append(np.hstack([np.zeros(52), -np.eye(26)[i], -np.eye(26)[j], np.array([1])]))
            b.append(0)
    sol = linprog(radii_objective, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return sol.x[52:], -sol.fun

def run_de_search(seed):
    bounds = [(0, 1)] * 52 + [(0, 0.25)] * 26
    result = differential_evolution(de_objective, bounds, strategy='intermediate', maxiter=2000, tol=1e-6, seed=seed)
    return result.x[:52]

def run_packing():
    best_sum = 0
    best_centres = None
    for _ in range(2):
        centres_best = run_de_search(np.random.randint(0, 10000, 52))
        if best_sum < solve_lps_for_centres(centres_best)[1]:
            best_sum = solve_lps_for_centres(centres_best)[1]
            best_centres = centres_best
    radii, total_sum = solve_lps_for_centres(best_centres)
    radii[radii < 0] = 0  # clip negative radii to 0
    return best_centres[:52].reshape(-1, 2), radii, total_sum

centres, radii, sum_radii = run_packing()
print(sum_radii)
