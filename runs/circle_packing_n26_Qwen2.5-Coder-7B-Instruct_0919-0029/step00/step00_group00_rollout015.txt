import numpy as np

def lp_optimum(centres):
    n = len(centres)
    A_ub = np.zeros((325 + 4 * n, n))
    b_ub = np.zeros(325 + 4 * n)
    c = -np.ones(n)

    for i in range(n):
        x, y = centres[i]
        radius_condition = min(x, 1 - x, y, 1 - y)
        boundary_constraints = [0, 1 - x, y, 1 - y]
        b_ub[i] = radius_condition - 1e-12
        A_ub[i, i] = 1

    for i in range(n):
        x, y = centres[i]
        d_ij = np.sqrt(np.sum((centres[i] - centres[i + 1:]) ** 2, axis=1))
        row_start = 26 + 4 * i
        for j in range(i + 1, n):
            A_ub[row_start + j - i - 1, i] = 1
            A_ub[row_start + j - i - 1, j] = 1
            b_ub[row_start + j - i - 1] = d_ij[j - i - 1] - 1e-12

    result = np.linalg.solve(A_ub, b_ub)
    radii = result
    return radii, -np.sum(result)

def run_packing():
    np.random.seed(42)
    n = 26
    step = 0.1
    step_min = 0.001
    decay_factor = 0.995
    NUM_ITERATIONS = 2000
    NUM_STARTS = 30

    best_sum = -np.inf
    best_centres = None
    best_radii = None

    def clip_position(pos):
        return np.clip(pos, 0, 1)

    for start in range(NUM_STARTS):
        centres = np.random.rand(n, 2) * 0.9  # avoid edge
        for it in range(NUM_ITERATIONS):
            idx = np.random.randint(n)
            delta = np.random.uniform(low=-step, high=step, size=2)
            new_c = clip_position(centres[idx] + delta)
            new_centres = centres.copy()
            new_centres[idx] = new_c

            radii, new_sum = lp_optimum(new_centres)

            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                best_radii = radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centres)
    return best_centres, final_radii, final_sum
