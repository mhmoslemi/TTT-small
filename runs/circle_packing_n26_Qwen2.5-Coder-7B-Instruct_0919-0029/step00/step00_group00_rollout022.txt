from scipy.optimize import linprog

# Fixed constants
N = 26
epsilon = 1e-12
# Number of starts for the random search part of the algorithm (each will have a number of iterations)
NUM_STARTS = 25
NUM_ITERS = 2000
step_min = 1e-6
decay_factor = 0.995

def compute_lp_radii(centers):
    distances = np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1)

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i*2, i], A_ub[i*2, j] = 1, 1
            A_ub[i*2+1, i], A_ub[i*2+1, j] = -1, -1
            b_ub[i*2] = distances[i, j] - epsilon
            b_ub[i*2+1] = distances[i, j] - epsilon
            for k in [0, 1]:
                A_ub[i*2+k, n+i*n+k] = -1
                b_ub[i*2+k] = centers[i, k] - epsilon
        A_ub[i*2+n*n+k, i*n+k] = 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise Exception("Optimization failed.")
    radii = -result.x
    sum_radii = np.sum(radii)
    return radii, sum_radii

def run_packing():
    best_centers = np.random.rand(N, 2)
    best_sum = -np.inf
    best_radii = np.zeros(N)
    step = 0.05
    
    for start in range(NUM_STARTS):
        centers = np.random.rand(N, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(0, N)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c
            new_radii, new_sum = compute_lp_radii(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii
            centers = new_centers
        step = max(step_min, step * decay_factor)
    
    final_radii, final_sum = compute_lp_radii(best_centers)
    return best_centers, final_radii, final_sum

# Add a function to validate the packing
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
