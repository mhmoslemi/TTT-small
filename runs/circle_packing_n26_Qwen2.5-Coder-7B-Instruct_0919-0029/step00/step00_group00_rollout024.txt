def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centres = None
    best_radii = None
    best_sum = -np.inf
    NUM_STARTS = 25
    INIT_R = 0.01
    SIM_ITERS = 2500
    
    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)  # uniform sample
        radii = np.full(26, INIT_R)

        # --- physics simulation ---
        for iter in range(SIM_ITERS):
            # 1. compute pairwise distances
            dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
            overlap = np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - dists)
            non_overlap = overlap == 0

            # 2. resolve overlaps by displacements
            for i in range(26):
                for j in range(i + 1, 26):
                    if overlap[i, j] > 0:
                        direction = (centres[i] + centres[j]) / 2
                        moving = (centres[i] + centres[j]) / 2 - centres[i]
                        new_pos_i = centres[i] + 0.5 * overlap[i, j] * moving / np.linalg.norm(moving)
                        new_pos_j = centres[j] - 0.5 * overlap[i, j] * moving / np.linalg.norm(moving)
                        
                        if 0 <= new_pos_i[0] <= 1 and 0 <= new_pos_i[1] <= 1:
                            centres[i] = new_pos_i
                        if 0 <= new_pos_j[0] <= 1 and 0 <= new_pos_j[1] <= 1:
                            centres[j] = new_pos_j

            # 3. boundary pushes
            for i in range(26):
                if centres[i, 0] - radii[i] < 0:
                    centres[i, 0] = radii[i]
                if centres[i, 0] + radii[i] > 1:
                    centres[i, 0] = 1 - radii[i]
                if centres[i, 1] - radii[i] < 0:
                    centres[i, 1] = radii[i]
                if centres[i, 1] + radii[i] > 1:
                    centres[i, 1] = 1 - radii[i]

            # 4. attempt radius growth
            delta_r = INIT_R * 0.9 ** (iter // 100)
            new_radii = radii + delta_r
            for i in range(26):
                for j in range(i + 1, 26):
                    if overlap[i, j] > 0:
                        new_radii[i] -= delta_r
                        new_radii[j] -= delta_r
                        break
                for j in range(26):
                    if non_overlap[i, j]:
                        break
                if j == 26:
                    radii = new_radii
                    break

            # 5. damping / cooling
            if iter % 100 == 0:
                delta_r *= 0.9

            # 6. break if converged
            if np.sum(np.abs(radii - new_radii)) < 1e-6:
                radii = new_radii
                break

        # --- LP post-processing ---
        opt_radii = lp_for_centres(centres)
        opt_sum = np.sum(opt_radii)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centres = centres.copy()
            best_radii = opt_radii.copy()

    # final LP to lock in best configuration
    final_radii = lp_for_centres(best_centres)
    return best_centres, final_radii, np.sum(final_radii)
