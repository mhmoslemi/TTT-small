import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centres = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    initial_radii = np.full(n, 0.09)
    
    def make_constraints(centers, radii):
        m = n
        constraints = []
        for i in range(m):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] + x[3*i+1] - 1})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+1]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i] - x[3*i+2]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+1] - x[3*i+2]})
        for i in range(m):
            for j in range(i+1, m):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt((x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2) - x[3*i+2] - x[3*j+2]})
        return constraints

    def obj(x):
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        return -np.sum(radii)  # Max sum of radii -> Min -sum of radii

    def grow_step(centers, radii):
        for i in range(n):
            r = radii[i]
            while True:
                new_r = r + 0.001
                if all(np.sqrt(np.sum((centers[i] - centers[j])**2)) >= new_r + radii[j] for j in range(n) if i != j) and new_r <= 0.5:
                    radii[i] = new_r
                else:
                    break
        return centers, radii

    def shake_and_refine(centers, radii):
        for _ in range(20):
            for i in range(n):
                dx, dy = np.random.uniform(-5e-4, 5e-4, 2)
                new_center = centers[i] + np.array([dx, dy])
                new_center = np.clip(new_center, 0, 1)
                new_centers = np.copy(centers)
                new_centers[i] = new_center
                if all(np.sqrt(np.sum((new_centers[i] - new_centers[j])**2)) >= radii[i] + radii[j] for j in range(n) if i != j):
                    centers = new_centers
                    radii, _ = grow_step(new_centers, radii)
                    break
        return centers, radii

    best_objective = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(5):
        x0 = np.concatenate((np.random.rand(2*n) * 2 - 1, initial_radii))
        x0 = np.clip(x0, -1, 1)
        res = minimize(obj, x0, method='SLSQP', bounds=[(0, 1)] * 2*n + [(0, 0.5)] * n, constraints=make_constraints(initial_centres, initial_radii), options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
        if res.fun > best_objective:
            best_objective = res.fun
            best_centers = res.x[:2*n].reshape((n, 2))
            best_radii = res.x[2*n:]
    
    best_centers, best_radii = shake_and_refine(best_centers, best_radii)
    total_radius = np.sum(best_radii)
    return best_centers, best_radii, total_radius
