import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k: int):
    solver = pywraplp.Solver.CreateSolver('GLOP')
    grid_vars = np.zeros((k, k, 2), dtype=np.object)

    for i in range(k):
        for j in range(k):
            grid_vars[i, j, 0] = solver.NumVar(0.0, k - i, f'x_{i}_{j}')
            grid_vars[i, j, 1] = solver.NumVar(0.0, k - j, f'y_{i}_{j}')
            r_max = min(i + 1, k - i, j + 1, k - j)
            grid_vars[i, j, 2] = solver.NumVar(0.0, r_max, f'r_{i}_{j}')
            solver.Add(grid_vars[i, j, 2] <= grid_vars[i, j, 0] * r_max)
            solver.Add(grid_vars[i, j, 2] <= grid_vars[i, j, 1] * r_max)

    solver.Add(solver.Sum([grid_vars[i, j, 2] for i in range(k) for j in range(k)]) == 26)

    for i1 in range(k):
        for j1 in range(k):
            for i2 in range(i1 + 1, k):
                for j2 in range(j1 + 1, k):
                    distance_squared = (grid_vars[i1, j1, 0] - grid_vars[i2, j2, 0]) ** 2 + \
                                      (grid_vars[i1, j1, 1] - grid_vars[i2, j2, 1]) ** 2
                    r1 = grid_vars[i1, j1, 2]
                    r2 = grid_vars[i2, j2, 2]
                    d_max = min(k - i1, i1 + 1, k - j1, j1 + 1) + min(k - i2, i2 + 1, k - j2, j2 + 1)
                    if np.sqrt(distance_squared) >= d_max:
                        continue
                    M = np.sqrt(distance_squared) - d_max + 1e-6
                    solver.Add(r1 + r2 <= np.sqrt(distance_squared) + M * (2 - grid_vars[i1, j1, 2] - grid_vars[i2, j2, 2]))

    return solver, grid_vars

def solve_milp(solver, time_limit: int):
    solver.SetTimeLimit(time_limit * 1000)
    solver.Solve()
    if solver.Status() not in [pywraplp.Solver.OPTIMAL, pywraplp.Solver.FEASIBLE]:
        raise ValueError("Solver did not find a feasible solution")
    best_sol = np.zeros_like(grid_vars, dtype=np.float64)
    for i in range(k):
        for j in range(k):
            best_sol[i, j, 0] = grid_vars[i, j, 0].solution_value()
            best_sol[i, j, 1] = grid_vars[i, j, 1].solution_value()
            best_sol[i, j, 2] = grid_vars[i, j, 2].solution_value()
    return solver.Status(), best_sol

            if best_sol[i, j, 2] > 0:

def polish(centers, radii):
    n = centers.shape[0]
    delta_r = 1e-4
    epsilon = 1e-4
    while True:
        improved = False
        for i in range(n):
            r_try = radii[i] + delta_r
            if r_try <= radii.max() and validate_center_place(centers[i], radii, r_try):
                radii[i] = r_try
                improved = True
            for j in range(8):
                dx, dy = [epsilon, -epsilon, epsilon, -epsilon, 0, 0, epsilon, -epsilon][j], \
                         [0, 0, epsilon, -epsilon, epsilon, -epsilon, 0, 0][j]
                if validate_center_place(centers[i] + [dx, dy], radii, r_try):
                    centers[i] += [dx, dy]
                    improved = True
        if not improved:
            break
    return centers, radii

def validate_center_place(center, radii, radius):
    for i in range(centers.shape[0]):
        if i == np.where(np.all(centers == center, axis=1))[0][0]:
            continue
        dist_squared = np.sum((center - centers[i])**2)
        if dist_squared <= (radii[i] + radius)**2 - 1e-12:
            return False
    return center[0] + radius <= 1 and center[0] - radius >= 0 and \
           center[1] + radius <= 1 and center[1] - radius >= 0
