import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    d_wall = np.array([[min(i, 1 - i, j, 1 - j) for j in range(1, 26)] for i in range(1, 26)])
    d_pair = np.ones((n, n)) * np.inf
    np.fill_diagonal(d_pair, 0)

    def lp(radii):
        A = np.vstack([np.eye(n), -np.eye(n), d_wall[0:num_radius, 1::2], d_wall[1::2, 0:num_radius]])
        b = np.hstack([np.zeros(n), 0])
        c = -radii
        res = linprog(c, A_ub=A, b_ub=b, method='highs')
        return res.fun, radii

    def initial_centers():
        centers = np.array([[0.1+0.2*i, 0.1+0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
        return centers

    def simulated_annealing(centers):
        num_radius = len(centers)
        d_pair = np.array([[np.linalg.norm(centers[i] - centers[j]) for j in range(num_radius)] for i in range(num_radius)])
        for T in np.logspace(0, -6, 100):
            for _ in range(2000):
                i = np.random.randint(num_radius)
                new_center = centers[i] + np.random.normal(0, 0.1/T)
                new_center = np.clip(new_center, 0, 1)
                d_pair[i, :] = np.linalg.norm(new_center - centers, axis=1)
                d_pair[:, i] = np.linalg.norm(new_center - centers, axis=1)
                new_centers = np.array(centers)
                new_centers[i] = new_center
                _, new_radii = lp(d_pair.new_centers[:, 2])
                if np.random.rand() < np.exp((np.sum(new_radii) - np.sum(centers[:, 2])) / T):
                    centers = new_centers
        return centers, np.array([d_wall[d_pair[:, i].argmin(), i] for i in range(num_radius)])

    centers = initial_centers()
    best_centers, best_radii = centers, np.array([d_wall[d_pair[:, i].argmin(), i] for i in range(n)])
    best_sum = np.sum(best_radii)

    for _ in range(100_000):
        centers = simulated_annealing(centers)
        _, radii = lp(centers[:, 2])
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_centers, best_radii = centers, radii
            best_sum = current_sum

    # Clip negative radii to zero and scale down slightly
    best_radii = np.clip(best_radii, 0, 1)
    best_radii *= 0.9999999

    return best_centers[:, :2], best_radii, np.sum(best_radii)
