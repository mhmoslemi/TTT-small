import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, differential_evolution

def initial centers():
    grid = np.array([[0.1 + i * 0.2, 0.1 + j * 0.2] for i in range(5) for j in range(5)])
    central = np.array([[0.5, 0.5]])
    return np.vstack((grid, central))

def boundary_distances(centers):
    return np.minimum(centers[:, 0], 1 - centers[:, 0])[:, np.newaxis], np.minimum(centers[:, 1], 1 - centers[:, 1])[:, np.newaxis]

def pairwise_distances(centers):
    sq_dists = squareform(pdist(centers))
    return sq_dists

def max_radii(centers):
    wall_distances, x_distances = boundary_distances(centers)
    pair_distances = pairwise_distances(centers)
    A = np.vstack((-np.ones(26), -pair_distances + pair_distances.diagonal(), -wall_distances, -x_distances))
    b = -np.ones(26)
    c = np.ones(26)
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    return result.x

def simulated_annealing():
    np.random.seed(42)
    best_centers = np.copy(initial_centers())
    best_sum = 0.0

    for T in np.geomspace(1.0, 1e-6, num=100):
        for _ in range(2000):
            i = np.random.randint(26)
            perturbation = (np.random.randn(2) * max(1e-6, T * 0.1))
            new_centers = np.copy(best_centers)
            new_centers[i] += perturbation
            new_centers[np.any(new_centers < 0, axis=1), :] = 0
            new_centers[np.any(new_centers > 1, axis=1), :] = 1

            new_radii = max_radii(new_centers)
            new_sum = np.sum(new_radii)

            if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
                best_sum = new_sum
                best_centers = new_centers
            else:
                new_centers[i] -= perturbation * 2

        for _ in range(100):
            i = np.random.randint(26)
            direction = np.sign(np.random.randn(2))
            step_size = np.sqrt((new_radii[i] * (1 - new_radii[i])) * 1e-6)
            new_centers = np.copy(best_centers)
            new_centers[i] += direction * step_size
            new_centers[np.any(new_centers < 0, axis=1), :] = 0
            new_centers[np.any(new_centers > 1, axis=1), :] = 1

            new_radii = max_radii(new_centers)
            new_sum = np.sum(new_radii)

            if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
                best_sum = new_sum
                best_centers = new_centers

    return best_centers, max_radii(best_centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centers, radii = simulated_annealing()
    return best_centers, radii, np.sum(radii)
