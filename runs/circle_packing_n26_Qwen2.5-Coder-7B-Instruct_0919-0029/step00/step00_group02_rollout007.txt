from ortools.linear_solver import pywraplp
import numpy as np
from scipy.spatial import distance_matrix

def build_milp_grid(k):
    Δ = 1 / k
    grid = np.array([(i * Δ, j * Δ) for i in range(k+1) for j in range(k+1)])
    R_max = np.minimum(grid[:, 0], 1 - grid[:, 0], grid[:, 1], 1 - grid[:, 1])
    R_max = R_max.reshape(-1, 1)
    vars_by_grid = {}
    for i, g in enumerate(grid):
        vars_by_grid[tuple(g)] = {'b': None, 'r': None}
    model = pywraplp.Solver('Circle Packing', pywraplp.Solver.CBC_MIXED_INTEGER_PROGRAMMING)
    for g, var in vars_by_grid.items():
        var['b'] = model.NewBoolVar(f'b_{g}')
        var['r'] = model.NewContinuousVar(0, R_max[i], f'r_{g}')
        model.Add(var['b'] * R_max[i] >= var['r'])
    model.Add(model.Sum(vars_by_grid[g]['r'] for g in vars_by_grid.keys()) == 26)
    pair_filter = distance_matrix(grid, grid)
    pair_filter.setdiag(np.inf)
    M = 2.0
    for i, g in enumerate(grid):
        for j, h in enumerate(grid):
            if pair_filter[i, j] < R_max[i] + R_max[j] - 1e-6:
                model.Add(vars_by_grid[tuple(g)]['r'] + vars_by_grid[tuple(h)]['r'] <= 
                         pair_filter[i, j] + M * (2 - vars_by_grid[tuple(g)]['b'] - vars_by_grid[tuple(h)]['b']))
            else:
                model.Add(vars_by_grid[tuple(g)]['r'] + vars_by_grid[tuple(h)]['r'] <= 10000)
    model.Maximize(model.Sum(vars_by_grid[g]['r'] for g in vars_by_grid.keys()))
    return model, vars_by_grid

def solve_milp(model, time_limit):
    model.SetTimeLimit(time_limit * 1000)  # milliseconds
    solver = model.Solve()
    return solver, model

def extract_solution(best_solution, vars_by_grid):
    centers = []
    radii = []
    for g, var in vars_by_grid.items():
        if var['b'].solution_value() > 0.5:
            centers.append(np.array(g))
            radii.append(var['r'].solution_value())
    return np.array(centers), np.array(radii)

def polish_solution(centers, radii):
    δr = 1e-4
    ε = 1e-4
    n = len(centers)
    for _ in range(20):
        for i in range(n):
            for dr in [δr, -δr]:
                new_radius = min(max(radii[i] + dr, 0), 1 - max(abs(centers[i][0]), abs(centers[i][1])))
                if is_valid_packing(centers, radii, i, new_radius):
                    radii[i] = new_radius
        for i in range(n):
            for j in range(n):
                if i != j:
                    for dx, dy in [(ε, 0), (-ε, 0), (0, ε), (0, -ε)]:
                        new_centers = np.array(centers)
                        new_centers[i, :] += np.array([dx, dy])
                        if is_valid_packing(new_centers, radii, i, radii[i]):
                            centers[i, :] = new_centers[i, :]
    return centers, radii

def is_valid_packing(centers, radii, affected_idx, affected_radius):
    n = len(centers)
    for i in range(n):
        dist = np.linalg.norm(centers[i] - centers[affected_idx])
        if i != affected_idx and dist < radii[i] + affected_radius - 1e-12:
            return False
    return True
