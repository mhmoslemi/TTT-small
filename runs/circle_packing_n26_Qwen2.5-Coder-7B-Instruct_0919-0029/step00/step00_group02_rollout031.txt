import numpy as np
from scipy.spatial import distance

def build_random_feasible(n, r0=0.05):
    while True:
        centers = np.random.rand(n, 2)
        radii = np.full(n, r0)
        for i in range(n):
            count = 0
            while count < len(centers):
                if np.all(distance.cdist([centers[i]], centers[:i], 'euclidean') >= radii[i] + radii[:i]):
                    count += 1
                else:
                    centers[i] = np.random.rand(2)
                    radii[i] = np.full(1, r0)
                    count = 0
        if distance.cdist(centers, centers[:i], 'euclidean').max() > 2 * radii[i]:
            return centers, radii

def simulated_annealing(centers, radii, best_sum):
    T = 1.0
    cool = 0.9995
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    n_shift = 1000

        i = np.random.randint(0, len(centers))
        if np.random.rand() < move_prob:

            new_pos = np.clip(centers[i] + [dx, dy], radii[i], 1-radii[i])
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_r = max(0.0, min(radii[i] + dr, min(centers[i][0], 1-centers[i][0], centers[i][1], 1-centers[i][1])))
            new_pos = centers[i]

        tentative_state = np.copy(centers); tentative_state[i] = new_pos; tentative_state[i][2] = new_r
        if (np.all(tentative_state[:,0] - tentative_state[:,2] >= 0) and

            np.all(tentative_state[:,1] - tentative_state[:,2] >= 0) and 
            np.all(tentative_state[:,1] + tentative_state[:,2] <= 1) and
            np.all(distance.cdist(tentative_state, tentative_state[:i], 'euclidean') >= tentative_state[:,2][i] + tentative_state[:i,2]) and
            np.all(distance.cdist(tentative_state, tentative_state[i+1:], 'euclidean') >= tentative_state[:,2][i] + tentative_state[i+1:,2])):

            if ΔObj >= 0:
                centers = tentative_state.copy()
                radii = tentative_state[:, 2]

                centers = tentative_state.copy()
                radii = tentative_state[:, 2]
                best_sum = max(best_sum, radii.sum())
        if k % n_shift == 0:
            shift_max *= 0.95
            radius_max *= 0.95

def deterministic_polish(centers, radii):
    improved = True
    while improved:
        improved = False
        for i in range(len(centers)):
            r_try = radii[i] + 1e-4
            if r_try <= min(centers[i][0], 1-centers[i][0], centers[i][1], 1-centers[i][1]) and \
                    np.all(distance.cdist(centers[i:i+1], centers[:i+1], 'euclidean') >= r_try + radii[:i+1]):
                radii[i] = r_try
                improved = True

    for _ in range(10):
        i = np.random.randint(0, len(centers))
        for dx, dy in [(0.01, 0.0), (-0.01, 0.0), (0.0, 0.01), (0.0, -0.01), (0.01, 0.01), (0.01, -0.01), (-0.01, 0.01), (-0.01, -0.01)]:
            new_pos = np.clip(centers[i] + [dx, dy], radii[i], 1-radii[i])
            if np.all(distance.cdist([new_pos], centers[i+1:], 'euclidean') >= radii[i] + radii[i+1:]) and \
                    np.all(distance.cdist([new_pos], centers[:i], 'euclidean') >= radii[i] + radii[:i]):
                centers[i] = new_pos
                improved = True

    return centers, radii
