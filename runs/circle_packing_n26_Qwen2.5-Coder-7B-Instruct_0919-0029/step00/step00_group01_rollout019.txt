import numpy as np
from scipy.optimize import differential_evolution, linprog

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def penalty(v, n):
    x, y, r = v[:n, :2], v[:n, 2:], v[n:]
    penalty = 0
    penalty += np.sum(np.maximum(0, -x)**2 + np.maximum(0, x-1)**2 + np.maximum(0, -y)**2 + np.maximum(0, y-1)**2)
    penalty += np.sum(np.maximum(0, -r)**2)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((x[i] - x[j])**2 + (y[i] - y[j])**2))
            penalty += np.maximum(0, r[i] + r[j] - dist)**2
    return penalty

def objective(v, n):
    r = v[n:]
    return -np.sum(r) + 10000000 * penalty(v, n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = {(0, 1), (0, 1), (0, 0.25)]
    for _ in range(2):
        result = differential_evolution(
            lambda v: objective(v, n),
            bounds * n,
            popsize=50,
            strategy='intermediate',
            mutation=0.8,
            maxiter=2000
        )
        if result.success:
            centers, radii = result.x[:n, :2], result.x[:n, 2:], result.x[n:]
            _, radii, sum_radii = validate_packing(centers, radii)
            if sum_radii > 2.637:
                continue
        break
    centers, radii = result.x[:n, :2], result.x[:n, 2:], result.x[n:]
    if not np.isnan(centers).any() or not np.isnan(radii).any():
        for i in range(n):
            if radii[i] < 0:
                radii[i] = 0
    for i in range(n):
        for j in range(i + 1, n):
            if radii[i] + radii[j] > np.sqrt(np.sum((centers[i] - centers[j])**2)):
                radii[i] = np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[j]
    return centers, radii, np.sum(radii)

centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}")
print(f"Radii:\n{radii}")
print(f"Sum of Radii: {sum_radii}")
