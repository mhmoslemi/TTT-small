import numpy as np
from scipy.optimize import differential_evolution, linprog

def penalty(v, n_circles):
    center_indices = range(n_circles)
    centers = v[:2*n_circles].reshape(n_circles, 2)
    radii = v[2*n_circles:]
    
    boundary_penalty = sum(np.maximum(0, -centers[:, 0])**2)
    boundary_penalty += sum(np.maximum(0, centers[:, 0] - 1)**2)
    boundary_penalty += sum(np.maximum(0, -centers[:, 1])**2)
    boundary_penalty += sum(np.maximum(0, centers[:, 1] - 1)**2)
    
    neg_r_penalty = sum(np.maximum(0, -radii)**2)
    
    overlap_penalty = 0
    for i in center_indices:
        for j in center_indices:
            if i < j:
                rd = radii[i] + radii[j]
                dx, dy = centers[i] - centers[j]
                d = np.sqrt(dx**2 + dy**2)
                overlap_penalty += (rd - d)**2
    
    return boundary_penalty + neg_r_penalty + overlap_penalty * 1e7

def objective(v):
    return -np.sum(v[2*n_circles:]) + penalty(v, n_circles)

def refine_centers(circles, n_circles):
    def lp_objective(r):
        return -np.sum(r)
    
    bounds = [(0, None) for _ in range(n_circles)]
    constraints = []
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circles[i, 0] + r[i] - 1})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circles[i, 0] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circles[i, 1] + r[i] - 1})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circles[i, 1] - r[i]})
        for j in range(n_circles):
            if i < j:
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: circles[i, 0] + r[i] + circles[j, 0] + r[j] - (circles[i, 0] - circles[j, 0])**2 - (circles[i, 1] - circles[j, 1])**2})

    result = linprog(c=np.ones(n_circles), bounds=bounds, A_ub=[c for c in constraints if c['type'] == 'ineq'], b_ub=[c['fun']([]) for c in constraints if c['type'] == 'ineq'], method='highs')
    refined_radii = result['x']
    return refined_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    bounds = [(0, 1) for _ in range(2*n_circles)] + [(0, 0.25) for _ in range(n_circles)]
    init_pop = np.random.rand(50, 52)
    init_pop[:, :2*n_circles] = np.clip(init_pop[:, :2*n_circles], 0, 1)
    init_pop[:, 2*n_circles:] = np.clip(init_pop[:, 2*n_circles:], 0, 0.25)

    de_result_1 = differential_evolution(objective, bounds, init='latinhypercube', popsize=50, maxiter=2000)
    de_result_2 = differential_evolution(objective, bounds, init='latinhypercube', popsize=50, maxiter=2000)

    best_centers = np.stack([de_result_1.x[:2*n_circles], de_result_2.x[:2*n_circles]])
    best_radii = np.maximum(0, np.clip(np.stack([de_result_1.x[2*n_circles:], de_result_2.x[2*n_circles:]]), 0, None))
    best_sum_radii = np.sum(np.maximum(0, best_radii))

    if best_sum_radii < 2.637:
        # If the initial results are not good, run again
        de_result_3 = differential_evolution(objective, bounds, init='latinhypercube', popsize=50, maxiter=2000)
        de_result_4 = differential_evolution(objective, bounds, init='latinhypercube', popsize=50, maxiter=2000)
        best_centers = np.stack([de_result_3.x[:2*n_circles], de_result_4.x[:2*n_circles]])
        best_radii = np.maximum(0, np.clip(np.stack([de_result_3.x[2*n_circles:], de_result_4.x[2*n_circles:]]), 0, None))
        best_sum_radii = np.sum(np.maximum(0, best_radii))

    # Use the better of the two results
    best_sum_radii = np.max(np.array([best_sum_radii, np.sum(np.maximum(0, refine_centers(best_centers, n_circles)))]))
    centers = best_centers[np.argmax(np.array([best_sum_radii, np.sum(np.maximum(0, refine_centers(best_centers, n_circles)))]), axis=0)[:2*n_circles]]
    refined_radii = refine_centers(centers, n_circles)
    
    return centers, refined_radii, np.sum(refined_radii)
