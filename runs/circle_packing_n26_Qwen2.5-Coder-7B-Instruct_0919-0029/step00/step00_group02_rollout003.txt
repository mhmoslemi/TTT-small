import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k: int):
    delta = 1 / k
    grid_points = [(i * delta, j * delta) for i in range(k) for j in range(k)]
    R_max_values = np.array([[min(i * delta, (k - i - 1) * delta, j * delta, (k - j - 1) * delta) for j in range(k)] for i in range(k)])
    grid_vars = {'b': {}, 'r': {}}

    for idx, (i, j) in enumerate(grid_points):
        grid_vars['b'][idx] = np.array([(i * delta, j * delta, R_max_values[i, j])])

    return solver, grid_vars

def solve_milp(solver, time_limit):
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL and status != pywraplp.Solver.FEASIBLE:
        raise Exception(f"Solver failed: {status}")

    best_sol = {key: value for key, value in grid_vars.items()}
    return status, best_sol

def extract_from_sol(best_sol, grid_vars):
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    count = 0

    for idx, (i, j), r_max in grid_vars['b'].values():
        if best_sol['b'][idx] > 0:
            center = np.array([i, j])
            centers[count] = center
            radii[count] = min(best_sol['r'][idx], r_max)
            count += 1

    return centers, radii[:26]

def polish(centers, radii):
    epsilon = 1e-4
    delta_radius = 1e-4

    def is_valid(center, radius, all_centers, all_radii):
        for other_center, other_radius in zip(all_centers, all_radii):
            if np.linalg.norm(center - other_center) <= radius + other_radius:
                return False
        return True

    def try_grow(i):
        r = radii[i]
        while r + delta_radius < 1 - centers[i][0] and r + delta_radius < 1 - centers[i][1] and is_valid(centers[i], r + delta_radius, centers, radii):
            r += delta_radius
        radii[i] = r

    def try_shift(i):
        dx, dy = epsilon, 0
        for dx, dy in [(dx, dy), (-dx, dy), (dy, dx), (-dy, dx), (dx, -dy), (-dx, -dy), (dy, -dx), (-dy, -dx)]:
            new_center = centers[i] + np.array([dx, dy])
            if is_valid(new_center, radii[i], centers, radii):
                centers[i] = new_center
                break

    improved = True
    while improved:
        improved = False
        for i in range(len(centers)):
            improved |= not is_valid(centers[i], radii[i], centers, radii)
            try_grow(i)
            if not is_valid(centers[i], radii[i], centers, radii):
                improved |= True
                try_shift(i)

    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    k = 20
    solver = pywraplp.Solver.CreateSolver('CBC')
    grid_vars = {'b': {}, 'r': {}}

    for idx, (i, j), r_max in build_milp_grid(k).items():
        binary_var = solver.BoolVar(f'b_{idx}')
        continuous_var = solver.NumVar(0, 10, f'r_{idx}')
        grid_vars['b'][idx] = binary_var
        grid_vars['r'][idx] = continuous_var
        solver.Add(continuous_var <= binary_var * r_max)

    solver.Add(sum(grid_vars['b'].values()) == 26)

    for i, b_i in enumerate(grid_vars['b'].values()):
        for j, b_j in enumerate(grid_vars['b'].values()):
            if i == j:
                continue
            r_i = grid_vars['r'][i]
            r_j = grid_vars['r'][j]
            distance = np.sqrt(np.sum((b_i[0] - b_j[0]) ** 2))
            d_max = min(b_i[2], b_j[2])
            M = abs(distance - d_max) + 1e-6
            solver.Add(r_i + r_j <= distance + (2 - grid_vars['b'][i] - grid_vars['b'][j]) * M)

    objective = solver.Objective()
    for r in grid_vars['r'].values():
        objective.SetCoefficient(r, 1)
    objective.SetMaximization()

    status, best_sol = solve_milp(solver, time_limit=600)
    centers, radii = extract_from_sol(best_sol, grid_vars)
    centers, radii = polish(centers, radii)

    validate_packing(centers, radii)
    return centers, radii, radii.sum()
