import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def compute_penalty(z, λ, p):
    n = z.shape[0] // 3
    centres = z[:3*n].reshape(n, 3)
    r = centres[:, 2]

    penalty = -np.sum(r)
    d_ij = cdist(centres[:, :2], centres[:, :2])
    eta_ij = np.maximum(0, r[:, None] + r - d_ij)
    penalty += λ * np.sum(eta_ij**p)

    beta_x = np.maximum(0, r - centres[:, 0])
    beta_xp = np.maximum(0, r - (1 - centres[:, 0]))
    beta_y = np.maximum(0, r - centres[:, 1])
    beta_yp = np.maximum(0, r - (1 - centres[:, 1]))

    penalty += λ * (np.sum(beta_x**p) 
                   + np.sum(beta_xp**p) 
                   + np.sum(beta_y**p)
                   + np.sum(beta_yp**p))
    return penalty

def compute_gradient(z, λ, p):
    n = z.shape[0] // 3
    centres = z[:3*n].reshape(n, 3)
    radii = centres[:, 2]
    
    n_grad = z.shape[0]
    grad = np.zeros(n_grad)

    d_ij = cdist(centres[:, :2], centres[:, :2])
    eta_ij = np.maximum(0, radii[:, None] + radii - d_ij)

    for i in range(n):
        for j in range(i + 1, n):
            if eta_ij[i, j] > 0:
                avg_grad = np.zeros(6)
                avg_grad[:2] = (centres[i, :2] - centres[j, :2]) / d_ij[i, j]
                avg_grad *= 2 * λ * p * eta_ij[i, j]**(p-1)
                grad[3*i:(3*i+3)] += avg_grad[:3]
                grad[3*j:(3*j+3)] += avg_grad[3:]
            else:
                grad[3*i + 2] += -λ * p * np.sum(eta_ij[:, i]**(p-1))
                grad[3*j + 2] += -λ * p * np.sum(eta_ij[:, j]**(p-1))

    beta_x = np.maximum(0, radii - centres[:, 0])
    beta_xp = np.maximum(0, radii - (1 - centres[:, 0]))
    beta_y = np.maximum(0, radii - centres[:, 1])
    beta_yp = np.maximum(0, radii - (1 - centres[:, 1]))

    for i in range(n):
        if beta_x[i] > 0:
            grad[3*i] += -λ * p * beta_x[i]**(p-1)
        if beta_xp[i] > 0:
            grad[3*i + 1] += -λ * p * beta_xp[i]**(p-1)
        if beta_y[i] > 0:
            grad[3*i + 2] += -λ * p * beta_y[i]**(p-1)
        if beta_yp[i] > 0:
            grad[3*i + 3] += -λ * p * beta_yp[i]**(p-1)

    return grad

def lp_optimize_centres(centres):
    n = centres.shape[0]
    r = np.ones(n, dtype=np.float64) / n

    for i in range(n):
        r[i] = 0

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centres[i,0]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: centres[i,0] - 1 + r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centres[i,1]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: centres[i,1] - 1 + r[i]})
        for j in range(i):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[i]+r[j]-cdist([centres[i,:2]], [centres[j,:2]])})

    result = minimize(lambda r: -np.sum(r), r, bounds=[(0, 1)] * n, constraints=constraints, method='SLSQP')
    return result.x

def run_packing():
    n = 26
    z = np.full(n * 3, 0.01)  # initial guess: log-uniform distribution
    z.fill(1e-3)  # centre coordinates
    z[2::3] = np.random.rand(n) / 100  # radii; initialize so radii not close to limit

    best_sum = -1e9
    best_z = None
    best_centres = None

    for _ in range(18):  # try numerous random starts
        result = minimize(compute_penalty, z, args=(5e4, 2), bounds=[(0, 1)] * n * 3, jac=compute_gradient, method='L-BFGS-B', options={'maxfun': 2000, 'disp': False})

        if result.success and np.allclose(result.fun, compute_penalty(result.x, 5e4, 2)):
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_z = result.x
                best_centres = best_z[:3*n]

    centres = best_centres.reshape(n, 3)
    final_r = lp_optimize_centres(centres[:, :2])  # [x, y] -> [r]
    final_sum = np.sum(final_r)
    x = best_centres.reshape(n, 3)
    return x[:, :2], final_r, final_sum
