import numpy as np
from ortools.linear_solver import pywraplp as _
from approxeng.local import Local
import scipy.spatial as spatial

# Constants
k = 20
N = k ** 2
δr = 1e-4
ε = 1e-4
time_limit = 600
gap_tolerance = 1e-6

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Set up the MILP solver
    solver = _.Solver('MiLP', _.Solver.CBC_MIXED_INTEGER_PROGRAMMING)
    solver.SetNumThreads(1)
    solver.SetTimeLimit(time_limit)
    solver.EnableOutput(logs=False)
    solver.SetObjectiveTimeZoneOffset(0)

    # Create grid points
    Δ = 1 / k
    grid_points = np.array([(i * Δ, j * Δ) for i in range(k) for j in range(k)])

    # Compute maximum allowed radii
    R_max = np.minimum(np.minimum(grid_points[:, 0], 1 - grid_points[:, 0]), np.minimum(grid_points[:, 1], 1 - grid_points[:, 1]))

    # Create grid variables
    binary_vars = [solver.IntVar(0, 1, f'b_{i,j}') for i in range(k) for j in range(k)]
    continuous_vars = [solver.NumVar(0, None, f'r_{i,j}') for i in range(k) for j in range(k)]
    grid_vars = {f'b_{i,j}': binary_vars[i * k + j] for i in range(k) for j in range(k)}
    grid_vars.update({f'r_{i,j}': continuous_vars[i * k + j] for i in range(k) for j in range(k)})

    # Add constraints
    for p in grid_points:
        i, j = np.where((grid_points == p).all(axis=1))
        solver.Add(grid_vars[f'r_{i[0], j[0]}'] <= grid_vars[f'b_{i[0], j[0]}'] * R_max[i[0], j[0]])

    solver.Add(solver.Sum(grid_vars[f'r_{i,j}'] for i in range(k) for j in range(k)) <= 2.637)

    for p in grid_points:
        i_p, j_p = np.where((grid_points == p).all(axis=1))
        for q in grid_points:
            if (i_p, j_p) != (i_q, j_q):
                d_max = min(R_max[i_p, j_p], R_max[i_q, j_q])
                if np.linalg.norm(p - q) >= d_max:
                    continue
                i_q, j_q = np.where((grid_points == q).all(axis=1))
                M_pq = np.linalg.norm(p - q) - d_max + ε
                solver.Add(grid_vars[f'r_{i_p[0], j_p[0]}'] + grid_vars[f'r_{i_q[0], j_q[0]}'] <= np.linalg.norm(p - q) + M_pq * (2 - grid_vars[f'b_{i_p[0], j_p[0]}'] - grid_vars[f'b_{i_q[0], j_q[0]}']))

    solver.Maximize(solver.Sum(grid_vars[f'r_{i,j}'] for i in range(k) for j in range(k)))

    # Solve the problem
    status = solver.Solve()
    if status != _.Solver.OPTIMAL and status != _.Solver.FEASIBLE:
        raise Exception(f"Solver status: {status}")

    # Extract solutions
    sol = {(i, j): (grid_vars[f'b_{i,j}'].solution_value(), grid_vars[f'r_{i,j}'].solution_value()) for i in range(k) for j in range(k)}
    used_positions = [p for p in grid_points if sol[p][0] > 0.5]
    used_radii = [sol[p][1] for p in used_positions]

    # Post-process the solution
    centers, radii = polish(np.array(used_positions), np.array(used_radii))
    validate_packing(centers, radii)
    return centers, radii, np.sum(radii)

def polish(centers, radii):
    while True:
        more = False
        for i, (x, y) in enumerate(centers):
            r = radii[i]
            for dx, dy in [(-ε, 0), (ε, 0), (0, -ε), (0, ε), (-ε, -ε), (-ε, ε), (ε, -ε), (ε, ε)]:
                try_r = r + δr
                try_x, try_y = x + dx, y + dy
                if 0 <= try_x <= 1 and 0 <= try_y <= 1:
                    if all(np.sqrt((try_x - other_x) ** 2 + (try_y - other_y) ** 2) >= try_r + other_r for other_x, other_y, other_r in zip(centers, centers, radii)):
                        centers[i] = np.array([try_x, try_y])
                        radii[i] = try_r
                        more = True
        if not more:
            break
    return centers, radii
