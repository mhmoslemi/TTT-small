import numpy as np
from scipy.optimize import minimize, LinprogMessage, linprog

def compute_penalty(z, λ, p):
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    n = len(r)
    d_ij = np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2)
    d_ij[range(n), range(n)] = np.inf
    eta_ij = np.maximum(0, r[:, None] + r[None, :] - d_ij)
    beta_x = np.maximum(0, r - x)
    beta_x_prime = np.maximum(0, r - (1 - x))
    beta_y = np.maximum(0, r - y)
    beta_y_prime = np.maximum(0, r - (1 - y))
    return -(np.sum(r) + λ * (np.sum(eta_ij**p) + np.sum((beta_x + beta_x_prime + beta_y + beta_y_prime)**p)))

def compute_gradient(z, λ, p):
    n = z.shape[0] // 3
    d_ij = np.sqrt(np.maximum((z[0::3, None, :] - z[0::3, :])**2 + (z[1::3, None, :] - z[1::3, :])**2, 1e-12))
    d_ij[range(n), range(n)] = np.inf
    eta_ij = np.maximum(0, z[2::3, None, :] + z[2::3, :][:, None] - d_ij)
    beta_x = np.maximum(0, z[2::3] - z[0::3])
    beta_x_prime = np.maximum(0, z[2::3] - (1 - z[0::3]))
    beta_y = np.maximum(0, z[2::3] - z[1::3])
    beta_y_prime = np.maximum(0, z[2::3] - (1 - z[1::3]))
    n_ij = np.sign(eta_ij)
    eta_loss = λ * p * eta_ij**(p - 1) * n_ij
    n_x = np.where(beta_x > 0, -1, 0)
    beta_x_loss = λ * p * beta_x**(p - 1) * n_x
    n_x_prime = np.where(beta_x_prime > 0, 1, 0)
    beta_x_prime_loss = λ * p * beta_x_prime**(p - 1) * n_x_prime
    n_y = np.where(beta_y > 0, -1, 0)
    beta_y_loss = λ * p * beta_y**(p - 1) * n_y
    n_y_prime = np.where(beta_y_prime > 0, 1, 0)
    beta_y_prime_loss = λ * p * beta_y_prime**(p - 1) * n_y_prime
    grad = np.zeros_like(z)
    np.add.at(grad, np.r_[0::3, 1::3, 2::3], eta_loss.sum(axis=1) + beta_x_loss + beta_x_prime_loss + beta_y_loss + beta_y_prime_loss + (-np.ones_like(r)))
    np.add.at(grad[0::3], np.where(eta_ij > 0), d_ij * (z[2::3] + eta_ij - d_ij) / d_ij**2)
    np.add.at(grad[1::3], np.where(eta_ij > 0), d_ij * (z[2::3] + eta_ij - d_ij) / d_ij**2)
    np.add.at(grad[2::3], np.where(eta_ij > 0), 2 * (z[2::3] + eta_ij - d_ij) / d_ij)
    np.add.at(grad[2::3], np.where(beta_x > 0), -1)
    np.add.at(grad[0::3], np.where(beta_x > 0), -1)
    np.add.at(grad[2::3], np.where(beta_x_prime > 0), 1)
    np.add.at(grad[0::3], np.where(beta_x_prime > 0), 1)
    np.add.at(grad[2::3], np.where(beta_y > 0), -1)
    np.add.at(grad[1::3], np.where(beta_y > 0), -1)
    np.add.at(grad[2::3], np.where(beta_y_prime > 0), 1)
    np.add.at(grad[1::3], np.where(beta_y_prime > 0), 1)
    return grad

def lp_optimize_centres(centres):
    n = centres.shape[0]
    A = []
    b = []
    A.append(np.hstack([np.eye(n), np.zeros((n, n)), np.zeros((n, n))]))
    b.append(1)
    A.append(np.hstack([np.zeros((n, 2)), np.eye(n), np.zeros((n, n))]))
    b.append(1)
    A.append(np.hstack([np.zeros((n, 2)), np.zeros((n, n)), np.eye(n)]))
    b.append(1)
    A.append(np.hstack([np.zeros((n, 2)), np.zeros((n, n)), np.zeros((n, n))]))
    b.append(1)
    for i in range(n):
        for j in range(i + 1, n):
            A.append(np.hstack([np.zeros((1, 2)), np.zeros((1, 2*i)), np.ones((1, 2)), np.zeros((1, (n - 2*i - 2)))]))
            b.append(2 * centres[i, 2] + 2 * centres[j, 2] - np.linalg.norm(centres[i, :2] - centres[j, :2]))
    A = np.array(A)
    b = np.array(b)
    res = linprog([2] * n + [0] * n, A_eq=A, b_eq=b, bounds=(0, 1), method='highs')
    if res.message == LinprogMessage.SUCCESS:
        return res.x[n:], res.fun
    else:
        return np.zeros(n), np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    λ = 5e4
    p = 2
    tol = 1e-6
    maxfun = 2000
    init_starts = [np.random.rand(n * 3) for _ in range(18)]
    init_starts.extend([np.zeros(n * 3) for _ in range(3)])
    init_starts.extend([(np.arange(n) % 4 == i) + np.random.rand(n) * 0.001 for i in range(4)])
    best_sum = float('-inf')
    best_centres = None
    best_radii = None
    for start in init_starts:
        res = minimize(compute_penalty, start, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)] * n + [(0, 1)] * n + [(0, 1)] * n, options={'maxfun': maxfun, 'gtol': tol})
        if np.isclose(compute_gradient(res.x, λ, p).dot(res.x - start), 0, atol=tol) and compute_penalty(res.x, λ, p) < best_sum:
            best_sum = compute_penalty(res.x, λ, p)
            best_centres = res.x[0::3].reshape(n, 2)
            best_radii = np.maximum(0, res.x[2::3])
    if best_centres is None:
        best_centres = np.zeros((n, 2))
        best_radii = np.zeros(n)
    best_sum = lp_optimize_centres(best_centres)[1]
    return best_centres, best_radii, best_sum
