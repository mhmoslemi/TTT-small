import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def compute_penalty(z, λ, p):
    n = 26
    x = z[:2*n]
    y = z[2*n:]
    r = z[4*n:]
    
    eta = np.maximum(0, r[:,None] + r - distance.cdist(x.reshape(n, 2), x.reshape(n, 2)).flatten())
    beta_x = np.maximum(0, r - x)
    beta_x_prime = np.maximum(0, r - (1 - x))
    beta_y = np.maximum(0, r - y)
    beta_y_prime = np.maximum(0, r - (1 - y))
    
    return -(r.sum()) + λ * (np.power(eta, p).sum() + np.power(beta_x, p).sum() + np.power(beta_x_prime, p).sum() + 
                          np.power(beta_y, p).sum() + np.power(beta_y_prime, p).sum())

def compute_gradient(z, λ, p):
    n = 26
    x = z[:2*n]
    y = z[2*n:]
    r = z[4*n:]
    
    def dcdz(i, j, x, y, r):
        dx = (x[i] - x[j]) / distance.cdist(x[i:i+1, None], x[j:j+1, None])[0][0]
        dy = (y[i] - y[j]) / distance.cdist(y[i:i+1, None], y[j:j+1, None])[0][0]
        return np.array([dx, dy, dx, dy])

    dpenalty = np.zeros(4*n)
    
    for i in range(n):
        for j in range(i+1, n):
            if (r[i] + r[j] - distance.cdist(x[i:i+1, None], x[j:j+1, None])[0][0]) > 1e-6:
                dpenalty[i*2:i*2+2] += dcdz(i, j, x, y, r)
                dpenalty[j*2:j*2+2] += dcdz(j, i, x, y, r)

        if r[i] - x[i] > 1e-6:
            dpenalty[i*2] += 1.0
        
        if x[i] - r[i] > 1e-6:
            dpenalty[i*2] -= 1.0

        if r[i] - y[i] > 1e-6:
            dpenalty[i*2+1] += 1.0

        if y[i] - r[i] > 1e-6:
            dpenalty[i*2+1] -= 1.0

    return -dpenalty * λ * p

def lp_optimize_centres(centres):
    n = 26
    r = np.zeros(n)
    for i in range(n):
        x, y = centres[i]
        margin = min(x, 1-x, y, 1-y)
        r[i] = margin
    from scipy.optimize import linprog
    A = np.zeros((2*n+n, 2*n+n))
    b = np.zeros(2*n+n)
    
    for i in range(n):
        A[i, 2*i:2*i+2] = -np.array([1, 0])
        A[i, 4*i+2*n] = -1
        b[i] = 0
        A[i+n, 2*i:2*i+2] = -np.array([0, 1])
        A[i+n, 4*i+2*n] = -1
        b[i+n] = 0
        A[i+2*n, 2*i:2*i+2] = np.array([1, 0])
        A[i+2*n, 4*i+2*n] = 1
        B_x, B_y = np.meshgrid(range(n), range(n))
        B_x = B_x.flatten()
        B_y = B_y.flatten()
        mask = B_x != B_y
        A[i+2*n+sum(mask), 2*B_x[mask] + 2:B_x[mask] + 3] = 1
        A[i+2*n+sum(mask), 2*B_y[mask] + 2:B_y[mask] + 3] = 1
        A[i+2*n+sum(mask), 4*B_x[mask]+2*n] = 1
        B_x, B_y = np.meshgrid(range(n), range(n))
        B_x = B_x.flatten()
        B_y = B_y.flatten()
        mask = B_x != B_y
        A[i+2*n+sum(mask), 2*B_x[mask] + 2:B_x[mask] + 3] = 1
        A[i+2*n+sum(mask), 2*B_y[mask] + 2:B_y[mask] + 3] = 1
        A[i+2*n+sum(mask), 4*B_x[mask]+2*n] = 1
    
    res = linprog(-r, A_ub=A, b_ub=b, method='highs')
    return res.x

def run_packing() -> (np.ndarray, np.ndarray, float):
    λ = 5e4
    p = 2
    max_evals = 2000
    n_trials = 18
    best_sum = 0
    best_centers = None
    best_r = None
    
    def objective(z):
        return compute_penalty(z, λ, p)
    
    def constraint(z):
        return compute_penalty(z, λ, p) >= 0
    
    def callback(xk):
        nonlocal best_sum, best_centers, best_r
        penalty = compute_penalty(xk, λ, p)
        if penalty < best_sum:
            best_sum = penalty
            best_centers = xk[:4*n].reshape(2*n, 2)
            best_r = xk[4*n:]
    
    bounds = [(0, 1) for _ in range(78)]
    initial_conditions = [
        np.concatenate([np.random.random(2*n), 0.01*np.ones(26)]),  # Random
        np.concatenate([1./3*np.arange(2*n).reshape(2*n, 2), 0.01*np.ones(26)]),  # Hexagonal
        np.concatenate([np.array([[0.2, 0.2], [0.2, 0.8], [0.8, 0.2], [0.8, 0.8]]).reshape(2*n, 2), 0.01*np.ones(26)])  # Corner-heavy
    ]
    
    for seed in np.random.randint(0, 1000, n_trials):
        np.random.seed(seed)
        for centers in initial_conditions:
            result = minimize(objective, centers, bounds=bounds, method='L-BFGS-B', options={'maxfun': max_evals}, callback=callback)
            if result.fun < best_sum:
                best_sum = result.fun
                best_centers = result.x[:4*n].reshape(2*n, 2)
                best_r = result.x[4*n:]
    
    lp_r = lp_optimize_centres(best_centers)
    return best_centers, lp_r, -best_sum
