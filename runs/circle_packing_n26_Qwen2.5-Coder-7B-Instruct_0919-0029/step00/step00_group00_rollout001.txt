import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def run_packing() -> (np.ndarray, np.ndarray, float):
    NUM_STARTS   = 25
    INIT_R       = 0.01
    SIM_ITERS    = 2500
    EPS          = 1e-12

    def solve_lp_for_centres(centres):
        n = centres.shape[0]
        A_eq = np.zeros((n, n))
        for i in range(n):
            A_eq[i, i] = -1
            for j in range(i + 1, n):
                dist = np.linalg.norm(centres[i] - centres[j])
                A_eq[i, j] = 1
                A_eq[j, i] = 1
        A_eq = np.append(A_eq, np.eye(n), axis=1)
        b_eq = np.zeros(n) - EPS
        b_eq = np.append(b_eq, np.ones(n))
        c = np.ones(n)
        
        res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
        if res.success:
            return res.x[:n], res.fun
        else:
            return np.full(n, 0), 0.0

    best_centres = np.empty((0, 2))
    best_radii = np.empty((0,))
    best_sum = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)
        radii = np.full(26, INIT_R)
        prev_centres = np.zeros((26, 2))
        
        for iter in range(SIM_ITERS):
            dists = cdist(centres, centres, 'euclidean')
            overlaps = (dists - np.eye(26) * (radii[:, np.newaxis] + radii)) > 0
            overlap_dirs = np.where(overlaps[:, :, np.newaxis],
                                   (centres[:, np.newaxis, :] - centres) / dists[:, :, np.newaxis],
                                   np.zeros_like(centres[:, :, np.newaxis]))
            overlap_shifts = 0.5 * overlaps * overlap_dirs
            shift = np.sum(overlap_shifts, axis=1) / 2
            centres += shift
            
            # Boundary correction
            centroids = np.sum(overlap_shifts, axis=1) / np.sum(overlaps, axis=1)[:, np.newaxis]
            bounds = np.clip(radii[:, np.newaxis] - (centres - centroids), 0, None)
            bounds[:, 0] = np.clip(bounds[:, 0], 0, centroids[:, 0] - radii)
            bounds[:, 1] = np.clip(bounds[:, 1], centroids[:, -1] + radii, 1)
            bounds[:, 2] = np.clip(bounds[:, 2], 0, centroids[:, -1] - radii)
            bounds[:, 3] = np.clip(bounds[:, 3], centroids[:, 0] + radii, 1)
            right_shifts = bounds[:, 0] - radii
            left_shifts = bounds[:, 1] - (centres[:, 0] - centroids[:, 0])
            top_shifts = bounds[:, 2] - radii
            bottom_shifts = bounds[:, 3] - (centres[:, 1] - centroids[:, 1])
            
            been_adjacent = overlaps
            for i in range(4):
                need_adj = np.any(been_adjacent, axis=0)
                if been_adjacent.shape[1] > 0:
                    popped_indices = np.argwhere(np.all(np.any(been_adjacent, axis=0), axis=1)[:, 0] == 0)
                    if possibly_non_empty(popped_indices[-1]):
                        enable_right, left, top, bottom, fans_out, optimal_r = step(clockwise, candidates, new_positions, adjacents, diags, remaining_capacity)
                        candidate_pos = choose_adjacent(right_ne and r < optimal_r, top_ne and top < optimal_r, bottom_ne and bottom < optimal_r, left_ne and left < optimal_r)
                        candidate_pos = np.array([candidate_pos])
                    else:
                        candidate_pos = optimal_positions
                    if not np.array_equal(candidate_pos, new_positions):
                        replace_people(candidate_pos, adjacents, remaining_capacity, clock, candidate_grid, total_population, new_state)
                    been_adjacent = new_adjacents

            overlap_remaining = overlaps.any()
            if not overlap_remaining:
                break
            
        opt_radii, opt_sum = solve_lp_for_centres(centres)
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centres = centres.copy()
            best_radii = opt_radii.copy()

    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
