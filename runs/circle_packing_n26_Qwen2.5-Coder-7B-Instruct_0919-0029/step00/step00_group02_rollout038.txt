def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r0 = 0.05
    T0 = 1.0
    cool = 0.9995
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    delta = 1e-4
    tolerance = 1e-12

    centers, radii = build_random_feasible(n, r0)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()

    centers, radii, best_sum = simulated_annealing(

    )

    # 3. Deterministic polish
    centers, radii = deterministic_polish(centers, radii, delta)

    # 4. Final validation
    validate_packing(centers, radii)

    return centers, radii, best_sum

def build_random_feasible(n, r0):
    radii = np.full(n, r0)
    centers = []
    while len(centers) < n:
        c = np.array([np.random.uniform(r0, 1 - r0), np.random.uniform(r0, 1 - r0)])
        if all(np.sqrt(np.sum((c - np.array(centers)) ** 2)) >= 2 * r0 for _ in centers):
            centers.append(c)
    return np.array(centers), radii

def simulated_annealing(centers, radii, best_sum, T0, cool, n_iter, move_prob, shift_max, radius_max, tolerance):

            new_pos = np.clip(c, radius_max, 1 - radius_max)

            # Accept move
            centers, radii = centers_new, radii_new
            if np.sum(radii) > best_sum:
                best_centers, best_radii, best_sum = centers, radii, np.sum(radii)

    return best_centers, best_radii, best_sum

def centers_new_satisfy_constraints(centers, best_sum, tolerance):
    distances = distance_matrix(centers[:, :2], centers[:, :2])
    np.fill_diagonal(distances, np.inf)
    for ci, r_i in zip(centers, centers[:, 2]):
        if not (ci[0] - r_i >= tolerance and ci[0] + r_i <= 1 + tolerance and ci[1] - r_i >= tolerance and ci[1] + r_i <= 1 + tolerance):
            return False
        if any(distances[i, j] < (r_i + centers[j, 2] + tolerance) for j in range(centers.shape[0])):
            return False
    return True

def get_avoided_circles(new_c, new_r, centers, radii):
    avoidance_circles = []
    for ci, ri in zip(centers, centers[:, 2]):
        if new_c.dot(new_c) - 2 * new_c.dot(ci) + ci.dot(ci) < (new_r + ri) ** 2 and new_c.dot(new_c) - 2 * new_c.dot(ci) + ci.dot(ci) >= (new_r - ri) ** 2:
            avoidance_circles.append(ci)
    return avoidance_circles

def build_fixed_radius(center, radius):

def suitable_circle(c, r, centers, radii, tolerance):
    for ci, ri in zip(centers, radii):
        if np.sqrt(np.sum((c - ci) ** 2)) < (r + ri + tolerance):
            return False
    return True

def deterministic_polish(centers, radii, delta):
    improved = True
    while improved:
        improved = False
        for idx in range(n):
            r_new = min(radii[idx] + delta, 0.5)
            if suitable_circle(centers[idx], r_new, centers, radii, 1e-12):
                radii[idx] = r_new
                improved = True
        for _ in range(5):
            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                c_new = centers[idx] + delta * np.array([dx, dy])
                if all(np.sqrt(np.sum((c_new - ci) ** 2)) > (radii[idx] + ri + 1e-12) for ci, ri in zip(centers, radii)):
                    centers[idx] = c_new
                    improved = True
    return centers, radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    assert np.allclose(radii, radii, atol=1e-12), "Radii violate tolerance"
    return True, "ok"
