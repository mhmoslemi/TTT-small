import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    c = np.reshape(centers, (n*2,))
    A = np.empty((n, 2*n))
    for i in range(n):
        A[i, 2*i] = 1
        A[i, 2*i + 1] = -1
        for j in range(n):
            if i != j:
                A[i, 2*j] = -1
                A[i, 2*j + 1] = -1
    
    b = np.empty(n)
    for i in range(n):
        b[i] = np.minimum(np.min(np.abs(centers[i] - np.array([0, 1]))), np.linalg.norm(centers[i] - centers))

    result = linprog(-b, A_ub=A, b_ub=-b, bounds=(0, None))
    if result.success:
        return -result.fun, centers
    else:
        return 0, centers

def gradient_step(centers, radius_sum):
    n = centers.shape[0]
    eps = 1e-4
    positions = np.reshape(centers, (n*2,))
    gradients = np.zeros_like(positions)
    
    for i in range(n*2):
        e = np.zeros_like(positions)
        e[i] = eps
        positions_plus = positions + e
        positions_minus = positions - e
        r_plus, centers_plus = compute_radii(positions_plus.reshape((n, 2)))
        r_minus, centers_minus = compute_radii(positions_minus.reshape((n, 2)))
        gradients[i] = (r_plus - r_minus) / (2 * eps)
    
    alpha = 0.01
    new_positions = positions + alpha * gradients
    new_positions = np.clip(new_positions, 0, 1)
    new_centers = new_positions.reshape((n, 2))
    return new_centers, radius_sum
