import numpy as np
import random
from scipy.spatial import distance

def build_random_feasible(n: int, r0: float):
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    attempts = 0
    while attempts < 10000:
        for i in range(n):
            while True:
                center = [random.random(), random.random()]
                if all(center[j] - radii[j] >= 0 and 
                       center[j] + radii[j] <= 1 for j in range(2)):
                    if all(distance.euclidean(center, centers[j]) >= radii[j] + radii[i] for j in range(i)):
                        centers[i] = center
                        break
        return centers, radii
    raise Exception("Failed to find a feasible start")

def simulated_annealing(centers, radii, best_sum):
    T0, cool, n_iter, move_prob, shift_max, radius_max, n_shift = 1.0, 0.9995, 200_000, 0.7, 0.1, 0.08, 1000
    best_centers, best_radii = centers.copy(), radii.copy()
    N = centers.shape[0]
    
    for k in range(n_iter):
        T = T0 * (cool ** k)
        i = random.randint(0, N - 1)
        if random.random() < move_prob:
            # Shift move
            dx, dy = random.uniform(-shift_max, shift_max), random.uniform(-shift_max, shift_max)
            new_pos = [centers[i][0] + dx, centers[i][1] + dy]
            if all(new_pos[j] - radii[i] >= 0 and new_pos[j] + radii[i] <= 1 for j in range(2)):
                if all(distance.euclidean(new_pos, centers[j]) >= radii[j] + radii[i] for j in range(i)):
                    centers[i] = new_pos
        else:
            # Radius move
            dr = random.uniform(-radius_max, radius_max)
            new_r = max(0.0, radii[i] + dr)
            new_r = min(new_r, min(centers[i][0], 1 - centers[i][0],
                                   centers[i][1], 1 - centers[i][1]))
            if all(new_r < radii[j] for j in range(i) if i != j) and all(distance.euclidean(centers[j], centers[i]) >= new_r + radii[j] - 1e-12 for j in range(i)):
                radii[i] = new_r
        
        if validate_packing(centers, radii)[0]:
            new_sum = radii.sum()
            if new_sum > best_sum:
                best_sum, best_centers, best_radii = new_sum, centers, radii.copy()
            elif random.random() < np.exp((new_sum - best_sum) / T):
                best_sum = new_sum
        else:
            centers[i] = best_centers[i]
            radii[i] = best_radii[i]
        
        if k % n_shift == n_shift - 1:
            shift_max *= 0.95
            radius_max *= 0.95
    
    return best_centers, best_radii, best_sum

def deterministic_polish(centers, radii):
    n = centers.shape[0]
    improved = True
    while improved:
        improved = False
        for i in range(n):
            while True:
                r_try = radii[i] + 1e-4
                if all(centers[i][j] - r_try >= 0 and centers[i][j] + r_try <= 1 for j in range(2)) and all(distance.euclidean(centers[i], centers[j]) >= r_try + radii[j] for j in range(n) if i != j):
                    radii[i] = r_try
                    improved = True
                else:
                    break
    return centers, radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = distance.euclidean(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii = build_random_feasible(26, r0=0.05)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()
    centers, radii, best_sum = simulated_annealing(centers, radii, best_sum)
    centers, radii = deterministic_polish(centers, radii)
    validate_packing(centers, radii)
    return centers, radii, best_sum
