import numpy as np
from scipy.optimize import linprog, minimize
from itertools import combinations
import math

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def energy(z, lambda_val=5e4, p=2):
    n = int(len(z) / 3)
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    total_energy = -np.sum(radii)
    overlap_penalties = 0
    boundary_penalties = 0
    
    for i, j in combinations(range(n), 2):
        dist = np.linalg.norm(centers[i] - centers[j])
        if dist < radii[i] + radii[j]:
            overlap_penalties += (radii[i] + radii[j] - dist)**p
    
    for i in range(n):
        for coordinate in [0, 1]:
            penalty = max(0, radii[i] - min(centers[i, coordinate], 1 - centers[i, coordinate]))
            boundary_penalties += penalty

    return total_energy + lambda_val * (overlap_penalties + boundary_penalties)

def initial_state(shape, seed=None):
    np.random.seed(seed)
    radii = 0.01 * np.ones(26)
    centers = np.random.rand(26, 2)
    return np.concatenate([centers[:, 0], centers[:, 1], radii])

def find_centers(max_iter=20000, stepsize=4e-1, lambda_val=5e4, p=2, restarts=20):
    best_centers = None
    best_energy = float('inf')

    for _ in range(restarts):
        z = initial_state('hexagonal', seed=np.random.randint(0, 2**32 - 1))
        initial_energy = energy(z, lambda_val, p)

        def objective(z):
            return energy(z, lambda_val, p)

        # Simulated Annealing
        T = 0.5
        for k in range(max_iter):
            z_new = z.copy()
            idx = np.random.randint(0, 26)
            dx, dy = np.random.normal(0, stepsize, size=2)
            new_center = np.clip(z_new[idx*2:idx*2+2] + dx, 0, 1)
            new_radius = np.clip(z_new[idx*2+2] + np.random.normal(0, stepsize), 0, 1)
            z_new[idx*2:idx*2+2] = new_center
            z_new[idx*2+2] = new_radius
            new_energy = energy(z_new, lambda_val, p)

            if new_energy <= initial_energy:
                z = z_new.copy()
                initial_energy = new_energy
            elif np.random.rand() < math.exp(-(new_energy - initial_energy) / T):
                z = z_new.copy()
                initial_energy = new_energy

            T *= 0.9995

            if new_energy < best_energy:
                best_energy = new_energy
                best_centers = z

    return best_centers

def final_optimization_fixed_centers(centers, lambda_val=5e4):
    n = centers.shape[0]
    constraints = []
    bounds = [(0, None) for _ in range(n)]  # Non-negativity of radii

    for i, j in combinations(range(n), 2):
        dist = np.linalg.norm(centers[i] - centers[j])
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, dist=dist: dist - r[i] - r[j]})
    
    for idx in range(n):
        for coordinate in [0, 1]:
            constraints.append({'type': 'ineq', 'fun': lambda r, idx=idx, coordinate=coordinate: r[idx] - min(centers[idx, coordinate], 1 - centers[idx, coordinate])})
    
    initial_radius_guess = [0.01] * n
    initial_guess = np.concatenate([centers.flatten(), initial_radius_guess])
    
    res = minimize(lambda x: -lambda_val * energy(x, lambda_val, p=2), initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)
    best_radii = res.x[-n:]
    return best_radii

def run_packing():
    best_centers = find_centers()
    best_centers = best_centers.reshape(26, 3)[:,:2]
    best_radii = final_optimization_fixed_centers(best_centers)
    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii
