import numpy as np
from scipy.optimize import minimize, SLSQP


def _make_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i] - x[3 * i + 2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3 * i] - x[3 * i + 2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i + 1] - x[3 * i + 2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3 * i + 1] - x[3 * i + 2]})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.hypot(x[3 * i] - x[3 * j],
                                                                            x[3 * i + 1] - x[3 * j + 1]) - x[
                                                                 3 * i + 2] - x[3 * j + 2]})
    return cons


def _obj(x):
    return -np.sum(x[2::3])


def _grow_step(x, idx):
    n = x.shape[0] // 3
    i, j = idx
    delta = 1e-4
    while True:
        x_new = x.copy()
        x_new[3 * i + 2] += delta
        if all([0 <= x_new[3 * k] - x_new[3 * k + 2] for k in range(n)]) and all(
                [x_new[3 * k] + x_new[3 * k + 2] <= 1 for k in range(n)]) and all(
                [np.hypot(x_new[3 * k] - x_new[3 * j], x_new[3 * k + 1] - x_new[3 * j + 1]) - x_new[3 * k + 2] - x_new[
                    3 * j + 2] >= 0 for k in range(n) if k != i] + [np.hypot(x_new[3 * i] - x_new[3 * k], x_new[3 * i + 1] - x_new[3 * k + 1]) - x_new[3 * i + 2] - x_new[
                    3 * k + 2] >= 0 for k in range(n) if k != j]):
            return x_new
        delta -= 1e-5


def _shake_and_refine(x):
    n = x.shape[0] // 3
    for _ in range(20):
        for i in range(n):
            dx = 5e-4 * np.random.uniform(-1, 1)
            dy = 5e-4 * np.random.uniform(-1, 1)
            x_new = x.copy()
            x_new[3 * i] += dx
            x_new[3 * i + 1] += dy
            if all([0 <= x_new[3 * k] - x_new[3 * k + 2] for k in range(n)]) and all(
                    [x_new[3 * k] + x_new[3 * k + 2] <= 1 for k in range(n)]) and all(
                    [np.hypot(x_new[3 * k] - x_new[3 * l], x_new[3 * k + 1] - x_new[3 * l + 1]) - x_new[3 * k + 2] - x_new[
                        3 * l + 2] >= 0 for k in range(n) if k != i] for l in range(n) if l != i]:
                x = _grow_step(x_new, (i, i))
    return x


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r0 = 0.09
    centers = np.zeros((n, 2))
    radii = np.zeros(n) + r0
    x0 = np.concatenate((centers.flatten(), radii))
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    for k in range(5):
        x0 = x0.copy()
        for i in range(n):
            x0[3 * i:3 * i + 2] += (0.4 * np.random.rand(2) - 0.2)
            x0[3 * i + 2] = max(0.01, min(0.15, x0[3 * i + 2]))
        res = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                     constraints=_make_constraints(n), options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
        x0 = res.x
    x0 = _shake_and_refine(x0)
    best_centers = x0[:2 * n].reshape((n, 2))
    best_radii = x0[2 * n:]
    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii
