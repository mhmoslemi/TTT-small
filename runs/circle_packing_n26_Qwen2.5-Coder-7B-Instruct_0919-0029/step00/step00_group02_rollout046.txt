import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    grid_size = 1 / k
    centers = np.array([(i * grid_size, j * grid_size) for i in range(k) for j in range(k)])
    R_max = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1]))
    solver = pywraplp.Solver.CreateSolver('SCIP')
    b_vars = {p: solver.IntVar(0, 1, f'b_{p}') for p in centers}
    r_vars = {p: solver.NumVar(0, R_max[grid centres.index(p)], f'r_{p}') for p in centers}
    
    for p in centers:
        solver.Add(r_vars[p] <= b_vars[p] * R_max[grid centres.index(p)])
    
    solver.Add(solver.Sum([b_vars[p] for p in centers]) == 26)
    
    M = 2.0
    for p, q in get_potentially_colliding_pairs(centers, R_max):
        solver.Add(r_vars[p] + r_vars[q] <= np.linalg.norm(p - q) + M * (2 - b_vars[p] - b_vars[q]))
    
    objective = solver.Objective()
    objective.SetMaximization()
    for p in centers:
        objective.SetCoefficient(r_vars[p], 1)
    
    return solver, r_vars, b_vars

def get_potentially_colliding_pairs(centers, R_max):
    pairs = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < 2 * R_max[i] - 1e-12 and dist * 2 < r_max[i] + r_max[j]:
                pairs.append((centers[i], centers[j]))
    return pairs

def solve_milp(solver, time_limit=600):
    solver.SetTimeLimit(time_limit * 1000)  # Convert to milliseconds
    status = solver.Solve()
    best_solution = {p: (r_vars[p].solution_value(), b_vars[p].solution_value()) for p in centers}
    return status, best_solution

def extract_solution(best_solution, vars_by_grid):
    centers = []
    radii = []
    for p, (r, b) in best_solution.items():
        if b > 0:
            centers.append(p)
            radii.append(r)
    return np.array(centers), np.array(radii)

def polish_solution(centers, radii, delta_r=1e-4, epsilon=1e-4):
    new_centers, new_radii = centers.copy(), radii.copy()
    for i, (c, r) in enumerate(zip(centers, radii)):
        while True:
            new_r = r + delta_r
            valid = True
            for j, (c2, r2) in enumerate(zip(centers, radii)):
                if j != i and overlap(c, r, c2, r2):
                    valid = False
                    break
            if valid:
                new_radii[i] = new_r
            else:
                break
        for _ in range(3):  # Let's say 3 passes for completeness
            new_c = c + np.array([np.random.uniform(-epsilon, epsilon), np.random.uniform(-epsilon, epsilon)])
            for j, (c2, r2) in enumerate(zip(centers, radii)):
                if j != i and overlap(new_c, new_radii[i], c2, r2):
                    new_c -= 2 * (new_c - c2) / np.linalg.norm(new_c - c2)  # Force to stay valid
            new_centers[i] = np.clip(new_c, new_radii[i], 1 - new_radii[i])
    return new_centers, new_radii

def overlap(c1, r1, c2, r2):
    dist = np.linalg.norm(c1 - c2)
    return dist < (r1 + r2) - 1e-12

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
