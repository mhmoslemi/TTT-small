    centers = np.array([
        [0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)
    ] + [[0.5, 0.5]])
    radii = np.zeros(n)
    
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    eps = 1e-9
    
    for i in range(max_iter):
        new_radii = radii + g
        
        conflicts = []
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(centers[i] - centers[j])
                if new_radii[i] + new_radii[j] > d_ij:
                    conflicts.append((i, j, new_radii[i] + new_radii[j] - d_ij))
            r_i = new_radii[i]
            if r_i > centers[i][0] or r_i > 1 - centers[i][0] or r_i > centers[i][1] or r_i > 1 - centers[i][1]:
                conflicts.append((i, None, r_i))
        
        for i, j, _ in conflicts:
            if j is not None:
                # Find the tightest violating circle neighbour and wall
                r_j = new_radii[j]
                c_ij = centers[i] - centers[j]
                dist_ij = np.linalg.norm(c_ij)
                r_ij_max = dist_ij / 2

                r_i = min(min(np.linalg.norm(centers[i] - centers[k]) / 2 for k in range(n) if k != i), centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])

        else:
