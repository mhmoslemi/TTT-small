import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r0 = 0.09
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    
    def _make_constraints(centers, radii):
        bounds = [(0, 1)] * 2 * n + [(0, r0)] * n
        
        def _distance(i, j):
            return np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2)
        
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i] - x[3 * i + 2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3 * i] - x[3 * i + 2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i + 1] - x[3 * i + 2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3 * i + 1] - x[3 * i + 2]})
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: _distance(i, j) - x[3 * i + 2] - x[3 * j + 2]})
        return cons, bounds
    
    def _obj(x):
        centers_flat = x[:2 * n]
        centers = centers_flat.reshape((n, 2))
        radii = x[2 * n:]
        return -np.sum(radii)  # Maximize sum of radii
    
    def optimize_with_initial_guess(x0):
        centers_flat = x0[:2 * n]
        centers = centers_flat.reshape((n, 2))
        cons, bounds = _make_constraints(centers, radii)
        res = minimize(_obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 10000, 'disp': True})
        return res
    
    best_centers = centers.flatten()
    best_radii = radii
    best_sum = -_obj(best_centers)
    
    for _ in range(5):
        centers = np.zeros((n, 2))
        for i in range(5):
            for j in range(5):
                centers[i * 5 + j] = np.array([0.1 + 0.2 * i, 0.1 + 0.2 * j])
        centers[25] = np.array([0.5, 0.5])
        radii = np.full(n, r0)
        centers_flat = centers.flatten()
        x0 = np.concatenate((centers_flat, radii))
        res = optimize_with_initial_guess(x0)
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x[:2 * n]
            best_radii = res.x[2 * n:]
    
    centers = best_centers.reshape((n, 2))
    radii = best_radii
    
    # Local refinement
    for _ in range(20):
        grow_step = np.array([np.random.uniform(-1e-4, 1e-4, size=2) for _ in range(n)])
        new_centers = centers + grow_step
        new_centers = np.clip(new_centers, 0, 1)  # Keep within bounds
        good_indices = np.all(_make_constraints(new_centers, radii)[0], axis=1)
        centers[good_indices] = new_centers[good_indices]
        
        shake_step = np.random.uniform(-5e-4, 5e-4, size=(n, 2))
        new_centers = centers + shake_step
        new_centers = np.clip(new_centers, 0, 1)  # Keep within bounds
        good_indices = np.all(_make_constraints(new_centers, radii)[0], axis=1)
        centers[good_indices] = new_centers[good_indices]
    
    return centers, radii, np.sum(radii)

# Run the packing algorithm and print the result
centers, radii, sum_radii = run_packing()
print("Centers:\n", centers)
print("Radii:\n", radii)
print("Sum of Radii:", sum_radii)
