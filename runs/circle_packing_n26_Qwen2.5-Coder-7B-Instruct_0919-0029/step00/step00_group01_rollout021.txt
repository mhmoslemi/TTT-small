import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2).clip(max=1)   # in bounds for Euclidean
    constraints = []
    for i in range(n):
        constraints.append([-1, 0, borders[i]])
    for i in range(n):
        for j in range(i+1, n):
            constraints.append([1, 1, distances[i,j]**2])
    A_ub = np.vstack(constraints)
    b_ub = np.hstack([-borders, -distances**2])
    c = -np.ones(n)
    bounds = np.array([[0, None] for _ in range(n)])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def random_move(centers, step):
    rng = np.random.default_rng(42)
    i = rng.choice(centers.shape[0])
    x = centers[i][0]
    y = centers[i][1]
    dx = rng.normal(scale=step, size=1)
    dy = rng.normal(scale=step, size=1)
    nx = np.clip(x + dx, 0, 1)
    ny = np.clip(y + dy, 0, 1)
    new_centers = centers.copy()
    new_centers[i] = [nx, ny]
    return new_centers

def simulated_annealing_centers(n, max_iter, step):
    rng = np.random.default_rng(42)
    centers = rng.random((n, 2))
    best_centers = centers.copy()
    current_radii, best_sum = solve_lp(centers)
    best_sum = best_sum + 1e-6  # tiny to bypass rounding with radii result
    current_sum = best_sum
    T = 0.02
    T_end = 1e-5
    step_size = step

    for iter in range(max_iter):
        new_centers = random_move(centers, step_size)
        new_radii, sum_radii = solve_lp(new_centers)
        
        if sum_radii > current_sum:
            centers = new_centers
            current_sum = sum_radii
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = new_centers
        else:
            if rng.random() < np.exp((current_sum - sum_radii) / T):
                centers = new_centers

        T = T * (T_end / T) ** (iter / max_iter)
        if iter % 1000 == 0:
            step_size *= 0.95  # Adapting step size for each 1000 iteration.

    return best_radii, best_sum, best_centers

def run_packing():
    n = 26
    max_iter = 30000
    step = 0.05
    radii, sum_radii, centers = simulated_annealing_centers(n, max_iter, step)
    return centers, radii, sum_radii
