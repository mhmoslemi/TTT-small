import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = -np.inf

    def calculate_energy(z):
        x = z[:n]
        y = z[n:2*n]
        r = z[2*n:]
        E = -np.sum(r)
        penalties = np.sum((r[:, None] + r[None, :] - np.linalg.norm(x[:, None] - x[None, :], 2, 1))**2 * 
                           (r[:, None] + r[None, :] > np.linalg.norm(x[:, None] - x[None, :], 2, 1) - 1e-12), (0, 1)) +
                 np.sum(np.clip(r[:, None] - x[None, :], a_min=0, a_max=None), 1) +
                 np.sum(np.clip(r[:, None] - (1 - x[None, :]), a_min=0, a_max=None), 1) +
                 np.sum(np.clip(r[:, None] - y[None, :], a_min=0, a_max=None), 1) +
                 np.sum(np.clip(r[:, None] - (1 - y[None, :]), a_min=0, a_max=None), 1)
        return E + 5e4 * penalties

    def proposal(centers, radii, temp):
        i = np.random.randint(n)
        sigma = 0.04 * temp
        tau = 0.01 * temp
        new_centers = centers.copy()
        new_radii = radii.copy()
        new_centers[i] += np.random.normal(0, sigma, 2)
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        new_radii[i] += np.random.normal(0, tau, 1)
        new_radii[i] = np.clip(new_radii[i], 0, 1)
        return new_centers, new_radii

    def is_valid(centers, radii):
        for i in range(n):
            if not (radii[i] >= 0):
                return False
            if not (0 <= centers[i, 0] <= 1 and 0 <= centers[i, 1] <= 1):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                if np.linalg.norm(centers[i] - centers[j]) < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def sa_restart(centers, radii, temperature):
        for _ in range(20000):
            new_centers, new_radii = proposal(centers, radii, temperature)
            new_energy = calculate_energy(np.concatenate((new_centers.flatten(), new_radii)))
            current_energy = calculate_energy(np.concatenate((centers.flatten(), radii)))
            if new_energy <= current_energy or np.random.rand() < np.exp((current_energy - new_energy) / temperature):
                centers = new_centers.copy()
                radii = new_radii.copy()
                if is_valid(centers, radii):
                    if -new_energy > best_sum_radii:
                        best_sum_radii = -new_energy
                        best_centers = centers.copy()
                        best_radii = radii.copy()
        return centers, radii, best_sum_radii

    restarts = [
        np.random.rand(n, 2),
        np.array([np.cos(2*np.pi*i/6)*0.15, np.sin(2*np.pi*i/6)*0.15] for i in range(26)),
        np.array([[0, 0], [0, 1], [1, 0], [1, 1], [0.5, 0.5], *np.random.rand(20, 2)])
    ]
    for restart in restarts:
        centers = restart * (1 + 0.01 * np.random.randn(n, 2))
        radii = 0.01 * np.ones(n)
        centers, radii, best_sum_radii = sa_restart(centers, radii, 0.5)

    best_rays = linprog(-radii, A_ub=np.hstack((np.eye(n), -np.eye(n))), 
                         b_ub=np.sum(np.linalg.norm(best_centers[:, None] - best_centers[None, :], 2, 1), axis=0),
                         A_eq=np.hstack((best_centers, np.ones((n, 1)))), 
                         b_eq=n, 
                         A_bound=np.vstack((np.ones(n), np.zeros(n), np.ones(n), np.zeros(n))),
                         b_bound=(best_centers.flatten(), np.ones(n), 1 - best_centers.flatten(), np.ones(n)),
                         method='highs')
    best_radii = -best_rays.x
    best_sum_radii = -best_rays.fun

    return best_centers, best_radii, best_sum_radii
