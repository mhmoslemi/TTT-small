import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    eps = 1e-9

    np.random.seed(12345)

    # Seed the configuration
    centers = np.zeros((N, 2))
    i, j = 0, 0
    for k in range(N):
        if k == 25:
            centers[k] = np.array([0.5, 0.5])
        else:
            centers[k] = np.array([0.1 + 0.2 * i, 0.1 + 0.2 * j])
            i += 1
            if i == 5:
                i = 0
                j += 1

    radii = np.zeros(N)

    for iter in range(max_iter):
        # Proposed growth
        radii_new = radii + g

        # Constraint checking
        conflicts = []
        for i in range(N):
            for j in range(i + 1, N):
                d_ij = np.linalg.norm(centers[i] - centers[j])
                if radii_new[i] + radii_new[j] > d_ij:
                    conflicts.append((i, j, d_ij, radii_new[i], radii_new[j]))
            d_x = centers[i][0] + radii_new[i]
            if d_x > 1:
                conflicts.append((i, j, 1, radii_new[i], 0))
            d_x = 1 - centers[i][0] - radii_new[i]
            if d_x > 0:
                conflicts.append((i, j, 1, 0, radii_new[i]))
            d_y = centers[i][1] + radii_new[i]
            if d_y > 1:
                conflicts.append((i, j, 1, radii_new[i], 0))
            d_y = 1 - centers[i][1] - radii_new[i]
            if d_y > 0:
                conflicts.append((i, j, 1, 0, radii_new[i]))

        # Resolve conflicts
        for i, j, d_ij, r1_new, r2_new in conflicts:
            d_min = 1e-12
            if r1_new + r2_new < d_ij:
                continue
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            if dist_ij < 1e-12:
                continue
            d_over = (r1_new + r2_new) - dist_ij
            if d_over > d_min:
                dx = (centers[i] - centers[j])[0] / dist_ij * d_over
                dy = (centers[i] - centers[j])[1] / dist_ij * d_over
                if radii[r1_new] + radii[r2_new] < d_ij - 2 * d_over:
                    dx = 0
                    dy = 0
                centers[i][0] += dx
                centers[i][1] += dy
                centers[j][0] -= dx
                centers[j][1] -= dy
                r1_new = min(min(np.linalg.norm(centers[i] - centers[k]) for k in range(N) if k != i), 1 - centers[i][0], centers[i][0], 1 - centers[i][1], centers[i][1])
                radii[i] = r1_new

        # Check for convergence
        sum_radii_new = np.sum(radii_new)
        if iter > 0 and np.abs(sum_radii_new - sum_radii) < 1e-8:
            break
        sum_radii = sum_radii_new

        # Optional Shake-Period
        if iter % shake_rounds == 0 and iter > 0:
            k = np.random.randint(N)
            shake_x = np.random.uniform(-0.02, 0.02)
            shake_y = np.random.uniform(-0.02, 0.02)
            shake_x = np.clip(shake_x, 0, 1)
            shake_y = np.clip(shake_y, 0, 1)
            centers[k] += np.array([shake_x, shake_y])
            r1_new = min(min(np.linalg.norm(centers[k] - centers[l]) for l in range(N) if l != k), 1 - centers[k][0], centers[k][0], 1 - centers[k][1], centers[k][1])
            radii[k] = r1_new

    return centers, radii, np.sum(radii)

# Validation check (optional)
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

centers, radii, sum_radii = run_packing()
validity, message = validate_packing(centers, radii)
if not validity:
    print(message)
else:
    print("Validation passed.")
