import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)

    I = np.triu_indices(n, k=1)
    J = I[1]
    I = I[0]

    A_ub = np.zeros((n + I.size, n))
    b_ub = np.zeros(n + I.size)

    A_ub[:n, I] = 1
    A_ub[:n, J] = 1
    b_ub[:n] = borders

    # Pairwise constraints (radius sum <= distance)
    A_ub[n:, I] = 1
    A_ub[n:, J] = 1
    b_ub[n:] = distances[I, J]

    c = -np.ones(n)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
        if result.success:
            return result.x, -result.fun
        else:
            return np.zeros(n), 0
    except Exception:
        return np.zeros(n), 0

def random_move(centers, step):
    i = np.random.randint(26)
    new_pos = centers[i] + 0.01*(np.random.randn(2))
    new_pos = np.clip(new_pos, 0, 1)
    return new_pos

def simulated_annealing_centers(max_iter, rng):
    step = 0.05
    centers = rng.random((26, 2))
    best_state = (centers, None, None)

    for iter in range(max_iter):
        i = np.random.randint(26)
        new_centers = centers.copy()
        new_centers[i] = random_move(centers, step)

        radii, sum_radii = solve_lp(new_centers)

        if sum_radii > best_state[2]:
            best_state = (new_centers, radii, sum_radii)
        elif np.random.rand() < np.exp((sum_radii - best_state[2])/0.02):
            best_state = (new_centers, radii, sum_radii)

        step *= 0.9995

    return best_state[0], best_state[1], best_state[2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    max_iter = 30000
    best_centers, best_radii, best_sum = simulated_annealing_centers(max_iter, rng)
    return best_centers, best_radii, best_sum
