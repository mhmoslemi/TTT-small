import numpy as np
from scipy.optimize import minimize

def _make_constraints(n):
    def constraints(x):
        cons = []
        r = x[2::3]
        x = x[:2*n].reshape(-1, 2)
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i, 0] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i, 0] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i, 1] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i, 1] - r[i]})
        for i in range(n):
            for j in range(i + 1, n):
                mid_idx = 3 * i + 6 * j
                cons.append({
                    'type': 'ineq',
                    'fun': lambda x, mid_idx=mid_idx, i=i, j=j: np.hypot(x[i, 0]-x[j, 0], x[i, 1]-x[j, 1]) - (r[i] + r[j])
                })
        return cons
    return constraints

def _obj(x):
    return -np.sum(x[2::3])

def run_packing():
    n = 26
    r0 = 0.09
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * 26
    centers = np.array([(i + 0.1) * 0.2 for i in range(5)] + [(0.5, 0.5)] + [(i + 0.1) * 0.2 for j in range(5) for i in range(5)])
    radii = np.array([r0] * n)
    x0 = np.hstack((centers.flatten(), radii))
    
    def objective_and_constraints(x):
        x_reshaped = np.reshape(x[:2*n], (26, 2))
        r = x[2*n:]
        dists_squared = -np.sum(np.square(x_reshaped.reshape(-1, 1, 2) - x_reshaped.reshape(1, -1, 2)), axis=-1)
        centers_inside = (x_reshaped[:, 0] >= 0) & (x_reshaped[:, 0] <= 1) & (x_reshaped[:, 1] >= 0) & (x_reshaped[:, 1] <= 1)
        radii_nonnegative = r >= 0
        constraints = np.hstack((x_reshaped - r, 1 - x_reshaped - r, centers_inside.reshape(-1, 1) * radii_nonnegative.reshape(-1, 1), dists_squared - r[None, :] - r[:, None] - 1e-12)).flatten()
        return -np.sum(r), constraints
    
    res = minimize(objective_and_constraints, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda x: x}, options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
    centers = res.x[:2*n].reshape(-1, 2)
    radii = res.x[2*n:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
