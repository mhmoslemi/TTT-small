import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.ones((n*4, n)) * np.inf
    A_ub[:n, np.arange(n)] = 1
    A_ub[n:2*n, np.arange(n)] = -1
    A_ub[2*n:3*n, np.arange(n)] = -1
    A_ub[3*n:, np.arange(n)] = 1

    b_ub = np.zeros(4*n)
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1-x, y, 1-y)
        b_ub[n+i] = r_max

    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    d_ij = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            d_ij[i, j] = d_ij[j, i] = np.sqrt((x1-x2)**2 + (y1-y2)**2)
            A_eq[i*2, j] = -1
            A_eq[i*2, i] = 1
            A_eq[j*2, i] = -1
            A_eq[j*2, j] = 1

    res = linprog(-A_eq.sum(axis=1), A_ub=np.minimum(A_eq, A_ub), b_ub=b_eq, b_eq=np.zeros(n), bounds=(0, None), method='highs')
    radii = -res.fun
    return radii, res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.05
    step_min = 1e-6
    decay_factor = 0.995
    NUM_STARS = 30
    NUM_ITERS = 2000

    best_centers, best_sum, best_radii = np.random.rand(n,2), -np.inf, np.zeros(n)

    for start in range(NUM_STARS):
        centres = np.random.rand(n,2)
        for it in range(NUM_ITERS):
            idx   = np.random.randint(0, n)
            delta = np.random.uniform([-step,step], size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centres)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centres
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
