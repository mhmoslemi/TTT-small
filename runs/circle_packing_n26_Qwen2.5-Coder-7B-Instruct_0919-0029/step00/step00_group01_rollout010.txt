import numpy as np
from scipy.optimize import differential_evolution, minimize


def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    # boundary
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None)**2
                          + np.clip(xs - 1, 0, None)**2
                          + np.clip(-ys, 0, None)**2
                          + np.clip(ys - 1, 0, None)**2)
    if np.any(rs < 0):
        penalty += 1e7 * np.sum(np.clip(-rs, 0, None)**2)
    # pairwise
    for i in range(26):
        for j in range(i + 1, 26):
            d = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += 1e7 * gap**2
    return penalty


def de_objective(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    return -np.sum(rs) + compute_penalty(v)


def solve_lps_for_centres(centres):

    constraints = []
    for i in range(26):
        constraints.extend([
            {'type': 'ineq', 'fun': lambda r, i=i: np.min([centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1]]) - r},
            {'type': 'ineq', 'fun': lambda r, i=i: rs[i] - r, 'jac': lambda r, i=i: -1}
        ])
        for j in range(i + 1, 26):
            constraints.append({
                'type': 'ineq', 'fun': lambda r, i=i, j=j: R[i, j] - r[i] - r[j], 'jac': lambda r, i=i, j=j: np.array([-1, -1])
            })
    result = minimize(lambda r: -np.sum(r), rs, method='SLSQP', bounds=[(0, 0.5)] * 26, constraints=constraints)
    return result.x, -result.fun


def run_de_search(seed):
    bounds = [(0, 1)] * 52 + [(0, 0.25)] * 26
    result = differential_evolution(de_objective, bounds, strategy='intermediate', popsize=50, maxiter=2000, tol=1e-6, seed=seed)
    return result.x[:52]
