import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "Ok"

def objective(x):
    centers = x[:52].reshape(26, 2)
    radii = x[52:]
    return -np.sum(radii)

def constraint_pairwise_distances(x):
    centers = x[:52].reshape(26, 2)
    radii = x[52:]
    n = centers.shape[0]
    constr = []
    for i in range(n):
        for j in range(i + 1, n):
            distance = np.linalg.norm(centers[i] - centers[j])
            constr.append(radii[i] + radii[j] - distance)
    return np.array(constr)

min_bounds = (np.zeros(52), np.full(52, 1))
max_bounds = ((np.full(52, 0.5), np.full(52, 1)))

x0 = np.empty(78)
x0[:52] = np.array([(0.1 + 0.2 * (i % 5), 0.1 + 0.2 * (i // 5)) for i in range(25)])[[7*i + 2 for i in range(25)]] + [0.5, 0.5]
x0[52:] = np.repeat(0.1, 26)

constraints = []
constraints.append({'type': 'ineq', 'fun': lambda x: x[:52]}), # x_i >= 0
constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x[:52]}), # 1 - x_i >= 0
constraints.append({'type': 'ineq', 'fun': lambda x: x[52:]}), # r_i >= 0
constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x[52:]}), # 1 - r_i >= 0
constraints.append({'type': 'ineq', 'fun': constraint_pairwise_distances})

result = minimize(objective, x0, method='SLSQP', bounds=min_bounds + max_bounds, constraints=constraints, tol=1e-6)

radii = result['x'][52:]
centers = result['x'][:52].reshape(26, 2)

final_radii = np.minimum(radii, np.minimum(np.minimum(np.minimum(centers[:, 0] + radii, 1 - centers[:, 0] - radii),
                                                         np.minimum(centers[:, 1] + radii, 1 - centers[:, 1] - radii)),
                                                    np.minimum(np.min(np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :])**2, axis=2)), axis=1)/2, radii)))
radii = final_radii
sum_radii = np.sum(radii)
centers clip = np.clip(centers, 0, 1)
radii clip = np.clip(radii, 0, np.minimum(np.minimum(centers clip[:, 0] + radii clip, 1 - centers clip[:, 0] - radii clip),
                                          np.minimum(centers clip[:, 1] + radii clip, 1 - centers clip[:, 1] - radii clip)))

if not validate_packing(centers clip, radii clip)[0]:
    radii clip = radii clip * 0.999999

return centers clip, radii clip, sum_radii
