import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    Delta = 1 / k
    Rmax = np.minimum(np.arange(k), 1 - np.arange(k))[:, None]
    points = np.array([(i * Delta, j * Delta) for i in range(k) for j in range(k)])
    num_points = k**2
    model = pywraplp.Solver.CreateSolver('SCIP')
    r = [model.NumVar(0, Rmax.flat[i], f'r_{i}') for i in range(num_points)]
    b = [model.IntVar(0, 1, f'b_{i}') for i in range(num_points)]
    for g in range(num_points):
        model.Add(r[g] <= b[g] * Rmax.flat[g])
    model.Add(sum(b) == 26)
    for i in range(num_points):
        for j in range(i):
            d_max = Rmax.flat[i] + Rmax.flat[j]
            dist = np.sqrt(np.sum((points[i] - points[j])**2))
            if dist - d_max > 1e-6:
                M = 2.0
            else:
                M = dist - d_max + 1e-6
            model.Add(r[i] + r[j] <= dist + M * (2 - b[i] - b[j]))
    model.Maximize(sum(r))
    return model, {i: (points[i], r[i]) for i in range(num_points)}

def solve_milp(model, time_limit=600):
    model.SetTimeLimit(time_limit * 1000)
    result_status = model.Solve()
    best_solution = {g: (np.nan, 0.0) for g in range(400)}
    if result_status in [pywraplp.Solver.OPTIMAL, pywraplp.Solver.FEASIBLE]:
        for g, (point, var) in model.vars_by_grid.items():
            if var.solution_value() > 1e-6:
                best_solution[g] = (point, var.solution_value())
    return result_status, best_solution

def extract_solution(best_solution, vars_by_grid):
    centers = np.array([point for point, _ in best_solution.values() if not np.isnan(point[0])])
    radii = np.array([radius for _, radius in best_solution.values() if radius > 1e-6])
    return centers, radii

def clipped(arr, min_val, max_val):
    return np.clip(arr, min_val, max_val)

def polish_solution(centers, radii):
    delta_r = 1e-4
    epsilon = 1e-4
    r_max = radii.max()
    while True:
        any_grown = False
        for i in range(len(radii)):
            r = radii[i]
            center = centers[i]
            if r + delta_r > r_max:
                continue
            new_center = center
            for dr in [-delta_r, delta_r]:
                for d in range(max(0, int(center[0] - dr - r) - int(center[0] + dr + r)), min(int(center[0] - dr - r) + 2, 2 * (center[0] + dr + r) + 1)):
                    for d2 in range(max(0, int(center[1] - dr - r) - int(center[1] + dr + r)), min(int(center[1] - dr - r) + 2, 2 * (center[1] + dr + r) + 1)):
                        new_center = np.array([center[0] + d - (center[0] + epsilon), center[1] + d2 - (center[1] + epsilon)])
                        if np.any(np.linalg.norm(centers - new_center, axis=1) <= r + delta_r):
                            continue
                        # Check distance to all other circles
                        for j in range(len(radii)):
                            if i == j:
                                continue
                            if np.linalg.norm(new_center - centers[j]) <= radii[j] + delta_r:
                                break
                        else:
                            radii[i] += delta_r
                            center = new_center
                            any_grown = True
                            break
            if any_grown:
                break
        if not any_grown:
            break
    return centers, radii
