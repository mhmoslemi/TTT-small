import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import pdist, squareform

def calculate_late_penalty(centers, radii):
    distances = pdist(centers, 'euclidean')
    max_dist = np.max(distances)
    penalty = 0.0
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            if radii[i] + radii[j] > 1e-6:
                penalty += (radii[i] + radii[j] - distances[(i, j)])
    return penalty * 1e4

def objective(z, n=26, lambda_=1e4, growth_step=0.02):
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    
    # Calculate overlap and boundary penalties
    distances = squareform(pdist(centers, 'euclidean'))
    overlap_penalty = np.sum((radii[:, np.newaxis] + radii[np.newaxis, :] - np.clip(distances, 0, None)) ** 2 * (distances > 0))
    boundary_penalty = np.sum(np.maximum(0, 1 - radii))
    
    # Reward large radii but add constraints for valid placement
    return -(np.sum(radii) - overlap_penalty - lambda_ * boundary_penalty)

def check_constraints(centers, radii):
    distances = pdist(centers, 'euclidean')
    if np.any(distances < 2e-6):
        return False
    if np.any(radii + distances > 1 + 2e-6):
        return False
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            if radii[i] + radii[j] > 1 + 1e-6 and distances[(i, j)] < 1e-6:
                return False
    return True

def run_packing() -> (np.ndarray, np.ndarray, float):
    n = 26
    restarts = 18
    best_energy = np.inf
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for _ in range(restarts):
        initial_centers = np.random.rand(n, 2)
        initial_radii = np.random.rand(n) * 0.01
        
        # run differential evolution
        result = differential_evolution(objective, 
                                        bounds=[((0,1),) * 26, (0,2) * 26],
                                        args=(n, 1e4, 0.02),
                                        strategy='best1bin',
                                        popsize=100, 
                                        tol=1e-6,
                                        maxiter=20000)

        centers = result.x[: n * 2].reshape((n, 2))
        radii = result.x[n * 2:]
        
        # final refinement via LP
        res = linprog(-radii, 
                      A_ub=pdist(centers, 'euclidean')[:, np.newaxis] - radii[np.newaxis, :] - np.eye(n),
                      b_ub=-1e-6,
                      A_eq=np.column_stack((np.eye(n), np.zeros(n))),
                      b_eq=radii,
                      method='highs',
                      bounds=(0, 1))

        if res.success and res.x is not None and check_constraints(centers, res.x):
            energy = objective(res.x)
            if energy < best_energy:
                best_energy = energy
                best_centers = centers
                best_radii = res.x

    return best_centers, best_radii, -best_energy
