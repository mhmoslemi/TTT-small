import numpy as np
from scipy.optimize import linprog
from scipy.spatial import Voronoi, voronoi_plot_2d
import matplotlib.pyplot as plt

def calculate_voronoi_circles(centers):
    vor = Voronoi(centers)
    radii = np.zeros(len(centers))
    for i, point in enumerate(vor.points):
        distances = np.linalg.norm(vor.vertices - point, axis=1)
        min_distance = np.min(distances)
        side_distance = min(point[0], 1 - point[0], point[1], 1 - point[1])
        radii[i] = min(min_distance / 2, side_distance)
    return radii

def maximize_sum_radii(centers):
    n = centers.shape[0]
    A = np.zeros((n * 2, n * 2))
    b = np.zeros(n * 2)
    c = np.zeros(n * 2)
    
    for i in range(n):
        A[i * 2, i], A[i * 2 + 1, i] = 1, 1
        b[i * 2], b[i * 2 + 1] = 0, centers[i, 0]
    
    for i in range(n):
        A[i * 2, n + i], A[i * 2 + 1, n + i] = 1, 1
        b[i * 2], b[i * 2 + 1] = 0, 1 - centers[i, 0]
    
    for i in range(n):
        A[i * 2, n + n + i], A[i * 2 + 1, n + n + i] = 1, 1
        b[i * 2], b[i * 2 + 1] = 0, centers[i, 1]
    
    for i in range(n):
        A[i * 2, n + n + n + i], A[i * 2 + 1, n + n + n + i] = 1, 1
        b[i * 2], b[i * 2 + 1] = 0, 1 - centers[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            distance = np.sqrt(dx * dx + dy * dy)
            A[n * 2 + i * n + j, i] = 1
            A[n * 2 + i * n + j, j] = -1
            b[n * 2 + i * n + j] = distance - radii[i] - radii[j]
            A[n * 2 + i * n + j + n, i] = -1
            A[n * 2 + i * n + j + n, j] = 1
            b[n * 2 + i * n + j + n] = distance - radii[i] - radii[j]
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='simplex')
    return result.x[:n], result.fun

                gradients_x = (S_plus_x - S_minus_x) / (2 * eps)
                
                radii_plus_y = maximize_sum_radii(centers.copy()[:i, :])
