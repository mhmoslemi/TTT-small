import numpy as np
from scipy.optimize import minimize, linprog
import random

def calculate_distances(centers):
    n = centers.shape[0]
    distances = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            distances[i, j] = np.linalg.norm(centers[i] - centers[j])
    return distances + distances.T

def evaluate_lps(centers):
    n = centers.shape[0]
    distances = calculate_distances(centers)
    r = np.array([min(x, 1 - x, y, 1 - y) for x, y in centers])
    A_eq = np.vstack([r[:, None] + r[i] for i in range(n)])
    A_eq = A_eq - distances
    A_eq = A_eq[~np.all(A_eq > 0, axis=1)]
    b_eq = np.zeros(len(A_eq)).reshape(-1, 1)
    A_eq = np.concatenate([np.eye(n), -np.eye(n), -np.eye(n), np.eye(n), -A_eq], axis=0)
    b_eq = np.concatenate([np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), -np.ones(len(A_eq)).reshape(-1, 1)], axis=0)
    result = linprog([-np.sum(r)], A_ub=A_eq, b_ub=b_eq, bounds=[(0, None)] * 2 * n, method='highs')
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    initial_radii = np.full(n, 0.13)
    initial_z = np.concatenate([initial_centers.flatten(), initial_radii])
    
    def objective(z):
        return -np.sum(z[2 * n:])
    
    def constraints(z):
        n = len(z) // 3
        centers = z[:2 * n].reshape(-1, 2)
        radii = z[2 * n:]
        g = np.zeros(4 * n + (n * (n - 1)) // 2)
        g[:4 * n] = np.concatenate([radii - np.min(centers[:, [0, 1]], axis=1),
                                    radii - (1 - np.min(centers[:, [0, 1]], axis=1)),
                                    radii - np.min(centers[:, [0, 1]], axis=1),
                                    radii - (1 - np.min(centers[:, [0, 1]], axis=1))])
        distances = calculate_distances(centers)
        g[4 * n:] = distances - (radii + radii[:, None]).reshape(-1)
        return g
    
    cons = {'type': 'ineq', 'fun': constraints}
    
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    
    res = minimize(objective, initial_z, method='COBYLA', bounds=bounds, constraints=cons, options={'rhobeg': 0.1, 'rhoend': 1e-8, 'maxfun': 20000})
    centers = res.x[:2 * n].reshape(-1, 2)
    radii, max_sum = evaluate_lps(centers)
    return centers, radii, max_sum

centers, radii, sum_radii = run_packing()
