import numpy as np
from scipy.optimize import linprog, minimize
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def energy_function(z, centers, radii):
    n = len(centers)
    E = 0
    for i in range(n):
        E -= radii[i]
        for j in range(i + 1, n):
            r_i, r_j = radii[i], radii[j]
            d_ij = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if r_i + r_j > d_ij:
                E += (r_i + r_j - d_ij) ** 2

def simulated_annealing(centers, radii, lambda_, p, T0, alpha, sigma, tau, N):
    n = len(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_energy = energy_function(np.hstack((centers, radii)), centers, radii)
    T = T0
    for k in range(N):
        i = random.randint(0, n - 1)
        old_pos = centers[i, :]
        old_radius = radii[i]
        new_pos = old_pos + np.random.normal(0, sigma)
        new_radius = old_radius + np.random.normal(0, tau)
        new_pos = np.clip(new_pos, 0, 1)
        new_radius = np.clip(new_radius, 0, 1)
        new_centers = centers.copy()
        new_radii = radii.copy()
        new_centers[i, :] = new_pos
        new_radii[i] = new_radius
        new_energy = energy_function(np.hstack((new_centers, new_radii)), new_centers, new_radii)
        if new_energy < best_energy:
            centers, radii = new_centers, new_radii
            best_centers, best_radii = new_centers, new_radii
            best_energy = new_energy
        else:
            deltaE = new_energy - best_energy
            if random.random() < np.exp(-deltaE / T):
                centers, radii = new_centers, new_radii
            else:
                centers[i, :] = old_pos
                radii[i] = old_radius
        T *= alpha
    return best_centers, best_radii, -best_energy
