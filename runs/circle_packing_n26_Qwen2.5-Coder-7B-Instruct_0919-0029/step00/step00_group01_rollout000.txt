import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]])
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(distances, np.inf)
    A_ub = []
    for i in range(n):
        row_i = np.zeros(n)
        row_i[i] = 1
        A_ub.append(np.concatenate((row_i, -borders)))
    for i in range(n):
        for j in range(i + 1, n):
            row_ij = np.zeros(n)
            row_ij[i] = 1
            row_ij[j] = 1
            A_ub.append(np.concatenate((row_ij, -distances[i, j])))
    A_ub = np.array(A_ub)
    c = -np.ones(n)  # to minimize -sum(r_i), hence maximize sum(r_i)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, bounds=bounds, method="highs")
    return res.x, -res.fun

def random_move(centers, step):
    i = np.random.randint(n)
    new_pos = centers[i] + step * (np.random.randn(2) - 0.5)
    new_pos = np.clip(new_pos, 0, 1)
    return np.array(centers[:i], centers[i+1:]).tolist() + [new_pos.tolist()]

def simulated_annealing_centers(n, max_iter, rng):
    centers = rng.uniform(0, 1, (n, 2))
    best_centers = centers.copy()
    best radii, best_sum = solve_lp(best_centers)
    step = 0.05
    T = 0.02
    T_end = 1e-5
    for _ in range(max_iter):
        new_centers = random_move(best_centers, step)
        new_radii, new_sum = solve_lp(new_centers)
        if new_sum > best_sum or np.exp((new_sum - best_sum) / T) > rng.rand():
            best_centers = new_centers
            best_sum = new_sum
        T = T * (T_end / T) ** (1 / max_iter)
        step = step * 0.999  # adaptive step size reduction
    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 30000
    rng = np.random.default_rng(42)
    best_centers, best_sum = simulated_annealing_centers(n, max_iter, rng)
    best_radii, _ = solve_lp(best_centers)
    best_radii = np.maximum(best_radii, 0)  # ensure feasibility
    return np.array(best_centers, dtype=np.float64), np.array(best_radii, dtype=np.float64), best_sum
