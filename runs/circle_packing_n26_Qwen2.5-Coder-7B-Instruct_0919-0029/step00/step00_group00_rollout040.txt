import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p):
    n = 26
    r = z[:n]
    x = z[n:2*n]
    y = z[2*n:]
    sum_r = np.sum(r)
    overlap = 0
    boundary = 0
    
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            eta_ij = np.maximum(0, r[i] + r[j] - d_ij)
            overlap += np.maximum(0, eta_ij)**p
    
    for i in range(n):
        boundary += np.maximum(0, r[i] - x[i])**p
        boundary += np.maximum(0, r[i] - (1 - x[i]))**p
        boundary += np.maximum(0, r[i] - y[i])**p
        boundary += np.maximum(0, r[i] - (1 - y[i]))**p
    
    return -sum_r + λ * (overlap + boundary)

def compute_gradient(z, λ, p):
    n = 26
    r = z[:n]
    x = z[n:2*n]
    y = z[2*n:]
    grad_r = np.zeros(n)
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            eta_ij = np.maximum(0, r[i] + r[j] - d_ij)
            if eta_ij > 0:
                grad_r[i] += p * eta_ij**(p-1)
                grad_r[j] += p * eta_ij**(p-1)
                grad_x[i] += p * eta_ij**(p-1) * (x[i] - x[j]) / d_ij
                grad_y[i] += p * eta_ij**(p-1) * (y[i] - y[j]) / d_ij
                grad_x[j] -= p * eta_ij**(p-1) * (x[i] - x[j]) / d_ij
                grad_y[j] -= p * eta_ij**(p-1) * (y[i] - y[j]) / d_ij
    
    boundary = 0
    for i in range(n):
        boundary += p * np.maximum(0, r[i] - x[i])**(p-1)
        boundary += p * np.maximum(0, r[i] - (1 - x[i]))**(p-1)
        boundary += p * np.maximum(0, r[i] - y[i])**(p-1)
        boundary += p * np.maximum(0, r[i] - (1 - y[i]))**(p-1)
    
    neg_grad_r = np.diff(grad_r)
    neg_grad_r = np.r_[neg_grad_r, neg_grad_r[0]]
    
    neg_grad_x = np.diff(grad_x)
    neg_grad_x = np.r_[neg_grad_x, neg_grad_x[0]]
    
    neg_grad_y = np.diff(grad_y)
    neg_grad_y = np.r_[neg_grad_y, neg_grad_y[0]]
    
    return np.concatenate((neg_grad_r, neg_grad_x, neg_grad_y))
