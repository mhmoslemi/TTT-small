import numpy as np
from ortools.linear_solver import pywraplp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    k = 20
    delta = 1 / k
    M = 1000  # Big M value for constraint strengthening

    # Build the MILP problem
    solver = pywraplp.Solver.CreateSolver('CBC')
    grid_vars = {}
    for i in range(k):
        for j in range(k):
            b_ij = solver.IntVar(0, 1, f'b_{i}_{j}')
            r_ij = solver.NumVar(0, 10, f'r_{i}_{j}')  # Upper bound is arbitrary
            grid_vars[(i, j)] = (b_ij, r_ij)
            solver.Add(r_ij <= b_ij * (min(i * delta, (k - i) * delta, j * delta, (k - j) * delta)))

    # Cardinality constraint
    solver.Add(sum(b_ij for b_ij, r_ij in grid_vars.values()) == 26)

    # Non-overlap constraints
    for i in range(k):
        for j in range(k):
            if i == j:
                continue
            d_ij = np.sqrt((i * delta - j * delta) ** 2 + (j * delta - i * delta) ** 2)
            d_max_ij = min((i * delta, (k - i) * delta, j * delta, (k - j) * delta))
            if d_ij >= d_max_ij:
                continue
            solver.Add(solver.Sum([grid_vars[(i, j)][1], grid_vars[(i, m)][1] for m in range(k)]) <= d_ij + M * (1 - grid_vars[(i, j)][0] - grid_vars[(i, m)][0]))
            solver.Add(solver.Sum([grid_vars[(i, j)][1], grid_vars[(m, j)][1] for m in range(k)]) <= d_ij + M * (1 - grid_vars[(i, j)][0] - grid_vars[(m, j)][0]))

    # Objective: maximize the sum of radii
    objective = solver.Objective()
    for _, (b_ij, r_ij) in grid_vars.items():
        objective.SetCoefficient(r_ij, 1)
    objective.Maximize()

    # Solve the MILP problem
    status = solver.Solve()
    if status != pywraplp.Solver.OPTIMAL and status != pywraplp.Solver.FEASIBLE:
        raise Exception(f"Solve failed: {solver.Status()}")
      
    # Extract the solution
    centers = []
    radii = []
    for (i, j), (b_ij, r_ij) in grid_vars.items():
        if b_ij.solution_value() > 0.5:
            centers.append((i * delta + delta / 2, j * delta + delta / 2))
            radii.append(r_ij.solution_value())

    # Polish the packing
    centers, radii = polish(np.array(centers), np.array(radii))
    
    validate_packing(np.array(centers), np.array(radii))

    return np.array(centers), np.array(radii), sum(radii)

def polish(centers, radii):
    tol = 1e-12
    delta = 1e-4
    epsilon = 1e-4
    while True:
        changed = False
        # Greedy radius grow
        for i in range(len(centers)):
            r_i = radii[i]
            while True:
                r_try = r_i + delta
                too_close = False
                for j in range(len(centers)):
                    if i != j and np.linalg.norm(centers[i] - centers[j]) < (r_i + r_try + tol):
                        too_close = True
                        break
                if too_close:
                    break
                r_i = r_try
                changed = True
            radii[i] = r_i
        
        # Minor centre shifts
        for _ in range(5):
            for i in range(len(centers)):
                for dx in [-epsilon, 0, epsilon]:
                    for dy in [-epsilon, 0, epsilon]:
                        too_close = False
                        for j in range(len(centers)):
                            if i != j and np.linalg.norm(centers[i] + np.array([dx, dy]) - centers[j]) < (radii[i] + radii[j] + tol):
                                too_close = True
                                break
                        if not too_close:
                            centers[i] += np.array([dx / 3, dy / 3])
                            changed = True
        
        if not changed:
            break
    
    return centers, radii
