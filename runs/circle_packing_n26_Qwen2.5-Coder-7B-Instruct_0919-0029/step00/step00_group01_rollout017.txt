import numpy as np
import scipy.optimize
from scipy.spatial.distance import pdist, squareform

def compute_penalty(centres, radii):
    penalty = 0.0
    for i in range(26):
        penalty += max(0, -centres[i, 0])**2 + max(0, centres[i, 0]-1)**2
        penalty += max(0, -centres[i, 1])**2 + max(0, centres[i, 1]-1)**2
        if radii[i] < 0:
            penalty += 1e7 * max(0, -radii[i])**2
    for i in range(26):
        for j in range(i+1, 26):
            dist = np.linalg.norm(centres[i] - centres[j])
            gap = radii[i] + radii[j] - dist
            if gap > 0:
                penalty += 1e7 * gap**2
    return penalty

def de_objective(v):
    centres = v[:52].reshape((26, 2))
    radii = v[52:]
    objective_value = -np.sum(radii) + compute_penalty(centres, radii)
    return objective_value

def solve_lps_for_centres(centres):
    half_edge = 0.5
    radii = np.zeros(26)
    for i in range(26):
        radius_bound = min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
        radii[i] = radius_bound
    return radii, np.sum(radii)

def run_de_search(seed):
    bounds_radius = [(0, 0.25) for _ in range(26)]
    bounds_position = [(0, 1) for _ in range(52)]
    bounds = bounds_position + bounds_radius
    x0 = np.random.rand(52)
    result = scipy.optimize.differential_evolution(func=de_objective, bounds=bounds, maxiter=2000, popsize=50, seed=seed, tol=1e-6, strategy='intermediate')
    return result.x

def run_packing():
    np.random.seed(0)
    seeds = np.random.randint(0, 10000, size=2)  # Two random seeds for DE
    results = [run_de_search(seed) for seed in seeds]
    best_result_idx = np.argmax([de_objective(result) for result in results])
    best_centres = results[best_result_idx][:52].reshape((26, 2))
    final_radii, best_sum_radii = solve_lps_for_centres(best_centres)
    return best_centres, final_radii, best_sum_radii

# Example usage:
centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}")
print(f"Radii:\n{radii}")
print(f"Sum of radii: {sum_radii}")
