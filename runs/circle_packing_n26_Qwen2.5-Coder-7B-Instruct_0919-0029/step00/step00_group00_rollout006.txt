import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p):
    n = 26
    x = z[:n].reshape(n, 2)
    r = z[n:]
    d = np.sqrt(np.sum((x[:, np.newaxis] - x[np.newaxis, :])**2, axis=-1))
    overlaps = np.where(d > 0)[0]
    overlap_penalties = np.maximum(0, (r[overlaps[0::2]] + r[overlaps[1::2]]) - d[overlaps]).astype(float)**p
    boundary_penalties = np.sum(np.maximum(0, np.column_stack((r - x[:, 0], r - 1 + x[:, 0], r - x[:, 1], r - 1 + x[:, 1]))).astype(float)**p, axis=1)
    return -np.sum(r) + λ * (np.sum(overlap_penalties) + boundary_penalties)

def compute_gradient(z, λ, p):
    n = 26
    x = z[:n].reshape(n, 2)
    r = z[n:]
    d = np.sqrt(np.sum((x[:, np.newaxis] - x[np.newaxis, :])**2, axis=-1))
    overlaps = np.where(d > 0)[0]
    overlap_penalties = np.maximum(0, (r[overlaps[0::2]] + r[overlaps[1::2]]) - d[overlaps]).astype(float)**p
    boundaries = np.column_stack((r - x[:, 0], r - 1 + x[:, 0], r - x[:, 1], r - 1 + x[:, 1]))
    boundary_penalties = (np.maximum(0, boundaries).astype(float)**p) / (np.abs(boundaries) + 1e-12)
    dr = -(np.sum([r[overlaps[0::2]] + r[overlaps[1::2]] - d[overlaps]*0.5] * p * overlap_penalties) + np.sum(np.sum(boundary_penalties, axis=1)))
    dc = -(np.sum([0.5*(r[overlaps[i*2]] + r[overlaps[i*2+1]]) / d[overlaps[i*2]] * np.array([(x[i*2][0] - x[i*2+1][0]), (x[i*2][1] - x[i*2+1][1])]) * overlap_penalties[i*2] for i in range(len(overlaps)//2)]) +
          np.sum(np.column_stack([x, np.zeros_like(x), np.zeros_like(x)]) * boundary_penalties, axis=1))
    return np.concatenate((dc.ravel(), dr))

def lp_optimize_centres(centres):
    n = 26
    r = np.ones(n) * 0.01
    A = np.hstack((np.eye(n, n, k=+0), -np.eye(n, n, k=+0), np.eye(n, n, k=-26), -np.eye(n, n, k=-26)))
    b = np.ones(2*n) * np.inf
    G = np.vstack((np.ones((n, n)), -np.eye(n)))
    h = np.hstack(([1]*n, np.zeros(n)))
    res = minimize(lambda r: np.sum(r), r, bounds=[(0, 1)] * n, method='highs',
                   constraints={'type': 'eq', 'fun': lambda r: np.dot(A, np.concatenate((centres, r))) - b},
                   options={'disp': False})
    r_opt = res.x
    return r_opt, np.sum(r_opt)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    penalty_weight = 5e4
    penalty_exponent = 2
    initial_radii = 0.01
    tolerance = 1e-6
    max_evaluations = 2000
    num_starts = 18

    best_solution = None
    best_sum = -np.inf

    for _ in range(num_starts):
        if _ % 4 == 0:
            centres = np.zeros((26, 2))
            centres[::2] = np.array([[1/2, 0], [0, 1/2], [1/2, 1], [1, 1/2]])
            centres[1::2] = np.random.rand(12, 2) * 0.45 + 0.25
        elif _ % 4 == 1:
            centres = np.array([[i/5, 0] for i in range(3)] + [[0, j/5] for j in range(3)] + [[i/5, 1] for i in range(3)] + [[1, j/5] for j in range(3)])
            r = np.ones(26) * initial_radii
        elif _ % 4 == 2:
            centres = np.array([[0.1, 0.1], [0.9, 0.9]])
            r = np.ones(26) * initial_radii
        else:
            centres = np.random.rand(26, 2)
            r = np.ones(26) * initial_radii

        start_z = np.concatenate((centres.ravel(), r))
        res = minimize(lambda z: compute_penalty(z, penalty_weight, penalty_exponent), start_z, method='L-BFGS-B', jac=lambda z: compute_gradient(z, penalty_weight, penalty_exponent), bounds=[(0, 1)] * 2*n + [(0, 1)] * n, options={'maxfun': max_evaluations, 'gtol': tolerance})
        if res.fun > best_sum:
            best_sum = res.fun
            best_solution = (centres, r)

    if best_solution is None:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    
    best_centres, best_r = best_solution
    best_r_opt, best_sum_opt = lp_optimize_centres(best_centres)
    return best_centres, best_r_opt, best_sum_opt
