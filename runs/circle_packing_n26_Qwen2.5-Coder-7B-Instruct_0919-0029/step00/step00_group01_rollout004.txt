import numpy as np
from scipy.optimize import differential_evolution, linprog

def packing_objective(v):
    n = 26
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            penalty += max(0, radii[i] + radii[j] - dist) ** 2
        penalty += max(0, -centers[i][0]) ** 2
        penalty += max(0, centers[i][0] - 1) ** 2
        penalty += max(0, -centers[i][1]) ** 2
        penalty += max(0, centers[i][1] - 1) ** 2
        penalty += max(0, -radii[i]) ** 2
    return -(np.sum(radii) + 1e8 * penalty)

def lp_radius_optimization(centers):
    n = 26
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([-1] * n)
        b_ub.append(-centers[i][0])
        A_ub.append([-1] * n)
        b_ub.append(-(1 - centers[i][0]))
        A_ub.append([-1] * n)
        b_ub.append(-centers[i][1])
        A_ub.append([-1] * n)
        b_ub.append(-(1 - centers[i][1]))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([1] * n + [1] * n)
            b_ub.append(centers[i][0] - centers[j][0] + centers[i][1] - centers[j][1])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    radii = result.x
    return np.maximum(radii, 0), -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1), (0, 1)] * 26 + [(0, 0.25)] * 26
    result1 = differential_evolution(packing_objective, bounds=bounds, popsize=50, mutation=0.8, crossover=0.7, maxiter=2000, polish=False)
    result2 = differential_evolution(packing_objective, bounds=bounds, popsize=50, mutation=0.8, crossover=0.7, maxiter=2000, polish=False)
    if result1.fun > result2.fun:
        center_best = result1.x[:2*26].reshape(26, 2)
        result_best = result1
    else:
        center_best = result2.x[:2*26].reshape(26, 2)
        result_best = result2
    radii_best, sum_radii = lp_radius_optimization(center_best)
    return center_best, radii_best, sum_radii
