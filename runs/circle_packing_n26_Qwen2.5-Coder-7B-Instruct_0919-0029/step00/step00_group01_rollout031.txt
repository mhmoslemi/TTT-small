import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)
    
    A_ub = np.array([np.concatenate((np.eye(n), -np.eye(n)), axis=0).flatten(),
                     *distances.flatten()], dtype=float)
    b_ub = np.concatenate((borders, distances.flatten()), axis=0)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        radii = result.x
        return radii, np.sum(radii)
    else:
        return np.zeros(n), 0.0

def random_move(centers, step):
    n = centers.shape[0]
    i = np.random.randint(n)
    new_pos = centers[i] + np.random.normal(scale=step, size=2)
    new_pos = np.clip(new_pos, 0, 1)
    return new_pos, i

def simulated_annealing_centers(n, max_iter, rng):
    centers = rng.uniform(0, 1, size=(n, 2))
    best_centres = centers.copy()
    best_radii, best_sum = solve_lp(centers)
    
    step = 0.05
    T = 0.02
    T_end = 1e-5
    
    for iter in range(max_iter):
        new_pos, i = random_move(centers, step)
        centers[i] = new_pos
        
        radii, new_sum = solve_lp(centers)
        delta = new_sum - best_sum
        
        if delta > 0 or rng.rand() < np.exp(delta / T):
            if new_sum > best_sum:
                best_centres = centers.copy()
                best_sum = new_sum
                best_radii = radii.copy()
        
        T = T_end * (T / T_end) ** (iter / max_iter)
    
    return best_centres, best_radii, best_sum

def run_packing():
    np.random.seed(42)  # Ensures reproducibility for grading
    n = 26
    max_iter = 30000
    result = simulated_annealing_centers(n, max_iter, np.random.default_rng())
    return result

centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
