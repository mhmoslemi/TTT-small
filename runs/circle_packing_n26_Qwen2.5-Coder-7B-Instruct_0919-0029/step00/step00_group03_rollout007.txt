import numpy as np
from scipy.optimize import minimize, linprog, cobyla

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial grid + one in center
    centers = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    centers = np.vstack((centers, (0.5, 0.5)))
    radii = np.full(n, 0.12)  # Minimal wall distance heuristic

    def cobyla_objective(z):
        return -z[n:]  # Maximize radii, negative because COBYLA minimizes

    def cobyla_constraints(z):
        xs = z[:n//2]
        ys = z[n//2:2*n//2]
        rs = z[2*n//2:]
        constraints = [
            g + 1e-12 for g in [
                xs - rs,
                1 - xs - rs,
                ys - rs,
                1 - ys - rs
            ]
        ]
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(np.linalg.norm(center_center(xs[i], ys[i], xs[j], ys[j])) - (rs[i] + rs[j]))
        return constraints

    def center_center(x1, y1, x2, y2):
        return np.array([x1, y1]) - np.array([x2, y2])

    cobyla_result = minimize(cobyla_objective, np.concatenate((centers.flatten(), radii)), method='COBYLA', constraints={'type': 'ineq', 'fun': cobyla_constraints}, options={'rhobeg': 0.1, 'rhoend': 1e-8, 'maxfun': 20000, 'tol': 1e-7})

    if not cobyla_result.success:
        return np.zeros((n, 2)), np.zeros(n), 0

    centers_opt = cobyla_result.x[:n//2].reshape((n//2, 2))
    radii_opt = cobyla_result.x[2*n//2:]

    def psor_objective(v):
        xs = v[:n//2]
        ys = v[n//2:]
        dists = np.array([np.linalg.norm(center_center(xs[i], ys[i], xs[j], ys[j])) for i in range(n) for j in range(i + 1, n)])
        ws = np.minimum(xs, 1 - xs)
        ws = np.minimum(ws, ys)
        ws = np.minimum(ws, 1 - ys)
        rs = np.maximum((ws / 2) + 1e-12, (dists - ws) / 2)
        return -np.sum(rs)

    xs0 = np.random.rand(n//2)
    ys0 = np.random.rand(n//2)
    psor_result = minimize(psor_objective, np.concatenate((xs0, ys0)), method='SLSQP', bounds=[(0, 1) for _ in range(n)], options={'maxiter': 2000})

    if not psor_result.success:
        return np.zeros((n, 2)), np.zeros(n), 0

    centers_psor = psor_result.x[:n//2].reshape((n//2, 2))
    radii_psor = -psor_result.fun

    final_centers = np.hstack((centers_psor, centers_psor[:, np.newaxis] + (centers_psor[0] - centers_psor[1])))
    final_radii = np.maximum(radii_psor, 1e-12)

    validation, message = validate_packing(final_centers, final_radii)
    if not validation:
        final_radii = np.maximum(final_radii * 0.9999999, 1e-12)
        validation, message = validate_packing(final_centers, final_radii)

    return final_centers, final_radii, np.sum(final_radii)
