import numpy as np
from scipy.optimize import linprog
from scipy.spatial import Voronoi

def lp_solve(centers):
    n = centers.shape[0]
    constraints = []
    bounds = []

    for i in range(n):
        bounds.append((0, None))

        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx**2 + dy**2)
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: d - (r[i] + r[j])})

        for coord in 'xy':
            constraints.append({'type': 'ineq', 'fun': lambda r, coord=coord: centers[i, ['x', 'y'].index(coord)] - r[i]})
            constraints.append({'type': 'ineq', 'fun': lambda r, coord=coord: 1 - centers[i, ['x', 'y'].index(coord)] - r[i]})

    c = np.ones(n)
    result = linprog(c, A_ub=constraints, bounds=bounds, method='simplex')
    if result.success:
        return result.x
    else:
        return np.zeros(n)

def gradient(centers, radii):
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    eps = 1e-4

    for i in range(n):
        tmp = centers[i, 0]
        centers[i, 0] = tmp + eps
        r_plus = lp_solve(centers)
        centers[i, 0] = tmp - eps
        r_minus = lp_solve(centers)
        grad[i, 0] = (-r_plus + r_minus) / (2 * eps)

        centers[i, 0] = tmp
        tmp = centers[i, 1]
        centers[i, 1] = tmp + eps
        r_plus = lp_solve(centers)
        centers[i, 1] = tmp - eps
        r_minus = lp_solve(centers)
        grad[i, 1] = (-r_plus + r_minus) / (2 * eps)

        centers[i, 1] = tmp

    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    centers = np.random.rand(26, 2)
    max_iter = 200
    tol = 1e-5
    alpha = 0.01
    best_centers = None
    best_sum = 0

    for run in range(2):
        for iter in range(max_iter):
            radii = lp_solve(centers.copy())
            sum_radii = np.sum(radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers.copy()
            grad = gradient(centers, radii)
            step = alpha * grad
            centers += step
            centers = np.clip(centers, 0, 1)
    
        alpha *= 0.95

    # Final local update
    for k in range(10):
        for i in range(26):
            radii_updated = radii.copy()
            radii_updated[i] += 1e-3
            sum_radii_updated = np.sum(lp_solve(centers))
            if sum_radii_updated > sum_radii:
                sum_radii = sum_radii_updated

    return best_centers, lp_solve(best_centers), sum_radii

centers, radii, sum_radii = run_packing()
centers = centers[~np.isnan(radii), :]
radii = radii[~np.isnan(radii)]
