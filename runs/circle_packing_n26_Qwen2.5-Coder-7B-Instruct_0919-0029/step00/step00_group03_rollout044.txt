def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    max_iter = 200000
    shake_rounds = 20
    g = 1e-6
    delta = 5e-7
    eps = 1e-9

    # Seed the configuration
    centers = np.array([[0.1 + 0.2 * i for i in range(5)] + [0.5],

    centers = np.array([centers[1], centers[0], centers[2], centers[3], centers[4],

                        centers[24], centers[25], centers[17], centers[23]])
    radii = np.zeros(num_circles)

    for _ in range(max_iter):
        # Proposed growth
        new_radii = radii + g

        # Constraint checking and resolution
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[j] - centers[i])
                if new_radii[i] + new_radii[j] > dist:
                    conflict = True
                    break
                conflict = False

                break
        if conflict:
            nearest = np.argmin([
                np.linalg.norm(centers[j] - centers[i], 2) - radii[j] - radii[i]

            d_cur = radii[i] + radii[nearest]
            d_new = radii[i] + g

                direction = (centers[nearest] - centers[i]) / d_new
                centers[i] += direction * delta

                    centers[i, 0] -= delta
                    radii[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

                    centers[i, 1] += delta
                    radii[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

                    centers[i, 1] -= delta
                    radii[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

            break

    for shake_r in range(shake_rounds):

        epsilon = 0.02 * np.random.rand(2)

        radii[k] = min(np.min(np.linalg.norm(centers - centers[k], 2, axis=0) - 2 * radii), centers[k, 0], 1 - centers[k, 0], centers[k, 1], 1 - centers[k, 1])
