import numpy as np
from scipy.optimize import differential_evolution, linprog

def circle_overlap_centered(circles):
    num_circles = len(circles)
    overlaps = np.zeros((num_circles, num_circles))
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            overlaps[i, j] = overlaps[j, i] = np.sqrt(np.sum((circles[i] - circles[j]) ** 2))
    return overlaps + np.eye(num_circles) * 1e-6

def penalty(de_parameter):
    centers = de_parameter[:26].reshape(-1, 2)
    radii = de_parameter[26:]
    boundary_penalty = np.sum(np.maximum(0, -centers[:, 0]) ** 2 + np.maximum(0, centers[:, 0] - 1) ** 2
                             + np.maximum(0, -centers[:, 1]) ** 2 + np.maximum(0, centers[:, 1] - 1) ** 2)
    negative_radius_penalty = np.sum(np.maximum(0, -radii) ** 2)
    overlap_penalty = np.sum(np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - circle_overlap_centered(centers)) ** 2)
    return -(np.sum(radii) + 1e-6 * (boundary_penalty + negative_radius_penalty + overlap_penalty))

def linear_programming_optimization(centers):
    num_circles = len(centers)
    objective_func = lambda radii: -np.sum(radii)
    constraints = {'type': 'ineq', 'fun': lambda radii: np.zeros(num_circles)}
    for i in range(num_circles):
        constraints['fun'] += lambda radii, i=i: centers[i, 0] - radii[i] - (np.abs(centers[i, 0]) < 1e-6)
        constraints['fun'] += lambda radii, i=i: 1 - centers[i, 0] - radii[i] - (np.abs(1 - centers[i, 0]) < 1e-6)
        constraints['fun'] += lambda radii, i=i: centers[i, 1] - radii[i] - (np.abs(centers[i, 1]) < 1e-6)
        constraints['fun'] += lambda radii, i=i: 1 - centers[i, 1] - radii[i] - (np.abs(1 - centers[i, 1]) < 1e-6)
        for j in range(i + 1, num_circles):
            constraints['fun'] += lambda radii, i=i, j=j: max(0, radii[i] + radii[j] - np.sqrt(np.sum((centers[i] - centers[j]) ** 2)))
    res = linprog([1] * num_circles, A_ub=constraints['fun'], b_ub=np.zeros(num_circles), bounds=(0, None), method='highs')
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    seeds = [42, 51]
    best_radii = None
    for seed in seeds:
        result = differential_evolution(penalty, bounds=[(0, 1)] * 52, popsize=50, mutation=0.8, crossover=0.7, max_iter=2000, seed=seed)
        if result.success:
            circles_2d = result.x[:26].reshape(-1, 2)
            refined_radii = linear_programming_optimization(circles_2d)
            if best_radii is None or np.sum(refined_radii) > np.sum(best_radii):
                best_radii = refined_radii
                best_centers = circles_2d
    if best_radii is None:
        raise ValueError("No valid solution found by differential evolution.")
    return (best_centers, best_radii, np.sum(best_radii))
