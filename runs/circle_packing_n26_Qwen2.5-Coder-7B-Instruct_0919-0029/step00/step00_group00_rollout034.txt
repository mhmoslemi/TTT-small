import numpy as np
from scipy.optimize import linprog

def generate_initial_centers(n, guess_type):
    if guess_type == 'uniform':
        centers = np.random.rand(n, 2)
    elif guess_type == 'hexagonal':
        side_length = 0.15
        grid_points = np.arange(-side_length, 1 + side_length, side_length * np.sqrt(3))
        centers = np.array([(x, y) for x in grid_points for y in grid_points if x + y >= 0 and (-x + 1 + y) >= 0 and (x + y - 1) >= 0 and (-x + y - 1) >= 0])
        np.random.shuffle(centers)
        centers = centers[:n]
    elif guess_type == 'corner':
        centers = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
        centers = np.vstack([centers, np.random.rand(n - 4, 2)])
    else:
        raise ValueError(f"Unknown guess type: {guess_type}")
    return centers

def calculate_penalties(centers, radii):
    n = centers.shape[0]
    d = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    overlaps = np.where((radii[:, np.newaxis] + radii[np.newaxis, :] - d) > 0, radii[:, np.newaxis] + radii[np.newaxis, :] - d, 0)
    boundary_penalties = np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - (1 - centers[:, 0])) + np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii - (1 - centers[:, 1]))
    return overlaps, boundary_penalties

def energy(z, lambda_):
    n = 26
    sigma = 0.04 * z[-1]
    tau = 0.01 * z[-1]
    
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:3*n]
    
    proposal = x**2 + y**2 + r**2
    
    dx = np.random.normal(0, sigma, n)
    dy = np.random.normal(0, sigma, n)
    
    db_x = np.clip(x + dx, 0, 1)
    db_y = np.clip(y + dy, 0, 1)
    
    new_x = (1 - db_x) * db_y / np.sqrt(db_x * (1 - db_x))
    new_y = (1 - db_y) * db_x / np.sqrt(db_y * (1 - db_y))
    
    r_new = r * (1 + np.random.normal(0, tau, n))
    r_new = np.clip(r_new, 0, 1)
    
    centers = np.column_stack((new_x, new_y))
    overlaps, penalties = calculate_penalties(centers, r_new)
    
    E = -np.sum(r_new) + lambda_ * np.sum(overlaps + penalties)
    
    return E

def run_packing():
    np.random.seed(42)
    n = 26
    lambda_ = 5e4
    iterations_per_run = 20000
    restarts = 18
    optimal_centers = None
    optimal_sum_radii = -np.inf
    
    for _ in range(restarts):
        guess_type = np.random.choice(['uniform', 'hexagonal', 'corner'])
        initial_centers = generate_initial_centers(n, guess_type)
        r = np.full(n, 0.01)
        
        initial_state = np.concatenate((initial_centers.flatten(), r))
        best_E = np.inf
        best_state = None
        
        for _ in range(iterations_per_run):
            current_state = best_state if best_state is not None else initial_state
            current_centers = current_state[:3*n].reshape(n, 3)[:, :2]
            current_r = current_state[:3*n].reshape(n, 3)[:, 2]
            
            if current_r.max() > 1:
                continue
            
            new_E = energy(current_state[:-1], lambda_)
            
            if new_E < best_E:
                best_E = new_E
                best_state = current_state
            else:
                alpha = np.exp(-new_E / current_state[-1])
                if np.random.rand() < alpha:
                    best_state = current_state
                    
        if best_E > optimal_sum_radii:
            optimal_centers = best_state[:3*n].reshape(n, 3)[:, :2]
            optimal_sum_radii = -best_E
    
    fixed_centers = optimal_centers
    constraints = []
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: -r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (fixed_centers[i, 0] - r[i])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (fixed_centers[i, 1] - r[i])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (1 - fixed_centers[i, 0] - r[i])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (1 - fixed_centers[i, 1] - r[i])})
        
        for j in range(i + 1, n):
            d = np.linalg.norm(fixed_centers[i] - fixed_centers[j])
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: (r[i] + r[j] - d)})
    
    result = linprog([1 if i < 26 else -lambda_ for i in range(3 * n)], A_ub=np.array([c['fun'] for c in constraints]), b_ub=-np.array([c['fun'](-np.inf) for c in constraints]), bounds=(0, 1), method='highs')
    
    if result.success and result.x[-1] > 1e-6:
        optimal_radii = result.x[:26]
        return optimal_centers, optimal_radii, np.sum(optimal_radii)
    else:
        return optimal_centers, np.full(n, 0.01), 0.01 * n
