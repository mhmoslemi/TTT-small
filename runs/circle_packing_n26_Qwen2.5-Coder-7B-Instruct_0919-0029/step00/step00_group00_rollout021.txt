import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((429, n))
    b_ub = np.zeros(429)
    
    # Pairwise constraints
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub[row] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            row += 1
    
    # Boundary constraints
    for i in range(n):
        r_i = np.where([centers[i, 0] - centers[i, 1], centers[i, 0] + centers[i, 1],
                        centers[i, 1] - centers[i, 0], centers[i, 1] + centers[i, 0]], 
                       [ centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]], 
                       [np.inf, np.inf, np.inf, np.inf])
        r_i = np.maximum(r_i, 0)
        A_ub[row:] = np.eye(n)[i, :][np.newaxis, :]
        b_ub[row:] = r_i[i] - 1e-12
        row += n
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        raise ValueError("LP solver failed to converge")
    
    return -res.fun, res.x

def random_initialisation():
    n = 26
    best_centres = np.random.rand(n, 2)
    best_sum, best_radii = lp_optimum(best_centres)
    return best_centres, best_sum, best_radii

def random_centres():
    return np.random.rand(26, 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NUM_STARS = 30
    NUM_ITERS = 2000
    step = 0.05
    step_min = 1e-6
    decay_factor = 0.995
    
    best_centres, best_sum, _ = random_initialisation()

    for start in range(NUM_STARS):
        centres = random_centres()
        for it in range(NUM_ITERS):
            idx = np.random.randint(0, 26)
            delta = np.random.uniform([-step, step], size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = np.array(centres)
            new_centres[idx] = new_c
            new_radii, new_sum = lp_optimum(new_centres)
            
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                
            step = max(step_min, step * decay_factor)
                
    final_radii, final_sum = lp_optimum(best_centres)
    return best_centres, final_radii, final_sum
