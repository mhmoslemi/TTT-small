import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p):
    x = z[::3]
    y = z[1::3]
    r = z[2::3]
    n = x.shape[0]
    penalty = -np.sum(r)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2))
            penalty += λ * (max(0, (r[i] + r[j]) - dist)) ** p
        penalty += λ * (max(0, r[i] - x[i])) ** p + λ * (max(0, r[i] - 1 + x[i])) ** p + λ * (max(0, r[i] - y[i])) ** p + λ * (max(0, r[i] - 1 + y[i])) ** p
    return penalty

def compute_gradient(z, λ, p):
    x = z[::3]
    y = z[1::3]
    r = z[2::3]
    n = x.shape[0]
    dx = np.zeros_like(x)
    dy = np.zeros_like(y)
    dr = np.zeros_like(r)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2))
            delta = max(0, (r[i] + r[j]) - dist)
            if delta > 0:
                dx[i] -= λ * p * delta ** (p - 1) * (x[i] - x[j]) / dist
                dy[i] -= λ * p * delta ** (p - 1) * (y[i] - y[j]) / dist
                dx[j] += λ * p * delta ** (p - 1) * (x[i] - x[j]) / dist
                dy[j] += λ * p * delta ** (p - 1) * (y[i] - y[j]) / dist
            dx[i] += -λ * p * (max(0, r[i] - x[i])) ** (p - 1) * (-1)
            dy[i] += -λ * p * (max(0, r[i] - y[i])) ** (p - 1) * (-1)

def lp_optimize_centres(centres):
    n = centres.shape[0]
    A = np.zeros((3 * n, n))
    b = np.ones(3 * n)
    for i in range(n):
        A[i, i] = 1
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = centres[i] - centres[j]
            d = np.sqrt(dx**2 + dy**2)
            A[2 * n + i, i] = 1
            A[2 * n + i, j] = -1
            b[2 * n + i] = d - 1e-12
            A[2 * n + j, i] = -1
            A[2 * n + j, j] = 1
            b[2 * n + j] = d - 1e-12
    result = minimize(lambda r: -np.sum(r), [0.25] * n, method='highs', bounds=[(0, 1)] * n, constraints={'type': 'ineq', 'fun': lambda r: b - np.dot(A, r)})
    return result.x, -result.fun

def run_packing():
    np.random.seed(0)
    λ = 5e4
    p = 2
    radius_min = 0.01
    gradient_tolerance = 1e-6
    max_evaluations = 2000
    num_starts = 18
    max_violation = 1e-6

    best_sum = 0
    best_centers = None
    best_radii = None

    for _ in range(num_starts):
        initial_centers = np.random.rand(num_starts * 3).reshape(num_starts, 3)
        initial_radii = np.random.rand(num_starts) * radius_min
        initial_z = np.concatenate((initial_centers, initial_radii[:, None]), axis=1).flatten()
        
        result = minimize(compute_penalty, initial_z, args=(λ, p), method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)] * 3 * num_starts + [(0, 1)] * num_starts, options={'maxiter': max_evaluations, 'gtol': gradient_tolerance})
        if result.success and np.all(result.fun < max_violation):
            final_centers = result['x'][::3].reshape(-1, 2)
            final_radii = lp_optimize_centres(final_centers)[0]
            final_sum = np.sum(final_radii)
            if final_sum > best_sum:
                best_sum = final_sum
                best_centers = final_centers
                best_radii = final_radii

    return best_centers, best_radii, best_sum
