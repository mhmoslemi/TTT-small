import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def build_random_feasible(n, r0=0.05):
    centers = []
    while len(centers) < n:
        x, y = np.random.uniform(0, 1, 2)
        c = np.array([x, y])
        if all(np.sqrt(np.sum((c - np.array(c1))**2)) >= r0 + r1 
               for c1, r1 in centers):
            centers.append((c, r0))
    return zip(*centers)

def check_validity(centers, radii):
    for i, (x, y) in enumerate(centers):
        if not (radii[i] > 0):
            return False
        if not (x - radii[i] >= 0):
            return False
        if not (x + radii[i] <= 1):
            return False
        if not (y - radii[i] >= 0):
            return False
        if not (y + radii[i] <= 1):
            return False
    dists = cdist(centers, centers)**2
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            if (dists[i, j] <= (radii[i] + radii[j])**2 - 1e-12):
                return False
    return True

def objective_function(params, centers):
    radii = np.sqrt(params[:len(centers)])
    centers = np.array([params[len(centers):], params[len(centers):+2*len(centers)]]).T
    return -radii.sum()  # Minimize negative sum for maximization

def simulated_annealing(centers, radii, best_sum):
    n = len(centers)
    T = 1.0
    cool = 0.9995
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    
    params = np.concatenate(([r**2 for r in radii], centers.flatten()))
    bounds = [(0, radii.max() + 1/5) for r in radii] + [(0, 1) for _ in centers.ravel()]
    step_size = (shift_max / T) * (T / 1.0)
    
    best_params = params
    best_sum = -best_sum
    
    for k in range(n_iter):
        T *= cool
        i = np.random.randint(n)
        if np.random.rand() < move_prob:
            dx, dy = np.random.uniform(-shift_max, shift_max, 2)
            new_pos = np.clip(centers[i] + (dx, dy), 0, 1)
            new_params = np.concatenate((params[:len(centers)], new_pos.flatten(), params[2*len(centers):]))
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_r = np.clip(radii[i] + dr, 0, 1)
            new_params = np.concatenate((params[:i*2+2], [new_r], params[i*2+3:]))
        
        if check_validity(np.array([new_params[len(centers):], new_params[len(centers):+2*len(centers)]]).T,
                           np.sqrt(new_params[:len(centers)])):
            ΔObj = -new_params[:len(centers)].sum() + best_sum
            if ΔObj >= 0 or np.random.rand() < np.exp(ΔObj / T):
                best_params = new_params
                best_sum = -new_params[:len(centers)].sum()
                step_size *= 0.95
        
        if k % 1000 == 0:
            shift_max *= 0.95
            radius_max *= 0.95
    
    best_centers = np.array([best_params[len(centers):], best_params[len(centers):+2*len(centers)]]).T
    best_radii = np.sqrt(best_params[:len(centers)])
    return best_centers, best_radii, -best_sum

def deterministic_polish(centers, radii):
    n = len(centers)
    improved = True
    while improved:
        improved = False
        for i in range(n):
            while True:
                r_try = radii[i] + 1e-4
                if any(r_try + radii[j] > np.sqrt(np.sum((centers[i] - centers[j])**2)) - 1e-12 
                       for j in range(n) if i != j):
                    break
                radii[i] = r_try
                improved = True
    
    for _ in range(10):
        for i in range(n):
            for dx, dy in [(0.001, 0), (0, 0.001), (-0.001, 0), (0, -0.001)]:
                new_pos = np.clip(centers[i] + (dx, dy), 0, 1)
                if check_validity(new_pos, radii):
                    centers[i] = new_pos
                    radii[i] += 1e-4
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Feasible initialization
    centers, radii = build_random_feasible(26, r0=0.05)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()

    best_centers, best_radii, best_sum = simulated_annealing(centers, radii, best_sum)

    # Step 3: Deterministic polishing
    best_centers, best_radii = deterministic_polish(best_centers, best_radii)

    # Step 4: Final validation
    validate_packing(best_centers, best_radii)
