import numpy as np
from scipy.optimize import linprog

def lp_optimum(centres):
    n = centres.shape[0]
    A_ub = np.zeros((2 * n + n * (n - 1) // 2, n))
    b_ub = np.zeros(2 * n + n * (n - 1) // 2)
    c = -np.ones(n)
    
    for i in range(n):
        r_i = centres[i, 0]
        x_i = centres[i, 1]
        b_ub[2 * i] = x_i - 1e-12
        b_ub[2 * i + 1] = 1 - x_i - 1e-12
        A_ub[2 * i, i] = 1
        A_ub[2 * i + 1, i] = 1
    
    centre_pairs = []
    dist_pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            centre_pairs.extend([i, j])
            centre_pairs.extend([i, j])
            dist_pairs.extend([dist - 1e-12] * 2)
    
    for i in range(n):
        r_i = centres[i, 0]
        y_i = centres[i, 1]
        b_ub[2 * n + 2 * i] = y_i - 1e-12
        b_ub[2 * n + 2 * i + 1] = 1 - y_i - 1e-12
        A_ub[2 * n + 2 * i, i] = 1
        A_ub[2 * n + 2 * i + 1, i] = 1
    
    for i in range(n):
        for j in range(i + 1, n):
            r_i = centres[i, 0]
            y_i = centres[i, 1]
            r_j = centres[j, 0]
            y_j = centres[j, 1]
            b_ub[2 * n + 2 * i + 4 * i + 2 * j] = r_i + r_j - 1e-12
            b_ub[2 * n + 2 * i + 4 * i + 2 * j + 1] = r_i + r_j - 1e-12
            A_ub[2 * n + 2 * i + 4 * i + 2 * j, i] = 1
            A_ub[2 * n + 2 * i + 4 * i + 2 * j, j] = 1
            A_ub[2 * n + 2 * i + 4 * i + 2 * j + 1, i] = 1
            A_ub[2 * n + 2 * i + 4 * i + 2 * j + 1, j] = 1
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return result.x, -result.fun

def run_packing():
    np.random.seed(42)
    n = 26
    step = 0.1
    step_min = 0.001
    decay_rate = 0.995
    best_sum = -np.inf
    best_centres = np.empty((0, 2))
    best_radii = np.empty((0,))

    for start in range(25):
        centres = np.random.rand(n, 2)
        for it in range(2000):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = np.copy(centres)
            new_centres[idx] = new_c
            new_radii, new_sum = lp_optimum(new_centres)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                best_radii = new_radii
        step *= decay_rate

    final_radii, final_sum = lp_optimum(best_centres)
    return best_centres, final_radii, final_sum
