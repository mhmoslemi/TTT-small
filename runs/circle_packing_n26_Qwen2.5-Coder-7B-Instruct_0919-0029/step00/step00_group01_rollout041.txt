import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:, 0], 1-centers[:, 0], centers[:, 1], 1-centers[:, 1]])
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(distances, np.inf)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([-1]*i + [1] + [0]*(n-i-1))
        b_ub.append(borders[i])
        for j in range(i+1, n):
            A_ub.append([-1]*i + [0]*(j-i-1) + [1] + [0]*(n-j-1))
            b_ub.append(distances[i, j])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x.clip(0), -result.fun

def random_move(centers, step):
    n = centers.shape[0]
    i = np.random.randint(n)
    new_pos = centers[i] + step * np.random.randn(2)
    new_pos = np.clip(new_pos, 0, 1)
    return new_pos

def simulated_annealing_centers(n, max_iter, rng):
    step = 0.05
    T_initial = 0.02
    T_final = 1e-5
    centers = rng.uniform(0, 1, size=(n, 2))
    best_centers = centers.copy()
    best_radii, best_sum = solve_lp(centers)
    
    for iter in range(max_iter):
        i = np.random.randint(n)
        new_center = random_move(centers, step)
        centers[i] = new_center
        current_radii, current_sum = solve_lp(centers)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
        else:
            if np.random.rand() < np.exp((best_sum - current_sum) / T_initial):
                best_sum = current_sum
                best_centers = centers.copy()
        step *= 0.9999
        
        if iter % 1000 == 0:
            T_initial *= 0.9995
    
    return best_centers, best_radii, best_sum

def run_packing():
    n = 26
    max_iter = 30000
    rng = np.random.default_rng(42)
    centers, radii, sum_radii = simulated_annealing_centers(n, max_iter, rng)
    return centers, radii, sum_radii
