import numpy as np
from scipy.optimize import differential_evolution, linprog

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def penalty(v):
    n = 26
    x = v[:n*n].reshape(n, n)
    r = v[n*n:].reshape(n,)
    boundary = np.sum(np.maximum(0, -x[:,:2].flatten())**2 + np.maximum(0, x[:,:2].flatten()-1)**2)
    negative_radius = np.sum(np.maximum(0, -r)**2)
    overlap = np.sum(np.maximum(0, r[:,None] + r[None,:] - np.sqrt(np.sum((x[:,None,:2] - x[None,:,:2])**2, axis=2))))**2)
    return boundary + negative_radius + overlap

def objective(v):
    radius_penalty = penalty(v)
    return -np.sum(v[n*n:]), radius_penalty

def stage1():
    bounds = [(0, 1)] * 2 * 26 + [(0, 0.25)] * 26
    res1 = differential_evolution(objective, bounds, strategy='intermediate', popsize=50, mutation=0.8, crossover=0.7, maxiter=2000, seed=42)
    res2 = differential_evolution(objective, bounds, strategy='intermediate', popsize=50, mutation=0.8, crossover=0.7, maxiter=2000, seed=51)
    return res1 if res1.fun < res2.fun else res2

def stage2(centers):
    n = 26
    bounds = [(0, None)] * 26
    objective = lambda r: -np.sum(r)
    constraints = [{'type': 'ineq', 'fun': lambda r: np.sin(np.pi * np.abs(r - centers[:,0])) * (0.5 - np.abs(r - centers[:,0]))} for centers[:,0] in centers] + [{'type': 'ineq', 'fun': lambda r: np.cos(np.pi * np.abs(r - centers[:,1])) * (0.5 - np.abs(r - centers[:,1]))} for centers[:,1] in centers] + [{'type': 'ineq', 'fun': lambda r, i=i: centers[i][0] - r[i] - 1} for i in range(n)] + [{'type': 'ineq', 'fun': lambda r, i=i: centers[i][1] - r[i] - 1} for i in range(n)] + [{'type': 'ineq', 'fun': lambda r, i=i, j=j: centers[i][0] - centers[j][0] - r[i] - r[j] + 1e-12 for j in range(i+1, n)}]
    r = linprog(objective, A_ub=np.array(constraints), b_ub=np.array([0]*len(constraints)), bounds=bounds, method='highs')
    if r.success:
        return r.x
    return np.zeros(n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    res = stage1()
    centers = res.x[:26*2].reshape(26, 2)
    radii = stage2(centers)
    return centers, radii, np.sum(radii)

# Final result unpacking
centers, radii, sum_radii = run_packing()
print((centers, radii, sum_radii))
