import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize
from scipy.spatial import ConvexHull

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    N = 26
    max_iter = 2000
    tol = 1e-4
    lower_bound = np.zeros(N)
    upper_bound = 0.5 * np.ones(N)
    bounds = np.vstack((lower_bound, upper_bound)).T
    W = 2000
    num_restarts = 20

    def cost_function(z, W):
        positions = z[:N]
        radii = z[N:]
        edge_cases = np.where(radii > 0.5)[0]
        if len(edge_cases):
            for e in edge_cases:
                radii[e] = 0.4
        dist_matrix = cdist(positions.reshape(-1, 2), positions.reshape(-1, 2), 'euclidean')
        overlap_penalties = np.maximum(0, radii[:, None] + radii - dist_matrix).sum()

        boundary_penalties = np.zeros_like(radii)
        for i in range(N):
            boundary_penalties[i] = sum([
                max(0, radii[i] - positions[i, 0]),
                max(0, radii[i] - (1 - positions[i, 0])),
                max(0, radii[i] - positions[i, 1]),
                max(0, radii[i] - (1 - positions[i, 1]))
            ])
        total_penalty = W * (overlap_penalties + boundary_penalties.sum())
        return -np.sum(radii) + total_penalty

    def check_adaptation(soln, W):
        positions = soln[:N]
        radii = soln[N:]
        dist_matrix = cdist(positions.reshape(-1, 2), positions.reshape(-1, 2), 'euclidean')
        violations = np.maximum(0, radii[:, None] + radii - dist_matrix).flatten()
        max_violation = np.max(violations)
        return max_violation < tol, max_violation

    best_radii_sum = 0
    best_solution = None

    for _ in range(num_restarts):
        centers = (np.arange(4).reshape(-1, 1) + np.arange(4)) / 7.5
        perimeter_points = []
        for i in range(4):
            t = np.linspace(0, 1, 10)
            perimeter_points.extend(list(zip(0.1 * t + 0.9 *(i & 1) , 0.1 * (1-t) + 0.9 * (not (i & 1)) )))
            perimeter_points.extend(list(zip(0.1 * (1-t) + 0.9* (not (i & 1)) , 0.1 * t + 0.9*(i & 1) )))
        centers = np.array(perimeter_points)
        remaining_points = np.random.uniform(0.25, 0.75, size=(N-4, 2))
        centers = np.concatenate([centers, remaining_points], axis=0)
        radii = np.random.uniform(0.05, 0.15, N)  # It is re assigned inside the code
        initial_guess = np.concatenate([centers.flatten(), radii])

        result = minimize(cost_function, initial_guess, args=(W,), method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
        if result.success and check_adaptation(result.x, W):
            radii = result.x[N:]
            if radii.sum() > best_radii_sum:
                best_solution = result.x.copy()
                best_radii_sum = radii.sum()
        else:
            W *= 10  # Lower than threshold increase penalty
            new_result = minimize(cost_function, initial_guess, args=(W,), method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})
            if new_result.success and check_adaptation(new_result.x, W):
                radii = new_result.x[N:]
                if radii.sum() > best_radii_sum:
                    best_solution = new_result.x.copy()
                    best_radii_sum = radii.sum()

    if best_solution is None:
        raise ValueError("Optimization did not converge or resulted in invalid solution")

    final_positions = best_solution[:N]
    final_radii = best_solution[N:]
    assert validate_packing(final_positions.reshape(-1, 2), final_radii)[0] == True, "Validation failed, invalid packing"

    return final_positions.reshape(-1, 2), final_radii, final_radii.sum()
