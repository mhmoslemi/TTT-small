import numpy as np
from scipy.optimize import differential_evolution as de, minimize_slsqp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    λ = 1e6
    N = 26
    maxiter_de = 200
    popsize_de = 20
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    X0 = np.array([0.5] * (3 * N))

    def penalty(X):
        penalty_sum = 0
        for i in range(N):
            penalty_sum += np.maximum(0, X[3*i] - X[3*i+2])**2
            penalty_sum += np.maximum(0, 1 - X[3*i] - X[3*i+2])**2
            penalty_sum += np.maximum(0, X[3*i+1] - X[3*i+2])**2
            penalty_sum += np.maximum(0, 1 - X[3*i+1] - X[3*i+2])**2
            for j in range(i + 1, N):
                d2 = (X[3*i] - X[3*j])**2 + (X[3*i+1] - X[3*j+1])**2
                penalty_sum += np.maximum(0, (X[3*i+2] + X[3*j+2])**2 - d2)**2
        return λ * penalty_sum

    def f_global(X):
        return -(X[3*N-3] + X[3*N-2] + X[3*N-1]) + penalty(X)

    # Run DE to find a good starting configuration
    result_de = de(f_global, X0, bounds, popsize=popsize_de, maxiter=maxiter_de, workers=1)
    X_global = result_de.x

    # Local optimization using SLSQP to find a feasible configuration
    def f_local(X):
        all_radius_sum = np.sum(X[3*N-3] + X[3*N-2] + X[3*N-1])
        return -all_radius_sum

    def g_left(X, i): return X[3*i] - X[3*i+2]
    def g_right(X, i): return 1 - X[3*i] - X[3*i+2]
    def g_bot(X, i): return X[3*i+1] - X[3*i+2]
    def g_top(X, i): return 1 - X[3*i+1] - X[3*i+2]
    def g_pair(X, i, j):
        d2 = (X[3*i] - X[3*j])**2 + (X[3*i+1] - X[3*j+1])**2
        return ((X[3*i+2] + X[3*j+2])**2 - d2)

    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': g_left, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': g_right, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': g_bot, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': g_top, 'args': (i,)})
    for i in range(N-1):
        for j in range(i+1, N):
            constraints.append({'type': 'ineq', 'fun': g_pair, 'args': (i, j)})

    result_slow = minimize_slsqp(f_local, X_global, bounds=bounds, constraints=constraints,
                                ftol=1e-8, maxiter=200, iprint=0)[0]

    # Extract the final packing
    centers = result_slow.reshape((N, 3))[:, :2]
    radii = result_slow.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    assert sum_radii > 2.637
    return centers, radii, sum_radii
