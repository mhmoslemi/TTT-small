import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i in range(n):
        for j in range(i+1, n):
            A_ub[i*(n-1)//2+j-2*i-1, i] = 1
            A_ub[i*(n-1)//2+j-2*i-1, j] = 1
            b_ub[i*(n-1)//2+j-2*i-1] = np.linalg.norm(centers[i] - centers[j]) - 1e-12

    for i in range(n):
        x_i, y_i = centers[i]
        r_i = n * (i + 0.5)
        if i % 2 == 0:
            A_ub[len(A_ub) - 4*n + 4*i, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i] = x_i - r_i
            A_ub[len(A_ub) - 4*n + 4*i + 1, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 1] = 1 - x_i
            A_ub[len(A_ub) - 4*n + 4*i + 2, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 2] = y_i - r_i
            A_ub[len(A_ub) - 4*n + 4*i + 3, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 3] = 1 - y_i
        else:
            A_ub[len(A_ub) - 4*n + 4*i, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i] = x_i - r_i
            A_ub[len(A_ub) - 4*n + 4*i + 1, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 1] = 1 - x_i
            A_ub[len(A_ub) - 4*n + 4*i + 2, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 2] = y_i - r_i
            A_ub[len(A_ub) - 4*n + 4*i + 3, i] = 1
            b_ub[len(b_ub) - 4*n + 4*i + 3] = 1 - y_i

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def evaluate(centers_flat):
    centres = centres_flat.reshape((26, 2))
    radii, total = lp_optimum(centres)
    if np.any(radii < 0):
        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))
    return -total

def run_packing():
    n = 26
    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est

    lattice = []
    for ky in range(int(1/h) + 1):
        y = ky * h + r_est
        if y > 1 - r_est:
            break
        for kx in range(int(1/s) + 1):
            x = kx * s + r_est
            if kx % 2 == 1:
                x += r_est / 2
            if x >= 1 - r_est:
                continue
            lattice.append([x, y])

    initial_centres = np.array(lattice[:n]) + 0.5 * (np.random.rand(26, 2) - 0.5)
    result = differential_evolution(evaluate, bounds=[(0, 1)]*2*n, popsize=50, strategy="best1bin", mutation=0.5, crossover=0.7, maxiter=80, init='latinhypercube')
    best_centres = result.x.reshape((26, 2))
    best_radii, best_sum = lp_optimum(best_centres)

    from scipy.optimize import minimize
    def refine(centres_flat):
        centres = centres_flat.reshape((26, 2))
        radii, total = lp_optimum(centres)
        return -total

    result = minimize(refine, best_centres.flatten(), method='Nelder-Mead', bounds=[(0, 1)]*2*n)
    refined_centres = result.x.reshape((26, 2))
    refined_radii, refined_sum = lp_optimum(refined_centres)

    return refined_centres, refined_radii, refined_sum
