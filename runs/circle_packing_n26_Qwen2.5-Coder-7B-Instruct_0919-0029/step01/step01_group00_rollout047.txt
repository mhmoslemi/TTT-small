import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

# Helper function to compute pairwise distances of centers
def compute_pairwise_distances(centers):
    return squareform(pdist(centers))

# Linear Programming solver for computing optimal radii given fixed centers
def lp_optimum(centers, tolerance=1e-12):
    n = centers.shape[0]
    A = np.zeros((n, n))
    for i in range(n):
        A[i, i] = 2

    b = np.zeros(n)
    c = -np.ones(n)

    centers_shifted = centers.reshape(-1)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if not result.success:
        return np.full(n, -1), -1

    radii = np.sqrt(result.x)
    while True:
        distances = compute_pairwise_distances(centers)
        overlaps = distances < radii[:, np.newaxis] + radii[np.newaxis, :] - tolerance
        distances[overlaps] -= 2 * tolerance
        radii = np.sqrt(np.maximum(result.x - (distances / 2), 0))
        if np.allclose(radii, result.x, atol=tolerance):
            break
    return radii, -result.fun
