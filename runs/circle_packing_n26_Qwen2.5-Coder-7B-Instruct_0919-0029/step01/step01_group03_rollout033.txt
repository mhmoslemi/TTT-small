from pulp import LpProblem, LpVariable, lpSum, LpMaximize, value
import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    nx, ny = 14, 14
    dx, dy = 1/nx, 1/ny
    step = 0.075

    P = nx * ny
    X = np.arange(dx/2, 1, dx).reshape(-1,1)
    Y = np.arange(dy/2, 1, dy)
    points = X[:,None] + Y[None,:]

    grid = np.vstack((points.reshape(-1,2), points[:, ::-1]))
    grid = grid[np.argsort(grid[:,0])]

    problem = LpProblem("CirclePacking", LpMaximize)

    y = LpVariable.dicts("y", range(P), cat='Binary')
    r = LpVariable.dicts("r", range(P), lowBound=0, upBound=0.5)

    problem += lpSum(r[i] for i in range(P))

    # Exactly 26 circles
    problem += lpSum(y[i] for i in range(P)) == 26

    # No-overlap constraints
    for i in range(P):
        for j in range(i+1, P):
            d = np.sqrt(np.sum((grid[i] - grid[j])**2))
            problem += r[i] + r[j] <= d + 0.5 * (1 - y[i]) + 0.5 * (1 - y[j])

    # Boundary containment constraints
    for i in range(P):
        problem += r[i] <= grid[i,0]
        problem += r[i] <= 1 - grid[i,0]
        problem += r[i] <= grid[i,1]
        problem += r[i] <= 1 - grid[i,1]

    for i in range(P):
        problem += r[i] <= 0.5 * y[i]

    solver = problem.solve()
