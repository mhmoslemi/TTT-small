def initialize_queue(circles):
    queue = []
    for i in range(len(circles)):
        for j in range(i+1, len(circles)):
            for k in range(j+1, len(circles)):
                curvatures = [1/circle["radius"] if circle["radius"] != 0 else np.inf for circle in circles[i:j+1]]

                    queue.append((new_curvature, center, circles[i:j+1]))
    queue = [(k, c, circles[i:j+1]) for k, c, circles[i:j+1] in queue]

def create_new_circles(circles, gap):

def is_fillable(circles, gap):
    center_i, center_j, center_k = gap["centers"]
    new_curvature = descartes_curvature([1/c["radius"] for c in circles])
    return new_curvature > 0

def generate_gaps(circles, gap):

    new_gaps = []

            new_gaps.append({
                "centers": [center_k, center_j, circles[i]],
                "curvature": descartes_curvature([1/c["radius"] for c in (center_k, center_j, circles[i])])
            })
    return new_gaps

def descartes_curvature(k):
    return (sum(k) + 2*np.sqrt(sum(k[i]*k[j] for i in range(len(k)) for j in range(i+1, len(k)))))**(1/2)

def calculate_center(circles, new_curvature):
    center_i, center_j = circles[:2]
    side_i = center_i["center"] - center_j["center"]
    r_i, r_j = center_i["radius"], center_j["radius"]
    r_new = 1/new_curvature
    center = center_i["center"] + (-side_i[0]*r_i + side_i[1]*r_i - side_i[0]*r_new + side_i[1]*new_curvature + r_j**2 - r_new**2 - side_i[0]*r_i - side_i[1]*new_curvature)**2
    return center

def refine_with_lp(centres):

        A_lhs[i][i] = 1
        for j in range(i):
            A_lhs[i][j] = centres[i] - centres[j]

def push_largest_circles(centres, radii):
    largest_circle_index = np.argmax(radii)
    largest_circle_radius = radii[largest_circle_index]
    if largest_circle_radius * 1.001 > 0.5:
        return centres, radii, sum(radii)
    smallest_radius = np.min(radii)
    radius_increment = min(0.001, (0.5 - largest_circle_radius) / (1 - smallest_radius))
    for i in range(centres.shape[0]):
        if i == largest_circle_index:
            centres[i, largest_circle_index] += radius_increment
        else:
            centres[i, largest_circle_index] -= radius_increment * radii[i] / smallest_radius
    return centres, radii, sum(radii)
