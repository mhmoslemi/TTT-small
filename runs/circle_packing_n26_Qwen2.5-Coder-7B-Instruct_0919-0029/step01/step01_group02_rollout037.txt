import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26
eps = 1e-12

def objective(x):
    radii = x[2::3]
    penalty = np.sum(np.maximum(0, radii - x[0::3]))
    penalty += np.sum(np.maximum(0, radii - (1 - x[0::3])))
    penalty += np.sum(np.maximum(0, radii - x[1::3]))
    penalty += np.sum(np.maximum(0, radii - (1 - x[1::3])))
    for i in range(N):
        for j in range(i+1, N):
            d2 = (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2
            v = (radii[i]**2 + radii[j]**2 - d2)
            if v > 0:
                penalty += v**2
    return -(np.sum(radii) + 1e10 * penalty)

def boundary_left(x, idx):
    return x[idx*3] - (x[idx*3+2] + eps)

def boundary_right(x, idx):
    return (1 - x[idx*3]) - (x[idx*3+2] + eps)

def boundary_bottom(x, idx):
    return x[idx*3+1] - (x[idx*3+2] + eps)

def boundary_top(x, idx):
    return (1 - x[idx*3+1]) - (x[idx*3+2] + eps)

def pair_non_overlap(x, i, j):
    d2 = (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2
    return (radii[i]**2 + radii[j]**2 - d2)

def run_packing():
    bounds = [(0,1)] * 2*np.??? of 26 plus [ (0,0.5)] for radii etc.
    bounds = [(0, 1)]*52 + [(0, 0.5)]*26

    res = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False, workers=1)
    best_x = res.x
    radii = best_x[2::3]
    x_coords = best_x[0::3]
    y_coords = best_x[1::3]

    constraints = []
    for i in range(N):
        constraints.append({'type':'ineq','fun': boundary_left, 'args': (i, )})
        constraints.append({'type':'ineq','fun': boundary_right, 'args': (i, )})
        constraints.append({'type':'ineq','fun': boundary_bottom, 'args': (i, )})
        constraints.append({'type':'ineq','fun': boundary_top, 'args': (i, )})

    for i in range(N):
        for j in range(i+1, N):
            constraints.append({'type':'ineq', 'fun': pair_non_overlap, 'args': (i, j)})

    res_slsqp = minimize(lambda y: objective(y) + 1e10 * penalty(y), x0=best_x, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol':eps, 'maxiter':200})
    centers = res_slsqp.x[0::3]
    radii = res_slsqp.x[2::3]
    sum_radii = np.sum(radii)
    return centers.reshape((N, 2)), radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        x, y = centers[i]
        r = radii[i]
        if x - r < -eps or x + r > 1 + eps or y - r < -eps or y + r > 1 + eps:
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - eps:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
