import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_points = np.array([(i*0.01, j*0.01) for i in range(101) for j in range(101)])
    
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    active_indices = []
    active_radii = np.zeros(26)

    def growth_time(idx):
        point = grid_points[idx]
        x, y = point
        r = radii[idx]
        fx = min(x - r, 1 - x - r, y - r, 1 - y - r)
        fx_safe = max(0, min(fx, min(np.sqrt(np.sum((point - centers) ** 2, axis=1)) - active_radii - r)))
        return fx_safe

    for _ in range(26):
        times = growth_time(np.arange(10201))
        idx = np.argmin(times)
        point = grid_points[idx]
        centers[len(active_indices)] = point
        active_indices.append(len(active_indices))
        
        radius = growth_time(idx)
        active_radii[len(active_indices) - 1] = radius
        radii += radius

    for _ in range(200):
        pressures = np.zeros_like(centers)
        for i, (xi, yi) in enumerate(centers):
            ri = active_radii[i]
            x_out = min(xi, 1 - xi - ri)
            y_out = min(yi, 1 - yi - ri)
            for j in active_indices:
                xj, yj = centers[j]
                rj = active_radii[j]
                dist = np.sqrt((xj - xi) ** 2 + (yj - yi) ** 2)
                if dist <= ri + rj:
                    pressures[i] += (ri + rj - dist) * (xj - xi) / dist
                    pressures[j] += (ri + rj - dist) * (xi - xj) / dist
            pressures[i] += -2 * x_out if x_out > 0 else 2 * x_out if x_out < 0 else 0
            pressures[i] += -2 * y_out if y_out > 0 else 2 * y_out if y_out < 0 else 0
        
        alpha = 0.05
        for k in range(26):
            centers[k] += max(-0.495, min(0.495, pressures[k] * alpha))
            centers[k] = np.clip(centers[k], 0, 1)

        for i, (xi, yi) in enumerate(centers):
            if active_radii[i] > 0:
                x_out = min(xi, 1 - xi - active_radii[i])
                y_out = min(yi, 1 - yi - active_radii[i])
                if x_out > 0 or y_out > 0:
                    min_r = min(np.sqrt(np.sum((centers - centers[i]) ** 2, axis=1)), active_radii[i]) - active_radii[i]
                    centers[i] += max(-0.495, min(0.495, pressures[i] * alpha))
                    centers[i] = np.clip(centers[i], 0, 1)
                    
        for i in range(26):
            xi, yi = centers[i]
            r = active_radii[i]
            fx = min(xi, 1 - xi - r, yi, 1 - yi - r)
            fx_safe = max(0, min(fx, min(np.sqrt(np.sum((centers[:i] - centers[i]) ** 2, axis=1)) - active_radii[:i] - r)))
            if fx_safe < 1e-9:
                active_radii[i] = 0
                active_indices.remove(i)
                centers[i] = np.zeros(2)
                radii -= r
            else:
                active_radii[i] += fx_safe
                radii += fx_safe
                
    for i in range(26):
        xi, yi = centers[i]
        r = active_radii[i]
        fx = min(xi, 1 - xi - r, yi, 1 - yi - r)
        fx_safe = max(0, min(fx, min(np.sqrt(np.sum((centers[:i] - centers[i]) ** 2, axis=1)) - active_radii[:i] - r)))
        active_radii[i] += fx_safe
        radii += fx_safe

    for i in range(26):
        if active_radii[i] > 0:
            active_radii[i] -= 1e-9
            radii -= 1e-9

    return centers, radii, np.sum(radii)
