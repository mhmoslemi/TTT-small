import numpy as np
import pulp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    Delta = 0.075
    nx = int(1 / Delta)
    ny = nx
    P = nx * ny
    
    X = np.arange(0, 1 + Delta, Delta)[:nx]
    Y = np.arange(0, 1 + Delta, Delta)[:ny]
    
    # Set up the linear programming problem
   prob = pulp.LpProblem("Circle_Packing", pulp.LpMaximize)
    
    # Decision variables
    y = [pulp.LpVariable(f'y_{k}', cat='Binary') for k in range(P)]
    r = [pulp.LpVariable(f'r_{k}', lowBound=0, upBound=0.5) for k in range(P)]
    
    # Constraint that exactly 26 circles are chosen
    prob += pulp.lpSum(y) == 26, "Layout_26_circles"
    
    # Overlap constraints
    distances = [[np.sqrt((X[i] - X[j]) ** 2 + (Y[i] - Y[j]) ** 2) for j in range(P)] for i in range(P)]
    for k in range(P):
        for l in range(k + 1, P):
            prob += r[k] + r[l] <= distances[k][l] + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l]), f"No_overlap_k_{k}_l_{l}"
    
    # Boundary constraints
    for k in range(P):
        prob += r[k] <= X[k]
        prob += r[k] <= 1 - X[k]
        prob += r[k] <= Y[k]
        prob += r[k] <= 1 - Y[k]
    
    # Objective: maximize total radius of chosen circles
    prob += pulp.lpSum(r), "Maximize_total_radius"
    
    # Solve problem
    solver = pulp.getSolver("CBC", msg=1)
    solver.options['timeLimit'] = 30
    prob.solve(solver=solver)
    
    # Extract results
    centers = np.zeros((26, 2))
    radii = np.zeros((26,))
    count = 0
    for k in range(P):
        if y[k].varValue > 0:
            centers[count, 0] = X[k]
            centers[count, 1] = Y[k]
            radii[count] = r[k].varValue
            count += 1
    
    return centers[:26, :], radii[:26], np.sum(radii)
