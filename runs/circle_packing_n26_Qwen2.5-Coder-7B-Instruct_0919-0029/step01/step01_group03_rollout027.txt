import numpy as np
import pulp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    nx = ny = 14
    delta = 1 / (nx - 1)
    X = np.arange(nx) * delta
    Y = np.arange(ny) * delta
    P = nx * ny

   isin = np.fromfunction(lambda i, j: 
                            (X[i] - X[j], Y[i] - Y[j]), (P, P, 2))
    dkl = np.sqrt(np.sum(isin ** 2, axis=2))

    y = pulp.LpVariable.dicts("y", range(P), cat="Binary")
    r = pulp.LpVariable.dicts("r", range(P), lowBound=0, upBound=0.5)

    lp = pulp.LpProblem("circle_packing", pulp.LpMaximize)

    objective = pulp.lpSum(r[k] for k in range(P))
    lp += objective

    for k in range(P):
        lp += pulp.lpSum(r[l] for l in range(P) if l != k) <= dkl[k, k] + 0.5 * (1 - y[k])
        for l in range(k + 1, P):
            lp += pulp.lpSum(r[l]) <= dkl[k, l] + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])

    for k in range(P):
        lp += r[k] <= X[k]
        lp += r[k] <= 1 - X[k]
        lp += r[k] <= Y[k]
        lp += r[k] <= 1 - Y[k]
        lp += r[k] <= 0.5 * y[k]

    lp += pulp.lpSum(y[k] for k in range(P)) == 26

    solver = pulp.Solver("CBC", pulp.apis.PulpSolverCommand())
    solver.options['timeLimit'] = 30

    warm_start = {f"y_{k}": 1 if i * ny + j + 1 in [31, 37, 43, 49, 111, 117, 123, 129] else 0
                   for k, (i, j) in enumerate(zip(X, Y))}
    warm_start.update({f"r_{k}": 0.1 if warm_start[f"y_{k}"] else 0.0
                        for k in range(P)})
    lp.setInitialSolution(warm_start)

    status = lp.solve(solver)

    centers = []
    radii = []
    for k in range(P):
        if y[k].varValue == 1:
            centers.append([X[k], Y[k]])
            radii.append(r[k].varValue)

    if pulp.LpStatus[status] == "Optimal":
        return np.array(centers), np.array(radii), sum(radii)
    else:
        return np.array([]), np.array([]), 0.0
