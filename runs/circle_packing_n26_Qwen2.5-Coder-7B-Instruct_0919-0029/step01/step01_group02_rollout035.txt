import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    g = 0.01
    grid_points = np.array([(i*g, j*g) for i in range(101) for j in range(101)])
    n = 26
    
    radii = np.zeros(n)
    active_points = set()
    
    while len(active_points) < n:
        distances_to_borders = np.min(np.abs(grid_points[:, np.newaxis, :] - np.array([0, 1])[:, np.newaxis]), axis=1)
        distances_to_neighbours = np.inf * np.ones((n, n))
        
        for i in range(n):
            if radii[i] > 0:
                continue
            valid_points = grid_points[np.all(grid_points - grid_points[i] > 2 * radii[i], axis=1)]
            valid_distances = np.min(np.abs(valid_points[:, np.newaxis, :] - grid_points[i]), axis=1)
            distances_to_neighbours[i, :len(valid_distances)] = valid_distances
        
        growth_times = np.minimum(distances_to_borders, distances_to_neighbours[np.arange(n), active_points])
        new_active_points = np.argmin(growth_times)
        
        if new_active_points in active_points:
            continue
        
        min_growth_time = growth_times[new_active_points]
        active_points.add(new_active_points)
        radii[new_active_points] = radii[new_active_points] + min_growth_time
        for i in active_points:
            if i != new_active_points:
                radii[i] = radii[i] + min_growth_time
    
    centres = grid_points[list(active_points)]
    
    def pressure(circle_index):
        x, y = centres[circle_index]
        r = radii[circle_index]
        p = [0, 0]
        for j in active_points:
            if j != circle_index:
                dx, dy = centres[j] - centres[circle_index]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r + radii[j]:
                    p[0] += np.sign(dx) / dist
                    p[1] += np.sign(dy) / dist
        if x <= r:
            p[0] -= 1
        elif x >= 1 - r:
            p[0] += 1
        if y <= r:
            p[1] -= 1
        elif y >= 1 - r:
            p[1] += 1
        return np.array(p)
    
    alpha = 0.05
    for _ in range(200):
        for i in range(n):
            p = pressure(i)
            new_x = np.clip(centres[i, 0] + alpha * p[0], 0, 1)
            new_y = np.clip(centres[i, 1] + alpha * p[1], 0, 1)
            centres[i] = [new_x, new_y]
    
    sum_radii = np.sum(radii)
    return centres, radii, sum_radii
