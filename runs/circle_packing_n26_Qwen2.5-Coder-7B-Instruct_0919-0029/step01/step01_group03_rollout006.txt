import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    # Bounding box for each variable (x, y, radius)
    bounds = [(0, 1)] * N + [(0, 0.4)] * N
    
    W_init = 2000
    radius_bound = 0.4
    
    def objective(z, W):
        # Split z into positions and radii
        centers = z[:N].reshape((N, 2))
        radii = z[N:]
        
        # Compute distances matrix
        diff = centers[:, None] - centers[None, :]
        dist = np.sqrt(np.sum(diff ** 2, axis=2))
        
        # Boundary penalties
        boundary_penalties = np.sum(np.maximum(0, radii[:, None] - centers[:, None][:, 0]) ** 2, axis=0)
        boundary_penalties += np.sum(np.maximum(0, radii[:, None] - (1 - centers[:, None][:, 0])) ** 2, axis=0)
        boundary_penalties += np.sum(np.maximum(0, radii[:, None] - centers[:, None][:, 1]) ** 2, axis=0)
        boundary_penalties += np.sum(np.maximum(0, radii[:, None] - (1 - centers[:, None][:, 1])) ** 2, axis=0)
        
        # Overlap penalties
        overlap_penalties = np.sum(np.maximum(0, radii[:, None] + radii[None, :] - dist) ** 2, axis=0) / 2
        
        # Objective
        return -np.sum(radii) + W * (np.sum(boundary_penalties) + np.sum(overlap_penalties))
    
    def initial_radius_guess():
        return np.random.uniform(0.05, 0.15, N)
    
    def generate_initial_position():
        centre_corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
        centre_perimeter = np.linspace(0.1, 0.9, 7)[:-1]
        centre_interior = np.random.uniform(0.25, 0.75, (N - 8, 2))
        return np.vstack([centre_corners, centre_perimeter.reshape(-1, 2), centre_perimeter[::-1].reshape(-1, 2), centre_interior])
    
    best_solution = None
    best_value = float('inf')
    
    for _ in range(30):
        centers_init = generate_initial_position()
        radii_init = initial_radius_guess()
        z0 = np.hstack([centers_init.flatten(), radii_init])
        res = minimize(objective, z0, args=(W_init,), method='L-BFGS-B', bounds=bounds, options={'maxfun': 2000, 'ftol': 1e-9, 'gtol': 1e-5})
        if res.success and res.fun < best_value:
            best_solution = res.x
            best_value = res.fun
    
    if best_solution is None:
        return np.zeros((N, 2)), np.zeros(N), 0.0  # Fallback in case of failure
    
    centers = best_solution[:N].reshape((N, 2))
    radii = best_solution[N:]
    
    # Ensure exact feasibility via LP refinement
    from scipy.optimize import linprog
    
    b = np.sums(radii)
    c = np.full(N, -1.0)
    A = np.hstack((-np.eye(N), np.abs(centers[:, None, :] - centers[None, :, :]), np.eye(N)))
    b_eq = np.full(N, 0.0).reshape(-1, 1)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * N, method='highs')
    if result.success and result.fun >= 0:
        optimal_radii = -result.fun
    else:
        optimal_radii = np.max(0.0, radii)
    
    return centers, optimal_radii, np.sum(optimal_radii)
