import numpy as np
import scipy.spatial as spatial
from scipy.optimize import minimize

def lp_optimum(centers):
    n = centers.shape[0]
    d = scipy.spatial.distance_matrix(centers, centers)
    pairs = np.triu_indices(n, k=1)
    c = -np.ones(n)
    A = np.zeros((2 * n - 2, n))
    A[:n, pairs[0]] = -1
    A[:n, pairs[1]] = -1
    A[n:, pairs[0]] = 1
    A[n:, pairs[1]] = 1
    b = -d[pairs]
    bounds = [(0, None)] * n
    result = minimize(c, np.ones(n), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda x: -b + np.dot(A, x)})
    radii = result.x
    sum_radii = -result.fun
    return radii, sum_radii

def run_packing():
    n = 26
    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    M = 32
    P = []
    for k in range(M):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        x_off = 0 if k % 2 == 0 else r_est
        for n in range(M):
            x = r_est + x_off + n * s
            if x > 1 - r_est:
                break
            P.append((x, y))
