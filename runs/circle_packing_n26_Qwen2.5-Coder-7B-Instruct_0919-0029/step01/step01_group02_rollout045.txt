import numpy as np

g = 0.01
nx, ny = int(1.0 / g) + 1, int(1.0 / g) + 1
grid_points = np.array([(i * g, j * g) for i in range(nx) for j in range(ny)])

def min_distance_to_boundary(center):
    x, y = center
    return min(x, 1 - x, y, 1 - y)

def min_distance_to_circle(center, circles, radii):
    return np.inf if not circles else np.min(np.sqrt(np.sum((center - circles) ** 2, axis=1)) - radii)

def find_next_point_to_activate(circles, radii):
    grid_indices = np.arange(len(grid_points))
    active_indices = np.where(radii > 0)[0]
    inactive_indices = np.setdiff1d(grid_indices, active_indices)
    distances_to_active_circles = min_distance_to_circle(grid_points[inactive_indices], circles[active_indices], radii[active_indices])
    distances_to_boundary = min_distance_to_boundary(grid_points[inactive_indices])
    times_to_contact = np.minimum(distances_to_boundary - radii[inactive_indices], distances_to_active_circles - radii[inactive_indices, np.newaxis])
    times_to_contact[np.isinf(times_to_contact)] = np.inf
    min_time_to_contact = np.min(times_to_contact)
    min_indexses = np.where(times_to_contact == min_time_to_contact)[0]
    next_point_index = inactive_indices[min_indexses[np.random.choice(len(min_indexses))]]
    return next_point_index

def grow_circles(circles, radii):
    while np.sum(radii) < 2.637:
        next_point_index = find_next_point_to_activate(circles, radii)
        dist_to_boundary = min_distance_to_boundary(grid_points[next_point_index])
        dist_to_circles = min_distance_to_circle(grid_points[next_point_index], circles, radii)
        t_min = min(dist_to_boundary - radii[next_point_index], dist_to_circles - radii)
        circles[next_point_index] += [t_min]
        radii[next_point_index] += t_min
        if np.sum(radii) >= 2.637:
            break
        active_indices = np.where(radii > 0)[0]
        for i in active_indices:
            circles[i] += [t_min]
            radii[i] += t_min
    return circles, radii

def optimize_radii(circles, radii):
    for _ in range(200):
        pressures = np.zeros_like(circles)
        for i in range(len(circles)):
            pressure = np.zeros(2)
            x, y = circles[i]
            if x >= 1 or x <= 0:
                pressure[0] -= np.sign(x) * g / 2
            if y >= 1 or y <= 0:
                pressure[1] -= np.sign(y) * g / 2
            for j in range(len(circles)):
                if i != j:
                    dx, dy = circles[i] - circles[j]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < abs(radii[i] + radii[j]):
                        pressure += (radii[i] + radii[j] - dist) * np.array([dx, dy]) / 2
            pressures[i] = pressure
        for i in range(len(circles)):
            circles[i] += pressures[i] * 0.05
            circles[i] = np.clip(circles[i], 0, 1)
            r = min(min_distance_to_boundary(circles[i]), min_distance_to_circle(circles[i], circles, radii))
            if r > radii[i]:
                radii[i] = r
    return circles, radii
