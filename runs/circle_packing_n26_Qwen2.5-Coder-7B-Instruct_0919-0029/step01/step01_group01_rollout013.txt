import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def acceptance_probability(delta_s, temperature):
    return np.exp(delta_s / temperature)

def run_packing():
    # Initial setup
    num_circles = 26
    steps_per_T = 500
    T0 = 0.1
    Tf = 1e-4
    alpha = 0.9995
    initial_step_size = 0.1
    
    # Step 1: Construct initial geometry
    step = initial_step_size
    centers = np.zeros((num_circles, 2))
    radius_init = 0.14
    hexagon_edge = np.sqrt(3) * radius_init
    x_offset = hexagon_edge / 2
    y_offset = hexagon_edge * np.sqrt(3) / 2
    row_count = 11
    col_count = 6

    for i in range(row_count):
        row_offset = y_offset * i
        for j in range(col_count):
            if (i + j) % 2 != 0:
                continue
            x = x_offset * (j + 1 / 2)
            y = row_offset + y_offset / 2
            if i < row_count - 1:
                y += y_offset / 2
            centers[i * col_count + j] = [x, y]

    # Step 2: Evaluate initial configuration
    initial_radii, initial_sum = lp_optimum(centers)
    max_sum = initial_sum
    best_centers = centers.copy()

    # Step 3: Main loop
    T = T0
    iteration_count = int(np.log(Tf / T0) / np.log(alpha))
    for k in range(iteration_count):
        for _ in range(steps_per_T):
            i = np.random.randint(num_circles)
            delta = np.random.randn(2) * step
            centers_new = centers.copy()
            centers_new[i] += delta
            centers_new[i] = np.clip(centers_new[i], 0, 1)
            new_radii, new_sum = lp_optimum(centers_new)
            delta_s = new_sum - max_sum
            
            if delta_s > 0 or np.random.rand() < acceptance_probability(delta_s, T):
                centers = centers_new
                max_sum = new_sum
                
            if max_sum > best_sum:
                best_sum = max_sum
                best_centers = centers.copy()

        T *= alpha
        step *= 0.999  # Optionally reduce step size with temperature
        
    # Step 4: Final polishing
    result = minimize(lambda x: -lp_optimum(x[:26].reshape(num_circles, 2))[1], 
                    centers.flatten(), 
                    constraints='ineq', 
                    bounds=[np.zeros(52), np.inf*52], 
                    method='L-BFGS-B')
    final_radii, final_sum = lp_optimum(result.x[:26].reshape(num_circles, 2))
    
    return best_centers, final_radii, max_sum
