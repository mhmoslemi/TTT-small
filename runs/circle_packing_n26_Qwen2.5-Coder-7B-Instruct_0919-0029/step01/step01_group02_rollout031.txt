import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Circle center positions
    centers = np.array([
            [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
            [0.13, 0.3], [0.33, 0.3], [0.53, 0.3], [0.73, 0.3], [0.93, 0.3],
            [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
            [0.13, 0.7], [0.33, 0.7], [0.53, 0.7], [0.73, 0.7], [0.93, 0.7],
            [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
            [0.5, 1.0]
        ])
    N = centers.shape[0]

    # Objective vector: maximize sum of radii
    c = np.ones(N)

    # Constraint matrix A from non-negativity and boundary conditions
    A = np.zeros((2*N, N))
    A[:N, range(N)] = 1  # Non-negativity
    A[N:, range(N)] = -1
    A[N:, range(N)] *= np.column_stack([centers, centers])

    # Compute maximum allowed radius for each circle based on boundary conditions
    max_radius = np.array([min(x, 1-x, y, 1-y) for x, y in centers])

    # Convert max_radius to a linear inequality matrix
    b = np.concatenate((-max_radius, max_radius))

    # Solve the LP problem
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if result.success:
        radii = result.x
        # Trim radii to avoid boundary violations
        radii -= 1e-9
    else:
