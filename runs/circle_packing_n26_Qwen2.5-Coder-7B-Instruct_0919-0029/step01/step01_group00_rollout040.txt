import math
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import NonlinearConstraint, Bounds

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 2*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = 2*(1 - 1e-12)
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - 1e-12
        idx += 1
        A_ub[idx, i] = -1
        b_ub[idx] = -1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = minimize(c, x0=np.ones(n)/n, method='highs', bounds=bounds, constraints=A_ub <= b_ub)
    radius_vector = result['x']
    sum_radii = -result['fun']
    return radius_vector, sum_radii

def run_packing():
    n = 26
    r_est = math.sqrt(0.9069 / (math.pi * n))
    h = math.sqrt(3) * r_est
    s = 2 * r_est
    centers = []
    for k in range(int((1 - r_est)/h) + 1):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        x_even = r_est + np.arange(int((1 - 2*r_est - y)/s), dtype=float) * s
        centers.extend(zip(x_even, [y] * x_even.size))
        if (k % 2 == 1):
            x_odd = r_est + r_est + np.arange(int((1 - 2*r_est - y)/s), dtype=float) * s
            centers.extend(zip(x_odd, [y] * x_odd.size))
    centers = np.array(centers)
    
    best_sum = -np.inf
    best_centers = None
    
    for subset in combinations(centers, n):
        subset = np.array(subset)
        radii, sum_radii = lp_optimum(subset)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = subset
    
    def ref_obj(x):
        return -lp_optimum(x.reshape(n, 2))[1]
    
    initial_simplex = np.vstack((best_centers.flatten(), (best_centers + np.random.uniform(-0.005, 0.005, size=best_centers.shape)).flatten())).T
    result = minimize(ref_obj, initial_simplex, method='Nelder-Mead', bounds=Bounds(0, 1), options={'maxiter': 30, 'tol': 1e-4})
    final_radii, final_sum = lp_optimum(result['x'].reshape(n, 2))
    
    if final_sum < 2.637:
        raise Exception("Cannot deliver better results for 26 circles")
    
    return best_centers, final_radii, final_sum

from itertools import combinations
print(run_packing())
