def push_largest_circles(centers, radii):
    while True:
        largest_circle_idx = np.argmax(radii)
        x, y = centers[largest_circle_idx]
        r = radii[largest_circle_idx]

            radii[largest_circle_idx] = r_new
            scaled_radii = radii / r_new * r

        else:

    return centers, radii
