import numpy as np
from pulp import LpProblem, LpMaximize, LpBinary, LpVariable, value
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize parameters
    delta = 0.075
    nx = ny = 14
    P = nx * ny
    X = np.arange(0, 1 + delta, delta)
    Y = np.arange(0, 1 + delta, delta)

    # Create variables
    prob = LpProblem("CirclePacking", LpMaximize)
    y = [LpVariable(f'y_{k}', cat='Binary') for k in range(P)]
    r = [LpVariable(f'r_{k}', lowBound=0, upBound=0.5, cat='Continuous') for k in range(P)]

    # Constraints
    prob += sum(y[k] for k in range(P)) == 26
    for k in range(P):
        for l in range(k + 1, P):
            dist_km = np.sqrt((X[k] - X[l]) ** 2 + (Y[k] - Y[l]) ** 2)
            prob += r[k] + r[l] <= dist_km + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])
        prob += r[k] <= X[k]
        prob += r[k] <= 1 - X[k]
        prob += r[k] <= Y[k]
        prob += r[k] <= 1 - Y[k]
        prob += r[k] <= 0.5 * y[k]

    # Objective
    prob += sum(r[k] for k in range(P))

    # Warm start from greedy solution
    greedy_solution = find_greedy_solution(X, Y)
    for k in range(P):
        prob.setLowBound(r[k].name, greedy_solution[k][2] if greedy_solution[k][1] else 0)
        prob.setInitialValue(r[k].name, greedy_solution[k][2] if greedy_solution[k][1] else 0)
        prob.setLowBound(y[k].name, 0 if greedy_solution[k][1] else 1)

    # Solve problem with time limit
    prob.solve(timeLimit=30)

    # Extract results
    centers = []
    radii = []
    for k in range(P):
        if value(y[k]) == 1:
            centers.append([X[k], Y[k]])
            radii.append(value(r[k]))

    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def find_greedy_solution(X, Y):
    greedy_centers = []
    distances = np.zeros((len(X), len(X)))
    for i in range(len(X)):
        for j in range(i + 1, len(X)):
            distances[i, j] = distances[j, i] = np.sqrt((X[i] - X[j]) ** 2 + (Y[i] - Y[j]) ** 2)

    while len(greedy_centers) < 26:
        max_radius = 0
        next_center = None
        for k in range(len(X)):
            if k not in set(j for _, j, _ in greedy_centers):
                candidates = [(distances[k, j], j) for j in range(len(X)) if j not in set(j for _, j, _ in greedy_centers)]
                if candidates:
                    d_min = min(d for d, _ in candidates)
                    radius = (d_min / 2) - 1e-12
                    if radius > max_radius:
                        max_radius = radius
                        next_center = (X[k], Y[k], radius)
        greedy_centers.append(next_center)

    return greedy_centers

# Example usage
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of Radii:", sum_radii)
