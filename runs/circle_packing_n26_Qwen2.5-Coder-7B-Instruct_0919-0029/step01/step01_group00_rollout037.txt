import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def calculate_centered_lattice(density, radius, num_circles):
    side_length = np.sqrt(radius * 8 * num_circles / density)
    lattice = np.arange(0, side_length, radius * np.sqrt(2))
    lattice = np.meshgrid(lattice, lattice)
    return np.vstack(lattice).T

def objective(radii):
    return -np.sum(radii)

def constraint_matrix(centers, radii):
    n = len(radii)
    A = np.zeros((n + n + n, n))
    b = np.zeros(n + n + n)
    for i in range(n):
        A[i, i] = 1
        b[i] = np.sqrt(2) * radii[i]
    for i, j in zip(*np.triu_indices(n, k=1)):
        A[n + 2*i, j] = radii[i]
        A[n + 2*i + 1, j] = -radii[i]
        A[n + 2*j, i] = radii[j]
        A[n + 2*j + 1, i] = -radii[j]
        b[n + 2*i + 2] = np.linalg.norm(centers[i] - centers[j])
    return A, b

def run_packing():
    num_circles = 26
    initial_guess = 0.05 * np.ones(num_circles)
    initial_centers = calculate_centered_lattice(20, 0.05, num_circles)
    
    def objective_with_centers(radii):
        centers = initial_centers * (radii / np.max(radii))
        return objective(radii)
    
    def constraint_with_centers(radii):
        centers = initial_centers * (radii / np.max(radii))
        A, b = constraint_matrix(centers, radii)
        return {'type': 'ineq', 'fun': lambda r: b - np.dot(A, r)}
    
    constraints = [constraint_with_centers(initial_guess)]
    bounds = [(0, None)] * num_circles
    result = minimize(objective_with_centers, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)
    optimized_radii = result.x
    optimized_centers = initial_centers * (optimized_radii / np.max(optimized_radii))
    return optimized_centers, optimized_radii, -result.fun
