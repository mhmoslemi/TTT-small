import numpy as np

def construct_initial_circles():
    mid = 1 / np.sqrt(2)
    C1 = (np.array([0, 0]), mid)
    C2 = (np.array([1, 0]), mid)
    C3 = (np.array([0, 1]), mid)
    C4 = (np.array([1, 1]), mid)
    return [C1, C2, C3, C4]

def apollonian_packing(circles, target_count=26):
    q = []
    for i in range(len(circles)):
        for j in range(i+1, len(circles)):
            for k in range(j+1, len(circles)):
                x1, r1 = circles[i]
                x2, r2 = circles[j]
                x3, r3 = circles[k]
                a = 2 * ((r1 + r2) * (r1 + r3)**2 + (r2 + r3) * (r2 + r1)
                           **2 + (r3 + r1) * (r3 + r2)**2 - 3 * r1**2 * r2**2
                           - 3 * r2**2 * r3**2 - 3 * r3**2 * r1**2)
                b = -4 * (x1[0]*(r1**2 + r2**2 + r1 * r2) + x2[0] *
                           (r2**2 + r3**2 + r2 * r3) + x3[0]*(r3**2 + r1**2 + r3 * r1))
                c = 4 * (r1**2 * r2**2 * r3**2 + 2 * x1[0] * x2[0]:
                           (r1**3 * r2 + r1 * r2**3) + 2 * x2[0] * x3[0]:
                           (r2**3 * r3 + r2 * r3**3) + 2 * x3[0] * x1[0]:
                           (r3**3 * r1 + r3 * r1**3))
                if a > 0:
                    k_new = (-b + np.sqrt(b**2 - 4 * a * c)) / (2 * a)
                    if k_new > 0 and k_new <= 1.0:
                        y1, y2, y3 = x1 + r1 * np.array([1, 1]), x2 + \
                                    r2 * np.array([1, -1]), x3 + r3 * np.array([-1, 1])
                        if 0 <= y1[0] <= 1 and 0 <= y1[1] <= 1 and \
                           0 <= y2[0] <= 1 and 0 <= y2[1] <= 1 and \
                           0 <= y3[0] <= 1 and 0 <= y3[1] <= 1:
                            new_circle = (y1, 1 / (2 * k_new))
                            circles.append(new_circle)
                            q.append((-new_circle[1], (i, j, k, new_circle)))
    q.sort()
    queue = q[:target_count-4]
    circles = circles[:4] + [x[3] for x in queue]
    return circles

def refine_with_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    for i in range(n):
        A_ineq.append(np.concatenate((np.zeros(n), -np.eye(n)[i])))
        A_ineq[-1][:i] = -1
        A_ineq[-1][i+1:] = -1
        A_ineq[-1][n+i] = 1
        A_ineq.append(np.concatenate((np.zeros(n), np.eye(n)[i])))
        A_ineq[-1][:i] = -1
        A_ineq[-1][i+1:] = -1
        A_ineq[-1][n+i] = -1
        for j in range(i+1, n):
            A_ineq.append(np.concatenate((np.zeros(n), np.zeros(n),
                                         np.eye(n)[i],
                                         attempted_zeros = np.zeros(n-i-1),
                                         np.eye(n)[j],
                                         attempted_zeros = np.zeros(n-j-1))))
            A_ineq[-1][:i] = -1
            A_ineq[-1][i+1:] = -1
            A_ineq[-1][n+i] = 1
            A_ineq[-1][2*n+i] = -1
            A_ineq[-1][2*n+j] = 1
            A_ineq[-1][3*n+i] = dists[i, j] - 1e-12

def push_largest_circles(centers, radii):
    max_radius_index = np.argmax(radii)
    max_circle_center, max_circle_radius = centers[max_radius_index], radii[max_radius_index]
    dist_x1, dist_y1, dist_x2, dist_y2 = max_circle_center[0] - max_circle_radius, \
                                        max_circle_center[1] - max_circle_radius, \
                                        max_circle_center[0] + max_circle_radius, \
                                        max_circle_center[1] + max_circle_radius
    push_radius = 1e-4
    sum_radii_before = np.sum(radii)
    while dist_x1 >= 0 and dist_y1 >= 0 and dist_x2 <= 1 and dist_y2 <= 1:
        new_radii = radii.copy()
        new_radii[max_radius_index] += push_radius
        updated_radii, sum_radii_after = refine_with_lp(centers)
        if sum_radii_after > sum_radii_before and np.any(updated_radii < 0):
            push_radius /= 2
        else:
            return updated_radii, sum_radii_after
    return radii, sum_radii_before

def run_packing():
    initial_circles = construct_initial_circles()
    circles = apollonian_packing(initial_circles, 50)
    circles = circles[:26]
    centers = np.array([c[0] for c in circles])
    radii = np.array([c[1] for c in circles])
    summed_r = np.sum(radii)
    refined_radii, refined_sum_radii = refine_with_lp(centers)
    if refined_sum_radii > summed_r - 1e-4:
        radii = refined_radii
        summed_r = refined_sum_radii
    final_radii, final_sum_radii = push_largest_circles(centers, radii)
    return centers, final_radii, final_sum_radii
