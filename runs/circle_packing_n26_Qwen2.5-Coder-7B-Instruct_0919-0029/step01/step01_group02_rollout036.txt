import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the skeleton centers
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.13, 0.3], [0.33, 0.3], [0.53, 0.3], [0.73, 0.3], [0.93, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.13, 0.7], [0.33, 0.7], [0.53, 0.7], [0.73, 0.7], [0.93, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
        [0.5, 1.0]
    ])
    
    # Define the problem
    N = len(centers)
    c = np.ones(N)
    A = []
    b = []
    
    # Boundary constraints
    for i in range(N):
        x, y = centers[i]
        A.append([x, -1, 0 if x != 0 else np.inf, 0 if x != 1 else np.inf, 0])
        A.append([0, y, 0 if y != 0 else np.inf, 0 if y != 1 else np.inf, -1])
        b.extend([x, y, -1e-12, -1e-12, 0])
