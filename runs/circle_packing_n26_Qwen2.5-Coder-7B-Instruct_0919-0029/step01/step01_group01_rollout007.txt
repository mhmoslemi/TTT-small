import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def sa_optimization(centers):
    n = centers.shape[0]
    T0 = 0.1
    Tf = 1e-4
    alpha = 0.9995
    delta = 0.1
    steps_per_T = 500
    max_iter = 50000

    s_best = -np.inf
    r_best = np.zeros(n)
    centers_best = np.zeros_like(centers)

    current_centers = centers.copy()
    s_current, _ = lp_optimum(current_centers)
    r_current, _ = lp_optimum(current_centers)

    for T in np.logspace(np.log10(T0), np.log10(Tf), num=int(max_iter / steps_per_T)):
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            delta_x = np.random.normal(0, delta)
            delta_y = np.random.normal(0, delta)
            new_c = np.clip(np.array([current_centers[i, 0] + delta_x, current_centers[i, 1] + delta_y]), 0, 1)
            new_centers = current_centers.copy()
            new_centers[i] = new_c

            r_new, s_new = lp_optimum(new_centers)
            delta_s = s_new - s_current

            if delta_s >= 0 or np.random.rand() < np.exp(delta_s / T):
                current_centers = new_centers
                s_current = s_new
                r_current = r_new

            if s_new > s_best:
                s_best = s_new
                r_best = r_new
                centers_best = current_centers.copy()

    return centers_best, r_best, s_best

    for _ in range(5):  # Refine the initial configuration with greedy radius optimization

    if result.success:
