import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    r = 0.15
    n = 26
    alpha = 0.02

    def get_center_positions():
        x, y = np.meshgrid(np.arange(0.5, 1, 0.2), np.arange(0.5, 1, 0.2 * np.cos(np.pi / 3)))
        points = np.vstack([x.ravel(), y.ravel()]).T
        np.random.shuffle(points)
        return points[:n, :]

    def greedy_assign_radii(order, centers):
        radii = np.zeros(n)
        for k, i in enumerate(order):
            xi, yi = centers[i]
            rho_left = xi
            rho_right = 1 - xi
            rho_bottom = yi
            rho_top = 1 - yi
            r_max = min(rho_left, rho_right, rho_bottom, rho_top)
            for j in order[:k]:
                xj, yj = centers[j]
                d_ij = np.sqrt((xi - xj)**2 + (yi - yj)**2) - radii[j]
                r_max = min(r_max, d_ij)
            radii[i] = max(r_max, 1e-6)
        return radii

    def accept_move(order_new, order_old, current_sum, new_sum):
        temperature = 0.995
        if new_sum >= current_sum:
            return True
        return np.random.rand() < np.exp((new_sum - current_sum) / temperature)

    def refine_radii(order, centers):
        c = np.vstack(centers[order])
        A = []
        b = []
        for i in range(n):
            A.append(np.hstack([np.eye(n)[i], [1] * n]))
            b.append(1)
            A.append(np.hstack([np.eye(n)[i], -np.eye(n)[i], [1] * (2 * n)]))
            b.append(np.inf)
            A.append(np.hstack([-np.eye(n)[i], np.eye(n)[i], [1] * (2 * n)]))
            b.append(np.inf)
        for i in range(n):
            for j in range(i + 1, n):
                A.append(np.hstack([np.eye(n)[i], np.eye(n)[j], -2 * np.eye(n)[i] - 2 * np.eye(n)[j], [1] * (4 * n)]))
                b.append(np.sqrt(np.sum((centers[order[i]] - centers[order[j]])**2) - 1e-4)
        A = np.array(A)
        b = np.array(b)
        res = linprog([-r] * n, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            raise ValueError("LP refinement failed")
        return res.x

    np.random.seed(42)
    best_order = None
    best_sum = 0
    T = 0.5
    N_steps = 30 * 1000

    for _ in range(N_steps):
        order_new = np.random.permutation(n)
        radii_new = greedy_assign_radii(order_new, get_center_positions())
        new_sum = np.sum(radii_new)
        if accept_move(order_new, best_order, best_sum, new_sum):
            best_order = order_new
            best_sum = new_sum

    centers = get_center_positions()[best_order]
    final_radii = refine_radii(best_order, centers)
    return centers, final_radii, np.sum(final_radii)
