import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import minimize

def shuffle_points(points):
    np.random.shuffle(points)
    return points

def density(f):
    def wrapper(centers):
        return -f(centers)
    return wrapper

def get_initial_centers():
    return np.random.rand(26, 2)

def optimize_centers(centers, target_radius):
    def objective_function(x):
        return -np.sum(x)

    n = centers.shape[0]
    triangulation = Delaunay(centers)
    simplices = triangulation.simplices
    vertices = triangulation.points[simplices]
    dists = np.sqrt(np.square(vertices[:, :, 0, np.newaxis] - vertices[:, :, :, np.newaxis, 0]) + 
                    np.square(vertices[:, :, 1, np.newaxis] - vertices[:, :, :, np.newaxis, 1]))

    A = np.zeros((n + 2*n*(n-1)//2, n))
    b = np.zeros(n + 2*n*(n-1)//2)
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0] - 1e-12
        A[i + n, i] = 1
        b[i + n] = 1 - centers[i, 0] - 1e-12
        A[i + n*2, i] = 1
        b[i + n*2] = centers[i, 1] - 1e-12
        A[i + n*2 + 1, i] = 1
        b[i + n*2 + 1] = 1 - centers[i, 1] - 1e-12

    for i in range(n):
        for j in range(i + 1, n):
            if i != j:
                idx = i * n + j
                A[idx, i] = 1
                A[idx, j] = 1
                A[idx + n*(n-1)//2, i] = 2
                A[idx + n*(n-1)//2, j] = 2
                b[idx] = dists[i][j] - 1e-12
                b[idx + n*(n-1)//2] = 2 * target_radius - 1e-12

    return objective_function, A, b

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_starts = 20
    num_iter = 1000
    step = 0.05
    decay_factor = 0.995
    best_centers = None
    best_radii = None
    best_sum_radii = -np.inf

    for _ in range(num_starts):
        centers = get_initial_centers()
        centers = shuffle_points(centers)
        for _ in range(num_iter):
            idx = np.random.randint(26)
            delta = np.random.uniform(-step, step, size=2)
            new_centers = centers.copy()
            new_centers[idx] = np.clip(centers[idx] + delta, 0, 1)
            radii, _, sum_radii = optimize_radii(new_centers, step)
            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = new_centers
                best_radii, _ = optimize_radii(best_centers, step)

        step = max(step * decay_factor, 0.00001)

    if best_centers is not None:
        final_radii, _ = optimize_radii(best_centers, step)
        return best_centers, final_radii, np.sum(final_radii)
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0
