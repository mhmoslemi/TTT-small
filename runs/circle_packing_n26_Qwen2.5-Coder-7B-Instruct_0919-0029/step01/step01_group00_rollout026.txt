import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A = np.zeros((n * (n - 1) // 2 + 2 * n, n))
    b = np.zeros(A.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = 0.1  # Using a heuristic estimate for the radius step
        A[idx, i] = 1
        b[idx] = x + r
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - x + r
        idx += 1
        A[idx, i] = 1
        b[idx] = y + r
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - y + r
        idx += 1

    c = -np.ones(n)
    bounds = [(0, 1) for _ in range(n)]  # Radii constrained to [0, 1]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x, -res.fun
