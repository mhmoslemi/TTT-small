import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial import distance_matrix

global N
N = 26

def objective(x):
    arr = x.reshape((N, 3))
    x_coords, y_coords, r = arr[:, 0], arr[:, 1], arr[:, 2]
    sum_r = np.sum(r)
    penalty = 0.0
    for i in range(N):
        penalty += np.maximum(0.0, r[i] - x_coords[i])**2
        penalty += np.maximum(0.0, r[i] - (1 - x_coords[i]))**2
        penalty += np.maximum(0.0, r[i] - y_coords[i])**2
        penalty += np.maximum(0.0, r[i] - (1 - y_coords[i]))**2
    dists = distance_matrix(arr[:, :2], arr[:, :2])
    for i in range(N):
        for j in range(i+1, N):
            d2 = dists[i, j]**2
            v = (r[i] + r[j])**2 - d2
            if v > 0:
                penalty += 1e3 * (v**0.5)
    return -sum_r + 1e4 * penalty

def boundary_left(x, idx):
    return x[idx*3] - x[idx*3+2]

def boundary_right(x, idx):
    return 1 - x[idx*3] - x[idx*3+2]

def boundary_bottom(x, idx):
    return x[idx*3+1] - x[idx*3+2]

def boundary_top(x, idx):
    return 1 - x[idx*3+1] - x[idx*3+2]

def pair_non_overlap(x, i, j):
    d2 = distance_matrix([x[i*3:3*(i+1)], x[j*3:3*(j+1)]], [x[i*3:3*(i+1)], x[j*3:3*(j+1)]])**2
    return (x[i*3+2] + x[j*3+2])**2 - d2

constraints = [{'type':'ineq', 'fun': boundary_left, 'args': (i,)} for i in range(N)]
for i in range(N):
    constraints.append({'type':'ineq', 'fun': boundary_right, 'args': (i,)})
    constraints.append({'type':'ineq', 'fun': boundary_bottom, 'args': (i,)})
    constraints.append({'type':'ineq', 'fun': boundary_top, 'args': (i,)})

for i in range(N):
    for j in range(i+1, N):
        constraints.append({'type':'ineq', 'fun': pair_non_overlap, 'args': (i, j)})

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    initradii = np.random.rand(N) * 0.5
    initg = np.random.rand(N, 2) * 0.8 + 0.1
    initx = np.concatenate((initg, initradii[:, np.newaxis]), axis=1).flatten()
    bounds = [(0, 1)]*2*N + [(0, 0.5)]*N
    res = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False, workers=-1)
    best_x = np.maximum(0, res.x.reshape((N, 3)), 1e-9)
    sum_r = np.sum(best_x[:, 2])
    valid, _ = validate_packing(best_x[:, :2], best_x[:, 2])
    if not valid:
        res = minimize(objective, best_x.flatten(), method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol':1e-6, 'maxiter':200})
        best_x = np.maximum(0, res.x.reshape((N, 3)), 1e-9)
        sum_r = np.sum(best_x[:, 2])
    return best_x[:, :2], best_x[:, 2], sum_r
