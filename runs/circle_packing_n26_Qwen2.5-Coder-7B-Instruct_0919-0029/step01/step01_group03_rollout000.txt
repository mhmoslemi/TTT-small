import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

N_CIRCLES = 26
CIRCLE_RADIUS_MIN = 0.01

def lp_radii(centers):
    radii_from_centers = np.zeros(N_CIRCLES)
    boundary_dist = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1]))
    radii_from_centers = np.maximum(radii_from_centers, boundary_dist)
    pairwise_dist = cdist(centers, centers)
    np.fill_diagonal(pairwise_dist, 1)  # Avoid division by 0
    pairwise_radius_overlap = np.maximum(0, pairwise_dist - radii_from_centers[:, np.newaxis] - radii_from_centers[np.newaxis, :])
    c = -np.sum(pairwise_radius_overlap, axis=1)
    A_ub = np.concatenate((np.eye(N_CIRCLES), -np.eye(N_CIRCLES)), axis=1)
    b_ub = np.zeros(N_CIRCLES * 2)
    bounds = [(CIRCLE_RADIUS_MIN, None)] * N_CIRCLES
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    best_radii = result.x, np.sum(result.fun)
    return best_radii

def jitter_layout(centers, sigma=0.05, scale=1.0):
    n = centers.shape[0]
    perturb = np.random.normal(0, sigma, (n, 2))
    jittered = centers + scale * perturb
    jittered = np.clip(jittered, 0, 1)
    return jittered

def run_packing() -> tuple:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = np.zeros((N_CIRCLES, 2))
    
    for _ in range(100):  # Number of random restarts
        centers = np.random.rand(N_CIRCLES, 2)
        for _ in range(10):  # Number of locally best attempts per restart
            jittered = jitter_layout(centers)
            _, sum_radii = lp_radii(jittered)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = jittered
    
    _, final_rdius = lp_radii(best_centers)
    return best_centers, final_rdius, best_sum
