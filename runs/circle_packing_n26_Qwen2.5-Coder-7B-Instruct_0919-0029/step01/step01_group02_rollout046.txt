import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

centers = np.array([
    [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
    [0.13, 0.3], [0.33, 0.3], [0.53, 0.3], [0.73, 0.3], [0.93, 0.3],
    [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
    [0.13, 0.7], [0.33, 0.7], [0.53, 0.7], [0.73, 0.7], [0.93, 0.7],
    [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
    [0.5, 1.0]
])

def run_packing():
    n = centers.shape[0]
    c = np.ones(n)
    A_eq = np.eye(n * 2)
    b_eq = np.hstack([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]])
    A_ineq = squareform(pdist(centers, 'euclidean')) - c * c
    b_ineq = np.zeros(n * (n - 1) // 2)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=(0, None))
    radii = result.x
    radii -= 1e-9
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
