def hexagonal_packing(n, side_length):
    k_hex = 2 / np.sqrt(3)
    r = side_length / 2
    h_row = k_hex * r
    w_half_row = r * 1.5
    rows = int(np.ceil(n / (3 / 4)))
    centers = []
    for i in range(rows):
        y = (i + 0.5) * h_row
        if i % 2 == 0:
            x_offset = 0
        else:
            x_offset = w_half_row
        for j in range(int(np.ceil(n / rows))):
            x = x_offset + j * w_half_row
            if len(centers) >= n:
                break
            centers.append([x, y])
    return np.array(centers), np.full(n, r)

def run_packing():
    n = 26
    side_length = np.sqrt(n / (2 * np.sqrt(3) / 3))
    centers, radii = hexagonal_packing(n, side_length)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def lp_optimum(centers, radii):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Center constraints
    for i in range(n):
        A_ub[idx, i] = 2 * radii[i]
        b_ub[idx] = 2
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

    n = 26

    return centers, radii, sum_radii
