import numpy as np
from scipy.spatial import KDTree
from scipy.optimize import linprog, minimize
from scipy.stats import norm

def compute_radii_radii(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    radii = np.ones(n) * (0.5 * (dists - 1e-12) / np.sqrt(3)).mean(axis=1)
    return radii

def compute_sum_of_radii(centers, fixed_radii, centres_fixed_indices):
    error = np.zeros(10)
    for i, r in enumerate(fixed_radii):
        if r == 0:
            error[i] = centers[centres_fixed_indices[i], 0] - 0.5 + 1e-5

    if not np.any(error):
        return 0, None
    
    centers_new = centers.copy()
    centers_new[centres_fixed_indices, 0] += np.clip(error, 0, None)

    radii_new = compute_radii_radii(centers_new)
    sum_of_radii = radii_new.sum()

    return sum_of_radii, centers_new

def check_validity(centers, radii):
    if np.any(radii < 0):
        return False, "Negative radii"
    
    if not (0 <= centers[:, 0] <= 1) or not (0 <= centers[:, 1] <= 1):
        return False, "Centres out of bounds"
    
    if len(radii[radii > 1e-5]) > 26:
        return False, "Too many large radii"
    
    if (radii[:, np.newaxis] + radii[np.newaxis, :] - centers[:, np.newaxis, 0][:, np.newaxis, :] - centers[np.newaxis, :, 0][:, :, np.newaxis] - centers[:, np.newaxis, 1][:, np.newaxis, :] - centers[np.newaxis, :, 1][:, :, np.newaxis] < 1e-5).any():
        return False, "Overlap detected"

    return True, "Valid"

def grid_search():
    numstarts = 20
    num_iters = 2000
    step = 0.05
    step_min = 0.001
    decay_factor = 0.995
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centers = np.random.rand(26, 2)
        for it in range(num_iters):
            idx = np.random.randint(26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            sum_of_radii, new_centers = compute_sum_of_radii(new_centers, best_radii, best_centers)
            
            if sum_of_radii > best_sum:
                best_sum = sum_of_radii
                best_centers = new_centers
                best_radii = compute_radii_radii(best_centers)
                
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centers, best_radii, best_sum = grid_search()

    return best_centers, best_radii, best_sum
