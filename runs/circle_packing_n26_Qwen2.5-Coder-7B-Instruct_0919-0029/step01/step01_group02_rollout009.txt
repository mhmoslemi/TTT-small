import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalty(X):
    n = len(X) // 3
    penalty = 0
    for i in range(n):
        x, y, r = X[3*i], X[3*i+1], X[3*i+2]
        penalty += max(0, r - x)**2
        penalty += max(0, r - (1 - x))**2
        penalty += max(0, r - y)**2
        penalty += max(0, r - (1 - y))**2
        for j in range(i + 1, n):
            dx = X[3*j] - X[3*i]
            dy = X[3*j+1] - X[3*i+1]
            d2 = dx**2 + dy**2
            penalty += max(0, (r + X[3*j+2])**2 - d2)**2
    return penalty

def f_global(X):
    n = len(X) // 3
    return -(sum(X[3*i+2] for i in range(n))) + 1e5 * penalty(X)

def pack_circles_global():
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26
    result = differential_evolution(f_global, bounds, maxiter=200, popsize=20, seed=42, workers=1)
    return result.x

def g_left(X, i):
    return X[3*i] - X[3*i+2]

def g_right(X, i):
    return 1 - X[3*i] - X[3*i+2]

def g_bot(X, i):
    return X[3*i+1] - X[3*i+2]

def g_top(X, i):
    return 1 - X[3*i+1] - X[3*i+2]

def g_pair(X, i, j):
    dx = X[3*j] - X[3*i]
    dy = X[3*j+1] - X[3*i+1]
    d2 = dx**2 + dy**2
    return ((X[3*i+2] + X[3*j+2])**2) - d2

def pack_circles_local(X_global):
    n = len(X_global) // 3
    constraints = []
    for i in range(n):
        constraints.append({'type':'ineq', 'fun': g_left ,  'args': (i,)} )
        constraints.append({'type':'ineq', 'fun': g_right , 'args': (i,)} )
        constraints.append({'type':'ineq', 'fun': g_bot   , 'args': (i,)} )
        constraints.append({'type':'ineq', 'fun': g_top   , 'args': (i,)} )
    for i in range(n-1):
        for j in range(i+1, n):
            constraints.append({'type':'ineq', 'fun': g_pair , 'args': (i, j)} )
    
    result = minimize(lambda X: -(sum(X[3*i+2] for i in range(n))), X_global, bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0})
    return result.x

def pack_circles():
    X_global = pack_circles_global()
    X_refined = pack_circles_local(X_global)
    centers = X_refined.reshape((26, 3))[:, :2]
    radii = X_refined.reshape((26, 3))[:, 2]
    radii = np.clip(radii, 0.0, None)
    radii -= np.clip(radii - 1e-9, 0.0, None)
    sum_radii = radii.sum()
    assert validate_packing(centers, radii)[0]
    return centers, radii, sum_radii

def run_packing():
    return pack_circles()

from scipy.optimize import validate_packing

# Example usage:
centers, radii, sum_radii = run_packing()
print("Sum of Radii:", sum_radii)
print("Validation Result:", validate_packing(centers, radii))
