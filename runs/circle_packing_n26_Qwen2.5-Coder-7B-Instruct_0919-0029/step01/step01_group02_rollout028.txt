import numpy as np
from scipy.optimize import minimize, differential_evolution, NonlinearConstraint

def penalty(X):
    N = len(X) // 3
    penalty_sum = 0.0
    for i in range(N):
        r, x, y = X[3*i:3*i+3]
        penalty_sum += np.maximum(0, r - x)**2
        penalty_sum += np.maximum(0, r - (1 - x))**2
        penalty_sum += np.maximum(0, r - y)**2
        penalty_sum += np.maximum(0, r - (1 - y))**2
        for j in range(i + 1, N):
            dx, dy = X[3*i:3*i+2] - X[3*j:3*j+2]
            d2 = dx**2 + dy**2
            r1, r2 = X[3*i+2], X[3*j+2]
            penalty_sum += np.maximum(0, (r1 + r2)**2 - d2)**2
    return penalty_sum

def objective(X):
    N = len(X) // 3
    return -X[2::3].sum() + 1e5 * penalty(X)

def g_left(X, i):
    r, x, y = X[3*i:3*i+3]
    return x - r

def g_right(X, i):
    r, x, y = X[3*i:3*i+3]
    return 1 - x - r

def g_bot(X, i):
    r, x, y = X[3*i:3*i+3]
    return y - r

def g_top(X, i):
    r, x, y = X[3*i:3*i+3]
    return 1 - y - r

def g_pair(X, i, j):
    dx, dy = X[3*i:3*i+2] - X[3*j:3*j+2]
    d2 = dx**2 + dy**2
    r1, r2 = X[3*i+2], X[3*j+2]
    return (r1 + r2)**2 - d2

def run_packing():
    N = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    # Initial guess: equally spaced small circles
    X_initial = np.zeros(3*N)
    for i in range(N):
        x = i * 1.0 / N + 1.0 / (2 * N)
        y = i * 1.0 / N + 1.0 / (2 * N)
        r = 1.0 / (2 * np.sqrt(2 * N))
        X_initial[3*i:3*i+3] = [x, y, r]

    result = differential_evolution(objective, bounds, popsize=20, maxiter=200, seed=42)
    
    if not result.success:
        print(f"DE failed: {result.message}")
        return np.zeros((N, 2)), np.zeros(N), 0.0

    X_global = result.x
    N = len(X_global) // 3
    
    constraints = []
    for i in range(N):
        constraints.extend([NonlinearConstraint(g_left, 0, np.inf, args=(i,)), 
                            NonlinearConstraint(g_right, 0, np.inf, args=(i,)), 
                            NonlinearConstraint(g_bot, 0, np.inf, args=(i,)), 
                            NonlinearConstraint(g_top, 0, np.inf, args=(i,))])
    for i in range(N-1):
        for j in range(i+1, N):
            constraints.append(NonlinearConstraint(g_pair, 0, np.inf, args=(i, j)))
    
    result_refined = minimize(objective, X_global, bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0})
    
    if not result_refined.success:
        print(f"SLSQP failed: {result_refined.message}")
        return np.zeros((N, 2)), np.zeros(N), 0.0
    
    X_refined = result_refined.x
    centers = X_refined.reshape((N, 3))[:, :2]
    radii = X_refined.reshape((N, 3))[:, 2] - 1e-9
