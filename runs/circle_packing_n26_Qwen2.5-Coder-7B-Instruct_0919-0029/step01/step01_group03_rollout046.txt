            # Distance to boundaries
            dist_left = x
            dist_right = 1 - x
            dist_bottom = y
            dist_top = 1 - y

            min_dist = np.inf
            for j in range(i):
                dist = np.linalg.norm(centers[i] - centers[j])
                min_dist = min(min_dist, dist - radii[j])

            r_max = min(dist_left, dist_right, dist_bottom, dist_top)
            r_max = min(r_max, min_dist)
