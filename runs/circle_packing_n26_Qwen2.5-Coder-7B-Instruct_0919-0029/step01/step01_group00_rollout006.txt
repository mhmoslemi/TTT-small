import numpy as np
from scipy.optimize import linprog, minimize, NelderMead
from scipy.spatial import distance_matrix

# Original lp_optimum function, slightly modified for robustness
def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            if dist < 0:
                return np.zeros(n), -np.inf
            b_ub[idx] = dist
            idx += 1

    for i in range(n):
        x, y = centers[i]
        r = np.min([x, y, 1 - x, 1 - y]) - 1e-12
        if r < 0:
            return np.zeros(n), -np.inf
        A_ub[idx, i] = 1
        b_ub[idx] = r
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        return np.zeros(n), -np.inf
    return result.x, -result.fun
