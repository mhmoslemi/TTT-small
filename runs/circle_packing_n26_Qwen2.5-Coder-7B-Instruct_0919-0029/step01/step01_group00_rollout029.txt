import numpy as np
from scipy.spatial import Delaunay, ConvexHull
from scipy.optimize import minimize, linprog
from itertools import combinations

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            a = centers[i] - centers[j]
            b = np.sqrt(np.sum(a**2)) - 1e-12
            A_ub[idx, i] -= a[0] / b
            A_ub[idx, j] -= a[1] / b
            b_ub[idx] = np.sum(a**2) - 2*a[0]*(centers[i,0]/b + centers[j,0]/b) - 2*a[1]*(centers[i,1]/b + centers[j,1]/b)
            idx += 1

    for i in range(n):
        for coord in range(2):
            A_ub[idx, i] = 1 if coord == 0 else -1
            b_ub[idx] = centers[i, coord] - 1e-12
            idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun
