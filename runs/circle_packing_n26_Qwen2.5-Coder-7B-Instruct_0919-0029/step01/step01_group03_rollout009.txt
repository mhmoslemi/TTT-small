def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    import numpy as np
    from pulp import *

    # Parameters
    Δ = 0.075
    nx, ny = 14, 14
    P = nx * ny

    # Grid points
    X = np.arange(0, 1, Δ)
    Y = np.arange(0, 1, Δ)

    # Distance matrix
    D = np.sqrt((X[:, None] - X[None, :]) ** 2 + (Y[:, None] - Y[None, :]) ** 2)

    # MILP formulation
    prob = LpProblem("CirclePackingMILP", LpMaximize)

    # Variables
    y = LpVariable.dicts("y", range(P), cat='Binary')
    r = LpVariable.dicts("r", range(P), lowBound=0, upBound=0.5)

    # Objective function
    prob += lpSum(r[k] for k in range(P))

    # Constraints
    prob += lpSum(y[k] for k in range(P)) == 26

    for k in range(P):
        prob += r[k] <= X[k]
        prob += r[k] <= 1 - X[k]
        prob += r[k] <= Y[k]
        prob += r[k] <= 1 - Y[k]
        prob += r[k] <= 0.5 * y[k]

    for k in range(P):
        for l in range(k + 1, P):
            d_kl = D[k, l]
            prob += r[k] + r[l] <= d_kl + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])

    # Solver
    prob.solve()

    # Extract solutions
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    count = 0
    for k in range(P):
        if y[k].varValue > 0.5:
            centers[count] = np.array([X[k], Y[k]])
            radii[count] = r[k].varValue
            count += 1

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
