import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26

    def lp_radii(centers):
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        A_ub = []
        b_ub = []
        for i in range(n):
            A_ub.append([1 if k == i else 0 for k in range(n)])
            b_ub.append(dist[i, i])
        for i in range(n):
            for j in range(i + 1, n):
                A_ub.append([0] * i + [1] + [0] * (n - i - 1) + [0] * j + [1] + [0] * (n - j - 1))
                b_ub.append(dist[i, j])
        for i in range(n):
            x, y = centers[i]
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(x)
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(1 - x)
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(y)
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(1 - y)
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        return res.x, -res.fun

    def compute_initial_layout():
        centers = np.zeros((n, 2))
        radius = 0.125
        centers[0] = [radius, radius]
        centers[1] = [1 - radius, radius]
        centers[2] = [radius, 1 - radius]
        centers[3] = [1 - radius, 1 - radius]
        for i in range(9):
            centers[4 + i] = [0.05 + i * 3/10, 0.8 - (3 + i % 2) * 3/20]
        for i in range(5):
            centers[14 + i] = [0.2 + i * 2/5, 0.3 + i * 3/10]
        return centers

    best_centers = None
    best_sum = -np.inf

    for restart in range(100):
        current_centers = compute_initial_layout()
        for _ in range(10):
            jitter = 0.03 * np.random.randn(n, 2)
            current_centers += jitter
            current_centers = np.clip(current_centers, 0, 1)
            radii, current_sum = lp_radii(current_centers)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = current_centers.copy()

    return best_centers, lp_radii(best_centers)[0], best_sum
