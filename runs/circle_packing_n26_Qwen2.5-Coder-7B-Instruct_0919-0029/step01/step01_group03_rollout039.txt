import numpy as np
from pulp import *

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Parameters
    n = 26
    r_max = 0.15
    alpha = 0.02
    T0 = 0.5
    N_steps = 30000
    
    # Build the initial perturbed lattice
    def create_perturbed_lattice(r, alpha):
        delta_x = r * 2
        delta_y = np.sqrt(3) * r
        points = [(r + k_x * delta_x, r + k_y * delta_y + (k_x % 2) * delta_y / 2)
                 for k_x in range(-int(np.ceil(1 / delta_x)) + 1, int(2 * np.ceil(1 / delta_x)))
                 for k_y in range(-int(1 / delta_y) + 1, int(2 * np.ceil(1 / delta_y)))]
        selected_points = points[:n]
        for i, (x, y) in enumerate(selected_points):
            delta = np.random.uniform(-alpha, alpha, 2)
            x_new, y_new = x + delta[0], y + delta[1]
            x_new = max(0, min(1, x_new))
            y_new = max(0, min(1, y_new))
            selected_points[i] = (x_new, y_new)
        return selected_points
    
    C = create_perturbed_lattice(r_max, alpha)
    
    # Get the initial order (random permutation)
    pi = np.random.permutation(n)
    
    # Simulated Annealing (SA) optimization to find the best order
    best_pi = pi.copy()
    best_sum = -np.inf
    current_sum = -np.inf
    T = T0
    for _ in range(N_steps):
        proposed_pi = best_pi.copy()
        proposed_i, proposed_j = np.random.permutation(n)[:2]
        proposed_pi[proposed_i], proposed_pi[proposed_j] = proposed_pi[proposed_j], proposed_pi[proposed_i]
        
        # Compute the sum of radii for the new order
        current_sum, _ = compute_sum(pi, C)
        _, proposed_sum = compute_sum(proposed_pi, C)
        
        if proposed_sum > current_sum or np.random.rand() < np.exp((proposed_sum - current_sum) / T):
            current_sum = proposed_sum
            pi = proposed_pi.copy()
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_pi = pi.copy()
        
        T *= 0.995
    
    # Refine the radii for the best order with Linear Programming (LP)
    x = C.T
    y = np.ones((n, 1))
    A, b = build_constraint_matrix(C, best_pi)
    c = -np.ones(n)  # Minimize by multiplying with -1
    res = linprog(c, A_ub=A, b_ub=b, bounds=(1e-6, None), method='highs')
    
    centers = np.vstack((x, y)).T
    radii = -res.x
    
    return centers, radii, np.sum(radii)

def compute_sum(pi, C):
    n = len(pi)
    radii = np.zeros(n)
    for i in pi:
        x, y = C[i]
        for j in pi[:i]:
            xj, yj = C[j]
            dist = np.sqrt((x - xj)**2 + (y - yj)**2)
            if dist > radii[j] + radii[i]:
                radii[i] = dist - radii[j]
            if radii[i] > r_max:
                radii[i] = r_max
    return np.sum(radii), radii

def build_constraint_matrix(C, pi):
    n = len(pi)
    A = []
    b = []
    for i in pi:
        row = np.zeros(n)
        row[pi.tolist().index(i)] = 1
        A.append(row)
        b.append(1)
    for i in pi:
        x, y = C[i]
        row = [0] * n
        row[pi.tolist().index(i)] = -1
        A.append(row)
        b.append(-x)
        row = [0] * n
        row[pi.tolist().index(i)] = -1
        A.append(row)
        b.append(-y)
        row = [0] * n
        row[pi.tolist().index(i)] = -1
        A.append(row)
        b.append(-(1 - x))
        row = [0] * n
        row[pi.tolist().index(i)] = -1
        A.append(row)
        b.append(-(1 - y))
    
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = C[pi[i]]
            x2, y2 = C[pi[j]]
            row = [0] * n
            row[pi.tolist().index(i)] = -1
            row[pi.tolist().index(j)] = -1
            A.append(row)
            b.append(-(x2 - x1) ** 2 - (y2 - y1) ** 2 + 1e-12)
    
    return np.array(A), np.array(b)
