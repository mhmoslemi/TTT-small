import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimize(centers, radii):
    n = centers.shape[0]
    A = np.zeros((2 * n, n))
    b = np.zeros(2 * n)
    for i in range(n):
        A[i, i] = 2
        b[i] = 1

    A_eq = np.zeros((n, n))
    for i in range(n):
        A_eq[i, i] = 1

    c = -radii + centers.flatten()
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=np.zeros(n), bounds=bounds, method='highs')
    return result.x, -result.fun

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = radii
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    centres = np.random.rand(n, 2)
    radii, sum_radii = lp_optimize(centres, np.ones(n) * 1e-3)
    return centres, radii, sum_radii
