import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A = np.zeros((n * (n - 1) // 2 + 2 * n, n))
    b = np.zeros(A.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    for i in range(n):
        distllx = np.clip(centers[i, 0] - 0.5, 0, None)
        disturx = np.clip(0.5 - centers[i, 0], 0, None)
        distlly = np.clip(centers[i, 1] - 0.5, 0, None)
        distrly = np.clip(0.5 - centers[i, 1], 0, None)
        A[idx, i] = 1
        b[idx] = distllx
        idx += 1
        A[idx, i] = 1
        b[idx] = disturx
        idx += 1
        A[idx, i] = 1
        b[idx] = distlly
        idx += 1
        A[idx, i] = 1
        b[idx] = distrly
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_eq=A, b_eq=b, A_ub=None, b_ub=None, bounds=bounds, method='highs')
    return result.x, -result.fun

def evaluate(centroid):
    centres = centroid.reshape(26, 2)
    radii, total = lp_optimum(centres)
    return -total

def local_search(centre_all):
    bounds = [(0, 1)] * 52
    result = minimize(evaluate, centre_all, bounds=bounds, method='Nelder-Mead', options={'maxiter': 20})
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    r_N = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_N
    h = np.sqrt(3) * r_N

    points = []
    for k in range(6):
        offset = 0 if k % 2 == 0 else r_N / 2
        x_center = r_N + offset + (s * np.arange(5))
        y_center = k * h + r_N
        for x in x_center:
            if 0 < x < 1:
                points.append([x, y_center])

    initial_population = points[:26] * 50 + np.random.rand(26 * 50, 2) * np.array([1 - s, 1 - s])

    res = differential_evolution(evaluate, bounds=[(0, 1)] * 52, popsize=50, strategy='best1bin', mutation=0.5, 
                                recombination=0.7, maxiter=80, polish=False, init=initial_population)

    centre_all = local_search(res.x)
    best_centres = centre_all.reshape(26, 2)
    radii, total_sum = lp_optimum(best_centres)

    final_optimization = minimize(lambda x: -evaluate(x), centre_all, method="Nelder-Mead", options={'maxiter': 20})
    centres = final_optimization.x.reshape(26, 2)
    _, final_sum = lp_optimum(centres)

    return centres, radii, final_sum
