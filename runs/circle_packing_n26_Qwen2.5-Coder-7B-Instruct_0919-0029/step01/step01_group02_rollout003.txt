import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26
bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
penalty_lambda = 1e6
maxiter_de = 200
maxiter_slsqp = 200
epsilon = 1e-9

def objective_global(X):
    radii = X[2::3]
    penalty = np.sum(np.maximum(0, X[2::3] - X[0::3])) + np.sum(np.maximum(0, X[2::3] - (1 - X[0::3]))) + np.sum(np.maximum(0, X[2::3] - X[1::3])) + np.sum(np.maximum(0, X[2::3] - (1 - X[1::3])))
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((X[3*i:3*i+2] - X[3*j:3*j+2]) ** 2)
            penalty += np.maximum(0, (X[2*i] + X[2*j])**2 - d2)**2
    return -(np.sum(radii)) + penalty_lambda * penalty

def constraint_radius_centered(x, i):
    return x - x[3*i+2]

def constraint_radius(x, i):
    return 1 - x - x[3*i+2]

def constraint_unscaled_radius(x, i):
    return x[3*i+2] - epsilon

constraints_global = [{'type': 'ineq', 'fun': constraint_radius_centered, 'args': (i,)} for i in range(N)] + \
                     [{'type': 'ineq', 'fun': constraint_radius, 'args': (i,)} for i in range(N)] + \
                     [{'type': 'ineq', 'fun': constraint_unscaled_radius, 'args': (i,)} for i in range(N)]

def objective_local(X):
    return -(np.sum(X[2::3]))

def constraint_boundary_left(x, i):
    return x[3*i]

def constraint_boundary_right(x, i):
    return 1 - x[3*i]

def constraint_boundary_bottom(x, i):
    return x[3*i+1]

def constraint_boundary_top(x, i):
    return 1 - x[3*i+1]

def constraint_non_overlap(x, i, j):
    d2 = np.sum((x[3*i:3*i+2] - x[3*j:3*j+2]) ** 2)
    return (x[3*i+2] + x[3*j+2])**2 - d2

constraints_slsqp = []
for i in range(N):
    constraints_slsqp.append({'type': 'ineq', 'fun': constraint_boundary_left, 'args': (i,)} )
    constraints_slsqp.append({'type': 'ineq', 'fun': constraint_boundary_right, 'args': (i,)} )
    constraints_slsqp.append({'type': 'ineq', 'fun': constraint_boundary_bottom, 'args': (i,)} )
    constraints_slsqp.append({'type': 'ineq', 'fun': constraint_boundary_top, 'args': (i,)} )
for i in range(N-1):
    for j in range(i+1, N):
        constraints_slsqp.append({'type': 'ineq', 'fun': constraint_non_overlap, 'args': (i, j)} )

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    result_global = differential_evolution(objective_global, bounds=bounds, constraints=constraints_global, cops=1, popsize=20, maxiter=maxiter_de, tol=1e-8, seed=42, workers=1)
    X_global = result_global.x
    result_local = minimize(objective_local, X_global, method='SLSQP', bounds=bounds, constraints=constraints_slsqp, options={'ftol': 1e-8, 'maxiter': maxiter_slsqp, 'iprint': 0})
    X_refined = result_local.x
    centers = X_refined.reshape((N, 3))[:, :2]
    radii = X_refined.reshape((N, 3))[:, 2]
    radii = np.clip(radii, 0.0, None)
    radii = np.clip(radii - epsilon, 0.0, None)
    sum_radii = radii.sum()
    assert validate_packing(centers, radii)[0], "Validation failed"
    return centers, radii, sum_radii
