import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    distances = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(distances, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(distances[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing():
    n = 26

    # Start with a near-hexagonal packing
    centers = np.zeros((n, 2))
    radius = 0.14
    x0 = np.linspace(0.05, 0.85, 5)
    x1 = np.linspace(0.15, 0.75, 4)
    x2 = np.linspace(0.25, 0.95, 4)
    x1, x2 = np.meshgrid(x1, x2)
    x1.ravel().sort()
    x2.ravel().sort()
    x = np.concatenate([x0, x1.ravel(), x2.ravel()])
    x_random = np.random.permutation(n)
    centers[:, 0] = x[x_random][:n]
    centers[:, 1] = np.zeros(n)
    for i in range(1, int(np.sqrt(n)) + 1):
        centers[int(np.sqrt(n)) * i + i // 2 - i % 2, 1] = radius * i

    # Annealing parameters
    T0 = 0.1
    Tf = 1e-4
    alpha = 0.9995
    Delta = 0.1
    steps_per_T = 500
    max_iter = steps_per_T * int(np.log(T0 / Tf) / np.log(alpha))

    best_centers = centers.copy()
    best_sum = np.inf

    # Annealing loop
    for T in np.linspace(T0, Tf, max_iter):
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            delta = np.random.randn(2) * Delta
            delta[delta < 0] = delta[delta < 0] * max(Delta / 10, 1 - T / 10e5)
            delta[delta > 0] = delta[delta > 0] * max(Delta / 10, 1 - T / 10e5)
            new_centers = centers.copy()
            new_centers[i] += delta
            new_centers[i] = np.clip(new_centers[i], 0, 1)

            try:
                radii, sum_radii = lp_optimum(new_centers)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = new_centers
                    Delta *= 1.1  # Increase step size
                elif sum_radii < best_sum and np.random.rand() < np.exp((sum_radii - best_sum) / T):
                    best_sum = sum_radii
                    centers = new_centers
                    Delta *= 1.1  # Increase step size
            except Exception as e:
                pass  # accept failure to escape; assume only permissible failures occur

    # Final polishing with L-BFGS-B
    result = minimize(lambda r: -np.sum(r), 
                      np.ones(n), 
                      method='L-BFGS-B', 
                      jac=lambda r: -np.ones(n),
                      constraints={'type': 'ineq', 'fun': lambda r: r},
                      options={'maxiter': 1000})
    best_radii = result.x
    best_sum = -result.fun

    return best_centers, best_radii, best_sum
