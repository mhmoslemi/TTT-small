import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26

    # Step 1: Generate perturbed lattice centres
    r = 0.15
    alpha = 0.02
    delta_x = 2 * r
    delta_y = np.sqrt(3) * r

    # Generate 30 lattice points
    lattice_x = np.arange(2 * r, 1 + 1e-12, delta_x)
    lattice_y = np.arange(r, 1 + 1e-12, delta_y / 2)
    lattice = np.array([[x, y] for x in lattice_x for y in lattice_y if x <= 1 and y <= 1])

    # Sample 26 points from the lattice
    np.random.shuffle(lattice)
    centers = lattice[:n]

    # Step 2: Simulated annealing to find the best order
    def get_radius_order(order, centers):
        radii = np.zeros(n)
        for k, i in enumerate(order):
            r_max = min(centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1])
            for j in order[:k]:
                r_max = min(r_max, np.linalg.norm(centers[i] - centers[j]) - radii[j])
            radii[i] = max(r_max, 1e-6)
        return radii, np.sum(radii)

    def annealing(initial_order):
        order = np.copy(initial_order)
        T = 0.5
        N_steps = 30_000
        best_order = np.copy(order)
        best_sum = -np.inf

        for step in range(N_steps):
            i, j = np.random.choice(n, 2, replace=False)
            order[i], order[j] = order[j], order[i]
            radii, sum_radii = get_radius_order(order, centers)

            if sum_radii >= best_sum or np.random.rand() < np.exp((sum_radii - best_sum) / T):
                best_order, best_sum = np.copy(order), sum_radii
            else:
                order[i], order[j] = order[j], order[i]

            T *= 0.995

        return best_order, get_radius_order(best_order, centers)

    initial_order = np.arange(n)
    best_order, (radii, best_sum) = annealing(initial_order)

    # Step 3: Final refinement with linear programming
    A = np.zeros((2 * n, n))
    b = np.zeros(2 * n)

    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0]
        A[n + i, i] = -1
        b[n + i] = 1 - centers[i, 0]

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A[2 * n + i, j] = 1
            A[2 * n + i, i] = 1
            b[2 * n + i] = dist
            A[2 * n + j, i] = -1
            b[2 * n + j] = dist

    res = linprog(-radii, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    radii_optimized = -res.x

    return centers, radii_optimized, np.sum(radii_optimized)
