import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Build a 0.01-spaced grid of 10,201 points.
    grid_points = [(i*0.01, j*0.01) for i in range(101) for j in range(101)]
    grid_points = np.array(grid_points)

    # Step 2: Initialise all radii to 0, mark all points inactive.
    radii = np.zeros(10201)
    active = np.zeros(10201, dtype=bool)

    # Step 3: While < 26 circles active:
    while np.sum(active) < 26:
        # a. Compute growth time τ to the next contact for every inactive point.
        growth_times = np.full(10201, np.inf)
        for i in range(10201):
            if active[i]:
                continue
            x, y = grid_points[i]
            rx, ry = radii[i], radii[i]
            boundary_distances = np.array([x - rx, 1 - x - rx, y - ry, 1 - y - ry])
            circle_distances = np.min(np.sqrt(np.sum((grid_points - grid_points[i])**2, axis=1)) - radii - radii[i], axis=0)
            growth_times[i] = np.minimum(boundary_distances, circle_distances).min()
        
        # b. Identify point(s) that achieve the minimal τ.
        min_growth_index = np.argmin(growth_times)
        
        # c. Activate them, increase all radii by τ, mark tempered.
        active[min_growth_index] = True
        growth_time = growth_times[min_growth_index]
        radii += growth_time

    # Step 4: Extract the 26 active centres and their radii.
    active_indices = np.where(active)[0]
    centers = grid_points[active_indices]

    # Step 5: Run a 200-step gradient-descent on the centres, recomputing radii each time.
    alpha = 0.05
    for _ in range(200):
        pressures = np.zeros((10201, 2))
        for i in range(10201):
            if not active[i]:
                continue
            x, y = centers[i]
            rx, ry = radii[i], radii[i]
            boundary_forces = np.array([[-1, 0], [1, 0], [0, -1], [0, 1]])
            boundary_directions = np.argmin(np.array([x - rx, 1 - x - rx, y - ry, 1 - y - ry]))
            boundary_force = boundary_forces[boundary_directions]
            neighbors = grid_points[(np.sum((grid_points - centers[i])**2, axis=1) < (rx + radii)**2) & ~active]
            if len(neighbors) > 0:
                neighbor_distances = np.sqrt(np.sum((neighbors - centers[i])**2, axis=1)) - (radii + radii)
                neighbor_forces = np.sum((neighbors - centers[i]) / np.array([1, 1]) / neighbor_distances * neighbor_distances, axis=0) / 2
                pressures[i] = -boundary_force - neighbor_forces
            centers[i] += pressure * alpha
            centers[i] = np.clip(centers[i], 0, 1)

    # Step 6: Subtract a tiny epsilon from all positive radii.
    epsilon = 1e-9
    radii -= epsilon * (radii > 0)

    # Step 7: Return (centres, radii, sum_radii).
    sum_radii = np.sum(radii)
    return centers[:26], radii[:26], sum_radii
