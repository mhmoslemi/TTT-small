import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

def obj_func(x):
    radii = x[2::3]
    penalty = 0.0
    penalty += np.sum(np.maximum(0.0, radii - x[0::3]))
    penalty += np.sum(np.maximum(0.0, radii - (1 - x[0::3])))
    penalty += np.sum(np.maximum(0.0, radii - x[1::3]))
    penalty += np.sum(np.maximum(0.0, radii - (1 - x[1::3])))
    for i in range(N-1):
        for j in range(i+1, N):
            d2 = (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2
            v = (radii[i] + radii[j])**2 - d2
            if v > 0:
                penalty += v**2
    return -np.sum(radii) + 1e8 * penalty

def boundary_left(x, idx):
    return x[idx*3] - x[idx*3+2]

def boundary_right(x, idx):
    return 1 - x[idx*3] - x[idx*3+2]

def boundary_bottom(x, idx):
    return x[idx*3+1] - x[idx*3+2]

def boundary_top(x, idx):
    return 1 - x[idx*3+1] - x[idx*3+2]

def pair_non_overlap(x, i, j):
    return (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2 - (x[i*3+2] + x[j*3+2])**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    x0 = np.concatenate([np.random.rand(52), 0.1 * np.ones(26)])
    
    result = differential_evolution(obj_func, bounds, maxiter=200, popsize=20, disp=False, workers=1)
    best_x = result.x
    
    res = minimize(obj_func, best_x, method='SLSQP', bounds=bounds, constraints=[], options={'ftol':1e-6, 'maxiter':200})
    centers = best_x[:52].reshape((N, 2))
    radii = best_x[52:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum Radii:", sum_radii)
