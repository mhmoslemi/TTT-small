```python
import numpy as np
from scipy.optimize import differential_evolution, minimize

def g_left(X, i):   return X[3*i] - X[3*i+2] 
def g_right(X, i):  return 1 - X[3*i] - X[3*i+2] 
def g_bot(X, i):    return X[3*i+1] - X[3*i+2] 
def g_top(X, i):    return 1 - X[3*i+1] - X[3*i+2] 
def g_pair(X, i, j): return ((X[3*i] - X[3*j])**2 + (X[3*i+1] - X[3*j+1])**2) - (X[3*i+2] + X[3*j+2])**2 

def penalty(X):
   /N = 26
    return sum(max(0, g_left (X, i))**2 
               + max(0, g_right(X, i))**2 
               + max(0, g_bot  (X, i))**2 
               + max(0, g_top  (X, i))**2 
               for i in range(N)) + \
           sum(max(0, g_pair (X, i, j))**2 for i in range(N-1) for j in range(i+1, N))

def f_global(X):
    return -(np.sum(X[2*N:3*N])) + 1e5 * penalty(X)

def f_local(X):
    return -(np.sum(X[2*N:3*N]))

N = 26
bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

# Global search
result_DE = differential_evolution(f_global, bounds, popsize=20, maxiter=200, seed=42, workers=1)

# Local refinement
X_refined = minimize(f_local, result_DE.x, bounds=bounds, method='SLSQP', args=(penalty,), 
                      constraints=[], options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0}).x

# Post-processing
centers = X_refined.reshape((N, 3))[:, :2]
radii = np.clip(X_refined.reshape((N, 3))[:, 2], 0.0, None)

# Epsilon correction
radii = np.clip(radii - 1e-9, 0.0, None)

# Sum of radii
sum_radii = radii.sum()

# Validation
import validate_packing # make sure to define the function in the environment
is_valid, message = validate_packing(centers, radii)

print("Validation:", is_valid, message)
print("Sum of radii:", sum_radii)

return centers, radii, sum_radii
```
