import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_point_step = 0.01
    grid_points = np.array([(x, y) for x in np.arange(0, 1 + grid_point_step, grid_point_step) for y in np.arange(0, 1 + grid_point_step, grid_point_step)])
    
    n = grid_points.shape[0]
    radii = np.zeros(n)
    active_indices = np.zeros(n, dtype=int)
    
    while np.sum(active_indices) < 26:
        # Compute growth times
        growth_times = []  # This will store the growth times for each grid point
        for i in range(n):
            if active_indices[i] == 0:
                x, y = grid_points[i]
                to_boundaries = np.array([x, 1 - x, y, 1 - y]) - radii
                to_neighbours = np.min([np.sqrt(np.sum((grid_points[i,:] - grid_points[j,:])**2, axis=1)) - radii[j] for j in range(n) if active_indices[j] == 0], initial=np.inf)
                t_min = np.min(np.concatenate([to_boundaries, to_neighbours]))
                growth_times.append((i, t_min))
            else:
                growth_times.append((i, np.inf))
        # Find the point with the minimum growth time
        next_point, t_min = min(growth_times, key=lambda x: x[1])
        # Activate the point and update radii
        if t_min < np.inf:
            active_indices[next_point] = 1
            radii[next_point] += t_min
            for j in range(n):
                if active_indices[j] == 1:
                    dist = np.sqrt(np.sum((grid_points[next_point,:] - grid_points[j,:])**2))
                    if dist < radii[next_point] + radii[j]:
                        radii[j] = (dist - radii[next_point]) / 2
    centres = grid_points[active_indices == 1]
    return centres, radii[active_indices == 1], np.sum(radii[active_indices == 1])
