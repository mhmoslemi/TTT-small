import numpy as np
from numba import njit, prange

eps = 1e-12

def dist(c1, c2):
    return np.sqrt(np.sum((c1 - c2) ** 2 + eps))

def construct_initial_circles():
    r = 0.5 * np.sqrt(2)
    C_L = np.array([0, r, r])
    C_R = np.array([1, 1-r, r])
    C_T = np.array([r, 1, r])
    return [C_L, C_R, C_T]

def apollonian_packing(initial_circles, max_count=26):
    circles = initial_circles
    queue = []
    for i in range(len(circles)):
        for j in range(i + 1, len(circles)):
            for k in range(j + 1, len(circles)):
                dx, dy = circles[i][1] - circles[j][1], circles[i][2] - circles[j][2]
                dy, dx = circles[j][1] - circles[k][1], circles[j][2] - circles[k][2]
                a, b, c = dx * dy - dy * dx, -1 * dx * dx - dy * dy, 1
                discriminant = a ** 2 - 4 * b * c
                if discriminant >= 0:
                    t = (-a + np.sqrt(discriminant)) / (2 * b)
                    if t > 0:
                        k_new = t * (1 / (4 * dx) + 1 / (4 * dy))
                        if k_new > 0:
                            cx = circles[i][1] / 2 + circles[j][1] / 2 + dx * t / 2
                            cy = circles[i][2] / 2 + circles[j][2] / 2 + dy * t / 2
                            r_new = 1 / (4 * k_new)
                            new_circle = np.array([cx, cy, r_new])

                                circles.append(new_circle)
                                queue.append(new_circle)
    while len(circles) > max_count:
        queue.pop(0)
        circles.pop()
    return circles
