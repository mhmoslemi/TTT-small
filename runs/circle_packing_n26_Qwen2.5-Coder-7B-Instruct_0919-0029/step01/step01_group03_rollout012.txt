import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def objective(z):
    M, N = 26, 2
    c = z[:M*N].reshape(M, N)
    r = z[M*N:].clip(0, 0.5)
    W = 2500.0
    
    dists = cdist(c, c)
    overlp = np.maximum(0, r[:, None] + r[None, :] - dists)
    bound_viol = np.maximum(0, r - c[:, 0]) + np.maximum(0, r - (1 - c[:, 0])) + \
                np.maximum(0, r - c[:, 1]) + np.maximum(0, r - (1 - c[:, 1]))
    
    return -np.sum(r - W * (np.sum(overlp**2) + np.sum(bound_viol**2)))

def gradient(z):
    M, N = 26, 2
    c = z[:M*N].reshape(M, N)
    r = z[M*N:].clip(0, 0.5)
    W = 2500.0
    
    dists = cdist(c, c)
    overlp = np.maximum(0, r[:, None] + r[None, :] - dists)
    bound_viol = np.maximum(0, r - c[:, 0]) + np.maximum(0, r - (1 - c[:, 0])) + \
                np.maximum(0, r - c[:, 1]) + np.maximum(0, r - (1 - c[:, 1]))
    
    overlp penalty = W * 2 * overlp
    bound_viol penalty = W * 2 * bound_viol
    
    grad_c = np.zeros_like(z)
    grad_r = -1 + np.sum(overlp penalty[:, None, :] - overlp penalty[None, :, :], axis=2) + \
              bound_viol penalty[:, :] - bound_viol penalty[:, :]
    
    grad_c[:M*N] = grad_r * 2 * (c[:, np.newaxis, 0] - c[np.newaxis, :, 0])
    grad_c[:M*N] += grad_r * 2 * (c[:, np.newaxis, 1] - c[np.newaxis, :, 1])
    grad_c[M*N:] = -1 + np.sum(overlp penalty[:, None, :] - overlp penalty[None, :, :], axis=1)
    
    return grad_c.ravel() + grad_r.ravel()

bounds = [(0, 1)] * (26 * 2) + [(0, 0.5)] * 26
initial_guess = np.random.rand(26 * 2 + 26) * (0.75 - 0.5) + 0.25

result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, jac=gradient, options={'maxfun': 4000})
centers = result.x[:26*2].reshape(26, 2)
radii = result.x[26*2:].clip(0, 0.5)
sum_radii = np.sum(radii)

centers, radii, sum_radii
