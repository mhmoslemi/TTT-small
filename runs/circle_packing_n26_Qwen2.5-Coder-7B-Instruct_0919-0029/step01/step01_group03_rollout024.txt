import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def cost_and_grad(z, W):
    N = 26
    centers = z[:N*N].reshape(N, 2)
    radii = z[N*N:].reshape(N, 1)
    dist = cdist(centers, centers, 'euclidean')
    penalty_bound = np.maximum(0, radii - centers[:, 0]) ** 2 + \
                    np.maximum(0, radii - (1 - centers[:, 0])) ** 2 + \
                    np.maximum(0, radii - centers[:, 1]) ** 2 + \
                    np.maximum(0, radii - (1 - centers[:, 1])) ** 2
    penalty_overlap = np.maximum(0, radii + radii[:, None] - dist) ** 2
    penalty = W * (np.sum(penalty_bound) + np.sum(penalty_overlap))
    obj = -np.sum(radii) - penalty
    return obj, gradient(obj, centers, radii, W, dist)

def gradient(obj, centers, radii, W, dist):
    N = centers.shape[0]
    grad = np.zeros_like(centers.flatten())
    for i in range(N):
        for j in range(N):
            if i != j:
                g = 2 * W * (radii[i] + radii[j] - dist[i, j] - 1) * (radii[i] + radii[j] > dist[i, j])
                if i == 0:
                    grad[i*N:i*N+2] -= g * np.array([1, 0])
                if i == 1:
                    grad[i*N:i*N+2] -= g * np.array([1, 0])
                if i == N-2:
                    grad[i*N:i*N+2] += g * np.array([-1, 0])
                if i == N-1:
                    grad[i*N:i*N+2] += g * np.array([-1, 0])
                if j == 0:
                    grad[j*N:j*N+2] -= g * np.array([1, 0])
                if j == 1:
                    grad[j*N:j*N+2] -= g * np.array([1, 0])
                if j == N-2:
                    grad[j*N:j*N+2] += g * np.array([-1, 0])
                if j == N-1:
                    grad[j*N:j*N+2] += g * np.array([-1, 0])

                grad[i*N+i*N+2*N] += g * 2 * W if i == j else g * 2 * W * (radii[i] + radii[j] > dist[i, j])
                grad[j*N+j*N+2*N] += g * 2 * W if i == j else g * 2 * W * (radii[i] + radii[j] > dist[i, j])

    return grad
