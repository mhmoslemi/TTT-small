import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(X, λ):
    sum_radii = np.sum(X[N::3])
    penalty = 0.0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((X[3*i:3*i+2] - X[3*j:3*j+2]) ** 2)
            penalty += np.maximum(0, (X[3*i+2] + X[3*j+2]) ** 2 - d2) ** 2
        penalty += np.maximum(0, X[3*i+2] - X[3*i]) ** 2  # x - r
        penalty += np.maximum(0, 1 - X[3*i] - X[3*i+2]) ** 2  # (1-x) - r
        penalty += np.maximum(0, X[3*i+2] - X[3*i+1]) ** 2  # y - r
        penalty += np.maximum(0, 1 - X[3*i+1] - X[3*i+2]) ** 2  # (1-y) - r
    return -(sum_radii - λ * penalty)

def penalty(X):
    penalty = 0.0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((X[3*i:3*i+2] - X[3*j:3*j+2]) ** 2)
            penalty += np.maximum(0, (X[3*i+2] + X[3*j+2]) ** 2 - d2) ** 2
        penalty += np.maximum(0, X[3*i+2] - X[3*i]) ** 2  # x - r
        penalty += np.maximum(0, 1 - X[3*i] - X[3*i+2]) ** 2  # (1-x) - r
        penalty += np.maximum(0, X[3*i+2] - X[3*i+1]) ** 2  # y - r
        penalty += np.maximum(0, 1 - X[3*i+1] - X[3*i+2]) ** 2  # (1-y) - r
    return penalty

def run_packing():
    N = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    λ = 1e5
    
    DE_result = differential_evolution(objective, bounds, args=(λ,), popsize=20, maxiter=200, seed=42, workers=1)
    DE_X = DE_result.x

    def objective_local(X):
        return -(np.sum(X[N::3]))
    
    SLSQP_result = minimize(objective_local, DE_X, constraints=[], bounds=bounds, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0});
    X_ref = SLSQP_result.x
    
    centers = X_ref.reshape((N, 3))[:, :2]
    radii = X_ref.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    radii = np.clip(radii, 0.0, None)
    for i in range(N):
        if radii[i] > 0:
            radii[i] -= 1e-9
    
    return centers, radii, sum_radii
