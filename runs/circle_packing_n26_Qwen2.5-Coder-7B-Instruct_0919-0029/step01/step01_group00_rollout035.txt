import numpy as np
from scipy.optimize import linprog, minimize
from itertools import combinations

# Function to build a triangular lattice
def build_lattice(r_est, s, h):
    C = []
    for k in range((int(1 / h) + 1)):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        for n in range((int(1 / s) + 1)):
            x = r_est + n * s
            if x > 1 - r_est:
                continue
            C.append((x, y))
            if (k % 2 == 1):
                x += r_est
                C.append((x, y))
    return np.array(C)


# Function to compute radii and sum for given centres
def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = 0.1  # Initial guess for radius (will update in Nelder-Mead)
        A_ub[idx + 4*i, i] = 1
        b_ub[idx + 4*i] = x - r
        A_ub[idx + 5*i, i] = 1
        b_ub[idx + 5*i] = 1 - x - r
        A_ub[idx + 6*i, i] = 1
        b_ub[idx + 6*i] = y - r
        A_ub[idx + 7*i, i] = 1
        b_ub[idx + 7*i] = 1 - y - r
        idx += 8

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), -np.inf

# Function to run the packing algorithm
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est

    candidates = build_lattice(r_est, s, h)

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for combo in combinations(candidates, 26):

        if sumR > best_sum:

    result = minimize(lambda x: -sum(lp_optimum(x.reshape(26, 2)).radi)[0],

                      bounds=bounds,
