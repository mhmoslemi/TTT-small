import numpy as np
from scipy.optimize import linprog

def lp_radii(centers):
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diffs, axis=2)
    A_ub = np.zeros((n * (n - 1), n))
    b_ub = dists.flatten()
    for i in range(n):
        A_ub[i * (n - 1):(i + 1) * (n - 1), i] = 1
        A_ub[i * (n - 1):(i + 1) * (n - 1), range(i + 1, n)] = 1
    
    for i in range(n):
        A_ub[i * (n - 1):(i + 1) * (n - 1), i] = 1
        b_ub[i * (n - 1):(i + 1) * (n - 1)] = centers[i, 0]
        A_ub[i * (n - 1):(i + 1) * (n - 1), i] = 1
        b_ub[i * (n - 1):(i + 1) * (n - 1)] = 1 - centers[i, 0]
        A_ub[i * (n - 1):(i + 1) * (n - 1), i] = 1
        b_ub[i * (n - 1):(i + 1) * (n - 1)] = centers[i, 1]
        A_ub[i * (n - 1):(i + 1) * (n - 1), i] = 1
        b_ub[i * (n - 1):(i + 1) * (n - 1)] = 1 - centers[i, 1]
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x, res.success

def create_initial_layout():
    # Initial layout of centres - deterministic template
    corners = [(0.12, 0.12), (0.12, 0.88), (0.88, 0.12), (0.88, 0.88)]
    centers = np.array(corners)
    side_length = 0.7 / 3  # Distance between centers along a side
    for i in range(4):
        for j in range(2):
            x = 0.12 + (1 - 2 * i) * side_length / 2 + j * side_length
            y = 0.12 + (2 * i + 1) * side_length / 2 + (i % 2) * side_length
            centers = np.vstack((centers, np.array((x, y))))
    # Remaining points in a smaller interior square
    interior_points = np.random.rand(14, 2) * 0.6 + 0.2
    centers = np.vstack((centers, interior_points))
    return centers

def jitter(centers, scale):
    jittered = centers + (scale * np.random.randn(*centers.shape))
    return np.clip(jittered, 0, 1)

def run_packing() -> tuple:
    np.random.seed(42)
    num_restarts = 200
    num_iterations_per_restart = 10
    best_centers = None
    best_sum = -np.inf
    best_radius_success = False
    
    for _ in range(num_restarts):
        # Create initial layout (deterministically)
        centers = create_initial_layout()
        
        for _ in range(num_iterations_per_restart):
            # Jitter the layout
            jittered_centers = jitter(centers, 0.03)
            # Solve the linear programme
            radii, success = lp_radii(jittered_centers)
            radii_sum = np.sum(radii)
            
            if success and radii_sum > best_sum:
                best_centers, best_sum, best_radius_success = jittered_centers, radii_sum, success
    
    if not best_radius_success:
        return np.zeros((26, 2)), np.zeros(26), 0
    
    return best_centers, best_radii, best_sum
