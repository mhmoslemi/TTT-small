import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 8*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    # Diagonal constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = np.sqrt(2) * (np.sqrt(2)/2 + 1e-12) - 1
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def random_center():
    return np.array([np.random.uniform(0, 1), np.random.uniform(0, 1)])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.1, 0.001
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centers = np.array([random_center() for _ in range(n)])
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_center = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_center

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
