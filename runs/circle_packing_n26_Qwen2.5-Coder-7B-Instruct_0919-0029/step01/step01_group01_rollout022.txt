```python
import numpy as np
from scipy.spatial import Delaunay, Voronoi, voronoi_plot_2d
from scipy.optimize import linprog

def calculate_voronoi_radii(centers):
    vor = Voronoi(centers)
    radii = np.zeros(centers.shape[0])
    for i in range(centers.shape[0]):
        region = vor.regions[vor.point_region[i]]
        if -1 in region:
            continue  # ignore infinite regions
        polygon = np.array([vor.vertices[j] for j in region])
        circum_center, circum_radius = calculate_circle_from_polygon(polygon)
        if circum_center[0] >= 0 and circum_center[0] <= 1 and circum_center[1] >= 0 and circum_center[1] <= 1:
            radii[i] = min(circum_radius, 1 - np.min(np.abs(centers[i] - np.array([[0, 0], [1, 0], [0, 1], [1, 1]]))))
    return radii

def calculate_circle_from_polygon(polygon):
    x = polygon[:, 0]
    y = polygon[:, 1]
    x_mean = np.mean(x)
    y_mean = np.mean(y)
    x_centered = x - x_mean
    y_centered = y - y_mean
    A = np.vstack([x_centered, y_centered, np.ones(len(x_centered))]).T
    b = x_centered**2 + y_centered**2
    center = np.linalg.lstsq(A, b, rcond=None)[0]
    radius = np.sqrt(center[0]**2 + center[1]**2 + center[2])
    return center[:2], radius

def lp_optimum(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing():
    numstarts = 50
    num_iters = 2000
    step = 0.01
    step_min = 0.001
    decay_factor = 0.995
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centers = np.random.rand(26, 2)
        for it in range(num_iters):
            idx = np.random.randint(26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            try:
                radii, sum_radii = lp_optimum(new_centers)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = new_centers
                    best_radii = radii
            except Exception as e:
                pass  # accept failure to escape; assume only permissible failures occur

            step = max(step_min, step * decay_factor)

    try:
        final_radii, final_sum = lp_optimum(best_centers)
    except Exception as e:
        return np.zeros((26, 2)), np.zeros(26), 0.0  # degenerate

    return best_centers, final_radii, final_sum

# Example usage
centers, radii, sum_radii = run_packing()
print(sum_radii)
```

This code uses the Voronoi diagram to find the circumcircle for each region and sets the radii based on the circumradius and the boundaries of the unit square. It then uses a linear programming optimizer to ensure non-overlap and boundary constraints. The process involves multiple random starts and iterations to find a good packing arrangement.
