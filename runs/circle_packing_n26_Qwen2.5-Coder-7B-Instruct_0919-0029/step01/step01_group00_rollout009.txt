from scipy.optimize import minimize, differential_evolution, linprog
from math import sqrt

        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))

def lp_optimum(centers):
    n = centers.shape[0]
    A = np.concatenate([np.eye(n) * 2, -np.eye(n)], axis=0)
    b = np.zeros(2*n)
    for i in range(n):
        r_lower = np.array([-centers[i, 0], -centers[i, 1], -1 + centers[i, 0], -1 + centers[i, 1]])
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j]) + 1e-12
            A = np.vstack([A, np.array([1, 1, -dist, -dist])])
            b = np.append(b, 0)
        A = np.vstack([A, r_lower])
        b = np.append(b, np.zeros(4))

    res = linprog(None, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x[:n], -res.fun

def run_packing():

    centres_best = result.x.reshape((26, 2))
    result = minimize(eval_funcs, centres_best.flatten(), method="Nelder-Mead", bounds=bounds)

    centres_best = result.x.reshape((26, 2))
    radii, total = lp_optimum(centres_best)

    return centres_best, radii, total
