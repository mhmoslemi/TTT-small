def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    import pulp as pl
    import numpy as np

    def d(x, y):
        return np.sqrt((x[0] - y[0])**2 + (x[1] - y[1])**2)

    # Define grid
    Δ = 0.075
    nx = ny = int(1/Δ) + 1
    X = np.arange(0, 1, Δ)
    Y = np.arange(0, 1, Δ)
    P = len(X) * len(Y)

    # Indexing
    I = np.arange(P).reshape((nx, ny))

    # Distance matrix
    dist_matrix = np.zeros((P, P))
    for i in range(P):
        for j in range(i + 1, P):
            dist_matrix[j, i] = dist_matrix[i, j] = d((X[i // ny], X[i % ny]), (X[j // ny], X[j % ny]))

    # Define problem
    prob = pl.LpProblem("circle_packing", pl.LpMaximize)

    # Decision variables
    y = [pl.LpVariable(f'y_{k}', cat='Binary') for k in range(P)]
    r = [pl.LpVariable(f'r_{k}', lowBound=0, upBound=0.5) for k in range(P)]

    # Objective
    prob += pl.lpSum(r[k] for k in range(P))

    # Constraints
    prob += pl.lpSum(y[k] for k in range(P)) == 26
    for k in range(P):
        prob += r[k] <= X[k]
        prob += r[k] <= 1 - X[k]
        prob += r[k] <= Y[k]
        prob += r[k] <= 1 - Y[k]
        prob += r[k] <= 0.5 * y[k]
    for k in range(P):
        for l in range(k + 1, P):
            prob += r[k] + r[l] <= dist_matrix[k, l] + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])

    # Solve problem
    prob.solve(pl.PULP_CBC_CMD(msg=False, timeLimit=30))

    # Retrieve solution
    centers = np.array([(X[i // ny], X[i % ny]) if pl.value(y[i]) > 0.5 else None for i in range(P)])
    radii = np.array([pl.value(r[i]) for i in range(P)])

    # Filter out inactive circles
    centers = centers[centers[:, 0] is not None]
    radii = radii[centers[:, 0] is not None]

    # Sum of radii
    sum_radii = sum(radii)

    if len(centers) < 26:
        raise Exception("Feasible solution not found.")

    return centers, radii, sum_radii
