import numpy as np
import scipy.optimize
from scipy.spatial.distance import pdist, squareform

def lp_optimum(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    radii = np.ones(n)
    for i in range(n):
        for j in range(i + 1, n):
            r_ij = dists[i, j] - 1e-12
            if r_ij > 0:
                x, y = centers[i]
                x2, y2 = centers[j]
                radius = 0.5 * (r_ij + x + x2 - np.sqrt((x + x2)**2 + (y - y2)**2))
                r_i = max(0, radius)
                r_j = max(0, r_ij - r_i)
                radii[i] = r_i
                radii[j] = r_j
    return radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        if x < 0 or x > 1 or y < 0 or y > 1:
            return False, f"Circle {i} at ({x},{y}) outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def evaluate(centres_flat):
    centres = centres_flat.reshape((26, 2))
    radii, total = lp_optimum(centres)
    if np.any(radii < 0):
        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))
    return -total

def run_packing():
    n = 26
    s = 0.2110
    h = 0.1825
    centres_flat = np.array([
        [x, y] for y in range(0, 1, int(np.ceil(1/h))) for x in range(0, 1, int(np.ceil(1/s))) 
        if x % 2 == y % 2 and x >= s and x <= 1-s and y >= s and y <= 1-s
    ]).flatten()
    centres_flat = centres_flat[:2*26].reshape((26, 2)).flatten()
    np.random.shuffle(centres_flat)
    bounds = [(0, 1)] * 2 * n
    result = scipy.optimize.differential_evolution(evaluate, bounds, maxiter=80, strategy="best1bin", popsize=50, mutation=0.5, tell=True)
    best_centers_flat = result.x.reshape((26, 2))
    best_radii, best_sum = lp_optimum(best_centers_flat)
    scipy.optimize.minimize(evaluate, best_centers_flat.flatten(), method="Nelder-Mead", bounds=bounds, options={"maxiter": 20})
    final_radii, final_sum = lp_optimum(best_centers_flat)
    return best_centers_flat.reshape((26, 2)), final_radii, final_sum

centers, radii, sum_radii = run_packing()
print((centers, radii, sum_radii))
