import numpy as np
from scipy.spatial.distance import cdist

def lp_optimum(centers):
    n = centers.shape[0]
    radii = Variable(n)
    constraints = [radii >= 0]
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            constraints.append(radii[i] + radii[j] <= dist)
        constraints.append(radii[i] <= centers[i, 0] - 1e-12)
        constraints.append(radii[i] <= 1 - centers[i, 0] - 1e-12)
        constraints.append(radii[i] <= centers[i, 1] - 1e-12)
        constraints.append(radii[i] <= 1 - centers[i, 1] - 1e-12)
    
    objective = Minimize(cp_sum(radii))
    prob = Problem(objective, constraints)
    result = prob.solve()
    return radii.value, -result

def evaluate(centres_flat):
    centres = centres_flat.reshape((26, 2))
    radii, total = lp_optimum(centres)
    if np.any(radii < 0):
        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))
    return -total
