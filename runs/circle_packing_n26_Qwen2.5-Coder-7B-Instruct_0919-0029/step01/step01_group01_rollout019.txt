import numpy as np
from scipy.optimize import linprog, minimize
from math import exp

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]**2, axis=2)))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def simulated_annealing(centers, init_radius=0.1, T0=0.1, Tf=1e-4, alpha=0.9995, delta=0.1, steps_per_T=500):
    n = centers.shape[0]
    r = np.full(n, init_radius)
    s_current = np.sum(r)
    best_centers = centers.copy()
    best_sum_radius = s_current

    T = T0
    for k in range(int(np.log(Tf / T0) / np.log(alpha)) + 1):
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            delta_i = np.random.normal(0, delta)
            centers[i] += (1, 1) * delta_i
            centers[i] = np.clip(centers[i], 0, 1)
            try:
                radii, sum_radii = lp_optimum(centers)
                centers[i] -= (1, 1) * delta_i
                radii, sum_radii = lp_optimum(centers)
                s_new = sum_radii
                delta_s = s_new - s_current
                if delta_s >= 0 or exp(delta_s / T) > np.random.rand():
                    r[i] = radii[i]
                    s_current = sum_radii
                    if s_new > best_sum_radius:
                        best_sum_radius = s_new
                        best_centers = centers.copy()
                else:
                    centers[i] -= (1, 1) * delta_i
            except Exception as e:
                pass
        T *= alpha
        
    return best_centers, r, best_sum_radius

def initial_hexagonal_lattice(n):
    circles_per_row = 1
    rows = 0
    while circles_per_row * (circles_per_row - 1) // 2 < n:
        circles_per_row += 1
    rows = circles_per_row - 1
    radius = 1 / (2 * np.sqrt(3))
    side = 2 * radius
    y_center = rows // 2 * side * np.sqrt(3) / 2
    x_offset = side * (rows - 1) / 2
    centers = np.zeros((n, 2))
    pos = 0
    for i in range(rows):
        for j in range(circles_per_row - i):
            centers[pos, 0] = x_offset + 2 * radius * i + j * side
            centers[pos, 1] = y_center + (i) * side * np.sqrt(3)
            pos += 1
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = initial_hexagonal_lattice(n)
    best_centers, best_radii, best_sum_radius = simulated_annealing(initial_centers)
    return best_centers, best_radii, best_sum_radius

**Explanation:** This script defines a strategy to optimize the placement and radii of 26 circles within a unit square by using simulated annealing based optimization. The positions are initialized in the shape of a hexagonal lattice that fits inside the unit square, followed by optimization steps using both a linear program to find the optimal radii for a given set of center positions and simulated annealing to explore new positions. The result is returned as a tuple of the final centre positions, the computed radii for each circle, and the sum of radii.
