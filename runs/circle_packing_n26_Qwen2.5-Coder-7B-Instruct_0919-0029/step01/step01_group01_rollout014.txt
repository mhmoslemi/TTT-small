import numpy as np
from heapq import heappop, heappush

curvature_to_radius = lambda k: 1 / np.sqrt(k)
radius_to_curvature = lambda r: 1.0 / (r * r)
interval_to_curvatures = lambda a, b, c, d: [
    [-1.0, -(a + b - c - d), b, c],
    [-1.0, -(a + c - b - d), c, d],
    [-1.0, -(b + c - a - d), a, d],
    [1.0, -(a + b + c - d), a, b, c],
    [1.0, -(a + b + d - c), a, b, d]
]

def enclosing_circle(c1, c2, c3):
    x1, x2, x3 = c1[0], c2[0], c3[0]
    y1, y2, y3 = c1[1], c2[1], c3[1]
    r1, r2, r3 = c1[2], c2[2], c3[2]

    bc = np.prod(np.array([
        [x2 - x3, y2 - y3, (r1 * r1 - r2 * r2) + (x2 - x3) ** 2 + (y2 - y3) ** 2],
        [x2 - x1, y2 - y1, (r2 * r2 - r1 * r1) + (x2 - x1) ** 2 + (y2 - y1) ** 2],
        [x3 - x1, y3 - y1, (r3 * r3 - r1 * r1) + (x3 - x1) ** 2 + (y3 - y1) ** 2]
    ]), axis=0)
    
    c = bc[:3] / 6.0
    r = bc[3] / (4.0 * np.linalg.norm(bc[:2]))

    return c, r

def construct_initial_circles():
    return [
        [0, 0, 1/2],
        [1, 0, 1/2],
        [0, 1, 1/2],
        [1, 1, 1/2]
    ]

def apollonian_packing(initial_circles, target_count=26):
    centre_to_gaps = {}

    for i in range(len(initial_circles) - 2):
        for j in range(i + 1, len(initial_circles) - 1):
            for k in range(j + 1, len(initial_circles)):
                c1, r1 = initial_circles[i]
                c2, r2 = initial_circles[j]
                c3, r3 = initial_circles[k]
                for circle in [c1, c2, c3]:
                    if circle not in centre_to_gaps:
                        centre_to_gaps[circle] = []
                enc_c, enc_r = enclosing_circle(c1, c2, c3)
                centre_to_gaps[c1].append([-enc_r, enc_c[0], enc_c[1]])
                centre_to_gaps[c2].append([-enc_r, enc_c[0], enc_c[1]])
                centre_to_gaps[c3].append([-enc_r, enc_c[0], enc_c[1]])

    queue = []

    for circle in centre_to_gaps:
        for gap in centre_to_gaps[circle]:
            heappush(queue, gap)

    circles = list(initial_circles)

    while len(circles) < target_count and queue:
        _, x, y, r = heappop(queue)
        if r < 1e-6:
            continue
        circles.append([x, y, r])
        centre_to_gaps = {}

        for i in range(len(circles) - 2):
            for j in range(i + 1, len(circles) - 1):
                for k in range(j + 1, len(circles)):
                    c1, r1 = circles[i]
                    c2, r2 = circles[j]
                    c3, r3 = circles[k]
                    for circle in [c1, c2, c3]:
                        if circle not in centre_to_gaps:
                            centre_to_gaps[circle] = []
                    enc_c, enc_r = enclosing_circle(c1, c2, c3)
                    centre_to_gaps[c1].append([-enc_r, enc_c[0], enc_c[1]])
                    centre_to_gaps[c2].append([-enc_r, enc_c[0], enc_c[1]])
                    centre_to_gaps[c3].append([-enc_r, enc_c[0], enc_c[1]])

        for circle in centre_to_gaps:
            for gap in centre_to_gaps[circle]:
                heappush(queue, gap)

    circles.sort(key=lambda c: -c[2])
    circles = circles[:target_count]
    trimmed_circles = [c[:3] for c in circles]
    return trimmed_circles

def refine_with_lp(centres):
    n = len(centres)
    nonneg = [(0, 1) for _ in range(n)]
    
    c = -np.array([c[2] for c in centres])
    A_eq = np.vstack([np.array([0] * i + [1] + [0] * (n - i - 1)) for i in range(n)])
    b_eq = np.array([np.min([c[i][1] - c[i][2], 1 - c[i][1], c[i][0] - c[i][2], 1 - c[i][0]]) if i % 2 == 0 else
                     np.min([c[i - 1][0] - c[i][2], 1 - c[i - 1][0], c[i][1] - c[i][2], 1 - c[i][1]]) for i in range(n)])

    lower_bounds = [0] * n
    upper_bounds = [np.inf] * n

    x = np.linalg.solve(np.vstack([A_eq, np.eye(n)]), np.hstack([b_eq, np.array([c[i] for i in range(n)])]))
    r = x[:n]
    valid = np.all(r >= 0) and np.all(np.linalg.norm(x[4*i:4*i+2] - x[4*j:4*j+2], 2) >= (x[2*i] + x[2*j]) for i in range(n) for j in range(i + 1, n))
    
    if valid:
        return r, n * np.min(np.array([c[i][2] for i in range(n)]))
    else:
        return np.array([0] * n), 0

def push_largest_circles(centres, radii):
    n = len(centres)
    centre_to_radii = {tuple(c): r for c, r in zip(centres, radii)}
    boundary_circles = []
    for i, c in enumerate(centres):
        if c[0] - radii[i] < 0 or c[0] + radii[i] > 1 or c[1] - radii[i] < 0 or c[1] + radii[i] > 1:
            boundary_circles.append(i)

    delta = 1e-4
    while True:
        smallest_change = float('inf')
        for i in boundary_circles:
            for j in range(len(centres)):
                if i == j:
                    continue
                d = np.linalg.norm(np.array(centres[i][:2]) - np.array(centres[j][:2]))
                if d < 2 * radii[i] + 2 * radii[j] - 2 * delta:
                    centre_to_radii[(centres[i][0], centres[i][1])] = radii[i]
                    centre_to_radii[(centres[j][0], centres[j][1])] = radii[j] + delta
                    smallest_change = 0
                    break

            if smallest_change == 0:
                break

        if smallest_change == float('inf'):
            break

        centres = [[c[0], c[1], centre_to_radii[(c[0], c[1])]] for c in centres]
        radii = [centre_to_radii[(c[0], c[1])] for c in centres]

    return centres, radii

def run_packing():
    initial_circles = construct_initial_circles()
    circles = apollonian_packing(initial_circles)
    refined_r, sum_radii = refine_with_lp(np.array(circles))
    if sum_radii < 0.5:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    centres, radii = np.array(circles), refined_r
    return centres, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    is_valid, msg = validate_packing(centers, radii)
    print(f"Valid: {is_valid}, Message: {msg}")
    print(f"Sum of radii: {sum_radii:.6f}")
