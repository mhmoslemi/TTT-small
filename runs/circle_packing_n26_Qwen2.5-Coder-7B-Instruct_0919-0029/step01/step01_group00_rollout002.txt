# Precompute a dense triangular lattice within the unit square
r_est = np.sqrt(0.9069 / (np.pi * 26))
s = 2 * r_est
h = np.sqrt(3) * r_est
triangular_lattice = []
for y in np.arange(r_est, 1 - r_est + h / 2, h):
    for x in np.arange(r_est, 1 - r_est + s / 2, s):
        if x >= r_est and x <= 1 - r_est and y >= r_est and y <= 1 - r_est:
            triangular_lattice.append([x, y])

# Since we need 26 centers, Duplicate the lattice twice and take the first 26 points
centers_start = np.array(triangular_lattice + triangular_lattice[:26])
