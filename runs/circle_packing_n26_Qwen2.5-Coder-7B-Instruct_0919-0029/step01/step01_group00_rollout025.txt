import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = np.sqrt(x**2 + y**2)
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    # Inner boundary at r = 0
    for i in range(n):
        x, y = centers[i]
        r = np.sqrt(x**2 + y**2)
        A_ub[idx, i] = 1
        b_ub[idx] = -r + 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    K = 36
    h = 0.41
    x = np.array([(.3 + 2 * i) % h for i in range(K/2)] * 2 + [h * 3.5 + (.3 + 2 * i) % h for i in range(K/2)])
    y = np.array([(-.7 + 2 * i) % h for i in range(K)] * 2)
    grid = np.stack((x, y))
    centers = grid.T[:26]

    num_subsets = np.math.comb(K, 26)
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    def evaluate_subset(subset):
        centers_subset = centers[subset]
        return lp_optimum(centers_subset)

    for subset in np.array(np.meshgrid(*[range(K)] * 26)).T.reshape(num_subsets, 26):
        radii, total = evaluate_subset(subset)
        if np.isnan(radii).any() or np.isnan(total):
            continue
        if total > best_sum:
            best_sum = total
            best_centers = centers[subset]
            best_radii = radii

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
print(sum_radii)
