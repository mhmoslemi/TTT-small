import numpy as np
import cvxpy as cp

def lp_optimum(centers):
    n = centers.shape[0]
    r = cp.Variable((n,), nonneg=True)

    objective = cp.Maximize(cp.sum(r))

    constraints = []
    for i in range(n):
        constraints.append(r[i] <= centers[i, 0])  # left boundary
        constraints.append(r[i] <= 1 - centers[i, 0])  # right boundary
        constraints.append(r[i] <= centers[i, 1])  # bottom boundary
        constraints.append(r[i] <= 1 - centers[i, 1])  # top boundary

    for i in range(n):
        for j in range(i+1, n):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= dist)

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.HSOLvers, verbose=False)
    return r.value, -objective.value
