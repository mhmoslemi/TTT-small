def construct_initial_circles():
    r = 0.25

def apollonian_packing(initial_circles, target_count=26):

    queue = []

            c4 = (c1[0] - c2[0] + c3[0]) / 3, 1 / k4_negative
            circle = (c4, c4[1])
            circles.append(circle)
            for i in [0, 1, 2]:
                queue.append((circle, circles[i], circles[(i+1)%3]))
