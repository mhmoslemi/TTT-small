import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    g = 0.01
    grid_points = np.array([(i*g, j*g) for i in range(101) for j in range(101)])
    
    radii = np.zeros(10001)
    active_indices = []
    
    while len(active_indices) < 26:
        min_growth_times = np.inf * np.ones_like(radii)
        contact_points = []

        for i, (x, y) in enumerate(grid_points):
            if radii[i] > 0:
                continue

            distances_to_boundaries = np.minimum(g - x, g - y)
            distances_to_active = np.sqrt(
                np.sum((grid_points[active_indices] - np.array([x, y])) ** 2, axis=1)
            )
            distances_to_active = np.clip(distances_to_active - radii[active_indices], 0, None) - g

            min_growth_times[i] = np.minimum(
                np.minimum(distances_to_boundaries, distances_to_active)
            )
            if min_growth_times[i] == distances_to_boundaries.min():
                contact_points.append((x, y))
        
        contact_point_index = np.argmin(min_growth_times)
        if np.isinf(min_growth_times[contact_point_index]):
            break
        
        active_indices.append(contact_point_index)
        radii += min_growth_times
        radii[contact_point_index] = 0

    active_circles = grid_points[active_indices]
    active_circles[:, 0] = np.clip(active_circles[:, 0], g, 1 - g)
    active_circles[:, 1] = np.clip(active_circles[:, 1], g, 1 - g)
    
    for _ in range(200):
        pressures = np.zeros((active_circles.shape[0], 2))
        for i, (x, y) in enumerate(active_circles):
            boundary_pressure = np.array([-1 if j < g else (1 - j) if j > 1 - g else 0 for j in x, y]).clip(min=0)
            neighbor_indices = np.flatnonzero(np.sum((active_circles - np.array([x, y])) ** 2, axis=1) < (radii[active_indices] + radii) ** 2)
            neighbor_pressures = np.array([np.array([j, k]) for j, k in active_circles[neighbor_indices]])
            relative_distances = active_circles[neighbor_indices] - np.array([x, y])
            repulsion_pressures = np.sum(relative_distances / (radii[active_indices] + radii)[:, np.newaxis], axis=0)
            total_pressure = boundary_pressure * 5 + np.sum(repulsion_pressures, axis=0)
            pressures[i] = total_pressure

        active_circles += pressures * 0.05
        active_circles[:, 0] = np.clip(active_circles[:, 0], g, 1 - g)
        active_circles[:, 1] = np.clip(active_circles[:, 1], g, 1 - g)
        radii = np.min(np.array([[min(r, np.maximum(np.sqrt(np.sum((active_circles - np.array([x, y])) ** 2, axis=1)), np.zeros(len(active_circles))[i]) - r) for x, y in active_circles] for r in radii]), axis=0)

    return active_circles[:26], radii[:26], np.sum(radii[:26])

centers, radii, sum_radii = run_packing()
