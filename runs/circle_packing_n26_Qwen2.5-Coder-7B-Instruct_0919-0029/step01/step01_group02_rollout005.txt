import numpy as np
import scipy.spatial

def grow_circles(grid_points, n_circles=26):
    centres = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    active = np.zeros(n_circles, dtype=bool)
    count = 0

    while count < n_circles:
        remaining_points = grid_points[~active]
        dt = np.inf * np.ones(len(remaining_points))
        
        for i in range(4):
            dist = np.abs(remaining_points[:, i] - np.clip(remaining_points[:, i], 0, 1 - radii[active]))
            dt = np.minimum(dt, dist)
        
        for i in range(n_circles):
            if active[i]:
                dist = scipy.spatial.distance.cdist(remaining_points, remaining_points[i None, :]
                dt = np.minimum(dt, np.maximum(dist[:, 1] - radii[active[i]] - radii[i], 0))

        idx = np.argmin(dt)
        if dt[idx] < 1e-9:
            break

        cent_x, cent_y = remaining_points[idx]
        r = dt[idx] - 1e-9  # Subtract a tiny epsilon to ensure feasibility
        centre = np.array([cent_x, cent_y])
        centre[r < 1e-9] = 1e-9  # Enforce minimal non-zero radius

        centres[count] = centre
        radii[count] = r
        active[count] = True
        count += 1

    return centres, radii

def refine_positions(centres, radii, alpha=0.05, max_iter=200):
    for _ in range(max_iter):
        pressures = np.zeros_like(centres)
        
        for i, centre in enumerate(centres):
            if radii[i] > 1e-7:
                r = radii[i]
                dx = (1 / r) * max(0, (np.sum(np.abs(centre - centres[np.where(active])[:, :1]) - r[np.where(active)], axis=1)) - 1)
                dy = (1 / r) * max(0, (np.sum(np.abs(centre - centres[np.where(active)][:, 1:]) - r[np.where(active)], axis=1)) - 1)
                pressures[i] = np.array([np.sum(dx), np.sum(dy)])
        
        centres += alpha * pressures
        centres = np.clip(centres, 0, 1)
        
        radii = np.minimum(np.sum(np.sqrt(np.sum(np.square(centres[:, np.newaxis, :] - centres[np.newaxis, :, :]), axis=2)) - radii[:, np.newaxis] - radii[np.newaxis, :], axis=1) / 2 - 1e-9, radii)

    return centres, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_points = np.array([(i * 0.01, j * 0.01) for i in range(101) for j in range(101)])
    centres, radii = grow_circles(grid_points)
    centres, radii = refine_positions(centres, radii)
    sum_radii = np.sum(radii)
    return centres, radii, sum_radii
