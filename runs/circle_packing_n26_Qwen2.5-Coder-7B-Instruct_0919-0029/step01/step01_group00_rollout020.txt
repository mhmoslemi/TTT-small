import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, minimize

# Define LP optimization to compute radii for given centers
def lp_optimum(centers):
    n = centers.shape[0]
    A = np.zeros((n * (n - 1) // 2, n))
    b = np.zeros(A.shape[0])

    # Pairwise distance squared constraints (using <= d^2 to avoid underflow)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 2
            A[idx, j] = 2
            b[idx] = np.sum((centers[i] - centers[j]) ** 2)
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = x
        A[idx, i] = 2
        b[idx] = 2 * (1 - x)
        idx += 1
        A[idx, i] = 2
        b[idx] = 2 * (x - 0)
        idx += 1
        A[idx, i] = 2
        b[idx] = 2 * (1 - y)
        idx += 1
        A[idx, i] = 2
        b[idx] = 2 * (y - 0)
        idx += 1

    # Set up objective and bounds
    c = np.ones(n)
    bounds = [(0, None)] * n

    # Solve the LP
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        radii = np.sqrt(result.x)
    else:
        radii = np.zeros(n)

    return radii, result.fun

def run_packing():
    n = 26
    M = 32

    # Generate triangular lattice of candidate centers
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    
    centers = []
    for k in range(M):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        for n in range(int((1 - y) / s + 1)):
            if ((k % 2 == 0) and (n % 2 == 1)) or ((k % 2 == 1) and (n % 2 == 0)):
                x = r_est + n * s
                centers.append(np.array([x, y]))
                x = r_est + r_est + n * s
                if x > 1 - r_est:
                    break
                centers.append(np.array([x, y]))
    
    centers = np.array(centers)

    best_sum = -np.inf
    best_centres = None
    best_radii = None

    # Enumerate all 26-subsets of candidate centers
    from itertools import combinations
    for subset in combinations(range(M), n):
        subset_centers = centers[list(subset), :]
        radii, total = lp_optimum(subset_centers)
        if np.any(radii < 0):
            continue
        if total > best_sum:
            best_sum = total
            best_centres = subset_centers.copy()
            best_radii = radii.copy()

    # Continuous refinement of the best discrete solution
    def ref_obj(centres_flat):
        centres_arr = centres_flat.reshape((n, 2))
        radii, total = lp_optimum(centres_arr)
        return -total  # Minimize negative sum to maximize sum

    initial_flat = best_centres.flatten()
    bounds = [(0, None)] * (2 * n)
    result = minimize(ref_obj, initial_flat, method='Nelder-Mead', bounds=bounds)
    refined_centres = result.x.reshape((n, 2))
    final_radii, final_sum = lp_optimum(refined_centres)

    if final_sum < 2.637:
        raise ValueError("Grid search strategy cannot deliver less than 2.637 for 26 circles")

    return best_centres, final_radii, final_sum
