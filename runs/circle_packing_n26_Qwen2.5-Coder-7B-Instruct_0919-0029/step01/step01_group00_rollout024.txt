import numpy as np
import itertools
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    C = []

    k = 0
    while True:
        y = r_est + k * h
        if y > 1 - r_est:
            k = 0
            break
        n = 0
        while True:
            x = r_est + n * s if k % 2 == 0 else r_est + r_est + n * s
            if x > 1 - r_est:
                break
            C.append(np.array([x, y]))
            n += 1
        k += 1

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for subset in itertools.combinations(C, 26):
        radi, sumR = lp_optimum(subset)
        if sumR > best_sum:
            best_sum = sumR
            best_centers = np.array(subset)
            best_radii = radi

    def refine(centers):
        bounds = [(0, None)] * len(centers) * 2
        constraints = []
        for i in range(len(centers)):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i] - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - 1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i+1] - 1e-12})

        def objective(x):
            return -np.sum(x[:len(centers)])

        result = linprog(c=-np.ones(len(centers)), A_eq=None, b_eq=None, A_ub=constraints, b_ub=[np.inf]*len(constraints), bounds=bounds, method='Nelder-Mead')
        return result.x[:len(centers)]

    refined_centers = refine(best_centers)
    refined_radii, refined_sum = lp_optimum(refined_centers)
    return refined_centers, refined_radii, refined_sum

centers, radii, sum_radii = run_packing()
