import numpy as np
from scipy.optimize import linprog

def construct_initial_circles():
    res = []
    for x in [0, 0.5, 1]:
        for y in [0, 0.5, 1]:
            if len(res) >= 3:
                break
            if x != 0.5 or y != 0.5:
                curv = np.sqrt(2) / x
                res.append((np.array([x, y]), 1 / curv))
    for i in range(3):
        res.append((np.array([0.5, 0.5]), 1 / (2 * np.sqrt(2))))
    return res

def apollonian_packing(initial_circles, target_count=26):
    num_circles = len(initial_circles)
    while True:
        new_circle = False
        for i in range(num_circles):
            for j in range(i+1, num_circles):
                for k in range(j+1, num_circles):
                    gap = calc_gap(initial_circles[i][0], initial_circles[j][0], initial_circles[k][0])
                    if gap[2] > 0 and gap[1] <= 1:
                        curv = gap[2]
                        c = new_circle_center(initial_circles[i][0], initial_circles[j][0], initial_circles[k][0], curv)
                        r = 1 / curv
                        initial_circles.append((c, r))
                        num_circles += 1
                        new_circle = True
                        break
                if new_circle:
                    break
            if new_circle:
                break
        if not new_circle or num_circles >= target_count:
            break
    initial_circles.sort(key=lambda x: -x[1])
    return initial_circles[:target_count]

def calc_gap(c1, c2, c3):
    dx12 = c1[0] - c2[0]
    dy12 = c1[1] - c2[1]
    dx13 = c1[0] - c3[0]
    dy13 = c1[1] - c3[1]
    dist12 = np.sqrt(dx12 ** 2 + dy12 ** 2)
    dist13 = np.sqrt(dx13 ** 2 + dy13 ** 2)
    s4 = 2 * np.sqrt((dist12 ** 2 + dist13 ** 2) - (dx12 * dx13 + dy12 * dy13))
    if s4 == 0:
        return None
    curv = (s4 - (1 / dist12 + 1 / dist13)) / (2 * (1 / dist12 - 1 / dist13))
    return (s4, dist12, curv)

def new_circle_center(c1, c2, c3, curv):
    dx12 = c1[0] - c2[0]
    dy12 = c1[1] - c2[1]
    dist12 = np.sqrt(dx12 ** 2 + dy12 ** 2)
    A = np.array([[dx12, dy12, -dist12],
                  [dx12 * dist12, dy12 * dist12, -(dist12 ** 3 - dist12 ** 2)],
                  [2 / (4 * curv), 2 / (4 * curv), 0]])
    B = np.array([-(2 * dist12 * curv), -(dist12 ** 3 * curv - dist12 ** 2 * curv), dist12 ** 3 * curv])
    return np.linalg.solve(A, B)

def refine_with_lp(centres):
    n = centres.shape[0]
    D = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(D, np.inf)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    A_ineq = [[1]] * n + [[-1]] * n + [[0, 1, 0, -1]] * n + [[1, 0, 0, -1]] * n
    b_ineq = np.array([1] * n + [0] * n + [0] * n + [1] * n)
    for i in range(n):
        for j in range(i+1, n):
            A_ineq.append([0, 0, 1, 1])
            b_ineq.append(D[i, j])
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def push_largest_circles(centres, radii):
    while any(k < 0.5 * k_j for k, k_j in zip(radii, np.max(radii))):
        idx = np.argmax(radii)
        r_max = radii[idx]
        i_max = np.argmax(np.dot(centres[idx] - centres, radii))
        r_i_max = radii[i_max]
        push_radius = np.min([0.0001 * r_max, 0.0001 * r_i_max])
        radii[idx] += push_radius
        radii[i_max] -= push_radius
    return radii, np.sum(radii)

def run_packing():
    initial_circles = construct_initial_circles()
    apollonian_set = apollonian_packing(initial_circles)
    centres = np.array([c[0] for c in apollonian_set])
    radii, sum_radii = refine_with_lp(centres)
    if sum_radii < 2.637:
        radii, sum_radii = push_largest_circles(centres, radii)
    return centres, radii, sum_radii
