import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def construct_initial_circles():
    r = 0.5
    left_center = (0, r)
    bottom_center = (r, 0)
    top_center = (0.5, 0.5)
    return [
        {'center': left_center, 'radius': r},
        {'center': bottom_center, 'radius': r},
        {'center': top_center, 'radius': r},
    ]

def apollonian_packing(initial_circles, target_count=26):
    objects = initial_circles[:]
    queue = []
    for i, obj1 in enumerate(objects):
        for j, obj2 in enumerate(objects[i:], start=i):
            for obj3 in objects[j:]:
                if i < j < len(objects):
                    k = compute_k(obj1, obj2, obj3)
                    if k > 0:
                        center = compute_center(obj1, obj2, obj3, k)
                        if 0 <= center[0] <= 1 and 0 <= center[1] <= 1:
                            circle = {'center': center, 'radius': 1 / k}
                            queue.append(circle)

    circles = initial_circles[:]
    while len(circles) < 26 and queue:
        new_circle = max(queue, key=lambda c: c['radius'])
        queue.remove(new_circle)
        circles.append(new_circle)
        for obj in circles:
            if not overlaps(obj, new_circle):
                k = compute_k(obj, new_circle)
                if k > 0:
                    new_center = compute_center(obj, new_circle, new_circle['center'], k)
                    new_circle['center'] = new_center
                    queue.append({'center': new_center, 'radius': 1 / k})

    return circles

def compute_k(obj1, obj2, obj3):
    k1, k2, k3 = obj1['radius'], obj2['radius'], obj3['radius']
    px, py = obj1['center'][0], obj1['center'][1]
    qx, qy = obj2['center'][0], obj2['center'][1]
    rx, ry = obj3['center'][0], obj3['center'][1]
    a = 1 / k1
    b = 1 / k2
    c = 1 / k3
    m = a * ((qy - ry) * (px - qx) - (py - qy) * (qx - rx)) + \
        b * ((ry - py) * (qx - rx) - (ry - qy) * (px - qx)) + \
        c * ((py - qy) * (rx - qx) - (ry - py) * (qx - px))
    if a == 0:
        a = b
        b = 0
        np = 1 / k2
        nq = -1 / k3
    elif b == 0:
        np = -1 / k1
        nq = 1 / k3
    elif c == 0:
        np = -1 / k1
        nq = -1 / k2

    d = np * px + nq * py + a * (-px**2 + ry**2 - rx**2 + qx**2) + b * (-py**2 + ry**2 - qx**2 + rx**2)
    m = np**2 / a**2 + nq**2 / b**2 - c / a
    if m <= 0:
        return -np.sqrt(c / a)
    return np.sqrt(c / a) + np.sqrt(m)

def compute_center(obj1, obj2, obj3, k):
    px, py = obj1['center']
    qx, qy = obj2['center']
    rx, ry = obj3['center']
    r1, r2, r3 = obj1['radius'], obj2['radius'], obj3['radius']
    np = k * (1 - r1 * r1 - r2 * r2 + r3 * r3) / (-r3)
    nq = k * (1 - r1 * r1 - r3 * r3 + r2 * r2) / (-r2)
    dx = nq * (py - qy) + np * (qx - rx)
    dy = np * (qx - rx) + nq * (ry - py)
    return px + dx / k, py + dy / k

def overlaps(obj1, obj2):
    return np.linalg.norm(np.array(obj1['center']) - np.array(obj2['center'])) < obj1['radius'] + obj2['radius']

def refine_with_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ineq = np.zeros((3 * n, 2 * n))
    b_ineq = np.zeros(3 * n)
    for i in range(n):
        x, y = centers[i]
        r = 0.5
        A_ineq[i] = [1, 0]
        b_ineq[i] = x - 1e-12
        A_ineq[n + i] = [-1, 0]
        b_ineq[n + i] = 1 - x - 1e-12
        A_ineq[2 * n + i] = [0, 1]
        b_ineq[2 * n + i] = y - 1e-12
        A_ineq[3 * n + i] = [0, -1]
        b_ineq[3 * n + i] = 1 - y - 1e-12
        for j in range(n):
            if i < j:
                dist = np.linalg.norm(centers[i] - centers[j])
                A_ineq[i + 3 * n, 2 * i] = 1
                A_ineq[i + 3 * n, 2 * j + 1] = 1
                b_ineq[i + 3 * n] = dist - 1e-12

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=[(0, None)] * (2 * n))
    if not result.success:
        raise Exception("LP optimization failed")
    return result.x[:n], -result.fun

def push_largest_circles(centers, radii, tolerance=1e-4, max_iterations=100):
    while max(radii) - tolerance > np.linalg.norm(centers, axis=1):
        new_radii, sum_radii = refine_with_lp(centers)
        if sum_radii < sum(radii):
            return centers, new_radii, sum_radii
        radii = new_radii
    return centers, radii, sum(radii)

def run_packing():
    initial_circles = construct_initial_circles()
    circles = apollonian_packing(initial_circles)
    centers = np.array([c['center'] for c in circles])
    radii = np.array([c['radius'] for c in circles])
    refined_radii, refined_sum = refine_with_lp(centers)
    final_centers, final_radii, final_sum = push_largest_circles(centers, refined_radii)
    return final_centers, final_radii, final_sum
