import numpy as np
from scipy.spatial import Delaunay, ConvexHull
from scipy.optimize import minimize

def optimize_radii(centers, initial_radii):
    n = centers.shape[0]
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        in_sphere = np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2) <= 4 * np.expand_dims(radii, 1) * np.expand_dims(radii, 0)
        np.fill_diagonal(in_sphere, False)
        return np.sum(in_sphere)

    x0 = initial_radii
    cons = ({'type': 'ineq', 'fun': constraint})
    bounds = [(0, None)] * n
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.05, 0.001
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centres = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c

            initial_radii = np.ones(n)
            new_radii, new_sum = optimize_radii(new_centres, initial_radii)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centres
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = optimize_radii(best_centers, best_radii)
    return best_centers, final_radii, final_sum
