import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def sa_optimization(centers, radii):
    n = len(centers)
    obj = sum(radii)
    final_centers = centers.copy()
    final_radii = radii.copy()
    best_obj = obj
    T = 0.1
    Tf = 1e-4
    alpha = 0.9995
    Delta = 0.1
    steps_per_T = 500
    max_iter = steps_per_T * int(np.log(Tf / T) / np.log(alpha))
    for _ in range(max_iter):
        if T <= Tf:
            T = Tf
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            perturbation = np.random.normal(0, Delta, 2)
            perturbation = np.clip(perturbation, -0.5, 0.5)
            new_center = centers[i] + perturbation
            new_center = np.clip(new_center, 0, 1)
            new_centers = centers.copy()
            new_centers[i] = new_center
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > obj or np.random.rand() < np.exp((new_sum - obj) / T):
                centers = new_centers
                radii = new_radii
                obj = new_sum
                if new_sum > best_obj:
                    best_obj = new_sum
                    final_centers = new_centers
                    final_radii = new_radii
        Delta *= alpha
        T *= alpha
    return final_centers, final_radii, best_obj
