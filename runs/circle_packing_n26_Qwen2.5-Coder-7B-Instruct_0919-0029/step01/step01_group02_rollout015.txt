def g_left(X, i):
    return X[3*i] - X[3*i+2]  # x - r

def g_right(X, i):
    return 1 - X[3*i] - X[3*i+2]  # (1-x) - r

def g_bot(X, i):
    return X[3*i+1] - X[3*i+2]  # y - r

    return 1 - X[3*i+1] - X[3*i+2]  # (1-y) - r
