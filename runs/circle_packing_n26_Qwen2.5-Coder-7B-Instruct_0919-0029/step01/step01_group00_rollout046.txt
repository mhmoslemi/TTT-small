import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def refine(centers, radii):
    n = len(centers)
    cons = [{'type': 'ineq', 'fun': lambda x, r=i: x[r*2] - (x[r*2 + 1] - 0.05)}
            for r in range(n)]
    for r in range(n):
        for s in range(r + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, r=r, s=s: 0.15 - np.sqrt((x[r*2] - x[s*2])**2 + (x[r*2 + 1] - x[s*2 + 1])**2 - (radii[r]**2 + radii[s]**2))})
    res = minimize(lambda x: -np.sum(radii), centers.flatten(), method='Nelder-Mead', bounds=[(0, 1), (0, 1)] * n, constraints=cons)
    return res.x.reshape((n, 2)), lp_optimum(res.x.reshape((n, 2)))[0]

def run_packing():
    n = 26
    spacing = np.sqrt(0.9069 / (np.pi * n)) * 2
    centers = []
    for k in range(int(1 / spacing) + 1):
        y = spacing * (k + 0.5)
        if y <= 1 - spacing / 2:
            for j in range(int(1 / (2 * spacing)) + 1):
                x = spacing * j
                centers.append([x, y])
                if k % 2 == 1:
                    x += spacing / 2
                    centers.append([x, y])
    centers = np.array(centers[:n])

    best_sum = float('-inf')
    for comb in combinations(centers, n):
        comb = np.array(comb)
        radii, total = lp_optimum(comb)
        if total > best_sum:
            best_sum = total
            best_centers = comb

    final_radii, final_sum = refine(best_centers.flatten(), radii)
    if final_sum < 2.637:
        raise ValueError("Final sum is below the target")
    return best_centers, final_radii, final_sum
