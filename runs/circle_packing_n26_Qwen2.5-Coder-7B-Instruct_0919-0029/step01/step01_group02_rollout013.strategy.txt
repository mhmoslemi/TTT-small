def sphere_packing_obj(x):
    n = 26
    # reshape: shape (n, 3)
    arr = x.reshape((n,3))
    x_coords = arr[:,0]; y_coords=arr[:,1]; r = arr[:,2]
    sum_r = np.sum(r)
    violation = 0.0
    # boundary
    violation += np.sum(np.maximum(0.0, r - x_coords)**2)
    violation += np.sum(np.maximum(0.0, r - (1 - x_coords))**2)
    violation += np.sum(np.maximum(0.0, r - y_coords)**2)
    violation += np.sum(np.maximum(0.0, r - (1 - y_coords))**2)

            if v > 0:

    return -sum_r + 1e4 * violation
