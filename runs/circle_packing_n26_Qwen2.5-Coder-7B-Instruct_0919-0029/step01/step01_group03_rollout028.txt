import numpy as np
from scipy.optimize import linprog

def lp_radii(centers):
    n = len(centers)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    A_ub = np.array([np.concatenate(([1 if k==i or k==j else 0] * n)) for i in range(n) for j in range(i+1, n)])
    b_ub = dist.flatten()
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    scale = 0.05

    # Initial layout
    centers = np.zeros((26, 2))
    centers[:4, :] = np.array([[0.015, 0.015], [0.015, 0.985], [0.985, 0.985], [0.985, 0.015]])
    for i in range(4, 8):
        centers[i, :] = centers[0, :] + np.array([0.25, 0]) if i == 4 else centers[-1, :] + np.array([0.25, 0])
    indices = np.arange(8, 26)
    np.random.shuffle(indices)
    centers[indices, :] = centers[4:8, :] + np.array([[0, 0.25], [0, 0.5], [0, 0.75], [0, 0.25], [0, 0.5], [0, 0.75], [0.5, 0.5], [0.75, 0.5], [0.5, 0.25], [0.5, 0.75], [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.25, 0.45], [0.75, 0.45], [0.45, 0.25], [0.45, 0.75]]

    best_centers = None
    best_sum = -np.inf

    for _ in range(100):
        if _ == 0:
            candidate_centers = centers.copy()
        else:
            candidate_centers = best_centers + np.random.normal(0, scale, size=centers.shape)
            candidate_centers = np.clip(candidate_centers, 0, 1)

        candidate_radii = lp_radii(candidate_centers)
        if candidate_radii is not None:
            candidate_sum = np.sum(candidate_radii)
            if candidate_sum > best_sum:
                best_sum = candidate_sum
                best_centers = candidate_centers

    return best_centers, lp_radii(best_centers), best_sum
