import itertools
import numpy as np
from scipy.optimize import linprog, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def construct_lattice():
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    centers = []
    for k in range(int(1 / h) + 1):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        for n in range(int((1 - y) / s) + 1):
            x = r_est + n * s
            if x > 1 - r_est:
                break
            centers.append([x, y])
        for n in range(int((1 - x) / s) + 1):
            x = r_est + r_est + n * s
            if x > 1 - r_est:
                break
            centers.append([x, y])
    return np.array(centers)

def lsa_optimize_step(subset):
    def obj(radii):
        n = subset.shape[0]
        A = np.zeros((n, n))
        for i in range(n):
            for j in np.arange(i+1, n):
                dist = np.linalg.norm(subset[i] - subset[j])
                A[i, j] = A[j, i] = dist - radii[i] - radii[j]
        return -np.sum(radii) + 1e6 * np.sum(np.maximum(A, 0)**2)

    x0 = np.mean([c for c in centers for i in subset])
    res = minimize(obj, x0, method='Nelder-Mead', bounds=[(0, None), (0, None)] * len(subset))
    new_radii = res.x[-len(subset):]
    new_centers = centers[:, subset]
    new_centres = (new_centers.T * (np.max(new_radii) / np.max(res.x)) + np.min(res.x)).T
    new_radii = np.maximum(new_radii, 0)
    return new_centers.T, new_radii, np.sum(new_radii)

def run_packing():
    M = 32
    candidate_centers = construct_lattice()[:M]
    best_sum = -np.inf
    best_subset = None

    for subset in combinations(candidate_centers, 26):
        subset = np.array(subset)
        subres = lp_optimum(subset)
        if subres[1] > best_sum:
            best_sum = subres[1]
            best_subset = subset

    best_centers, best_radii, best_sum = lsa_optimize_step(best_subset)
    return best_centers, best_radii, best_sum

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(centers.shape, radii.shape, sum_radii)
