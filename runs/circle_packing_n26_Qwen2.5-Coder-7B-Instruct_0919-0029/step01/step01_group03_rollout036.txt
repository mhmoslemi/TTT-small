import numpy as np
from scipy.optimize import linprog

def linear_program(radii_from_centers):
    n = len(radii_from_centers)
    # Pairwise distances
    diff = radii_from_centers[:, None, :] - radii_from_centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    
    # Build the upper-bound constraints matrix A_ub
    A_ub = np.zeros((3 * n + n * (n - 1) // 2, n))
    b_ub = np.zeros(3 * n + n * (n - 1) // 2)
    
    # Boundary constraints
    for i in range(n):
        A_ub[i] = [1 if j == i else 0 for j in range(n)]
        b_ub[i] = radii_from_centers[i]
        A_ub[n + i] = [1 if j == i else 0 for j in range(n)]
        b_ub[n + i] = 1 - radii_from_centers[i]
        A_ub[2 * n + i] = [1 if j == i else 0 for j in range(n)]
        b_ub[2 * n + i] = radii_from_centers[i]
        A_ub[3 * n + i] = [1 if j == i else 0 for j in range(n)]
        b_ub[3 * n + i] = 1 - radii_from_centers[i]
    
    # Pairwise constraints
    k = 3 * n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k] = [1 if c == i else 0 for c in range(n)] + [1 if c == j else 0 for c in range(n)]
            b_ub[k] = dist[i, j]
            k += 1
    
    c = -np.ones(n)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    radii = res.x
    return radii, res.fun
