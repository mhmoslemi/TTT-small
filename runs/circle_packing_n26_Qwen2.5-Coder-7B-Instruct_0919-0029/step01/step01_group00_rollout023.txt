import numpy as np
from scipy.optimize import linprog
from itertools import combinations
from scipy.optimize import minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def refine_positions(centers, initial_radii):
    initial_sum = np.sum(initial_radii)
    def objective(new_positions):
        new_radii, _ = lp_optimum(new_positions)
        return -np.sum(new_radii)
    
    constraints = [{'type': 'ineq', 'fun': lambda p, i=i: p[i, 0] - 1e-12},
                   {'type': 'ineq', 'fun': lambda p, i=i: 1 - p[i, 0] - 1e-12},
                   {'type': 'ineq', 'fun': lambda p, i=i: p[i, 1] - 1e-12},
                   {'type': 'ineq', 'fun': lambda p, i=i: 1 - p[i, 1] - 1e-12}]
    
    bounds = [(0, None)] * 52
    result = minimize(objective, centers.flatten(), method='Nelder-Mead', bounds=bounds, constraints=constraints)
    refined_centers = result.x.reshape((26, 2))
    refined_radii, refined_sum = lp_optimum(refined_centers)
    return refined_centers, refined_radii, refined_sum

def run_packing():
    n = 26
    candidate_positions = []
    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    y = r_est
    while y <= 1 - r_est:
        if y >= 0.15 or y <= 1 - 0.15:
            x_even = r_est
            while x_even <= 1 - r_est:
                candidate_positions.append((x_even, y))
                x_even += s
            x_odd = r_est + r_est
            while x_odd <= 1 - r_est:
                candidate_positions.append((x_odd, y))
                x_odd += s
        y += h

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for subset in combinations(candidate_positions, n):
        centers = np.array(subset)
        radii, sumR = lp_optimum(centers)
        if sumR > best_sum:
            best_sum = sumR
            best_centers = centers
            best_radii = radii

    final_centers, final_radii, final_sum = refine_positions(best_centers, best_radii)
    return final_centers, final_radii, final_sum
