import numpy as np
import pulp

def build_grid(step_size):
    n = int(1 / step_size)
    x = np.arange(0, 1 + step_size, step_size)
    y = np.arange(0, 1 + step_size, step_size)
    X, Y = np.meshgrid(x, y)
    return X.reshape(-1), Y.reshape(-1)

def euclidean_distance(x1, y1, x2, y2):
    return np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)

def build_milp(X, Y):
    P = len(X)
    y = pulp.LpVariable.dicts("y", range(P), cat="Binary")
    r = pulp.LpVariable.dicts("r", range(P), lowBound=0, upBound=0.5)

    prob = pulp.LpProblem("Circle_Packing", pulp.LpMaximize)

    # Objective: maximize the sum of radii
    prob += pulp.lpSum(r[k] for k in range(P))

    # Constraint: exactly 26 circles
    prob += pulp.lpSum(y[k] for k in range(P)) == 26

    # Overlap constraints
    for k in range(P):
        for l in range(k + 1, P):
            if k != l:
                dist_kl = euclidean_distance(X[k], Y[k], X[l], Y[l])
                prob += r[k] + r[l] <= dist_kl + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])

    # Boundary containment constraints
    for k in range(P):
        prob += r[k] <= np.minimum(X[k], 1 - X[k], Y[k], 1 - Y[k])

    return prob, y, r

def solve_milp(prob, begin):
    for k, yk in enumerate(begin):
        prob += y[k] == int(yk > 0.5)

    prob.solve()

    return pulp.value(prob.objective), dict(prob.variables_dict())

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    step_size = 0.075
    X, Y = build_grid(step_size)
    
    prob, y, r = build_milp(X, Y)
    
    # Greedy initial solution
    greedy_solution = np.zeros(len(X))
    max_area = np.pi * 0.25**2 * 25 / 26
    for k in range(26):
        covered = set()
        for i in range(len(X)):
            if i in covered:
                continue
            radius = min(np.minimum(X[i], 1 - X[i], Y[i], 1 - Y[i]) / 2, np.pi * r[i].varValue / np.pi - 1e-12)
           greedy_solution[i] = max(greedy_solution[i], radius)
            for j in range(len(X)):
                if j != i and j not in covered and euclidean_distance(X[i], Y[i], X[j], Y[j]) <= 2 * radius:
                    covered.add(j)
    begin = greedy_solution > 0
    
    sum_radii, sol = solve_milp(prob, begin)
    centers, radii = [], []
    for k in range(len(X)):
        if sol[f'y_{k}'].varValue > 0.5:
            centers.append([X[k], Y[k]])
            radii.append(sol[f'r_{k}'].varValue)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    return centers, radii, sum_radii
