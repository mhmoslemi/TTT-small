import numpy as np
from scipy.optimize import minimize

def objective(z):
    N = 26
    x = z[:N]
    y = z[N:2*N]
    r = z[2*N:]
    W = 2500  # Penalty weight
    
    # Distance matrix
    diff = x[:, None] - x[None, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-10)
    
    # Feasibility penalties
    phi_bound = np.maximum(0, r - x) + np.maximum(0, r - (1 - x)) + \
               np.maximum(0, r - y) + np.maximum(0, r - (1 - y))
    phi_ovlp = np.maximum(0, r[:, None] + r[None, :] - dist[None, :, :] + 1e-10)
    
    # Objective function
    obj = -np.sum(r) - W * (np.sum(phi_bound**2) + np.sum(phi_ovlp**2))
    
    return obj

def gradient(z):
    N = 26
    x = z[:N]
    y = z[N:2*N]
    r = z[2*N:]
    W = 2500  # Penalty weight
    
    # Distance matrix
    diff = x[:, None] - x[None, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-10)
    
    # Feasibility penalties
    phi_bound = np.maximum(0, r - x) + np.maximum(0, r - (1 - x)) + \
               np.maximum(0, r - y) + np.maximum(0, r - (1 - y))
    phi_ovlp = np.maximum(0, r[:, None] + r[None, :] - dist[None, :, :] + 1e-10)
    
    # Gradient of objective function
    G = np.zeros_like(z)
    G[:N] = -2 * W * phi_bound  # grad wrt x
    G[N:2*N] = -2 * W * phi_bound  # grad wrt y
    G[2*N:] = -1 + np.sum(phi_ovlp > 0, axis=0) - lengthy  # grad wrt r
    
    return G

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    bounds = [(0, 1)] * (3 * N)
    initial_positions = np.array([
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.1],
        [0.9, 0.9],
        [0.3, 0.1],
        [0.3, 0.3],
        [0.7, 0.1],
        [0.7, 0.3],
        [0.1, 0.3],
        [0.1, 0.7],
        [0.9, 0.3],
        [0.9, 0.7],
        [0.3, 0.7],
        [0.3, 0.9],
        [0.7, 0.7],
        [0.7, 0.9],
        [0.15, 0.15],
        [0.15, 0.85],
        [0.85, 0.15],
        [0.85, 0.85],
        [0.45, 0.15],
        [0.45, 0.45],
        [0.55, 0.15],
        [0.55, 0.45],
        [0.15, 0.55],
        [0.15, 0.85],
        [0.85, 0.55],
        [0.85, 0.85]
    ])
    initial_radii = np.random.uniform(0.05, 0.15, N)
    initial_z = np.concatenate([initial_positions.flatten(), initial_radii])
    
    best_res = minimize(objective, initial_z, method='L-BFGS-B', bounds=bounds, jac=gradient)
    
    for _ in range(2):
        if best_res.success:
            break
        W *= 2
        best_res = minimize(objective, initial_z, method='L-BFGS-B', bounds=bounds, jac=gradient)
    
    if not best_res.success:
        raise Exception("Optimization failed")
    
    best_centers = best_res.x[:N].reshape(-1, 2)
    best_radii = best_res.x[2*N:]
    return best_centers, best_radii, -best_res.fun

centers, radii, sum_radii = run_packing()
print("Best Centers:", centers)
print("Best Radii:", radii)
print("Sum of Radii:", sum_radii)
