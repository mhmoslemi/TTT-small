import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def construct_initial_circles():
    r = 1.0 / 2
    centers = np.array([
        [r, r],
        [1 - r, r],
        [1 - r, 1 - r]
    ])
    return centers, np.full(3, r)

def apply_descartes(curvatures, centers):
    curvatures = curvatures.reshape(-1, 4)
    centers = centers.reshape(-1, 4, 2)
    curvatures[:, 1:] = 0  # Treat lines as circles of curvature 0
    new_curvatures = curvatures[:, 0] + curvatures[:, 1] + curvatures[:, 2] - 2 * np.sqrt(curvatures[:, 0] * curvatures[:, 1] + curvatures[:, 1] * curvatures[:, 2] + curvatures[:, 2] * curvatures[:, 0])
    new_centers = centers[:, 0][:, None] + centers[:, 1][:, None] + centers[:, 2][:, None]  # Placeholder, not actually used
    return new_curvatures, new_centers

def apollonian_packing(centres, target_count=26):
    circles = []
    queue = [(np.array([0, 0]), np.array([1, 0]), np.array([0, 1])), 
            (np.array([0, 0]), np.array([1, 0]), np.array([1, 1])), 
            (np.array([0, 0]), np.array([0, 1]), np.array([1, 1]))]
    while len(circles) < target_count and queue:
        _, _, queue = zip(*sorted(queue, key=lambda x: np.sum(x[2] ** 2), reverse=True))
        circle = apply_descartes(queue[0][2], queue[0][:3])
        circles.append((queue[0][:3], circle[0]))
        queue = [x for x in queue[1:] if np.isclose(circle[1], np.array([[0, 0]]), atol=1e-12).all(axis=1).any() == False]
    while len(circles) < target_count:
        circles.append((np.array([0, 0]), 0))
    return circles

def refine_with_lp(centres):
    n = centres.shape[0]
    dists = distance_matrix(centres, centres, Euclidean=True)
    np.fill_diagonal(dists, np.inf)
    A_ineq = np.zeros((n*(n-1), n))
    b_ineq = np.zeros(n*(n-1))
    c = np.ones(n)
    A_eq = np.zeros((n, n))
    b_eq = np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    
    count = 0
    for i in range(n):
        A_eq[i, i] = 1
        for j in range(i+1, n):
            A_ineq[count, :] = 2 * np.array([centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1], 1])
            b_ineq[count] = dists[i, j] + 1e-12
            count += 1

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def push_largest_circles(centres, radii):
    for _ in range(100):
        gaps = radii < 1 - centres[:, 0]
        if not np.any(gaps):
            break
        largest_circle = np.argmax(radii[gaps])
        radii[largest_circle] += 1e-4
        target_sum = np.sum(radii)
        result = refine_with_lp(centres)
        radii, sum_radii = result
        if sum_radii <= target_sum:
            break
    return radii, sum_radii

def run_packing():
    circles = apollonian_packing(*construct_initial_circles())
    circles = sorted(circles, key=lambda x: x[1], reverse=True)
    circles = circles[:26]
    centres = np.array([x[0] for x in circles])
    radii, sum_radii = push_largest_circles(centres, np.array([x[1] for x in circles]))
    return centres, radii, sum_radii
