import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import minimize
from itertools import combinations

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = minimize(lambda radii: -np.sum(radii),
                      bounds=bounds,
                      method='SLSQP',
                      hess=None,
                      x0=np.ones(n),
                      constraint={'type': 'ineq', 'fun': lambda radii, centers=centers: A_ub @ radii - b_ub})
    return result.x, -result.fun

def run_packing():
    n = 26

    # Step 1: Generate candidate lattice of centers
    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    g = np.arange(0, 1, h)
    l = np.array([x + y % 2 * s / 2 for x in g for y in g if x + y < 1])
    P = np.array([[x, y] for x, y in zip(l, g) if 0 < x < 1 and 0 < y < 1])

    # Step 2: Enumerate all 26-subsets and find the best
    best_sum = -np.inf
    best_subset = None
    for subset in combinations(P, n):
        radii, total_sum = lp_optimum(subset)
        if total_sum > best_sum or (total_sum == best_sum and np.max(radii) < np.max(best_subset.radii)):
            best_subset = np.array(subset)
            best_sum = total_sum

    # Step 3: Local refinement using Nelder-Mead
    def objective(x):
        centers = x.reshape((n, 2))
        return -np.sum(lp_optimum(centers)[1])

    initial_guess = best_subset.flatten()
    bounds = [(0, None)] * (2 * n)
    result = minimize(objective, initial_guess, bounds=bounds, method='Nelder-Mead')
    refined_centers = result.x.reshape((n, 2))
    refined_radii, refined_sum = lp_optimum(refined_centers)

    return refined_centers, refined_radii, refined_sum
