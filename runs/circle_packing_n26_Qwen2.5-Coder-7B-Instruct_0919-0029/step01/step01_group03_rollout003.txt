import numpy as np
import pulp        
from scipy.spatial import Delaunay

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the grid
    Δ = 0.075
    nx, ny = 14, 14
    grid_x = np.arange(Δ/2, 1, Δ)
    grid_y = np.arange(Δ/2, 1, Δ)
    X, Y = np.meshgrid(grid_x, grid_y)
    X, Y = X.flatten(), Y.flatten()
    P = len(X)
    
    # Define the problem
    problem = pulp.LpProblem("Circle_packing", pulp.LpMaximize)
    
    # Decision variables
    y = {k: pulp.LpVariable(f'y_{k}', cat='Binary') for k in range(P)}
    r = {k: pulp.LpVariable(f'r_{k}', lowBound=0, upBound=0.5) for k in range(P)}
    
    # Constraints
    # 1. Exactly 26 circles
    problem += pulp.lpSum(y[k] for k in range(P)) == 26
    
    # 2. No-overlap
    for k in range(P):
        for l in range(k+1, P):
            d = np.sqrt((X[k] - X[l])**2 + (Y[k] - Y[l])**2)
            problem += r[k] + r[l] <= d + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l])
    
    # 3. Boundary containment
    for k in range(P):
        problem += r[k] <= X[k]
        problem += r[k] <= 1 - X[k]
        problem += r[k] <= Y[k]
        problem += r[k] <= 1 - Y[k]
    
    # Objective function
    problem += pulp.lpSum(r[k] for k in range(P))
    
    # Solve the problem
    solver = pulp.Solver('CBC', problem)
    solver.solve()
    
    # Extract the solution
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    idx = 0
    for k in range(P):
        if y[k].varValue > 1e-12:
            centers[idx, 0] = X[k]
            centers[idx, 1] = Y[k]
            radii[idx] = r[k].varValue
            idx += 1
    
    return centers, radii, np.sum(radii)
