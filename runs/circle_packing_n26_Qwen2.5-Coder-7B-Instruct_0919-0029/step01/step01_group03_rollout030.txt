import numpy as np
from scipy.optimize import minimize

# Constants
N = 26
radius_bounds = (0, 0.4)
weight = 2000
penalty_limit = 1e-4
max_adaptive_steps = 5
max_fun = 2000
ftol = 1e-9
gtol = 1e-5

bnd = np.tile(radius_bounds, (2 * N, 1))

def initial_config(seed):
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.random.uniform(0.05, 0.15, size=N)
    
    # Corner seeds
    r = 0.12
    centers[0] = [r, r]
    centers[1] = [1 - r, r]
    centers[2] = [r, 1 - r]
    centers[3] = [1 - r, 1 - r]
    
    # Perimeter band
    spacing = 0.25 / np.sqrt(2)
    for i in range(2):
        centers[4 + i] = [0.5 - spacing / 2, r + spacing * i]
        centers[4 + 2 + i] = [0.5 - spacing / 2, 0.5 - r - spacing * i]
        centers[4 + 4 + i] = [r + spacing * i, 0.5 - spacing / 2]
        centers[4 + 4 + 2 + i] = [0.5 - r - spacing * i, 0.5 - spacing / 2]
    
    # Interior core
    centers[16:] = np.random.uniform(0.25, 0.75, size=(10, 2))
    
    return centers.flatten(), radii.flatten()

def cost_function(z):
    centers = z[:N*2].reshape(-1, 2)
    radii = z[N*2:]
    overlap = np.maximum(0, radii[:, None] + radii[None, :] - np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2)))
    boundary = np.maximum(0, radii[:, None] - centers[:, None, 0]) + np.maximum(0, radii[:, None] - (1 - centers[:, None, 0])) + np.maximum(0, radii[:, None] - centers[:, None, 1]) + np.maximum(0, radii[:, None] - (1 - centers[:, None, 1]))
    return -np.sum(radii) + weight * np.sum(overlap**2) + weight * np.sum(boundary**2)

def gradient(z):
    centers = z[:N*2].reshape(-1, 2)
    radii = z[N*2:]
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    overlap = np.maximum(0, radii[:, None] + radii[None, :] - dist)
    boundary = np.maximum(0, radii[:, None] - centers[:, None, 0]) + np.maximum(0, radii[:, None] - (1 - centers[:, None, 0])) + np.maximum(0, radii[:, None] - centers[:, None, 1]) + np.maximum(0, radii[:, None] - (1 - centers[:, None, 1]))
    
    grad_center_i = np.einsum('igj,ig,ij->ij', diff, np.divide(overlap, dist), np.ones((26, 26, 26)))[:, :, 1:].sum(axis=1) - np.einsum('igj,ig,ij->ij', diff, np.divide(boundary, centers[:, None, 0]), np.arange(26)).sum(axis=1)
    grad_center_j = np.einsum('igj,ig,ij->ij', diff, np.divide(overlap, dist), np.ones((26, 26, 26)))[:, :, 0:-1].sum(axis=1)
    grad_radius = overlap.sum(axis=1) + boundary.sum(axis=1)
    
    return np.concatenate([grad_center_i, grad_center_j, -grad_radius])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_z = None
    best_score = -np.inf
    
    for _ in range(30):
        z0, r0 = initial_config _
        res = minimize(cost_function, z0, jac=gradient, method='L-BFGS-B', bounds=bnd, options={'maxfun': max_fun, 'ftol': ftol, 'gtol': gtol})
        
        while True:
            score = cost_function(res.x)
            if score > best_score:
                best_z = res.x
                best_score = score
            else:
                if np.max(overlap) <= penalty_limit:
                    break
                weight *= 2
                res = minimize(cost_function, res.x, jac=gradient, method='L-BFGS-B', bounds=bnd, options={'maxfun': max_fun, 'ftol': ftol, 'gtol': gtol})
                if weight > 1e6:
                    raise ValueError('Maximum weight reached without convergence')
    
    centers = best_z[:N*2].reshape(-1, 2)
    radii = best_z[N*2:]
    validate_status, message = validate_packing(centers, radii)
    if not validate_status:
        raise ValueError(f'Validation failed: {message}')
    
    return centers, radii, -best_score / weight

# Run the packing
centers, radii, sum_radii = run_packing()
print(f'Centers: {centers}')
print(f'Radii: {radii}')
print(f'Sum of radii: {sum_radii}')
