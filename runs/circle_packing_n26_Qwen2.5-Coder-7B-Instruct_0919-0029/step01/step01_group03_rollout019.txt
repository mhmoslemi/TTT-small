import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n_circles = 26
    r_min = 1e-6
    
    # Define a small perturbation parameter
    alpha = 0.02
    
    # Define the radius of the lattice
    r = 0.15
    
    # Build the initial hexagonal lattice
    delta_x = 2 * r
    delta_y = np.sqrt(3) * r
    k_max_x = int(1 / delta_x)
    k_max_y = int(np.sqrt(3) / delta_y)
    
    # Initialize the set of lattice points
    lattice_points = []
    for k_x in range(k_max_x + 1):
        for k_y in range(k_max_y + 1):
            x = r + k_x * delta_x
            y = r + k_y * delta_y + (k_x % 2) * (delta_y / 2)
            if 0 <= x <= 1 and 0 <= y <= 1:
                lattice_points.append((x, y))
    
    # Select 26 points from the lattice
    selected_indices = np.random.choice(range(len(lattice_points)), n_circles, replace=False)
    centers = np.array([lattice_points[i] for i in selected_indices])
    
    # Perturb the lattice points
    for i in range(n_circles):
        perturbation = np.random.uniform(-alpha, alpha, 2)
        centers[i] = np.clip(centers[i] + perturbation, 0, 1)
    
    # Define the annealing parameters
    n_steps = 30000
    initial_temperature = -np.log(0.3 / (1 - 0.3)) / n_steps
    temperature_schedule = initial_temperature * (0.995 ** np.arange(n_steps))
    
    # Initialize variables to store the best solution
    best_centers = centers
    best_radii = np.repeat(r_min, n_circles)
    best_sum_radii = 0
    
    # Simulated Annealing optimization of the insertion order
    for step in range(n_steps):
        # Generate a random permutation of the circle indices
        current_order = np.random.permutation(n_circles)
        current_radii = np.zeros(n_circles)
        current_sum = 0
        
        for k in range(n_circles):
            i = current_order[k]
            
            # Compute distances to boundaries
            rho_left = centers[i, 0]
            rho_right = 1 - centers[i, 0]
            rho_bottom = centers[i, 1]
            rho_top = 1 - centers[i, 1]
            r_max = min(rho_left, rho_right, rho_bottom, rho_top)
            
            # Compute distances to previously placed circles
            for j in range(k):
                r_j = current_radii[j]
                d_ij = np.linalg.norm(centers[i] - centers[j]) - r_j
                r_max = min(r_max, d_ij)
            
            # Set the radius for the current circle
            current_radii[i] = max(r_max, r_min)
            current_sum += current_radii[i]
        
        # Accept/reject the new state
        if step == 0 or (current_sum > best_sum_radii or
                        (np.random.rand() < np.exp((current_sum - best_sum_radii) / temperature_schedule[step]))):
            centers = best_centers.copy()
            best_centers = centers.copy()
            best_radii = current_radii.copy()
            best_sum_radii = current_sum
    
    # Run the LP refinement of the radii for the best centers
    bounds = [(0, None) for _ in range(n_circles)]
    constraints = []
    for i in range(n_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r: 1 - best_centers[i, 0] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r: best_centers[i, 0] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r: 1 - best_centers[i, 1] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r: best_centers[i, 1] - r[i]})
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.linalg.norm(best_centers[i] - best_centers[j]) - r[i] - r[j]})
    
    result = linprog(-best_radii, A_ub=np.array([[1] * n_circles]), b_ub=np.array([1]), method='highs', bounds=bounds, constraints=constraints)
    best_radians = -result.x
    final_sum_radii = result.fun * -1
    
    return best_centers, best_radians, final_sum_radii
