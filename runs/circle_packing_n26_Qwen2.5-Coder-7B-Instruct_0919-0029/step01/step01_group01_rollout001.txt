import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = []
    b_ub = []

    A_ub.extend([[1, 0] + [0]*n]*n)
    b_ub.extend([0.0, 1.0]*n)
    
    x_min = [0]*n
    y_min = [0]*n
    x_max = [1]*n
    y_max = [1]*n

    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinite on the diagonal

    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([0, 0, 1]*n + [dists[i, j]]*n)
            b_ub.append(dists[i, j])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def de_objective(flat_centres):
    centre = flat_centres.reshape(26, 2)
    theoretical_radii, _ = solve_radius_lp(centre)
    total_radius = np.sum(theoretical_radii)
    penalty = 0
    for x, y, r in zip(centre[:, 0], centre[:, 1], theoretical_radii):
        penalty += np.log(max(0, x - r)) + np.log(max(0, y - r)) + np.log(max(0, 1 - x - r)) + np.log(max(0, 1 - y - r))
    return -total_radius + penalty

def run_packing():
   bounds = [(0.0,1.0)] * 52
    result = differential_evolution(
                de_objective,
                bounds,
                maxiter=120,
                popsize=12,
                tol=1e-6,
                polish=True,
                seed=42)
    centres = result.x.reshape(26, 2)
    theoretical_radii, final_sum = solve_radius_lp(centres)
    return centres, theoretical_radii, final_sum
