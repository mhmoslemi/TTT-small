import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def de_objective(flat_centres):
    centres = flat_centres.reshape(26, 2)
    _, s = solve_radius_lp(centres)
    return -s  # DE minimises

def run_packing():
    bounds = [(0.0, 1.0)] * 52  # one bound per coordinate
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=120,  # about 120*popsize evaluations
        popsize=15,   # 15*52 ≈ 780 individuals
        tol=1e-6,
        polish=True,  # final local refine with L-BFGS
        seed=42
    )
    best_centers = result.x.reshape(26, 2)
    final_radii, final_sum = solve_radius_lp(best_centers)
    return best_centers, final_radii, final_sum
