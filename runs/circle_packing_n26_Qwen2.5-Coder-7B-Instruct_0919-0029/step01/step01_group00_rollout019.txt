import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist
from scipy.spatial import Delaunay

def lp_optimum(centers):
    n = centers.shape[0]
    A = np.vstack([centers[:, 0], centers[:, 1]]).T
    c = np.ones(n)

    A_eq = np.zeros((n, 2 * n))
    for i in range(n):
        A_eq[i, 2 * i] = 1
        A_eq[i, 2 * i + 1] = centers[i, 1]
        A_eq[i, 2 * (centers.shape[0] - i - 1)] = centers[i, 0]
        A_eq[i, 2 * (centers.shape[0] - i - 1) + 1] = 1

    b_eq = centers[:, 0] + centers[:, 1]

    A_ub = np.zeros((n * (n - 1) // 2, 2 * n))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[n * (i - 1) // 2 + j - i - 1, 2 * i] = 1
            A_ub[n * (i - 1) // 2 + j - i - 1, 2 * i + 1] = -1
            A_ub[n * (i - 1) // 2 + j - i - 1, 2 * j] = -1
            A_ub[n * (i - 1) // 2 + j - i - 1, 2 * j + 1] = 1

    b_ub = -np.sqrt(np.sum((centers[i] - centers[j]) ** 2 for i in range(n) for j in range(i + 1, n)))

    bounds = [(0, None)] * 2 * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        radii = result.x / 2
        return radii, np.sum(radii)
    return np.ones(n), 0

def continuous_refinement(centers):
    def ref_obj(centers_flat):
        centres_arr = centres_flat.reshape((26, 2))
        return -lp_optimum(centres_arr)[1]

    initial_simplex = np.vstack([centers, centers + np.random.uniform(-0.005, 0.005, size=(26, 2))])
    result = minimize(ref_obj, initial_simplex.ravel(), method='Nelder-Mead', options={'maxiter': 30, 'xatol': 1e-4})
    return result.x.reshape((26, 2)), -result.fun

def run_packing():
    n = 26
    lattice_radius = np.sqrt(0.9069 / (np.pi * n))
    lattice_step_size_h = np.sqrt(3) * lattice_radius
    lattice_step_size_w = 2 * lattice_radius

    lattice = []
    for k in range(int(1 / lattice_step_size_h) + 1):
        if lattice_step_size_h * k > 1 - lattice_radius:
            break
        for m in range(int((1 - lattice_radius) / lattice_step_size_w) + 1):
            if lattice_step_size_w * m > 1 - lattice_radius:
                break
            if m % 2 == 0:
                lattice.append([lattice_radius + m * lattice_step_size_w, lattice_radius + k * lattice_step_size_h])
            else:
                lattice.append([lattice_radius + m * lattice_step_size_w + lattice_radius, lattice_radius + k * lattice_step_size_h])

    lattice = np.array(lattice)
    lattice = lattice[:n]

    best_sum = -np.inf
    best_centres = None

    for subset in np.array(np.meshgrid(range(n), range(n), range(n), range(n), range(n))).T.reshape(-1, n):
        subset = subset[np.random.permutation(n)][:n]
        centres = lattice[subset]
        radii, total = lp_optimum(centres)
        if total > best_sum:
            best_sum = total
            best_centres = centres.copy()

    best_centres, best_sum = continuous_refinement(best_centres)
    final_radii, final_sum = lp_optimum(best_centres)

    return best_centres, final_radii, final_sum
