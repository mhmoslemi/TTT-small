import numpy as np
from itertools import combinations

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            B_sq = np.sum((centers[i] - centers[j])**2)
            A_ub[idx, idx] = -2*(centers[i] - centers[j])
            b_ub[idx] = -B_sq
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 2
        b_ub[idx] = 0
        idx += 1
        A_ub[idx, i] = -2
        b_ub[idx] = 0
        idx += 1
        A_ub[idx, i] = 2
        b_ub[idx] = 0
        idx += 1
        A_ub[idx, i] = -2
        b_ub[idx] = 0
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP solver failed with message: " + result.message)

def packet_voronoi(n, d=None):
    if d is None:
        d = np.sqrt(3 * np.pi / (2*n))

    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est

    centers = []
    for k in range(40):  # Upper bound to ensure we have enough points
        y = r_est + k * h
        if y > 1 - r_est:
            break
        if k % 2 == 0:
            for n in range(40):
                x = r_est + n * s
                if x > 1 - r_est:
                    break
                centers.append([x, y])
        else:
            for n in range(40):
                x = r_est + r_est + n * s
                if x > 1 - r_est:
                    break
                centers.append([x, y])

    return np.array(centers)

def run_packing():
    centers = packet_voronoi(26)
    best_sum = -np.inf
    best_subset = None

    for subset in combinations(range(centers.shape[0]), 26):
        subset_centers = centers[subset]
        try:
            radii, sum_r = lp_optimum(subset_centers)
            if sum_r > best_sum:
                best_sum = sum_r
                best_subset = subset
        except Exception:
            continue

    if best_subset is None:
        raise Exception("No valid packing found")

    refined_centers = best_subset
    for _ in range(50):
        best_sum = -np.inf
        for subset in combinations(refined_centers, 26):
            subset_centers = centers[subset]
            try:
                radii, sum_r = lp_optimum(subset_centers)
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_subset = subset
            except Exception:
                continue

        refined_centers = best_subset

    best_radii, _ = lp_optimum(refined_centers)
    return best_centers[refined_centers], best_radii, best_sum
