import numpy as np
from scipy.spatial import ConvexHull
from scipy.optimize import linprog

def lp_optimisation(centres):
    n = centres.shape[0]
    coords = np.vstack((centres[:, 0], centres[:, 1])).T
    hull = ConvexHull(coords)
    interior_points = np.array([coords[i] for i in hull.vertices])
    x_min, y_min = interior_points.min(axis=0)
    x_max, y_max = interior_points.max(axis=0)
    padding = 0.01
    x_min, y_min = x_min - padding, y_min - padding
    x_max, y_max = x_max + padding, y_max + padding
    ranges = ((x_min, x_max), (y_min, y_max))

    A_ineq, b_ineq = [], []
    for i in range(n):
        A_ineq.append([1, 0])
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([-1, 0])
        b_ineq.append(-centres[i, 0] - 1e-12)
        A_ineq.append([0, 1])
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0, -1])
        b_ineq.append(-centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            A_ineq.append([1, 0])
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimisation failed")

def run_packing():
    numstarts = 20
    num_iters = 2000
    step = 0.05
    step_min = 0.001
    decay_factor = 0.995
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centres = np.random.rand(26, 2)
        for it in range(num_iters):
            idx = np.random.randint(26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c

            try:
                radii, sum_radii = lp_optimisation(new_centres)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centres = new_centres
                    best_radii = radii
            except Exception as e:
                pass  # accept failure to escape; assume only permissible failures occur

            step = max(step_min, step * decay_factor)

    try:
        final_radii, final_sum = lp_optimisation(best_centres)
    except Exception as e:
        return np.zeros((26, 2)), np.zeros(26), 0.0  # degenerate

    return best_centres, final_radii, final_sum
