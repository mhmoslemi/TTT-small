import numpy as np
from scipy.optimize import linprog

def lp_optimum(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing():
    numstarts = 20
    num_iters = 2000
    step = 0.05
    step_min = 0.001
    decay_factor = 0.995
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        # Start with a hexagonal packing pattern
        dime = np.sqrt(3) / 3
        circle_diameter = dime * 2 / 3
        num_cols = int(1 / circle_diameter) + 1
        num_rows = int(np.sqrt(3 * 1 / dime)) + 1

        centers = np.zeros((num_cols * num_rows, 2))
        row_count = 0
        for col in range(num_cols):
            for row in range(num_rows):
                if (col + row) % 2 == 0:
                    centers[row_count, 0] = col * circle_diameter
                    centers[row_count, 1] = row * circle_diameter
                    row_count += 1
                else:
                    centers[row_count, 0] = col * circle_diameter + dime
                    centers[row_count, 1] = row * circle_diameter
                    row_count += 1

        # Trim unnecessary center points
        centers = centers[:26]

        for it in range(num_iters):
            idx = np.random.randint(26)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            try:
                radii, sum_radii = lp_optimum(new_centers)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = new_centers
                    best_radii = radii
            except Exception as e:
                pass  # accept failure to escape; assume only permissible failures occur

            step = max(step_min, step * decay_factor)

    try:
        final_radii, final_sum = lp_optimum(best_centers)
    except Exception as e:
        return np.zeros((26, 2)), np.zeros(26), 0.0  # degenerate

    return best_centers, final_radii, final_sum
