import numpy as np
from scipy.spatial import Delaunay, distance_matrix
from scipy.optimize import linprog

def compute_k_curvature(x, y):
    k = np.sqrt(2) / np.sqrt(x**2 + y**2)
    return k

def construct_initial_circles():
    # Construct the initial circles
    circles = [
        {'center': (0.5, 0.5), 'radius': 0.5},
        {'center': (0, 1), 'radius': 1},
        {'center': (1, 1), 'radius': 1},
        {'center': (0.5, 1), 'radius': np.sqrt(2)/2}
    ]
    return circles

def apollonian_packing(initial_circles, target_count=26):
    circles = initial_circles[:]
    gaps = []

    # Define lines (as circles with infinitely large radius)
    lines = [{'center': (0, y), 'radius': 0} for y in [0, 1]]
    lines += [{'center': (x, 0), 'radius': 0} for x in [0, 1]]

    # Define the initial center-circles
    lines += [{'center': (c['center'][0], c['center'][1]), 'radius': 0} for c in circles]

    # Compute gaps and insert circles
    while len(circles) < target_count:
        # Find the most promising gap
        curvatures = []
        for gap in gaps:
            k_prev = gap['k']
            if k_prev < 0:
                continue
            cx, cy, r = gap['center']
            for circle in circles:
                cx1, cy1 = circle['center']
                r1 = circle['radius']
                delta_x = cx - cx1
                delta_y = cy - cy1
                if delta_x**2 + delta_y**2 <= (r + r1)**2:
                    break
            else:
                k = compute_k_curvature(delta_x, delta_y) - (r + r1)
                curvatures.append((k, cx, cy, r, circle['center'], circle['radius']))
        if not curvatures:
            break
        k, cx, cy, r, center1, radius1 = sorted(curvatures, key=lambda x: x[0])[0]

        # Calculate new circle parameters
        cx_new = cx - r1 * (cx - cx1) / np.sqrt(delta_x**2 + delta_y**2)
        cy_new = cy - r1 * (cy - cy1) / np.sqrt(delta_x**2 + delta_y**2)
        r_new = np.abs(cy_new - cy1) / np.sqrt(2)

        # Add new circle
        circles.append({'center': (cx_new, cy_new), 'radius': r_new})

        # Update gaps
        gaps.remove(gap)
        for line in lines:
            cx2, cy2 = line['center']
            r2 = line['radius']
            delta_x = cx - cx2
            delta_y = cy - cy2
            if (cx_new - cx2)**2 + (cy_new - cy2)**2 <= (r_new + r2)**2:
                if k < 0 and (cx_new - cx2)**2 + (cy_new - cy2)**2 == (r_new + r2)**2:
                    continue
                new_gap = {'center': (cx_new, cy_new), 'k': k + r2, 'radius': r_new + r2}
                gaps.append(new_gap)
            for circle in circles:
                cx3, cy3 = circle['center']
                if (cx_new - cx3)**2 + (cy_new - cy3)**2 <= (r_new + circle['radius'])**2:
                    break
            else:
                new_gap = {'center': (cx_new, cy_new), 'k': k + circle['radius'], 'radius': r_new + circle['radius']}
                gaps.append(new_gap)

        # Add lines
        gaps += [{'center': (cx_new, cy_new), 'k': -np.inf, 'radius': 1} for _ in range(4)]

    # Sort circles by radius
    circles.sort(key=lambda x: x['radius'], reverse=True)
    
    # Truncate to target_count circles
    circles = circles[:target_count]

    return circles

def refine_with_lp(centres):
    n = centres.shape[0]
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1])

        for j in range(i + 1, n):
            delta_x = centres[i, 0] - centres[j, 0]
            delta_y = centres[i, 1] - centres[j, 1]
            dist = np.sqrt(delta_x**2 + delta_y**2)
            if dist < 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
            A_ineq.append([delta_x, delta_y, 1])
            b_ineq.append(dist)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def push_largest_circles(centres, radii):
    n = centres.shape[0]
    idx = np.argmax(radii)
    for _ in range(100):
        dx = np.random.uniform(-1e-4, 1e-4)
        dy = np.random.uniform(-1e-4, 1e-4)
        new_c = (centres[idx, 0] + dx, centres[idx, 1] + dy)
        if new_c[0] >= 0 and new_c[0] <= 1 and new_c[1] >= 0 and new_c[1] <= 1:
            new_radii, new_sum = refine_with_lp(centres)
            if new_sum > max(radii):
                centres[idx, 0] = new_c[0]
                centres[idx, 1] = new_c[1]
                radii[idx] = new_radii[idx]
                break
    return centres, radii

def run_packing():
    initial_circles = construct_initial_circles()
    circles = apollonian_packing(initial_circles)
    centres = np.array([[c['center'][0], c['center'][1]] for c in circles])
    radii = np.array([c['radius'] for c in circles])
    sum_radii = np.sum(radii)

    centres, radii, sum_radii = refine_with_lp(centres)

    centres, radii = push_largest_circles(centres, radii)

    return centres, radii, sum_radii
