import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    W = 2000
    radius_bounds = [(0, 0.4)] * N
    position_bounds = [(0, 1)] * (2 * N)
    bounds = radius_bounds + position_bounds
    
    def penalty(z):
        radii = z[:N]
        centers = z[N:].reshape(N, 2)
        dist = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
        bound_penalty = np.sum(np.maximum(0, radii - centers[:, 0]) ** 2 +
                              np.maximum(0, radii - (1 - centers[:, 0])) ** 2 +
                              np.maximum(0, radii - centers[:, 1]) ** 2 +
                              np.maximum(0, radii - (1 - centers[:, 1])) ** 2)
        overlap_penalty = np.sum(np.maximum(0, radii + radii.T - dist) ** 2)
        return W * (bound_penalty + overlap_penalty)
    
    def objective(z):
        return -np.sum(z[:N]) - penalty(z)
    
    def penalty_gradient(z):
        radii = z[:N]
        centers = z[N:].reshape(N, 2)
        dist = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
        grad = -np.ones_like(z)
        B = np.zeros((2 * N, 2 * N))
        
        for i in range(N):
            grad[radii >= 0] = 0
            grad[N:i*N + 2*N:i*N + 2] += -2 * np.maximum(0, radii - centers[:, 0]) * (radii - centers[:, 0] > 0)
            grad[N:i*N + 2*N:i*N + 2] += -2 * np.maximum(0, radii - (1 - centers[:, 0])) * (radii - (1 - centers[:, 0]) > 0)
            grad[N:i*N + 2*N:i*N + 2] += -2 * np.maximum(0, radii - centers[:, 1]) * (radii - centers[:, 1] > 0)
            grad[N:i*N + 2*N:i*N + 2] += -2 * np.maximum(0, radii - (1 - centers[:, 1])) * (radii - (1 - centers[:, 1]) > 0)
            
            grad[i] = -1
            for j in range(N):
                grad[j] = -np.sum(np.maximum(0, radii + radii[j] - dist[i, j]) * (radii[j] + radii[i] > 0)) * 2
        
        B[:N, :N] += np.eye(N)
        B[N:, :N] = centers
        B[:N, N:] = centers
        B[N:, N:] -= np.eye(N)
        grad = B @ grad
        
        return grad
    
    def initial_positions():
        core = np.random.uniform(0.25, 0.75, (14, 2))
        corners = np.array([
            [0.12, 0.12],
            [1 - 0.12, 0.12],
            [0.12, 1 - 0.12],
            [1 - 0.12, 1 - 0.12]
        ])
        perimeter = np.array([
            [0.1, 0.375],
            [0.375, 0.1],
            [0.625, 0.1],
            [0.89, 0.375],
            [0.89, 0.625],
            [0.89, 0.89],
            [0.625, 0.89],
            [0.375, 0.89],
            [0.1, 0.625],
            [0.1, 0.375]
        ])
        return np.vstack((corners, perimeter, core))
    
    def run_single_restart():
        z0 = np.zeros(2 * N)
        z0[:N] = np.random.uniform(0.05, 0.15, N)
        z0[N:] = initial_positions().flatten()
        res = minimize(objective, z0, method='L-BFGS-B', bounds=bounds, jac=penalty_gradient, options={'maxfun': 2000, 'ftol': 1e-9, 'gtol': 1e-5})
        if res.success:
            return res.x
    
    best_z = None
    best_value = -np.inf
    
    for _ in range(20):
        z = run_single_restart()
        value = objective(z)
        if value > best_value:
            best_value = value
            best_z = z
    
    radii = best_z[:N]
    centers = best_z[N:].reshape(N, 2)
    
    def lp_solve(centers):
        from scipy.optimize import linprog
        obj = -np.ones(N)
        A = []
        b = []
        
        # Boundary constraints
        for i in range(N):
            A.append([1 if j == i else 0 for j in range(N)] + list((1 - centers[i, :])) * 2)
            b.append(0)
        
        # Overlap constraints
        for i in range(N):
            for j in range(i + 1, N):
                A.append([0] * N + [-1, 0, -1 if j == i else 0 for _ in centers[i, :]] + [1, 0, 1 if j == i else 0 for _ in centers[j, :]])
                A.append([0] * N + [-1, 0, 1 if j == i else 0 for _ in centers[i, :]] + [-1, 0, -1 if j == i else 0 for _ in centers[j, :]])
                b.append(radii[i] + radii[j] - np.sqrt(((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)))
                b.append(radii[i] + radii[j] - np.sqrt(((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)))
        
        res = linprog(obj, A_ub=A, b_ub=b, bounds=[(0, 0.5)] * N)
        if res.success:
            return res.x
        return None
    
    final_radii = lp_solve(centers)
    if final_radii is not None:
        return centers, final_radii, np.sum(final_radii)
    return centers, radii, np.sum(radii)
