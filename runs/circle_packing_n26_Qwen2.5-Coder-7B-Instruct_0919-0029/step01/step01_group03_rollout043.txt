        b[i + n] = 1 - centers[i, 0] if i % 2 == 0 else centers[i, 0]
        A[n*2 + i*2, i] = 1
        b[n*2 + i*2] = 1 - centers[i, 1] if i % 2 == 1 else centers[i, 1]
        for j in range(i + 1, n):
            idx = dists.size - (j - i - 1) // 2 if j > i else 0
            A[2*n + idx, i] = 1
            A[2*n + idx, j] = 1
            b[2*n + idx] = dist_matrix[i, j]

    c = np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x, res.success
