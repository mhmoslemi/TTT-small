import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import minimize

def objective_function(x, dists):
    n = int(np.sqrt(len(x)))
    radii = x.reshape(n, n)
    return -np.sum(radii)

def constraint_function(x, dists):
    n = int(np.sqrt(len(x)))
    radii = x.reshape(n, n)
    overlaps = radii + radii.T - dists - 1e-12
    return np.sum(np.maximum(0, overlaps))

def run_packing():
    n = 26
    side_length = np.sqrt(n)
    centers = np.random.rand(n, 2)
    dists = squareform(pdist(centers))
    initial_radii = np.zeros(n)
    initial_x = initial_radii.flatten()

    constraints = {'type': 'ineq', 'fun': constraint_function, 'args': (dists,)}

    bounds = [(0, None)] * n**2

    result = minimize(objective_function, initial_x, args=(dists,), method='SLSQP', bounds=bounds, constraints=constraints)

    if result.success:
        radii = result.x.reshape(n, n)
        valid_radii = radii diagonal
        centers = centers[result.success_indices]
        radii = valid_radii[result.success_indices]
        return centers, radii
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0
