import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x):
    n = 26
    arr = x.reshape((n, 3))
    x_coords, y_coords, r = arr[:, 0], arr[:, 1], arr[:, 2]
    sum_r = np.sum(r)
    violation = 0.0
    # boundary
    violation += np.sum(np.maximum(0.0, r - x_coords) ** 2)
    violation += np.sum(np.maximum(0.0, r - (1 - x_coords)) ** 2)
    violation += np.sum(np.maximum(0.0, r - y_coords) ** 2)
    violation += np.sum(np.maximum(0.0, r - (1 - y_coords)) ** 2)
    # pairwise
    for i in range(n-1):
        xi, yi, ri = x_coords[i], y_coords[i], r[i]
        for j in range(i+1, n):
            d2 = (xi - x_coords[j])**2 + (yi - y_coords[j])**2
            v = (ri + r[j])**2 - d2
            if v > 0:
                violation += v ** 2
    return -sum_r + 1e8 * violation

def boundary_left(x, idx):
    return x[idx*3] - x[idx*3+2]

def boundary_right(x, idx):
    return 1 - x[idx*3] - x[idx*3+2]

def boundary_bottom(x, idx):
    return x[idx*3+1] - x[idx*3+2]

def boundary_top(x, idx):
    return 1 - x[idx*3+1] - x[idx*3+2]

def pair_non_overlap(x, i, j):
    return ((x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2) - (x[i*3+2] + x[j*3+2])**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 3 * n + [(0, 0.5)] * n
    x0 = np.random.rand(3 * n + n)
    
    result = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False, workers=1)
    best_x = result.x
    
    def refine(x0):
        constraints = [{'type': 'ineq', 'fun': boundary_left, 'args': (i,)} for i in range(n)]
        constraints += [{'type': 'ineq', 'fun': boundary_right, 'args': (i,)} for i in range(n)]
        constraints += [{'type': 'ineq', 'fun': boundary_bottom, 'args': (i,)} for i in range(n)]
        constraints += [{'type': 'ineq', 'fun': boundary_top, 'args': (i,)} for i in range(n)]
        for i in range(n-1):
            for j in range(i+1, n):
                constraints.append({'type': 'ineq', 'fun': pair_non_overlap, 'args': (i, j)})
        res = minimize(lambda y: objective(y) + 1e10 * sum(max(0, constraint(y)) for constraint in constraints), x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'maxiter': 200})
        return res

    refine_result = refine(best_x)
    best_x = refine_result.x
    r = best_x[2::3]
    r = np.maximum(0, r - 1e-9)
    centers = best_x[:2*n].reshape((n, 2))
    return centers, r, np.sum(r)
