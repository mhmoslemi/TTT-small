import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = np.zeros((104 + 325, n))
    b_ub = np.zeros(104 + 325)

    # Side constraints: r_i <= min(c_{i,x}, 1 - c_{i,x}, c_{i,y}, 1 - c_{i,y})
    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])

    # Pairwise distance constraints: r_i + r_j <= \|c_i - c_j\|
    idx = 104
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1

    result = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    return result.x, -result.fun

def de_objective(flat_centres):

def run_packing():
