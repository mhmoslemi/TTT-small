import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, minimize

def run_packing():
    # Define centres
    centres = np.array([
        (0.1, 0.1), (0.3, 0.1), (0.5, 0.1), (0.7, 0.1), (0.9, 0.1),
        (0.13, 0.3), (0.33, 0.3), (0.53, 0.3), (0.73, 0.3), (0.93, 0.3),
        (0.1, 0.5), (0.3, 0.5), (0.5, 0.5), (0.7, 0.5), (0.9, 0.5),
        (0.13, 0.7), (0.33, 0.7), (0.53, 0.7), (0.73, 0.7), (0.93, 0.7),
        (0.1, 0.9), (0.3, 0.9), (0.5, 0.9), (0.7, 0.9), (0.9, 0.9),
        (0.5, 1.0)
    ])

    # Number of circles and centres
    N = len(centres)

    # Distance matrix
    distances = squareform(pdist(centres))
    
    # Define the objective function
    def objective(radii):
        return -np.sum(radii)  # We negate because linprog minimizes
    
    # Define bounds for radii
    bounds = [(0, None) for _ in range(N)]
    
    # Non-overlap constraints matrix and vector
    A_non_overlap = []
    b_non_overlap = []
    for i in range(N):
        for j in range(i + 1, N):
            x1, y1 = centres[i]
            x2, y2 = centres[j]
            d = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            A_non_overlap.append([0] * i + [1] + [0] * (j - i - 1) + [1] + [-(d - 1e-12)])
            b_non_overlap.append(0)
    
    # Boundary constraints matrix and vector
    A_boundary = []
    b_boundary = []
    for i in range(N):
        x, y = centres[i]
        A_boundary.append([-1, 0])
        b_boundary.append(-x)
        A_boundary.append([1, 0])
        b_boundary.append(-(1 - x))
        A_boundary.append([0, -1])
        b_boundary.append(-y)
        A_boundary.append([0, 1])
        b_boundary.append(-(1 - y))
    
    # Combine all constraints
    A = np.vstack((np.eye(N), A_non_overlap, A_boundary))
    b = np.hstack((np.zeros(N), b_non_overlap, b_boundary))
    
    # Initial guess for radii (all circles of the same small radius)
    initial_guess = np.ones(N) * 0.1
    
    # Solve the linear problem
    result = linprog(c=objective(initial_guess), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    # Extract the solution
    radii = -result.x
    
    # Trim the radii to stay within boundaries
    radii = np.maximum(0, np.minimum(radii, np.minimum(np.minimum(centres[:, 0], 1 - centres[:, 0]), np.minimum(np.minimum(centres[:, 1], 1 - centres[:, 1]))))) - 1e-9
    
    # Compute the sum of the radii
    sum_radii = np.sum(radii)
    
    return centres, radii, sum_radii
