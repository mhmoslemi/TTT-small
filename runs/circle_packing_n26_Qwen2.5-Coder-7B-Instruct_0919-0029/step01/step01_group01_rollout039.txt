import numpy as np
from scipy.optimize import linprog, minimize

# Function to compute the sum of radii for a given center set
def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Define the inequality constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), -result.fun

# Function to perform simulated annealing
def simulated_annealing():
    n = 26
    # Start with a near-hexagonal packing
    centers = np.zeros((n, 2))
    for i in range(n):
        theta = np.pi * (3 + np.sqrt(5)) * i
        centers[i, 0] = 0.5 + 0.9 * np.cos(theta)
        centers[i, 1] = 0.5 + 0.9 * np.sin(theta)
    centers = centers + 0.1 * np.random.normal(size=centers.shape)
    centers = np.clip(centers, 0, 1)

    T0, Tf = 0.1, 1e-4
    alpha = 0.9995
    delta = 0.1
    steps_per_T = 500
    max_iter = 10000

    T = T0
    current_radii, current_sum = lp_optimum(centers)
    best_centers, best_sum, best_radii = centers, current_sum, current_radii

    for _ in range(max_iter):
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            delta_c = delta * np.random.randn(2)
            new_c = centers[i] + delta_c
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[i] = new_c
            new_radii, new_sum = lp_optimum(new_centers)

            if new_sum >= current_sum or np.exp((new_sum - current_sum) / T) > np.random.rand():
                centers = new_centers
                current_radii, current_sum = new_radii, new_sum

                if new_sum > best_sum:
                    best_centers, best_sum, best_radii = centers, new_sum, new_radii

            T *= alpha

        # Final L-BFGS polish on the best configuration so far
        result = minimize(lambda x: -np.sum(x[:n]), best_centers.flatten(), bounds=[(0, None)] * n, method='L-BFGS-B')
        best_centers = result.x.reshape((n, 2))
        best_radii, best_sum = lp_optimum(best_centers)

    return best_centers, best_radii, best_sum

def run_packing():
    return simulated_annealing()
