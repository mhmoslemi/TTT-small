from scipy.spatial.distance import cdist
from scipy.spatial import Voronoi

def voronoi_center(radii, points):
    vor = Voronoi(points)
    num_vertices = len(vor.vertices)
    num_circles = len(radii)
    centers = np.copy(vor.vertices)

    return centers

def initial_placement(num_circles):

    fractions = []
    for i in range(1, num_circles + 1):
        f = 0.0
        base = 2
        while i > 0:
            f /= base
            i, remainder = divmod(i, base)
            f += remainder / base
        fractions.append(f)
    coords = np.array(fractions)
    x = coords * 0.5 + 0.25
    y = coords * 0.5 + 0.25
    return np.vstack((x, y)).T

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    max_radius = 1 / np.sqrt(2 * num_circles)
    initial_points = initial_placement(num_circles)
    radii = np.ones(num_circles) * max_radius
    centers = initial_points
