        bound_penalties = np.maximum(0, radii[:, None] - centers[:, None, 0]) ** 2
        bound_penalties += np.maximum(0, radii[:, None] - (1 - centers[:, None, 0])) ** 2
        bound_penalties += np.maximum(0, radii[:, None] - centers[:, None, 1]) ** 2
        bound_penalties += np.maximum(0, radii[:, None] - (1 - centers[:, None, 1])) ** 2
        overlap_penalties = np.maximum(0, radii[:, None] + radii[None, :] - dist) ** 2
        overlap_penalties = np.triu(overlap_penalties, k=1)
        return -(np.sum(radii) - 2000 * (bound_penalties.sum() + overlap_penalties.sum()))
    
    def penalize(out_of_bounds, overlaps, bound_penalty_weight=2000):
        out_penalties = np.multiply(out_of_bounds, bound_penalty_weight)
        overlap_penalties = np.multiply(overlaps, 2000)
        return out_penalties.sum() + overlap_penalties.sum()

        r = p[2::3]
        x = p[0::3]
        y = p[1::3]
        return r - x, r - (1 - x), r - y, r - (1 - y)

        x = p[0::3]
        y = p[1::3]
