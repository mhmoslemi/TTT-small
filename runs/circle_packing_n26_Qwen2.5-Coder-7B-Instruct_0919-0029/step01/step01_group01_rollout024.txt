import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = np.zeros((4*n + 325, n))
    b_ub = np.zeros(4*n + 325)
    
    # Side constraints: ri <= ci,x, ri <= 1-ci,x, ri <= ci,y, ri <= 1-ci,y
    for i in range(n):
        A_ub[4*i, i] = 1
        A_ub[4*i + 1, i] = -1
        A_ub[4*i + 2, i] = 1
        A_ub[4*i + 3, i] = -1
        b_ub[4*i] = centres[i, 0] - 1e-12
        b_ub[4*i + 1] = 1 - centres[i, 0] - 1e-12
        b_ub[4*i + 2] = centres[i, 1] - 1e-12
        b_ub[4*i + 3] = 1 - centres[i, 1] - 1e-12

    # Pairwise constraints: ri + rj <= ||ci - cj||
    k = 4*n
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centres[i]-centres[j])
            A_ub[k, i] = 1
            A_ub[k, j] = 1
            b_ub[k] = dist - 1e-12
            k += 1

    result = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not result.success:
        raise ValueError("LP optimization failed")
    return result.x, -result.fun

def de_objective(flat_centres):
    centres = flat_centres.reshape(26,2)
    _, s = solve_radius_lp(centres)
    return -s    # DE minimises

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0,1.0)] * 52
    result = differential_evolution(
                de_objective,
                bounds,
                maxiter=60,
                popsize=12,
                tol=1e-6,
                polish=True,
                seed=42)
    if not result.success:
        raise ValueError("DE optimisation failed")
    centres = result.x.reshape(26, 2)
    radii, sum_radii = solve_radius_lp(centres)
    return centres, radii, sum_radii
