import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing():
    n_circles = 26
    numstarts = 20
    steps_per_T = 500
    max_iter = 10 * steps_per_T * numstarts
    T0 = 0.1
    Tf = 1e-4
    alpha = 0.9995
    delta = 0.1

    def objective(x):
        centers = x[:2 * n_circles].reshape((n_circles, 2))
        radii, sum_radii = lp_optimum(centers)
        return -sum_radii

    def objective_gradient(x):
        centers = x[:2 * n_circles].reshape((n_circles, 2))
        radii, sum_radii = lp_optimum(centers)

        # Precompute distances
        dists = cdist(centers, centers, 'euclidean')

        # Compute row sums of laplacian matrix
        laplacian = np.zeros((n_circles, n_circles))
        for i in range(n_circles):
            laplacian[i] = -np.sum(dists[i] / (radii[i] + radii!) - 1 / radii[i]) + np.sum(dists[:, i] / (radii[:, i] + radii[i]) - 1)
        
        return -np.dot(laplacian, radii)

    best_centers = None
    best_sum = -np.inf
    best_radii = None

    for _ in range(numstarts):
        centers = np.random.rand(n_circles, 2)

        # Convert centers to parameter vector
        x0 = centers.flatten()

        # Run SA
        T = T0
        for _ in range(max_iter):
            idx = np.random.randint(n_circles)
            delta_vec = np.random.randn(2) * delta
            new_c = np.clip(centers[idx] + delta_vec, 0, 1)
            new_x = x0.copy()
            new_x[2 * idx: 2 * idx + 2] = new_c
            new_centers = new_x.reshape((n_circles, 2))
            new_radii, new_sum_radii = lp_optimum(new_centers)
            if new_sum_radii > best_sum:
                best_sum = new_sum_radii
                best_centers = new_x
                best_radii = new_radii
            else:
                acceptance_prob = 0 if new_sum_radii < best_sum else np.exp((best_sum - new_sum_radii) / T)
                if np.random.rand() < acceptance_prob:
                    best_sum = new_sum_radii
                    best_centers = new_x
                    best_radii = new_radii
            T = T * alpha

    # Final L-BFGS polish
    if best_centers is not None:
        final_result = minimize(objective, best_centers.flatten(), method='L-BFGS-B', jac=objective_gradient, options={'maxiter': steps_per_T})
        best_centers = final_result.x.reshape((n_circles, 2))
        radii, sum_radii = lp_optimum(best_centers)
        best_sum = sum_radii

    return best_centers, radii, best_sum
