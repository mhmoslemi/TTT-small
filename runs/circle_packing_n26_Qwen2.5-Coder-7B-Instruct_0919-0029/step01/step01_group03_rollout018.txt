import numpy as np
from scipy.optimize import minimize

def fun(z):
    N = 26
    x = z[:N]
    y = z[N:2*N]
    r = z[2*N:]
    
    phi_bound = np.sum(np.maximum(0, r - x)**2 + np.maximum(0, r - (1 - x))**2 + np.maximum(0, r - y)**2 + np.maximum(0, r - (1 - y))**2)
    
    diff = x[:,None] - x[None,:]
    dist = np.sqrt((diff**2 + (y[:,None] - y[None,:])**2))
    
    phi_ovlp = np.sum(np.maximum(0, r[:,None] + r[None,:] - dist)**2)
    
    return -(np.sum(r) - 2000 * (phi_bound + phi_ovlp))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    bounds = [(0, 1), ] * N + [(0, 1), ] * N + [(0, 0.4), ] * N
    
    np.random.seed(42)
    W = 2000
    results = []
    
    for _ in range(20):
        centers_seed = np.array([
            (0.1, 0.1),
            (0.1, 0.9),
            (0.9, 0.1),
            (0.9, 0.9),
            *np.random.uniform(0.25, 0.75, (4, 2)),
            *np.random.uniform(0.25, 0.75, (22, 2))
        ])
        
        radii_seed = np.random.uniform(0.05, 0.15, (26,))
        
        z0 = np.hstack((centers_seed.flatten(), radii_seed))
        
        res = minimize(fun, z0, method='L-BFGS-B', bounds=bounds, options={'maxfun': 1000, 'ftol': 1e-9, 'gtol': 1e-5})
        if res.success:
            results.append(res.x)
    
    best_z = max(results, key=lambda z: np.sum(z[2*N:]))
    x = best_z[:N]
    y = best_z[N:2*N]
    
    diff = x[:,None] - x[None,:]
    dist = np.sqrt((diff**2 + (y[:,None] - y[None,:])**2))
    
    phi_bound = np.sum(np.maximum(0, best_z[2*N:] - x)**2 + np.maximum(0, best_z[2*N:] - (1 - x))**2 + np.maximum(0, best_z[2*N:] - y)**2 + np.maximum(0, best_z[2*N:] - (1 - y))**2)
    
    phi_ovlp = np.sum(np.maximum(0, best_z[2*N:][:,None] + best_z[2*N:][None,:] - dist)**2)
    
    if phi_bound + phi_ovlp > 1e-4:
        W *= 2
        res = minimize(fun, best_z, method='L-BFGS-B', bounds=bounds, options={'maxfun': 1000, 'ftol': 1e-9, 'gtol': 1e-5})
        if res.success:
            best_z = res.x
    
    x = best_z[:N]
    y = best_z[N:2*N]
    r = best_z[2*N:]
    
    for _ in range(5):
        if fun(best_z) > 0:
            W *= 2
            res = minimize(fun, best_z, method='L-BFGS-B', bounds=bounds, options={'maxfun': 1000, 'ftol': 1e-9, 'gtol': 1e-5})
            if res.success:
                best_z = res.x
    
    return x.reshape(-1,2), r, np.sum(r)
