import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

bounds = [(0, 1)] * (3 * N) + [(0, 0.5)] * N

def sphere_packing_obj(x):
    n = 26
    arr = x.reshape((n, 3))
    x_coords = arr[:, 0]
    y_coords = arr[:, 1]
    r = arr[:, 2]
    sum_r = np.sum(r)
    violation = 0.0
    for i in range(n):
        violation += np.maximum(0.0, r[i] - x_coords[i]) ** 2
        violation += np.maximum(0.0, r[i] - (1 - x_coords[i])) ** 2
        violation += np.maximum(0.0, r[i] - y_coords[i]) ** 2
        violation += np.maximum(0.0, r[i] - (1 - y_coords[i])) ** 2
    for i in range(n - 1):
        xi, yi, ri = x_coords[i], y_coords[i], r[i]
        for j in range(i + 1, n):
            d2 = (xi - x_coords[j]) ** 2 + (yi - y_coords[j]) ** 2
            v = (ri + r[j]) ** 2 - d2
            if v > 0:
                violation += v ** 2
    return -sum_r + 1e10 * violation

def boundary_left(x, idx):
    return x[idx * 3] - x[idx * 3 + 2]

def boundary_right(x, idx):
    return 1 - x[idx * 3] - x[idx * 3 + 2]

def boundary_bottom(x, idx):
    return x[idx * 3 + 1] - x[idx * 3 + 2]

def boundary_top(x, idx):
    return 1 - x[idx * 3 + 1] - x[idx * 3 + 2]

def pair_non_overlap(x, i, j):
    return (x[i * 3] - x[j * 3]) ** 2 + (x[i * 3 + 1] - x[j * 3 + 1]) ** 2 - (x[i * 3 + 2] + x[j * 3 + 2]) ** 2

constraints = []
for i in range(N):
    constraints.append({'type': 'ineq', 'fun': boundary_left, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': boundary_right, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': boundary_bottom, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': boundary_top, 'args': (i,)})

for i in range(N - 1):
    for j in range(i + 1, N):
        constraints.append({'type': 'ineq', 'fun': pair_non_overlap, 'args': (i, j)})
