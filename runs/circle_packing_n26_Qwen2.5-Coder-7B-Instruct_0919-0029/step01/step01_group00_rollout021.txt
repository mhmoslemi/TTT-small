import numpy as np
from scipy.spatial import voronoi
from scipy.optimize import linprog

def calculate_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < 2 * radii[i] + 2 * radii[j]:
                return None
            radii[i] = max(radii[i], (dist - radii[j]) / 2)
            radii[j] = max(radii[j], (dist - radii[i]) / 2)
        radii[i] = max(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    return radii

def lp_optimize(centers, radii):
    n = centers.shape[0]
    A_ub = np.zeros((2*n*(n-1), n))
    b_ub = np.zeros(2*n*(n-1))

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 2 * (radii[i] + radii[j])
            A_ub[idx, j] = 2 * (radii[i] + radii[j])
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 2 * (radii[i] + 0.01)
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 2 * (radii[i] + 0.01)
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 2 * (radii[i] + 0.01)
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 2 * (radii[i] + 0.01)
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def voronoi_packing():
    n = 26
    step, step_min = 0.05, 0.001
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centres = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c

            new_radii = calculate_radii(new_centres)
            if new_radii is not None:
                new_sum = np.sum(new_radii)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centres
                    best_radii = new_radii

        step = max(step_min, step * decay_factor)

    return best_centers, best_radii, np.sum(best_radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    return voronoi_packing()
