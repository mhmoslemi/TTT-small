import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_points = np.array([(i/100.0, j/100.0) for i in range(101) for j in range(101)])
    n = 26
    radii = np.zeros(n)
    centres = np.zeros((n, 2))
    
    def growth_rate(point, centres, radii):
        distances = np.sqrt(np.sum((point - centres) ** 2, axis=1))
        distances[distances < 1e-12] = np.inf  # Avoid division by zero
        return np.min([np.min([point[0], 1-point[0], point[1], 1-point[1]]), np.min(distances - radii)])
    
    while len(centres) < n:
        growth_times = np.array([growth_rate(point, centres, radii) for point in grid_points if point not in centres])
        if np.all(growth_times == np.inf):
            break
        min_growth_time = np.min(growth_times)
        new_point = grid_points[np.argmin(growth_times)]
        new_radius = min_growth_time
        centres = np.vstack((centres, new_point))
        radii = np.hstack((radii, new_radius))
    
    def relaxation(centres, radii):
        alpha = 0.05
        for _ in range(200):
            pressures = np.zeros_like(centres)
            for i in range(n):
                pressures[i, 0] += np.sum(np.maximum(0, (1-centres[i, 0]) - (centres[:, 0] - centres[i, 0]) - (radii + radii[:, None])).clip(min=0))
                pressures[i, 0] -= np.sum(np.maximum(0, (centres[:, 0] - centres[i, 0]) - (1-centres[i, 0]) - (radii[:, None] + radii[i])).clip(min=0))
                pressures[i, 0] += np.sum(np.maximum(0, ( centres[i, 0] - (centres[:, 0] - centres[i, 0]) - (radii + radii[:, None])).clip(min=0)))
                pressures[i, 1] += np.sum(np.maximum(0, (1-centres[i, 1]) - (centres[:, 1] - centres[i, 1]) - (radii + radii[:, None])).clip(min=0))
                pressures[i, 1] -= np.sum(np.maximum(0, (centres[:, 1] - centres[i, 1]) - (1-centres[i, 1]) - (radii[:, None] + radii[i])).clip(min=0))
                pressures[i, 1] += np.sum(np.maximum(0, ( centres[i, 1] - (centres[:, 1] - centres[i, 1]) - (radii + radii[:, None])).clip(min=0)))
                pressures[i, 0] = max(pressures[i, 0], 0)
                pressures[i, 0] = min(pressures[i, 0], 2*(centres[i, 0] - 0.01))
                pressures[i, 1] = max(pressures[i, 1], 0)
                pressures[i, 1] = min(pressures[i, 1], 2*(centres[i, 1] - 0.01))
                centres[i] += alpha * pressures[i] / np.linalg.norm(pressures[i])
        return centres, radii
    
    centres, radii = relaxation(centres, radii)
    radii -= 1e-9
    
    return centres, radii, np.sum(radii)
