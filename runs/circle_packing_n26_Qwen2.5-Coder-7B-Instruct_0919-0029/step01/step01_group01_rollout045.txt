import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append([0] * n)
        A_ub[-1][i] = 1
        b_ub.append(centres[i, 0])
        A_ub.append([0] * n)
        A_ub[-1][i] = 1
        b_ub.append(1 - centres[i, 0])
        A_ub.append([0] * n)
        A_ub[-1][i] = 1
        b_ub.append(centres[i, 1])
        A_ub.append([0] * n)
        A_ub[-1][i] = 1
        b_ub.append(1 - centres[i, 1])

    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    A_ub.extend(dists.flatten())
    b_ub.extend(np.ones(n*(n-1)//2))

    A_ub = np.array(A_ub).T
    b_ub = np.array(b_ub)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def de_objective(flat_centres):
    centres = flat_centres.reshape(26, 2)
    _, s = solve_radius_lp(centres)
    return -s

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * 52
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=120,
        popsize=12,
        tol=1e-6,
        polish=True,
        seed=42
    )

    best_centres = result.x.reshape(26, 2)
    final_radii, final_sum = solve_radius_lp(best_centres)

    return best_centres, final_radii, final_sum
