import numpy as np
import pulp

def build_grid(step_size):
    nx = round(1 / step_size)
    ny = round(1 / step_size)
    x = np.arange(nx) * step_size
    y = np.arange(ny) * step_size
    X, Y = np.meshgrid(x, y)
    return X.flatten(), Y.flatten(), nx, ny

def calculate_distances(X, Y):
    n = X.shape[0]
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((X[i] - X[j])**2 + (Y[i] - Y[j])**2)
            D[i, j] = dist
            D[j, i] = dist
    return D

def run_packing():
    X, Y, nx, ny = build_grid(0.075)
    P = X.shape[0]

    D = calculate_distances(X, Y)
    m = pulp.LpProblem("Circle_Packing", pulp.LpMaximize)

    y = [pulp.LpVariable(f"y_{k}", cat='Binary') for k in range(P)]
    r = [pulp.LpVariable(f"r_{k}", lowBound=0, upBound=0.5) for k in range(P)]

    m += pulp.lpSum(r[k] for k in range(P)), "Objective"

    m += pulp.lpSum(y[k] for k in range(P)) == 26, "Exactly_26"

    for k in range(P):
        for l in range(k + 1, P):
            m += r[k] + r[l] <= D[k, l] + 0.5 * (1 - y[k]) + 0.5 * (1 - y[l]), f"No_overlap_{k}_{l}"

    for k in range(P):
        m += r[k] <= X[k], f"Boundary_containment_left_{k}"
        m += r[k] <= 1 - X[k], f"Boundary_containment_right_{k}"
        m += r[k] <= Y[k], f"Boundary_containment_bottom_{k}"
        m += r[k] <= 1 - Y[k], f"Boundary_containment_top_{k}"
        m += r[k] <= 0.5 * y[k], f"Nonzero_radius_when_selected_{k}"

    m.solve(solver=pulp.PULP_CBC_CMD(msg=0, timeLimit=30))

    centers = []
    radii = []
    for k in range(P):
        if y[k].varValue >= 0.5:
            centers.append((X[k].varValue, Y[k].varValue))
            radii.append(r[k].varValue)

    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
