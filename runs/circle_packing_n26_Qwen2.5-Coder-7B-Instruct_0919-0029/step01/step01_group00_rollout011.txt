import numpy as np
from scipy.optimize import highs
from itertools import combinations
from scipy.spatial import distance_matrix

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = distance_matrix(centers[[i]], centers[[j]])[0] - 1e-12
            idx += 1

    for i in range(n):
        for j in range(2):
            A_ub[idx, i] = 1
            b_ub[idx] = centers[i, j] - 1e-12
            idx += 1
            A_ub[idx, i] = 1
            b_ub[idx] = 1 - centers[i, j] - 1e-12
            idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = highs(highs.formulate_highs(c, A_ub, b_ub, "maximize"))
    return result["x"], -result["opt_val"]
