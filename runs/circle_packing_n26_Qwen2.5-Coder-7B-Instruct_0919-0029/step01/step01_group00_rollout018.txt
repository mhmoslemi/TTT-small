import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import nnls, minimize

def lp_optimum(centers):
    n = centers.shape[0]
    A = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    A = np.sum(A**2, axis=-1)
    A[np.diag_indices(n)] = 1e10
    A /= 2
    b = np.zeros(n)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = minimize(lambda r: c @ r + np.sum(np.linalg.norm(centers - centers[r==r.min()].reshape(-1, 2), axis=1) - A[r==r.min()][:, r==r.min()].flatten()), np.ones(n)/n, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda r: A[r==r.min()][:, r==r.min()].flatten()}}
    return result.x, -result.fun

def generate_lattice(n_points):
    r_est = np.sqrt(0.9069 / (np.pi * n_points))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    candidates = []
    for k in range(int(1 / h) + 1):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        for n in range(int(1 / s) + 1):
            x = r_est + n * s
            if x > 1 - r_est:
                break
            candidates.append([x, y])
            if (k + n) % 2 == 1:
                candidates.append([x + r_est, y])
    return np.array(candidates)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = generate_lattice(32)
    subset_indices = list(np.random.permutation(len(centers)))
    best_sum = -np.inf
    best_subset_indices = None

    for start in range(10):
        np.random.shuffle(subset_indices)
        for i in range(len(subset_indices)):
            subset = subset_indices[:i+1]
            if len(subset) == 26:
                radii, sum_r = lp_optimum(centers[subset])
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_subset_indices = subset
                    break
            else:
                continue

    refined_centers = centers[best_subset_indices]
    refined_radii, refined_sum = lp_optimum(refined_centers)
    return refined_centers, refined_radii, refined_sum
