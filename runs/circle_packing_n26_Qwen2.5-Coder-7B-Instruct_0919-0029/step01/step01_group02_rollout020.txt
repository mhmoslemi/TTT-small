import numpy as np
import cvxpy as cp
import scipy.optimize as opt

def solve_lp(x, y):
    N = len(x)
    c = np.ones(N)
    A = np.vstack([
        -np.eye(N),
        np.ones(N) + np.eye(N),
        x - np.ones(N),
        -x,
        y - np.ones(N),
        -y
    ])
    b = np.hstack([np.zeros(N), 2*np.sqrt(2)*np.ones(N), x, -x, y, -y])
    r = cp.Variable(N)
    problem = cp.Problem(cp.Maximize(c @ r), [A @ r <= b, r >= 0])
    problem.solve()
    return r.value
