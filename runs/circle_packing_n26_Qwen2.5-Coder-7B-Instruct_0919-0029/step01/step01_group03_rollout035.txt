import numpy as np
from scipy.optimize import linprog
from typing import Tuple

def linspace_with_center(base: float, offset: float, num: int) -> np.ndarray:
    return np.linspace(base + offset, 1 - offset, num=num)

def hex_lattice(y_min: float, y_max: float, x_min: float, x_max: float, num_rows: int, num_cols: int) -> np.ndarray:
    row_offset = (x_max - x_min) / (num_cols * np.sqrt(3))
    x = np.array([linspace_with_center(x_min, row_offset * i % (x_max - x_min) / 4, num_cols) for i in range(num_rows)])
    y = np.array([np.linspace(y_min, y_max, num=num_cols) for _ in range(num_rows)])
    x = x.flatten()[:, None]
    y = y.flatten()[:, None]
    return np.hstack([x, y])

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Define initial layout
    centers = np.zeros((26, 2))
    
    # Corner points
    centers[0] = np.array([0.125, 0.125])  # top-left
    centers[1] = np.array([0.875, 0.125])  # top-right
    centers[2] = np.array([0.125, 0.875])  # bottom-left
    centers[3] = np.array([0.875, 0.875])  # bottom-right
    
    # Face centers
    centers[4:8] = hex_lattice(0.125, 0.875, 0.125, 0.875, 3, 2)
    centers[8:12] = hex_lattice(0.125, 0.875, 0.875, 0.125, 2, 3)
    centers[12:16] = hex_lattice(0.875, 0.125, 0.125, 0.875, 2, 3)
    centers[16:20] = hex_lattice(0.875, 0.125, 0.875, 0.125, 3, 2)
    
    # Hex centers
    centers[20:26] = hex_lattice(0.3, 0.7, 0.3, 0.7, 2, 2)  # March-shaped centers in the center of the square
    
    num_iter = 10
    best_centers = None
    best_sum = -np.inf
    
    for _ in range(30):  # Multiple random restarts
        for _ in range(num_iter):  # Local search per restart
            jitter = np.random normal(loc=0, scale=0.03, size=(26, 2))
            jittered_centers = np.clip(centers + jitter, 0, 1)
            
            radii = linear_program(jittered_centers)[0]
            if np.any(radii > 0):
                current_sum = np.sum(radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = jittered_centers
    
    return best_centers, radii, best_sum

def linear_program(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    n = len(centers)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    
    A_ub = np.zeros((n * n + n * 8, n))
    b_ub = np.zeros(n * n + n * 8)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * n + j * 2 - 2, i] = 1
            A_ub[i * n + j * 2 - 2, j] = 1
            b_ub[i * n + j * 2 - 2] = dist[i, j]
            
            A_ub[i * n + j * 2 - 1, i] = -1
            A_ub[i * n + j * 2 - 1, j] = -1
            b_ub[i * n + j * 2 - 1] = 0
    
    A_ub[26 * n:, :n] = np.eye(n)
    b_ub[26 * n:] = np.ones(n)
    
    A_ub[26 * n + n:, :n] = -np.eye(n)
    b_ub[26 * n + n:] = np.zeros(n)
    
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs', options={'display': False})
    
    if res.success:
        return res.x, res.fun
    else:
        return np.zeros(n), 0.0
