import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i in range(n):
        for j in range(i+1, n):
            A_ub[i*(n-1)//2 + j - i - 1, i] = 1
            A_ub[i*(n-1)//2 + j - i - 1, j] = 1
            b_ub[i*(n-1)//2 + j - i - 1] = np.linalg.norm(centers[i] - centers[j]) - 1e-12

    for i in range(n):
        A_ub[n*(n-1)//2 + 4*i, i] = 1
        b_ub[n*(n-1)//2 + 4*i] = centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 1, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 1] = 1 - centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 2, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 2] = centers[i, 1] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 3, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 3] = 1 - centers[i, 1] - 1e-12

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def evaluate(centres_flat):
    centres = centres_flat.reshape((26, 2))
    radii, total = lp_optimum(centres)
    if np.any(radii < 0):
        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))
    return -total

def refine(centres):
    n = centres.shape[0]
    def local_objective(x):
        x = x.reshape((n, 2))
        return -evaluate(x.flatten())

    bounds = [(0, 1)] * n * 2
    initial = centres.flatten()
    result = minimize(local_objective, initial, method="Nelder-Mead", bounds=bounds)
    return result.x.reshape((n, 2))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    points = []
    for k in range(6):
        for i in range(3 if k % 2 == 0 else 4):
            x = (i + (k % 2) / 2) * s + s
            y = h * k
            if x >= r_est and x <= 1 - r_est:
                points.append([x, y])
            y += h / 2
            if x >= r_est and x <= 1 - r_est:
                points.append([x, y])
    centres = np.array(points[:26])
    population = np.tile(centres, (50, 1))
    np.random.shuffle(population)

    result = differential_evolution(evaluate, bounds, args=(), 
                                   popsize=50, strategy="best1bin", 
                                   mutation=0.5, crossover=0.7, 
                                   max_iter=80, tol=1e-6, popmut=0.7, 
                                   polish=True, init="random")
    centres_best = result.x.reshape((26, 2))

    centres_best = refine(centres_best)
    radii, total = lp_optimum(centres_best)
    return centres_best, radii, total

# Run the function
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of radii: {sum_radii}")
