```python
import numpy as np
from scipy.optimize import differential_evolution, minimize


def penalty(X):
    N = len(X) // 3
    penalty_value = 0.0
    for i in range(N):
        x, y, r = X[3*i], X[3*i + 1], X[3*i + 2]
        penalty_value += max(0, r - x)**2 + max(0, r - (1 - x))**2
        penalty_value += max(0, r - y)**2 + max(0, r - (1 - y))**2
        for j in range(i + 1, N):
            dx, dy = np.abs(X[3*i] - X[3*j]), np.abs(X[3*i + 1] - X[3*j + 1])
            penalty_value += max(0, (r + X[3*j + 2])**2 - dx**2 - dy**2)**2
    return penalty_value


def objective(X):
    N = len(X) // 3
    radii = X[2::3]
    return -(radii.sum()) + 1e6 * penalty(X)


bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26


def g_left(X, i):
    return X[3*i] - X[3*i + 2]


def g_right(X, i):
    return 1 - X[3*i] - X[3*i + 2]


def g_bot(X, i):
    return X[3*i + 1] - X[3*i + 2]


def g_top(X, i):
    return 1 - X[3*i + 1] - X[3*i + 2]


def g_pair(X, i, j):
    dx, dy = np.abs(X[3*i] - X[3*j]), np.abs(X[3*i + 1] - X[3*j + 1])
    return (X[3*i + 2] + X[3*j + 2])**2 - dx**2 - dy**2


constraints = []
for i in range(26):
    constraints.append({'type': 'ineq', 'fun': g_left, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': g_right, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': g_bot, 'args': (i,)})
    constraints.append({'type': 'ineq', 'fun': g_top, 'args': (i,)})
for i in range(25):
    for j in range(i + 1, 26):
        constraints.append({'type': 'ineq', 'fun': g_pair, 'args': (i, j)})


def run_packing():
    X_global = differential_evolution(objective, bounds=bounds, maxiter=200,
                                     popsize=20, polish=False, workers=1)
    X_refined = minimize(objective, X_global.x, bounds=bounds,
                        constraints=constraints, method='SLSQP',
                        options={'eps': 1e-7, 'maxiter': 200})
    
    X_refined = X_refined['x']
    centers = X_refined.reshape((26, 3))[:, :2]
    radii = X_refined.reshape((26, 3))[:, 2]
    radii = np.clip(radii, 0.0, None)
    radii[:10] -= 1e-9
    sum_radii = radii.sum()
    assert validate_packing(centers, radii)[0]
    return centers, radii, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

centers, radii, sum_radii = run_packing()
print((centers, radii, sum_radii))
```
