import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    def lp_radii(centers):
        n = centers.shape[0]
        # pairwise distances
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.linalg.norm(diff, axis=2)
        
        A_ub = []
        b_ub = []
        
        # boundary constraints
        for i in range(n):
            x, y = centers[i]
            A_ub.append([0]*i + [1] + [0]*(n-i-1))
            b_ub.append(x)
            A_ub.append([0]*i + [1] + [0]*(n-i-1))
            b_ub.append(1-x)
            A_ub.append([0]*i + [1] + [0]*(n-i-1))
            b_ub.append(y)
            A_ub.append([0]*i + [1] + [0]*(n-i-1))
            b_ub.append(1-y)
        
        # pair constraints
        for i in range(n):
            for j in range(i+1, n):
                A_ub.append([-1 if k==i or k==j else 0 for k in range(n)])
                b_ub.append(0)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = -np.ones(n)
        bounds = [(0, None)]*n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            raise ValueError("LP failed to find a feasible solution")
        
        return res.x, res.fun
    
    #-initial layout
    centers = []
    center_radius = 0.12
    for row in range(4):
        for col in range(4):
            centers.append([col * 0.2 + 0.1, row * 0.2 + 0.1])
    for row in range(4):
        centers.append([0.1, row * 0.25 + 0.75])
        centers.append([1 - 0.1, row * 0.25 + 0.75])
        centers.append([col * 0.2 + 0.1, 0.1])
        centers.append([col * 0.2 + 0.1, 1 - 0.1])
    for i in range(14):
        while True:
            candidate_center = np.random.uniform(0.2, 0.8, 2)
            valid = True
            for center in centers:
                if np.linalg.norm(candidate_center - center) < 0.1:
                    valid = False
                    break
            if valid:
                centers.append(candidate_center)
                break
    
    best_centers = np.array(centers)
    best_sum = -np.inf
    
    for _ in range(200):
        # Perturb centers
        jittered_centers = best_centers + np.random.normal(0, 0.03, size=best_centers.shape)
        jittered_centers = np.clip(jittered_centers, 0, 1)
        
        # Local gradient step (optional)
        radii, objective = lp_radii(jittered_centers)
        if objective > best_sum:
            best_centers, best_sum = jittered_centers, objective
    
    return best_centers, radii, best_sum
