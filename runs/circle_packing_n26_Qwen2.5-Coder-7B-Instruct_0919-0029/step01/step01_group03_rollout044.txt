import numpy as np
from scipy.optimize import linprog

# Precompute distances between pairs of centers
def compute_lp(centers):
    n = len(centers)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i+1, n):
            A_ub.append([1 if k==i or k==j else 0 for k in range(n)])
            b_ub.append(dist[i,j])
    for i in range(n):
        x, y = centers[i]
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(x)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1-x)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(y)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1-y)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    radii = res.x
    return radii

# Calculate the sum of radii and validate the packing
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    num_restart = 100
    num_jitter = 10
    best_sum = -np.inf
    best_centers = None
    
    for _ in range(num_restart):
        if _ == 0:
            # Initial layout from hexagonal lattice near centre
            x = np.array([0.4, 0.4, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.4, 0.45, 0.5, 0.55,
                         0.6, 0.65, 0.8, 0.8, 0.8, 0.85, 0.85, 0.55, 0.5, 0.45, 0.95, 0.9, 0.85])
            y = np.array([0.4, 0.55, 0.7, 0.7, 0.7, 0.7, 0.7, 0.7, 0.5, 0.5, 0.5, 0.45,
                         0.45, 0.45, 0.8, 0.85, 0.85, 0.9, 0.9, 0.85, 0.8, 0.75, 0.8, 0.8, 0.85])
            centers_0 = np.vstack((x, y)).T
        else:
            centers_0 = best_centers + 0.1 * np.random.randn(26, 2)
            centers_0 = np.clip(centers_0, 0.05, 0.95)
            
        for _ in range(num_jitter):
            centers = centers_0 + 0.1 * np.random.randn(26, 2)
            centers = np.clip(centers, 0.05, 0.95)
            radii = compute_lp(centers)
            if np.any(radii < 0):
                continue
            sum_radii = np.sum(radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers
                
    if validate_packing(best_centers, radii) != (True, "ok"):
        raise ValueError("Invalid packing produced")
                
    return best_centers, radii, best_sum
