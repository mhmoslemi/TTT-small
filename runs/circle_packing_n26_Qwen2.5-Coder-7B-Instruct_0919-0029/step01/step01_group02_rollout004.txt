import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    g = 0.01
    grid_points = [(i*g, j*g) for i in range(101) for j in range(101)]
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    active = np.zeros(26, dtype=bool)

    for i in range(26):
        while True:
            t_min = 1000.0
            min_idx = -1
            for idx, (x, y) in enumerate(grid_points):
                if active[idx] or sum((centers[j] - (x, y)) ** 2 for j in range(26) if j != i).sum() ** 0.5 - radii[j] < 0.0001:
                    continue
                dist_to_bound = min(x, 1-x, y, 1-y, min(dist - r - 0.0001 for dist, r, active in zip([x, 1-x, y, 1-y], radii, active)))
                radii_option = min(dist_to_bound, min(dist - r - 0.0001 for dist, r, active in zip([(x, y) - centers[j] for j in range(26) if j != i], radii, active)))
                if radii_option < t_min:
                    t_min = radii_option
                    min_idx = idx
            if min_idx == -1:
                break
            centers[i] = grid_points[min_idx]
            radii[i] += t_min
            active[i] = True

    for iteration in range(200):
        net_pressures = np.zeros((26, 2))
        for i in range(26):
            r_i = radii[i]
            for j in range(26):
                if i == j:
                    continue
                r_j = radii[j]
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < r_i + r_j:
                    net_pressures[i] += ( centers[i] - centers[j] ) * ( r_i + r_j - dist + 0.0001 )
            net_pressures[i] += ( centers[i] - np.array(centers[i] >= 0) * centers[i] ) * -1
            net_pressures[i] += ( centers[i] - np.array(centers[i] < 1) * (centers[i] - 1) ) * 1
        centers += 0.05 * (net_pressures / np.linalg.norm(net_pressures, axis=1, keepdims=True))
        centers = np.clip(centers, 0, 1)
        radii = np.clip(radii - 1e-9, 0, None)
        for i in range(26):
            r_i = radii[i]
            for j in range(26):
                if i == j:
                    continue
                r_j = radii[j]
                dist = np.linalg.norm(centers[i] - centers[j])
                radii[i] = min(dist - r_j - 1e-9, r_i)

    sum_radii = radii.sum()
    return centers, radii, sum_radii
