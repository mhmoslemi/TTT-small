import numpy as np
from scipy.optimize import differential_evolution, linprog

# Helper function to compute pairwise distances
def get_distances(centers):
    n = centers.shape[0]
    distances = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            distances[i, j] = np.linalg.norm(centers[i] - centers[j])
    return distances

# Helper function to evaluate the objective function (negative sum of radii)
def eval_func(centers_flat):

    distances = get_distances(centers)
    A = np.zeros((26 * 26, 26))
    b = np.zeros(26 * 26)
    for i in range(26):
        for j in range(26):

                idx = 26 * i + j
                A[idx, i] = 1
                A[idx, j] = 1
                b[idx] = distances[i, j]

    bounds = [(0, None) for _ in range(26)]
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
