import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def eval_funcs(centres_flat):
    centres = centres_flat.reshape((26, 2))
    radii, total = lp_optimum(centres)
    if np.any(radii < 0):
        return 1e9 + 10 * np.sum(-np.minimum(radii, 0))
    return -total

def run_packing():
    centers = 0.5 * (np.random.rand(50, 2) + 1)  # x,y in [0,1]
    result = differential_evolution(eval_funcs, bounds=[(0, 1)]*52, popsize=50, strategy='best1bin', mutation=0.5, crossover=0.7, maxiter=80)
    centres_best = result.x.reshape((26, 2))
    radii, total = lp_optimum(centres_best)
    return centres_best, radii, total

if __name__ == "__main__":
    centres, radii, total = run_packing()
    print("Centres:\n", centres)
    print("Radii:\n", radii)
    print("Total sum of radii:", total)
