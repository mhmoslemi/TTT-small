import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

def objective(x):
    arr = x.reshape((N, 3))
    radii = arr[:, 2]
    sum_r = np.sum(radii)
    violation = 0.0
    for i in range(N):
        r = radii[i]
        x_i = arr[i, 0]
        y_i = arr[i, 1]
        violation += np.maximum(0.0, r - x_i)**2
        violation += np.maximum(0.0, r - (1 - x_i))**2
        violation += np.maximum(0.0, r - y_i)**2
        violation += np.maximum(0.0, r - (1 - y_i))**2
    for i in range(N-1):
        for j in range(i+1, N):
            d2 = (arr[i, 0] - arr[j, 0])**2 + (arr[i, 1] - arr[j, 1])**2
            v = (radii[i] + radii[j])**2 - d2
            if v > 0:
                violation += v**2
    return -sum_r + 1e6 * violation

bounds = [(0, 1)]*3*N + [(0, 0.5)]*N

res = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False, workers=1)

x0 = res.x

def refine(x0):
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3] - x[i*3+2]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*3] - x[i*3+2]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3+1] - x[i*3+2]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*3+1] - x[i*3+2]})
    for i in range(N-1):
        for j in range(i+1, N):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2 - (x[i*3+2] + x[j*3+2])**2})
    arr = x0.reshape((N, 3))
    x0_refine = x0.copy()
    res_refine = minimize(objective, x0_refine, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'maxiter': 200})
    return res_refine
