import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Fixed centers based on a hexagonal-3-row grid skeleton
    centres = np.array([
        (0.1,  0.1), (0.3, 0.1), (0.5,  0.1), (0.7, 0.1), (0.9,  0.1),
        (0.13, 0.3), (0.33, 0.3), (0.53, 0.3), (0.73, 0.3), (0.93, 0.3),
        (0.1,  0.5), (0.3, 0.5), (0.5,  0.5), (0.7, 0.5), (0.9,  0.5),
        (0.13, 0.7), (0.33, 0.7), (0.53, 0.7), (0.73, 0.7), (0.93, 0.7),
        (0.1,  0.9), (0.3, 0.9), (0.5,  0.9), (0.7, 0.9), (0.9,  0.9),
        (0.5,  1.0)
    ])
    N = centres.shape[0]

    # Pairwise euclidean distances
    distances = cdist(centres, centres)

    # Construct LP problem
    c = np.ones(N, dtype=np.float64)
    A = np.zeros((N**2, N), dtype=np.float64)
    b = np.zeros(N**2, dtype=np.float64)
    for i in range(N):
        for j in range(i + 1, N):
            A[i * N + j, i] = 1
            A[i * N + j, j] = 1
            b[i * N + j] = distances[i, j]

    # Boundary constraints
    for i in range(N):
        A[N**2 + 2 * i, i] = 1
        A[N**2 + 2 * i + 1, i] = -1
        b[N**2 + 2 * i] = 1
        b[N**2 + 2 * i + 1] = centers[i, 0]

        A[N**2 + 2 * i + 2, i] = 1
        A[N**2 + 2 * i + 3, i] = -1
        b[N**2 + 2 * i + 2] = 1
        b[N**2 + 2 * i + 3] = 1 - centers[i, 0]

        A[N**2 + 2 * i + 4, i] = 1
        A[N**2 + 2 * i + 5, i] = -1
        b[N**2 + 2 * i + 4] = 1
        b[N**2 + 2 * i + 5] = centers[i, 1]

        A[N**2 + 2 * i + 6, i] = 1
        A[N**2 + 2 * i + 7, i] = -1
        b[N**2 + 2 * i + 6] = 1
        b[N**2 + 2 * i + 7] = 1 - centers[i, 1]

    # Non-negativity constraints
    bounds = Bounds(0, None, dtype=np.float64)
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        raise ValueError(f"LP not successful: {result.message}")

    radii = result.x
    sum_radii = np.sum(radii)

    # Postprocessing
    centres[centres + radii >= 1] -= 1e-9
    if not validate_packing(centres, radii)[0]:
        raise ValueError("Invalid packing after fixing to [0,1]")

    return centres, radii, sum_radii
