```python
import numpy as np
import scipy.optimize

def run_packing():
    N = 26
    BOUNDARIES = [0, 1]
    
    def objective(z):
        centers = z[:2*N].reshape(N, 2)
        radii = z[2*N:]
        penalties = np.zeros_like(radii)
        
        # Boundary penalties
        penalties += np.maximum(0, radii - centers[:, 0]) ** 2
        penalties += np.maximum(0, radii - (1 - centers[:, 0])) ** 2
        penalties += np.maximum(0, radii - centers[:, 1]) ** 2
        penalties += np.maximum(0, radii - (1 - centers[:, 1])) ** 2
        
        # Overlap penalties
        dist = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        penalties += np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - dist) ** 2
        
        return -np.sum(radii) + 1e4 * np.sum(penalties)
    
    def feasibility(z):
        centers = z[:2*N].reshape(N, 2)
        radii = z[2*N:]
        overlap = np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2)))
        boundary = np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - (1 - centers[:, 0])) + np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii - (1 - centers[:, 1]))
        return overlap, boundary
    
    bounds = [(0, 1) for _ in range(2*N)] + [(0, 0.5) for _ in range(N)]
    
    best_z = None
    best_radius_sum = -np.inf
    
    for _ in range(30):
        z0 = np.zeros(3*N)
        z0[:2*N] = np.random.uniform(0.25, 0.75, size=2*N)
        z0[2*N:] = np.random.uniform(0.05, 0.15, size=N)
        
        res = scipy.optimize.minimize(objective, z0, method='L-BFGS-B', bounds=bounds, options={'maxfun': 2000, 'ftol': 1e-9, 'gtol': 1e-5})
        
        if res.success:
            overlaps, boundaries = feasibility(res.x)
            if np.all(overlaps < 1e-4) and np.all(boundaries < 1e-4):
                current_radius_sum = -res.fun
                if current_radius_sum > best_radius_sum:
                    best_z = res.x
                    best_radius_sum = current_radius_sum
    
    if best_z is None:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    
    centers = best_z[:2*N].reshape(N, 2)
    radii = best_z[2*N:]
    
    return centers, radii, best_radius_sum
```
