import numpy as np
import itertools
from scipy.optimize import linprog
from scipy.spatial import Delaunay
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    candidates = []

    # Generate candidate points in a triangular lattice
    r_est = np.sqrt(0.9069 / (np.pi * n))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    
    for k in range(30):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        x_offset = r_est if k % 2 == 0 else r_est + r_est
        for n in range(20):
            x = x_offset + n * s
            if x > 1 - r_est:
                break
            candidates.append([x, y])

    best_sum = -np.inf
    best_centers = None

    # Enumerate all combinations of 26 candidate points
    for subset in itertools.combinations(candidates, n):
        subset = np.array(subset)
        radii, sum_radii = lp_optimum(subset)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = subset

    # Local refinement using Nelder-Mead simplex
    result = linprog(-radii, A_ub=None, b_ub=None, bounds=None, method='nelder-mead', x0=subset, options={'maxiter': 30})
    best_centers = result.x
    best_radii, best_sum = lp_optimum(best_centers)

    return best_centers, best_radii, best_sum
