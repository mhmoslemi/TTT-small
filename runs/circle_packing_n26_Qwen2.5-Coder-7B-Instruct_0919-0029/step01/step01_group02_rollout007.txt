import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

def objective(x):
    global N
    n = N*3
    arr = x.reshape((n // 3, 3))
    x_coords = arr[:, 0]
    y_coords = arr[:, 1]
    radii = arr[:, 2]
    sum_r = np.sum(radii)
    penalty = 0.0
    for i in range(N):
        penalty += max(0.0, x_coords[i] - radii[i])**2
        penalty += max(0.0, 1 - x_coords[i] - radii[i])**2
        penalty += max(0.0, y_coords[i] - radii[i])**2
        penalty += max(0.0, 1 - y_coords[i] - radii[i])**2
    for i in range(N-1):
        for j in range(i+1, N):
            d2 = (x_coords[i] - x_coords[j])**2 + (y_coords[i] - y_coords[j])**2
            v = radii[i]**2 + radii[j]**2 - d2
            if v > 0:
                penalty += v**2
    return -sum_r + 1e10 * penalty

def boundary_left(x, idx):
    return x[idx*3] - x[idx*3+2]

def boundary_right(x, idx):
    return 1 - x[idx*3] - x[idx*3+2]

def boundary_bottom(x, idx):
    return x[idx*3+1] - x[idx*3+2]

def boundary_top(x, idx):
    return 1 - x[idx*3+1] - x[idx*3+2]

def pair_non_overlap(x, i, j):
    return (x[i*3] - x[j*3])**2 + (x[i*3+1] - x[j*3+1])**2 - (x[i*3+2] + x[j*3+2])**2

bounds = [(0, 1)] * 3 * N + [(0, 0.5)] * N

constraints += [{'type': 'ineq', 'fun': boundary_top, 'args': (i,)} for i in range(N)]

for i in range(N-1):
