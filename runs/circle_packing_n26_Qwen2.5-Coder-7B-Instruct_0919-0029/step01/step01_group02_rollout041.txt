import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26

def objective(x):
    radii = x[2::3]
    sum_r = np.sum(radii)
    violation = 0.0
    # boundary
    violation += np.sum(np.maximum(0.0, radii - x[0::3]**2)**2)
    violation += np.sum(np.maximum(0.0, radii - (1 - x[0::3]**2))**2)
    violation += np.sum(np.maximum(0.0, radii - x[1::3]**2)**2)
    violation += np.sum(np.maximum(0.0, radii - (1 - x[1::3]**2))**2)
    # pairwise
    for i in range(N - 1):
        xi, yi = x[0::3][i], x[1::3][i]
        for j in range(i + 1, N):
            dist = np.sqrt((xi - x[0::3][j])**2 + (yi - x[1::3][j])**2)
            v = radii[i] + radii[j] - dist
            if v > 0:
                violation += v**2
    return -sum_r + 1e4 * violation

def boundary_left(x, idx):
    return x[idx*3] - x[idx*3+2]

def boundary_right(x, idx):
    return (1 - x[idx*3]) - x[idx*3+2]

def boundary_bottom(x, idx):
    return x[idx*3+1] - x[idx*3+2]

def boundary_top(x, idx):
    return (1 - x[idx*3+1]) - x[idx*3+2]

def pair_non_overlap(x, i, j):
    xi, yi = x[0::3][i], x[1::3][i]
    xj, yj = x[0::3][j], x[1::3][j]
    dist = (xi - xj)**2 + (yi - yj)**2
    return dist - (x[2::3][i] + x[2::3][j])**2

def slqsproot(x0):
    bounds = [(0, 1)] * 6 * N
    constraints = [{'type': 'ineq', 'fun': boundary_left, 'args': (i,)} for i in range(N)]
    constraints.extend({'type': 'ineq', 'fun': boundary_right, 'args': (i,)} for i in range(N))
    constraints.extend({'type': 'ineq', 'fun': boundary_bottom, 'args': (i,)} for i in range(N))
    constraints.extend({'type': 'ineq', 'fun': boundary_top, 'args': (i,)} for i in range(N))
    for i in range(N - 1):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': pair_non_overlap, 'args': (i, j)})
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'maxiter': 200})
    return res

def run_packing():
    bounds = [(0, 1)] * 6 * N
    x0 = np.random.rand(N * 6)
    result = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False)
    refined_result = slqsproot(result.x)
    if refined_result.fun * -1 < 2.637:
        x0 = np.random.rand(N * 6)
        result = differential_evolution(objective, bounds, maxiter=200, popsize=20, disp=False)
        refined_result = slqsproot(result.x)
    centers = refined_result.x[0::3]
    radii = refined_result.x[2::3]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
