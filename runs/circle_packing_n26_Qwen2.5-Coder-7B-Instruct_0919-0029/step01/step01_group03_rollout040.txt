import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance_matrix
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    lattice_size = 0.3
    num_points = 30
    alpha = 0.02
    num_circles = 26

    # Build a hexagonal lattice
    x = np.arange(0, 1, lattice_size)
    y = np.arange(0, 1, 2 * lattice_size / np.sqrt(3))
    X, Y = np.meshgrid(x, y)
    centers = np.vstack((X.flatten(), Y.flatten())).T
    centers = centers[:num_points]

    # Add small perturbation
    centers += np.random.uniform(-alpha, alpha, (num_points, 2))
    centers = np.clip(centers, 0, 1)

    # Select 26 points and pad with placeholder
    centers = centers[:num_circles]

    best_order = None
    best_sumradii = 0
    best_radii = np.zeros(num_circles)

    def greedy_insertion(order):
        radii = np.zeros(num_circles)
        for i, idx in enumerate(order):
            xi, yi = centers[idx]
            rhomin = min(xi, 1 - xi, yi, 1 - yi)
            rhomin = min(rhomin, min(np.abs(xi - xj) - rj for xj, yj, rj in zip(centers[:i, 0], centers[:i, 1], radii[:i])))
            radii[idx] = max(rhomin, 1e-6)
        return np.sum(radii)

    temperature = 0.5
    for _ in range(30000):
        order = np.arange(num_circles)
        np.random.shuffle(order)
        sumradii = greedy_insertion(order)
        if sumradii > best_sumradii:
            best_sumradii = sumradii
            best_order = order
            best_radii = np.zeros(num_circles)
            for i, idx in enumerate(best_order):
                xi, yi = centers[idx]
                rhomin = min(xi, 1 - xi, yi, 1 - yi)
                rhomin = min(rhomin, min(np.abs(xi - xj) - rj for xj, yj, rj in zip(centers[:i, 0], centers[:i, 1], best_radii[:i])))
                best_radii[idx] = max(rhomin, 1e-6)
        previous_order = order
        order1, order2 = random.sample(range(num_circles), 2)
        order[order1], order[order2] = order[order2], order[order1]
        sumradii = greedy_insertion(order)
        if sumradii > best_sumradii:
            best_sumradii = sumradii
            best_order = order
            best_radii = np.zeros(num_circles)
            for i, idx in enumerate(best_order):
                xi, yi = centers[idx]
                rhomin = min(xi, 1 - xi, yi, 1 - yi)
                rhomin = min(rhomin, min(np.abs(xi - xj) - rj for xj, yj, rj in zip(centers[:i, 0], centers[:i, 1], best_radii[:i])))
                best_radii[idx] = max(rhomin, 1e-6)
        elif random.random() < np.exp((sumradii - greedy_insertion(previous_order)) / temperature):
            best_order = order
            best_radii = np.zeros(num_circles)
            for i, idx in enumerate(best_order):
                xi, yi = centers[idx]
                rhomin = min(xi, 1 - xi, yi, 1 - yi)
                rhomin = min(rhomin, min(np.abs(xi - xj) - rj for xj, yj, rj in zip(centers[:i, 0], centers[:i, 1], best_radii[:i])))
                best_radii[idx] = max(rhomin, 1e-6)
        temperature *= 0.995

    # LP refinement
    Aeq = np.eye(num_circles)
    beq = np.ones(num_circles)
    bounds = [(0, None) for _ in range(num_circles)]
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({"type": 'ineq', "fun": lambda r, i=i, j=j: np.sum((centers[i] - centers[j]) ** 2) - (r[i] + r[j])})
    constraints += [{"type": 'ineq', "fun": lambda r, i=i: centers[i, 0] - r[i]}, {"type": 'ineq', "fun": lambda r, i=i: 1 - centers[i, 0] - r[i]},
                   {"type": 'ineq', "fun": lambda r, i=i: centers[i, 1] - r[i]}, {"type": 'ineq', "fun": lambda r, i=i: 1 - centers[i, 1] - r[i]}]
    result = linprog(-best_radii, A_ub=None, b_ub=None, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
    optimal_radii = -result.x

    return centers, optimal_radii, np.sum(optimal_radii)
