import numpy as np
from scipy.optimize import linprog

def generate_positions(num_outer, num_inner):
    outer_positions = []
    x_step = 0.33
    y_step = 0.33
    for x in np.arange(0, 1, x_step):
        for y in np.arange(0, 1, y_step):
            outer_positions.append((x, y))
    inner_positions = [(0.5, 0.5)]
    return outer_positions[:num_outer] + inner_positions[:num_inner]

def compute_distances(centers):
    diff = centers[:, None, :] - centers[None, :, :]
    return np.linalg.norm(diff, axis=2)

def lp_radii(centers):
    n = centers.shape[0]
    dist = compute_distances(centers)
    A_ub = np.zeros((n * n, n))
    b_ub = np.zeros(n * n)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * n + j] = np.array([1 if k == i or k == j else 0 for k in range(n)])
            b_ub[i * n + j] = dist[i, j]
    A_eq = None
    b_eq = None
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple:
    num_outer = 12
    num_inner = 14
    positions = generate_positions(num_outer, num_inner)
    best_centers = np.array(positions)
    best_sum = 0.0
    num_iter = 500
    best_centers = None
    for _ in range(num_iter):
        perturbed_centers = best_centers + np.random.normal(0, 0.03, best_centers.shape)
        perturbed_centers = np.clip(perturbed_centers, 0.05, 0.95)
        radii, sum_radii = lp_radii(perturbed_centers)
        if sum_radii > best_sum:
            best_centers = perturbed_centers
            best_sum = sum_radii
    return best_centers.astype(np.float64), radii.astype(np.float64), best_sum
