import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) / 100  # Initial random small radii
    radii.fill(0.01)  # Set initial radii to be very small

    for _ in range(1000):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0:
                x = radii[i] + 1e-9
            if x + radii[i] > 1:
                x = 1 - radii[i] - 1e-9
            if y - radii[i] < 0:
                y = radii[i] + 1e-9
            if y + radii[i] > 1:
                y = 1 - radii[i] - 1e-9

            new_centers = centers.copy()
            new_centers[i] = np.array([x, y])

            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(new_centers[i] - new_centers[j])
                    if dist < radii[i] + radii[j] - 1e-9:
                        new_centers[i, 0] = (new_centers[i, 0] + new_centers[j, 0]) / 2
                        new_centers[i, 1] = (new_centers[i, 1] + new_centers[j, 1]) / 2

            centers = new_centers

        radii = np.random.rand(n) / 100  # Reset radii for next iteration

    return centers, radii, np.sum(radii)
