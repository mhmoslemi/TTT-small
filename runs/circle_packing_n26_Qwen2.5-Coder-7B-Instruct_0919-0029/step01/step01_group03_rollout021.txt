import numpy as np
from scipy.optimize import linprog

def lp_radii(centers):
    n = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])
    idx = 0
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist[i, j]
            idx += 1
        
        for k in range(4):
            A_ub[idx, k] = 1
            A_ub[idx, i] = max(0, centers[i, k] - 1)
            b_ub[idx] = centers[i, k]
            idx += 1
    
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n_circles = 26
    max_iter = 15
    restarts = 50
    
    def random_layout():
        def hex_positions():
            core = [(0.5, 0.5), (1/3, 1/3), (2/3, 1/3), (1/3, 2/3), (2/3, 2/3), (1/2, 1/6), (1/2, 1/2), (1/2, 5/6)]
            perimeter = [(0.05, 0.175), (0.95, 0.175), (0.05, 0.725), (0.95, 0.725), 
                          (0.175, 0.05), (0.825, 0.05), (0.175, 0.95), (0.825, 0.95), 
                          (0.05, 0.45), (0.95, 0.45), (0.45, 0.05), (0.45, 0.95), 
                          (0.5, 0.3), (0.5, 0.7), (0.3, 0.5), (0.7, 0.5)]
            return core + perimeter
        
        def random_positions():
            return [(np.random.rand(), np.random.rand()) for _ in range(n_circles - len(hex_positions()))]
        
        hex_pos = hex_positions()
        random_pos = random_positions()
        return hex_pos + random_pos
    
    best_centers = None
    best_sum = -np.inf
    
    for _ in range(restarts):
        centers = np.array(random_layout())
        for _ in range(max_iter):
            jitter_amount = 0.05 * np.mean(np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2)))
            jitter = np.random.normal(scale=jitter_amount, size=centers.shape)
            jitter = np.clip(jitter, -2, 2)
            new_centers = centers + jitter
            new_radii, new_sum = lp_radii(new_centers)
            if new_sum > best_sum and not np.isnan(new_sum):
                best_centers, best_sum = new_centers, new_sum
    
    return best_centers, new_radii, best_sum
