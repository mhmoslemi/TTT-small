import numpy as np
from scipy.optimize import linprog

# Define initial circles based on Apollonian construction
def construct_initial_circles():
    circle_L = (0.25, 0.25, 0.25)
    circle_R = (0.75, 0.25, 0.25)
    circle_T = (0.5, 0.75, 0.25)
    return [circle_L, circle_R, circle_T]

# Construct an Apollonian packing of circles in the unit square
def apollonian_packing(initial_circles, target_count=26):
    lines = [(0, 0), (0, 1), (1, 0), (1, 1)]
    initial_objects = initial_circles + lines
    gaps = []

    # Add gaps based on initial circles and lines
    for i in range(len(initial_objects)):
        for j in range(i + 1, len(initial_objects)):
            for k in range(j + 1, len(initial_objects)):
                x1, y1, _ = initial_objects[i]
                x2, y2, _ = initial_objects[j]
                x3, y3, _ = initial_objects[k]
                k1 = (x2 * y3 - x3 * y2) / (x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2))
                k2 = (x3 * y1 - x1 * y3) / (x2 * (y3 - y1) + x3 * (y1 - y2) + x1 * (y2 - y3))
                k3 = (x1 * y2 - x2 * y1) / (x3 * (y1 - y2) + x1 * (y2 - y3) + x2 * (y3 - y1))
                k = (k1 + k2 + k3 + 2 * np.sqrt(k1 * k2 + k2 * k3 + k3 * k1)) * 2
                if k > 0 and 0 <= x1 + x2 + x3 <= 2 and 0 <= y1 + y2 + y3 <= 2:
                    gaps.append((initial_objects[i], initial_objects[j], initial_objects[k], k))

    while len(initial_circles) < target_count:
        gaps.sort(key=lambda gap: gap[3], reverse=True)
        if not gaps:
            break
        c1, c2, c3, k = gaps.pop(0)
        x1, y1, r1 = c1
        x2, y2, r2 = c2
        x3, y3, r3 = c3
        x_new = (x1 + x2 + x3) / 3
        y_new = (y1 + y2 + y3) / 3
        r_new = k / 2
        initial_circles.append((x_new, y_new, r_new))

    return initial_circles

# Refine the radius distribution with an LP
def refine_with_lp(centres):
    n = centres.shape[0]
    radii = np.zeros(n)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    A_ub = []
    b_ub = []

    for i in range(n):
        r_i = radii[i]
        for j in range(i + 1, n):
            r_j = radii[j]
            dist = np.sqrt((centres[i, 0] - centres[j, 0])**2 + (centres[i, 1] - centres[j, 1])**2)
            A_ub.append([r_i, r_j, dist])
            b_ub.append(r_i + r_j)

        r_i = radii[i]
        dist_to_left = centres[i, 0]
        dist_to_right = 1 - centres[i, 0]
        dist_to_bottom = centres[i, 1]
        dist_to_top = 1 - centres[i, 1]
        A_ub.append([r_i, 0, dist_to_left])
        b_ub.append(dist_to_left)
        A_ub.append([r_i, 0, dist_to_right])
        b_ub.append(dist_to_right)
        A_ub.append([r_i, 0, dist_to_bottom])
        b_ub.append(dist_to_bottom)
        A_ub.append([r_i, 0, dist_to_top])
        b_ub.append(dist_to_top)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

# Perform an optional push of the largest circles
def push_largest_circles(centres, radii, push_amount=1e-4):
    max_radius = np.max(radii)
    indices = np.where(radii == max_radius)[0]
    push_indices = indices[np.argmax(centres[indices, 1])]
    remaining_circles = np.where(radii != max_radius)
    new_radii = radii.copy()
    new_radii[push_indices] += push_amount
    new_radii[remaining_circles] *= 1 - push_amount / (np.sum(new_radii[remaining_circles]) + push_amount)
    return centres, new_radii

# Finalize the packing process
def run_packing():
    initial_circles = construct_initial_circles()
    apollonian_circles = apollonian_packing(initial_circles)
    centres = np.array([[c[0], c[1], c[2]] for c in apollonian_circles[:26]])
    initial_radii, initial_sum = refine_with_lp(centres)
    final_radii, final_sum = push_largest_circles(centres, initial_radii)
    return centres[:, :2], final_radii, final_sum

# Run the packing
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)

The provided code completes the implementation of the Apollonian packing method to optimize the sum of radii for 26 circles in a unit square, followed by an LP refinement and optionally a push. This approach offers a deterministically constructed configuration, local LP polish, and a mechanism to optionally push the largest circles, significantly differing from previous random walk or SA-based methods.
