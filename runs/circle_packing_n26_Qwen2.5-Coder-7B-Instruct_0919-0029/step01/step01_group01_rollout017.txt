import numpy as np
from scipy.optimize import differential_evolution, linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = np.empty((400, 26))
    b_ub = np.empty(400)

    for i in range(n):
        A_ub[i*4] = [1, 0] + [0]*i + [0]*(n-i-1)
        b_ub[i*4] = centres[i, 0]
        A_ub[i*4+1] = [-1, 0] + [0]*i + [0]*(n-i-1)
        b_ub[i*4+1] = 1 - centres[i, 0]
        A_ub[i*4+2] = [0, 1] + [0]*i + [0]*(n-i-1)
        b_ub[i*4+2] = centres[i, 1]
        A_ub[i*4+3] = [0, -1] + [0]*i + [0]*(n-i-1)
        b_ub[i*4+3] = 1 - centres[i, 1]

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[n*4 + (i*(n-i)+j-1)*2] = [0] * i + [1, 0] + [1, 0] + [0]*(n-j-2)
            b_ub[n*4 + (i*(n-i)+j-1)*2] = np.linalg.norm(centres[j] - centres[i])
            A_ub[n*4 + (i*(n-i)+j-1)*2 + 1] = [0] * (i+1) + [1, 0] + [0]*(n-j-1)
            b_ub[n*4 + (i*(n-i)+j-1)*2 + 1] = np.linalg.norm(centres[j] - centres[i])

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def de_objective(flat_centres):
    centres = flat_centres.reshape(26,2)
    _, s = solve_radius_lp(centres)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0,1.0)] * 52
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=120,
        popsize=12,
        tol=1e-6,
        polish=True,
        seed=42
    )
    best_centres = result.x.reshape(26,2)
    final_radii, final_sum = solve_radius_lp(best_centres)
    return best_centres, final_radii, final_sum
