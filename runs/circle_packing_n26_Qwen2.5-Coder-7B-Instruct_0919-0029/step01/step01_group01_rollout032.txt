import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import Delaunay

def lp_optimum(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # infinity on the diagonal

    # Collect constraints
    A_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        b_ineq.append(centers[i, 1] - 1e-12)
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def initial_layout(n):
    radius = 0.14
    circle_positions = []
    center_x = np.linspace(0.05, 0.95, 3)
    center_y = np.linspace(0.05, 0.95, 9)
    for x in center_x:
        for y in center_y:
            circle_positions.append([x, y])
    return np.array(circle_positions[:n]), np.full(n, radius)

def sa_optimization(centers, n, T0=0.1, Tf=1e-4, alpha=0.9995, delta=0.1, steps_per_T=500, max_iter=None):
    if max_iter is None:
        max_iter = int(-np.log(Tf) / np.log(alpha)) * steps_per_T

    best_centers = centers.copy()
    best_sum = lp_optimum(centers)[1]

    for k in range(int(-np.log(Tf) / np.log(alpha))):
        T = T0 * alpha ** k
        for _ in range(steps_per_T):
            i = np.random.randint(n)
            dx, dy = np.random.randn(2) * delta
            dx, dy = min(max(dx, 0), 1) - centers[i, 0], min(max(dy, 0), 1) - centers[i, 1]
            new_centers = centers.copy()
            new_centers[i] += [dx, dy]
            new_centers = np.clip(new_centers, 0, 1)
            radii, sum_radii = lp_optimum(new_centers)
            delta_s = sum_radii - best_sum
            if delta_s > 0 or np.random.rand() < np.exp(delta_s / T):
                centers = new_centers
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = centers

    return best_centers, best_sum

def final_polish(centers, m):
    def objective_function(vars):
        return -sum(vars[:m])
    
    def constraint(vars):
        centers_opt = centers.copy()
        centers_opt[:, 0] = vars[m:m+26]
        centers_opt[:, 1] = vars[m+26:]
        _, sum_radii = lp_optimum(centers_opt)
        return sum_radii - m
    
    centers_opt = centers.copy()
    x0 = np.concatenate([np.zeros(m), centers.flatten()])
    cons = {'type': 'ineq', 'fun': constraint}
    bounds = [(0, 1)] * 26 + [(0, None)] * 26
    res = minimize(objective_function, x0, method='L-BFGS-B', bounds=bounds, constraints=cons)
    
    return res.x[m:m+26], res.x[m+26:], -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    m = 26
    centers, _ = initial_layout(m)
    centers, sum_radii = sa_optimization(centers, m)
    centers, radii, sum_radii = final_polish(centers, m)
    return centers, radii, sum_radii
