import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Fixed centres in a 3-row hexagonal lattice
    centers = np.array([
        (0.1,   0.1), (0.3,   0.1), (0.5,   0.1), (0.7,   0.1), (0.9,   0.1),
        (0.13,  0.3), (0.33, 0.3), (0.53, 0.3), (0.73, 0.3), (0.93, 0.3),
        (0.1,   0.5), (0.3,   0.5), (0.5,   0.5), (0.7,   0.5), (0.9,   0.5),
        (0.13,  0.7), (0.33, 0.7), (0.53, 0.7), (0.73, 0.7), (0.93, 0.7),
        (0.1,   0.9), (0.3,   0.9), (0.5,   0.9), (0.7,   0.9), (0.9,   0.9),
        (0.5,   1.0)
    ])
    
    # Number of centres
    N = centers.shape[0]
    
    # Calculated boundary constraints max(r_i) <= x_i or y_i
    ub = np.minimum(centers[:, 0], np.ones(N))  # clamp x coordinates (0 to 1)
    ub = np.minimum(ub, np.ones(N))  # clamp y coordinates (0 to 1)
    
    # Precompute distances between all pairs of circles
    dists = cdist(centers, centers)
    
    # Create coefficients matrix for circle radius constraints (LP objective is to maximize radii sum)
    A = np.zeros((N * (N - 1) // 2, N))
    b = np.zeros(A.shape[0])
    
    # Put non-overlap and boundary constraints into the matrix A and b
    row_count = 0
    for i in range(N):
        for j in range(N):
            if i < j:
                A[row_count, i] = 1
                A[row_count, j] = 1
                b[row_count] = dists[i, j] - 1e-12
                row_count += 1
    
    # Define bounds for radii
    bounds = Bounds(0, np.inf)
    
    # Solve the linear programming problem to find optimal radii
    result = linprog(-np.ones(N), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    radii = -result.x  # convert maximization problem to minimization by multiplying by -1
    
    # Apply epsilon trimming to ensure circles stay inside the unit square
    epsilon = 1e-9
    radii = np.clip(radii, 0, 1 - epsilon)
