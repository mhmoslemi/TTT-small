import numpy as np
from scipy.optimize import linprog

def linear_program(centers):
    n = len(centers)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i+1, n):
            A_ub.append([1 if k==i or k==j else 0 for k in range(n)])
            b_ub.append(dist[i,j])
    for i in range(n):
        x, y = centers[i]
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(x)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1-x)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(y)
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1-y)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("Linear program infeasible")
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n_circles = 26
    radius_guess = np.full(n_circles, 0.1)
    centers_best = np.random.rand(n_circles, 2)
    sum_radii_best = -np.inf
    
    def initial_hex_grid():
        radii = np.full(n_circles, 0.12)
        x_corner = np.array([0.125, 0.125, 0.875, 0.875])
        y_corner = np.array([0.125, 0.875, 0.125, 0.875])
        centers = np.column_stack((np.repeat(x_corner, 4), np.repeat(y_corner, 4)))
        interior_spacing = 0.3
        interior = np.random.rand(2*n_circles, 2)
        interior[:, 0] += 0.15 + 0.3 * (interior[:, 0] - 0.5)
        interior[:, 1] += 0.15 + 0.3 * (interior[:, 1] - 0.5)
        centers = np.vstack((centers, np.clip(interior, 0, 1)))
        np.random.shuffle(centers)
        return centers, radii
    
    def jitter(centers, scale):
        jitter = scale * np.random.randn(n_circles, 2)
        jitter[:, 0] = np.clip(jitter[:, 0], -0.05, 0.05)
        jitter[:, 1] = np.clip(jitter[:, 1], -0.05, 0.05)
        return centers + jitter
    
    for restart in range(100):
        centers, radius_guess = initial_hex_grid()
        for step in range(15):
            centers = jitter(centers, 0.05)
            try:
                radius_guess, sum_radii = linear_program(centers)
                if sum_radii > sum_radii_best:
                    sum_radii_best = sum_radii
                    centers_best = centers
            except RuntimeError:
                continue
    
    _, sum_radii = linear_program(centers_best)
    return centers_best[:n_circles], np.clip(radius_guess, 0, None), sum_radii_best
