import numpy as np
import pulp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters
    Δ = 0.075
    nx = nx_ny = 14

    # Grid points
    X, Y = np.meshgrid(np.linspace(0, 1, nx+1), np.linspace(0, 1, nx+1))
    grid_points = np.vstack((X.flatten(), Y.flatten())).T

    # Distance between points
    D = np.sqrt(np.sum((grid_points[:, np.newaxis, :] - grid_points[np.newaxis, :, :]) ** 2, axis=-1))

    # Optimize
    model = pulp.LpProblem("Circle Packing", pulp.LpMaximize)
    binary_vars = [pulp.LpVariable(f'y_{k}', 0, 1, pulp.LpBinary) for k in range(len(grid_points))]
    continuous_vars = [pulp.LpVariable(f'r_{k}', 0, 0.5) for k in range(len(grid_points))]

    # Constraints
    model += pulp.lpSum(continuous_vars)  # Maximize sum of radii
    model += pulp.lpSum(binary_vars) == 26  # Exactly 26 circles

    for i in range(len(grid_points)):
        for j in range(i + 1, len(grid_points)):
            if D[i, j] < 0.5:
                break  # Avoid unnecessary constraints
            model += continuous_vars[i] + continuous_vars[j] <= D[i, j] + 0.5 * (1 - binary_vars[i]) + 0.5 * (1 - binary_vars[j])

    for k in range(len(grid_points)):
        model += continuous_vars[k] <= grid_points[k][0]
        model += continuous_vars[k] <= 1 - grid_points[k][0]
        model += continuous_vars[k] <= grid_points[k][1]
        model += continuous_vars[k] <= 1 - grid_points[k][1]

    model += pulp.lpSum(binary_vars[k] * continuous_vars[k] for k in range(len(grid_points)))  # Only affect chosen circles

    # Solver setup
    solver = pulp.PULP_CBC_CMD(timeLimit=30)
    model.solve(solver=solver)

    # Extract results
    centers = grid_points[:, [pulp.value(Y[k]) for k in range(len(grid_points)) if pulp.value(binary_vars[k]) == 1]]
    radii = np.array([pulp.value(continuous_vars[k]) for k in range(len(grid_points)) if pulp.value(binary_vars[k]) == 1])

    return centers, radii, np.sum(radii)
