import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    r_est = np.sqrt(0.9069 / (np.pi * 26))
    s = 2 * r_est
    h = np.sqrt(3) * r_est
    points = []
    for k in range(-4, 5):
        y = r_est + k * h
        if y > 1 - r_est:
            break
        if k % 2 == 0:
            for i in range(int((1 - 2 * r_est - y) / s) + 1):
                x = r_est + i * s
                points.append([x, y])
        else:
            for i in range(-1, int((1 - 2 * r_est - y) / s) + 1):
                x = r_est + (2 * i + 1) * s / 2
                points.append([x, y])

    points = np.array(points)[:32]
    best_sum = -np.inf
    best_centres = None

        centres = points[list(subset)]
        radii, total = lp_optimum(centres)
        if total > best_sum:
            best_sum = total
            best_centres = centres.copy()

    def ref_obj(centres_flat):
        centres_arr = centres_flat.reshape((n, 2))
        radii, total = lp_optimum(centres_arr)
        return -total

    x0 = best_centres.ravel()
    result = minimize(ref_obj, x0, method='nelder-mead', options={'maxfev': 30, 'tol': 1e-4})
    best_centres = result.x.reshape((n, 2))
    final_radii, final_sum = lp_optimum(best_centres)
    return best_centres, final_radii, final_sum
