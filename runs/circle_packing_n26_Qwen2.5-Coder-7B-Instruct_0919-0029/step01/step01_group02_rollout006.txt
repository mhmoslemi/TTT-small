import numpy as np
from scipy.spatial.distance import pdist, squareform

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    N = 26
    δ = 0.02
    ε = 1e-9

    # Base hexagonal lattice skeleton
    row1 = np.array([1/6, 1/3, 1/2, 2/3, 5/6]) - 1/2 * np.array([1, 1, 0, 0, 1])
    row2 = np.array([1/6, 1/3, 1/2, 2/3, 5/6]) - 1/2 * np.array([1, 0, 1, 0, 1])
    row3 = np.array([1/6, 1/3, 1/2, 2/3, 5/6]) - 1/2 * np.array([1, 1, 1, 1, 1])
    row4 = np.array([1/6, 1/3, 1/2, 2/3, 5/6]) + 1/2 * np.array([1, 1, 0, 0, 1])
    row5 = np.array([1/6, 1/3, 1/2, 2/3, 5/6]) + 1/2 * np.array([1, 0, 1, 0, 1])
    row6 = np.array([1/2, 5/6]) - 1/2 * np.array([1, 1])
    row7 = np.array([1/2, 5/6]) + 1/2 * np.array([1, 1])

    centers = np.array(list(row1) + list(row2) + list(row3) + list(row4) + list(row5) + list(row6) + list(row7))

    # LP to maximize radii
    c = np.ones(N)
    offsets = squareform(pdist(centers)) * 0.5 - np.eye(N)
    A = offsets.reshape(N*(N-1)//2, N)
    b = np.zeros(N*(N-1)//2)

    result = np.linalg.solve(A, b)
    radii = result

    if np.any(radii < 0):
        return np.zeros((N, 2)), np.zeros(N), 0.0

    # Local refinement
    M = 15
    best_radii = radii
    best_centers = centers
    best_sum = np.sum(radii)

    for _ in range(M):
        perturbed_centers = centers + np.random.uniform(-δ, δ, size=(N, 2))
        perturbed_centers = np.clip(perturbed_centers, 0, 1)
        perturbed_radii = np.linalg.solve(A, b)
        if np.any(perturbed_radii < 0):
            continue
        perturbed_sum = np.sum(perturbed_radii)
        if perturbed_sum > best_sum or (perturbed_sum == best_sum and np.min(perturbed_radii) > np.min(best_radii)):
            best_radii = perturbed_radii
            best_centers = perturbed_centers
            best_sum = perturbed_sum

    # Ensure fesibility
    best_centers -= best_radii
    best_centers = np.clip(best_centers, 0, 1)
    best_centers += best_radii

    # Trim epsilon
    best_radii -= ε
    best_radii = np.clip(best_radii, 0, 0.5)

    return best_centers, best_radii, np.sum(best_radii)
