import numpy as np
import scipy.optimize

def run_packing():
    np.random.seed(42)
    
    def scaffold_radii(a, centers, radius):
        r = np.ones_like(a) * radius
        return r
    
    def radii_from_lp(centers):
        N = 26
        obj = np.ones(N)
        A_ub = np.zeros((N * (N - 1) // 2, N))
        b_ub = np.zeros(N * (N - 1) // 2)
        
        for i in range(N):
            A_ub[i * (N - 1) // 2, i] = -1
            b_ub[i * (N - 1) // 2] = -centers[i, 0]
            A_ub[i * (N - 1) // 2 + (N - 1), i] = 1
            b_ub[i * (N - 1) // 2 + (N - 1)] = 1 - centers[i, 0]
            A_ub[i * (N - 1) // 2 + (N - 1) * 2, i] = -1
            b_ub[i * (N - 1) // 2 + (N - 1) * 2] = -centers[i, 1]
            A_ub[i * (N - 1) // 2 + (N - 1) * 3, i] = 1
            b_ub[i * (N - 1) // 2 + (N - 1) * 3] = 1 - centers[i, 1]
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx**2 + dy**2)
                A_ub[i * (N - 1) // 2 + j * (N - i - 1), i] = 1
                A_ub[i * (N - 1) // 2 + j * (N - i - 1), j] = 1
                b_ub[i * (N - 1) // 2 + j * (N - i - 1)] = d
        
        res = scipy.optimize.linprog(c=-obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        return -res.x
    
    def objective(x, centers):
        r = radii_from_lp(centers)
        return np.sum(r)
    
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.13, 0.3], [0.33, 0.3], [0.53, 0.3], [0.73, 0.3], [0.93, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.13, 0.7], [0.33, 0.7], [0.53, 0.7], [0.73, 0.7], [0.93, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
        [0.5, 1.0]
    ])
    
    def constrain_boundaries(x, centers):
        return np.clip(x, 0.1, 0.9)
    
    perturb_strength = 0.02
    max_iter = 15
    best_centers = None
    best_sum_r = -np.inf
    
    for _ in range(max_iter):
        perturbed_centers = centers + np.random.uniform(-perturb_strength, perturb_strength, (26, 2))
        perturbed_centers = np.clip(perturbed_centers, 0.0, 1.0)
        sum_r = objective(perturbed_centers, centers)
        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_centers = perturbed_centers
    
    r = radii_from_lp(best_centers)
    r -= 1e-9
    r = np.clip(r, 0.0, 0.5)
    centers = np.round(best_centers, 6)
    return centers, r, np.sum(r)
