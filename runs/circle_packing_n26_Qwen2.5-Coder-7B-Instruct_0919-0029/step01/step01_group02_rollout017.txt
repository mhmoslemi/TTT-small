import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    g = 0.01
    grid_points = np.array([(i*g, j*g) for i in range(101) for j in range(101)])
    num_circles = 26
    max_iterations = 200
    epsilon = 1e-9
    
    centres = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    active_count = 0
    while active_count < num_circles:
        tau_min = np.inf
        candidate_indices = []
        for i, point in enumerate(grid_points):
            if np.any(radii == 0):
                distances_to_boundaries = np.array([point[i] for i in [0, 1, 2, 3]]) - np.array([radii[j] for j in range(num_circles)])
                distances_to_active_circles = np.inf
                if active_count > 0:
                    distances_to_active_circles = np.min(np.sqrt(np.sum((grid_points[:active_count] - point)**2, axis=1)) - (radii[:active_count] + radii[i]))
                tau = np.min([np.maximum(distances_to_boundaries, 0), np.maximum(distances_to_active_circles, 0)])
                if tau < tau_min:
                    tau_min = tau
                    candidate_indices = [i]
                elif tau == tau_min:
                    candidate_indices.append(i)
            else:
                break
        for i in candidate_indices:
            if radii[i] == 0:
                radii[i] += tau_min
                active_count += 1
    
    for it in range(max_iterations):
        pressures = np.zeros((num_circles, 2))
        for i in range(num_circles):
            forces = np.zeros((num_circles, 2))
            for j in range(num_circles):
                if i != j:
                    diff = centres[i] - centres[j]
                    dist = np.linalg.norm(diff)
                    overlap = dist - radii[i] - radii[j]
                    if overlap > 0:
                        forces[i] += diff / dist * overlap
            boundary_forces = np.array([np.minimum(1-x, x) - r, np.minimum(1-y, y) - r for x, y, r in zip(centres[:, 0], centres[:, 1], radii)])
            pressures[i] = forces[i] + boundary_forces
        
        for i in range(num_circles):
            new_centre = np.clip(centres[i] - pressures[i] * 0.05, 0, 1)
            distances_to_active_circles = np.inf
            if active_count > 0:
                distances_to_active_circles = np.min(np.sqrt(np.sum((new_centre - centres[:active_count])**2, axis=1)) - (radii[:active_count] + radii[i]))
            if distances_to_active_circles > 0:
                centres[i] = new_centre
                radii[i] = min(1 - min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1]), sums(get_max_radius(circles(np.zeros((num_circles, 2)), np.zeros(num_circles)))) + epsilon)
    
    return centres, radii, np.sum(radii)

def circles(centres, radii):
    return np.array([(c[0] - r, c[1] - r, c[0] + r, c[1] + r) for c, r in zip(centres, radii)])

def get_max_radius(centre, radius):
    n = centre.shape[0]
    max_radius = np.zeros(n)
    for i in range(n):
        max_radius[i] = np.min([1 - min(centre[i, 0], 1 - centre[i, 0], centre[i, 1], 1 - centre[i, 1]), (centre[i, 1] - centre[i, 0] + radius[i] * np.sqrt(2)) / 2])
    return max_radius
