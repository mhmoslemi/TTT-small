def local_search(centers, radii):
    best = (centers, radii, lp_optimum(centers)[1])
    improved = True
    step = 0.05
    while improved:
        improved = False
        for i in range(26):
            for dx in [-step, 0, step]:
                for dy in [-step, 0, step]:
                    if dx==0 and dy==0: continue
                    new_center = best[0][i].copy()
                    new_center += np.array([dx, dy])
                    new_center = np.clip(new_center, 0, 1)
                    new_centers = best[0].copy()
                    new_centers[i] = new_center
                    radii_new, sum_new = lp_optimum(new_centers)
                    if sum_new > best[2] + 1e-12:
                        best = (new_centers, radii_new, sum_new)
                        improved = True
