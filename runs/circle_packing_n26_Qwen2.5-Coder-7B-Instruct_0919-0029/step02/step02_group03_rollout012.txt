import numpy as np
from scipy.optimize import minimize

def solve_radii(centers):
    n = centers.shape[0]
    c = centers.flatten()
    A = np.zeros((n*(n-1)//2, 3*n + 2*n))
    b = np.zeros(n*(n-1)//2)
    for i in range(n):
        A[i*(n-1)//2, 6*i:6*i+2] = 2 * centers[i]
        A[i*(n-1)//2, 6*i+2] = -1
        b[i*(n-1)//2] = 1
    for i in range(n):
        for j in range(i+1, n):
            A[i*(n-1)//2 + (j-i-1), 6*i:6*i+2] = centers[i]
            A[i*(n-1)//2 + (j-i-1), 6*j:6*j+2] = centers[j]
            A[i*(n-1)//2 + (j-i-1), 6*i+2] = 1
            A[i*(n-1)//2 + (j-i-1), 6*j+2] = 1
            b[i*(n-1)//2 + (j-i-1)] = np.linalg.norm(centers[i] - centers[j])
    r = np.linalg.solve(A, b)
    return r[:n]

def sum_of_radii(centers):
    radii = solve_radii(centers)
    return np.sum(radii)

def objective(Z):
    n = int(len(Z) / 2)
    centers = Z.reshape((n, 2))
    return -sum_of_radii(centers)

def gradient(Z):
    n = int(len(Z) / 2)
    centers = Z.reshape((n, 2))
    h = 1e-4
    grad = np.zeros_like(Z)
    for i in range(n * 2):
        Z_p = Z.copy()
        Z_n = Z.copy()
        Z_p[i] += h
        Z_n[i] -= h
        grad[i] = (objective(Z_p) - objective(Z_n)) / (2 * h)
    return grad

def run_packing():
    n = 26
    centers = np.array([[(x, y) for x in np.linspace(0, 1, 7)[:-1] for y in np.linspace(0, 1, 7)[:-1] if (x, y) != (0.5, 0.5)] for _ in range(2)]).reshape((n, 2))
    perturbation = 0.01 * np.random.randn(n, 2)
    initial_coords = centers + perturbation
    initial_coords = np.clip(initial_coords, 0, 1)
    initial_Z = initial_coords.flatten()
    result = minimize(
        objective,
        initial_Z,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0.0, 1.0)] * (2 * n),
        options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
    )
    best_Z = result.x
    final_coords = best_Z.reshape((n, 2))
    final_coords = np.clip(final_coords, 0, 1)
    radii = solve_radii(final_coords)
    sum_radii = sum_of_radii(final_coords)
    if sum_radii > 2.637:
        sum_radii *= 0.999999
    return final_coords, np.maximum(radii, 0), sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
