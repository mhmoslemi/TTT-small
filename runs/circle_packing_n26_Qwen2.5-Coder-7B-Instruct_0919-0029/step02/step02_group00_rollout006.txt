import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog, ECOS, OSQP
import cvxpy as cp

def hexagonal_lattice_in_unit_square():
    n = 26
    m = np.sqrt(3 * n / np.pi)
    m = int(np.floor(m))
    k = int(np.floor((n - m * (m - 1) * 3 / 2) / (3 * m / 2)))
    x = np.linspace(k * 1.0 / m, (2 * k + m - 1) * 1.0 / m, m)
    y = np.concatenate([np.zeros(k + 1), np.linspace((k + 1) * 1.0 / (2 * m), (2 * k + 3 * m - 2) * 1.0 / (2 * m), m - 1)])
    center_indices = np.argsort(x**2 + y**2)[:n]
    centers = np.vstack((x[center_indices] * np.cos(np.linspace(0, 2 * np.pi, n, endpoint=False)),
                           y[center_indices] * np.sin(np.linspace(0, 2 * np.pi, n, endpoint=False)))).T
    centers = centers + np.random.normal(0, 0.01, size=centers.shape)
    centers = centers - np.min(centers, axis=0)
    centers = centers / np.max(centers, axis=0)
    return centers

def solve_LP(centers, tol=1e-9):
    n = centers.shape[0]
    objective = np.ones(n)
    A_sparse = np.zeros((n * (n - 1) // 2 + n * 4, n))
    b_sparse = np.ones(n * (n - 1) // 2 + n * 4)
    idx = 0
    
    for i in range(n):
        r = cp.Variable()
        A_sparse[idx, i] = 1
        b_sparse[idx] = centers[i, 0]
        A_sparse[idx + 1, i] = 1
        b_sparse[idx + 1] = 1 - centers[i, 0]
        A_sparse[idx + 2, i] = 1
        b_sparse[idx + 2] = centers[i, 1]
        A_sparse[idx + 3, i] = 1
        b_sparse[idx + 3] = 1 - centers[i, 1]
        idx += 4
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_sparse[idx, i] = 1
            A_sparse[idx, j] = 1
            b_sparse[idx] = dist
            idx += 1
    
    problem = cp.Problem(cp.Maximize(objective @ r), [cp.Minimize(A_sparse @ r - b_sparse) <= 0])
    problem.solve(solver=ECOS, eps=tol, max_iter=2000)
    return r.value

def solve_SOCP(radii, centers):
    n = len(radii)
    objective = cp.Minimize(cp.quad_form(centers, np.eye(n)))
    constraints = [radii[i] <= 0.5 * (centers[i, 0] + 1), radii[i] <= 0.5 * (-centers[i, 0] + 1),
                    radii[i] <= 0.5 * (centers[i, 1] + 1), radii[i] <= 0.5 * (-centers[i, 1] + 1)]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(cp.sqrt(cp.sum_squares(centers[i] - centers[j])) >= radii[i] + radii[j])
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=OSQP, eps=1e-8, max_iter=2000)
    return centers.value

def run_packing():
    centers = hexagonal_lattice_in_unit_square()
    radii = np.zeros(26)
    lam = np.zeros(26)
    rho = 10.0
    obj_prev = -np.inf

    for k in range(30):
        # Radius update (LP)
        r_lp = solve_LP(centers)
        r = r_lp + (lam - rho * r_lp) / rho

        # Centre update (SOCP)
        centers = solve_SOCP(r, centers)

        # Dual update
        lam = lam + rho * (r - r)

        obj = np.sum(r)
        if abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj

    # Final polishing
    r_final = solve_LP(centers)
    return centers, r_final, np.sum(r_final)
