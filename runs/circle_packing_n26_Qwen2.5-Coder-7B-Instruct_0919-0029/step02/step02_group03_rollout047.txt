import numpy as np
from scipy.optimize import linprog, minimize

# Constants
N = 26
BOUNDARY = 1.0
GRID_SIZE = 0.1
INITIAL_STEP_SIZE = 1e-4
MAX_ITERATIONS = 200
NUM_RESTARTS = 10
TOLERANCE = 1e-9

# Helper function to solve the radii for given centers
def solve_radii(centers):
    c = np.zeros(2 * N)
    A = []
    b = []
    for i in range(N):
        A.append([1, 0])
        b.append(centers[i, 0])
        A.append([0, 1])
        b.append(centers[i, 1])
        A.append([-1, -1])
        b.append(-BOUNDARY)
        A.append([-1, 1])
        b.append(-BOUNDARY)
        A.append([1, -1])
        b.append(-BOUNDARY)
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            A.append([1, 1, -d2])
            b.append(0)
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    if res.success:
        return res.x
    else:
        return np.zeros(N)

# Objective function to maximize the sum of radii
def objective(Z):
    centres = Z.reshape((N, 2))
    radii = solve_radii(centres)
    return -radii.sum()

# Gradient of the objective function
def gradient(Z):
    centres = Z.reshape((N, 2))
    radii = solve_radii(centres)
    
    grad = np.zeros(52)
    for k in range(52):
        h = INITIAL_STEP_SIZE
        Z_plus = Z.copy()
        Z_plus[k] += h
        Z_minus = Z.copy()
        Z_minus[k] -= h
        grad[k] = (objective(Z_plus) - objective(Z_minus)) / (2 * h)
    
    return grad

# Run multiple restarts to find the best packing
best_sum_radii = -np.inf
best_centres = None
best_radii = None

for restart in range(NUM_RESTARTS):
    # Initial grid-based packing
    hex_grid = np.zeros((N, 2))
    for i in range(N):
        x = (i % 3) * GRID_SIZE + np.random.normal(0, 0.01)
        y = np.floor(i / 3) * np.sqrt(3) * GRID_SIZE + np.random.normal(0, 0.01)
        hex_grid[i] = [x, y]
    
    # Run L-BFGS-B optimization
    initial_radius = np.random.rand(N) * 0.1
    initial_Z = np.concatenate((hex_grid.flatten(), initial_radius))
    initial_bounds = [(0, 1)] * 52
    
    result = minimize(objective, initial_Z, method='L-BFGS-B', jac=gradient, bounds=initial_bounds, options={'ftol': TOLERANCE, 'maxiter': MAX_ITERATIONS})
    if result.success:
        centres = result.x.reshape((N, 2))
        radii = solve_radii(centres)
        sum_radii = radii.sum()
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centres = centres
            best_radii = radii

# Final adjustments
if best_centres is not None:
    best_centres = np.clip(best_centres, 0, BOUNDARY)
    for i in range(N):
        best_radii[i] = max(best_radii[i], 0)
        while True:
            r_new = solve_radii(best_centres)[i]
            if r_new > best_radii[i]:
                break
            best_centres[i] = np.clip(best_centres[i], 0, BOUNDARY)
            best_radii[i] = r_new
