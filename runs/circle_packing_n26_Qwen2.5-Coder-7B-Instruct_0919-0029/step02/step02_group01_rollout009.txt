import numpy as np
from scipy.linalg import cholesky
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.25, 0.01
    decay_factor = 0.95
    NUM_RESTARTS = 50
    NUM_ITER = 4000
    RADIUS_TOLERANCE = 1e-12

    def initial_grid_plus_center():
        grid_points = [np.array([0.125 + 0.25 * i, 0.125 + 0.25 * j]) for i in range(4) for j in range(4)]
        additional_points = [np.array([0.25 * (i % 2), 0.25 * (j % 2)]) for i in range(10) for j in range(10)]
        return np.array(grid_points + additional_points + [np.array([0.5, 0.5])])

    def gaussian_perturb(centers, sigma=0.02):
        noise = np.random.normal(0, sigma, size=(centers.shape[0], 2))
        return np.clip(centers + noise, 0, 1)

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for seed in range(NUM_RESTARTS):
        np.random.seed(seed)
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        for _ in range(NUM_ITER):
            idx = np.random.randint(n)
            delta = np.random.normal(0, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, np.maximum(final_radii, 0), final_sum
