import numpy as np
from scipy.optimize import minimize, linear_sum_assignment
from scipy.spatial.distance import cdist
from scipy.linalg import lstsq

def hexagonal_lattice seed(num_points):
    a = np.array([1.0, 0.0])
    b = np.array([0.5, np.sqrt(3) / 2.0])
    p = np.outer(range(-1, 2), a) + np.outer(range(-2, 3), b)
    p = p[np.all(np.linalg.norm(p, axis=1)[:, None] > 0.5, axis=0)]
    p = p[np any(np.abs(p[:, 0]) + np.abs(p[:, 1]) < 1.5, axis=1)]
    return p[:num_points]

def high_lambda_penalty(violations, lambda_val=1e7):
    return lambda_val * np.sum(violations**2)

def penalty(objective, penalties):
    return objective + sum(penalties)

def boundary_penalties(z, n):
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    dx, dy = x - 1, y - 1
    x_boundary = np.maximum(0, r - x)
    x_boundary += np.maximum(0, 1 - x - r)
    y_boundary = np.maximum(0, r - y)
    y_boundary += np.maximum(0, 1 - y - r)
    return np.concatenate((x_boundary, y_boundary))

def overlap_penalties(z, n):
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    d = cdist(np.column_stack((x, y)), np.column_stack((x, y)))
    np.fill_diagonal(d, np.inf)
    return np.maximum(0, r[:, None] + r - d)

def combine_penalties(z, n):
    return penalty(np.sum(z[2*n:]), [high_lambda_penalty(boundary_penalties(z, n)), 
                                      high_lambda_penalty(overlap_penalties(z, n))])

def cost_function(z, n):
    return combine_penalties(z, n)

def encode_network(z):
    n = int((np.sqrt(8*z[-1] + 1) + 1) / 2)
    d = cdist(np.column_stack((z[:n], z[n:n*2])), np.column_stack((z[:n], z[n:n*2])))
    np.fill_diagonal(d, np.inf)
    return d < z[n*2:]

def decode_z(zstar):
    n = int((np.sqrt(8*zstar[-1] + 1) + 1) / 2)
    x = zstar[:n]
    y = zstar[n:n*2]
    r = lstsq(x[:, None] + y[:, None], zstar[n*2:], rcond=None)[0]
    return x, y, np.minimum(r, 1 - np.maximum(x, y))

def run_packing():
    n = 26
    seed = hexagonal_lattice_seed(n)
    initial_centers = np.column_stack((seed[:, 0] * 0.5 + 0.25, seed[:, 1] * 0.5 + 0.25))
    initial_radii = np.zeros(n)
    initial_z = np.concatenate((initial_centers.flatten(), initial_radii))

    bounds = [(0, 1)] * n * 2 + [(0, None)] * n
    options = {'maxiter': 300, 'disp': False, 'adaptive': True}

    def objective_with_penalty(z):
        return cost_function(z, n)

    result = minimize(objective_with_penalty, initial_z, bounds=bounds, method='SLSQP', options=options)
    zstar = result.x

    x, y, r = decode_z(zstar)
    return np.column_stack((x, y)), r, np.sum(r)

def validate_solution(centers, radii):
    return validate_packing(centers, radii)

centers, radii, sum_radii = run_packing()
print("Sum of radii:", sum_radii)
print("Centers:\n", centers)
print("Radii:\n", radii)
