import numpy as np
import scipy.optimize as opt

def lp_optimum(centers):
    n = centers.shape[0]
    A = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b = np.zeros(A.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A[idx, i] = 1
        b[idx] = centers[i, 0] - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = centers[i, 1] - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x, -res.fun

def run_packing():
    n = 26
    NUM_ITERS = 4000
    NUM_STARS = 20
    step = 0.1
    step_min = 1e-4
    decay = 0.995
    best_centers, best_radii, best_sum = None, None, 0

    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        cur_centers = centers.copy() 
        cur_sum = 0 
        cur_radii = np.zeros(n)

        while step > step_min:
            for _ in range(NUM_ITERS):
                idx = np.random.randint(n)
                dir = np.random.randn(2) 
                dir /= np.linalg.norm(dir)
                delta = dir * step
                new_c = cur_centers[idx] + delta
                new_c = np.clip(new_c, 0, 1)

                new_radii, new_sum = lp_optimum(cur_centers)
                if new_sum > cur_sum:
                    cur_centers[idx] = new_c
                    cur_sum = new_sum
                    cur_radii = new_radii
            step *= decay

        if cur_sum > best_sum:
            best_centers, best_radii, best_sum = cur_centers.copy(), cur_radii.copy(), cur_sum

    return best_centers, best_radii, best_sum
