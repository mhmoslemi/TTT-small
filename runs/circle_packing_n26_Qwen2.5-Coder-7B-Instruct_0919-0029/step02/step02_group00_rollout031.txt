```python
import numpy as np
import cvxpy as cp
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

# Constants
N = 26
EPSILON = 1e-6
MAX_ITER = 20

# LP for radius optimization
def solve_radius_lp(centers):
    radii = cp.Variable(N, nonneg=True)
    constraints = [radii <= centers[:, 0], radii <= centers[:, 1],
                    radii <= 1 - centers[:, 0], radii <= 1 - centers[:, 1]]
                    # Adjacent distance >= sum of radii
    dists = cdist(centers, centers)
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(dists[i, j] >= radii[i] + radii[j])
    problem = cp.Problem(cp.Maximize(cp.sum(radii)), constraints)
    problem.solve(solver=cp.HS)  # Use high performance linear programming solver
    return radii.value, -problem.value  # Return - because linprog is a minimizer

# SOCP for center optimization (hexagonal positions and quadratic cost)
def solve_center_socp(radii, centers):
    x = cp.Variable(N)
    y = cp.Variable(N)
    objective = cp.Minimize(cp.sum_squares(x - centers[:, 0]) + 
                            cp.sum_squares(y - centers[:, 1]))
    constraints = [radii <= x, radii <= 1 - x,
                    radii <= y, radii <= 1 - y]
                    # Adjacent distance >= sum of radii
    dists = cdist(centers, centers)
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2) >= radii[i] + radii[j])
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS)  # Use ECOS for SOCP
    return np.column_stack((x.value, y.value))

def perturb_hexagonal_centers():
    side_length = np.sqrt(2) / 2
    centers = []
    for r in range(int(np.ceil(N / 3)) + 1):
        center_y = r * side_length
        for c in range(r + 1):
            center_x = (c - (r % 2) / 2) * side_length
            if center_x + side_length / 2 <= 1:
                centers.append([center_x, center_y])
    centers = np.array(centers)
    if centers.shape[0] < N:
        offset = np.array([0.5, 0.25 * np.sqrt(3)])
        centers = np.concatenate([centers, centers + offset], axis=0)
        centers = centers[:N, :]
    return centers

def run_packing():
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    radii_init = np.zeros(N)

    # Create three initial configurations with perturbations
    centers_init = perturb_hexagonal_centers()
    for _ in range(3):
        centers_perturbed = centers_init + np.random.normal(0, 0.01, centers_init.shape)
        centers_perturbed = np.clip(centers_perturbed, 0, 1)

        for _ in range(MAX_ITER):
            radii, sum_radii = solve_radius_lp(centers_perturbed)
            centers_perturbed = solve_center_socp(radii, centers_perturbed)

        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers_perturbed
            best_radii = radii

        # Check if we need to restart due to naive optimization
        if sum_radii < best_sum_radii - EPSILON:
            break

    # Final polishing
    _, final_radii = solve_radius_lp(best_centers)

    return best_centers, final_radii, np.sum(final_radii)
```
