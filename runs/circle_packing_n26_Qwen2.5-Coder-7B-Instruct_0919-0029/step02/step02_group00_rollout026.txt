import numpy as np
from scipy.optimize import linprog, minimize
from cvxpy import Variable, Problem, Minimize, SubjectTo, sum_squares, norm, cones, diag

def build_lp_matrix(centres):
    n = centres.shape[0]
    A = np.zeros((2 * n, n))
    for i in range(n):
        A[2 * i, i] = 1
        A[2 * i + 1, i] = 1
    return A

def solve_lp(centres, r):
    n = centres.shape[0]
    A_ineq = np.vstack([build_lp_matrix(centres),
                          -build_lp_matrix(centres),
                          -build_lp_matrix(centres),
                          -build_lp_matrix(centres)])
    b_ineq = np.hstack([centres[:, 0] - r,
                          1 - (centres[:, 0] - r),
                          centres[:, 1] - r,
                          1 - (centres[:, 1] - r)])
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    if not result.success:
        raise ValueError("LP optimization failed")
    return result.x

def solve_socp(centres, r):
    n = centres.shape[0]
    r = Variable(n)
    c = Variable(n, 2)
    constraints = []
    constraints.append(norm(c[:, 0] - c[:, 1]) >= r + r)
    for i in range(n):
        constraints.append(norm(c[:, i] - np.array([0, 0])) <= r[i])
        constraints.append(norm(c[:, i] - np.array([1, 0])) <= r[i])
        constraints.append(norm(c[:, i] - np.array([0, 1])) <= r[i])
        constraints.append(norm(c[:, i] - np.array([1, 1])) <= r[i])
    obj = Minimize(sum_squares(c[:, 0] - c[:, 1]))
    prob = Problem(obj, constraints)
    prob.solve(solver=cones.SCS, eps=1e-8, max_iters=2000)
    if prob.status != prob.SUCCEEDED:
        raise ValueError("SOCP optimization failed")
    return c.value, r.value

def admm(centres, max_iter=30, rho=10.0, tol=1e-7):
    n = centres.shape[0]
    r_k = np.zeros(n)
    r_c_k = np.zeros(n)
    lambda_k = np.zeros(n)
    obj_prev = -np.inf

    for k in range(max_iter):
        # Radius update
        r_k = solve_lp(centres, r_c_k + lambda_k / rho)
        r_k = r_k + (lambda_k - rho * r_c_k) / rho

        # Centre update
        c_k, r_c_k = solve_socp(centres, r_k)
        lambda_k += rho * (r_k - r_c_k)

        obj = np.sum(r_k)
        if np.abs(obj - obj_prev) < tol:
            break
        obj_prev = obj

    return c_k, r_k, np.sum(r_k)

def run_packing():
    centres = np.random.rand(26, 2)
    return admm(centres)
