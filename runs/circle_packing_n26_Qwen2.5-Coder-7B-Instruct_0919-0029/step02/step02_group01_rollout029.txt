import numpy as np
from scipy.optimize import linprog

def create_initial_grid_and_add_center():
    step = 0.20
    grid_points = np.arange(step, 1, step)
    centers = np.array([(x, y) for x in grid_points for y in grid_points] + [(0.5, 0.5)])
    return centers

def perturb(centers, epsilon):
    n = len(centers)
    centers += np.random.uniform(-epsilon, epsilon, size=(n, 2))
    centers = np.clip(centers, 0, 1)
    return centers

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise RuntimeError("LP solver failed")
    radii = -result.fun
    return radii, result.fun

def move_center(centers, k, step):
    new_centers = centers.copy()
    new_c = new_centers[k] + step * np.random.random(2) / 2
    new_c = np.clip(new_c, 0, 1)
    new_centers[k] = new_c
    return new_centers

def random_accept(delta, temperature):
    if delta > 0:
        return True
    return np.random.rand() < np.exp(delta / temperature)

NUM_RESTARTS = 30
NUM_ITER = 4000
TAU_STEP = 0.997
TAU_TEMP = 0.995

def run_packing():
    best_sum = -np.inf
    best_centers = None
    for restart in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb(centers, 0.05)

        radii, sum_r = solve_lp(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(len(centers))
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = solve_lp(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_sum = centers.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

    best_radii, _ = solve_lp(best_centers)
    return best_centers, best_radii, best_sum
