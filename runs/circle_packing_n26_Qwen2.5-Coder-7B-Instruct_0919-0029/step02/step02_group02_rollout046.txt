def penaltyfunc(x, alpha, beta):
    n = x.shape[0] // 3
    circles, radii = x[:n], x[n:2*n]
    coords = circles.reshape(n, 2)
    penalty = 0

    # Distance penalties
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(coords[i] - coords[j])
            penalty += np.maximum(0, radii[i] + radii[j] - d) ** 2 / alpha

    # Boundary penalties
    for i in range(n):
        r, x_i, y_i = radii[i], coords[i, 0], coords[i, 1]
        penalty += np.maximum(0, r - x_i) ** 2 / beta + \
                   np.maximum(0, r - (1 - x_i)) ** 2 / beta + \
                   np.maximum(0, r - y_i) ** 2 / beta + \
                   np.maximum(0, r - (1 - y_i)) ** 2 / beta

    return penalty
