import numpy as np
from scipy.optimize import differential_evolution, trust_constr

def objective(p, A_constraints, b_constraints, lambda_penalty):
    radii = p[26:]
    sum_radii = -np.sum(radii)
    sum_penalties = lambda_penalty * np.sum(np.maximum(0, A_constraints @ p - b_constraints)**2)
    return sum_radii + sum_penalties

def gradient(p, A_constraints, b_constraints, lambda_penalty):
    n = 26
    X, Y, r = p[:26], p[26:26 + n], p[-n:]
    grad_sum_radii = -np.ones(len(r))
    grad_penalties = lambda_penalty * 2 * A_constraints.T @ np.maximum(0, A_constraints @ p - b_constraints)
    grad_x = np.zeros(26)
    grad_y = np.zeros(26)
    for i in range(n):
        x_i, y_i = X[i], Y[i]
        for j in range(i + 1, n):
            x_j, y_j = X[j], Y[j]
            dist_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            penalty_ij = dist_ij - (r[i] + r[j])
            if penalty_ij < 0:
                continue
            grad_x[i] += (x_i - x_j) / dist_ij
            grad_y[i] += (y_i - y_j) / dist_ij
            grad_x[j] -= (x_i - x_j) / dist_ij
            grad_y[j] -= (y_i - y_j) / dist_ij
    return np.concatenate([grad_x, grad_y, grad_sum_radii + grad_penalties])
