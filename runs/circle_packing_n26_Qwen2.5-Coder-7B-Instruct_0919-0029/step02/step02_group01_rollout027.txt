import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[i*(n-i)//2 + j-i-1, i] = 1
            A_ub[i*(n-i)//2 + j-i-1, j] = 1
            b_ub[i*(n-i)//2 + j-i-1] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
    
    for i in range(n):
        A_ub[n*(n-1)//2 + i, i] = 1
        b_ub[n*(n-1)//2 + i] = centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + n + i, i] = 1
        b_ub[n*(n-1)//2 + n + i] = 1 - centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + 2*n + i, i] = 1
        b_ub[n*(n-1)//2 + 2*n + i] = centers[i, 1] - 1e-12
        A_ub[n*(n-1)//2 + 3*n + i, i] = 1
        b_ub[n*(n-1)//2 + 3*n + i] = 1 - centers[i, 1] - 1e-12
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    np.random.seed(42)
    n = 26
    step_start = 0.1
    step_min = 1e-4
    decay_factor = 0.995
    NUM_ITER_PER_RESTART = 1000
    NUM_STARS = 20
    
    best_sum = -np.inf
    best_centers, best_radii = None, None
    
    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        new_centers = centers.copy()
        new_radii, new_sum = lp_optimum(centers)
        
        cur_centers = new_centers.copy()
        cur_radii = new_radii.copy()
        cur_sum = new_sum
        
        step = step_start
        for iter in range(NUM_ITER_PER_RESTART):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            cur_centers[idx] += delta
            cur_centers[idx] = np.clip(cur_centers[idx], 0, 1)
            
            new_radii, new_sum = lp_optimum(cur_centers)
            
            if new_sum > cur_sum:
                cur_centers = cur_centers.copy()
                cur_sum = new_sum
                cur_radii = new_radii.copy()
            
            step = max(step_min, step * decay_factor)
        
        if best_sum < cur_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()
    
    final_radii, final_sum = lp_optimum(best_centers)
    
    return best_centers, final_radii, final_sum
