import numpy as np
import cvxpy as cp

def compute_sum_radii(centers):
    radii = np.sqrt((centers[:,0][None] - centers[:,0][:,None])**2 + (centers[:,1][None] - centers[:,1][:,None])**2)
    np.fill_diagonal(radii, 2)
    return np.sum(np.maximum(radii - centers[:,2][:,None] - centers[:,2][None,:], 0))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N)
    
    constraints = [r[i] >= 0 for i in range(N)]
    constraints += [x[i] + r[i] <= 1 for i in range(N)]
    constraints += [x[i] - r[i] >= 0 for i in range(N)]
    constraints += [y[i] + r[i] <= 1 for i in range(N)]
    constraints += [y[i] - r[i] >= 0 for i in range(N)]
    
    for i in range(N):
        for j in range(i + 1, N):
            constraints += [cp.norm(cp.array([x[i], y[i]] - cp.array([x[j], y[j]]))) >= r[i] + r[j]]
    
    objective = cp.Maximize(cp.sum(r))
    
    prob = cp.Problem(objective, constraints)
    
    if prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-7, reltol=1e-7) > 0:
        centers = np.vstack((x.value, y.value)).T
        radii = np.clip(r.value, 0, None)
        sum_radii = compute_sum_radii(centers)
        lp_solve_constraints = np.abs(np.sum(radii)**2 - np.sum(compute_sum_radii(centers)**2))
        if lp_solve_constraints < 1e-6:
            return centers, radii, sum_radii
    
    if prob.solve(solver=cp.SCS, verbose=False) > 0:
        centers = np.vstack((x.value, y.value)).T
        radii = np.clip(r.value, 0, None)
        sum_radii = compute_sum_radii(centers)
        lp_solve_constraints = np.abs(np.sum(radii)**2 - np.sum(compute_sum_radii(centers)**2))
        if lp_solve_constraints < 1e-6:
            return centers, radii, sum_radii
    
    # If both solvers fail, fall back to differential evolution
    return run_packing()

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
