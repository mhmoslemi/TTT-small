import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    step, step_min = 0.1, 1e-6
    decay_factor = 0.3
    NUM_ITERS = 300
    NUM_STARS = 2

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        if start == 0:
            centers = np.array([
                [0.25, 0.25], [0.5, 0.25], [0.75, 0.25],
                [0.25, 0.5],  [0.5, 0.5],   [0.75, 0.5],
                [0.25, 0.75], [0.5, 0.75],  [0.75, 0.75]
            ])
        else:
            centers = np.random.rand(n, 2) + np.array([0.1, 0.1])

        for it in range(NUM_ITERS):
            grad = np.zeros(2*n)
            xi, yi, wi = centers[:,0], centers[:,1], 1e-5

            dX = (xi[:, np.newaxis] + wi - xi[np.newaxis, :]) / np.maximum(np.linalg.norm(xi[:, np.newaxis] - xi[np.newaxis, :], axis=2), 1e-12)
            dY = (yi[:, np.newaxis] + wi - yi[np.newaxis, :]) / np.maximum(np.linalg.norm(yi[:, np.newaxis] - yi[np.newaxis, :], axis=2), 1e-12)
            for i in range(n):
                idx = np.where(np.linalg.norm(xi - xi[i], axis=0) < 1)  # Find nearest neighbors
                wi_i = 1 - np.abs(xi[i] - xi[idx])  # Weight by inverse distance
                grad[(i, 0)] += wi_i * np.sum(dX[i] * np.clip(centers[idx, 0] - failcheck(xi[i]}, 0, 1) - centers[i, 0], 0, 1))
                grad[2*i+1] += wi_i * np.sum(dY[i] * np.clip(centers[idx, 1] - failcheck(yi[i]}, 0, 1) - centers[i, 1], 0, 1))

                grad[(i, 0)] += (wi * np.clip(centers[:, 0] - xi[i], 0, 1) - xi[i]) / centers[:, 0] * wi
                grad[(i, 1)] += (wi * np.clip(centers[:, 1] - yi[i], 0, 1) - yi[i]) / centers[:, 1] * wi

            dx = grad[:n] * step
            dy = grad[n:] * step
            new_centers =centers + np.array([dx, dy]).T
            new_centers = np.clip(new_centers, 0, 1)

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum

def failcheck(x):
    if any(x < 0) or any(x > 1):
        raise ValueError("Out of bounds")
    return x
