import numpy as np
import cvxpy as cp

def f_local(X):
    N = int(len(X) / 3)
    x = X[0:N]
    y = X[N:2*N]
    r = X[2*N:]
    sum_r = np.sum(r)
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            if dist < r[i] + r[j]:
                return np.inf
        if r[i] - x[i] < 1e-12 or r[i] - y[i] < 1e-12 or r[i] - (1 - x[i]) < 1e-12 or r[i] - (1 - y[i]) < 1e-12:
            return np.inf
    return -sum_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    centers = np.random.rand(N, 2) * 0.98 + 0.01
    radii = np.zeros(N)
    sum_radii = 0.0
    for attempt in range(10):
        x = centers[:, 0]
        y = centers[:, 1]
        r = np.array(radii)
        prob = cp.Problem(cp.Maximize(cp.sum(r)), 
                          [r >= 0, 
                           r <= x, 
                           r <= 1 - x, 
                           r <= y, 
                           r <= 1 - y, 
                           cp.norm(cp.hstack((x[i] - x[j] for j in range(i+1, N))), 2) >= cp.sum(r[i] + r[j] for i in range(i+1, N)) for i in range(N)])
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-7, reltol=1e-7)
        if prob.status == cp.OPTIMAL:
            x = x.value
            y = y.value
            r = r.value
            Radii = np.clip(r, 0, None)
            centers = np.vstack((x, y)).T
            sum_radii = np.sum(Radii)
        else:
            centers = np.random.rand(N, 2) * 0.98 + 0.01
            radii = np.zeros(N)
            sum_radii = 0.0
    return centers, np.clip(radii, 0, None), sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
