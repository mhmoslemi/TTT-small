import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def hexagonal_lattice(size):
    a = np.array([1, 0])
    b = np.array([0.5, np.sqrt(3)/2])
    points = []
    for i in range(-1, size+2):
        for j in range(-1, size+2):
            p = i*a + j*b
            if 0 <= p.dot(a) <= size and 0 <= p.dot(b) <= size:
                points.append(p)
    return np.array(points)

def penalty_function(z, lambda_min, lambda_max, lambda_step):
    n = 26
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    penalty = -np.sum(r)
    for i in range(n):
        penalty += lambda_min * np.sum(np.maximum(x - r[i], 0)) \
                   + lambda_min * np.sum(np.maximum(x + r[i] - 1, 0)) \
                   + lambda_min * np.sum(np.maximum(y - r[i], 0)) \
                   + lambda_min * np.sum(np.maximum(y + r[i] - 1, 0)) \
                   + lambda_min * np.sum(np.maximum(np.sqrt(squareform(pdist(np.column_stack((x, y))), squared=True) - (r + r[:, np.newaxis])), 0).reshape(-1))
    return penalty

def pso_objective(z, lambda_min, lambda_max, lambda_step):
    return penalty_function(z, lambda_min, lambda_max, lambda_step)

def solve_radius_lp_z(z):
    n = 26
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    At = []
    b = []
    for i in range(n):
        At.append([-1, 0, 0])
        b.append(-x[i] + 1e-12)
        At.append([0, -1, 0])
        b.append(-y[i] + 1e-12)
        At.append([0, 0, 1])
        b.append(-r[i] + 1e-12)
        At.append([0, 0, -1])
        b.append(r[i] + 1e-12)

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            At.append([0, 0, 0, 0, 0, 0, 2*(x[i] - x[j]), 2*(y[i] - y[j]), -2])
            b.append(dist + x[i] + x[j] -y[i] - y[j] - 1e-12)

    At = np.array(At)
    b = np.array(b)

    c = np.concatenate([np.zeros(2*26), -np.ones(26)])
    bounds = [(0, None)] * 2 * 26 + [(0, None)] * 26

    result = linprog(c, A_ub=At, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        raise ValueError("LP optimization failed")

    r_opt = result.x[-26:]
    return r_opt, -result.fun

def run_packing():
    size = 4
    centers = hexagonal_lattice(size)[:26]
    z = np.hstack((centers.flatten(), np.zeros(26)))
    lambda_min = 1e4
    lambda_max = 1e7
    lambda_step = 1e3

    for _ in range(2):
        bounds = [(0, None), (0, None)] * 2 * len(centers) + [(0, None), (0, None)] * len(centers)
        options = {'maxiter': 300}
        res = minimize(pso_objective, z, args=(lambda_min, lambda_max, lambda_step), bounds=bounds, options=options, method='L-BFGS-B')
        z = res.x
        r_opt, sum_radii = solve_radius_lp_z(z)
        z = np.hstack((centers.flatten(), r_opt))

    centers = z[:26].reshape(26, 2)
    return centers, r_opt, sum_radii

if __name__ == '__main__':
    centers, radii, sum_radii = run_packing()
    print(f"Centers:\n{centers}")
    print(f"Radii:\n{radii}")
    print(f"Sum of radii: {sum_radii}")
