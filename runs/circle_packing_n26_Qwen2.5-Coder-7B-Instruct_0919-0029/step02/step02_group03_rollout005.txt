import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    radii = np.zeros(N)
    A_ub = []
    b_ub = []
    for i in range(N):
        A_ub.append(np.zeros(N))
        A_ub[-1][i] = 1
        b_ub.append(centers[i, 0]) # x_i <= 1-x_i
        A_ub.append(np.zeros(N))
        A_ub[-1][i] = 1
        b_ub.append(centers[i, 1]) # y_i <= 1-y_i
        A_ub.append(np.zeros(N))
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i, 0]) # 1-x_i
        A_ub.append(np.zeros(N))
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i, 1]) # 1-y_i
    for i in range(N):
        for j in range(i+1, N):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub.append(np.zeros(N))
            A_ub[-1][i] = 1
            A_ub[-1][j] = 1
            b_ub.append(dist)
    A_ub = np.vstack(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(N)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if res.success:
        return res.x, res.x.sum()
    else:
        return np.zeros(N), -np.inf

def objective(X):
    centers = X[:52].reshape((N, 2))
    _, radii_sum = compute_sum_radii(centers)
    return -radii_sum

def simulated_annealing(centers, max_iter=4000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum, _ = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum, _ = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing():
    bounds = [(0, 1)] * 52

    de_result = differential_evolution(objective, bounds, popsize=10, maxiter=200, strategy='best1bin', tol=1e-6, seed=42)
    
    best_centers = de_result.x[:52].reshape((N, 2))
    best_sum = -de_result.fun
    
    final_centers = simulated_annealing(best_centers, max_iter=4000)
    _, final_sum = compute_sum_radii(final_centers)
    
    return final_centers, final_centers[:, 2], final_sum

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
