import numpy as np
from scipy.optimize import differential_evolution, minimize, NonlinearConstraint

def f_global(X):
    N = int(len(X) // 3)
    radii = X[2::3]
    penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((X[i * 3: i * 3 + 2] - X[j * 3: j * 3 + 2]) ** 2)
            penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
        penalty += max(radii[i] - X[i * 3], 0) ** 2
        penalty += max(radii[i] - X[i * 3 + 1], 0) ** 2
        penalty += max(radii[i] - (1 - X[i * 3]), 0) ** 2
        penalty += max(radii[i] - (1 - X[i * 3 + 1]), 0) ** 2
    return -(radii.sum()) + 1e6 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    result = differential_evolution(f_global, bounds, popsize=30, maxiter=300, seed=42, workers=1)
    
    X = result.x
    centers = X.reshape((N, 3))[:, :2]
    radii = X.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    
    def f_local(X):
        return -(radii.sum())
    
    def cons(X):
        N = int(len(X) // 3)
        constraints = []
        for i in range(N):
            constraints.append(np.sum(X[i * 3: i * 3 + 2] ** 2) - (radii[i] ** 2))
            constraints.append(1 - np.sum(X[i * 3: i * 3 + 2] ** 2) - (radii[i] ** 2))
        for i in range(N-1):
            for j in range(i+1, N):
                d2 = np.sum((X[i * 3: i * 3 + 2] - X[j * 3: j * 3 + 2]) ** 2)
                constraints.append(d2 - (radii[i] + radii[j]) ** 2)
        return constraints
    
    nlc = NonlinearConstraint(cons, 0, np.inf)

    result = minimize(f_local, X, method='SLSQP', bounds=bounds, constraints=nlc, options={'ftol': 1e-8, 'maxiter': 500, 'iprint': 0})
    X = result.x
    centers = X.reshape((N, 3))[:, :2]
    radii = X.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    
    if sum_radii <= 2.637:
        for i in range(N):
            radii[i] *= 0.999999
        sum_radii = radii.sum()
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
