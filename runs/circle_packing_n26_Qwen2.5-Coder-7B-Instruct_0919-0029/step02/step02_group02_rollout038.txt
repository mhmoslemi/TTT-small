import numpy as np
from cvxpy import Variable, Problem, Maximize, sum, Parameter, quad_form, sqrt, perturb
from scipy.spatial import Delaunay, ConvexHull
from scipy.optimize import line_search

def lp_optimum(centers):
    n = centers.shape[0]
    r = Variable(n)
    d = np.empty((n, n))
    for i in range(n):
        for j in range(i+1, n):
            d[i, j] = np.linalg.norm(centers[i, :] - centers[j, :])
            d[j, i] = d[i, j]
    dual_pairs = []
    for i in range(n):
        r_i = r[i]
        for j in range(i+1, n):
            r_j = r[j]
            distance = d[i, j]
            dual_pairs.append((r_i + r_j <= distance, distance))
    objective = Maximize(sum(r))
    subject_to = [x <= 1 for x in r] + [x >= 0 for x in r]
    for r_i, distance in dual_pairs:
        subject_to.append(r_i)
    prob = Problem(objective, subject_to)
    prob.solve(solver='ECOS', verbose=False)
    return r.value, prob.value

def compute_gradient(centers, radii):
    n = centers.shape[0]
    r = Variable(n)
    d = np.empty((n, n))
    for i in range(n):
        for j in range(i+1, n):
            d[i, j] = np.linalg.norm(centers[i, :] - centers[j, :])
            d[j, i] = d[i, j]
    dual_pairs = []
    for i in range(n):
        r_i = r[i]
        for j in range(i+1, n):
            r_j = r[j]
            distance = d[i, j]
            dual_pairs.append((r_i + r_j <= distance, distance))
    objective = Maximize(sum(r))
    subject_to = [x <= 1 for x in r] + [x >= 0 for x in r]
    for r_i, distance in dual_pairs:
        subject_to.append(r_i)
    prob = Problem(objective, subject_to)
    prob.solve(solver='ECOS', dual_eps=1e-9, primal_eps=1e-9)
    dual_values = prob.dual_value
    grad = np.zeros(52)
    for i in range(n):
        x, y = centers[i, :]
        for j in range(i+1, n):
            distance = d[i, j]
            dual_val = dual_values[dual_pairs.index((r[i] + r[j]<=distance, distance))]
            if dual_val > 0:
                dx = (x[j] - x[i]) / distance
                dy = (y[j] - y[i]) / distance
                grad[2*i] += (dual_val * dx)
                grad[2*i+1] += (dual_val * dy)
            if x[i] > x[j]:
                grad[2*i] += dual_values[dual_pairs.index((r[i]<=x[i], r[i]))]
                grad[2*i] -= dual_values[dual_pairs.index((r[i]<=1-x[i], r[i]))]
        if x[i] < 1e-12:
            grad[2*i] += dual_values[dual_pairs.index((r[i]<=x[i], r[i]))]
        if abs(1 - x[i]) < 1e-12:
            grad[2*i] -= dual_values[dual_pairs.index((r[i]<=1-x[i], r[i]))]
        if y[i] > y[j]:
            grad[2*i+1] += dual_values[dual_pairs.index((r[i]<=y[i], r[i]))]
            grad[2*i+1] -= dual_values[dual_pairs.index((r[i]<=1-y[i], r[i]))]
        if y[i] < 1e-12:
            grad[2*i+1] += dual_values[dual_pairs.index((r[i]<=y[i], r[i]))]
        if abs(1 - y[i]) < 1e-12:
            grad[2*i+1] -= dual_values[dual_pairs.index((r[i]<=1-y[i], r[i]))]
    return grad / np.linalg.norm(grad)

def run_packing():
    n = 26
    step = 0.1
    step_min = 1e-6
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

            grad = compute_gradient(new_centers, new_radii)
            alpha = line_search(lp_optimum, compute_gradient, new_centers, -grad)[0]
            alpha = max(alpha, step_min / np.linalg.norm(grad))
            new_centers -= alpha * grad
            new_centers = np.clip(new_centers, 0, 1)

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
