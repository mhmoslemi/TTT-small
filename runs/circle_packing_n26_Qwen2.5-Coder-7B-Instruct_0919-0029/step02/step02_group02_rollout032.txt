import numpy as np
from scipy.optimize import highs

def lp_optimal_radii(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist2 = np.sum((centers[i] - centers[j]) ** 2)
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.sqrt(dist2) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = highs(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
    duals = res['xn'][n:]
    return duals, -res['optimal_value']

def compute_gradient(centers, duals):
    n = centers.shape[0]
    gradient = np.zeros((n, 2))
    for i in range(n):
        dist2 = np.zeros(n)
        for j in range(n):
            if j != i:
                dist2[j] = np.sum((centers[i] - centers[j]) ** 2)
        dist2 = np.sqrt(dist2)
        valid_indices = dist2 > 1e-12
        avg_dist = np.mean(dist2[valid_indices])
        gradient[i, 0] = np.sum((centers[i, 0] - centers[valid_indices, 0]) * duals[valid_indices] / dist2[valid_indices] / avg_dist)
        gradient[i, 0] += duals[i] if centers[i, 0] - 1e-12 <= 0 else 0
        gradient[i, 0] -= duals[i] if 1 - centers[i, 0] - 1e-12 <= 0 else 0
        gradient[i, 1] = np.sum((centers[i, 1] - centers[valid_indices, 1]) * duals[valid_indices] / dist2[valid_indices] / avg_dist)
        gradient[i, 1] += duals[i] if centers[i, 1] - 1e-12 <= 0 else 0
        gradient[i, 1] -= duals[i] if 1 - centers[i, 1] - 1e-12 <= 0 else 0
    return gradient

def project_to_bounds(centres):
    row, col = np.where(centres < 0)
    centres[row, col] = 0
    row, col = np.where(centres > 1)
    centres[row, col] = 1
    return centres

def jitter(centres, std_dev=0.005):
    return centres + np.random.normal(0, std_dev, centres.shape)

def run_packing():
    n = 26
    initial_patterns = [
        (np.arange(5).astype(float) * 0.2 + 0.1, np.linspace(0.1, 0.9, 5), np.arange(5).astype(float) * 0.2 + 0.1),
        (np.arange(6).astype(float) * 0.2 + 0.2, np.linspace(0.1, 0.8, 6), np.linspace(0.2, 1.0, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.3, np.linspace(0.1, 0.7, 6), np.linspace(0.3, 0.9, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.4, np.linspace(0.1, 0.6, 6), np.linspace(0.4, 0.8, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.5, np.linspace(0.1, 0.5, 6), np.linspace(0.5, 0.7, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.6, np.linspace(0.1, 0.4, 6), np.linspace(0.6, 0.6, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.7, np.linspace(0.2, 0.3, 6), np.linspace(0.7, 0.5, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.8, np.linspace(0.3, 0.2, 6), np.linspace(0.8, 0.3, 6)),
        (np.arange(6).astype(float) * 0.2 + 0.9, np.linspace(0.4, 0.1, 6), np.linspace(0.9, 0.1, 6)),
    ]
    best_sum = -np.inf
    best_centers = None

    for pattern in initial_patterns:
        x = jitter(np.meshgrid(pattern[0], pattern[1])[0])
        y = jitter(np.meshgrid(pattern[0], pattern[2])[1])
        centers = np.vstack((x.flatten(), y.flatten())).T
        for it in range(500):
            duals, current_sum = lp_optimal_radii(centers)
            grad = compute_gradient(centers, duals)
            step_size = 0.1
            while True:
                new_centers = project_to_bounds(centers + step_size * grad)
                new_duals, new_sum = lp_optimal_radii(new_centers)
                if new_sum > current_sum:
                    centers = new_centers
                    duals = new_duals
                    current_sum = new_sum
                    step_size *= 1.05
                    break
                step_size *= 0.9
            if step_size < 1e-6:
                break

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers

    radii, _ = lp_optimal_radii(best_centers)
    return best_centers, radii, best_sum
