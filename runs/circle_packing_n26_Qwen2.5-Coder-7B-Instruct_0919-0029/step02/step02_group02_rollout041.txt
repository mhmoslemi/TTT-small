import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def hill_climb(centers):
    n = centers.shape[0]
    step = 0.01
    decay_factor = 0.99
    NUM_ITERS = 1000

    best_sum = -np.inf
    best_centers = centers

    for it in range(NUM_ITERS):
        centers = best_centers.copy()
        for i in range(n):
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[i] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[i] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers

        step = max(step * decay_factor, 1e-8)

    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)

    # Differential Evolution optimization
    bounds = [(0, 1)] * (2 * n)
    result = differential_evolution(lambda x: -lp_optimum(x.reshape(-1, 2))[1], bounds, popsize=50, maxiter=500)
    centers = result.x.reshape(-1, 2)

    # Local hill-climbing refinement
    centers, sum_radii = hill_climb(centers)

    # Compute radii
    _, radii = lp_optimum(centers)

    return centers, radii, sum_radii
