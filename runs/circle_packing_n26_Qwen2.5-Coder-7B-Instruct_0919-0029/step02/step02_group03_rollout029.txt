import cvxpy as cp
import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N, nonneg=True)

    # Boundary containment
    constraints = [r[i] <= x[i] for i in range(N)]
    constraints += [r[i] <= 1 - x[i] for i in range(N)]
    constraints += [r[i] <= y[i] for i in range(N)]
    constraints += [r[i] <= 1 - y[i] for i in range(N)]
    
    # Non-overlap constraints
    for i in range(N):
        for j in range(i+1, N):
            constraints += [cp.norm([x[i] - x[j], y[i] - y[j]]) >= r[i] + r[j]]
    
    # Objective
    objective = cp.Maximize(cp.sum(r))
    
    # Solve with ECOS, fallback to SCS if ECOS fails
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-7, reltol=1e-7)
        if np.isinf(prob.value):
            raise Exception("ECOS returned infinity; falling back to SCS.")
    except Exception:
