import numpy as np
from scipy.optimize import differential_evolution, NonlinearConstraint, LinearConstraint, Bounds, trust_constr

def objective(p):
    x, y, r = p[:26], p[26:52], p[52:]
    return -np.sum(r) + 1e8 * np.sum(np.maximum(0, r - x) + np.maximum(0, r - y) + np.maximum(0, r - (1 - x)) + np.maximum(0, r - (1 - y)))

def gradient(p):
    x, y, r = p[:26], p[26:52], p[52:]
    n = len(r)
    grad = np.zeros_like(p)
    for i in range(n):
        edge_term = np.array([np.gradient(np.maximum(0, r[i] - x[j]))[0] for j in range(n) if j != i])
        edge_term += np.array([np.gradient(np.maximum(0, r[i] - y[j]))[0] for j in range(n) if j != i])
        edge_term += np.array([np.gradient(np.maximum(0, r[i] - (1 - x[j]))) for j in range(n) if j != i])
        edge_term += np.array([np.gradient(np.maximum(0, r[i] - (1 - y[j]))) for j in range(n) if j != i])
        grad[:26] += edge_term
        grad[26:52] += np.where(r[i] > x[i], 1, 0)
        grad[26:52] += np.where(r[i] > y[i], 1, 0)
        grad[26:52] += np.where(r[i] > (1 - x[i]), 1, 0)
        grad[26:52] += np.where(r[i] > (1 - y[i]), 1, 0)
    return grad

def constraints(p):
    x, y, r = p[:26], p[26:52], p[52:]
    n = len(r)
    return [np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2) - (r[i] + r[j]) for i in range(n) for j in range(i + 1, n)]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)

    # Initial guess
    x0 = np.concatenate([np.linspace(0.1, 0.9, 5)**2 * 0.5 + 0.05, np.linspace(0.1, 0.9, 5)**2 * 0.5 + 0.05, np.full(5, 0.0999)])
    x0 = np.concatenate([x0, [0.5], [0.0999]])

    # Nonlinear constraints
    nl_cons = [NonlinearConstraint(constraints, 0, np.inf)]

    # Bounds on R and X Y
    r_upper = np.minimum(np.minimum(x0[:26], 1 - x0[:26]), np.minimum(y0[:26], 1 - y0[:26]))
    bounds = Bounds(0, 1)
    for _ in range(26):
        bounds = bounds + Bounds(0, r_upper[_])

    # Run DE
    res_de = differential_evolution(objective, bounds, args=(), constraints=nl_cons, popsize=15, max_iter=400)
    p_DE = res_de.x

    # Run trust-constr refinement
    res_constr = trust_constr(objective, p_DE, args=(), bounds=bounds, constraints=nl_cons, jac=gradient, callback=None)

    # Transform back to centers and radii
    centers = res_constr.x[:52].reshape(-1, 2)
    radii = res_constr.x[52:]
    sum_radii = -res_constr.fun

    return centers, radii, sum_radii
