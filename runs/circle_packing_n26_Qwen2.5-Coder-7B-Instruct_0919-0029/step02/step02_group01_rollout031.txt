import numpy as np
from scipy.optimize import linprog

def lp_solve(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_grid_plus_center():
    centers = np.array([(i + 0.125) * 0.25 for i in range(4)] * 4)
    centers = centers.reshape((4, 4, 2)).swapaxes(0, 2)
    centers = centers.flatten()
    centers = centers[:25]  # Remove the central point
    centers = np.append(centers, 0.5)
    centers = centers.reshape((26, 2))
    return centers

def gaussian_perturb(centers, sigma=0.02):
    noise = np.random.normal(0, sigma, size=centers.shape)
    new_centers = np.clip(centers + noise, 0, 1)
    return new_centers

def run_packing():
    N_restart = 40
    N_iter = 4000
    step_min = 1e-4
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(N_restart):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        for it in range(N_iter):
            k = np.random.randint(26)
            new_centers = centers.copy()
            new_centers[k] += np.random.normal(0, step_min)
            new_centers[k] = np.clip(new_centers[k], 0, 1)

            new_radii, new_sum = lp_solve(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

            step_min = max(step_min, step_min * 0.995)

    best_radii, best_sum = lp_solve(best_centers)
    best_radii = np.maximum(best_radii, 0)
    best_radii = np.round(best_radii, 10)
    return best_centers, best_radii, best_sum
