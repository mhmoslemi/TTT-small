import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    m = n*(n-1)//2 + 4*n
    A = np.empty((m, n))
    b = np.empty(m)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i]-centers[j]) - 1e-12
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1; b[idx] = x - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = 1 - x - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = y - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = 1 - y - 1e-12; idx += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x, -res.fun

def proposal_move(centers):
    idx = np.random.randint(centers.shape[0])
    delta = np.random.uniform(-0.05, 0.05, size=2)
    new_c = centers[idx] + delta
    new_c = np.clip(new_c, 0, 1)
    return idx, new_c

def simulated_annealing(centers, step, n_iters, T0):
    best_centers, best_sum = centers.copy(), None
    cur_centers = centers.copy()
    cur_radii, cur_sum = lp_optimum(centers)
    best_sum = cur_sum
    T = T0
    for k in range(n_iters):
        idx, new_c = proposal_move(cur_centers)
        cur_centers[idx] = new_c
        new_radii, new_sum = lp_optimum(cur_centers)
        if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / T):
            cur_centers, cur_sum = new_centers, new_sum
            if new_sum > best_sum:
                best_centers, best_sum = new_centers, new_sum
        T *= 0.99
    return best_centers, best_sum

def run_packing():
    np.random.seed(42)
    NUM_STARS = 20
    NUM_ITERS = 4000
    step = 0.1
    best_centers, best_radii, best_sum = None, None, -np.inf

    for seed in range(NUM_STARS):
        np.random.seed(seed)
        centers = np.random.rand(26, 2)
        cur_centers, cur_sum = simulated_annealing(centers, step, NUM_ITERS, 1.0)
        if cur_sum > best_sum:
            best_centers, best_sum = cur_centers, cur_sum
            radius, _ = lp_optimum(best_centers)
            best_radii = radius

    return best_centers, best_radii, best_sum
