import numpy as np
import scipy.optimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def gradient_function(x, alpha=10**4, beta=10**6):
    n = int(np.sqrt(x.shape[0]))
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    n_pairs = n * (n - 1) // 2
    n_sides = n

    dist_penalty = np.zeros(n_pairs)
    boundary_penalty = np.zeros(n_sides)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx * dx + dy * dy)
            p_ij = max(0, radii[i] + radii[j] - dist)
            dist_penalty[idx] = p_ij
            idx += 1

    idx = 0
    for i in range(n):
        p_L = max(0, radii[i] - centers[i, 0])
        p_R = max(0, radii[i] - (1 - centers[i, 0]))
        p_B = max(0, radii[i] - centers[i, 1])
        p_T = max(0, radii[i] - (1 - centers[i, 1]))
        boundary_penalty[idx] = p_L
        boundary_penalty[idx + 1] = p_R
        boundary_penalty[idx + 2] = p_B
        boundary_penalty[idx + 3] = p_T
        idx += 4

    dist_grad = np.zeros(2*n)
    boundary_grad = np.zeros(2*n)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist > 1e-12:
                dist_grad[2*i] += 2 * alpha * dist_penalty[idx] * dx / dist
                dist_grad[2*i + 1] += 2 * alpha * dist_penalty[idx] * dy / dist
                dist_grad[2*j] += 2 * alpha * dist_penalty[idx] * dx / dist
                dist_grad[2*j + 1] += 2 * alpha * dist_penalty[idx] * dy / dist
            idx += 1

    idx = 0
    for i in range(n):
        radii_idx = 2*n + i
        p_L = boundary_penalty[idx]
        p_R = boundary_penalty[idx + 1]
        p_B = boundary_penalty[idx + 2]
        p_T = boundary_penalty[idx + 3]
        boundary_grad[2*i] += 2 * beta * p_L * (radii[i] - centers[i, 0])
        boundary_grad[2*i + 1] += 2 * beta * p_L * dy
        boundary_grad[2*i] += 2 * beta * p_R * (radii[i] - (1 - centers[i, 0]))
        boundary_grad[2*i + 1] += 2 * beta * p_R * dy
        boundary_grad[2*i] += 2 * beta * p_B * (radii[i] - centers[i, 1])
        boundary_grad[2*i + 1] += 2 * beta * p_B * dx
        boundary_grad[2*i] += 2 * beta * p_T * (radii[i] - (1 - centers[i, 1]))
        boundary_grad[2*i + 1] += 2 * beta * p_T * dx
        idx += 4

    grad = dist_grad + boundary_grad - np.ones(2*n)
    return grad

def run_packing():
    n = 26
    centers = np.zeros(2*n)
    fontsize = 6

    font_title = 16
    
    n_attempts = 10
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for attempt in range(n_attempts):
        np.random.seed(attempt)
        centers[:2*n] = np.random.rand(n, 2)
        for i in range(n):
            centers[2*i] -= 0.005 * np.random.rand()
            centers[2*i + 1] -= 0.005 * np.random.rand()
            centers[2*i] += 0.005 * np.random.rand()
            centers[2*i + 1] += 0.005 * np.random.rand()
        centers[2*n:] = 1e-3 * np.random.rand(n)

        def objective(x):
            _, sum_radii = lp_optimum(centers[:2*n].reshape(n, 2))
            return sum_radii

        def objective_with_jitter(x):
            return -objective(x[2*n:])

        result = scipy.optimize.minimize(objective_with_jitter, centers[2*n:], bounds=[(0, None)] * n, jac=gradient_function, method='L-BFGS-B', options={'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': 2000})
        sum_radii = -result.fun
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers[:2*n].reshape(n, 2)
            best_radii = lp_optimum(best_centers)[1]

    return best_centers, best_radii, best_sum
