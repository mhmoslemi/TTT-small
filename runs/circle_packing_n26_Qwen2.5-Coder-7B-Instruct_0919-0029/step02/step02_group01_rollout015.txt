import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_initial_grid():
    step = 0.20
    grid = np.array([[x + step / 2, y + step / 2] for x in np.linspace(0, 1 - step, 5) for y in np.linspace(0, 1 - step, 5)])
    return grid

def add_center_randomly(grid):
    return np.concatenate((grid, np.array([0.5, 0.5])[np.newaxis, :]))

def perturb_randomly(centers, epsilon0):
    epsilon = epsilon0
    centers = centers.copy()
    for i in range(centers.shape[0]):
        u = np.random.uniform(-1, 1, 2)
        u /= np.linalg.norm(u)
        centers[i] += u * epsilon
        centers[i] = np.clip(centers[i], 0, 1)
        epsilon *= 0.99
    return centers

def move_center(centers, k, step):
    new_center = centers[k] + np.random.uniform(-step, step, 2)
    new_center = np.clip(new_center, 0, 1)
    return np.array([centers[i] for i in range(centers.shape[0]) if i != k] + [new_center])

def random_accept(delta_sum, temperature):
    return np.random.rand() < np.exp(delta_sum / temperature)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NUM_RESTARTS = 30
    NUM_ITER = 4000
    TAU_STEP = 0.997
    TAU_TEMP = 0.995

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        grid = generate_initial_grid()
        centers = add_center_randomly(grid)
        centers = perturb_randomly(centers, 0.05)

        radii, sum_r = lp_optimum(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(centers.shape[0])
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = new_centers.copy(), new_radii.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

    # Fine-tuning
    step, temp = 1e-4, 0.5
    while step > 1e-5:
        for _ in range(NUM_ITER // 10):
            k = np.random.randint(centers.shape[0])
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = new_centers.copy(), new_radii.copy(), new_sum
        step *= TAU_STEP
        temp *= TAU_TEMP

    # Fine-tune final lp solve
    best_radii, best_sum = lp_optimum(best_centers)

    return best_centers, best_radii, best_sum
