def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

            A_ineq[-1][i] = 1

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun

        raise Exception("LP optimization failed")

def solve_socp_centre(radii, centres):
    n = centres.shape[0]

    c = np.ones(n)
    M = np.zeros((n, 2*n))
    for i in range(n):
        M[i, i] = 1
        M[i, n+i] = 1

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda centres, i=i: centres[i, 0] - radii[i]})
        constraints.append({'type': 'ineq', 'fun': lambda centres, i=i: 1 - centres[i, 0] - radii[i]})
        constraints.append({'type': 'ineq', 'fun': lambda centres, i=i: centres[i, 1] - radii[i]})
        constraints.append({'type': 'ineq', 'fun': lambda centres, i=i: 1 - centres[i, 1] - radii[i]})

        for j in range(n):
            if i != j:
                constraints.append({'type': 'ineq', 'fun': lambda centres, i=i, j=j: radii[i] + radii[j] - np.linalg.norm(centres[i] - centres[j], ord=2) + 1e-9})

    problem = cp.Problem(cp.Minimize(cp.sum_squares(M @ centres)), constraints)

    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if not problem.status == cp.OPTIMAL:
            problem.solve(solver=cp.OSQP, verbose=False)
    except Exception as e:
        raise Exception("SOCP optimization failed with error: " + str(e))

    if problem.status == cp.OPTIMAL:
        return problem.value, problem.variables()
    else:
        raise Exception("SOCP optimization failed")
