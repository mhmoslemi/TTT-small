import numpy as np
import cvxpy as cp
from scipy.spatial import cKDTree

def pair_distances(centres):
    tree = cKDTree(centres)
    dist_matrix = tree.sparse_distance_matrix(tree, np.inf).toarray()
    np.fill_diagonal(dist_matrix, np.inf)
    return dist_matrix

def build_lp(centres):
    n = centres.shape[0]
    rs = cp.Variable(n, nonneg=True)
    dists = pair_distances(centres)
    cons = [rs[i] + rs[j] <= dists[i, j] for i in range(n) for j in range(i + 1, n)]
    cons += [rs[i] <= centres[i, 0] for i in range(n)]
    cons += [rs[i] <= 1 - centres[i, 0] for i in range(n)]
    cons += [rs[i] <= centres[i, 1] for i in range(n)]
    cons += [rs[i] <= 1 - centres[i, 1] for i in range(n)]
    prob = cp.Problem(cp.Maximize(cp.sum(rs)), cons)
    return prob, rs

def solve_lp(prob):
    prob.solve(solver=cp.HS, verbose=False)
    dual_pair = {tuple(sorted((i, j))): prob.constraints[k].dual for k, (i, j) in enumerate(prob.model.constraints)}
    dual_left = {i: prob.constraints[k].dual for k, i in enumerate(prob.model.constraints) if prob.constraints[k].id in ["{0}-leq-{1}".format(i, "x") for i in range(prob.model.n_var)]}
    dual_right = {i: prob.constraints[k].dual for k, i in enumerate(prob.model.constraints) if prob.constraints[k].id in ["{0}-leq-{1}".format(i, "1-x") for i in range(prob.model.n_var)]}
    dual_bottom = {i: prob.constraints[k].dual for k, i in enumerate(prob.model.constraints) if prob.constraints[k].id in ["{0}-leq-{1}".format(i, "y") for i in range(prob.model.n_var)]}
    dual_top = {i: prob.constraints[k].dual for k, i in enumerate(prob.model.constraints) if prob.constraints[k].id in ["{0}-leq-{1}".format(i, "1-y") for i in range(prob.model.n_var)]}
    return rs.value, dual_pair, dual_left, dual_right, dual_bottom, dual_top

def compute_gradient(centres, radii, dual_pair, dual_left, dual_right, dual_bottom, dual_top):
    n = centres.shape[0]
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    dists = pair_distances(centres)
    for i in range(n):
        for j in range(n):
            if i == j: continue
            d_ij = dists[i, j]
            if d_ij > 1e-12:
                grad_x[i] += dual_pair[(i, j)] * (centres[i, 0] - centres[j, 0]) / d_ij
                grad_y[i] += dual_pair[(i, j)] * (centres[i, 1] - centres[j, 1]) / d_ij
        d_left = dual_left[i]
        d_right = dual_right[i]
        d_bottom = dual_bottom[i]
        d_top = dual_top[i]
        if 1e-12 < d_left < 1e-10: d_left = 0
        if 1e-12 < d_right < 1e-10: d_right = 0
        if 1e-12 < d_bottom < 1e-10: d_bottom = 0
        if 1e-12 < d_top < 1e-10: d_top = 0
        grad_x[i] += d_left - d_right
        grad_y[i] += d_bottom - d_top
    return grad_x, grad_y

def run_packing():
    MAX_ITERS = 500
    n = 26
    centres = np.empty((n, 2))

    # Initial pattern: rotated hexagonal lattice
    s = 0.2
    tx = 0.5 + s * 0.5 + s * np.sqrt(3) / 2 * 0.5
    ty = s + s * np.sqrt(3) / 2 * 0.5
    centre_indices = np.arange(n).reshape(5, 5)
    centre_indices[2, 0], centre_indices[2, 4], centre_indices[0, 2], centre_indices[4, 2] = -1, -1, -1, -1
    centre_indices[1, [1, 3]] = -1
    centre_indices[3, [1, 3]] = -1

    for i in range(5):
        for j in range(5):
            if centre_indices[i, j] != -1:
                x = centre_indices[i, j] * s
                y = (i + (j % 2)) * np.sqrt(3) / 2 * s / 2 + ty
                centres[centre_indices[i, j], :] = [x, y]

    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(MAX_ITERS):
        prob, rs = build_lp(centres)
        radii, dual_pair, dual_left, dual_right, dual_bottom, dual_top = solve_lp(prob)
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centres.copy(), radii.copy()

        grad_x, grad_y = compute_gradient(centres, radii, dual_pair, dual_left, dual_right, dual_bottom, dual_top)
        norm = np.sqrt((grad_x**2 + grad_y**2).sum())
        if norm < 1e-12: break
        grad = np.vstack((grad_x, grad_y)).T / norm

        new_centres = np.clip(centres + step * grad, 0, 1)
        prob, rs = build_lp(new_centres)
        new_radii, _, _, _, _, _ = solve_lp(prob)
        new_obj = new_radii.sum()

        if new_obj > obj:
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3

        if step < 1e-6: break

    return best_centres, best_radii, best_obj
