import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    radii = np.zeros(N)
    A = []
    b = []
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0] * N
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dist)
    for i in range(N):
        A.append([1 if k == i else 0 for k in range(N)])
        b.append(centers[i, 0])
        A.append([1 if k == i else 0 for k in range(N)])
        b.append(centers[i, 1])
        A.append([1 if k == i else 0 for k in range(N)])
        b.append(1 - centers[i, 0])
        A.append([1 if k == i else 0 for k in range(N)])
        b.append(1 - centers[i, 1])
    c = -np.ones(N)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * N, method='highs')
    if res.success:
        radii = res.x
        sum_radii = radii.sum()
        return sum_radii, radii
    else:
        return -np.inf, np.zeros(N)

def objective(X):
    centers = X.reshape((N, 2))
    sum_radii, _ = compute_sum_radii(centers)
    return -sum_radii

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_sum, _ = compute_sum_radii(current_centers)
    best_centers = current_centers.copy()
    T = initial_temperature
    for _ in range(max_iter):
        i = np.random.randint(N)
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum, _ = compute_sum_radii(new_centers)
        if new_sum > best_sum:
            current_centers = new_centers
            best_sum = new_sum
            best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 52
    result = differential_evolution(objective, bounds, popsize=12, maxiter=200, strategy='best1bin')
    centers = result.x.reshape((N, 2))
    best_centers = simulated_annealing(centers)
    _, radii = compute_sum_radii(best_centers)
    sum_radii = radii.sum()
    return best_centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
