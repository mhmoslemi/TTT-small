import numpy as np
import cvxpy as cp

def pair_distances(centres):
    n = centres.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.linalg.norm(centres[i] - centres[j])
    return dists

def build_lp(centres):
    n = centres.shape[0]
    r = cp.Variable(n, nonneg=True)
    dists = pair_distances(centres)
    constraints = [r[i] + r[j] <= dists[i, j] for i in range(n) for j in range(i + 1, n)]
    for i in range(n):
        constraints += [r[i] <= centres[i, 0], r[i] <= 1 - centres[i, 0], r[i] <= centres[i, 1], r[i] <= 1 - centres[i, 1]]
    return cp.Problem(cp.Maximize(r.sum()), constraints)

def solve_lp(prob):
    prob.solve(solver=cp.HIGHS)
    radii = prob.variables()[0].value
    duals = {constraint: constraint.dual_value for constraint in prob.constraints}
    return radii, duals

def compute_gradient(centres, duals):
    n = centres.shape[0]
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    dists = pair_distances(centres)
    for i in range(n):
        for j in range(i + 1, n):
            if not np.isclose(dists[i, j], 0):
                grad_x[i] += duals[f"Constr{2*(i*n + j)}"] * (centres[i, 0] - centres[j, 0]) / dists[i, j]
                grad_y[i] += duals[f"Constr{2*(i*n + j)}"] * (centres[i, 1] - centres[j, 1]) / dists[i, j]
        left = duals[f"Constr{i*n + n}"] - duals[f"Constr{i*n + n+1}"]
        right = duals[f"Constr{i*n + 2*n}"] - duals[f"Constr{i*n + 2*n+1}"]
        bottom = duals[f"Constr{i*n + 3*n}"] - duals[f"Constr{i*n + 3*n+1}"]
        top = duals[f"Constr{i*n + 4*n}"] - duals[f"Constr{i*n + 4*n+1}"]
        grad_x[i] += left - right
        grad_y[i] += bottom - top
    return grad_x, grad_y

def initial_pattern():
    # Create a structured pattern (e.g., hexagonal lattice with one missing corner)
    centers = np.array([
        [0.25, 0], [0.75, 0],
        [0.125, 0.433], [0.375, 0.433], [0.625, 0.433], [0.875, 0.433],
        [0.25, 0.866], [0.75, 0.866],
        [0.125, 1.299], [0.375, 1.299], [0.625, 1.299], [0.875, 1.299],
        [0.5, 1.472],
        [0.125, 1.645], [0.375, 1.645], [0.625, 1.645], [0.875, 1.645],
        [0.75, 1.818], [0.25, 1.818]
    ])
    return centers + 0.01 * np.random.randn(26, 2)

def run_packing():
    centres = initial_pattern()
    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None
    max_iter = 500

    for iter in range(max_iter):
        radii, duals = solve_lp(build_lp(centres))
        obj = radii.sum()

        if obj > best_obj:
            best_obj = obj
            best_centres = centres.copy()
            best_radii = radii.copy()

        grad_x, grad_y = compute_gradient(centres, duals)
        norm = np.sqrt((grad_x**2 + grad_y**2).sum())
        if norm < 1e-12:
            break
        grad = np.vstack((grad_x, grad_y)).T / norm

        new_centres = np.clip(centres + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))

        if new_obj > obj:  # accepted
            centres = new_centres
            step *= 0.9
        else:  # rejected
            step *= 0.3

        if step < 1e-6:
            break

    return best_centres, best_radii, best_obj
