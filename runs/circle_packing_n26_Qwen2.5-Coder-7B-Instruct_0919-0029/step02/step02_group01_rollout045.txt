```python
import numpy as np
from scipy.optimize import linprog

# Constants
NUM_RESTARTS = 30
NUM_ITER = 4000
STEP_INIT = 0.15
STEP_MIN = 1e-3
TAU_STEP = 0.997
TAU_TEMP = 0.995

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        raise RuntimeError("LP solve failed")

def move_center(centers, k, step):
    direction = np.random.randn(2)
    direction /= np.linalg.norm(direction)
    new_c = centers[k].copy()
    new_c += direction * step
    new_c = np.clip(new_c, 0, 1)
    return new_c

def random_accept(delta_sum, temperature):
    return np.random.rand() < np.exp(delta_sum / temperature)

def create_initial_grid_and_add_center():
    step = 0.20
    centers = np.array([[x, y] for y in [0.1, 0.3, 0.5, 0.7, 0.9] for x in [0.1, 0.3, 0.5, 0.7, 0.9]])
    centers = np.vstack([centers, [0.5, 0.5]])
    return centers

def perturb_randomly(centers, epsilon0=0.05):
    epsilon = epsilon0
    centers += np.random.uniform(-epsilon, epsilon, size=centers.shape)
    centers = np.clip(centers, 0, 1)
    return centers

def run_packing():
    best_centers, best_radii, best_sum = None, None, np.inf

    for restart in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, epsilon0=0.05)

        try:
            radii, sum_r = solve_lp(centers)
            step, temp = STEP_INIT, 0.5

            for _ in range(NUM_ITER):
                k = np.random.randint(centers.shape[0])
                new_centers = centers.copy()
                new_centers[k] = move_center(centers, k, step)
                new_radii, new_sum = solve_lp(new_centers)
                if new_sum < sum_r and random_accept(sum_r - new_sum, temp):
                    centers, radii, sum_r = new_centers.copy(), new_radii.copy(), new_sum
                    if new_sum < best_sum:
                        best_sum, best_centers, best_radii = new_sum, new_centers.copy(), new_radii.copy()
                step *= TAU_STEP
                temp *= TAU_TEMP

            if best_sum < best_sum:
                best_sum, best_centers, best_radii = sum_r, centers.copy(), radii.copy()

        except RuntimeError:
            pass

    # Final fine-tune
    final_radii, final_sum = solve_lp(best_centers)
    return best_centers, final_radii, final_sum

# Run the packing algorithm
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
```
