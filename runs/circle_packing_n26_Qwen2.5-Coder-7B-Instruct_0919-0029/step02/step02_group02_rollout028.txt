import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((2 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i in range(n):
        A_ub[i + n, i] = 1
        b_ub[i + n] = centers[i, 0] - 1e-12
        A_ub[i, i] = 1
        b_ub[i] = 1 - centers[i, 0] - 1e-12

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    result = differential_evolution(lambda x: lp_optimum(x.reshape(n, -1))[1], bounds=[(0, 1)] * (2 * n))
    best_centers = result.x.reshape(n, -1)
    best_radii, best_sum = lp_optimum(best_centers)
    return best_centers, best_radii, best_sum
