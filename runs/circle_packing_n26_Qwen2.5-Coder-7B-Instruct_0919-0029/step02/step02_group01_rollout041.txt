import numpy as np
from scipy.optimize import linprog
from scipy.stats import norm
from scipy.spatial.distance import cdist

# Helper functions
def initial_grid_plus_center():
    grid = np.array([(i/4 + 0.125, j/4 + 0.125) for i in range(4) for j in range(4)])
    centers = np.vstack((grid, (0.5, 0.5)))
    return centers

def gaussian_perturb(centers, sigma):
    noise = norm.rvs(scale=sigma, size=centers.shape)
    new_centers = np.clip(centers + noise, 0, 1)
    return new_centers

def lp_solve(centers):
    n = centers.shape[0]
    radii_bounds = [(0, None)] * n
    distances = cdist(centers, centers)
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = distances[i, j] - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        A_ub[idx+1, i] = 1
        b_ub[idx+1] = 1 - x - 1e-12
        A_ub[idx+2, i] = 1
        b_ub[idx+2] = y - 1e-12
        A_ub[idx+3, i] = 1
        b_ub[idx+3] = 1 - y - 1e-12
        idx += 4

    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=radii_bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26
   N_restart = 40
    N_iter = 4000
    step_min = 1e-4

    best_sum = -np.inf
    best_centers, best_radii = None, None

    np.random.seed(42)

    for restart in range(N_restart):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        for it in range(N_iter):
            k = np.random.randint(n)
            new_centers = centers.copy()
            new_centers[k] += norm.rvs(scale=0.15)
            new_centers[k] = np.clip(new_centers[k], 0, 1)

            new_radii, new_sum = lp_solve(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

    final_radii, final_sum = lp_solve(best_centers)
    final_radii = np.maximum(final_radii, 0)
    return best_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
