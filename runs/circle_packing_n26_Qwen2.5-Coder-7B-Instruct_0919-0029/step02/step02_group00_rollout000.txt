# Define input and helper functions

def calculate_pairwise_indices(num_circles):
    """Indices for all unique circle-to-circle R^2 distances."""
    i = np.arange(num_circles)[:, None]
    j = np.arange(num_circles)[None, :]
    return (i, j)

def objective_function(variables):
    """Total negative radii sum for maximization."""
    _, radii = variables.shape[0]//2, variables.shape[1]
    return -np.sum(radii)

def boundary_constraints(variables):
    """Constraints that ensure circles do not exceed unit square."""
    _, num_circles, num_vars = variables.shape[0]//2, variables.shape[1][1], variables.shape[2]
    boundary_idx = ((variables[:, 0, 0] - variables[:, :, 1]) < 0) & \
                  ((variables[:, 0, 0] + variables[:, :, 1]) > 1) & \
                  ((variables[:, 0, 1] - variables[:, :, 1]) < 0) & \
                  ((variables[:, 0, 1] + variables[:, :, 1]) > 1)
    return boundary_idx

def non_overlap_constraints(variables):
    """Constraints that ensure no two circles overlap."""
    _, num_circles, num_vars = variables.shape[0]//2, variables.shape[1][1], variables.shape[2]
    square_idx = (variables[:, 0, 0] - variables[:, :, 1]) >= 0 & \
                 (variables[:, 0, 0] + variables[:, :, 1]) <= 1 & \
                 (variables[:, 0, 1] - variables[:, :, 1]) >= 0 & \
                 (variables[:, 0, 1] + variables[:, :, 1]) <= 1
    non_overlap_idx = np.logical_and(square_idx[:, :, None], square_idx[:, None, :])
    pairwise_r2 = np.sum((variables[:, :-1, :2] - variables[:, 1:, :2])**2, axis=2)
    overlap_idx = pairwise_r2 < (variables[:, :-1, -1][:, None, :] + variables[:, 1:, -1])[non_overlap_idx]
    return overlap_idx

    """Penalty for overlapping circles."""

    return penalty
