import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    radii = result.x if result.success else np.zeros(n)
    sum_radii = -result.fun if result.success else 0
    return radii, sum_radii

def run_packing():
    n = 26
    step = 0.15
    decay_factor = 0.997
    NUM_RESTARTS = 30
    NUM_ITER = 4000

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = np.hstack((np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))) + [0.5, 0.5]
        centers = centers.reshape(26, 2)
        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
        centers = np.clip(centers, 0, 1)

        for it in range(NUM_ITER):
            k = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[k] + delta, 0, 1)
            centers[k] = new_c

            new_radii, new_sum = solve_lp(centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii

            step *= decay_factor

    final_radii, _ = solve_lp(best_centers)
    final_radii = np.maximum(final_radii, 0.0)
    return best_centers, final_radii, np.sum(final_radii)
