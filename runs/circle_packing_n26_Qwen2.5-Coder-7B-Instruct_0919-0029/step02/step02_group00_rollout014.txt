import numpy as np
import cvxpy as cp
from scipy.optimize import linprog

def solve_radius_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def solve_center_soc(radii, centers):
    n = len(radii)
    v = cp.Variable(n, complex=True)

    objective = cp.Minimize(cp.sum_squares(v.real) + cp.sum_squares(v.real))
    constraints = []
    for i in range(n):
        constraints.append(v.real[i] >= radii[i])
        constraints.append(v.real[i] <= 1 - radii[i])
        constraints.append(v.real[i] >= radii[i])
        constraints.append(v.real[i] <= 1 - radii[i])

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(cp.norm(cp.vstack((v.real[i] - v.real[j], v.imag[i] - v.imag[j])), p=2) >= radii[i] + radii[j])

    problem = cp.Problem(objective, constraints)

    problem.solve(solver=cp.ECOS)

    if problem.status == cp.OPTIMAL or problem.status == cp.OPTIMAL_INACCURATE:
        return v.value.real, v.value.imag
    else:
        raise Exception("SOCP optimization failed")

def run_packing():
    n = 26
    init_centers = np.zeros((n, 2))
    init_centers[::2, 0] = np.linspace(1/n, 2-1/n, n//2)
    init_centers[1::2, 0] = np.linspace(1/(2*n), 2-1/(2*n), n//2)
    init_centers[:, 1] = np.concatenate((np.tile(1/3, n//2), np.tile(2/3, n//2)))

    best_sum = 0
    best_centres = None
    best_radii = None

    for _ in range(3):
        radii, sum_radii = solve_radius_lp(init_centers)
        updated_centers = np.zeros((n, 2))
        for i in range(n):
            updated_centers[i] = np.array([np.mean(radii[:i+1]), np.mean(radii[:i+1])])
        radii, sum_radii = solve_radius_lp(updated_centers)
        centres, _ = solve_center_soc(radii, init_centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centres = centres
            best_radii = radii

        np.random.seed(_)
        init_centers += np.random.normal(0, 0.01, size=[n, 2])
        init_centers = np.clip(init_centers, 0, 1)

    _, sum_radii_final = solve_radius_lp(best_centres)
    return best_centres, best_radii, sum_radii_final
