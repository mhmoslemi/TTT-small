import numpy as np
from scipy.optimize import minimize

def generate_bounds(n):
    bounds = [([0, 1], [0, None])] * n
    return bounds

def generate_initial_state(n):
    centers = np.zeros((n, 2))
    radii = np.zeros((n,))
    for i in range(n):
        centers[i, 0] = 0.2 * (i % 5) + 0.1 * (i // 5)
        centers[i, 1] = 0.2 * (i // 5)
        centers[i] += 0.005 * (np.random.rand(2) * 2 - 1)
        radii[i] = 1e-3
    return centers, radii

def penalty_function(x, n):
    centers, radii = x.reshape((n, 2)), x.reshape((n,))[n:]
    pen = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d_sq = np.sum((centers[i] - centers[j]) ** 2)
            r_sum = radii[i] + radii[j]
            pen += max(0, r_sum - np.sqrt(d_sq)) ** 2
        for side in ['L', 'R', 'B', 'T']:
            r_x = centers[i, 0] if side == 'L' or side == 'R' else centers[i, 1]
            bound = 1 if side == 'R' or side == 'T' else 0
            pen += max(0, r_x - bound - radii[i]) ** 2
    return pen

def objective_function(x, alpha, beta):
    n = len(x) // 2
    pen = penalty_function(x, n)
    size = -np.sum(x[n:])
    return size + alpha * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    alpha, beta = 10**4, 10**6
    centers, radii = generate_initial_state(n)
    initial_state = np.concatenate((centers.flatten(), radii))
    bounds = generate_bounds(n)
    result = minimize(objective_function, initial_state, args=(alpha, beta), method='L-BFGS-B', jac=None, bounds=bounds, options={'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': 2000})
    final_state = result.x
    centers, radii = final_state.reshape((n, 2)), final_state.reshape((n,))[n:]
    _, final_radii, _ = lp_optimum(centers)
    return centers, final_radii, -result.fun

from scipy.spatial.distance import cdist
def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = cdist([centers[i]], [centers[j]], 'euclidean').item() - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = minimize(c, x0=np.zeros(n), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda x: b_ub - np.dot(A_ub, x)})
    return result.x, -result.fun

centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
