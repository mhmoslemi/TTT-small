import numpy as np
from scipy.optimize import minimize

def objective_function(variables, alpha, beta):
    n = variables.shape[0]
    centers = variables[:n].reshape(n, 2)
    radii = variables[n:]
    
    size_penalty = -np.sum(radii)
    distance_penalties = 0
    boundary_penalties = 0
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            distance_penalties += 0.5 * alpha * ( np.maximum(0, radii[i] + radii[j] - d_ij) )**2
    
    for i in range(n):
        boundary_penalties += 0.5 * beta * ( np.maximum(0, radii[i] - centers[i, 0]) )**2
        boundary_penalties += 0.5 * beta * ( np.maximum(0, radii[i] - (1 - centers[i, 0])) )**2
        boundary_penalties += 0.5 * beta * ( np.maximum(0, radii[i] - centers[i, 1]) )**2
        boundary_penalties += 0.5 * beta * ( np.maximum(0, radii[i] - (1 - centers[i, 1])) )**2
    
    return size_penalty + distance_penalties + boundary_penalties

def gradient_function(variables, alpha, beta):
    n = variables.shape[0]
    centers = variables[:n].reshape(n, 2)
    radii = variables[n:]
    
    grad_size = -np.ones(n)
    grad_distance = np.zeros(n * 2)
    grad_boundary = np.zeros(n * 2)
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            if d_ij < 1e-12:
                continue
            d_ij_inv = 1 / d_ij
            grad_distance[i] += alpha * np.maximum(0, radii[i] + radii[j] - d_ij) * d_ij_inv
            grad_distance[j] += alpha * np.maximum(0, radii[i] + radii[j] - d_ij) * d_ij_inv
            grad_boundary[i] += beta * np.maximum(0, radii[i] - centers[i, 0])
            grad_boundary[j] += beta * np.maximum(0, radii[i] - (1 - centers[i, 0]))
            grad_boundary[i] += beta * np.maximum(0, radii[i] - centers[i, 1])
            grad_boundary[j] += beta * np.maximum(0, radii[i] - (1 - centers[i, 1]))
    
    grad = np.concatenate([grad_distance, grad_boundary])
    
    return grad

def initialize_centers(n):
    row_sizes = [5, 5, 6, 5, 5]
    delta = 0.2
    jitter = 0.005
    
    centers = np.zeros((n, 2))
    
    for i, size in enumerate(row_sizes):
        for k in range(size):
            centers[i*size + k] = [(i*size + k + 0.5)*delta + (size > 5) * (delta / 2), 0.1 + (i == 2) * delta * (k + 1)]
    
    centers += jitter * np.random.uniform(-1, 1, centers.shape)
    
    return centers.flatten()

def initialize_radii(n):
    return 1e-3 * np.ones(n)

def refine_solution(centers, alpha, beta):
    n = centers.shape[0]
    variables = np.concatenate([centers, initialize_radii(n)])
    options = {
        'method': 'L-BFGS-B',
        'bounds': [(0, 1) for _ in range(2*n)] + [(0, None) for _ in range(n)],
        'jac': lambda vars: gradient_function(vars, alpha, beta),
        'options': {'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': 2000}
    }
    
    result = minimize(lambda vars: objective_function(vars, alpha, beta), variables, **options)
    
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    alpha = 10**4
    beta = 10**6
    
    centers = initialize_centers(n)
    new_centers = refine_solution(centers, alpha, beta)
    
    lp_centers, lp_radii, lp_sum = lp_optimum(new_centers.reshape(n, 2))
    
    return lp_centers, lp_radii, lp_sum

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = minimize(c, centers.flatten(), method='highs', bounds=bounds, constraints={'type': 'ineq', 'fun': A_ub.dot, 'args': (b_ub,)}, options={'maxiter': 2000})
    return result.x.reshape(n, 2), -result.fun

# Run the packing algorithm
centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
