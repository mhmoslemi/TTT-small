import numpy as np
from scipy.optimize import minimize

def penalty(centers, radii, alpha=10**4, beta=10**6):
    n = centers.shape[0]
    pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        pen += np.max(0, r - x)
        pen += np.max(0, r - (1 - x))
        pen += np.max(0, r - y)
        pen += np.max(0, r - (1 - y))
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            pen += np.max(0, r + radii[j] - d) ** 2
    return pen / alpha + pen / beta

def objective(x):
    centers = x[:52].reshape((26, 2))
    radii = x[52:]
    f_size = -np.sum(radii)
    f_pen = penalty(centers, radii)
    return f_size + f_pen
