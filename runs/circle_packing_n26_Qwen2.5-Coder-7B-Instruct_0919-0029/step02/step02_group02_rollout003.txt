import numpy as np
from scipy.optimize import linprog
from scipy.spatial import distance_matrix

def lp_optimum(centers):
    n = centers.shape[0]
    d = distance_matrix(centers, centers)
    r = np.ones(n)
    A_ub = np.zeros((n ane-1), n)) - np.eye(n)
    A_ub = np.vstack((A_ub, np.hstack((-A_ub[:, np.newaxis], -A_ub[1:, :]))))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[d[i, j] <= 1][i*n + j] = 1
    b_ub = np.zeros(A_ub.shape[0])
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, np.inf), method='highs')
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    centers = np.random.rand(26, 2)
    step = 0.1
    max_iter = 250
    best_centers = centers.copy()
    best_sum = 0.0
    for _ in range(max_iter):
        new_centers = centers + np.random.normal(0, 0.005, size=(26, 2)) * step
        new_centers = np.clip(new_centers, 0, 1)
        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
    return best_centers, best_radii, best_sum
