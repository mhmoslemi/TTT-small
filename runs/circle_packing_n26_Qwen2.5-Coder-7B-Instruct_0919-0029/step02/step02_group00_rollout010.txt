import numpy as np
from scipy import optimize
import cvxpy as cp

def hexagonal_lattice_in_unit_square():
    n = 26
    r = np.sqrt(2/3)
    centers = []

    for i in range(2):
        for j in range(3 * i + ((i + 1) % 2), 3 * i + (i + 1) % 2 + 2):
            x = (2 * j + (i % 2)) * r / 2
            y = (i + j // 3) * np.sqrt(3) * r / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    return np.array(centers)[:n]

def solve_LP(centers, rho=10.0, eps=1e-7, max_iter=1000):
    r = cp.Variable(centers.shape[0], nonneg=True)
    constraints = [
        r[i] <= centers[i, 0] for i in range(centers.shape[0])
    ] + [
        r[i] <= 1 - centers[i, 0] for i in range(centers.shape[0])
    ] + [
        r[i] <= centers[i, 1] for i in range(centers.shape[0])
    ] + [
        r[i] <= 1 - centers[i, 1] for i in range(centers.shape[0])
    ] + [
        r[i] + r[j] <= cp.sqrt(np.sum((centers[i] - centers[j])**2)) for i in range(centers.shape[0]) for j in range(i + 1, centers.shape[0])
    ]
    problem = cp.Problem(cp.Maximize(sum(r)), constraints)
    problem.solve(solver=cp.ECOS, verbose=False, eps=eps, max_iter=max_iter)
    return r.value

def solve_SOCP(centers, radii, rho=10.0, eps=1e-6, max_iter=2000):
    r = cp.Variable(centers.shape[0], nonneg=True)
    c = cp.Variable((centers.shape[0], centers.shape[1]), nonneg=True)
    constraints = [
        r[i] <= c[i, 0] for i in range(centers.shape[0])
    ] + [
        r[i] <= 1 - c[i, 0] for i in range(centers.shape[0])
    ] + [
        r[i] <= c[i, 1] for i in range(centers.shape[0])
    ] + [
        r[i] <= 1 - c[i, 1] for i in range(centers.shape[0])
    ] + [
        cp.norm(c[i] - c[j], p=2) >= r[i] + r[j] for i in range(centers.shape[0]) for j in range(i + 1, centers.shape[0])
    ]
    problem = cp.Problem(cp.Minimize(sum(cp.norm(c - centers, p=2, axis=1)**2)), constraints)
    problem.solve(solver=cp.ECOS, verbose=False, eps=eps, max_iter=max_iter)
    return c.value

def admm_packing(max_iter=30, rho=10.0, eps=1e-7):
    centers = hexagonal_lattice_in_unit_square()
    radii = np.zeros(centers.shape[0])
    dual = np.zeros(centers.shape[0])
    obj_prev = -np.inf

    for k in range(max_iter):
        r_lp = solve_LP(centers, rho)
        radii = r_lp + (dual - rho * radii) / rho

        centers = solve_SOCP(centers, radii, rho)

        dual = dual + rho * (radii - radii)
        obj = sum(radii)

        if abs(obj - obj_prev) < eps: break
        obj_prev = obj

    r_final = solve_LP(centers, rho)
    return centers, r_final, sum(r_final)

centers, radii, sum_radii = admm_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of Radii:", sum_radii)
