import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import differential_evolution, linprog
from scipy.integrate import quad

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def de_objective(flat_centres):
    centres = flat_centres.reshape(26, 2)
    _, s = solve_radius_lp(centres)
    return -s  # DE minimises

def perturb_solution(centres, radii, sigma=0.01):
    dx = np.random.normal(0, sigma, 52)
    dy = np.random.normal(0, sigma, 52)
    new_centres = centres.copy()
    new_centres[:, 0] += dx
    new_centres[:, 1] += dy
    new_centres[new_centres < 0] = 0
    new_centres[new_centres > 1] = 1
    new_radii = radii.copy()
    new_radii[new_radii < 0] = 0
    return new_centres, new_radii

def run_packing():
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    # Step 1: Initialize with hexagonal lattice
    delta = np.sqrt(3) / 2
    centers = np.zeros((26, 2))
    x, y = 0.16666667, 0.5
    r = x / (2 * delta)
    for i in range(13):
        for j in range(1, 14):
            offset = 1 if i % 2 == j % 2 else 0
            centers[i * 13 + j - offset] = np.array([x, y])
            x += delta
        y += 0.5 + offset * 0.25
        x = 0.16666667 if i % 2 == 0 else 0.5 + (i % 2) * 0.25

    # Step 2: Run PSO
    bounds = [(0.0, 1.0)] * 52
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=120,
        popsize=12,
        tol=1e-6,
        polish=True,
        seed=42
    )
    best_centres = result.x.reshape(26, 2)

    # Step 3: Perturb and re-run PSO
    best_centres, _ = perturb_solution(best_centres, np.zeros(26))
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=120,
        popsize=12,
        tol=1e-6,
        polish=True,
        seed=43
    )
    if result.fun < -1e-6:
        best_centres = result.x.reshape(26, 2)

    # Step 4: Solve LP to refine radii
    final_radii, final_sum = solve_radius_lp(best_centres)

    return best_centres, final_radii, final_sum

if __name__ == "__main__":
    np.random.seed(42)
    result = run_packing()
    validate_packing(*result)
    print(f"Sum of radii: {result[2]}")
