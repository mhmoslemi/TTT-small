import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    m = n * (n - 1) // 2 + 4 * n
    A = np.empty((m, n))
    b = np.empty(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1
        b[idx] = x - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - x - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = y - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - y - 1e-12
        idx += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x, -res.fun
