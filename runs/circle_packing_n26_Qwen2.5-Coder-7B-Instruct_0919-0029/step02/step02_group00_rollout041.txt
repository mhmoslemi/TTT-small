import numpy as np
from scipy.optimize import minimize

def solve_radius_lp(centre):
    n = centre.shape[0]
    midpoints = (centre[:, 0] + centre[:, 1])[:, None] / 2
    rs = np.sum(centre[:, 0]**2 + centre[:, 1]**2 - midpoints**2, axis=1)

    r = np.max(rs)
    areas = np.pi * r**2
    return areas.sum()

def non_overlap_constraint(centre, r_idx):
    def constraint(params):
        centre_i = centre[r_idx]
        r_i = params[(r_idx * 3):((r_idx + 1) * 3)]
        diffs = centre - centre_i
        dist = np.sum(diffs**2, axis=1)
        overlap = dist + r_i[0] + r_i[2] - r_i[1] - 1e-10
        return np.maximum(overlap, 0)
    return constraint

def boundary_constraint(centre, r_idx):
    def constraint(params):
        centre_i = centre[r_idx]
        r_i = params[(r_idx * 3):((r_idx + 1) * 3)]
        x_min = centre_i[0] - r_i[0] - 1e-10
        x_max = centre_i[0] + r_i[2] - 1e-10
        y_min = centre_i[1] - r_i[0] - 1e-10
        y_max = centre_i[1] + r_i[2] - 1e-10
        return np.maximum([0, 0, 0, 0],
                          [np.maximum(0, -x_min), np.maximum(0, 1 - x_max),
                           np.maximum(0, -y_min), np.maximum(0, 1 - y_max)])
    return constraint

def objective(params, centre):
    r = np.sqrt(params[(0, 1, 2)] + params[(120, 121, 122)] + params[(240, 241, 242)])
    areas = r**2 * np.pi
    penalty = 1e-7 * (sum(areas) - 2.6)
    return -(np.sum(areas) + penalty)

def run_packing():
    np.random.seed(42)
    n = 26
    stick_radius = 0.1
    circle_area = np.pi * stick_radius**2

    centre = np.zeros((n, 2))
    j = 0
    for x in np.linspace(0.0, 1.0, 8):
        for y in np.linspace(0.0, 1.0, 8):
            centre[j] = (x, y)
            j += 1
    centre = centre[:n]

    r = stick_radius
    areas = np.pi * r**2 * np.ones(n)
    discrete_centre = centre * 6
    discrete_centre = np.round(discrete_centre).astype(int)
    discrete_centre = discrete_centre % 6
    centre = discrete_centre / 6
    r = stick_radius

    res = []
    for _ in range(3):
        centre += np.random.normal(0, 0.01, centre.shape)
        centre = np.clip(centre, 0, 1)
        objective_params = np.hstack([centre.flatten(), np.array([2.6 / np.sqrt(np.pi), 0])])
        bnds = [(0, 1), (0, 1), (0, 1)]
        for i in range(n):
            bnds_ = bnds.copy()
            bnds_[i * 3] = (centre[i, 0] - r, centre[i, 0] + r)
            bnds_[i * 3 + 1] = (centre[i, 0] - r, centre[i, 0] + r)
            bnds_[i * 3 + 2] = (r, r)
            constraint_i = [(non_overlap_constraint(centre, i), np.inf) for i in range(n)]
            constraint_i.extend([(boundary_constraint(centre, i), np.inf) for i in range(n)])
            res_ = minimize(objective, objective_params.copy(), args=(centre,), constraints=constraint_i, bounds=bnds_)
            objective_params = res_.x
            centre = objective_params[0:n*3].reshape(-1, 3)[:, 0:2]
            r = objective_params[0:3]
        res.append(res_)

    best_idx = np.argmax([res[i]['fun'] for i in range(3)])
    best_res = res[best_idx]
    centre = best_res['x'][0:n*3].reshape(-1, 3)[:, 0:2]
    r = best_res['x'][0:3]
    centre, r = np.clip(centre, 0, 1), np.abs(r)
    area = np.pi * r**2 * np.ones(n)
    r = np.sqrt(area / np.pi)
    return centre, r, np.sum(r)
