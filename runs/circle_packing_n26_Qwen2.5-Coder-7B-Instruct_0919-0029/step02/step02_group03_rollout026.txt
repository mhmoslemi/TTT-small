import numpy as np
from scipy.optimize import differential_evolution, funmin

N = 26

def compute_sum_radii(centers):
    n = centers.shape[0]
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([0]*n)
        A_ub[-1][i] = 1
        b_ub.append(centers[i,0])
        A_ub.append([0]*n)
        A_ub[-1][i] = 1
        b_ub.append(centers[i,1])
        A_ub.append([0]*n)
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i,0])
        A_ub.append([0]*n)
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i,1])
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0]*n
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
    A = np.vstack(A_ub)
    b = np.array(b_ub)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
    if res.success:
        return np.maximum(res.x, 0).sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N,2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing():
    bounds = [(0.0, 1.0) for _ in range(52)]
    centers_notradii = np.zeros(52)
    result = differential_evolution(objective, bounds=bounds, popsize=12, maxiter=200, strategy='best1bin', tol=1e-6)
    best_centers = result.x.reshape((N,2))
    best_radii = np.maximum(-compute_sum_radii(best_centers), 0)
    final_sum_radii = np.sum(best_radii)
    if final_sum_radii > 2.637:
        return best_centers, best_radii, final_sum_radii
    else:
        centers = best_centers.copy()
        for i in range(N):
            for j in range(i+1, N):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d > 2 * min(centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1], centers[j, 0], centers[j, 1], 1 - centers[j, 0], 1 - centers[j, 1]):
                    centers[i] = centers[i] * 0.999999
                    centers[j] = centers[j] * 0.999999
                    break
        best_radii = np.maximum(-compute_sum_radii(centers), 0)
        final_sum_radii = np.sum(best_radii)
        return centers, best_radii, final_sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
