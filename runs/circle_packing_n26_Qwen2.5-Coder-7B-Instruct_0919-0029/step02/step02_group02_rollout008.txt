import numpy as np
from cvxpy import Variable, Problem, Solver, Parameter, Minimize, Nonneg
from operator import itemgetter

# Geometry helpers
def pair_distances(centers):
    n = centers.shape[0]
    M = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            M[i, j] = M[j, i] = np.linalg.norm(centers[i] - centers[j])
    return M

def build_lp(centers):
    n = centers.shape[0]
    r = Variable(n)
    problem = Problem(
        Minimize(sum(r)),
        [
            Nonneg(r),
            *([r[i] + r[j] <= M[i, j] for i in range(n) for j in range(i+1, n)]),
            *[0 <= r[i] <= centers[i, 0] for i in range(n)],
            *[0 <= r[i] <= 1 - centers[i, 0] for i in range(n)],
            *[0 <= r[i] <= centers[i, 1] for i in range(n)],
            *[0 <= r[i] <= 1 - centers[i, 1] for i in range(n)],
        ]
    )
    return problem

def solve_lp(prob):
    prob.solve(solver=Solver.HIGHS, verbose=False)
    return prob.value, {('pair', i, j): prob.constraints[k].dual_value for k, (i, j) in enumerate(itemgetter(*[np.triu_indices(n, k=1)])(prob.constraints))}

# Main packing function
def run_packing():
    n = 26
    step = 0.1
    MAX_ITERS = 500

    # Initial pattern (5x5 hex lattice)
    centers = np.array([
        [0.2, 0.5], [0.4, 0.5], [0.6, 0.5], [0.8, 0.5], [1.0, 0.5],
        [0.25, 0.0], [0.35, 0.0], [0.75, 0.0], [0.85, 0.0], [0.25, 1.0], [0.35, 1.0],
        [0.75, 1.0], [0.85, 1.0],
        [0.3, 0.3], [0.3, 0.7], [0.7, 0.3], [0.7, 0.7],
        [0.22, 0.22], [0.22, 0.78], [0.78, 0.22], [0.78, 0.78],
        [0.44, 0.14], [0.44, 0.86], [0.86, 0.14], [0.86, 0.86]
    ])
    centers += 0.01 * np.random.randn(*centers.shape)

    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(MAX_ITERS):
        radii, duals = solve_lp(build_lp(centers))
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centers.copy(), radii.copy()

        grad_x, grad_y, grad_r = compute_gradient(centers, duals)
        norm = np.sqrt((grad_x**2 + grad_y**2 + grad_r**2).sum())
        if norm < 1e-12: break
        grad = np.vstack((grad_x / norm, grad_y / norm, grad_r / norm))

        new_centres = np.clip(centers + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))

        if new_obj > obj:                          # accepted
            centres = new_centres
            step *= 0.9
        else:                                      # rejected
            step *= 0.3

        if step < 1e-6: break

    return best_centres, best_radii, best_obj

def compute_gradient(centers, duals):
    n = centers.shape[0]
    M = pair_distances(centers)
    grad_r = np.array([sum(duals[('pair', i, j)] for j in range(n) if i != j) for i in range(n)])
    grad_x = grad_y = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            dist = M[i, j]
            if dist > 1e-12:
                grad_x[i] += (centers[i, 0] - centers[j, 0]) * duals[('pair', i, j)] / dist
                grad_y[i] += (centers[i, 1] - centers[j, 1]) * duals[('pair', i, j)] / dist
                grad_x[j] -= (centers[i, 0] - centers[j, 0]) * duals[('pair', i, j)] / dist
                grad_y[j] -= (centers[i, 1] - centers[j, 1]) * duals[('pair', i, j)] / dist
        grad_x[i] += (centers[i, 0] * duals[('lower', i)] - (1 - centers[i, 0]) * duals[('upper', i)]) / M[i, i]
        grad_y[i] += (centers[i, 1] * duals[('lower', i)] - (1 - centers[i, 1]) * duals[('upper', i)]) / M[i, i]
    return grad_x, grad_y, grad_r

# Main packing function call
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of radii: {sum_radii}")

This program implements the strategy described in the plan:
- Initializes the centre locations in a structured pattern (hexagonally for a grid of 5x5 and one missing corner).
- Uses the dual information from an initial linear programming solution to compute the gradient of the objective function with respect to the centre coordinates.
- Performs gradient descent updates with a step size adaptation mechanism.
- Ensures that the updated centre locations stay within the feasible region [0, 1] using the clip function.
- Regularly checks for convergence based on the step size and objective value improvement.

This method is distinct from the previous strategies and leverages exact dual sensitivities to guide gradient-based updates, aiming to produce a high-quality packing close to your objective of maximizing the sum of radii.
