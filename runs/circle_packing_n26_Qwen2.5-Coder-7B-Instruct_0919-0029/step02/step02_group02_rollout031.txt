import numpy as np
from scipy.optimize import linprog

def calc_gradient(c, centers, d_ij, mu):
    n = len(centers)
    grad = np.zeros(2*n)
    for i in range(n):
        grad_i = np.zeros(n)
        for j in range(n):
            if i != j:
                dx_ij, dy_ij = c[i] - c[j], centers[i] - centers[j]
                r_ij = np.linalg.norm(dx_ij)
                if r_ij > 1e-12:
                    grad_i[j] = (dx_ij / r_ij) * d_ij[i][j]
        grad[2*i] = grad[i] = sum(grad_i) + mu[i*2] - mu[i*2 + 1]
    return grad

def run_packing() -> (np.ndarray, np.ndarray, float):
    def lp_optimum(centers):
        n = centers.shape[0]
        A_eq = np.zeros((2*n, n))
        b_eq = np.zeros(2*n)
        for i in range(n):
            for j in range(i + 1, n):
                A_eq[j - i - 1, i] = 1
                A_eq[j - i - 1, j] = 1
                b_eq[j - i - 1] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_eq[2*n-4+i*2, i] = 1
            b_eq[2*n-4+i*2] = centers[i][0]
            A_eq[2*n-3+i*2, i] = 1
            b_eq[2*n-3+i*2] = 1 - centers[i][0]
            A_eq[2*n-2+i*2, i] = 1
            b_eq[2*n-2+i*2] = centers[i][1]
            A_eq[2*n-1+i*2, i] = 1
            b_eq[2*n-1+i*2] = 1 - centers[i][1]
        c = np.ones(n)
        result = linprog(c, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
        return -result.x, -result.fun

    def init_centers():
        Δ = 0.2
        centers = np.array([
            [Δ/2, Δ/2], [2*Δ/2, Δ/2], [3*Δ/2, Δ/2], [4*Δ/2, Δ/2], [5*Δ/2, Δ/2],
            [1*Δ, 1*Δ], [2*Δ, 1*Δ], [3*Δ, 1*Δ], [4*Δ, 1*Δ], [5*Δ, 1*Δ],
            [1*Δ, 2*Δ], [2*Δ, 2*Δ], [3*Δ, 2*Δ], [4*Δ, 2*Δ], [5*Δ, 2*Δ],
            [1*Δ, 3*Δ], [2*Δ, 3*Δ], [3*Δ, 3*Δ], [4*Δ, 3*Δ], [5*Δ, 3*Δ],
            [1*Δ, 4*Δ], [2*Δ, 4*Δ], [3*Δ, 4*Δ], [4*Δ, 4*Δ], [5*Δ, 4*Δ],
            [2*Δ, 5*Δ], [3*Δ, 5*Δ], [4*Δ, 5*Δ]
        ])
        return centers + np.random.uniform(-5e-3, 5e-3, (26, 2))

    def calc_d_ij(centers):
        n = len(centers)
        d_ij = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                d_ij[i][j] = d_ij[j][i] = np.linalg.norm(centers[i] - centers[j])
        return d_ij

    centers = init_centers()
    d_ij = calc_d_ij(centers)

    mu = np.zeros(2*len(centers))
    step, step_min = 0.1, 1e-6
    decay_factor = 0.995
    best_sum = -np.inf
    best_centers = None
    best_d_ij = None
    best_mu = None

    for start in range(5):
        for _ in range(500):
            idx = np.random.randint(len(centers))
            delta = np.random.uniform(-step, step, 2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c
            new_d_ij = calc_d_ij(new_centers)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_d_ij = new_d_ij
                best_mu = calc_gradient(new_c, new_centers, new_d_ij, mu)

        mu = calc_gradient(best_centers.reshape(-1), best_centers, best_d_ij, mu)
        step = max(step_min, step * decay_factor)

    return best_centers, best_radii, best_sum
