import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize

# Constants
NUM_CIRCLES = 26
GRID(side_length=10):
    n = side_length ** 2
    x = np.linspace(0, 1, side_length)
    y = np.linspace(0, 1, side_length)
    centers = []
    for xi in x:
        for yi in y:
            centers.append((xi, yi))
    return np.array(centers)
    
def solve_radii(centers):
    n = NUM_CIRCLES
    A_ub = np.vstack([
        -np.eye(n),  centers[:, 0] - np.array([1] * n),
        np.eye(n),   centers[:, 0],
        -np.eye(n),  centers[:, 1] - np.array([1] * n),
        np.eye(n),   centers[:, 1],
    ])
    b_ub = np.repeat([0] * 2 * n, 2)
    c = np.zeros(6 * n)
    c[:n] = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(0, 1)] * n)
    return 1 - result.x if result.success else np.zeros(n)

def objective(Z):
    n = NUM_CIRCLES
    centers = Z[:2*n].reshape((n, 2))
    radii = solve_radii(centers)
    return -np.sum(radii)

def gradient(Z):
    n = NUM_CIRCLES
    h = 1e-4
    J = np.zeros((2*n, n))
    e = np.eye(n)
    for i in range(n):
        J[:, i], _ = obj_and_grad(Z, i, h)
    return J

def obj_and_grad(Z, i, h):
    zpos, _ = objective_and_gradient(Z.copy(), 2*n + 2*i)
    Z[2*n + 2*i] += h
    zneg, _ = objective_and_gradient(Z, 2*n + 2*i)
    Z[2*n + 2*i] -= h
    grad = (zpos - zneg) / (2 * h)
    return zpos, grad

def run_packing():
    initial_centers = np.random.rand(NUM_CIRCLES, 2)
    best_result = None
    for _ in range(10):
        centers = initial_centers + 0.01 * np.random.randn(NUM_CIRCLES, 2)
        centers = np.clip(centers, 0, 1)
        result = minimize(objective, centers.flat, jac=gradient, method='L-BFGS-B', bounds=[(0, 1)] * (2 * NUM_CIRCLES))
        if best_result is None or result.fun > best_result.fun:
            best_result = result
    centers = best_result.x.reshape((NUM_CIRCLES, 2))
    radii = solve_radii(centers)
    radii = np.maximum(radii, 0)
    return centers, radii, np.sum(radii)

# Execute function and print result
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}, Radii: {radii}, Sum Radii: {sum_radii}")
