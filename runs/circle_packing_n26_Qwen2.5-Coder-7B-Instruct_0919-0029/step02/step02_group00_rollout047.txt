import numpy as np
import cvxpy as cp
from scipy.optimize import linprog
from scipy.spatial.distance import cdist
import math

def solve_LP(C):
    n = C.shape[0]
    r = cp.Variable(n, nonneg=True)
    c = np.ones(n)

    # Define the constraints
    x_lower = C[:, 0] - r
    x_upper = C[:, 0] + r
    y_lower = C[:, 1] - r
    y_upper = C[:, 1] + r

    # Add the hexagonal contacts
    D = cdist(C, C, 'euclidean')
    non_diag_indices = ~np.eye(n, dtype=bool)
    constraints = [
        r + r <= D[non_diag_indices] - 1e-6,
        -x_lower <= 0,
        x_upper - 1 <= 0,
        -y_lower <= 0,
        y_upper - 1 <= 0
    ]
    
    # Solve the LP
    problem = cp.Problem(cp.Maximize(c.T @ r), constraints)
    problem.solve(solver=cp.HS, verbose=False)
    
    return np.maximum(r.value, 1e-8)

def solve_SOCP(r, C):
    n = C.shape[0]
    C_new = cp.Variable((n, 2), complex=False)
    
    # Create a quadratic cost term
    Q = np.eye(n)
    R = np.concatenate([C_new.real, C_new.imag], axis=1)
    cost = cp.quad_form(R, Q)
    
    # Define constraints based on r and C
    contr1 = C_new[:, 0] + C_new[:, 1] <= 1 + 1e-6
    contr2 = C_new[:, 0] - r <= 0
    contr3 = r - C_new[:, 0] <= 0
    contr4 = C_new[:, 1] + C_new[:, 1] <= 1 + 1e-6
    
    # Define distance constraints
    distances = cdist(C_new, C_new, 'euclidean')
    contacts = [(i, j) for i in range(n) for j in range(i + 1, n)]
    lower_bounds = r + r
    constraints = [
        distances[contacts] >= lower_bounds, 
        contr1, contr2, contr3, contr4
    ]
  
    # Solve the SOCP
    prob = cp.Problem(cp.Minimize(cost), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    return C_new.value

def run_packing():
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    n = 26
    rho = 10.0
    iterations = 30

    # Initialize centers and radii
    C = np.zeros((n, 2))
    r = np.zeros(n)
    lambd = np.zeros(n)

    obj_prev = -np.inf

    for k in range(1, iterations + 1):
        # Radius update
        r_new = solve_LP(C)
        r = r_new + (lambd - rho * r_new) / rho
        
        # Center update
        C = solve_SOCP(r, C)
        
        # Dual update
        lambd = lambd + rho * (r - r)

        obj = np.sum(r)
        if abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj

    # Final polishing
    r_final = solve_LP(C)
        
    return C, r_final, np.sum(r_final)

# Execute the function to get the result
centres, radii, sum_radii = run_packing()
print(f"Centres: {centres}")
print(f"Radii: {radii}")
print(f"Sum of Radii: {sum_radii}")
