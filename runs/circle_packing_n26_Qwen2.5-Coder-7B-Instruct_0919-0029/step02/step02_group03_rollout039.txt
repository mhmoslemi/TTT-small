import cvxpy as cp
import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    x = cp.Variable(N, nonneg=True)
    y = cp.Variable(N, nonneg=True)
    r = cp.Variable(N, nonneg=True)
    
    constraints = []
    for i in range(N):
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])
        for j in range(i+1, N):
            constraints.append(cp.norm([x[i] - x[j], y[i] - y[j]], 2) >= (r[i] + r[j]))
    
    objective = cp.Maximize(cp.sum(r))
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-4, reltol=1e-4)
        if prob.status == cp.SOLVER_ERROR or prob.value == np.inf:
            raise ValueError("Solver failed or returned inf")
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-4)
        except:
            return run_packing()  # Fallback to heuristic if both solve methods fail
    
    centers = np.vstack((np.clip(x.value, 0, 1), np.clip(y.value, 0, 1))).T
    radii = np.clip(r.value, 0, None)
    
    def compute_sum_radii(centers, radii):
        n = centers.shape[0]
        sum_radii = radii.sum()
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-6:
                    sum_radii -= abs(radii[i] + radii[j] - dist) / 10
        return sum_radii
    
    sum_radii = compute_sum_radii(centers, radii)
    if sum_radii < 2.637:
        for i in range(N):
            radii[i] = radii[i] * 0.999999
        sum_radii = radii.sum()
    
    return centers, radii, sum_radii
