import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    n = len(centers)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(centers[i][0])  # x <= x_i
        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(centers[i][1])  # y <= y_i
        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(1 - centers[i][0])  # 1-x
        A_ub.append([1 if j == i else 0 for j in range(n)])
        b_ub.append(1 - centers[i][1])  # 1-y
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [1 if k == i else 0 for k in range(n)] + [1 if k == j else 0 for k in range(n)]
            A_ub.append(row)
            b_ub.append(dist)
    c = -np.ones(2 * n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * (2 * n), method='highs')
    if res.success:
        radii = np.sqrt(-res.fun / (np.array(b_ub[:N]) - np.array(b_ub[N:])))
        return radii.sum()
    return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum >= 0 and (new_sum > best_sum or np.exp((new_sum - best_sum) / T) > np.random.rand()):
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers, best_sum

def run_packing():
    bounds = [(0.0, 1.0) for _ in range(52)]
    result_DE = differential_evolution(objective, bounds=bounds, popsize=10, maxiter=200, strategy='best1bin')
    best_centers, best_sum = result_DE.x[:N*2].reshape((-1, 2)), -result_DE.fun
    sim_result = simulated_annealing(best_centers, max_iter=5000)
    final_centers, final_sum = sim_result[0], simulate_annealing(best_centers, max_iter=4000)[1]
    final_radii = compute_sum_radii(final_centers)
    return final_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
