import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2 * n)
    options = {
        'disp': False,
        'polish': True,
        'workers': -1
    }
    result = differential_evolution(
        lambda x: lp_optimum(x.reshape(n, 2))[1], bounds, options=options
    )
    centers, _ = lp_optimum(result.x.reshape(n, 2))
    radii, sum_radii = lp_optimum(result.x.reshape(n, 2))
    return result.x.reshape(n, 2), radii, sum_radii
