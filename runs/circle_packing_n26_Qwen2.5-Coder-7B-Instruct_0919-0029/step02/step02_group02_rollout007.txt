import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def fix_bounds(x, bounds):
    return np.clip(x, bounds[:, 0], bounds[:, 1])

def objective(x, bounds):
    centers = x.reshape(-1, 2)
    circles = np.concatenate([centers, lp_optimum(centers)[0].reshape(-1, 1)], axis=1)
    return -circles[:, 2].sum()

def run_packing():
    n = 26
    bounds = np.vstack((np.zeros(2 * n), np.ones(2 * n))).T
    result = differential_evolution(objective, bounds, popsize=50, maxiter=500, verbose=0)
    centers = np.reshape(result.x[:2 * n], (-1, 2))
    radii, sum_radii = lp_optimum(centers)
    return centers, radii, sum_radii
