import cvxpy as cp
import numpy as np

def f_global(X):
    # Variables for center coordinates and radii
    N = int(len(X) / 3)
    x = X[:N]
    y = X[N:2*N]
    r = X[2*N:]

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints:
    constraints = []

    # Boundary constraints
    for i in range(N):
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])
        constraints.append(r[i] >= 0)

    # Non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.norm([x[i] - x[j], y[i] - y[j]], 2) >= r[i] + r[j])

    # Formulate and solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, verbose=False)

    if np.isinf(problem.value):
       problem.solve(solver=cp.SCS, verbose=False)

    if np.isinf(problem.value):
        return False, "infeasible"

    return problem.value, r.value, x.value, y.value
