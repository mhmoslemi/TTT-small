import numpy as np
from scipy.optimize import linprog
from scipy.stats import norm

def initial_grid_plus_center():
    grid_size = 4
    step = 0.25
    half_step = step / 2
    grid_points = np.arange(half_step, 1.0, step)
    centers = np.array([[x, y] for x in grid_points for y in grid_points])
    center = np.array([[0.5, 0.5]])
    centers = np.concatenate([centers, center])
    return centers

def gaussian_perturb(centers, sigma=0.02):
    noise = np.random.normal(0, sigma, centers.shape)
    new_centers = centers + noise
    return np.clip(new_centers, 0, 1)

def lp_solve(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26
    step, step_min = 0.15, 1e-3
    decay_factor = 0.995
    NUM_ITER = 4000
    NUM_RESTARTS = 40

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        for _ in range(50):  # Pre-solve for better starting points
            _, sum = lp_solve(centers)
            if sum > best_sum:
                best_sum = sum
                best_centers = centers.copy()

        cur_centers, cur_sum, cur_radii = centers.copy(), 0, np.zeros(n)

        for it in range(NUM_ITER):
            k = np.random.randint(n)
            step_k = step * max(1e-3, 1 - it / NUM_ITER)
            new_centers = cur_centers.copy()
            new_centers[k] += norm.ppf(np.random.random()) * step_k
            new_centers[k] = np.clip(new_centers[k], 0, 1)

            new_radii, new_sum = lp_solve(new_centers)

            if new_sum >= cur_sum or np.random.random() < np.exp((new_sum - cur_sum) / step):
                cur_centers, cur_sum, cur_radii = new_centers, new_sum, new_radii
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centers.copy()

            step = max(step_min, step * decay_factor)

    best_radii, _ = lp_solve(best_centers)
    best_radii = np.maximum(best_radii, 0)
    best_sum = np.sum(best_radii)
    return best_centers, best_radii, best_sum
