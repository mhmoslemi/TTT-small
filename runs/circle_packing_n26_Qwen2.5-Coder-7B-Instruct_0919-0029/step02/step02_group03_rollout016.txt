import numpy as np
from scipy.optimize import differential_evolution

def f_global(X):
    N = int(len(X) / 3)
    radii = X[2::3]
    penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = (X[0 + 3 * i] - X[0 + 3 * j]) ** 2 + (X[1 + 3 * i] - X[1 + 3 * j]) ** 2
            penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
        penalty += max(radii[i] - X[0 + 3 * i], 0) ** 2
        penalty += max(radii[i] - X[1 + 3 * i], 0) ** 2
        penalty += max(radii[i] - (1 - X[0 + 3 * i]), 0) ** 2
        penalty += max(radii[i] - (1 - X[1 + 3 * i]), 0) ** 2
    return -(radii.sum()) + 1e6 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    result = differential_evolution(f_global, bounds, popsize=20, maxiter=300, seed=42, workers=1)
    
    X = result.x
    centers = X.reshape((N, 3))[:, :2]
    radii = X.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
