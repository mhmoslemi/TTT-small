import numpy as np
import cvxpy as cp

def run_packing():
    N = 26
    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N)
    
    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints: boundary containment and non-overlap
    constraints = []
    for i in range(N):
        constraints.append(cp.Le(r[i], x[i]))
        constraints.append(cp.Le(r[i], 1 - x[i]))
        constraints.append(cp.Le(r[i], y[i]))
        constraints.append(cp.Le(r[i], 1 - y[i]))
        constraints.append(r[i] >= 0)
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.Le(cp.norm(cp.array([x[i] - x[j], y[i] - y[j]]), 2), r[i] + r[j]))
    
    # Combine objective and constraints
    problem = cp.Problem(objective, constraints)
    
    # Solve the problem
    result = problem.solve(solver=cp.ECOS, verbose=False, abstol=1e-7, reltol=1e-7)
    if np.isinf(result):
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.SCS, verbose=False, abstol=1e-7, reltol=1e-7)
        if np.isinf(result):

                        penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
                    penalty += max(radii[i] - X[0 + 3 * i], 0) ** 2
                    penalty += max(radii[i] - X[1 + 3 * i], 0) ** 2
                    penalty += max(radii[i] - (1 - X[0 + 3 * i]), 0) ** 2
                    penalty += max(radii[i] - (1 - X[1 + 3 * i]), 0) ** 2
                return -(radii.sum()) + 1e6 * penalty

            bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
            result = differential_evolution(f_global, bounds, popsize=20, maxiter=200, seed=42, workers=1)
            X = result.x

        radii = r.value
        sum_radii = np.sum(radii)
        
    # Clip negative values to zero and boundary overshoots to boundary
    radii = np.clip(radii, 0, None)
    centers = np.clip(centers, 0, 1)
    
    # Repack to ensure no significant overlaps
    from scipy.spatial.distance import pdist, squareform
    D = squareform(pdist(centers))
    overlaps_mask = (D < 2 * (radii[:, None] + radii))[np.triu_indices(N, k=1)]
