import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def create_initial_grid_and_add_center():
    centers = np.array([(i*0.2, j*0.2) for i in range(5) for j in range(5)] + [(0.5, 0.5)])
    return centers

def perturb_randomly(centers, epsilon0):
    n = centers.shape[0]
    epsilon = epsilon0 * np.random.rand(n, 2)
    epsilon -= epsilon.mean(axis=0)
    epsilon /= np.linalg.norm(epsilon, axis=1, keepdims=True)
    perturbed_centers = np.clip(centers + epsilon, 0, 1)
    return perturbed_centers

def move_center(centers, k, step):
    theta = step * np.random.uniform()
    direction = np.array([np.cos(theta), np.sin(theta)])
    new_center = centers[k] + direction
    new_center = np.clip(new_center, 0, 1)
    centers[k] = new_center
    return centers

def random_accept(delta, temp):
    return np.random.rand() < np.exp(delta / temp)

def run_packing():
    n = 26
    num_starts = 30
    num_iter = 4000
    step = 0.15
    step_min = 1e-3
    tau_step = 0.997
    tau_temp = 0.997
    temp = 0.5

    best_centers = None
    best_sum = -np.inf

    for start in range(num_starts):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, epsilon0=step)
        radii, sum_r = lp_optimum(centers)

        for _ in range(num_iter):
            k = np.random.randint(n)
            new_centers = move_center(centers.copy(), k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if sum_r > best_sum:
                    best_centers = centers.copy()
                    best_sum = sum_r
            step = max(step_min, step * tau_step)
            temp *= tau_temp

        if sum_r > best_sum:
            best_centers = centers.copy()
            best_sum = sum_r

    final_radii, _ = lp_optimum(best_centers)
    return best_centers, final_radii, best_sum
