import numpy as np
from scipy.optimize import differential_evolution, linprog
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def hill_climbing(centers, max_iter=1000, step=0.01, decay_factor=0.995, min_step=0.001):
    n = centers.shape[0]
    best_centers = centers
    best_sum = lp_optimum(best_centers)[1]

    for _ in range(max_iter):
        idx = random.randint(0, n - 1)
        delta = np.random.uniform(-step, step, size=2)
        new_c = np.clip(best_centers[idx] + delta, 0, 1)
        new_centers = best_centers.copy()
        new_centers[idx] = new_c

        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
            step = max(step * decay_factor, min_step)

    return best_centers, lp_optimum(best_centers)[1]

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2 * n)
    objective = lambda x: -lp_optimum(x.reshape(n, 2))[1]
    initial_guess = np.random.rand(2 * n)

    result = differential_evolution(objective, bounds=bounds, ngen=500, popsize=50, strategy='best2bin')
    centers = result.x.reshape(n, 2)
    best_sum = lp_optimum(centers)[1]
    
    refined_centers, refined_sum = hill_climbing(centers)
    refined_sum = lp_optimum(refined_centers)[1]

    return refined_centers, lp_optimum(refined_centers)[0], refined_sum
