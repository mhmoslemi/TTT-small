import numpy as np
import cvxpy as cp
from scipy.optimize import linprog
from scipy.spatial.distance import squareform

def solve_radius_lp(centre):
    n = centre.shape[0]
    dists = squareform(np.sqrt(np.sum((centre - centre[:, np.newaxis, :]) ** 2, axis=2)))
    np.fill_diagonal(dists, np.inf)

    c = -np.ones(n)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.extend([[0]*n, [0]*n, [0]*n, [0]*n])
        b_ineq.extend([centre[i, 0] - 1e-12, 1 - centre[i, 0] - 1e-12, centre[i, 1] - 1e-12, 1 - centre[i, 1] - 1e-12])

        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs', options={"feastol": 1e-9, "abstol": 1e-9, "mip": True})
    
    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def solve_socp(radii, centre):
    n = len(radii)
    centre = np.array(centre)
    reference =centre

    r = cp.Variable(n)
    x = cp.Variable(n)
    y = cp.Variable(n)
    objective = cp.Minimize(cp.sum square((np.hstack((x, y)) - reference) ** 2))
    constraints = []
    
    for i in range(n):
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])

    dist_matrix = squareform(np.sqrt(np.sum((centre[:, np.newaxis, :] - centre[np.newaxis, :, :]) ** 2, axis=2)))
    np.fill_diagonal(dist_matrix, np.inf)
    
    for i, j in zip(*np.triu_indices(n, k=1)):
        constraints.append(cp.norm(cp.hstack((x[i], y[i])) - cp.hstack((x[j], y[j]))) >= radii[i] + radii[j])

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, eps_abs=1e-8, eps_rel=1e-8, max_iter=2000)

    if problem.status == cp.OPTIMAL:
        return np.hstack((x.value, y.value)), problem.value
    else:
        raise Exception("SOCP optimization failed")
