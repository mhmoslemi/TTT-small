import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def penalty(penalties, alpha, beta):
    return np.sum(penalties * penalties) / alpha + \
           np.sum(penalties * penalties[:, None] / beta, axis=0)

def objective(z, alpha, beta):
    n = len(z) // 2
    centers = z[:n].reshape((-1, 2))
    radii = z[n:]
    dist_matrix = squareform(pdist(centers))
    penalties = np.maximum(0, radii[np.triu_indices(n, k=1)] + radii[np.triu_indices(n, k=1)] - dist_matrix[np.triu_indices(n, k=1)])
    penalties += np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - (1 - centers[:, 0])) + np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii - (1 - centers[:, 1]))
    return -(np.sum(radii) + penalty(penalties, alpha, beta))

def initial_solution(n, alpha, beta):
    # Structured start
    x = np.tile(np.linspace(0.2, 0.8, 5), (5, 1))
    x[2:5, 1::2] += 0.1
    y = np.tile(np.linspace(0.2, 0.8, 5), 5).reshape(-1, 2)
    jitter = np.random.uniform(-0.005, 0.005, size=(n, 2))
    centers = x + y + jitter
    # Tiny random jitter
    while True:
        radii = np.full(n, 1e-3)
        penalties = np.maximum(0, radii[np.triu_indices(n, k=1)] + radii[np.triu_indices(n, k=1)] - squareform(pdist(centers[np.triu_indices(n, k=1)])))
        penalties += np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - (1 - centers[:, 0])) + np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii - (1 - centers[:, 1]))
        if np.all(penalties < 0.01):
            break
        radii *= 0.5
    return np.vstack((centers, radii))

def solve_packing_problem(n, alpha, beta):
    centers = initial_solution(n, alpha, beta)
    result = minimize(objective, centers.flatten(), args=(alpha, beta), method='L-BFGS-B', jac=None, bounds=[(0, 1)] * 2 * n)
    centers = result.x[:n].reshape((-1, 2))
    radii = result.x[n:]
    return centers, radii, -result.fun
