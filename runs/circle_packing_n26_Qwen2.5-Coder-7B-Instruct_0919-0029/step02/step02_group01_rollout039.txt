import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise RuntimeError("LP solver did not converge")
    return result.x, -result.fun

def create_initial_grid_and_add_center():
    grid = np.array([(x + 0.1, y + 0.1) for x in np.arange(0, 1, 0.2) for y in np.arange(0, 1, 0.2)])
    center = (0.5, 0.5)
    return np.vstack((grid, center))

def perturb_randomly(centers, epsilon0):
    n = centers.shape[0]
    epsilon = epsilon0
    new_centers = centers.copy()
    for _ in range(50):
        k = np.random.randint(n)
        step = epsilon * np.random.uniform(0.1, 1.0)
        direction = np.random.uniform(-1.0, 1.0, size=2)
        direction /= np.linalg.norm(direction)
        new_c = centers[k] + step * direction
        new_c = np.maximum(np.minimum(new_c, 1.0), 0.0)
        new_centers[k] = new_c
        if epsilon > 1e-3:
            epsilon *= 0.95
    return new_centers

def move_center(centers, k, step):
    direction = np.random.uniform(-1.0, 1.0, size=2)
    direction /= np.linalg.norm(direction)
    new_c = centers[k] + step * direction
    new_c = np.maximum(np.minimum(new_c, 1.0), 0.0)
    new_centers = centers.copy()
    new_centers[k] = new_c
    return new_centers

def random_accept(delta, temp):
    return np.random.random() < np.exp(delta / temp)

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_centers, best_radii, best_sum = None, None, -np.inf

    for restart in range(30):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, 0.05)

        try:
            radii, sum_r = solve_lp(centers)
        except RuntimeError:
            continue

        step, temp = 0.15, 0.5
        for _ in range(2000):
            k = rng.integers(0, len(centers))
            new_centers = move_center(centers, k, step)
            try:
                new_radii, new_sum = solve_lp(new_centers)
            except RuntimeError:
                continue
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = new_centers.copy(), new_radii.copy(), new_sum
            step *= 0.997
            temp *= 0.997

    # final fine-tune
    best_radii, best_sum = solve_lp(best_centers)

    return best_centers, best_radii, best_sum
