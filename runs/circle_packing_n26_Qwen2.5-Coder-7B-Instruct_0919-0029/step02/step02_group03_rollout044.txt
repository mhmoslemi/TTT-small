import numpy as np
from scipy.optimize import differential_evolution, dual_annealing, linprog

N = 26

def compute_sum_radii(centers):
    n = centers.shape[0]
    A_ub = []
    b_ub = []
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(centers[i, 0])
        A_ub.append(row)
        b_ub.append(centers[i, 1])
        A_ub.append(row)
        b_ub.append(1 - centers[i, 0])
        A_ub.append(row)
        b_ub.append(1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
    A = np.vstack(A_ub)
    b = np.array(b_ub)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing():
    centers = np.random.rand(26, 2)  # Initial random centers
    centers = 0.5 + (centers - 0.5) * 0.95  # Perturb to keep near center
    result = differential_evolution(objective, bounds=[(0, 1)] * 52, popsize=12, maxiter=200, strategy='best1bin')
    X = result.x
    initial_centers = X.reshape((N, 2))
    final_centers = simulated_annealing(initial_centers)
    radii = compute_sum_radii(final_centers)
    sum_radii = radii.sum()
    return final_centers, radii, sum_radii
