import numpy as np
from scipy.optimize import differential_evolution, NonlinearConstraint, LinearConstraint, Bounds

def penalty_fcn(x):
    n = 26
    centers = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    pairs = np.array([(i, j) for i in range(n) for j in range(i+1, n)])
    g = np.sqrt((centers[pairs[:, 0], 0] - centers[pairs[:, 1], 0])**2 +
               (centers[pairs[:, 0], 1] - centers[pairs[:, 1], 1])**2) - (radii[pairs[:, 0]] + radii[pairs[:, 1]])
    violate = np.maximum(g, 0)
    return -radii.sum() + 1e8 * violate.sum()

def residual_constraint(x):
    n = 26
    centers = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    pairs = np.array([(i, j) for i in range(n) for j in range(i+1, n)])
    return np.sqrt((centers[pairs[:, 0], 0] - centers[pairs[:, 1], 0])**2 +
                   (centers[pairs[:, 0], 1] - centers[pairs[:, 1], 1])**2) - (radii[pairs[:, 0]] + radii[pairs[:, 1]])

def radius_bounds_constraint(x):
    n = 26
    radii = x[2*n:]
    return np.maximum(np.minimum(radii, 0.5), 0) - radii

def bounds_constraint(x):
    n = 26
    centers = x[:2*n].reshape((n, 2))
    return np.maximum(np.maximum(centers, 0), 1 - np.minimum(centers, 1)) - centers

def run_packing():
    n = 26
    initial_guess = np.zeros(3*n)
    initial_guess[:n] = 0.5
    initial_guess[n:2*n] = 0.5
    initial_guess[2*n:] = 0.0999
    initial_guess[2*(n-1)//2] = 0.5
    initial_guess[2*(n-1)//2 + 1] = 0.5
  
    bounds = Bounds(np.concatenate([np.full(2*n, 0), np.full(n, 0)]),
                    np.concatenate([np.full(2*n, 1), np.full(n, 0.5)]))

    nlcon_nonlin = NonlinearConstraint(residual_constraint, 0, np.inf, jac=lambda x: 
                                        2 * (x[:2*n].reshape((n, 2)) - x[(2*n//2):].reshape((n, 1, 2))).flatten() /
                                        np.sqrt((x[:2*n].reshape((n, 2)) - x[(2*n//2):].reshape((n, 1, 2))).sum(axis=2) +
                                            1e-16)[:, :, None])

    nlcon_radius = NonlinearConstraint(radius_bounds_constraint, 0, 0, jac=lambda x: 
                                      np.concatenate((np.eye(2*n, n, dtype=int), -np.eye(n)), axis=1))

    nlcon_bounds = NonlinearConstraint(bounds_constraint, 0, 0, jac=lambda x: 
                                       np.concatenate((np.eye(2*n, n, dtype=int), -np.eye(n), -np.eye(n)), axis=1))

    res = differential_evolution(penalty_fcn, bounds=bounds, method='ML', maxiter=400, popsize=15, seeds=10, seed=42)
    centers = res.x[:2*n].reshape((n, 2))
    radii = res.x[2*n:]
    sum_radii = -res.fun
    return centers, radii, sum_radii
