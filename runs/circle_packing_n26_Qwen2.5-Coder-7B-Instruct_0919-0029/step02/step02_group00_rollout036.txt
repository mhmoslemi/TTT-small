import numpy as np
from scipy.spatial import minkowski_distance
from scipy.optimize import minimize
from scipy.spatial import ConvexHull

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def create_initial_centers():
    a = np.array([1, 0])
    b = np.array([0.5, np.sqrt(3)/2])
    vertices = []
    for i in range(-3, 10):
        for j in range(-3, 10):
            p = i*a + j*b
            if np.all(p >= -0.5) and np.all(p <= 1.5):
                pm = p % 1
                if pm[0] >= 0.5: pm[0] -= 1
                if pm[1] >= 0.5: pm[1] -= 1
                vertices.append(pm)
    vertices = np.array(vertices)[:26]
    return vertices, np.zeros(26)

def add_penalties(z, lambda_pen):
    n = z.shape[0] // 3
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    penalties = 0
    for i in range(n):
        penalties += np.maximum(0, r[i] - x[i]) + np.maximum(0, x[i] + r[i] - 1)
        penalties += np.maximum(0, r[i] - y[i]) + np.maximum(0, y[i] + r[i] - 1)
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((x[i] - x[j])**2 + (y[i] - y[j])**2))
            penalties += np.maximum(0, r[i] + r[j] - dist)
    return penalties

def PSO(objective, x0, bounds, lambda_pen, n_particles=200, max_iter=300, v_max=[0.05, 0.05, 0.02]):
    from pyswarms.single import GlobalBestPSO
    optimizer = GlobalBestPSO(n_particles, dimensions=78, options={'c1': 0.5, 'c2': 0.3, 'w':0.9})
    best_cost, best_pos = optimizer.optimize(lambda x: objective(x, lambda_pen), iters=max_iter, bounds=bounds)
    return best_cost, best_pos

def objective(z, lambda_pen):
    return -np.sum(z[26:]) + lambda_pen * add_penalties(z, lambda_pen)

def refine_solution(z):
    n = z.shape[0] // 3
    x, y, r = z[:n], z[n:2*n], z[2*n:]
    A_ineq = np.zeros((n, 3*n))
    b_ineq = np.zeros(n)
    for i in range(n):
        A_ineq[i, i] = 1
        A_ineq[i, n+i] = 1
        A_ineq[i, 2*n+i] = -1
        A_ineq[i, n+i] = 1
        A_ineq[i, n+n+i] = -1
        A_ineq[i, 2*n+i] = -1
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((x[i]-x[j])**2 + (y[i]-y[j])**2))
            A_ineq[i, n+j] = 1
            A_ineq[i, n+n+j] = 1
            A_ineq[i, 2*n+i] = -1
            A_ineq[i, n+j] = 1
            A_ineq[i, n+n+j] = -1
            A_ineq[i, 2*n+i] = -1
    res = minimize(lambda r: -np.sum(r) + np.sum(np.maximum(0, dist - r)), r, bounds=[(0, None)]*n, constraints={'type': 'ineq', 'fun': lambda r: q_exp*r})
    return x, y, res.x

def q_exp(x):
    return np.exp(-x)

def find_contact_graph(centers):
    n = centers.shape[0]
    dist_matrix = minkowski_distance(centers, centers, p=1)
    # Add a small epsilon to ensure strict inequality for convex hull
    dist_matrix -= np.eye(n) * 1e-3
    hull = ConvexHull(dist_matrix)
    contact_graph = np.zeros((n, n))
    for simplex in hull.simplices:
        for i in range(len(simplex)):
            for j in range(i+1, len(simplex)):
                contact_graph[i, j] = 1
                contact_graph[j, i] = 1
    return contact_graph

def run_packing():
    centers, initial_radii = create_initial_centers()
    x0 = np.hstack((centers.reshape(-1), initial_radii))
    bounds = [(-1e-12, 1), (-1e-12, 1)] * 26 + [(0, 1)] * 26
    c = 0
    lambda_pen = 1e4
    while c < 3:
       lambda_pen *= 10
        cost, best_pos = PSO(objective, x0, bounds, lambda_pen, max_iter=300)
        contact_graph = find_contact_graph(best_pos[:26].reshape(-1, 2))
        A_ineq = np.zeros((n, 3*n))
        b_ineq = np.zeros(n)
        for i in range(n):
            A_ineq[i, i] = 1
            A_ineq[i, n+i] = 1
            A_ineq[i, 2*n+i] = -1
            A_ineq[i, n+i] = 1
            A_ineq[i, n+n+i] = -1
            A_ineq[i, 2*n+i] = -1
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((best_pos[n+i] - best_pos[n+j])**2 + (best_pos[2*n+i] - best_pos[2*n+j])**2))
                A_ineq[i, n+j] = 1
                A_ineq[i, n+n+j] = 1
                A_ineq[i, 2*n+i] = -1
                A_ineq[i, n+j] = 1
                A_ineq[i, n+n+j] = -1
                A_ineq[i, 2*n+i] = -1
        res = minimize(lambda r: -np.sum(r), np.ones(n), bounds=[(0, None)]*n, constraints={'type': 'eq', 'fun': lambda r: np.sum(r*r) - np.sum(A_ineq @ np.vstack((np.ones(n), best_pos[n:26].reshape(-1, 2), (contact_graph @ best_pos[26:]**0.5).reshape(-1, 1)) ** 2))}
        new_centers, new_radii, _ = refine_solution(np.hstack((best_pos[:26], res.x)))
        c += 1
    return new_centers, new_radii, -objective(new_centers, new_radii)

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print((centers, radii, sum_radii))
    assert validate_packing(centers, radii)[0], "Validation failed"
