import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    radii_lower_bounds = np.clip(centers[:, 0], 0, 1)
    radii_upper_bounds = np.clip(1 - centers[:, 0], 0, 1)
    radii_lower_bounds = np.clip(centers[:, 1], 0, 1)
    radii_upper_bounds = np.clip(1 - centers[:, 1], 0, 1)

    dists = squareform(pdist(centers))
    A_ub = np.zeros((n * (n - 1) // 2, 2 * n))
    b_ub = np.zeros(n * (n - 1) // 2)

    for i in range(n):
        for j in range(i + 1, n):
            idx = (i * (n - 1) // 2 + j - i - 1) * 2
            A_ub[idx, 2 * i:2 * i + 2] = 1
            A_ub[idx + 1, 2 * j:2 * j + 2] = 1
            b_ub[idx] = dists[i, j] - 1e-12
            b_ub[idx + 1] = dists[i, j] - 1e-12

    A = np.zeros((4 * n, 2 * n))
    b = np.zeros(4 * n)

    for i in range(n):
        A[4 * i, 2 * i] = 1
        A[4 * i, 2 * i + 1] = 0
        b[4 * i] = radii_lower_bounds[i]
        A[4 * i + 1, 2 * i] = -1
        A[4 * i + 1, 2 * i + 1] = 0
        b[4 * i + 1] = -radii_upper_bounds[i]
        A[4 * i + 2, 2 * i] = 0
        A[4 * i + 2, 2 * i + 1] = 1
        b[4 * i + 2] = radii_lower_bounds[i]
        A[4 * i + 3, 2 * i] = 0
        A[4 * i + 3, 2 * i + 1] = -1
        b[4 * i + 3] = -radii_upper_bounds[i]

    bounds = [(0, None)] * 2 * n
    result = linprog([1] * n + [0] * n, A_ub=A_ub, b_ub=b_ub, A=A, b=b, bounds=bounds, method='highs')
    radii_x, _ = result.x[:n], result.x[n:]
    radii = np.sqrt(radii_x**2 + radii_y**2)
    return radii, -result.fun

def dual_based_gradient(centers, duals):
    n = centers.shape[0]
    grad_centers = np.zeros((n, 2))
    radii = lp_optimum(centers)[0]

    distances = squareform(pdist(centers))
    for j in range(n):
        for i in range(j):
            if distances[i, j] > 1e-12 and duals[j * (n - 1) // 2 + i] > 0:
                grad_centers[j] += duals[j * (n - 1) // 2 + i] * (centers[i] - centers[j]) / distances[i, j]
        grad_centers[j, 0] += duals[4 * j] * (radii[j] ** 2 - centers[j, 0] ** 2) / (2 * radii[j])
        grad_centers[j, 0] -= radii[j] * seen[j] if seen[j] > 0 else 0

        grad_centers[j, 1] += duals[4 * j + 2] * (radii[j] ** 2 - centers[j, 1] ** 2) / (2 * radii[j])
        grad_centers[j, 1] -= radii[j] * seen[j] if seen[j] > 0 else 0

    seen = np.sign(centers[:, 0] - radii[centers[:, 0] - radii > 0]) + np.sign(centers[:, 1] - radii[centers[:, 1] - radii > 0])
    return grad_centers / n

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radius_grads = np.zeros((n, n))
    sum_radii = 0.0
    step = 0.1
    step_min = 1e-10
    decay_factor = 0.995
    NUM_ITERS = 250
    best_sum = -np.inf
    best_centers = None

    for start in range(30):
        centers = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            grad_centers = dual_based_gradient(centers, duals)
            normalized_grad = grad_centers / np.linalg.norm(grad_centers)
            new_centers = centers + step * normalized_grad
            new_centers = np.clip(new_centers, 0, 1)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()

            step = max(step_min, step * decay_factor)

    return best_centers, lp_optimum(best_centers)[0], best_sum

centers, radii, sum_radii = run_packing()
