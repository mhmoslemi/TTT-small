import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i in range(n):
        for j in range(i+1, n):
            A_ub[i*(n-1)//2 + 2*i - i*(i+1)//2 + j - i - 1, i] = 1
            A_ub[i*(n-1)//2 + 2*i - i*(i+1)//2 + j - i - 1, j] = 1
            b_ub[i*(n-1)//2 + 2*i - i*(i+1)//2 + j - i - 1] = np.linalg.norm(centers[i] - centers[j]) - 1e-12

    for i in range(n):
        A_ub[n*(n-1)//2 + 4*i, i] = 1
        b_ub[n*(n-1)//2 + 4*i] = centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 1, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 1] = 1 - centers[i, 0] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 2, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 2] = centers[i, 1] - 1e-12
        A_ub[n*(n-1)//2 + 4*i + 3, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 3] = 1 - centers[i, 1] - 1e-12

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def propose_move(centers):
    n = centers.shape[0]
    idx = np.random.randint(n)
    step = 0.2
    delta = np.random.randn(2)
    delta /= np.linalg.norm(delta)
    delta *= step
    new_c = centers[idx] + delta
    new_c = np.clip(new_c, 0, 1)
    return new_c

def hill_climbing(centers, step, n_iters):
    best_centers, best_sum = centers.copy(), None
    cur_centers = centers.copy()
    cur_radii, cur_sum = lp_optimum(centers)
    best_sum = cur_sum
    for _ in range(n_iters):
        new_c = propose_move(cur_centers)
        new_radii, new_sum = lp_optimum(np.vstack((cur_centers[:idx], new_c, cur_centers[idx+1:])))
        if new_sum > cur_sum:
            cur_centers = np.vstack((cur_centers[:idx], new_c, cur_centers[idx+1:]))
            cur_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_sum = cur_centers.copy(), new_sum
    return best_centers, best_sum

def run_packing():
    np.random.seed(42)
    
    NUM_STARS = 20
    NUM_ITERS = 4000
    step = 0.1
    
    best_sum = -np.inf
    best_centers, best_radii = None, None
    
    for _ in range(NUM_STARS):
        centers = np.random.rand(26, 2)
        cur_centers = centers.copy()
        cur_sum = hill_climbing(cur_centers, step, NUM_ITERS)[1]
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers, best_radii = hill_climbing(cur_centers, step, NUM_ITERS)
    
    initial_radii, initial_sum = lp_optimum(best_centers)
    return best_centers, initial_radii, initial_sum
