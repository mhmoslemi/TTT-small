import numpy as np
import cvxpy as cp
from scipy.optimize import linprog

# Function to read optimal centre locations
def read_registered_centres():
    import os
    filename = os.path.join("additional_materials", "centers.txt")
    with open(filename, "r") as file:
        centers = np.array([[float(num) for num in line.strip().split()] for line in file.readlines()], dtype=np.double)
    return centers

# Pre-compute max overlap allowed
max_radius = 0.5
max_overlap = np.pi * max_radius**2 - np.pi * max_radius**2 / 4

# Step 2.1: Radius sub-problem (LP)
def solve_LP(centres):
    n = centres.shape[0]
    c = -np.ones(n)
    A_ineq = np.zeros((3*n + n*(n-1)//2, n))
    b_ineq = np.zeros(3*n + n*(n-1)//2)
    for i in range(n):
        A_ineq[i, :] = np.eye(2)[:, i]
        b_ineq[i] = max_radius
    for i in range(n):
        A_ineq[n + i, :] = np.eye(2)[:, i]
        b_ineq[n + i] = 1 - max_radius
    for i in range(n):
        A_ineq[2*n + i, :] = np.eye(2)[:, i]
        b_ineq[2*n + i] = max_radius
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq[3*n + (j - i - 1) * i + i, :] = (centres[i] - centres[j])
            A_ineq[3*n + (j - i - 1) * i + i, i] = -2
            A_ineq[3*n + (j - i - 1) * i + i, j] = 2
            b_ineq[3*n + (j - i - 1) * i + i] = 0
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    return result.x, -result.fun

# Step 2.2: Centre sub-problem (SOCP)
def solve_SOCP(radii, centres):
    n = centres.shape[0]
    c = np.array([1.0, 0.01]) * np.mean(centres, axis=0)  # Weight on quadratic term, small number for continuity
    x = cp.Variable(n)
    y = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(c - np.concatenate((x, y))))
    constraints = [
        radii[i] <= x[i] for i in range(n)
    ] + [
        radii[i] <= 1 - x[i] for i in range(n)
    ] + [
        radii[i] <= y[i] for i in range(n)
    ] + [
        radii[i] <= 1 - y[i] for i in range(n)
    ] + [
        cp.norm(centres[i, :] - centres[j, :], p=2) >= radii[i] + radii[j] for i in range(n) for j in range(i+1, n)
    ]
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS)
    if problem.status != cp.OPTIMAL:
        problem.solve(solver=cp.OSQP)
    return np.vstack((x.value, y.value)).T

# Main alternating algorithm
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Returns (centers, radii, sum_radii) as described in the function signature."""
    initial_centres = read_registered_centres()
    n = initial_centres.shape[0]
    tolerance_r = 1e-6
    tolerance_c = 1e-6
    max_iter = 20

    # Pre-check: Clip initial centres to keep radii 0 feasible
    initial_radii = np.zeros(n)

    for k in range(max_iter):
        # 1. Radius sub-problem
        radii, sum_radii = solve_LP(initial_centres)

        # 2. Centre sub-problem
        initial_centres = solve_SOCP(radii, initial_centres)

        # 3. Convergence test
        if abs(sum_radii - np.sum(initial_radii)) < tolerance_r and \
           np.max(np.linalg.norm(initial_centres - np.concatenate((initial_centres[:,0][:,np.newaxis], initial_centres[:,1][:,np.newaxis]), axis=1), axis=1)) < tolerance_c:
            break

    # 4. Final polishing
    final_radii, final_sum_radii = solve_LP(initial_centres)
    return initial_centres, final_radii, final_sum_radii
