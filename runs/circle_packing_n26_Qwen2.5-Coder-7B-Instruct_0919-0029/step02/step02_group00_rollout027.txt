import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def objective(vars, pair_distance_upper_bound):
    centers = vars[:52].reshape((26, 2))
    radii = vars[52:]
    return -np.sum(radii)

def constraint(vars, pair_distance_upper_bound):
    centers = vars[:52].reshape((26, 2))
    radii = vars[52:]
    # Boundary constraints
    x_low = 0.0
    x_high = 1.0
    y_low = 0.0
    y_high = 1.0
    bounds_x_low = x_low - radii
    bounds_y_low = y_low - radii
    bounds_x_high = x_high + radii
    bounds_y_high = y_high + radii
    violated_constraints = []
    for idx in range(len(bounds_x_low)):
        if bounds_x_low[idx] < 0:
            violated_constraints.append(bounds_x_low[idx])
        if bounds_x_high[idx] > 1:
            violated_constraints.append(bounds_x_high[idx])
        if bounds_y_low[idx] < 0:
            violated_constraints.append(bounds_y_low[idx])
        if bounds_y_high[idx] > 1:
            violated_constraints.append(bounds_y_high[idx])

    # Pairwise distance constraints
    distances = pdist(centers)
    violating_distances = distances[distances < radii[np.argmin(distances)] + radii[np.argmax(distances)] - 1e-12]
    violated_constraints += (violating_distances**2 < (radii[np.argmin(distances)] + radii[np.argmax(distances)])**2).astype(int)

    return violated_constraints

def run_packing():
    np.random.seed(42)
    # Stage 1: Structured initialization
    hex_radius = np.sqrt(3) / 3
    hexx = np.arange(-hex_radius, 1+hex_radius, 0.5*hex_radius)
    hexy = np.arange(-hex_radius, 1+hex_radius, 1.0*hex_radius)
    points = np.zeros((0, 2))
    for y in hexy:
        if y % 1.0 != 0:
            x_off = 0.5 * hex_radius
        else:
            x_off = 0.0
        for x in hexx + x_off:
            if np.hypot(x, y) <= 1.0:
                points = np.append(points, [[x, y]], axis=0)
    centres_initial = points[:26]
    radii_initial = np.zeros(26)
    vars_initial = np.concatenate((centres_initial.flatten(), radii_initial))

    # Stage 2: Non-Linear refinement
    pair_distance_upper_bound = np.sqrt(2)  # Diameter of the bounding box
    bnds = [(0.0, 1.0)] * 52 + [(0.0, None)] * 26
    constraints = [{'type': 'ineq', 'fun': constraint, 'args': (pair_distance_upper_bound,)}]
    result_opt = minimize(objective, vars_initial, method='trust-constr', bounds=bnds, constraints=constraints, jac=None, options={'maxiter': 200})
    centers_opt = result_opt['x'][:52].reshape((26, 2))
    radii_opt = result_opt['x'][52:]
    final_sum_radii = -result_opt['fun']

    # Stage 3: Stochastic restarts & polishing
    best_result_opt = result_opt
    for _ in range(2):
        centres_initial = points[:26] + np.random.normal(scale=0.01, size=(26, 2))
        centres_initial = np.clip(centres_initial, 0.0, 1.0)
        vars_initial = np.concatenate((centres_initial.flatten(), radii_initial))
        result_opt = minimize(objective, vars_initial, method='trust-constr', bounds=bnds, constraints=constraints, jac=None, options={'maxiter': 200})
        if -result_opt['fun'] > -best_result_opt['fun']:
            best_result_opt = result_opt

    centers_opt = best_result_opt['x'][:52].reshape((26, 2))
    radii_opt = best_result_opt['x'][52:]
    final_sum_radii = -best_result_opt['fun']

    return centers_opt, radii_opt, final_sum_radii
