import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    radii = np.zeros(N)
    x, y = zip(*centers)
    A = []
    b = []
    for i in range(N):
        row = np.zeros(N)
        row[i] = 1
        A.append(row)
        b.append(x[i])
        A.append(row)
        b.append(y[i])
        A.append(row)
        b.append(1 - x[i])
        A.append(row)
        b.append(1 - y[i])
    for i in range(N):
        for j in range(i+1, N):
            dist = np.linalg.norm(np.array(centers[i]) - np.array(centers[j]))
            row = np.zeros(N)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dist)
    A = np.vstack(A)
    b = np.array(b)
    c = -np.ones(N)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*N, method='highs')
    if res.success:
        radii[:N//2] = res.x[:N//2]
        radii[N//2:] = res.x[N//2:]
    return sum(radii), radii
