import numpy as np
import cvxpy as cp

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    r = cp.Variable(n, nonneg=True)
    constraints = []
    constraints += [r[i] <= centres[i, 0] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 0] for i in range(n)]
    constraints += [r[i] <= centres[i, 1] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 1] for i in range(n)]
    constraints += [dists[i, j] >= r[i] + r[j] for i in range(n) for j in range(i + 1, n)]

    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.HIGHS, verbose=False)
    
    return r.value, -problem.value

def solve_socp(radii, centres):
    n = centres.shape[0]
    r = cp.Variable(n, nonneg=True)
    c = cp.Variable(n, complex)

    constraints = []
    constraints += [r[i] <= centres[i, 0] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 0] for i in range(n)]
    constraints += [r[i] <= centres[i, 1] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 1] for i in range(n)]
    constraints += [cp.norm(c[i] - c[j]) >= r[i] + r[j] for i in range(n) for j in range(i + 1, n)]

    X = cp.sum_hermitian_product(c.reshape(n, 1, 1), c.reshape(1, n, 1)) - np.eye(n)
    objective = cp.Minimize(cp.norm(cp.real(X), p='nuclear'))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, verbose=False)
    
    return c.value.real, -problem.value

def run_packing():
    np.random.seed(42)
    initial_centres = np.array([[(i + j) / 27.0 for i in range(6)] for j in range(4)] + [[x, y] for x in [0, 0.5, 1] for y in [0, 0.5, 1]])
    initial_centres = initial_centres[initial_centres[:, 0] < 1][:26]

    best_centres = np.copy(initial_centres)
    best_radii = np.zeros(26)
    best_sum_radii = 0

    for _ in range(3):
        np.random.seed()
        gap = 0.25
        while gap > 1e-6:
            gap = 0.0
            niter = 10

            for _ in range(niter):
                radii, sum_radii = solve_radius_lp(best_centres)
                if sum_radii > best_sum_radii:
                    best_centres, best_sum_radii = best_centres, sum_radii
                    gap *= 0.9

                positions, _ = solve_socp(radii, best_centres)
                diff = positions - best_centres
                gap = gap if gap < np.max(abs(diff)) else np.max(abs(diff))
                best_centres += diff * gap / 2

            radii, sum_radii = solve_radius_lp(best_centres)
            if sum_radii > best_sum_radii:
                best_centres, best_sum_radii = best_centres, sum_radii

    final_radii, _ = solve_radius_lp(best_centres)
    return best_centres, final_radii, best_sum_radii
