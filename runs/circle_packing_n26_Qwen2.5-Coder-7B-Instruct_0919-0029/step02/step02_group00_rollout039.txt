def hexagonal_lattice_in_unit_square():

    for i in range(3):
        for j in range(3):
            indices = np.array([i*3 + j*3 + np.array([0, 1, 2, 3, 4, 5])])
            centers[indices % 26] = np.array([np.outer(x, [1, 1, 1, 0.5, 0, 0])[i] + y[i] + 0.5 * (indices % 2) / 3,

            centers[indices % 26] = centers[indices % 26] / np.sqrt(np.sum(centers[indices % 26] ** 2, axis=1))[:, np.newaxis]
    centers = centers[centers[:, 0] > 0]
    np.random.seed(42)
    centers = centers[indices[(np.random.rand(len(indices)) < len(indices)/26))]
    centers = centers[centers[:, 0] > 0][:26]
    return centers

def solve_LP(centers):
    n = centers.shape[0]
    r = Variable(n)

    objective = Maximize(sum(r))
    constraints = [
        r <= centers[:, 0],
        r <= 1 - centers[:, 0],
        r <= centers[:, 1],
        r <= 1 - centers[:, 1],
        sum(r) <= 2 * (1 - np.mean(centers[:, 1]))
    ]
    lp = Problem(objective, constraints)
    lp.solve(method='highs')
    return np.array(r.value)
