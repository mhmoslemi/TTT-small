import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize
from scipy.stats import norm

def evaluate_solution(centers, radii):
    n = centers.shape[0]
    distances = cdist(centers, centers)
    penalty = 0.0
    for i in range(n):
        if radii[i] < 0:
            penalty += max(0, -radii[i])
        if radii[i] + centers[i, 0] > 1:
            penalty += max(0, radii[i] + centers[i, 0] - 1)
        if radii[i] + centers[i, 1] > 1:
            penalty += max(0, radii[i] + centers[i, 1] - 1)
        if radii[i] + centers[i, 0] < 0:
            penalty += max(0, -radii[i] - centers[i, 0])
        if radii[i] + centers[i, 1] < 0:
            penalty += max(0, -radii[i] - centers[i, 1])
        for j in range(i + 1, n):
            if distances[i, j] < radii[i] + radii[j]:
                penalty += max(0, radii[i] + radii[j] - distances[i, j])
    return -sum(radii) + penalty

def lp_refine(centers):
    n = centers.shape[0]
    distances = cdist(centers, centers)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)
        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(distances[i, j] - 1e-12)
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    result = minimize(c, np.ones(n), method='highs', constraints={'type': 'ineq', 'fun': A_ineq, 'jac': np.zeros((2*n, n))), bounds=bounds, options={'feastol': 1e-9})
    if result.success:
        return result.x
    else:
        return None

def psa_init(seed=None):
    if seed is not None:
        np.random.seed(seed)
    delta = 2 / np.sqrt(3)
    initial_centers = np.array([
        (i * delta, j * delta) for i in range(4) for j in range(4)
    ])
    initial_radii = np.zeros_like(initial_centers[:, 0])
    return np.vstack((initial_centers.flatten(), initial_radii.flatten()))

def run_packing():
