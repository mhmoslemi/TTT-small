import numpy as np
from scipy.optimize import differential_evolution, minimize, linprog, dual_annealing

# Constants
N = 26

def compute_sum_radii(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 0])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 1])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 0])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [1 if k == i else 0 if k == j else 0 for k in range(n)]
            A.append(row)
            b.append(dist)
    A = np.vstack(A)
    b = np.array(b)
    c = np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n)
    if res.success:
        return res.x.sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        new_pos = current_centers[i] + np.random.uniform(-0.05, 0.05, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0)] * N
    result = differential_evolution(objective, bounds, popsize=12, maxiter=200, strategy='best1bin', tol=1e-6, seed=42)
    
    X = result.x
    centers = X.reshape((N, 2))
    current_sum = compute_sum_radii(centers)
    best_centers = simulated_annealing(centers)
    
    result = minimize(lambda X: -compute_sum_radii(X.reshape((N, 2))), result.x, method='Nelder-Mead', bounds=bounds)
    X = result.x
    centers = X.reshape((N, 2))
    radii = compute_sum_radii(centers)
    
    return centers, np.array([radii]), radii.sum()

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)

This code creates a function `run_packing` that solves for the best-possible packing of 26 circles within a unit square using differential evolution to locate the optimal circle centers and simulated annealing to adjust them further. It uses a custom `compute_sum_radii` function to compute the sum of the radii given a set of circle centers, and an `objective` function to use with the DE algorithm. The resulting final centers and radii are returned by the function.
