import numpy as np
import random
from scipy.spatial.distance import cdist
from scipy.optimize import minimize, Bounds, NonlinearConstraint

def initialize_centres():
    # Generate a hexagonal grid within the unit square
    num_rows = int(np.sqrt(26 * np.pi / np.sqrt(3)))
    num_cols = int(np.round(np.sqrt(26 * 3 / np.pi)))
    centres = []
    for r in range(num_rows):
        offset = (r % 2) * 0.5 * np.sqrt(3)
        for c in range(num_cols):
            x = 0.5 + offset + c * np.sqrt(3)
            y = 0.25 * (1 + np.sqrt(3)) + r
            if x < 1 and y < 1:
                centres.append([x, y])
    # Shuffle the points to randomize the initial centers
    random.shuffle(centres)
    return np.array(centres[:26])

def objective(vars, constraint_dict):
    n = len(vars)//3
    centres = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    return -np.sum(radii) + constraint_dict['penalty'] * constraint_dict['constraint_sum']

def boundary_constraints(vars, constraint_dict):
    n = len(vars)//3
    centres = vars[:2*n].reshape(n, 2)
    return np.hstack([centres[:, 0] - constraint_dict['radii'], 1 - constraint_dict['radii'] - centres[:, 0],
                       centres[:, 1] - constraint_dict['radii'], 1 - constraint_dict['radii'] - centres[:, 1]])

def non_overlap_constraints(vars, constraint_dict):
    n = len(vars)//3
    centres = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    return 1e-6 + cdist(centres, centres)**2 - (2*radii)[:,None] - (2*radii)[None,:]

def solve_packing():
    hex_centres = initialize_centres()
    hypothesis_radii, _ = solve_radius_lp(hex_centres)
    best_sol = (hex_centres, hypothesis_radii, 0)
    for _ in range(3):
        perturbed_centres = hex_centres + np.random.normal(scale=0.01, size=hex_centres.shape)
        perturbed_centres = np.clip(perturbed_centres, 0, 1)
        initial_vars = np.hstack([perturbed_centres.reshape(-1), hypothesis_radii])
        constraint_dict = {
            'constraint_sum': 0,
            'penalty': 1e-3,
            'radii': hypothesis_radii
        }
        boundary_nc = NonlinearConstraint(boundary_constraints, 0, 0, args=(constraint_dict,))
        non_overlap_nc = NonlinearConstraint(non_overlap_constraints, 0, np.inf, args=(constraint_dict,))
        result = minimize(objective, initial_vars, args=(constraint_dict,), method='trust-constr',
                         constraints=[boundary_nc, non_overlap_nc],
                         bounds=Bounds(lo=np.hstack([0]*len(hex_centres)+[0]*26), hi=np.hstack([1]*len(hex_centres)+[np.inf]*26)),
                         jac=lambda x, callback: None, options={'maxiter': 200})
        if result.success:
            centres, radii = result.x[:2*n].reshape(n, 2), result.x[2*n:]
            final_radii, _ = solve_radius_lp(centres)
            if np.sum(final_radii) > best_sol[2]:
                best_sol = (centres, final_radii, np.sum(final_radii))
            constraint_dict['constraint_sum'] = -np.sum(final_radii)  # update constraint sum for next iteration

    final_radii, _ = solve_radius_lp(best_sol[0])  # polish the final radii for the best centres
    return best_sol[0], final_radii, np.sum(best_sol[1])

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = cdist(centres, centres)
    np.fill_diagonal(dists, np.inf)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)
        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    result = minimize(lambda x: -np.sum(x), np.ones(n), method='SLSQP', bounds=bounds,
                     constraints={'type': 'ineq', 'fun': lambda x: b_ineq - np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2)) + 1e-12})
    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing():
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    return solve_packing()
