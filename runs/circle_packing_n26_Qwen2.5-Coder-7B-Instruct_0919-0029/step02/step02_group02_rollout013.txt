import numpy as np
import cvxpy as cp
from scipy.spatial import distance

def pair_distances(centers):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
            dists[j, i] = dists[i, j]
    return dists

def build_lp(centers):
    n = centers.shape[0]
    r = cp.Variable(n, nonneg=True)
    constraints = [r[i] + r[j] <= dist for i in range(n) for j in range(i + 1, n)]
    constraints.extend([r[i] <= centers[i, 0] for i in range(n)])
    constraints.extend([r[i] <= 1 - centers[i, 0] for i in range(n)])
    constraints.extend([r[i] <= centers[i, 1] for i in range(n)])
    constraints.extend([r[i] <= 1 - centers[i, 1] for i in range(n)])

    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    return prob

def solve_lp(prob):
    prob.solve(solver=cp.HS)
    radii = prob.variables('r').value
    dual_pair = {}
    dual_left = {}
    dual_right = {}
    dual_bottom = {}
    dual_top = {}
    
    for i in range(len(prob.constraints)):
        constraint = prob.constraints[i]
        if isinstance(constraint, cp.Le):
            try:
                dual_i, dual_j = constraint.variables()

                    j_idx = int(re.search(r'\[([0-9]+)\]', dual_j.name()).group(1))

                elif dual_j.name() == 'r':
                    j_idx = int(re.search(r'\[([0-9]+)\]', dual_j.name()).group(1))
                    r_idx = int(re.search(r'\[([0-9]+)\]', dual_i.name()).group(1))
                    dual_pair[(r_idx, j_idx)] = constraint.dual_value
            except AttributeError:
                continue
        
        if 'left' in constraint.name():
            r_idx = int(re.search(r'\[([0-9]+)\]', constraint.name()).group(1))
            dual_left[r_idx] = constraint.dual_value
        elif 'right' in constraint.name():
            r_idx = int(re.search(r'\[([0-9]+)\]', constraint.name()).group(1))
            dual_right[r_idx] = constraint.dual_value
        elif 'bottom' in constraint.name():
            r_idx = int(re.search(r'\[([0-9]+)\]', constraint.name()).group(1))
            dual_bottom[r_idx] = constraint.dual_value
        elif 'top' in constraint.name():
            r_idx = int(re.search(r'\[([0-9]+)\]', constraint.name()).group(1))
            dual_top[r_idx] = constraint.dual_value

    return radii, dual_pair, dual_left, dual_right, dual_bottom, dual_top

def compute_gradient(centres, duals):
    n = centres.shape[0]
    grad_x, grad_y = np.zeros(n), np.zeros(n)
    for i in range(n):
        dists = np.maximum(duals['dual_pair'][i] * (centres[i] - centres[:, :n]).T / np.minimum(distance.cdist(np.array([centres[i]]), centres, 'euclidean'), 1e-12).T, 0)
        grad_x[i] = np.sum(dists[:, i]) + duals['dual_left'][i] - duals['dual_right'][i] if 'dual_left' in duals and 'dual_right' in duals else np.sum(dists[:, i])
        grad_y[i] = np.sum(dists[:, n+i]) + duals['dual_bottom'][i] - duals['dual_top'][i] if 'dual_bottom' in duals and 'dual_top' in duals else np.sum(dists[:, n+i])
    
    return grad_x, grad_y

def initial_pattern():
    radius = 1/7
    centers = np.array([
        [0, 0], [0, 1], [0, 2], [0, 3], [1, 2],
        [1, 3], [1, 4], [1, 5], [1, 6], [2, 3],
        [2, 4], [2, 5], [2, 6], [2, 7], [3, 4],
        [3, 5], [3, 6], [3, 7], [3, 8], [3, 9],
        [4, 5], [4, 6], [4, 7], [4, 8], [4, 9],
        [5, 6], [5, 7], [5, 8], [5, 9], [6, 7],
        [6, 8], [6, 9], [7, 8], [7, 9]
    ])
    return centers * (1 + 2 * radius)

MAX_ITERS = 500
def run_packing():
    centers = initial_pattern()
    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(MAX_ITERS):
        radii, duals, dual_left, dual_right, dual_bottom, dual_top = solve_lp(build_lp(centers))
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centers.copy(), radii.copy()

        grad_x, grad_y = compute_gradient(centers, {'dual_pair': duals, 'dual_left': dual_left, 'dual_right': dual_right, 'dual_bottom': dual_bottom, 'dual_top': dual_top})
        norm = np.sqrt((grad_x**2+grad_y**2).sum())
        if norm < 1e-12: break
        grad = np.vstack((grad_x, grad_y)).T / norm

        new_centres = np.clip(centers + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))

        if new_obj > obj:                          # accepted
            centers = new_centres
            step *= 0.9
        else:                                      # rejected
            step *= 0.3

        if step < 1e-6: break

    return best_centres, best_radii, best_obj

centers, radii, best_obj = run_packing()
print("Best objective:", best_obj)
