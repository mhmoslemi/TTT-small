```python
import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def objective(x):
    n = int(len(x) // 3)
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    penal = 0
    for i in range(n):
        for j in range(i + 1, n):
            d2 = cdist([centers[i]], [centers[j]])[0][0]
            penal += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
        penal += max(radii[i] - centers[i, 0], 0) ** 2
        penal += max(radii[i] - centers[i, 1], 0) ** 2
        penal += max(radii[i] - (1 - centers[i, 0]), 0) ** 2
        penal += max(radii[i] - (1 - centers[i, 1]), 0) ** 2
    return -(radii.sum()) + 1e6 * penal

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    result = minimize(objective, np.random.rand(3 * n), bounds=bounds, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0})
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = radii.sum()
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
```
