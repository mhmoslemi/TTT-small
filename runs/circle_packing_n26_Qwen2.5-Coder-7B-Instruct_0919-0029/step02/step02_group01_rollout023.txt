import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        radii = np.maximum(result.x, 0.0)
        sum_radii = -result.fun
        return radii, sum_radii
    else:
        return np.zeros(n), 0.0

def create_initial_grid_and_add_center():
    center_grid = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    center_middle = np.array([0.5, 0.5])
    centers = np.vstack((center_grid, center_middle))
    return centers

def perturb_randomly(centers, epsilon0):
    eps = epsilon0
    n, _ = centers.shape
    for i in range(n):
        delta = np.random.uniform(-eps, eps, size=2)
        new_c = np.clip(centers[i] + delta, 0, 1)
        centers[i, :] = new_c
    return centers

def move_center(centers, k, step):
    u = np.random.rand(2)
    theta = step * np.random.rand()
    new_c = centers[k, :] + theta * u / np.linalg.norm(u)
    new_c = np.clip(new_c, 0, 1)
    centers[k, :] = new_c
    return centers

def random_accept(delta_sum, temp):
    if delta_sum > 0:
        return True
    else:
        return np.random.rand() < np.exp(delta_sum / temp)

def run_packing():
    n = 26
    NUM_RESTARTS = 30
    NUM_ITER = 4000
    TAU_STEP = 0.997
    TAU_TEMP = 0.997

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, 0.05)

        radii, sum_r = lp_optimum(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(n)
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = new_centers.copy(), new_radii.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

        final_radii, final_sum = lp_optimum(best_centers)
        if final_sum > best_sum:
            best_centers, best_radii, best_sum = best_centers.copy(), final_radii.copy(), final_sum

    return best_centers, best_radii, best_sum
