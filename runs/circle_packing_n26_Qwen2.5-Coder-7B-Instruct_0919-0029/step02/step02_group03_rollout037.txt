import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import matrix_similarity

def solve_radii(centres):
    r = np.array(matrix_similarity(centres.T, np.identity(centres.shape[1])) * 0.1, dtype=float)
    A = []
    b = []
    for i in range(centres.shape[0]):
        for j in range(centres.shape[0]):
            if i == j:
                A.append([-1, 0])
                b.append(-r[i])
                A.append([0, -1])
                b.append(-r[i])
                A.append([1, 0])
                b.append(r[i])
                A.append([0, 1])
                b.append(r[i])
            else:
                A.append([-1, 0, -1, 0])
                b.append(-r[i] - r[j])
                A.append([0, -1, 0, -1])
                b.append(-r[i] - r[j])
                A.append([1, 0, 1, 0])
                b.append(r[i] + r[j])
                A.append([0, 1, 0, 1])
                b.append(r[i] + r[j])
    A = np.array(A)
    b = np.array(b)
    res = minimize(lambda r: -np.sum(r), r, method='highs', constraints={'type': 'ineq', 'fun': lambda r: -A @ r + b})
    if not res.success:
        return np.zeros_like(r)
    return np.maximum(res.x, 0)

def sum_of_radii(centres):
    return np.sum(solve_radii(centres))

def objective(Z):
    centres = Z.reshape(26, 2)
    return -sum_of_radii(centres)

def gradient(Z):
    h = 1e-4
    grad = np.zeros_like(Z)
    for k in range(Z.size):
        perturbation = np.zeros_like(Z)
        perturbation[k] = h
        grad[k] = (objective(Z + perturbation) - objective(Z - perturbation)) / (2 * h)
    return grad

def initial_grid(N):
    grid_size = int(np.sqrt(N))
    centres = []
    for i in range(grid_size):
        for j in range(grid_size):
            if i + j < grid_size:
                centres.append([i + 0.5, j + 0.5])
            else:
                x_offset = (i - j + grid_size) * 0.5
                y_offset = (j - grid_size + i) * 0.5
                centres.append([0.5 + x_offset, 0.5 + y_offset])
        if len(centres) >= N:
            break
    random_offset = np.random.normal(size=(N, 2)) * 0.01
    return np.array(centres[:N]) + random_offset
