import numpy as np
import scipy.spatial
import scipy.optimize

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = scipy.spatial.distance_matrix(centres, centres)
    
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([-1 if j == i else 0 for j in range(n)])
        b_ineq.append(-(centres[i, 0] - 1e-12))
        A_ineq.append([1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([-1 if j == i else 0 for j in range(n)])
        b_ineq.append(-(centres[i, 1] - 1e-12))
        A_ineq.append([1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            dx, dy = centres[i] - centres[j]
            d = np.sqrt(dx**2 + dy**2)
            A_ineq.append([1 if j == i else 0 if j == k else 0 for k in range(n)])
            A_ineq[-1].append(-2*dx**2 / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(-2*dx * dy / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(-2*dx**2 / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(-2 * dx * dy / (d**2 + (d + 1e-12)**2))
            A_ineq.append([1 if j == i or j == k else 0 for k in range(n)])
            A_ineq[-1].append(2*dx)
            A_ineq[-1].append(2*dy)
            A_ineq.append(0 for _ in range(2*n))
            b_ineq.append(0)
            A_ineq.append([1 if j == i else 0 if j == k else 0 for k in range(n)])
            A_ineq[-1].append(-2 * dy**2 / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(2*dx * dy / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(-2 * dy**2 / (d**2 + (d + 1e-12)**2))
            A_ineq[-1].append(2*dx * dy / (d**2 + (d + 1e-12)**2))
            A_ineq.append([1 if j == i or j == k else 0 for k in range(n)])
            A_ineq[-1].append(2*dy)
            A_ineq[-1].append(-2*dx)
            A_ineq.append(0 for _ in range(2*n))
            b_ineq.append(0)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = scipy.optimize.linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def pso_objective(flat_centres, N=26):
    centres = flat_centres.reshape(N, 3)
    radii = centres[:, 2]
    constraints = []
    for i in range(N):
        constraints.append((radii[i], None))  # r_i >= 0
        constraints.append((centres[i, 0] - radii[i], None))  # x_i - r_i >= 0
        constraints.append((1 - centres[i, 0] - radii[i], None))  # 1 - x_i - r_i >= 0
        constraints.append((centres[i, 1] - radii[i], None))  # y_i - r_i >= 0
        constraints.append((1 - centres[i, 1] - radii[i], None))  # 1 - y_i - r_i >= 0
        for j in range(i + 1, N):
            dx, dy = centres[i, :2] - centres[j, :2]
            d = np.sqrt(dx**2 + dy**2)
            constraints.append((d - (radii[i] + radii[j]), None))  # d >= r_i + r_j
    cons = scipy.optimize.NonlinearConstraint(lambda z: np.array(constraints), -np.inf, np.inf)
    return -np.sum(radii)

def run_packing():
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    N = 26
    centres = np.zeros((N, 3))
    centres[N-12:N, :2] = np.meshgrid(np.linspace(0, 1, 4), np.linspace(0, 1, 4))[::-1].reshape(-1, 2)
    np.random.shuffle(centres[N-12:N, :2])
    centres[:, 2] = np.zeros(N)

    bounds = [(0, 1) for _ in range(3*N)]
    
    result = scipy.optimize.minimize(pso_objective, centres.flatten(), bounds=bounds, constraints=[])

    best_centres = result.x.reshape(N, 3)
    centre_positions = best_centres[:,:2]
    final_radii, final_sum = solve_radius_lp(centre_positions)

    return centre_positions, final_radii, final_sum

centres, radii, sum_radii = run_packing()
print((centres, radii, sum_radii))
