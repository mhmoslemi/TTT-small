import numpy as np
from scipy.optimize import particle_swarm, linprog, Bounds

def validate_packing(centers, radii):

def de_objective(flat_centres):
    centres = flat_centres.reshape(26, 2)
    _, s = solve_radius_lp(centres)
    return -s

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1])

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j])

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    bounds = [(0.0, 1.0)] * 52

    lambda_val = 1e7
    for _ in range(2):
        result = particle_swarm(
            de_objective,
            bounds=bounds,
            maxiter=1000,
            version="original",
            init="random",
            c1=0.729848, c2=0.149618, w=0.448808,
            omega_min=0.0, omega_max=0.9,
            bounds=bounds,
            mutation=(2, 2),
            repair="random",
            post_update="bounds",
            verbose=False
        )

        if result[1] < 0:
            break

        centres = result[0].reshape(26, 2)
        radii_0, _ = solve_radius_lp(centres)
        radiiỏ = lambda_val * radii_0

        ball = np.array([(0.5, 0.5), (0.5, 0.25), (0.5, 0.75), (0.75, 0.5), (0.25, 0.5)])
        for b in ball:
            for r_i in radii_ỏ:
                if any(np.sum((b - centres) ** 2, axis=1) < (r_i + radii) ** 2 for radii in radii_ỏ):
                    radii_ỏ[radii_ỏ.tolist().index(r_i)] *= 0.95

        centres_plus_radii = np.hstack([centres, radii_ỏ])
        new_centres_plus_radii = particle_swarm(
            lambda z: de_objective(z) + np.sum(np.maximum(radii_ỏ - 0.05, 0.05)),
            bounds=bounds,
            maxiter=1000,
            version="original",
            init="random",
            c1=0.729848, c2=0.149618, w=0.448808,
            omega_min=0.0, omega_max=0.9,
            bounds=bounds,
            mutation=(2, 2),
            repair="random",
            post_update="bounds",
            verbose=False
        )[0]
        centres_plus_radii += new_centres_plus_radii
        centres, radii_0 = centres_plus_radii[:, :2], centres_plus_radii[:, 2:]

        radii, _ = solve_radius_lp(centres)
        result = (np.vstack([centres, radii]), result[1])

    return result[0], result[1]
