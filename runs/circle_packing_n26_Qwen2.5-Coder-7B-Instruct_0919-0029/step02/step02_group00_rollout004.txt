```python
import numpy as np
from scipy.optimize import linprog, minimize
import cvxpy as cp

# Precompute indices for all pairs (i, j)
n = 26
pairs = np.array([[i, j] for i in range(n) for j in range(i + 1, n)])

def solve_radius_lp(centers):
    n = centers.shape[0]
    r = cp.Variable(n, nonneg=True)
    dists = cp.norm(centers[pairs[:, 0]] - centers[pairs[:, 1]], axis=1)
    constraints = [r <= cp.min(cp.vstack([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]), axis=0)]
    for d, r_i, r_j in zip(dists, r[pairs[:, 0]], r[pairs[:, 1]]):
        constraints.append(d >= r_i + r_j)
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return r.value, prob.value

def solve_position_soc(radii, centers):
    n = centers.shape[0]
    x, y = cp.Variable(n), cp.Variable(n)
    ref = centers.copy()
    constraints = [
        radii <= x,
        radii <= 1 - x,
        radii <= y,
        radii <= 1 - y,
        cp.norm(x - x[pairs[:, 0]], axis=1) + cp.norm(y - y[pairs[:, 1]], axis=1) >= radii + radii[pairs[:, 1]],
        cp.norm(x - ref[:, 0], axis=1)**2 + cp.norm(y - ref[:, 1], axis=1)**2
    ]
    objective = cp.Minimize(cp.sum((x - ref[:, 0])**2 + (y - ref[:, 1])**2))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return x.value, y.value, prob.value

def run_packing():
    hexagonal_center = np.array([(1 / 3, 1 / 2), (2 / 3, 1 / 2), (1 / 2, 0), (1 / 2, 1), (0, 1 / 3), (0, 2 / 3), (1, 1 / 3), (1, 2 / 3)])
    hexagonal_radius = np.array([1 / 3] * 8)
    candidate_centers = np.random.rand(26, 2)
    candidate_radius = np.zeros(26)
    best_centers = None
    best_radius = 0
    
    restarts = [hexagonal_center, 0.01 * np.random.randn(26, 2), np.full((26, 2), 0.001)]
    
    for restart in restarts:
        current_centers = restart
        current_radius = candidate_radius
        for _ in range(10):
            current_radius, _ = solve_radius_lp(current_centers)
            x, y, _ = solve_position_soc(current_radius, current_centers)
            current_centers = np.column_stack((x, y))
        
        radii, sum_radii = solve_radius_lp(current_centers)
        if sum_radii > best_radius:
            best_centers = current_centers
            best_radius = sum_radii
    
    final_radii, final_sum = solve_radius_lp(best_centers)
    
    return best_centers, final_radii, final_sum
```
