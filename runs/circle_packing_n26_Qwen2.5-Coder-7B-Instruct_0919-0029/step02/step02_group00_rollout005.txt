import numpy as np
from scipy.optimize import linprog, solve
from cvxpy import Variable, constraints, objective, Problem, minimize
import scipy.spatial.distance as dist
from scipy.spatial import Delaunay

def lp_solve(C):
    n = C.shape[0]
    r = Variable(n, nonneg=True)
    constraints = [
        r <= C[:, 0],
        r <= 1 - C[:, 0],
        r <= C[:, 1],
        r <= 1 - C[:, 1]
    ]
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = Variable()
            constraints += [d_ij >= r[i] + r[j], d_ij <= dist.pdist(C[[i, j]])[0], d_ij >= 0]
    objective = objective(sum(r))
    prob = Problem(objective, constraints)
    prob.solve(solver=linprog.HS)  # Use highs as the solver
    if prob.status == 'optimal':
        return r.value
    else:
        return np.zeros(n)

def socp_solve(C, new_r):
    n = C.shape[0]
    r = new_r  # Get radii from lp_solve
    x, y = Variable(n), Variable(n)
    constraints = [
        r <= x,
        r <= 1 - x,
        r <= y,
        r <= 1 - y
    ]
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraints += [d_ij >= r[i] + r[j], d_ij >= 0]
    objective = minimize(np.sum([(x[i] - y[i]) ** 2 for i in range(n)]))
    prob = Problem(objective, constraints)
    prob.solve(solver=solve.ECOS)  # ECOS can handle SOCP
    centres = np.vstack((x.value, y.value)).T
    if prob.status == 'optimal':
        return centres
    else:
        return C  # Use previous centers

def admm_packing():
    C = np.array([[i / 6, j / 6] for i in range(8) for j in range(8) if i == 0 or i == 7 or j == 0 or j == 7])[:26]
    r = np.zeros(26)
    rho = 10.0
    max_iter = 30
    obj_prev = -np.inf

    for k in range(max_iter):
        # Radius update
        r = lp_solve(C)
        # Centre update
        C = socp_solve(C, r)
        # Dual update
        obj = np.sum(r)
        if abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj
    return C, r, obj

def run_packing():
    return admm_packing()
