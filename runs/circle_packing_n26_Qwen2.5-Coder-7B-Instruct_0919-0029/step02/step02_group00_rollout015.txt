# Function to create a hexagonal lattice in the unit square
def hexagonal_lattice(n):
    theta = np.linspace(0, 2 * np.pi, n + 1)[:-1]
    r = np.sqrt(3) * np.linspace(0, 1 / (2 * np.tan(np.pi / (2 * n + 2))), n)
    x = np.sum(r[:, None] * np.cos(theta), axis=0)
    y = np.sum(r[:, None] * np.sin(theta), axis=0)
    x = x / np.max(x) * 1.0
    y = y / np.max(y) * 1.0
    return np.column_stack((x, y))

# Function to solve the LP for finding the radii with fixed centres
def solve_LP(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    radii_upper_bound = np.minimum(centers[:, 0], np.minimum(1 - centers[:, 0], np.minimum(centers[:, 1], 1 - centers[:, 1])))
    A_ineq = np.zeros((n*(n-1) + 4 * n, n))
    b_ineq = np.zeros(n*(n-1) + 4 * n)
    for i in range(n):
        A_ineq[i, i] = 1
        b_ineq[i] = radii_upper_bound[i]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ineq[n*(n-1) + 2*(i-n) + (j-i-1), i] = 1
            A_ineq[n*(n-1) + 2*(i-n) + (j-i-1), j] = 1
            b_ineq[n*(n-1) + 2*(i-n) + (j-i-1)] = dist - radii_upper_bound[i] - radii_upper_bound[j]
    A_ineq[n*(n-1) + 2*n: n*(n-1) + 4*n, 0:n] = -np.eye(n)
    b_ineq[n*(n-1) + 2*n: n*(n-1) + 4*n] = np.zeros(n)
    sol = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=(radii_upper_bound, np.inf), method='highs')
    return sol.x

# Function to solve the packing problem using ADMM
def admm_packing(n, max_iter=30, rho=10.0):
    centers = hexagonal_lattice(n)
    radii = np.zeros(n)
    lambda_ = np.zeros(n)
    obj_prev = -np.inf

    for k in range(max_iter):
        # Radius update
        r_next = solve_LP(centers)
        r = r_next + (lambda_ - rho * radii) / rho

        # Centre update
        centers = solve_SOCP(r, centers)

        # Dual update
        lambda_ = lambda_ + rho * (r - radii)

        obj = np.sum(r)
        if np.abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj

    radii_final = solve_LP(centers)
    return centers, radii_final, np.sum(radii_final)
