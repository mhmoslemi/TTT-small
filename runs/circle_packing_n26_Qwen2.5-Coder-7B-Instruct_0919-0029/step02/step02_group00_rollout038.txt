import numpy as np
import cvxpy as cp
from scipy.optimize import linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def solve_center_soc(radii, centers):
    n = centers.shape[0]
    
    x = cp.Variable(n)
    y = cp.Variable(n)
    
    obj = cp.Minimize(cp.sum_squares(x - centers[:, 0]) + cp.sum_squares(y - centers[:, 1]))
    
    constraints = [radii[i] <= x[i] for i in range(n)],
    constraints += [radii[i] <= 1 - x[i] for i in range(n)],
    constraints += [radii[i] <= y[i] for i in range(n)],
    constraints += [radii[i] <= 1 - y[i] for i in range(n)],
    constraints += [cp.norm(x[i] - x[j], 2) + cp.norm(y[i] - y[j], 2) >= radii[i] + radii[j] for i in range(n) for j in range(i + 1, n)]
    
    problem = cp.Problem(obj, constraints)

    problem.solve(solver=cp.ECOS)

    if problem.status == cp.OPTIMAL:
        return np.column_stack((x.value, y.value))
    else:
        raise Exception("SOCP optimization failed")

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    initial_centers = centers.copy()
    
    # Generate initial hexagonal lattice
    h = np.sqrt(3) * 0.05
    w = 0.1
    for i in range(n):
        centers[i, 0] = 0.5 * (i % 3) + (i // 3) * 0.5 * h
        centers[i, 1] = (i // 3) * h + (i % 3) // 3 * 0.5 * w

    centers = centers % 1

    best_centers = initial_centers.copy()
    best_sum_radii = 0

    for _ in range(3):  # Three restarts
        np.random.seed(42)
        initial_centers += 0.01 * np.random.randn(n, 2)
        initial_centers = np.clip(initial_centers, 0, 1)

        centers = initial_centers.copy()
        for _ in range(20):
            radii, sum_radii = solve_radius_lp(centers)
            centers = solve_center_soc(radii, centers)

            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = centers.copy()

    final_radii, _ = solve_radius_lp(best_centers)
    final_centers = solve_center_soc(final_radii, best_centers)
    sum_radii = sum(final_radii)

    return best_centers, final_radii, sum_radii
