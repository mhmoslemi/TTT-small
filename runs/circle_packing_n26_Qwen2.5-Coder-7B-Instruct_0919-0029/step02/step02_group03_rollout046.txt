```python
import numpy as np
from scipy.optimize import minimize

def f_local(X):
    N = int(len(X) / 3)
    radii = X[2::3]
    penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = (X[0 + 3 * i] - X[0 + 3 * j]) ** 2 + (X[1 + 3 * i] - X[1 + 3 * j]) ** 2
            penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
        penalty += max(radii[i] - X[0 + 3 * i], 0) ** 2
        penalty += max(radii[i] - X[1 + 3 * i], 0) ** 2
        penalty += max(radii[i] - (1 - X[0 + 3 * i]), 0) ** 2
        penalty += max(radii[i] - (1 - X[1 + 3 * i]), 0) ** 2
    return -(radii.sum()) + 1e6 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    initial_guess = np.random.rand(3 * N)
    initial_guess[:2*N] = np.random.rand(2*N)  # random centers in [0,1]^2
    initial_guess[2*N:] = np.random.rand(N) / 10  # small initial radii
    
    constraints = []
    for i in range(N):
        constraints.append({'type':'ineq', 'fun': lambda X, i=i: X[3*i] - X[3*i+2]})
        constraints.append({'type':'ineq', 'fun': lambda X, i=i: 1 - X[3*i] - X[3*i+2]})
        constraints.append({'type':'ineq', 'fun': lambda X, i=i: X[3*i+1] - X[3*i+2]})
        constraints.append({'type':'ineq', 'fun': lambda X, i=i: 1 - X[3*i+1] - X[3*i+2]})
    for i in range(N-1):
        for j in range(i+1, N):
            constraints.append({'type':'ineq', 'fun': lambda X, i=i, j=j: ((X[3*i] - X[3*j])**2 + (X[3*i+1] - X[3*j+1])**2) - (X[3*i+2] + X[3*j+2])**2})
    
    result = minimize(f_local, initial_guess, method='SLSQP', bounds=None, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0})
    X = result.x
    centers = X.reshape((N, 3))[:, :2]
    radii = X.reshape((N, 3))[:, 2]
    sum_radii = radii.sum()
    
    if sum_radii <= 2.637:
        for i in range(N):
            radii[i] *= 0.999999
        sum_radii = radii.sum()
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
```
