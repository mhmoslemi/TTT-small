import numpy as np
from scipy.optimize import linprog, minimize
from itertools import product

def solve_radii(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1, 0])
        A.append([-1, 0])
        A.append([0, 1])
        A.append([0, -1])
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            A.append([1, 0, 1, 0, -1, 0, 1, 0])
            A.append([0, 1, -1, 0, 1, 0, -1, 0])
            A.append([0, 0, 0, 0, 1, 0, 1, 0])
            A.append([0, 0, 0, 0, 0, 1, -1, 0])
            A.append([0, 0, 0, 0, 0, 0, 0, 0])
            A[-5].append(-d2)
            A[-4].append(d2)
            A[-3].append(d2)
            A[-2].append(d2)
            A[-1].append(-d2)
    A = np.array(A).reshape(len(A), 8 * n)
    c = np.array([-1] * n + [0] * 4 * n)
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    return res.x[:n]

def sum_of_radii(centers):
    return np.sum(solve_radii(centers))

def objective(Z):
    n = int(len(Z) / 4)
    centers = Z.reshape((n, 2))
    return -sum_of_radii(centers)

def gradient(Z):
    epsilon = 1e-4
    n = int(len(Z) / 4)
    grad = np.zeros(len(Z))
    for i in range(len(Z)):
        Z_p = Z.copy()
        Z_m = Z.copy()
        Z_p[i] += epsilon
        Z_m[i] -= epsilon
        grad[i] = (objective(Z_p) - objective(Z_m)) / (2 * epsilon)
    return grad

def run_packing():
    HexGrid = np.array([[0.5, 0.5],
                        [1/np.sqrt(3), 1.5/np.sqrt(3)],
                        [1.5/np.sqrt(3), 2/np.sqrt(3)],
                        [1/np.sqrt(3), 1.0/np.sqrt(3)],
                        [0.5,   0.5],
                        [1/np.sqrt(3), 0.5/np.sqrt(3)],
                        [1.5/np.sqrt(3), 1/np.sqrt(3)],
                        ])
    extra_cells = [[0, 0], [1, 0], [1, 1]]
    centers = np.array([coord for cell in extra_cells for coord in HexGrid])
    centers *= np.random.normal(1, 0.01, size=(26, 2))
    top_level_bounds = [(0.0, 1.0)] * 52
    for i in range(5):
        res = minimize(objective, centers.flatten(), method='L-BFGS-B', jac=gradient, bounds=top_level_bounds, options={'maxiter': 200})
        centers = res.x.reshape((26, 2))
    best_sum = np.inf
    best_centers = None
    for _ in range(10):
        offsets = np.random.normal(0, 0.01, size=(26, 2))
        final_centers = centers + offsets
        np.clip(final_centers, 0, 1, out=final_centers)
        final_sum = sum_of_radii(final_centers)
        if final_sum < best_sum:
            best_sum = final_sum
            best_centers = final_centers
    return best_centers, solve_radii(best_centers), best_sum
