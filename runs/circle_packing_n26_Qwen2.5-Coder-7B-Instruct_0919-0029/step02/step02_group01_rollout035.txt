import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    dist_matrix = squareform(pdist(centers))
    A_ub = np.hstack((dist_matrix[:, :, np.newaxis], 
                      np.eye(n)[:, :, np.newaxis] * centers[:, 0, np.newaxis],
                      np.eye(n)[:, :, np.newaxis] * centers[:, 1, np.newaxis],
                      np.eye(n)[:, :, np.newaxis] * (1 - centers[:, 0, np.newaxis]),
                      np.eye(n)[:, :, np.newaxis] * (1 - centers[:, 1, np.newaxis])))
    A_ub = A_ub.reshape(-1, 5)
    b_ub = dist_matrix.flatten() + (centers[:, 0, np.newaxis] - centers[:, 0, np.newaxis].T).flatten() + \
           (centers[:, 1, np.newaxis] - centers[:, 1, np.newaxis].T).flatten() - \
           (1 - centers[:, 0, np.newaxis] - centers[:, 0, np.newaxis].T).flatten() - \
           (1 - centers[:, 1, np.newaxis] - centers[:, 1, np.newaxis].T).flatten() + 1e-12
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_grid_plus_center():
    centers = np.array([(0.125 + 0.25 * i, 0.125 + 0.25 * j) for i in range(4) for j in range(4)])
    center = np.array([[0.5, 0.5]])
    centers = np.vstack((centers, center))
    return centers

def gaussian_perturb(centers, sigma=0.02):
    noise = np.random.normal(0, sigma, centers.shape)
    new_centers = centers + noise
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers

def run_packing():
    n = 26
    step, step_min = 0.15, 1e-3
    decay_factor = 0.995
    NUM_ITERS = 4000
    NUM_RESTARTS = 40

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.randn(2) * step
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, np.maximum(final_radii, 0), final_sum
