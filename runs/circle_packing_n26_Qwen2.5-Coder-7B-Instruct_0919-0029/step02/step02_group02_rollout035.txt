import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def de_optimization(centers):
    def objective(x):
        centers_new = x.reshape((26, 2))
        return -lp_optimum(centers_new)[1]

    bounds = [(0, 1)] * (26 * 2)
    result = differential_evolution(objective, bounds, popsize=30, maxiter=500)
    return result.x

def local_refinement(centers, n=26, max_iter=1000, decay_factor=0.99, step_start=0.05, step_min=1e-5):
    step = step_start
    while step > step_min and max_iter > 0:
        idx = np.random.randint(n)
        delta = np.random.uniform(-step, step, size=2)
        new_center = np.clip(centers[idx] + delta, 0, 1)
        new_centers = np.copy(centers)
        new_centers[idx] = new_center

        new_radii, new_sum_radii = lp_optimum(new_centers)
        if new_sum_radii > lp_optimum(centers)[1]:
            centers = new_centers
        step *= decay_factor
        max_iter -= 1

    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    result = de_optimization(np.random.rand(n, 2))
    centers = result.reshape((n, 2))
    centers = local_refinement(centers)
    radii, sum_radii = lp_optimum(centers)
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print("Centers:\n", centers)
    print("Radii:\n", radii)
    print("Sum of Radii:", sum_radii)
