def objective(centers_vect):

def gradient(centers_vect):
    h = 1e-4
    n = 26
    grad = np.zeros(len(centers_vect))
    controls = centers_vect.reshape((n, 2))
    for k in range(2 * n):
        d = np.zeros_like(centers_vect)

        grad[k] = (f_plus - f_minus) / (2 * h)
    return grad

def generate_initial_grid(n):
    controls = np.zeros((n, 2))
    radius_per_cell = 1.0 / np.sqrt(7)
    row_size = round(np.sqrt(n))
    col_size = n // row_size

            x = (j + 0.5) * (3 * radius_per_cell + 1e-4)
            y = (i + 0.5) * (3 * radius_per_cell + 1e-4)
            controls[i * col_size + j] = [x, y]
    if controls.shape[0] < n:
