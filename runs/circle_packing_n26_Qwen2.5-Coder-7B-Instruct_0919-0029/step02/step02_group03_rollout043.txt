import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    n = centers.shape[0]
    A_eq = np.zeros((n*(n-1)//2, 3*n))
    b_eq = np.zeros(n*(n-1)//2)
    for i in range(n):
        for j in range(i+1, n):
            A_eq[i*(n-1)//2 + j - i - 1, 3*i] = 1
            A_eq[i*(n-1)//2 + j - i - 1, 3*i+1] = 1
            A_eq[i*(n-1)//2 + j - i - 1, 3*i+2] = -1
            A_eq[i*(n-1)//2 + j - i - 1, 3*j] = 1
            A_eq[i*(n-1)//2 + j - i - 1, 3*j+1] = 1
            A_eq[i*(n-1)//2 + j - i - 1, 3*j+2] = -1
            b_eq[i*(n-1)//2 + j - i - 1] = np.linalg.norm(centers[i] - centers[j])
    c = np.ones(3*n)
    bounds = [(0, None) for _ in range(3*n)]
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return result.x[:n]

def sum_of_radii(centers):
    r = solve_radii(centers)
    r[r < 0] = 0
    return np.sum(r)

def objective(Z):
    n = int(len(Z) / 2)
    centers = Z.reshape((n, 2))
    return -sum_of_radii(centers)

def gradient(Z):
    n = int(len(Z) / 2)
    def gradient_i(i):
        eps = 1e-4
        Z_minus = Z.copy()
        Z_minus[2*i:2*i+2] -= eps
        Z_plus = Z.copy()
        Z_plus[2*i:2*i+2] += eps
        f_minus = objective(Z_minus)
        f_plus = objective(Z_plus)
        return (f_plus - f_minus) / (2 * eps)
    grad = np.hstack([gradient_i(i) for i in range(n)])
    return grad

def run_packing():
    n = 26
    initial_grid_side = 0.1
    centers = np.stack([
        np.outer(np.linspace(0.5, 1.5, n//2), np.linspace(0.5, 1.5, n//2))
        - (np.sqrt(n/2) * 2 * initial_grid_side / 2),
        2 * (np.arange(n//2) % n//2) * 2 * initial_grid_side / 2
    ], axis=2).reshape((n, 2))
    centers[0::2, 0] += initial_grid_side + 0.01
    centers[1::2, 1] += initial_grid_side + 0.01
    centers[1::2, 0] -= initial_grid_side + 0.01
    centers[0::2, 1] -= initial_grid_side + 0.01
    centers[25] = center(shape=(1, 2), ratios=[initial_grid_side] * 2, constraints=[(0, 1), (0, 1)], padding=0.01)
    centers = centers[:n]
    Z_initial = centers.flatten()
    best_result = None
    for _ in range(10):
        Z_initial += np.random.normal(0, 0.01, 26 * 2)
        result = minimize(objective, Z_initial, method='L-BFGS-B', jac=gradient, bounds=[(0.0, 1.0)] * 52, options={'ftol': 1e-9, 'maxiter': 200, 'disp': False})
        if best_result is None or -result.fun > -best_result.fun:
            best_result = result
    centers = best_result.x.reshape((n, 2))
    r = solve_radii(centers)
    r[r < 0] = 0
    return centers, r, np.sum(r)

def center(shape, ratios, constraints, padding=0):
    assert len(ratios) == len(constraints)
    M = len(ratios)
    n_vars = 2 * M
    A = []
    if padding:
        A.append([-1] * n_vars + [padding] * n_vars)
        A.append([1] * n_vars + [padding] * n_vars)
        A.append([-1] * n_vars + [-padding])
        A.append([1] * n_vars + [-padding])
    
    for i in range(M):
        A.append([-1, 0, 0, 1] + [0] * (2 * i))
        A.append([-1, 0, 1, 0] + [0] * (2 * i))
        A.append([1, 0, 0, 1] + [0] * (2 * i))
        A.append([1, 0, 1, 0] + [0] * (2 * i))
        A.append([0, -1, 0, 0] + [0] * (2 * i))
        A.append([0, -1, 1, 0] + [0] * (2 * i))
        A.append([0, 1, 0, 0] + [0] * (2 * i))
        A.append([0, 1, 1, 0] + [0] * (2 * i))
    
    b = [0] * (n_vars + 2)
    for ratio, constraint in zip(ratios, constraints):
        b.append(constraint)
        b.append(constraint)
    
    c = [-1] * n_vars
    bounds = [(0, None) for _ in range(n_vars)]
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    centers = result.x
    return centers[:2 * M].reshape((M, 2))

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
