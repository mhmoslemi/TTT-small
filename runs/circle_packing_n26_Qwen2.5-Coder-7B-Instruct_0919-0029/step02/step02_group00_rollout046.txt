import numpy as np
import cvxpy as cp
from scipy.optimize import linprog

def solve_LP(C):
    n = len(C)
    r = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(sum(r))
    con = [
        r <= C[:, 0],  # Left boundary
        r <= 1 - C[:, 0],  # Right boundary
        r <= C[:, 1],  # Bottom boundary
        r <= 1 - C[:, 1]  # Top boundary
    ]
    for i in range(n):
        for j in range(i + 1, n):
            con.append(r[i] + r[j] <= np.linalg.norm(C[i] - C[j]))
    prob = cp.Problem(obj, con)
    prob.solve(solver=cp.HS, verbose=False, max_iter=1000)
    return r.value

def solve_SOCP(r, C):
    n = len(r)
    C_new = C + 1e-6 * np.random.randn(n, 2)  # Add small noise
    r_new = cp.Variable(n, nonneg=True)
    obj = cp.Minimize(sum((C_new - C)**2))
    con = [
        r_new <= C_new[:, 0],  # Left boundary
        r_new <= 1 - C_new[:, 0],  # Right boundary
        r_new <= C_new[:, 1],  # Bottom boundary
        r_new <= 1 - C_new[:, 1],  # Top boundary
        cp.SOC(r_new + r), cp.SOC(C_new[:, 0] - 1 - r),
        cp.SOC(1 - C_new[:, 0] - r), cp.SOC(C_new[:, 1] - 1 - r)
    ]
    for i in range(n):
        for j in range(i + 1, n):
            con.append(cp.SOC(r_new[i] + r_new[j] + C_new[i] - C_new[j]))
    prob = cp.Problem(obj, con)
    prob.solve(solver=cp.SCS, verbose=False, max_iter=2000)
    return C_new, r_new.value
