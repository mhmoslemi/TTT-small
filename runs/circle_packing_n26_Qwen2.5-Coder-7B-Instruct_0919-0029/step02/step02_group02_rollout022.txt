import numpy as np
import scipy.optimize

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def differential_evolution_optimization(num_dimensions=52, num_positions=30, num_iterations=1000):
    bounds = [(0, 1)] * num_dimensions
    objective = lambda centers: -lp_optimum(centers.reshape((26, 2)))[1]
    result = scipy.optimize.differential_evolution(objective, bounds, population_size=num_positions, maxiter=num_iterations, strategy='best2bin')
    return result.x.reshape((26, 2)), -result.fun

def local_search(centers, tolerable_difference=1e-6):
    tolerance = 1000
    decrease_factor = 2000.0
    while tolerance > tolerable_difference:
        for _ in range(tolerance):
            i = np.random.randint(26)
            delta = np.random.uniform(-1, 1, 2)
            new_positions = centers.copy()
            new_positions[i] += delta
            new_positions = np.clip(new_positions, 0, 1)
            radii, sum_radii = lp_optimum(new_positions)
            if sum_radii > lp_optimum(centers)[1]:
                centers = new_positions
                break
        tolerance = max(1, tolerance / decrease_factor)
    return centers

def run_packing():
    centers, sum_radii = differential_evolution_optimization()
    centers = local_search(centers)
    final_radii, _ = lp_optimum(centers)
    return centers, final_radii, sum_radii
