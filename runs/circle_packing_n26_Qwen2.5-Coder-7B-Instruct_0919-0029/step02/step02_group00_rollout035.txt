import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.spatial.distance import pdist, squareform

def objective(x):
    n = 26
    centres = x[:52].reshape((n, 2))
    radii = x[52:]
    return -np.sum(radii)  # Negative because SLSQP minimizes

def constraint_non_overlap(x):
    n = 26
    centres = x[:52].reshape((n, 2))
    dists = squareform(pdist(centres))
    radii = x[52:]
    non_overlap = (dists - (radii[:, np.newaxis] + radii[np.newaxis, :])).flatten()
    non_overlap[range(dists.size), range(dists.size)] = np.inf  # Remove diagonal
    return non_overlap

def constraint_bounds(x):
    n = 26
    r_min = np.zeros(n)
    return np.concatenate((r_min, x[52:]))

bounds_radius = [(0, None)] * 26
bounds_centres = [(0, 1)] * 52
bounds = bounds_centres + bounds_radius

constraint1 = LinearConstraint(constraint_bounds, bounds[:, 0], bounds[:, 1])
constraint2 = LinearConstraint(constraint_non_overlap, np.full(325, 1e-12), np.inf)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0
    best_solution = (np.zeros(52), np.zeros(26))
    for _ in range(3):
        initial_guess = np.zeros(78)
        initial_guess[:52] = np.random.uniform(0, 1, 52)  # Hexagonal grid centres
        initial_guess[52:] = np.random.uniform(0.1, 0.3, 26)  # Initial radii guess
        result = minimize(objective, initial_guess, method='trust-constr', jac='2-point', constraints=[constraint1, constraint2], bounds=bounds, options={'maxiter': 200, 'gtol': 1e-6})
        radius_solution = -result.fun
        if radius_solution > best_sum:
            best_sum = radius_solution
            best_solution = (result.x[:52].reshape((26, 2)), result.x[52:])
    
    # Final polishing step with LP for radii
    final_centres = best_solution[0]
    lap_radii = solve_radius_lp(final_centres)
    final_radii = lap_radii[0]
    final_sum = np.sum(final_radii)
    return final_centres, final_radii, final_sum

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = squareform(pdist(centres))
    np.fill_diagonal(dists, np.inf)

    A_eq = np.ones((1, n))
    b_eq = np.sum(1e-6 * np.ones(n))  # Small positive value to ensure feasibility

    A_ineq = []
    b_ineq = []
    for i in range(n):
        b_ineq.append(centres[i, 0] - 1e-12)
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        b_ineq.append(centres[i, 1] - 1e-12)
        b_ineq.append(1 - centres[i, 1] - 1e-12)
        for j in range(i + 1, n):
            b_ineq.append(dists[i, j] - 1e-12)
    
    A_ineq = np.array([A_ineq]).T @ np.ones((1, len(b_ineq)))
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = minimize(c, np.ones(n), method='highs', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - b_eq}, {'type': 'ineq', 'fun': A_ineq.flatten @ x - b_ineq.flatten @ x}])

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")
