import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)
        A_ub[idx, i] = r
        b_ub[idx] = 0
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def proposal_move(centers):
    n = centers.shape[0]
    idx = np.random.randint(n)
    delta = np.random.uniform(-0.1, 0.1, size=2)
    new_c = np.clip(centers[idx] + delta, 0, 1)
    return idx, new_c

def hill_climbing(centers, steps):
    best_centers = centers.copy()
    best_sum_radii = lp_optimum(centers)[1]
    for step in range(steps):
        idx, new_c = proposal_move(centers)
        centers[idx] = new_c
        current_sum_radii = lp_optimum(centers)[1]
        if current_sum_radii > best_sum_radii:
            best_sum_radii = current_sum_radii
            best_centers = centers.copy()
        else:
            centers[idx] = best_centers[idx]
    return best_centers, best_sum_radii

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_centers, best_radii = None, None
    NUM_ITERS = 4000
    NUM_STARS = 20
    for _ in range(NUM_STARS):
        centers = np.random.rand(26, 2)
        final_centers, final_sum_radii = hill_climbing(centers, NUM_ITERS)
        if final_sum_radii > best_sum:
            best_sum = final_sum_radii
            best_centers = final_centers

    final_radii, final_sum_radii = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum_radii
