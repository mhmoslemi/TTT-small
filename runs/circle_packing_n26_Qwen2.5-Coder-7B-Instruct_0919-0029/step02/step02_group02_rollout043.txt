import numpy as np
import cvxpy as cp

def pair_distances(centres):
    n = centres.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.linalg.norm(centres[i] - centres[j])
    return dists

def build_lp(centres):
    n = centres.shape[0]
    r = cp.Variable(n, nonneg=True)
    
    obj = cp.Maximize(cp.sum(r))
    
    constraints = []
    dists = pair_distances(centres)
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= dists[i, j])
        constraints.append(r[i] <= centres[i, 0])
        constraints.append(r[i] <= 1 - centres[i, 0])
        constraints.append(r[i] <= centres[i, 1])
        constraints.append(r[i] <= 1 - centres[i, 1])
    
    prob = cp.Problem(obj, constraints)
    return prob

def solve_lp(prob):
    prob.solve(solver='highs', verbose=False)
    r_opt = np.array(prob.variables()[0].value).reshape(-1)
    dual_pair = {f"pair_{i}_{j}": 0 for i in range(len(r_opt)) for j in range(i + 1, len(r_opt))}
    dual_left = np.zeros(len(r_opt))
    dual_right = np.zeros(len(r_opt))
    dual_bottom = np.zeros(len(r_opt))
    dual_top = np.zeros(len(r_opt))
    for constr in prob.constraints[0:len(r_opt)*(len(r_opt)-1)//2]:
        i, j = map(int, constr.name().replace('pair_','').split('_'))
        dual_pair[(i, j)] = constr.dual_value if constr.dual_value >= 0 else 0
    for constr in prob.constraints[len(r_opt)*(len(r_opt)-1)//2:2*len(r_opt)*(len(r_opt)-1)//2]:
        i = int(constr.name().replace('left_',''))
        dual_left[i] = constr.dual_value if constr.dual_value >= 0 else 0
    for constr in prob.constraints[2*len(r_opt)*(len(r_opt)-1)//2:3*len(r_opt)*(len(r_opt)-1)//2]:
        i = int(constr.name().replace('right_',''))
        dual_right[i] = constr.dual_value if constr.dual_value >= 0 else 0
    for constr in prob.constraints[3*len(r_opt)*(len(r_opt)-1)//2:4*len(r_opt)*(len(r_opt)-1)//2]:
        i = int(constr.name().replace('bottom_',''))
        dual_bottom[i] = constr.dual_value if constr.dual_value >= 0 else 0
    for constr in prob.constraints[4*len(r_opt)*(len(r_opt)-1)//2:5*len(r_opt)*(len(r_opt)-1)//2]:
        i = int(constr.name().replace('top_',''))
        dual_top[i] = constr.dual_value if constr.dual_value >= 0 else 0
    dual_values = {"pair": dual_pair, "left": dual_left, "right": dual_right, "bottom": dual_bottom, "top": dual_top}
    return r_opt, dual_values

def compute_gradient(centres, dual_values):
    n = centres.shape[0]
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    dists = pair_distances(centres)
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] > 1e-12:
                grad_x[i] += dual_values["pair"][f"pair_{i}_{j}"] * (centres[i, 0] - centres[j, 0]) / dists[i, j]
                grad_x[j] += dual_values["pair"][f"pair_{i}_{j}"] * (centres[j, 0] - centres[i, 0]) / dists[i, j]
                grad_y[i] += dual_values["pair"][f"pair_{i}_{j}"] * (centres[i, 1] - centres[j, 1]) / dists[i, j]
                grad_y[j] += dual_values["pair"][f"pair_{i}_{j}"] * (centres[j, 1] - centres[i, 1]) / dists[i, j]
        if dual_values["left"][i] > 0:
            grad_x[i] += dual_values["left"][i]
        if dual_values["right"][i] > 0:
            grad_x[i] -= dual_values["right"][i]
        if dual_values["bottom"][i] > 0:
            grad_y[i] += dual_values["bottom"][i]
        if dual_values["top"][i] > 0:
            grad_y[i] -= dual_values["top"][i]
    return grad_x, grad_y

def run_packing():
    n = 26
    Iterations = 100
    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    centres = np.random.rand(n, 2) + 0.01
    for _ in range(Iterations):
        r_opt, dual_values = solve_lp(build_lp(centres))
        obj = r_opt.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centres.copy(), r_opt.copy()

        grad_x, grad_y = compute_gradient(centres, dual_values)
        norm = np.linalg.norm(np.vstack([grad_x, grad_y]))
        if norm < 1e-12: break
        grad = np.vstack([grad_x, grad_y]) / norm

        new_centres = np.clip(centres + step * grad, 0, 1)
        new_r_opt, _ = solve_lp(build_lp(new_centres))
        new_obj = new_r_opt.sum()

        if new_obj > obj:
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3

        if step < 1e-6: break

    return best_centres, best_radii, best_obj

centres, radii, obj = run_packing()
print(f"Centres: \n {centres}\nRadii: \n {radii}\nObjective: {obj}")
