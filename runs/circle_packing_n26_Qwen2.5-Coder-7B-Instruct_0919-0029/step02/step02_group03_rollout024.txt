import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import minimize

def f_global(X):
    N = int(len(X) / 2)
    radii = X[N:]
    centers = X[:N]
    penalty = 0
    for i in range(N):
        centers[i] = np.clip(centers[i], radii[i], 1 - radii[i])
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sum((centers[i] - centers[j]) ** 2)
            penalty += max((radii[i] + radii[j]) ** 2 - dist, 0) ** 2
    return -(radii.sum()) + 1e6 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    initial_centers = np.random.rand(N, 2)
    initial_radii = np.random.rand(N) * (1 / 2)
    initial_X = np.concatenate((initial_centers.flatten(), initial_radii.flatten()))
    result = minimize(f_global, initial_X, method='L-BFGS-B', bounds=[(0, 1)] * 2 * N + [(0, 0.5)] * N, options={'ftol': 1e-8, 'maxiter': 200})
    X = result.x
    centers = X[:N].reshape((N, 2))
    radii = X[N:].reshape((N,))
    sum_radii = radii.sum()
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
