import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def negate_sum_radii(centers):
    radii, sum_radii = lp_optimum(centers)
    return -sum_radii

def refine_solution(centers):
    n = centers.shape[0]
    for _ in range(100):
        idx = np.random.randint(n)
        delta = np.random.uniform(-0.01, 0.01, size=2)
        new_c = np.clip(centers[idx] + delta, 0, 1)
        new_centres = centers.copy()
        new_centres[idx] = new_c
        new_radii, new_sum = lp_optimum(new_centres)
        if new_sum > sum_radii:
            centers = new_centres
            sum_radii = new_sum
    return centers, sum_radii

def run_packing():
    n = 26
    bounds = [(0, 1), (0, 1)] * n
    result = differential_evolution(negate_sum_radii, bounds=bounds, population_size=50, maxiter=200)
    best_centers = result.x.reshape((n, 2))
    best_radii, best_sum = lp_optimum(best_centers)
    best_centers, best_sum = refine_solution(best_centers)
    return best_centers, best_radii, best_sum
