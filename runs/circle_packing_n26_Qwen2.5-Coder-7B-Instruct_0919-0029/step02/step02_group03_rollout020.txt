import numpy as np
from scipy.optimize import linprog

N = 26

# Compute radii for given centers (only radii)
def compute_sum_radii(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1 if k==i else 0 for k in range(n)])
        b.append(centers[i,0])
        A.append([1 if k==i else 0 for k in range(n)])
        b.append(centers[i,1])
        A.append([1 if k==i else 0 for k in range(n)])
        b.append(1 - centers[i,0])
        A.append([1 if k==i else 0 for k in range(n)])
        b.append(1 - centers[i,1])
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A.append([1 if k==i else 0 for k in range(n)] + [1 if k==j else 0 for k in range(n)])
            b.append(dist)
    A = np.vstack(A).astype(np.float64)
    b = np.array(b).astype(np.float64)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        radii = res.x
        sum_radii = np.sum(radii)
    else:
        return np.array([0]*n), 0.0
    return radii, sum_radii

# Main objective function used by DE
def objective(X):
    centers = X.reshape((N, 2))
    _, sum_radii = compute_sum_radii(centers)
    return -sum_radii

# Simulated annealing to refine SA
def simulated_annealing(centers, steps=10000):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    for step in range(steps):
        i = np.random.randint(N)
        offset = np.random.uniform(-0.02, 0.02, size=2)
        current_centers[i] = np.clip(current_centers[i] + offset, 0, 1)
        _, current_sum = compute_sum_radii(current_centers)
        _, best_sum = compute_sum_radii(best_centers)
        if current_sum > best_sum:
            best_centers = current_centers.copy()
    return best_centers

# Main packing function
def run_packing():
    # Initial hexagonal pattern centers
    hexagon_centers = np.array([
        [1/3, 1/6], [2/3, 1/6], [1/3, 1/2], [2/3, 1/2], [0, 1/3], [1, 1/3], [0, 2/3], [1, 2/3],
        [1/4, 1/4], [3/4, 1/4], [1/4, 3/4], [3/4, 3/4], [1/3, 0], [2/3, 0], [1/3, 1], [2/3, 1],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.1, 0.5], [0.9, 0.5]
    ])
    centers = np.random.rand(N, 2) 
    centers = hexagon_centers[:N] + (centers - hexagon_centers[:N].mean(axis=0)) * 0.8

    # Differential Evolution to find best centers
    bounds = [(0, 1)]*2*N
    result = minimize(objective, centers.flatten(), method='L-BFGS-B', bounds=bounds)
    final_centers = result.x.reshape((N, 2))

    # Simulated Annealing to fine-tune the centers
    best_centers = simulated_annealing(final_centers)

    # Compute final radii and sum for the best centers
    final_radii, sum_radii = compute_sum_radii(best_centers)

    return best_centers, final_radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
