import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26
bounds = [(0.0, 1.0)] * 52

def compute_sum_radii(centers):
    radii_sum = 0.0
    for i in range(N):
        x_i, y_i, r_i = centers[3 * i], centers[3 * i + 1], centers[3 * i + 2]
        if r_i <= 0:
            return -np.inf
        if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or 
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
            return -np.inf
        for j in range(i + 1, N):
            x_j, y_j, r_j = centers[3 * j], centers[3 * j + 1], centers[3 * j + 2]
            dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
            if dist < r_i + r_j - 1e-12:
                return -np.inf
        radii_sum += r_i
    return radii_sum

def objective(X):
    centers = X.reshape((N, 3))[:, :3]
    return compute_sum_radii(centers)

def run_packing():
    # DE global optimization to get initial centres and radii
    result = differential_evolution(objective, bounds, popsize=12, maxiter=200, strategy='best1bin', tol=1e-6, seed=42)
    centers = result.x.reshape((N, 3))[:, :2]
    radii = result.x.reshape((N, 3))[:, 2]

    result = minimize(sa_objective, centers.ravel(), method='L-BFGS-B', bounds=[(0.0, 1.0)] * 52, options={'ftol': 1e-8, 'maxiter': 4000})
    centers = result.x.reshape((N, 3))[:, :2]
    radii = result.x.reshape((N, 3))[:, 2]
    
    # Ensure no negative radii and correct sum
