import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import pdist, squareform

# Hexagonal grid initialization
def hex_grid():
    n = np.sqrt(26)
    r = np.sqrt(3) / 2 * 0.1
    grid = np.zeros((int(np.ceil(n)), int(np.ceil(n)), 2), dtype=np.float64)
    for i in range(grid.shape[0]):
        for j in range(grid.shape[1]):
            x = (i + (j % 2)) * 0.5 * np.sqrt(3) * r
            y = j * 0.75 * 0.2 * r
            grid[i, j] = np.array([x, y])
    grid = grid[:int(n), :int(n)]
    flat_grid = grid.flatten().reshape(-1, 2)
    flat_grid = flat_grid[:26]
    flat_grid += np.random.normal(0, 0.01, size=flat_grid.shape)
    flat_grid = np.clip(flat_grid, 0, 1)
    return flat_grid

# Solve for radii via LP
def solve_radii(centres):
    n = centres.shape[0]
    A = np.zeros((n * (n - 1) // 2 + 2 * n, n))
    b = np.zeros(n * (n - 1) // 2 + 2 * n)
    for i in range(n):
        A[i * (n - 1) // 2 + i, i] = -1
        A[i * (n - 1) // 2 + i, i] = -1
        A[i * (n - 1) // 2 + n + i, i] = 1
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            a = np.zeros(n)
            a[i], a[j] = 1, 1
            A[n * (n - 1) // 2 + i * (n - 1 + 1) // 2 + j - 1, :] = a
            b[n * (n - 1) // 2 + i * (n - 1 + 1) // 2 + j - 1] = dist
    res = optimize.linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None))
    return res.x

# Objective and gradient functions
def objective(Z):
    centres = Z.reshape(26, 2)
    radii = solve_radii(centres)
    return -radii.sum()

def gradient(Z):
    n = Z.shape[0] // 2
    centre_vec = Z.reshape(2, -1)
    grad = np.zeros(Z.shape)  # 2 x 52
    for k in range(n):
        e_k = np.zeros(n * 2)
        e_k[k] += 1
        e_k[n + k] += 1
        f_plus = objective(Z + 1e-4 * e_k)
        f_minus = objective(Z - 1e-4 * e_k)
        grad[:, k], grad[:, n + k] = (f_plus - f_minus) / (2 * 1e-4), (f_plus - f_minus) / (2 * 1e-4)
    return grad.flatten()

# Run main packing algorithm
def run_packing():
    best_centers = None
    best_radii = None
    best_sum_radii = float('-inf')

    for restarts in range(10):
        initial_grid = hex_grid()
        result = optimize.minimize(
            objective,
            initial_grid,
            method='L-BFGS-B',
            jac=gradient,
            bounds=[(0.0, 1.0)] * 52,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result.success:
            radii = solve_radii(result.x.reshape(26, 2))
            current_sum_radii = radii.sum()
            if current_sum_radii > best_sum_radii:
                best_sum_radii = current_sum_radii
                best_centers = result.x.reshape(26, 2)
                best_radii = radii
    
    return best_centers, best_radii, best_sum_radii

# Execute the algorithm and get results
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
