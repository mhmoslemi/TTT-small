import numpy as np
from scipy.optimize import differential_evolution, linprog

# Global constants
N = 26

def compute_sum_radii(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 0])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 1])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 0])
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 1])
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A.append([1 if k == i else 0 for k in range(n)] + [1 if k == j else 0 for k in range(n)])
            b.append(dist)
    A = np.vstack(A)
    b = np.array(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x.sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        step_size = 0.05
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum > best_sum:
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Generate initial centers using hex grid pattern
    def hex_grid(n_rows):
        centers = []
        offset = np.sqrt(3) / 2
        y_offset = 0
        step = 2 / (3 * n_rows)
        for row in range(n_rows):
            for col in range(row % 2, n_rows, 2):
                x = col * step + step / 2 * np.cos(row * np.pi / 3)
                y = y_offset + row * step * np.sin(np.pi / 3)
                centers.append([x, y])
            y_offset += step * np.sin(np.pi / 3)
        return np.array(centers) * (1 / (1 + offset))
    
    initial_centers = np.array(hex_grid(9))
    initial_centers = np.random.rand(26, 2)  # Randomly perturb
    
    # Differential Evolution to find good centers
    result = differential_evolution(objective, [(0, 1)] * (2 * N), popsize=12, maxiter=200, strategy='best1bin', tol=1e-6)
    best_centers = result.x.reshape((N, 2))
    
    # Simulated Annealing to fine-tune
    best_centers = simulated_annealing(best_centers)
    
    # Final LP to compute radii and sum
    final_sum_radii = compute_sum_radii(best_centers)
    final_radii = compute_sum_radii(best_centers, approximate=True)
    
    return best_centers, final_radii, final_sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
