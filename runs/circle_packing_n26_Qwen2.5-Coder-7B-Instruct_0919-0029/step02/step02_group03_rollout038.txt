import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    A_ub = []
    b_ub = []
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub.append([0] * N)
            A_ub[-1][i] = 1
            A_ub[-1][j] = 1
            b_ub.append(dist)
    for i in range(N):
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(centers[i,0])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(centers[i,1])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i,0])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(1 - centers[i,1])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(N)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res.success:
        return res.x.sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, max_iter=5000, initial_temperature=0.02, cooling_rate=0.995):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for i in range(max_iter):
        idx = np.random.randint(N)
        step = 0.02
        new_pos = current_centers[idx] + np.random.uniform(-step, step, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[idx] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0)] * N

    birdseed = 42  # pseudo-random seed to provide stable runs (change if desired)
    np.random.seed(birdseed)

    centers_start = 0.5 * np.random.rand(N, 2)

    res_DE = differential_evolution(objective, bounds, maxiter=200, popsize=12, seed=birdseed)
    initial_state = res_DE.x.reshape((N, 3))[:, :2]

    final_state, final_sum_radii = simulated_annealing(initial_state, max_iter=4000)

    c = -np.ones(N)
    A_ub = []
    b_ub = []
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.linalg.norm(final_state[i] - final_state[j])
            A_ub.append([0] * N)
            A_ub[-1][i] = 1
            A_ub[-1][j] = 1
            b_ub.append(dist)
    for i in range(N):
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(final_state[i,0])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(final_state[i,1])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(1 - final_state[i,0])
        A_ub.append([0] * N)
        A_ub[-1][i] = 1
        b_ub.append(1 - final_state[i,1])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(N)
    res_LP = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if res_LP.success:
        final_radii = res_LP.x
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0

    return final_state, final_radii, final_sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
