import numpy as np
from scipy.optimize import linprog, minimize, Bounds

def solve_radii(centers):
    n = centers.shape[0]
    c = np.ones(n, dtype=float)
    A = np.empty((n*(n-1)//2 + 4*n, 2*n))
    A[:n*(n-1)//2, :n] = centers[:-1, :]
    A[:n*(n-1)//2, n:] = centers[1:, :]
    A[:n*(n-1)//2, :n] -= centers[:-1, :]
    A[:n*(n-1)//2, n:] -= centers[1:, :]
    A[:n*(n-1)//2] *= -2
    A[n*(n-1)//2:n*(n-1)//2+4*n, :n] = np.eye(n)
    A[n*(n-1)//2:n*(n-1)//2+4*n, n:] = -np.eye(n)
    b = np.hstack((np.ones(n*(n-1)//2), np.full(4*n, 1.0)))
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, 1), method='highs')
    return -res.x if res.success else np.zeros(n)

def sum_of_radii(centers):
    return solve_radii(centers).sum()

def objective(Z):
    Z = Z.reshape(-1, 2)
    return -sum_of_radii(Z)

def gradient(Z):
    n = int(Z.size / 2)
    Z = Z.reshape(-1, 2)
    radii = solve_radii(Z)
    grad = np.zeros_like(Z)
    h = 1e-4
    for col in range(2):
        x_plus = Z.copy()
        x_minus = Z.copy()
        x_plus[:, col] += h
        x_minus[:, col] -= h
        grad[:, col] = 0.5 * (objective(x_plus) - objective(x_minus)) / h
    return grad.flatten()

def run_packing():
    initial_grid_spacing = 0.1
    n = 26
    hex_grid = np.array([[i/5.0, j/5.0 + i % 2 * 0.3] for i in range(5) for j in range(5)])[[3, 2, 1], [1, 0, -1], [3, 2, 1]]
    hex_grid = np.tile(hex_grid, (3, 1))
    hex_grid = hex_grid[:n, :2]
    hex_grid /= np.sqrt(3) * initial_grid_spacing
    hex_grid += 0.5
    hex_grid -= np.min(hex_grid)
    hex_grid /= np.max(hex_grid)
    hex_grid *= 0.95
    hex_grid += 0.01 * np.random.normal(size=hex_grid.shape)
    
    best_centers = np.zeros((n, 2))
    best_sum = 0
    for _ in range(10):
        centered = np.max(hex_grid, axis=0)
        centered = centered / np.max(centered) * 0.95
        centers = hex_grid - np.mean(hex_grid, axis=0) + centered
        bounds = Bounds(0, 1)
        result = minimize(
            objective,
            centers.flatten(),
            method='L-BFGS-B',
            jac=gradient,
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
        )
        centers = result.x.reshape(-1, 2)
        radii = solve_radii(centers)
        r_sum = radii.sum()
        if r_sum > best_sum:
            best_sum = r_sum
            best_centers = centers.copy()
    
    return best_centers, solve_radii(best_centers), best_sum
