import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j]) + 1e-12
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def project(centers):
    return np.clip(centers, 0, 1)

def run_packing():
    n = 26
    centers = np.random.rand(n, 2) + 0.005  # add small jitter
    step = 0.1
    step_min = 1e-6
    decay_factor = 0.995
    max_iter = 2000
    tol = 1e-9
    tol_stall = 1e-8
    noise_std = 1e-3

    for _ in range(max_iter):
        step *= decay_factor
        if step < step_min:
            break

        g = np.zeros(2 * n)
        radii, _ = lp_optimum(centers)
        duals = linprog(c=radii, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')()['dual']

        for i in range(n):
            idx = 2 * i
            for j in range(n):
                if j == i:
                    continue
                d_ij = np.linalg.norm(centers[i] - centers[j])
                if d_ij < 1e-12:
                    continue
                g[idx] += duals[i * n + j] * (centers[i, 0] - centers[j, 0]) / d_ij
                g[idx + 1] += duals[i * n + j] * (centers[i, 1] - centers[j, 1]) / d_ij

            mu_l = duals[i * n]
            mu_r = duals[i * n + 1]
            mu_b = duals[i * n + 2]
            mu_t = duals[i * n + 3]

            g[idx] += -mu_l + mu_r
            g[idx + 1] += -mu_b + mu_t

        projected = project(centers + step / 2 * g)
        projected_radii, new_obj = lp_optimum(projected)
        if new_obj > 0:
            centers = project(projected + step * g)
            if new_obj - np.sum(lp_optimum(centers)[1]) < tol:
                step *= 1.05
                if step > 0.2:
                    step = 0.2
        else:
            step *= 0.3

        if step < step_min:
            break

        if np.linalg.norm(g) < tol_stall:
            centers += noise_std * np.random.randn(n, 2)
            step *= 1.5
            if step > 0.1:
                step = 0.1

    final_radii, final_sum = lp_optimum(centers)
    return centers, final_radii, final_sum

# Call the function to run the packing
centers, radii, sum_radii = run_packing()
print(f"Final sum of radii: {sum_radii}")
