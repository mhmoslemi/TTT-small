import numpy as np
import cvxpy as cp

def hexagonal_lattice_in_unit_square(n=26):
    side_length = np.sqrt(3) / 3
    rows = np.ceil(2 / side_length).astype(int)
    cols = np.ceil(4 / side_length).astype(int)
    lattice = np.array([[(x + y * 0.5) * side_length, y * side_length] 
                       for x in np.arange(cols) 
                       for y in np.arange(rows) 
                       if x % 2 == y % 2])
    return lattice[:n, :]

def solve_LP(C):
    n = C.shape[0]
    r = cp.Variable(n, nonneg=True)
    constraints = [r <= C[:, 0], r <= 1 - C[:, 0], 
                   r <= C[:, 1], r <= 1 - C[:, 1]]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= cp.norm(C[i] - C[j], 2))
    problem = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    result = problem.solve(solver=cp.HS, verbose=False, warm_start=True)
    if problem.status != cp.OPTIMAL:
        raise Exception("LP returns invalid status: " + str(problem.status))
    return r.value

def solve_SOCP(r, C):
    n = r.shape[0]
    C_new = cp.Variable((n, 2))
    constraints = [C_new[:, 0] >= 0, C_new[:, 1] >= 0, C_new[:, 0] <= 1, C_new[:, 1] <= 1,
                   cp.norm(C_new[i] - C_new[j], 2) >= r[i] + r[j] for i in range(n) for j in range(i + 1, n)]
    problem = cp.Problem(cp.Minimize(cp.sum((C_new - C)**2, axis=1)), constraints)
    result = problem.solve(solver=cp.ECOS, verbose=False, warm_start=True)
    if problem.status != cp.OPTIMAL:
        raise Exception("SOCP returns invalid status: " + str(problem.status))
    return C_new.value, r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    C = hexagonal_lattice_in_unit_square()
    r = np.zeros(n)
    λ = np.zeros(n)
    ρ = 10.0
    obj_prev = -np.inf
    
    for k in range(30):
        # Radius update
        r_lp = solve_LP(C)
        r = r_lp + (λ - ρ * r) / ρ
        
        # Centre update
        C, r = solve_SOCP(r, C)
        
        # Dual update
        λ = λ + ρ * (r - r)
        
        obj = np.sum(r)
        if np.abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj
    
    r_final = solve_LP(C)
    return C, r_final, np.sum(r_final)
