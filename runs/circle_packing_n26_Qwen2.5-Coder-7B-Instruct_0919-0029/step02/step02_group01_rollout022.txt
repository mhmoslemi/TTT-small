import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def create_initial_grid_and_add_center():
    initial_grid = np.array([
        (0.1, 0.1), (0.3, 0.1), (0.5, 0.1), (0.7, 0.1), (0.9, 0.1),
        (0.1, 0.3), (0.3, 0.3), (0.5, 0.3), (0.7, 0.3), (0.9, 0.3),
        (0.1, 0.5), (0.3, 0.5),                   (0.7, 0.5), (0.9, 0.5),
        (0.1, 0.7), (0.3, 0.7), (0.5, 0.7), (0.7, 0.7), (0.9, 0.7),
        (0.1, 0.9), (0.3, 0.9), (0.5, 0.9), (0.7, 0.9), (0.9, 0.9),
    ])
    centers = 0.2 * initial_grid + 0.1
    center = np.array([0.5, 0.5])
    centers = np.vstack((centers, center))
    return centers

def perturb_randomly(centers, epsilon0):
    n = centers.shape[0]
    epsilon = epsilon0
    for i in range(n):
        delta = np.random.uniform(-epsilon, epsilon, size=2)
        new_c = np.clip(centers[i] + delta, 0, 1)
        centers[i] = new_c
    return centers

def move_center(centers, k, step):
    c = centers[k]
    u = 2 * np.random.rand(2) - 1
    u /= np.linalg.norm(u)
    theta = step
    new_c = c + theta * u
    new_c = np.clip(new_c, 0, 1)
    new_centers = centers.copy()
    new_centers[k] = new_c
    return new_centers

def random_accept(delta_E, T):
    if delta_E > 0:
        return True
    else:
        return np.random.rand() < np.exp(delta_E / T)

NUM_RESTARTS = 30
NUM_ITER = 4000
TAU_STEP = 0.997
TAU_TEMP = 0.997

def run_packing():
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, epsilon0=0.05)

        radii, sum_r = lp_optimum(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(centers.shape[0])
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = centers.copy(), new_radii.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

        if best_sum > 2.637:
            break

    # Fine-tuning
    while True:
        step = 1e-4
        for _ in range(1000):
            k = np.random.randint(centers.shape[0])
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum

        if abs(new_sum - sum_r) < 1e-10:
            break
        sum_r = new_sum

    return best_centers, best_radii, best_sum
