import numpy as np
from scipy.optimize import linprog
import math

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun


def get_gradient(centers, radii, d_ij):
    grad = np.zeros(52)
    for i in range(26):
        df = np.zeros(52)
        for j in range(26):
            if j != i and d_ij[i, j] > 1e-12:
                grad_i = (centers[i, 0] - centers[j, 0]) / d_ij[i, j]
                grad_j = (centers[i, 1] - centers[j, 1]) / d_ij[i, j]
                df[2 * i] += radii[j] * grad_i
                df[2 * i + 1] += radii[j] * grad_j
                df[2 * j] -= radii[j] * grad_i
                df[2 * j + 1] -= radii[j] * grad_j
        grad[2 * i] += radii[i] * (1 - centers[i, 0])
        grad[2 * i + 1] += radii[i] * (1 - centers[i, 1])
        grad[2 * i] -= radii[i] * centers[i, 0]
        grad[2 * i + 1] -= radii[i] * centers[i, 1]
        grad[2 * i] += np.clip(radii[i] + radii[i] * radii[i], 0, None)
        grad[2 * i + 1] += np.clip(radii[i] + radii[i] * radii[i], 0, None)
        for s in [0, 1]:
            grad[s * 26] += radii[i] * (0 if centers[i, s] < 0 else np.inf)
            grad[s * 26 + 1] += radii[i] * (np.inf if centers[i, s] > 1 else 0)
        grad[2 * i] -= np.clip(radii[i], 0, None)
        grad[2 * i + 1] -= np.clip(radii[i], 0, None)
        for s in [0, 1]:
            grad[s * 26] -= np.clip(radii[i], 0, None)
            grad[s * 26 + 1] -= np.clip(radii[i], 0, None)
        grad += df
    return grad


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.1
    step_min = 1e-6
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    centers = np.random.rand(n, 2)
    for s in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > np.sum(radii):
                centers = new_centers
                radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(centers)
    return centers, final_radii, final_sum
