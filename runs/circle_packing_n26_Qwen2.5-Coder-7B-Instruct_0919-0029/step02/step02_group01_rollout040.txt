import numpy as np
from scipy.optimize import linprog

def initial_grid_plus_center():
    base_grid = np.array([(0.125 + 0.25*i, 0.125 + 0.25*j) for i in range(4) for j in range(4)])
    extra_positions = np.array([(0.5, 0.5), (0.25, 0.25), (0.25, 0.75), (0.75, 0.25), (0.75, 0.75),
                                (0.125, 0.125), (0.125, 0.875), (0.875, 0.125), (0.875, 0.875),
                                (0.375, 0.375), (0.375, 0.625), (0.625, 0.375), (0.625, 0.625)])
    return np.vstack((base_grid, extra_positions))

def gaussian_perturb(centers, sigma=0.02):
    perturbation = np.random.normal(0, sigma, size=(centers.shape[0], 2))
    return np.clip(centers + perturbation, 0, 1)

def lp_solve(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j], axis=0) - 1e-10
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-10
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-10
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-10
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-10
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    N_restart = 40
    N_iter = 4000
    step_min, temp_min = 1e-4, 1e-6

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(N_restart):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        cur_centers, cur_sum, cur_radii = centers.copy(), np.inf, np.zeros(n)
        step, temp = 0.15, 0.2

        for it in range(N_iter):
            k = np.random.randint(n)
            new_centers = cur_centers.copy()
            new_centers[k] = np.clip(cur_centers[k] + np.random.normal(0, step), 0, 1)

            new_radii, new_sum = lp_solve(new_centers)
            new_radii = np.maximum(new_radii, 0)

            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / temp):
                cur_centers, cur_sum, cur_radii = new_centers, new_sum, new_radii
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers, best_radii = cur_centers, new_radii

            step = max(step_min, step * 0.995)
            temp = max(temp_min, temp * 0.995)

    final_radii, final_sum = lp_solve(best_centers)
    final_radii = np.maximum(final_radii, 0)
    return best_centers, final_radii, final_sum

best_centers, best_radii, final_sum = run_packing()
print("Best Sum:", final_sum)
print("centers:", best_centers)
print("radii:", best_radii)
