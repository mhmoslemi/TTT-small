import numpy as np
from scipy.optimize import minimize, trust constr, differential_evolution

def hexagonal_grid(n, side_length):
    t = np.sqrt(3) / 2 * side_length
    h = 2 * t
    w = 3 * side_length / 2
    rows = int(np.ceil(1.0 / h))
    cols = int(np.ceil(2.0 / w))
    centers = np.zeros((0, 2))
    for row in range(rows):
        offset = side_length / 2 if row % 2 == 1 else 0
        for col in range(cols):
            x = col * w + offset
            y = row * h
            if np.all([x > 0, x < 1, y > 0, y < 1]):
                centers = np.vstack([centers, [x, y]])
    centers = centers[:n]
    return centers

def objective(vars, n, pairs, target_area):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    sum_radii = np.sum(radii)
    return sum_radii - 1e-6 * np.sum((radii - target_area / n)**2)  # penalty for large radii

def constraint(vars, n, pairs):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    constraints = [np.add(radii, radii.T) - dists]
    constraints.append(radii)
    constraints += [np.clip(centers[:, i] + radii[i], 0, 1) - centers[:, i] for i in range(2)]
    constraints += [centers[:, i] - (centers[:, i] - radii[i]) for i in range(2)]
    return constraints

def run_packing():
    n = 26
    side_length = 1 / np.sqrt(n)
    pairs = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    target_area = np.pi * (1.0 / n)**2
    centers_hex = hexagonal_grid(n, side_length)
    best_centers = centers_hex
    best_radius_sum = 0
    for _ in range(3):
        centers_init = best_centers + np.random.normal(0, 0.01, size=best_centers.shape)
        centers_init = np.clip(centers_init, 0, 1)
        bounds_r = [(0, None) for _ in range(n)]
        bounds_c = [(0, 1) for _ in range(2 * n)]
        bounds = bounds_c + bounds_r
        constraints = {'type': 'ineq', 'fun': constraint, 'args': (n, pairs)}
        res = minimize(objective, np.concatenate([centers_init.ravel(), [1e-3] * n]), args=(n, pairs, target_area),
                       method='trust-constr', bounds=bounds, constraints=constraints, options={'maxiter': 200})
        centers, radii = res.x[:2*n].reshape(n, 2), res.x[2*n:]
        radius_sum = np.sum(radii)
        if radius_sum > best_radius_sum:
            best_centers, best_radius_sum = centers, radius_sum
    final_radii, _ = np.clip(radii, 0, 1), best_radius_sum
    return best_centers, final_radii, best_radius_sum
