import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    c = [-1] * 26
    A = np.zeros((4 * 26, 52))
    b = np.zeros(4 * 26)
    for i in range(26):
        A[4 * i + 0, 2 * i + 0] = 1
        A[4 * i + 0, 2 * i + 1] = 0
        A[4 * i + 1, 2 * i + 0] = 0
        A[4 * i + 1, 2 * i + 1] = 1
        A[4 * i + 2, 2 * i + 0] = -1
        A[4 * i + 2, 2 * i + 1] = 0
        A[4 * i + 3, 2 * i + 0] = 0
        A[4 * i + 3, 2 * i + 1] = -1
        for j in range(26):
            if i < j:
                d2 = np.sum((centers[i] - centers[j]) ** 2)
                A[4 * i + 4 + 4 * (j - i - 1), 2 * i + 0:2 * i + 2] = [1, 1]
                A[4 * i + 5 + 4 * (j - i - 1), 2 * j + 0:2 * j + 2] = [-1, -1]
                b[4 * i + 4 + 4 * (j - i - 1)] = np.sqrt(d2)
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    if not res.success:
        return np.zeros(26)
    return res.x

def sum_of_radii(centers):
    return solve_radii(centers).sum()

def objective(Z):
    centers = Z.reshape((26, 2))
    return -sum_of_radii(centers)

def gradient(Z):
    h = 1e-4
    n = Z.shape[0]
    grad = np.zeros(n)
    for i in range(n):
        Z_plus = Z.copy()
        Z_plus[i] += h
        grad[i] += (objective(Z_plus) - objective(Z)) / (2 * h)
        Z_minus = Z.copy()
        Z_minus[i] -= h
        grad[i] -= (objective(Z_minus) - objective(Z)) / (2 * h)
    return grad

def run_packing():
    hex_grid = np.array([
        [0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], # Core 4
        [0.5, 0.125], [0.5, 0.875], # Horizontals 1
        [0.125, 0.5], [0.875, 0.5], # Verticals 1
        [0.375, 0.5], [0.625, 0.5], # Horizontals 2
        [0.5, 0.375], [0.5, 0.625], # Verticals 2
        [0.375, 0.125], [0.625, 0.125], [0.375, 0.875], [0.625, 0.875], # Horizontals 3
        [0.125, 0.375], [0.125, 0.625], [0.875, 0.375], [0.875, 0.625]  # Verticals 3
    ])
    hex_grid = hex_grid + np.random.normal(0, 0.01, hex_grid.shape)
    Z_initial = hex_grid.ravel()
    best_centers = hex_grid
    best_sum_radii = sum_of_radii(hex_grid)
    for _ in range(10):
        Z = hex_grid.ravel() + np.random.normal(0, 0.01, hex_grid.shape).ravel()
        result = minimize(objective, Z, method='L-BFGS-B', jac=gradient, bounds=[(0.0, 1.0)] * 52)
        centers = result.x.reshape((26, 2))
        sum_radii = sum_of_radii(centers)
        if sum_radii > best_sum_radii:
            best_centers = centers
            best_sum_radii = sum_radii
    best_rad = solve_radii(best_centers)
    return best_centers, best_rad, best_sum_radii
