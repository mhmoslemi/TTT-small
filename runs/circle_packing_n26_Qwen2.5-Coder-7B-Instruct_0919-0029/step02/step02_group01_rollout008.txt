import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def create_initial_grid_and_add_center():
    grid = np.array([[0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
                     [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
                     [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
                     [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
                     [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]])
    centers = np.vstack((grid, np.array([0.5, 0.5])))
    return centers

def perturb_randomly(centers, epsilon0):
    n, _ = centers.shape
    step = epsilon0 * (2 ** np.random.randint(0, np.log2(1 / epsilon0) + 1))
    u = np.random.uniform(-1, 1, size=(n, 2))
    eta = step * (u / np.linalg.norm(u, axis=1, keepdims=True))
    new_centers = np.clip(centers + eta, 0, 1)
    return new_centers

def move_center(centers, k, step):
    c = centers[k]
    theta = step * np.random.uniform(0, 1)
    u = np.array([np.cos(theta), np.sin(theta)])
    new_c = np.clip(c + u, 0, 1)
    new_centers = centers.copy()
    new_centers[k] = new_c
    return new_centers

def random_accept(delta, temp):
    return np.random.rand() < np.exp(delta / temp)

def run_packing():
    NUM_RESTARTS = 30
    NUM_ITER = 4000
    TAU_STEP = 0.997
    TAU_TEMP = 0.997
    RND_SEED = 42

    np.random.seed(RND_SEED)
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, epsilon0=0.05)
        radii, sum_r = lp_optimum(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(0, centers.shape[0])
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = centers.copy(), new_radii.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, np.round(final_sum, 10)

centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}")
print(f"Radii:\n{radii}")
print(f"Sum of Radii: {sum_radii}")
