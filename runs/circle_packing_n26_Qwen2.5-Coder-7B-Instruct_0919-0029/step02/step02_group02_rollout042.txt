from cvxpy import discs, Variable, Problem, Minimize, Exp
from cvxpy import norm, solve
import numpy as np

MAX_ITERS = 500
INITIAL_step = 0.1
step = INITIAL_step
tolerance = 1e-9

def pair_distances(centres):
    n = centres.shape[0]
    distances = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            distances.append(dist)
    return np.array(distances)

def build_lp(centres):
    n = centres.shape[0]
    r = Variable(n, nonneg=True)
    dist = pair_distances(centres)
    d_pairs = dist.reshape(n, n // 2)[:, :-1].flatten()
    r_pairs = Variable(n, nonneg=True)
    p = centres[:, 0]
    q = centres[:, 1]
    
    # Constraints: r_i >= 0 and r_j >= 0
    constraints = [ r[i] >= 0 for i in range(n) ]
    constraints.extend( [r[j] >= 0 for j in range(n)] )
    
    # Constraints: r_i + r_j <= d_ij
    constraints.extend( [r[i] + r[j] <= d_pairs[j] for j in range(n//2)] )
    
    # Constraints: r_i <= p_i and r_i <= 1-p_i
    constraints.extend( [r[i] <= p[i] for i in range(n)] )
    constraints.extend( [r[i] <= 1-p[i] for i in range(n)] )
    
    # Constraints: r_i <= q_i and r_i <= 1-q_i
    constraints.extend( [r[i] <= q[i] for i in range(n)] )
    constraints.extend( [r[i] <= 1-q[i] for i in range(n)] )

    # Objective function: max sum r_i
    objective = Minimize( sum(r) )
    
    problem = Problem(objective, constraints)
    
    return problem, r

def solve_lp(problem):
    problem.solve(solver='MOSEK')
    r = problem.variables()
    objectives = problem.objective()
    return r, objectives

def compute_gradient(centres, duals):
    n = centres.shape[0]
    d_pairs = pair_distances(centres)
    p = centres[:, 0]
    q = centres[:, 1]
    grad_p = np.zeros(n)
    grad_q = np.zeros(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            if j < i and j < n//2:
                if d_pairs[j] >= 1e-12:
                    grad_p[i] += duals[j] * (centres[j][0] - centres[i][0]) / d_pairs[j]
                    grad_q[i] += duals[j] * (centres[j][1] - centres[i][1]) / d_pairs[j]
                    grad_p[j] -= duals[j] * (centres[j][0] - centres[i][0]) / d_pairs[j]
                    grad_q[j] -= duals[j] * (centres[j][1] - centres[i][1]) / d_pairs[j]
                else:
                    grad_p[i] -= duals[j]
                    grad_p[j] += duals[j]
            else:
                if j < n // 2:
                    if d_pairs[j] >= 1e-12:
                        grad_p[i] += duals[j] * (centres[j][0] - centres[i][0]) / d_pairs[j]
                        grad_q[i] += duals[j] * (centres[j][1] - centres[i][1]) / d_pairs[j]
                    else:
                        grad_p[i] -= duals[j]
                else:
                    if d_pairs[j] >= 1e-12:
                        grad_p[j] -= duals[j] * (centres[j][0] - centres[i][0]) / d_pairs[j]
                        grad_q[j] -= duals[j] * (centres[j][1] - centres[i][1]) / d_pairs[j]
                    else:
                        grad_q[j] += duals[j]
        
        if i < n // 2:
            if p[i] >= 1e-12:
                grad_p[i] -= duals[i+n//2]
            if 1-p[i] >= 1e-12:
                grad_p[i] -= duals[i+n]
        else:
            if q[i] >= 1e-12:
                grad_q[i] -= duals[i]
            if 1-q[i] >= 1e-12:
                grad_q[i] -= duals[i+n]
                
    return grad_p, grad_q

def initial_pattern():
    n = 26
    centres = np.array([
        [0.1, 0.1], [0.1, 0.4], [0.1, 0.7], [0.4, 0.1], [0.4, 0.7],
        [0.7, 0.1], [0.7, 0.4], [0.7, 0.7], [0.1, 0.2], [0.1, 0.3],
        [0.1, 0.5], [0.1, 0.9], [0.4, 0.2], [0.4, 0.3], [0.4, 0.5],
        [0.4, 0.9], [0.7, 0.2], [0.7, 0.3], [0.7, 0.5], [0.7, 0.9],
        [0.2, 0.1], [0.3, 0.1], [0.5, 0.1], [0.2, 0.9], [0.3, 0.9],
        [0.5, 0.9]
    ])

    return centres

def run_packing():
    centres = initial_pattern()
    step = INITIAL_step
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(MAX_ITERS):
        problem, r = build_lp(centres)
        duals, objectives = solve_lp(problem)
        r = np.array(r.value)
        obj = objectives.value
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centres.copy(), r.copy()

        grad_p, grad_q = compute_gradient(centres, duals)
        norm = np.sqrt((grad_p**2+grad_q**2).sum())
        if norm < tolerance: break
        grad = np.vstack((grad_p, grad_q)).T / norm

        new_centres = np.clip(centres + step * grad, 0, 1)
        problem, r = build_lp(new_centres)
        duals, objectives = solve_lp(problem)
        new_r = np.array(r.value)
        new_obj = objectives.value

        if new_obj > obj:                          # accepted
            centres = new_centres
            step *= 0.9
        else:                                      # rejected
            step *= 0.3

        if step < tolerance: break

    return best_centres, best_radii, best_obj

centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of Radii:", sum_radii)
