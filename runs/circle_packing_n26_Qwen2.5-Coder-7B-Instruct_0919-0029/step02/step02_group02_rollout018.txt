import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij - 1e-12
            idx += 1

    for i in range(n):
        d_li = centers[i, 0]
        A_ub[idx, i] = 1
        b_ub[idx] = d_li - 1e-12
        idx += 1
        d_ri = 1 - centers[i, 0]
        A_ub[idx, i] = 1
        b_ub[idx] = d_ri - 1e-12
        idx += 1
        d_bi = centers[i, 1]
        A_ub[idx, i] = 1
        b_ub[idx] = d_bi - 1e-12
        idx += 1
        d_ti = 1 - centers[i, 1]
        A_ub[idx, i] = 1
        b_ub[idx] = d_ti - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    step = 0.1
    step_min = 1e-6
    decay_factor = 0.995
    NUM_ITERS = 2000
    NUM_STARS = 30

    def nearest_rectangular_pattern():
        step = 0.2
        jitter = np.random.uniform(-0.005, 0.005, size=(n, 2))
        centers = np.empty((n, 2))
        for i, j in np.ndindex((5, 5)):
            if i == 2:
                centers[i * 5 + j] = np.array([0.5 * step + j * step + jitter[i * 5 + j, 0], 0.5 * step + i * step + jitter[i * 5 + j, 1]])
            elif i != 4:
                centers[i * 5 + j] = np.array([j * step + jitter[i * 5 + j, 0], i * step + jitter[i * 5 + j, 1]])
        return centers

    def nearest_hexagon_pattern():
        step = np.sqrt(3) * 0.1
        jitter = np.random.uniform(-0.005, 0.005, size=(n, 2))
        centers = np.empty((n, 2))
        counter = 0
        for i in range(3):
            y = i * step + 0.125
            for j in range(2 * i + 1):
                x = j * step * 0.5 + step / 2 * (i % 2)
                centers[counter] = np.array([x + jitter[counter, 0], y + jitter[counter, 1]])
                counter += 1
        return centers

    best_sum = -np.inf
    best_centers = nearest_hexagon_pattern() if nearest_hexagon_pattern().max() > nearest_rectangular_pattern().max() else nearest_rectangular_pattern()

    for _ in range(NUM_STARS):
        centers = best_centers.copy()
        for _ in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step *= decay_factor if step > step_min else 1

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
