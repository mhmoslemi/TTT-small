import numpy as np
from scipy.optimize import minimize, linear_sum_assignment

def distance_matrix(c1, c2):
    return np.sqrt(np.sum((c1[:, np.newaxis, :] - c2[np.newaxis, :, :]) ** 2, axis=2))

def penalty_function(z, lambda_):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    p_boundary = np.maximum(0, r - x) + np.maximum(0, x + r - 1) + np.maximum(0, r - y) + np.maximum(0, y + r - 1)
    p_overlap = np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - np.sqrt((x[:, np.newaxis] - x[np.newaxis, :]) ** 2 + (y[:, np.newaxis] - y[np.newaxis, :]) ** 2))
    return np.sum(p_boundary) + np.sum(p_overlap ** 2) * lambda_

def constraint_movement(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    return [x, y, r]

def run_packing():
    n = 26
    a = np.array([1, 0])
    b = np.array([0.5, np.sqrt(3)/2])
    hex_points = []
    for i in range(-1, n+3):
        for j in range(-1, n+3):
            p = i * a + j * b
            if 0 <= p[0] < 2 and 0 <= p[1] < np.sqrt(3):
                hex_points.append(p)
    hex_points = hex_points[:n]
    hex_points = np.array(hex_points)

    # Initial guess (hexagonal lattice with truncated points)
    centres = np.copy(hex_points)
    radii = np.zeros(n)
    initial_guess = np.concatenate((centres.flatten(), radii.flatten()))

    # PSO settings
    options = {'c1': 0.5, 'c2': 0.3, 'w': 0.9}
    bounds = np.array([
        (0, 1), (0, 1), (0, 1)] * n)
    penalty_lambda = 1e6
    
    # PSO optimization
    result = minimize(penalty_function, initial_guess, args=(penalty_lambda,),
                      constraints=[{'type': 'ineq', 'fun': constraint_movement}],
                      method='SLSQP', bounds=bounds, options=options)
    
    centres = result.x[:n*2].reshape((n, 2))
    radii = result.x[n*2:]
    
    # Refinement with exact resolution
    dist_mtx = distance_matrix(centres, centres)
    max_len = dist_mtx.sum(axis=1).min(keepdims=True)
    lambda_ = 1e-6
    A_ineq = []
    b_ineq = []

    # Boundary constraints
    for i in range(n):
        A_ineq.append([-1, 0, 1] + [0] * (3*n-5))
        b_ineq.append(0)
        A_ineq.append([0, -1, 1] + [0] * (3*n-5))
        b_ineq.append(1)

    # Overlap constraints
    for i, j in linear_sum_assignment(dist_mtx - max_len):
        if i != j:
            A_ineq.append([0] * (3*i + [1]*(2*n) + [1]*(3*n-2*i-3*j))] + [-1] * (2*j) + [1] * (2*n-2*j) + [0] * n)
            b_ineq.append(-1.001 * (dist_mtx[i, j] + 1e-6))

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    # LP to solve for optimal radii
    c = np.sum((radii) ** 2) * -1
    bounds = [(0, None) for _ in range(n)]

    result = linear_sum_assignment(dist_mtx - max_len)
    solution = solve_radius_LP(result, centres, dist_mtx, lambda_)

    return solution

def solve_radius_LP(result, centres, dist_mtx, lambda_):
    n = 26
    A = np.array([np.concatenate((np.eye(n), -np.eye(n)))])
    b = np.array([np.zeros(n)])
    A_ineq = []
    b_ineq = []

    for i, j in result:
        A_ineq.append(np.concatenate((np.zeros(i), -1e-10, np.zeros(j), 1e-10, np.zeros(n-j-1))))
        b_ineq.append(-lambda_ * (dist_mtx[i, j] + 2e-12))

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = np.sum((np.random.rand(n, n) - 0.5) ** 2) * -1
    bounds = [(0, 1)] * n

    solution, sum_radii = minimize(c, np.zeros(n), bounds=bounds, constraints={'type': 'ineq', 'fun': lambda r, dist_matrix: dist_matrix[:, result] - np.append(r, r)[:, None]})
