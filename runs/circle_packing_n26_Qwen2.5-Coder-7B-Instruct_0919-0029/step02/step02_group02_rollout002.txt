import numpy as np
import cvxpy as cp
import scipy.optimize as opt

def pair_distances(centers):
    n = centers.shape[0]
    d = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d[i, j] = np.linalg.norm(centers[i] - centers[j])
            d[j, i] = d[i, j]
    return d

def build_lp(centers):
    n = centers.shape[0]
    d = pair_distances(centers)
    r = cp.Variable(n, nonneg=True)

    constraints = []
    for i in range(n):
        constraints.append(r[i] <= centers[i, 0])
        constraints.append(r[i] <= 1 - centers[i, 0])
        constraints.append(r[i] <= centers[i, 1])
        constraints.append(r[i] <= 1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= d[i, j])

    prob = cp.Problem(cp.Maximize(r.sum()), constraints)
    return prob

def solve_lp(prob):
    prob.solve(solver=cp.HQSOL)
    dual_pair = {}
    dual_left = np.zeros((26,))
    dual_right = np.zeros((26,))
    dual_bottom = np.zeros((26,))
    dual_top = np.zeros((26,))
    for i in range(26):
        for j in range(i + 1, 26):
            dual_pair[(i, j)] = prob.constraints[j - 1].dual_value
    for i in range(26):
        dual_left[i] = prob.constraints[52 + i].dual_value
        dual_right[i] = prob.constraints[52 + 26 + i].dual_value
        dual_bottom[i] = prob.constraints[52 + 52 + i].dual_value
        dual_top[i] = prob.constraints[52 + 52 + 26 + i].dual_value
    return prob.value, dual_pair, dual_left, dual_right, dual_bottom, dual_top

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    for i in range(5):
        for j in range(5):
            if i == 0 and j == 0:
                centers[i * 5 + j] = np.array([0.1, 0.1])
            elif i == 0 and j == 4:
                centers[i * 5 + j] = np.array([0.9, 0.1])
            elif i == 4 and j == 0:
                centers[i * 5 + j] = np.array([0.1, 0.9])
            elif i == 4 and j == 4:
                centers[i * 5 + j] = np.array([0.9, 0.9])
            elif i == 2 and j == 2:
                centers[i * 5 + j] = np.array([0.5, 0.5])
            else:
                centers[i * 5 + j] = np.array([0.2 + (i - 2) * 0.25, 0.2 + (j - 2) * 0.25])

    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(500):
        obj, dual_pair, dual_left, dual_right, dual_bottom, dual_top = solve_lp(build_lp(centers))
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centers.copy(), None

        grad_x = np.zeros(n)
        grad_y = np.zeros(n)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                norm = np.linalg.norm(centers[i] - centers[j])
                if norm > 1e-12:
                    grad_x[i] += dual_pair[(i, j)] * (centers[i, 0] - centers[j, 0]) / norm
                    grad_y[i] += dual_pair[(i, j)] * (centers[i, 1] - centers[j, 1]) / norm
            grad_x[i] += dual_left[i] - dual_right[i]
            grad_y[i] += dual_bottom[i] - dual_top[i]

        norm = np.linalg.norm(np.vstack([grad_x, grad_y]))
        if norm < 1e-12:
            break
        grad = np.vstack([grad_x, grad_y]).T / norm

        new_centers = np.clip(centers + step * grad, 0, 1)
        new_obj, _, _, _, _, _ = solve_lp(build_lp(new_centers))

        if new_obj > obj:
            centers = new_centers
            step *= 0.9
        else:
            step *= 0.3

        if step < 1e-6:
            break

    prob = cp.Problem(cp.Maximize(r.sum()), constraints)
    prob.solve(solver=cp.HQSOL)
    return best_centres, best_radii, best_obj
