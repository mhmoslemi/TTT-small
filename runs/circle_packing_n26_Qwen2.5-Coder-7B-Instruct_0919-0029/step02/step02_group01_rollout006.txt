import numpy as np

def lp_optimum(centers):
    n = centers.shape[0]
    m = n * (n - 1) // 2 + 4 * n
    A = np.empty((m, n))
    b = np.empty(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1; b[idx] = x - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = 1 - x - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = y - 1e-12; idx += 1
        A[idx, i] = 1; b[idx] = 1 - y - 1e-12; idx += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    np.random.seed(42)
    n = 26
    step = 0.1
    step_min = 1e-4
    decay = 0.995
    NUM_ITER_PER_RESTART = 4000
    NUM_STARS = 20
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        radii, new_sum = lp_optimum(centers)
        cur_centers = centers.copy()
        cur_sum = new_sum
        cur_radii = radii
        for iter in range(NUM_ITER_PER_RESTART):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / (iter + 1)):
                cur_centers, cur_sum = new_c, new_sum
                cur_radii = new_radii

            step *= decay
            step = max(step_min, step)

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers
            best_radii = cur_radii

    return best_centers, best_radii, best_sum
