import numpy as np
from scipy.optimize import particle_swarm, linprog

def distance_matrix(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    return dists

def penalty(centers, radii):
    n = centers.shape[0]
    dists = distance_matrix(centers)
    boundary_penalty = np.sum(np.maximum(0, radii - centers[:, 0] + 1e-12) +
                              np.maximum(0, centers[:, 0] + radii - 1 - 1e-12) +
                              np.maximum(0, radii - centers[:, 1] + 1e-12) +
                              np.maximum(0, centers[:, 1] + radii - 1 - 1e-12))
    overlap_penalty = np.sum(np.maximum(0, r_i + r_j - d_ij) ** 2
                             for r_i, r_j, d_ij in zip(radii, radii[:, None], dists))
    return boundary_penalty + overlap_penalty

def objective(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    return -(np.sum(radii) + 1e-6 * penalty(centers, radii))

def refine_radii(centers):
    n = centers.shape[0]
    dists = distance_matrix(centers)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    if not result.success:
        raise Exception("LP optimization failed")
    return result.x

def initial_positions():
    a = np.array([1, 0])
    b = np.array([0.5, np.sqrt(3)/2])
    points = set()
    for i in range(-5, 6):
        for j in range(-5, 6):
            point = i*a + j*b
            if 0 <= point[0] <= 1 and 0 <= point[1] <= 1:
                points.add(tuple(point))
    points -= {(1, 0), (1, 1)}
    return np.array(list(points))

def run_packing():
    centers = initial_positions()
    bounds = [(0, 1)] * 2 * len(centers) + [(0, None)] * len(centers)
    init_guess = np.concatenate((centers.flatten(), np.zeros(len(centers))))
    adapted_pso = lambda x: objective(x) + 1e-4 * np.sum((x - init_guess) ** 2)
    options = {'maxiter': 300, 'bounds': bounds}
    best, _ = particle_swarm(adapted_pso, bounds, options=options)
    centers = best[:2 * len(centers)].reshape(-1, 2)
    radii = refine_radii(centers)
    return centers, radii, np.sum(radii)
