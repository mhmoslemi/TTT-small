import numpy as np
from scipy.optimize import differential_evolution, linprog

N = 26

def compute_sum_radii(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 0])  # x <= ...
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(centers[i, 1])  # y <= ...
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 0])  # 1-x
        A.append([1 if k == i else 0 for k in range(n)])
        b.append(1 - centers[i, 1])  # 1-y
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0] * n
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dist)
    A = np.vstack(A)
    b = np.array(b)
    c = -np.ones(n)  # maximize: minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x.sum()
    else:
        return -np.inf

def objective(X):
    centers = X.reshape((N, 2))
    return -compute_sum_radii(centers)

def simulated_annealing(centers, initial_temperature=0.02, cooling_rate=0.995, max_iter=5000, step_size=0.05):
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    best_sum = compute_sum_radii(current_centers)
    T = initial_temperature
    for it in range(max_iter):
        i = np.random.randint(len(current_centers))
        new_pos = current_centers[i] + np.random.uniform(-step_size, step_size, 2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_sum = compute_sum_radii(new_centers)
        if new_sum == -np.inf:
            continue
        delta = new_sum - best_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            current_centers = new_centers
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
        T *= cooling_rate
        if T < 1e-5:
            break
    return best_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([(0.3 + 0.2 * i) * 0.8 for i in range(5)] + [(0.1 + 0.2 * i, 0.2732 + 0.1732 * j) for j in range(7) for i in range(4 if j % 2 == 0 else 5)] + [(0.5, 0.5), (0.5, 0.55), (0.55, 0.5)])
    bounds = [(0.0, 1.0)] * 52
    result = differential_evolution(objective, bounds, maxiter=200, popsize=12, strategy='best1bin')
    best_centers = result.x.reshape((N, 2))
    best_centers = simulated_annealing(best_centers)
    final_ris = linprog(-np.ones(N), A_ub=[*np.eye(N), *[centers[i] - centers[j] for i in range(N) for j in range(i + 1, N)], *[[1 - centers[i, 0], 1 - centers[i, 1], centers[i, 0], centers[i, 1]] for i in range(N)]], b_ub=[0] * N + [np.inf] * (N * (N - 1) + 4 * N), bounds=[(0, None)] * N).x
    sum_radii = final_ris.sum()
    return best_centers, final_ris, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
