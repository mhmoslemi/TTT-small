import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
from numpy.random import default_rng

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")


def solve_centre_soc(radii, centres):
    n = centres.shape[0]
    objective = cp.Minimize(cp.sum_squares(centres - centres))

    constraints = []
    for i in range(n):
        constraints.append(cp.Ge(centres[:, i], radii[i]))
        constraints.append(cp.Le(centres[:, i], 1 - radii[i]))

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(cp.Ge(cp.norm(centres[:, i] - centres[:, j]), radii[i] + radii[j]))

    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, eps_abs=1e-8, eps_rel=1e-8, max_iter=2000)

    return problem.value, centres.value


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    rng = default_rng(42)
    candidate_starts = [
        rng.random((n, 2)) * 0.95 + 0.025,
        rng.normal(0.5, 0.01, (n, 2)).clip(0.025, 0.975),
        rng.random((n, 2)) * 0.9 + 0.05
    ]

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for start in candidate_starts:
        centers = start.copy()
        for _ in range(10):
            radii, _ = solve_radius_lp(centers)
            _, updated_centers = solve_centre_soc(radii, centers)
            centers = updated_centers
        radii, sum_radii = solve_radius_lp(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers, best_radii = centers, radii

    return best_centers, best_radii, best_sum
