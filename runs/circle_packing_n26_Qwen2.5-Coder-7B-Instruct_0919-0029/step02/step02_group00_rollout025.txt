import numpy as np
from scipy.optimize import linprog, NonlinearConstraint, minimize
import cvxpy as cp

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def solve_center_socp(radii, centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    
    objective = cp.Minimize(cp.sum_squares(centers - cp.Parameter(n, n) @ cp.Parameter(n)))
    
    constraints = [
        radii[i] <= centers[i, 0],
        radii[i] <= 1 - centers[i, 0],
        radii[i] <= centers[i, 1],
        radii[i] <= 1 - centers[i, 1],
    ]
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(dists[i, j] >= radii[i] + radii[j])
    
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, verbose=False)
    
    return problem.value, centers.value

def admm_packing(max_iter=30, rho=10.0, tol=1e-7):
    n = 26
    centres = np.full((n, 2), 0.5)
    r = np.zeros(n)
    lambda_ = np.zeros(n)

    for k in range(max_iter):
        r_prev = r

        # Radius update (LP)
       objective = r - 0.5 * rho * np.sum((r - r_prev)**2)
        constraints = [
            r <= centres[:, 0],
            r <= 1 - centres[:, 0],
            r <= centres[:, 1],
            r <= 1 - centres[:, 1],
        ]
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append((centres[i] - centres[j])**2 >= (r[i] + r[j]) ** 2)

        result = linprog(objective, method='highs', Bounds=[(None, None)] * n, A_ub=np.row_stack(constraints), b_ub=[c[2] for c in constraints])
        
        r = result.x
        
        # Centre update (SOCP)
        centers = minimize(solve_center_socp, centres, args=(r,), method='SLSQP', options={'maxiter': 200})

        # Dual update
        lambda_ += rho * (r - r)

        if np.linalg.norm(r - r_prev) < tol:
            break

    r_final, _ = solve_radius_lp(centers)
    return centres, r_final, np.sum(r_final)

def run_packing():
    centres, radii, sum_radii = admm_packing()
    return centres, radii, sum_radii
