```python
import numpy as np
from scipy.optimize import minimize

def objective_function(X):
    N = int(len(X) // 3)
    radii = X[2::3]
    overlap_penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((X[0 + 3 * i] - X[0 + 3 * j]) ** 2 + (X[1 + 3 * i] - X[1 + 3 * j]) ** 2)
            overlap_penalty += max((radii[i] + radii[j]) ** 2 - dist ** 2, 0) ** 2
        overlap_penalty += max(radii[i] - X[0 + 3 * i], 0) ** 2
        overlap_penalty += max(radii[i] - X[1 + 3 * i], 0) ** 2
        overlap_penalty += max(radii[i] - (1 - X[0 + 3 * i]), 0) ** 2
        overlap_penalty += max(radii[i] - (1 - X[1 + 3 * i]), 0) ** 2
    return -(radii.sum()) + 1e6 * overlap_penalty

def constraint(X):
    N = int(len(X) // 3)
    return np.full(N, -np.inf)

bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26
initial_guess = np.random.rand(26 * 3) * 0.95 + 0.025

constraints = {'type': 'ineq', 'fun': constraint}
result = minimize(objective_function, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 200, 'iprint': 0})

if result.success:
    X = result.x
    centers = X.reshape((26, 3))[:, :2]
    radii = X.reshape((26, 3))[:, 2]
    sum_radii = radii.sum()
else:
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    sum_radii = 0

print(centers, radii, sum_radii)
```
