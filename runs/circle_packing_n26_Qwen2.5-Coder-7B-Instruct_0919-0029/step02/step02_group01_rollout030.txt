import numpy as np
from scipy.optimize import linprog

def lp_solve(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = dist - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26
    N_restart = 40
    step, step_min = 0.15, 1e-3
    NUM_ITERS = 4000

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(N_restart):
        if restart % 10 == 0:
            np.random.seed(restart)
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, sigma=0.02)

        cur_centers, cur_sum, cur_radii = centers.copy(), lp_solve(centers)
        step, temp = step, 0.2

        for it in range(NUM_ITERS):
            k = np.random.randint(n)
            delta = np.random.normal(0, step, size=2)
            new_centers = cur_centers.copy()
            new_centers[k] = np.clip(new_centers[k] + delta, 0, 1)
            new_radii, new_sum = lp_solve(new_centers)

            if new_sum >= cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / temp):
                cur_centers, cur_sum, cur_radii = new_centers, new_sum, new_radii
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers, best_radii = cur_centers.copy(), cur_radii.copy()

            step = max(step_min, step * 0.995)
            temp = temp * 0.995

    centers, best_radii, best_sum = cur_centers.copy(), cur_radii.copy(), cur_sum
    best_radii = np.clip(np.round(best_radii, 10), 0, np.inf)
    return best_centers, best_radii, best_sum

def initial_grid_plus_center():
    grid = np.array([
        (0.125 + i * 0.25, 0.125 + j * 0.25) for i in range(4) for j in range(4)
    ])
    center = np.array([0.5, 0.5])
    points = np.vstack((grid, center))
    return points

def gaussian_perturb(centers, sigma):
    return np.clip(centers + np.random.normal(0, sigma, size=centers.shape), 0, 1)
