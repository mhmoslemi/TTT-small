import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def annealing(centers):
    n = centers.shape[0]
    step, step_min = 0.05, 0.001
    decay_factor = 0.995
    NUM_ITERS = 1000

    for it in range(NUM_ITERS):
        idx = np.random.randint(n)
        delta = np.random.uniform(-step, step, size=2)
        new_c = np.clip(centers[idx] + delta, 0, 1)
        new_centres = centers.copy()
        new_centres[idx] = new_c

        new_radii, new_sum = lp_optimum(new_centres)
        if new_sum > lp_optimum(centers)[1]:
            centers = new_centres
            grid = new_radii

        step = max(step_min, step * decay_factor)

    return centers, grid, lp_optimum(centers)[1]

def run_packing():
    startx = np.random.rand(26, 52)
    bounds = [(0, 1)] * 52
    result = differential_evolution(lambda x: lp_optimum(x.reshape(26, 2))[1], bounds, maxiter=300)
    best_centers = result.x.reshape(26, 2)
    best_radii, best_sum = lp_optimum(best_centers)
    optimized_centers, optimized_radii, final_sum = annealing(best_centers)
    return optimized_centers, optimized_radii, final_sum

best_centers, best_radii, final_sum = run_packing()
print(f"Sum of radii: {final_sum}")
