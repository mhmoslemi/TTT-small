import numpy as np
from scipy.optimize import minimize

def penalty_term(x):
    n = int(np.sqrt(len(x) // 2))
    radii = x[:n]
    centers = x[n:].reshape(n, 2)
    pen = 0
    alpha = 1.0e4
    beta = 1.0e6
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            p_ij = np.maximum(0, radii[i] + radii[j] - d_ij)
            if d_ij < 1.0e-12:
                continue
            pen += 0.5 * alpha * p_ij ** 2
        p_L = np.maximum(0, radii[i] - centers[i, 0])
        p_R = np.maximum(0, radii[i] - (1 - centers[i, 0]))
        p_B = np.maximum(0, radii[i] - centers[i, 1])
        p_T = np.maximum(0, radii[i] - (1 - centers[i, 1]))
        pen += 0.5 * beta * (p_L ** 2 + p_R ** 2 + p_B ** 2 + p_T ** 2)
    return pen

def gradient_function(x):
    n = int(np.sqrt(len(x) // 2))
    radii = x[:n]
    centers = x[n:].reshape(n, 2)
    delta = 1.0e-6
    grad = np.zeros_like(x)
    h = np.inf
    for i in range(n):
        grad[i] = -1.0
        for j in range(n):
            if i == j:
                continue
            d_ij = np.linalg.norm(centers[i] - centers[j])
            p_ij = np.maximum(0, radii[i] + radii[j] - d_ij)
            if p_ij > 0 and d_ij >= 1.0e-12:
                grad[n + 2 * i] += 2.0 * (p_ij * (centers[i, 0] - centers[j, 0])) / d_ij
                grad[n + 2 * i + 1] += 2.0 * (p_ij * (centers[i, 1] - centers[j, 1])) / d_ij
                grad[n + 2 * j] += 2.0 * (p_ij * (centers[j, 0] - centers[i, 0])) / d_ij
                grad[n + 2 * j + 1] += 2.0 * (p_ij * (centers[j, 1] - centers[i, 1])) / d_ij
            else:
                grad[n + 2 * i] = 0.0
                grad[n + 2 * i + 1] = 0.0
                grad[n + 2 * j] = 0.0
                grad[n + 2 * j + 1] = 0.0
        for side in ['L', 'R', 'B', 'T']:
            p = 0.0
            if side == 'L':
                p = np.maximum(0, radii[i] - centers[i, 0])
            elif side == 'R':
                p = np.maximum(0, radii[i] - (1 - centers[i, 0]))
            elif side == 'B':
                p = np.maximum(0, radii[i] - centers[i, 1])
            elif side == 'T':
                p = np.maximum(0, radii[i] - (1 - centers[i, 1]))
            if p > 0:
                if side == 'L':
                    grad[n + 2 * i] += 2.0 * p
                elif side == 'R':
                    grad[n + 2 * i] += -2.0 * p
                elif side == 'B':
                    grad[n + 2 * i + 1] += 2.0 * p
                elif side == 'T':
                    grad[n + 2 * i + 1] += -2.0 * p
        grad[i] = 0.0
    return -grad

def run_packing():
    n = 26
    jitter_magnitude = 5.0e-3
    alpha = 1.0e4

    ftol = 1.0e-9
    gtol = 1.0e-7
    maxiter = 2000
    num_restarts = 3
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(num_restarts):
        centers = np.random.rand(n, 2)
        centers += jitter_magnitude * (np.random.rand(n, 2) - 0.5)
        radii = np.ones(n) * 1.0e-3
        x0 = np.concatenate((radii, centers.flatten()))
        bounds = [(0, None)] * n + [(0, 1)] * (2 * n)
        result = minimize(penalty_term, x0, method='L-BFGS-B', jac=gradient_function, bounds=bounds, options={'ftol': ftol, 'gtol': gtol, 'maxiter': maxiter})
        if result.fun > best_sum:
            best_sum = result.fun
            best_centers = result.x[n:].reshape(n, 2)

    true_radii = np.zeros_like(best_centers)
    for i in range(n):
        A = 0.0
        B = 0.0
        C = 0.0
        D = 0.0

        for side in ['L', 'R', 'B', 'T']:

            elif side == 'T':
                p = np.maximum(0, best_radii[i] - (1 - best_centers[i, 1]))
            if p > 0:
                if side == 'L':
                    A += 2.0 * p
                elif side == 'R':
                    A += -2.0 * p
                elif side == 'B':
                    B += 2.0 * p
                elif side == 'T':
                    B += -2.0 * p
        for side in ['L', 'R', 'B', 'T']:

                p = np.maximum(0, best_radii[i] - best_centers[i, 0])

                p = np.maximum(0, best_radii[i] - (1 - best_centers[i, 1]))
            if p > 0:
                if side == 'L':

                elif side == 'R':
                    C += -2.0 * p
                elif side == 'B':
                    D += 2.0 * p

        B, C = C, B
        determinant = A * D - B ** 2
        if determinant > 0:
