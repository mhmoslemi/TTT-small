import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centres):
    n = centres.shape[0]
    c = np.ones(n)
    A_eq = np.zeros((n, 2*n))
    for i in range(n):
        A_eq[i, 2*i] = 1
        A_eq[i, 2*i+1] = 1
    A_eq = np.vstack((A_eq, np.eye(2*n), -np.eye(2*n)))
    b_eq = np.hstack((np.ones(n), np.array([1]*n+[0]*n)))
    A_side = np.zeros((3*n, 2*n))
    for i in range(n):
        for j in range(i+1, n):
            A_side[i*(n+1)+j-1, 2*i] = 1
            A_side[i*(n+1)+j-1, 2*j] = 1
            A_side[i*(n+1)+j-1, 2*i+1] = -1
            A_side[i*(n+1)+j-1, 2*j+1] = -1
    A_side = np.hstack((A_side, np.eye(2*n), -np.eye(2*n)))
    b_side = np.hstack((np.ones(n-1)*2, np.array([2]*(n-1)), np.array([2]*n+[0]*n)))
    bounds = [(0, None) for _ in range(2*n)]
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_side, b_ub=b_side, bounds=bounds, method='highs')
    radii = result.x
    return radii

def sum_of_radii(centres):
    return solve_radii(centres).sum()

def objective(Z):
    n = Z.shape[0] // 2
    centres = Z.reshape((n, 2))
    return -sum_of_radii(centres)

def gradient(Z):
    n = Z.shape[0] // 2
    centres = Z.reshape((n, 2))
    h = 1e-4
    gr = np.zeros_like(Z)
    for k in range(Z.size):
        Z_pos = Z.copy()
        Z_neg = Z.copy()
        Z_pos[k] += h
        Z_neg[k] -= h
        gr[k] = (objective(Z_pos) - objective(Z_neg)) / (2*h)
    return gr

def run_packing():
    n = 26
    def initial_grid(scale=0.1):
        coords = np.array([
            (i, j) for i in np.linspace(0.5, 1 iteration=n-1, num=n) for j in np.linspace(0.5, 1 iteration=n, num=n)
        ])
        np.random.shuffle(coords)
        grid = np.zeros((43, 2))
        for i in range(n):
            grid[i] = coords[i]
            grid[i+n] = np.array([grid[i][0], 1-grid[i][1]])
            grid[i+2*n] = np.array([1-grid[i][0], grid[i][1]])
            grid[i+3*n] = np.array([1-grid[i][0], 1-grid[i][1]])
        grid = grid[:n]
        grid += np.random.normal(0, scale, (n, 2))
        grid[:, 0] = np.clip(grid[:, 0], 0, 1)
        grid[:, 1] = np.clip(grid[:, 1], 0, 1)
        return grid.flatten()
    
    best_result = None
    for _ in range(10):
        centres = initial_grid()
        result = minimize(objective, centres, method='L-BFGS-B', jac=gradient, bounds=[(0.0, 1.0)] * 52, options={'ftol': 1e-9, 'maxiter': 200})
        results = (result.x.reshape((n, 2)), solve_radii(result.x.reshape((n, 2))), result.fun * -1)
        if best_result is None or results[2] > best_result[2]:
            best_result = results
    
    centres, radii, sum_radii = best_result
    return centres, radii, sum_radii
