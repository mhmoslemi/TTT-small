import numpy as np

def save_centers_and_radii(centers, radii, filename):
    np.savez(filename, centers=centers, radii=radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_lattice_points():
    a = np.array([1.0, 0.0])
    b = np.array([0.5, 0.5 * np.sqrt(3)])
    points = []
    for i in range(-1, 2):
        for j in range(-1, 3):
            p = np.dot(a, i) + np.dot(b, j)
            if 0 <= p[0] <= 1 and 0 <= p[1] <= 1:
                points.append(p)
    return np.array(points)

def initial_roads(centers, radii):
    n = centers.shape[0]
    dist_matrix = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dist_matrix, np.inf)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0.0] * n)
        A_ineq[-1][i] = 1.0
        b_ineq.append(1.0)
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j]:
                A_ineq.append([0.0] * n)
                A_ineq[-1][i] = 1.0
                A_ineq[-1][j] = 1.0
                b_ineq.append(dist_matrix[i, j])
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    return A_ineq, b_ineq

def objective_function(centers, radii):
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        r = radii[i]
        penalty += max(0.0, r - centers[i, 0]) + max(0.0, centers[i, 0] + r - 1.0) + \
                   max(0.0, r - centers[i, 1]) + max(0.0, centers[i, 1] + r - 1.0)
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            penalty += max(0.0, r + radii[j] - dist)
    return -np.sum(radii) + penalty * 1e6

def optimize_centers_and_radii():
    points = compute_lattice_points()
    if points.shape[0] < 26:
        np.random.seed(42)
        points = np.random.rand(26, 2)
    initial_radii = np.zeros(26)
    initial_centers = points[:26]

    # Perform the first PSO run
    bounds = [(0.0, 1.0)] * 78
    from pyswarm import minimize
    result = minimize(objective_function, initial_centers.flatten(), args=(initial_radii,),
                      bounds=bounds, options={'maxiter': 300, 'ftol': 1e-5})
    # initial_centres, initial_radius = result.x[:26].reshape(-1, 2), result.x[26:]
    

    # Perform the second PSO run
    from scipy.optimize import shgo
   .bounds = [(0.0, 1.0)] * 78
    result = shgo(objective_function, bounds, args=(initial_radii,),
                  options={'ftol': 1e-5, 'maxiter': 300})
    centres, radii = result.x[:26].reshape(-1, 2), np.abs(result.x[26:])
    is_valid, message = validate_packing(centres, radii)
    if not is_valid:
        print(message)
    # Perform a local refinement with LP
    A_ineq, b_ineq = compute_lattice_points()
    from scipy.optimize import linprog
    c = -np.ones(26)
    bounds = [(0.0, None)] * 26
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    
    refined_radii = np.abs(result.x)
    
    return centres, refined_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centres, refined_radii = optimize_centers_and_radii()
    is_valid, message = validate_packing(centres, refined_radii)
    if not is_valid:
        print(message)
        return np.zeros((26, 2)), np.zeros(26), 0.0
    sum_radii = np.sum(refined_radii)
    return centres, refined_radii, sum_radii

if __name__ == '__main__':
    centres, refined_radii, sum_radii = run_packing()
    filename = './output_main.npz'
    save_centers_and_radii(centres, refined_radii, filename)
    print(filename)
    print(centres)
    print(refined_radii)
    print(sum_radii)
