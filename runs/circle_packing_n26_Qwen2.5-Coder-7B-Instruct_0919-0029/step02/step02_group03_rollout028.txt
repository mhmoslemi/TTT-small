import cvxpy as cp
import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N)
    
    objective = cp.Maximize(cp.sum(r))
    
    constraints = []
    for i in range(N):
        constraints.append(r[i] >= 0)
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])
    
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.norm([x[i] - x[j], y[i] - y[j]], 2) >= r[i] + r[j])
    
    prob = cp.Problem(objective, constraints)
    
    solvers = [cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, abstol=1e-7, reltol=1e-7)
            if prob.status == cp.OPTIMAL:
                break
        except Exception as e:
            pass
                
    if solvers[solvers.index(solver)].status != cp.OPTIMAL:
        # Fallback to differential evolution
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
        result = opt.differential_evolution(f_global, bounds, popsize=20, maxiter=200, seed=42)
        X = result.x
        centers = X.reshape((N, 3))[:, :2]
        radii = X.reshape((N, 3))[:, 2]
        sum_radii = radii.sum()
    else:
        centers = np.vstack((x.value, y.value)).T
        radii = r.value
        sum_radii = cp.sum(r).value
    
    # Clip values to ensure feasible solution
    centers = np.clip(centers, 0, 1)
    radii = np.clip(radii, 0, None)
    
    # Verify sum of radii
    redefine_radii = compute_sum_radii(centers, radii)
    if not np.isclose(redefine_radii, sum_radii, atol=1e-6):
        radii *= redefine_radii / sum_radii
        sum_radii = redefine_radii
    
    return centers, radii, sum_radii


def compute_sum_radii(centers, radii):
    n = centers.shape[0]
    sum_radii = np.sum(radii)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                sum_radii -= radii[i] + radii[j] - dist + 1e-6
    return sum_radii

def f_global(X):
    N = int(len(X) / 3)
    radii = X[2::3]
    penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = (X[0 + 3 * i] - X[0 + 3 * j]) ** 2 + (X[1 + 3 * i] - X[1 + 3 * j]) ** 2
            penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
        penalty += max(radii[i] - X[0 + 3 * i], 0) ** 2
        penalty += max(radii[i] - X[1 + 3 * i], 0) ** 2
        penalty += max(radii[i] - (1 - X[0 + 3 * i]), 0) ** 2
        penalty += max(radii[i] - (1 - X[1 + 3 * i]), 0) ** 2
    return -(radii.sum()) + 1e6 * penalty
