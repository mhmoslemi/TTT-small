import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint
from scipy.spatial import distance_matrix

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    dists = distance_matrix(centers, centers)
    np.fill_diagonal(dists, np.inf)

    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def placement_cost(z):
    n = 26
    x = z[:26]
    y = z[26:52]
    r = z[52:]

    terms = -np.sum(r)
    for i in range(n):
        terms += 1e6 * np.maximum(0, r[i] - x[i])  # Left boundary
        terms += 1e6 * np.maximum(0, r[i] - (1 - x[i]))  # Right boundary
        terms += 1e6 * np.maximum(0, r[i] - y[i])  # Bottom boundary
        terms += 1e6 * np.maximum(0, r[i] - (1 - y[i]))  # Top boundary

    dists = distance_matrix(np.vstack((x, y)).T, np.vstack((x, y)).T)
    np.fill_diagonal(dists, np.inf)
    for i in range(n):
        for j in range(i + 1, n):
            terms += 1e6 * np.maximum(0, r[i] + r[j] - dists[i, j])  # Overlap

    return terms

def placement_gradient(z):
    n = 26
    x = z[:26]
    y = z[26:52]
    r = z[52:]

    grad = np.zeros_like(z)
    for i in range(n):
        grad[i] -= 1  # Derivative of max(0, r[i] - x[i])
        grad[26 + i] -= 1  # Derivative of max(0, r[i] - (1 - x[i]))
        grad[i + 52] -= 1  # Derivative of max(0, r[i] - y[i])
        grad[26 + i + 52] -= 1  # Derivative of max(0, r[i] - (1 - y[i]))

    dists = distance_matrix(np.vstack((x, y)).T, np.vstack((x, y)).T)
    np.fill_diagonal(dists, np.inf)
    for i in range(n):
        for j in range(i + 1, n):
            grad[i] += 1e6 * np.minimum(0, dists[i, j] - r[i] - r[j])  # Overlap
            grad[26 + i] += 1e6 * np.minimum(0, dists[i, j] - r[i] - r[j])  # Overlap
            grad[i + 52] += 1e6 * np.minimum(0, dists[i, j] - r[i] - r[j])  # Overlap
            grad[26 + i + 52] += 1e6 * np.minimum(0, dists[i, j] - r[i] - r[j])  # Overlap

    return grad

def refine_packing(x, y):
    n = 26
    dists = distance_matrix(np.vstack((x, y)).T, np.vstack((x, y)).T)
    np.fill_diagonal(dists, np.inf)

    A_ineq = []

        A_ineq.append([-1] * n)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * n)
        A_ineq[-1][i] = -x[i]
        A_ineq.append([0] * n)
        A_ineq[-1][i] = -y[i]
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1 - x[i]
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1 - y[i]

        for j in range(i + 1, n):
            A_ineq.append([-1] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            A_ineq.append([0] * n)
            A_ineq[-1][i] = -1
            A_ineq[-1][j] = -1
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = -1
            A_ineq.append([0] * n)
            A_ineq[-1][i] = -1
            A_ineq[-1][j] = 1

    bounds = [
        (0, None) for _ in range(n)  # x coordinates
    ] + [(0, None) for _ in range(n)]  # y coordinates
    bounds += [(0, None) for _ in range(n)]  # radii

    lin_con = LinearConstraint(np.eye(n * 4), 0, None)
    nlin_con = NonlinearConstraint(
        lambda z: -z[n * 2:] + distance_matrix(
            np.vstack((z[:n], z[n:(n * 2)] + z[n * 2:])).T,
            np.vstack((z[:n], z[n:(n * 2)] + z[n * 2:])).T
        ).flatten()
        - np.concatenate([2 * z[n * 2:], np.ones(n * 2)]),
        0, None
    )

    res = minimize(
        placement_cost, np.concatenate((x, y, np.zeros(n))),
        method='SLSQP',
        jac=placement_gradient,
        bounds=bounds,
        constraints=[lin_con, nlin_con],
        tol=1e-6
    )

    if res.success:
        return res.x
    else:
        raise Exception("Refinement failed")
