import numpy as np
import cvxpy as cp
import scipy.spatial
from scipy.spatial.distance import cdist
import math

def pair_distances(centres):
    return cdist(centres, centres)

def build_lp(centres):
    n = centres.shape[0]
    r = cp.Variable(n, nonneg=True)
    dists = pair_distances(centres)
    constraints = [r[i] + r[j] <= dists[i, j] for i in range(n) for j in range(i+1, n)]
    constraints += [r[i] <= centres[i, 0] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 0] for i in range(n)]
    constraints += [r[i] <= centres[i, 1] for i in range(n)]
    constraints += [r[i] <= 1 - centres[i, 1] for i in range(n)]
    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)
    return problem, r, constraints

def solve_lp(problem):
    problem.solve(solver=cp.HS11)
    radii = problem.variables()[0].value
    duals = {constr: constr.dual_value for constr in problem.constraints}
    return radii, duals

def compute_gradient(centres, duals, dists):
    n = centres.shape[0]
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < 1e-12:
                continue
            grad_x[i] += duals[(i, j)] * (centres[i, 0] - centres[j, 0]) / dists[i, j]
            grad_y[i] += duals[(i, j)] * (centres[i, 1] - centres[j, 1]) / dists[i, j]
            grad_x[j] -= duals[(i, j)] * (centres[i, 0] - centres[j, 0]) / dists[i, j]
            grad_y[j] -= duals[(i, j)] * (centres[i, 1] - centres[j, 1]) / dists[i, j]
        
        grad_x[i] += duals[(i, i, 'left')] - duals[(i, i, 'right')]
        grad_y[i] += duals[(i, i, 'bottom')] - duals[(i, i, 'top')]
    
    return grad_x, grad_y

def initial_pattern():
    centres = np.array([
        [0.5, 0.5],
        [0.25, 0.25],
        [0.75, 0.25],
        [0.25, 0.75],
        [0.75, 0.75],
        [0.125, 0.125],
        [0.375, 0.125],
        [0.625, 0.125],
        [0.875, 0.125],
        [0.125, 0.375],
        [0.375, 0.375],
        [0.625, 0.375],
        [0.875, 0.375],
        [0.125, 0.625],
        [0.375, 0.625],
        [0.625, 0.625],
        [0.875, 0.625],
        [0.125, 0.875],
        [0.375, 0.875],
        [0.625, 0.875],
        [0.875, 0.875]
    ])
    return centres

def run_packing():
    n = 26
    centres = initial_pattern()
    step = 0.1
    max_obj = -np.inf
    best_centres = None
    best_radii = None

    for _ in range(2000):
        radii, duals = solve_lp(build_lp(centres))
        obj = np.sum(radii)
        if obj > max_obj:
            max_obj, best_centres, best_radii = obj, centres.copy(), radii.copy()
        
        dists = pair_distances(centres)
        grad_x, grad_y = compute_gradient(centres, duals, dists)
        norm = np.linalg.norm(np.vstack([grad_x, grad_y]))
        if norm < 1e-12: break
        grad = np.vstack([grad_x, grad_y]) / norm
        
        new_centres = np.clip(centres + step * grad, 0, 1)
        new_radii, _ = solve_lp(build_lp(new_centres))
        
        if np.sum(new_radii) > np.sum(radii):
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3
        
        if step < 1e-6: break

    return best_centres, best_radii, np.sum(best_radii)
