from scipy.optimize import differential_evolution, NonlinearConstraint, LinearConstraint, Bounds
import numpy as np

def objective(p):
    x, y, r = p[:26], p[26:52], p[52:]
    return -np.sum(r)

def constraint(p):
    x, y, r = p[:26], p[26:52], p[52:]
    constraint_set = []
    for i in range(len(x)):
        for j in range(i + 1, len(x)):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraint_set.append(dist - (r[i] + r[j]))
        constraint_set.append(np.minimum(x[i] - r[i], r[i], 1 - x[i] + r[i], r[i], 1 - y[i] + r[i], r[i]) - r[i])
    return np.array(constraint_set)

def penalty(p):
    x, y, r = p[:26], p[26:52], p[52:]
    pen = 0
    for i in range(len(x)):
        for j in range(i + 1, len(x)):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            pen += max(0, dist - (r[i] + r[j]))**2
        pen += max(0, np.minimum(x[i] - r[i], r[i], 1 - x[i] + r[i], r[i], 1 - y[i] + r[i], r[i]) - r[i])**2
    return 1e8 * pen

def jac(p):
    x, y, r = p[:26], p[26:52], p[52:]
    dx = np.zeros_like(p)
    dy = np.zeros_like(p)
    dr = np.zeros_like(p)

    for i in range(len(x)):
        for j in range(i + 1, len(x)):
            dx[i] += (x[i] - x[j]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            dy[i] += (y[i] - y[j]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            dr[i] -= 1
            dr[j] -= 1
            dx[j] -= (x[j] - x[i]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            dy[j] -= (y[j] - y[i]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            dr[i] -= 1
            dr[j] -= 1

        dx[i] += np.maximum(0, np.minimum(0, 1 - 2 * x[i], -x[i] + 2 * r[i], -x[i] + 2 * r[i], 0)) / r[i]
        dy[i] += np.maximum(0, np.minimum(0, 1 - 2 * y[i], -y[i] + 2 * r[i], -y[i] + 2 * r[i], 0)) / r[i]
        dr[i] -= np.minimum(x[i], y[i], 1 - x[i], 1 - y[i])
        
        dx[j] += np.maximum(0, np.minimum(0, 1 - 2 * x[j], -x[j] + 2 * r[j], -x[j] + 2 * r[j], 0)) / r[j]
        dy[j] += np.maximum(0, np.minimum(0, 1 - 2 * y[j], -y[j] + 2 * r[j], -y[j] + 2 * r[j], 0)) / r[j]
        dr[j] -= np.minimum(x[j], y[j], 1 - x[j], 1 - y[j])
        
    return np.hstack((dx, dy, dr))

def run_packing():
    np.random.seed(42)
    n = 26
    step = 0.2
    x_center_init = np.linspace(0.1, 0.9, 5)
    y_center_init = np.linspace(0.1, 0.9, 5)
    centers_initial = np.array([(x, y) for x in x_center_init for y in y_center_init])
    radii_initial = np.full(n, 0.0999)
    initial_guess = np.hstack((centers_initial.flatten(), radii_initial))
    
    DE_result = differential_evolution(objective + penalty, bounds=Bounds([0, 0, 0], [1, 1, 0.5]), args=(None,), jac=jac, tol=1e-3, popsize=15, maxiter=400)
    
    lin_constraints = LinearConstraint([[1 if i == 26 + j * 3 or i == 27 + j * 3 or i == 28 + j * 3 else 0 for j in range(n)] for i in range(3 * n)], np.zeros(n), np.ones(n))
    nl_constraints = NonlinearConstraint(constraint, 0, np.inf, jac=jac)
    
    final_result = differential_evolution(objective, bounds=Bounds([0, 0, 0], [1, 1, 0.5]), args=(None,), constraints=[lin_constraints, nl_constraints], method='trust-constr', options={'disp': False, 'maxiter': 200, 'hess': None, 'xtol': 1e-8})
    
    centers = final_result.x[:52].reshape((n, 3))[:, :2]
    radii = -final_result.fun + penalty(final_result.x)
    
    return centers, radii, np.sum(radii)

centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
