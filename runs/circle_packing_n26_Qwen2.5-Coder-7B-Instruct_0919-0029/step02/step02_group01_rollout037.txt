import numpy as np
import scipy.optimize as opt

def optimize_circle_packing():
    n = 26
    initial_guess = np.zeros((n, 3))
    initial_guess[:, :2] = np.array([(0.1 + i * 0.2, 0.1 + j * 0.2) for i in range(5) for j in range(5)])[16:]
    initial_guess[15, :2] = 0.5
    initial_guess[:, 2] = 0.0999

    bounds = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n

    def objective(p):
        return -np.sum(p[2*n:])
    
    def penalty(p):
        d = np.sqrt((p[:n, :2] - p[n:, :2])**2).sum(axis=1)
        return np.sum(np.maximum(0, d - p[:n, 2] - p[n:, 2]) ** 2)
    
    penalty_weight = 1e8

    def equality(p):
        return p[2*n:]
    
    def jac_equality(p):
        return np.eye(n)
    
    def nonlinear(p):
        return penalty(p)
    
    def jac_nonlinear(p):
        d = np.sqrt((p[:n, :2] - p[n:, :2])**2).sum(axis=1)
        penalty_gradient = 2 * np.maximum(0, d - p[:n, 2] - p[n:, 2]) * (p[:n, :2] - p[n:, :2]) / (d + 1e-12)
        penalty_gradient[:n] -= np.eye(n)
        penalty_gradient[n:] -= np.eye(n)
        return penalty_gradient.flatten()

    constraints = [
        {'type': 'ineq', 'fun': nonlinear, 'jac': jac_nonlinear},
        {'type': 'eq', 'fun': equality, 'jac': jac_equality}
    ]

    solution = opt.minimize(objective, initial_guess.flatten(), method='SLSQP', constraints=constraints, bounds=bounds, jac=jac_nonlinear, tol=1e-8)
    
    if not solution.success:
        raise Exception("Optimization failed")
    
    centers = solution.x[:2*n].reshape(n, 2)
    radii = solution.x[2*n:]
    
    return centers, radii, -solution.fun

centers, radii, sum_radii = optimize_circle_packing()
print(centers)
print(radii)
print(sum_radii)
