import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def hill_climb(centers, num_iterations, decay_factor, step_min):
    n = centers.shape[0]
    best_sum = lp_optimum(centers)[1]
    best_centers = centers.copy()

    for it in range(num_iterations):
        idx = np.random.randint(n)
        delta = np.random.uniform(-step, step, size=2)
        new_c = np.clip(centers[idx] + delta, 0, 1)
        new_centres = centers.copy()
        new_centres[idx] = new_c

        new_radii, new_sum = lp_optimum(new_centres)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centres
        elif np.random.rand() < np.exp((best_sum - new_sum) / (step * decay_factor)):
            best_sum = new_sum
            best_centers = new_centres

        step = max(step_min, step * decay_factor)

    return best_centers, best_sum

def run_packing():
    n = 26
    num_iterations = 2000
    decay_factor = 0.995
    step_min = 0.001
    step = 0.05

    num_starts = 30
    best_sum = -np.inf
    best_centers = None

    for start in range(num_starts):
        centres = np.random.rand(n, 2)
        result = differential_evolution(lambda x: -lp_optimum(x.reshape(n, 2))[1], bounds=[(0, 1)] * (2 * n), polish=False)
        final_radii, final_sum = lp_optimum(result.x.reshape(n, 2))
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = result.x.reshape(n, 2)

    final_radii, final_sum = hill_climb(best_centers, num_iterations, decay_factor, step_min)
    return best_centers, final_radii, final_sum
