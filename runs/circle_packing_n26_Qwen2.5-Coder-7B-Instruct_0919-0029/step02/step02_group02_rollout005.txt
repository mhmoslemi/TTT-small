import numpy as np
from scipy.optimize import minimize

def objective_and_gradient(x, n, alpha, beta):
    radii = x[:n]
    centers = x[n:].reshape((n, 2))
    
    # Size penalty
    F_size = -np.sum(radii)
    
    # Distance penalties
    F_pen = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            p_ij = np.maximum(0, radii[i] + radii[j] - d_ij)
            F_pen += p_ij
    
    # Boundary penalties
    F_be = 0
    for i in range(n):
        p_L = np.maximum(0, radii[i] - centers[i, 0])
        p_R = np.maximum(0, radii[i] - (1 - centers[i, 0]))
        p_B = np.maximum(0, radii[i] - centers[i, 1])
        p_T = np.maximum(0, radii[i] - (1 - centers[i, 1]))
        F_be += p_L**2 + p_R**2 + p_B**2 + p_T**2
    
    F_pen *= 1 / alpha
    F_be *= 1 / beta
    F_total = F_size + F_pen + F_be
    
    # Gradient
    grad_size = -np.ones(n)
    grad_pen = np.zeros(n * 2)
    grad_be = np.zeros(n * 2)
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            if d_ij > 1e-12:
                p_ij = np.maximum(0, radii[i] + radii[j] - d_ij)
                grad_size[i] += p_ij
                grad_size[j] += p_ij
                grad_pen[i * 2] = 2 / alpha * p_ij * (centers[i, 0] - centers[j, 0]) / d_ij
                grad_pen[i * 2 + 1] = 2 / alpha * p_ij * (centers[i, 1] - centers[j, 1]) / d_ij
                grad_pen[j * 2] = 2 / alpha * p_ij * (centers[j, 0] - centers[i, 0]) / d_ij
                grad_pen[j * 2 + 1] = 2 / alpha * p_ij * (centers[j, 1] - centers[i, 1]) / d_ij
    
    grad_be[i * 2] = 2 / beta * p_L
    grad_be[i * 2 + 1] = 2 / beta * p_R
    grad_be[i * 2 + 2] = 2 / beta * p_B
    grad_be[i * 2 + 3] = 2 / beta * p_T
    
    grad = np.concatenate((grad_size, grad_pen, grad_be))
    return F_total, grad

def run_packing():
    n = 26
    α = 10**4
    β = 10**6
    step_min = 5e-3
    NUM_ITERS = 2000
    
    jitter = np.random.uniform(-step_min, step_min, size=n * 2)
    centers = np.array([
        [0.1, 0.1],
        [0.3, 0.3],
        [0.5, 0.1],
        [0.7, 0.3],
        [0.9, 0.1],
        [0.25, 0.35],
        [0.75, 0.35],
        [0.5, 0.55],
        [0.1, 0.75],
        [0.3, 0.95],
        [0.5, 0.75],
        [0.7, 0.95],
        [0.9, 0.75],
        [0.25, 0.6],
        [0.75, 0.6],
        [0.5, 0.8],
        [0.1, 0.4],
        [0.3, 0.6],
        [0.5, 0.8],
        [0.7, 1.0],
        [0.9, 0.8],
        [0.25, 0.2],
        [0.75, 0.2],
        [0.5, 0.4],
        [0.1, 0.6],
        [0.3, 0.8],
        [0.5, 1.0],
        [0.7, 0.8]
    ])
    centers += jitter
    centers = np.clip(centers, 0, 1)
    
    initial_radii = 1e-3 * np.ones(n)
    initial_r_and_c = np.concatenate((initial_radii, centers.flatten()))
    
    result = minimize(objective_and_gradient, initial_r_and_c, args=(n, α, β), method='L-BFGS-B', jac=True, bounds=[(0, None)] * n + [(0, 1)] * (n * 2), options={'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': NUM_ITERS})
    
    radii_optimal = result.x[:n]
    centers_optimal = result.x[n:].reshape((n, 2))
    
    radii_optimal, sum_radii_optimal = run_lp_on_final_solution(centers_optimal)
    
    return centers_optimal, radii_optimal, sum_radii_optimal

def run_lp_on_final_solution(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(A_ub.shape[0])
    
    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result_lp = minimize(c, centers.flatten(), method='highs', ArgumentExceptions=A_ub, b_ub=b_ub, bounds=bounds)
    
    radii_cleaned = -result_lp.fun
    centers_optimal = result_lp.x.reshape((n, 2))
    
    return radii_cleaned, np.sum(radii_cleaned)
