import numpy as np
import cvxpy as cp

def f_global(centers, radii):
    N = centers.shape[0]
    penalty = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
    
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        penalty += max(r - x, 0) ** 2
        penalty += max(r - y, 0) ** 2
        penalty += max(r - (1 - x), 0) ** 2
        penalty += max(r - (1 - y), 0) ** 2
        
    return -(radii.sum()) + 1e6 * penalty

def check_feasibility(centers, radii):
    N = centers.shape[0]
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if not (r >= 0 and r <= x and r <= y and r <= 1 - x and r <= 1 - y):
            return False
    
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                return False
    
    return True
