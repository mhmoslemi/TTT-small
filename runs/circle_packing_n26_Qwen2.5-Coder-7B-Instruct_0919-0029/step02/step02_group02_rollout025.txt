import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
from scipy.spatial.distance import cdist

def gradient_function(params):
    n = params.shape[0] // 3
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    n_pairs = n * (n - 1) // 2
    c2c = cdist(centers, centers)
    off diagonals = c2c[off diagonals]
    off diagonals_mask = off diagonals > 0
    off diagonals[off diagonals_mask] = 1 / off diagonals[off diagonals_mask]
    
    grad_sums = -np.ones(n)
    grad_penalties = np.zeros(2 * n + n_pairs)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        grad_sums[i] = -r
        grad_penalties[2*i] = 2 * (r - x)
        grad_penalties[2*i + 1] = 2 * (r - (1 - x))
        grad_penalties[2*n + i] = 2 * np.maximum(0, r - y)
        grad_penalties[2*n + i + 1] = 2 * np.maximum(0, r - (1 - y))
    
    for i in range(n):
        for j in range(i + 1, n):
            r1 = radii[i]
            r2 = radii[j]
            dist = c2c[i, j]
            excess = r1 + r2 - dist
            if excess > 0:
                factor = 2 * excess * off diagonals[i, j]
                grad_penalties[2*i] += factor
                grad_penalties[2*j] += factor
                grad_penalties[2*n + j] += factor
                grad_penalties[2*n + i] += factor
    
    return np.hstack((grad_sums, grad_penalties))

def objective_function(params):
    n = params.shape[0] // 3
    radii = params[2*n:]
    return -np.sum(radii)
