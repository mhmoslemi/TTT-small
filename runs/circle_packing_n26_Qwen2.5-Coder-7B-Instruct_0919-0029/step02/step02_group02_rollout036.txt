import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def pair_distances(centres):
    n = centres.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dists[i, j] = np.linalg.norm(centres[i] - centres[j])
            dists[j, i] = dists[i, j]
    return dists

def build_lp(centres):
    n = centres.shape[0]
    r = cp.Variable(n)
    dists = pair_distances(centres)
    constraints = [r[i] >= 0 for i in range(n)]
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(r[i] + r[j] <= dists[i, j])
        constraints.append(r[i] <= centres[i, 0])
        constraints.append(r[i] <= 1 - centres[i, 0])
        constraints.append(r[i] <= centres[i, 1])
        constraints.append(r[i] <= 1 - centres[i, 1])
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    return prob

def solve_lp(prob):
    prob.solve(solver=cp.HS107, verbose=False, eps=1e-8)
    if prob.status != cp.OPTIMAL:
        raise ValueError("LP did not converge")

def compute_gradient(centres, duals):
    n = centres.shape[0]
    dists = pair_distances(centres)
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for i in range(n):
        for j in range(n):
            if i != j:
                d_ij = dists[i, j]
               grad_x[i] += duals[0][i, j] * (centres[i, 0] - centres[j, 0]) / d_ij
                grad_y[i] += duals[0][i, j] * (centres[i, 1] - centres[j, 1]) / d_ij
        # Boundary constraints
        grad_x[i] += duals[1][i] - duals[2][i]
        grad_y[i] += duals[3][i] - duals[4][i]
    return grad_x, grad_y

def gradient_step(centres, duals):
    n = centres.shape[0]
    step = 0.05
    max_iter = 500
    tol = 1e-6
    
    best_obj = -np.inf
    best_centres = centres.copy()
    best_radii = None
    
    for iter in range(max_iter):
        radii, duals = solve_lp(build_lp(centres))
        obj = radii.sum()
        if obj > best_obj:
            best_obj = obj
            best_centres = centres.copy()
            best_radii = radii.copy()
        
        grad_x, grad_y = compute_gradient(centres, duals)
        norm = np.sqrt((grad_x**2 + grad_y**2).sum())
        if norm < 1e-12: break
        grad = np.vstack((grad_x, grad_y)).T / norm
        
        new_centres = np.clip(centres + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))
        
        if new_obj > obj:
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3
        
        if step < tol: break
    
    return best_centres, best_radii, best_obj
