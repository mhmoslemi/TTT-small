import numpy as np
from scipy.optimize import linprog

def initial_grid_plus_center():
    grid_points = np.array([(0.125 + 0.25 * i, 0.125 + 0.25 * j) for i in range(4) for j in range(4)])
    extra_points = np.array([(0.125 + 0.25 * i, 0.205 + 0.25 * j) for i in range(4) for j in range(4)])
    centers = np.vstack((grid_points, extra_points))
    centers = centers + 0.1 * np.random.randn(26, 2)
    centers = np.clip(centers, 0, 1)
    centers[25] = 0.5, 0.5
    return centers

def lp_solve(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * (n - 1) // 2 + j - i - 1, i] = 1
            A_ub[i * (n - 1) // 2 + j - i - 1, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[i * (n - 1) // 2 + j - i - 1] = dist - 1e-12

    for i in range(n):
        A_ub[n * (n - 1) // 2 + 4 * i, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i] = centers[i, 0] - 1e-12
        A_ub[n * (n - 1) // 2 + 4 * i + 1, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 1] = 1 - centers[i, 0] - 1e-12
        A_ub[n * (n - 1) // 2 + 4 * i + 2, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 2] = centers[i, 1] - 1e-12
        A_ub[n * (n - 1) // 2 + 4 * i + 3, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 3] = 1 - centers[i, 1] - 1e-12

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26
    N_restart = 40
    N_iter = 4000
    step_min = 1e-3
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(N_restart):
        centers = initial_grid_plus_center()
        for it in range(N_iter):
            k = np.random.randint(n)
            delta = 0.15 * (2 * np.random.rand(2) - 1)
            new_c = np.clip(centers[k] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[k] = new_c

            new_radii, new_sum = lp_solve(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, 0.15 * (0.995 ** (restart + 1)))

    final_radii, final_sum = lp_solve(best_centers)
    return best_centers, final_radii, final_sum
