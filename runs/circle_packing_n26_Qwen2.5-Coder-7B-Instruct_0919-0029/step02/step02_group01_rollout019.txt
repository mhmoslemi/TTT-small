import numpy as np
from scipy.optimize import differential_evolution, trust_constr

def objective(p):
    n = len(p) // 3
    x, y, r = p[:n], p[n:2*n], p[2*n:]
    return -np.sum(r) + 1e8 * np.sum(np.maximum(g(x, y, r), -1)**2)

def g(x, y, r):
    n = len(x)
    G = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx**2 + dy**2)
            G[i, j] = dist - (r[i] + r[j])
    return G

def constraint_radius_bounds(p):
    n = len(p) // 3
    x, y, r = p[:n], p[n:2*n], p[2*n:]
    return np.concatenate([r - x, r - y, x - r, y - r])

def constraint_pairwise_non_overlap(p):
    n = len(p) // 3
    x, y, r = p[:n], p[n:2*n], p[2*n:]
    return np.maximum(g(x, y, r), -1)

n = 26
bounds = [(0, 1), (0, 1), (0, 0.5)] * n

initial_guess = np.concatenate([
    np.linspace(0.1, 0.9, 5)[np.newaxis].T.repeat(5, axis=0).flatten(),
    np.linspace(0.1, 0.9, 5)[np.newaxis].T.repeat(5, axis=0).flatten(),
    np.full(n, 0.0999)
])

result_DE = differential_evolution(
    objective, bounds, method="differential_evolution", maxiter=400, popsize=15
)

p_DE = result_DE.x
bounds_radius = np.array([
    [0, bounds[i]] for i in range(3*n) if (i // n) % 3 == 2
] * n).flatten()

result_TC = trust_constr(
    objective, p_DE, bounds=bounds_radius, constraints=[
        {'type': 'ineq', 'fun': constraint_radius_bounds},
        {'type': 'ineq', 'fun': constraint_pairwise_non_overlap}
    ], jac=(lambda p: np.array([
        np.gradient(objective, p, axis=i) for i in range(len(p))
    ]).T)
)

x_opt, y_opt, r_opt = result_TC.x[:n], result_TC.x[n:2*n], result_TC.x[2*n:]
sum_radii = np.sum(r_opt)

centers = np.column_stack((x_opt, y_opt))
return centers, r_opt, round(sum_radii, 10)
