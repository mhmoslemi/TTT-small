import numpy as np
from scipy.optimize import differential_evolution, LinearConstraint, NonlinearConstraint
from scipy.spatial import distance

def penalty_sum(p):
    x, y, r = p.reshape((26, 2)), p[52:].reshape((26,))
    return -np.sum(r) + 1e8 * np.sum(np.clip(distance.pdist(x), 0, None) - (r[:, None] + r[None, :]) <= 0)

def bounds(N):
    return [(0, None)] * (2 * N) + [(0, 0.5)] * N

con1 = LinearConstraint(np.eye(78), bounds=[0, 0.5] * 26)
con2 = NonlinearConstraint(
    lambda p: np.clip(distance.pdist(p[:52].reshape((26, 2))), 0, None) - (p[52:].reshape((26,))[:, None] + p[52:].reshape((26,))[None, :]),
    0,
    np.inf,
    jac='2-point',
    hess=None
)

def run_packing():
    n = 26
    initial = np.array([x for(px,py) in [(0.1+0.2*i,0.1+0.2*j) for i in range(5) for j in range(5)] for x in (px,py)] + [0.5,0.5] * 2 + [0.0999] * n)
    result = differential_evolution(penalty_sum, bounds(bounds(n)), constraints=[con1, con2] * 2, tol=1e-6, maxiter=500, popsize=15)
    final_r = result.x[52:]
    final_x = result.x[:52].reshape((26, 2))
    final_sum = np.sum(final_r)
    return final_x, final_r, final_sum
