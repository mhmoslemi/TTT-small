        A_ineq[-1][i] = 1

            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(2 - dists[i, j])

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    bounds = [(0, None) for _ in range(n)]

    result = opt('SLSQP', np.ones(n), args=(A_ineq, b_ineq, bounds), 
                fun=lambda radii: -np.sum(radii), method='trust-constr')
    
    if not result.success:
        raise Exception("LP optimization failed")
    
    return result.x, -result.fun
