def gradient_function(x):

    y_coords = x[n:2*n]
    r_coords = x[2*n:]
    
    n_pairs = n * (n - 1) // 2
    gradients = np.zeros(3 * n)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = x_coords[i] - x_coords[j]
            dy = y_coords[i] - y_coords[j]
            d = np.sqrt(dx**2 + dy**2)
            if d > 0:
                p_ij = max(0, r_coords[i] + r_coords[j] - d)
                if p_ij > 0:
                    gradients[2 * i] += 2 * p_ij * dx / d
                    gradients[2 * i + 1] += 2 * p_ij * dy / d
                    gradients[2 * j] += 2 * p_ij * dx / d
                    gradients[2 * j + 1] += 2 * p_ij * dy / d
                    gradients[2 * n + i] += 2 * r_coords[i] * p_ij / d
                    gradients[2 * n + j] += 2 * r_coords[j] * p_ij / d
    
    for i in range(n):
        p_L = max(0, r_coords[i] - x_coords[i])
        p_R = max(0, r_coords[i] - (1 - x_coords[i]))
        p_B = max(0, r_coords[i] - y_coords[i])
        p_T = max(0, r_coords[i] - (1 - y_coords[i]))
        gradients[2 * i] += 2 * p_L / r_coords[i] if p_L > 0 else 0
        gradients[2 * i + 1] += 2 * p_B / r_coords[i] if p_B > 0 else 0
        gradients[2 * n + i] += 2 * p_L if p_L > 0 else 0
        gradients[2 * n + i] += 2 * p_R if p_R > 0 else 0
        gradients[2 * n + i] += 2 * p_B if p_B > 0 else 0
        gradients[2 * n + i] += 2 * p_T if p_T > 0 else 0
    
    for i in range(n):
        gradients[2 * n + i] -= 1
    
    return gradients
