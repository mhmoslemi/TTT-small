import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    PA = squareform(pdist(centers))
    PA = PA * 2 - np.eye(n)  # Adjust to get boundary distances directly
    c = np.ones(n)
    A_ub = np.hstack([PA, np.zeros((n, 4*n)), np.eye(n)])
    b_ub = np.vstack([PA.ravel(), np.zeros((4*n,)), np.ones(n)])
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x[0:n], -result.fun
