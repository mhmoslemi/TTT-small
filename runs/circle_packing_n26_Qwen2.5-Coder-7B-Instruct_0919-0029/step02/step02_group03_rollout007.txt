import cvxpy as cp
import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N)

    objective = cp.Maximize(cp.sum(r))

    constraints = []
    for i in range(N):
        constraints.append(r[i] >= 0)
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])
        
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.norm([x[i] - x[j], y[i] - y[j]], 2) >= r[i] + r[j])

    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-7, reltol=1e-7)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False, abstol=1e-7, reltol=1e-7)
        except:
            # Fallback to previous approach if both solvers fail
            return None, None, None

    centers = np.vstack((x.value, y.value)).T
    radii = r.value
    radii = np.clip(radii, 0, None)
    centers = np.clip(centers, 0, 1)
    
    def compute_sum_radii(centers, radii):
        return np.sum(radii)

    min_sum_radii = np.inf
    for i in range(10):
        lp_centers = cp.Variable((N, 2))
        lp_r = cp.Variable(N)
        lp_objective = cp.Minimize(cp.sum(lp_r))
        lp_constraints = [lp_r >= 0, lp_r <= cp.maximum(cp.maximum(cp.maximum(lp_centers[:, 0], 1 - lp_centers[:, 0]), lp_centers[:, 1]), 1 - lp_centers[:, 1])]
        for j in range(N):
            for k in range(j+1, N):
                lp_constraints.append(cp.norm(lp_centers[j] - lp_centers[k], 2) >= lp_r[j] + lp_r[k])
        lp_prob = cp.Problem(lp_objective, lp_constraints)
        lp_prob.solve(solver=cp.SCS, verbose=False)
        sum_radii = compute_sum_radii(lp_centers.value, lp_r.value)
        if sum_radii < min_sum_radii:
            min_sum_radii = sum_radii
            refined_centers = lp_centers.value
            refined_radii = lp_r.value

    refined_radii = np.clip(refined_radii, 0, None)
    refined_centers = np.clip(refined_centers, 0, 1)
    refined_sum_radii = compute_sum_radii(refined_centers, refined_radii)
    
    return refined_centers, refined_radii, refined_sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
