import cvxpy as cp
import numpy as np

def compute_sum_radii(centers, radii):
    N = centers.shape[0]
    sum_radii = np.sum(radii)
    constraints = []
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j])
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)
    opt = cp.Problem(cp.Maximize(sum_radii), constraints)
    opt.solve(solver=cp.ECOS, verbose=False)
    sum_radii_opt = opt.value
    return sum_radii_opt - sum_radii if np.abs(opt.value - sum_radii) > 1e-6 else 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    x = cp.Variable(N)
    y = cp.Variable(N)
    r = cp.Variable(N)
    
    objective = cp.Maximize(cp.sum(r))
    
    # Radius positivity
    constraints = [r >= 0.0]
    
    # Boundary conditions
    for i in range(N):
        constraints.extend([r[i] <= x[i], r[i] <= 1 - x[i], r[i] <= y[i], r[i] <= 1 - y[i]])
    
    # Overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2) >= r[i] + r[j])
    
    problem = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status != cp.OPTIMAL:
            raise Exception('Solver did not find an optimal solution (status: {})'.format(problem.status))
    
        radii = np.maximum(r.value, 0.0)
        centers = np.vstack((x.value, y.value)).T
    except Exception as e:
        print("Error occurred: ", str(e))
    
    # Fallback to evasion if solver fails
    if radii is None:
        centers, radii, _ = run_packing_fallback()
    
    # Re$/,olve LP for the valid final radii
    sum_radii = compute_sum_radii(centers, radii)
    
    # Clip outputs
    centers = np.clip(centers, 0, 1)
    return centers, radii, sum_radii

def run_packing_fallback() -> tuple[np.ndarray, np.ndarray, float]:
    # Fallback to differential evolution if the optimization fails
    import numpy as np
    from scipy.optimize import differential_evolution
    def evaluation(centers):
        X = centers.reshape((26, 3))
        radii = X[:, 2]
        penalty = 0
        for i in range(26):
            for j in range(i + 1, 26):
                d2 = (X[i, 0] - X[j, 0]) ** 2 + (X[i, 1] - X[j, 1]) ** 2
                penalty += max((radii[i] + radii[j]) ** 2 - d2, 0) ** 2
            penalty += max(radii[i] - X[i, 0], 0) ** 2
            penalty += max(radii[i] - X[i, 1], 0) ** 2
            penalty += max(radii[i] - (1 - X[i, 0]), 0) ** 2
            penalty += max(radii[i] - (1 - X[i, 1]), 0) ** 2
        return -(radii.sum()) + 1e6 * penalty

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * 26
    result = differential_evolution(evaluation, bounds, popsize=20, maxiter=200, seed=42, workers=1)
    
    X = result.x
    centers = X.reshape((26, 3))[:, :2]
    radii = X.reshape((26, 3))[:, 2]
    sum_radii = radii.sum()
    return centers, radii, sum_radii

# Output the result of the packing function
centers, radii, sum_radii = run_packing()
