import numpy as np
import cvxpy as cp
from scipy.optimize import linprog, solve_ivp

def solve_radius_lp(centres):
    n = centres.shape[0]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 0] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(centres[i, 1] - 1e-12)
        A_ineq.append([0]*n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centres[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0]*n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def solve_position_socp(radii, centres, prev_centres):
    n = centres.shape[0]

    # Define the variables for the quadratic cost
    diff = centres - prev_centres
    Q = 2 * np.dot(diff.T, diff)

    # Define the SOCP constraints
    M = cp.Variable((2*n, 2*n))
    constraints = []
    for i in range(n):
        constraints.append(cp.norm(cp.vstack([centres[i, 1:], r - centres[i, 0]])) >= radii[i])
        constraints.append(0 <= centres[i, 0] <= 1)
        constraints.append(0 <= centres[i, 1] <= 1)

    # Solve the SOCP problem
    problem = cp.Problem(cp.Minimize(cp.quad_form(diff, Q)), constraints)
    problem.solve(solver=cp.ECOS)

    if problem.status != cp.OPTIMAL:
        raise Exception("SOCP optimization failed")

    c = problem.variables()
    return np.array([cp.value(c[i]) for i in range(2*n)]).reshape(n, 2), cp.value(problem.objective)

def run_packing():
    """Returns (centres, radii, sum_radii) as described in the function signature."""
    # Initialize the centres with a hexagonal lattice truncated to the unit square
    n = 26
    r = np.sqrt(3/4) / 2
    side_length = np.sqrt(3) * r
    num_rows = 14
    num_cols = 27
    centres = np.zeros((n, 2))
    
    cx = np.cumsum(np.concatenate(([0], np.arange(1, num_cols) * side_length, [2*num_rows * side_length - side_length])))
    cy = np.concatenate([np.arange(num_rows) * side_length for _ in range(num_cols // 2 + 1)])
    cx = cx[:n]
    cy = cy[:n]
    centres[:, 0] = cx / (2 * np.sqrt(3) * r * np.cos(np.pi / 6))
    centres[:, 1] = cy / (2 * np.sqrt(3) * r * np.cos(np.pi / 6))
    
    # Random perturbations
    noise = 0.01 * np.random.randn(n, 2)
    noise[:, 0] = noise[:, 0] * (1 / np.cos(np.pi / 6))
    noise[:, 1] = noise[:, 1] * (1 / np.cos(np.pi / 6))
    centres[11:] += noise[11:]
    centres[8:] += noise[8:]
    centres[5:] += noise[5:]
    centres[2:] += noise[2:]
    centres = np.clip(centres, 0, 1)

    # Perform alternating suboptimal optimisation
    tolerance_r = 1e-6
    tolerance_c = 1e-6
    iterations = 20
    best_r = None
    best_c = None
    max_sum_radii = -np.inf

    for _ in range(3):  # Restart the process three times
        r = np.zeros(n)
        prev_c = np.zeros((n, 2))
        for _ in range(iterations):
            # Solve the radius subproblem
            r, sum_radii = solve_radius_lp(prev_c)
            # Solve the position subproblem
            c, _ = solve_position_socp(r, prev_c, prev_c)

            # Check for convergence
            if abs(sum_radii - max_sum_radii) < tolerance_r and np.all(np.linalg.norm(c - prev_c, axis=1) < tolerance_c):
                break

            prev_c = c
            max_sum_radii = sum_radii

        r, _ = solve_radius_lp(prev_c)
        if sum(r) > max_sum_radii:
            max_sum_radii = sum(r)
            best_r = r
            best_c = prev_c

    return best_c, best_r, max_sum_radii
