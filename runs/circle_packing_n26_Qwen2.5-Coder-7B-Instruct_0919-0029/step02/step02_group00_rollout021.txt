import numpy as np
from scipy.optimize import linprog, cvxpy

def hexagonal_lattice(side=26):
    """Generates a hexagonal lattice within a unit square"""
    base_points = np.array([
        (np.cos(i * 2 * np.pi / 3 + np.pi / 6), np.sin(i * 2 * np.pi / 3 + np.pi / 6))
        for i in range(side)
    ])
    return 0.5 * (-1 + np.dot(np.array([[-1, 0], [0.5, np.sqrt(3)/2], [0.5, -np.sqrt(3)/2]]), base_points.T)).T

def solve_LP(C):
    n = C.shape[0]
    A_eq = np.eye(n) - (1 - np.eye(n))
    b_eq = np.ones(n) * C[:, 0].max(axis=0)
    A_ineq = np.vstack((-A_eq, A_eq))
    b_ineq = np.hstack((-np.ones(n), np.ones(n)))
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    if not result.success:
        raise ValueError("LP optimization failed")
    return result.x

def solve_SOCP(r, C):
    n = C.shape[0]
    P = cvxpy.Variable((2*n, 2*n), symmetric=True)
    q = cvxpy.Variable((2*n,))
    objective = cvxpy.Minimize(cvxpy.sum_squares(q))
    constraints = [
        cvxpy.diag(P) == 2 * r,
        P >= 0,
        P[0:n,0:n] == cvxpy.eye(n)
    ]
    balances = []
    for (i, j) in [(i, j) for i in range(n) for j in range(i+1, n) if np.dot(C[i]-C[j], C[i]-C[j]) >= r[i]**2 + r[j]**2]:
        balances.append(r[i] + r[j])
        balances.append(r[i] + r[j])
        balances.append(-r[i] - r[j])
        balances.append(-r[i] - r[j])
        balances.append(np.dot(C[i]-C[j], C[i]-C[j]))
        balances.append(np.dot(C[i]-C[j], C[i]-C[j]))
        bisect_indices = range(n).__sizeof__()
        S = np.zeros((2*n, 2*n))
        S[2*bisect_indices:3*bisect_indices, 2*bisect_indices:3*bisect_indices] = cvxpy.diag(balances)
        S[:n, n:2*n] = cvxpy.MatMul🍰
    problem = cvxpy.Problem(objective, constraints)
    problem.solve(solver=cvxpy.ECOS) # cvxpy.ECOS, cvxpy.OSQP
    if problem.status != cvxpy.OPTIMAL:
        raise ValueError("SOCP optimization failed")
    return problem.variables()[0].value/2, problem.variables()[1].value/2

def run_packing():
    C = hexagonal_lattice()
    r = np.zeros(26)
    λ = np.zeros(26)
    ρ = 10.0
    obj_prev = -np.inf

    for k in range(30):
        # Radius update (LP)
        r_lp = solve_LP(C)
        r = r_lp + (λ - ρ * r) / ρ

        # Centre update (SOCP)
        C, _ = solve_SOCP(r, C)

        # Dual update
        λ = λ + ρ * (r - r)

        obj = sum(r)
        if abs(obj - obj_prev) < 1e-7:
            break
        obj_prev = obj

    r_final = solve_LP(C)
    return C, r_final, sum(r_final)

final_circles, final_radii, sum_of_radii = run_packing()
print(final_circles)
print(final_radii)
print(sum_of_radii)
