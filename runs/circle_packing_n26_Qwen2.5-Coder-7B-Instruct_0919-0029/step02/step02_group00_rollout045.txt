def solve_centre_soc(radii, centres, eps=1e-8):
    n = centres.shape[0]
    C = cp.Variable((n, 2))
    Q = cp.Variable((n, n)) + cp.Variable((n, n)).T
    D = 2 * (cp.multiply(radii.reshape(n, 1), radii.reshape(1, n)) + eps)
    constraints = []
    constraints += [c >= 0 for c in C]
    constraints += [c <= 1 for c in C.T]
    constraints += [norm(c_i - c_j, 2) >= r_i + r_j for c_i, c_j, r_i, r_j in zip(C, C.T, radii, radii.T)]
    constraints += [Q == C.T @ (D - cp.multiply(C, C.T))]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(C - centres)), constraints)
    prob.solve(solver=cp.ECOS)  # Use ECOS for a robust solver
    return C.value, prob.value

def run_packing():
    initial_centers = np.array([[i/5, j/5] for i in range(6) for j in range(5)])[:26]  # Hexagonal lattice truncated to [0,1]^2
    best_radius_sum = 0
    best_centers = None
    for _ in range(3):  # Three independent restarts
        centers = initial_centers + np.random.randn(26, 2) * 0.01  # Gaussian noise-based perturbations
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)  # Ensure within bounds

        for _ in range(10):  # 10 iterations per restart

                best_radius_sum = opt_val
                best_centers = new_centers
        final_radii, final_opt_val = solve_radius_lp(best_centers)
        if final_opt_val > best_radius_sum:
