```python
import numpy as np
from scipy.optimize import linprog

NUM_RESTARTS = 30
NUM_ITERATIONS = 4000
TAU_STEP = 0.997
TAU_TEMP = 0.997
EPSILON0 = 0.05

def create_initial_grid_and_add_center():
    grid_points = np.array([[0.1 * (i + 0.5), 0.1 * (j + 0.5)] for i in range(5) for j in range(5)])
    centers = np.vstack([grid_points, [0.5, 0.5]])
    return centers

def perturb_randomly(centers, epsilon0):
    epsilon = EPSILON0 * np.ones(2)
    noise = np.random.uniform(-1, 1, size=(26, 2)) * epsilon
    perturbed_centers = np.clip(centers + noise, 0, 1)
    return perturbed_centers

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if not result.success:
        return np.zeros(n), -np.inf

    return -result.x, result.fun

def move_center(centers, k, step):
    from math import pi
    angle = np.random.uniform(0, 2*pi)
    delta = step * np.array([np.cos(angle), np.sin(angle)])
    new_centers = centers.copy()
    new_centers[k] += delta
    new_centers[k] = np.clip(new_centers[k], 0, 1)
    return new_centers

def random_accept(delta, temp):
    if delta > 0:
        return True
    return np.random.rand() < np.exp(delta / temp)

def run_packing():
    best_centers, best_radii, best_sum = None, None, -np.inf

    for restart in range(NUM_RESTARTS):
        centers = create_initial_grid_and_add_center()
        centers = perturb_randomly(centers, EPSILON0)

        radii, sum_r = solve_lp(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITERATIONS):
            k = np.random.randint(26)
            new_centers = move_center(centers, k, step)
            new_radii, new_sum = solve_lp(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if new_sum > best_sum:
                    best_centers, best_radii, best_sum = new_centers.copy(), new_radii.copy(), new_sum
            step *= TAU_STEP
            temp *= TAU_TEMP

    best_radii, best_sum = solve_lp(best_centers)
    return best_centers, best_radii, best_sum
```
