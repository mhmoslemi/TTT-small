import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = dist  # exact, not offset by -1e-12 for now
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        radii = np.zeros(n)
    else:
        radii = np.maximum(result.x, 0.0)
    return radii, -result.fun if result.success else 0

def random_accept(delta, temp):
    if delta > 0:
        return True
    else:
        return np.random.uniform() < np.exp(delta / temp)

def move_center(centers, k, step):
    direction = np.random.uniform(size=2)
    direction /= np.linalg.norm(direction)
    new_c = centers[k] + step * direction
    new_c = np.clip(new_c, 0, 1)
    centers[k] = new_c
    return centers

        centers[25] = np.array([0.5, 0.5])
        centers += np.random.uniform(-epsilon0, epsilon0, size=centers.shape)
        centers = np.clip(centers, 0, 1)
        radii, sum_r = solve_lp(centers)

        step, temp = 0.15, 0.5
        for _ in range(NUM_ITER):
            k = np.random.randint(26)
            new_centers = move_center(centers.copy(), k, step)
            new_radii, new_sum = solve_lp(new_centers)
            if new_sum > sum_r or random_accept(new_sum - sum_r, temp):
                centers, radii, sum_r = new_centers, new_radii, new_sum
                if sum_r > best_sum:
                    best_centers, best_radii, best_sum = centers.copy(), radii.copy(), sum_r
            step *= tau_step
            temp *= tau_temp

        best_newRadii, best_newSum = solve_lp(best_centers)
        if best_newSum > best_sum:
            best_radii, best_sum = best_newRadii, best_newSum

    return best_centers, best_radii, best_sum
