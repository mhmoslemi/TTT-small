import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    m = n*(n-1)//2 + 4*n
    A = np.empty((m,n))
    b = np.empty(m)
    idx=0
    for i in range(n):
        for j in range(i+1,n):
            A[idx,i]=1
            A[idx,j]=1
            b[idx]=np.linalg.norm(centers[i]-centers[j]) - 1e-12
            idx+=1
    for i in range(n):
        x, y = centers[i]
        A[idx,i]=1; b[idx] = x-1e-12; idx+=1
        A[idx,i]=1; b[idx] = 1 - x -1e-12; idx+=1
        A[idx,i]=1; b[idx] = y-1e-12; idx+=1
        A[idx,i]=1; b[idx] = 1 - y -1e-12; idx+=1
    c = -np.ones(n)
    bounds = [(0,None)]*n
    res = linprog(c,A,b, bounds=bounds,method='highs')
    return res.x, - res.fun

def run_packing():
    np.random.seed(42)
    NUM_ITERS = 4000
    NUM_STARS = 20
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centers = np.random.rand(26,2)
        radii, sum_radii = lp_optimum(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = radii

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
