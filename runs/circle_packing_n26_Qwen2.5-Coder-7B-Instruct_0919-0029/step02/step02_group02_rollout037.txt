import numpy as np
import cvxpy as cp

def pair_distances(centres):
    n = centres.shape[0]
    return np.linalg.norm(centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=2)

def build_lp(centres):
    n = centres.shape[0]
    r = cp.Variable(n)
    d = pair_distances(centres)
    
    constraints = [0 <= r, r <= 0.5]
    for i in range(n):
        constraints.extend([r[i] <= centres[i, 0], r[i] <= 1 - centres[i, 0],
                            r[i] <= centres[i, 1], r[i] <= 1 - centres[i, 1]])

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= d[i, j])

    obj = cp.Maximize(cp.sum(r))
    prob = cp.Problem(obj, constraints)
    return prob, d

def solve_lp(prob):
    prob.solve(solver=cp.HS)
    r = prob.variables()[0].value
    dual_pair = {}
    dual_left = prob.constraints[4*n:5*n].dual_value
    dual_right = prob.constraints[5*n:6*n].dual_value
    dual_bottom = prob.constraints[6*n:7*n].dual_value
    dual_top = prob.constraints[7*n:].dual_value
    
    for i in range(4*n, 8*n):
        for j in range(i + 1, 8*n):
            dual_pair[(i-4*n, j-4*n)] = prob.constraints[i].dual_value[j-4*n]
            dual_pair[(j-4*n, i-4*n)] = prob.constraints[i].dual_value[j-4*n]

    return r, dual_pair, dual_left, dual_right, dual_bottom, dual_top

def compute_gradient(centres, duals):
    n = centres.shape[0]
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    d = pair_distances(centres)
    
    for i in range(n):
        for j in range(i + 1, n):
            if d[i, j] < 1e-12:
                continue
            grad_x[i] += dual_pair[(i, j)] * (centres[i, 0] - centres[j, 0]) / d[i, j]
            grad_x[j] -= dual_pair[(i, j)] * (centres[i, 0] - centres[j, 0]) / d[i, j]
            grad_y[i] += dual_pair[(i, j)] * (centres[i, 1] - centres[j, 1]) / d[i, j]
            grad_y[j] -= dual_pair[(i, j)] * (centres[i, 1] - centres[j, 1]) / d[i, j]
        
        if dual_left[i] > 1e-10:
            grad_x[i] += dual_left[i]
        if dual_right[i] > 1e-10:
            grad_x[i] -= dual_right[i]
        if dual_bottom[i] > 1e-10:
            grad_y[i] += dual_bottom[i]
        if dual_top[i] > 1e-10:
            grad_y[i] -= dual_top[i]
    
    return grad_x, grad_y

def initial_pattern():
    centres = np.array([
        [0.5, 0.5], [0.5, 0.2], [0.5, 0.8],
        [0.2, 0.5], [0.8, 0.5],
        [0.3334, 0.3334], [0.3334, 0.6666],
        [0.6666, 0.3334], [0.6666, 0.6666],
        [0.25, 0.5], [0.75, 0.5],
        [0.5, 0.25], [0.5, 0.75],
        [0.25, 0.25], [0.25, 0.75],
        [0.75, 0.25], [0.75, 0.75],
        [0.125, 0.125], [0.125, 0.375],
        [0.375, 0.125], [0.375, 0.375],
        [0.875, 0.125], [0.875, 0.375],
        [0.625, 0.625], [0.625, 0.875],
        [0.875, 0.625], [0.875, 0.875]
    ])
    return centres + 0.01 * (2 * np.random.rand(n, 2) - 1)

def run_packing():
    MAX_ITERS = 500
    step = 0.1
    best_obj = -inf
    best_centres = None
    best_radii = None

    for iter in range(MAX_ITERS):
        centres = initial_pattern()
        radii, dual_pair, dual_left, dual_right, dual_bottom, dual_top = solve_lp(build_lp(centres))
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centres.copy(), radii.copy()
        
        grad_x, grad_y = compute_gradient(centres, dual_pair, dual_left, dual_right, dual_bottom, dual_top)
        norm = np.sqrt((grad_x**2 + grad_y**2).sum())
        if norm < 1e-12:
            break
        grad = np.vstack((grad_x, grad_y)).T / norm
        
        new_centres = np.clip(centres + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))
        
        if new_obj > obj:
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3
        
        if step < 1e-6:
            break

    return best_centres, best_radii, best_obj

centres, radii, sum_radii = run_packing()
