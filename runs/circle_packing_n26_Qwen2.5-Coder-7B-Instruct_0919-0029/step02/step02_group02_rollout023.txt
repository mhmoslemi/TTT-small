import numpy as np
import cvxpy as cp
from scipy.spatial import distance_matrix

def pair_distances(centers):
    return distance_matrix(centers, centers)

def build_lp(centers):
    n = centers.shape[0]
    dual_pair = cp.Variable((n, n), name='dual_pair', nonneg=True)
    dual_left = cp.Variable((n,), name='dual_left', nonneg=True)
    dual_right = cp.Variable((n,), name='dual_right', nonneg=True)
    dual_bottom = cp.Variable((n,), name='dual_bottom', nonneg=True)
    dual_top = cp.Variable((n,), name='dual_top', nonneg=True)
    radii = cp.Variable(n, name='radii', nonneg=True)
    distances = distance_matrix(centers, centers)
    left = cp.Bounded(-cp.Infinity, cp.Dot(centers[:, 0], cp.ones(n)), cp.Infinity)
    right = cp.Bounded(-cp.Infinity, cp.Dot(1 - centers[:, 0], cp.ones(n)), cp.Infinity)
    bottom = cp.Bounded(-cp.Infinity, cp.Dot(centers[:, 1], cp.ones(n)), cp.Infinity)
    top = cp.Bounded(-cp.Infinity, cp.Dot(1 - centers[:, 1], cp.ones(n)), cp.Infinity)
    objective = cp.Maximize(cp.sum(radii))
    constraints = [radii + radii <= distances, left <= radii, right >= radii, bottom <= radii, top >= radii]
    prob = cp.Problem(objective, constraints)
    return prob, dual_pair, dual_left, dual_right, dual_bottom, dual_top

def solve_lp(prob):
    prob.solve(solver=cp.HS, verbose=False)
    if prob.value is None:
        raise ValueError('LP solution did not converge')
    return prob.variables()['radii'].value, {
        (i, j): prob.constraints[(n*(n-1)//2 + i*n + j).value].dual
        for i in range(n) for j in range(i, n)
    }

def compute_gradient(centers, duals):
    n = centers.shape[0]
    dists = distance_matrix(centers, centers)
    grad_x, grad_y = [0.] * n, [0.] * n
    for i in range(n):
        for j in range(i, n):
            if j == i:
                continue
            d_ij = dists[i, j]
            if d_ij < 1e-12:
                continue
            grad_x[i] += duals[i, j] * (centers[i, 0] - centers[j, 0]) / d_ij
            grad_x[j] -= duals[i, j] * (centers[i, 0] - centers[j, 0]) / d_ij
            grad_y[i] += duals[i, j] * (centers[i, 1] - centers[j, 1]) / d_ij
            grad_y[j] -= duals[i, j] * (centers[i, 1] - centers[j, 1]) / d_ij
        grad_x[i] += duals[i, i] - duals[i, i]
        grad_y[i] += duals[i, i] - duals[i, i]
    return grad_x, grad_y

def initial_pattern():
    centers = np.array([
        [0.15, 0.65], [0.35, 0.15], [0.45, 0.35], [0.65, 0.55], [0.75, 0.65],
        [0.15, 0.35], [0.25, 0.15], [0.35, 0.35], [0.45, 0.55], [0.55, 0.65],
        [0.05, 0.35], [0.15, 0.15], [0.25, 0.35], [0.35, 0.55], [0.45, 0.65],
        [0.25, 0.65], [0.35, 0.45], [0.45, 0.55], [0.55, 0.65], [0.45, 0.65],
        [0.55, 0.35], [0.65, 0.15], [0.75, 0.35], [0.85, 0.55], [0.95, 0.65]
    ])
    np.random.seed(1)
    noise = np.random.uniform(-0.01, 0.01, centers.shape)
    centers += noise
    return centers

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    centers = initial_pattern()
    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for _ in range(1000):
        radii, duals = solve_lp(build_lp(centers))
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centers.copy(), radii.copy()
            
        grad_x, grad_y = compute_gradient(centers, duals)
        grad_norm = np.linalg.norm(grad_x)
        if grad_norm < 1e-12:
            break
        grad = np.vstack((grad_x, grad_y)) / grad_norm

        new_centres = np.clip(centers + step * grad, 0, 1)
        new_radii, _ = solve_lp(build_lp(new_centres))
        new_obj = new_radii.sum()

        if new_obj > obj:
            centers = new_centres
            step *= 0.9
        else:
            step *= 0.3

        if step < 1e-6:
            break

    return best_centres, best_radii, best_obj
