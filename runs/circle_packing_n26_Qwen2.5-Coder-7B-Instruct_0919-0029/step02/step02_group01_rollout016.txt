import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_grid_plus_center():
    centers = np.array([(0.125 + 0.25 * i, 0.125 + 0.25 * j) for i in range(4) for j in range(4)] + [(0.5, 0.5)])
    return centers[~np.all(centers == (0.5, 0.5), axis=1)]

def gaussian_perturb(centers, σ=0.02):
    noise = np.random.normal(0, σ, size=(centers.shape[0], 2))
    centers += noise
    centers = np.clip(centers, 0, 1)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_RESTARTS = 40
    NUM_ITERS = 4000
    step, step_min = 0.15, 1e-3
    decay_factor = 0.995
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for restart in range(NUM_RESTARTS):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, σ=0.02)

        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.normal(0, step, size=2)
            new_c = np.clip(centers[idx] + delta, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
