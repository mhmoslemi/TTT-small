import numpy as np
import cvxpy as cp

def pair_distances(centres):
    n = centres.shape[0]
    d = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d[i, j] = np.linalg.norm(centres[i] - centres[j])
    d += d.T
    np.fill_diagonal(d, 0)
    return d

def build_lp(centres):
    n = centres.shape[0]
    radii = cp.Variable(n, nonneg=True)
    distances = pair_distances(centres)
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(radii[i] + radii[j] <= distances[i, j])
    constraints.append(0 <= radii)
    constraints.append(radii <= centres[:, 0])
    constraints.append(radii <= 1 - centres[:, 0])
    constraints.append(radii <= centres[:, 1])
    constraints.append(radii <= 1 - centres[:, 1])
    obj = cp.Maximize(radii.sum())
    prob = cp.Problem(obj, constraints)
    return prob, radii, distances

def solve_lp(prob):
    prob.solve(solver=cp.HSLS)
    duals = {c: c.dual for c in prob.constraints[:n * (n - 1) // 2]}
    for i in range(n):
        duals.update({c: [c.dual] for c in prob.constraints[n * (n - 1) // 2+n*i:n * (n - 1) // 2+n*(i+1)]})
    return prob.variables()[0].value, duals

def compute_gradient(centres, duals):
    n = centres.shape[0]
    d = pair_distances(centres)
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = d[i, j]
            if d_ij > 0:
                grad_x[i] += duals[(i, j)] * (centres[i, 0] - centres[j, 0]) / d_ij
                grad_x[j] -= duals[(i, j)] * (centres[i, 0] - centres[j, 0]) / d_ij
                grad_y[i] += duals[(i, j)] * (centres[i, 1] - centres[j, 1]) / d_ij
                grad_y[j] -= duals[(i, j)] * (centres[i, 1] - centres[j, 1]) / d_ij
        grad_x[i] += duals[f'x{i}']
        grad_x[i] += duals[f'1 - x{i}']
        grad_y[i] += duals[f'y{i}']
        grad_y[i] += duals[f'1 - y{i}']
    return grad_x, grad_y

def initial_pattern():
    n = 26
    pattern = np.array(
        [[0.5, 0.1, 0.5, 0.1, 0.5, 0.1, 0.5, 0.1],
         [0.1, 0.5, 0.1, 0.5, 0.1, 0.5, 0.1, 0.5],
         [0.5, 0.1, 0.5, 0.1, 0.5, 0.1, 0.5, 0.1],
         [0.1, 0.5, 0.1, 0.5, 0.1, 0.5, 0.1, 0.5],
         [0.5, 0.1, 0.5, 0.1, 0.5, 0.1, 0.5, 0.1]]
    )
    centres = np.vstack((pattern, pattern[:, ::-1], pattern[::-1], pattern[:, ::-1]))
    np.random.shuffle(centres)
    return centres

def run_packing():
    centres = initial_pattern() + 0.01 * np.random.rand(26, 2)
    step = 0.1
    best_obj = -np.inf
    best_centres = None
    best_radii = None

    for iter in range(600):
        radii, duals = solve_lp(build_lp(centres))
        obj = radii.sum()
        if obj > best_obj:
            best_obj, best_centres, best_radii = obj, centres.copy(), radii.copy()

        grad_x, grad_y = compute_gradient(centres, duals)
        norm = np.sqrt((grad_x**2 + grad_y**2).sum())
        if norm < 1e-12: break
        grad = np.vstack((grad_x, grad_y)).T / norm

        new_centres = np.clip(centres + step * grad, 0, 1)
        new_radii, new_obj = solve_lp(build_lp(new_centres))

        if new_obj > obj:
            centres = new_centres
            step *= 0.9
        else:
            step *= 0.3

        if step < 1e-6: break

    return best_centres, best_radii, best_obj
