import numpy as np
from scipy.optimize import linprog, HighsStatus

class PackingOptimizer:
    def __init__(self):
        self.n = 26
        self.step = 0.1
        self.step_min = 0.001
        self.decay_factor = 0.995
        self.NUM_ITERS = 2000
        self.NUM_STARS = 30

    def lp_optimum(self, centers):
        n = self.n
        A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
        b_ub = np.zeros(A_ub.shape[0])

        # Pairwise distance constraints
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[idx, i] = 1
                A_ub[idx, j] = 1
                b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
                idx += 1

        # Boundary constraints
        for i in range(n):
            A_ub[idx, i] = 1
            b_ub[idx] = centers[i, 0] - 1e-12
            idx += 1
            A_ub[idx, i] = 1
            b_ub[idx] = 1 - centers[i, 0] - 1e-12
            idx += 1
            A_ub[idx, i] = 1
            b_ub[idx] = centers[i, 1] - 1e-12
            idx += 1
            A_ub[idx, i] = 1
            b_ub[idx] = 1 - centers[i, 1] - 1e-12
            idx += 1

        c = -np.ones(n)
        bounds = [(0, None)] * n
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.status == HighsStatus.OPTIMAL:
            return result.x, -result.fun
        else:
            return np.zeros(n), 0.0

    def run_packing(self):
        best_sum = -np.inf
        best_centers, best_radii = None, None

        for start in range(self.NUM_STARS):
            centers = self.generate_initial_centers()
            for it in range(self.NUM_ITERS):
                idx = np.random.randint(self.n)
                delta = np.random.uniform(-self.step, self.step, size=2)
                new_c = np.clip(centers[idx] + delta, 0, 1)
                new_centers = centers.copy()
                new_centers[idx] = new_c

                new_radii, new_sum = self.lp_optimum(new_centers)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centers
                    best_radii = new_radii

            self.step = max(self.step_min, self.step * self.decay_factor)

        final_radii, final_sum = self.lp_optimum(best_centers)
        return best_centers, final_radii, final_sum

    def generate_initial_centers(self):
        Δ = 0.2
        centers = np.array([
            [(2 * k + 1) * Δ / 2, (2 * l) * Δ / 2]
            for k in range(5)
            for l in range(5)
        ])
        centers = centers[:(self.n // 2 + 1)]
        centers = np.append(centers, [
            [(2 * k + 1) * Δ / 2, (2 * (l + 1)) * Δ / 2]
            for k in range(5)
            for l in range(5)
        ])
        centers = centers[:(self.n // 2 + 1):2]
        centers += np.random.uniform(-0.005, 0.005, size=(self.n, 2))
        return centers

opt = PackingOptimizer()
centers, radii, sum_radii = opt.run_packing()
print(centers)
print(radii)
print(sum_radii)
