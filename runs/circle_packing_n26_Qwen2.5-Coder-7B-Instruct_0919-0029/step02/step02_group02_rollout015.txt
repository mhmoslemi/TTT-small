import numpy as np
from scipy.optimize import minimize

def objective(vars, centers, alpha=1e4, beta=1e6):
    n = len(centers)
    radii = vars[n:]
    sum_radii = -np.sum(radii)
    p_ij = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            p_ij.append(max(0, radii[i] + radii[j] - dist))
    
    p_boundary = []
    for i in range(n):
        p_boundary.append(max(0, radii[i] - centers[i, 0]))
        p_boundary.append(max(0, radii[i] - (1 - centers[i, 0])))
        p_boundary.append(max(0, radii[i] - centers[i, 1]))
        p_boundary.append(max(0, radii[i] - (1 - centers[i, 1])))
    
    penalty = sum(p_ij)**2 / alpha + sum(p_boundary)**2 / beta
    return sum_radii + penalty

def gradient(vars, centers, alpha=1e4, beta=1e6):
    n = len(centers)
    radii = vars[n:]
    grad_size = -np.ones(n + 2 * n)
    
    p_ij = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist > 1e-12:
                grad_size[i] += (2 * alpha * max(0, radii[i] + radii[j] - dist) * (centers[i, 0] - centers[j, 0]) / dist)
                grad_size[j] += (2 * alpha * max(0, radii[i] + radii[j] - dist) * (centers[i, 1] - centers[j, 1]) / dist)
    
    p_boundary = []
    for i in range(n):
        if radii[i] > 1e-12:
            grad_size[i + n] += 2 * beta * max(0, radii[i] - centers[i, 0])
            grad_size[i + n + 1] += 2 * beta * max(0, radii[i] - (1 - centers[i, 0]))
            grad_size[i + n + 2] += 2 * beta * max(0, radii[i] - centers[i, 1])
            grad_size[i + n + 3] += 2 * beta * max(0, radii[i] - (1 - centers[i, 1]))
    
    return grad_size

def run_packing():
    n = 26
    radii_initial = np.ones(n) * 1e-3
    centers_initial = np.zeros((n, 2))
    for i in range(5):
        for j in range(5):
            centers_initial[i*5 + j, 0] = (i + 0.5 * (i % 2)) * 0.2
            centers_initial[i*5 + j, 1] = j * 0.2 + np.random.uniform(-0.005, 0.005)
    
    np.random.shuffle(centers_initial[:, 1])
    centers_initial[:, 1] = np.clip(centers_initial[:, 1], 0, 1)
    
    initial_state = np.concatenate([centers_initial.flat, radii_initial])
    bounds = [(0, 1)] * (2 * n) + [(0, None)] * n
    
    result = minimize(objective, initial_state, jac=gradient, bounds=bounds, method='L-BFGS-B', options={'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': 2000})
    
    final_centers = result['x'][:2*n].reshape((-1, 2))
    final_radii = np.maximum(result['x'][2*n:], 0)
    final_sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum_radii
