import numpy as np
from scipy.optimize import minimize, Bounds

def lp_sumsuffix(vars):
    n = 26
    centres, radii = vars[:52].reshape(n, 2), vars[52:]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    
    lp_matrix = np.zeros((n, n))
   	lp_matrix[np.triu_indices(n, k=1)] = -radii
   	lp_matrix += lp_matrix.T
   	lp_matrix[np.diag_indices(n)] += 2 * (1 - np.clip(centres[:, 0], 0, 1) ** 2)
    
   	lp_rhs = np.max(np.hstack([np.clip(centres[:, 0], 0, 1) - 1, np.clip(centres[:, 1], 0, 1) - 1]), axis=1)
    
    res = linprog(np.ones(n), A_ub=lp_matrix, b_ub=lp_rhs)
    return res.x, -res.fun

def objective(vars):
    centres, radii = vars[:52].reshape(26, 2), vars[52:]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    overlap = dists < radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-12
    
    return -np.sum(radii) + 1e-6 * np.sum(overlap)

def constraint(vars):
    centres, radii = vars[:52].reshape(26, 2), vars[52:]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    overlap = dists < radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-12
    return np.sum(overlap)

def run_packing():
    num_starts = 3
    max_iter = 200
    
    best_results = []
    
    for _ in range(num_starts):
        init_centres = (np.array([[i*4/25, j*4/25] for j in range(7) for i in range(7)]).flatten() + 0.1 * np.random.randn(52))
        init_radii = np.ones(26)
        
        initial_guess = np.hstack([init_centres, init_radii])
        bounds = Bounds(lower=np.hstack([np.zeros(52), np.zeros(26)]), upper=np.hstack([np.ones(52), np.ones(26)]))
        
        constraints = {'type': 'ineq', 'fun': constraint}
        
        result = minimize(objective, initial_guess, bounds=bounds, methods={'trust-constr'}, constraints=constraints, jac=True, options={'maxiter': max_iter})
        
        for _ in range(5):
            result = minimize(objective, result.x, bounds=bounds, methods={'trust-constr'}, constraints=constraints, jac=True, options={'maxiter': max_iter})
        centres, radii = lp_sumsuffix(result.x)
        s = np.sum(radii)
        best_results.append((centres, radii, s))
    
    bests = max(best_results, key=lambda x: x[2])

    return bests
