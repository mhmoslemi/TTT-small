def solve_center_soc(radii):
    n = len(radii)
    centers = cp.Variable((n, 2))
    
    pairwise_dist_matrix = cp.Parameter((n, n), nonneg=True)
    constraints = [
        pairwise_dist_matrix[i, j] >= radii[i] + radii[j] for i in range(n) for j in range(i + 1, n)
    ]
    constraints += [
        centers[:, 0] <= 1 - radii,
        centers[:, 1] <= 1 - radii,
        centers[:, 0] >= radii,
        centers[:, 1] >= radii
    ]
    
    objective = cp.Minimize(cp.sum_squares(centers - np.mean(centers, axis=0)))
    prob = cp.Problem(objective, constraints)
    pairwise_dist_matrix.value = np.eye(n)
    prob.solve()
    return centers.value, prob.value
