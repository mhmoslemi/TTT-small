import numpy as np
from scipy.optimize import linprog
from scipy.stats import norm

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_grid_plus_center():
    grid_size = 4
    step = 0.25
    center = np.array([0.5, 0.5])
    grid = np.array([
        [i*step + np.random.normal(0, 0.03) for i in range(grid_size)], 
        [j*step + np.random.normal(0, 0.03) for j in range(grid_size)]
    ])
    central_grid = grid.flatten()
    central_grid = central_grid.reshape(grid_size, grid_size)
    central_grid = central_grid.reshape(-1, 2)
    central_grid = np.vstack((central_grid, center))
    return central_grid

def gaussian_perturb(centers, sigma):
    return np.clip(centers + norm.rvs(scale=sigma, size=centers.shape), 0, 1)

def run_packing():
    n = 26
    np.random.seed(42)
    best_sum = -np.inf
    best_centers, best_radii = None, None

    N_restart = 40
    N_iter = 4000
    step_min = 1e-4
    step, temp = 0.15, 0.2

    for restart in range(N_restart):
        centers = initial_grid_plus_center()
        centers = gaussian_perturb(centers, 0.02)

        for it in range(N_iter):
            k = np.random.randint(n)
            new_centers = centers.copy()
            new_centers[k] = np.clip(new_centers[k] + norm.rvs(scale=step), 0, 1)

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii
                step = max(step_min, step * 0.995)
                temp = temp * 0.995

    centers = best_centers
    radii, sum_radii = lp_optimum(centers)

    return centers, np.maximum(radii, 0), sum_radii
