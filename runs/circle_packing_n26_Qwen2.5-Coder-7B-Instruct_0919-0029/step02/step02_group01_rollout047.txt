import numpy as np
from scipy.optimize import differential_evolution, NonlinearConstraint, Bounds, minimize, trust_constr

def objective_and_penalty(p):
    x, y, r = p[:26], p[26:52], p[52:]
    penalty = 0
    for i in range(26):
        if x[i] < 0 or x[i] > 1 or y[i] < 0 or y[i] > 1:
            penalty += 1e8
        if r[i] <= 0 or r[i] > 0.5:
            penalty += 1e8
        for j in range(i + 1, 26):
            dx, dy = x[i] - x[j], y[i] - y[j]
            dist = np.sqrt(dx**2 + dy**2)
            penalty += max(0, (r[i] + r[j] - dist))**2
    return -np.sum(r), penalty

def run_optimization():
    n = 26
    initial_guess = np.tile(np.array([0.4, 0.4, 0.0999]), (n, 1))
    initial_guess.ravel()

    bounds = Bounds(np.zeros(78), np.array([1, 1, 0.5]) * np.ones(78))

    def constraint(p):
        x, y, r = p[:26], p[26:52], p[52:]
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: 0.5 - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: x[i] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - x[i] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: y[i] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - y[i] - r[i]})
            for j in range(i + 1, n):
                dx, dy = x[i] - x[j], y[i] - y[j]
                dist = np.sqrt(dx**2 + dy**2)
                cons.append({'type': 'ineq', 'fun': lambda p, i=i, j=j: dist - r[i] - r[j]})
        return cons

    # Differential Evolution Phase
    de_result = differential_evolution(
        objective_and_penalty, bounds=bounds, popsize=15, maxiter=400
    )

    # Trust-Region Phase
    tr_bounds = Bounds(np.zeros(78), np.array([1, 1, 0.5]) * np.ones(78))
    tr_constraints = constraint(de_result.x)
    tr_result = minimize(
        objective_and_penalty, de_result.x, bounds=tr_bounds, constraints=tr_constraints, method='trust-constr', jac=True
    )

    return tr_result.x[:n], tr_result.x[n:52], tr_result.x[52:], -tr_result.fun

centers, radii, sum_radii = run_optimization()
sum_radii = round(sum_radii, 10)
print(sum_radii)
