import numpy as np
from scipy.optimize import minimize
import itertools

def sum_radii_neg(xs):
    n = 26
    centres = xs[:n * 2].reshape((n, 2))
    radii = xs[n * 2:]
    return -(np.sum(radii))

def centre_bounds(n):
    return [(0.0, 1.0)] * (2 * n)

def centre_and_radius_bounds(n):
    return centre_bounds(n) + [(0.0, 1.0)] * n

def radius_non_overlap_cons(xs):
    n = 26
    centres = xs[:n * 2].reshape((n, 2))
    radii = xs[n * 2:]
    dists = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)
    cons = {}
    for (i, j) in itertools.combinations(range(n), 2):
        cons[(i, j)] = dists[i, j] - radii[i] - radii[j] >= 0
    return [cons[(i, j)] for (i, j) in cons]

def non_overlap_cons(xs):
    return sum(radius_non_overlap_cons(xs))

def boundary_cons(xs):
    n = 26
    centres = xs[:n * 2].reshape((n, 2))
    radii = xs[n * 2:]
    cons = [np.all(centres[:, 0] - radii >= 0),
            np.all(centres[:, 0] + radii <= 1),
            np.all(centres[:, 1] - radii >= 0),
            np.all(centres[:, 1] + radii <= 1)]
    return sum(cons)

def objective(xs):
    return sum_radii_neg(xs) - boundary_cons(xs) - non_overlap_cons(xs)

def initial_guess(n):
    side_length = np.sqrt(26 / np.pi)
    radius = side_length / (2 * np.sqrt(3))
    hex_radius = np.sqrt(3) * radius / 2
    centre_x, centre_y = 1 / (2 * np.sqrt(3)), 1 / 3
    points = [(centre_x + 2 * radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * radius * np.sin(2 * np.pi * i / 3))
              for i in range(3)]
    centre_x, centre_y = 1 / 2, 1 / 2
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 1.5 / 2, 2 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = -1 / 2, -1 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 0.5 / 2, -2 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 1 / (2 * np.sqrt(3)), -1 / 3
    points += [(centre_x + 2 * radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 1 / 2, -1 / 2
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 1.5 / 2, -2 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = -1 / 2, 2 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    centre_x, centre_y = 0.5 / 2, 2 / 3
    points += [(centre_x + 2 * hex_radius * np.cos(2 * np.pi * i / 3), centre_y + 2 * hex_radius * np.sin(2 * np.pi * i / 3))
               for i in range(3)]
    flat_points = np.array(points)
    flat_points = np.concatenate((flat_points, flat_points[:, 2] * np.ones((n, 1))))
    return flat_points.flatten() + 0.01 * np.random.randn(2 * n + n)

def run_packing():
    n = 26
    bounds = centre_and_radius_bounds(n)
    centres = initial_guess(n)
    best_value = np.inf
    best_soln = np.array([])
    for seed in range(3):
        np.random.seed(seed)
        soln = minimize(objective, centres, method='trust-constr', bounds=bounds, options={'maxiter': 200, 'disp': False})
        value = -soln.fun + boundary_cons(soln.x) + non_overlap_cons(soln.x)
        if value < best_value:
            best_value = value
            best_soln = soln.x
    centres = best_soln[:n * 2].reshape((n, 2))
    radii = best_soln[n * 2:]
    return centres, radii, np.sum(radii)
