import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij - 1e-12
            idx += 1

    for i in range(n):
        r = centers[i, 0]
        A_ub[idx, i] = 1
        b_ub[idx] = r - 1e-12
        idx += 1
        r = 1 - centers[i, 0]
        A_ub[idx, i] = 1
        b_ub[idx] = r - 1e-12
        idx += 1
        r = centers[i, 1]
        A_ub[idx, i] = 1
        b_ub[idx] = r - 1e-12
        idx += 1
        r = 1 - centers[i, 1]
        A_ub[idx, i] = 1
        b_ub[idx] = r - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x, -result.fun

def dual_gradient_descent(centers, step=0.1, step_min=1e-6, max_steps=300, accept_thresh=1e-9):
    n = centers.shape[0]
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(30):  # 30 stars instead of 30 as suggested previously
        centres = np.random.rand(n, 2)
        centres += 0.005 * np.random.randn(n, 2)  # Add small random jitter

        for _ in range(max_steps):
            radii, current_sum = lp_optimum(centres)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centres
                best_radii = radii

            grad_x = np.zeros(2 * n)
            grad_y = np.zeros(2 * n)

            for i in range(n):
                row = 0
                for j in range(n):
                    if j <= i:
                        continue

                    d_ij = np.linalg.norm(centres[i] - centres[j])
                    if d_ij < 1e-12:
                        continue

                    grad_x[2 * i] += radii[j] * (centres[i, 0] - centres[j, 0]) / d_ij
                    grad_x[2 * j] -= radii[i] * (centres[i, 0] - centres[j, 0]) / d_ij

                    grad_y[2 * i] += radii[j] * (centres[i, 1] - centres[j, 1]) / d_ij
                    grad_y[2 * j] -= radii[i] * (centres[i, 1] - centres[j, 1]) / d_ij

                mu_l = np.where(centres[i, 0] - radii[i], radii[i], 0)
                mu_r = np.where(1 - centres[i, 0] - radii[i], radii[i], 0)
                mu_b = np.where(centres[i, 1] - radii[i], radii[i], 0)
                mu_t = np.where(1 - centres[i, 1] - radii[i], radii[i], 0)

                grad_x[2 * i] += mu_l - mu_r
                grad_y[2 * i] += mu_b - mu_t

            step_modifier = 1.0
            for _ in range(100):
                dp_x = grad_x * step_modifier
                dp_y = grad_y * step_modifier

                new_centres = centres + np.vstack([dp_x, dp_y]).T
                new_centres = np.clip(new_centres, 0, 1)

                new_radii, new_sum = lp_optimum(new_centres)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centres
                    best_radii = new_radii
                    step += 0.05  # Increase step size if improvement is made
                else:
                    step *= 0.3  # Reduce step size if no improvement
                    step_modifier /= 2  # Reduce step modifier if rejection occurs
                    if step_modifier < 1e-6:
                        step_modifier = 1e-6
                    if step < step_min:
                        step = step_min
                    break

    return best_centers, best_radii, best_sum - max_steps * 0.25

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centres, radii, sum_radii = dual_gradient_descent(np.random.rand(26, 2), step=0.1)
    return centres, radii, sum_radii
