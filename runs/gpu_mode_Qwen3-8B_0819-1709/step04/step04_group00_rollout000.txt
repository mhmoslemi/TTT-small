import torch
import triton
import triton.language as tl
import math
import sys

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen, bs = input_tensor.shape[1], input_tensor.shape[0]
    output = torch.zeros(bs, seqlen, seqlen, dim, device=input_tensor.device, dtype=input_tensor.dtype)

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        x_ptr, mask_ptr, weights, output_ptr,
        B, N, C, H, mask_pad,
        BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr
    ):
        pid = tl.program_id(axis=0)
        b = pid // (N * N)
        n = (pid // N) % N
        m = pid % N
        block_b = b % BLOCK_B
        block_n = n % BLOCK_N
        block_m = m % BLOCK_M

        # Load inputs
        x_row = tl.load(x_ptr + (b * N * N + n * N + m) * C, mask=tl.full((C,), True, dtype=tl.int32))
        mask_val = tl.load(mask_ptr + (b * N * N + n * N + m), mask=tl.full((1,), True, dtype=tl.int32))
        if mask_val == 0:
            return

        # Initialize output for this block
        acc = tl.zeros((BLOCK_C,), dtype=tl.float32)

        # Iterate over all k's
        for k in range(N):
            # Load inputs for k
            x_k = tl.load(x_ptr + (b * N * N + k * N + m) * C, mask=tl.full((C,), True, dtype=tl.int32))
            mask_k = tl.load(mask_ptr + (b * N * N + k * N + m), mask=tl.full((1,), True, dtype=tl.int32))
            if mask_k == 0:
                continue

            # Compute the dot product
            dot = tl.dot(x_row, x_k, bias=0.0)
            acc += dot

        # Store the result
        tl.store(output_ptr + (b * N * N + m * N + n) * C, acc)

    # Launch kernel
    grid = (B * N * N,)
    tri_mul_kernel[grid](input_tensor, mask, weights, output, B, N, dim, hidden_dim, 1, BLOCK_B=1, BLOCK_N=1, BLOCK_C=dim)
    
    return output

import torch
import triton
import triton.language as tl
import math
import sys
import numpy as np

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen, bs = input_tensor.shape[1], input_tensor.shape[0]
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Apply LayerNorm to input tensor
    x = input_tensor
    x = x * norm_weight + norm_bias

    # Compute projections and gates
    left = x @ left_proj_weight.t()
    right = x @ right_proj_weight.t()
    left_gate = x @ left_gate_weight.t()
    right_gate = x @ right_gate_weight.t()
    out_gate = x @ out_gate_weight.t()

    # Apply masks
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul using einsum
    out = torch.einsum('bikd, bjdk -> bijd', left, right)

    # Apply LayerNorm to output
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply final gate and output projection
    out_gate = torch.sigmoid(out_gate)
    out = out * out_gate
    out = out @ to_out_weight.t()

    return out
