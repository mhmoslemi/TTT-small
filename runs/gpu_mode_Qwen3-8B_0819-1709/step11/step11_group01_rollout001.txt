import triton
import triton.language as tl
import torch

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    B, N, _, _ = input_tensor.shape
    num_block = 4  # 4 blocks for better utilization on H100

    # Access weights
    # LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # LayerNorm (output)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Final Linear
    to_out_weight = weights["to_out.weight"]

    # Reshape to [B, N, N, C] -> [B, N, N, C] (no change)
    input_tensor = input_tensor.contiguous()
    mask = mask.unsqueeze(-1).contiguous()

    # Perform the TriMul
    # Step 1: LayerNorm on input
    def layer_norm(input, weight, bias):
        input = input.view(-1, dim)
        mean = torch.mean(input, dim=1, keepdim=True)
        var = torch.var(input, dim=1, keepdim=True)
        input = (input - mean) * torch.rsqrt(var + 1e-6)
        input = input * weight + bias
        return input.view(B, N, N, dim)

    x = layer_norm(input_tensor, norm_weight, norm_bias)

    # Step 2: Project to hidden_dim
    def linear(input, weight):
        return input @ weight.t()

    left = linear(x, left_proj_weight)
    right = linear(x, right_proj_weight)
    left_gate = linear(x, left_gate_weight)
    right_gate = linear(x, right_gate_weight)
    out_gate = linear(x, out_gate_weight)

    # Apply mask and gates
    left = left * mask
    right = right * mask
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Step 3: Compute the TriMul (out = left @ right^T)
    # Note: The einsum is equivalent to i,j,k -> i,j
    # So we'll compute (B, N, N, hidden_dim) @ (B, N, N, hidden_dim) -> (B, N, N, hidden_dim)

    # Create output
    output = torch.zeros(B, N, N, hidden_dim, device=input_tensor.device)

    # Define triton kernel
    @triton.jit
    def triton_trimul_kernel(
        x_ptr, y_ptr, out_ptr,
        B, N, H, 
        stride_xb, stride_xn, stride_xc,
        stride_yb, stride_yn, stride_yc,
        stride_outb, stride_outn, stride_outc,
        BLOCK_SIZE: tl.constexpr,
    ):
        # Each thread block computes a block of output
        pid = tl.program_id(axis=0)
        # Iterate over batch dimension
        b = pid // (N * N)
        # Remaining after batch
        remaining = pid % (N * N)
        # Iterate over output indices
        i = remaining // N
        j = remaining % N

        # Offset into the output tensor
        out_pos = b * stride_outb + i * stride_outn + j * stride_outc
        out_ptr = out_ptr + out_pos

        # Each thread block computes H elements of the output
        # We'll compute H elements in a loop
        for k in range(0, H, BLOCK_SIZE):
            # Block index in H dimension
            block_k = k // BLOCK_SIZE
            # Offset in H dimension
            offset_k = k % BLOCK_SIZE

            # Load x and y data
            # x is (B, N, N, H), y is (B, N, N, H)
            # x is accessed as [b][i][k][...] and y is [b][j][k][...]
            # So, we have to loop over the H dimension
            # For each block, we need to load a chunk of data
            # We'll need to loop over k for the H dimension
            # This is the main loop
            # To avoid memory issues, we'll load data in blocks of size BLOCK_SIZE
            # and compute the sum for each k in [block_k * BLOCK_SIZE : block_k * BLOCK_SIZE + BLOCK_SIZE)

            # For this example, we're using a block size of 64 for H
            # We'll load x and y in chunks of 64 and compute the dot product
            # The following is a simplified version
            # For each position in the H dimension, compute the dot product
            # This is the main loop for the H dimension

            # We can implement this efficiently using triton
            # and the fact that the H dimension is small
            # For each k in 0 to H-1
            # Compute out[i][j] += x[b][i][k] * y[b][j][k]
            # But for large H, we need to use a block-based approach
            # We'll use a block size of 64
            # So, we'll loop over k in chunks of 64
            # and compute the dot product for each chunk

            # This is the main loop

            # Now, let's implement the actual computation
            # For each k in 0 to H-1, compute the dot product
            # We'll use a block size of 64 and compute in blocks
            # We'll loop over k and accumulate the result

            # This is the main computation loop
            # We'll use the fact that the H dimension is small (up to 384)
            # and use a block size of 16 or 32
            # This is the main part of the kernel
            # We'll compute the dot product using the triton dot operator
            # The code below uses the triton dot

            # We'll load x and y data
            # for x, it's [b][i][k][...], for y it's [b][j][k][...]
            # So, we load x as a vector and y as a vector

            # Load x and y
            x = tl.load(x_ptr, tl.bytes(BLOCK_SIZE * H))
            y = tl.load(y_ptr, tl.bytes(BLOCK_SIZE * H))
            # Compute the dot product
            out = tl.dot(x, y)
            # Store the result
            tl.store(out_ptr, out)

        # End of the main loop
        return

    # Launch the kernel
    grid = (B * N * N, )
    triton_trimul_kernel[grid](left, right, output, B, N, hidden_dim, 
                               left.stride(0), left.stride(1), left.stride(2),
                               right.stride(0), right.stride(1), right.stride(2),
                               output.stride(0), output.stride(1), output.stride(2),
                               BLOCK_SIZE=64)

    # Step 4: Apply to_out_norm (LayerNorm on output)
    out = output.view(-1, hidden_dim)
    mean = torch.mean(out, dim=1, keepdim=True)
    var = torch.var(out, dim=1, keepdim=True)
    out = (out - mean) * torch.rsqrt(var + 1e-6)
    out = out * to_out_norm_weight + to_out_norm_bias
    out = out.view(B, N, N, hidden_dim)

    # Step 5: Apply out_gate
    out = out * out_gate.sigmoid()

    # Step 6: Final Linear
    out = out @ to_out_weight.t()

    # Ensure output is in float32
    out = out.to(torch.float32)

    return out
