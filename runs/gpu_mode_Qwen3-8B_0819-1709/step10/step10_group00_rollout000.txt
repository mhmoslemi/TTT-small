import triton
import triton.language as tl
import torch
import math
import sys
import os
import time

# This kernel implements the 'outgoing' version of the TriMul operator
# from AlphaFold3, optimized using Triton for performance on NVIDIA H100 (sm_90)
# The kernel processes a 4D tensor [B, N, N, C] with mask and model weights

@triton.jit
def trit_mul_kernel(
    input_ptr, mask_ptr, out_ptr,
    norm_weight, norm_bias,
    left_proj_weight, right_proj_weight,
    left_gate_weight, right_gate_weight, out_gate_weight,
    to_out_norm_weight, to_out_norm_bias,
    to_out_weight,
    B, N, K, C, H, 
    stride_input_n, stride_input_k, stride_input_c,
    stride_mask_n, stride_mask_k,
    stride_out_n, stride_out_k, stride_out_c,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr, 
    NUM_THREADS: tl.constexpr, 
    NUM_BLOCKS: tl.constexpr
):
    # Each thread handles a block of (block_n, block_k, block_c)
    pid = tl.program_id(axis=0)
    block_n = pid % NUM_BLOCKS
    block_k = (pid // NUM_BLOCKS) % NUM_BLOCKS
    block_c = (pid // (NUM_BLOCKS * NUM_BLOCKS)) % NUM_BLOCKS
    
    # Compute the offsets for the current block
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = block_k * BLOCK_K + tl.arange(0, BLOCK_K)
    offs_c = block_c * BLOCK_C + tl.arange(0, BLOCK_C)
    
    # Load the mask into shared memory for faster access
    mask = tl.load(mask_ptr + offs_n[:, None] * stride_mask_n + offs_k[None, :], mask=tl.full((BLOCK_N, BLOCK_K), True, dtype=tl.bool))
    mask = tl.where(mask, 1.0, 0.0)
    
    # Load input and weights for left and right projections
    input_left = tl.load(input_ptr + offs_n[:, None] * stride_input_n + offs_k[None, :] * stride_input_k + offs_c[None, :] * stride_input_c, 
                         mask=tl.full((BLOCK_N, BLOCK_K, BLOCK_C), True, dtype=tl.bool))
    input_right = tl.load(input_ptr + offs_n[:, None] * stride_input_n + offs_k[None, :] * stride_input_k + offs_c[None, :] * stride_input_c, 
                          mask=tl.full((BLOCK_N, BLOCK_K, BLOCK_C), True, dtype=tl.bool))
    
    # Apply weights for left and right projections
    input_left = input_left * left_proj_weight[offs_c, offs_k]
    input_right = input_right * right_proj_weight[offs_c, offs_k]
    
    # Apply masks to left and right projections
    input_left = input_left * mask
    input_right = input_right * mask
    
    # Apply gate matrices (simplified with sigmoid)
    # Note: The full sigmoid is not implemented as it's a nonlinear operation; 
    # this is an approximation for the purpose of the kernel
    gate_left = tl.sigmoid(input_left)
    gate_right = tl.sigmoid(input_right)
    gate_out = tl.sigmoid(input_left)  # Simplified for kernel
    
    # Apply gates
    input_left = input_left * gate_left
    input_right = input_right * gate_right
    
    # Compute the TriMul operation
    # This is a contraction of (block_n, block_k, block_c) x (block_n, block_k, block_c)
    # Resulting in a (block_n, block_n, block_c) tensor
    
    # Initialize output accumulator
    acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_C), dtype=tl.float32)
    
    # Perform the TriMul operation with nested loops over the k dimension
    for k in range(BLOCK_K):
        input_left_k = input_left[offs_n, k, offs_c]
        input_right_k = input_right[offs_n, k, offs_c]
        acc = acc + tl.dot(input_left_k, input_right_k)
    
    # Apply LayerNorm weights
    acc = (acc - tl.reduce(acc, axis=(0,1), keepdims=True)) * to_out_norm_weight[offs_c] + to_out_norm_bias[offs_c]
    
    # Apply output gate
    acc = acc * gate_out
    
    # Write the result back to output tensor
    tl.store(out_ptr + offs_n[:, None] * stride_out_n + offs_n[None, :] * stride_out_k + offs_c[None, :] * stride_out_c,
             acc, mask=tl.full((BLOCK_N, BLOCK_N, BLOCK_C), True, dtype=tl.bool))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B = config["bs"]
    N = config["seqlen"]
    C = config["dim"]
    H = config["hidden_dim"]

    # Ensure input and mask are on the correct device
    input_tensor = input_tensor.contiguous()
    mask = mask.contiguous()
    
    # Get the input strides
    stride_input_n = input_tensor.stride(1)
    stride_input_k = input_tensor.stride(2)
    stride_input_c = input_tensor.stride(3)

    # Get mask strides
    stride_mask_n = mask.stride(1)
    stride_mask_k = mask.stride(2)

    # Allocate output tensor
    out = torch.empty((B, N, N, C), device=input_tensor.device, dtype=input_tensor.dtype)

    # Get output strides
    stride_out_n = out.stride(1)
    stride_out_k = out.stride(2)
    stride_out_c = out.stride(3)

    # Get the weights from the weights dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute the number of blocks and thread configuration
    # We use a block size of 128 for N, K, C
    # Adjusted for the number of threads and blocks to fit within shared memory and registers
    BLOCK_N = 128
    BLOCK_K = 128
    BLOCK_C = 128
    NUM_BLOCKS = (N + BLOCK_N - 1) // BLOCK_N
    NUM_THREADS = 256

    # Launch the kernel
    grid = (NUM_BLOCKS * NUM_BLOCKS * NUM_BLOCKS,)
    trit_mul_kernel[grid](
        input_tensor, mask, out,
        norm_weight, norm_bias,
        left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias,
        to_out_weight,
        B, N, K, C, H,
        stride_input_n, stride_input_k, stride_input_c,
        stride_mask_n, stride_mask_k,
        stride_out_n, stride_out_k, stride_out_c,
        BLOCK_N, BLOCK_K, BLOCK_C, 
        NUM_THREADS, 
        NUM_BLOCKS
    )
    
    return out
