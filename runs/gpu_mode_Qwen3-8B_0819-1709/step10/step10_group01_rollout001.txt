import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Implements the TriMul operator from AlphaFold3, optimized using Triton for large-scale protein structure prediction.

    This kernel performs the following steps:
    1. Applies LayerNorm to input tensor
    2. Projects input through linear layers for left and right components
    3. Applies sigmoid gates to these components
    4. Performs a pairwise multiplication and sum operation (TriMul)
    5. Applies a second LayerNorm and final projection

    The operation is memory efficient and optimized for large sequence lengths up to 1024.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights, config)
    Returns:
        output: [bs, seq_len, seq_len, dim] tensor
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Access weights (assuming they are preloaded on the GPU)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure mask is on the same device as input_tensor
    mask = mask.to(input_tensor.device)

    # Precompute mask-related parameters for fast access
    mask = mask.unsqueeze(-1)  # shape [bs, seq_len, seq_len, 1]

    # Define kernel
    @triton.jit
    def tr_mul_kernel(
        input_ptr, left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        norm_weight_ptr, norm_bias_ptr,
        left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr, right_gate_weight_ptr,
        out_gate_weight_ptr, to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        mask_ptr,
        output_ptr,
        bs, seq_len, dim, hidden_dim, num_blocks,
        BLOCK_SIZE: tl.constexpr, NUM_BLOCK_M: tl.constexpr, NUM_BLOCK_N: tl.constexpr
    ):
        # Get the block index and thread index
        pid = tl.program_id(axis=0)
        block_idx = pid // (seq_len * seq_len)
        i = (pid // seq_len) % seq_len
        j = pid % seq_len

        # Load the input tensor for this (i, j) pair
        input = tl.load(input_ptr + tl.arange(0, BLOCK_SIZE), axis=0)

        # Multiply with mask
        mask_val = tl.load(mask_ptr + block_idx * seq_len * seq_len + i * seq_len + j)
        input = input * mask_val

        # Apply norm
        norm_input = input * norm_weight_ptr + norm_bias_ptr
        norm_input = tl.where(tl.isnan(norm_input), norm_input, norm_input)

        # Apply projection and gate operations
        # Left projection
        left = tl.dot(norm_input, left_proj_weight_ptr, axis=1)
        left = left * left_gate_ptr
        left = left * mask_val

        # Right projection
        right = tl.dot(norm_input, right_proj_weight_ptr, axis=1)
        right = right * right_gate_ptr
        right = right * mask_val

        # Compute TriMul sum over k
        out = tl.zeros((BLOCK_SIZE, ), dtype=tl.float32)
        for k in range(seq_len):
            a = tl.load(left_ptr + k * seq_len * seq_len + i * seq_len + j)
            b = tl.load(right_ptr + k * seq_len * seq_len + j * seq_len + i)
            out += a * b

        # Apply gate
        out_gate = tl.dot(norm_input, out_gate_weight_ptr, axis=1)
        out_gate = out_gate * mask_val
        out = out * out_gate

        # Apply second norm
        out = out * to_out_norm_weight_ptr + to_out_norm_bias_ptr

        # Apply final projection
        output = tl.dot(out, to_out_weight_ptr, axis=1)

        # Write output
        tl.store(output_ptr + block_idx * seq_len * seq_len + i * seq_len + j, output)

    # Define kernel parameters
    num_blocks = seq_len * seq_len
    BLOCK_SIZE = 128
    NUM_BLOCK_M = 1
    NUM_BLOCK_N = 1

    # Create buffers for all tensors in the kernel
    input_buffer = input_tensor.view(bs * seq_len * seq_len, dim)
    left_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    right_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    left_gate_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    right_gate_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    out_gate_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    to_out_norm_buffer = torch.zeros(bs * seq_len * seq_len, hidden_dim, device=input_tensor.device)
    to_out_buffer = torch.zeros(bs * seq_len * seq_len, dim, device=input_tensor.device)

    # Execute the kernel
    tr_mul_kernel(
        input_tensor.data_ptr(), left_buffer.data_ptr(), right_buffer.data_ptr(),
        left_gate_buffer.data_ptr(), right_gate_buffer.data_ptr(), out_gate_buffer.data_ptr(),
        to_out_norm_buffer.data_ptr(), to_out_buffer.data_ptr(),
        norm_weight.data_ptr(), norm_bias.data_ptr(),
        left_proj_weight.data_ptr(), right_proj_weight.data_ptr(), left_gate_weight.data_ptr(),
        right_gate_weight.data_ptr(), out_gate_weight.data_ptr(), to_out_norm_weight.data_ptr(),
        to_out_norm_bias.data_ptr(), to_out_weight.data_ptr(),
        mask.data_ptr(),
        to_out_buffer.data_ptr(),
        bs, seq_len, dim, hidden_dim, num_blocks,
        BLOCK_SIZE, NUM_BLOCK_M, NUM_BLOCK_N
    )

    # Reshape output and return
    output = to_out_buffer.view(bs, seq_len, seq_len, dim)
    return output
