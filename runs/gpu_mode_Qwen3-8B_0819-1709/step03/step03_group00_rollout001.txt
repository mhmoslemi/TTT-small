import triton
import triton.language as tl
import torch

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Input and output dimensions
    B, N, _, D = input_tensor.shape
    output_shape = (B, N, N, dim)
    output = torch.empty(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # Main kernel
    @triton.jit
    def tril_mul_kernel(
        input_ptr, mask_ptr, weights, output_ptr,
        B, N, D, hidden_dim,
        stride_in_b, stride_in_n, stride_in_c,
        stride_mask_b, stride_mask_n, stride_mask_c,
        stride_out_b, stride_out_n, stride_out_c,
        BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_B: tl.constexpr
    ):
        # Compute the thread position
        pid = tl.program_id(axis=0)

        # Compute the batch and output indices
        batch_idx = pid // (N * N)
        output_idx = pid % (N * N)
        i = output_idx // N
        j = output_idx % N

        # Compute the input indices
        # The input is [B, N, N, D]
        # We loop over all k in [0, N)
        k = tl.arange(0, N)
        k_offsets = k + tl.arange(0, BLOCK_N)[:, None] * N

        # Load the mask
        mask = tl.load(mask_ptr + batch_idx * stride_mask_b + i * stride_mask_n + j * stride_mask_c)
        if mask != 1.0:
            return

        # Load the input tensor
        input = tl.load(input_ptr + batch_idx * stride_in_b + i * stride_in_n + k_offsets * stride_in_c)
        input = input * mask

        # Apply the linear projections
        # left = self.left_proj(x) -> [B, N, N, hidden_dim]
        # right = self.right_proj(x) -> [B, N, N, hidden_dim]
        # We use the following to compute left and right
        # left = (input * left_proj_weight).sum(dim=-1)
        # right = (input * right_proj_weight).sum(dim=-1)

        # Compute left
        left = tl.dot(input, left_proj_weight.T)  # [N, hidden_dim]
        left = left + tl.zeros((N, hidden_dim), dtype=tl.float32)
        # Compute right
        right = tl.dot(input, right_proj_weight.T)  # [N, hidden_dim]
        right = right + tl.zeros((N, hidden_dim), dtype=tl.float32)

        # Apply the gate
        # left_gate = self.left_gate(x).sigmoid()
        # right_gate = self.right_gate(x).sigmoid()
        # out_gate = self.out_gate(x).sigmoid()
        # left = left * left_gate
        # right = right * right_gate

        # Compute left_gate
        left_gate = tl.dot(input, left_gate_weight.T)  # [N, hidden_dim]
        left_gate = left_gate + tl.zeros((N, hidden_dim), dtype=tl.float32)
        left_gate = left_gate.sigmoid()

        # Compute right_gate
        right_gate = tl.dot(input, right_gate_weight.T)  # [N, hidden_dim]
        right_gate = right_gate + tl.zeros((N, hidden_dim), dtype=tl.float32)
        right_gate = right_gate.sigmoid()

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # For each i and j, compute the sum over k of left[i][k] * right[j][k]
        # This is equivalent to left @ right.T (transposed)
        # So we can compute it as a matrix multiplication

        # We perform this in a more efficient way by transposing right
        right_T = right.T
        out = tl.dot(left, right_T)  # [N, N] for each i and j

        # Apply the to_out_norm and to_out
        # out = self.to_out_norm(out)
        # out = out * out_gate
        # return self.to_out(out)

        # Compute to_out_norm (LayerNorm over hidden_dim)
        # mean = out.mean(dim=-1)
        # var = out.var(dim=-1)
        # out = (out - mean) / sqrt(var + 1e-5)
        # out = out * to_out_norm_weight + to_out_norm_bias
        # out = out * out_gate
        # out = out @ to_out_weight.T

        # Compute mean
        mean = out.mean(axis=-1, keepdim=True)
        # Compute variance
        var = (out - mean).pow(2).mean(axis=-1, keepdim=True)
        # Normalize
        out = (out - mean) / (tl.sqrt(var + 1e-5))
        # Apply to_out_norm
        out = out * to_out_norm_weight + to_out_norm_bias
        # Apply out_gate
        out_gate = tl.dot(input, out_gate_weight.T)
        out_gate = out_gate + tl.zeros((N, hidden_dim), dtype=tl.float32)
        out_gate = out_gate.sigmoid()
        out = out * out_gate
        # Output
        out = out @ to_out_weight.T  # [N, N] @ [hidden_dim, dim] = [N, dim]

        # Write the result to the output
        output_idx_2d = i * N + j
        out_flat = out.reshape(N * dim)
        output_ptr += batch_idx * stride_out_b + output_idx_2d * stride_out_c
        tl.store(output_ptr, out_flat)

    # Launch the kernel
    grid = (B * N * N,)
    tril_mul_kernel[
        grid,
        (N,),
        (N, hidden_dim, B,)
    ](
        input_tensor, mask, weights, output,
        B, N, D, hidden_dim,
        input_tensor.stride(0), input_tensor.stride(1), input_tensor.stride(2),
        mask.stride(0), mask.stride(1), mask.stride(2),
        output.stride(0), output.stride(1), output.stride(2),
        256, 128, 2
    )

    return output
