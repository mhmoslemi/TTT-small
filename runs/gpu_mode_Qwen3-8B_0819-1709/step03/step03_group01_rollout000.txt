import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len = input_tensor.shape[0], input_tensor.shape[1]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output shape: [batch_size, seq_len, seq_len, dim]
    output_shape = (batch_size, seq_len, seq_len, dim)
    output = torch.empty(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # LayerNorm
    def layer_norm(x):
        mean = torch.mean(x, dim=-1, keepdim=True)
        variance = torch.var(x, dim=-1, keepdim=True, unbiased=False)
        x_norm = (x - mean) * (variance + 1e-5).sqrt()
        return x_norm

    # Initialize output
    output = layer_norm(input_tensor)
    output = output * mask.unsqueeze(-1)

    # Compute left and right projections
    left = torch.matmul(output, left_proj_weight.t())
    right = torch.matmul(output, right_proj_weight.t())

    # Apply masks
    left = left * mask.unsqueeze(-1)
    right = right * mask.unsqueeze(-1)

    # Apply gates
    left_gate = torch.matmul(output, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(output, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(output, out_gate_weight.t()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute the einsum operation: out = left @ right
    # This is equivalent to out[i][j] = sum_k (left[i][k] * right[j][k])
    # Define kernel for this operation
    @triton.jit
    def tr_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, hidden_dim,
        BLOCK_SEQ: tl.constexpr, BLOCK_DIM: tl.constexpr,
        stride_lbs, stride_lsd, stride_rbs, stride_rsd, stride_obs, stride_osd,
    ):
        # Compute the block index
        pid = tl.program_id(axis=0)
        block_idx = pid
        # Compute the offset for this block
        block_size = BLOCK_SEQ * BLOCK_SEQ
        block_idx_in_batch = block_idx // (seq_len * seq_len)
        block_idx_in_seq = (block_idx % (seq_len * seq_len)) // seq_len
        block_idx_in_seq2 = (block_idx % (seq_len * seq_len)) % seq_len
        
        # Offset in the tensor
        off_bs = block_idx_in_batch
        off_s = block_idx_in_seq
        off_s2 = block_idx_in_seq2
        off_d = tl.arange(0, BLOCK_DIM)

        # Load left and right
        left = tl.load(left_ptr + off_bs * stride_lbs + off_s * stride_lsd + off_d[None, :], 
                      shape=(BLOCK_SEQ, BLOCK_DIM), 
                      mask=off_d[None, :] < hidden_dim)
        right = tl.load(right_ptr + off_bs * stride_rbs + off_s2 * stride_rsd + off_d[None, :], 
                        shape=(BLOCK_SEQ, BLOCK_DIM), 
                        mask=off_d[None, :] < hidden_dim)

        # Compute the matrix multiplication
        acc = tl.dot(left, right)

        # Write back to output
        tl.store(out_ptr + off_bs * stride_obs + off_s * stride_osd + off_d[None, :], 
                 acc, 
                 mask=off_d[None, :] < hidden_dim)

    # Launch the kernel
    num_block_seq = seq_len
    num_block_dim = (hidden_dim + BLOCK_DIM - 1) // BLOCK_DIM
    grid = (num_block_seq, )
    tr_mul_kernel[grid](
        left, right, output,
        batch_size, seq_len, hidden_dim,
        num_block_seq, num_block_dim,
        left.stride(0), left.stride(1), right.stride(0), right.stride(1), output.stride(0), output.stride(1)
    )

    # Apply output gate and normalization
    output = output * out_gate
    output = torch.matmul(output, to_out_norm_weight.t()) + to_out_norm_bias
    output = torch.matmul(output, to_out_weight.t())

    return output
