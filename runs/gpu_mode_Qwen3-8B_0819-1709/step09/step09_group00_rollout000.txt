import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Custom Triton kernel for the outgoing version of the Triangle Multiplicative Update (TriMul) operator.
    This kernel implements the forward pass of TriMul, with efficient computation using Tensor Cores (wgmma)
    on NVIDIA H100 (sm_90) hardware.
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    
    # Ensure all tensors are on the correct device
    input_tensor = input_tensor.to("cuda")
    mask = mask.to("cuda")
    batch_size, seq_len, _, _ = input_tensor.shape
    
    # Access weights
    norm_weight = weights["norm.weight"].to("cuda")
    norm_bias = weights["norm.bias"].to("cuda")
    left_proj_weight = weights["left_proj.weight"].to("cuda")
    right_proj_weight = weights["right_proj.weight"].to("cuda")
    left_gate_weight = weights["left_gate.weight"].to("cuda")
    right_gate_weight = weights["right_gate.weight"].to("cuda")
    out_gate_weight = weights["out_gate.weight"].to("cuda")
    to_out_norm_weight = weights["to_out_norm.weight"].to("cuda")
    to_out_norm_bias = weights["to_out_norm.bias"].to("cuda")
    to_out_weight = weights["to_out.weight"].to("cuda")

    # Output tensor
    output = torch.zeros(batch_size, seq_len, seq_len, dim, device="cuda", dtype=torch.float32)
    
    # Step 1: Normalize input using LayerNorm
    norm_input = input_tensor
    if dim != 1:
        norm_input = torch.nn.functional.layer_norm(norm_input, (dim,), norm_weight, norm_bias, eps=1e-5)
    
    # Step 2: Compute left and right projections
    left_input = norm_input
    right_input = norm_input
    left_proj = torch.nn.functional.linear(left_input, left_proj_weight, None)
    right_proj = torch.nn.functional.linear(right_input, right_proj_weight, None)
    
    # Apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    
    # Step 3: Compute gates
    left_gate_input = norm_input
    right_gate_input = norm_input
    out_gate_input = norm_input
    
    left_gate = torch.nn.functional.linear(left_gate_input, left_gate_weight, None).sigmoid()
    right_gate = torch.nn.functional.linear(right_gate_input, right_gate_weight, None).sigmoid()
    out_gate = torch.nn.functional.linear(out_gate_input, out_gate_weight, None).sigmoid()
    
    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate
    
    # Step 4: Compute the TriMul operation
    # This is the key operation, which is equivalent to a batched matrix multiplication
    # out[i][j][d] = sum_k (left[i][k][d] * right[j][k][d])
    # We implement this using a Triton kernel for optimized performance
    
    # Allocate output
    out = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device="cuda", dtype=torch.float32)
    
    # Define the Triton kernel
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, D,  # batch, seq_len, hidden_dim
        stride_left_b, stride_left_n, stride_left_d,
        stride_right_b, stride_right_n, stride_right_d,
        stride_out_b, stride_out_n, stride_out_d,
        BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        # Map the thread to the indices
        pid = tl.program_id(axis=0)  # batch
        n = tl.program_id(axis=1)    # sequence i
        k = tl.program_id(axis=2)    # sequence k
        d = tl.program_id(axis=3)    # dimension

        # Calculate the offset for the current thread
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        # Offsets for the left and right projections
        left_offs = (pid * stride_left_b + offs_n[:, None, None] * stride_left_n +
                     offs_k[None, :, None] * stride_left_d +
                     offs_d[None, None, :])
        right_offs = (pid * stride_right_b + offs_k[None, :, None] * stride_right_n +
                      offs_n[:, None, None] * stride_right_d +
                      offs_d[None, None, :])
        # Load the data
        left = tl.load(left_ptr + left_offs)
        right = tl.load(right_ptr + right_offs)
        # Compute the dot product over k
        out = tl.dot(left, right, axis=1)
        # Store the result
        tl.store(out_ptr + (pid * stride_out_b + offs_n[:, None] * stride_out_n + offs_d[None, :]), out)

    # Launch the Triton kernel
    # Choose the best block sizes for performance on H100
    # We use tiling to match the Tensor Core sizes (e.g., 16x16 blocks)
    # Note: We assume that the hidden_dim is small enough for our tiling strategy
    # For this example, we set block sizes as follows:
    block_N = 16
    block_D = 16
    block_K = 16

    # Launch the kernel
    grid = (batch_size, seq_len, seq_len, hidden_dim)
    triton_trimul_kernel[grid](
        left_proj.contiguous(),
        right_proj.contiguous(),
        out,
        batch_size, seq_len, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        block_N, block_D, block_K
    )

    # Step 5: Apply LayerNorm and out_gate
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight, None)

    # Ensure output is in float32
    out = out.float()
    
    return out
