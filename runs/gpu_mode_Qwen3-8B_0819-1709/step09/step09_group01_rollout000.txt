import triton
import triton.language as tl
import torch
import math
import torch.nn.functional as F

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]
    output_shape = (batch_size, seqlen, seqlen, dim)
    output = torch.empty(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # Access weights directly from the dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply LayerNorm to input
    input_norm = (input_tensor - norm_bias[None, None, None, None]) * norm_weight[None, None, None, None]

    # Compute projections and gates
    input_norm = input_norm.contiguous()
    left_proj = input_norm @ left_proj_weight.transpose(0, 1)
    right_proj = input_norm @ right_proj_weight.transpose(0, 1)
    left_gate = input_norm @ left_gate_weight.transpose(0, 1)
    right_gate = input_norm @ right_gate_weight.transpose(0, 1)
    out_gate = input_norm @ out_gate_weight.transpose(0, 1)

    # Apply activation functions and apply mask
    if mask is not None:
        left_proj = left_proj * mask.unsqueeze(-1)
        right_proj = right_proj * mask.unsqueeze(-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Apply the TriMul operation: einsum('... i k d, ... j k d -> ... i j d')
    # This is a O(N^3) operation, must be optimized in Triton

    # Define the Triton kernel
    @triton.jit
    def tri_mul_kernel(
        left_proj_ptr, right_proj_ptr, out_ptr,
        seqlen, batch_size, hidden_dim,
        stride_lk, stride_lr, stride_ly, stride_lz,
        stride_ rk, stride_rr, stride_ry, stride_rz,
        stride_op, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(axis=0)
        off_b = pid * BLOCK
        mask = off_b < batch_size
        if not mask:
            return

        offs_b = tl.arange(0, BLOCK)
        offs_i = tl.arange(0, seqlen)
        offs_j = tl.arange(0, seqlen)

        # Load left_proj[b, i, k, d]
        l_offsets = (offs_b[:, None, None] * stride_lk) + (offs_i[None, :, None] * stride_ly) + (offs_k[None, None, :] * stride_lr)
        left_block = tl.load(left_proj_ptr + l_offsets, mask=offs_k[None, None, :] < seqlen, other=0.0)

        # Compute the output[i][j] for each j
        for j in range(seqlen):
            # Load right_proj[b, j, k, d]
            r_offsets = (offs_b[:, None, None] * stride_rk) + (offs_j[None, None, :] * stride_ry) + (offs_k[None, None, :] * stride_rr)
            right_block = tl.load(right_proj_ptr + r_offsets, mask=offs_k[None, None, :] < seqlen, other=0.0)

            # Compute dot product
            out_block = tl.dot(left_block, right_block, axis=2)

            # Store the result to output[b, i, j, d]
            o_offsets = (offs_b[:, None, None] * stride_op) + (offs_i[None, :, None] * stride_oy) + (offs_j[None, None, :] * stride_oz)
            tl.store(out_ptr + o_offsets, out_block, mask=offs_i[None, :, None] < seqlen)

    # Launch the kernel
    grid = (batch_size, )
    tri_mul_kernel[(grid, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, ), (seqlen, )]

    # Apply LayerNorm and gate to the output
    output = output.contiguous()
    output_norm = (output - to_out_norm_bias[None, None, None, None]) * to_out_norm_weight[None, None, None, None]
    output_norm = output_norm * out_gate
    output = output_norm @ to_out_weight.transpose(0, 1)

    return output
