import triton
import triton.language as tl
import torch
import math
import time

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Extract tensor metadata
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Output initialization
    output = torch.empty((B, N, N, dim), device=device, dtype=dtype)

    # Define the kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        output_ptr,
        B: tl.constexpr, N: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
        mask_val: tl.constexpr,
        BLOCK: tl.constexpr,
    ):
        # Each thread block computes a 2D tile of the output
        pid = tl.program_id(axis=0)
        offset = pid * BLOCK
        i = offset + tl.arange(0, BLOCK)
        j = tl.arange(0, BLOCK)

        # Load mask and apply it
        mask = tl.load(mask_ptr + i * N + j, mask_val)

        # Load input data
        input = tl.load(input_ptr + i * N * dim + j * dim, mask_val, 0)
        input = input.to(tl.float32)

        # Apply LayerNorm
        mean = tl.sum(input) / (N * N)
        var = tl.sum((input - mean) * (input - mean)) / (N * N)
        input = (input - mean) / tl.sqrt(var + 1e-6)
        input = input * norm_weight + norm_bias
        input = input.to(dtype)

        # Compute left and right projections
        left = tl.dot(input, left_proj_weight_ptr)  # [B, N, N, hidden_dim]
        right = tl.dot(input, right_proj_weight_ptr)  # [B, N, N, hidden_dim]

        # Apply mask
        left = left * mask
        right = right * mask

        # Compute gates
        left_gate = tl.dot(input, left_gate_weight_ptr)  # [B, N, N, hidden_dim]
        right_gate = tl.dot(input, right_gate_weight_ptr)  # [B, N, N, hidden_dim]
        out_gate = tl.dot(input, out_gate_weight_ptr)  # [B, N, N, hidden_dim]

        left_gate = left_gate.sigmoid()
        right_gate = right_gate.sigmoid()
        out_gate = out_gate.sigmoid()

        # Apply gates to left and right
        left = left * left_gate
        right = right * right_gate

        # Compute TriMul: einsum('... i k d, ... j k d -> ... i j d')
        acc = tl.zeros((BLOCK, dim), dtype=tl.float32)
        for k in range(N):
            a = tl.load(input_ptr + i * N * dim + k * dim, mask_val, 0)
            b = tl.load(input_ptr + j * N * dim + k * dim, mask_val, 0)
            temp = tl.dot(a, b)  # [N, dim] @ [N, dim] = [dim]
            acc += temp
        acc = acc * out_gate

        # Apply output LayerNorm
        mean_output = tl.sum(acc) / (N * N)
        var_output = tl.sum((acc - mean_output) * (acc - mean_output)) / (N * N)
        acc = (acc - mean_output) / tl.sqrt(var_output + 1e-6)
        acc = acc * to_out_norm_weight + to_out_norm_bias
        acc = acc.to(dtype)

        # Apply output projection
        output = tl.dot(acc, to_out_weight_ptr)  # [N, dim] @ [hidden_dim, dim] = [N, dim]

        # Store output
        output = output.to(tl.float32)
        tl.store(output_ptr + i * N * dim + j * dim, output)

    # Launch the kernel
    num_blocks = (N + BLOCK - 1) // BLOCK
    grid = (num_blocks, )

    tri_mul_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output,
        B, N, dim, hidden_dim,
        1.0,
        256,
    )

    return output
