import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]
    
    # Access the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Output tensor shape: [bs, seqlen, seqlen, dim]
    output_shape = (batch_size, seqlen, seqlen, dim)
    output = torch.empty(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)
    
    # Apply layer norm (only on the feature dimension)
    input_tensor = input_tensor * norm_weight + norm_bias
    
    # Compute projections
    left_proj = torch.matmul(input_tensor, left_proj_weight.t())
    right_proj = torch.matmul(input_tensor, right_proj_weight.t())
    left_gate = torch.matmul(input_tensor, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_tensor, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_tensor, out_gate_weight.t()).sigmoid()
    
    # Apply masks
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    
    # Compute the TriMul operation
    # Note: This is the main computational part and will be implemented as a Triton kernel
    # We'll implement it in the `tritondot_kernel` below
    
    # Compute the TriMul using a Triton kernel
    tritondot_kernel[0, 0](left_proj, right_proj, output, seqlen, seqlen, seqlen, hidden_dim,
                           batch_size, input_tensor.device, input_tensor.dtype)
    
    # Apply LayerNorm and gate
    output = torch.matmul(output, to_out_norm_weight.t()) + to_out_norm_bias
    output = output * out_gate
    
    # Final linear projection
    output = torch.matmul(output, to_out_weight.t())
    
    return output

@triton.jit
def tritondot_kernel(
    left_ptr, right_ptr, out_ptr,
    seqlen, seqlen2, seqlen3, hidden_dim,
    batch_size, device, dtype,
):
    # Set the desired data type
    if dtype == torch.float32:
        dtype = tl.float32
    elif dtype == torch.float16:
        dtype = tl.float16
    else:
        dtype = tl.float32  # default for torch.float64
    
    # Define block sizes
    BLOCK = 64  # adjust this to maximize throughput

    # Compute the total number of tasks
    num_tasks = batch_size * seqlen * seqlen

    # Launch grid of threads
    grid = (num_tasks,)
    offset = tl.program_id(axis=0)
    i = offset // (seqlen * seqlen)
    j = (offset // seqlen) % seqlen
    k = offset % seqlen

    # Load the left and right projections
    # Here, we assume the data is stored in a 4D format: [batch, seq, seq, hidden_dim]
    # We need to compute out[i, j, d] = sum_k left[i, k, d] * right[j, k, d]

    # Use shared memory for faster access
    shared_left = tl.tensor([0.0] * (BLOCK * hidden_dim), dtype=dtype)
    shared_right = tl.tensor([0.0] * (BLOCK * hidden_dim), dtype=dtype)
    
    # Load the current block of left and right
    # Convert i, j, k to pointers in the input tensors
    left_idx = i * seqlen * seqlen * hidden_dim + k * seqlen * hidden_dim + j * hidden_dim
    right_idx = j * seqlen * seqlen * hidden_dim + k * seqlen * hidden_dim + i * hidden_dim

    # Fetch data from global to shared memory
    for d in range(hidden_dim):
        shared_left[d] = tl.load(left_ptr + left_idx + d)
        shared_right[d] = tl.load(right_ptr + right_idx + d)
    
    # Compute the dot product over the hidden dimension
    acc = tl.zeros((1, 1), dtype=dtype)
    for d in range(hidden_dim):
        acc += shared_left[d] * shared_right[d]
    
    # Store the result back to output
    out_idx = i * seqlen * seqlen * hidden_dim + j * seqlen * hidden_dim + k * hidden_dim
    tl.store(out_ptr + out_idx, acc)
