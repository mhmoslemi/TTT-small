import triton.language as tl
import torch
import math

def custom_kernel(data):

    Triton kernel for the outgoing version of the Triangle Multiplicative Update (TriMul) operator.
    
    This implementation computes the TriMul forward pass using a sequence of matrix operations
    involving projections, gating, and pairwise multiplication. The mask is applied to suppress
    invalid entries during the matrix multiplication. The operations are optimized using Triton
    to support high performance on NVIDIA H100 GPUs.
    """

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Constants
    N = seq_len
    B = bs
    C = dim
    H = hidden_dim

    # Output tensor shape
    output = torch.zeros((B, N, N, C), device=input_tensor.device, dtype=input_tensor.dtype)

    # Define kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        output_ptr,
        N: tl.constexpr, B: tl.constexpr, C: tl.constexpr, H: tl.constexpr,
        BLOCK_N: tl.constexpr, BLOCK_B: tl.constexpr, BLOCK_C: tl.constexpr
    ):
        # Compute the batch, i, j, and c indices for this thread
        pid = tl.program_id(axis=0)
        batch_idx = pid // (N * N)
        i = (pid // N) % N
        j = pid % N

        # Compute offsets for the current batch and i,j
        batch_offsets = batch_idx * N * N
        i_offsets = i * N
        j_offsets = j * N

        # Load the mask (with broadcasting to H dimension)
        mask_val = tl.load(mask_ptr + batch_idx + (i * N + j))
        mask = tl.broadcast_to(tl.reshape(tl.broadcast_to(tl.reshape(tl.broadcast_to(mask_val, (1, N, N)), (1, N, N, 1)), (1, 1, N, N, H)), (B, N, N, 1, H)), (B, N, N, H))

        # Load input and apply LayerNorm
        input = tl.load(input_ptr + batch_offsets + i_offsets + tl.arange(0, N))
        input = (input - tl.broadcast_to(norm_bias, (N, N, C))) * tl.broadcast_to(norm_weight, (N, N, C))

        # Compute left and right projection
        left = tl.einsum("ijkd,dkh->ijh", input, left_proj_weight)
        right = tl.einsum("ijkd,dkh->ijh", input, right_proj_weight)

        # Apply mask
        left = left * mask
        right = right * mask

        # Compute gates
        left_gate = tl.einsum("ijkd,dkh->ijh", input, left_gate_weight)
        right_gate = tl.einsum("ijkd,dkh->ijh", input, right_gate_weight)
        out_gate = tl.einsum("ijkd,dkh->ijh", input, out_gate_weight)

        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.sigmoid(out_gate)

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute the TriMul operation
        # Use einsum to compute the pairwise product over k
        # This is analogous to: out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # Here, for each i and j, we sum over k: left[i,k,:] * right[j,k,:]
        out = tl.zeros((N, H), dtype=tl.float32)
        for k in range(N):
            l = tl.load(left_ptr + batch_offsets + i_offsets + k * H)
            r = tl.load(right_ptr + batch_offsets + j_offsets + k * H)
            out += l * r

        # Apply the output normalization and gate
        out = (out - tl.broadcast_to(to_out_norm_bias, (N, H))) * tl.broadcast_to(to_out_norm_weight, (N, H))
        out = out * out_gate

        # Apply the final linear projection
        out = tl.einsum("ijkh, hkd -> ijd", out, to_out_weight)

        # Store the result
        out = tl.reshape(out, (N, N, C))
        tl.store(output_ptr + batch_offsets + i_offsets + j_offsets, out)

    # Launch kernel
    grid = (B * N * N,)
    triton.kernel_launcher(tri_mul_kernel, input_tensor, mask, left_proj_weight, right_proj_weight,
                           left_gate_weight, right_gate_weight, out_gate_weight,
                           to_out_norm_weight, to_out_norm_bias, to_out_weight,
                           output, N, B, C, H, 64, 8, 128)

    return output
