import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output shape
    output_shape = (bs, seq_len, seq_len, dim)
    output = torch.empty(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # Compute the TriMul operation
    # We will implement this in a triton kernel that efficiently computes:
    # for each batch, for each i, j, compute sum over k of (left[i, k] * right[j, k])

    # Kernel definition
    @triton.jit
    def trit_mul_kernel(
        input_ptr, mask_ptr, left_proj_w_ptr, right_proj_w_ptr,
        left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
        to_out_norm_w_ptr, to_out_norm_b_ptr, to_out_w_ptr,
        output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        num_blocks_m, num_blocks_n,
        BLOCK: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
    ):
        # Each threadblock computes a M x N block of the output
        pid = tl.program_id(axis=0)
        block_m = pid // num_blocks_n
        block_n = pid % num_blocks_n

        # Compute the block's starting index
        # For the outgoing TriMul, the i and j indices are both in [0, seq_len)
        # We will compute the block's M (i) and N (j) ranges
        m_start = block_m * BLOCK_M
        n_start = block_n * BLOCK_N

        # Load the output into shared memory for faster access
        offs_m = tl.arange(0, BLOCK_M)
        offs_n = tl.arange(0, BLOCK_N)

        # Compute the offsets for the output
        m_offsets = m_start + offs_m
        n_offsets = n_start + offs_n

        # Initialize output
        out = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # The hidden_dim is the dimension of the intermediate layer
        # So, for each i, j, we will compute sum over k of (left[i, k] * right[j, k])

        # We need to compute left[i, k] and right[j, k] for all k in [0, seq_len)

        # For this kernel, we will use tiling to compute the sum over k
        # We will use a block size for k that fits into the shared memory

        # The number of blocks for k
        num_k_blocks = tl.cdiv(seq_len, BLOCK)

        # Loop over k blocks
        for k_block in range(num_k_blocks):
            # Compute the offset for k
            k_offsets = k_block * BLOCK + tl.arange(0, BLOCK)

            # Compute the indices for the inputs
            # left[i, k] = left_proj(input[i, k]) * mask[i, k] * left_gate(input[i, k])
            # Similarly for right[j, k]

            # Load left and right projections
            # Left projections: [bs, seq_len, seq_len, hidden_dim]
            # We'll load for i in m_offsets, k in k_offsets
            # So we can compute this as input[i, k] * left_proj_weight
            # Then apply the mask and left_gate

            # Similarly for right projections

            # But since this is a triton kernel, we need to compute these using pointer arithmetic

            # First, compute the left projections
            # This part is very complex and involves loading from memory and applying weights

            # We will load all the weights for the left projections
            # left_proj_weights: [hidden_dim, dim]
            # We can compute left[i, k] = (input[i, k] @ left_proj_weights) * mask[i, k] * left_gate_weights @ input[i, k]

            # Let's assume that the weights are stored in column-major format, which is typically the case in PyTorch

            # Compute the left projections
            left_input = tl.load(
                input_ptr + (m_offsets[:, None] * seq_len * seq_len * dim + k_offsets[None, :] * seq_len * dim),
                mask=tl.where(tl.arange(0, seq_len) >= seq_len, 0, 1).astype(tl.float32),
            )
            left_proj = tl.dot(left_input, left_proj_weight)
            left_gate = tl.sigmoid(tl.dot(left_input, left_gate_weight))
            left_val = left_proj * left_gate * tl.load(mask_ptr + m_offsets[:, None] * seq_len + k_offsets[None, :], dtype=tl.float32)

            # Similarly for right projections
            right_input = tl.load(
                input_ptr + (n_offsets[:, None] * seq_len * seq_len * dim + k_offsets[None, :] * seq_len * dim),
                mask=tl.where(tl.arange(0, seq_len) >= seq_len, 0, 1).astype(tl.float32),
            )
            right_proj = tl.dot(right_input, right_proj_weight)
            right_gate = tl.sigmoid(tl.dot(right_input, right_gate_weight))
            right_val = right_proj * right_gate * tl.load(mask_ptr + n_offsets[:, None] * seq_len + k_offsets[None, :], dtype=tl.float32)

            # Now compute the product
            # This is the key step: sum over k of left[i, k] * right[j, k]
            # For this, we can compute the product for all i, j, k

            # We need to sum across k
            # For this, we'll perform a matrix multiplication across k

            temp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
            for k in range(BLOCK):
                # Load left and right values for this k
                left_k = tl.load(
                    left_val_ptr + (m_offsets[:, None] * seq_len + k_offsets[None, :] * seq_len),
                    mask=tl.where(tl.arange(0, seq_len) >= seq_len, 0, 1).astype(tl.float32),
                )
                right_k = tl.load(
                    right_val_ptr + (n_offsets[:, None] * seq_len + k_offsets[None, :] * seq_len),
                    mask=tl.where(tl.arange(0, seq_len) >= seq_len, 0, 1).astype(tl.float32),
                )

                # Multiply left and right for this k
                temp += tl.dot(left_k, right_k)

            # Now, apply the normalization and the out_gate
            out = out * tl.sigmoid(tl.dot(input_ptr + (m_offsets[:, None] * seq_len * seq_len * dim + n_offsets[None, :] * seq_len * dim), out_gate_weight))
            out = out * tl.nn.functional.layer_norm(out, to_out_norm_weight, to_out_norm_bias)
            out = out * tl.dot(out, to_out_weight)

            # Store the computed output
            tl.store(output_ptr + (m_offsets[:, None] * seq_len + n_offsets[None, :] * seq_len), out)

        # Return the output
        return out

    # Launch the kernel
    num_blocks_m = tl.cdiv(seq_len, BLOCK_M)
    num_blocks_n = tl.cdiv(seq_len, BLOCK_N)
    num_blocks_total = num_blocks_m * num_blocks_n
    grid = (num_blocks_total, 1, 1)

    output = trit_mul_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output,
        bs, seq_len, dim, hidden_dim,
        num_blocks_m, num_blocks_n,
        BLOCK=BLOCK, BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N
    )

    return output
