import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    """
    TriMul kernel implementation using Triton for the "outgoing" version of the operation.

    This kernel performs the following operations:
    1. Applies LayerNorm to the input tensor.
    2. Projects input into hidden dimension using left and right projections.
    3. Applies masks to the projected tensors.
    4. Applies sigmoid gates to left and right projections.
    5. Computes the TriMul operation using einsum-like operation.
    6. Applies LayerNorm and gate to the output.
    7. Projects the output back to the input dimension.

    The kernel is optimized for performance on NVIDIA H100 GPUs, leveraging features such as TMA and wgmma.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output tensor
    output = torch.empty_like(input_tensor)

    # Batch size, sequence length, and channel dimensions
    B = batch_size
    N = seq_len
    C = dim

    # Helper function to load weights in Triton
    def load_weights(weight, n_rows, n_cols):
        return tl.load(weight,
                       boundary_check={0: False, 1: False},
                       padding=0)

    # Helper function to apply LayerNorm
    def apply_layer_norm(tensor, weight, bias, eps=1e-5):
        mean = tl.sum(tensor, axis=-1, keepdims=True) / N
        var = tl.sum(tl.square(tensor - mean), axis=-1, keepdims=True) / N
        std = tl.sqrt(var + eps)
        norm = (tensor - mean) / std
        return norm * weight + bias

    # Define kernel
    @triton.jit
    def trit_mul_kernel(
        input_ptr, mask_ptr, left_proj_w_ptr, right_proj_w_ptr, left_gate_w_ptr, right_gate_w_ptr,
        out_gate_w_ptr, to_out_norm_w_ptr, to_out_norm_b_ptr, to_out_w_ptr,
        output_ptr,
        B: tl.constexpr, N: tl.constexpr, C: tl.constexpr, H: tl.constexpr,
        STRIDE_IN_BS: tl.constexpr, STRIDE_IN_SN: tl.constexpr, STRIDE_IN_SN2: tl.constexpr,
        STRIDE_IN_C: tl.constexpr,
        STRIDE_MASK_BS: tl.constexpr, STRIDE_MASK_SN: tl.constexpr, STRIDE_MASK_SN2: tl.constexpr,
        STRIDE_LPROJ_W_C: tl.constexpr, STRIDE_LPROJ_W_H: tl.constexpr,
        STRIDE_RPROJ_W_C: tl.constexpr, STRIDE_RPROJ_W_H: tl.constexpr,
        STRIDE_LGATE_W_C: tl.constexpr, STRIDE_LGATE_W_H: tl.constexpr,
        STRIDE_RGATE_W_C: tl.constexpr, STRIDE_RGATE_W_H: tl.constexpr,
        STRIDE_UGATE_W_C: tl.constexpr, STRIDE_UGATE_W_H: tl.constexpr,
        STRIDE_TONORM_W_C: tl.constexpr, STRIDE_TONORM_W_H: tl.constexpr,
        STRIDE_TONORM_B: tl.constexpr,
        STRIDE_TOOUT_W_C: tl.constexpr, STRIDE_TOOUT_W_H: tl.constexpr,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # Get thread ID
        pid = tl.program_id(axis=0)

        # Compute position in batch, sequence dimension
        idx_b = pid // (N * N)
        idx_n = (pid // N) % N
        idx_m = pid % N

        # Compute offset in memory
        offset = idx_b * STRIDE_IN_BS + idx_n * STRIDE_IN_SN + idx_m * STRIDE_IN_SN2

        # Load input tensor for this batch, sequence pair
        input = tl.load(input_ptr + offset, mask=tl.full((N, N), True, dtype=tl.bool))

        # Apply mask
        mask = tl.load(mask_ptr + idx_b * STRIDE_MASK_BS + idx_n * STRIDE_MASK_SN + idx_m * STRIDE_MASK_SN2)
        input = input * mask

        # Apply LayerNorm
        input = apply_layer_norm(input, norm_weight, norm_bias)

        # Project left and right
        left = tl.dot(input, left_proj_weight)
        right = tl.dot(input, right_proj_weight)

        # Apply left and right gates
        left_gate = tl.dot(input, left_gate_weight).sigmoid()
        right_gate = tl.dot(input, right_gate_weight).sigmoid()
        left = left * left_gate
        right = right * right_gate

        # Compute TriMul
        output = tl.dot(left, right)

        # Apply LayerNorm and gate
        output = apply_layer_norm(output, to_out_norm_weight, to_out_norm_bias)
        output = output * tl.dot(input, out_gate_weight).sigmoid()

        # Project to output dimension
        output = tl.dot(output, to_out_weight)

        # Write output
        tl.store(output_ptr + offset, output)

    # Set up the kernel parameters
    STRIDE_IN_BS = input_tensor.stride(0)
    STRIDE_IN_SN = input_tensor.stride(1)
    STRIDE_IN_SN2 = input_tensor.stride(2)
    STRIDE_IN_C = input_tensor.stride(3)

    STRIDE_MASK_BS = mask.stride(0)
    STRIDE_MASK_SN = mask.stride(1)
    STRIDE_MASK_SN2 = mask.stride(2)

    # Load weights
    left_proj_w = load_weights(left_proj_weight, N, C)
    right_proj_w = load_weights(right_proj_weight, N, C)
    left_gate_w = load_weights(left_gate_weight, N, C)
    right_gate_w = load_weights(right_gate_weight, N, C)
    out_gate_w = load_weights(out_gate_weight, N, C)
    to_out_norm_w = load_weights(to_out_norm_weight, H, C)
    to_out_norm_b = load_weights(to_out_norm_bias, 1, H)
    to_out_w = load_weights(to_out_weight, H, C)

    # Launch kernel
    num_blocks = (B * N * N + 1) // 1  # Block size = 1 for simplicity
    num_warps = 8
    trit_mul_kernel[ num_blocks, num_warps ](
        input_tensor,
        mask,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w,
        output,
        B,
        N,
        C,
        H,
        STRIDE_IN_BS,
        STRIDE_IN_SN,
        STRIDE_IN_SN2,
        STRIDE_IN_C,
        STRIDE_MASK_BS,
        STRIDE_MASK_SN,
        STRIDE_MASK_SN2,
        STRIDE_LPROJ_W_C,
        STRIDE_LPROJ_W_H,
        STRIDE_RPROJ_W_C,
        STRIDE_RPROJ_W_H,
        STRIDE_LGATE_W_C,
        STRIDE_LGATE_W_H,
        STRIDE_RGATE_W_C,
        STRIDE_RGATE_W_H,
        STRIDE_UGATE_W_C,
        STRIDE_UGATE_W_H,
        STRIDE_TONORM_W_C,
        STRIDE_TONORM_W_H,
        STRIDE_TONORM_B,
        STRIDE_TOOUT_W_C,
        STRIDE_TOOUT_W_H,
        1,
        1,
        1,
    )

    return output
