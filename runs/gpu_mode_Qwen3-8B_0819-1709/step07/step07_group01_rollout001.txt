import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    
    batch_size, seq_len, _, _ = input_tensor.shape
    # Input: [bs, seq_len, seq_len, dim]
    # Mask: [bs, seq_len, seq_len]
    
    # Access the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Perform LayerNorm on the input
    input_norm = input_tensor * norm_weight + norm_bias
    
    # Project input to hidden space
    left_proj = input_norm @ left_proj_weight.t()
    right_proj = input_norm @ right_proj_weight.t()
    
    # Apply mask to left and right projections
    mask = mask.unsqueeze(-1).to(torch.bool)
    left_proj = left_proj.masked_fill(~mask, 0)
    right_proj = right_proj.masked_fill(~mask, 0)
    
    # Compute gate activations (sigmoid)
    left_gate = torch.sigmoid(input_norm @ left_gate_weight.t())
    right_gate = torch.sigmoid(input_norm @ right_gate_weight.t())
    out_gate = torch.sigmoid(input_norm @ out_gate_weight.t())
    
    # Element-wise multiplication with gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate
    
    # Compute TriMul using einsum-like operation
    # out[b, i, j] = sum_k (left[b, i, k] * right[b, j, k])
    # We'll implement this with a Triton kernel
    out = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    
    # Triton kernel for TriMul
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len, batch_size, hidden_dim,
        BLOCK_SIZE: tl.constexpr,
    ):
        # Each thread handles a single (i, j) pair (i.e., a single output position)
        i = tl.program_id(0)
        j = tl.program_id(1)
        # Each thread handles a full block of k positions
        block_k = tl.program_id(2)
        # Compute the base indices for the thread
        block_start_k = block_k * BLOCK_SIZE
        block_end_k = block_start_k + BLOCK_SIZE
        # Compute the offset indices for each position in the block
        offset_i = i * seq_len + j
        offset_j = j * seq_len + i

        # Compute the offset for the left and right projections
        # left_ptr: [batch, seq_len, seq_len, hidden_dim]
        # right_ptr: [batch, seq_len, seq_len, hidden_dim]
        # out_ptr: [batch, seq_len, seq_len, hidden_dim]
        # Each thread accesses the same batch index
        batch_index = tl.program_id(3)

        # Compute the base indices for the current batch
        base_left = batch_index * batch_size * seq_len * seq_len * hidden_dim
        base_right = base_left
        base_out = base_left

        # Compute the offsets for the current (i,j) pair
        # left[i, k] = batch_index, i, k, *
        left_offsets = (base_left + (i * seq_len + k) * hidden_dim) for k in range(seq_len)
        right_offsets = (base_right + (j * seq_len + k) * hidden_dim) for k in range(seq_len)
        out_offsets = (base_out + (i * seq_len + j) * hidden_dim) for _ in range(hidden_dim)
        
        # Compute the sum over k: left[i][k] * right[j][k]
        acc = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k in range(seq_len):
            # Load the left and right vectors
            left_vec = tl.load(left_ptr + (i * seq_len + k) * hidden_dim, nolock=True)
            right_vec = tl.load(right_ptr + (j * seq_len + k) * hidden_dim, nolock=True)
            # Element-wise product
            product = left_vec * right_vec
            # Add to accumulation
            acc += product
        
        # Apply the out_gate
        out_gate_vec = tl.load(out_gate_ptr + (batch_index * seq_len * seq_len * hidden_dim) + (i * seq_len + j) * hidden_dim, nolock=True)
        acc *= out_gate_vec
        
        # Store the result
        tl.store(out_ptr + (i * seq_len + j) * hidden_dim, acc, nolock=True)

    # Launch the kernel
    num_blocks = triton.cdiv(seq_len * seq_len * batch_size, 128)
    num_threads_per_block = 128
    grid = (seq_len, seq_len, seq_len, batch_size)
    tri_mul_kernel[grid](left_proj, right_proj, out,
                         seq_len, batch_size, hidden_dim,
                         num_threads_per_block)

    # Apply LayerNorm to the output
    out_norm = out * to_out_norm_weight + to_out_norm_bias
    # Apply the final linear projection
    out_final = out_norm @ to_out_weight.t()
    
    return out_final
