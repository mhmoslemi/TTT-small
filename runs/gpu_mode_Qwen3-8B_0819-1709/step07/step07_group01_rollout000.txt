import torch
import triton
import triton.language as tl
import math
from typing import Dict, Tuple

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]
    
    # Define the output tensor
    output = torch.empty((batch_size, seqlen, seqlen, dim), dtype=torch.float32, device=input_tensor.device)
    
    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm (can be optimized further if needed)
    input_tensor = input_tensor * norm_weight + norm_bias
    
    # Project to hidden dimension
    left_proj = torch.matmul(input_tensor, left_proj_weight.t())
    right_proj = torch.matmul(input_tensor, right_proj_weight.t())
    
    # Apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left_gate = torch.matmul(input_tensor, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_tensor, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_tensor, out_gate_weight.t()).sigmoid()

    # Apply gates to projections
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate
    
    # Define the kernel
    @triton.jit
    def _triu_kernel(
        left_ptr, right_ptr, output_ptr,
        seqlen, batch_size, hidden_dim,
        stride_lbs, stride_lsd, stride_lsd2,
        stride_rbs, stride_rsd, stride_rsd2,
        stride_obs, stride_ods, stride_odsd,
        BLOCK: tl.constexpr,
    ):
        # Map the thread to a position in the batch
        pid_batch = tl.program_id(0)
        # Map the thread to a position in the sequence
        pid_seq = tl.program_id(1)
        # Compute the offset in the matrix
        offset = pid_seq * seqlen + tl.arange(0, BLOCK)
        # Compute the indices
        i = offset + tl.arange(0, BLOCK)
        j = offset
        # Compute the mask for the upper triangle
        mask = i >= j
        
        # Load data
        left = tl.load(left_ptr + pid_batch * stride_lbs + i[:, None] * stride_lsd + tl.arange(0, BLOCK)[None, :] * stride_lsd2, mask=mask)
        right = tl.load(right_ptr + pid_batch * stride_rbs + j[:, None] * stride_rsd + tl.arange(0, BLOCK)[None, :] * stride_rsd2, mask=mask)
        
        # Compute sum over k
        acc = tl.sum(left * right, axis=1)
        
        # Store output
        tl.store(output_ptr + pid_batch * stride_obs + i[:, None] * stride_ods + tl.arange(0, BLOCK)[None, :] * stride_odsd, acc)

    # Launch the kernel
    num_warps = 4
    grid = (batch_size, seqlen)
    _triu_kernel[grid](left_proj, right_proj, output,
                       seqlen, batch_size, hidden_dim,
                       left_proj.stride(0), left_proj.stride(1), left_proj.stride(2),
                       right_proj.stride(0), right_proj.stride(1), right_proj.stride(2),
                       output.stride(0), output.stride(1), output.stride(2),
                       BLOCK=seqlen // 32)
    
    # Apply to_out_norm
    output = output * to_out_norm_weight + to_out_norm_bias
    
    # Apply out_gate
    output = output * out_gate
    
    # Final projection
    output = torch.matmul(output, to_out_weight.t())
    
    return output

    in the AlphaFold3 paper. Uses Triton for efficient computation of
