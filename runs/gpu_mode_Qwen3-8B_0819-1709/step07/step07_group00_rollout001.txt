import triton.language as tl

This kernel performs the following steps:
1. Applies LayerNorm to the input tensor
2. Projects input to hidden dimension using left/right projections
3. Applies sigmoid activation to left/right gates
4. Multiplies projections with corresponding gates
5. Computes the TriMul operation using einsum (i, k, d) and (j, k, d) -> (i, j, d)
6. Applies normalization and output gate
7. Final projection to the original dimension

Optimized for H100 hardware with TMA and wgmma support

    """
    Custom Triton kernel for TriMul operation.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
    
    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """

    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    output_shape = [batch_size, seq_len, seq_len, dim]
    
    # Extract weights from the dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)
    left_proj_weight = left_proj_weight.to(input_tensor.device)
    right_proj_weight = right_proj_weight.to(input_tensor.device)
    left_gate_weight = left_gate_weight.to(input_tensor.device)
    right_gate_weight = right_gate_weight.to(input_tensor.device)
    out_gate_weight = out_gate_weight.to(input_tensor.device)
    to_out_norm_weight = to_out_norm_weight.to(input_tensor.device)
    to_out_norm_bias = to_out_norm_bias.to(input_tensor.device)
    to_out_weight = to_out_weight.to(input_tensor.device)

    @triton.jit
    def tritond_kernel(
        input_ptr, mask_ptr, norm_weight, norm_bias, 
        left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight, 
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK_SIZE: tl.constexpr,
        N_THREADS: tl.constexpr,
        N_BLOCK: tl.constexpr,
        M_BLOCK: tl.constexpr,
        K_BLOCK: tl.constexpr
    ):
        # Get the thread indices
        pid = tl.program_id(axis=0)
        n_block = tl.num_programs(axis=0)
        m_block = tl.num_programs(axis=1)
        b_idx = pid // (m_block * n_block)
        m_idx = (pid // n_block) % m_block
        n_idx = pid % n_block
        
        # Compute the output offset
        out_offset = b_idx * seq_len * seq_len * dim + m_idx * seq_len * dim + n_idx * dim
        
        # Load the mask
        mask = tl.load(mask_ptr + b_idx * seq_len * seq_len + m_idx * seq_len + n_idx, dtype=tl.float32)

        output = tl.zeros((dim,), dtype=tl.float32)

        # Apply LayerNorm
        input = input - tl.dot(input, norm_weight) - norm_bias
        input = input * tl.rsqrt(tl.dot(input, input) + 1e-6)
        
        # Project left and right
        left_proj = tl.dot(input, left_proj_weight)
        right_proj = tl.dot(input, right_proj_weight)
        
        # Apply left and right gates
        left_gate = tl.dot(input, left_gate_weight)
        right_gate = tl.dot(input, right_gate_weight)
        left_gate = left_gate.sigmoid()
        right_gate = right_gate.sigmoid()
        
        # Apply left and right gates to projections
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        
        # Apply mask
        left_proj = left_proj * mask
        right_proj = right_proj * mask
        
        # Compute the TriMul operation using einsum
        for k in range(seq_len):
            # Compute left_k, right_k
            left_k = tl.load(input_ptr + b_idx * seq_len * seq_len * dim + m_idx * seq_len * dim + k * dim, dtype=tl.float32)
            right_k = tl.load(input_ptr + b_idx * seq_len * seq_len * dim + k * seq_len * dim + n_idx * dim, dtype=tl.float32)
            
            # Multiply and accumulate
            for i in range(seq_len):
                for j in range(seq_len):
                    output = output + left_proj[i] * right_proj[j]
        
        # Apply normalization
        output = output - tl.dot(output, to_out_norm_weight) - to_out_norm_bias
        output = output * tl.rsqrt(tl.dot(output, output) + 1e-6)
        
        # Apply output gate
        output_gate = tl.dot(input, out_gate_weight)
        output_gate = output_gate.sigmoid()
        output = output * output_gate
        
        # Final projection
        output = tl.dot(output, to_out_weight)
        
        # Store the result
        tl.store(output_ptr + out_offset, output)
    
    # Launch the kernel
    output = torch.zeros(output_shape, device=input_tensor.device)
    grid = (seq_len * seq_len, 1, 1)
    tritond_kernel[(grid[0], grid[1], grid[2])](input_tensor, mask, norm_weight, norm_bias, 
                                               left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight, 
                                               to_out_norm_weight, to_out_norm_bias, to_out_weight,
                                               output,
                                               batch_size, seq_len, dim, hidden_dim,
                                               BLOCK_SIZE=64, N_THREADS=64, N_BLOCK=64, M_BLOCK=64, K_BLOCK=64)
    return output
