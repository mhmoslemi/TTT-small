import torch
import triton
import triton.language as tl
import math
from triton import Config, compile


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    
    # Extract relevant weights
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output normalization
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Final output projection
    to_out_weight = weights["to_out.weight"]

    # Extract input shape
    B, N, _, _ = input_tensor.shape
    output_shape = (B, N, N, dim)

    # Initialize output
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=input_tensor.dtype)

    # Define a Triton kernel function
    @triton.jit
    def tritond_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        output_ptr,
        batch, seq_len, dim, hidden_dim,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
        STRIDE_INNER: tl.constexpr, STRIDE_OUTER: tl.constexpr,
        STRIDE_INNER_MASK: tl.constexpr, STRIDE_OUTER_MASK: tl.constexpr
    ):
        # Tile indices
        pid = tl.program_id(axis=0)
        pid_n = pid // (seq_len // BLOCK_N)
        pid_k = pid % (seq_len // BLOCK_N)

        # Local offsets
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
        offs_d = tl.arange(0, BLOCK_D)

        # Load data
        input = tl.load(
            input_ptr + offs_n[:, None] * STRIDE_INNER + offs_k[None, :] * STRIDE_OUTER + offs_d[None, None, :],
            mask=tl.where(tl.load(mask_ptr + offs_n[:, None] * STRIDE_INNER_MASK + offs_k[None, :], 0) == 1,
                          1, 0)
        )
        # Project
        left_proj = tl.dot(input, left_proj_weight, axis=-1)
        right_proj = tl.dot(input, right_proj_weight, axis=-1)
        # Apply gates
        left_gate = tl.dot(input, left_gate_weight, axis=-1).sigmoid()
        right_gate = tl.dot(input, right_gate_weight, axis=-1).sigmoid()
        out_gate = tl.dot(input, out_gate_weight, axis=-1).sigmoid()
        # Apply masks
        left_proj = left_proj * tl.load(mask_ptr + offs_n[:, None] * STRIDE_INNER_MASK + offs_k[None, :], 0)
        right_proj = right_proj * tl.load(mask_ptr + offs_n[:, None] * STRIDE_INNER_MASK + offs_k[None, :], 0)
        # Apply gates
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        # Compute TriMul
        out = tl.einsum('...i k d, ...j k d -> ...i j d', left_proj, right_proj)
        # Normalize
        out = tl.dot(out, to_out_norm_weight, axis=-1) + to_out_norm_bias
        out = out * out_gate
        # Apply final projection
        out = tl.dot(out, to_out_weight, axis=-1)

        # Write output
        tl.store(
            output_ptr + offs_n[:, None] * STRIDE_INNER + offs_k[None, :] * STRIDE_OUTER + offs_d[None, None, :],
            out
        )

    # Launch Kernel
    # Compute the number of work items (total number of N^2 tiles to process)
    total_tiles = (seq_len * seq_len)
    num_warps = 4
    num_stages = 8

    # Define kernel parameters
    STRIDE_INNER = (dim * seq_len)
    STRIDE_OUTER = seq_len
    STRIDE_INNER_MASK = seq_len
    STRIDE_OUTER_MASK = 1

    # Launch kernel
    tritond_kernel[
        (total_tiles, 1), (num_warps, num_stages), (num_warps, num_stages)
    ](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_weight,
        output,
        B, N, dim, hidden_dim,
        BLOCK_N=128, BLOCK_K=128, BLOCK_D=dim,
        STRIDE_INNER=STRIDE_INNER,
        STRIDE_OUTER=STRIDE_OUTER,
        STRIDE_INNER_MASK=STRIDE_INNER_MASK,
        STRIDE_OUTER_MASK=STRIDE_OUTER_MASK
    )

    return output
