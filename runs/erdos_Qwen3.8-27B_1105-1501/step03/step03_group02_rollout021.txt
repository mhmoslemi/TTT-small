def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0

        w /= wsum

    for r in range(2 * n - 1):
        if w[r] > 1e-10:
            g += w[r] * _score_grad(h, r - (n - 1))
