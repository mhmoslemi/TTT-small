def _line_minimal(h, d, A, B, C, tlo, thi):
    """Find min of max_r (A[r] + B[r]*t + C[r]*t^2) over t in [tlo, thi]"""
    n = len(h)
    M = 2 * n - 1
    
    # Collect candidate t values
    candidates = [tlo, thi, 0.0]
    
    # Stationary points of individual quadratics (only if C > 0)
    for r in range(M):
        if C[r] > 1e-15:
            t0 = -B[r] / (2.0 * C[r])
            if tlo <= t0 <= thi:
                candidates.append(t0)
    
    # Pairwise crossings
    # For pair (i, j): (C_i - C_j)t^2 + (B_i - B_j)t + (A_i - A_j) = 0
    for i in range(M):
        for j in range(i+1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:
                    t0 = -c / b
                    if tlo <= t0 <= thi:
                        candidates.append(t0)
            else:
                disc = b*b - 4.0*a*c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b + sqrt_disc) / (2.0 * a)
                    t2 = (-b - sqrt_disc) / (2.0 * a)
                    if tlo <= t1 <= thi:
                        candidates.append(t1)
                    if tlo <= t2 <= thi:
                        candidates.append(t2)
    
    # Deduplicate
    candidates = np.array(candidates)
    candidates = np.sort(candidates)
    # Remove duplicates within tolerance
    tol = max(1e-12, 1e-10 * max(abs(tlo), abs(thi), 1.0))
    unique = [candidates[0]]
    for val in candidates[1:]:
        if val - unique[-1] > tol:
            unique.append(val)
    
    # Evaluate envelope at each candidate
    best_t = tlo
    best_val = np.inf
    for t in unique:
        vals = A + B * t + C * t * t
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_t = t
    
    return best_t, best_val

from scipy.optimize import linprog
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _repair(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    s = float(np.sum(v))
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _quad_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for l in range(-(n - 1), n):
        r = n - 1 + l
        if l == 0:
            A[r] = float(np.sum(h * (1.0 - h)))
            B[r] = float(np.sum(d * (1.0 - 2.0 * h)))
            C[r] = -float(np.sum(d * d))
        elif l > 0:
            m = n - l
            hh = h[l:l+m]
            hh1 = 1.0 - h[:m]
            dd = d[l:l+m]
            dd1 = d[:m]
            A[r] = float(np.sum(hh * hh1))
            B[r] = float(np.sum(dd * hh1 - hh * dd1))
            C[r] = -float(np.sum(dd * dd1))
        else:
            g = -l
            m = n - g
            hh = h[:m]
            hh1 = 1.0 - h[g:g+m]
            dd = d[:m]
            dd1 = d[g:g+m]
            A[r] = float(np.sum(hh * hh1))
            B[r] = float(np.sum(dd * hh1 - hh * dd1))
            C[r] = -float(np.sum(dd * dd1))
    return A, B, C

def _feasible_interval(h, d):
    tlo = 0.0
    thi = np.inf
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if np.any(pos_mask):
        thi = min(thi, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        thi = min(thi, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    if thi < 0:
        thi = 0.0
    return tlo, thi

def _line_min(h, d, A, B, C, tlo, thi):
    n = len(h)
    M = 2 * n - 1
    if thi - tlo < 1e-14:
        vals = A + B * tlo + C * tlo * tlo
        return tlo, float(np.max(vals))
    candidates = [tlo, thi]
    for r in range(M):
        if C[r] > 1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if tlo <= t0 <= thi:
                candidates.append(t0)
    for i in range(M):
        Ci, Bi, Ai = C[i], B[i], A[i]
        for j in range(i + 1, M):
            a = Ci - C[j]
            b = Bi - B[j]
            c = Ai - A[j]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    t0 = -c / b
                    if tlo <= t0 <= thi:
                        candidates.append(t0)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-14:
                    if disc < 0:
                        disc = 0.0
                    sd = np.sqrt(disc)
                    t1 = (-b + sd) / (2.0 * a)
                    t2 = (-b - sd) / (2.0 * a)
                    if tlo <= t1 <= thi:
                        candidates.append(t1)
                    if tlo <= t2 <= thi:
                        candidates.append(t2)
    cands = np.array(candidates, dtype=float)
    cands = np.sort(cands)
    tol = max(1e-12, 1e-9 * max(abs(tlo), abs(thi), 1.0))
    unique = [cands[0]]
    for val in cands[1:]:
        if val - unique[-1] > tol:
            unique.append(val)
    unique = np.array(unique)
    vals = A[None, :] + B[None, :] * unique[:, None] + C[None, :] * (unique[:, None] ** 2)
    env = vals.max(axis=1)
    idx = int(np.argmin(env))
    return unique[idx], float(env[idx])

def _score_grad(h, r):
    n = len(h)
    l = r - (n - 1)
    g = np.zeros(n)
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        m = n - l
        g[:m] += -h[l:l+m]
        g[l:] += (1.0 - h[:m])
    else:
        gg = -l
        m = n - gg
        g[:m] += (1.0 - h[gg:gg+m])
        g[gg:] += -h[:m]
    return g

def _smooth_max_dir(h, scores, beta=100.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    ws = float(np.sum(w))
    if ws < 1e-300:
        idx = int(np.argmax(scores))
        d = -_score_grad(h, idx)
    else:
        w /= ws
        d = np.zeros(n)
        for r in range(len(scores)):
            d += w[r] * _score_grad(h, r)
        d = -d
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _lp_dir(h, scores, top_k=40):
    n = len(h)
    M = 2 * n - 1
    idx = np.argsort(scores)[::-1][:top_k]
    G = np.zeros((len(idx), n))
    for i, r in enumerate(idx):
        G[i] = _score_grad(h, r)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(idx))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            d -= np.mean(d)
            norm = float(np.max(np.abs(d)))
            if norm < 1e-13:
                return None
            return d / norm
    except Exception:
        pass
    return None

def _pair_dir(h, scores, rng, n_pairs=8):
    n = len(h)
    dirs = []
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:min(20, len(sorted_idx))]
    for _ in range(n_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = np.zeros(n)
        sign = 1.0 if rng.random() < 0.5 else -1.0
        d[i] = sign
        d[j] = -sign
        d -= np.mean(d)
        norm = float(np.max(np.abs(d)))
        if norm > 1e-15:
            dirs.append(d / norm)
        i2 = int(active[int(rng.integers(len(active)))] if len(active) > 0 else rng.integers(n))
        j2 = int(rng.integers(n))
        if i2 != j2:
            d2 = np.zeros(n)
            s2 = 1.0 if rng.random() < 0.5 else -1.0
            d2[i2] = s2
            d2[j2] = -s2
            d2 -= np.mean(d2)
            n2 = float(np.max(np.abs(d2)))
            if n2 > 1e-15:
                dirs.append(d2 / n2)
    return dirs

def _random_dir(n, rng):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _top_grad_dirs(h, scores, n_top=10):
    n = len(h)
    idx = np.argsort(scores)[::-1][:n_top]
    dirs = []
    for r in idx:
        d = -_score_grad(h, r)
        d -= np.mean(d)
        norm = float(np.max(np.abs(d)))
        if norm > 1e-14:
            dirs.append(d / norm)
    return dirs

def _try_direction(h, d, current_raw, target, deadline):
    d = d - np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None, current_raw
    d = d / norm
    tlo, thi = _feasible_interval(h, d)
    if thi - tlo < 1e-13:
        return None, current_raw
    A, B, C = _quad_coeffs(h, d)
    best_t, best_val = _line_min(h, d, A, B, C, tlo, thi)
    if best_val >= current_raw - 1e-12:
        return None, current_raw
    h_new = h + best_t * d
    h_new = _repair(h_new, target)
    new_raw = float(np.max(_scores(h_new)))
    if new_raw < current_raw - 1e-12:
        return h_new, new_raw
    return None, current_raw

def _local_descent(h, deadline, rng, max_stall=30):
    n = len(h)
    target = n / 2.0
    cur_raw = float(np.max(_scores(h)))
    best_h = h.copy()
    best_raw = cur_raw
    stall = 0
    it = 0
    while time.time() < deadline and stall < max_stall:
        it += 1
        scores = _scores(h)
        cur_raw = float(np.max(scores))
        dirs = []
        sm = _smooth_max_dir(h, scores, beta=50.0)
        if sm is not None:
            dirs.append(sm)
        sm2 = _smooth_max_dir(h, scores, beta=200.0)
        if sm2 is not None:
            dirs.append(sm2)
        lp = _lp_dir(h, scores, top_k=30)
        if lp is not None:
            dirs.append(lp)
        dirs.extend(_top_grad_dirs(h, scores, n_top=6))
        dirs.extend(_pair_dir(h, scores, rng, n_pairs=4))
        rd = _random_dir(n, rng)
        if rd is not None:
            dirs.append(rd)
        improved = False
        for d in dirs:
            if time.time() > deadline:
                break
            h_new, raw_new = _try_direction(h, d, cur_raw, target, deadline)
            if h_new is not None:
                h = h_new
                cur_raw = raw_new
                if cur_raw < best_raw:
                    best_raw = cur_raw
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            stall += 1
            pert = h + rng.normal(0, 0.02, n)
            h = _repair(pert, target)
            cur_raw = float(np.max(_scores(h)))
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = h.copy()
    return best_h, best_raw

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new / 2.0)

    if raw1 < best_raw:

    # Phase 2: Multiple perturbations and local descent
    dl2 = t0 + budget_s * 0.65
    scale = 0.04
    attempts = 0
    while time.time() < dl2:
        attempts += 1
        if best_c5 < 0.38080:
            break
        scale = 0.04 * (0.93 ** min(attempts, 25))
        if scale < 0.01:
            scale = 0.01
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _repair(h_p, best_n / 2.0)
        per_dl = min(time.time() + max(5.0, (dl2 - time.time()) * 0.25), dl2)
        h_try, raw_try = _local_descent(h_p, per_dl, rng, max_stall=25)
        if raw_try * (2.0 / best_n) < best_c5 - 1e-13:
            best_raw = raw_try
            best_h = h_try.copy()
            best_c5 = raw_try * (2.0 / best_n)
        elif attempts % 5 == 0:
            scale = min(0.08, scale * 1.15)

    # Phase 3: Try parent transforms
    if time.time() < t0 + budget_s * 0.75:
        transforms = [best_h, np.ones(best_n) - best_h, best_h[::-1],
                      np.ones(best_n) - best_h[::-1]]
        for tf in transforms:
            if time.time() > deadline:
                break
            tf = _repair(tf, best_n / 2.0)
            h_t, raw_t = _local_descent(tf, min(time.time() + 20, t0 + budget_s * 0.78), rng, max_stall=20)

                best_c5 = raw_t * (2.0 / best_n)

    # Phase 4: Resolution changes
    if time.time() < t0 + budget_s * 0.85:
        for dn in [2, -2, 4, -4, 8, -8]:

                break
            nt = best_n + dn

                best_n = nt
                best_c5 = c5_r

    if time.time() < deadline:

        if c5_f < best_c5 - 1e-13:

            best_h = h_f.copy()
            best_c5 = c5_f

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5

    return best_h, final_c5, best_n
