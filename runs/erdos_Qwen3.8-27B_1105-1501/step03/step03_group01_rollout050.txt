def _project(v, target):
    # Project onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fix rounding
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _topk_gradient(h, k=5, tau=50.0):
    """Generate a descent direction using top-k lag weights."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # These are the actual C5 values for each lag
    
    # Softmax weights over the scores
    max_s = np.max(scores)
    exp_s = np.exp(-tau * (max_s - scores))
    w = exp_s / np.sum(exp_s)
    
    # Gradient of the weighted sum of f_k with respect to h
    # f_k = dx * sum_i h[i] * (1-h[i+k])  (with boundary handling)
    # df_k/dh[i] = dx * [(1-h[i+k]) - h[i-k]] (approximately)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)  # lag
        # f_r = dx * sum over valid i of h[i] * (1-h[i+d])
        # d f_r / d h[i] = dx * [(1-h[i+d]) if i+d valid] - dx * [h[i-d] if i-d valid]
        
        # First term: contribution to h[i] from h[i]*(1-h[i+d])
        if d >= 0:
            m = n - d
            grad[:m] += w[r] * (1.0 - h[d:d+m])
            # Second term: -h[i-d]*h[i] contribution
            if d > 0:
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            # h[i]*(1-h[i-g]) for i >= g
            grad[g:] += w[r] * (1.0 - h[:m])
            # -h[i+g]*h[i] for i < n-g
            grad[:m] += w[r] * (-h[g:])
    
    # Make mean-zero
    grad -= np.mean(grad)
    
    # Normalize
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    return grad / norm

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, c5_cur, n_grid=25, n_golden=20):
    n = len(h)
    am = 1e10
    for j in range(n):
        if p[j] > 1e-14:
            am = min(am, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            am = min(am, h[j] / (-p[j]))
    am = min(am, 1.0)
    if am < 1e-10:
        return 0.0, c5_cur
    best_a = 0.0
    best_c5 = c5_cur
    grid = np.linspace(-am, am, n_grid)
    for a in grid:
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(-am, best_a - am * 0.2)
    hi_a = min(am, best_a + am * 0.2)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_golden):
        if hi_a - lo_a < 1e-11:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _weighted_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(-tau * (max_s - scores))
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wk = w[r]
        if wk < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:])
                if d > 0:
                    grad[d:] += wk * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * (1.0 - h[:m])
                grad[:m] += wk * (-h[g:])
    grad -= np.mean(grad)
    return grad

def _grad_descent(h, deadline, tau=50.0, lr_init=0.08, momentum=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        grad = _weighted_gradient(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        mom = momentum * mom + (1.0 - momentum) * grad
        p = -mom
        am = np.linalg.norm(p)
        if am < 1e-13:
            break
        p = p / am
        alpha, new_c5 = _line_search(h, p, cur_c5, n_grid=20, n_golden=18)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.85
            if stall > 50:
                break
            if stall % 20 == 0:
                h = best_h + np.random.default_rng(int(time.time()*1000)).normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine_exact(h, deadline, max_passes=80, n_grid=12, n_golden=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9 or h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                p = np.zeros(n)
                p[j] = 1.0
                p[k] = -1.0
                alpha, c5_try = _line_search(h, p, best_c5, n_grid=n_grid, n_golden=n_golden)
                if c5_try < best_c5 - 1e-13 and abs(alpha) > 1e-10:
                    h = h + alpha * p
                    h = _project(h, target)
                    c5_new = _exact_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _evolutionary(h_init, deadline, pop_size=12, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.03
    while time.time() < deadline:
        idx1 = rng.integers(0, pop_size, size=3)
        i1 = idx1[int(np.argmin([fits[i] for i in idx1]))]
        idx2 = rng.integers(0, pop_size, size=3)
        i2 = idx2[int(np.argmin([fits[i] for i in idx2]))]
        child = 0.5 * (pop[i1] + pop[i2]) + rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.94)
            else:
                mut_scale = min(0.1, mut_scale * 1.04)
        else:
            mut_scale = min(0.1, mut_scale * 1.02)
    return best_h, best_f

def _block_init(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    nblocks = rng.integers(2, 6)
    positions = np.sort(rng.permutation(n))[:nblocks]
    for idx, pos in enumerate(positions):
        width = max(2, int(n / (2 * nblocks) + rng.integers(-3, 4)))
        start = pos - width // 2
        end = start + width
        h[start:end] = 1.0
    h = _project(h, n / 2.0)
    return h

def _smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for _ in range(rng.integers(2, 5)):
        center = rng.uniform(0, 2)
        width = rng.uniform(0.2, 0.8)
        h += np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h = np.clip(h, 0, 1)
    h = _project(h, n / 2.0)
    return h
