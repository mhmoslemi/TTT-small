import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)

        return np.zeros(n)
    if target >= n:

    return x[0] - corr

    h = x[1:]
    return np.sum(h) - len(h) / 2.0

def _grad_of_lag(h, d):
    """Compute gradient of g[n-1+d] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    # d = k - (n-1), so the lag
    # g[n-1+d] = sum_j h[j]*(1-h[j-d]) for valid j
    # dg/dh[m] = (1-h[m-d])*I(0<=m-d<n) - h[m+d]*I(0<=m+d<n)
    for m in range(n):
        val = 0.0
        if 0 <= m - d < n:
            val += (1.0 - h[m - d])
        if 0 <= m + d < n:
            val -= h[m + d]
        grad[m] = val
    return grad

def _subgradient_step(h, top_k=5, step_size=0.01, deadline=None):
    """One step of the subgradient method using top-k active lags."""
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Find top-k lags
    idx = np.argsort(corr)[::-1][:top_k]
    weights = corr[idx] / (np.sum(corr[idx]) + 1e-15)
    
    # Compute combined gradient
    grad = np.zeros(n)
    for w, i in zip(weights, idx):
        d = i - (n - 1)
        grad += w * _grad_of_lag(h, d)
    
    # Remove mean to preserve integral (approximately)
    grad -= np.mean(grad)
    
    # Line search
    cur_c5 = _exact_c5(h)
    best_a, best_c5 = 0.0, cur_c5
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy(), cur_c5
    
    grad = grad / norm
    for t in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:

        c5_t = _exact_c5(h_t)

    if best_a > 0:

    return h.copy(), cur_c5

                                h_t = _project(h_t, target)
                                c5_t = _exact_c5(h_t)
                                if c5_t < best_c5 - 1e-13:
                                    best_c5 = c5_t
                                    best_h = h_t.copy()
                                    h = h_t
                                    improved = True
                                    break
                            if improved:
                                break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP epigraph (25% budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _slsqp_solve(best_h, maxiter=300)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    # Second SLSQP pass
    if time.time() < d1:
        h1b, c5_1b = _slsqp_solve(best_h, maxiter=200)
        if c5_1b < best_c5:
            best_c5, best_h = c5_1b, h1b.copy()

    # Phase 2: L-BFGS smooth (10% budget)
    d2 = t0 + budget_s * 0.35
    for alpha in [20.0, 50.0, 100.0]:
        if time.time() > d2:
            break
        h2, c5_2 = _lbfgs_smooth(best_h, alpha, maxiter=150)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Subgradient (15% budget)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _subgradient_loop(best_h, d3, max_iters=500, top_k=5, temperature=10.0)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _subgradient_loop(best_h, d3, max_iters=300, top_k=3, temperature=20.0)
        if c5_3b < best_c5:
            best_c5, best_h = c5_3b, h3b.copy()

    # Phase 4: SLSQP again (10% budget)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _slsqp_solve(best_h, maxiter=300)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: More subgradient with different params (10% budget)
    d5 = t0 + budget_s * 0.70
    h5, c5_5 = _subgradient_loop(best_h, d5, max_iters=400, top_k=7, temperature=5.0)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Random restarts with SLSQP (15% budget)
    d6 = t0 + budget_s * 0.85
    scale = 0.02
    for _ in range(500):
        if time.time() > d6:
            break
        if best_c5 < 0.3800:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        h_try, c5_try = _slsqp_solve(h_p, maxiter=100)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.15)

    # Phase 7: Symmetry variants (5% budget)
    d7 = t0 + budget_s * 0.90
    sym_list = [1.0 - best_h, best_h[::-1], 1.0 - best_h[::-1]]
    for sp in sym_list:
        if time.time() > d7:
            break
        sp = _project(sp, best_n / 2.0)
        h_try, c5_try = _slsqp_solve(sp, maxiter=150)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    # Phase 8: Pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.95:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.95)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 9: Final SLSQP (5% budget)
    if time.time() < t0 + budget_s * 0.99:
        hf, c5f = _slsqp_solve(best_h, maxiter=200)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
