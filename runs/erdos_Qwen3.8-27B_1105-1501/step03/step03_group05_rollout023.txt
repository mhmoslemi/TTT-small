def _lp_direction(h, scores, radius):
    """Find best LP direction within Linf ball."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = _build_G(h)  # This includes dx scaling
    
    # Minimize t s.t. G @ d <= t - scores[i], |d[i]| <= radius, sum(d) = 0
    # Variables: d (n) and t (1) -> n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((L + 2*n, n + 1))
    b_ub = np.zeros(L + 2*n)
    
    # G @ d - t <= -scores  =>  G @ d + (-1)*t <= -scores
    A_ub[:L, :n] = G
    A_ub[:L, n] = -1.0
    b_ub[:L] = -scores
    
    # d[i] <= radius
    for i in range(n):
        A_ub[L+i, i] = 1.0
        b_ub[L+i] = radius
        A_ub[L+n+i, i] = -1.0
        b_ub[L+n+i] = radius
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    try:

                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

        c5_t = _exact_c5(h_try)

            best_a = a

        return 0.0, cur_c5
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max / n_samples)
    hi = min(alpha_max, best_a + alpha_max / n_samples)

    c1 = hi - gr * (hi - lo)

            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project_box_sum(h + c2 * d, n / 2.0))

    opt_c5 = _exact_c5(h_opt)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

        p_hi = min(n - 1, n - 1 - d)
        for p in range(p_lo, p_hi + 1):

def _smooth_gradient(h, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')

    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / w_sum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:

        if d >= 0:

            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])

                else:
                    a_max = min(a_max, h[k] / (-dk))
                if a_max < 1e-10:
                    continue
                for frac in [0.3, 0.6, 0.85, 1.0]:
                    a = a_max * frac
                    h_try = h.copy()
                    h_try[j] += dj * a
                    h_try[k] += dk * a
                    h_try = _project_box_sum(h_try, target)
                    c5_t = _exact_c5(h_try)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
                if len(interior) < 2:
                    break
        if not improved:
            break
    return h, best_c5

def _binary_swap_search(h, deadline, max_iters=30000, seed=0):
    n = len(h)
    target = n / 2.0
    idx_order = np.argsort(-h)
    h_bin = np.zeros(n)
    k = n // 2
    h_bin[idx_order[:k]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        if not ones or not zeros:
            break
        best_swap = None
        best_val = cur_c5
        batch = min(20, len(ones) * len(zeros))
        for _ in range(batch):
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_stop = t0 + 0.97 * budget_s
    rng = np.random.default_rng(seed)

    # Initialize
    h0 = np.asarray(initial_h_values, dtype=float)
    h0 = np.clip(h0, 0.0, 1.0)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization at original resolution (25%)
    deadline1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_loop(h, deadline1, radius_init=0.12, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: CVXPY trust-region (15%)
    deadline2 = t0 + budget_s * 0.32
    h2, c5_2 = _cvx_loop(best_h, deadline2, rho_init=0.06, max_iters=150)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement after CVXPY (10%)
    deadline3 = t0 + budget_s * 0.40
    h3, c5_3 = _lp_loop(best_h, deadline3, radius_init=0.08, max_iters=4000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Continuation gradient (10%)
    deadline4 = t0 + budget_s * 0.48
    h4, c5_4 = _continuation_opt(best_h, deadline4, beta_start=5.0, beta_end=150.0, n_steps=10)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Diversification - try perturbed starts (18%)
    deadline5 = t0 + budget_s * 0.62
    n_div = 6
    per_div_time = (deadline5 - time.time()) / max(n_div, 1)
    for div_i in range(n_div):
        if time.time() > deadline5:
            break
        div_deadline = min(deadline5, time.time() + per_div_time)
        h_div = _feasible_perturb(best_h, seed + 100 + div_i * 7, n_moves=40, max_delta=0.12)
        h_div, c5_div = _lp_loop(h_div, div_deadline, radius_init=0.1, max_iters=1500)
        if c5_div < best_c5:
            best_c5 = c5_div
            best_h = h_div.copy()

    # Phase 6: Binary swap search (12%)
    deadline6 = t0 + budget_s * 0.72
    hb, c5b = _binary_swap_search(best_h, deadline6, max_iters=40000, seed=seed)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # If binary was better, refine it
    if c5b < best_c5 + 1e-8:
        deadline6b = min(deadline6 + budget_s * 0.04, t0 + budget_s * 0.76)
        h6b, c5_6b = _lp_loop(best_h, deadline6b, radius_init=0.08, max_iters=2000)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution search (15%)
    deadline7 = t0 + budget_s * 0.88
    n_candidates = [best_n]
    for dn in [-8, -4, 4, 8, 12, 16, -12, 20, -20]:
        nn = best_n + dn
        if 32 <= nn <= 160 and nn not in n_candidates:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    for n_try in n_candidates:
        if time.time() > deadline7:
            break
        remaining = deadline7 - time.time()
        per_n = remaining / max(len([x for x in n_candidates if x >= n_try]), 1)
        if per_n < 5:
            continue
        n_try = int(n_try)
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline7, time.time() + per_n * 0.6)
        h_r, c5_r = _lp_loop(h_r, dl_r, radius_init=0.1, max_iters=1000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Quick CVXPY polish
            if time.time() < deadline7:
                dl_cv = min(deadline7, time.time() + per_n * 0.25)
                h_cv, c5_cv = _cvx_loop(best_h, dl_cv, rho_init=0.05, max_iters=50)
                if c5_cv < best_c5:
                    best_c5 = c5_cv
                    best_h = h_cv.copy()

    # Phase 8: Pair polish (7%)
    deadline8 = t0 + budget_s * 0.95
    hp, c5p = _pair_polish(best_h, deadline8, max_passes=100, n_trials=200)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 9: Final LP polish (rest)
    deadline9 = safe_stop
    h_f, c5_f = _lp_loop(best_h, deadline9, radius_init=0.04, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)

    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    # Safety: ensure we don't return worse than initial
    if final_c5 > best_c5:
        final_c5 = best_c5
        best_h = _project_box_sum(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
