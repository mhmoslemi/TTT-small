def _anti_periodic_transform(h):
    """Check/apply anti-periodic structure: h[i+n/2] = 1 - h[i]"""
    n = len(h)
    if n % 2 != 0:
        return h
    half = n // 2
    # Project to anti-periodic subspace
    y = np.zeros(half)
    for i in range(half):
        y[i] = (h[i] - (1.0 - h[i+half])) / 2.0 + 0.5
    y = np.clip(y, 0, 1)
    # Reconstruct
    h_ap = np.zeros(n)
    h_ap[:half] = y
    h_ap[half:] = 1.0 - y
    return _project(h_ap, n / 2.0)

def _resample(h, n_new):

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thr = (lo + hi) / 2.0
    x = np.clip(v - thr, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m <= 0:
            return grad
        grad[d:] += (1.0 - h[:m])
        grad[:m] += -h[d:]
    else:
        g = -d
        m = n - g
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[g:g+m])
        grad[g:] += -h[:m]
    return grad

def _active_set_direction(h, scores, n_active=None, tolerance=1e-6):
    """Compute direction: negative average gradient over active lags"""
    n = len(h)
    max_score = float(np.max(scores))
    if n_active is not None:
        idx = np.argsort(scores)[::-1][:n_active]
    else:
        idx = np.where(scores >= max_score - tolerance)[0]
    if len(idx) == 0:
        idx = [int(np.argmax(scores))]
    direction = np.zeros(n)
    for k in idx:
        d = int(k) - (n - 1)
        direction += _score_grad(h, d)
    direction /= len(idx)
    direction = -direction  # Negative gradient
    return direction

def _feasible_line_search(h, direction, deadline, max_iter=50):
    """Golden section line search along direction, keeping h in [0,1] and sum fixed"""
    n = len(h)
    direction = direction - np.mean(direction)  # Make sum-zero
    
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    if np.max(np.abs(direction)) < 1e-15:
        return 0.0, _exact_c5(h)
    
    lo = 0.0
    hi = alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = np.clip(h + c1 * direction, 0.0, 1.0)
    h1 = _project(h1, n / 2.0)
    f1 = _exact_c5(h1)
    h2 = np.clip(h + c2 * direction, 0.0, 1.0)
    h2 = _project(h2, n / 2.0)
    f2 = _exact_c5(h2)
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if hi - lo < 1e-10 * max(1.0, alpha_max):
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            h1 = np.clip(h + c1 * direction, 0.0, 1.0)
            h1 = _project(h1, n / 2.0)
            f1 = _exact_c5(h1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            h2 = np.clip(h + c2 * direction, 0.0, 1.0)
            h2 = _project(h2, n / 2.0)
            f2 = _exact_c5(h2)
    
    opt_a = (lo + hi) / 2.0
    h_opt = np.clip(h + opt_a * direction, 0.0, 1.0)
    h_opt = _project(h_opt, n / 2.0)
    opt_c5 = _exact_c5(h_opt)
    
    # Also check the two interior points
    if f1 < opt_c5:

    if f2 < opt_c5:

    return opt_a, opt_c5

    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):

            break
        if stall > 80:
            break
        
        scores = _scores(h)
        direction = _active_set_direction(h, scores, n_active, tol)
        
        if np.max(np.abs(direction)) < 1e-14:
            stall += 1
            # Try with more active lags
            if n_active is None:
                tol = tol * 0.5
            else:
                n_active = min(n_active + 5, 2 * n - 1)
            if tol < 1e-12 and (n_active is not None and n_active >= 2 * n - 1):
                break
            continue
        
        alpha, new_c5 = _feasible_line_search(h, direction, deadline, max_iter=40)

        h_new = _project(h_new, n / 2.0)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0

        if remaining < 1.0:
            break
        h_new, c5_new = _lbfgs_solve(cur_h, t0_global + remaining, betas[i], rhos[i],
                                      maxiter=min(200, int(remaining * 50)))
        cur_h = h_new
        cur_c5 = c5_new
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
    return best_h, best_c5

t0_global = 0.0

def _active_set_direction(h, scores, n_active=None, tolerance=1e-8):
    n = len(h)
    max_score = float(np.max(scores))
    if n_active is not None:
        idx = np.argsort(scores)[::-1][:n_active]
    else:
        idx = np.where(scores >= max_score - tolerance)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(scores))])
    direction = np.zeros(n)
    for k in idx:
        d = int(k) - (n - 1)
        direction += _score_grad(h, d)
    direction /= len(idx)
    direction = -direction
    return direction

def _feasible_line_search(h, direction, deadline, max_iter=45):
    n = len(h)
    direction = direction - np.mean(direction)
    if float(np.max(np.abs(direction))) < 1e-15:
        return 0.0, _exact_c5(h)
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-12)
    lo = 0.0
    hi = alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = np.clip(h + c1 * direction, 0.0, 1.0)
    h1 = _project(h1, n / 2.0)
    f1 = _exact_c5(h1)
    h2 = np.clip(h + c2 * direction, 0.0, 1.0)
    h2 = _project(h2, n / 2.0)
    f2 = _exact_c5(h2)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if hi - lo < 1e-11 * max(1.0, alpha_max):
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            h1 = np.clip(h + c1 * direction, 0.0, 1.0)
            h1 = _project(h1, n / 2.0)
            f1 = _exact_c5(h1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            h2 = np.clip(h + c2 * direction, 0.0, 1.0)
            h2 = _project(h2, n / 2.0)
            f2 = _exact_c5(h2)
    opt_a = (lo + hi) / 2.0
    h_opt = np.clip(h + opt_a * direction, 0.0, 1.0)
    h_opt = _project(h_opt, n / 2.0)
    opt_c5 = _exact_c5(h_opt)
    best_a, best_f = opt_a, opt_c5
    if f1 < best_f:
        best_a, best_f = c1, f1
    if f2 < best_f:
        best_a, best_f = c2, f2
    return best_a, best_f

def _active_set_loop(h, deadline, n_active=None, tol=1e-8, max_iters=400):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        direction = _active_set_direction(h, scores, n_active, tol)
        if float(np.max(np.abs(direction))) < 1e-14:
            stall += 1
            if n_active is None:
                tol *= 0.5
            else:
                n_active = min(n_active + 3, 2 * n - 1)
            if tol < 1e-14:
                break
            continue
        alpha, new_c5 = _feasible_line_search(h, direction, deadline, max_iter=40)
        h_new = np.clip(h + alpha * direction, 0.0, 1.0)
        h_new = _project(h_new, n / 2.0)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0:
                scores2 = _scores(h)
                direction2 = _active_set_direction(h, scores2, None, tol * 0.3)
                alpha2, new_c5_2 = _feasible_line_search(h, direction2, deadline, max_iter=20)
                h_tmp = np.clip(h + alpha2 * direction2, 0.0, 1.0)
                h_tmp = _project(h_tmp, n / 2.0)
                new_c5_2 = _exact_c5(h_tmp)
                if new_c5_2 < cur_c5 - 1e-13:
                    h = h_tmp
                    cur_c5 = new_c5_2
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                else:
                    tol *= 0.7
                    if n_active is not None:
                        n_active = max(3, n_active // 2)
    return best_h, best_c5

def _pair_optimize(h, i, j, deadline):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return h.copy(), _exact_c5(h)
    best_c5 = _exact_c5(h)
    best_val = h[i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = h.copy()
    h1[i] = c1
    h1[j] = s - c1
    f1 = _exact_c5(h1)
    h2 = h.copy()
    h2[i] = c2
    h2[j] = s - c2
    f2 = _exact_c5(h2)
    for _ in range(35):
        if time.time() > deadline:
            break
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            h1 = h.copy()
            h1[i] = c1
            h1[j] = s - c1
            f1 = _exact_c5(h1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            h2 = h.copy()
            h2[i] = c2
            h2[j] = s - c2
            f2 = _exact_c5(h2)
    for val in [lo, hi, (lo + hi) * 0.5]:
        h_t = h.copy()
        h_t[i] = val
        h_t[j] = s - val
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_val = val
    h_best = h.copy()
    h_best[i] = best_val
    h_best[j] = s - best_val
    return h_best, best_c5

def _pair_sweep(h, deadline, n_pairs=300, seed=123):
    n = len(h)
    rng = np.random.default_rng(seed)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        h_new, c5_new = _pair_optimize(best_h, i, j, deadline)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _round_robin_pairs(h, deadline, max_rounds=5):
    """Systematically try all pairs in rounds"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(best_h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                h_new, c5_new = _pair_optimize(best_h, i, j, deadline)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
    return best_h, best_c5

def _anti_periodic_project(h):
    """Project to anti-periodic subspace h[i+n/2] = 1-h[i]"""
    n = len(h)
    if n % 2 != 0:
        return h
    half = n // 2
    y = np.zeros(half)
    for i in range(half):
        y[i] = 0.5 * (h[i] + 1.0 - h[i + half])
    y = np.clip(y, 0.0, 1.0)
    h_ap = np.zeros(n)
    h_ap[:half] = y
    h_ap[half:] = 1.0 - y
    return _project(h_ap, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    t0 = t0_global
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B continuation (smooth max)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _lbfgs_continuation(best_h, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Active set gradient (broad active set)
    dl2 = t0 + budget_s * 0.42
    h2, c5_2 = _active_set_loop(best_h, dl2, n_active=None, tol=1e-4, max_iters=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: Pair sweep (random)
    dl3 = t0 + budget_s * 0.52
    h3, c5_3 = _pair_sweep(best_h, dl3, n_pairs=400, seed=seed)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Phase 4: Tight active set + pair alternating
    dl4 = t0 + budget_s * 0.65
    for _ in range(12):
        if time.time() >= dl4:
            break
        chunk1 = time.time() + max(1.5, (dl4 - time.time()) * 0.35)
        h4, c5_4 = _active_set_loop(best_h, chunk1, n_active=8, tol=1e-7, max_iters=100)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
        chunk2 = time.time() + max(1.0, (dl4 - time.time()) * 0.30)
        h4p, c5_4p = _pair_sweep(best_h, chunk2, n_pairs=80, seed=int(time.time() * 1000) % 100000)
        if c5_4p < best_c5:
            best_c5 = c5_4p
            best_h = h4p

    # Phase 5: L-BFGS-B with high beta
    dl5 = t0 + budget_s * 0.75
    remaining5 = dl5 - time.time()
    if remaining5 > 3.0:
        prob_hi = SmoothMaxObj(best_n, 800.0, 5000.0)
        bounds5 = [(0.0, 1.0)] * best_n
        try:
            res5 = minimize(prob_hi.fun, best_h, jac=prob_hi.grad,
                            bounds=bounds5, method='L-BFGS-B',
                            options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
            h5 = _project(np.clip(res5.x, 0.0, 1.0), best_n / 2.0)
            c5_5 = _exact_c5(h5)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5
        except Exception:
            pass

    # Phase 6: Random restarts
    dl6 = t0 + budget_s * 0.88
    scale = 0.025
    attempt = 0
    while time.time() < dl6:
        attempt += 1
        if best_c5 < 0.38075:
            break
        scale = 0.025 * (0.94 ** min(attempt, 60))
        per_dl = time.time() + max(1.5, min(10.0, (dl6 - time.time()) * 0.3))
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        h_try, c5_try = _active_set_loop(h_pert, per_dl, n_active=10, tol=1e-6, max_iters=80)
        h_try, c5_try = _pair_sweep(h_try, min(per_dl + 2.0, dl6), n_pairs=50, seed=seed + attempt * 7919)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            scale = max(0.005, scale * 0.85)

    # Phase 7: Multi-resolution
    dl7 = t0 + budget_s * 0.95
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 12, -12, 16, -16, 24, -24]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 256 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        per_dl = time.time() + max(2.0, min(8.0, (dl7 - time.time()) * 0.3))
        h_r, c5_r = _active_set_loop(h_r, per_dl, n_active=10, tol=1e-5, max_iters=150)
        h_r, c5_r = _pair_sweep(h_r, min(per_dl + 2.0, dl7), n_pairs=80, seed=1000 + n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
            best_n = n_try

    # Phase 8: Final polish
    dl8 = t0 + budget_s * 0.995
    h8, c5_8 = _active_set_loop(best_h, dl8, n_active=5, tol=1e-9, max_iters=200)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8
    h8p, c5_8p = _pair_sweep(best_h, dl8, n_pairs=150, seed=seed + 4242)
    if c5_8p < best_c5:
        best_c5 = c5_8p
        best_h = h8p
    # One more L-BFGS with very high beta
    if time.time() < dl8:
        prob_final = SmoothMaxObj(best_n, 2000.0, 20000.0)
        bounds_f = [(0.0, 1.0)] * best_n
        try:
            res_f = minimize(prob_final.fun, best_h, jac=prob_final.grad,
                             bounds=bounds_f, method='L-BFGS-B',
                             options={'maxiter': 100, 'ftol': 1e-17, 'gtol': 1e-13, 'disp': False})
            h_f = _project(np.clip(res_f.x, 0.0, 1.0), best_n / 2.0)
            c5_f = _exact_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
        except Exception:
            pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        # Revert if projection hurt
        best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
