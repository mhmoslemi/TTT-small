        return opt_a, opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.2, max_iters=5000):
    if top_k_list is None:
        top_k_list = [5, 8, 10, None, 5, 3]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.6)
        else:
            stall += 1
            radius *= 0.65
            ti += 1
            if stall > 25 and radius < 0.02:
                radius = 0.12
    return best_h, best_c5

def _pnorm_optimize(h, deadline, p_list=None, n_steps=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    if p_list is None:
        p_list = [8, 20, 50, 120, 300, 800]
    p_idx = 0
    p = p_list[0]
    stagnation = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        sc = scores + 1e-15
        log_w = (1.0 / p) * (np.log(sc) - np.log(max_s + 1e-15))
        w = np.exp(p * (sc / (max_s + 1e-15) - 1.0))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.08
        improved = False
        for _ in range(50):
            h_new = _project(h + alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.6
            if alpha < 1e-12:
                break
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20:
                p_idx = min(p_idx + 1, len(p_list) - 1)
                p = p_list[p_idx]
                stagnation = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        scores = _scores(h)
        top_idx = np.argsort(scores)[-6:]
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 0.02 or h[j] > 0.98:
                continue
            for k in range(j + 1, min(j + 15, n)):
                if abs(k - j) > 14:
                    continue
                if h[k] < 0.02 or h[k] > 0.98:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _structure_restart(h, rng, n):
    """Create diverse structured perturbations."""
    method = rng.integers(0, 5)
    if method == 0:
        scale = rng.uniform(0.03, 0.12)
        h_pert = h + rng.normal(0, scale, n)
    elif method == 1:
        freq = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        pattern = np.sin(2 * np.pi * freq * x + phase)
        amp = rng.uniform(0.05, 0.15)
        h_pert = h + amp * pattern
    elif method == 2:
        n_flip = rng.integers(4, max(5, n // 4))
        idx = rng.choice(n, n_flip, replace=False)
        h_pert = h.copy()
        for i in idx:
            h_pert[i] = 1.0 - h_pert[i]
    elif method == 3:
        shift = rng.integers(1, n // 3)
        h_pert = np.roll(h, shift)
    else:
        n_blocks = rng.integers(2, 5)
        h_pert = h.copy()
        for b in range(n_blocks):
            start = int(b * n / n_blocks)
            end = int((b + 1) * n / n_blocks)
            delta = rng.uniform(-0.1, 0.1)
            h_pert[start:end] += delta
    return _project(h_pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Aggressive LP with expanding radius
    dl1 = t0 + budget_s * 0.18
    h, c5 = _lp_optimize(h, dl1, top_k_list=[5, 8, 10, 5, None, 3], radius_init=0.3, max_iters=4000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: p-norm smooth optimization
    if time.time() < t0 + budget_s * 0.32:
        dl2 = t0 + budget_s * 0.35
        h, c5 = _pnorm_optimize(best_h, dl2, p_list=[10, 30, 80, 200, 500, 1200], n_steps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Refined LP
    if time.time() < t0 + budget_s * 0.45:
        dl3 = t0 + budget_s * 0.50
        h, c5 = _lp_optimize(best_h, dl3, top_k_list=[3, 4, 5, 8, None], radius_init=0.08, max_iters=5000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Diverse restarts
    dl4 = t0 + budget_s * 0.72
    scale = 0.05
    for attempt in range(600):
        if time.time() > dl4:
            break
        if best_c5 <= 0.3807:
            break
        h_pert = _structure_restart(best_h, rng, best_n)
        sub_dl = min(dl4, time.time() + budget_s * 0.02)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[5, 8, 10, None], radius_init=0.15, max_iters=500)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)

    # Phase 5: Multi-resolution sweep
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.85
        for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 5, -6, 6, 8, -8, 10, -10, 12]:
            if time.time() > dl5:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 160 or n_try == best_n:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[5, 10, None], radius_init=0.12, max_iters=400)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 6: Pair polish
    if time.time() < t0 + budget_s * 0.88:
        dl6 = t0 + budget_s * 0.90
        hp, c5p = _pair_polish(best_h, dl6, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Final ultra-fine LP
    if time.time() < t0 + budget_s * 0.95:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.99, top_k_list=[3, 4, 5, 6], radius_init=0.02, max_iters=8000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
