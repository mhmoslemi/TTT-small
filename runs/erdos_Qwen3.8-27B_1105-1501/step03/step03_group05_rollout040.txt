```python
def _lp_step(h, scores, radius):
    n = len(h)
    dx = 2.0/n
    L = 2*n - 1
    # Build gradient matrix G (L x n) where G[r] = grad of score r
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n-d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    # LP: minimize t
    # s.t. G @ (x-h) + scores - t <= 0  =>  G@x - t <= -scores + G@h
    #      0 <= x <= 1
    #      sum(x) = n/2
    #      |x_i - h_i| <= radius (trust region)
    
    # Variables: x (n) + t (1) = n+1 vars
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: G@x - t <= -scores + G@h
    # => G@x - [row of -1 for t] <= -scores + G@h
    A_ub = np.zeros((L, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores + G @ h
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: max(0, h_i - radius) <= x_i <= min(1, h_i + radius), t free
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
```

```python
    """Find the best swap (i,j) that minimizes max(scores)."""
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    # remove_delta[i][d] = h[i+d]*[valid, d≠0] - (1-h[i-d])*[valid, d≠0]
    remove_deltas = np.zeros((len(ones), 2*n-1))

            if 0 <= i+d < n:
                remove_deltas[k, d+n-1] += h[i+d]
            if 0 <= i-d < n:
                remove_deltas[k, d+n-1] -= (1.0 - h[i-d])

    # add_delta[j][d] = -(h[j+d])*[valid, d≠0] + (1-h[j-d])*[valid, d≠0]
    add_deltas = np.zeros((len(zeros), 2*n-1))
    for k, j in enumerate(zeros):
        for d in range(-(n-1), n):
            if d == 0: continue
            if 0 <= j+d < n:

            if 0 <= j-d < n:
                add_deltas[k, d+n-1] += (1.0 - h[j-d])

    # This is (len(ones) x len(zeros) x (2n-1)) which is 32*32*127 = 130K for n=64
    # Compute as: scores[:, None, :] ... no, let me do it more memory-efficiently
    
    best_val = np.inf

        # new_scores for all j with this i
        tmp = scores + remove_deltas[k]  # shape (2n-1,)
```

            cur_max = float(scores.max())
    return best_h, best_c5

def _lp_refine(h, deadline, rng, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.12
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        L = 2 * n - 1
        G = np.zeros((L, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            lo = max(0, -d)
            hi = min(n, n - d)
            for p in range(lo, hi):
                q = p + d
                G[r, q] += dx * (1.0 - h[p])
                G[r, p] -= dx * h[q]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores + G.dot(h)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = []
        for i in range(n):
            lo_b = max(0.0, h[i] - radius)
            hi_b = min(1.0, h[i] + radius)
            if lo_b < hi_b:
                bounds.append((lo_b, hi_b))
            else:
                bounds.append((h[i], h[i]))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        if not res.success:
            stall += 1
            radius *= 0.7
            continue
        x_cand = res.x[:n]
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            x_try = _project(h + frac * (x_cand - h), target)
            c5_try = _c5(x_try)
            if c5_try < cur_c5 - 1e-12:
                h = x_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                radius = min(radius * 1.15, 0.3)
                break
        if not improved:
            stall += 1
            radius *= 0.7
            if radius < 0.005:
                break
    return best_h, best_c5

def _smooth_refine(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for beta in np.geomspace(5, 300, 10):
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            scores = np.correlate(h, 1.0 - h, mode='full')
            mx = scores.max()
            w = np.exp(beta * (scores - mx))
            ws = w.sum()
            if ws < 1e-300:
                break
            w = w / ws
            grad = np.zeros(n)
            dx = 2.0 / n
            for d in range(-(n-1), n):
                wd = w[n-1+d]
                if abs(wd) < 1e-14:
                    continue
                lo = max(0, -d)
                hi = min(n, n - d)
                for p in range(lo, hi):
                    q = p + d
                    grad[q] += wd * (1.0 - h[p])
                    grad[p] -= wd * h[q]
            grad *= dx
            grad -= grad.mean()
            gn = np.max(np.abs(grad))
            if gn < 1e-14:
                break
            d_dir = -grad / gn
            # Line search
            a_max = 0.3
            for j in range(n):
                if d_dir[j] > 1e-10:
                    a_max = min(a_max, (1.0 - h[j]) / d_dir[j])
                elif d_dir[j] < -1e-10:
                    a_max = min(a_max, h[j] / (-d_dir[j]))
            a_max = max(a_max, 1e-8)
            best_a = 0.0
            best_f = cur_c5
            for a in np.linspace(0, a_max, 10):
                if a < 1e-10:
                    continue
                h_t = _project(h + a * d_dir, target)
                f_t = _c5(h_t)
                if f_t < best_f:
                    best_f = f_t
                    best_a = a
            if best_a > 0 and best_f < cur_c5 - 1e-13:
                h = _project(h + best_a * d_dir, target)
                cur_c5 = _c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _c5(h0)
    best_n = n0

    # Phase 1: Binary steepest descent from parent projection
    deadline1 = t0 + budget_s * 0.30
    hb, cb = _binary_search(best_h, deadline1, rng, max_restarts=100)

        best_c5 = cb
        best_h = hb.copy()

    deadline2 = t0 + budget_s * 0.50
    hl, cl = _lp_refine(best_h, deadline2, rng, max_iters=15000)
    if cl < best_c5:
        best_c5 = cl
        best_h = hl.copy()

    # Phase 3: Smooth gradient refinement
    deadline3 = t0 + budget_s * 0.60
    hs, cs = _smooth_refine(best_h, deadline3)
    if cs < best_c5:
        best_c5 = cs
        best_h = hs.copy()

    # Phase 4: Another LP pass
    deadline4 = t0 + budget_s * 0.70
    hl2, cl2 = _lp_refine(best_h, deadline4, rng, max_iters=10000)
