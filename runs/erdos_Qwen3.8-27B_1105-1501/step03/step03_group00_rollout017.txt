    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, :m] += -(1.0 - h[d:d+m])
            G[r, d:d+m] += h[:m]
        else:
            g = -d
            G[r, :m] += (1.0 - h[g:g+m])
            G[r, g:g+m] += -(h[:m])
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:

        G_k = G[idx]

        scores_k = scores
        G_k = G

    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
