                best_a = a
        if best_a > 0 and best_c5_try < cur - 1e-12:
            h = _project(h + best_a * p, target)
            cur = _c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, seed_val=0):
    """Create diverse structured initializations."""
    rng = np.random.default_rng(seed_val)
    x = np.arange(n) / n
    candidates = []
    # Sine-based
    for freq in [1, 2, 3, 5]:
        for amp in [0.15, 0.25, 0.35]:
            h = 0.5 + amp * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
            candidates.append(_project(h, n/2.0))
    # Triangular
    h = 0.5 + 0.3 * (1.0 - 2.0 * np.abs(x - 0.5))
    candidates.append(_project(h, n/2.0))
    # Best: return the one with lowest C5
    best = candidates[0]
    bc = _c5(best)
    for c in candidates[1:]:
        cc = _c5(c)
        if cc < bc:
            bc = cc
            best = c
    return best

def _multi_active_descent(h, deadline, max_iters=15000):
    """Descend using weighted combination of top lags with adaptive weights."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best_c5 = cur
    lr = 0.1
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _corr(h)
        m = np.max(scores)
        # Power weighting: sharper focus
        p = 20 + 10 * it / max_iters
        w = np.power(np.maximum(scores - 0.95*m, 0.0), p)
        ws = w.sum()
        if ws < 1e-10:
            break
        w = w / ws
        idx_all = np.arange(-(n-1), n)
        grad = np.zeros(n)
        for d, wt in zip(idx_all, w):
            if wt < 1e-10:
                continue
            if d >= 0:
                mm = n - d
                if mm > 0:
                    grad[:mm] += wt * (1.0 - h[d:])
                    grad[d:] -= wt * h[:mm]
            else:
                g = -d
                mm = n - g
                if mm > 0:
                    grad[:mm] += wt * (1.0 - h[g:])
                    grad[g:] -= wt * h[:mm]
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn
        improved = False
        alpha = lr
        for _ in range(18):
            h_try = h + alpha * direction
            if np.any(h_try < -0.15) or np.any(h_try > 1.15):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _c5(h_try)
            if c5_try < cur - 1e-14:
                h = h_try
                cur = c5_try
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.4)
                improved = True
                break
            else:
                alpha *= 0.65
        if not improved:
            stall += 1
            lr *= 0.82
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Multi-active descent on parent
    deadline1 = t0 + budget_s * 0.20
    h, c5 = _multi_active_descent(h, deadline1, max_iters=12000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Top-k descent
    deadline2 = t0 + budget_s * 0.35
    h, c5 = _topk_descent(best_h, deadline2, k=5, max_iters=15000, lr_init=0.08)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: LP polish
    deadline3 = t0 + budget_s * 0.50
    h, c5 = _lp_polish(best_h, deadline3, radius_init=0.06, max_iters=3000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Simulated annealing for global exploration
    deadline4 = t0 + budget_s * 0.65
    h, c5 = _simulated_anneal(best_h, deadline4, T_start=0.2, T_end=0.001, max_iters=40000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Perturbation restarts
    deadline5 = t0 + budget_s * 0.80
    scale = 0.05
    for attempt in range(200):
        if time.time() > deadline5:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        t_per = max(0.1, (deadline5 - time.time()) / max(200 - attempt, 1))
        h_try, c5_try = _topk_descent(h_pert, time.time() + min(t_per, 8.0),
                                       k=4, max_iters=2000, lr_init=0.06)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)

    # Phase 6: Try different n values
    deadline6 = t0 + budget_s * 0.90
    for dn in [0, -8, -4, -12, 4, 8, -16, 12]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 128:
            continue
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _topk_descent(h_r, min(deadline6, time.time() + 5.0),
                                   k=4, max_iters=3000, lr_init=0.08)
        h_r, c5_r = _lp_polish(h_r, min(deadline6, time.time() + 4.0),
                                radius_init=0.05, max_iters=500)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Block descent
    deadline7 = t0 + budget_s * 0.95
    h, c5 = _block_descent(best_h, deadline7, block_size=3, n_passes=30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 8: Final tight polish
    if time.time() < t0 + budget_s * 0.985:
        h, c5 = _topk_descent(best_h, t0 + budget_s * 0.985,
                               k=6, max_iters=8000, lr_init=0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() < t0 + budget_s * 0.998:
            h, c5 = _lp_polish(best_h, t0 + budget_s * 0.998,
                                radius_init=0.02, max_iters=1000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
