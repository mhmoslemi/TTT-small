    lin_con = LinearConstraint(A, target, target)
    bnds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(lag_obj, h0, jac=lag_grad, method='SLSQP',
                   bounds=bnds, constraints=[lin_con],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return _project(res.x, target)
    return h0

def _anneal_lse(h0, deadline, base_T=0.05, n_anneal=8):
    """Anneal LSE temperature and solve at each level"""
    n = len(h0)
    target = n / 2.0
    cur = h0.copy()
    best = h0.copy()
    best_c5 = _exact_c5(h0)
    for i in range(n_anneal):
        if time.time() > deadline:
            break
        frac = 1.0 - (i + 1) / (n_anneal + 1)
        T = base_T * (0.15 ** i) + 0.0005
        cur = _solve_lse_slsqp(cur, T, maxiter=120, ftol=1e-12)
        c5 = _exact_c5(cur)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = cur.copy()
    return best, best_c5

def _targeted_refine(h0, deadline):
    """Iteratively target the worst lag and re-minimize"""
    n = len(h0)
    target = n / 2.0
    cur = h0.copy()
    best = h0.copy()
    best_c5 = _exact_c5(h0)
    for _ in range(30):
        if time.time() > deadline:
            break
        vals = np.correlate(cur, 1.0 - cur, mode='full')
        worst = int(np.argmax(vals))
        cur = _solve_single_lag(cur, worst, maxiter=150)
        c5 = _exact_c5(cur)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = cur.copy()
        else:
            # Try second worst
            vals2 = vals.copy()
            vals2[worst] = -1e10
            worst2 = int(np.argmax(vals2))
            cur2 = _solve_single_lag(best, worst2, maxiter=100)
            c52 = _exact_c5(cur2)
            if c52 < best_c5 - 1e-14:
                best_c5 = c52
                best = cur2.copy()
            else:
                break
    return best, best_c5

def _pair_polish(h0, deadline):
    """Coordinate pair descent for fine tuning"""
    n = len(h0)
    cur = h0.copy()
    best = h0.copy()
    best_c5 = _exact_c5(h0)
    rng = np.random.default_rng(123)
    stall = 0
    while time.time() < deadline and stall < 200:
        improved = False
        # Compute gradient to guide
        vals = np.correlate(cur, 1.0 - cur, mode='full')
        dx = 2.0 / n
        m = float(np.max(vals))
        T = 0.01
        w = np.exp(np.clip((vals - m) / T, -500, 0))
        w /= w.sum()
        g = np.zeros(n)
        for pos in range(2 * n - 1):
            if w[pos] < 1e-6:
                continue
            d = pos - (n - 1)
            c = w[pos] * dx
            if d >= 0:
                L = n - d
                g[:L] += c * (1.0 - cur[d:d + L])
                g[d:d + L] -= c * cur[:L]
            else:
                L = n + d
                g[-L:] += c * (1.0 - cur[:L])
                g[:L] -= c * cur[-L:]
        # Top coordinates by gradient
        top = np.argsort(-np.abs(g))[:16]
        for ii in range(len(top)):
            if time.time() > deadline:
                return best, best_c5
            j = int(top[ii])
            if cur[j] <= 1e-8 or cur[j] >= 1.0 - 1e-8:
                continue
            for jj in range(ii + 1, len(top)):
                k = int(top[jj])
                if k == j or cur[k] <= 1e-8 or cur[k] >= 1.0 - 1e-8:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    for sign in [1, -1]:
                        a_max_j = (1.0 - cur[j]) if sign > 0 else cur[j]
                        a_max_k = (1.0 - cur[k]) if sign < 0 else cur[k]
                        a = min(a_max_j, a_max_k) * frac
                        if a < 1e-10:
                            continue
                        cur[j] += sign * a
                        cur[k] -= sign * a
                        c5 = _exact_c5(cur)
                        if c5 < best_c5 - 1e-13:
                            best_c5 = c5
                            best = cur.copy()
                            improved = True
                            break
                        else:
                            cur[j] -= sign * a
                            cur[k] += sign * a
                    if improved:
                        break
                if improved:
                    break
        # Random pair moves
        if not improved:
            for _ in range(100):
                if time.time() > deadline:
                    return best, best_c5
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j == k or cur[j] <= 1e-7 or cur[j] >= 1 - 1e-7 or cur[k] <= 1e-7 or cur[k] >= 1 - 1e-7:
                    continue
                sign = 1.0 if g[j] * g[k] < 0 else (-1.0 if rng.random() < 0.5 else 1.0)
                a_max_j = (1.0 - cur[j]) if sign > 0 else cur[j]
                a_max_k = (1.0 - cur[k]) if sign < 0 else cur[k]
                a = min(a_max_j, a_max_k) * 0.5
                if a < 1e-10:
                    continue
                cur[j] += sign * a
                cur[k] -= sign * a
                c5 = _exact_c5(cur)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best = cur.copy()
                    improved = True
                    break
                else:
                    cur[j] -= sign * a
                    cur[k] += sign * a
            if not improved:
                cur = best.copy()
                stall += 1
        else:
            stall = 0
    return best, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_lse(h0, deadline, rng, n_starts=8):
    """Try different perturbations with LSE optimization"""
    n = len(h0)
    best = h0.copy()
    best_c5 = _exact_c5(h0)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        alloc = min(remaining * 0.8 / (n_starts - s), 60.0)
        sub_dl = time.time() + alloc
        r = rng.random()
        if r < 0.3:
            h_try = h0 + rng.normal(0, 0.04, n)
        elif r < 0.5:
            h_try = h0.copy()
            idx = rng.choice(n, size=max(1, n // 6), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.65:
            shift = int(rng.integers(1, max(2, n // 10)))
            h_try = np.roll(h0, shift)
        elif r < 0.8:
            h_try = 0.5 + (h0 - 0.5) * (0.7 + 0.5 * rng.random())
        else:
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.6 * h0 + 0.4 * h_rand
        h_try = _project(h_try, n / 2.0)
        h_opt, c5_opt = _anneal_lse(h_try, sub_dl, base_T=0.04, n_anneal=5)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best = h_opt.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LSE annealing from parent (30% budget)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _anneal_lse(h, dl1, base_T=0.06, n_anneal=10)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Targeted refinement on worst lags (15%)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = _targeted_refine(best_h, dl2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-start with LSE (20%)
    dl3 = t0 + budget_s * 0.65
    h3, c5_3 = _multi_start_lse(best_h, dl3, rng, n_starts=10)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Resolution exploration (20%)
    dl4 = t0 + budget_s * 0.85
    for dn in [8, -8, 16, -16, 24, -24, 32, -32]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 160:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + min(30.0, (dl4 - time.time()) * 0.3))
        h_r, c5_r = _anneal_lse(h_r, sub_dl, base_T=0.05, n_anneal=6)
        if c5_r < best_c5 - 1e-9:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: LSE + targeted on best resolution (10%)
    dl5 = t0 + budget_s * 0.95
    h5, c5_5 = _anneal_lse(best_h, dl5, base_T=0.03, n_anneal=6)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5.copy()
    h5b, c5_5b = _targeted_refine(best_h, min(dl5, time.time() + budget_s * 0.03))
    if c5_5b < best_c5:
        best_c5, best_h = c5_5b, h5b.copy()

    # Phase 6: Pair polish (5%)
    dl6 = t0 + budget_s * 0.99
    h6, c5_6 = _pair_polish(best_h, dl6)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    s = float(np.sum(best_h))
    diff = best_n / 2.0 - s
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
