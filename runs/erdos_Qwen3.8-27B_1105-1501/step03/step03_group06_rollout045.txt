    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for s in range(-(n - 1), n):
        r = n - 1 + s
        # h part: j where 0 <= j+s < n
        jlo = max(0, -s)
        jhi = min(n - 1, n - 1 - s)
        if jlo <= jhi:
            G[r, jlo:jhi+1] += omh[jlo+s:jhi+s+1]
        # (1-h) part: j where 0 <= j-s < n
        jlo2 = max(0, s)
        jhi2 = min(n - 1, s + n - 1)
        if jlo2 <= jhi2:
            G[r, jlo2:jhi2+1] += -h[jlo2-s:jhi2-s+1]
    return G

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = _corr_of(h)
    m = float(np.max(corr))
    vals = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_v = np.exp(vals)
    f = m + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
    w = exp_v / float(np.sum(exp_v))
    G = _grad_matrix(h)
    g = G.T @ w
    g -= np.mean(g)
    return f, g

def _smooth_optimize(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj_and_grad, h, args=(alpha,),
                   jac=True, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40, 'iprint': -1})
    x = np.clip(res.x, 0.0, 1.0)

        c5 = cmax * 2.0 / n

        G = _grad_matrix(h)

            return z[n]
        
        def jac_obj(z):
            gj = np.zeros(n + 1)
            gj[n] = 1.0
            return gj

                    return jv
