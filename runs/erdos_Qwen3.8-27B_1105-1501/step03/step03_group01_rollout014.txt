def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _correlation_rows(h, d):
    """Compute S0, T, C for the line h + t*d.
    S0 = corr(h, 1-h)
    T = corr(d, 1-h) - corr(h, d)  [linear coefficient]
    C = corr(d, d)  [quadratic coefficient, enters as -t^2*C]
    """
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    return S0, T, C

def _line_objective(t, S0, T, C, dx):
    """Evaluate F(t) = dx * max_r (S0_r + t*T_r - t^2*C_r)"""
    vals = S0 + t * T - t * t * C
    return float(np.max(vals)) * dx

def _feasible_interval(h, d):

    # Ensure 0 is in the interval

    t_hi = max(t_hi, 0.0)

def _softmax_direction(h, dx, tau):
    """Compute softmax-weighted gradient direction (to decrease the max)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx  # These are the actual C5 contributions
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # Gradient of each row w.r.t. h
    # Row r corresponds to shift d = r - (n-1)
    # f_r(h) = sum_i h[i] * (1-h[i+d]) for valid i
    # df_r/dh[j] = (1-h[j+d]) for the first sum (when i=j)
    #            - h[j-d] for the second sum (when i+d=j, i.e., i=j-d)
    # So df_r/dh[j] = (1-h[j+d]) - h[j-d] (with boundary handling)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        # For shift d: f_r = sum_i h[i]*(1-h[i+d]) where i and i+d are in [0,n-1]
        # df_r/dh[j]:
        #   From term where i=j: (1-h[j+d]) if j+d valid
        #   From term where i+d=j, i=j-d: -h[j-d] if j-d valid
        # For positive d: i ranges from 0 to n-1-d, so j+d in [0,n-1] means j in [0,n-1-d]
        #   Contribution: (1-h[j+d]) for j in [0, n-1-d]
        #   And -h[j-d] for j-d in [0, n-1-d], i.e., j in [d, n-1]
        # For negative d, let g = -d > 0: i ranges from g to n-1, j+d = j-g
        #   Contribution: (1-h[j-g]) for j in [g, n-1]
        #   And -h[j+g] for j+g in [g, n-1], i.e., j in [0, n-1-g]
        
        if d >= 0:
            m = n - d
            if m > 0:
                for j in range(m):
                    grad[j] += w[r] * (1.0 - h[j + d])
                for j in range(d, n):
                    grad[j] += w[r] * (-h[j - d])
        else:
            g = -d
            m = n - g
            if m > 0:
                for j in range(m):
                    grad[j] += w[r] * (-h[j + g])
                for j in range(g, n):
                    grad[j] += w[r] * (1.0 - h[j - g])
    
    # Make mean zero
    grad -= np.mean(grad)
    return grad

def _topk_direction(h, dx, k, tau):
    """Direction using only top-k correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    # Get top k indices
    idx = np.argsort(scores)[::-1][:k]
    subset_scores = scores[idx]
    exp_s = np.exp(tau * (subset_scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    w_full = np.zeros(2 * n - 1)
    w_full[idx] = w
    
    grad = np.zeros(n)
    for r in idx:
        if w_full[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                for j in range(m):
                    grad[j] += w_full[r] * (1.0 - h[j + d])
                for j in range(d, n):
                    grad[j] += w_full[r] * (-h[j - d])
        else:
            g = -d
            m = n - g
            if m > 0:
                for j in range(m):
                    grad[j] += w_full[r] * (-h[j + g])
                for j in range(g, n):
                    grad[j] += w_full[r] * (1.0 - h[j - g])
    
    grad -= np.mean(grad)
    return grad

def _pair_direction(h, i, j):
    """Direction that moves mass from position i to position j (or vice versa)."""
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _block_direction(h, rng):
    """Random block transfer direction."""
    n = len(h)
    # Pick two random intervals and transfer mass between them
    a = rng.integers(0, n - 1)
    b = rng.integers(0, n - 1)
    length = max(2, n // 10)
    # Source block
    s_start = a % (n - length)
    # Destination block
    d_start = b % (n - length)
    if s_start == d_start:
        return None
    s_end = min(s_start + length, n)
    d_end = min(d_start + length, n)
    
    d = np.zeros(n)
    s_sum = h[s_start:s_end].sum()
    d_sum = h[d_start:d_end].sum()
    # Transfer proportional
    if s_sum > 1e-10:
        d[s_start:s_end] = -h[s_start:s_end] / s_sum
    if d_sum > 1e-10:
        d[d_start:d_end] = h[d_start:d_end] / d_sum
    # The total transfer is min(s_sum, d_sum) but we normalize
    # Actually this doesn't preserve sum. Let me fix.
    # We want sum(d) = 0 and d represents a valid direction.
    # Transfer amount = 1 (normalized)
    # from source to dest
    src_mask = np.zeros(n, dtype=bool)
    dst_mask = np.zeros(n, dtype=bool)
    src_mask[s_start:s_end] = True
    dst_mask[d_start:d_end] = True
    
    n_src = src_mask.sum()
    n_dst = dst_mask.sum()
    if n_src == 0 or n_dst == 0:
        return None
    d = np.zeros(n)
    d[src_mask] = -1.0 / n_src
    d[dst_mask] = 1.0 / n_dst
    d -= d.mean()  # ensure mean zero (should already be close)
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_descent(h, deadline, tau_vals=(50, 100, 200), lr_init=0.02, momentum=0.85):

    target = n / 2.0

    best_h = h.copy()

    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    tau_idx = 0
    
    while time.time() < deadline:

        d = _softmax_direction(h, dx, tau)
        # Normalize direction
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            tau_idx += 1
            mom = np.zeros(n)
            continue
        d = d / d_norm
        
        # Momentum
        mom = momentum * mom + (1.0 - momentum) * d
        mom_norm = np.linalg.norm(mom)
        if mom_norm < 1e-12:
            continue
        mom = mom / mom_norm
        
        # Line search with trust region

                # Try different tau
                tau_idx += 1
                if stall > 80:

def _structured_search(h, deadline):

        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Perturbation + re-optimization (12%)
    d5 = t0 + budget_s * 0.82
    scale = 0.035
    for attempt in range(120):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.14, scale * 1.12)

    # Phase 6: Pair refinement (8%)
    d6 = t0 + budget_s * 0.90
    h_p, c5_p = _pair_refine(best_h, d6, max_passes=35)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 7: Final LP polish
    d7 = t0 + budget_s * 0.97
    h_f, c5_f = _lp_loop(best_h, d7, radius_init=0.04)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
