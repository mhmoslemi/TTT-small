def _project(v, target):

        else:

def _grad_matrix(h):

                G[r, :m] += -h[d:]
        else:

            m = n - g

                G[r, :m] += (1.0 - h[g:])

def _sim_anneal(h, deadline, rng, n_iters=50000, t_start=0.004, t_end=0.0001):

        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        # Transfer amount

            delta = min(h[j] - h[k], rng.uniform(0, 0.15))

            hk_new = h[k] + delta

        h_old_j = h[j]
        h_old_k = h[k]

        h[k] = hk_new

def _resample(h, n_new):

        if len(active) < 2:
            break
        G = _grad_matrix(h)
        G_a = G[active]

        # s.t. sum(p) = 0

        # Using augmented system
        lam = 0.5

        M[:n, n] = 1.0

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _softmax_weights(scores, tau):
    mx = np.max(scores)
    e = np.exp((scores - mx) / tau)
    s = np.sum(e)
    if s < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = e / s
    return w

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _resample(h, n_new):

            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None], radius_init=0.08, max_iters=150)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r; best_h = h_r.copy(); best_n = n_try

    # Phase 10: Final ultra-fine polish
    if time.time() < t0 + budget_s * 0.94:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.995, top_k_list=[3, 5, 4], radius_init=0.008, max_iters=4000)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
