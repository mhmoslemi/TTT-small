```python
def _corr_jac(h):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    one_minus_h = 1.0 - h
    for d in range(-(n-1), n):
        i = d + n - 1
        j_min = max(0, -d)
        j_max = min(n-1, n-1-d)
        m = j_max - j_min + 1
        if m <= 0:
            continue
        # Terms where j+d = k, i.e., k = j+d for j in [j_min, j_max]
        # k ranges over [j_min+d, j_max+d]
        k_start1 = j_min + d  # = max(0,-d) + d
        k_end1 = j_max + d    # = min(n-1, n-1-d) + d
        # Terms where j = k, i.e., k in [j_min, j_max]
        k_start2 = j_min
        k_end2 = j_max
        
        # Vectorized: for k in [k_start1, k_end1], add (1-h[k-d])
        # For k in [k_start2, k_end2], subtract h[k+d]
        
        # First term
        idx1 = np.arange(k_start1, k_end1 + 1)
        G[i, idx1] += one_minus_h[idx1 - d]
        
        # Second term
        idx2 = np.arange(k_start2, k_end2 + 1)
        G[i, idx2] -= h[idx2 + d]
    
    return G
```

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_jac(h):
    """Jacobian of the full correlation vector w.r.t. h."""
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    omh = 1.0 - h
    for d in range(-(n - 1), n):
        i = d + n - 1
        jm = max(0, -d)
        jM = min(n - 1, n - 1 - d)
        if jM < jm:
            continue
        k1 = np.arange(jm + d, jM + d + 1)
        G[i, k1] += omh[k1 - d]
        k2 = np.arange(jm, jM + 1)
        G[i, k2] -= h[k2 + d]
    return G


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
