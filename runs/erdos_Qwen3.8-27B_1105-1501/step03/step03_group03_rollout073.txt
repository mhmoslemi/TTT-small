    h_test[j] = h[j] - c1
    f1 = np.max(_scores(h_test))
    h_test[i] = h[i] + c2
    h_test[j] = h[j] - c2
    f2 = np.max(_scores(h_test))

            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            h_test[i] = h[i] + c1
            h_test[j] = h[j] - c1
            f1 = np.max(_scores(h_test))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            h_test[i] = h[i] + c2
            h_test[j] = h[j] - c2
            f2 = np.max(_scores(h_test))
    
    opt_t = (lo + hi) / 2.0
    h_test[i] = h[i] + opt_t
    h_test[j] = h[j] - opt_t
    opt_score = np.max(_scores(h_test))
    
    if opt_score < best_score:
        best_t = opt_t
        best_score = opt_score
    
    if best_t > 1e-14:
        h[i] += best_t
        h[j] -= best_t
    return best_t, h
