            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_step(h, scores, G, K_active, radius):
    """Solve LP for descent direction using top K active lags."""
    n = len(h)
    s_max = float(np.max(scores))
    # Select active set
    if K_active >= 2*n - 1:
        active = np.arange(2*n - 1)
    else:
        active = np.argsort(scores)[::-1][:K_active]
    
    G_act = G[active, :]
    s_act = scores[active]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < s_max - 1e-10:
            return res.x[:n], res.fun
    except Exception:
        pass
    return None, None
