import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_gradient(h, p, scores):
    n = len(h)
    G = _grad_matrix(h)
    eps = 1e-15
    s = np.maximum(scores, eps)
    w = s ** (p - 1.0)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _line_search(h, d, target, c5_cur, n_coarse=16, n_fine=28):
    n = len(h)
    if float(np.max(np.abs(d))) < 1e-15:
        return 0.0, c5_cur
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-14)
    alphas = np.linspace(0.0, alpha_max, n_coarse)
    vals = np.array([_exact_c5(h + a * d) for a in alphas])
    bi = int(np.argmin(vals))
    if bi == 0:
        return 0.0, c5_cur
    if bi == n_coarse - 1:
        return alphas[-1], vals[-1]
    lo = alphas[bi - 1]
    hi = alphas[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(n_fine):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    best_a = alphas[bi]
    best_v = vals[bi]
    if opt_c5 < best_v:
        return opt_a, opt_c5
    return best_a, best_v

def _pnorm_anneal(h, deadline, p_start=6, p_end=300, n_steps=500, target_c5=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if target_c5 is not None and best_c5 <= target_c5:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        grad = _pnorm_gradient(h, p, scores)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-16:
            stall += 5
            continue
        d = -grad / gn
        alpha, new_c5 = _line_search(h, d, target, cur_c5)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 30:
                h = _project(h + 0.01 * np.random.randn(n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
        if stall > 80:
            break
    return best_h, best_c5

def _active_direction(h, top_k=6):
    n = len(h)
    scores = _scores(h)
    idx = np.argsort(scores)[-top_k:]
    G = _grad_matrix(h)
    A = G[idx]
    d = -A.sum(axis=0)
    d -= np.mean(d)
    dn = float(np.linalg.norm(d))
    if dn < 1e-15:
        return None
    d /= dn
    return d

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _binary_sa(h_start, deadline, n_iters=80000, T0=None):
    n = len(h_start)
    if n % 2 != 0:
        return h_start.copy(), _exact_c5(h_start)
    target = n // 2
    b = (h_start > 0.5).astype(float)
    if int(b.sum()) > target:
        idx = np.where(b > 0.5)[0]
        np.random.shuffle(idx)
        for i in range(int(b.sum()) - target):
            b[idx[i]] = 0.0
    elif int(b.sum()) < target:
        idx = np.where(b < 0.5)[0]
        np.random.shuffle(idx)
        for i in range(target - int(b.sum())):
            b[idx[i]] = 1.0
    cur_c5 = _exact_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    if T0 is None:
        T0 = 0.003 * cur_c5
    T = T0
    decay = (1e-6 * cur_c5 / T0) ** (1.0 / n_iters)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        if b[i] > 0.5:
            j = np.random.choice(np.where(b < 0.5)[0])
            b[i] = 0.0
            b[j] = 1.0
        else:
            j = np.random.choice(np.where(b > 0.5)[0])
            b[i] = 1.0
            b[j] = 0.0
        new_c5 = _exact_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()
        else:
            if b[i] > 0.5:
                b[i] = 1.0
                b[j] = 0.0
            else:
                b[i] = 0.0
                b[j] = 1.0
        T *= decay
    return best_b, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _make_structured_init(n, rng, variant):

        h = np.zeros(n)

    elif variant == 1:
        h = np.zeros(n)
        period = rng.integers(3, 8)
        duty = rng.uniform(0.3, 0.7)
        for i in range(n):
            if (i % period) < int(duty * period):
                h[i] = 1.0
        return _project(h, n / 2.0)
    elif variant == 2:
        h = np.zeros(n)
        k = rng.integers(2, 5)
        offsets = rng.integers(0, n, size=k)
        lengths = [int(rng.uniform(0.1, 0.3) * n) for _ in range(k)]
        for off, ln in zip(offsets, lengths):
            for i in range(ln):
                h[(off + i) % n] = 1.0
        return _project(h, n / 2.0)
    else:
        h = rng.uniform(0, 1, n)
        return _project(h, n / 2.0)

            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None], radius_init=0.1, max_iters=2000)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final: ultra-fine LP polish
    if time.time() < t0 + budget_s * 0.995:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.998, top_k_list=[3, 4, 5], radius_init=0.01, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
