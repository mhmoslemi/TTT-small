def _exact_c5(h):

def _lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

        h0 = _gen_binary(64, rng)
    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = _project(h0, target0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: LP descent from initial (30% of budget)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_descent(best_h, dl1, target0, R_init=3.0, max_iters=10000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair improvement (10%)
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _pair_improve(best_h, dl2, best_n / 2.0, max_rounds=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP descent again from improved point (15%)
    dl3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_descent(best_h, dl3, best_n / 2.0, R_init=2.0, max_iters=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-start (20%)
    dl4 = t0 + budget_s * 0.75
    n_starts = 12
    for s in range(n_starts):
        if time.time() > dl4:
            break
        remaining = dl4 - time.time()
        if remaining < 2.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 15.0)
        sub_dl = time.time() + alloc

        r = rng.random()
        if r < 0.3:
            h_try = _gen_binary(best_n, rng)
        elif r < 0.5:
            h_try = _gen_smooth(best_n, rng)
        elif r < 0.7:
            h_try = _gen_perturb(best_h, rng, scale=0.08)
        elif r < 0.85:
            h_try = _gen_perturb(best_h, rng, scale=0.15)
        else:
            h_rand = (rng.random(best_n) > 0.5).astype(float)
            h_try = _project(0.5 * best_h + 0.5 * h_rand, best_n / 2.0)

        h_try = _project(h_try, best_n / 2.0)
        h_opt, c5_opt = _lp_descent(h_try, sub_dl, best_n / 2.0, R_init=2.5, max_iters=3000)
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Resolution sweep (12%)
    dl5 = t0 + budget_s * 0.87
    try_ns = []
    for delta in [-2, -4, -8, -16, 2, 4, 8, 16, -12, 12, -6, 6]:
        nt = best_n + delta
        if 20 <= nt <= 160 and nt % 2 == 0:
            try_ns.append(nt)
    for nt in try_ns:
        if time.time() > dl5:
            break
        remaining = dl5 - time.time()
        if remaining < 3.0:
            break
        sub_dl = time.time() + min(remaining * 0.6, 12.0)
        h_r = _resample(best_h, nt)
        h_r, c5_r = _lp_descent(h_r, sub_dl, nt / 2.0, R_init=2.0, max_iters=2000)
        # Quick pair polish
        h_r, c5_r = _pair_improve(h_r, min(dl5, time.time() + 3.0), nt / 2.0, max_rounds=10)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt

    # Phase 6: Final LP polish (10%)
    dl6 = t0 + budget_s * 0.97
    h6, c5_6 = _lp_descent(best_h, dl6, best_n / 2.0, R_init=1.5, max_iters=5000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final pair polish
    dl7 = t0 + budget_s * 0.99
    h7, c5_7 = _pair_improve(best_h, dl7, best_n / 2.0, max_rounds=20)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        # Fallback
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
