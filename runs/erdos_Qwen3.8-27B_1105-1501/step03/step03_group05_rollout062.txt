```python
def _build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_minus_h = 1.0 - h
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += dx * one_minus_h[:m]
                G[r, :m] -= dx * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += dx * one_minus_h[:m]
                G[r, :m] -= dx * h[g:]
    return G
```

- Terms where p = j: +δ * (1-h[j+d]) (if j+d is in range)
- Terms where p+d = j, i.e., p = j-d: h[j-d] * (1-(h[j]+δ)) - h[j-d] * (1-h[j]) = -h[j-d] * δ (if j-d is in range)

So the total change in score at lag d is:
ΔS(d) = δ * [h[i-d] * I(i-d valid) - (1-h[i+d]) * I(i+d valid) + (1-h[j+d]) * I(j+d valid) - h[j-d] * I(j-d valid)]

This is linear in δ with coefficient:
c(d) = h[i-d]*I(i-d valid) - (1-h[i+d])*(i+d valid) + (1-h[j+d])*(j+d valid) - h[j-d]*I(j-d valid)

(all with appropriate boundary conditions)

Then the new score at lag d is: S(d) + c(d) * δ (times dx for the scaled version)

We want to minimize max_d (S(d) + c(d)*δ) * dx over δ ∈ [δ_min, δ_max].

This is a 1D convex optimization (max of linear functions = convex). The minimum is at a breakpoint or a boundary.

To find the minimum:
1. Compute all breakpoints: δ where S(d) + c(d)*δ = S(e) + c(e)*δ for some d ≠ e
   → δ = (S(e) - S(d)) / (c(d) - c(e)) for c(d) ≠ c(e)
2. Filter breakpoints to [δ_min, δ_max]
3. Evaluate the max at each breakpoint and at the boundaries
4. Take the argmin

This is O(L²) per pair, where L = 2n-1. For n=64, L=127, so L² ≈ 16k per pair. With ~2000 pairs, that's ~32 million operations per full pass. In Python, this might take a few seconds.

```python
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = h + a * d
        h_t = _project_box_sum(h_t, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5
    lo = max(0.0, best_a - alpha_max / 16)
    hi = min(alpha_max, best_a + alpha_max / 16)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h1 = _project_box_sum(h + c1 * d, target)
    h2 = _project_box_sum(h + c2 * d, target)
    f1 = _exact_c5(h1)
    f2 = _exact_c5(h2)
    for _ in range(14):
        if hi - lo < 1e-10 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            h1 = _project_box_sum(h + c1 * d, target)
            f1 = _exact_c5(h1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            h2 = _project_box_sum(h + c2 * d, target)
            f2 = _exact_c5(h2)
    opt_a = (lo + hi) / 2.0
    h_opt = _project_box_sum(h + opt_a * d, target)
    opt_c5 = _exact_c5(h_opt)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _gradient_loop(h, deadline, beta_start=5.0, beta_end=200.0, n_steps=12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    betas = np.geomspace(beta_start, beta_end, n_steps)
    for beta in betas:
        if time.time() > deadline:
            break
        step_size = 0.15
        stall = 0
        for _ in range(3000):
            if time.time() > deadline:
                break
            g = _smooth_grad(h, beta)
            g -= np.mean(g)
            norm = float(np.max(np.abs(g)))
            if norm < 1e-14:
                break
            d = -g / norm * step_size
            a, new_c5 = _line_search(h, d, cur_c5, target)
            if new_c5 < cur_c5 - 1e-12:
                h = h + a * d
                h = _project_box_sum(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step_size = min(step_size * 1.08, 0.5)
            else:
                stall += 1
                step_size *= 0.7
                if stall > 80:
                    break
    return best_h, best_c5

def _pair_sweep(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    while time.time() < deadline and no_improve < 5:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                if h[j] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-10:
                    continue
                best_d = 0.0
                best_val = cur_c5
                for frac in [0.5, 0.3, 0.7, 0.15, 0.85]:
                    delta = delta_max * frac
                    h_t = h.copy()
                    h_t[i] -= delta
                    h_t[j] += delta
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_val - 1e-13:
                        best_val = c5_t
                        best_d = delta
                if best_d > 0:
                    h[i] -= best_d
                    h[j] += best_d
                    cur_c5 = best_val
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    return best_h, best_c5

def _binary_sa(h, deadline, rng, n_iter=50000):
    n = len(h)
    target = n / 2.0
    idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    cur_c5 = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    ones = list(np.where(h_b == 1.0)[0])
    zeros = list(np.where(h_b == 0.0)[0])
    temp = 0.05
    no_improve = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        c5_new = _exact_c5(h_b)
        delta_c5 = c5_new - cur_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(temp, 1e-10)):
            cur_c5 = c5_new
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
                no_improve = 0
            temp *= 0.9997
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def _resample_integral(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old_edges = np.linspace(0, 2, n_old + 1)
    x_new_edges = np.linspace(0, 2, n_new + 1)
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a, b = x_new_edges[j], x_new_edges[j + 1]
        mass = 0.0
        for i in range(n_old):
            lo = max(a, x_old_edges[i])
            hi = min(b, x_old_edges[i + 1])
            if hi > lo:
                mass += (hi - lo) * h_old[i]
        h_new[j] = mass / (b - a)
    return _project_box_sum(h_new, n_new / 2.0)

def _refine_continuous_from_binary(h_bin, deadline, target):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    for _ in range(5):
        if time.time() > deadline:
            break
        h, c5 = _cvx_loop(h, min(deadline, time.time() + 30), sigma_start=0.04)
        h, c5_g = _gradient_loop(h, min(deadline, time.time() + 30), beta_start=10, beta_end=100, n_steps=6)
        c5_after = _exact_c5(h)
        if c5_after < _exact_c5(np.zeros(n)):
            break
    return h, _exact_c5(h)

def _lp_local(h, deadline, radius_init=0.08):
    from scipy.optimize import linprog
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 40:
        scores = _scores_scaled(h)
        G = _build_G(h)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = []
        for j in range(n):
            lb = max(-radius, -h[j])
            ub = min(radius, 1.0 - h[j])
            bounds.append((lb, ub))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
            if res.success:
                p = res.x[:n]
                if float(np.max(np.abs(p))) > 1e-12:
                    a, new_c5 = _line_search(h, p, cur_c5, target)
                    if new_c5 < cur_c5 - 1e-11:
                        h = h + a * p
                        h = _project_box_sum(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.15, 0.3)
                        continue
        except Exception:
            pass
        stall += 1
        radius *= 0.75
        if radius < 1e-5:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_end = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h = (np.arange(64) % 2).astype(float)

    n = len(h)
    target = n / 2.0
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: CVXPY trust-region (main new method)
    deadline1 = min(t0 + budget_s * 0.30, safe_end)
    h1, c5_1 = _cvx_loop(h, deadline1, sigma_start=0.07, sigma_max=0.35)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient refinement
    deadline2 = min(t0 + budget_s * 0.42, safe_end)
    h2, c5_2 = _gradient_loop(best_h, deadline2, beta_start=5.0, beta_end=150.0, n_steps=12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: CVXPY again from improved point
    deadline3 = min(t0 + budget_s * 0.55, safe_end)
    h3, c5_3 = _cvx_loop(best_h, deadline3, sigma_start=0.04, sigma_max=0.25)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP local refinement
    deadline4 = min(t0 + budget_s * 0.65, safe_end)
    h4, c5_4 = _lp_local(best_h, deadline4, radius_init=0.1)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair sweep
    deadline5 = min(t0 + budget_s * 0.75, safe_end)
    h5, c5_5 = _pair_sweep(best_h, deadline5, target)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary SA + continuous refinement
    deadline6 = min(t0 + budget_s * 0.82, safe_end)
    hb, c5b = _binary_sa(best_h, deadline6, rng, n_iter=40000)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()
        if time.time() < safe_end:
            deadline6b = min(t0 + budget_s * 0.86, safe_end)
            hr, c5r = _refine_continuous_from_binary(best_h, deadline6b, target)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr.copy()

    # Phase 7: Multi-resolution
    deadline7 = min(t0 + budget_s * 0.93, safe_end)
    n_candidates = sorted(set(x for x in [best_n - 8, best_n - 4, best_n + 4,
                                            best_n + 8, best_n + 16, 48, 56, 80, 96]
                               if 24 <= x <= 128 and x % 2 == 0))
    per_n_time = budget_s * 0.02
    for n_try in n_candidates:
        if time.time() > deadline7:
            break
        h_r = _resample_integral(best_h, n_try)
        deadline_r = min(deadline7, time.time() + per_n_time)
        h_r, c5_r = _cvx_loop(h_r, deadline_r, sigma_start=0.05, sigma_max=0.3)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final CVXPY polish
    if time.time() < t0 + budget_s * 0.96:
        h_f, c5_f = _cvx_loop(best_h, min(t0 + budget_s * 0.98, safe_end),
                               sigma_start=0.02, sigma_max=0.1)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final projection and validation
    best_n = len(best_h)
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, best_n
```
