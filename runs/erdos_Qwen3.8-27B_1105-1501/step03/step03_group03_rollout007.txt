def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    # d = 0
    G[n-1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        # positive lag d
        G[n-1+d, :m] -= h[d:]
        G[n-1+d, d:] += 1.0 - h[:m]
        # negative lag -d
        G[n-1-d, :m] += 1.0 - h[d:]
        G[n-1-d, d:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -10.0, 10.0
    for _ in range(100):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)

            x = np.clip(x, 0.0, 1.0)
