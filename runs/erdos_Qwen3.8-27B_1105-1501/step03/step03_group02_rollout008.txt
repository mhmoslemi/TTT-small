    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_active(h, deadline, top_k=40, maxiter=150, ftol=1e-12):
    """SLSQP with growing active set."""
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-4

    active_lags = [int(k - (n - 1)) for k in active_idx]
    
    def build_constraints(lags):
        cons = [{
            'type': 'eq',
            'fun': lambda x: np.sum(x[:n]) - target,
            'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
        }]
        for d in lags:
            cons.append({
                'type': 'ineq',
                'fun': lambda x, dd=d: x[n] - _eval_score(x[:n], dd),
                'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])

            break

        z0 = float(np.max(scores)) + 1e-5

        def obj(x):
            return x[n]
        
        def obj_jac(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def cb(xk):
            if time.time() > deadline:
                raise RuntimeError("timeout")
        
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                          callback=cb)
        except (RuntimeError, Exception):
            break

        if abs(np.sum(h_new) - target) > 1e-6:
            h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        c5_old = _exact_c5(h_cur)
        
        if c5_new < c5_old - 1e-14:
            h_cur = h_new
        else:
            break

def _eval_score(h, d):

    elif d > 0:

    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _smoothed_gradient_step(h, deadline, beta=30.0, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        w = np.exp(beta * (scores - max_s))
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        w /= wsum
        
        # Weighted gradient
        grad = np.zeros(n)
        for idx in range(len(scores)):
            if w[idx] > 1e-6:
                d = idx - (n - 1)
                grad += w[idx] * _score_grad(h, d)
        
        # Project onto sum=0
        grad -= np.mean(grad)
        nrm = float(np.sqrt(np.sum(grad**2)))
        if nrm < 1e-15:
            break
        grad /= nrm  # unit direction
        
        # Step direction is -grad (decrease smoothed max)
        # Find max step
        pos = grad > 1e-12
        neg = grad < -1e-12
        alpha_max = 1.0
        if np.any(pos):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / grad[pos])))
        if np.any(neg):
            alpha_max = min(alpha_max, float(np.min((-h[neg]) / grad[neg])))
        alpha_max = max(alpha_max, 1e-10)
        
        # Try several step sizes
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a = alpha_max * frac
            h_try = h - a * grad
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            beta = max(5.0, beta * 0.9)
            if beta < 5.0:
                break
    
    return best_h, best_c5

```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] -= h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] -= h[:m]
    return grad

def _eval_score(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.5
    hi = float(v.max()) + 1.5
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _slsqp_round(h, deadline, top_k=40, maxiter=100, ftol=1e-12):
    """Single SLSQP solve with given active set."""
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-4
    active_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in active_idx]
    
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
    
    obj = lambda x: x[n]
    obj_jac = lambda x: np.concatenate([np.zeros(n), [1.0]])
    
    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    }]
    for d in active_lags:
        cons.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - _eval_score(x[:n], dd),
            'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        })
    
    def cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                      callback=cb)
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        if abs(np.sum(h_new) - target) > 1e-6:
            h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    except Exception:
        return None

def _slsqp_growing(h, deadline, top_k_init=30, max_rounds=8):
    """SLSQP with growing active set."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    top_k = top_k_init
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        result = _slsqp_round(h_cur, min(deadline, time.time() + 30), top_k=top_k, maxiter=80, ftol=1e-11)
        if result is None:
            break
        h_new, c5_new = result
        if c5_new < best_c5 - 1e-14:
            h_cur = h_new
            best_c5 = c5_new
            best_h = h_cur.copy()
            # Grow active set
            top_k = min(2 * n - 1, top_k + 10)
        else:
            break
    
    return best_h, best_c5

def _pairwise_opt(h, deadline, max_sweeps=300, n_trials=100, rng=None):
    """Pairwise coordinate descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if rng is None:
        rng = np.random.default_rng(999)
    
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline or stall > 30:
            break
        improved = False
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-12:
                continue
            best_t = 0.0
            best_local = best_c5
            for frac in [0.2, 0.4, 0.5, 0.6, 0.8]:
                t = t_lo + frac * (t_hi - t_lo)
                h[i] += t
                h[j] -= t
                c5_t = _exact_c5(h)
                if c5_t < best_local:
                    best_local = c5_t
                    best_t = t
                h[i] -= t
                h[j] += t
            if best_t != 0.0 and best_local < best_c5 - 1e-13:
                h[i] += best_t
                h[j] -= best_t
                best_c5 = best_local
                best_h = h.copy()
                improved = True
                stall = 0
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
