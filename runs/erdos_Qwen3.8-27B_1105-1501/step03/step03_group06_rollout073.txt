    """Gradient of corr[k_idx] with respect to h.
    corr = np.correlate(h, 1-h, 'full')
    corr[k] = sum_n h[n+k_offset]*(1-h[n]) where k_offset = k - (n-1)
    """

    d = k_idx - (n - 1)  # the lag

        m = n - d
        # d/dh[i] of this:
        # If i appears as the first factor (h[n+d] where n+d=i, so n=i-d):
        #   contributes (1-h[i-d]) if 0 <= i-d < m, i.e., d <= i < d+m = n
        # If i appears as the second factor (1-h[n] where n=i):
        #   contributes -h[i+d] if 0 <= i < m and i+d < n, i.e., 0 <= i < m and i < n-d
        #   Since i < m = n-d, i+d < n is automatic.
        if d > 0:

            g[d:] += (1.0 - h[:m])
            # Second factor: i from 0 to m-1, contributes -h[i+d]
            g[:m] += -h[d:]
        else:

            if stall % 15 == 0 and radius < 0.03:
                radius = 0.12
    
    return best_h, best_c5


def _lp_loop_strict(h, deadline, radius_init=0.15, max_iters=3000, stall_limit=60):
    """Strict LP loop (parent-style)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        scores = _corr(h)
        G = _grad_matrix(h)
        p, t_val = _lp_strict(h, G, scores, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.08
    
    return best_h, best_c5


def _binary_round(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b


def _binary_local_search(b, deadline, max_passes=300):
    """Local search on binary sequences by swapping 1s and 0s."""
    n = len(b)
    b = b.copy()
    best_c5 = _exact_c5(b)
    rng = np.random.default_rng(123)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Generate random pairs to try
        n_try = min(2000, len(ones) * len(zeros))
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            b[i], b[j] = b[j], b[i]
            c5 = _exact_c5(b)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
            else:
                b[i], b[j] = b[i], b[j]
        if not improved:
            # Try all pairs (deterministic)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            any_improve = False
            for i in ones:
                for j in zeros:
                    b[i], b[j] = b[j], b[i]
                    c5 = _exact_c5(b)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        any_improve = True
                        improved = True
                        ones = np.where(b == 1)[0]
                        zeros = np.where(b == 0)[0]
                    else:
                        b[i], b[j] = b[i], b[j]
            if not any_improve:
                break
    return b, best_c5


def _smooth_lag_step(h, alpha, maxiter=150):
    """Smoothed max optimization step."""
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        s = _corr(x)
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -400.0, 400.0)
        ev = np.exp(vals)
        sv = float(np.sum(ev))
        f = ms + (1.0 / alpha) * np.log(sv)
        w = ev / sv
        G = _grad_matrix(x)
        g = G.T @ w
        g -= np.mean(g)
        return f, g
    
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 10:
        return None
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _coordinate_descent(h, deadline, max_passes=50):
    """Try moving each coordinate to 0 or 1 (nearest boundary)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for target_val in [0.0, 1.0]:
                if abs(h[j] - target_val) < 1e-10:
                    continue
                h_t = h.copy()
                h_t[j] = target_val
                h_t = _project(h_t, target)
                c5_t = _exact_c5(h_t)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    h = h_t
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Relaxed LP optimization (15% budget)
    d1 = t0 + budget_s * 0.15
    h, c5 = _lp_loop_relaxed(h, d1, radius_init=0.2, max_iters=3000, stall_limit=60)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    
    # Phase 2: Strict LP from current best (10% budget)
    d2 = t0 + budget_s * 0.25
    h, c5 = _lp_loop_strict(best_h, d2, radius_init=0.1, max_iters=2000, stall_limit=50)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    
    # Phase 3: Smooth optimization with multiple alphas (15% budget)
    d3 = t0 + budget_s * 0.40
    for alpha in [10.0, 25.0, 50.0, 100.0, 200.0]:
        if time.time() > d3:
            break
        h_try, c5_try = _smooth_lag_step(best_h, alpha, maxiter=100)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
        # Follow with LP
        if time.time() < d3:
            sub_d = min(d3, time.time() + budget_s * 0.02)
            h_try, c5_try = _lp_loop_relaxed(best_h, sub_d, radius_init=0.08, max_iters=500, stall_limit=30)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
    
    # Phase 4: Binary local search to find good combinatorial structures (15% budget)
    d4 = t0 + budget_s * 0.55
    b = _binary_round(best_h)
    b, c5_b = _binary_local_search(b, d4, max_passes=200)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, b.copy(), len(b)
    
    # Also try a random binary starting point
    if time.time() < d4:
        b_rand = np.zeros(n0)
        perm = rng.permutation(n0)
        b_rand[perm[:n0 // 2]] = 1.0
        b_r, c5_r = _binary_local_search(b_rand, d4, max_passes=100)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, b_r.copy(), len(b_r)
    
    # Phase 5: Multi-restart with diverse perturbations (20% budget)
    d5 = t0 + budget_s * 0.75
    scale = 0.05
    for attempt in range(500):
        if time.time() > d5:
            break
        if best_c5 <= 0.38080:
            break
        # Diverse perturbation
        r = rng.random()
        if r < 0.4:
            h_p = best_h + rng.normal(0, scale, best_n)
        elif r < 0.7:
            h_p = best_h + rng.normal(0, scale * 0.5, best_n)
            h_p = np.where(h_p > 0.3 and h_p < 0.7, 0.5, h_p)
        else:
            # Binary perturbation
            b_p = _binary_round(best_h)
            n_flip = max(1, best_n // 16)
            flip_idx = rng.choice(best_n, size=n_flip, replace=False)
            b_p[flip_idx] = 1.0 - b_p[flip_idx]
            h_p = b_p.astype(float)
        
        h_p = _project(h_p, best_n / 2.0)
        c5_p = _exact_c5(h_p)
        
        # Quick LP step
        sub_d = min(d5, time.time() + budget_s * 0.02)
        h_try, c5_try = _lp_loop_relaxed(h_p, sub_d, radius_init=0.12, max_iters=800, stall_limit=40)
        if c5_try < best_c5 - 1e-12:
            best_c5, best_h = c5_try, h_try.copy()
            scale = max(0.01, scale * 0.9)
        else:
            scale = min(0.25, scale * 1.1)
    
    # Phase 6: Resolution exploration (15% budget)
    d6 = t0 + budget_s * 0.90
    for dn in [2, -2, 4, -4, 6, 8, -6, 16, -8, 32, -16, 12, 24, -12, -24, 48]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 300:
            continue
        h_r = _resample(best_h, n_try)
        if h_r is None:
            continue
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _smooth_lag_step(h_r, 30.0, maxiter=80)
        if time.time() < d6:
            sub_d2 = min(d6, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop_relaxed(h_r, sub_d2, radius_init=0.1, max_iters=400, stall_limit=30)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try
    
    # Phase 7: Coordinate descent + final LP (10% budget)
    d7 = t0 + budget_s * 0.97
    h_cd, c5_cd = _coordinate_descent(best_h, d7, max_passes=30)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()
    
    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop_relaxed(best_h, t0 + budget_s * 0.995,
                                     radius_init=0.05, max_iters=2000, stall_limit=100)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()
    
    # Final strict LP
    if time.time() < t0 + budget_s * 0.999:
        h_f2, c5_f2 = _lp_loop_strict(best_h, t0 + budget_s * 0.999,
                                      radius_init=0.03, max_iters=1000, stall_limit=50)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
