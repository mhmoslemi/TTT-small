        if c5_opt < best_c5 - 1e-14:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 4: Different resolutions (15%)
    dl4 = t0 + budget_s * 0.90
    for dn in [-8, -4, 4, 8, -12, 12, -16, 16]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 120 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + min(20.0, (dl4 - time.time()) * 0.35))
        h_r, c5_r = _pair_optimize(h_r, sub_dl, n_try / 2.0, max_rounds=60)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Final intensification (5%)
    dl5 = t0 + budget_s * 0.97
    h5, c5_5 = _pair_optimize(best_h, dl5, best_n / 2.0, max_rounds=100)
    if c5_5 < best_c5 - 1e-14:
        best_c5, best_h = c5_5, h5.copy()
    if time.time() < dl5:
        h5b, c5_5b = _slsqp_polish(best_h, dl5, best_n / 2.0, T=0.1)
        if c5_5b < best_c5 - 1e-14:
            best_c5, best_h = c5_5b, h5b.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    s = float(np.sum(best_h))
    if abs(s - best_n / 2.0) > 1e-8:
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
