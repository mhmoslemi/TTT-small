3. Try all possible single-bit flips (change h[j] from 1 to 0 or 0 to 1, maintaining sum constraint by also flipping another bit)
4. Accept the flip that most reduces the maximum overlap

Actually, for binary with fixed sum, a "move" is: choose positions i (where h[i]=1) and j (where h[j]=0), swap them. This changes h[i] from 1 to 0 and h[j] from 0 to 1.

The effect on overlap at offset d:
- If the move affects positions i and j, it changes a_d by some amount depending on the neighbors of i and j.

For efficiency, I can precompute the "impact" of each possible swap. But for n=64, there are C(32,1) * C(32,1) = 1024 possible swaps, and each requires O(n) to evaluate. So each full pass is O(n³) = O(262144), which is fast.

import numpy as np
import time
import math
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lsq_grad(h, beta):
    """Analytical gradient of log-sum-exp smoothed C5"""
    n = len(h)
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * out * dx
    maxv = float(np.max(vals))
    exp_v = np.exp(vals - maxv)
    w = exp_v / float(np.sum(exp_v))
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            if d == 0:
                grad[:] += w[r] * dx * (1.0 - 2.0 * h)
            else:
                grad[d:] += w[r] * dx * (1.0 - h[:m])
                grad[:m] -= w[r] * dx * h[d:]
        else:
            m = -d
            ml = n - m
            if ml <= 0:
                continue
            grad[:ml] += w[r] * dx * (1.0 - h[m:])
            grad[m:] -= w[r] * dx * h[:ml]
    
    return grad

def _slsqp_optimize(h, target, deadline, beta_start=30.0, max_beta=500.0):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    constraints = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    beta = beta_start
    while time.time() < deadline:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            vals = beta * corr * (2.0 / n)
            maxv = float(np.max(vals))
            return float(np.log(np.sum(np.exp(vals - maxv))) + maxv) / beta
        
        res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-12})
        
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_try = _project(res.x, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try.copy()
        
        beta *= 2.0
        if beta > max_beta:
            break
    
    return best_h, best_c5

def _binary_search(h, target, deadline, max_passes=500):
    """Greedy best-improvement binary swap search"""
    n = len(h)
    h = _project(h, target)
    # Binarize
    h_b = (h >= 0.5).astype(float)
    s = int(np.round(np.sum(h_b)))
    if s > int(target):
        ones = np.where(h_b > 0.5)[0]
        # Remove ones with highest "cost"
        for idx in ones[int(target):]:
            h_b[idx] = 0.0
    elif s < int(target):
        zeros = np.where(h_b < 0.5)[0]
        for idx in zeros[:int(target) - s]:
            h_b[idx] = 1.0
    
    best_h = h_b.copy()
    best_c5 = _exact_c5(h_b)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        ones = list(np.where(h_b > 0.5)[0])
        zeros = list(np.where(h_b < 0.5)[0])
        if not ones or not zeros:
            break
        
        best_delta = 1e-10
        best_move = None
        
        for i in ones:
            for j in zeros:
                h_b[i] = 0.0
                h_b[j] = 1.0
                c5_new = _exact_c5(h_b)
                delta = c5_new - best_c5
                if delta < best_delta:

                    best_move = (i, j)

                h_b[j] = 0.0
        
        if best_move is not None:
            i, j = best_move
            h_b[i] = 0.0
            h_b[j] = 1.0
            best_c5 += best_delta
            best_h = h_b.copy()
        else:
            break
    
    return best_h, best_c5

def _milp_binary(n, time_limit=30):
    """Solve binary MIQP for given n using cvxpy"""
    import cvxpy as cp
    
    target = n // 2
    x = cp.Variable(n, boolean=True)
    M = cp.Variable(nonneg=True)
    
    constraints = [cp.sum(x) == target]
    
    # For each offset d = 1 to n-1, compute a_d and a_{-d}
    # a_d = sum_{i=0}^{n-1-d} x[i]*(1-x[i+d]) = sum x[i] - sum x[i]*x[i+d]
    # a_{-d} = sum_{i=0}^{n-1-d} x[i+d]*(1-x[i]) = sum x[i+d] - sum x[i]*x[i+d]
    
    for d in range(1, n):
        m = n - d
        # y[i,d] = x[i] * x[i+d]
        y = cp.Variable(m, boolean=True)
        # Linearization
        constraints += [y <= x[:m]]
        constraints += [y <= x[d:]]
        constraints += [y >= x[:m] + x[d:] - 1]
        
        sum_x_front = cp.sum(x[:m])
        sum_x_back = cp.sum(x[d:])
        sum_y = cp.sum(y)
        
        # a_d = sum_x_front - sum_y ≤ M
        constraints += [sum_x_front - sum_y <= M]
        # a_{-d} = sum_x_back - sum_y ≤ M
        constraints += [sum_x_back - sum_y <= M]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=time_limit)
        except Exception:
            return None, None
    
    if x.value is None or not np.all(np.isfinite(x.value)):
        return None, None
    
    h = np.round(x.value).astype(float)
    # Verify
    if abs(float(np.sum(h)) - target) > 1e-6:
        h = _project(h, target)
        if abs(float(np.sum(h)) - target) > 0.01:
            return None, None
    
    c5 = _exact_c5(h)
    return h, c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
