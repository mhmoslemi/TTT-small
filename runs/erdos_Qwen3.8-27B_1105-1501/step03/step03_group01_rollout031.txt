import numpy as np
from scipy.optimize import minimize, NonlinearConstraint, LinearConstraint, linprog
import time

def _make_feasible(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

    return x[n]

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _make_feasible(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall % 8 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        j_lo = max(0, d)
        j_hi = min(n - 1, d + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[idx2] += wk * (-h[idx2 + d])
    return grad

def _grad_search(h, deadline, tau=100.0, lr=0.012, momentum_beta=0.88):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad = _soft_grad(h, tau)
        mom = momentum_beta * mom + (1.0 - momentum_beta) * grad
        h_new = h - lr * mom
        h_new = _make_feasible(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.04, 0.08)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
            if stall % 15 == 0:
                h = best_h + np.random.default_rng(int(time.time()*1000)).normal(0, 0.015, n)
                h = _make_feasible(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _make_feasible(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=20):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-7 or h[k] >= 1.0 - 1e-7:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = min((1.0 - h[j]) / max(dj, 1e-15), h[j] / max(-dj, 1e-15) if dj < 0 else 1e18,
                             (1.0 - h[k]) / max(dk, 1e-15) if dk > 0 else h[k] / max(-dk, 1e-15))
                    if am < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def _generate_seeds(h_best, n_best, rng, n_seeds=8):
    seeds = []
    seeds.append(h_best.copy())
    h_sym = 0.5 * (h_best + 1.0 - h_best[::-1])
    seeds.append(_make_feasible(h_sym, n_best / 2.0))
    for i in range(n_seeds - 2):
        scale = 0.03 + 0.04 * i
        h_p = h_best + rng.normal(0, scale, n_best)
        seeds.append(_make_feasible(h_p, n_best / 2.0))
    return seeds

def _evolutionary(h_init, deadline, pop_size=10, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_make_feasible(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.04
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2)
        child += rng.normal(0, mut_scale, n)
        child = _make_feasible(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.93)
            else:
                mut_scale = min(0.1, mut_scale * 1.05)
        else:
            mut_scale = min(0.12, mut_scale * 1.04)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _make_feasible(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Epigraph solve (trust-constr) on initial
    d1 = t0 + budget_s * 0.12
    h_e, c5_e = _epigraph_solve(h, d1, max_iter=200)
    if c5_e < best_c5:
        best_c5 = c5_e
        best_h = h_e.copy()

    # Phase 2: LP refinement
    d2 = t0 + budget_s * 0.28
    h_l, c5_l = _lp_loop(best_h, d2, radius_init=0.12)
    if c5_l < best_c5:
        best_c5 = c5_l
        best_h = h_l.copy()

    # Phase 3: Gradient search warmup then epigraph again
    d3 = t0 + budget_s * 0.38
    h_g, c5_g = _grad_search(best_h, d3, tau=120.0, lr=0.01, momentum_beta=0.85)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
    h_e2, c5_e2 = _epigraph_solve(best_h, t0 + budget_s * 0.45, max_iter=200)
    if c5_e2 < best_c5:
        best_c5 = c5_e2
        best_h = h_e2.copy()

    # Phase 4: Multi-start with epigraph
    d4 = t0 + budget_s * 0.70
    seeds = _generate_seeds(best_h, best_n, rng, n_seeds=10)
    for s in seeds:
        if time.time() > d4:
            break
        sub1 = min(d4, time.time() + budget_s * 0.04)
        h_s, c5_s = _epigraph_solve(s, sub1, max_iter=150)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        sub2 = min(d4, time.time() + budget_s * 0.02)
        h_sl, c5_sl = _lp_loop(h_s, sub2, radius_init=0.06)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 5: Multi-resolution search
    d5 = t0 + budget_s * 0.82
    offsets = [-8, -4, 0, 4, 8, 12, 16, -12, -16, 20, -20]
    tried_n = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 140:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub1 = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = _epigraph_solve(h_r, sub1, max_iter=180)
        if time.time() < d5:
            sub2 = min(d5, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_loop(h_r, sub2, radius_init=0.06)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Perturbation + epigraph (annealing scale)
    d6 = t0 + budget_s * 0.90
    scale = 0.025
    for attempt in range(60):
        if time.time() > d6:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _make_feasible(h_p, best_n / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_t, c5_t = _epigraph_solve(h_p, sub_dl, max_iter=120)
        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.2)

    # Phase 7: Evolutionary
    if time.time() < t0 + budget_s * 0.94:
        d7 = t0 + budget_s * 0.94
        h_ev, c5_ev = _evolutionary(best_h, d7, pop_size=10, seed=seed + 11)
        if c5_ev < best_c5 - 1e-11:
            best_c5 = c5_ev
            best_h = h_ev.copy()

    # Phase 8: Final LP + pair polish
    if time.time() < t0 + budget_s * 0.97:
        d8 = t0 + budget_s * 0.97
        h_f, c5_f = _lp_loop(best_h, d8, radius_init=0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < t0 + budget_s * 0.99:
        h_pr, c5_pr = _pair_refine(best_h, t0 + budget_s * 0.99, max_passes=12)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()

    # Final verification
    best_h = _make_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
