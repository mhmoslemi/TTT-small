import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining discrepancy
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    """Compute gradient of smooth max of correlation scores"""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    # Softmax weights
    A = beta * (scores - max_s)
    A = A - np.max(A)
    W = np.exp(A)
    W = W / np.sum(W)
    
    # Gradient: for each score index r, d = r - (n-1)
    # ds[r]/dh[j] = (1-h[j-d])*[0<=j-d<n] - h[j+d]*[0<=j+d<n]
    grad = np.zeros(n)
    for r in range(2*n - 1):
        d = r - (n-1)
        w = W[r]
        if w < 1e-15:
            continue
        # Positions j where j-d is valid: j from max(0,d) to min(n-1, n-1+d)
        # Positions j where j+d is valid: j from max(0,-d) to min(n-1, n-1-d)
        # Contribution to grad[j]: (1-h[j-d]) for valid j-d, minus h[j+d] for valid j+d
        # Vectorized:
        # (1-h[j-d]) for j where 0<=j-d<n: j in [max(0,d), min(n-1, n-1+d)]
        lo1 = max(0, d)
        hi1 = min(n, n+d)  # exclusive
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-d:hi1-d])
        lo2 = max(0, -d)
        hi2 = min(n, n-d)
        if hi2 > lo2:
            grad[lo2:hi2] += -w * h[lo2+d:hi2+d]
    
    # Center for sum constraint
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return None, scores
    return -grad / norm, scores
