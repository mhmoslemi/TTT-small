import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute Jacobian of all lag scores wrt h."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                # score = dx * sum_{i=0}^{m-1} h[i+d]*(1-h[i])
                # d/dh[j] for j in [0, m-1]: -dx*h[j+d] (from (1-h[i]) term when j=i)
                # d/dh[j] for j in [d, n-1]: dx*(1-h[j-d]) (from h[i+d] term when j=i+d)
                G[r, :m] += -dx * h[d:d+m]  # from (1-h[i]) derivative
                G[r, d:n] += dx * (1.0 - h[:m])  # from h[i+d] derivative
        else:
            g = -d
            m = n - g
            if m > 0:
                # score = dx * sum_{i=g}^{n-1} h[i]*(1-h[i-g])
                # d/dh[j] for j in [g, n-1]: dx*(1-h[j-g]) (from h[i] term)
                # d/dh[j] for j in [0, m-1]: -dx*h[j+g] (from (1-h[i-g]) term)
                G[r, g:n] += dx * (1.0 - h[:m])  # from h[i] derivative
                G[r, :m] += -dx * h[g:g+m]  # from (1-h[i-g]) derivative
    return G

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _vertex_direction(h, w, G):
    """Find the vertex of the feasible polytope that minimizes w^T G^T v = (G^T w)^T v."""
    n = len(h)
    g = G.T @ w  # gradient direction
    # To minimize g^T v over {0<=v<=1, sum(v)=n/2}, set v=1 where g is smallest
    T = n / 2.0
    if T == int(T):
        T = int(T)
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:T]] = 1.0
        return v - h
    else:
        # Fractional case
        T_int = int(T)
        frac = T - T_int
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:T_int]] = 1.0
        if frac > 1e-12:
            v[order[T_int]] = frac
        return v - h

def _momentum_optimize(h, deadline, n_steps=500):
    """Subgradient with Nesterov momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h_prev = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Adaptive step size
    step = 0.02
    t = 1.0
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        t += 1.0
        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 1.0)
        y = h + momentum * (h - h_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        scores = _scores(y_clipped)
        M = np.max(scores)
        G = _grad_matrix(y_clipped)
        
        # Softmax weights
        beta = 20.0 + 80.0 * (it / max(1, n_steps))
        w = np.exp(beta * (scores - M))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        
        grad = G.T @ w
        # Remove mean to preserve sum constraint better
        grad -= np.mean(grad)
        
        h_prev = h.copy()
        h_new = h - step * grad
        h_new = _project(h_new, target)
        
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            step = min(step * 1.02, 0.1)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            step *= 0.7
            h = h_prev.copy()  # revert
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_block_init(n, n_blocks):
    """Create a block-structured initialization."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for k in range(n_blocks):
        if k % 2 == 0:
            start = k * block_size
            end = start + block_size
            h[start:end] = 1.0
    # Smooth a bit
    kernel = np.array([0.25, 0.5, 0.25])
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    return _project(h_smooth, n / 2.0)

def _make_wave_init(n):
    """Create a wave initialization."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(np.pi * x)
    return _project(h, n / 2.0)

def _exact_c5(h):

    return float(np.max(corr) * dx)

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += -dx * h[d:d+m]
                G[r, d:n] += dx * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:n] += dx * (1.0 - h[:m])
                G[r, :m] += -dx * h[g:g+m]
    return G

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

                if a_max < 1e-10:
                    continue
                lo, hi = 0.0, a_max
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                h1 = h.copy(); h1[j] += sign * c1; h1[k] -= sign * c1
                h1 = np.clip(h1, 0, 1)
                f1 = _exact_c5(h1)
                h2 = h.copy(); h2[j] += sign * c2; h2[k] -= sign * c2
                h2 = np.clip(h2, 0, 1)
                f2 = _exact_c5(h2)
                for _ in range(15):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        h1 = h.copy(); h1[j] += sign * c1; h1[k] -= sign * c1
                        h1 = np.clip(h1, 0, 1)
                        f1 = _exact_c5(h1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        h2 = h.copy(); h2[j] += sign * c2; h2[k] -= sign * c2
                        h2 = np.clip(h2, 0, 1)
                        f2 = _exact_c5(h2)
                a_opt = (lo + hi) / 2.0
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Strong LP from initial (22% time)
    dl = t0 + budget_s * 0.22
    h1, c5_1 = _lp_optimize(h, dl, top_k_list=[None, 15, 10, 5, 8, None, 20], radius_init=0.25, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Vertex directions (10%)
    if time.time() < t0 + budget_s * 0.32:
        dl = t0 + budget_s * 0.32
        h2, c5_2 = _vertex_optimize(best_h, dl, n_iters=4000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Refined LP small radius (12%)
    if time.time() < t0 + budget_s * 0.44:
        dl = t0 + budget_s * 0.44
        h3, c5_3 = _lp_optimize(best_h, dl, top_k_list=[5, 8, 10, 12, None], radius_init=0.06, max_iters=6000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Momentum (8%)
    if time.time() < t0 + budget_s * 0.52:
        dl = t0 + budget_s * 0.52
        h4, c5_4 = _momentum_optimize(best_h, dl, n_steps=400)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Gradient-informed restarts (22%)
    dl = t0 + budget_s * 0.74
    scale = 0.018
    for attempt in range(800):
        if time.time() > dl:
            break
        if best_c5 < 0.3806:
            break
        # Gradient-informed perturbation
        scores = _scores(best_h)
        M = np.max(scores)
        G = _grad_matrix(best_h)
        w = np.exp(20.0 * (scores - M))
        sw = np.sum(w)
        if sw > 1e-300:
            w = w / sw
            grad = G.T @ w
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn > 1e-12:
                grad = grad / gn
            else:
                grad = None
        else:
            grad = None
        if grad is not None and rng.random() < 0.5:
            noise = rng.normal(0, scale, best_n)
            h_pert = best_h + 0.3 * scale * grad + noise * 0.7
        else:
            h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.018)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[10, 5, None], radius_init=0.1, max_iters=400)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.15)

    # Phase 6: Resolution sweep (8%)
    if time.time() < t0 + budget_s * 0.82:
        dl = t0 + budget_s * 0.82
        for dn in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10, 12]:
            if time.time() > dl:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 180:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None], radius_init=0.12, max_iters=500)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Pair polish + single coord (7%)
    if time.time() < t0 + budget_s * 0.89:
        dl = t0 + budget_s * 0.89
        hp, c5p = _pair_polish(best_h, dl, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
        h_sc, c5_sc = _single_coord(best_h, t0 + budget_s * 0.92)
        if c5_sc < best_c5:
            best_c5 = c5_sc
            best_h = h_sc.copy()

    # Phase 8: Final ultra-fine LP (8%)
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.995,
                                 top_k_list=[3, 4, 5, 6, None], radius_init=0.012, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
