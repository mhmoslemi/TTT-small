    Returns best t and best raw value."""
    n = len(h)
    A, B, C = _line_coeffs(h, d)
    M = len(A)
    
    # Find feasible interval for t
    tlo = 0.0
    thi = 0.0
    for i in range(n):
        if d[i] > 1e-15:
            tlo = max(tlo, -h[i] / d[i])  # h[i] + t*d[i] >= 0
            thi = min(thi, (1.0 - h[i]) / d[i])  # h[i] + t*d[i] <= 1
        elif d[i] < -1e-15:
            tlo = max(tlo, (1.0 - h[i]) / d[i])
            thi = min(thi, -h[i] / d[i])
    
    # Since t=0 is feasible, tlo <= 0 <= thi
    tlo = max(tlo, -1.0)  # safety
    thi = min(thi, 1.0)  # safety
    
    if thi - tlo < 1e-14:
        return 0.0, current_raw
    
    # Collect candidates
    candidates = [tlo, thi]
    
    # Stationary points of each quadratic
    for r in range(M):
        if abs(C[r]) > 1e-15:
            t0 = -B[r] / (2.0 * C[r])
            if tlo <= t0 <= thi:
                candidates.append(t0)

    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-15:
                # Linear: b*t + c = 0
                if abs(b) > 1e-15:
                    t0 = -c / b
                    if tlo <= t0 <= thi:
                        candidates.append(t0)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b + sqrt_disc) / (2.0 * a)
                    t2 = (-b - sqrt_disc) / (2.0 * a)
                    if tlo <= t1 <= thi:
                        candidates.append(t1)
                    if tlo <= t2 <= thi:
                        candidates.append(t2)
    
    if len(candidates) > 50000:
        candidates = candidates[:50000]
    
    cand_arr = np.array(candidates)
    cand_arr = np.clip(cand_arr, tlo, thi)
    
    # Sort and deduplicate
    cand_arr = np.sort(cand_arr)
    if len(cand_arr) > 1:
        diffs = np.diff(cand_arr)
        mask = np.concatenate([[True], diffs > 1e-10])
        cand_arr = cand_arr[mask]

    best_t = 0.0
    best_raw = current_raw
    for t in cand_arr:
        raw = float(np.max(A + B * t + C * t * t))
        if raw < best_raw - 1e-12:
            best_raw = raw
            best_t = t
    
    return best_t, best_raw

            dirs.append(d_lp)

        improved = False
        for d in dirs:
            if time.time() > deadline:
                break
            if d is None:
                continue
            dsum = float(np.sum(d))
            if abs(dsum) > 1e-10:
                d = d - dsum / n

            best_t, new_max, h_new = _line_search(h, d, deadline)
            if best_t != 0.0 and new_max < cur_max - 1e-10:
                h = h_new
                cur_max = new_max
                scores = _exact_raw(h)
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1

    return best_h, best_max * (2.0 / n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_seed(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    for _ in range(20):
        i = rng.integers(n)
        j = rng.integers(n)
        if i != j:
            h[i], h[j] = 0.5, 0.5
    return _project(h, n / 2.0)

def _edge_seed(n, rng):
    h = np.zeros(n)
    k = n // 4
    h[:k] = 1.0
    h[n-k:] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Intensive local polish of parent (30% time)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _local_descent(h, dl1, rng, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n0

    if best_c5 <= 0.38080:
        return best_h, best_c5, best_n

    # Phase 2: Try parent transforms and local polish (15% time)
    dl2 = t0 + budget_s * 0.45
    transforms = [
        1.0 - best_h,
        best_h[::-1].copy(),
        1.0 - best_h[::-1].copy(),
    ]
    for t_h in transforms:
        if time.time() > dl2:
            break
        t_h = _project(t_h, best_n / 2.0)
        t_c5 = _exact_c5(t_h)
        if t_c5 < best_c5:
            best_c5 = t_c5
            best_h = t_h.copy()
        h2, c5_2 = _local_descent(t_h, min(dl2, time.time() + 20.0), rng, max_iters=500)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    if best_c5 <= 0.38080:
        return best_h, best_c5, best_n

    # Phase 3: Population exploration (25% time)
    dl3 = t0 + budget_s * 0.70
    seeds = [best_h.copy()]

    # Perturbed versions
    for scale in [0.03, 0.06, 0.10, 0.15]:
        for _ in range(2):
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _project(h_p, best_n / 2.0)
            seeds.append(h_p)

    # Binary seeds
    for _ in range(3):
        seeds.append(_binary_seed(best_n, rng))

    # Edge seeds
    seeds.append(_edge_seed(best_n, rng))

    # Sort by objective
    seed_objs = [(s, _exact_c5(s)) for s in seeds]
    seed_objs.sort(key=lambda x: x[1])

    per_seed_time = max(5.0, min(25.0, (dl3 - time.time()) / max(1, len(seed_objs))))
    for s_h, s_obj in seed_objs:
        if time.time() > dl3:
            break
        if s_obj >= best_c5 + 0.005:
            continue
        per_dl = min(dl3, time.time() + per_seed_time)
        h3, c5_3 = _local_descent(s_h, per_dl, rng, max_iters=2000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = len(h3)
            if best_c5 <= 0.38080:
                return best_h, best_c5, best_n

    # Phase 4: Resolution sweep (10% time)
    dl4 = t0 + budget_s * 0.80
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8, 10, -10, 12, -12]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _local_descent(h_r, min(dl4, time.time() + 15.0), rng, max_iters=1000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if best_c5 <= 0.38080:
                return best_h, best_c5, best_n

    # Phase 5: Final intensive polish (18% time)
    dl5 = hard_deadline
    # Multiple rounds of local descent with increasing direction diversity
    for round_i in range(6):
        if time.time() > dl5:
            break
        per_dl = min(dl5, time.time() + max(10.0, (dl5 - time.time()) / (6 - round_i)))
        h5, c5_5 = _local_descent(best_h, per_dl, rng, max_iters=3000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = len(h5)
        elif c5_5 >= best_c5:
            # No improvement; try perturbation
            if time.time() > dl5 - 5.0:
                break
            pert = best_h + rng.normal(0, 0.02, best_n)
            pert = _project(pert, best_n / 2.0)
            h5b, c5_5b = _local_descent(pert, min(dl5, time.time() + 15.0), rng, max_iters=1000)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    if final_c5 > best_c5 + 1e-6:
        # Projection degraded; try the pre-projection if it was feasible
        pre = np.clip(best_h, 0.0, 1.0)
        if abs(float(np.sum(pre)) - best_n / 2.0) < 1e-6:
            final_c5 = _exact_c5(pre)
            best_h = pre

    return best_h, final_c5, best_n
