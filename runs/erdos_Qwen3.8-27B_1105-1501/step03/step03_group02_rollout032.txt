def _repair(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-10:
        return v
    # Bisection on shift
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _score_gradient(h, k):

    n = len(h)
    grad = np.zeros(n)
    m = n - abs(k)

        return grad

        s_sel = scores[idx]
        max_sel = float(np.max(s_sel))
        w[idx] = np.exp(beta * (s_sel - max_sel))
    else:

        w[int(np.argmax(scores))] = 1.0

    d -= np.mean(d)  # make zero-sum
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm  # descent direction

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-10:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _line_interval(h, d):
    t_lo, t_hi = -np.inf, np.inf
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    t_lo = max(t_lo, -10.0)
    t_hi = min(t_hi, 10.0)
    return t_lo, t_hi

def _line_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for r in range(M):
        k = r - (n - 1)
        if k == 0:
            A[r] = np.dot(h, 1.0 - h)
            B[r] = np.dot(d, 1.0 - 2.0 * h)
            C[r] = -np.dot(d, d)
        elif k > 0:
            m = n - k
            A[r] = np.dot(h[k:k+m], 1.0 - h[:m])
            B[r] = np.dot(d[k:k+m], 1.0 - h[:m]) - np.dot(h[k:k+m], d[:m])
            C[r] = -np.dot(d[k:k+m], d[:m])
        else:
            g = -k
            m = n - g
            A[r] = np.dot(h[:m], 1.0 - h[g:g+m])
            B[r] = np.dot(d[:m], 1.0 - h[g:g+m]) - np.dot(h[:m], d[g:g+m])
            C[r] = -np.dot(d[:m], d[g:g+m])
    return A, B, C

def _line_min(h, d):
    """Find best t for direction d, return (best_t, best_raw) where raw = max score."""
    t_lo, t_hi = _line_interval(h, d)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(_scores(h)))
    
    A, B, C = _line_coeffs(h, d)
    n = len(h)
    M = 2 * n - 1
    
    # Verify orientation
    s0 = _scores(h)
    if not np.allclose(A, s0, atol=1e-9):
        # Try reversed
        if np.allclose(A[::-1], s0, atol=1e-9):
            A, B, C = A[::-1], B[::-1], C[::-1]
        else:
            # Fallback: golden section
            return _line_min_golden(h, d, t_lo, t_hi)
    
    # Collect candidates
    cands = [t_lo, t_hi, 0.0]
    
    # Stationary points
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo - 1e-12 <= t0 <= t_hi + 1e-12:
                cands.append(t0)
    
    # Pairwise roots (vectorized)
    i_idx, j_idx = np.triu_indices(M, k=1)
    ad = C[i_idx] - C[j_idx]
    bd = B[i_idx] - B[j_idx]
    cd = A[i_idx] - A[j_idx]
    
    quad_mask = np.abs(ad) > 1e-14
    if np.any(quad_mask):
        disc = bd[quad_mask]**2 - 4.0 * ad[quad_mask] * cd[quad_mask]
        valid = disc >= -1e-14
        if np.any(valid):
            disc_v = np.maximum(disc[valid], 0.0)
            sq = np.sqrt(disc_v)
            a_v = ad[quad_mask][valid]
            b_v = bd[quad_mask][valid]
            t1 = (-b_v + sq) / (2.0 * a_v)
            t2 = (-b_v - sq) / (2.0 * a_v)
            cands.extend(t1.tolist())
            cands.extend(t2.tolist())
    
    lin_mask = (~quad_mask) & (np.abs(bd) > 1e-14)
    if np.any(lin_mask):
        t_lin = -cd[lin_mask] / bd[lin_mask]
        cands.extend(t_lin.tolist())
    
    cands = np.array(cands)
    cands = cands[np.isfinite(cands)]
    cands = cands[(cands >= t_lo - 1e-10) & (cands <= t_hi + 1e-10)]
    cands = np.clip(cands, t_lo, t_hi)
    
    if len(cands) == 0:
        return 0.0, float(np.max(A))
    
    # Deduplicate
    cands = np.unique(np.round(cands * 1e10).astype(np.int64).astype(float) / 1e10)
    
    # Cap to avoid memory issues
    if len(cands) > 50000:
        # Keep endpoints and sample
        cands = np.unique(np.concatenate([cands[:100], cands[-100:], np.linspace(t_lo, t_hi, 5000)]))
    
    # Evaluate F(t) = max_r (A_r + B_r*t + C_r*t^2)
    t_arr = cands[:, None]
    vals = A[None, :] + B[None, :] * t_arr + C[None, :] * t_arr * t_arr
    F = np.max(vals, axis=1)
    
    best_idx = int(np.argmin(F))
    best_t = float(cands[best_idx])
    best_raw = float(F[best_idx])
    
    return best_t, best_raw

def _line_min_golden(h, d, t_lo, t_hi):
    """Golden section fallback"""
    n = len(h)
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    c = hi - gr * (hi - lo)
    d_val = lo + gr * (hi - lo)
    fc = _exact_c5(np.clip(h + c * d, 0, 1))
    fd = _exact_c5(np.clip(h + d_val * d, 0, 1))
    best_t, best_c5 = c, fc
    if fd < fc:
        best_t, best_c5 = d_val, fd
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d_val
            d_val = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _exact_c5(np.clip(h + c * d, 0, 1))
        else:
            lo = c
            c = d_val
            fc = fd
            d_val = lo + gr * (hi - lo)
            fd = _exact_c5(np.clip(h + d_val * d, 0, 1))
        tc = (lo + hi) / 2.0
        tc5 = _exact_c5(np.clip(h + tc * d, 0, 1))
        if tc5 < best_c5:
            best_c5 = tc5
            best_t = tc
    return best_t, best_c5 / dx

def _all_grads(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for r in range(M):
        k = r - (n - 1)
        g = np.zeros(n)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            m = n - k
            g[:m] = -h[k:k+m]
            g[k:] = 1.0 - h[:m]
        else:
            ga = -k
            m = n - ga
            g[:m] = 1.0 - h[ga:ga+m]
            g[ga:] = -h[:m]
        G[r] = g
    return G

def _smooth_dir(h, scores, beta=30.0, top_k=None):
    n = len(h)
    M = len(scores)
    max_s = float(np.max(scores))
    if top_k and top_k < M:
        idx = np.argsort(scores)[::-1][:top_k]
        w = np.zeros(M)
        ws = np.exp(beta * (scores[idx] - max_s))
        w[idx] = ws
    else:
        w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        return None
    w /= wsum
    G = _all_grads(h)
    d = G.T @ w
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def _pair_dir(h, i, j):
    """Direction: transfer from i to j (d[i]=-1, d[j]=+1)"""
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d

def _pair_scores(h, scores, top_n=80):
    """Find top pairs by directional derivative of max score"""
    n = len(h)
    s = _scores(h)
    max_s = float(np.max(s))
    # For each pair (i,j), compute how much it reduces the max
    # Directional derivative of score r in direction d_ij:
    # = -G[r][i] + G[r][j]
    # We want to minimize the max score, so we want pairs where
    # the active scores decrease most.
