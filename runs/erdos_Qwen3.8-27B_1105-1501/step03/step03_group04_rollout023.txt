            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def get_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradient(h, weights):
    """Compute gradient of weighted sum of scores."""
    n = len(h)
    dgrad = np.zeros(n)
    # Lag 0 at index n-1
    dgrad += weights[n-1] * (1.0 - 2.0*h)
    for d in range(1, n):
        # full[n-1+d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        w1 = weights[n-1+d]
        dgrad[d:] += w1 * (1.0 - h[:n-d])
        dgrad[:n-d] -= w1 * h[d:]
        # full[n-1-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        w2 = weights[n-1-d]
        dgrad[:n-d] += w2 * (1.0 - h[d:])
        dgrad[d:] -= w2 * h[:n-d]
    dgrad -= np.mean(dgrad)
    return dgrad

def soft_gradient_descent(h, deadline, beta_start=50, beta_end=10000, max_iters=5000, lr_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        sc = get_scores(h)
        m = float(np.max(sc))
        exps = np.exp(beta * (sc - m))
        wsum = float(np.sum(exps))
        if wsum < 1e-300:
            break
        weights = exps / wsum
        grad = compute_gradient(h, weights)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        improved = False
        alpha = lr
        for _ in range(25):
            h_try = h + alpha * direction
            if np.any(h_try < -0.2) or np.any(h_try > 1.2):
                alpha *= 0.5
                continue
            h_try = project(h_try, target)
            c5_try = exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.02, 0.5)
                improved = True
                break
            alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.8
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def binary_sa(n, deadline, rng, max_iters=200000):
    """Simulated annealing for binary sequences."""
    target = n // 2
    # Start with random binary
    h = np.zeros(n)
    idx_best = rng.choice(n, size=target, replace=False)
    h[idx_best] = 1.0
    cur_score = float(np.max(get_scores(h)))
    best_h = h.copy()
    best_score = cur_score
    T = 5.0
    T_min = 0.01
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        T = max(T_min, T * 0.9999)
        # Move: flip 2-4 random positions maintaining sum
        n_flip = rng.integers(2, 5)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < n_flip or len(zeros) < n_flip:
            n_flip = min(n_flip, len(ones), len(zeros))
        if n_flip < 1:
            break
        flip_ones = rng.choice(ones, size=n_flip, replace=False)
        flip_zeros = rng.choice(zeros, size=n_flip, replace=False)
        h[flip_ones] = 0.0
        h[flip_zeros] = 1.0
        new_score = float(np.max(get_scores(h)))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[flip_ones] = 1.0
            h[flip_zeros] = 0.0
    return best_h, best_score * (2.0 / n)

def relax_continuous(h_bin, deadline, max_iters=3000):
    """Take a binary sequence and relax to continuous."""
    n = len(h_bin)
    target = n / 2.0
    # Add small noise to break symmetry
    h = h_bin + 0.05 * (np.random.default_rng(0).random(n) - 0.5)
    h = project(h, target)
    return soft_gradient_descent(h, deadline, beta_start=100, beta_end=20000, max_iters=max_iters, lr_init=0.05)

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

Actually, let me try yet another approach: **alternating minimization**.

Fix all but a few coordinates, optimize those coordinates (which becomes a small convex problem since the objective becomes piecewise linear in those coordinates), then move to the next set.

If I fix all but 2 coordinates, the problem becomes: minimize max_d (a_d + b_d1*x + b_d2*y) subject to constraints on x,y. This is a 2D convex minimization, solvable by checking O(n^2) extreme points or by a 2D LP.

This is getting complex. Let me just go with a practical approach:

1. Start from parent
2. Use a modified gradient descent with:
   - Nesterov momentum
   - Adaptive β (softmax temperature)
   - Line search with golden section
   - Periodic "jumps" (add noise and re-optimize)
3. Try multiple n values
4. At the end, do a coordinate-wise 1D optimization

- Using momentum (Nesterov)
- Using a "top-K" weighting instead of full softmax (more aggressive focus on the worst lags)
- Periodic binary projection (project to nearest binary, then relax)

import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def get_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    g += weights[n-1] * (1.0 - 2.0*h)
    for d in range(1, n):
        w1 = weights[n-1+d]
        g[d:] += w1 * (1.0 - h[:n-d])
        g[:n-d] -= w1 * h[d:]
        w2 = weights[n-1-d]
        g[:n-d] += w2 * (1.0 - h[d:])
        g[d:] -= w2 * h[:n-d]
    g -= np.mean(g)
    return g

def topk_weights(scores, k, beta):
    """Compute weights focusing on top-k scores."""
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    # Zero out all but top-k
    topk_idx = np.argsort(scores)[-k:]
    mask = np.zeros(len(scores), dtype=bool)
    mask[topk_idx] = True
    exps[~mask] = 0.0
    s = float(np.sum(exps))
    if s < 1e-300:
        return exps / max(1, len(scores))
    return exps / s

    stall = 0

        x_look = x - momentum * v
        x_look = np.clip(x_look, 0.0, 1.0)
