def _coord_descend(h, target, max_cycles=200, deadline=None, verbose=False):

    h = h.copy()
    best_h = h.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0
    
    for cycle in range(max_cycles):
        if deadline and time.time() > deadline:
            break
        prev_best = best_c5

            # Compute slope by evaluating at h[i]=0 and h[i]=1
            tmp = h[i]
            h[i] = 0.0
            s0 = _scores(h)
            h[i] = 1.0
            s1 = _scores(h)
            slope = s1 - s0
            h[i] = tmp
            
            # Golden section on f(v) = max(s0 + v * slope)
            lo_v, hi_v = 0.0, 1.0
            c1 = hi_v - gr * (hi_v - lo_v)
            c2 = lo_v + gr * (hi_v - lo_v)
            f1 = float(np.max(s0 + c1 * slope))
            f2 = float(np.max(s0 + c2 * slope))
            for _ in range(25):
                if f1 < f2:
                    hi_v, c2, f2 = c2, c1, f1
                    c1 = hi_v - gr * (hi_v - lo_v)
                    f1 = float(np.max(s0 + c1 * slope))
                else:
                    lo_v, c1, f1 = c1, c2, f2
                    c2 = lo_v + gr * (hi_v - lo_v)
                    f2 = float(np.max(s0 + c2 * slope))
            h[i] = (lo_v + hi_v) / 2.0

        cur_c5 = _exact_c5(h)

            best_h = h.copy()
            stall = 0
        else:

def _to_binary(h, rng):

    w = n // 2
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:w]] = 1.0

def _golden_pattern(n):
    """Generate a quasi-periodic pattern using golden ratio"""
    w = n // 2
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    vals = np.array([((i * phi) % 1.0) for i in range(n)])
    # Take the top w values as 1s
    order = np.argsort(-vals)
    h[order[:w]] = 1.0
    return h

def _cubic_pattern(n, k):
    """Generate pattern based on cubic residue"""
    w = n // 2
    h = np.zeros(n)
    if k <= 1:
        # Simple alternating
        h[::2] = 1.0
        # Adjust to get exactly w ones
        cnt = int(np.sum(h))
        if cnt > w:
            order = np.where(h == 1)[0]
            h[order[-(cnt-w):]] = 0.0
        elif cnt < w:
            order = np.where(h == 0)[0]
            h[order[:w-cnt]] = 1.0
    else:
        # Use modular arithmetic
        vals = np.array([(i**3) % n for i in range(n)], dtype=float)
        order = np.argsort(vals)
        h[order[:w]] = 1.0
    return h

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Extended coordinate descent from best (30% time)
    dl2 = t0 + budget_s * 0.42
    h_cd, c5_cd = _coord_descend(best_h, target0, max_cycles=800, deadline=dl2)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: Multi-start with perturbations (28% time)
    dl3 = t0 + budget_s * 0.70
    n_starts = 12
    for s in range(n_starts):
        if time.time() > dl3:
            break
        remaining = dl3 - time.time()
        if remaining < 5:
            break
        alloc = min(remaining / max(1, n_starts - s), 35.0)
        sub_dl = time.time() + alloc

        r = rng.random()
        if r < 0.25:
            h_try = _perturb(best_h, target0, rng, scale=0.04)
        elif r < 0.5:
            h_try = _perturb_sparse(best_h, target0, rng, n_pert=6)
        elif r < 0.7:
            h_try = _kick(best_h, target0, rng, k=3)
        elif r < 0.85:
            h_try = _kick(best_h, target0, rng, k=5)
            h_try = _perturb(h_try, target0, rng, scale=0.03)
        else:
            if best_bin_b is not None:
                h_try = _project(best_bin_b + rng.normal(0, 0.2, n0), target0)
            else:
                h_try = _perturb(best_h, target0, rng, scale=0.1)

        h_opt, c5_opt = _coord_descend(h_try, target0, max_cycles=150, deadline=sub_dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Resolution sweep (15% time)
    dl4 = t0 + budget_s * 0.85
    n_res = len(best_h)
    resolutions = []
    for delta in [-6, 6, -10, 10, -14, 14, -20, 20, -28, 28, 36, -36, 44, -44]:
        nt = n_res + delta
        if 32 <= nt <= 160 and nt not in resolutions:
            resolutions.append(nt)

    for n_try in resolutions:
        if time.time() > dl4:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + 12.0)
        h_r, c5_r = _coord_descend(h_r, n_try / 2.0, max_cycles=60, deadline=sub_dl)
        if time.time() < dl4:
            h_r, c5_r = _coord_descend(h_r, n_try / 2.0, max_cycles=100, deadline=min(dl4, time.time() + 8.0))
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final polish (13% time)
    dl5 = t0 + budget_s * 0.99
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final, c5_final = _coord_descend(best_h, target_final, max_cycles=400, deadline=dl5)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    # One more coordinate descent pass on the very best
    if time.time() < dl5:
        h_final2, c5_final2 = _coord_descend(best_h, target_final, max_cycles=200, deadline=dl5)
        if c5_final2 < best_c5:
            best_c5 = c5_final2
            best_h = h_final2.copy()

    # Final validation
    best_h = _project(best_h, target_final)
    final_c5 = _exact_c5(best_h)
    n_out = len(best_h)

    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
        n_out = len(h0)

    return best_h, final_c5, n_out
