def _project_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    n = len(h)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    # Use bisection to find the threshold
    lo, hi = -1.0, 2.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0

    x = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

The soft_subgrad with the inner Python loop is O(n * 2n) = O(n²) per call, which for n=64 is about 8000 iterations. In Python, this might take ~1ms per call, so we can do ~10000 iterations in 10 seconds. That's reasonable.

def _soft_subgrad_v(h, beta=30.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    weights = exps / np.sum(exps)
    
    sg = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        d = k - (n - 1)
        w = weights[k]
        if d >= 0:
            sg[d:] += w * (1.0 - h[:n-d])
            sg[:n-d] += w * (-h[d:])
        else:
            e = -d
            sg[:n-e] += w * (1.0 - h[e:])
            sg[e:] += w * (-h[:n-e])
    return sg
```
