import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-11:
        if resid > 0:
            mask = x < 1.0 - 1e-10
        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_lags_and_grads(h):
    n = len(h)
    nh = 1.0 - h
    corr = np.correlate(h, nh, mode='full')
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grads[k, :m] += nh[d:d+m]
                grads[k, d:d+m] -= h[:m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grads[k, l:l+m] += nh[:m]
                grads[k, :m] -= h[l:l+m]
    return corr, grads

def _lp_direction(h, corr, grads, n, R, lam, top_k=None):
    """Solve LP to find optimal tangent direction."""
    n_lags = 2 * n - 1
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(n_lags)
    n_constr = len(idx)
    
    n_vars = 1 + 2 * n  # t, p_0..p_{n-1}, q_0..q_{n-1}
    
    # Objective: minimize t + lam * (sum p + sum q)
    c = np.zeros(n_vars)
    c[0] = 1.0
    c[1:] = lam
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag s: -t + sum grad_s[j]*p_j - sum grad_s[j]*q_j <= -J_s
    # Plus: sum(p + q) <= R
    n_rows = n_constr + 1
    A_ub = np.zeros((n_rows, n_vars))
    b_ub = np.zeros(n_rows)
    
    for i, s in enumerate(idx):
        A_ub[i, 0] = -1.0
        A_ub[i, 1:1+n] = grads[s]
        A_ub[i, 1+n:1+2*n] = -grads[s]
        b_ub[i] = -corr[s]
    
    # l1 norm constraint
    A_ub[n_constr, 1:] = 1.0
    b_ub[n_constr] = R
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    eps_b = 1e-6
    for j in range(n):
        if h[j] >= 1.0 - eps_b:
            bounds.append((0.0, 0.0))    # p_j = 0 (can't increase)
            bounds.append((0.0, None))    # q_j free
        elif h[j] <= eps_b:
            bounds.append((0.0, None))    # p_j free
            bounds.append((0.0, 0.0))    # q_j = 0 (can't decrease)
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    p = result.x[1:1+n]
    q = result.x[1+n:1+2*n]
    d = p - q
    # Remove mean to ensure sum = 0
    d -= np.mean(d)
    
    norm_d = float(np.linalg.norm(d))
    if norm_d < 1e-10:
        return None
    
    return d / norm_d

def _line_search(h, d, target, c_cur, max_alpha=1.0):

    n = len(h)
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = min(max(alpha_max, 1e-8), 3.0)
    
    best_a = 0.0
    best_c = c_cur
    # Grid search
    alphas = np.linspace(0.02, alpha_max, 15)
    for a in alphas:
        h_t = _project(h + a * d, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-13:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c_cur
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.6)
    hi = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))

            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * d, target))

    return best_a, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    dl1 = t0 + budget_s * 0.45
    h1, c1 = _lp_opt(h, dl1, target0, max_it=20000)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    dl2 = t0 + budget_s * 0.72
    n_starts = 12
    for s in range(n_starts):
        if time.time() > dl2:
            break
        rem = dl2 - time.time()
        if rem < 8.0:
            break
        sub_dl = time.time() + min(rem * 0.85 / (n_starts - s), 25.0)
        r = rng.random()
        if r < 0.25:
            ht = _project(best_h + rng.normal(0, 0.05, best_n), target0)
        elif r < 0.45:
            idx = rng.choice(best_n, size=max(2, best_n // 5), replace=False)
            ht = best_h.copy()
            ht[idx] += rng.normal(0, 0.12, len(idx))
            ht = _project(ht, target0)
        elif r < 0.65:
            sh = int(rng.integers(1, max(2, best_n // 5)))
            ht = _project(np.roll(best_h, sh), target0)
        elif r < 0.80:
            hr = (rng.random(best_n) > 0.5).astype(float)
            ht = _project(0.5 * best_h + 0.5 * hr, target0)
        else:
            half = best_n // 2
            u = 0.5 * (best_h[:half] + best_h[half:])
            ht = _project(np.concatenate([u, u]), target0)
        ho, co = _lp_opt(ht, sub_dl, target0, max_it=8000)
        if co < best_c5 - 1e-12:
            best_c5 = co
            best_h = ho.copy()

    dl3 = t0 + budget_s * 0.86
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8, -12, 12, -16, 16]:
        if time.time() > dl3:
            break
        nt = best_n + dn
        if nt < 16 or nt > 128:
            continue
        hr = _resample(best_h, nt)
        sub = min(dl3, time.time() + 12.0)
        hr, cr = _lp_opt(hr, sub, nt / 2.0, max_it=5000)
        if cr < best_c5 - 1e-10:
            best_c5 = cr
            best_h = hr.copy()
            best_n = nt

    dl4 = t0 + budget_s * 0.95
    h4, c4 = _lp_opt(best_h, dl4, best_n / 2.0, max_it=20000)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()

    dl5 = t0 + budget_s * 0.985
    h5, c5 = _pair_polish(best_h, dl5, best_n / 2.0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    best_h = _project(best_h, best_n / 2.0)
    fc5 = _exact_c5(best_h)
    if fc5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        fc5 = _exact_c5(best_h)
    return best_h, fc5, best_n
