The key insight from the strategy is to use exact one-dimensional line search where:
- For a direction v (zero-sum), the raw scores are quadratic in alpha
- We can find the exact minimum of max(quadratics) by checking stationary points and pairwise intersections
- This is more powerful than gradient methods or LP because it can jump to envelope cusps

Let me design a clean, efficient implementation:

1. **Exact line search**: For direction v, compute A, B, C coefficients, find feasible alpha interval, enumerate all critical points (stationary + pairwise intersections), pick best.

2. **Direction generation**: 
   - Active gradient directions (using exponential weighting of top scores)
   - Random feasible directions
   - Pair directions (e_i - e_j)

3. **Transformations**: Flip, roll, threshold, mix

4. **Multi-resolution**: Resample to different n and optimize

Let me think about the key mathematical details:

For the line h + alpha*v:
- S_d(alpha) = sum_p (h[p]+alpha*v[p+d])*(1-h[p]-alpha*v[p])

- = A_d + alpha*B_d + alpha²*C_d

where:
- A_d = correlate(h, 1-h, 'full')[n-1+d]
- B_d = correlate(v, 1-h, 'full')[n-1+d] - correlate(h, v, 'full')[n-1+d]
- C_d = -correlate(v, v, 'full')[n-1+d]

Wait, let me be more careful. The correlate mode="full" gives:
- np.correlate(a, b, 'full')[k] = sum_j a[j] * b[k-j] for appropriate range

Actually let me just verify: for the line h(a) = h + a*v:
- np.correlate(h+a*v, 1-(h+a*v), 'full')
- = np.correlate(h+a*v, (1-h)-a*v, 'full')
- = np.correlate(h, (1-h)-a*v, 'full') + a*np.correlate(v, (1-h)-a*v, 'full')

So: A = correlate(h, 1-h, 'full')
    B = -correlate(h, v, 'full') + correlate(v, 1-h, 'full')
    C = -correlate(v, v, 'full')

This is correct.

Now for the feasible interval:
- For each i where v[i] > 0: alpha >= -h[i]/v[i], alpha <= (1-h[i])/v[i]
- For each i where v[i] < 0: alpha >= (1-h[i])/v[i], alpha <= -h[i]/v[i]

The key optimization: for n=64, we have 127 scores. Pairwise intersections: C(127,2) = 8001 pairs. For each pair, solve a quadratic (or linear if DeltaC≈0). This is fast.

For n=128: 255 scores, C(255,2) = 32385 pairs. Still manageable.

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for water-filling
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_coeffs(h, v):
    """Compute A, B, C coefficients for line h + alpha*v."""
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    return A, B, C

def _feasible_interval(h, v):
    """Compute feasible alpha interval for h + alpha*v staying in [0,1]."""
    pos = v > 1e-15
    neg = v < -1e-15
    lo = 0.0
    hi = 0.0
    if np.any(pos):
        lo = max(lo, float(np.min(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if np.any(neg):
        lo = max(lo, float(np.min((1.0 - h[neg]) / v[neg])))
        hi = min(hi, float(np.min(-h[neg] / v[neg])))
    # Ensure 0 is in interval
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if lo >= hi:
        return None
    return lo, hi

def _exact_line_min(h, v, cur_c5):
    """Exact 1D line minimization. Returns (alpha_best, h_new, c5_new) or None."""
    n = len(h)
    interval = _feasible_interval(h, v)
    if interval is None:
        return None
    lo, hi = interval
    if hi - lo < 1e-14:
        return None
    
    A, B, C = _line_coeffs(h, v)
    ns = len(A)  # 2n-1
    
    # Candidate alphas
    candidates = [lo, hi, 0.0]
    
    # Stationary points of each parabola
    for r in range(ns):
        if abs(C[r]) > 1e-18:
            a_star = -B[r] / (2.0 * C[r])
            if lo <= a_star <= hi:
                candidates.append(a_star)
    
    # Pairwise intersections (exact for n <= 96)
    if ns <= 160:
        for i in range(ns):
            for j in range(i + 1, ns):
                dC = C[i] - C[j]
                dB = B[i] - B[j]
                dA = A[i] - A[j]
                if abs(dC) > 1e-18:
                    disc = dB * dB - 4.0 * dC * dA
                    if disc >= -1e-14:
                        if disc < 0:
                            disc = 0.0
                        sq = np.sqrt(disc)
                        a1 = (-dB + sq) / (2.0 * dC)
                        a2 = (-dB - sq) / (2.0 * dC)
                        if lo <= a1 <= hi:
                            candidates.append(a1)
                        if lo <= a2 <= hi:
                            candidates.append(a2)
                elif abs(dB) > 1e-18:
                    a1 = -dA / dB
                    if lo <= a1 <= hi:
                        candidates.append(a1)
    
    # Grid points for numerical robustness
    n_grid = min(50, max(10, int((hi - lo) * 200)))
    grid = np.linspace(lo, hi, n_grid + 1)
    candidates.extend(grid.tolist())
    
    # Deduplicate
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= lo - 1e-12) & (candidates <= hi + 1e-12)]
    candidates = np.clip(candidates, lo, hi)
    candidates = np.unique(np.round(candidates, 12))
    
    # Evaluate raw cost at each candidate
    best_raw = np.inf
    best_a = 0.0
    for a in candidates:
        scores = A + a * B + a * a * C
        cost = float(np.max(scores))
        if cost < best_raw:
            best_raw = cost
            best_a = a
    
    # Directly evaluate the candidate
    h_new = h + best_a * v
    # Repair tiny numerical issues
    if h_new.min() < -1e-12 or h_new.max() > 1.0 + 1e-12:
        h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_box_sum(h_new, n / 2.0)
    
    c5_new = _exact_c5(h_new)
    if c5_new < cur_c5 - 1e-14:
        return best_a, h_new, c5_new
    return None

def _project_box_sum(v, target):

    if target <= 0:

def _line_search(h, v, cur_c5):
    """Exact 1D line minimization along h + alpha*v."""
    n = len(h)
    pos = v > 1e-15
    neg = v < -1e-15
    lo = 0.0
    hi = 0.0
    if np.any(pos):
        lo = max(lo, float(np.min(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))

def _grad_direction(h, beta):
    """Active gradient direction with exponential weighting."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    
    # Build gradient
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] -= wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[g:]
    
    v = -grad
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

def _random_direction(rng, n, scale=1.0):
    v = rng.standard_normal(n) * scale
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

def _pair_direction(n, i, j):
    v = np.zeros(n)
    v[i] = 1.0
    v[j] = -1.0
    return v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _local_line_descend(h, deadline, rng, n_betas=5, n_random=20):
    """Intensive local line descent."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    betas = [5.0, 15.0, 40.0, 100.0, 200.0, 400.0]
    stall = 0
    max_stall = 30
    
    while time.time() < deadline and stall < max_stall:
        improved = False
        
        # Gradient directions
        for beta in betas:
            if time.time() > deadline:
                break
            v = _grad_direction(h, beta)
            if v is None:
                continue
            res = _line_search(h, v, cur_c5)
            if res is not None:
                h, c5_new = res
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        
        # Random directions
        for _ in range(n_random):
            if time.time() > deadline:
                break
            v = _random_direction(rng, n)
            if v is None:
                continue
            res = _line_search(h, v, cur_c5)
            if res is not None:
                h, c5_new = res
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        
        if not improved:
            stall += 1
        else:
            stall = 0
    
    return best_h, best_c5

def _active_grad_direction(h, eps=1e-6):
    """Direction that reduces the active (near-maximum) scores."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    # Find active scores (within eps of max)
    active = scores >= max_s - eps
    if not np.any(active):
        active = np.zeros(len(scores), dtype=bool)
        active[np.argmax(scores)] = True
    
    # Equal weight on active scores
    w = np.zeros(len(scores))
    w[active] = 1.0 / np.sum(active)
    
    # Gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if w[r] < 1e-14:
            continue
        wd = w[r]
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] -= wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[g:]
    
    v = -grad
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

    scores_mat = A[np.newaxis, :] + cands[:, np.newaxis]*B[np.newaxis, :] + (cands[:, np.newaxis]**2)*C[np.newaxis, :]
    costs = scores_mat.max(axis=1)
    idx_best = int(np.argmin(costs))
    best_a = float(cands[idx_best])
    if abs(best_a) < 1e-14:
        return None
    h_new = h + best_a * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_box_sum(h_new, n / 2.0)
    c5_new = _exact_c5(h_new)
    if c5_new < cur_c5 - 1e-14:
        return h_new, c5_new
    return None

def _smooth_grad(h, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] -= wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[g:]
    v = -grad
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

def _active_grad(h, eps_frac=1e-4):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    thresh = max_s - eps_frac * max(1.0, max_s)
    active = scores >= thresh
    if not np.any(active):
        active = np.zeros(len(scores), dtype=bool)
        active[np.argmax(scores)] = True
    w = np.zeros(len(scores))
    w[active] = 1.0 / float(np.sum(active))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] -= wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[g:]
    v = -grad
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

def _rand_dir(rng, n):
    v = rng.standard_normal(n)
    v -= np.mean(v)
    norm = float(np.linalg.norm(v))
    if norm < 1e-14:
        return None
    return v / norm

def _pair_dir(n, i, j):
    v = np.zeros(n)
    v[i] = 1.0
    v[j] = -1.0
    return v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _local_descend(h, deadline, rng, n_rand=20, n_betas=4):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    betas = [5.0, 15.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1600.0]
    stall = 0
    max_stall = 25
    while time.time() < deadline and stall < max_stall:
        improved = False
        for beta in betas[:n_betas]:
            if time.time() > deadline:
                break
            v = _smooth_grad(h, beta)
            if v is None:
                continue
            res = _line_search(h, v, cur_c5)
            if res is not None:
                h, cur_c5 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        v_a = _active_grad(h)
        if v_a is not None:
            res = _line_search(h, v_a, cur_c5)
            if res is not None:
                h, cur_c5 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        for _ in range(n_rand):
            if time.time() > deadline:
                break
            v = _rand_dir(rng, n)
            if v is None:
                continue
            res = _line_search(h, v, cur_c5)
            if res is not None:
                h, cur_c5 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _pair_polish(h, deadline, rng, max_passes=50):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n_pairs = n * (n - 1) // 2
    for p in range(max_passes):
        if time.time() > deadline:
            break
        pairs = list(range(n_pairs))
        rng.shuffle(pairs)
        any_improve = False
        for pi in pairs:
            if time.time() > deadline:
                break
            i = pi // (n - 1)
            j = pi % (n - 1)
            if j >= i:
                j += 1
            if i == j or i >= n or j >= n:
                continue
            v = _pair_dir(n, i, j)
            res = _line_search(h, v, cur_c5)
            if res is not None:
                h, cur_c5 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                any_improve = True
        if not any_improve:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.97
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = _project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Intensive line descent at parent resolution
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _local_descend(h, dl1, rng, n_rand=40, n_betas=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Exhaustive pair polish
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _pair_polish(best_h, dl2, rng, max_passes=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More line descent
    dl3 = t0 + budget_s * 0.50
    h3, c5_3 = _local_descend(best_h, dl3, rng, n_rand=30, n_betas=8)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair polish again
    dl4 = t0 + budget_s * 0.60
    h4, c5_4 = _pair_polish(best_h, dl4, rng, max_passes=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Transformations
    if time.time() < t0 + budget_s * 0.70:
        transforms = [
            best_h[::-1].copy(),
            np.roll(best_h, 1).copy(),
            np.roll(best_h, -1).copy(),
            np.roll(best_h, n // 2).copy(),
            np.roll(best_h, n // 3).copy(),
            np.roll(best_h, n // 4).copy(),
        ]
        for t_h in transforms:
            if time.time() > t0 + budget_s * 0.70:
                break
            t_h = _project_box_sum(t_h, n / 2.0)
            c5_t = _exact_c5(t_h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = t_h.copy()
                dl_t = min(t0 + budget_s * 0.70, time.time() + 15)
                ht, ct = _local_descend(best_h, dl_t, rng, n_rand=10, n_betas=4)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = ht.copy()

    # Phase 6: Multi-resolution
    dl6 = t0 + budget_s * 0.85
    n_tries = sorted(set(x for x in [n-8, n-4, n-2, n+2, n+4, n+8, n+16, 48, 56, 72, 80, 96, 112] if 24 <= x <= 192 and x != n))
    for n_try in n_tries:
        if time.time() > dl6:
            break
        h_r = _resample(best_h, n_try)
        dl_r = min(dl6, time.time() + 25)
        h_r, c5_r = _local_descend(h_r, dl_r, rng, n_rand=15, n_betas=4)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if time.time() < dl6:
                hp, cp = _pair_polish(best_h, min(dl6, time.time() + 15), rng, max_passes=10)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = hp.copy()

    # Phase 7: Final intensive line descent
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.93
        h7, c5_7 = _local_descend(best_h, dl7, rng, n_rand=40, n_betas=8)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final pair polish
    if time.time() < t0 + budget_s * 0.96:
        dl8 = t0 + budget_s * 0.96
        h8, c5_8 = _pair_polish(best_h, dl8, rng, max_passes=20)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
