```python
def line_search_exact(h, d, S0, a, b, t_min, t_max):
    """Find t in [t_min, t_max] that minimizes max_k (S0[k] + t*a[k] - t^2*b[k]).
    Returns (best_t, best_value)."""
    n = len(h)
    dx = 2.0 / n
    
    def f(t):
        vals = S0 + t * a - t * t * b
        return float(np.max(vals)) * dx
    
    t0 = (t_min + t_max) / 2
    f0 = f(0)
    
    # Grid search
    n_grid = 40
    ts = np.linspace(t_min, t_max, n_grid)
    fs = np.array([f(t) for t in ts])
    i_best = int(np.argmin(fs))
    
    # Golden section around best
    lo = ts[max(0, i_best-2)]
    hi = ts[min(n_grid-1, i_best+2)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)
    
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    t_best = (lo + hi) / 2
    f_best = f(t_best)
    
    # Compare with t=0
    if f_best < f0:
        return t_best, f_best
    return 0.0, f0
```

This is very efficient! Each evaluation of f(t) is O(n) (just a vectorized max), and we do about 90 evaluations per line search.

```python
def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _lag_grad(h, k):
    """Gradient of lag k correlation w.r.t. h."""
    n = len(h)
    shift = k - (n - 1)
    grad = np.zeros(n)
    m_start = max(0, shift)
    m_end = min(n, n + shift)
    if m_start < m_end:
        idx = np.arange(m_start, m_end)
        grad[idx] = (1.0 - h[idx - shift])
    m_start2 = max(0, -shift)
    m_end2 = min(n, n - shift)
    if m_start2 < m_end2:
        idx2 = np.arange(m_start2, m_end2)
        grad[idx2] += (-h[idx2 + shift])
    return grad

def _subgradient_direction(h, S0, n_active_max=5):
    """Compute negated mean-zero subgradient of max correlation."""
    n = len(h)
    max_val = float(np.max(S0))
    # Find lags close to max
    active = np.where(S0 >= max_val - 1e-8)[0]
    if len(active) > n_active_max:
        # Take top
        order = np.argsort(S0[:len(active)])[::-1][:n_active_max]
        active = active[order]
    
    g = np.zeros(n)
    for k in active:
        g += _lag_grad(h, int(k))
    g /= len(active)
    # Negate for descent
    d = -g
    # Make mean-zero
    d -= np.mean(d)
    return d

    d = rng.standard_normal(n)
    d -= np.mean(d)
    # Scale: make sure it's not too large
    norm = np.linalg.norm(d)
    if norm > 1e-10:

    return d
```
