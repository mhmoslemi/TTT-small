def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid
    tau = (lo + hi) / 2.0

        else:

        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success:
            h_new = _project(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)


def _lp_direction_fast(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        if m > 0:
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] -= h[d:]
    for d in range(1, n):
        r = n - 1 - d
        m = n - d
        G[r, :m] += (1.0 - h[d:])
        G[r, d:] -= h[:m]
    c = np.zeros(n)
    c[:] = 1.0
    A_ub = G
    b_ub = -scores
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_polish(h, deadline, radius_init=0.06, max_iters=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction_fast(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.85
            if radius < 1e-6:
                break
            continue
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-12:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a = 0.0
        hi_a = min(alpha_max, 2.0)
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(30):
            if hi_a - lo_a < 1e-12:
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1
                f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo_a = c1
                c1 = c2
                f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo_a + hi_a) / 2.0
        h_new = _project(h + opt_a * p, target)
        cur_c5_new = _exact_c5(h_new)
        if cur_c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = cur_c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.3)
        else:
            stall += 1
            radius *= 0.8
    return best_h, best_c5


def _freq_perturb(h, rng, strength=0.15):
    n = len(h)
    h_ac = h - np.mean(h)
    H = np.fft.rfft(h_ac)
    pert = rng.normal(0, 1, len(H))
    mask = np.ones(len(H))
    mask[0] = 0.0
    for k in range(1, len(H)):
        mask[k] = strength / (1.0 + 0.3 * k)
    H_new = H + pert * mask
    h_new = np.fft.irfft(H_new, n) + np.mean(h)
    return _project(h_new, n / 2.0)


def _shift_perturb(h, rng):
    n = len(h)
    s = rng.integers(1, max(2, n // 4))
    h_new = np.roll(h, int(s))
    return _project(h_new, n / 2.0)


def _random_mutation(h, rng, scale=0.08):
    n = len(h)
    r = rng.random()
    if r < 0.4:
        h_pert = h + rng.normal(0, scale, n)
    elif r < 0.6:
        return _freq_perturb(h, rng, strength=scale * 2)
    elif r < 0.75:
        return _shift_perturb(h, rng)
    else:
        mask = rng.random(n) < 0.3
        h_pert = h.copy()
        h_pert[mask] += rng.normal(0, scale * 1.5, int(np.sum(mask)))
    return _project(h_pert, n / 2.0)


def _local_refine(h, deadline):
    n = len(h)
    h = h.copy()
    c5 = _exact_c5(h)
    # Short gradient steps
    dl1 = min(time.time() + 0.5, deadline)
    for i in range(500):
        if time.time() > dl1:
            break
        beta = 20.0 + 50.0 * (i / 500.0)
        h, c5_new = _grad_step_with_line(h, beta, lr_init=0.3)
        if c5_new < c5:
            c5 = c5_new
        else:
            break
    # LP polish
    dl2 = min(time.time() + 2.0, deadline)
    h, c5 = _lp_polish(h, dl2, radius_init=0.04, max_iters=500)
    # SLSQP
    if time.time() < deadline - 1.0:
        h, c5 = _slsqp_refine(h, deadline, beta=100.0, max_iter=100)
    return h, c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new <= 2:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _thue_morse_init(n, amplitude=0.3):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial point
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    best_h = _project(h0, target0)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    # Phase 1: Adam descent from parent (15% budget)
    dl1 = t0 + budget_s * 0.15
    h, c5 = _adam_descent(best_h, dl1, lr=0.2, b1=0.85, b2=0.999,
                          beta_start=5.0, beta_end=6000.0, soft_iters=1500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP polish + SLSQP (5% budget)
    dl2 = t0 + budget_s * 0.20
    h, c5 = _lp_polish(best_h, dl2, radius_init=0.05, max_iters=3000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < dl2:
        h, c5 = _slsqp_refine(best_h, dl2, beta=120.0, max_iter=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Basin hopping with diverse mutations (55% budget)
    dl3 = t0 + budget_s * 0.75
    scale = 0.06
    accept_rate = 0
    total_iters = 0
    while time.time() < dl3:
        total_iters += 1
        # Mutate
        h_pert = _random_mutation(best_h, rng, scale)
        # Local refine
        time_left = dl3 - time.time()
        if time_left < 5.0:
            break
        alloc = min(time_left * 0.15, 8.0)
        h_try, c5_try = _local_refine(h_pert, time.time() + alloc)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            accept_rate += 1
            scale = max(0.01, scale * 0.9)
        else:
            scale = min(0.3, scale * 1.08)
            if total_iters > 20 and accept_rate < 3:
                scale = min(0.4, scale * 1.5)

    # Phase 4: Multi-n exploration (15% budget)
    dl4 = t0 + budget_s * 0.90
    n_list = []
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, 12, -12]:
        nt = best_n + dn
        if 32 <= nt <= 128:
            n_list.append(nt)
    n_list = list(dict.fromkeys(n_list))
    for n_try in n_list:
        if time.time() > dl4:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        alloc = min((dl4 - time.time()) / max(len(n_list), 1), 6.0)
        if alloc < 2.0:
            break
        h_r, c5_r = _adam_descent(h_r, time.time() + alloc * 0.6,
                                  lr=0.15, b1=0.85, b2=0.999,
                                  beta_start=10.0, beta_end=4000.0, soft_iters=800)
        h_r, c5_r = _lp_polish(h_r, time.time() + alloc * 0.3,
                               radius_init=0.04, max_iters=800)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Final intensive polish (10% budget)
    dl5 = t0 + budget_s * 0.99
    # LP polish with very small radius
    h, c5 = _lp_polish(best_h, min(dl5, time.time() + budget_s * 0.03),
                       radius_init=0.02, max_iters=5000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    # SLSQP final
    if time.time() < dl5 - 2.0:
        h, c5 = _slsqp_refine(best_h, min(dl5, time.time() + 15.0),
                              beta=200.0, max_iter=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
