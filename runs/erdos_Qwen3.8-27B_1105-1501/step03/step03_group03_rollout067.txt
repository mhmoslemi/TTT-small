import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_grad_vec(h):
    n = len(h)
    L = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:n] += (1.0 - h[:m])
            G[r, :m] -= h[d:n]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:n])
            G[r, g:n] -= h[:m]
    return scores, G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, scores, G, radius):
    n = len(h)
    L = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = c5_cur
    # Coarse scan
    n_eval = 30
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Fine golden section
    span = alpha_max * 0.15
    lo = max(0.0, best_a - span)
    hi = min(alpha_max, best_a + span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores, G = _scores_grad_vec(h)
        
        # Try LP direction
        p = _lp_direction(h, scores, G, radius)
        
        if p is not None and float(np.max(np.abs(p))) > 1e-12:
            # Try forward
            alpha, new_c5 = _line_search(h, p, cur_c5)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.1, 0.5)
                continue
            
            # Try backward
            alpha_b, new_c5_b = _line_search(h, -p, cur_c5)
            if new_c5_b < cur_c5 - 1e-13:
                h = h - alpha_b * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.1, 0.5)
                continue
        
        stall += 1
        radius *= 0.85
        if radius < 1e-6:
            break
        if stall % 15 == 0 and radius < 0.02:
            radius = 0.05
    
    return best_h, best_c5

def _subgrad_descent(h, deadline, max_iters=3000, step_init=0.005):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        
        # Active set: scores within 2% of max
        eps = 0.02 * s_max + 1e-6
        active = scores > (s_max - eps)
        if not np.any(active):
            active = scores == s_max
        
        _, G = _scores_grad_vec(h)
        grad_avg = G[active].mean(axis=0)
        grad_avg -= np.mean(grad_avg)  # zero-sum
        
        norm = np.sqrt(np.sum(grad_avg**2))
        if norm < 1e-12:
            stall += 1
            continue
        
        # Try asymmetric active sets
        for frac in [0.5, 0.25]:
            eps_f = frac * s_max * 0.05
            active_f = scores > (s_max - eps_f)
            if not np.any(active_f):
                continue
            grad_f = G[active_f].mean(axis=0)
            grad_f -= np.mean(grad_f)
            norm_f = np.sqrt(np.sum(grad_f**2))
            if norm_f < 1e-12:
                continue
            p = -grad_f / norm_f
            h_try = _project(h + step * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                step *= 1.05
                stall = 0
                break
        else:
            # Try standard direction
            p = -grad_avg / norm
            h_try = _project(h + step * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                step *= 1.05
                stall = 0
            else:
                step *= 0.7
                stall += 1
    
    return best_h, best_c5
