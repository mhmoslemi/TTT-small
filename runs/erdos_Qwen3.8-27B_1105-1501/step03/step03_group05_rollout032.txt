def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    x = np.clip(v - shift, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_quad_coeffs(h, i, j):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _binary_random_perturb(h, n, rng, frac=0.1):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin == 1.0))
    n_swap = max(1, int(n * frac))
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    for _ in range(n_swap):
        if not ones or not zeros:
            break
        oi = rng.integers(len(ones))
        ji = rng.integers(len(zeros))
        o = ones[oi]
        j = zeros[ji]
        h_bin[o] = 0.0
        h_bin[j] = 1.0
        ones[oi] = j
        zeros[ji] = o
    return h_bin
