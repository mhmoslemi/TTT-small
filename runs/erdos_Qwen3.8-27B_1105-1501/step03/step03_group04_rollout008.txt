```python
def _grad_for_lags(h, lags):
    """Compute gradient contributions for specific lags."""
    n = len(h)
    grad = np.zeros(n)
    one_m_h = 1.0 - h
    for d in lags:
        # d ranges from -(n-1) to n-1
        if d >= 0:
            # Contribution from (1-h_{j-d}) for j=d..n-1: grad[d:] += one_m_h[:n-d]
            # Contribution from -h_{j+d} for j=0..n-1-d: grad[:n-d] -= h[d:]
            grad[d:] += one_m_h[:n-d]
            grad[:n-d] -= h[d:]
        else:
            g = -d
            # Contribution from (1-h_{j+g}) for j=0..n-1-g: grad[:n-g] += one_m_h[g:]
            # Contribution from -h_{j-g} for j=g..n-1: grad[g:] -= h[:n-g]
            grad[:n-g] += one_m_h[g:]
            grad[g:] -= h[:n-g]
    return grad
```

This is O(M*n) for M lags, which is much faster than O(n²) when M << n.

∂/∂h_j (∑_k c_k^p) = p * ∑_k c_k^{p-1} * (∂c_k/∂h_j)

So the gradient is: p * (weights @ G) where weights_k = c_k^{p-1} and G[k,j] = ∂c_k/∂h_j.

With the selective gradient computation, this becomes:
grad = p * ∑_{k in top_M} c_k^{p-1} * (∂c/∂h)_k

Which is O(M*n) per iteration.

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}."""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
