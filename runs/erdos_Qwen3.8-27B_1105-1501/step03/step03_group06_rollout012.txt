import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _corr_grads(h):
    """Return all gradients of correlation values w.r.t. h.
    Shape: (2n-1, n) - gradient of each correlation value."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _slsqp_epigraph(h, T_init, top_k=10, maxiter=300, deadline=None):
    """SLSQP on epigraph with top_k active constraints."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    # Build initial constraint indices
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Sort by value, take top_k
    idx = np.argsort(corr)[-top_k:]
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    constraints = []
    for ci in idx:
        def make_constraint(ci):
            def con(x):
                h_arr = x[1:]
                corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
                return x[0] - corr[ci]
            def con_grad(x):
                h_arr = x[1:]
                G = _corr_grads(h_arr)
                g = np.zeros(n + 1)
                g[0] = 1.0
                g[1:] -= G[ci]
                return g
            return {'type': 'ineq', 'fun': con, 'jac': con_grad}
        constraints.append(make_constraint(ci))
    
    # Equality constraint: sum(h) = target
    con_eq_fun = lambda x: np.sum(x[1:]) - target
    con_eq_grad = lambda x: np.concatenate([[0.0], np.ones(n)])
    constraints.append({'type': 'eq', 'fun': con_eq_fun, 'jac': con_eq_grad})
    
    bounds = [(T_init * 0.5, T_init * 2.0)] + [(0.0, 1.0)] * n
    
    x0 = np.concatenate([[T_init], h])
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h.copy()

def _active_set_grad(h, alpha, top_frac=0.1):
    """Compute active-set gradient direction (negative)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    # Soft weights
    vals = np.clip(alpha * (corr - ms), -100.0, 100.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    G = _corr_grads(h)
    d = -(G.T @ w)
    d -= np.mean(d)  # maintain sum constraint
    return d, ms, corr

def _line_search(h, d, c5_cur, max_step=1.0):
    """Line search along direction d with projection."""
    n = len(h)
    target = n / 2.0
    
    # Find max step before hitting bounds
    step_max = max_step
    for j in range(n):
        if d[j] > 1e-14:
            step_max = min(step_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            step_max = min(step_max, h[j] / (-d[j]))
    step_max = max(step_max, 1e-10)
    
    best_a, best_c5 = 0.0, c5_cur
    
    # Coarse search
    for t in np.linspace(0.0, 1.0, 16):
        a = t * step_max
        if a < 1e-10:
            continue
        h_t = np.clip(h + a * d, 0.0, 1.0)
        h_t = _project(h_t, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    
    # Fine golden section search
    if best_a > 0:
        lo = max(0.0, best_a - step_max * 0.1)
        hi = best_a + step_max * 0.1
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(np.clip(h + c1 * d, 0.0, 1.0), target))
        f2 = _exact_c5(_project(np.clip(h + c2 * d, 0.0, 1.0), target))
        for _ in range(20):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(np.clip(h + c1 * d, 0.0, 1.0), target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(np.clip(h + c2 * d, 0.0, 1.0), target))

        c5_t = _exact_c5(_project(np.clip(h + a * d, 0.0, 1.0), target))

def _active_set_loop(h, deadline, max_iters=2000):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

    cur_c5 = best_c5
    alpha = 10.0
    stall = 0
    step_scale = 0.1

        if time.time() > deadline:
            break

            break
        
        d, ms, corr = _active_set_grad(h, alpha)
        norm = float(np.linalg.norm(d))

def _lp_active(h, deadline, max_iters=2000, top_k=8):
    """LP-based direction finding on active set."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        ms = float(np.max(corr))
        # Get top_k active constraints
        idx = np.argsort(corr)[-top_k:]
        
        G = _corr_grads(h)
        # LP: find direction d that reduces all top_k correlations
        # minimize -sum of reductions (or max reduction of worst)
        # s.t. G[idx] @ d <= -eps (reduce each active by at least eps)
        #      sum(d) = 0
        #      |d| <= radius
        
        radius = 0.1
        # Objective: minimize -sum(d_reductions) = maximize sum of reductions
        # Actually: minimize -1^T * (G[idx] @ d) = minimize -G[idx].sum(axis=0) @ d
        c = np.zeros(n + 1)
        c[:n] = -G[idx].sum(axis=0)
        c[n] = 0  # free variable for slack
        
        # Constraints:
        # G[idx] @ d + [0,1,0,...] * t <= -eps  (reduce each by at least eps)
        eps = 0.01
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = G[idx]
        b_ub = -eps
        
        # Symmetry constraint: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                a, new_c5 = _line_search(h, p, cur_c5, max_step=1.0)
                if new_c5 < cur_c5 - 1e-13:
                    h = _project(np.clip(h + a * p, 0.0, 1.0), target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()
                    stall = 0
                else:
                    stall += 1
                    eps *= 0.5
                    if eps < 0.001:
                        break
            else:
                stall += 1
                eps *= 0.5
                if eps < 0.001:
                    break
        except Exception:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
