            continue
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5
    lo = max(0.0, best_a - alpha_max / n_steps)
    hi = min(alpha_max, best_a + alpha_max / n_steps)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(12):
        if hi - lo < 1e-9 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _gradient_refine(h, deadline, beta_start=5.0, beta_end=300.0, n_beta_steps=12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta_list = np.geomspace(beta_start, beta_end, n_beta_steps)
    for beta in beta_list:
        if time.time() > deadline:
            break
        step_size = 0.1
        stall = 0
        for _ in range(5000):
            if time.time() > deadline or stall > 80:
                break
            g = _smooth_grad(h, beta)
            g -= np.mean(g)
            norm = float(np.max(np.abs(g)))
            if norm < 1e-14:
                break
            d = -g / norm * step_size
            a, new_c5 = _line_search(h, d, cur_c5)
            if new_c5 < cur_c5 - 1e-12:
                h = _project_box_sum(h + a * d, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step_size = min(step_size * 1.1, 0.5)
            else:
                stall += 1
                step_size *= 0.7
    return best_h, best_c5

def _binary_local_search(h, deadline, max_iters=60000, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline or no_improve > 1500:
            break
        if not ones or not zeros:
            break
        best_swap = None
        best_val = cur_c5
        batch = min(20, len(ones) * len(zeros))
        for _b in range(batch):
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def _block_pattern(n, rng):
    h = np.zeros(n)
    block_size = int(rng.integers(4, max(5, n // 4)))
    num_blocks = max(1, (n // 2) // block_size)
    positions = rng.choice(n, size=num_blocks, replace=False)
    for pos in positions:
        start = pos % n
        for k in range(block_size):
            h[(start + k) % n] = 1.0
    return _project_box_sum(h, n / 2.0)

def _pair_transfer_refine(h, deadline, max_iters=30000, rng=None):
    if rng is None:
        rng = np.random.default_rng(1)
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline or no_improve > 500:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = float(rng.uniform(0.05, 0.3))
        h[i] -= delta
        h[j] += delta
        c5_new = _exact_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] += delta
            h[j] -= delta
            no_improve += 1
    return best_h, best_c5

def _lp_direction(h, A, F, radius, top_k=50):
    n = len(h)
    idx = np.argsort(-F)[:top_k]
    A_sub = A[idx]
    F_sub = F[idx]
    G = A_sub
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(F_sub), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -F_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 300})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15, max_iters=2000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    for _ in range(max_iters):
        if time.time() > deadline or stall > 40:
            break
        A, F = _build_grad_matrix(h)
        p = _lp_direction(h, A, F, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        a, new_c5 = _line_search(h, p, cur_c5, n_steps=12)
        if new_c5 < cur_c5 - 1e-11:
            h = _project_box_sum(h + a * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_end = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)
    solver_list = ['CLARABEL', 'ECOS', 'SCS']

    # Get initial point
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def _accept(h_new, c5_new):
        nonlocal best_h, best_c5, best_n
        if h_new is not None and c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = len(h_new)

    # Validate and set initial
    if h0 is not None and len(h0) > 0:
        h0 = _project_box_sum(h0, len(h0) / 2.0)
        _accept(h0, _exact_c5(h0))
    else:
        n_default = 64
        h0 = _random_binary(n_default, rng)
        _accept(h0, _exact_c5(h0))

    # Phase 1: Trust-region refinement from parent (25% budget)
    deadline1 = t0 + budget_s * 0.25
    h1, c5_1 = _refine_trust(best_h, deadline1, rho_init=1.5, n_active=30, solver_list=solver_list)
    _accept(h1, c5_1)

    # Phase 2: Gradient refinement (10%)
    deadline2 = t0 + budget_s * 0.35
    h2, c5_2 = _gradient_refine(best_h, deadline2, beta_start=5.0, beta_end=200.0, n_beta_steps=10)
    _accept(h2, c5_2)

    # Phase 3: LP refinement (10%)
    deadline3 = t0 + budget_s * 0.45
    h3, c5_3 = _lp_loop(best_h, deadline3, radius_init=0.15, max_iters=2000)
    _accept(h3, c5_3)

    # Phase 4: Trust-region again (10%)
    deadline4 = t0 + budget_s * 0.55
    h4, c5_4 = _refine_trust(best_h, deadline4, rho_init=0.8, n_active=25, solver_list=solver_list)
    _accept(h4, c5_4)

    # Phase 5: Binary local search (10%)
    deadline5 = t0 + budget_s * 0.65
    hb, c5b = _binary_local_search(best_h, deadline5, max_iters=50000, rng=rng)
    _accept(hb, c5b)

    # Phase 6: Multi-start with different resolutions (15%)
    deadline6 = t0 + budget_s * 0.80
    n_candidates = sorted(set([48, 52, 56, 60, 64, 68, 72, 76, 80]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 128]
    per_n_time = budget_s * 0.02
    for n_try in n_candidates:
        if time.time() > deadline6:
            break
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline6, time.time() + per_n_time)
        h_r, c5_r = _refine_trust(h_r, dl_r, rho_init=1.0, n_active=20, solver_list=solver_list)
        _accept(h_r, c5_r)
        if time.time() < deadline6:
            dl_r2 = min(deadline6, time.time() + per_n_time * 0.5)
            h_r2, c5_r2 = _gradient_refine(h_r, dl_r2, beta_start=10.0, beta_end=100.0, n_beta_steps=5)
            _accept(h_r2, c5_r2)

    # Phase 7: Pair transfer refinement (5%)
    deadline7 = t0 + budget_s * 0.88
    hp, c5p = _pair_transfer_refine(best_h, deadline7, max_iters=20000, rng=rng)
    _accept(hp, c5p)

    # Phase 8: Final trust-region polish on best (5%)
    if time.time() < safe_end:
        deadline8 = safe_end
        h8, c5_8 = _refine_trust(best_h, deadline8, rho_init=0.5, n_active=15, solver_list=solver_list)
        _accept(h8, c5_8)

    # Phase 9: Final gradient polish
    if time.time() < safe_end - 5:
        deadline9 = safe_end
        h9, c5_9 = _gradient_refine(best_h, deadline9, beta_start=50.0, beta_end=500.0, n_beta_steps=5)
        _accept(h9, c5_9)

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = _exact_c5(best_h)
        best_n = len(best_h)

    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
