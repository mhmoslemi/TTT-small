    n = len(h)
    # Minimize t s.t. G @ (h + p) <= scores + t*0 ... 
    # Actually: minimize max_k (scores[k] + G[k] @ p)
    # Linearize: min t s.t. G[k] @ p + scores[k] - scores_max <= t - scores_max
    # i.e., G[k] @ p - t <= scores_max - scores[k] for all k
    # with ||p||_inf <= radius, sum(p) = 0
    
    smax = float(np.max(scores))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = smax - scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline, radius_init=0.2, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline or stall > 80:
            break
        scores = _scores(h)
        G = _all_grads(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        # Project p to maintain sum
        p -= np.mean(p)
        a, new_c5 = _line_search(h, p, cur_c5, 1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.8
            if stall % 20 == 0 and radius < 0.05:
                radius = 0.1
    return best_h, best_c5

def _tiny_perturb(h, deadline, n_tries=5000):
    """Try very small random perturbations and keep improvements."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        # Perturb 1-3 random coordinates
        n_p = rng.integers(1, 4)
        idx = rng.choice(n, size=n_p, replace=False)
        h_try = h.copy()
        h_try[idx] += rng.normal(0, 0.02, n_p)
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    
    return best_h, best_c5
