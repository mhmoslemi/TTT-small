    Returns (optimal_alpha, optimal_value_before_scaling).

        val = np.max(corr + alpha * slope)
        return alpha, float(val)

    fa = float(np.max(corr + a * slope))
    fb = float(np.max(corr + b * slope))

    # At endpoint a: if all active slopes >= 0, minimum is at a
    # At endpoint b: if all active slopes <= 0, minimum is at b

    max_a = float(np.max(active_a))

        return b, max_b
    
    # Bisection for interior minimum
    lo, hi = a, b
    for _ in range(60):

def _pair_gradient(h, i, j):

    Returns array of length 2n-1, indexed by d + (n-1).

    n = len(h)

    slope = np.zeros(2 * n - 1)
    for d in range(-nm1, nm1 + 1):
        rd = d + nm1
        # d corr[d]/d h[k] = (1-h[k-d])*I[0<=k-d<n] - h[k+d]*I[0<=k+d<n]
        # For k = i:
        s_i = 0.0

        if 0 <= j - d < n:
            s_j += (1.0 - h[j - d])
        if 0 <= j + d < n:
            s_j += (-h[j + d])
        slope[rd] = s_i - s_j
    return slope

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_slope(h, i, j):
    """Compute slope[d] for transfer h[i]+=alpha, h[j]-=alpha."""
    n = len(h)
    nm1 = n - 1
    d = np.arange(-nm1, nm1 + 1)
    
    id_arr = i - d
    mask1 = (id_arr >= 0) & (id_arr < n)
    iplus = i + d
    mask2 = (iplus >= 0) & (iplus < n)
    
    grad_i = np.zeros(2 * n - 1)
    grad_i[mask1] = 1.0 - h[id_arr[mask1]]
    grad_i[mask2] += -h[iplus[mask2]]
    
    jd_arr = j - d
    mask3 = (jd_arr >= 0) & (jd_arr < n)
    jplus = j + d
    mask4 = (jplus >= 0) & (jplus < n)
    
    grad_j = np.zeros(2 * n - 1)
    grad_j[mask3] = 1.0 - h[jd_arr[mask3]]
    grad_j[mask4] += -h[jplus[mask4]]
    
    return grad_i - grad_j

def _opt_alpha(corr, slope, a, b):
    """Minimize max_d(corr[d] + alpha * slope[d]) over alpha in [a, b].
    Returns (opt_alpha, opt_value).
    """
    if b - a < 1e-15:
        val = float(np.max(corr + a * slope))
        return a, val
    
    # Check if minimum is at left endpoint
    fa_arr = corr + a * slope
    fa = float(np.max(fa_arr))
    active_a = fa_arr >= fa - 1e-13
    if np.all(slope[active_a] >= -1e-13):
        return a, fa
    
    # Check if minimum is at right endpoint
    fb_arr = corr + b * slope
    fb = float(np.max(fb_arr))
    active_b = fb_arr >= fb - 1e-13
    if np.all(slope[active_b] <= 1e-13):
        return b, fb
    
    # Bisection
    lo, hi = a, b
    for _ in range(50):
        mid = (lo + hi) / 2.0
        arr = corr + mid * slope
        mv = float(np.max(arr))
        am = arr >= mv - 1e-12
        ms = float(np.max(slope[am]))
        mns = float(np.min(slope[am]))
        if ms < -1e-13:
            lo = mid
        elif mns > 1e-13:
            hi = mid
        else:
            break
    
    alpha = (lo + hi) / 2.0
    val = float(np.max(corr + alpha * slope))
    return alpha, val

def _pair_descent_pass(h, deadline, rng, order='sequential'):
    """One pass of pairwise exact descent. Returns (h, improved)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    corr = _scores(h)
    improved = False
    total_improvement = 0.0
    
    if order == 'random':
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for (i, j) in pairs:
        if time.time() > deadline:
            break
        # Feasible range for alpha (h[i] += alpha, h[j] -= alpha)
        a = max(-h[i], h[j] - 1.0)
        b = min(1.0 - h[i], h[j])
        if b - a < 1e-14:
            continue
        
        slope = _pair_slope(h, i, j)
        opt_a, opt_val = _opt_alpha(corr, slope, a, b)
        
        # Need to recompute the actual C5 after the move (because changing h
        # changes the scores for all lags, not just via the linear approximation)
        # Wait - the scores change linearly with alpha for a FIXED pair transfer.
        # But after applying the move, the scores for the NEXT pair will be different.
        # So I need to update corr after each accepted move.
        
        # The linear approximation is EXACT for the scores after the move:
        # new_corr[d] = corr[d] + opt_a * slope[d]
        # This is because each score is linear in each h[k], and we're changing
        # exactly two coordinates by ±alpha.
        
        new_corr = corr + opt_a * slope
        new_c5 = float(np.max(new_corr)) * (2.0 / n)
        cur_c5 = float(np.max(corr)) * (2.0 / n)
        
        if new_c5 < cur_c5 - 1e-14:
            h[i] += opt_a
            h[j] -= opt_a
            h = np.clip(h, 0.0, 1.0)
            # Re-exactify the sum
            h = _project_box_sum(h, target)
            # Recompute exact scores
            corr = _scores(h)
            improved = True
            total_improvement += (cur_c5 - new_c5)
    
    return h, improved

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_feasible(n, rng):
    """Generate a random feasible vector."""
    h = rng.random(n)
    return _project_box_sum(h, n / 2.0)

def _random_binary(n, rng):
    """Generate a random binary feasible vector."""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def _opt_alpha(corr, slope, q, a, b):
    """Minimize max_d(corr[d] + alpha*slope[d] + alpha^2*q[d]) over alpha in [a,b].
    q[d] is 0 or 1 (1 for the two special lags).
    """
    if b - a < 1e-15:
        val = float(np.max(corr + a * slope + a * a * q))
        return a, val
    
    # Check left endpoint
    fa_arr = corr + a * slope + a * a * q
    fa = float(np.max(fa_arr))
    active_a = fa_arr >= fa - 1e-12
    # Derivatives of active functions at a
    deriv_a = slope[active_a] + 2 * a * q[active_a]
    if np.all(deriv_a >= -1e-12):
        return a, fa
    
    # Check right endpoint
    fb_arr = corr + b * slope + b * b * q
    fb = float(np.max(fb_arr))
    active_b = fb_arr >= fb - 1e-12
    deriv_b = slope[active_b] + 2 * b * q[active_b]
    if np.all(deriv_b <= 1e-12):
        return b, fb
    
    # Bisection
    lo, hi = a, b
    for _ in range(60):
        mid = (lo + hi) / 2.0
        arr = corr + mid * slope + mid * mid * q
        mv = float(np.max(arr))
        am = arr >= mv - 1e-11
        deriv = slope[am] + 2 * mid * q[am]
        if np.max(deriv) < -1e-12:
            lo = mid
        elif np.min(deriv) > 1e-12:
            hi = mid
        else:
            break
    
    alpha = (lo + hi) / 2.0
    val = float(np.max(corr + alpha * slope + alpha * alpha * q))

def _pair_descent_pass(h, deadline, rng, order='sequential'):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    corr = _scores(h)
    improved = False
    
    if order == 'random':
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for (i, j) in pairs:
        if time.time() > deadline:
            break
        a = max(-h[i], h[j] - 1.0)
        b = min(1.0 - h[i], h[j])
        if b - a < 1e-14:
            continue
        
        slope = _pair_slope(h, i, j)
        # Compute q: 1 at positions d=j-i and d=i-j (if valid lags)
        q = np.zeros(2 * n - 1)
        nm1 = n - 1
        d_pos = j - i  # this is d = j-i, index is d_pos + nm1
        if -nm1 <= d_pos <= nm1:
            # The term at m=i: h[j]*(1-h[i]) exists always (i,j in range)
            q[d_pos + nm1] = 1.0
        d_neg = i - j
        if -nm1 <= d_neg <= nm1:
            q[d_neg + nm1] = 1.0
        
        opt_a, opt_val = _opt_alpha(corr, slope, q, a, b)
        
        new_corr = corr + opt_a * slope + opt_a * opt_a * q
        new_c5 = float(np.max(new_corr)) * (2.0 / n)
        cur_c5 = float(np.max(corr)) * (2.0 / n)
        
        if new_c5 < cur_c5 - 1e-14:
            h[i] += opt_a
            h[j] -= opt_a
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, target)
            corr = _scores(h)
            improved = True
    
    return h, improved

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_slope(h, i, j):
    n = len(h)
    nm1 = n - 1
    d = np.arange(-nm1, nm1 + 1)
    id_arr = i - d
    mask1 = (id_arr >= 0) & (id_arr < n)
    iplus = i + d
    mask2 = (iplus >= 0) & (iplus < n)
    grad_i = np.zeros(2 * n - 1)
    grad_i[mask1] = 1.0 - h[id_arr[mask1]]
    grad_i[mask2] += -h[iplus[mask2]]
    jd_arr = j - d
    mask3 = (jd_arr >= 0) & (jd_arr < n)
    jplus = j + d
    mask4 = (jplus >= 0) & (jplus < n)
    grad_j = np.zeros(2 * n - 1)
    grad_j[mask3] = 1.0 - h[jd_arr[mask3]]
    grad_j[mask4] += -h[jplus[mask4]]
    return grad_i - grad_j

def _pair_q(i, j, n):
    nm1 = n - 1
    q = np.zeros(2 * n - 1)
    d_pos = j - i
    if -nm1 <= d_pos <= nm1:
        q[d_pos + nm1] = 1.0
    d_neg = i - j
    if -nm1 <= d_neg <= nm1:
        q[d_neg + nm1] = 1.0
    return q

def _opt_alpha(corr, slope, q, a, b):
    if b - a < 1e-15:
        val = float(np.max(corr + a * slope + a * a * q))
        return a, val
    fa_arr = corr + a * slope + a * a * q
    fa = float(np.max(fa_arr))
    active_a = fa_arr >= fa - 1e-12
    deriv_a = slope[active_a] + 2.0 * a * q[active_a]
    if np.all(deriv_a >= -1e-12):
        return a, fa
    fb_arr = corr + b * slope + b * b * q
    fb = float(np.max(fb_arr))
    active_b = fb_arr >= fb - 1e-12
    deriv_b = slope[active_b] + 2.0 * b * q[active_b]
    if np.all(deriv_b <= 1e-12):
        return b, fb
    lo, hi = a, b
    for _ in range(60):
        mid = (lo + hi) / 2.0
        arr = corr + mid * slope + mid * mid * q
        mv = float(np.max(arr))
        am = arr >= mv - 1e-11
        deriv = slope[am] + 2.0 * mid * q[am]
        if np.max(deriv) < -1e-12:
            lo = mid
        elif np.min(deriv) > 1e-12:
            hi = mid
        else:
            break
    alpha = (lo + hi) / 2.0
    val = float(np.max(corr + alpha * slope + alpha * alpha * q))
    return alpha, val

def _pair_descent_pass(h, corr, deadline, rng, order='random'):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    corr = corr.copy()
    improved = False
    n_accept = 0
    
    if order == 'random':
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for (i, j) in pairs:
        if time.time() > deadline:
            break
        a = max(-h[i], h[j] - 1.0)
        b = min(1.0 - h[i], h[j])
        if b - a < 1e-14:
            continue
        slope = _pair_slope(h, i, j)
        q = _pair_q(i, j, n)
        opt_a, opt_val = _opt_alpha(corr, slope, q, a, b)
        cur_c5 = float(np.max(corr))
        if opt_val < cur_c5 - 1e-14:
            h[i] += opt_a
            h[j] -= opt_a
            corr = corr + opt_a * slope + opt_a * opt_a * q
            improved = True
            n_accept += 1
            if n_accept % 200 == 0:
                corr = _scores(h)
    
    return h, corr, improved

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Main pair descent from initial point
    deadline1 = t0 + budget_s * 0.35
    corr = _scores(h)
    stall = 0
    while time.time() < deadline1 and stall < 5:
        h, corr, improved = _pair_descent_pass(h, corr, deadline1, rng, order='random')
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
        if not improved:
            stall += 1
            # Try sequential order as well
            h, corr, improved2 = _pair_descent_pass(h, corr, deadline1, rng, order='sequential')
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
            if not improved2:
                stall += 1
        else:
            stall = 0
    
    # Phase 2: Multi-resolution search
    deadline2 = t0 + budget_s * 0.65
    n_options = [n0, n0 - 8, n0 - 4, n0 + 4, n0 + 8, n0 + 16]
    n_options = sorted(set(x for x in n_options if 32 <= x <= 128))
    remaining = max(deadline2 - time.time(), 0)
    per_n = remaining / max(1, len(n_options))
    
    for n_try in n_options:
        if time.time() > deadline2:
            break
        sub_dl = min(deadline2, time.time() + per_n)
        h_r = _resample(best_h, n_try)
        corr_r = _scores(h_r)
        stall_r = 0
        while time.time() < sub_dl and stall_r < 3:
            h_r, corr_r, imp = _pair_descent_pass(h_r, corr_r, sub_dl, rng, order='random')
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
            if not imp:
                stall_r += 1
    
    # Phase 3: Random restarts on best_n
    deadline3 = t0 + budget_s * 0.85
    n_restarts = 8
    restart_time = max(budget_s * 0.015, (deadline3 - time.time()) / n_restarts)
    for _ in range(n_restarts):
        if time.time() > deadline3:
            break
        h_r = _random_binary(best_n, rng)
        corr_r = _scores(h_r)
        sub_dl = min(deadline3, time.time() + restart_time)
        stall_r = 0
        while time.time() < sub_dl and stall_r < 2:
            h_r, corr_r, imp = _pair_descent_pass(h_r, corr_r, sub_dl, rng, order='sequential')
            if not imp:
                stall_r += 1
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 4: Final refinement
    deadline4 = t0 + budget_s * 0.99
    corr_best = _scores(best_h)
    stall = 0
    while time.time() < deadline4 and stall < 5:
        best_h, corr_best, imp = _pair_descent_pass(best_h, corr_best, deadline4, rng, order='random')
        c5 = _exact_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        if not imp:
            best_h, corr_best, imp2 = _pair_descent_pass(best_h, corr_best, deadline4, rng, order='sequential')
            c5 = _exact_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
            if not imp2:
                stall += 1
    
    # Final
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n

def _binary_swap_search(h, deadline, rng):
    """Local search on binary vectors: swap a 1 to a 0."""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[idx[:n//2]] = 1.0
    
    ones = set(np.where(h_bin == 1.0)[0].tolist())
    zeros = set(np.where(h_bin == 0.0)[0].tolist())

        best_val = cur_c5

        for _ in range(min(50, len(ones) * len(zeros))):
            i = list(ones)[rng.integers(len(ones))]
