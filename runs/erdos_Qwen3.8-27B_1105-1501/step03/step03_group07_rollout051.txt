def _lp_tangent_step(h, R, mode='full', top_k=None, lambda_pen=0.0):
    """
    Solve the tangent LP to find a descent direction.
    
    Returns (direction, predicted_t, success)
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_lags = 2 * n - 1
    current_max = np.max(corr)
    
    # Select which lags to include
    if mode == 'full':
        lag_indices = np.arange(n_lags)
    elif mode == 'top':
        threshold = np.sort(corr)[-top_k]
        lag_indices = np.where(corr >= threshold - 1e-10)[0]
    else:
        lag_indices = np.arange(n_lags)
    
    # Compute gradients for selected lags
    # Variable layout: [t, p_0, p_1, ..., p_{n-1}, q_0, q_1, ..., q_{n-1}]
    # d_i = p_i - q_i
    nv = 1 + 2 * n  # total variables
    
    # Objective: minimize t + lambda * sum(p + q)
    c = np.zeros(nv)
    c[0] = 1.0
    if lambda_pen > 0:
        c[1:1+n] = lambda_pen
        c[1+n:1+2*n] = lambda_pen
    
    # Constraints:
    # 1. For each selected lag r: -t + sum_j grad_r[j] * (p_j - q_j) <= -corr[r]
    #    i.e., -t + sum_j grad_r[j]*p_j - sum_j grad_r[j]*q_j <= -corr[r]
    # 2. sum_i d_i = 0: sum_i p_i - sum_i q_i = 0
    # 3. sum_i (p_i + q_i) <= R
    
    m_k = len(lag_indices)
    A_ub = np.zeros((m_k + 1, nv))
    b_ub = np.zeros(m_k + 1)
    
    for ii, r in enumerate(lag_indices):
        d = int(r) - (n - 1)
        grad = _lag_gradient(h, d)
        A_ub[ii, 0] = -1.0
        A_ub[ii, 1:1+n] = grad  # coefficient of p
        A_ub[ii, 1+n:1+2*n] = -grad  # coefficient of q
        b_ub[ii] = -corr[r]
    
    # L1 norm constraint: sum(p + q) <= R
    A_ub[m_k, 1:1+n] = 1.0
    A_ub[m_k, 1+n:1+2*n] = 1.0
    b_ub[m_k] = R
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    eps = 1e-8
    for i in range(n):
        if h[i] <= eps:
            # Can only increase: q_i = 0
            bounds.append((0, None))  # p_i
            bounds.append((0, 0))     # q_i = 0
        elif h[i] >= 1.0 - eps:
            # Can only decrease: p_i = 0
            bounds.append((0, 0))     # p_i = 0
            bounds.append((0, None))  # q_i
        else:
            bounds.append((0, None))  # p_i
            bounds.append((0, None))  # q_i
    
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, current_max, False
    
    t_val = res.x[0]
    p_vec = res.x[1:1+n]
    q_vec = res.x[1+n:1+2*n]
    d_vec = p_vec - q_vec
    
    # Ensure sum is zero
    d_vec -= np.mean(d_vec)
    
    # Check if prediction is better
    if t_val >= current_max - 1e-10:
        return d_vec, t_val, False
    
    return d_vec, t_val, True
