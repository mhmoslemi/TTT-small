            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, n, deadline, radius_init=0.1, max_iters=2000, stall_limit=50):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, n, radius)
        if p is None or np.max(np.abs(p)) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _lp_line_search(h, p, cur_c5, n)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _elapsed():
        return time.time() - t0

    # Phase 1: LSE smoothing warm-up (15% budget)
    d1 = t0 + budget_s * 0.15
    alphas = [10.0, 20.0, 40.0, 80.0]
    for alpha in alphas:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, maxiter=80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP epigraph (main, 30% budget)
    d2 = t0 + budget_s * 0.45
    for attempt in range(3):
        if time.time() > d2:
            break
        start = best_h.copy()
        if attempt > 0:
            start = _project(start + rng.normal(0, 0.02, n0), target)
        h_s, c5_s = _slsqp_solve(start, n0, maxiter=400, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        # Follow-up SLSQP
        if time.time() < d2:
            h_s2, c5_s2 = _slsqp_solve(h_s, n0, maxiter=300, ftol=1e-15)
            if c5_s2 < best_c5:
                best_c5, best_h = c5_s2, h_s2.copy()

    # Phase 3: LP polishing (20% budget)
    d3 = t0 + budget_s * 0.65
    h_lp, c5_lp = _lp_loop(best_h, n0, d3, radius_init=0.1,
                           max_iters=3000, stall_limit=40)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: More SLSQP from perturbed starts (15% budget)
    d4 = t0 + budget_s * 0.80
    scale = 0.03
    for attempt in range(200):
        if time.time() > d4:
            break
        if best_c5 <= 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, n0)
        h_p = _project(h_p, target)
        h_s, c5_s = _slsqp_solve(h_p, n0, maxiter=150, ftol=1e-13)
        if c5_s < best_c5 - 1e-10:
            best_c5, best_h = c5_s, h_s.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 5: Resolution search (12% budget)
    d5 = t0 + budget_s * 0.92
    for dn in [0, 1, 2, 3, 4, 6, -1, -2, -3, -4, -6, 8, -8]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        # Quick LSE
        sub_d = min(d5, time.time() + budget_s * 0.015)
        for alpha in [20.0, 50.0]:
            if time.time() > sub_d:
                break
            h_r, c5_r = _lse_step(h_r, n_try, alpha, maxiter=50)
        # SLSQP
        sub_d2 = min(d5, time.time() + budget_s * 0.02)
        h_s, c5_s = _slsqp_solve(h_r, n_try, maxiter=200, ftol=1e-13)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        # LP
        sub_d3 = min(d5, time.time() + budget_s * 0.015)
        h_lp, c5_lp = _lp_loop(h_r, n_try, sub_d3, radius_init=0.08,
                               max_iters=500, stall_limit=30)
        if c5_lp < c5_r:
            h_r, c5_r = h_lp, c5_lp
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Final polish (8% budget)
    d6 = t0 + budget_s * 0.98
    h_f, c5_f = _slsqp_solve(best_h, best_n, maxiter=300, ftol=1e-15)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    h_lp, c5_lp = _lp_loop(best_h, best_n, d6, radius_init=0.03,
                           max_iters=2000, stall_limit=60)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
