    """Compute gradient of correlation at lag d with respect to h.
    d is the offset: corr index = d + (n-1)"""
    n = len(h)
    grad = np.zeros(n)

        if m > 0:

            grad[d:] -= h[:m]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[l:] += (1.0 - h[:m])

    #   -step_limit <= d_i <= step_limit  (step size)
    
    # Variables: x = [d_0,...,d_{n-1}, t]

    HI = np.minimum(1.0 - h, step_limit)

            return d / d_norm, t

    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.06
    beta = 0.5
    T = T_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        grad = _softmax_grad(h, scores, T)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        grad /= gnorm
        v = beta * v - alpha * grad
        v -= np.mean(v)
        a, new_c5 = _line_search(h, v, target, cur_c5 if False else 1.0, max_alpha=1.0)
        if a > 0 and new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * v, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            T = max(0.03, T * 0.99)
            beta = min(0.9, beta + 0.001)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.9
            T = max(0.03, T * 0.95)
            if stall > 50:
                v *= 0.3
                alpha = 0.04
                T = 0.15
    return best_h, best_c5

def _pair_search(h, deadline, target, max_passes=50):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)
        grad = _softmax_grad(h, scores, 0.1)
        grad -= np.mean(grad)
        top_count = min(n, 16)
        top = np.argsort(-np.abs(grad))[:top_count]
        for ii in range(len(top)):
            if time.time() > deadline:
                return best_h, best_c5
            j = top[ii]
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for jj in range(len(top)):
                if jj == ii:
                    continue
                k = top[jj]
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                    a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                    a_max = min(a_max_j, a_max_k)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-12:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            for _ in range(100):
                if time.time() > deadline:
                    return best_h, best_c5
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j == k or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                dj = 1.0 if rng.random() > 0.5 else -1.0
                dk = -dj
                a_max = min((1 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                            (1 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                if a_max < 1e-10:
                    continue
                for frac in [0.3, 0.7, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-12:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_snap(h, n):
    """Create a binary version of h with exactly n/2 ones."""
    target = int(n / 2)
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target - 1]
    h_bin = (h >= threshold).astype(float)
    excess = int(np.sum(h_bin)) - target
    if excess > 0:
        at_thresh = np.where((h >= threshold - 1e-10) & (h <= threshold + 1e-10))[0]
        to_flip = at_thresh[:excess]
        h_bin[to_flip] = 0.0
    elif excess < 0:
        just_below = np.where(h < threshold - 1e-10)[0]
        h_sorted_below = np.sort(h[just_below])[::-1]
        to_flip = just_below[np.argsort(-h[just_below])[:(-excess)]]
        h_bin[to_flip] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_iters=20000):
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(30):
            i = ones[int(rng.integers(0, len(ones)))]
            j = zeros[int(rng.integers(0, len(zeros)))]
            h_try = h.copy()
            h_try[i] = 0.0
            h_try[j] = 1.0
            c_try = _exact_c5(h_try)
            if c_try < best_c5 - 1e-12:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _gen_structured(n, rng, kind):
    if kind == 0:
        return (np.arange(n) % 2).astype(float)
    elif kind == 1:
        h = np.zeros(n)
        period = n // 4 if n % 4 == 0 else n // 3
        for i in range(n):
            if (i % (2 * period)) < period:
                h[i] = 1.0
        return _project(h, n / 2)
    elif kind == 2:
        h = np.sin(2 * np.pi * 3 * np.arange(n) / n) > 0
        return _project(h.astype(float), n / 2)
    else:
        h = np.zeros(n)
        h[::2] = 1.0
        return _project(h, n / 2)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization (35% time)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_optimize(h, dl1, target0, max_iters=500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient refinement (15% time)
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _subgrad_optimize(best_h, dl2, best_n / 2.0, T_init=0.25, max_iters=8000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair search (10% time)
    dl3 = t0 + budget_s * 0.60
    h3, c5_3 = _pair_search(best_h, dl3, best_n / 2.0, max_passes=80)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary exploration (15% time)
    dl4 = t0 + budget_s * 0.75
    h_bin = _binary_snap(best_h, best_n)
    h_b, c5_b = _binary_local_search(h_bin, dl4, max_iters=15000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = len(h_b)

    # Also try LP from binary
    if time.time() < dl4:
        h_bl, c5_bl = _lp_optimize(h_b, dl4, best_n / 2.0, max_iters=200)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 5: Multi-start with diverse perturbations (10% time)
    dl5 = t0 + budget_s * 0.85
    n_starts = 12
    for s in range(n_starts):
        if time.time() > dl5:
            break
        remaining = dl5 - time.time()
        if remaining < 3.0:
            break
        alloc = min(remaining * 0.9 / max(1, n_starts - s), 15.0)
        sub_dl = time.time() + alloc

        r = rng.random()
        if r < 0.3:
            h_try = best_h + rng.normal(0, 0.04, best_n)
        elif r < 0.5:
            h_try = best_h.copy()
            idx = rng.choice(best_n, size=max(1, best_n // 6), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.65:
            h_try = (best_h + rng.normal(0, 0.06, best_n) > 0.5).astype(float)
        elif r < 0.8:
            h_try = _gen_structured(best_n, rng, int(rng.integers(0, 4)))
        else:
            h_try = 0.5 * best_h + 0.5 * (rng.random(best_n) > 0.5).astype(float)

        h_try = _project(h_try, best_n / 2.0)
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, best_n / 2.0, max_iters=150)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Resolution sweep (10% time)
    dl6 = t0 + budget_s * 0.95
    for dn in [4, -4, 8, -8, 16, -16, 24, -24]:
        if time.time() > dl6:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + min(12.0, (dl6 - time.time()) * 0.3))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, max_iters=150)
        h_r, c5_r = _pair_search(h_r, min(dl6, time.time() + 3.0), n_try / 2.0, max_passes=20)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final LP polish (5% time)
    dl7 = t0 + budget_s * 0.99
    h7, c5_7 = _lp_optimize(best_h, dl7, best_n / 2.0, max_iters=200)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
        best_n = n0

    return best_h, final_c5, best_n
