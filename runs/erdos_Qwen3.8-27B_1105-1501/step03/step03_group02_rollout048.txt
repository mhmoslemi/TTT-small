def _smooth_gradient(h, scores, beta=50.0):
    """Compute smooth approximation of subgradient of max scores"""
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= wsum
    
    # Gradient is weighted sum of individual score gradients
    grad = np.zeros(n)
    for r, wi in enumerate(w):
        if wi > 1e-10:
            d = r - (n - 1)
            grad += wi * _score_grad(h, d)
    return grad

        best_h = h_p.copy()

    # Phase 2: Gradient-based steps
    dl2 = t0 + budget_s * 0.50
    h_g, c5_g = _gradient_loop(best_h, dl2, rng, best_n / 2.0)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    # Phase 3: More pair swaps after gradient steps
    dl3 = t0 + budget_s * 0.65
    h_p2, c5_p2 = _pair_swap_loop(best_h, dl3, rng, max_rounds=1000, beta_start=20.0)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()

    # Phase 4: SLSQP refinement
    if elapsed() < budget_s * 0.80:
        dl4 = t0 + budget_s * 0.82
        h_s, c5_s = _slsqp_refine(best_h, dl4, top_k=15, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Binary local search for independent topology
    if elapsed() < budget_s * 0.88:
        dl5 = t0 + budget_s * 0.90
        bits, c5_b = _binary_local_search(best_h, dl5, rng)
        if c5_b < best_c5:
            # Relax the binary solution
            h_relax = bits.astype(float) + rng.normal(0, 0.05, best_n)
            h_relax = _project(h_relax, best_n / 2.0)
            h_relax, c5_r = _pair_swap_loop(h_relax, min(dl5 + 5, t0 + budget_s * 0.92), rng, max_rounds=200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_relax.copy()

    # Phase 6: Multi-resolution
    if elapsed() < budget_s * 0.94:
        dl6 = t0 + budget_s * 0.97
        for dn in [0, -4, 4, -8, 8, 12, -12]:
            if elapsed() > budget_s * 0.96:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 128 or n_try % 2 != 0:
                continue
            if n_try == best_n:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _pair_swap_loop(h_r, min(dl6, time.time() + 30), rng, max_rounds=300)
            if c5_r < best_c5 - 1e-5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final pair swap polish
    if elapsed() < budget_s * 0.99:
        h_f, c5_f = _pair_swap_loop(best_h, t0 + budget_s * 0.995, rng, max_rounds=500, beta_start=15.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final: ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    if final_c5 > best_c5 + 1e-8:
        # Projection degraded quality; try without final project if sum is close
        if abs(float(np.sum(best_h)) - best_n / 2.0) < 1e-8:
            pass
        else:
            best_h = _project(best_h, best_n / 2.0)
            final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
