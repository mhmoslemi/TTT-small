import numpy as np
from scipy.optimize import linprog
import time

def _repair(v, target):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _line_coeffs(h, d):
    """Compute A, B, C arrays for each lag on line h + t*d.
    f_r(t) = A[r] + B[r]*t + C[r]*t^2
    """
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    
    for r in range(M):
        k = r - (n - 1)
        if k == 0:
            # Q_0 = sum(h[i](1-h[i]))
            A[r] = np.sum(h * (1.0 - h))
            B[r] = np.sum(d * (1.0 - 2.0 * h))
            C[r] = -np.sum(d * d)
        elif k > 0:
            m = n - k
            A[r] = np.sum(h[k:] * (1.0 - h[:m]))
            B[r] = np.sum(d[k:] * (1.0 - h[:m]) - h[k:] * d[:m])
            C[r] = -np.sum(d[k:] * d[:m])
        else:
            g = -k
            m = n - g

            B[r] = np.sum(d[:m] * (1.0 - h[g:]) - h[:m] * d[g:])
            C[r] = -np.sum(d[:m] * d[g:])
    
    return A, B, C

def _feasible_interval(h, d):
    """Find t range where h[i] + t*d[i] in [0,1] for all i"""
    t_lo = 0.0
    t_hi = 0.0

        elif d[i] < -1e-15:
            t_lo = max(t_lo, -h[i] / d[i])
    # Ensure 0 is in interval (it should be since h is feasible)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    return t_lo, t_hi

def _exact_1d_min(h, d):
    """Find exact minimum of C5 along line h + t*d.
    Returns (best_t, best_h, best_c5) or (0, h, current_c5) if no improvement."""
    n = len(h)
    M = 2 * n - 1
    
    A, B, C = _line_coeffs(h, d)
    t_lo, t_hi = _feasible_interval(h, d)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, h.copy(), _c5(h)
    
    # Collect candidate t values
    candidates = set()
    candidates.add(t_lo)
    candidates.add(t_hi)
    candidates.add(0.0)
    
    # Individual stationary points
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t_star = -B[r] / (2.0 * C[r])
            if t_lo - 1e-12 <= t_star <= t_hi + 1e-12:
                candidates.add(max(t_lo, min(t_hi, t_star)))
    
    # Pair crossings
    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-14:
                # Linear: b*t + c = 0
                if abs(b) > 1e-14:
                    t_root = -c / b
                    if t_lo - 1e-12 <= t_root <= t_hi + 1e-12:
                        candidates.add(max(t_lo, min(t_hi, t_root)))
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-14:
                    if disc < 0:
                        disc = 0.0
                    sqrt_disc = np.sqrt(disc)
                    for t_root in [(-b + sqrt_disc) / (2.0 * a),
                                   (-b - sqrt_disc) / (2.0 * a)]:
                        if t_lo - 1e-12 <= t_root <= t_hi + 1e-12:
                            candidates.add(max(t_lo, min(t_hi, t_root)))
    
    # Evaluate at each candidate
    cands = np.array(list(candidates))
    if len(cands) == 0:
        return 0.0, h.copy(), _c5(h)
    
    # Evaluate E(t) = max_r (A[r] + B[r]*t + C[r]*t^2) for each candidate
    # Vectorized: shape (len(cands), M)
    vals = A[None, :] + B[None, :] * cands[:, None] + C[None, :] * (cands[:, None] ** 2)
    E = np.max(vals, axis=1)
    
    best_idx = np.argmin(E)
    best_t = float(cands[best_idx])
    best_raw = float(E[best_idx])
    
    if best_raw >= np.max(A) - 1e-14:  # No improvement
        return 0.0, h.copy(), _c5(h)
    
    h_new = h + best_t * d
    h_new = _repair(h_new, n / 2.0)
    return best_t, h_new, _c5(h_new)

def _pair_direction(h, scores):
    """Generate a pair exchange direction: +1 at i, -1 at j"""
    n = len(h)
    # Choose i where h[i] is low (room to increase) and j where h[j] is high (room to decrease)
    # Or use gradient info: increase i where gradient is negative, decrease j where positive
    # For now, just pick the most extreme low and high
    i = int(np.argmin(h))
    j = int(np.argmax(h))
    d = np.zeros(n)
    d[i] = 1.0

def _random_direction(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = np.sum(d**2)
    if norm < 1e-15:
        return None
    return d / np.sqrt(norm)

def _lp_direction(h, scores, top_k=20):
    """Use LP to find direction that decreases all top scores."""
    n = len(h)
    M = 2 * n - 1
    
    # Get gradients for top_k scores
    idx = np.argsort(scores)[::-1][:top_k]
    
    # Build gradient matrix for these scores
    G = np.zeros((top_k, n))
    for ii, r in enumerate(idx):
        k = r - (n - 1)
        if k == 0:
            G[ii] = 1.0 - 2.0 * h
        elif k > 0:
            m = n - k
            G[ii, :m] += -h[k:k+m]
            G[ii, k:] += (1.0 - h[:m])
        else:
            g = -k
            m = n - g
            G[ii, :m] += (1.0 - h[g:])
            G[ii, g:] += -h[:m]
    
    # LP: min t subject to G @ d <= t, sum(d) = 0, -1 <= d_i <= 1
    # Variables: d[0..n-1], t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0  # G@d - t <= 0
    b_ub = np.zeros(top_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] < -1e-8:
            d = res.x[:n]
            if np.sum(d**2) > 1e-15:
                return d / np.sqrt(np.sum(d**2))
    except Exception:
        pass
    return None

def _exact_1d_min_fast(h, d):
    n = len(h)
    M = 2 * n - 1
    A, B, C = _line_coeffs(h, d)
    t_lo, t_hi = _feasible_interval(h, d)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, h.copy(), _c5(h)
    
    candidates = [t_lo, t_hi, 0.0]
    
    # Individual stationary points (vectorized)
    mask = np.abs(C) > 1e-14
    if np.any(mask):
        t_stars = -B[mask] / (2.0 * C[mask])
        valid = (t_stars >= t_lo - 1e-12) & (t_stars <= t_hi + 1e-12)
        candidates.extend(np.clip(t_stars[valid], t_lo, t_hi).tolist())
    
    # Pair crossings (vectorized)
    C_diff = C[:, None] - C[None, :]
    B_diff = B[:, None] - B[None, :]
    A_diff = A[:, None] - A[None, :]
    
    # Upper triangle mask
    tri = np.triu_indices(M, k=1)
    a = C_diff[tri]
    b = B_diff[tri]
    c = A_diff[tri]
    
    # Quadratic case
    q_mask = np.abs(a) > 1e-12
    if np.any(q_mask):
        disc = b[q_mask]**2 - 4.0 * a[q_mask] * c[q_mask]
        real_mask = disc >= -1e-12
        if np.any(real_mask):
            disc_ok = np.clip(disc[real_mask], 0, None)
            sqrt_d = np.sqrt(disc_ok)
            a_ok = a[q_mask][real_mask]
            b_ok = b[q_mask][real_mask]
            roots1 = (-b_ok + sqrt_d) / (2.0 * a_ok)
            roots2 = (-b_ok - sqrt_d) / (2.0 * a_ok)
            for roots in [roots1, roots2]:
                valid = (roots >= t_lo - 1e-12) & (roots <= t_hi + 1e-12)
                candidates.extend(np.clip(roots[valid], t_lo, t_hi).tolist())
    
    # Linear case
    l_mask = np.abs(a) <= 1e-12
    if np.any(l_mask):
        b_l = b[l_mask]
        c_l = c[l_mask]
        lin_mask = np.abs(b_l) > 1e-12
        if np.any(lin_mask):
            roots = -c_l[lin_mask] / b_l[lin_mask]
            valid = (roots >= t_lo - 1e-12) & (roots <= t_hi + 1e-12)
            candidates.extend(np.clip(roots[valid], t_lo, t_hi).tolist())
    
    candidates = np.array(candidates)
    if len(candidates) == 0:
        return 0.0, h.copy(), _c5(h)
    
    # Deduplicate
    candidates = np.unique(np.round(candidates, 14))
    
    # Evaluate E(t) at all candidates
    vals = A[None, :] + B[None, :] * candidates[:, None] + C[None, :] * candidates[:, None]**2
