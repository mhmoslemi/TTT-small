import numpy as np
from scipy.optimize import differential_evolution, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            # h[k] appears as h[i+d] when i = k-d, valid for d <= k <= n-1
            # contribution: (1-h[k-d])
            # h[k] appears as h[i] when i = k, valid for 0 <= k <= n-1-d
            # contribution: -h[k+d]
            grad[d:] += (1.0 - h[:m])  # h[k] as h[i+d], i=k-d, k from d to n-1
            grad[:m] += -h[d:]  # h[k] as h[i], i=k, k from 0 to n-1-d
    else:
        g = -d
        m = n - g
        if m > 0:
            # score_{-g} = sum_{i=0}^{n-1-g} h[i] * (1-h[i+g])
            # h[k] as h[i] when i=k, valid for 0 <= k <= n-1-g:
            # contribution: (1-h[k+g])
            # h[k] as h[i+g] when i+g=k, i=k-g, valid for g <= k <= n-1:
            # contribution: -h[k-g]
            grad[:m] += (1.0 - h[g:])  # h[k] as h[i]
            grad[g:] += -h[:m]  # h[k] as h[i+g]
    return grad

def _project(v, target):
    """Project v to {x : 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad_direction(h, beta=50.0):
    """Compute smooth max gradient direction (for minimization)"""
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    # Avoid overflow
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wi = w[n-1+d]
        if wi > 1e-15:
            grad += wi * _score_grad(h, d)
    # Project to zero-sum
    grad -= np.mean(grad)
    return -grad, scores  # negative for descent

def _line_search_exact(h, p, c5_cur, deadline, max_iter=40):
    """Exact golden-section line search along h + alpha*p, p has zero sum"""
    n = len(h)
    # Find alpha_max
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 1.0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)
    f_c = _exact_c5(np.clip(h + c * p, 0.0, 1.0))
    f_d = _exact_c5(np.clip(h + d_ * p, 0.0, 1.0))
    
    best_a = 0.0
    best_f = c5_cur
    if f_c < best_f:
        best_f = f_c
        best_a = c
    if f_d < best_f:
        best_f = f_d
        best_a = d_
    
    for _ in range(max_iter):
        if hi - lo < 1e-12 or time.time() > deadline:
            break
        if f_c < f_d:
            hi = d_
            d_ = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = _exact_c5(np.clip(h + c * p, 0.0, 1.0))
            if f_c < best_f:
                best_f = f_c
                best_a = c
        else:
            lo = c
            c = d_
            f_c = f_d
            d_ = lo + gr * (hi - lo)
            f_d = _exact_c5(np.clip(h + d_ * p, 0.0, 1.0))
            if f_d < best_f:
                best_f = f_d
                best_a = d_
    
    mid_a = (lo + hi) / 2.0
    f_mid = _exact_c5(np.clip(h + mid_a * p, 0.0, 1.0))
    if f_mid < best_f:
        best_f = f_mid
        best_a = mid_a
    
    return best_a, best_f

def _local_refine(h, deadline, max_rounds=200, beta_init=30.0):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    beta = beta_init
    stall = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        # Compute smooth gradient direction
        p, scores = _smooth_grad_direction(h, beta)
        p_norm = np.linalg.norm(p)
        if p_norm < 1e-15:
            break
        p = p / p_norm
        
        # Line search
        alpha, new_c5 = _line_search_exact(h, p, c5, deadline)
        
        if new_c5 < c5 - 1e-14:

            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall = 0

def _pairwise_opt(h, deadline, max_rounds=100):

    n = len(h)
    target = n / 2.0
    h = h.copy()

        d_worst = max_idx - (n - 1)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anti_complement(h):
    """Project h onto the anti-complement subspace"""
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    half = n // 2
    # Average h[i] and 1-h[i+half] to get y[i]
    y = 0.5 * (h[:half] + (1.0 - h[half:]))
    # Reconstruct
    h_ac = np.zeros(n)
    h_ac[:half] = y
    h_ac[half:] = 1.0 - y
    return h_ac

def _lp_refine(h, deadline, max_iters=500):
    """LP-based direction finding with exact line search"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    radius = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        
        scores = _all_scores(h)
        
        # Build LP: minimize z s.t. score_d(h) + grad_d · t <= z, sum(t)=0, |t|<=radius, bounds
        # Actually, let's use the linearization: minimize z
        # s.t. for each d: score_d_linearized(h+t) <= z
        #      score_d(h) + grad_d(h) · t <= z
        #      sum(t) = 0
        #      -radius <= t_j <= radius
        #      0 <= h_j + t_j <= 1
        
        n_vars = n + 1  # t_0...t_{n-1}, z
        # Objective: minimize z
        c = np.zeros(n_vars)
        c[-1] = 1.0
        
        # Inequality constraints: grad_d · t - z <= -score_d for each d
        # i.e., grad_d · t + (-1)*z <= -score_d
        A_ub = np.zeros((2*n-1, n_vars))
        b_ub = np.zeros(2*n-1)
        for d in range(-(n-1), n):
            r = n - 1 + d
            g = _score_grad(h, d)
            A_ub[r, :n] = g
            A_ub[r, n] = -1.0
            b_ub[r] = -float(scores[r])
        
        # Box constraints on t: -radius <= t_j <= radius AND 0 <= h_j + t_j <= 1
        # t_j <= min(radius, 1-h_j) and t_j >= max(-radius, -h_j)
        t_lower = np.maximum(-radius, -h)
        t_upper = np.minimum(radius, 1.0 - h)
        
        # Equality: sum(t) = 0
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds_lp = [(t_lower[j], t_upper[j]) for j in range(n)] + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
            if res.success:
                t = res.x[:n]
                # Verify zero sum (should be exact from equality constraint)
                t -= np.mean(t)  # numerical safeguard
                # Line search along t direction
                alpha, new_c5 = _line_search_exact(h, t, c5, deadline, max_iter=30)
                if new_c5 < c5 - 1e-14:
                    h = np.clip(h + alpha * t, 0.0, 1.0)
                    h = _project(h, target)
                    c5 = _exact_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.3)
                else:
                    stall += 1
                    radius *= 0.8
            else:
                stall += 1
                radius *= 0.8
        except Exception:
            stall += 1
            radius *= 0.8
        
        if radius < 1e-6:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import differential_evolution, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad_direction(h, beta=50.0):
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wi = w[n - 1 + d]
        if wi > 1e-15:
            grad += wi * _score_grad(h, d)
    grad -= np.mean(grad)
    return -grad, scores

def _line_search_exact(h, p, c5_cur, deadline, max_iter=35):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-15:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-15:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 2.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    f_c = _exact_c5(np.clip(h + c * p, 0.0, 1.0))
    f_d = _exact_c5(np.clip(h + d_pt * p, 0.0, 1.0))
    best_a = 0.0
    best_f = c5_cur
    if f_c < best_f:
        best_f = f_c
        best_a = c
    if f_d < best_f:
        best_f = f_d
        best_a = d_pt
    for _ in range(max_iter):
        if hi - lo < 1e-12 or time.time() > deadline:
            break
        if f_c < f_d:
            hi = d_pt
            d_pt = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = _exact_c5(np.clip(h + c * p, 0.0, 1.0))
            if f_c < best_f:
                best_f = f_c
                best_a = c
        else:
            lo = c
            c = d_pt
            f_c = f_d
            d_pt = lo + gr * (hi - lo)
            f_d = _exact_c5(np.clip(h + d_pt * p, 0.0, 1.0))
            if f_d < best_f:
                best_f = f_d
                best_a = d_pt
    mid_a = (lo + hi) / 2.0
    f_mid = _exact_c5(np.clip(h + mid_a * p, 0.0, 1.0))
    if f_mid < best_f:
        best_f = f_mid
        best_a = mid_a
    return best_a, best_f

def _local_refine(h, deadline, max_rounds=200, beta_init=30.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    beta = beta_init
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 35:
            break
        p, scores = _smooth_grad_direction(h, beta)
        p_norm = float(np.linalg.norm(p))
        if p_norm < 1e-15:
            break
        p = p / p_norm
        alpha, new_c5 = _line_search_exact(h, p, c5, deadline)
        if new_c5 < c5 - 1e-14:
            h = np.clip(h + alpha * p, 0.0, 1.0)
            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
            beta = min(beta * 1.12, 600.0)
        else:
            stall += 1
            beta = max(beta * 0.75, 5.0)
    return best_h, best_c5

def _lp_refine(h, deadline, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    radius = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        scores = _all_scores(h)
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n_vars))
        b_ub = np.zeros(2 * n - 1)
        for d in range(-(n - 1), n):
            r = n - 1 + d
            g = _score_grad(h, d)
            A_ub[r, :n] = g
            A_ub[r, n] = -1.0
            b_ub[r] = -float(scores[r])
        t_lower = np.maximum(-radius, -h)
        t_upper = np.minimum(radius, 1.0 - h)
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds_lp = [(float(t_lower[j]), float(t_upper[j])) for j in range(n)] + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
            if res.success:
                t = res.x[:n].copy()
                t -= np.mean(t)
                t_norm = float(np.linalg.norm(t))
                if t_norm < 1e-14:
                    stall += 1
                    radius *= 0.8
                    continue
                alpha, new_c5 = _line_search_exact(h, t, c5, deadline, max_iter=25)
                if new_c5 < c5 - 1e-14:
                    h = np.clip(h + alpha * t, 0.0, 1.0)
                    h = _project(h, target)
                    c5 = _exact_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.4)
                else:
                    stall += 1
                    radius *= 0.8
            else:
                stall += 1
                radius *= 0.8
        except Exception:
            stall += 1
            radius *= 0.8
        if radius < 1e-7:
            break
    return best_h, best_c5

def _pairwise_opt(h, deadline, max_rounds=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 25:
            break
        scores = _all_scores(h)
        top_k = np.argsort(scores)[::-1][:5]
        grad_avg = np.zeros(n)
        for ki in top_k:
            d_k = int(ki) - (n - 1)
            g_k = _score_grad(h, d_k)
            g_k -= np.mean(g_k)
            grad_avg += g_k
        grad_avg /= float(len(top_k))
        direction = -grad_avg
        p_norm = float(np.linalg.norm(direction))
        improved = False
        if p_norm > 1e-15:
            p = direction / p_norm
            alpha, new_c5 = _line_search_exact(h, p, c5, deadline, max_iter=25)
            if new_c5 < c5 - 1e-14:
                h = np.clip(h + alpha * p, 0.0, 1.0)
                h = _project(h, target)
                c5 = _exact_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
        if not improved:
            # Try specific pairs
            dec_mask = h < 0.95
            inc_mask = h > 0.05
            if np.any(dec_mask) and np.any(inc_mask):
                dec_idx = np.where(dec_mask)[0]
                inc_idx = np.where(inc_mask)[0]
                dec_sorted = dec_idx[np.argsort(grad_avg[dec_idx])][:4]
                inc_sorted = inc_idx[np.argsort(-grad_avg[inc_idx])][:4]
                best_pair_c5 = c5
                best_pair = None
                for i in dec_sorted:
                    for j in inc_sorted:
                        if i == j:
                            continue
                        p = np.zeros(n)
                        p[int(i)] = -1.0
                        p[int(j)] = 1.0
                        a, sc = _line_search_exact(h, p, c5, deadline, max_iter=18)
                        if sc < best_pair_c5 - 1e-14:
                            best_pair_c5 = sc
                            best_pair = (int(i), int(j), a)
                if best_pair is not None:
                    i, j, a = best_pair
                    h[i] -= a
                    h[j] += a
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    c5 = _exact_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _de_objective(y, n):
    h = np.array(y, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = _project(h, n / 2.0)
    return _exact_c5(h)

def _de_optimize(n, deadline, seed, popsize=15, maxiter=80):
    target = n / 2.0
    def _obj(y):
        return _de_objective(y, n)
    bounds = [(0.0, 1.0)] * n
    try:
        result = differential_evolution(_obj, bounds, seed=seed,
                                         popsize=popsize, maxiter=maxiter,
                                         tol=1e-9, mutation=(0.4, 1.0),
                                         recombination=0.85,
                                         polish=False, init='halton-jittered')
        h = np.clip(result.x, 0.0, 1.0)
        h = _project(h, target)
        return h, _exact_c5(h)
    except Exception:
        return None, float('inf')

def _anti_complement(h):
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    half = n // 2
    y = 0.5 * (h[:half] + (1.0 - h[half:]))
    h_ac = np.zeros(n)
    h_ac[:half] = y
    h_ac[half:] = 1.0 - y
    return h_ac

def _subgradient_step(h, scores, beta=50.0):
    """Alternative direction: uniform over top-k scores"""
    n = len(h)
    top_k = np.argsort(scores)[::-1][:max(3, int(0.1 * n))]
    grad = np.zeros(n)
    for ki in top_k:
        d_k = int(ki) - (n - 1)
        grad += _score_grad(h, d_k)
    grad /= float(len(top_k))
    grad -= np.mean(grad)
    return -grad
