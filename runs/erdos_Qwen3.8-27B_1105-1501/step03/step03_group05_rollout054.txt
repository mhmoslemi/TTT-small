            m = n - g
            if m > 0:
                grad[g:] += wd * (-h[:m])
                grad[:m] += wd * (1.0 - h[g:])
    return dx * grad

    """Build the (2n-1) x n matrix where row r is gradient of score at lag r-n+1."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):

        if d >= 0:

            if m > 0:

                G[r, d:] += (1.0 - h[:m])
        else:

            m = n - g
            if m > 0:
                G[r, g:] += -h[:m]
                G[r, :m] += (1.0 - h[g:])
    G *= dx
    return G

def _lp_trust_region(h, scores, G, radius, target):
    """Solve LP trust region: min t s.t. scores + G*(x-h) <= t, 0<=x<=1, sum(x)=target, ||x-h||_inf <= radius"""
    n = len(h)
    # Variables: x[0..n-1], t
    # Actually, let's use delta = x - h, so:
    # min t s.t. scores + G*delta <= t, 0<=h+delta<=1, sum(delta)=0, ||delta||_inf <= radius
    # c = [0,...,0,1] (n+1 vars: delta_0..delta_{n-1}, t)
    c = np.zeros(n+1)
    c[-1] = 1.0
    # A_ub: scores + G*delta - t <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    # A_eq: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    # Bounds: max(-radius, -h[i]) <= delta[i] <= min(radius, 1-h[i])
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-14)
        hi = min(radius, 1.0 - h[i] - 1e-14)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 2000, 'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
