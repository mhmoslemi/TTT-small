def _grad_scores_from_h(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k_out in range(2 * n - 1):
        w = weights[k_out]
        if abs(w) < 1e-15:
            continue
        k = k_out - (n - 1)  # actual lag
        if k >= 0:
            # valid n range: [0, n-1-k]
            m = n - k  # number of valid n values
            if m > 0:
                # First term: (1-h[j-k]) for j in [k, n-1], i.e., j-k in [0, n-1-k]
                # This is: for i in [0, m-1], j = i+k, contribution (1-h[i])
                grad[k:] += w * (1.0 - h[:m])  # h[j-k] = h[i] for j=i+k
                # Second term: -h[j+k] for j in [0, m-1]
                grad[:m] += w * (-h[k:k+m])  # h[j+k] for j in [0,m-1]
        else:
            g = -k
            # valid n range: [g, n-1]
            m = n - g  # number of valid n values
            if m > 0:
                # First term: (1-h[j-k]) = (1-h[j+g]) for j in [g-1... wait
                # j-k = j+g must be in [g, n-1], so j in [0, n-1-g] = [0, m-1]
                # j+g in [g, n-1]
                # h[j-k] = h[j+g] for j in [0, m-1]
                # So: grad[j] += w * (1-h[j+g]) for j in [0, m-1]
                grad[:m] += w * (1.0 - h[g:g+m])  # h[j+g] for j in [0,m-1]

                grad[g:] += w * (-h[:m])  # h[j-g] for j in [g, n-1]
    return grad
