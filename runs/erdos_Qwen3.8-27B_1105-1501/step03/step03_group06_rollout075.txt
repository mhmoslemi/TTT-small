import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = float(np.sum(x)) - target
    if res > 1e-11:
        order = np.argsort(-x)
        for idx in order:
            d = min(res, x[idx])
            x[idx] -= d
            res -= d
            if res < 1e-12:
                break
    elif res < -1e-11:
        res = -res
        order = np.argsort(x)
        for idx in order:
            d = min(res, 1.0 - x[idx])
            x[idx] += d
            res -= d
            if res < 1e-12:
                break
    return x


def _grad_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    h1m = 1.0 - h
    for s in range(-(n - 1), n):
        r = n - 1 + s
        abs_s = abs(s)
        m = n - abs_s
        if m <= 0:
            continue
        G[r, abs_s:] += h1m[:m]
        G[r, :m] += -h[abs_s:]
    return G


def _pair_bisection(delta_lo, delta_hi, a, corr, max_iter=45):
    """Find delta in [delta_lo, delta_hi] minimizing max(a*delta + corr)."""
    if delta_hi - delta_lo < 1e-14:
        return (delta_lo + delta_hi) / 2.0
    lo = delta_lo
    hi = delta_hi
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        vals = a * mid + corr
        mx = float(np.max(vals))
        active = vals > mx - 1e-11
        if not np.any(active):
            break
        a_act = a[active]
        if a_act.size == 1:
            rd = ld = float(a_act[0])
        else:
            rd = float(np.max(a_act))
            ld = float(np.min(a_act))
        if rd > 1e-10:
            hi = mid
        elif ld < -1e-10:
            lo = mid
        else:
            break
        if hi - lo < 1e-13:
            break
    return (lo + hi) / 2.0


def _pairwise_cd(h, deadline, max_cycles=500, shuffle_seed=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = _corr_vec(h)
    cur_max = float(np.max(corr))
    rng = np.random.default_rng(shuffle_seed or 0)
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    stall = 0
    for cycle in range(max_cycles):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        perm = rng.permutation(len(all_pairs))
        improved = False
        for pi in perm:
            if time.time() > deadline:
                break
            i, j = all_pairs[pi]
            hi_val = h[i]
            hj_val = h[j]
            d_lo = max(-hi_val, hj_val - 1.0)
            d_hi = min(1.0 - hi_val, hj_val)
            if d_hi - d_lo < 1e-10:
                continue
            G = _grad_matrix(h)
            a = G[:, i] - G[:, j]
            delta = _pair_bisection(d_lo, d_hi, a, corr)
            if abs(delta) < 1e-13:
                continue
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            corr_new = _corr_vec(h)
            new_max = float(np.max(corr_new))
            if new_max < cur_max - 1e-11:
                corr = corr_new
                cur_max = new_max
                c5 = _exact_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                stall = 0
            else:
                h[i] = hi_val
                h[j] = hj_val
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, target)
        if not improved:
            break
    return best_h, best_c5


def _slsqp_step(h0, deadline):
    n = len(h0)
    target = n / 2.0
    corr0 = _corr_vec(h0)
    t0 = float(np.max(corr0)) + 0.05
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    def _obj(x):
        return x[n]

    def _obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def _ineq_fun(x):
        h = x[:n]
        t = x[n]
        return t - _corr_vec(h)

    def _ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def _eq_fun(x):
        return np.array([float(np.sum(x[:n])) - target])

    def _eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h0, [t0]])
    constraints = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    try:
        elapsed = time.time() - (deadline - 60.0) if False else 0
        maxit = 1500
        res = minimize(_obj, x0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxit, 'iprint': 0, 'ftol': 1e-10})
        if res.success or res.x is not None:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    return best_h, best_c5


def _binary_local_search(b, deadline, max_rounds=200):
    """b is a binary array with n/2 ones. Try swaps to reduce max corr."""
    n = len(b)
    target = n / 2.0
    b = b.copy()
    best_b = b.copy()
    curr = _corr_vec(b)
    best_max = float(np.max(curr))
    ones = set(np.where(b == 1)[0].tolist())
    zeros = set(np.where(b == 0)[0].tolist())
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        o_list = sorted(ones)
        z_list = sorted(zeros)
        for i in o_list:
            if time.time() > deadline:
                break
            for j in z_list:
                # Try swap: b[i] 1->0, b[j] 0->1
                b[i] = 0
                b[j] = 1
                new_corr = _corr_vec(b)
                new_max = float(np.max(new_corr))
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    curr = new_corr
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                    best_b = b.copy()
                    improved = True
                    o_list = sorted(ones)
                    z_list = sorted(zeros)
                    break
                else:
                    b[i] = 1
                    b[j] = 0
            if improved:
                continue
        if not improved:
            break
    best_b = np.where(best_b > 0.5, 1.0, 0.0)
    best_b = _project(best_b, target)
    return best_b, _exact_c5(best_b)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _lbfgs_smooth(h, alpha, maxiter=150):
    n = len(h)
    target = n / 2.0

    def _fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -400.0, 400.0)
        e = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(e)))
        w = e / float(np.sum(e))
        G = _grad_matrix(x)
        g = G.T @ w
        g -= np.mean(g)
        return f, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_fun, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                               'maxls': 30, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)
