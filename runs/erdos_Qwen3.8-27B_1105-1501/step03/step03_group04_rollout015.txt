import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_active(h, beta=80.0):
    n = len(h)
    s = _scores(h)
    m = float(np.max(s))
    exps = np.exp(beta * (s - m))
    w = exps / float(np.sum(exps))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                grad[:m2] += w[r] * (1.0 - h[d:])
                grad[d:] -= w[r] * h[:m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                grad[:m2] += w[r] * (1.0 - h[g:])
                grad[g:] -= w[r] * h[:m2]
    return grad

def _proj_grad(h, deadline, max_iters=200000, beta=80.0):

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):

        target_count = n_try // 2
        if cur_count < target_count:
            idx = np.where(bits == 0)[0]
            rng.shuffle(idx)
            for k in range(target_count - cur_count):
                bits[idx[k]] = 1.0
        elif cur_count > target_count:
            idx = np.where(bits == 1)[0]
            rng.shuffle(idx)
            for k in range(cur_count - target_count):
                bits[idx[k]] = 0.0
        # Multiple SA restarts
        n_restarts = max(1, int(min(40.0, time_left * 0.7 / 5.0)))
        for r in range(n_restarts):
            if time.time() > deadline2 - 10:
                break
            sub_deadline = min(deadline2 - 5, time.time() + 8)
            if r == 0:
                b, bc = _binary_hill(bits, sub_deadline, rng)
            else:
                # Perturbed start
                bp = bits.copy()
                for _ in range(3):
                    oi = np.where(bp == 1)[0]
                    zi = np.where(bp == 0)[0]
                    if len(oi) == 0 or len(zi) == 0:
                        break
                    ii = oi[rng.integers(len(oi))]
                    jj = zi[rng.integers(len(zi))]
                    bp[ii] = 0
                    bp[jj] = 1
                b, bc = _binary_sa(bp, sub_deadline, rng)
            if bc < best_c5:
                best_c5 = bc
                best_h = b.copy()
                best_n = n_try
                # Refine the binary solution with gradient
                hl = _proj_grad(b, min(deadline2 - 3, time.time() + 5), max_iters=20000)
                if hl[1] < best_c5:
                    best_c5 = hl[1]
                    best_h = hl[0].copy()

    # Phase 3: LP polish on current best
    deadline3 = t0 + budget_s * 0.60
    h, c5 = _lp_polish(best_h, deadline3, radius_init=0.06, max_iters=4000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Perturbation restarts with gradient
    deadline4 = t0 + budget_s * 0.80
    scale = 0.08
    for attempt in range(800):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        per_time = (deadline4 - time.time()) / max(800 - attempt, 1)
        sub_dl = min(deadline4, time.time() + min(per_time, 4.0))
        h_try, c5_try = _proj_grad(h_pert, sub_dl, max_iters=8000, beta=50.0)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.02, scale * 0.92)
        else:
            scale = min(0.3, scale * 1.08)

    # Phase 5: Snap and refine (encourage binarization)
    deadline5 = t0 + budget_s * 0.88
    h_snap, c5_snap = _snap_and_refine(best_h, deadline5, rng)
    if c5_snap < best_c5:
        best_c5 = c5_snap
        best_h = h_snap.copy()

    # Phase 6: Try resampling to nearby n and optimize
    deadline6 = t0 + budget_s * 0.93
    for dn in [-8, -4, -2, 2, 4, 8]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 28 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline6, time.time() + budget_s * 0.012)
        h_r, c5_r = _proj_grad(h_r, sub_dl, max_iters=15000)
        h_r, c5_r = _lp_polish(h_r, min(deadline6, time.time() + budget_s * 0.01),
                               radius_init=0.04, max_iters=800)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final LP polish
    deadline7 = t0 + budget_s * 0.97
    h_f, c5_f = _lp_polish(best_h, deadline7, radius_init=0.03, max_iters=3000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Phase 8: Coordinate swap
    deadline8 = t0 + budget_s * 0.995
    h_cd, c5_cd = _coord_swap(best_h, deadline8, max_passes=40)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
