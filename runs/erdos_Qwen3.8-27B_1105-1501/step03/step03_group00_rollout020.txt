    """Compute change in scores for swapping b[i] (1) and b[j] (0)"""
    delta = np.zeros(2*n-1)  # indexed by m = d + n - 1

        # Term 4: k=j-d, need j-d valid
        if 0 <= j-d <= n-1:

    return delta

def _compute_swap_deltas(b, n, i, j):
    """Compute change in all scores when swapping b[i] (1) and b[j] (0)"""
    # delta[m] where m = d + n - 1, d ranges from -(n-1) to n-1
    # out[n-1+d] = sum_k b[k] * (1 - b[k+d])
    # After swap at i (1→0) and j (0→1):
    # Change = -(1-b[i+d]) + (1-b[j+d]) + b[i-d] - b[j-d] (each if valid)
    
    deltas = {}  # sparse: d -> change
    # We only need to check d values where at least one term is non-zero
    # Term 1: -(1-b[i+d]) non-zero when b[i+d]=0 and 0<=i+d<n
    # Term 2: +(1-b[j+d]) non-zero when b[j+d]=0 and 0<=j+d<n
    # Term 3: +b[i-d] non-zero when b[i-d]=1 and 0<=i-d<n
    # Term 4: -b[j-d] non-zero when b[j-d]=1 and 0<=j-d<n
    
    # For efficiency, just iterate over all d
    for d in range(-(n-1), n):
        delta = 0.0
        if 0 <= i+d < n and b[i+d] == 0:
            delta -= 1.0
        if 0 <= j+d < n and b[j+d] == 0:
            delta += 1.0
        if 0 <= i-d < n and b[i-d] == 1:
            delta += 1.0
        if 0 <= j-d < n and b[j-d] == 1:
            delta -= 1.0
        if delta != 0:
            deltas[d + n - 1] = deltas.get(d + n - 1, 0) + delta
    
    return deltas  # dict from index to change

def _binary_greedy(b, max_rounds=200, deadline=None):
    """Find best improving swap and apply. Repeat."""
    n = len(b)
    b = b.copy()
    dx = 2.0 / n
    cur_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        best_delta = 0.0
        best_oi, best_oz = -1, -1

def _binary_sa(b, deadline, rng, init_T=None):
    """Simulated annealing on binary vectors"""
    n = len(b)
    b = b.copy()
    dx = 2.0 / n
    cur_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
    best_b = b.copy()
    best_max = cur_max
    
    if init_T is None:
        init_T = 0.05
    T = init_T
    
    while time.time() < deadline:
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        b[i] = 0
        b[j] = 1
        new_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
        delta = new_max - cur_max
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        
        T *= 0.9995
        if T < 0.005:
            T = 0.005  # floor to allow some exploration
    
    return best_b, best_max

def _continuous_optimize(h_init, deadline, rng):
    """Gradient-based continuous optimization with softmax weights"""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _eval_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    beta = 50.0
    step_size = 0.1
    
    while time.time() < deadline:
        scores = _eval_scores(h)
        max_s = float(np.max(scores))
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros_like(scores)
            w[np.argmax(scores)] = 1.0
        else:
            w = w / sw
        
        # Gradient: d(scores)/d(h) = correlate matrix transposed
        # score_d = sum_i h[i] * (1-h[i+d])
        # d(score_d)/d(h[i]) = (1-h[i+d]) - h[i-d]*(if valid) ... 
        # Actually let's use the chain rule properly:
        # score_d = sum_k h[k]*(1-h[k+d])
        # d(score_d)/d(h[i]) = (1-h[i+d]) if i+d valid  [from h[i] in first factor]
        #                       - h[i-d] if i-d valid    [from h[i] in second factor: 1-h[i]]
        
        # Build gradient as G[:, i] where G[m, i] = d(score_m)/d(h[i])
        # Then grad[i] = sum_m w[m] * G[m, i]
        
        grad = np.zeros(n)
        for d in range(-(n-1), n):
            m = n - 1 + d
            w_m = w[m]
            if w_m < 1e-15:
                continue
            for k in range(max(0, -d), min(n, n - d)):
                # h[k] appears in score_m as: h[k]*(1-h[k+d])
                # d/dh[k]: (1-h[k+d]) - h[k-d] (second term valid if k-d in [0,n-1])
                g = 1.0 - h[k + d]
                if 0 <= k - d <= n - 1:
                    g -= h[k - d]
                grad[k] += w_m * g
        
        # Project gradient to zero-sum subspace
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Line search
        grad = grad / gn
        best_alpha = 0.0
        best_val = cur_c5
        for alpha in [step_size * 0.5**k for k in range(15)]:
            h_try = _project(h + alpha * grad, target)
            c5_try = _eval_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = alpha
        
        if best_alpha > 0:
            h = _project(h + best_alpha * grad, target)
            cur_c5 = _eval_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
        
        # Adaptively increase beta to focus on worst lags
        beta *= 1.05
        step_size *= 0.95
