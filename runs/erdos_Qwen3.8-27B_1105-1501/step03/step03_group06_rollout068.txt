import numpy as np
from scipy.optimize import linprog, minimize
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            if m > 0:
                G[r, :m] += (1.0 - h[s:s + m])
                G[r, s:] += -h[:m]
        else:
            g = -s
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:g + m]
    return G


def _smooth_objective(x, alpha, target):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_step(h, alpha, maxiter=200, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_objective, h, args=(alpha, target),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _lsqp_obj(z, target, penalty):
    n = int(target * 2)
    h = z[:n]
    b = z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = b + penalty * np.sum(h * (1.0 - h))
    return obj


def _lsqp_obj_grad(z, target, penalty):
    n = int(target * 2)
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    # gradient of objective wrt h: penalty*(1-2h), wrt b: 1
    g_h = penalty * (1.0 - 2.0 * h)
    g_b = np.array([1.0])
    # constraint gradients added by SLSQP
    return np.concatenate([g_h, g_b])


def _lsqp_cons(z, target):
    n = int(target * 2)
    h = z[:n]
    b = z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # constraints: b - corr[r] >= 0 for all r
    return b - corr


def _lsqp_cons_jac(z, target):
    n = int(target * 2)
    h = z[:n]
    G = _grad_matrix(h)
    # d(b - corr[r])/dz = [-G[r,:], 1]
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _lsqp_optimize(h0, target, penalty, maxiter=300, deadline=None):
    n = len(h0)
    z0 = np.concatenate([h0, [float(np.max(np.correlate(h0, 1.0 - h0, mode='full')))]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    eq_cons = [{'type': 'eq',
                'fun': lambda z: np.sum(z[:n]) - target,
                'jac': lambda z: np.concatenate([np.ones(n), [0.0]])}]
    ineq_cons = [{'type': 'ineq',
                  'fun': _lsqp_cons,
                  'jac': _lsqp_cons_jac,
                  'args': (target,)}]
    try:
        res = minimize(_lsqp_obj, z0, args=(target, penalty),
                       method='SLSQP', bounds=bounds,
                       constraints=eq_cons + ineq_cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.success or hasattr(res, 'x'):
            x = np.clip(res.x[:n], 0.0, 1.0)
            x = _project(x, target)
            return x
    except Exception:
        pass
    return None


def _lp_direction(h, G, scores, radius, active_frac=1.0):
    n = len(h)
    num_corr = 2 * n - 1
    n_active = max(1, int(num_corr * active_frac))
    top_k = np.argpartition(scores, num_corr - n_active)[num_corr - n_active:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(top_k), n + 1))
    A_ub[:, :n] = G[top_k, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((-radius, radius))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 14):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(14):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t

    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.15, max_iters=4000, stall_limit=60,
             active_frac_init=0.5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    af = active_frac_init

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, G, scores, radius, af)

        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            af = min(1.0, af * 1.3)
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            af = max(0.3, af * 0.9)
            radius = min(radius * 1.12, 0.4)
        else:
            stall += 1
            af = min(1.0, af * 1.2)
            radius *= 0.72
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.08

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _freq_perturb(h, rng, scale):
    n = len(h)
    h_p = h.copy()
    for freq in [2, 3, 5, 8, 13]:
        amp = rng.uniform(-scale, scale)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        h_p += amp * np.sin(2 * np.pi * freq * t / n + phase)
    return h_p


def _block_perturb(h, rng, scale):
    n = len(h)
    h_p = h.copy()
    for _ in range(3):
        start = rng.integers(0, n - 2)
        length = rng.integers(2, min(8, n - start))
        h_p[start:start + length] += rng.uniform(-scale, scale)
    return h_p


def _balancing_step(h, G, scores, target):
    n = len(h)
    ms = float(np.max(scores))
    active = np.where(scores > ms - 0.5 * (ms - np.min(scores)))[0]
    if len(active) == 0:
        return h.copy(), _exact_c5(h)
    # Try to reduce active correlations by moving in -gradient direction
    avg_grad = np.mean(G[active, :], axis=0)
    avg_grad -= np.mean(avg_grad)
    norm = float(np.sqrt(np.sum(avg_grad ** 2)))
    if norm < 1e-14:
        return h.copy(), _exact_c5(h)
    d = -avg_grad / norm
    # Line search with small step
    best_a, best_c5 = 0.0, _exact_c5(h)
    for a in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
        h_t = _project(h + a * d, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        return _project(h + best_a * d, target), best_c5
    return h.copy(), _exact_c5(h)
