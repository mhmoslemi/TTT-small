    """Find optimal h[i] given all other coordinates fixed. Returns new h[i]."""
    n = len(h)
    dx = 2.0 / n
    n1 = n - 1

    # a_r = dx * [(1-h[i-d_r])*[0<=i-d_r<n] - h[i+d_r]*[0<=i+d_r<n]]

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-10:
        return v
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(x)) > target:
            lo = mid
        else:
            hi = mid
    level = (lo + hi) / 2.0
    x = np.clip(v - level, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _soft_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    return tau * float(np.log(np.sum(np.exp((scores - max_s) / tau)))) + max_s

def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    grad = np.zeros(n)
    n1 = n - 1
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - n1
        i_lo = max(0, d)
        i_hi = min(n - 1, d + n - 1)
        if i_lo <= i_hi:
            j_lo = i_lo - d
            j_hi = i_hi - d
            grad[i_lo:i_hi + 1] += wr * (1.0 - h[j_lo:j_hi + 1])
        i_lo2 = max(0, -d)
        i_hi2 = min(n - 1, n - 1 - d)
        if i_lo2 <= i_hi2:
            j_lo2 = i_lo2 + d
            j_hi2 = i_hi2 + d
            grad[i_lo2:i_hi2 + 1] += wr * (-h[j_lo2:j_hi2 + 1])
    return dx * grad

def _slsqp_obj(x, tau):
    return _soft_value(x, tau)

def _slsqp_jac(x, tau):
    return _soft_grad(x, tau)

def _con_fun(x):
    return np.sum(x) - _C_TARGET[0]

def _con_jac(x):
    return np.ones(len(x))
