        if time.time() > deadline or stall > 150:

        scores = np.correlate(h, 1.0 - h, mode='full')
        direction = _smooth_direction(h, scores, beta)
        norm_g = float(np.sqrt(np.sum(direction**2)))
        if norm_g < 1e-13:
            break
        direction = direction / norm_g
        if norm_g < best_norm * 0.3:
            best_norm = norm_g
            stall += 1
        else:
            best_norm = norm_g
        a_opt, f_new = _line_search(h, direction, target, cur_c5, deadline)
        if f_new < cur_c5 - 1e-13:
            h = np.clip(h + a_opt * direction, 0.0, 1.0)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            beta = min(beta * 1.4, 400.0)
            if beta > 200 and stall > 50:
                break
    return best_h, best_c5

def _basin_hopping(h, rng, deadline, target_c5=0.3808):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.04
    while time.time() < deadline and best_c5 > target_c5 - 1e-6:
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(time.time() + 15.0, deadline)
        h_opt, c5_opt = _smooth_opt(h_pert, sub_dl, beta_init=40.0, max_iters=800)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            scale = max(0.008, scale * 0.9)
        else:
            scale = min(0.12, scale * 1.15)
    return best_h, best_c5

def _de_antiperiodic(h_start, n, deadline, population=25, maxiter=60):
    half = n // 2
    y0 = h_start[:half].copy()
    bounds = [(0.0, 1.0)] * half
    
    def de_obj(y):
        h = np.zeros(n)
        h[:half] = y
        h[half:] = 1.0 - y
        return _exact_c5(h)
    
    c5_current = _exact_c5(h_start)
    
    def de_cb(xk, con):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        result = differential_evolution(de_obj, bounds, seed=42, maxiter=maxiter,
                                         popsize=population, tol=1e-9, mut=(0.3, 0.9),
                                         recomb=0.7, polish=False, init='latinhypercube',
                                         updating='deferred', callback=de_cb,
                                         popsize=population)
        if result.success:
            h_de = np.zeros(n)
            h_de[:half] = result.x
            h_de[half:] = 1.0 - result.x
            c5_de = _exact_c5(h_de)
            if c5_de < c5_current:
                return h_de, c5_de
    except RuntimeError:
        pass
    except Exception:
        pass
    return None, None

def _de_random(h_start, n, deadline, rng, population=20, maxiter=40):
    bounds = [(0.0, 1.0)] * n
    c5_current = _exact_c5(h_start)
    
    def de_obj(v):
        h = _project(v, n / 2.0)
        return _exact_c5(h)
    
    try:
        result = differential_evolution(de_obj, bounds, seed=int(rng.integers(100000)),
                                         maxiter=maxiter, popsize=population,
                                         tol=1e-8, mut=(0.4, 0.9), recomb=0.8,
                                         polish=False, init='latinhypercube',
                                         updating='immediate')
        if result.success:
            h_de = _project(result.x, n / 2.0)
            c5_de = _exact_c5(h_de)
            if c5_de < c5_current:
                return h_de, c5_de
    except Exception:
        pass
    return None, None

def _active_set_kkt(h, deadline, top_k=4, max_newton=50):
    """Try to solve KKT system for active set"""
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    idx_top = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in idx_top]
    
    # Build Jacobian of active scores w.r.t. h
    J = np.zeros((top_k, n))
    for i, d in enumerate(active_lags):
        J[i] = _score_grad(h, d)
    
    # Hessian of each score w.r.t. h
    # score_d = sum h[k+d]*(1-h[k]) = sum h[k+d] - sum h[k+d]*h[k]
    # d(score_d)/d(h[j]) is in J
    # d²(score_d)/d(h[j])d(h[j']) involves products
    
    # Use a simpler approach: iterate the active set QP via SLSQP-like
    from scipy.optimize import minimize
    
    c5_init = float(np.max(scores)) + 1e-6
    
    def obj_func(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    
    for d in active_lags:
        def cons_fun(x, dd=d):
            return x[n] - float(np.sum(np.correlate(x[:n], 1.0 - x[:n], mode='full'))[n-1+dd])
        def cons_jac(x, dd=d):
            return np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        cons.append({'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac})
    
    x0 = np.concatenate([h, [c5_init]])
    bds = [(0.0, 1.0)] * n + [(0.0, c5_init + 1.0)]
    
    def dl_cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        res = minimize(obj_func, x0, method='SLSQP', jac=obj_jac,
                       bounds=bds, constraints=cons,
                       options={'maxiter': max_newton, 'ftol': 1e-14, 'disp': False},
                       callback=dl_cb)
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        if abs(np.sum(h_new) - target) > 1e-4:
            h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            return h_new, c5_new
    except RuntimeError:
        pass
    except Exception:
        pass
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pairwise_sweep(h, deadline, max_sweeps=10):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle pair order for diversity
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            hi_val = h[i]
            hj_val = h[j]
            if hi_val < 1e-10 or hi_val > 1.0 - 1e-10:
                continue
            if hj_val < 1e-10 or hj_val > 1.0 - 1e-10:
                continue
            
            # Try transferring from i to j
            max_t = min(1.0 - hi_val, hj_val) * 0.99
            if max_t < 1e-8:
                continue
            
            # Quick check with a few samples
            best_t = 0.0
            best_f = cur_c5
            for frac in [0.25, 0.5, 0.75]:
                t = frac * max_t
                h[i] = hi_val + t
                h[j] = hj_val - t
                f = _exact_c5(h)
                if f < best_f:
                    best_f = f
                    best_t = t
                h[i] = hi_val
                h[j] = hj_val
            
            if best_t > 0 and best_f < cur_c5 - 1e-11:
                # Golden section refine
                lo_t, hi_t = 0.0, max_t
                for _ in range(12):
                    if hi_t - lo_t < 1e-10:
                        break
                    t1 = hi_t - gr * (hi_t - lo_t)
                    t2 = lo_t + gr * (hi_t - lo_t)
                    h[i] = hi_val + t1
                    h[j] = hj_val - t1
                    f1 = _exact_c5(h)
                    h[i] = hi_val + t2
                    h[j] = hj_val - t2
                    f2 = _exact_c5(h)
                    h[i] = hi_val
                    h[j] = hj_val
                    if f1 < best_f:
                        best_f = f1
                        best_t = t1
                    if f2 < best_f:
                        best_f = f2
                        best_t = t2
                    if f1 < f2:
                        hi_t = t2
                    else:
                        lo_t = t1
                h[i] = hi_val + best_t
                h[j] = hj_val - best_t
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Aggressive smooth gradient descent (0-100s)
    dl1 = t0 + budget_s * 0.09
    h1, c5_1 = _smooth_opt(h, dl1, beta_init=30.0, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # KKT polish
    dl_kkt = t0 + budget_s * 0.12
    h_kkt, c5_kkt = _active_set_kkt(best_h, dl_kkt, top_k=5, max_newton=80)
    if h_kkt is not None and c5_kkt < best_c5:
        best_c5 = c5_kkt
        best_h = h_kkt.copy()
        h_kkt2, c5_kkt2 = _active_set_kkt(best_h, dl_kkt, top_k=4, max_newton=60)
        if h_kkt2 is not None and c5_kkt2 < best_c5:
            best_c5 = c5_kkt2
            best_h = h_kkt2.copy()
    
    # Phase 2: Basin hopping (100-350s)
    dl2 = t0 + budget_s * 0.32
    h2, c5_2 = _basin_hopping(best_h, rng, dl2, target_c5=0.38079)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Differential evolution in anti-periodic subspace (350-500s)
    dl3 = t0 + budget_s * 0.45
    if time.time() < dl3:
        h_de, c5_de = _de_antiperiodic(best_h, best_n, dl3, population=30, maxiter=80)
        if h_de is not None and c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            # Refine DE result
            dl3b = min(t0 + budget_s * 0.52, dl3 + 40)
            h3, c5_3 = _smooth_opt(best_h, dl3b, beta_init=40.0, max_iters=2000)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
    
    # Phase 4: DE in full random space
    dl4 = t0 + budget_s * 0.52
    if time.time() < dl4:
        h_de2, c5_de2 = _de_random(best_h, best_n, dl4, rng, population=20, maxiter=40)
        if h_de2 is not None and c5_de2 < best_c5:
            best_c5 = c5_de2
            best_h = h_de2.copy()
            h4, c5_4 = _smooth_opt(best_h, min(t0 + budget_s * 0.56, time.time() + 30),
                                    beta_init=50.0, max_iters=1500)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
    
    # Phase 5: Multi-resolution search (500-750s)
    dl5 = t0 + budget_s * 0.68
    for dn in [0, -4, 4, -8, 8, -16, 16, -2, 2]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + 60.0)
        h_r, c5_r = _smooth_opt(h_r, sub_dl, beta_init=30.0, max_iters=2000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Try basin hopping at this resolution
            if time.time() < dl5:
                h_r2, c5_r2 = _basin_hopping(best_h, rng, min(dl5, time.time() + 60),
                                              target_c5=best_c5 - 0.0001)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()
    
    # Phase 6: More basin hopping at best resolution (750-900s)
    dl6 = t0 + budget_s * 0.82
    h6, c5_6 = _basin_hopping(best_h, rng, dl6, target_c5=best_c5 - 0.00005)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Phase 7: Final smooth optimization (900-1000s)
    dl7 = t0 + budget_s * 0.92
    h7, c5_7 = _smooth_opt(best_h, dl7, beta_init=80.0, max_iters=5000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Phase 8: Pairwise sweep (1000-1060s)
    dl8 = t0 + budget_s * 0.97
    h8, c5_8 = _pairwise_sweep(best_h, dl8, max_sweeps=15)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    
    # Phase 9: Very final polish
    dl9 = t0 + budget_s * 0.995
    h9, c5_9 = _smooth_opt(best_h, dl9, beta_init=100.0, max_iters=3000)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    # If projection hurt, try without
    h_no_proj = np.clip(h0 if best_n == n0 else best_h, 0.0, 1.0)
    if abs(np.sum(h_no_proj) - best_n / 2.0) < 1e-6:
        c5_np = _exact_c5(h_no_proj)
        if c5_np < final_c5:
            best_h = h_no_proj
            final_c5 = c5_np
    
    return best_h, final_c5, best_n
