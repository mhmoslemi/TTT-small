def _active_set_lp(h, radius, top_k=10, tol=1e-4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    C_max = np.max(corr)
    # Active set: correlations within tol of max
    active = np.where(corr >= C_max - tol)[0]
    # Limit to top_k
    if len(active) > top_k:
        # Sort by value and take top_k
        active_sorted = active[np.argsort(corr[active])[::-1][:top_k]]
    else:
        active_sorted = active
    
    G = _grad_matrix(h)
    G_a = G[active_sorted, :]  # shape (k, n)
    C_a = corr[active_sorted]  # shape (k,)
    k = len(active_sorted)
    
    # LP: minimize t
    # s.t. G_a @ p <= t - C_a  =>  G_a @ p - t <= -C_a
    #      sum(p) = 0
    #      -radius <= p_i <= radius
    #      t is free (or bounded below by 0)
    
    # Variables: [p_0, ..., p_{n-1}, t]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: G_a @ p - t <= -C_a
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -C_a
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _active_set_lp(h, radius, top_k=15, tol=1e-3):
    """Solve an LP to find a direction that reduces the top-k active correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    C_max = np.max(corr)
    active = np.where(corr >= C_max - tol)[0]
    if len(active) == 0:

    if len(active) > top_k:

    G_a = G[active, :]

    k = len(active)

    c[-1] = 1.0

    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -C_a

    bounds = [(-radius, radius)] * n + [(None, None)]

            if np.all(new_C <= C_a + 1e-10):

    except Exception:
        pass
    return None

def _line_search_c5(h, p, c5_cur, max_alpha=1.0):
    """Find the best step size alpha."""
    n = len(h)
    target = n / 2.0
    # Compute max step
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    best_a, best_c5 = 0.0, c5_cur
    # Coarse search
    for t in np.linspace(0.0, 1.0, 10):
        a = t * alpha_max
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    
    # Golden section refinement
    if best_a > 0 and best_a < alpha_max:
        lo = max(0.0, best_a - alpha_max * 0.2)
        hi = min(alpha_max, best_a + alpha_max * 0.2)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(16):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, target))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(_project(h + a * p, target))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    
    return best_a, best_c5

def _smooth_obj_and_grad(x, alpha, n):
    """Log-sum-exp smooth approximation of max correlation, with gradient."""
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    C_max = np.max(corr)
    vals = np.clip(alpha * (corr - C_max), -500.0, 500.0)
    exp_vals = np.exp(vals)
    log_sum = np.log(np.sum(exp_vals))
    f = dx * (C_max + log_sum / alpha)
    w = exp_vals / np.sum(exp_vals)
    G = _grad_matrix(x)
    g = dx * (G.T @ w)
    g -= np.mean(g)  # Ensure sum-free direction
    return f, g

def _lbfgs_smooth(h, alpha, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj_and_grad, h, args=(alpha, n),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_epigraph(h0, n, maxiter=200):
    """SLSQP with epigraph formulation on active constraints only."""

from scipy.optimize import minimize, linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

            lo = mid
        else:
            hi = mid
