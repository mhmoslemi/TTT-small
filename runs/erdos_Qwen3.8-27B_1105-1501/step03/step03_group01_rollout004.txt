def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_scores(h):

def _active_lp_direction(h, scores, radius, top_k=15):
    n = len(h)
    num_constr = len(scores)
    k = min(top_k, num_constr)
    top_idx = np.argsort(scores)[::-1][:k]
    G = np.zeros((k, n))
    for row, r in enumerate(top_idx):
        d = int(r) - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[row, d:] += (1.0 - h[:m])
                G[row, :m] += (-h[d:])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[row, :m] += (1.0 - h[g:])
                G[row, g:] += (-h[:m])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx].astype(float)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop_active(h, deadline, radius_init=0.1, top_k=15, max_stall=80):

def _penalty_gradient(h, deadline, mu=150.0, lr=0.012):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    beta = 0.9
    stall = 0
    while time.time() < deadline and stall < 60:
        scores = _c5_scores(h)
        max_s = float(np.max(scores))
        exp_s = np.exp(mu * (scores - max_s))
        wsum = float(np.sum(exp_s))
        if wsum < 1e-30:
            break
        w = exp_s / wsum
        grad = np.zeros(n)
        for r in range(len(w)):
            if w[r] < 1e-10:
                continue
            wr = w[r]
            d = int(r) - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[d:] += wr * (1.0 - h[:m])
                    grad[:m] += wr * (-h[d:])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += wr * (1.0 - h[g:])
                    grad[g:] += wr * (-h[:m])
        mom = beta * mom + (1.0 - beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.06)
        else:
            stall += 1
            lr *= 0.92
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue

                continue
            if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-10:
                    continue
                for frac in [0.5, 1.0]:
                    h_t = h.copy()

                    h_t[k] += dk * am * frac
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5
