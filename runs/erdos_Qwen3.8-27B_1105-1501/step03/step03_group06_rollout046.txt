import numpy as np
from scipy.optimize import linprog, minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _smooth_fun(x, alpha, target):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha, target),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15, max_iters=4000, stall_limit=60):

def _slsqp_epigraph(h0, deadline, top_k=8, maxiter=400):
    n = len(h0)
    target = n / 2.0
    K = 2 * n - 1
    h = h0.copy()
    s0 = np.correlate(h, 1.0 - h, mode='full')
    ms0 = float(np.max(s0))
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    active_idx = np.argsort(-s0)[:top_k]

    def _obj(x):
        return x[0]

    def _obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def _con_fun(x):
        hh = x[1:]
        T = x[0]
        s = np.correlate(hh, 1.0 - hh, mode='full')
        return T - s[active_idx]

    def _con_jac(x):
        hh = x[1:]
        G = _grad_matrix(hh)
        J = np.zeros((top_k, n + 1))
        J[:, 0] = 1.0
        for ii in range(top_k):
            J[ii, 1:] = -G[active_idx[ii], :]
        return J

    def _eq_fun(x):
        return np.sum(x[1:]) - target

    def _eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.zeros(n + 1)
    x0[0] = ms0 + 0.005
    x0[1:] = h

    bounds = [(None, None)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': _con_fun, 'jac': _con_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]

    try:
        res = minimize(_obj, x0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_new = _project(res.x[1:], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass

    # Try with updated active set
    if time.time() < deadline:
        s_new = np.correlate(best_h, 1.0 - best_h, mode='full')
        new_active = np.argsort(-s_new)[:top_k]
        if not np.array_equal(new_active, active_idx):

            x0[1:] = best_h
            x0[0] = float(np.max(s_new)) + 0.005
            try:

                    c5_new = _exact_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
            except Exception:
                pass

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    # Phase 3: Smooth optimization (10% budget)
    d3 = t0 + budget_s * 0.45
    h, c5 = _smooth_loop(best_h, d3, alphas=(20.0, 50.0, 100.0, 200.0))
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 4: LP again after smooth (10% budget)
    d4 = t0 + budget_s * 0.55
    h, c5 = _lp_loop(best_h, d4, radius_init=0.1, max_iters=4000, stall_limit=45)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 5: Restart with structured perturbations (25% budget)
    d5 = t0 + budget_s * 0.80
    scale = 0.03
    attempts = 0
    while time.time() < d5:
        attempts += 1
        if best_c5 < 0.3806:
            break
        r = rng.random()
        if r < 0.3:
            h_p = best_h + rng.normal(0, scale, best_n)
        elif r < 0.6:
            h_p = _sine_perturb(best_h, rng, (0.01, 0.06 + scale))
        elif r < 0.8:
            h_p = _shift_perturb(best_h, rng, max_shift=2)
        else:
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _sine_perturb(h_p, rng, (0.005, 0.03))

        h_p = _project(h_p, best_n / 2.0)

        # Quick SLSQP
        sub_d = min(d5, time.time() + budget_s * 0.03)
        h_try, c5_try = _slsqp_epigraph(h_p, sub_d, top_k=8, maxiter=200)

        if c5_try < best_c5 - 1e-11:
            # Follow with LP
            sub_d2 = min(d5, time.time() + budget_s * 0.03)
            h_try, c5_try = _lp_loop(h_try, sub_d2, radius_init=0.08,
                                     max_iters=500, stall_limit=35)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.005, scale * 0.85)
        elif c5_try < best_c5 - 1e-13:
            sub_d2 = min(d5, time.time() + budget_s * 0.02)
            h_try, c5_try = _lp_loop(h_try, sub_d2, radius_init=0.06,
                                     max_iters=300, stall_limit=30)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.005, scale * 0.9)
        else:
            scale = min(0.15, scale * 1.12)

    # Phase 6: Resolution search (10% budget)
    d6 = t0 + budget_s * 0.90
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 12, -12]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _slsqp_epigraph(h_r, sub_d, top_k=8, maxiter=200)
        if time.time() < d6:
            sub_d2 = min(d6, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_loop(h_r, sub_d2, radius_init=0.1,
                                 max_iters=800, stall_limit=40)
        if c5_r < best_c5 - 1e-11:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Pair polish (3% budget)
    if time.time() < t0 + budget_s * 0.93:
        d7 = t0 + budget_s * 0.94
        hp, c5p = _pair_polish(best_h, d7, max_passes=40)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 8: Final fine LP polish (6% budget)
    d8 = t0 + budget_s * 0.99
    h_f, c5_f = _lp_loop(best_h, d8, radius_init=0.03, max_iters=5000, stall_limit=100)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
        best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
