def _lp_direction(h, top_k=10, radius=0.1):
    n = len(h)
    scores = _scores(h)
    # Get top k lags by score
    k = min(top_k, len(scores))
    top_idx = np.argsort(scores)[-k:]
    
    # Build constraint matrix A_ub (k x n) for these lags
    A = np.zeros((k, n))
    for i, idx in enumerate(top_idx):
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            A[i, :m] = (1.0 - h[d:])
            A[i, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            A[i, :m] = (1.0 - h[g:])
            A[i, g:] += (-h[:m])
    
    # Scale by dx
    dx = 2.0 / n
    A *= dx
    
    # LP: minimize t
    # s.t. A @ p <= t - scores[top_idx]  →  A @ p - t <= -scores[top_idx]
    #      sum(p) = 0
    #      -radius <= p_i <= min(radius, 1-h[i]) and p_i >= max(-radius, -h[i])
    
    # Variables: [p_0, ..., p_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@p - t <= -scores
    b_ub = -scores[top_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None
