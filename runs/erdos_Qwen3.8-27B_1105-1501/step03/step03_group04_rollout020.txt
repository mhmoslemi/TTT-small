```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    s = np.sum(np.exp(beta * (scores - m)))

def _soft_descent_v2(h, deadline, beta_start=5.0, beta_end=200.0, max_iters=5000):
    """Soft descent with logsumexp objective and adaptive step size."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.5
    stall = 0
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = np.max(scores)
        exps = np.exp(beta * (scores - m))
        weights = exps / np.sum(exps)
        
        # Compute gradient
        n_pts = len(h)
        G = np.zeros((2 * n_pts - 1, n_pts))
        for d in range(-(n_pts - 1), n_pts):
            r = n_pts - 1 + d
            if d >= 0:
                mlen = n_pts - d
                if mlen > 0:
                    G[r, d:] += (1.0 - h[:mlen])
                    G[r, :mlen] -= h[d:]
            else:
                g = -d
                mlen = n_pts - g
                if mlen > 0:
                    G[r, g:] += (1.0 - h[:mlen])
                    G[r, :mlen] -= h[g:]
        
        grad = G.T @ weights
        grad -= np.mean(grad)  # Remove component along all-ones (preserves sum)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        
        # Line search
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 3.0)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.6
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def _slsqp_refine(h, deadline, target_c5=None):
    """Use SLSQP to minimize the max correlation."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    if target_c5 is None:
        target_c5 = _exact_c5(h)
    
    # Reformulate: minimize t subject to corr[k] <= t for all k
    # But SLSQP doesn't handle 'minimize max' directly well.
    # Instead: minimize sum(softplus(corr[k] - t)) + lambda * t
    # Or: minimize t with constraints corr[k] <= t
    # Variables: [h[0], ..., h[n-1], t]
    
    def objective(x):
        h_vec = x[:n]
        t = x[n]
        scores = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        # Penalize violations of t
        violations = np.maximum(0, scores - t)
        return t * dx + 100.0 * np.sum(violations**2) * dx
    
    def objective_grad(x):
        h_vec = x[:n]
        t = x[n]
        scores = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        grad_h = np.zeros(n)
        for d in range(-(n-1), n):
            r = n - 1 + d
            if d >= 0:
                mlen = n - d
                if mlen > 0:
                    s = scores[r]
                    if s > t:
                        w = 200.0 * (s - t)
                    else:
                        w = 0.0
                    grad_h[d:] += w * (1.0 - h_vec[:mlen])
                    grad_h[:mlen] += w * (-h_vec[d:])
            else:
                g = -d
                mlen = n - g
                if mlen > 0:
                    s = scores[r]
                    if s > t:
                        w = 200.0 * (s - t)
                    else:
                        w = 0.0
                    grad_h[g:] += w * (1.0 - h_vec[:mlen])
                    grad_h[:mlen] += w * (-h_vec[g:])
        grad_t = dx - 100.0 * dx * np.sum(np.maximum(0, scores - t))
        return np.concatenate([grad_h, [grad_t]])
    
    # Constraints
    def constraint_sum(x):
        return np.sum(x[:n]) - target
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    bounds = [(0, 1)] * n + [(0, 2.0/n * 64)]  # t upper bound
    
    x0 = np.concatenate([h, [_exact_c5(h) * 1.01]])
    
    try:
        result = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-10})
        if result.success:
            h_opt = _project(result.x[:n], target)
            c5_opt = _exact_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h, _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] -= h[g:]
    return G

def _random_perturb(h, rng, scale):
    """Add a random perturbation and project."""
    n = len(h)
    noise = rng.normal(0, scale, n)
    noise -= np.mean(noise)
    return _project(h + noise, n / 2.0)

def _structure_perturb(h, rng):
    """Apply structural perturbations (flips, shifts, mixtures)."""
    n = len(h)
    choice = rng.integers(0, 4)
    if choice == 0:
        # Flip a random subset
        mask = rng.random(n) < 0.3
        h_new = h.copy()
        h_new[mask] = 1.0 - h_new[mask]
        return _project(h_new, n / 2.0)
    elif choice == 1:
        # Circular shift
        shift = rng.integers(1, max(2, n // 8))
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] -= h[g:]
    return G

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Lipschitz constant estimate
    G0 = _grad_matrix(h)
    L = float(np.max(np.sum(G0**2, axis=1))) * dx * dx  # rough estimate
    L = max(L, 1.0)
    
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        scores_y = _scores(y)
        m = np.max(scores_y)
        exps = np.exp(beta * (scores_y - m))
        weights = exps / np.sum(exps)
        
        G_y = _grad_matrix(y)
        grad = (G_y.T @ weights) * dx
        
        # Forward step
        x_new = y - grad / L
        
        # Backward step (projection)
        x_new = _project(x_new, target)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        
        # Check bounds on y (soft clip to avoid wandering)
        if np.any(y < -1) or np.any(y > 2):
            y = x_new  # restart momentum
            t_k = 1.0
        else:
            t_k = t_new
        
        x = x_new
        
        # Track best
        c5_cur = _exact_c5(x)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = x.copy()
        
        # Periodic restart if no improvement
        if it % 500 == 0 and best_c5 == _exact_c5(h):
            break
    
    return best_h, best_c5
```

```python
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.max(scores_at_0)  # f(0)
    for t in candidates:
        val = np.max(scores_at_0 + t * deltas)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val
```

This is O(n²) per coordinate, which for n=64 and 127 scores is about 64 * 8000 = 512000 operations per pass. That's very fast.

Actually, I realize I should be more careful about the exact coordinate descent. Let me verify the linearity claim.

Score at lag d (d ≥ 0): S_d = ∑_{i=0}^{n-1-d} h[i+d] * (1-h[i])

If I fix all h except h[j], then:
- If j-d is in [0, n-1-d]: h[j] appears as h[(j-d)+d] = h[j], contributing h[j]*(1-h[j-d])
- If j is in [0, n-1-d]: h[j] appears as h[j], contributing h[j+d]*(1-h[j]) = h[j+d] - h[j+d]*h[j]

So S_d = (terms not involving h[j]) + h[j]*(1-h[j-d])*1[j-d valid] + h[j+d]*(1-h[j])*1[j valid]
= (terms not involving h[j]) + h[j]*(1-h[j-d])*1[j-d valid] + h[j+d]*1[j valid] - h[j+d]*h[j]*1[j valid]
= (terms not involving h[j]) + h[j]*[(1-h[j-d])*1[j-d valid] - h[j+d]*1[j valid]] + h[j+d]*1[j valid]

So S_d = A_d + h[j] * B_d where:
A_d = (terms not involving h[j]) + h[j+d]*1[j valid]
B_d = (1-h[j-d])*1[j-d valid] - h[j+d]*1[j valid]

This confirms S_d is affine in h[j]. 

So the coefficient of h[j] in S_d is B_d, and S_d(0) = A_d.

I can compute this by:
- Set h[j] = 0: S_d(0) = A_d
- Set h[j] = 1: S_d(1) = A_d + B_d
- So B_d = S_d(1) - S_d(0)

This is what I had before. Good.

Now, the 1D optimization: minimize max_d (A_d + t * B_d) over t ∈ [0,1].

This is a 1D linear programming problem (minimize the maximum of linear functions). The optimal t is at a point where the subgradient condition is met.

An efficient way to solve this:
1. Sort the indices by B_d (slope)
2. Process in order of decreasing slope
3. Maintain the "upper envelope" (convex hull of the lines)
4. The minimum is at the leftmost point of the upper envelope within [0,1]

But for simplicity, since we have at most 2n-1 ≈ 127 lines, we can just check all O(n²) intersection points plus the boundaries.

Let me code this up properly:

```python
def _find_opt_t(s0, deltas):
    """Find t in [0,1] that minimizes max_k (s0[k] + t * deltas[k])."""
    n_lines = len(s0)
    
    # Evaluate at boundaries
    best_t = 0.0
    best_val = s0.max()  # f(0)
    f1 = (s0 + deltas).max()  # f(1)
    if f1 < best_val:
        best_val = f1
        best_t = 1.0
    
    # Check intersections
    # For each pair (i, k) with deltas[i] > deltas[k]:
    #   Intersection at t = (s0[k] - s0[i]) / (deltas[i] - deltas[k])
    #   At this t, lines i and k have equal value
    #   We need this to be the maximum
    for i in range(n_lines):
        for k in range(n_lines):
            if i == k:
                continue
            dd = deltas[i] - deltas[k]
            if abs(dd) < 1e-14:
                continue
            t_star = (s0[k] - s0[i]) / dd
            if t_star < 0.0 or t_star > 1.0:
                continue
            # Check if at t_star, lines i and k are indeed the maximum
            vals = s0 + t_star * deltas
            if abs(vals[i] - vals[k]) < 1e-10 and vals.max() - vals[i] < 1e-10:
                if vals[i] < best_val:
                    best_val = vals[i]
                    best_t = t_star
    
    return best_t, best_val
```

This is O(n²) per coordinate, which for n_scores = 2*64-1 = 127 is about 16129 checks per coordinate, and 64 coordinates, so about 1 million checks per pass. Each check is O(n) to verify it's the max, so total is about 130 million operations per pass. That might be a bit slow for Python.

Let me optimize: instead of checking all pairs, I can:
1. Compute s0 and deltas
2. Sort by deltas
3. Use the fact that the upper envelope changes at most n times

Actually, a simpler optimization: instead of O(n²) pair checks, I can:
1. Evaluate f(t) at a set of candidate t values
2. The candidates are: 0, 1, and the intersection of the top 2 lines at each "regime change"

Or even simpler: since the function is convex and piecewise linear with at most n "kinks", I can find the minimum by:
1. Binary search on the subgradient (find where the subgradient changes sign)
2. The subgradient at t is the set of slopes of the active (maximizing) lines
3. We want the subgradient to contain 0

This is O(n log n) per step with binary search.

But for the scale of this problem (n=64, 127 lines), the O(n²) approach is fine. Let me just make it efficient with numpy:

```python
def _find_opt_t(s0, deltas):
    n = len(s0)
    best_t = 0.0
    best_val = float(s0.max())
    f1 = float((s0 + deltas).max())
    if f1 < best_val:
        best_val = f1
        best_t = 1.0
    
    # Vectorized: compute all intersection points
    # For pairs (i,k): t = (s0[k] - s0[i]) / (deltas[i] - deltas[k])
    # Use broadcasting: s0[k] - s0[i] for all i,k -> shape (n, n)
    # deltas[i] - deltas[k] for all i,k -> shape (n, n) [transposed]
    
    # Actually, let's just do it in a loop but with numpy for the verification
    s0_diff = np.subtract.outer(s0, s0)  # s0_diff[i,k] = s0[i] - s0[k]
    d_diff = np.subtract.outer(deltas, deltas)  # d_diff[i,k] = deltas[i] - deltas[k]
    
    # t[i,k] = s0_diff[k,i] / d_diff[i,k] = (s0[k]-s0[i]) / (deltas[i]-deltas[k])
    # = -s0_diff[i,k] / d_diff[i,k]
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = -s0_diff / d_diff
    
    # Valid where |d_diff| > eps and 0 <= t <= 1
    valid = (np.abs(d_diff) > 1e-12) & (t_mat >= 0) & (t_mat <= 1)
    
    # For each valid (i,k), check if it's the max
    # vals = s0 + t * deltas, check if vals[i] == vals[k] == vals.max()
    # This is expensive to do for all pairs...
    
    # Alternative: just evaluate f(t) at all valid t values
    t_candidates = np.unique(t_mat[valid])
    for t in t_candidates:
        val = float((s0 + t * deltas).max())
        if val < best_val:
            best_val = val
            best_t = float(t)
    
    return best_t, best_val
```

This is much faster! The `t_mat` computation is O(n²) but done in C via numpy. Then we just evaluate f(t) at each unique candidate.

The number of unique candidates is at most n²/(n-1) ≈ n (on average, many intersections coincide or are outside [0,1]). So this is very fast.

Actually, the number of unique values in t_mat[valid] can be up to n² in the worst case, but in practice it's much less. For n=127, the worst case is about 16000 unique values, and evaluating f(t) at each is O(n), so total is about 2 million operations. Fine.
