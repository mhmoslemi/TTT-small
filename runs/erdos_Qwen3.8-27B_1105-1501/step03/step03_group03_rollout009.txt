def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] -= h[d:]
                G[r, d:] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += 1.0 - h[g:]
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    """Project v onto {x : 0 <= x <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgrad_nesterov(h, target, deadline, lr_init=0.01, momentum=0.9, K=10):
    """Nesterov-accelerated subgradient method."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    T = 0.01  # temperature for simulated annealing
    
    while time.time() < deadline:
        # Nesterov lookahead
        y = h + momentum * v
        
        # Compute subgradient at y
        scores = _scores(y)
        G = _grad_scores(y)
        s_max = scores.max()
        # Active set: scores within 0.05 of max
        active = scores >= s_max - 0.05 * max(s_max, 1e-6)
        if not np.any(active):
            active = scores >= s_max - 1e-6
        
        # Average gradient over active set
        grad = G[active].mean(axis=0)
        grad -= np.mean(grad)  # center to help with sum constraint
        
        # Nesterov update
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)
        
        delta = c5_new - c5_cur
        accept = delta < 0

            if c5_new < best_c5:

                best_h = h_new.copy()
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.8
            v *= 0.5
        
        T *= 0.999
        if lr < 1e-8:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_scores(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] -= h[d:]
                G[r, d:] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += 1.0 - h[g:]
                G[r, g:] -= h[:m]
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _lp_optimize(h, target, deadline, radius_init=0.08):

    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

        if stall > 80:

            stall = 0
        
        scores = _scores(h)
        G = _grad_scores(h)
        s_max = float(np.max(scores))

        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []

            lo = max(-radius, -h[i])
            hi = min(radius, 1.0 - h[i])

        bounds.append((s_max - 1.0, s_max + 1.0))

def _subgrad_nesterov(h, target, deadline, lr_init=0.008, momentum=0.85, K_width=0.08):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    temp = 0.005
    rng = np.random.default_rng(123)
    
    while time.time() < deadline:
        y = h + momentum * v
        scores = _scores(y)
        G = _grad_scores(y)
        s_max = float(np.max(scores))
        
        # Active set
        width = K_width * max(s_max, 1.0)
        active = np.where(scores >= s_max - width)[0]
        if len(active) == 0:
            active = np.where(scores == s_max)[0]
        
        grad = G[active].mean(axis=0)
        grad -= np.mean(grad)
        
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)
        
        delta = c5_new - c5_cur
        accept = delta < -1e-14
        if not accept and temp > 1e-8:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        
        if accept:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            lr = min(lr * 1.005, 0.15)
        else:
            lr *= 0.75
            v *= 0.3
        
        temp *= 0.999
        if lr < 1e-8:
            break
    
    return best_h, best_c5


def _slsqp_epigraph(h, target, K=40, max_iter=80):
    n = len(h)
    scores = _scores(h)
    s_max = float(np.max(scores))
    x0 = np.concatenate([h, [s_max + 0.05]])
    
    # Get top K indices
    top_k = np.argsort(scores)[-K:]
    
    def func(x):
        return x[-1]
    
    def jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constr(x):
        h_p = x[:n]
        tau = x[-1]
        s = _scores(h_p)
        return tau - s[top_k]
    
    def constr_jac(x):
        h_p = x[:n]
        G = _grad_scores(h_p)
        J = np.zeros((K, n + 1))
        J[:, :n] = G[top_k]
        J[:, n] = -1.0
        return J
    
    def sum_fun(x):
        return np.sum(x[:n]) - target
    
    def sum_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, s_max + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac},
        {'type': 'ineq', 'fun': constr, 'jac': constr_jac}
    ]
    
    try:
        res = minimize(func, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and len(res.x) == n + 1:
            h_new = _project(res.x[:n], target)
            if _exact_c5(h_new) < _exact_c5(h):
                return h_new
    except Exception:
        pass
    return h.copy()


def _random_perturb(h, target, rng, scale):
    n = len(h)
    d = rng.standard_normal(n)
    d -= np.mean(d)
    d /= np.sqrt(np.sum(d**2))
    h_new = _project(h + scale * d, target)
    return h_new
