    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.02, 0.04, 0.08, 0.12, 0.2, 0.3]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for delta in deltas:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < -1e-10 or h[j] + d > 1.0 + 1e-10:
                        continue
                    comp_mask = np.ones(n, dtype=bool)
                    comp_mask[j] = False
                    if d > 0:
                        comp = int(np.argmax(np.where(comp_mask, h, -1)))
                    else:
                        comp = int(np.argmax(np.where(comp_mask, 1.0 - h, -1)))
                    if comp < 0 or h[comp] + (-d) < -1e-10 or h[comp] + (-d) > 1.0 + 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    h_try[comp] -= d
                    if np.any(h_try < -1e-10) or np.any(h_try > 1.0 + 1e-10):
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _soft_descent(h, deadline, beta_start=30.0, beta_end=20000.0, max_iters=15000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(scores.max())
        shifted = beta * (scores - m)
        shifted = np.clip(shifted, -500.0, 0.0)
        exps = np.exp(shifted)
        s_exp = float(np.sum(exps))
        if s_exp < 1e-300:
            w = np.zeros_like(exps)
            w[np.argmax(exps)] = 1.0
        else:
            w = exps / s_exp
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        direction = -grad / gnorm
        improved = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                improved = True
                lr = min(lr * 1.04, 0.3)
                break
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-6 or stall > 70:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: SLSQP with increasing beta (main optimization)
    deadline1 = t0 + budget_s * 0.45
    betas = [20.0, 80.0, 300.0, 1000.0, 5000.0, 15000.0, 50000.0]
    for beta in betas:
        if time.time() > deadline1:
            break
        remaining = deadline1 - time.time()
        if remaining < 5:
            break
        h_new = _slsqp_optimize(h, beta, maxiter=600)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    # Phase 2: Soft descent to further refine
    deadline2 = t0 + budget_s * 0.60
    h, c5 = _soft_descent(best_h, deadline2, beta_start=5000.0, beta_end=200000.0, max_iters=10000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Subgradient refinement on active set
    deadline3 = t0 + budget_s * 0.72
    h, c5 = _subgradient_refine(best_h, deadline3)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Pairwise coordinate descent
    deadline4 = t0 + budget_s * 0.85
    h_cd, c5_cd = _pairwise_coord(best_h, deadline4, max_passes=300)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 5: Resample to nearby n and optimize
    deadline5 = t0 + budget_s * 0.95
    for dn in [2, -2, 4, -4, 6, -6]:
        if time.time() > deadline5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        h_r = _slsqp_optimize(h_r, 5000.0, maxiter=400)
        h_r = _project(h_r, n_try / 2.0)
        c5_r = _exact_c5(h_r)
        # Further refine with soft descent
        remaining5 = deadline5 - time.time()
        if remaining5 > 3 and c5_r < best_c5 + 0.002:
            h_r, c5_r = _soft_descent(h_r, min(deadline5, time.time() + 15),
                                       beta_start=3000.0, beta_end=50000.0, max_iters=3000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final SLSQP polish on best solution
    if time.time() < t0 + budget_s * 0.985:
        h_f = _slsqp_optimize(best_h, 50000.0, maxiter=300)
        h_f = _project(h_f, best_n / 2.0)
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation and cleanup
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, best_n
