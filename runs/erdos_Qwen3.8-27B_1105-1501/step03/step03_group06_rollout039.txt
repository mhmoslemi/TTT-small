DC iteration:
1. Given h^(t), decompose g_k(h) = g_k^c(h) + g_k^v(h) where g_k^c is concave and g_k^v is convex.
2. Linearize g_k^c around h^(t): g_k^c(h) ≈ g_k^c(h^(t)) + ∇g_k^c(h^(t))^T (h - h^(t))
3. Solve: minimize max_k [g_k^v(h) + g_k^c(h^(t)) + ∇g_k^c(h^(t))^T (h - h^(t))]
   subject to 0 <= h <= 1, sum(h) = n/2
4. This is a convex problem (min of max of convex functions) → can be written as QP

The QP is:
minimize T
subject to: (1/4)||D_k h||^2 + (l_k + ∇c_k(h^(t)))^T h - c_k(h^(t)) - ∇c_k(h^(t))^T h^(t) <= T  for all k
0 <= h <= 1, sum(h) = n/2

where D_k h is the vector of differences h[i+k] - h[i] for valid i.

This QP has 2n-1 quadratic constraints. For n=64, that's 127 QCQPs (quadratically constrained QP).

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _grad_row(h, lag):
    n = len(h)
    d = lag - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:])
            grad[g:] += -h[:m]
    return grad


def _grad_rows(h, lags):
    return np.array([_grad_row(h, lag) for lag in lags])


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _resample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_step(h, active_lags, maxiter=300):
    n = len(h)
    target = n / 2.0
    T0 = float(np.max(_corr(h)))

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    cons = []
    for lag in active_lags:
        def cfun(x, lag=lag):
            return x[0] - _corr(x[1:])[lag]
        def cgrad(x, lag=lag):
            g = np.zeros(n + 1)
            g[0] = 1.0
            g[1:] = -_grad_row(x[1:], lag)
            return g
        cons.append({'type': 'ineq', 'fun': cfun, 'jac': cgrad})

    def eqfun(x):
        return float(np.sum(x[1:])) - target

    def eqgrad(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eqfun, 'jac': eqgrad})

    bounds = [(T0 * 0.3, None)] + [(0.0, 1.0)] * n
    x0 = np.concatenate([[T0], h])

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x[0] < T0 - 1e-12:
            h_new = _project(res.x[1:], target)
            c5_new = _c5(h_new)
            if c5_new < _c5(h):
                return h_new, c5_new
    except Exception:
        pass
    return h.copy(), _c5(h)

    c5 = _c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase D: Resolution search (25% budget)
    dD = t0 + budget_s * 0.90
    deltas = [-8, -4, -2, -1, 1, 2, 4, 8]
    for dn in deltas:
        if time.time() > dD:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(dD, time.time() + budget_s * 0.03)
        if HAS_CP:
            h_r, c5_r = _dc_loop(h_r, sub_d, max_iters=8, n_active=10)
        if time.time() < dD:
            sub_d2 = min(dD, time.time() + budget_s * 0.02)
            h_r, c5_r = _multi_slsqp(h_r, sub_d2, max_rounds=5)
        c5_r = _c5(h_r)
        if c5_r < best_c5 - 1e-8:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase E: Final SLSQP + DC polish (7% budget)
    dE = t0 + budget_s * 0.97
    if HAS_CP:
        h_f, c5_f = _dc_loop(best_h, min(dE, time.time() + budget_s * 0.03),
                              max_iters=10, n_active=10)
        c5_f = _c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if time.time() < dE:
        h_f, c5_f = _multi_slsqp(best_h, dE, max_rounds=8)
        c5_f = _c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase F: Pair polish (3% budget)
    if time.time() < t0 + budget_s * 0.99:
        h_p, c5_p = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=20)
        c5_p = _c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
