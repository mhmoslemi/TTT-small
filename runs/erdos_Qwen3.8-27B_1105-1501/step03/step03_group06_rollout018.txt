def _slsqp_epigraph(h_init, deadline, maxiter=10000):
    n = len(h_init)
    target = n / 2.0
    
    # x = [h_0, ..., h_{n-1}, t]
    x0 = np.append(h_init, _exact_c5(h_init) + 0.1)
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def equality_constraint(x):
        return np.sum(x[:n]) - target
    
    def equality_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Inequality constraints: t - corr_r(h) >= 0 for each r
    # We'll define them as a vector
    def ineq_constraints(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr  # should all be >= 0
    
    def ineq_jac(x):
        h = x[:n]
        # Compute grad matrix
        G = _grad_matrix(h)  # shape (2n-1, n), G[r, i] = d(corr_r)/d(h_i)
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': equality_constraint, 'jac': equality_jac},
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'iprint': 0})
    
    if res.success:
        h_opt = np.clip(res.x[:n], 0, 1)
        h_opt = _project(h_opt, target)
        return h_opt, _exact_c5(h_opt)
    return h_init, _exact_c5(h_init)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Residual repair
    residual = float(np.sum(x)) - target
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(x)[::-1]
            for idx in order:
                if residual <= 0:
                    break
                reducible = x[idx]
                amt = min(reducible, residual)
                x[idx] -= amt
                residual -= amt
        else:
            order = np.argsort(x)
            for idx in order:
                if residual >= 0:
                    break
                increasable = 1.0 - x[idx]
                amt = min(increasable, -residual)
                x[idx] += amt
                residual += amt
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += -h[g:]
                G[r, g:] += (1.0 - h[:m])
    return G


def _smooth_fun(x, alpha, target):
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha, n / 2.0),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, n / 2.0)
    return x, _exact_c5(x)


def _slsqp_solve(h_init, deadline, maxiter=3000):
    n = len(h_init)
    target = n / 2.0
    init_c5 = _exact_c5(h_init)
    x0 = np.append(h_init.copy(), init_c5 + 0.05)

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return float(np.sum(x[:n]) - target)

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr

    def ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'iprint': 0, 'ftol': 1e-16})
        if time.time() < deadline:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = _project(h_opt, target)
            c5 = _exact_c5(h_opt)
            if c5 < init_c5:
                return h_opt, c5
    except Exception:
        pass
    return h_init.copy(), init_c5


def _line_search_c5(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 16):
        a = t * alpha_max
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, n / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.2)
        hi = min(alpha_max, best_a + alpha_max * 0.2)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
        for _ in range(18):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(_project(h + a * p, n / 2.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t

    return best_a, best_c5


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.15, max_iters=5000, stall_limit=80):

    h = h.copy()

def _binary_sa(h_init, deadline, rng, max_iters=2000000):
    n = len(h_init)
    target = n // 2
    b = (h_init > 0.5).astype(float)
    if int(np.sum(b)) != target:
        idx = np.where(b > 0.5)[0]
        idx0 = np.where(b < 0.5)[0]
        need = target - int(np.sum(b))
        if need > 0:
            for i in range(min(need, len(idx0))):
                b[idx0[i]] = 1.0
        elif need < 0:
            for i in range(min(-need, len(idx))):
                b[idx[i]] = 0.0

    def _binary_max_corr(bv):
        corr = np.correlate(bv, 1.0 - bv, mode='full')
        return float(np.max(corr))

    cur_val = _binary_max_corr(b)
    best_b = b.copy()
    best_val = cur_val
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]

    T = 2.0
    T_end = 0.01
    iters_per_temp = max(1, max_iters // 20)

    for temp_i in range(20):
        T = T * (T_end / T) ** (1.0 / 20.0)
        for _ in range(iters_per_temp):
            if time.time() > deadline:
                return _project(best_b, n / 2.0), _exact_c5(_project(best_b, n / 2.0))
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            b[i], b[j] = 0.0, 1.0
            new_val = _binary_max_corr(b)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_val = new_val
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
                if cur_val < best_val:
                    best_val = cur_val
                    best_b = b.copy()
            else:
                b[i], b[j] = 1.0, 0.0

    h_binary = _project(best_b, n / 2.0)
    return h_binary, _exact_c5(h_binary)

        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) / (n_starts - i))
        h_try, c5_try = _slsqp_solve(h_pert, sub_deadline, maxiter=1000)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick smooth optimization (12% budget)
    d1 = t0 + budget_s * 0.12
    for alpha in [10.0, 30.0, 60.0]:
        if time.time() > d1:
            break
        h, c5 = _lbfgs_step(h, alpha, maxiter=150)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP epigraph (20% budget)
    d2 = t0 + budget_s * 0.32
    h_sls, c5_sls = _slsqp_solve(best_h, d2, maxiter=5000)
    if c5_sls < best_c5:
        best_c5, best_h = c5_sls, h_sls.copy()

    # Multi-start SLSQP
    if time.time() < d2:
        h_ms, c5_ms = _multi_start_slsqp(best_h, d2, rng, n_starts=5)
        if c5_ms < best_c5:
            best_c5, best_h = c5_ms, h_ms.copy()

    # Phase 3: Active-set LP polishing (20% budget)
    d3 = t0 + budget_s * 0.52
    h_lp, c5_lp = _active_set_lp(best_h, d3, radius_init=0.12, max_iters=4000, k_active=15)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Full LP polish
    if time.time() < d3:
        h_lp2, c5_lp2 = _lp_loop(best_h, d3, radius_init=0.1, max_iters=3000, stall_limit=60)
        if c5_lp2 < best_c5:
            best_c5, best_h = c5_lp2, h_lp2.copy()

    # Phase 4: Binary SA (10% budget)
    d4 = t0 + budget_s * 0.62
    h_bin, c5_bin = _binary_sa(best_h, d4, rng, max_iters=1000000)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin.copy()
        # Refine the binary solution with SLSQP + LP
        if time.time() < d4:
            h_br, c5_br = _slsqp_solve(best_h, d4, maxiter=2000)
            if c5_br < best_c5:
                best_c5, best_h = c5_br, h_br.copy()

    # Phase 5: Multi-resolution search (18% budget)
    d5 = t0 + budget_s * 0.80
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 16, -16]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.02)
        for alpha in [20.0, 50.0]:
            if time.time() > sub_d:
                break
            h_r, c5_r = _lbfgs_step(h_r, alpha, maxiter=80)
        if time.time() < d5:
            sub_d2 = min(d5, time.time() + budget_s * 0.015)
            h_r, c5_r = _slsqp_solve(h_r, sub_d2, maxiter=1500)
        if time.time() < d5:
            sub_d3 = min(d5, time.time() + budget_s * 0.015)
            h_r, c5_r = _active_set_lp(h_r, sub_d3, radius_init=0.08, max_iters=1000, k_active=12)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Final intensive LP (12% budget)
    d6 = t0 + budget_s * 0.95
    h_f, c5_f = _lp_loop(best_h, d6, radius_init=0.06, max_iters=5000, stall_limit=100)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
        best_n = len(best_h)

    # Final SLSQP polish
    if time.time() < t0 + budget_s * 0.97:
        h_fs, c5_fs = _slsqp_solve(best_h, t0 + budget_s * 0.97, maxiter=2000)
        if c5_fs < best_c5:
            best_c5, best_h = c5_fs, h_fs.copy()
            best_n = len(best_h)

    # Final LP
    if time.time() < t0 + budget_s * 0.99:
        h_fl, c5_fl = _active_set_lp(best_h, t0 + budget_s * 0.995,
                                     radius_init=0.03, max_iters=2000, k_active=10)
        if c5_fl < best_c5:
            best_c5, best_h = c5_fl, h_fl.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
