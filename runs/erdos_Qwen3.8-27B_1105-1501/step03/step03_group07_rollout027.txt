import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
    else:
        k = -s
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    return grad

def _lp_direction(h, radius, lam=0.0, top_k=None):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    n_lags = 2 * n - 1
    
    if top_k is not None and top_k < n_lags:
        sorted_raw = np.sort(raw)[::-1]
        thresh = sorted_raw[min(top_k, n_lags) - 1]
        active = np.where(raw >= thresh - 1e-12)[0]
    else:
        active = np.arange(n_lags)
    
    n_active = len(active)
    n_vars = 1 + 2 * n
    n_ub = n_active + 1  # lag constraints + norm
    
    A_ub = np.zeros((n_ub, n_vars))
    b_ub = np.zeros(n_ub)
    
    for idx, r in enumerate(active):
        s = int(r) - (n - 1)
        g = _grad_lag(h, s)
        A_ub[idx, 0] = -1.0
        A_ub[idx, 1:1+n] = g
        A_ub[idx, 1+n:1+2*n] = -g
        b_ub[idx] = -raw[r]
    
    # Norm: sum(p) + sum(q) <= radius
    A_ub[n_active, 1:1+n] = 1.0
    A_ub[n_active, 1+n:1+2*n] = 1.0
    b_ub[n_active] = radius
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(-10000, 10000)]
    for j in range(n):
        if h[j] <= 1e-8:
            bounds.append((0, None))  # p_j
            bounds.append((0, 0))     # q_j = 0
        elif h[j] >= 1 - 1e-8:
            bounds.append((0, 0))     # p_j = 0
            bounds.append((0, None))  # q_j
        else:
            bounds.append((0, None))
            bounds.append((0, None))
    
    c = np.zeros(n_vars)
    c[0] = 1.0
    if lam > 0:
        c[1:1+n] = lam
        c[1+n:1+2*n] = lam
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    p = res.x[1:1+n]
    q = res.x[1+n:1+2*n]
    d = p - q
    d[np.abs(d) < 1e-14] = 0.0
    d -= np.mean(d)
    
    if float(np.max(np.abs(d))) < 1e-14:
        return None
    
    return d

def _line_search(h, d, target, c5_cur):
    n = len(h)
    alpha_max = 5.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:

    best_c = c5_cur

    for a in alphas:
        h_t = _project(h + a * d, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section
    lo = max(0.0, best_a * 0.6)
    hi = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))
    
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)

        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * d, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
    else:
        k = -s
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    return grad

def _lp_direction(h, radius, lam=0.0, top_k=None, single_worst=False):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    n_lags = 2 * n - 1

    if single_worst:
        active = np.array([int(np.argmax(raw))])
    elif top_k is not None and top_k < n_lags:
        sorted_raw = np.sort(raw)[::-1]
        thresh = float(sorted_raw[min(top_k, n_lags) - 1])
        active = np.where(raw >= thresh - 1e-12)[0]
    else:
        active = np.arange(n_lags)

    n_active = len(active)
    n_vars = 1 + 2 * n
    n_ub = n_active + 1

    A_ub = np.zeros((n_ub, n_vars))
    b_ub = np.zeros(n_ub)

    for idx, r in enumerate(active):
        s = int(r) - (n - 1)
        g = _grad_lag(h, s)
        A_ub[idx, 0] = -1.0
        A_ub[idx, 1:1 + n] = g
        A_ub[idx, 1 + n:1 + 2 * n] = -g
        b_ub[idx] = -raw[r]

    A_ub[n_active, 1:1 + n] = 1.0
    A_ub[n_active, 1 + n:1 + 2 * n] = 1.0
    b_ub[n_active] = radius

    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:1 + 2 * n] = -1.0
    b_eq = np.array([0.0])

    bounds = [(-10000.0, 10000.0)]
    for j in range(n):
        if h[j] <= 1e-10:
            bounds.append((0, None))
            bounds.append((0, 0))
        elif h[j] >= 1.0 - 1e-10:
            bounds.append((0, 0))
            bounds.append((0, None))
        else:
            bounds.append((0, None))
            bounds.append((0, None))

    c = np.zeros(n_vars)
    c[0] = 1.0
    if lam > 0:
        c[1:1 + n] = lam
        c[1 + n:1 + 2 * n] = lam

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    p = np.asarray(res.x[1:1 + n], dtype=float)
    q = np.asarray(res.x[1 + n:1 + 2 * n], dtype=float)
    d = p - q
    d[np.abs(d) < 1e-14] = 0.0
    d -= np.mean(d)
    if float(np.max(np.abs(d))) < 1e-14:
        return None
    return d

def _line_search(h, d, target, c5_cur, max_alpha=3.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-6)
    alpha_max = min(alpha_max, max_alpha)

    best_a = 0.0
    best_c = c5_cur

    alphas = np.linspace(0.05, alpha_max, 10)
    for a in alphas:
        h_t = _project(h + a * d, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c:
            best_c = c_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo = max(0.0, best_a * 0.65)
    hi = min(alpha_max, best_a * 1.35)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))

    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))

    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * d, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _lp_optimize(h, deadline, target, max_stall=50):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    radii = [0.15 * n, 0.4 * n, 0.8 * n, 1.5 * n, 3.0 * n]
    lams = [0.0, 0.1, 1.0]
    mode_cycle = ['full', 'top12', 'top30', 'worst', 'full', 'top48']
    mode_cycle_idx = 0
    radii_idx = 0

    stall = 0
    rng = np.random.default_rng(12345)

    while time.time() < deadline and stall < max_stall:
        improved = False
        R = radii[radii_idx % len(radii)]
        mode = mode_cycle[mode_cycle_idx % len(mode_cycle)]

        if mode == 'full':
            tk = None
            sw = False
        elif mode == 'worst':
            tk = None
            sw = True
        else:
            tk = int(mode[3:])
            sw = False

        for lam in lams:
            if time.time() > deadline:
                return best_h, best_c5
            d = _lp_direction(h, radius=R, lam=lam, top_k=tk, single_worst=sw)
            if d is None:
                continue
            for sign in [1.0, -1.0]:
                if time.time() > deadline:
                    return best_h, best_c5
                a, c_t = _line_search(h, sign * d, target, cur_c5, max_alpha=2.0)
                if a > 0 and c_t < cur_c5 - 1e-13:
                    h = _project(h + a * sign * d, target)
                    cur_c5 = _exact_c5(h)
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()

        if improved:
            stall = 0
            radii_idx = (radii_idx + 1) % len(radii)
        else:
            stall += 1
            mode_cycle_idx = (mode_cycle_idx + 1) % len(mode_cycle)
            radii_idx = (radii_idx + 1) % len(radii)
            if stall % 8 == 0:
                h_try = _project(h + rng.normal(0, 0.06, n), target)
                c_try = _exact_c5(h_try)
                if c_try < best_c5:
                    h = h_try
                    cur_c5 = c_try
                else:
                    h = best_h.copy()
                    cur_c5 = best_c5

    return best_h, best_c5

def _push_binary(h, target, rng):
    n = len(h)
    h_int = np.floor(target).astype(int)
    h_frac = target - h_int
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:h_int]] = 1.0
    if h_frac > 1e-10 and h_int < n:
        h_bin[order[h_int]] = h_frac
    h_bin = _project(h_bin + rng.normal(0, 0.01, n), target)
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_lp(h0, deadline, n_starts=20, seed=42):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    rng = np.random.default_rng(seed)

    for s in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining * 0.8 / max(1, n_starts - s), 30.0)
        sub_dl = time.time() + alloc

        r = rng.random()
        if r < 0.25:
            h_try = h0 + rng.normal(0, 0.05, n)
        elif r < 0.4:
            h_try = h0.copy()
            idx = rng.choice(n, size=max(1, n // 6), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.55:
            h_try = _push_binary(h0, target, rng)
        elif r < 0.7:
            shift = rng.integers(1, max(2, n // 4))
            h_try = np.roll(h0, int(shift))
        elif r < 0.85:
            h_try = 0.5 + (h0 - 0.5) * (0.7 + 0.6 * rng.random())
        else:
            h_try = 0.5 * h0 + 0.5 * (rng.random(n) > 0.5).astype(float)

        h_try = _project(h_try, target)
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, target, max_stall=25)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5

def _pair_polish(h, deadline, target, max_passes=30):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        raw = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(raw))
        active = np.where(raw >= max_s - 0.01 * max_s)[0]

        for r in active[:8]:
            s = int(r) - (n - 1)
            g = _grad_lag(h, s)
            g -= np.mean(g)
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                continue
            g /= gn
            a, c_t = _line_search(h, -g, target, cur_c5, max_alpha=0.5)
            if a > 0 and c_t < best_c5 - 1e-13:
                h = _project(h + a * (-g), target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True

        if not improved:
            for _ in range(150):
                if time.time() > deadline:
                    return best_h, best_c5
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j == k or h[j] <= 1e-7 or h[j] >= 1 - 1e-7 or h[k] <= 1e-7 or h[k] >= 1 - 1e-7:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                    a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                    a_max = min(a_max_j, a_max_k)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
            if not improved:
                break

    return best_h, best_c5
