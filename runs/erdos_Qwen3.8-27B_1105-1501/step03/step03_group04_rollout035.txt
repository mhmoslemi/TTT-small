            num = num // int(base)
            f /= base

def _opt_t_max_linear(D, A, t_lo, t_hi):
    """Find t in [t_lo, t_hi] that minimizes max_k (D[k] + A[k] * t)
    D, A are arrays of length m.
    Returns optimal t and the minimum value.
    """
    m = len(D)
    # Collect candidate t values: boundaries + all crossing points in range
    candidates = [t_lo, t_hi]
    for i in range(m):
        for j in range(i+1, m):
            dA = A[i] - A[j]
            if abs(dA) < 1e-15:
                continue
            t_cross = (D[j] - D[i]) / dA
            if t_lo <= t_cross <= t_hi:
                candidates.append(t_cross)
    
    best_t = t_lo
    best_val = float('inf')
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        vals = D + A * t
        v = vals.max()
        if v < best_val:
            best_val = v
            best_t = t
    return best_t, best_val

def _overlaps_raw(h):
    """Returns raw overlap sums (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _coeffs_for_index(h, idx):
    """Compute c_k(idx) for all k: the coefficient of h[idx] in raw overlap k."""
    n = len(h)
    c = np.zeros(2*n-1)
    for k in range(2*n-1):
        d = k - (n-1)  # lag = d
        # h[idx] appears as h[i+k] when i = idx-k (in 0-based indexing, lag d means h[i+d])
        # In the correlate formulation:
        # For lag d >= 0: B_d = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        #   h[j] appears as h[i+d] when i+d=j, i=j-d: valid if 0<=j-d and j-d<=n-1-d, i.e., d<=j<=n-1
        #     contribution: (1-h[j-d])
        #   h[j] appears as h[i] when i=j: valid if 0<=j<=n-1-d
        #     contribution: -h[j+d]
        # For lag d < 0 (g = -d): B_{-g} = sum_{i=g}^{n-1} h[i-g]*(1-h[i])
        #   h[j] appears as h[i-g] when i-g=j, i=j+g: valid if g<=j+g<=n-1, i.e., 0<=j<=n-1-g
        #     contribution: (1-h[j+g])
        #   h[j] appears as h[i] when i=j: valid if g<=j<=n-1
        #     contribution: -h[j-g]
        
        if d >= 0:
            if d <= idx <= n-1:
                c[k] += (1.0 - h[idx - d])
            if 0 <= idx <= n-1-d:
                c[k] -= h[idx + d]
        else:
            g = -d
            if 0 <= idx <= n-1-g:
                c[k] += (1.0 - h[idx + g])
            if g <= idx <= n-1:
                c[k] -= h[idx - g]
    return c

Wait, this is O(n) per index and O(n*(2n-1)) for all indices. For n=64, that's 64*127 = 8128 operations. Very fast.

Actually, I can vectorize this. For a given index j, c_k(j) depends on:
- h[j-d] (or h[j+|d|]) and h[j+d] (or h[j-|d|])

Let me think of a vectorized way...

For d >= 0: c at row (n-1+d), col j:
  - (1-h[j-d]) if d <= j <= n-1
  - -h[j+d] if 0 <= j <= n-1-d

For d < 0, g = -d: c at row (n-1-d) = (n-1+g), col j:
  - (1-h[j+g]) if 0 <= j <= n-1-g
  - -h[j-g] if g <= j <= n-1

This is essentially a circulant-like structure but with boundaries. I'll just use the loop version since it's fast enough.

import numpy as np
import time
from scipy.special import softmax

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coeffs_all(h):
    """Compute c[k, j] = coefficient of h[j] in raw overlap k. Shape (2n-1, n)."""
    n = len(h)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                C[k, d:] += (1.0 - h[:m])
                C[k, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                C[k, :m] += (1.0 - h[g:])
                C[k, g:] -= h[:m]
    return C

def _soft_grad(h, beta):
    n = len(h)
    ov = _raw_overlaps(h)
    w = softmax(beta * (ov - ov.max()))
    C = _coeffs_all(h)
    return C.T @ w

def _opt_pair_t(D, A, t_lo, t_hi):
    """Minimize max_k(D[k] + A[k]*t) over t in [t_lo, t_hi]."""
    candidates = np.array([t_lo, t_hi])
    m = len(D)
    # Use only near-active lines for crossing points (speedup)
    t_mid = (t_lo + t_hi) / 2.0
    vals_mid = D + A * t_mid
    thr = vals_mid.max() - 0.5 * (vals_mid.max() - vals_mid.min())
    active = np.where(vals_mid >= thr)[0]
    if len(active) < 4:
        active = np.argsort(-vals_mid)[:8]
    for ii in range(len(active)):
        for jj in range(ii+1, len(active)):
            i, j = active[ii], active[jj]
            dA = A[i] - A[j]
            if abs(dA) < 1e-14:
                continue
            tc = (D[j] - D[i]) / dA
            if t_lo <= tc <= t_hi:
                candidates = np.append(candidates, tc)
    best_t = t_lo
    best_v = np.inf
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        v = np.max(D + A * t)
        if v < best_v:
            best_v = v
            best_t = t
    return best_t, best_v

def _pair_search(h, deadline, max_passes=50, tol=1e-12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        max_impr = 0.0
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[ii]
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for jj in range(ii+1, len(interior)):
                j = interior[jj]
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                C = _coeffs_all(h)
                ci = C[:, i].copy()
                cj = C[:, j].copy()
                A = ci - cj
                B = _raw_overlaps(h)
                D = B - h[i] * A
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                t_opt, v_opt = _opt_pair_t(D, A, t_lo, t_hi)
                if abs(t_opt - h[i]) < 1e-10:
                    continue
                h_try = h.copy()
                h_try[i] = t_opt
                h_try[j] = s - t_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - tol:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    max_impr = max(max_impr, improved_diff if (improved_diff := True) else False)
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_search_full(h, deadline, max_passes=20):
    """Pair search over ALL pairs, not just interior."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                C = _coeffs_all(h)
                ci = C[:, i]
                cj = C[:, j]
                A = ci - cj
                B = _raw_overlaps(h)
                D = B - h[i] * A
                t_opt, v_opt = _opt_pair_t(D, A, t_lo, t_hi)
                if abs(t_opt - h[i]) < 1e-10:
                    continue
                h_try = h.copy()
                h_try[i] = t_opt
                h_try[j] = s - t_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _subgrad(h, deadline, beta=200.0, lr_init=0.03, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = _soft_grad(h, beta)
        g -= g.mean()
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        h_new = h - lr * g
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.7
            if lr < 1e-6:
                beta = min(beta * 1.5, 5000.0)
                lr = lr_init * 0.3
    return best_h, best_c5

def _annealed_subgrad(h, deadline, beta_start=30.0, beta_end=3000.0, lr_init=0.05, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        g = _soft_grad(h, beta)
        g -= g.mean()
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        h_new = h - lr * g
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 0.2)
        else:
            stall += 1
            lr *= 0.75
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fft_init(n, rng):
    """Initialize with a sum of random Fourier modes."""
    x = np.arange(n) / n
    h = np.full(n, 0.5)
    for k in range(1, 6):
        amp = rng.uniform(0.02, 0.12) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    return _project(h, n / 2.0)

def _balanced_grid_init(n, rng):
    """Initialize with a structured pattern plus noise."""
    h = np.zeros(n)
    period = max(2, n // 8)
    for i in range(n):
        h[i] = 1.0 if (i % (2 * period)) < period else 0.0
    h += rng.normal(0, 0.15, n)
    return _project(h, n / 2.0)

def _optimize_h(h, deadline, aggressive=False):
    """Multi-stage local optimization starting from h."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    # Stage 1: Annealed subgradient
    dl1 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h1, c5_1 = _annealed_subgrad(best_h, dl1, beta_start=20, beta_end=2000, lr_init=0.04)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    # Stage 2: Targeted subgradient
    dl2 = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    h2, c5_2 = _subgrad(best_h, dl2, beta=300, lr_init=0.02)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
    # Stage 3: Pair search
    dl3 = deadline
    h3, c5_3 = _pair_search(best_h, dl3, max_passes=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
    # Stage 4: Another subgradient round
    if time.time() < deadline - 5:
        h4, c5_4 = _subgrad(best_h, deadline, beta=500, lr_init=0.015)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    # Final pair polish
    if time.time() < deadline - 2:
        h5, c5_5 = _pair_search(best_h, deadline, max_passes=10)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5
    return best_h, best_c5
