def _line_eval(h, D, t_arr):
    """Evaluate C5 at multiple t values using precomputed correlations."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full')
    Q1 = -np.correlate(h, D, mode='full')
    Q2 = -np.correlate(D, D, mode='full')

    # t_arr shape: (m,), S shape: (2n-1,)
    # Result: (m,)
    combined = B + Q1
    # For each t: values[k] = S[k] + t*combined[k] + t²*Q2[k]
    # max over k
    # Use broadcasting: (m, 2n-1)
    # But this might be too memory-heavy for large m and n
    # Instead, loop over t (or vectorize)
    results = np.zeros(len(t_arr))
    for i, t in enumerate(t_arr):
        vals = S + t * combined + t*t * Q2
        results[i] = float(np.max(vals) * dx)
    return results

    D = D - np.mean(D)
    max_abs = float(np.max(np.abs(D)))

        return None, None

        t_lo_neg = (1.0 - h[neg]) / D[neg]  # D[neg] < 0, so this is negative

        t_lo = max(t_lo, float(np.max(t_lo_neg)))

    left = max(0, best_idx - 2)
    right = min(len(t_grid) - 1, best_idx + 2)
    lo_t = t_grid[left]
    hi_t = t_grid[right]

    opt_f = _f(opt_t)
    if opt_f < best_f:
        best_t = opt_t
        best_f = opt_f
