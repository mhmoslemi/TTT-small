from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _max_alpha(h, p):
    n = len(h)
    alpha_max = 0.5
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    return max(alpha_max, 1e-12)

def _line_search(h, p):
    alpha_max = _max_alpha(h, p)
    n_samp = 25
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _pnorm_optimize(h, deadline, p_start=4.0, p_end=300.0, n_steps=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        eps = 1e-14
        s = scores + eps
        log_w = (p - 1.0) * np.log(s)
        log_w -= np.max(log_w)
        w = np.exp(log_w)
        w_sum = np.sum(w)
        if w_sum < 1e-100:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= w_sum
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        p_dir = -grad / gn  # normalize for consistent step size
        alpha_max = _max_alpha(h, p_dir)
        # Scale direction
        p_dir = p_dir * min(alpha_max, 0.1)  # cap the direction magnitude
        # Actually, let's not normalize - use the raw gradient direction
        p_dir = -grad
        alpha, c5_try = _line_search(h, p_dir)
        h_new = h + alpha * p_dir
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.1, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _binary_sa(h, deadline, n_steps=50000, T0=None, T_end=1e-5):
    """Simulated annealing on binary version of h."""
    n = len(h)
    # Create binary starting point: threshold at 0.5
    h_bin = (h >= 0.5).astype(float)
    # Ensure exact count
    target_count = n // 2
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    if len(ones) > target_count:
        # Remove excess ones
        excess = len(ones) - target_count
        # Remove ones with lowest h values
        to_remove = ones[np.argsort(h[ones])][:excess]
        h_bin[to_remove] = 0.0
    elif len(ones) < target_count:
        # Add ones
        deficit = target_count - len(ones)
        to_add = zeros[np.argsort(-h[zeros])][:deficit]
        h_bin[to_add] = 1.0
    
    if T0 is None:
        T0 = 0.01
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if step % 5000 == 0:
            T = T0 * (T_end / T0) ** (step / max(1, n_steps - 1))
        else:
            T = T0 * (T_end / T0) ** ((step + 1) / max(1, n_steps - 1))
        
        # Pick random swap
        pos_idx = np.where(h_bin == 1.0)[0]
        neg_idx = np.where(h_bin == 0.0)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        i = pos_idx[np.random.randint(len(pos_idx))]
        j = neg_idx[np.random.randint(len(neg_idx))]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = _exact_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _build_binary_seeds(n, rng, n_seeds=20):
    """Build diverse binary starting points."""
    seeds = []
    target = n // 2
    # Threshold-based
    for _ in range(n_seeds):
        bits = np.zeros(n)
        idx = rng.choice(n, size=target, replace=False)
        bits[idx] = 1.0
        seeds.append(bits)
    # Periodic patterns
    for period in [2, 3, 4, 5, 6, 7]:
        bits = np.zeros(n)
        for i in range(n):
            if (i % period) < period // 2:
                bits[i] = 1.0
        # Adjust to target count
        ones = int(np.sum(bits))
        if ones > target:
            excess = ones - target
            to_remove = np.where(bits == 1.0)[0]
            bits[to_remove[:excess]] = 0.0
        elif ones < target:
            deficit = target - ones
            to_add = np.where(bits == 0.0)[0]
            bits[to_add[:deficit]] = 1.0
        seeds.append(bits)
    return seeds

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _max_alpha(h, p):
    n = len(h)
    alpha_max = 0.5
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    return max(alpha_max, 1e-12)

def _line_search(h, p):
    alpha_max = _max_alpha(h, p)
    n_samp = 25
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _pnorm_optimize(h, deadline, p_start=4.0, p_end=300.0, n_steps=400,
                    use_momentum=False, momentum=0.6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    v = np.zeros(n) if use_momentum else None
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        eps = 1e-14
        s = np.maximum(scores, eps)
        log_w = (p - 1.0) * np.log(s)
        log_w -= np.max(log_w)
        w = np.exp(log_w)
        w_sum = np.sum(w)
        if w_sum < 1e-100:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= w_sum
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        if use_momentum:
            v = momentum * v - grad
            p_dir = -v
        else:
            p_dir = -grad
        alpha, c5_try = _line_search(h, p_dir)
        h_new = h + alpha * p_dir
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if use_momentum:
                v *= 0.5
        if stall > 50:
            break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.08, max_iters=4000):
    if top_k_list is None:
        top_k_list = [None, 12, 7, 4, 10]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.35)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 25 and radius < 0.015:
                radius = 0.06
    return best_h, best_c5

def _binary_sa(h, deadline, rng, n_steps=50000, T0=0.008, T_end=1e-5):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target_count = n // 2
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    if len(ones) > target_count:
        excess = len(ones) - target_count
        to_remove = ones[np.argsort(h[ones])][:excess]
        h_bin[to_remove] = 0.0
    elif len(ones) < target_count:
        deficit = target_count - len(ones)
        to_add = zeros[np.argsort(-h[zeros])][:deficit]
        h_bin[to_add] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (step / max(1, n_steps - 1))
        pos_idx = np.where(h_bin == 1.0)[0]
        neg_idx = np.where(h_bin == 0.0)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        i = pos_idx[int(rng.integers(len(pos_idx)))]
        j = neg_idx[int(rng.integers(len(neg_idx)))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = _exact_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish_top(h, deadline, max_passes=6, top_pairs=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        inner = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(inner) < 2:
            break
        n_inner = len(inner)
        n_pairs = min(top_pairs, n_inner * (n_inner - 1) // 2)
        rng_local = np.random.default_rng(_pass * 1000 + 42)
        for pi in range(n_pairs):
            if time.time() > deadline:
                break
            if _pass == 0 and pi < n_inner * (n_inner - 1) // 2 and pi < top_pairs:
                j_idx = pi // max(1, n_inner - 1 - pi // n_inner)
                k_idx = pi % n_inner
                if j_idx >= k_idx:
                    j_idx, k_idx = k_idx, j_idx
                if j_idx >= n_inner or k_idx >= n_inner:
                    continue
                j = int(inner[j_idx])
                k = int(inner[k_idx])
            else:
                j = int(inner[rng_local.integers(n_inner)])
                k = int(inner[rng_local.integers(n_inner)])
                if j == k:
                    continue
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[j], h[k])
                else:
                    a_max = min(h[j], 1.0 - h[k])
                if a_max < 1e-10:
                    continue
                lo, hi = 0.0, a_max
                for _ in range(12):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    h1 = h.copy()
                    h1[j] += sign * m1
                    h1[k] -= sign * m1
                    f1 = _exact_c5(h1)
                    h2 = h.copy()
                    h2[j] += sign * m2
                    h2[k] -= sign * m2
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                a_opt = (lo + hi) / 2.0
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _build_binary_seeds(n, rng, n_seeds=15):
    seeds = []
    target = n // 2
    for _ in range(n_seeds):
        bits = np.zeros(n)
        idx = rng.choice(n, size=target, replace=False)
        bits[idx] = 1.0
        seeds.append(bits)
    for period in [2, 3, 4, 5, 6, 7, 8]:
        bits = np.zeros(n)
        for i in range(n):
            if (i % period) < period // 2:
                bits[i] = 1.0
        ones = int(np.sum(bits))
        if ones > target:
            to_remove = np.where(bits == 1.0)[0]
            bits[to_remove[:ones - target]] = 0.0
        elif ones < target:
            to_add = np.where(bits == 0.0)[0]
            bits[to_add[:target - ones]] = 1.0
        seeds.append(bits)
    return seeds

def _adaptive_pnorm(h, deadline, max_rounds=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    p_current = 6.0
    stall_total = 0
    for rnd in range(max_rounds):
        steps_in_round = 120
        stall = 0
        for step in range(steps_in_round):
            if time.time() > deadline:
                break
            scores = _scores(h)
            eps = 1e-14
            s = np.maximum(scores, eps)
            log_w = (p_current - 1.0) * np.log(s)
            log_w -= np.max(log_w)
            w = np.exp(log_w)
            w_sum = np.sum(w)
            if w_sum < 1e-100:
                w = np.zeros(len(scores))
                w[np.argmax(scores)] = 1.0
            else:
                w /= w_sum
            G = _grad_matrix(h)
            grad = G.T @ w
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            p_dir = -grad
            alpha, c5_try = _line_search(h, p_dir)
            h_new = h + alpha * p_dir
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if stall > 100:
            p_current *= 2.0
        else:
            p_current *= 1.5
        stall_total += stall
        if time.time() > deadline:
            break
        if p_current > 2000:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

        h3, c5_3 = _pnorm_optimize(best_h, dl3, p_start=16.0, p_end=600.0,
                                   n_steps=350, use_momentum=True, momentum=0.5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Binary SA exploration (15%)

        h_bin, c5_bin = _binary_sa(best_h, dl4, rng, n_steps=80000, T0=0.006, T_end=5e-6)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Try a few random binary seeds
        if time.time() < dl4:
            seeds = _build_binary_seeds(best_n, rng, n_seeds=8)
            for s in seeds:

                    break
                sub_dl = min(dl4, time.time() + budget_s * 0.012)
                hb, cb = _binary_sa(s, sub_dl, rng, n_steps=8000, T0=0.004, T_end=3e-6)
                if cb < best_c5:
                    best_c5 = cb
                    best_h = hb.copy()
                    # Refine the binary solution continuously
                    if time.time() < dl4:
                        sub_dl2 = min(dl4, time.time() + budget_s * 0.01)
                        hr, cr = _pnorm_optimize(best_h, sub_dl2, p_start=8, p_end=150, n_steps=80)
                        if cr < best_c5:
                            best_c5 = cr
                            best_h = hr.copy()

    # Phase 5: Perturbation + p-norm restarts (12%)
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.83
        scale = 0.015
        for attempt in range(800):
            if time.time() > dl5:
                break
            if best_c5 < 0.3806:
                break
            h_p = best_h + rng.normal(0, scale, best_n)

            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.85)

        for dn in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10, -8, -10]:
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 120 or n_try == best_n:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + budget_s * 0.015)
            h_r, c5_r = _pnorm_optimize(h_r, sub_dl, p_start=8, p_end=200, n_steps=120)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Pair polish on top lags (5%)
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.95
        hp, c5p = _pair_polish_top(best_h, dl7, max_passes=5, top_pairs=150)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final ultra-fine LP (5%)
    if time.time() < t0 + budget_s * 0.955:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.99,

    # Final cleanup

    final_c5 = _exact_c5(best_h)
    best_n = len(best_h)
    return best_h, final_c5, best_n
