import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    target = float(target)
    # Binary search for the water-filling level
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_gradient(h):
    """Build gradient matrix G and score vector F for the trust-region subproblem."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return F, G

def _solve_trust_cvxpy(h, F, G, rho, timeout=5.0):
    """Solve the trust-region subproblem using CVXPY."""
    n = len(h)
    target = n / 2.0
    x = cp.Variable(n)
    t = cp.Variable()
    
    cons = [
        G @ x + (F - G @ h) <= t,  # linearized scores <= t
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h) <= rho
    ]
    
    prob = cp.Problem(cp.Minimize(t), cons)
    
    # Try solvers in order
    solvers = ['CLARABEL', 'SCS', 'ECOS', 'OSQP']
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.array(x.value, dtype=float)
        except Exception:
            continue
    return None

def _solve_lp_alternating(h_old):
    """Solve the LP: min t s.t. A_d @ h <= t for all d, 0<=h<=1, sum(h)=n/2"""
    n = len(h_old)
    dx = 2.0 / n
    L = 2 * n - 1
    target = n / 2.0
    
    # Build A matrix: A[r, q] = dx * (1 - h_old[q-d]) for valid q
    A = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        q_lo = max(0, d)
        q_hi = min(n, n + d)
        for q in range(q_lo, q_hi):
            p = q - d
            A[r, q] = dx * (1.0 - h_old[p])
    
    # LP: min t
    # Variables: h[0..n-1], t (total n+1 variables)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: A @ h - t <= 0  →  A @ h + (-1)*t <= 0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(L)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 10})
    if res.success:
        return res.x[:n]
    return None

            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Gradually reduce damping
            # eta = max(eta * 0.99, 0.1)
        else:
            stall += 1
            eta = max(eta * 0.9, 0.05)
    
    return best_h, best_c5

def _feasible_rotations(h, rng):
    """Generate feasible rotated/reversed variants."""
    n = len(h)
    variants = []

    variants.append(h[::-1].copy())
    # Cyclic shifts
    for s in [1, n//4, n//2, 3*n//4]:
        variants.append(np.roll(h, s).copy())
    return variants

def _mixture(h1, h2, alpha):
    """Feasible mixture of two feasible vectors."""
    return _project_box_sum(alpha * h1 + (1 - alpha) * h2, len(h1) / 2.0)

def _resample_integral(h, n_new):
    """Resample h to n_new points using integral averaging."""
    n = len(h)
    if n == n_new:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    
    # For each new point, compute the integral average of h over [j*dx_new, (j+1)*dx_new]
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        # Find overlap with old intervals
        for i in range(n):
            a_old = i * dx_old
            b_old = (i + 1) * dx_old
            overlap = max(0.0, min(b, b_old) - max(a, a_old))
            if overlap > 0:
                h_new[j] += h[i] * overlap / dx_new  # weighted average
    
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project_box_sum(h_new, n_new / 2.0)

        if abs(delta) < 1e-10:

        h[i] -= delta

            h[j] -= delta

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    target = float(target)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return F, G


def _solve_lp_gradient(h):
    n = len(h)
    target = n / 2.0
    F, G = _build_gradient(h)
    L = 2 * n - 1
    rhs = F - G @ h
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5, 'max_iter': 5000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _solve_trust_cvxpy(h, F, G, rho, timeout=4.0):
    n = len(h)
    target = n / 2.0
    rhs = F - G @ h
    x = cp.Variable(n)
    t = cp.Variable()
    cons = [
        G @ x <= t - rhs,
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h) <= rho
    ]
    prob = cp.Problem(cp.Minimize(t + 1e-6 * cp.sum_squares(x - h)), cons)
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if prob.status in ('optimal', 'optimal_inaccurate') and x.value is not None:
                return np.array(x.value, dtype=float)
        except Exception:
            continue
    return None

def _resample_integral(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        total = 0.0
        for i in range(n):
            a_old = i * dx_old
            b_old = (i + 1) * dx_old
            overlap = max(0.0, min(b, b_old) - max(a, a_old))
            if overlap > 1e-15:
                total += h[i] * overlap
        h_new[j] = total / dx_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project_box_sum(h_new, n_new / 2.0)

                        grad[d:] += -wd * h[:m]
                else:
                    g = -d
                    m = n - g
                    if m > 0:
                        grad[g:] += wd * (1.0 - h[:m])
                        grad[:m] += -wd * h[g:]
            grad -= np.mean(grad)
            norm = float(np.max(np.abs(grad)))
            if norm < 1e-14:
                break
            direction = -grad / norm * 0.3
            a, new_c5 = _line_search(h, direction, cur_c5,
                                     alphas=[1.0, 0.5, 0.3, 0.15])
            if new_c5 < cur_c5 - 1e-12:
                h = _project_box_sum(h + a * direction, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    safe_stop = t0 + budget_s * 0.96

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP gradient loop (fast, primary)
    deadline1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_loop(h, deadline1, max_iters=10000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth continuation to escape flat regions
    deadline2 = t0 + budget_s * 0.30
    h2, c5_2 = _smooth_continuation(best_h, deadline2, beta_start=3.0, beta_end=150.0, n_beta_steps=8)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More LP
    deadline3 = t0 + budget_s * 0.42
    h3, c5_3 = _lp_loop(best_h, deadline3, max_iters=10000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: CVXPY trust-region
    if HAS_CVXPY:
        deadline4 = t0 + budget_s * 0.55
        h4, c5_4 = _trust_region_loop(best_h, deadline4, rho_init=0.10)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: LP again after trust-region
    deadline5 = t0 + budget_s * 0.62
    h5, c5_5 = _lp_loop(best_h, deadline5, max_iters=8000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Diversification via rotations and mixtures
    deadline6 = t0 + budget_s * 0.75
    variants = []
    variants.append(np.roll(best_h, 1))
    variants.append(np.roll(best_h, -1))
    variants.append(best_h[::-1].copy())
    if best_n % 2 == 0:
        variants.append(np.roll(best_h, best_n // 2))
    for a_mix in [0.3, 0.5, 0.7]:
        variants.append(a_mix * best_h + (1 - a_mix) * best_h[::-1])
    for v in variants:
        if time.time() > deadline6:
            break
        v = _project_box_sum(v, len(v) / 2.0)
        v_c5 = _exact_c5(v)
        if v_c5 < best_c5 + 0.005:
            dl = min(deadline6, time.time() + budget_s * 0.025)
            hv, cv = _lp_loop(v, dl, max_iters=3000)
            if cv < best_c5 - 1e-10:
                best_c5 = cv
                best_h = hv.copy()

    # Phase 7: Random transfer search
    deadline7 = t0 + budget_s * 0.82
    hr, cr = _random_transfer_opt(best_h, deadline7, rng, n_trials=20000)
    if cr < best_c5:
        best_c5 = cr
        best_h = hr.copy()

    # Phase 8: Resolution exploration
    deadline8 = t0 + budget_s * 0.92
    n_cands = sorted(set([
        best_n - 8, best_n - 4, best_n + 4, best_n + 8,
        best_n + 16, best_n - 16, 48, 56, 72, 80, 96, 112
    ]))
    n_cands = [x for x in n_cands if 32 <= x <= 160]
    for n_try in n_cands:
        if time.time() > deadline8:
            break
        per_time = min(budget_s * 0.02, max(8.0, (deadline8 - time.time()) / max(1, len(n_cands))))
        h_r = _resample_integral(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.01:
            dl_r = min(deadline8, time.time() + per_time)
            hr2, cr2 = _lp_loop(h_r, dl_r, max_iters=3000)
            if cr2 < best_c5 - 1e-10:
                best_c5 = cr2
                best_h = hr2.copy()
                best_n = n_try
                if HAS_CVXPY and time.time() < deadline8:
                    hr3, cr3 = _trust_region_loop(best_h, min(deadline8, time.time() + per_time * 0.3), rho_init=0.05)
                    if cr3 < best_c5:
                        best_c5 = cr3
                        best_h = hr3.copy()

    # Phase 9: Final LP polish
    if time.time() < safe_stop:
        dl_final = safe_stop
        hf, cf = _lp_loop(best_h, dl_final, max_iters=6000)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Final projection and evaluation
    best_n = len(best_h)
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5

    return best_h, best_c5, best_n
