def _active_descent_dir(h, band_frac=0.10, gamma=2.0):

        # All scores are ~0, any direction works
        d = np.random.randn(n)
        d -= np.mean(d)
        return d, scores
    
    # Weight active scores by their proximity to max
    margin = band_frac * s_max
    active_mask = scores >= (s_max - margin)
    active_idx = np.where(active_mask)[0]
    
    # Weight: weights = (scores[active] - (s_max - margin))^gamma / normalizer
    diff = scores[active_idx] - (s_max - margin)
