```python
    n = len(h)
    target = n / 2.0

    cur_c5 = _exact_c5(h)

    beta_m = 0.9
    stall = 0
```

```python
    n = len(h)
    topk_vals = np.sort(scores)[-k:]
    topk_idx = np.argsort(scores)[-k:]

    w = np.exp(sharpness * (topk_vals - topk_vals.max()))
```

```python
    if abs(diff) > 1e-10:
        mask = (x < 1.0) if diff > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, d):
    """Gradient of correlation score at lag d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        grad[:m] = (1.0 - h[d:])
        grad[d:] = -h[:m]
    else:
        g = -d
        m = n - g
        grad[:m] = (1.0 - h[g:])
        grad[g:] = -h[:m]
    return grad

def _topk_gradient(h, scores, k=10, sharpness=0.0):
    """Weighted gradient of top-k scores."""
    n = len(h)
    idx = np.argsort(scores)[-k:]
    vals = scores[idx]
    if sharpness > 0:
        w = np.exp(sharpness * (vals - vals.max()))
        w = w / w.sum()
    else:
        w = np.ones(k) / k
    grad = np.zeros(n)
    for wi, si in zip(w, idx):
        d = si - (n - 1)
        grad += wi * _lag_gradient(h, d)
    grad -= np.mean(grad)
    return grad

def _gd_momentum(h, deadline, k_start=16, k_end=3, max_iters=100000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.02
    mom = np.zeros(n)
    beta = 0.9
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max_iters
        k = max(k_end, int(round(k_start * (1 - frac) + k_end * frac)))
        scores = _scores_arr(h)
        grad = _topk_gradient(h, scores, k, sharpness=2.0)
        mom = beta * mom - lr * grad
        gnorm = np.linalg.norm(mom)
        if gnorm < 1e-13:
            break
        direction = mom / gnorm
        alpha = 0.08
        improved = False
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if improved:
            lr = min(lr * 1.01, 0.1)
        else:
            stall += 1
            lr *= 0.85
            if stall > 80 or lr < 1e-7:
                break
    return best_h, best_c5
```

Now for the population search. The idea is:
- Maintain a list of P candidates
- Each iteration, pick a candidate, mutate it, and if it's better than the worst, replace
- With some probability, crossover two candidates

For the crossover: h_new = (1-t)*h_a + t*h_b, then project.

```python
def _population_search(h_init, deadline, pop_size=20, max_iters=50000):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    
    # Initialize population
    pop_h = [h_init.copy()]
    pop_c5 = [_exact_c5(h_init)]
    for _ in range(pop_size - 1):
        h_new = _project(h_init + rng.normal(0, 0.05, n), target)
        pop_h.append(h_new)
        pop_c5.append(_exact_c5(h_new))
    
    best_idx = int(np.argmin(pop_c5))
    best_h = pop_h[best_idx].copy()
    best_c5 = pop_c5[best_idx]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick a random member to mutate
        i = int(rng.integers(pop_size))
        h_cur = pop_h[i]
        c5_cur = pop_c5[i]
        
        # Mutation
        scores = _scores_arr(h_cur)
        k = int(rng.integers(3, 16))
        grad = _topk_gradient(h_cur, scores, k, sharpness=rng.uniform(0, 5))
        direction = -grad + rng.normal(0, 0.02, n)
        gnorm = np.linalg.norm(direction)
        if gnorm < 1e-12:
            continue
        direction /= gnorm
        alpha = rng.uniform(0.01, 0.1)
        h_new = _project(h_cur + alpha * direction, target)
        c5_new = _exact_c5(h_new)
        
        # With 10% probability, try crossover instead
        if rng.random() < 0.1:
            j = int(rng.integers(pop_size))
            t = rng.uniform(0.2, 0.8)
            h_new = _project((1-t) * pop_h[i] + t * pop_h[j], target)
            c5_new = _exact_c5(h_new)
        
        # Accept if better than worst
        worst_idx = int(np.argmax(pop_c5))
        if c5_new < pop_c5[worst_idx]:
            pop_h[worst_idx] = h_new
            pop_c5[worst_idx] = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                # Reset some population members near the best
                for ri in range(pop_size):
                    if rng.random() < 0.3:
                        pop_h[ri] = _project(best_h + rng.normal(0, 0.04, n), target)
                        pop_c5[ri] = _exact_c5(pop_h[ri])
    
    return best_h, best_c5
```

```python
def _resample(h, n_new):
```

```python
def _lp_polish_step(h, scores, radius=0.1):
    """Try to find an LP direction that reduces all scores."""
    n = len(h)
    # G @ p represents the change in scores when h → h + p
    # We want: scores + G @ p ≤ 0 (all scores non-positive, i.e., decreased)
    # Equivalently: G @ p ≤ -scores
    # With sum(p) = 0 and |p_i| ≤ radius
    
    # Build G matrix (gradient of scores w.r.t. h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            m = n - d
            G[r, :m] = (1.0 - h[d:])
            G[r, d:] = -h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] = (1.0 - h[g:])
            G[r, g:] = -h[:m]
    
    # LP: minimize 0 (feasibility) subject to G@p <= -scores, sum(p)=0, bounds
    c = np.zeros(n)
    A_ub = G
    b_ub = -scores
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except:
        pass
    return None
```

```python
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G
```
