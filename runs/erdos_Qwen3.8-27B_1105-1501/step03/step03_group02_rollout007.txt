            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad


def _all_score_grads(h):
    """Compute gradient for all lags at once. Returns array of shape (2n-1, n)."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        grads[r] = _score_grad(h, d)
    return grads


def _project_feasible(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-10:
        if resid > 0:
            mask = (x < 1.0 - 1e-9) & (x > 0.0)
        else:
            mask = (x > 1e-9) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
    return np.clip(x, 0.0, 1.0)
