def _project_onto_simplex_box(v, target_sum):
    """Project v onto {h : 0 <= h[i] <= 1, sum(h) = target_sum}"""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    # Bisection on the threshold
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fix sum exactly
    diff = target_sum - x.sum()
    if abs(diff) > 1e-14:
        # Adjust among interior points
        interior = (x > 1e-12) & (x < 1 - 1e-12)
        count = interior.sum()
        if count > 0:
            x[interior] += diff / count
            x = np.clip(x, 0, 1)

def _structured_start(n, rng):
    """Generate a structured initial guess"""
    r = rng.random()
    if r < 0.25:
        # Two smooth bumps
        c1, c2 = n * 0.25, n * 0.75
        w = n * 0.15
        x = np.arange(n)
        h = np.zeros(n)
        h += np.exp(-0.5 * ((x - c1) / w) ** 2)
        h += np.exp(-0.5 * ((x - c2) / w) ** 2)
        h = h / h.max()
    elif r < 0.4:
        # Cosine window
        x = np.arange(n)
        h = np.cos(np.pi * (x - n * 0.1) / (n * 0.8)) ** 2

        h[x > n * 0.9] = 0

    elif r < 0.7:

        if time.time() > dl4:
            break
        remaining = dl4 - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining / (n_starts - s), 20.0)
        sub_dl = time.time() + alloc
        if s == 0:
            h_try = _project(best_h + rng.normal(0, 0.02, best_n), best_n / 2.0)
        elif s == 1:
            h_try = _project(best_h + rng.normal(0, 0.05, best_n), best_n / 2.0)
        elif s == 2:
            h_try = _structured_start(best_n, rng)
        elif s == 3:
            h_try = (best_h + rng.normal(0, 0.08, best_n) > 0.45).astype(float)
            h_try = _project(h_try, best_n / 2.0)
        elif s == 4:
            shift = int(rng.integers(1, max(2, best_n // 4)))
            h_try = _project(np.roll(best_h, shift), best_n / 2.0)
        elif s == 5:
            h_try = _project(1.0 - best_h, best_n / 2.0)
        elif s == 6:
            h_try = _project(best_h[::-1], best_n / 2.0)
        else:
            h_try = _structured_start(best_n, rng)
        h_try = _project(h_try, best_n / 2.0)
        t1 = time.time() + alloc * 0.5
        h_opt, c5_opt = _optimize_smooth(h_try, t1, best_n / 2.0, use_soft_max=False, max_iters=8000)
        t2 = time.time() + alloc * 0.2
        h_opt, c5_opt = _optimize_smooth(h_opt, t2, best_n / 2.0, use_soft_max=True, T_init=0.2, max_iters=3000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Final polish (15%)
    dl5 = t0 + budget_s * 0.99
    # Alternating soft-max with decreasing T
    h_cur = best_h.copy()
    c5_cur = _exact_c5(h_cur)
    T_list = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02]
    for T in T_list:
        if time.time() > dl5:
            break
        sub_dl = min(dl5, time.time() + (dl5 - time.time()) / max(1, len(T_list)))
        h_cur, c5_cur = _optimize_smooth(h_cur, sub_dl, best_n / 2.0, use_soft_max=True, T_init=T, max_iters=5000)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Coordinate descent polish
    h_cd, c5_cd = _coord_descent(best_h, dl5, best_n / 2.0, max_rounds=100)
    if c5_cd < best_c5 - 1e-15:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
