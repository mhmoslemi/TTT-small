def _compute_grad(w, h):
    n = len(h)
    # w has length 2n-1
    # grad[j] = sum_d w[n-1+d] * ((1-h[j-d])*[valid] - h[j+d]*[valid])
    d = np.arange(-(n-1), n)  # d from -(n-1) to n-1
    j = np.arange(n)
    
    # Create meshgrid
    J, D = np.meshgrid(j, d, indexing='ij')  # J[i,j] = i, D[i,j] = d
    
    # First term: (1-h[J-D]) where 0 <= J-D < n
    idx1 = J - D
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(mask1, 1.0 - h[idx1.clip(0, n-1)], 0.0)
    
    # Second term: h[J+D] where 0 <= J+D < n
    idx2 = J + D
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(mask2, h[idx2.clip(0, n-1)], 0.0)
    
    # grad[j] = sum_d w[n-1+d] * (term1 - term2)
    W = w  # w[n-1+d] = w[d + n - 1], d from -(n-1) to n-1, so index 0 to 2n-2
    grad = (W[None, :] * (term1 - term2)).sum(axis=1)
    return grad

def _lp_polish(h, deadline, radius_init=0.2, max_iters=5000, k_active=15):

def _slsqp_polish(h, deadline, beta=50.0):
    """Use SLSQP to minimize log-sum-exp of scores."""
    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    
    def objective(x):
        scores = _scores(x)
        m = np.max(scores)
        return np.log(np.sum(np.exp(beta * (scores - m)))) / beta
    
    def objective_grad(x):
        scores = _scores(x)
        m = np.max(scores)
        exps = np.exp(beta * (scores - m))
        weights = exps / np.sum(exps)
        return _compute_grad(weights, x)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
         'jac': lambda x: np.ones(n)},
    ]
    bounds = [(0, 1)] * n
    
    try:
        result = minimize(objective, h0, method='SLSQP', jac=objective_grad,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-12})
        h_new = _project(result.x, target)
        c5 = _exact_c5(h_new)
        return h_new, c5
    except Exception:
        return h0, _exact_c5(h0)

            break

        r = rng.random()
        h_pert = best_h.copy()
        
        if r < 0.4:
            # Gaussian perturbation
            h_pert += rng.normal(0, scale, n)
        elif r < 0.6:
            # Sparse perturbation (modify k random coordinates)
            k = int(rng.integers(2, max(3, n // 4)))
            idx = rng.choice(n, size=k, replace=False)
            h_pert[idx] += rng.normal(0, scale * 2, k)
        elif r < 0.8:

            bsize = int(rng.integers(2, max(3, n // 8)))
            i = int(rng.integers(0, n - 2 * bsize))
            j = int(rng.integers(0, n - 2 * bsize))
            if j < i + bsize and j + bsize > i:
                continue
            h_pert[i:i+bsize], h_pert[j:j+bsize] = h_pert[j:j+bsize].copy(), h_pert[i:i+bsize].copy()
        else:

            shift = rng.normal(0, scale * 0.5)
            h_pert = h_pert + shift + rng.normal(0, scale * 0.3, n)
        
        h_pert = _project(h_pert, target)

            scale = max(0.02, scale * 0.85)
        else:

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt <= 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_grad(w, h):
    n = len(h)
    d = np.arange(-(n - 1), n)
    j = np.arange(n)
    J, D = np.meshgrid(j, d, indexing='ij')
    idx1 = J - D
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(mask1, 1.0 - h[idx1.clip(0, n - 1)], 0.0)
    idx2 = J + D
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(mask2, h[idx2.clip(0, n - 1)], 0.0)
    grad = (w[None, :] * (term1 - term2)).sum(axis=1)
    return grad

    stall = 0
    momentum = np.zeros(n)
    mu = 0.85

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        sw = float(np.sum(exps))
        if sw < 1e-300:
            break
        weights = exps / sw
        grad = _compute_grad(weights, h)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        momentum = mu * momentum - lr * grad / gnorm
        direction = -momentum
        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-14:
            break
        direction /= dnorm
        alpha = lr
        improved = False
        for _ls in range(22):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.02, 0.5)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.8
            momentum *= 0.5
            if lr < 1e-6:
                break
    return best_h, best_c5

def _lp_direction_active(h, scores, k_active=20, radius=0.2):
    n = len(h)
    idx_top = np.argsort(scores)[::-1][:k_active]
    one_minus_h = 1.0 - h
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k_active, n + 1))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    b_ub = np.zeros(k_active)
    for i, r in enumerate(idx_top):
        d = r - (n - 1)
        for j in range(n):
            g = 0.0
            ia = j - d
            if 0 <= ia < n:
                g += one_minus_h[ia]
            ib = j + d
            if 0 <= ib < n:
                g -= h[ib]
            A_ub[i, j] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -float(scores[r])
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] - 1e-12)
        hi = min(radius, 1.0 - h[j] + 1e-12)
        bounds.append((lo, hi))
    bounds.append((-radius, radius))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-10:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def _line_search_quad(h, p, c5_cur, target, n):
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 0.001)
    alphas = np.geomspace(0.01, min(alpha_max, 10.0), 20)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section refinement
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max * 1.5, best_a * 2.0)
    if hi_a <= lo_a:
        return best_a, best_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2, f2 = c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo_a = c1
            c1, f1 = c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.25, max_iters=8000, k_active=20):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        scores = _scores(h)
        p = _lp_direction_active(h, scores, k_active=k_active, radius=radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_quad(h, p, cur_c5, target, n)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.6)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.04:
                radius = 0.12
    return best_h, best_c5

def _slsqp_polish(h, deadline, beta=100.0):
    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    def _obj(x):
        sc = _scores(x)
        m = np.max(sc)
        ak = beta * (sc - m)
        ak = np.clip(ak, -50, 50)
        return float(np.log(np.sum(np.exp(ak))) / beta + m)
    def _obj_grad(x):
        sc = _scores(x)
        m = np.max(sc)
        exps = np.exp(np.clip(beta * (sc - m), -50, 50))
        weights = exps / np.sum(exps)
        return _compute_grad(weights, x)
    def _con_fun(x):
        return np.sum(x) - target
    def _con_jac(x):
        return np.ones(n)
    try:
        res = minimize(_obj, h0, method='SLSQP', jac=_obj_grad,
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': 400, 'ftol': 1e-14})
        h_new = _project(res.x, target)
        return h_new, _exact_c5(h_new)
    except Exception:
        return h0, _exact_c5(h0)

def _coord_polish(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    comp = int(np.argmax(1.0 - h_try) if d > 0 else np.argmax(h_try))
                    if comp == j:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        if d > 0:
                            comp = int(np.argmax(np.where(mask, 1.0 - h_try, -1)))
                        else:
                            comp = int(np.argmax(np.where(mask, h_try, -1)))
                    if comp == j:
                        continue
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hop(h_best, c5_best, deadline, rng):
    n = len(h_best)
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = c5_best
    scale = 0.08
    max_tries = 200
    for it in range(max_tries):
        if time.time() > deadline:
            break
        if best_c5 < 0.3805:
            break
        r = rng.random()
        h_pert = best_h.copy()
        if r < 0.35:
            h_pert += rng.normal(0, scale, n)
        elif r < 0.55:
            k = int(rng.integers(2, max(3, n // 3)))
            idx = rng.choice(n, size=k, replace=False)
            h_pert[idx] += rng.normal(0, scale * 2.5, k)
        elif r < 0.75:
            bsize = int(rng.integers(2, max(3, n // 6)))
            i = int(rng.integers(0, max(1, n - 2 * bsize)))
            j = int(rng.integers(0, max(1, n - 2 * bsize)))
            if abs(i - j) >= bsize:
                h_pert[i:i + bsize], h_pert[j:j + bsize] = h_pert[j:j + bsize].copy(), h_pert[i:i + bsize].copy()
        else:
            h_pert = best_h + rng.normal(0, scale * 0.4, n)
            h_pert += rng.normal(0, scale * 0.3, 1)
        h_pert = _project(h_pert, target)
        time_per = (deadline - time.time()) / max(max_tries - it, 1)
        t1 = min(time_per * 0.5, 4.0)
        h_try, c5_try = _softmax_descent(h_pert, time.time() + t1,
                                          beta_start=40.0, beta_end=6000.0, max_iters=3000)
        t2 = min((deadline - time.time()) * 0.3, 3.0)
        h_try, c5_try = _lp_polish(h_try, time.time() + t2,
                                    radius_init=0.15, max_iters=800, k_active=12)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.015, scale * 0.85)
        else:
            scale = min(0.3, scale * 1.15)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Softmax gradient descent (25%)
    deadline1 = t0 + budget_s * 0.25
    h, c5 = _softmax_descent(h, deadline1, beta_start=15.0, beta_end=6000.0, max_iters=15000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP polish (20%)
    deadline2 = t0 + budget_s * 0.45
    h, c5 = _lp_polish(h, deadline2, radius_init=0.25, max_iters=8000, k_active=20)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: SLSQP with increasing beta (10%)
    deadline3 = t0 + budget_s * 0.55
    for beta_val in [50.0, 100.0, 200.0]:
        if time.time() > deadline3:
            break
        h_s, c5_s = _slsqp_polish(best_h, min(deadline3, time.time() + budget_s * 0.04), beta=beta_val)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: Basin hopping (30%)
    deadline4 = t0 + budget_s * 0.85
    best_h, best_c5 = _basin_hop(best_h, best_c5, deadline4, rng)

    # Phase 5: Multi-resolution (7%)
    deadline5 = t0 + budget_s * 0.92
    for dn in list(range(-8, 9, 2)):
        if time.time() > deadline5:
            break
        n_try = best_n + dn
        if 32 <= n_try <= 128 and n_try != best_n:
            t_rem = deadline5 - time.time()
            if t_rem < 3:
                break
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _softmax_descent(h_r, time.time() + t_rem * 0.4,
                                          beta_start=30.0, beta_end=5000.0, max_iters=2000)
            h_r, c5_r = _lp_polish(h_r, time.time() + t_rem * 0.35,
                                    radius_init=0.12, max_iters=600, k_active=12)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 6: Final combined polish (8%)
    deadline6 = t0 + budget_s * 0.99
    h_f, c5_f = _lp_polish(best_h, min(deadline6, t0 + budget_s * 0.96),
                            radius_init=0.2, max_iters=6000, k_active=25)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    if time.time() < deadline6:
        h_s, c5_s = _slsqp_polish(best_h, min(deadline6, time.time() + budget_s * 0.02), beta=300.0)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    if time.time() < deadline6:
        h_cd, c5_cd = _coord_polish(best_h, deadline6, max_passes=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
