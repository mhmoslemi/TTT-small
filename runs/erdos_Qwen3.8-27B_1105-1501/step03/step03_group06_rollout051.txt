    shifted = alpha * (corr - mx)
    shifted = np.clip(shifted, -500, 500)

def _smooth_step(h, alpha, maxiter=100, deadline=None):
    """One L-BFGS-B step on the smoothed objective."""
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        f, g = _smooth_obj_and_grad(np.clip(x, 0, 1), alpha)
        return f, g
    
    try:
        if deadline is None or time.time() < deadline:
            res = minimize(fun, h, jac=True, method='L-BFGS-B',
                         bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': -1})
            x = _project(res.x, target)
    except Exception:
        x = h.copy()
    return x, _exact_c5(x)

def _active_gradient_step(h, n_active=5, step_size=0.01):

    target = n / 2.0

    # Get top n_active lags
    top_idx = np.argsort(corr)[::-1][:n_active]

    for r in top_idx:
        g += _grad_for_lag(h, r)

    g -= np.mean(g)
    
    # Line search

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_for_lag(h, r):

    d = r - (n - 1)

            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5


def _pair_swap_polish(h, deadline, max_rounds=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = [j for j in range(n) if 0.05 < h[j] < 0.95]
    if len(interior) < 2:
        return best_h, best_c5
    no_improve = 0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > 30:
            break
        improved = False
        i = interior[np.random.randint(len(interior))]
        j = interior[np.random.randint(len(interior))]
        if i == j:
            continue
        delta = h[i] - h[j]
        if abs(delta) < 0.01:
            continue
        for s in [0.5, 1.0, 0.3, 0.7, 0.15]:
            a = s * delta
            h[i] -= a
            h[j] += a
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                improved = True
                break
            else:
                h[i] += a
                h[j] -= a
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, target)
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            if 0.05 <= h[i] <= 0.95:
                if i not in interior:
                    interior.append(i)
            if 0.05 <= h[j] <= 0.95:
                if j not in interior:
                    interior.append(j)
            interior = [k for k in interior if 0.05 < h[k] < 0.95]
    return best_h, best_c5


def _smooth_loop(h, deadline, alphas=None):
    if alphas is None:
        alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 120.0, 200.0]
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        for _pass in range(2):
            if time.time() > deadline:
                break
            h, c5 = _lbfgs_smooth_step(h, alpha, maxiter=150, deadline=deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _scatter_perturb(h, rng, scale=0.15):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    # Identify regions contributing to high correlations
    h_pert = h.copy()
    # Move mass from positions that are "sources" of high correlation
    # to positions that are "sinks"
    for _ in range(5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        h_pert[i] -= scale * rng.random()
        h_pert[j] += scale * rng.random()
    return _project(h_pert, n / 2.0)


def _combined_optimize(h, deadline, n_active_start=3, n_active_end=10):
    """Interleave active gradient and smooth steps."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    n_iter = 0
    while time.time() < deadline:
        if n_iter % 5 == 0:
            t_frac = n_iter / max(1, 200)
            na = int(n_active_start + (n_active_end - n_active_start) * t_frac)
            h, c5 = _active_grad_step(h, n_active=na, base_step=0.015)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h, c5 = _lbfgs_smooth_step(h, 40.0, maxiter=50, deadline=deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        n_iter += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth continuation from parent (15% budget)
    d1 = t0 + budget_s * 0.15
    h, c5 = _smooth_loop(h, d1, alphas=[5, 10, 20, 40, 80])
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Combined active + smooth optimization (20% budget)
    d2 = t0 + budget_s * 0.35
    h, c5 = _combined_optimize(h, d2, n_active_start=3, n_active_end=10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Active gradient refinement (10% budget)
    d3 = t0 + budget_s * 0.45
    step = 0.02
    h = best_h.copy()
    for it in range(1000):
        if time.time() > d3:
            break
        h, c5 = _active_grad_step(h, n_active=4, base_step=step)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            step = min(step * 1.08, 0.05)
        else:
            step *= 0.92

    # Phase 4: Pair swap polish (10% budget)
    d4 = t0 + budget_s * 0.55
    h, c5 = _pair_swap_polish(best_h, d4, max_rounds=2000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Restart with perturbations (20% budget)
    d5 = t0 + budget_s * 0.75
    scale = 0.06
    for attempt in range(300):
        if time.time() > d5:
            break
        if best_c5 < 0.38065:
            break
        # Different perturbation strategies
        strat = attempt % 3
        if strat == 0:
            h_p = best_h + rng.normal(0, scale, best_n)
        elif strat == 1:
            h_p = _scatter_perturb(best_h, rng, scale=scale * 1.5)
        else:
            # Binary perturbation
            h_bin = (best_h + rng.normal(0, 0.1, best_n) >= 0.5).astype(float)
            h_bin = _project(h_bin, best_n / 2.0)
            h_p = h_bin * 0.3 + best_h * 0.7
            h_p = _project(h_p, best_n / 2.0)
        h_p = _project(h_p, best_n / 2.0)

        # Quick optimization
        sub_d = min(d5, time.time() + budget_s * 0.012)
        h_try, c5_try = _lbfgs_smooth_step(h_p, 30, maxiter=80, deadline=sub_d)
        if time.time() < d5 and c5_try < best_c5 + 0.001:
            sub_d2 = min(d5, time.time() + budget_s * 0.011)
            h_try, c5_try = _combined_optimize(h_try, sub_d2, 3, 7)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.015, scale * 0.9)
        else:
            scale = min(0.25, scale * 1.12)

    # Phase 6: Resolution sweep (12% budget)
    d6 = t0 + budget_s * 0.87
    offsets = [0, -8, 8, -16, 16, -32, 32, 64, -4, 4]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 320:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _smooth_loop(h_r, sub_d, alphas=[20, 60])
        if time.time() < d6 and c5_r < best_c5 + 0.002:
            sub_d2 = min(d6, time.time() + budget_s * 0.015)
            h_r, c5_r = _combined_optimize(h_r, sub_d2, 3, 8)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final fine-tuning (8% budget)
    d7 = t0 + budget_s * 0.95
    for alpha in [50, 100, 200, 400]:
        if time.time() > d7:
            break
        h_f, c5_f = _lbfgs_smooth_step(best_h, alpha, maxiter=300, deadline=d7)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Final active + pair polish (5% budget)
    d8 = t0 + budget_s * 0.99
    h_f, c5_f = _active_loop_final(best_h, d8)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n


def _active_loop_final(h, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    step = 0.01
    for it in range(500):
        if time.time() > deadline:
            break
        if it % 10 == 0:
            # Pair swap every 10 iterations
            h, c5 = _pair_swap_polish(h, min(deadline, time.time() + 0.3), max_rounds=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h, c5 = _active_grad_step(h, n_active=3, base_step=step)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                step = min(step * 1.05, 0.03)
            else:
                step *= 0.93
    return best_h, best_c5
