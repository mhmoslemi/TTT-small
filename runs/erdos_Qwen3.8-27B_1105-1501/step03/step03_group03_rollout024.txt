   ```python
   def _minimax_1d(a, b, lo, hi):
       """Minimize max_m (a[m]*x + b[m]) for x in [lo, hi]
       where a and b are arrays of length M."""
       M = len(a)
       # Find all breakpoints
       pts = [lo, hi]
       for i in range(M):
           for j in range(i+1, M):
               da = a[i] - a[j]
               if abs(da) > 1e-15:
                   x = (b[j] - b[i]) / da
                   if lo <= x <= hi:
                       pts.append(x)
       pts.append(lo)
       pts.append(hi)
       best_x = lo
       best_val = np.inf
       for x in pts:
           val = np.max(a * x + b)
           if val < best_val:
               best_val = val
               best_x = x
       return best_x, best_val
   ```
