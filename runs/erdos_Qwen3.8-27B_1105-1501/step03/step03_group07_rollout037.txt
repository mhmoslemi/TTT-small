    grad += 2.0 * mu * penalty * np.ones(len(h))

                    h_try = _project(h + sign * alpha * grad, target)
                    c_try = _exact_c5(h_try)
                    if c_try < best_c5 - 1e-13:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try
                        improved = True
                        break
        if not improved:
            active_mask = (h > 1e-6) & (h < 1.0 - 1e-6)
            active_idx = np.where(active_mask)[0]
            if len(active_idx) < 2:
                break
            n_try = min(200, len(active_idx) * len(active_idx))
            for _ in range(n_try):
                if time.time() > deadline:
                    return best_h, best_c5
                i = int(rng.integers(0, len(active_idx)))
                j = int(rng.integers(0, len(active_idx)))
                if i == j:
                    continue
                ii = active_idx[i]
                jj = active_idx[j]
                for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max_i = (1.0 - h[ii]) / di if di > 0 else h[ii] / (-di)
                    a_max_j = (1.0 - h[jj]) / dj if dj > 0 else h[jj] / (-dj)
                    a_max = min(a_max_i, a_max_j)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        a = a_max * frac
                        h_try = h.copy()
                        h_try[ii] += di * a
                        h_try[jj] += dj * a
                        c_try = _exact_c5(h_try)
                        if c_try < best_c5 - 1e-13:
                            best_c5 = c_try
                            best_h = h_try.copy()
                            h = h_try
                            improved = True
                            break
                    if improved:
                        break
        if not improved:
            h_try = _project(h + rng.normal(0, 0.02, n), target)
