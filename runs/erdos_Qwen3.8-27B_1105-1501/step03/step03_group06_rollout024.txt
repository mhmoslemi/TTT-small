    """Compute LSE smooth approximation and its gradient"""
    n = len(h)

    cmax = float(corr.max())
    vals = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = cmax + (1.0 / alpha) * np.log(float(exp_vals.sum()))
    w = exp_vals / float(exp_vals.sum())
    # Gradient
    g = np.zeros(n)

def _subgrad_method(h, deadline, n_iters=2000):
    """Exact subgradient method with adaptive step size"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    c5_cur = best_c5
    
    # Adaptive parameters
    step_init = 0.05
    decay = 0.999
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(corr.max())
        
        # Soft-max weighting (temperature-based)
        temp = max(0.1, cmax * 0.1)
        w = np.exp((corr - cmax) / temp)
        w_sum = float(w.sum())
        w = w / w_sum
        
        # Weighted gradient
        g = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-5:
                continue
            s = r - (n - 1)
            i_min = max(0, -s)
            i_max = min(n - 1, n - 1 - s)
            for i in range(i_min, i_max + 1):
                g[i] += w[r] * (1.0 - h[i + s])
                g[i + s] += w[r] * (-h[i])
        
        # Remove mean to preserve sum
        g -= np.mean(g)
        g_norm = float(np.sqrt(np.dot(g, g)))
        if g_norm < 1e-12:
            break
        
        step = step_init * (decay ** it)
        # Line search
        best_a = 0.0
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a = t * step * n / g_norm  # scale by n for consistent step
            h_t = _project(h - a * g, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5:

                best_a = a
        
        if best_a > 0:

            c5_cur = best_c5
            step = step_init * (decay ** it)  # keep step if improved

            step *= 0.5

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(n, n_ones, iterations=50000, t0=1.0, tf=0.01):
    """Simulated annealing for binary patterns"""
    rng = np.random.default_rng(123)
    # Start with alternating pattern
    h = np.zeros(n)
    positions = rng.permutation(n)[:n_ones]
    h[positions] = 1.0
    
    def eval_binary(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(corr.max())
    
    cur = eval_binary(h)
    best = cur
    best_h = h.copy()
    
    for it in range(iterations):
        if it % 1000 == 0 and time.time() > _sa_deadline:
            break
        T = t0 * (tf / t0) ** (it / iterations)
        
        # Random move: swap a 1 and a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new = eval_binary(h)
        
        if new < cur or rng.random() < np.exp((cur - new) / max(T, 1e-10)):
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best / (n / 2.0)  # Return C5 = max_corr * 2/n = best * 2/(n)
