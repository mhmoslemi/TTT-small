        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 20
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _equalize_direction(h, scores, G, top_k=8):
    n = len(h)
    n_lags = len(scores)
    k = min(top_k, n_lags)
    idx = np.argsort(scores)[-k:]
    Ga = G[idx]
    A = np.vstack([Ga, np.ones((1, n))])
    b = np.concatenate([-np.ones(k), [0.0]])
    try:
        ATA = A @ A.T
        rhs = A @ b
        nu = np.linalg.solve(ATA, rhs)
        p = A.T @ nu
    except Exception:
        return None
    if np.linalg.norm(p) < 1e-14:
        return None
    return p

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.12, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 16, 8, 4]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p = None
        if radius > 0.001:
            p, _ = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = _equalize_direction(h, scores, G, top_k=min(8, 2 * n - 1))
            if p is not None:
                p = np.clip(p, -radius, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=30):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    target = n / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)
        G = _grad_matrix(h)
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                a = G[:, j] - G[:, k]
                dmin = max(-h[j], h[k] - 1.0)
                dmax = min(1.0 - h[j], h[k])
                if dmax - dmin < 1e-10:
                    continue
                base = scores
                # f(delta) = max(base + delta * a)
                # convex PL, find min over [dmin, dmax]
                # Try endpoints and use ternary
                lo = dmin
                hi = dmax
                best_f = _exact_c5(h)
                best_d = 0.0
                # Sample a few points
                n_terr = 15
                tries = np.linspace(lo, hi, n_terr)
                best_t = tries[0]
                best_tf = best_f
                for tt in tries:
                    h_try = h.copy()
                    h_try[j] += tt
                    h_try[k] -= tt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    ft = _exact_c5(h_try)
                    if ft < best_tf:
                        best_tf = ft
                        best_t = tt
                # Refine with golden section
                idx_t = int(np.argmin(np.abs(tries - best_t)))
                lo2 = tries[max(0, idx_t - 1)]
                hi2 = tries[min(n_terr - 1, idx_t + 1)]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1v = hi2 - gr * (hi2 - lo2)
                c2v = lo2 + gr * (hi2 - lo2)
                h_t1 = h.copy(); h_t1[j] += c1v; h_t1[k] -= c1v
                f1 = _exact_c5(np.clip(h_t1, 0, 1))
                h_t2 = h.copy(); h_t2[j] += c2v; h_t2[k] -= c2v
                f2 = _exact_c5(np.clip(h_t2, 0, 1))
                for _ in range(20):
                    if hi2 - lo2 < 1e-14:
                        break
                    if f1 < f2:
                        hi2 = c2v; f2 = f1; c2v = c1v
                        c1v = hi2 - gr * (hi2 - lo2)
                        h_t1 = h.copy(); h_t1[j] += c1v; h_t1[k] -= c1v
                        f1 = _exact_c5(np.clip(h_t1, 0, 1))
                    else:
                        lo2 = c1v; f1 = f2; c1v = c2v
                        c2v = lo2 + gr * (hi2 - lo2)
                        h_t2 = h.copy(); h_t2[j] += c2v; h_t2[k] -= c2v
                        f2 = _exact_c5(np.clip(h_t2, 0, 1))
                d_opt = (lo2 + hi2) / 2.0
                d_opt = max(dmin, min(dmax, d_opt))
                h_try = h.copy()
                h_try[j] += d_opt
                h_try[k] -= d_opt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_gradient_optimize(h, deadline, n_steps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 8.0 + 200.0 * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        alpha = 0.04
        for _ in range(35):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5

def _anneal(h, deadline, n_iters=50000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    t = 0.003
    rng = np.random.default_rng(12345)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            t *= 0.85
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
            continue
        if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
            continue
        dmax = min(1.0 - h[j], h[k])
        if dmax < 1e-8:
            continue
        delta = rng.uniform(-dmax, dmax)
        if abs(delta) < 1e-9:
            continue
        h_try = h.copy()
        h_try[j] += delta
        h_try[k] -= delta
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 or rng.random() < np.exp((cur_c5 - c5_try) / t):
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _structured_perturb(h, rng, scale):
    n = len(h)
    target = n / 2.0
    # Add a smooth low-frequency perturbation
    x = np.arange(n) / n
    freq = rng.integers(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    smooth = scale * np.sin(2 * np.pi * freq * x + phase)
    # Add small high-freql noise
    smooth += scale * 0.3 * rng.normal(0, 1, n)
    h_new = _project(h + smooth, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization (aggressive)
    dl = t0 + budget_s * 0.18
    h, c5 = _lp_optimize(h, dl, top_k_list=[None, 16, 8], radius_init=0.25, max_iters=3000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Smooth gradient
    if time.time() < t0 + budget_s * 0.30:
        dl = t0 + budget_s * 0.33
        h, c5 = _smooth_gradient_optimize(best_h, dl, n_steps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Fine LP
    if time.time() < t0 + budget_s * 0.42:
        dl = t0 + budget_s * 0.47
        h, c5 = _lp_optimize(best_h, dl, top_k_list=[4, 6, 8, None], radius_init=0.05, max_iters=4000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Pair polish
    if time.time() < t0 + budget_s * 0.52:
        dl = t0 + budget_s * 0.58
        hp, c5p = _pair_polish(best_h, dl, max_passes=25)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 5: Structured perturbation restarts
    dl = t0 + budget_s * 0.72
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl:
            break
        if best_c5 < 0.3807:
            break
        h_pert = _structured_perturb(best_h, rng, scale)
        sub_dl = min(dl, time.time() + budget_s * 0.018)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[8, 4, None], radius_init=0.08, max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 6: Simulated annealing
    if time.time() < t0 + budget_s * 0.78:
        dl = t0 + budget_s * 0.82
        ha, c5a = _anneal(best_h, dl, n_iters=40000)
        if c5a < best_c5:
            best_c5 = c5a
            best_h = ha.copy()
        # Quick LP polish after annealing
        if time.time() < t0 + budget_s * 0.84:
            hl, c5l = _lp_optimize(best_h, t0 + budget_s * 0.84, top_k_list=[6, 4], radius_init=0.04, max_iters=2000)
            if c5l < best_c5:
                best_c5 = c5l
                best_h = hl.copy()

    # Phase 7: Resolution sweep
    if time.time() < t0 + budget_s * 0.90:
        dl = t0 + budget_s * 0.92
        for dn in list(range(-8, 9, 2)) + [10, -10, 12, -12]:
            if time.time() > dl:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 96:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None], radius_init=0.1, max_iters=400)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 8: Final pair polish + fine LP
    if time.time() < t0 + budget_s * 0.95:
        dl = t0 + budget_s * 0.96
        hp, c5p = _pair_polish(best_h, dl, max_passes=15)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    if time.time() < t0 + budget_s * 0.98:
        hf, c5f = _lp_optimize(best_h, t0 + budget_s * 0.995, top_k_list=[3, 4, 5], radius_init=0.01, max_iters=3000)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()
            best_n = len(best_h)

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
