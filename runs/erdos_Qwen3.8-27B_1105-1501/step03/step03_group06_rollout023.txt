def _project(v, target):

        return np.ones(n)
    lo = float(v.min()) - 2.0

def _grad_rows(h, idx):
    n = len(h)
    rows = np.zeros((len(idx), n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                rows[ii, d:] += (1.0 - h[:m])
                rows[ii, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                rows[ii, :m] += (1.0 - h[g:])
                rows[ii, g:] += -h[:m]
    return rows


def _subgrad_dir_opt(h, m):

    A_eq = np.zeros((1, m + 1))

def _grad_full(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))

            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _slsqp_epigraph(h, top_k, deadline, maxiter=200):
    n = len(h)
    target = n / 2.0
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-top_k:]
    corr_full = c.copy()

    def _obj_val(hx):
        cf = np.correlate(hx, 1.0 - hx, mode='full')
        return float(cf[top_idx].max())

    def _make_constraint(idx_k):
        def _con(x):
            hx = x[1:]
            t = x[0]
            cf = np.correlate(hx, 1.0 - hx, mode='full')
            return t - cf[idx_k]
        _con.idx_k = idx_k
        return _con

    x0 = np.zeros(n + 1)
    x0[0] = float(corr_full.max()) + 0.1
    x0[1:] = h

    cons_list = []
    for idx_k in top_idx:
        con_fn = _make_constraint(idx_k)
        cons_list.append({'type': 'ineq', 'fun': con_fn})
    cons_list.append({'type': 'eq', 'fun': lambda x: float(np.sum(x[1:])) - target})

    bounds = [(None, None)] + [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_val, x0, method='SLSQP',
                       args=(),
                       bounds=bounds,
                       constraints=cons_list,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.success or res.fun < x0[0]:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h.copy(), _exact_c5(h)


def _smooth_fun_obj(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -400.0, 400.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    return f


def _lbfgs_step(h, alpha, maxiter=200):

        res = minimize(_fun_and_grad, h, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _symmetrize(h):
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return _project(h_sym, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth L-BFGS (15% budget)
    d1 = t0 + budget_s * 0.15
    for alpha in [20.0, 50.0, 100.0, 200.0]:
        if time.time() > d1:
            break
        h_s, c5_s = _lbfgs_step(h, alpha, maxiter=150)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h = h_s

    # Phase 2: Subgradient minimax (25% budget)
    d2 = t0 + budget_s * 0.40
    m_sched = [4, 6, 8, 10, 15, 20, 15, 10, 8, 6, 5, 4, 5, 6, 8, 10, 8, 6, 5, 4]
    h, c5 = _minimax_subgrad_loop(h, d2, m_schedule=m_sched, max_iters=5000)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: LP trust-region polish (15% budget)
    d3 = t0 + budget_s * 0.55
    h, c5 = _lp_loop(h, d3, radius_init=0.2, max_iters=3000, stall_limit=70)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 4: SLSQP epigraph on top lags (10% budget)
    d4 = t0 + budget_s * 0.65
    for tk in [8, 12, 16, 10]:
        if time.time() > d4:
            break
        h_sl, c5_sl = _slsqp_epigraph(best_h, tk, d4, maxiter=100)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl.copy()
            best_n = len(best_h)

    # Phase 5: Restart with perturbations and re-optimize (15% budget)
    d5 = t0 + budget_s * 0.80
    scale = 0.06
    for attempt in range(200):
        if time.time() > d5:
            break
        if best_c5 < 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        # Quick subgradient
        sub_d = min(d5, time.time() + budget_s * 0.015)
        h_try, c5_try = _minimax_subgrad_loop(h_p, sub_d, m_schedule=[5, 8, 10, 8, 6, 5],
                                              max_iters=200)
        if c5_try < best_c5 - 1e-11:
            # Follow with LP
            sub_d2 = min(d5, time.time() + budget_s * 0.02)
            h_try, c5_try = _lp_loop(h_try, sub_d2, radius_init=0.1, max_iters=800, stall_limit=40)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.005, scale * 0.8)
        else:
            # Try symmetrized version
            h_sym = _symmetrize(h_p)
            sub_d3 = min(d5, time.time() + budget_s * 0.01)
            h_sym2, c5_sym = _minimax_subgrad_loop(h_sym, sub_d3, max_iters=100)
            if c5_sym < best_c5 - 1e-11:
                best_c5, best_h = c5_sym, h_sym2.copy()
                scale = max(0.005, scale * 0.85)
            else:
                scale = min(0.25, scale * 1.12)

    # Phase 6: Resolution search (10% budget)
    d6 = t0 + budget_s * 0.90
    for dn in [2, -2, 4, -4, 6, -6, 8, 12, -12, 16, -16, 32, -32]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _minimax_subgrad_loop(h_r, sub_d, max_iters=300)
        if time.time() < d6:
            sub_d2 = min(d6, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_d2, radius_init=0.12, max_iters=500, stall_limit=40)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Final combined polish (10% budget)
    d7 = t0 + budget_s * 0.99
    # Subgradient
    h_f, c5_f = _minimax_subgrad_loop(best_h, min(d7, time.time() + budget_s * 0.04),
                                      m_schedule=[3, 4, 5, 4, 3, 4, 5, 4, 3], max_iters=2000)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    # LP
    if time.time() < d7:
        h_f2, c5_f2 = _lp_loop(best_h, d7, radius_init=0.08, max_iters=3000, stall_limit=80)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
