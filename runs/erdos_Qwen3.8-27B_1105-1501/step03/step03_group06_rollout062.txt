from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_row(h, d):
    """Gradient of corr at lag d w.r.t. h. d ranges from -(n-1) to n-1."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        g_abs = -d
        m = n - g_abs
        if m > 0:
            g[:m] += (1.0 - h[g_abs:])
            g[g_abs:] += -h[:m]
    return g

def _all_grads(h):
    """Return matrix G where G[r] is gradient of corr at row r."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        G[r] = _grad_row(h, d)
    return G

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:

    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _slsqp_solve(h, active_indices, T_init, maxiter=200, ftol=1e-12):
    """Solve the epigraph problem with SLSQP."""
    n = len(h)
    k = len(active_indices)
    x0 = np.zeros(n + 1)
    x0[0] = T_init
    x0[1:] = h

        return x[0]

        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

        # Constraint: x[0] - corr_idx(x[1:]) >= 0
        # Gradient: [1, -G[idx, :]]

            return x[0] - float(_correlations(h_sub)[idx])

            g[1:] = -_grad_row(h_sub, idx - (n-1))  # idx is row, d = idx - (n-1)
            return g

def _smooth_obj(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    return f

def _smooth_grad(x, alpha, G):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    w = exp_vals / float(np.sum(exp_vals))
    g = G.T @ w
    g -= np.mean(g)  # remove mean to handle the equality constraint
    return g

def _lp_direction(h, scores, G, radius):

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))

    except Exception:

    return None

    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        ga = -d
        m = n - ga
        if m > 0:
            g[:m] += (1.0 - h[ga:])
            g[ga:] += -h[:m]
    return g

def _all_grads(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        G[n-1+d] = _grad_row(h, d)
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_fg(x):
    alpha = _gs['alpha']
    s = _correlations(x)
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _all_grads(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g

def _slq_obj(x):
    return x[0]

def _slq_obj_jac(x):
    g = np.zeros(len(x))
    g[0] = 1.0
    return g

def _slq_ineq(x):
    h_sub = x[1:]
    return x[0] - _correlations(h_sub)[_gs['active']]

def _slq_ineq_jac(x):
    n = _gs['n']
    h_sub = x[1:]
    k = len(_gs['active'])
    J = np.zeros((k, n + 1))
    J[:, 0] = 1.0
    for i in range(k):
        idx = _gs['active'][i]
        J[i, 1:] = -_grad_row(h_sub, idx - (n - 1))
    return J

def _slq_eq(x):
    return float(np.sum(x[1:]) - _gs['n'] / 2.0)

def _slq_eq_jac(x):
    g = np.zeros(_gs['n'] + 1)
    g[1:] = 1.0
    return g

def _slsqp_solve(h, active, T_init, maxiter=150):
    n = len(h)
    _gs['n'] = n
    _gs['active'] = active
    x0 = np.zeros(n + 1)
    x0[0] = T_init
    x0[1:] = h
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': _slq_ineq, 'jac': _slq_ineq_jac},
        {'type': 'eq', 'fun': _slq_eq, 'jac': _slq_eq_jac},
    ]
    try:
        res = minimize(_slq_obj, x0, jac=_slq_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.x is not None:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            return _project(h_new, n / 2.0)
    except Exception:
        pass
    return None

def _slsqp_loop(h, deadline, max_iters=60, k_active=12):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        scores = _correlations(h)
        T = float(np.max(scores))
        tol = max(0.05 * max(T, 1.0), 0.3)
        active = np.where(scores >= T - tol)[0]
        if len(active) > k_active:
            active = active[np.argsort(scores[active])[-k_active:]]
        h_new = _slsqp_solve(h, active, T + 0.1, maxiter=120)
        if h_new is not None:
            new_c5 = _exact_c5(h_new)
            if new_c5 < best_c5 - 1e-12:
                h = h_new
                best_h, best_c5 = h.copy(), new_c5
                k_active = max(4, int(k_active * 0.7))
            else:
                k_active = min(24, k_active + 2)
    return best_h, best_c5

def _lp_dir(h, scores, G, radius, top_k):
    n = len(h)
    T = float(np.max(scores))
    tol = max(0.01 * max(T, 1.0), 0.15)
    idx = np.where(scores >= T - tol)[0]
    if len(idx) > top_k:
        order = np.argsort(scores[idx])
        idx = idx[order[-top_k:]]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = G[idx, :]
    A_ub = np.column_stack([A_ub, -np.ones(len(idx))])
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bnds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bnds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 16):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.2)
        hi = best_a + alpha_max * 0.2
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(16):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.12, max_iters=5000, stall_limit=55, top_k=25):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _correlations(h)
        G = _all_grads(h)
        p = _lp_dir(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.35)
        else:
            stall += 1
            radius *= 0.72
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.07
    return best_h, best_c5

def _lbfgs_step(h, alpha, maxiter=80):
    n = len(h)
    _gs['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fg, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    return _project(x, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)

def _grad_step(h, alpha_smooth, n_steps=4):
    n = len(h)
    _gs['alpha'] = alpha_smooth
    h = h.copy()
    for _ in range(n_steps):
        scores = _correlations(h)
        ms = float(np.max(scores))
        vals = np.clip(alpha_smooth * (scores - ms), -500.0, 500.0)
        exp_vals = np.exp(vals)
        w = exp_vals / float(np.sum(exp_vals))
        G = _all_grads(h)
        g = G.T @ w
        g -= np.mean(g)
        norm = float(np.linalg.norm(g))
        if norm < 1e-14:
            break
        h = _project(h - 0.04 * g / norm, n / 2.0)
    return h
