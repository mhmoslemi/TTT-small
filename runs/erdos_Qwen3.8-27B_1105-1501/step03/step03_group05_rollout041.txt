    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_alpha(h, i, j):

    Returns (best_alpha, best_c5) or (0, current_c5) if no improvement."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores_raw(h)  # length 2n-1
    d1 = j - i  # positive (since i < j WLOG)
    d2 = i - j  # negative
    
    # Compute linear coefficients b_d for all lags
    # Index in scores array: lag d corresponds to index n-1+d
    b = np.zeros(2*n-1)
    
    # For each lag d, compute ∂S_d/∂h[i] - ∂S_d/∂h[j]
    # ∂S_d/∂h[x] = (1-h[x+d])*1_{valid_first(x,d)} - h[x-d]*1_{valid_second(x,d)}
    
    # For x=i and x=j, for all d:
    for x in [i, j]:
        sign = 1.0 if x == i else -1.0
        for d in range(-(n-1), n):
            idx = n - 1 + d
            # First factor: h[x] as first factor, h[x+d] as second
            if 0 <= x + d <= n - 1:
                b[idx] += sign * (1.0 - h[x + d])
            # Second factor: h[x-d] as first, h[x] as second (i.e., h[x] = h[(x-d)+d])
            if 0 <= x - d <= n - 1:
                b[idx] += sign * (-h[x - d])
    
    # Quadratic coefficients
    c = np.zeros(2*n-1)
    c[n-1] = -2.0  # d=0
    c[n-1 + d1] = 1.0  # d = j-i > 0
    c[n-1 + d2] = 1.0  # d = i-j < 0
    
    # Feasible alpha range
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-12:
        return 0.0, _exact_c5(h)

    # Use grid + golden section

    n_grid = 64
    alphas = np.linspace(a_lo, a_hi, n_grid)
    best_a = 0.0

        f_val = float(np.max(s_new)) * dx

            best_f = f_val

    step = (a_hi - a_lo) / n_grid
    lo = max(a_lo, best_a - step)
    hi = min(a_hi, best_a + step)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def f_of_a(a):
        s_new = scores + b * a + c * a * a
        return float(np.max(s_new)) * dx
    
    f1 = f_of_a(c1)
    f2 = f_of_a(c2)
    
    for _ in range(50):

        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f_of_a(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f_of_a(c2)
    
    best_a = (lo + hi) / 2.0
    best_f = f_of_a(best_a)
    
    # Verify with exact computation
    h_new = h.copy()
    h_new[i] += best_a
    h_new[j] -= best_a
    h_new = np.clip(h_new, 0.0, 1.0)
    exact_f = _exact_c5(h_new)
    
    if exact_f < best_f:
        best_f = exact_f
    # Actually use exact_f since it's the ground truth
    
    return best_a, exact_f

    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _pair_alpha(h, i, j):
    n = len(h)
    dx = 2.0 / n
    scores = _scores_raw(h)
    n_s = 2 * n - 1
    b = np.zeros(n_s)
    for x, sgn in [(i, 1.0), (j, -1.0)]:
        for d in range(-(n - 1), n):
            idx = n - 1 + d
            v = 0.0
            if 0 <= x + d < n:
                v += (1.0 - h[x + d])
            if 0 <= x - d < n:
                v += (-h[x - d])
            b[idx] += sgn * v
    c = np.zeros(n_s)
    c[n - 1] = -2.0
    d1 = j - i
    c[n - 1 + d1] = 1.0
    c[n - 1 - d1] = 1.0
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-12:
        return 0.0, float(np.max(scores) * dx)
    n_grid = 64
    alphas = np.linspace(a_lo, a_hi, n_grid)
    a2 = alphas ** 2
    S = scores[:, None] + b[:, None] * alphas[None, :] + c[:, None] * a2[None, :]
    f_vals = S.max(axis=0) * dx
    best_idx = int(f_vals.argmin())
    best_a = float(alphas[best_idx])
    best_f = float(f_vals[best_idx])
    cur_f = float(np.max(scores) * dx)
    if best_f >= cur_f - 1e-14:
        return 0.0, cur_f
    step = (a_hi - a_lo) / n_grid
    lo = max(a_lo, best_a - step)
    hi = min(a_hi, best_a + step)
    if hi - lo < 1e-12:
        h_new = h.copy()
        h_new[i] += best_a
        h_new[j] -= best_a
        return best_a, _exact_c5(np.clip(h_new, 0, 1))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float((scores + b * c1 + c * c1 * c1).max()) * dx
    f2 = float((scores + b * c2 + c * c2 * c2).max()) * dx
    for _ in range(40):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float((scores + b * c1 + c * c1 * c1).max()) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float((scores + b * c2 + c * c2 * c2).max()) * dx
    best_a = (lo + hi) / 2.0
    h_new = h.copy()
    h_new[i] += best_a
    h_new[j] -= best_a
    h_new = np.clip(h_new, 0.0, 1.0)
    return best_a, _exact_c5(h_new)

def _stochastic_pair_descent(h, deadline, rng, max_iters=20000, temp=0.0, full_scan_every=500):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    no_improve = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        do_full = (it % full_scan_every == 0)
        if do_full:
            best_imp = 0
            best_pair = None
            best_a = 0
            best_new = cur_c5
            for i in range(n):
                for j in range(i + 1, n):
                    if abs(h[i] - h[j]) < 1e-5:
                        continue
                    a, cn = _pair_alpha(h, i, j)
                    if cn < best_new - 1e-13:
                        best_new = cn
                        best_pair = (i, j)
                        best_a = a
            if best_pair is not None:
                i, j = best_pair
                h[i] += best_a
                h[j] -= best_a
                h = np.clip(h, 0.0, 1.0)
                h = _project_box_sum(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += full_scan_every
        else:
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                a, cn = _pair_alpha(h, i, j)
                if cn < cur_c5 - 1e-13:
                    h[i] += a
                    h[j] -= a
                    h = np.clip(h, 0.0, 1.0)
                    h = _project_box_sum(h, n / 2.0)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5 - 1e-14:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        no_improve = 0
                    break
            else:
                no_improve += 1
        if temp > 0 and no_improve % 200 == 199:
            perturb = rng.normal(0, temp, n)
            h_p = h + perturb
            h_p = _project_box_sum(h_p, n / 2.0)
            c5_p = _exact_c5(h_p)
            if c5_p < cur_c5 + temp * 2:
                h = h_p
                cur_c5 = c5_p
        if no_improve > 5000:
            break
    return best_h, best_c5

def _greedy_pair_descent(h, deadline, max_rounds=200):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_new = cur_c5
        best_pair = None
        best_a = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                a, cn = _pair_alpha(h, i, j)
                if cn < best_new - 1e-13:
                    best_new = cn
                    best_pair = (i, j)
                    best_a = a
        if best_pair is None:
            break
        i, j = best_pair
        h[i] += best_a
        h[j] -= best_a
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, n / 2.0)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _gradient_pair_descent(h, deadline, max_iters=10000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _scores_raw(h)
    max_s = float(np.max(scores))
    active = np.where(np.abs(scores * (2.0 / n) - max_s) < 1e-3)[0]
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores_raw(h)
        cur_c5 = float(np.max(scores) * (2.0 / n))
        max_s = float(np.max(scores))
        active = np.where(np.abs(scores - max_s) < 0.01 * max_s)[0]
        best_new = cur_c5
        best_pair = None
        best_a = 0
        for i in range(n):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                a, cn = _pair_alpha(h, i, j)
                if cn < best_new - 1e-13:
                    best_new = cn
                    best_pair = (i, j)
                    best_a = a
        if best_pair is None:
            break
        i, j = best_pair
        h[i] += best_a
        h[j] -= best_a
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, n / 2.0)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break
    return best_h, best_c5

def _binary_start(n, offset=0):
    h = np.zeros(n)
    ones_per_block = 1
    block_size = 2
    n_full = n // (block_size * ones_per_block)
    h[offset:offset + n_full * block_size * ones_per_block: block_size] = 1.0
    remaining = n - len(h) * 0
    rem = n - (offset + n_full * block_size * ones_per_block)
    target = n // 2
    cur_sum = int(h.sum())
    if cur_sum < target:
        zeros = np.where(h == 0)[0]
        for k in range(target - cur_sum):
            h[zeros[k % len(zeros)]] = 1.0
    elif cur_sum > target:
        ones = np.where(h == 1)[0]
        for k in range(cur_sum - target):
            h[ones[k % len(ones)]] = 0.0
    return h

def _multi_binary_starts(n, seed):
    rng = np.random.default_rng(seed)
    starts = []
    h_base = _binary_start(n)
    starts.append(h_base.copy())
    h_rev = h_base[::-1].copy()
    starts.append(h_rev)
    h_shift = np.roll(h_base, n // 4).copy()
    starts.append(_project_box_sum(h_shift, n // 2.0))
    for _ in range(4):
        h_rand = (rng.random(n) < 0.5).astype(float)
        h_rand = _project_box_sum(h_rand, n / 2.0)
        starts.append(h_rand)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Stochastic pair descent on parent (fast, 15% budget)
    dl1 = t0 + budget_s * 0.15
    h1, c5_1 = _stochastic_pair_descent(h, dl1, rng, max_iters=30000, full_scan_every=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Greedy full-scan pair descent (thorough, 10% budget)
    dl2 = t0 + budget_s * 0.25
    h2, c5_2 = _greedy_pair_descent(best_h, dl2, max_rounds=100)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (25% budget)
    dl3 = t0 + budget_s * 0.50
    n_candidates = sorted(set(x for x in [n0-8, n0-4, n0+4, n0+8, n0+12,
                                            32, 40, 48, 56, 64, 72, 80, 88, 96]
                               if 24 <= x <= 128))
    for n_try in n_candidates:
        if time.time() > dl3:
            break
        per_time = budget_s * 0.03
        h_r = _resample(best_h, n_try)
        dl_r = min(dl3, time.time() + per_time)
        h_r, c5_r = _stochastic_pair_descent(h_r, dl_r, rng, max_iters=8000, full_scan_every=200)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Quick greedy polish
            if time.time() < dl3:
                dl_p = min(dl3, time.time() + per_time * 0.3)
                h_p, c5_p = _greedy_pair_descent(best_h, dl_p, max_rounds=30)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Phase 4: Binary starts at various resolutions (15% budget)
    dl4 = t0 + budget_s * 0.65
    for n_try in [best_n, best_n - 4, best_n + 4, 64, 80]:
        if time.time() > dl4:
            break
        if n_try < 24 or n_try > 128:
            continue
        starts = _multi_binary_starts(n_try, seed + n_try)
        for k, h_s in enumerate(starts):
            if time.time() > dl4:
                break
            per_s = budget_s * 0.02
            dl_s = min(dl4, time.time() + per_s)
            h_s2, c5_s = _stochastic_pair_descent(h_s, dl_s, rng, max_iters=5000, full_scan_every=200)
            if c5_s < best_c5 - 1e-10:
                best_c5 = c5_s
                best_h = h_s2.copy()
                best_n = n_try

    # Phase 5: Perturbation and restart (10% budget)
    dl5 = t0 + budget_s * 0.75
    for trial in range(30):
        if time.time() > dl5:
            break
        noise_scale = 0.1 + 0.1 * rng.random()
        h_p = best_h + rng.normal(0, noise_scale, best_n)
        h_p = _project_box_sum(h_p, best_n / 2.0)
        c5_p0 = _exact_c5(h_p)
        if c5_p0 < best_c5:
            best_c5 = c5_p0
            best_h = h_p.copy()
        dl_p = min(dl5, time.time() + budget_s * 0.015)
        h_p2, c5_p2 = _stochastic_pair_descent(h_p, dl_p, rng, max_iters=3000, full_scan_every=100)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 6: Coarse-to-fine refinement (10% budget)
    dl6 = t0 + budget_s * 0.85
    n_coarse = max(24, best_n // 2)
    h_c = _resample(best_h, n_coarse)
    dl_c = min(dl6, time.time() + budget_s * 0.04)
    h_c, c5_c = _stochastic_pair_descent(h_c, dl_c, rng, max_iters=5000, full_scan_every=150)
    h_f = _resample(h_c, best_n)
    dl_f = min(dl6, time.time() + budget_s * 0.04)
    h_f, c5_f = _stochastic_pair_descent(h_f, dl_f, rng, max_iters=5000, full_scan_every=150)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    # Also try finer
    n_fine = min(128, best_n + 16)
    h_cf = _resample(best_h, n_fine)
    if time.time() < dl6:
        dl_cf = min(dl6, time.time() + budget_s * 0.03)
        h_cf, c5_cf = _stochastic_pair_descent(h_cf, dl_cf, rng, max_iters=3000, full_scan_every=100)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf.copy()
            best_n = n_fine

    # Phase 7: Final greedy polish (10% budget)
    dl7 = t0 + budget_s * 0.95
    h7, c5_7 = _greedy_pair_descent(best_h, dl7, max_rounds=150)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final stochastic polish
    if time.time() < t0 + budget_s * 0.99:
        h8, c5_8 = _stochastic_pair_descent(best_h, t0 + budget_s * 0.995, rng, max_iters=10000, full_scan_every=500)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            best_n = len(best_h)

    # Final projection and verification
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
