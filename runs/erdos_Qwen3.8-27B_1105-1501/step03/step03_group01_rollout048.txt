import numpy as np
from scipy.optimize import linprog

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _project(v, target):
    # Project v onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        s = np.sum(np.clip(v-mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)/2, 0, 1)
    # Fix sum
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x < 1.0) & (x > 0.0)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x

def _line_search(h, P, A, B, C, t_min, t_max, n_samples=30):

def _gen_direction_grad(h, A, tau):
    """Generate a mean-zero direction from softmax-gradient of max correlation."""
    n = len(h)
    dx = 2.0/n
    scores = A * dx
    max_s = np.max(scores)
    w = np.exp(tau*(scores - max_s))
    w /= w.sum()
    
    # Gradient: for each j, grad[j] = sum_k w[k] * dA_k/dh_j

            grad[js2] -= w[k] * h[js2+d]

    P = -grad  # descent direction
    P -= P.mean()
    return P
