                break
        else:
            # No improvement found, shrink velocity
            v *= 0.5

                row[d:] += omh[:m]
                row[:m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                row[:m] += omh[g:]
                row[g:] -= h[:m]
        G[r] = row * (2.0 / n)
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_momentum(h, deadline, beta_start=5.0, beta_end=300.0,
                     n_steps=5000, mu_start=0.3, mu_end=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    stall = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        mu = mu_start + (mu_end - mu_start) * frac
        
        scores = _scores(h)
        mx = np.max(scores)
        diff = beta * (scores - mx)
        w = np.exp(diff)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        
        v = mu * v + grad
        nv = np.linalg.norm(v)
        if nv < 1e-15:
            break
        v /= nv
        
        # Backtracking line search
        alpha = min(0.1, 1.0 / (nv + 1e-10))
        improved = False
        for _ in range(40):
            h_new = _project(h + alpha * v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-14:
                break
        if not improved:
            v *= 0.3
            stall += 1
        else:
            stall = 0
    
    return best_h, best_c5

def _anneal(h, deadline, rng, n_steps=50000, t_start=0.005, t_end=0.0001):
    """Simulated annealing with pair transfer moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Pick two random coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Random transfer amount
        delta = rng.uniform(-0.1, 0.1)
        # Ensure feasibility
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        
        h[i] = hi_new
        h[j] = hj_new
        c5_new = _exact_c5(h)
        
        dC = c5_new - cur_c5
        if dC < 0 or rng.random() < np.exp(-dC / T):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h[j] += sign * a_opt
                    h[k] -= sign * a_opt
                    h = np.clip(h, 0.0, 1.0)
                    c5_try = _exact_c5(h)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                    else:
                        # revert
                        h[k] += sign * a_opt
                        h[j] -= sign * a_opt
                        h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5
