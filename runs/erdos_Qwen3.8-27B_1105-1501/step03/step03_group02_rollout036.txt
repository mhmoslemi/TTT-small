def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _anti_to_full(y):
    """Convert anti-periodic parameter y to full h"""
    m = len(y)
    return np.concatenate([y, 1.0 - y])

def _c5_anti(y):
    """C5 for anti-periodic parameterization"""
    h = _anti_to_full(y)
    return _exact_c5(h)

def _c5_anti_smooth(y, beta=50.0):
    """Smoothed C5 for anti-periodic parameterization"""
    h = _anti_to_full(y)
    scores = _all_scores(h)
    dx = 2.0 / len(h)
    max_s = np.max(scores)
    # log-sum-exp for numerical stability
    shifted = beta * (scores - max_s)
    lse = max_s + np.log(np.exp(shifted).sum()) / beta
    return float(lse * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _anti_to_full(y):
    m = len(y)
    return np.concatenate([y, 1.0 - y])

def _smooth_max_obj(y, beta):
    h = _anti_to_full(y)
    scores = _all_scores(h)
    n = len(h)
    dx = 2.0 / n
    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    lse = max_s + float(np.log(np.sum(np.exp(shifted)))) / beta
    return lse * dx

def _smooth_max_grad_anti(y, beta):
    h = _anti_to_full(y)
    scores = _all_scores(h)
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    gh = np.zeros(n)
    for idx in range(len(scores)):
        if w[idx] < 1e-12:
            continue
        d = idx - (n - 1)
        gh += w[idx] * _score_grad(h, d)
    gy = gh[:m] - gh[m:]
    return gy * dx

def _lbfgsb_anti(y_init, deadline, betas=(15, 30, 60, 120, 250)):
    best_y = y_init.copy()
    best_c5 = _exact_c5(_anti_to_full(y_init))
    for beta in betas:
        if time.time() > deadline:
            break
        try:

                return _smooth_max_obj(y, beta)
            def grad_fun(y):
                return _smooth_max_grad_anti(y, beta)
            res = minimize(obj_fun, best_y, jac=grad_fun,
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(y_init),
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-11})
            y_try = np.clip(res.x, 0.0, 1.0)
            c5_try = _exact_c5(_anti_to_full(y_try))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_y = y_try
        except Exception:
            pass
    return best_y, best_c5

def _subgradient_step_dir(h, scores, beta=None, top_k=None):
    n = len(h)
    if top_k is not None:
        k = min(top_k, len(scores))
        idx = np.argsort(scores)[::-1][:k]
        w = np.zeros(len(scores))
        w[idx] = 1.0 / k
    elif beta is not None:
        max_s = float(np.max(scores))

        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
    else:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    gh = np.zeros(n)
    for idx in range(len(scores)):
        if w[idx] < 1e-12:
            continue
        d = idx - (n - 1)
        gh += w[idx] * _score_grad(h, d)
    gh -= np.mean(gh)
    gn = float(np.max(np.abs(gh)))
    if gn < 1e-14:
        return None
    return gh / gn

def _slsqp_active(h_init, deadline, top_k=20):
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    z0 = float(np.max(scores)) + 1e-5
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]
    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 5.0)]

    def _obj(x):
        return x[n]

    def _obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for d in active_lags:
        cons.append({'type': 'ineq',
                     'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
                     'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])})

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-13, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return h_init, _exact_c5(h_init)
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_quick_search(h_bin, deadline, max_iters=5000):
    n = len(h_bin)
    c5 = _exact_c5(h_bin)
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    for it in range(max_iters):
        if time.time() > deadline:
            break
        best_impr = 0
        best_pair = None
        for i, po in enumerate(ones):
            if i % 3 != 0 and it > 100:
                continue
            for pz in zeros:
                h_bin[po] = 0.0
                h_bin[pz] = 1.0
                new_c5 = _exact_c5(h_bin)
                h_bin[po] = 1.0
                h_bin[pz] = 0.0
                impr = c5 - new_c5
                if impr > best_impr:
                    best_impr = impr
                    best_pair = (po, pz)
        if best_pair is None:
            break
        po, pz = best_pair
        ones.remove(po)
        ones.append(pz)
        zeros.remove(pz)
        zeros.append(po)
        h_bin[po] = 0.0
        h_bin[pz] = 1.0
        c5 -= best_impr
    return h_bin, c5
