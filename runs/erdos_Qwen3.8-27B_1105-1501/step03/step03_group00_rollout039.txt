```python
def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    # G[m, i] = dx * [(1-h[i+d_m])*[0<=i+d_m<n] - h[i-d_m]*[0<=i-d_m<n]]
    # where d_m = n-1-m
    
    # For each m, d_m = n-1-m
    # G[m, i] = dx * [(1-h[i+d_m])*[valid] - h[i-d_m]*[valid]]
    
    # Let me compute this row by row (vectorized over i)
    G = np.zeros((n_lags, n))
    for m in range(n_lags):
        d = n - 1 - m
        # Term 1: (1-h[i+d]) for 0<=i+d<n, i.e., max(0,-d) <= i < n-d (if d>=0) or 0<=i<n+d (if d<0)
        # Actually: 0 <= i+d < n → -d <= i < n-d
        # Valid i range: [max(0, -d), min(n, n-d) - 1]
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if hi1 > lo1:
            G[m, lo1:hi1] = dx * (1.0 - h[lo1+d:hi1+d])
        
        # Term 2: -h[i-d] for 0<=i-d<n, i.e., d <= i < n+d
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if hi2 > lo2:
            G[m, lo2:hi2] -= dx * h[lo2-d:hi2-d]
    return G
```

This is O(n * n_lags) = O(n^2) with small constant. For n=64, n_lags=127, that's about 8K vector operations, each on arrays of length ≤64. Very fast.

        beta = beta_start * (beta_end / beta_start) ** frac

            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(y_new)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)

            break

        y_tried = y_new
        h_tried = h

                break
            alpha *= 0.5
        else:

        if _exact_c5(h) < best_c5 - 1e-14:
            best_c5 = _exact_c5(h)
            best_h = h.copy()
        y = h
    return best_h, best_c5
