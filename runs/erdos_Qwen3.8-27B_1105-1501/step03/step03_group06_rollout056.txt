∂/∂h[m]:

- m appears in h[i+k] when i+k = m, i.e., i = m-k (if 0 ≤ m-k ≤ n-k-1, i.e., k ≤ m ≤ n-1): contributes -h[m-k]

For k < 0, g = -k (j < n-1):

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5val(h):
    return float(np.max(_corr(h))) * 2.0 / len(h)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(-x)
            for i in idx:
                a = min(res, 1.0 - x[i])
                if a > 0: x[i] += a; res -= a
                if res < 1e-12: break
        else:
            idx = np.argsort(x)
            for i in idx:
                a = min(-res, x[i])
                if a > 0: x[i] -= a; res += a
                if res > -1e-12: break
    return x

def _corr_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            if m <= 0: continue
            G[r, :m] += (1.0 - h[s:m+s])  # (1-h[j+s]) for j=0..m-1
            G[r, s:] += -h[:m]  # -h[j-s] for j=s..n-1
        else:
            g = -s
            m = n - g
            if m <= 0: continue
            G[r, g:] += (1.0 - h[:m])  # (1-h[j-g]) for j=g..n-1
            G[r, :m] += -h[g:]  # -h[j+g] for j=0..m-1
    return G

def _active_set_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5val(h)
    best_h, best_c5 = h.copy(), cur_c5
    
    for iteration in range(1000):
        if time.time() > deadline: break
        
        corr = _corr(h)
        max_corr = corr.max()
        # Find active set: lags within 1% of max
        threshold = max_corr - 0.01 * max_corr
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0: active = np.array([np.argmax(corr)])
        
        # Compute gradient of max (approximate: average of active gradients)
        G = _corr_grad_matrix(h)
        g_avg = G[active].mean(axis=0)
        
        # Project gradient onto feasible directions (null space of sum constraint)
        g_avg = g_avg - g_avg.mean()  # ensure sum = 0
        
        # Line search
        best_a, best_val = 0.0, cur_c5
        for a in [0.01, 0.005, 0.002, 0.001, 0.5, 0.25, 0.1]:
            h_t = _project(h + a * (-g_avg), target)
            c5_t = _c5val(h_t)
            if c5_t < best_val:
                best_val, best_a = c5_t, a
        
        if best_a == 0: break
        h = _project(h - best_a * g_avg, target)
        cur_c5 = _c5val(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()
    
    return best_h, best_c5

def _run_slsqp(h0, t_init, maxiter=200):
    n = len(h0)
    target = n / 2.0
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    # Nonlinear inequality: t - corr_r(h) >= 0 for all r
    def ineq_fun(x):
        h = x[:n]
        return x[-1] - _corr(h)
    
    def ineq_jac(x):
        h = x[:n]
        G = _corr_grad_matrix(h)
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J
    
    # Equality: sum(h) - target = 0
    def eq_fun(x):
        return x[:n].sum() - target
    
    def eq_jac(x):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                   bounds=bounds,

                       {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                   ],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if res.success:
        h_opt = _project(res.x[:n], target)
        return h_opt, _c5val(h_opt)
    return h0, _c5val(h0)

Actually, due to the sum constraint, if I change h_i by δ, I must change h_j by -δ (to maintain the sum). So the 2D problem becomes 1D: find the optimal δ.

For a given δ:
- h_i' = h_i + δ
- h_j' = h_j - δ
- Need 0 ≤ h_i' ≤ 1 and 0 ≤ h_j' ≤ 1

The change in corr_r:
Δcorr_r = (h_i' - h_i) * a_r + (h_j' - h_j) * b_r + (h_i' - h_i)*(h_j' - h_j) * c_r

where a_r, b_r, c_r depend on the other h values and the shift r.

import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5val(h):
    return float(np.max(_corr(h))) * 2.0 / len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(-x)
            for i in idx:
                a = min(res, 1.0 - x[i])
                if a > 1e-15:
                    x[i] += a
                    res -= a
                if res < 1e-12:
                    break
        else:
            idx = np.argsort(x)
            for i in idx:
                a = min(-res, x[i])
                if a > 1e-15:
                    x[i] -= a
                    res += a
                if res > -1e-12:
                    break
    return x

def _corr_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            G[r, :m] = 1.0 - h[s:m + s]
            G[r, s:] -= h[:m]
        else:
            g = -s
            m = n - g
            if m <= 0:
                continue
            G[r, g:] = 1.0 - h[:m]
            G[r, :m] -= h[g:]
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _epigraph_solve(h0, maxiter=300):
    n = len(h0)
    target = n / 2.0
    t_init = float(np.max(_corr(h0))) + 0.01

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        return x[-1] - _corr(x[:n])

    def ineq_jac(x):
        G = _corr_grad_matrix(x[:n])
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J

    A_lin = np.zeros((1, n + 1))
    A_lin[0, :n] = 1.0
    lin = LinearConstraint(A_lin, target, target)
    nonlin = NonlinearConstraint(ineq_fun, np.zeros(2 * n - 1), np.full(2 * n - 1, np.inf), jac=ineq_jac)

    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='trust-constr',
                       constraints=[lin, nonlin], bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'xtol': 1e-12})
        h_opt = _project(res.x[:n], target)
        return h_opt, _c5val(h_opt)
    except Exception:
        return h0.copy(), _c5val(h0)

def _subgrad_polish(h, deadline, max_iter=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5val(h)
    stall = 0

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break

        corr = _corr(h)
        mx = float(corr.max())
        # Active set: lags within 0.5% of max
        thresh = mx - 0.005 * mx
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        G = _corr_grad_matrix(h)
        g = G[active].mean(axis=0)
        g = g - g.mean()  # project onto sum=0 subspace

        # Backtracking line search
        step = 0.5
        improved = False
        for _bs in range(30):
            h_t = _project(h - step * g, target)
            c5_t = _c5val(h_t)
            if c5_t < best_c5 - 1e-14:
                h = h_t
                best_c5 = c5_t
                best_h = h.copy()
                stall = 0
                improved = True
                break
            step *= 0.5
        if not improved:
            stall += 1

    return best_h, best_c5

def _smooth_lbfgs(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0

    def fun(x):
        s = _corr(x)
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -400.0, 400.0)
        e = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(e)))
        w = e / float(np.sum(e))
        G = _corr_grad_matrix(x)
        g = G.T @ w
        g = g - np.mean(g)
        return f, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': -1})
        x = _project(res.x, target)
    except Exception:
        x = h.copy()
    return x, _c5val(x)

def _binary_sa(h_init, deadline, n_iter=100000):
    n = len(h_init)
    target = int(n / 2)
    # Start from a binary version of h_init
    b = (h_init > 0.5).astype(float)
    # Adjust to have exactly target ones
    if int(b.sum()) != target:
        if int(b.sum()) > target:
            ones = np.where(b == 1)[0]
            np.random.shuffle(ones)
            for i in ones[:int(b.sum()) - target]:
                b[i] = 0
        else:
            zeros = np.where(b == 0)[0]
            np.random.shuffle(zeros)
            for i in zeros[:target - int(b.sum())]:
                b[i] = 1

    cur_corr = _corr(b)
    cur_max = float(cur_corr.max())
    best_b = b.copy()
    best_max = cur_max

    T0 = 2.0
    cooling = 0.99995
    T = T0
    rng = np.random.default_rng(12345)

    for _ in range(n_iter):
        if time.time() > deadline:
            break
        T = max(T * cooling, 0.01)

        # Pick random 1 and random 0
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        b[i] = 0
        b[j] = 1
        new_corr = _corr(b)
        new_max = float(new_corr.max())

        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0

    return best_b, best_max * 2.0 / n

def _pair_line_search(h, i, j, target):
    """Line search along direction e_i - e_j (preserves sum)."""
    n = len(h)
    hi = h[i]
    hj = h[j]
    # Feasible range for delta: hi + d in [0,1], hj - d in [0,1]
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_max - d_min < 1e-14:
        return h.copy(), _c5val(h)

    cur_c5 = _c5val(h)
    best_d = 0.0
    best_c5 = cur_c5
    direction = np.zeros(n)
    direction[i] = 1.0
    direction[j] = -1.0

    # Coarse search
    for frac in np.linspace(0, 1, 21):
        d = d_min + frac * (d_max - d_min)
        if abs(d) < 1e-14:
            continue
        h_t = h + d * direction
        h_t = np.clip(h_t, 0.0, 1.0)
        # Recompute sum to ensure exactness
        if abs(h_t.sum() - target) > 1e-10:
            h_t = _project(h_t, target)
        c5_t = _c5val(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_d = d

    if best_d == 0.0:
        return h.copy(), cur_c5

    # Fine search (golden section)
    lo = max(d_min, best_d - 0.1 * (d_max - d_min))
    hi_ = min(d_max, best_d + 0.1 * (d_max - d_min))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_ - gr * (hi_ - lo)
    c2 = lo + gr * (hi_ - lo)

    def eval_d(d):
        h_t = h + d * direction
        h_t = np.clip(h_t, 0.0, 1.0)
        if abs(h_t.sum() - target) > 1e-10:
            h_t = _project(h_t, target)
        return _c5val(h_t)

    f1 = eval_d(c1)
    f2 = eval_d(c2)
    for _ in range(30):
        if hi_ - lo < 1e-12:
            break
        if f1 < f2:
            hi_ = c2
            c2 = c1
            f2 = f1
            c1 = hi_ - gr * (hi_ - lo)
            f1 = eval_d(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi_ - lo)
            f2 = eval_d(c2)

    d_best = (lo + hi_) / 2.0
    h_best = h + d_best * direction
    h_best = np.clip(h_best, 0.0, 1.0)
    if abs(h_best.sum() - target) > 1e-10:
        h_best = _project(h_best, target)
    c5_best = _c5val(h_best)
    return h_best, c5_best

def _pair_cd_polish(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5val(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng_local = np.random.default_rng(_pass + 7)
        pairs = []
        for i in range(n):
            j = (_pass + i * 7 + 3) % n
            if j != i:
                pairs.append((i, j))
        # Also add some random pairs
        for _ in range(min(20, n)):
            a = rng_local.integers(n)
            b_ = rng_local.integers(n)
            if a != b_:
                pairs.append((a, b_))

        for (i, j) in pairs:
            if time.time() > deadline:
                break
            h_new, c5_new = _pair_line_search(h, i, j, target)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
        if not improved:
            break

    return best_h, best_c5

def _lp_polish(h, deadline, max_iters=2000):
    """LP-based trust region polish using exact gradients."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5val(h)
    radius = 0.1
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break

        corr = _corr(h)
        mx = float(corr.max())
        thresh = mx - 0.01 * mx
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        G = _corr_grad_matrix(h)
        # LP: min radius_s.t. G[active] @ d <= -1 for all active, |d| <= radius, sum(d) = 0
        # Simplified: just use the average gradient direction
        g = G[active].mean(axis=0)
        g = g - g.mean()
        norm = float(np.max(np.abs(g)))
        if norm < 1e-14:
            stall += 10
            continue
        d = -g / norm * radius

        # Line search
        best_a, best_c5_try = 0.0, best_c5
        for a in np.linspace(0, 1, 16):
            h_t = _project(h + a * d, target)
            c5_t = _c5val(h_t)
            if c5_t < best_c5_try:
                best_c5_try = c5_t
                best_a = a

        if best_a > 0 and best_c5_try < best_c5 - 1e-14:
            h = _project(h + best_a * d, target)
            best_c5 = _c5val(h)
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.3)
        else:
            stall += 1
            radius *= 0.7

    return best_h, best_c5
