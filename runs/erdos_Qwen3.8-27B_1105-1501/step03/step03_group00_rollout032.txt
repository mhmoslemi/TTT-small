def _smooth_grad(h, scores, tau):
    n = len(h)
    sh = (scores - scores.max()) / tau
    mx = sh.max()
    w = np.exp(sh - mx)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(scores)
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    dx = 2.0 / n
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wk = w[ki]
        if wk < 1e-15:
            continue
        jlo = max(0, k)
        jhi = min(n, n + k)
        if jhi > jlo:
            grad[jlo:jhi] += wk * (1.0 - h[jlo - k:jhi - k])
        jlo2 = max(0, -k)
        jhi2 = min(n, n - k)
        if jhi2 > jlo2:
            grad[jlo2:jhi2] -= wk * h[jlo2 + k:jhi2 + k]
    return grad * dx

def _smooth_descent(h, deadline, tau_list=None):
    if tau_list is None:
        tau_list = [0.08, 0.04, 0.02, 0.01, 0.005]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = _exact_c5(h)
    best_h = h.copy()
    cur = best
    for tau in tau_list:
        if time.time() > deadline:
            break
        for step in range(500):
            if time.time() > deadline:
                break
            scores = _scores(h)
            g = _smooth_grad(h, scores, tau)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            alpha = 0.15
            improved = False
            for _ in range(30):
                h_try = _project(h - alpha * g, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur - 1e-14:
                    h = h_try
                    cur = c5_try
                    if cur < best:
                        best = cur
                        best_h = h.copy()
                    improved = True
                    break
                alpha *= 0.65
            if not improved:
                break
    return best_h, best

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_dir(h, scores, G, radius, top_k=None):
    n = len(h)
    nl = len(scores)
    if top_k is not None and top_k < nl:
        idx = np.argsort(scores)[-top_k:]
        sk = scores[idx]
        Gk = G[idx]
    else:
        sk = scores
        Gk = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n + 1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_c5(h, p, cur_c5):
    n = len(h)
    a_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = max(a_max, 1e-12)
    alphas = np.linspace(0, a_max, 20)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    bi = int(np.argmin(vals))
    if bi == 0:
        return 0.0, cur_c5
    if bi == len(alphas) - 1:
        return alphas[-1], vals[-1]
    lo = alphas[bi - 1]
    hi = alphas[bi + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < vals[bi]:
        return opt_a, opt_c5
    return alphas[bi], vals[bi]

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur = _exact_c5(h)
    best = cur
    best_h = h.copy()
    stall = 0
    ti = 0
    nt = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % nt]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_dir(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_c5(h, p, cur)
        if new_c5 < cur - 1e-13:
            h = _project(h + alpha * p, target)
            cur = _exact_c5(h)
            if cur < best:
                best = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.4)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.06
    return best_h, best

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_starts(h0, n0):
    starts = [h0.copy()]
    comp = 1.0 - h0
    starts.append(comp.copy())
    rev = h0[::-1].copy()
    starts.append(rev)
    for shift in [1, 2, 3]:
        sh = np.roll(h0, shift)
        starts.append(sh)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            n0 = 64
            h0 = np.full(n0, 0.5)
            h0[:n0 // 2] = 1.0
            h0 = _project(h0, n0 / 2.0)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    best_n = n0

    def elapsed():
        return time.time() - t0

    def remaining():
        return max(0, budget_s - elapsed())

    # Phase 1: Smooth gradient descent (quick improvement)
    dl1 = t0 + budget_s * 0.06
    h_s, c5_s = _smooth_descent(best_h, dl1, tau_list=[0.06, 0.03, 0.015, 0.008])
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 2: Coordinate descent (find local min at current resolution)
    dl2 = t0 + budget_s * 0.18
    h_cd, c5_cd = _cd_full(best_h, dl2, passes=4)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: Simulated annealing (global exploration)
    dl3 = t0 + budget_s * 0.55
    h_an, c5_an = _anneal(best_h, dl3, rng, T_start=0.006, T_end=5e-6)
    if c5_an < best_c5:
        best_c5 = c5_an
        best_h = h_an.copy()

    # Phase 4: Coordinate descent after annealing
    dl4 = t0 + budget_s * 0.65
    h_cd2, c5_cd2 = _cd_full(best_h, dl4, passes=5)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 5: LP optimization (refined local search)
    dl5 = t0 + budget_s * 0.78
    h_lp, c5_lp = _lp_optimize(best_h, dl5, top_k_list=[8, 5, None, 10], radius_init=0.1, max_iters=4000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 6: Restart from structured variants + anneal
    dl6 = t0 + budget_s * 0.88
    variants = _structured_starts(best_h, best_n)
    for vi in range(1, len(variants)):
        if elapsed() > budget_s * 0.87:
            break
        hv = variants[vi]
        hv = _project(hv, best_n / 2.0)
        sub_dl = min(dl6, t0 + budget_s * 0.04)
        h_av, c5_av = _anneal(hv, sub_dl, rng, T_start=0.004, T_end=3e-6)
        if c5_av < best_c5:
            h_cd3, c5_cd3 = _cd_full(h_av, min(dl6, t0 + budget_s * 0.02), passes=2)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
            elif c5_av < best_c5:
                best_c5 = c5_av
                best_h = h_av.copy()

    # Phase 7: Resolution sweep
    dl7 = t0 + budget_s * 0.95
    for dn in [1, -1, 2, -2, 3, -3, 4, -4, 5, 6, 8, -5, -6, -8]:
        if elapsed() > budget_s * 0.94:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, t0 + budget_s * 0.018)
        h_r2, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[10, None], radius_init=0.12, max_iters=500)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r2.copy()
            best_n = n_try

    # Phase 8: Final ultra-fine polish
    dl8 = t0 + budget_s * 0.995
    h_f, c5_f = _cd_full(best_h, dl8, passes=6)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        # Fallback shouldn't happen but just in case
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
