    n = len(v)

        return np.zeros(n)

    n = len(h)

    if a_hi - a_lo < 1e-12:

    best_a = 0.0
    best_val = cur_c5

    vals = np.empty(len(alphas))
    for k, a in enumerate(alphas):

            continue
        h_new = h.copy()
        h_new[i] += a
        h_new[j] -= a
        vals[k] = _exact_c5(h_new)

        return 0.0, cur_c5

    left = max(a_lo, alphas[max(0, idx-3)])

        h_new[i] += a
        h_new[j] -= a
        v = _exact_c5(h_new)
        if v < best_refined_val:
            best_refined_val = v
            best_refined_a = a

    best_h = h.copy()
    best_c5 = cur_c5
    target = n / 2.0

        for i in range(n):

def _coord_descent_fast(h, deadline, n_grid=100, max_rounds=200):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_c5 = float(np.max(scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        round_improved = False
        for i in range(n):
            old = h[i]
            if old < 1e-12 or old > 1.0 - 1e-12:
                # At boundary, can only move inward
                pass
            
            # Compute affected lags
            # Lag k (from -(n-1) to n-1) is affected if i+k is in [0,n-1] or i-k is in [0,n-1]
            # i.e., k in [-i, n-1-i] ∪ [i-(n-1), i]
            
            # For k=0: new score = scores[n-1] + (t - old) - (t² - old²)
            # For k≠0: new score = scores[n-1+k] + (t - old) * (coeff_k)
            # where coeff_k = (1-h[i+k]) * [0<=i+k<n] - h[i-k] * [0<=i-k<n]
            
            # Evaluate on grid
            best_t = old
            best_local_c5 = cur_c5
            
            # Generate grid points
            if old < 0.01:
                ts = np.concatenate([np.array([old]), np.linspace(0.01, 1.0, n_grid)])
            elif old > 0.99:
                ts = np.concatenate([np.array([old]), np.linspace(0.0, 0.99, n_grid)])
            else:
                ts = np.linspace(0.0, 1.0, n_grid)
            
            for t in ts:
                if abs(t - old) < 1e-10:
                    continue
                # Compute new max score
                new_max = 0.0
                for lag_idx in range(2*n-1):
                    k = lag_idx - (n-1)
                    s_val = scores[lag_idx]
                    if k == 0:
                        s_val = s_val + (t - old) - (t*t - old*old)
                    else:
                        ck = 0.0
                        if 0 <= i + k < n:
                            ck += (1.0 - h[i+k])
                        if 0 <= i - k < n:
                            ck -= h[i-k]
                        s_val = s_val + (t - old) * ck
                    if s_val > new_max:
                        new_max = s_val
                local_c5 = new_max * dx
                if local_c5 < best_local_c5:
                    best_local_c5 = local_c5
                    best_t = t

                h = _project_box_sum(h, target)

                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                round_improved = True
        if not round_improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coord_descent_fast(h, deadline, n_grid=80):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    for rnd in range(500):
        if time.time() > deadline:
            break
        round_improved = False
        for i in range(n):
            old = h[i]
            if old < 1e-14 or old > 1.0 - 1e-14:
                pass
            ts = np.linspace(0.0, 1.0, n_grid)
            best_t = old
            best_local = cur_c5
            # Pre-compute coefficients for this i
            # For lag k (index lag_idx = n-1+k):
            #   k=0: score becomes scores[n-1] + (t-old) - (t^2 - old^2)
            #   k!=0: score becomes scores[n-1+k] + (t-old)*coeff_k
            #   coeff_k = (1-h[i+k])*[0<=i+k<n] - h[i-k]**[0<=i-k<n]
            coefs = np.zeros(2 * n - 1)
            for lag_idx in range(2 * n - 1):
                k = lag_idx - (n - 1)
                if k == 0:
                    continue
                ck = 0.0
                ipk = i + k
                imk = i - k
                if 0 <= ipk < n:
                    ck += (1.0 - h[ipk])
                if 0 <= imk < n:
                    ck -= h[imk]
                coefs[lag_idx] = ck
            for t in ts:
                if abs(t - old) < 1e-12:
                    continue
                dt = t - old
                # New scores
                new_max = 0.0
                # k=0 special
                s0 = scores[n - 1] + dt - (t * t - old * old)
                if s0 > new_max:
                    new_max = s0
                # k != 0
                for lag_idx in range(2 * n - 1):
                    if lag_idx == n - 1:
                        continue
                    sv = scores[lag_idx] + dt * coefs[lag_idx]
                    if sv > new_max:
                        new_max = sv
                lc5 = new_max * dx
                if lc5 < best_local - 1e-14:
                    best_local = lc5
                    best_t = t
            if abs(best_t - old) > 1e-10:
                h[i] = best_t
                h = _project_box_sum(h, target)
                scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
                cur_c5 = float(np.max(scores) * dx)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                round_improved = True
        if not round_improved:
            break
    return best_h, best_c5

def _greedy_pair_moves(h, deadline, n_alpha=40):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    max_no_improve = 200
    while time.time() < deadline and no_improve < max_no_improve:
        # Find best pair move
        best_overall_a = 0.0
        best_overall_val = cur_c5
        best_overall_ij = (-1, -1)
        # Sample pairs (not all to save time for large n)
        if n <= 80:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = [(i, j) for i in range(0, n, 2) for j in range(i + 1, n, 2)]
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            hi = h[i]
            hj = h[j]
            a_lo = max(-hi, hj - 1.0)
            a_hi = min(1.0 - hi, hj)
            if a_hi - a_lo < 1e-10:
                continue
            # Feasible range from box constraints
            # Precompute D_k for this pair
            D = np.zeros(2 * n - 1)
            for lag_idx in range(2 * n - 1):
                k = lag_idx - (n - 1)
                if k == 0:
                    D[lag_idx] = 2.0 * (hj - hi)  # linear part; quadratic is -2a^2
                    continue
                ci = 0.0
                cj = 0.0
                ipk = i + k
                imk = i - k
                jpk = j + k
                jmk = j - k
                if 0 <= ipk < n:
                    ci += (1.0 - h[ipk])
                if 0 <= imk < n:
                    ci -= h[imk]
                if 0 <= jpk < n:
                    cj += (1.0 - h[jpk])
                if 0 <= jmk < n:
                    cj -= h[jmk]
                D[lag_idx] = ci - cj
            # Evaluate on grid
            alphas = np.linspace(a_lo, a_hi, n_alpha)
            for a in alphas:
                if abs(a) < 1e-12:
                    continue
                # New scores
                new_max = 0.0
                # k=0: scores[n-1] + a*D[k=0] - 2a^2
                s0 = scores[n - 1] + a * D[n - 1] - 2.0 * a * a
                if s0 > new_max:
                    new_max = s0
                for lag_idx in range(2 * n - 1):
                    if lag_idx == n - 1:
                        continue
                    sv = scores[lag_idx] + a * D[lag_idx]
                    if sv > new_max:
                        new_max = sv
                lc5 = new_max * dx
                if lc5 < best_overall_val - 1e-14:
                    best_overall_val = lc5
                    best_overall_a = a
                    best_overall_ij = (i, j)
        
        if best_overall_ij != (-1, -1) and best_overall_val < cur_c5 - 1e-12:
            i, j = best_overall_ij
            h[i] += best_overall_a
            h[j] -= best_overall_a
            h = _project_box_sum(h, target)
            scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
            cur_c5 = float(np.max(scores) * dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def _subgradient_descent(h, deadline, lr_start=0.05, lr_decay=0.95):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_start
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
        max_s = float(np.max(scores))
        active = np.where(scores >= max_s - 1e-10)[0]
        # Compute subgradient (average of gradients of active scores)
        grad = np.zeros(n)
        for lag_idx in active:
            k = lag_idx - (n - 1)
            for i in range(n):
                if k == 0:
                    grad[i] += (1.0 - 2.0 * h[i])
                else:
                    gi = 0.0
                    ipk = i + k
                    imk = i - k
                    if 0 <= ipk < n:
                        gi += (1.0 - h[ipk])
                    if 0 <= imk < n:
                        gi -= h[imk]
                    grad[i] += gi
        grad /= len(active)
        norm = float(np.max(np.abs(grad)))
        if norm < 1e-14:
            break
        # Line search
        best_a = 0.0
        best_val = cur_c5
        for a in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            h_trial = h - a * grad / norm
            h_trial = _project_box_sum(h_trial, target)
            v = _exact_c5(h_trial)
            if v < best_val - 1e-14:
                best_val = v
                best_a = a
        if best_a == 0.0:
            lr *= lr_decay
            if lr < 1e-6:
                break
            continue
        h = h - best_a * grad / norm
        h = _project_box_sum(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            lr = min(lr * 1.1, 0.3)
        else:
            lr *= lr_decay
            if lr < 1e-6:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

    return h

def _anneal_pair_moves(h, deadline, rng, n_alpha=30, temp_start=0.05, temp_decay=0.9):
    """Simulated annealing style pair moves."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = temp_start
    iters = 0
    max_iters = 50000
    stall = 0
    while time.time() < deadline and iters < max_iters and stall < 500:
        iters += 1
        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        a_lo = max(-hi, hj - 1.0)
        a_hi = min(1.0 - hi, hj)
        if a_hi - a_lo < 1e-10:
            stall += 1
            continue
        a = rng.uniform(a_lo, a_hi)
        h[i] += a
        h[j] -= a
        h = _project_box_sum(h, target)
        scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_c5 = float(np.max(scores) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
        else:
            # Revert
            h[i] -= a
            h[j] += a
            h = _project_box_sum(h, target)
            scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
            cur_c5 = float(np.max(scores) * dx)
            stall += 1
        temp *= temp_decay
    return best_h, best_c5

    deadline5 = t0 + budget_s * 0.80
    h5, c5_5 = _subgradient_descent(best_h, deadline5, lr_start=0.03, lr_decay=0.93)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
        # Follow up with coordinate descent
        deadline5b = t0 + budget_s * 0.85
        h5b, c5_5b = _coord_descent_fast(best_h, deadline5b, n_grid=80)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()
    
    # Phase 6: Try different resolutions (12% budget)
    deadline6 = t0 + budget_s * 0.97
    n_candidates = list(set([best_n - 4, best_n, best_n + 4, best_n + 8,
                              best_n - 8, 48, 56, 72, 80, 96, 112, 128]))
    n_candidates = sorted(set(x for x in n_candidates if 32 <= x <= 160))
    per_n_budget = min(budget_s * 0.02, max(5.0, (deadline6 - time.time()) / max(len(n_candidates), 1)))
    for n_try in n_candidates:
        if time.time() > deadline6:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline6, time.time() + per_n_budget)
        # Quick coordinate descent
        h_r, c5_r = _coord_descent_fast(h_r, dl_r, n_grid=60)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Quick pair polish
            if time.time() < deadline6:
                h_rp, c5_rp = _greedy_pair_moves(best_h, min(deadline6, time.time() + per_n_budget * 0.5), n_alpha=30)
                if c5_rp < best_c5:
                    best_c5 = c5_rp
                    best_h = hp.copy() if False else h_rp.copy()
                    best_n = n_try
    
    # Phase 7: Final polish (3% budget)
    if time.time() < t0 + budget_s * 0.99:
        dl_final = t0 + budget_s * 0.995
        hf, c5f = _coord_descent_fast(best_h, dl_final, n_grid=100)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()
            best_n = len(best_h)
    
    # Final projection and validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, best_n
