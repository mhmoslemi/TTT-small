import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coord_optimize(h, j, current_scores):
    """Find optimal h[j] given all others fixed. Returns (new_val, new_scores)."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute A_k = score[k] when h[j]=1 minus score[k] when h[j]=0
    # (all other coordinates same)
    h0 = h.copy()
    h0[j] = 0.0
    h1 = h.copy()
    h1[j] = 1.0
    
    s0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    s1 = np.correlate(h1, 1.0 - h1, mode='full') * dx
    A = s1 - s0  # change per unit change in h[j]
    
    # Current scores: score[k] = s0[k] + h[j] * A[k]
    # New scores with h[j] = x: score_new[k] = s0[k] + x * A[k]
    # Minimize max_k (s0[k] + x * A[k]) over x in [0,1]
    
    # g(x) = max_k (s0[k] + x * A[k])
    # Convex piecewise linear in x
    
    # Quick check: if all A >= 0, optimum is x=0; if all A <= 0, optimum is x=1
    if np.all(A >= -1e-15):
        return 0.0, s0
    if np.all(A <= 1e-15):
        return 1.0, s1
    
    # Grid search + refinement
    xs = np.linspace(0.0, 1.0, 200)
    g_vals = np.max(s0[None, :] + xs[:, None] * A[None, :], axis=1)
    best_i = np.argmin(g_vals)
    best_x = xs[best_i]
    
    # Goldilocks refinement
    if 0 < best_i < len(xs) - 1:
        lo = xs[best_i - 1]
        hi = xs[best_i + 1]
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            g1 = np.max(s0 + m1 * A)
            g2 = np.max(s0 + m2 * A)
            if g1 < g2:
                hi = m2
            else:
                lo = m1
        best_x = (lo + hi) / 2.0
    
    new_scores = s0 + best_x * A
    return best_x, new_scores

def _cd_sweep(h, order=None, deadline=None):

    n = len(h)

        new_val, _ = _coord_optimize(h, j, None)
