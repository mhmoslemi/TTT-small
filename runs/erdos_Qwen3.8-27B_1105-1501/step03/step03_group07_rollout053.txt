The constraint J_d(h) <= T is a quadratic constraint (since J_d is quadratic in h). This is a non-convex feasibility problem in general.

However, we can use a **successive convex approximation** (SCA) approach:
1. Start with a feasible h
2. Linearize J_d(h) around the current point
3. Solve the resulting LP (which is convex)
4. Update h
5. Repeat

This is essentially the tangent LP approach I described earlier.

Let me also think about a **SOCP reformulation**. The constraint J_d(h) <= T can be written as:
sum_i h[i]*(1-h[i+d]) <= T

The term sum_i h[i]*h[i+d] is non-convex (product of variables). But if we fix one of them, it becomes linear.

```python
    dx = 2.0 / n
```

```python
    return float(np.max(corr) * dx)

def _project(vec, target):
    n = len(vec)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    vec = np.clip(np.nan_to_num(vec, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(vec - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(vec - mu, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-14:
        if resid > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, j):
    """Compute gradient of correlation coefficient j with respect to h.
    j is the index into np.correlate(h, 1-h, 'full'), ranging 0..2n-2.
    offset = j - (n-1)
    """
    n = len(h)
    offset = j - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        d = offset
        m = n - d
        if m > 0:
            # C = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # grad[j] = 1 for j in [d, n-1]
            # grad[j] -= h[j-d] for j in [d, n-1]
            # grad[j] -= h[j+d] for j in [0, m-1]
            grad[d:] += 1.0
            grad[d:] -= h[:m]       # h[j-d] for j in [d, n-1]
            grad[:m] -= h[d:]       # h[j+d] for j in [0, m-1]
    else:

        m = n - k
```

```python
def _lp_tangent_step(h, R=1.0, lam=0.001, top_k=None):
    """Solve tangent LP to find descent direction.
    
    Parameters:
    - h: current feasible point
    - R: L1 radius for the step
    - lam: penalty parameter
    - top_k: if set, only use top_k lags (closest to max)
    
    Returns: direction vector d (or None if LP fails)
    """
    n = len(h)
    m_total = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Select which lags to include
    if top_k is not None and top_k < m_total:
        # Use lags with highest scores
        top_indices = np.argsort(-scores)[:top_k]
    else:
        top_indices = np.arange(m_total)
    
    # Build gradient matrix for selected lags
    # G[s, j] = dC_s/dh_j
    G = np.zeros((len(top_indices), n))
    for idx, j in enumerate(top_indices):
        G[idx] = _corr_grad(h, int(j))
    
    # Current scores for selected lags
    J_sel = scores[top_indices]
    
    # LP: minimize t + lam * sum(p_i + q_i)
    # Variables: [t, p_0, q_0, p_1, q_1, ..., p_{n-1}, q_{n-1}]
    # n_vars = 1 + 2*n
    n_vars = 1 + 2 * n
    
    # Objective
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0  # minimize t
    for i in range(n):
        c_obj[1 + 2*i] = lam      # p_i
        c_obj[2 + 2*i] = lam      # q_i
    
    # Inequality constraints: J_s + G_s · d <= t  for each selected s
    # => -t + G_s · p - G_s · q <= -J_s
    A_ub = np.zeros((len(top_indices), n_vars))
    b_ub = np.zeros(len(top_indices))
    for idx, s in enumerate(range(len(top_indices))):
        A_ub[idx, 0] = -1.0
        A_ub[idx, 1::2] = G[s]       # p coefficients
        A_ub[idx, 2::2] = -G[s]      # q coefficients
        b_ub[idx] = -J_sel[idx]
    
    # Equality constraint: sum(d_i) = 0 => sum(p_i) - sum(q_i) = 0
    A_eq_sum = np.zeros((1, n_vars))
    A_eq_sum[0, 1::2] = 1.0
    A_eq_sum[0, 2::2] = -1.0
    
    # L1 bound: sum(p_i + q_i) <= R
    A_uneq_l1 = np.zeros((1, n_vars))
    A_uneq_l1[0, 1:] = 1.0
    
    # Combine inequality constraints
    A_ub_full = np.vstack([A_ub, A_uneq_l1])
    b_ub_full = np.concatenate([b_ub, [R]])
    
    # Combine equality constraints
    A_eq_full = np.array([A_eq_sum[0]])
    b_eq_full = np.array([0.0])
    
    # Bounds
    bounds = []
    bounds.append((-1000.0, 1000.0))  # t is free (bounded for solver)
    for i in range(n):
        # p_i >= 0, upper bound = 1 - h[i] (so h[i] + p_i <= 1)
        # q_i >= 0, upper bound = h[i] (so h[i] - q_i >= 0)
        p_ub = max(0.0, 1.0 - h[i])
        q_ub = max(0.0, h[i])
        bounds.append((0.0, p_ub if p_ub < 1e-10 else None))
        bounds.append((0.0, q_ub if q_ub < 1e-10 else None))
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                      A_eq=A_eq_full, b_eq=b_eq_full,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    # Extract direction
    d = np.zeros(n)
    for i in range(n):
        d[i] = res.x[1 + 2*i] - res.x[2 + 2*i]
    
    # Check if direction is meaningful
    if np.linalg.norm(d) < 1e-10:
        return None
    
    # Renormalize to have L1 norm = R
    l1_norm = np.sum(np.abs(d))
    if l1_norm > 1e-10:
        d = d / l1_norm * R
    
    return d
```

Now for the line search:

```python
def _line_search(h, direction, target, c5_cur, max_alpha=3.0):
    """Line search along direction from h."""
    n = len(h)
    
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), max_alpha)
    
    best_a = 0.0
    best_c = c5_cur
    
    # Try a grid of alphas
    alphas = np.linspace(0.05, alpha_max, 15)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-13:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.7)
    hi = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(25):
        if hi - lo < 1e-12 * max(1, alpha_max):
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c
```

def _lp_optimize(h, deadline, target, max_iters=5000):
    """LP tangent optimization."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    lam = 0.001
    
    for it in range(max_iters):

            break
        if stall > 50:
            break
        
        # Try different configurations
        configs = [
            (R, lam, None),        # full, current R
            (R * 0.5, lam, None),  # smaller step
            (R, lam * 0.1, None),  # less penalty
            (R, lam, 10),          # top 10 lags
            (R, lam, 15),          # top 15
            (R * 2, lam, None),    # larger step
        ]
        
        improved = False

                break
