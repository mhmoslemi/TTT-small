For each pair (i,j), the direction is D = e_i - e_j (move mass from j to i). The line along this direction is h + α(D), which is feasible for α in some interval [α_min, α_max]. At any point on this line, the objective is:

where G[k,j] = d(score_k)/d(h_j).

I can precompute G (the 2n-1 by n matrix) once, then for each pair, I just need to find the max of (score + α * delta) over k, which is a set of linear functions in α. The minimum over α of this max is a 1D piecewise linear minimization problem.

This is actually a small LP in 1D! The optimal α is where the maximum of the linear functions is minimized. I can find it by:
1. For the current α=0, find which k gives the max
2. Increase α in the direction that decreases the active max
3. Find the intersection with the next-highest line
4. Repeat

This is the "line method" for minimizing a max of linear functions. It's O(n) per query (to find the max) and the number of breakpoints is at most O(n) (since there are n-1 such functions... wait, 2n-1 linear functions).

Actually, for a pair (i,j), delta_k = G[k,i] - G[k,j]. The function we're minimizing is:

This is a convex piecewise-linear function in α. The minimum is at a point where two (or more) of the lines are equal and all others are below.

I can find this efficiently:
1. Compute all s_k + α*delta_k at α=0, find the active set
2. The subgradient at α is the convex hull of {delta_k : k active}
3. Move in the direction of -delta_active to reduce the active max
4. Find the next intersection

For practical purposes, I can just do a golden section search on this 1D function, evaluating f(α) in O(n) time (just compute max of s + α*delta). This is much faster than calling np.correlate each time!

This is a key optimization for the pair refinement. Let me implement it:

```python
def pair_line_score(s, delta, alpha):
    return float(np.max(s + alpha * delta))

def pair_line_search(s, delta, alpha_min, alpha_max):
    # Golden section search for min of max(s + alpha*delta)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_min, alpha_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = pair_line_score(s, delta, c1)
    f2 = pair_line_score(s, delta, c2)
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = pair_line_score(s, delta, c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = pair_line_score(s, delta, c2)
    return (a + b) / 2.0
```

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def scores_of(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        d = j - (n - 1)
        m_lo = max(0, d)
        m_hi = min(n - 1, n - 1 + d)
        if m_lo <= m_hi:
            G[j, m_lo:m_hi + 1] += (1.0 - h[m_lo - d:m_hi - d + 1])
        m_lo2 = max(0, -d)
        m_hi2 = min(n - 1, n - 1 - d)
        if m_lo2 <= m_hi2:
            G[j, m_lo2:m_hi2 + 1] += (-h[m_lo2 + d:m_hi2 + d + 1])
    G *= dx
    return G

def soft_obj_grad(h, tau):
    n = len(h)
    s = scores_of(h)
    max_s = float(np.max(s))
    exp_s = np.exp((s - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        m_lo = max(0, d)
        m_hi = min(n - 1, n - 1 + d)
        if m_lo <= m_hi:
            grad[m_lo:m_hi + 1] += wk * (1.0 - h[m_lo - d:m_hi - d + 1])
        m_lo2 = max(0, -d)
        m_hi2 = min(n - 1, n - 1 - d)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] += wk * (-h[m_lo2 + d:m_hi2 + d + 1])
    grad *= (2.0 / n)
    return val, grad

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new / 2.0)

def smooth_descent(h0, deadline, tau_list=None):

def pair_refine(h, deadline, max_passes=15, max_moves_per_pass=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c = exact_c5(h)
    s = scores_of(h)
    G = grad_matrix(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        moves = 0
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                    amax = min((1.0 - h[i]) / di if di > 0 else h[i] / (-di),
                               (1.0 - h[j]) / dj if dj > 0 else h[j]

                            continue

                    # Golden section on f(a) = max(s + a * delta)
                    a_lo, a_hi = 0.0, amax
                    c1 = a_hi - gr * (a_hi - a_lo)
                    c2 = a_lo + gr * (a_hi - a_lo)
                    f1 = float(np.max(s + c1 * delta))
                    f2 = float(np.max(s + c2 * delta))
                    for _ in range(35):

                            break
                        if f1 < f2:
                            a_hi = c2
                            c2 = c1
                            f2 = f1

                            f1 = float(np.max(s + c1 * delta))
                        else:

                            f1 = f2
                            c2 = a_lo + gr * (a_hi - a_lo)
                            f2 = float(np.max(s + c2 * delta))
