```python
def _lp_step(h, radius, all_lags=True):

    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = _grad_matrix(h)
    K = len(corr)  # 2n-1
    
    # Variables: h_new[0..n-1], t
    # Minimize t
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Constraints: corr[s] + G[s,:] @ (h_new - h) <= t
    # => G[s,:] @ h_new - t <= corr[s] - G[s,:] @ h
    # => G[s,:] @ h_new + (-1)*t <= corr[s] - G[s,:] @ h
    A_ub = np.zeros((K, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = corr - G @ h
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: max(0, h_i - radius) <= h_new_i <= min(1, h_i + radius)
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is unbounded
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:n]
        t_val = res.x[n]
        return h_new, t_val
    return None, None
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            order = np.argsort(-x)
            for idx in order:
                add = min(res, 1.0 - x[idx])
                x[idx] += add
                res -= add
                if abs(res) < 1e-12:
                    break
        else:
            order = np.argsort(x)
            for idx in order:
                sub = min(-res, x[idx])
                x[idx] -= sub
                res += sub
                if abs(res) < 1e-12:
                    break
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:d + m] += -h[:m]
        else:
            g = -d
            G[r, g:g + m] += (1.0 - h[:m])
            G[r, :m] += -h[g:g + m]
    return G


def _smooth_obj(x, alpha, target):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _smooth_step(h, alpha, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h.copy(), args=(alpha, target),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _smooth_loop(h, deadline, alphas=None):

        alphas = (5.0, 10.0, 20.0, 40.0, 80.0, 120.0, 200.0)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_new, c5_new = _smooth_step(h, alpha, maxiter=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
```

```python
        b_new[i], b_new[j] = 1.0 - b_new[i], 1.0 - b_new[j]
        c5_new = _exact_c5(b_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            b = b_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()

    return best_b, best_c5


def _slsqp_step(h, deadline, maxiter=200):
    """SLSQP on the epigraph formulation."""
    n = len(h)
    target = n / 2.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr0))

    def obj(z):
        h_sub = z[:n]
        t = z[n]
        return t

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def constraints_func(z):
        h_sub = z[:n]
        t = z[n]
        cc = np.correlate(h_sub, 1.0 - h_sub, mode='full')
        return t - cc

    def constraints_grad(z):
        h_sub = z[:n]
        ncc = len(constraints_func(z))
        G = _grad_matrix(h_sub)
        J = np.zeros((ncc, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    cons = [{'type': 'ineq', 'fun': constraints_func, 'jac': constraints_grad}]
    eq_con = [{'type': 'eq',
               'fun': lambda z: np.sum(z[:n]) - target,
               'jac': lambda z: np.append(np.ones(n), 0.0)}]
    bounds = [(0.0, 1.0)] * n + [(t0 * 0.5, t0 * 1.5)]

    z0 = np.append(h.copy(), t0)
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons + eq_con,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        x = res.x[:n]
        x = _project(x, target)
        c5_x = _exact_c5(x)
        if c5_x < _exact_c5(h):
            return x, c5_x
    except Exception:
        pass
    return h.copy(), _exact_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth optimization (20% budget)
    d1 = t0 + budget_s * 0.20
    h, c5 = _smooth_loop(h, d1, alphas=(5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 250.0, 400.0))
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: LP trust region (25% budget)
    d2 = t0 + budget_s * 0.45
    h, c5 = _lp_trust_region(h, d2, radius_init=0.15, max_iters=10000, stall_limit=150)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: SLSQP attempts (10% budget)
    d3 = t0 + budget_s * 0.55
    for _ in range(50):
        if time.time() > d3:
            break
        h_sl, c5_sl = _slsqp_step(best_h, d3, maxiter=100)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl.copy()
            # Follow with LP
            sub_d = min(d3, time.time() + budget_s * 0.03)
            h_lp, c5_lp = _lp_trust_region(h_sl, sub_d, radius_init=0.08,
                                           max_iters=2000, stall_limit=60)
            if c5_lp < best_c5:
                best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: Multi-resolution and restarts (25% budget)
    d4 = t0 + budget_s * 0.80
    scale = 0.04
    for attempt in range(10000):
        if time.time() > d4:
            break
        if best_c5 <= 0.3806:
            break

        strat = attempt % 5
        if strat == 0:
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _project(h_p, best_n / 2.0)
        elif strat == 1:
            # Binary pattern smoothed
            h_p = np.zeros(best_n)
            idx = rng.choice(best_n, best_n // 2, replace=False)
            h_p[idx] = 1.0
            h_p = uniform_filter(h_p, size=3)
            h_p = _project(h_p, best_n / 2.0)
        elif strat == 2:
            # Different resolution
            n_try = best_n + rng.choice(np.array([-8, -4, 4, 8, 12, -12, 16, -16, 24]))
            if 24 <= n_try <= 256:
                h_p = _resample(best_h, n_try)
            else:
                h_p = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
        elif strat == 3:
            # Binary annealing
            h_bin, c5_bin = _binary_search(best_h, best_n, min(d4, time.time() + budget_s * 0.02),
                                           rng, max_iters=3000)
            h_p = h_bin + rng.normal(0, 0.1, best_n)
            h_p = _project(h_p, best_n / 2.0)
        else:
            # Structured perturbation
            h_p = best_h.copy()
            block = rng.integers(4, 16)
            start = rng.integers(0, max(1, best_n - block))
            h_p[start:start + block] = 1.0 - h_p[start:start + block]
            h_p = _project(h_p, best_n / 2.0)

        # Quick smooth step
        h_try, c5_try = _smooth_step(h_p, 50.0, maxiter=80)
        if c5_try < best_c5 - 1e-7:
            # Follow with LP
            sub_d = min(d4, time.time() + budget_s * 0.025)
            h_try, c5_try = _lp_trust_region(h_try, sub_d, radius_init=0.08,
                                             max_iters=1500, stall_limit=50)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                best_n = len(best_h)
                scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.25, scale * 1.12)

    # Phase 5: Binary refinement if close (5% budget)
    if time.time() < t0 + budget_s * 0.88 and best_n >= 32:
        d5 = t0 + budget_s * 0.88
        h_bin, c5_bin = _binary_search(best_h, best_n, d5, rng, max_iters=50000)
        # Smooth the binary result
        h_sm = uniform_filter(h_bin, size=2)
        h_sm = _project(h_sm, best_n / 2.0)
        h_sm, c5_sm = _smooth_step(h_sm, 80.0, maxiter=100)
        if c5_sm < best_c5:
            best_c5, best_h = c5_sm, h_sm.copy()

    # Phase 6: Local moves (5% budget)
    if time.time() < t0 + budget_s * 0.95:
        d6 = t0 + budget_s * 0.95
        hp, c5p = _local_moves(best_h, d6, rng, max_ops=200000)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 7: Final LP polish (5% budget)
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_trust_region(best_h, t0 + budget_s * 0.995,
                                     radius_init=0.04, max_iters=15000, stall_limit=300)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
