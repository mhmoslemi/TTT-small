def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute the gradient matrix G where G[r, j] gives d(scores[r])/d(h[j])."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_gradient(h, p, top_k=None):

    scores = _scores(h)

def _active_equalize_direction(h, active_indices, target_scores):

    k = len(active_indices)

    # with mean(d) = 0 and minimize ||d||
    # This is: minimize d'd + λ*(G@d + 1)^2 ... 
    # Or simpler: minimize ||G[active] @ d - (-ones)||^2 + λ*||d||^2 subject to mean(d)=0
    A = G[active_indices].T  # shape (n, k)

    # Add regularization and mean constraint
    lam = 0.1

    # KKT: [A A^T + lam I, 1; 1^T, 0] [d; μ] = [A b; 0]
    # Wait, A here is k×n (rows of G for active indices)
    # We minimize ||G_act @ d - (-1)||^2 + lam * ||d||^2 s.t. mean(d) = 0
    # = min_d (G_act @ d + 1)^T (G_act @ d + 1) + lam * d^T d - 1^T d
    # gradient: 2 G_act^T (G_act d + 1) + 2 lam d - 1 = 0, with constraint 1^T d = 0
    # (G_act^T G_act + lam I) d = 0.5 * 1 - G_act^T * 1
    # With Lagrange multiplier for constraint:
    # (G_act^T G_act + lam I) d - μ * 1 = 0.5 * 1 - G_act^T * 1
    # 1^T d = 0
    Ga = G[active_indices]  # (k, n)
    rhs = 0.5 * np.ones(n) - Ga.T @ np.ones(k)
    M = Ga.T @ Ga + lam * np.eye(n)
    # Solve with constraint
    # From first eq: d = M^{-1} (rhs + μ * 1)
    # From second: 1^T M^{-1} (rhs + μ * 1) = 0
    # μ = -1^T M^{-1} rhs / (1^T M^{-1} 1)
    try:

    return d

def _line_search(h, p, c5_cur, max_iter=40):

    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    
    # Golden section refinement
    lo = float(alphas[max(0, best_i - 1)])
    hi = float(alphas[min(n_samp - 1, best_i + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c2 = lo + gr * (hi - lo)

def _lp_direction(h, scores, G, radius, top_k=None):

        scores_k = scores
        G_k = G

    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

    except:
        pass
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_start, n, deadline, seed=42):
    """Binary simulated annealing starting from thresholding h_start."""
    rng = np.random.default_rng(seed)
    target = n // 2
    # Initial binary: threshold at 0.5
    b = (h_start[:n] > 0.5).astype(float)
    # Fix count
    n_ones = int(b.sum())
    if n_ones > target:
        # Remove excess ones
        idx_ones = np.where(b == 1)[0]
        rng.shuffle(idx_ones)
        b[idx_ones[target:]] = 0
    elif n_ones < target:
        idx_zeros = np.where(b == 0)[0]
        rng.shuffle(idx_zeros)
        b[idx_zeros[:target - n_ones]] = 1
    
    cur_c5 = _exact_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    
    T = 0.002
    T_min = 1e-6
    cool = 0.995
    max_iter = 50000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a 1 and a 0 to swap
        idx_ones = np.where(b == 1)[0]
        idx_zeros = np.where(b == 0)[0]
        i = idx_ones[rng.integers(len(idx_ones))]
        j = idx_zeros[rng.integers(len(idx_zeros))]
        
        b[i] = 0
        b[j] = 1
        new_c5 = _exact_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
        T = max(T * cool, T_min)
        if T <= T_min * 1.01:
            break
    
    return best_b, best_c5

def _pair_polish(h, deadline, max_passes=8):

                break

                if k < 0 or k >= n:

                    continue

                    else:

                    lo, hi = 0.0, a_max

                        m1 = lo + (hi - lo) / 3.0

                        h1 = h.copy()
                        h1[j] += sign * m1

                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)

                            lo = m1
                    a_opt = (lo + hi) / 2.0

                    if c5_try < best_c5 - 1e-13:

def _structured_perturb(h, rng, scale):
    """Generate structured perturbations (not just Gaussian)."""
    n = len(h)
    strategy = rng.integers(3)
    if strategy == 0:
        # Gaussian noise
        return h + rng.normal(0, scale, n)
    elif strategy == 1:
        # Block perturbation: shift a random block
        block_len = rng.integers(n // 8, n // 3)
        start = rng.integers(0, n - block_len)
        shift = rng.integers(-block_len, block_len + 1)
        h_p = h.copy()
        h_p[start:start+block_len] += shift * scale / block_len
        return h_p
    else:
        # Sinusoidal perturbation
        freq = rng.integers(2, n // 4)
        phase = rng.random() * 2 * np.pi
        x = np.arange(n)
        h_p = h + scale * np.sin(2 * np.pi * freq * x / n + phase)
        return h_p

            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])

        h_f, c5_f = _hybrid_opt(best_h, t0 + budget_s * 0.99, radius_init=0.008, max_iters=15000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
