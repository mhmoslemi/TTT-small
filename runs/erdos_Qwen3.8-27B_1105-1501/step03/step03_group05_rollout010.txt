# Global state for SLSQP constraints
_SLQSTATE = {'n': 0, 'target': 0.0, 'dx': 0.0}

def _obj_slq(x):
    return x[0]

def _cons_ineq_slq(x):
    h_x = x[1:]
    s = np.correlate(h_x, 1.0 - h_x, mode='full') * _SLQSTATE['dx']
    return x[0] - s

def _cons_ineq_slq_jac(x):
    n = _SLQSTATE['n']
    dx = _SLQSTATE['dx']
    h_x = x[1:]
    jac = np.zeros((2*n-1, n+1))
    jac[:, 0] = 1.0
    d_vals = np.arange(-(n-1), n)
    for idx, d in enumerate(d_vals):
        r = (n-1) + idx
        for p in range(max(0, -d), min(n, n-d)):
            # ∂s_r/∂h_{p+d} += (1-h[p])
            jac[r, 1 + (p+d)] += dx * (1.0 - h_x[p])
            # ∂s_r/∂h_p -= h[p+d]
            jac[r, 1 + p] -= dx * h_x[p+d]
    return jac

def _cons_eq_slq(x):
    return np.sum(x[1:]) - _SLQSTATE['target']

def _cons_eq_slq_jac(x):
