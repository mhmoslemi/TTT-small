Then corr = np.correlate(a, b, mode='full').

The derivative of corr[r] with respect to a[j]:
np.correlate(a, b, mode='full')[r] = sum_i a[i] * b[i + (r - (n-1))]  (approximately, with the exact indexing depending on numpy's convention)

Actually, numpy's correlate is defined as:
correlate(a, v, mode='full')[k] = sum_i a[i+k] * conjugate(v[i]) for valid i

So correlate(h, 1-h, mode='full')[k] = sum_i h[i+k] * (1-h[i]) for valid i.

With the "full" mode, k ranges from -(n-1) to n-1, and the output has length 2n-1.
The index in the output array is k + (n-1), so output index r corresponds to k = r - (n-1).

For k = r - (n-1):
corr[r] = sum_i h[i+k] * (1-h[i]) where i and i+k are both in [0, n-1].

So: i ranges from max(0, -k) to min(n-1, n-1-k).

∂corr[r]/∂h[j]:
h[j] appears in the sum when:
- j = i+k (so i = j-k, valid if max(0,-k) ≤ j-k ≤ min(n-1, n-1-k))
- j = i (so h[j] appears in (1-h[i]) with i = j, valid if max(0,-k) ≤ j ≤ min(n-1, n-1-k))

Case 1: j = i+k, i = j-k
Valid if: max(0,-k) ≤ j-k AND j-k ≤ min(n-1, n-1-k)
= j ≥ k AND j-k ≥ 0 AND j-k ≤ n-1 AND j-k ≤ n-1-k
= j ≥ max(0,k) AND j ≤ n-1 AND j ≤ n-1+k
= for k ≥ 0: j ≥ k AND j ≤ n-1
= for k < 0: j ≥ 0 AND j ≤ n-1+k

Contribution: (1 - h[j-k])

Case 2: j = i
Valid if: max(0,-k) ≤ j ≤ min(n-1, n-1-k)
= for k ≥ 0: j ≤ n-1-k (and j ≥ 0)
= for k < 0: j ≥ -k (and j ≤ n-1)

Contribution: -h[j+k]

So:
∂corr[r]/∂h[j] = [j ≥ max(0,k) AND j ≤ n-1+k]* (1-h[j-k]) - [max(0,-k) ≤ j ≤ min(n-1, n-1-k)] * h[j+k]

where k = r - (n-1).

This is correct but complex to vectorize. For n ≤ 256, I'll just loop over r.

Actually, let me think of a simpler approach. Since I need to compute sum_r w_r * ∂corr[r]/∂h[j] for all j, and the gradient for each r has a specific structure, I can compute it as:

grad[j] = sum_over_valid_r_of_case1(w_r * (1-h[j-k_r])) - sum_over_valid_r_of_case2(w_r * h[j+k_r])

def _project(v, target):

    n = len(v)

        if w[r] < 1e-10:
            continue
        k = r - (n - 1)
        # Case 1: j = i+k contributes (1-h[i]) where i = j-k
        # Valid j: j >= max(0,k) and j <= n-1+k (and j in [0,n-1])
        # Contribution to grad[j] from this r: (1-h[j-k]) for valid j

        j_end2 = min(n-1, n-1-k)
        if j_end2 >= j_start2:
            i_vals2 = np.arange(j_start2, j_end2+1)  # i = j
            grad[j_start2:j_end2+1] -= w[r] * h[i_vals2 + k]
    
    obj = T * np.log(wsum)  # scaled by dx later
    return obj, grad

        T = max(T_final, T_init * (T_final/T_init) ** (it / max(1, max_iters)))

        y_new = h + (it / (it + 1 + beta)) * (h - y) if it > 0 else h

        # Let me use a simpler momentum approach

        grad_y = grad_y / n  # Scale gradient
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break
        
        # Step size (Polyak-like)
        # f(h) - f* <= ... but we don't know f*
        # Use decreasing step size
        a = alpha / (1 + 0.01 * it)
        
        h_new = _project(y - a * grad_y, target)
        f_new, _ = _smooth_obj_and_grad(h_new, T)
        
        # Check improvement
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if f_new < f_y + 0.5 * a * gnorm**2:  # descent condition (relaxed)
            y = h
            h = h_new
        else:
            y = h_new  # still update y
            h = h_new

            alpha = min(2.0, alpha * 1.05)
        else:
            alpha *= 0.9

    return best_h, best_c5

def _line_search(h, direction, target, c5_cur, max_alpha=2.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a, best_c = 0.0, c5_cur
    # Coarse search
    for a in np.linspace(0.01, alpha_max, 15):
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c:

            best_a = a
    
    if best_a == 0:
        return 0, c5_cur
    
    # Fine search (golden section)
    lo = max(0, best_a - 0.3 * best_a)
    hi = min(alpha_max, best_a + 0.3 * best_a)
    gr = (np.sqrt(5) - 1) / 2

    opt = (lo + hi) / 2
    opt_c = _exact_c5(_project(h + opt * direction, target))

        return opt, opt_c
    return best_a, best_c

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_softmax(h, T):
    n = len(h)
    corr = _corr_full(h)
    shifted = corr - np.max(corr)
    w = np.exp(shifted / T)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for r in range(len(w)):
        if w[r] < 1e-10:
            continue
        k = r - (n - 1)
        # Term 1: j = i+k, i = j-k
        js1 = max(0, k)
        je1 = min(n-1, n-1+k)
        if je1 >= js1:
            grad[js1:je1+1] += w[r] * (1.0 - h[np.arange(js1, je1+1) - k])
        # Term 2: i = j, contributes -h[j+k]
        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if je2 >= js2:
            grad[js2:je2+1] -= w[r] * h[np.arange(js2, je2+1) + k]
    return grad, corr

def _opt_subgrad(h0, deadline, T_init=0.5, T_min=0.03, lr=1.0, name=""):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    t = 1
    stall = 0
    
    while time.time() < deadline:
        T = max(T_min, T_init * np.exp(-0.003 * t))
        grad, corr = _grad_softmax(y, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_norm = grad / gnorm
        
        # Nesterov
        if t > 1:
            mu = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / 2.0
            y = h + ((t-1) / mu) * (h - y)
        
        h_new = _project(y - lr * grad_norm, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(3.0, lr * 1.03)
        else:
            h = h_new
            stall += 1
            lr *= 0.9
            if stall > 100:
                # Random perturbation to escape
                h = _project(h + 0.05 * np.random.default_rng(t).normal(0, 1, n), target)
                lr = 0.5
                stall = 0
        
        t += 1
    
    return best_h, best_c5

def _opt_line_search(h0, deadline, T=0.2, max_iters=2000):
    """Stochastic line search: pick random direction, line search on exact objective"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(55)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Get a direction from softmax gradient
        grad, _ = _grad_softmax(h, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        # Mix with random
        if rng.random() < 0.3:
            direction = 0.5 * grad + 0.5 * rng.normal(0, 1, n)
            direction -= np.mean(direction)
        else:
            direction = grad.copy()
            direction -= np.mean(direction)
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if direction[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
            elif direction[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-direction[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        best_a, best_c = 0.0, best_c5
        for a in np.linspace(0.05, alpha_max, 10):
            h_t = _project(h + a * direction, target)
            c_t = _exact_c5(h_t)
            if c_t < best_c:
                best_c = c_t
                best_a = a
        
        if best_a > 0:
            # Refine
            lo = max(0, best_a * 0.7)
            hi = min(alpha_max, best_a * 1.3)
            for _ in range(10):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                f1 = _exact_c5(_project(h + m1*direction, target))
                f2 = _exact_c5(_project(h + m2*direction, target))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            opt_a = (lo+hi)/2
            opt_c = _exact_c5(_project(h + opt_a*direction, target))
            if opt_c < best_c:
                best_c, best_a = opt_c, opt_a
            
            h = _project(h + best_a * direction, target)
            if best_c < best_c5:
                best_c5 = best_c
                best_h = h.copy()
            T = max(0.05, T * 0.995)
    
    return best_h, best_c5

def _opt_pairs(h0, deadline, target):
    """Pairwise coordinate descent"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(77)
    max_rounds = 200
    pairs_per_round = 100
    
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        
        # Get gradient for guidance
        grad, _ = _grad_softmax(h, 0.15)
        
        # Select top indices
        top = np.argsort(-np.abs(grad))[:min(20, n)]
        
        for _ in range(pairs_per_round):
            if time.time() > deadline:
                break
            if rng.random() < 0.6:
                i = top[rng.integers(0, len(top))]
                j = top[rng.integers(0, len(top))]
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            if i == j:
                continue
            if h[i] <= 1e-7 or h[i] >= 1-1e-7 or h[j] <= 1e-7 or h[j] >= 1-1e-7:
                continue
            
            for sign in [1.0, -1.0]:
                if sign > 0:
                    amt = min(h[i], 1.0 - h[j])
                else:
                    amt = min(h[j], 1.0 - h[i])
                if amt < 1e-10:
                    continue
                for frac in [0.5, 1.0]:
                    a = amt * frac
                    h_t = h.copy()
                    if sign > 0:
                        h_t[i] -= a
                        h_t[j] += a
                    else:
                        h_t[j] -= a
                        h_t[i] += a
                    c_t = _exact_c5(h_t)

                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        
        if not improved and _round > 10:
            break
    
    return best_h, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
