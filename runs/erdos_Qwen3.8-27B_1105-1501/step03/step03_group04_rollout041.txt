import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.clip(v - tau, 0.0, 1.0)
    # Fix any residual
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (x < 1.0) & (x > 0.5)
        else:
            mask = (x < 0.5) & (x > 0.0)
        if not np.any(mask):
            mask = x < 1.0 if diff > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient matrix G where G[r, j] = d(score_r)/d(h[j])."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    """Solve LP to find best descent direction for max score."""
    n = len(h)

    num_vars = n + 1

    A_ub[:, n] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, num_vars))

def _thue_morse_h(n, amplitude=0.3):
    """Generate Thue-Morse based initial h."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _crossover(h1, h2, rng):
    """Element-wise crossover with random selection."""
    n = len(h1)
    mask = rng.random(n) < 0.5
    h = np.where(mask, h1, h2)
    return _project(h, n / 2.0)

def _mutate(h, rng, scale=0.05):
    h_new = h + rng.normal(0, scale, len(h))
    return _project(h_new, len(h) / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = float(np.min(v)) - 3.0, float(np.max(v)) + 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = 1.0 - 2.0 * h
    for d in range(1, n):
        r = n - 1 + d
        m = n - d
        G[r, d:] += (1.0 - h[:m])
        G[r, :m] -= h[d:]
    for g in range(1, n):
        r = n - 1 - g
        m = n - g
        G[r, :m] += (1.0 - h[g:])
        G[r, g:] -= h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    A_ub = np.zeros((2 * n - 1, nv))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j] + 1e-12)
        hi_j = min(radius, 1.0 - h[j] - 1e-12)
        bounds.append((lo_j, hi_j) if lo_j <= hi_j else (0.0, 0.0))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, n):
    target = n / 2.0
    alpha_max = 2.0
    for j in range(n):
        if abs(p[j]) > 1e-14:
            if p[j] > 0:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            else:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    n_grid = 14
    best_a = 0.0
    best_c5 = c5_cur
    for a in np.linspace(0.02, alpha_max, n_grid):
        c5_t = _exact_c5(_project(h + a * p, target))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a < 0.01:
        return best_a, best_c5
    lo = max(0.0, best_a * 0.5)
    hi = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(22):
        if hi - lo < 1e-11:

        if f1 < f2:

    opt_c5 = _exact_c5(_project(h + opt_a * p, target))

def _softmax_descent(h, deadline, beta_start=20.0, beta_end=5000.0, max_iters=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.12
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(np.clip(beta * (scores - m), -500, 0))
        sw = float(np.sum(exps))
        if sw < 1e-300:
            break
        weights = exps / sw
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(18):

def _thue_morse_h(n, amplitude=0.25):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                h_try[j] = 0.0
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_delta = c5_try - best_c5
                    best_pair = (i, j)
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
                h_try[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Strong LP optimization from parent (30%)
    deadline1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_optimize(h, deadline1, radius_init=0.2, max_iters=20000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Softmax descent + LP (15%)
    deadline2 = t0 + budget_s * 0.45
    h2 = _softmax_descent(best_h.copy(), min(deadline2, time.time() + budget_s * 0.07),
                          beta_start=15.0, beta_end=4000.0, max_iters=5000)
    h2, c5_2 = _lp_optimize(h2, deadline2, radius_init=0.12, max_iters=5000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Structured restarts with diverse inits (20%)
    deadline3 = t0 + budget_s * 0.65
    n_list = sorted(set([best_n, best_n - 2, best_n + 2, best_n - 4,
                         max(16, best_n - 8), min(128, best_n + 8)]))
    scale = 0.08
    attempts = 0
    while time.time() < deadline3 and attempts < 60:
        attempts += 1
        r = rng.random()
        if r < 0.3:
            n_try = n_list[rng.integers(0, len(n_list))]
            h_try = _thue_morse_h(n_try, amplitude=rng.uniform(0.15, 0.35))
            time_sub = max(time.time() + budget_s * 0.005, min(deadline3, time.time() + budget_s * 0.012))
            h_try, c5_try = _lp_optimize(h_try, time_sub, radius_init=0.15, max_iters=3000)
        elif r < 0.6:
            h_try = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
            time_sub = min(deadline3, time.time() + budget_s * 0.008)
            h_try, c5_try = _lp_optimize(h_try, time_sub, radius_init=0.1, max_iters=2000)
        else:
            h_try = _project(best_h + 0.3 * (_thue_morse_h(best_n, 0.2) - 0.5), best_n / 2.0)
            time_sub = min(deadline3, time.time() + budget_s * 0.01)
            h_try, c5_try = _softmax_descent(h_try, time_sub, beta_start=10.0,
                                             beta_end=2000.0, max_iters=2000)
            h_try, c5_try = _lp_optimize(h_try, time_sub, radius_init=0.08, max_iters=1000)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
            scale = max(0.02, scale * 0.85)
        else:
            scale = min(0.3, scale * 1.15)

    # Phase 4: Binary local search for structure (10%)
    deadline4 = t0 + budget_s * 0.75
    h_bin, c5_bin = _binary_local_search(best_h, deadline4, max_rounds=300)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
        best_n = len(h_bin)
        h_bin, c5_bin = _lp_optimize(best_h, deadline4, radius_init=0.1, max_iters=2000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 5: Resolution search (10%)
    deadline5 = t0 + budget_s * 0.88
    for dn in [-6, -4, -2, 2, 4, 6, -8, 8, -10, 10]:
        if time.time() > deadline5:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        time_sub = min(deadline5, time.time() + budget_s * 0.007)
        h_r, c5_r = _lp_optimize(h_r, time_sub, radius_init=0.12, max_iters=2000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final intensive LP polish (7%)
    deadline6 = t0 + budget_s * 0.97
    h_f, c5_f = _lp_optimize(best_h.copy(), deadline6, radius_init=0.08, max_iters=15000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Phase 7: Final micro-polish (3%)
    if time.time() < t0 + budget_s * 0.99:
        h_m, c5_m = _lp_optimize(best_h.copy(), t0 + budget_s * 0.995,
                                 radius_init=0.03, max_iters=5000)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
