    """Compute gradient of soft-max of f_k(h) values."""
    n = len(h)
    scores = _scores(h)
    m = scores.max()
    w = np.exp(beta * (scores - m))
    w /= w.sum()
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for r in range(2*n-1):
        if w[r] < 1e-10:
            continue
        d = r - (n-1)
        if d >= 0:
            # ∂c_d/∂h[j] = (1-h[j-d]) for j in [d, n-1] - h[j+d] for j in [0, n-1-d]
            grad[d:] += w[r] * (1.0 - h[:n-d])
            grad[:n-d] -= w[r] * h[d:]
        else:
            g = -d
            # ∂c_{-g}/∂h[j] = (1-h[j+g]) for j in [0, n-1-g] - h[j-g] for j in [g, n-1]
            grad[:n-g] += w[r] * (1.0 - h[g:])
            grad[g:] -= w[r] * h[:n-g]
    return grad
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, weights):
    """Compute weighted sum of gradients of f_k.
    weights: array of length 2n-1, summing to 1.
    """
    n = len(h)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    # Subtract mean to maintain sum constraint
    grad -= grad.mean()
    return grad

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, direction, c5_cur, n):
    """Exponential bracketing + golden section line search."""
    target = n / 2.0
    # Find bracket
    alpha = 0.1
    best_a = 0.0
    best_c5 = c5_cur
    prev_c5 = c5_cur
    for i in range(20):
        h_try = h + alpha * direction
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_a = alpha
        if c5_try > prev_c5 + 1e-6:
            # Found bracket [prev_alpha, alpha]
            lo_a = alpha / 2.0
            hi_a = alpha
            break
        prev_c5 = c5_try
        alpha *= 1.6
        if alpha > 50:
            break
    else:
        return best_a, best_c5
    
    # Golden section in [lo_a, hi_a]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-10 * max(1, hi_a):
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _topk_weights(scores, topk=5, beta=50.0):
    """Compute weights: uniform over top-k with softmax tail."""
    sorted_idx = np.argsort(scores)[::-1]
    top = sorted_idx[:topk]
    rest = sorted_idx[topk:]
    
    # Uniform weight over top-k
    w = np.zeros(len(scores))
    w[top] = 1.0 / topk
    
    # Small softmax weight over rest
    if len(rest) > 0:
        s_rest = scores[rest]
        m = s_rest.max()
        exps = np.exp(beta * (s_rest - m))
        tail_w = exps / exps.sum() * 0.1  # 10% weight on tail
        w[rest] = tail_w
        w[top] *= 0.9  # 90% on top
    
    return w

def _soft_weights(scores, beta=100.0):
    """Standard softmax weights."""
    m = scores.max()
    exps = np.exp(beta * (scores - m))
    s = exps.sum()
    if s < 1e-300:
        return np.ones(len(scores)) / len(scores)
    return exps / s

def _descent_step(h, c5_cur, beta, topk, momentum, n):
    """Single descent step with momentum."""
    scores = _scores(h)
    if topk > 0:
        w = _topk_weights(scores, topk, beta)
    else:
        w = _soft_weights(scores, beta)
    grad = _grad_weighted(h, w)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h, c5_cur, 0.0, w
    direction = -grad / gnorm
    alpha, new_c5 = _line_search(h, direction, c5_cur, n)
    if alpha > 0:
        h_new = _project(h + alpha * direction, n / 2.0)
        return h_new, new_c5, alpha, w
    return h, c5_cur, 0.0, w

def _momentum_descent(h, deadline, beta_start=50.0, beta_end=2000.0, 
                      topk_start=3, topk_end=2, max_iters=10000):
    """Gradient descent with momentum and adaptive parameters."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)  # momentum
    mu = 0.85  # momentum coefficient
    lr = 0.3
    stall = 0
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        topk = max(1, round(topk_start + (topk_end - topk_start) * frac))
        
        scores = _scores(h)
        if topk > 0:
            w = _topk_weights(scores, topk, beta)
        else:
            w = _soft_weights(scores, beta)
        grad = _grad_weighted(h, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        
        # Momentum
        v = mu * v + (1 - mu) * direction
        v_norm = np.linalg.norm(v)
        if v_norm > 0:
            v = v / v_norm
        
        # Line search with momentum direction
        alpha, new_c5 = _line_search(h, v, cur_c5, n)
        
        if new_c5 < cur_c5 - 1e-13 and alpha > 0:
            h_new = _project(h + alpha * v, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 2.0)
        else:
            # Try without momentum
            alpha2, new_c5_2 = _line_search(h, direction, cur_c5, n)
            if new_c5_2 < cur_c5 - 1e-13 and alpha2 > 0:
                h_new = _project(h + alpha2 * direction, target)
                cur_c5 = _exact_c5(h_new)
                h = h_new
                v = 0.3 * v  # damp momentum
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                mu = max(0.5, mu * 0.97)
                if stall > 80:
                    break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_polish(h, deadline, radius_init=0.15, max_iters=2000):
    """LP-based polish: find direction that reduces all active lags."""
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        
        scores = _scores(h)
        m = scores.max()
        # Active set: lags within 5% of max
        active = np.where(scores >= m - 0.05 * m)[0]
        
        # Build LP: find p with |p| <= radius such that G @ p + scores <= 0
        # where G is the gradient matrix for all scores
        # Simplified: use only active lags
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize t
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        
        for idx, r in enumerate(active):
            d = r - (n - 1)
            # Gradient of f_d at h (without mean subtraction)
            g_row = np.zeros(n)
            if d >= 0:
                mm = n - d
                if mm > 0:
                    g_row[d:] = (1.0 - h[:mm])
                    g_row[:mm] -= h[d:]
            else:
                g = -d
                mm = n - g
                if mm > 0:
                    g_row[:mm] = (1.0 - h[g:])
                    g_row[g:] -= h[:mm]
            A_ub[idx, :n] = g_row
            A_ub[idx, n] = -1.0
            b_ub[idx] = -scores[r]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-12:
                    alpha, new_c5 = _line_search(h, p, cur_c5, n)
                    if new_c5 < cur_c5 - 1e-12:
                        h = _project(h + alpha * p, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
