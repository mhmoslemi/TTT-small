def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_and_grad(h, r):
    """Compute correlation row r and its gradient."""
    n = len(h)
    k = r - (n - 1)
    g = abs(k)
    # C_r = sum_j h[j](1-h[j+g]) for j = 0..n-1-g
    score = 0.0
    for j in range(n - g):
        score += h[j] * (1.0 - h[j + g])
    # Gradient: d[m] = (1-h[m+g])*[m <= n-1-g] - h[m-g]*[m >= g]
    grad = np.zeros(n)
    for m in range(n - g):
        grad[m] += (1.0 - h[m + g])
    for m in range(g, n):
        grad[m] -= h[m - g]
    return score, grad

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    K = 2*n - 1
    
    x0 = np.zeros(n + 1)
    x0[0] = float(np.max(np.correlate(h, 1-h, mode='full'))) * dx
    x0[1:] = h
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[0] = 1.0
        return g
    
    constraints = []
    for r in range(K):
        k = r - (n-1)
        g = abs(k)
        if g == 0:
            def cons(x, g=0, r=r):
                hv = x[1:]
                c = np.sum(hv) - np.sum(hv**2)
                return x[0] - dx * c
            def cons_grad(x, g=0, r=r):
                hv = x[1:]
                dg = np.zeros(n+1)
                dg[0] = 1.0
                dg[1:] = -dx * (1.0 - 2.0*hv)
                return dg
        else:
            def cons(x, g=g, r=r):
                hv = x[1:]
                c = 0.0
                for j in range(n-g):
                    c += hv[j] * (1.0 - hv[j+g])
                return x[0] - dx * c
            def cons_grad(x, g=g, r=r):
                hv = x[1:]
                dg = np.zeros(n+1)
                dg[0] = 1.0

                return dg

        return np.sum(x[1:]) - target

        g = np.zeros(n+1)

        return g

from scipy.optimize import minimize
import cvxpy as cp
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        g = abs(k)
        if g == 0:
            G[r, :] = 1.0 - 2.0 * h
        else:
            G[r, :n-g] += (1.0 - h[g:])
            G[r, g:] += -h[:n-g]
    return G

def _sm_fun(x, alpha, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full') * dx
    ms = float(np.max(scores))
    vals = np.clip(alpha * (scores - ms), -500.0, 500.0)
    ev = np.exp(vals)
    f = ms + np.log(float(np.sum(ev))) / alpha
    w = ev / float(np.sum(ev))
    G = _grad_matrix(x) * dx
    g = G.T @ w
    g -= np.mean(g)  # remove mean for sum constraint
    return f, g

                             method='L-BFGS-B', bounds=[(0,1)]*n,
                             options={'maxiter': 300, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
                h = np.clip(res.x, 0, 1)
            except Exception:

            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    scores = _all_scores(h)
    G = _grad_matrix(h)
    
    h_var = cp.Variable(n)
    T_var = cp.Variable()
    obj = cp.Minimize(T_var + mu * cp.sum_squares(h_var - h))
    cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1, T_var >= 0]
    for r in active_rows:
        lin = scores[r] + G[r, :] @ (h_var - h)
        cons.append(T_var >= dx * lin)
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000)

        try:

        except Exception:
            return None

    h_new = np.clip(np.array(h_var.value).flatten(), 0, 1)
    h_new = _project(h_new, target)
    return h_new

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        g = abs(k)
        if g == 0:
            G[r, :] = 1.0 - 2.0 * h
        else:
            G[r, :n - g] += (1.0 - h[g:])
            G[r, g:] += -h[:n - g]
    return G


def _sm_fun(x, alpha, n):
    dx = 2.0 / n
    scores = np.correlate(x, 1.0 - x, mode='full') * dx
    ms = float(np.max(scores))
    vals = np.clip(alpha * (scores - ms), -500.0, 500.0)
    ev = np.exp(vals)
    f = ms + np.log(float(np.sum(ev))) / alpha
    w = ev / float(np.sum(ev))
    G = _grad_matrix(x) * dx
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _smooth_opt(h, deadline, alphas):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        for _pass in range(2):
            if time.time() > deadline:
                break
            try:
                res = minimize(_sm_fun, h, args=(alpha, n), jac=True,
                               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 300, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
                h = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass
            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def _sqp_solve(h, active_rows, mu):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    scores = _all_scores(h)
    G = _grad_matrix(h)

    h_var = cp.Variable(n)
    T_var = cp.Variable()
    obj = cp.Minimize(T_var + mu * cp.sum_squares(h_var - h))
    cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1, T_var >= 0]
    for r in active_rows:
        lin = scores[r] + G[r, :] @ (h_var - h)
        cons.append(T_var >= dx * lin)
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000)
        except Exception:
            return None
    if h_var.value is None:
        return None
    h_new = np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
    h_new = _project(h_new, target)
    return h_new


def _sqp_loop(h, deadline, mu_init=1.0, max_active=20):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mu = mu_init
    stall = 0

    for it in range(3000):

            break
        if stall > 150:
            break

        scores = _all_scores(h)
        scaled = scores * dx
        ms = float(np.max(scaled))
        thresh = ms * 0.985
        active = set(int(x) for x in np.where(scaled >= thresh)[0])
        top_k = min(max_active, 2 * n - 1)
        top_idx = np.argsort(scaled[-1 * top_k:]) if False else np.argsort(scaled)
        top_idx = top_idx[-top_k:]
        for idx in top_idx:
            active.add(int(idx))
        active = sorted(active)

        h_new = _sqp_solve(h, active, mu)

def _penalty_fun(x, n, dx, lam):
    h = x
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    ms = float(np.max(scores))
    sq_pen = 0.0
    for r in range(len(scores)):
        v = scores[r] - ms
        if v > 0:
            sq_pen += v * v
    return ms + lam * sq_pen


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-7 or h[k] >= 1.0 - 1e-7:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for frac in (0.5, 1.0, 0.25, 0.75):
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_h, best_c5


def _coordinate_polish(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = [j for j in range(n) if 1e-6 < h[j] < 1.0 - 1e-6]

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        any_improve = False
        for j in interior:
            if time.time() > deadline:
                break
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            for direction in (1.0, -1.0):
                a_max = (1.0 - h[j]) if direction > 0 else h[j]
                if a_max < 1e-10:
                    continue
                # Try moving j by delta, distribute -delta to minimize max
                # Simple: split among all other interior points
                others = [k for k in range(n) if k != j and 1e-6 < h[k] < 1.0 - 1e-6]
                if len(others) == 0:
                    continue
                for frac in (0.1, 0.3, 0.5, 0.7, 1.0):
                    delta = direction * a_max * frac / len(others)
                    h_t = h.copy()
                    h_t[j] += direction * a_max * frac
                    for k in others:
                        h_t[k] -= delta
                    h_t = np.clip(h_t, 0.0, 1.0)
                    h_t = _project(h_t, target)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t
                        any_improve = True
                        break
        if not any_improve:
            break

    return best_h, best_c5

        h = np.zeros(n)
        h[x < 0.5] = 1.0
    else:
