def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _gnorm(h):
    """Compute gradient matrix G where G[k,j] = d_scores_k / d_h_j (includes dx factor)."""
    n = len(h)

    G = np.zeros((2 * n - 1, n))

            m = n - g

                G[k, :m] += -h[g:n]
    return G * dx

def _pnorm_direction(h, p):

    n = len(h)
    s = _scores(h)
    s_pos = np.maximum(s, 1e-15)

    ws = w.sum()

        w = np.ones(len(s)) / len(s)
    else:
        w /= ws
    G = _gnorm(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _line_search_exact(h, p, c5_cur, n_coarse=20):

    n = len(h)
    # Find max feasible step

    vals = np.array([_exact_c5(h + a * p) for a in alphas])

    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])

        return best_a, best_c5
    
    # Golden section refinement
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)

        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
