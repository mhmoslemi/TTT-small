def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        else:

    # d_i = p_i - q_i
    # Objective: minimize t + lam * sum(p + q)
    nv = 1 + 2 * n
    c = np.zeros(nv)
    c[0] = 1.0  # minimize t
    c[1:] = lam  # penalty on L1 norm of d
    
    # Constraints: for each active lag s:
    # raw[s] + grad[s] · d <= t
    # => t - grad[s] · (p - q) >= raw[s]
    # => -t + grad[s] · p - grad[s] · q <= -raw[s]
    A_ub = np.zeros((len(active), nv))
    b_ub = np.zeros(len(active))

        A_ub[i, 0] = -1.0
        g = grads[s]
        A_ub[i, 1:1+n] = g       # grad · p
        A_ub[i, 1+n:1+2*n] = -g  # -grad · q
        b_ub[i] = -raw[s]
    
    # L1 norm constraint: sum(p + q) <= rstep
    A_ub = np.vstack([A_ub, np.zeros((1, nv))])
    b_ub = np.append(b_ub, rstep)
    A_ub[-1, 1:] = 1.0
    
    # Sum constraint: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)

    bounds = [(-1e6, 1e6)]  # t

    dnorm = float(np.max(np.abs(d)))

def _project(v, target):
    n = len(v)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-11:
        if res > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _offset_data(h):
    n = len(h)
    m = 2 * n - 1
    raw = np.zeros(m)
    grads = np.zeros((m, n))
    for idx in range(m):
        d = idx - (n - 1)
        if d >= 0:
            length = n - d
            if length <= 0:
                continue
            raw[idx] = float(np.sum(h[:length] * (1.0 - h[d:d+length])))
            grads[idx, :length] += (1.0 - h[d:d+length])
            grads[idx, d:d+length] -= h[:length]
        else:
            l = -d
            length = n - l
            if length <= 0:
                continue
            raw[idx] = float(np.sum(h[l:l+length] * (1.0 - h[:length])))
            grads[idx, l:l+length] += (1.0 - h[:length])
            grads[idx, :length] -= h[l:l+length]
    return raw, grads

def _lp_direction(h, raw, grads, rstep, lam, gate, max_active=30):
    n = len(h)
    m = len(raw)
    raw_max = float(np.max(raw))
    threshold = raw_max - gate
    active = np.where(raw >= threshold)[0]
    if len(active) > max_active:
        order = np.argsort(-raw[active])[:max_active]
        active = active[order]
    if len(active) == 0:
        active = np.array([int(np.argmax(raw))])
    
    nv = 1 + 2 * n
    c = np.zeros(nv)
    c[0] = 1.0
    c[1:] = lam
    
    n_act = len(active)
    A_ub = np.zeros((n_act + 1, nv))
    b_ub = np.zeros(n_act + 1)
    
    for i, s in enumerate(active):
        A_ub[i, 0] = -1.0
        g = grads[s]
        A_ub[i, 1:1+n] = g
        A_ub[i, 1+n:1+2*n] = -g
        b_ub[i] = -raw[s]
    
    A_ub[n_act, 1:] = 1.0
    b_ub[n_act] = rstep
    
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(-500.0, 500.0)]
    for i in range(n):
        if h[i] < 1e-6:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif h[i] > 1.0 - 1e-6:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    x = res.x
    p = x[1:1+n]
    q = x[1+n:1+2*n]
    d = p - q
    d -= np.mean(d)
    dnorm = float(np.max(np.abs(d)))
    if dnorm < 1e-10:
        return None
    return d / dnorm

def _line_search(h, direction, target, c5_cur, alpha_limit):
    n = len(h)
    alpha_max = alpha_limit
    for j in range(n):
        if direction[j] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-12:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), 5.0)
    
    best_a = 0.0
    best_c = c5_cur
    alphas = np.linspace(0.02, alpha_max, 8)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-11:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    lo = max(0.0, best_a * 0.5)
    hi = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(22):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _lp_optimize(h, deadline, target, max_iters=300):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    configs = []
    for gate in [0.01, 0.03, 0.08, 0.15, 0.30]:
        for rstep in [0.04 * n, 0.1 * n, 0.25 * n, 0.6 * n]:
            for lam in [0.0, 0.005, 0.05]:
                configs.append((gate, rstep, lam))
    
    cfg_idx = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 20:
            break
        
        offsets_raw, offsets_grads = _offset_data(h)
        improved = False
        
        # Try a limited number of configs per iteration
        n_try_cfgs = min(len(configs), 6)
        for ci in range(n_try_cfgs):
            idx = (cfg_idx + ci) % len(configs)
            gate, rstep, lam = configs[idx]
            
            for ma in [15, 30]:
                d = _lp_direction(h, offsets_raw, offsets_grads, rstep, lam, gate, max_active=ma)
                if d is None:
                    continue
                for sign in [1.0, -1.0]:
                    a, c_t = _line_search(h, sign * d, target, cur_c5, rstep)
                    if a > 0 and c_t < best_c5 - 1e-10:
                        h = _project(h + a * sign * d, target)
                        cur_c5 = c_t
                        stall = 0
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            cfg_idx = (cfg_idx + n_try_cfgs) % len(configs)
        else:
            stall += 1
            # Subgradient fallback
            raw_max = float(np.max(offsets_raw))
            T = max(0.05, raw_max * 0.08)
            w = np.exp((offsets_raw - raw_max) / T)
            wsum = float(np.sum(w))
            if wsum > 1e-10:
                w /= wsum
                grad = np.zeros(n)
                for r in range(len(offsets_raw)):
                    if w[r] > 1e-6:
                        grad += w[r] * offsets_grads[r]
                grad -= np.mean(grad)
                gnorm = float(np.linalg.norm(grad))
                if gnorm > 1e-10:
                    d = grad / gnorm
                    a, c_t = _line_search(h, d, target, cur_c5, 0.2 * n)
                    if a > 0 and c_t < best_c5 - 1e-10:
                        h = _project(h + a * d, target)
                        cur_c5 = c_t
                        stall = 0
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, target, n_trials=5000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5 or h[k] <= 1e-5 or h[k] >= 1.0 - 1e-5:
            continue
        for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
            a_max = min((1.0 - h[j]) if dj > 0 else h[j],
                        (1.0 - h[k]) if dk > 0 else h[k])
            if a_max < 1e-7:
                continue
            for frac in [0.4, 0.8, 1.0]:
                a = a_max * frac
                h_t = h.copy()
                h_t[j] += dj * a
                h_t[k] += dk * a
                c_t = _exact_c5(h_t)
                if c_t < best_c5 - 1e-12:
                    best_c5 = c_t
                    best_h = h_t.copy()
                    h = h_t.copy()
                    break
    
    return best_h, best_c5

def _gen_random_feasible(n, rng, base=None):
    r = rng.random()
    if base is not None and r < 0.4:
        v = base + rng.normal(0, 0.06, n)
    elif r < 0.3:
        v = rng.random(n)
    elif r < 0.5:
        v = (rng.random(n) > 0.5).astype(float)
    elif r < 0.7:
        period = max(4, int(n // rng.integers(2, 6)))
        bp = rng.random(period)
        v = np.tile(bp, n // period + 1)[:n]
    elif r < 0.85:
        v = 0.5 + 0.5 * rng.normal(0, 0.3, n)
    else:
        sigma = rng.integers(n // 16 + 1, n // 4 + 1)
        v = rng.normal(0.5, 0.3, n)
        order = np.argsort(v)
        v = np.zeros(n)
        v[order[:n//2]] = 0.0
        v[order[n//2:]] = 1.0
    return _project(v, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP from initial (25%)
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_optimize(h, dl1, target0, max_iters=200)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-start with diverse perturbations (35%)
    dl2 = t0 + budget_s * 0.60
    n_starts = 25
    for s in range(n_starts):
        if time.time() > dl2:
            break
        remaining = dl2 - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 22.0)
        sub_dl = time.time() + alloc
        
        r = rng.random()
        if r < 0.15:
            h_try = _project(best_h + rng.normal(0, 0.04, best_n), best_n / 2.0)
        elif r < 0.3:
            h_try = _gen_random_feasible(best_n, rng, base=best_h)
        elif r < 0.45:
            shift = int(rng.integers(1, best_n // 4 + 1))
            h_try = _project(np.roll(best_h, shift), best_n / 2.0)
        elif r < 0.6:
            idx = rng.choice(best_n, size=best_n // 5, replace=False)
            h_t = best_h.copy()
            h_t[idx] += rng.normal(0, 0.1, len(idx))
            h_try = _project(h_t, best_n / 2.0)
        elif r < 0.75:
            h_try = _gen_random_feasible(best_n, rng, base=None)
        elif r < 0.9:
            # Anti-periodic structure
            half = best_n // 2
            u = 0.5 * (best_h[:half] + best_h[half:2*half]) if 2*half <= best_n else 0.5 * best_h[:half]
            if len(u) * 2 <= best_n:
                h_try = _project(np.tile(u, best_n // len(u) + 1)[:best_n], best_n / 2.0)
            else:
                h_try = _gen_random_feasible(best_n, rng, base=best_h)
        else:
            t = rng.random()
            hr = _gen_random_feasible(best_n, rng, base=None)
            h_try = _project(t * best_h + (1.0 - t) * hr, best_n / 2.0)
        
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, best_n / 2.0, max_iters=60)
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Resolution sweep (20%)
    dl3 = t0 + budget_s * 0.80
    n_try_list = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 12, -12, 16, -16]:
        nt = best_n + dn
        if 24 <= nt <= 160 and nt not in n_try_list:
            n_try_list.append(nt)
    
    for n_try in n_try_list:
        if time.time() > dl3:
            break
        remaining = dl3 - time.time()
        if remaining < 4.0:
            break
        alloc = min(remaining * 0.35, 18.0)
        sub_dl = time.time() + alloc
        
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, max_iters=40)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Final polish (15%)
    dl4 = t0 + budget_s * 0.985
    h4, c5_4 = _lp_optimize(best_h, min(dl4, time.time() + (dl4 - time.time()) * 0.5), best_n / 2.0, max_iters=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    h5, c5_5 = _pair_polish(best_h, dl4, best_n / 2.0, n_trials=8000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
