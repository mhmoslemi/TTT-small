    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_a = opt_a
    return h + best_a * p, best_c5


def _lp_direction(h, sc, radius, k_active=12):
    n = len(h)
    s_max = float(np.max(sc))
    if s_max < 1e-15:
        return None
    band = max(0.06 * s_max, 1e-6)
    thr = s_max - band
    active = np.where(sc >= thr)[0]
    if len(active) > k_active:
        order = np.argsort(sc[active])
        active = active[order[-k_active:]]
    if len(active) == 0:
        active = np.array([np.argmax(sc)])
    n1 = n - 1
    G = np.zeros((len(active), n))
    for i, r in enumerate(active):
        d = int(r) - n1
        G[i, :] = _grad_row(h, d)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -sc[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n].copy()
            p -= np.mean(p)
            if float(np.max(np.abs(p))) > 1e-14:
                return p
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.15, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        sc = _scores(h)
        p = _lp_direction(h, sc, radius, k_active=12)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.82
            if radius < 1e-6:
                break
            continue
        h_new, c5_new = _line_search(h, p, cur_c5)
        if c5_new < cur_c5 - 5e-14:
            h = _project(h_new, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.6)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.06
    return best_h, best_c5


def _pair_optimize(h, i, j):
    n = len(h)
    c = h[i] + h[j]
    u_lo = max(0.0, c - 1.0)
    u_hi = min(1.0, c)
    if u_hi - u_lo < 1e-12:
        return 0.0, 0.0
    n_samp = 25
    best_u = h[i]
    best_val = _exact_c5(h)
    h_work = h.copy()
    for u in np.linspace(u_lo + 1e-9, u_hi - 1e-9, n_samp):
        v = c - u
        h_work[i] = u
        h_work[j] = v
        val = _exact_c5(h_work)
        if val < best_val:
            best_val = val
            best_u = u
    if best_u == h[i]:
        return 0.0, 0.0
    span = (u_hi - u_lo) / n_samp
    lo = max(u_lo, best_u - span * 1.5)
    hi = min(u_hi, best_u + span * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h_work[i] = c1
    h_work[j] = c - c1
    f1 = _exact_c5(h_work)
    h_work[i] = c2
    h_work[j] = c - c2
    f2 = _exact_c5(h_work)
    for _ in range(35):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            h_work[i] = c1
            h_work[j] = c - c1
            f1 = _exact_c5(h_work)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            h_work[i] = c2
            h_work[j] = c - c2
            f2 = _exact_c5(h_work)
    opt_u = (lo + hi) / 2.0
    return opt_u, (lo + hi) / 2.0


def _pair_sweep(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        g, sc, s_max = _subgrad(h)
        idx = np.argsort(-np.abs(g))
        top_k = min(20, n)
        for pi in range(top_k):
            if time.time() > deadline:
                break
            i = idx[pi]
            if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
                continue
            for pj in range(pi + 1, top_k):
                j = idx[pj]
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                    continue
                c = h[i] + h[j]
                u_lo = max(0.0, c - 1.0)
                u_hi = min(1.0, c)
                if u_hi - u_lo < 1e-12:
                    continue
                n_samp = 20
                best_u = h[i]
                best_val = best_c5
                h_work = h.copy()
                for u in np.linspace(u_lo + 1e-9, u_hi - 1e-9, n_samp):
                    v = c - u
                    h_work[i] = u
                    h_work[j] = v
                    val = _exact_c5(h_work)
                    if val < best_val:
                        best_val = val
                        best_u = u
                if best_val < best_c5 - 1e-13:
                    span = (u_hi - u_lo) / n_samp
                    lo = max(u_lo, best_u - span)
                    hi = min(u_hi, best_u + span)
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    h_work[i] = c1
                    h_work[j] = c - c1
                    f1 = _exact_c5(h_work)
                    h_work[i] = c2
                    h_work[j] = c - c2
                    f2 = _exact_c5(h_work)
                    for _ in range(30):
                        if hi - lo < 1e-12:
                            break
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            h_work[i] = c1
                            h_work[j] = c - c1
                            f1 = _exact_c5(h_work)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            h_work[i] = c2
                            h_work[j] = c - c2
                            f2 = _exact_c5(h_work)
                    opt_u = (lo + hi) / 2.0
                    h[i] = opt_u
                    h[j] = c - opt_u
                    new_c5 = _exact_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _subgrad_line_search(h, deadline, n_max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(n_max_iters):
        if time.time() > deadline:
            break
        g, sc, s_max = _subgrad(h, band_frac=0.04, k_max=10)
        d = -g - np.mean(g)
        norm = float(np.sqrt(np.sum(d * d)))
        if norm < 1e-12:
            # Try random direction
            d = np.random.randn(n)
            d -= np.mean(d)
            norm = float(np.sqrt(np.sum(d * d)))
            if norm < 1e-12:
                break
            d = d / norm
        d = d / norm
        h_new, c5_new = _line_search(h, d, best_c5)
        if c5_new < best_c5 - 1e-13:
            h = _project(h_new, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
        else:
            # Try negative direction (ascent then descent)
            h_new2, c5_new2 = _line_search(h, -d, best_c5)
            if c5_new2 < best_c5 - 1e-13:
                h = _project(h_new2, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                # Add random component
                d_rand = np.random.randn(n)
                d_rand -= np.mean(d_rand)
                norm_r = float(np.sqrt(np.sum(d_rand * d_rand)))
                if norm_r > 1e-12:
                    d_rand = d_rand / norm_r
                    h_new3, c5_new3 = _line_search(h, 0.3 * d + 0.7 * d_rand, best_c5)
                    if c5_new3 < best_c5 - 1e-13:
                        h = _project(h_new3, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(64, 0.5)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP loop (main optimization)
    deadline1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_loop(best_h, deadline1, radius_init=0.15, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair coordinate descent
    deadline2 = t0 + budget_s * 0.50
    h2, c5_2 = _pair_sweep(best_h, deadline2, max_passes=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient line search
    deadline3 = t0 + budget_s * 0.62
    h3, c5_3 = _subgrad_line_search(best_h, deadline3, n_max_iters=2000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Another LP pass on improved solution
    deadline4 = t0 + budget_s * 0.72
    h4, c5_4 = _lp_loop(best_h, deadline4, radius_init=0.10, max_iters=4000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Random perturb + quick re-optimize
    deadline5 = t0 + budget_s * 0.85
    for attempt in range(50):
        if time.time() > deadline5:
            break
        if best_c5 < 0.38075:
            break
        # Random perturbation
        noise_scale = rng.uniform(0.005, 0.04)
        pert = np.random.randn(best_n) * noise_scale
        h_pert = _project(best_h + pert, best_n / 2.0)
        # Quick LP polish
        h_try, c5_try = _lp_loop(h_pert, min(deadline5, time.time() + budget_s * 0.015),
                                 radius_init=0.08, max_iters=600)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 6: Resolution exploration
    if time.time() < t0 + budget_s * 0.93:
        deadline6 = t0 + budget_s * 0.95
        for dn in [1, -1, 2, -2, 3, -3, 4, -4]:
            if time.time() > deadline6:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _lp_loop(h_r, min(deadline6, time.time() + budget_s * 0.018),
                                 radius_init=0.08, max_iters=800)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final pair polish
    if time.time() < t0 + budget_s * 0.975:
        deadline7 = t0 + budget_s * 0.98
        h7, c5_7 = _pair_sweep(best_h, deadline7, max_passes=15)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final LP polish
    if time.time() < t0 + budget_s * 0.995:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.998,
                             radius_init=0.03, max_iters=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
