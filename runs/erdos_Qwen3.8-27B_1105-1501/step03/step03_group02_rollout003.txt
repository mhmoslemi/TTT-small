def _best_pair_direction(h, scores):
    """Find the best pair (i,j) for transfer based on gradient."""
    n = len(h)
    # Find active lag(s)
    max_idx = int(np.argmax(scores))
    s_star = max_idx - (n - 1)
    grad = _score_grad(h, s_star)
    
    # For each pair (i, j), the directional derivative along (e_i - e_j) is:
    # grad[i] - grad[j] (times dx, but we can ignore the constant)
    # We want to minimize this, so we want grad[j] - grad[i] to be large
    # i.e., transfer from j (high grad) to i (low grad)
    
    # Sort by gradient
    order = np.argsort(grad)
    # Best pair: i = order[0] (lowest grad), j = order[-1] (highest grad)
    return order[0], order[-1]

    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))

        s = k - (n - 1)
        grad += w[k] * _score_grad(h, s)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        if m > 0:
            g[:m] -= h[s:]
            g[s:] += (1.0 - h[:m])
    else:
        ga = -s
        m = n - ga
        if m > 0:
            g[:m] += (1.0 - h[ga:])
            g[ga:] -= h[:m]
    return g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_grad(h, scores, beta):
    n = len(h)
    mx = float(np.max(scores))
    shifted = scores - mx
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        return _score_grad(h, int(idx - (n - 1)))
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-18:
            s = int(k - (n - 1))
            grad += w[k] * _score_grad(h, s)
    return grad

def _c5_after_transfer(h, i, j, t):
    old_i, old_j = h[i], h[j]
    h[i] = old_i + t
    h[j] = old_j - t
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    h[i] = old_i
    h[j] = old_j
    return val

def _golden_section_1d(f, a, b, max_iter=50, tol=1e-11):
    if b - a < tol:
        m = (a + b) / 2.0
        return m, f(m)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    m = (a + b) / 2.0
    return m, f(m)

def _gradient_phase(h, deadline, beta_start=20.0, beta_end=200.0, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best = h.copy()
    best_c5 = cur
    lr = 0.05
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, max_iters - 1))
        grad = _smooth_max_grad(h, scores, beta)
        
        # Project gradient to feasible direction (zero mean)
        grad = grad - np.mean(grad)
        gnorm = float(np.sqrt(np.sum(grad**2)))
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(30):
            h_try = _project(h - alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur - 1e-13:
                h = h_try
                cur = c5_try
                if cur < best_c5:
                    best_c5 = cur
                    best = h.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if improved:
            no_improve = 0
            lr = min(lr * 1.05, 0.5)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 50:
                break
    
    return best, best_c5

def _pair_phase(h, deadline, max_passes=500):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best = h.copy()
    best_c5 = cur
    stall = 0
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Build pair list sorted by gradient promise
        scores = _all_scores(h)
        grad = _smooth_max_grad(h, scores, beta=80.0)
        
        # Generate random pair order
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        np.random.shuffle(pairs)
        
        # Try each pair
        for (i, j) in pairs:
            if time.time() > deadline:
                return best, best_c5
            
            t_max = min(1.0 - h[i], h[j])
            t_min = -min(h[i], 1.0 - h[j])
            if t_max - t_min < 1e-10:
                continue
            
            # Quick evaluation at a few points
            t_test1 = (t_max - t_min) * 0.25 + t_min
            t_test2 = (t_max - t_min) * 0.75 + t_min
            
            f1 = _c5_after_transfer(h, i, j, t_test1)
            f2 = _c5_after_transfer(h, i, j, t_test2)
            
            if min(f1, f2) >= cur - 1e-12:
                continue
            
            # Full golden section search
            def f_t(t):
                return _c5_after_transfer(h, i, j, t)
            
            t_opt, f_opt = _golden_section_1d(f_t, t_min, t_max, max_iter=40)
            
            if f_opt < cur - 1e-13:
                h[i] += t_opt
                h[j] -= t_opt
                h[i] = max(0.0, min(1.0, h[i]))
                h[j] = max(0.0, min(1.0, h[j]))
                cur = f_opt
                improved = True
                if cur < best_c5:
                    best_c5 = cur
                    best = h.copy()
        
        if not improved:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0

def _slsqp_phase(h, deadline, top_k=30, maxiter=200):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-5
    
    active_lags = []
    for k in np.argsort(scores)[::-1][:top_k]:
        active_lags.append(int(k - (n - 1)))
    
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 5.0)]
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = [{'type': 'eq',
             'fun': lambda x: float(np.sum(x[:n])) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    
    for d in active_lags:
        cons.append({'type': 'ineq',
                     'fun': lambda x, dd=d: x[n] - float(np.sum(_all_scores(x[:n])[(n-1+dd):(n-1+dd)+1]) * 1.0),
                     'jac': None})
    
    # Actually, let me use the proper score function
    def _score_at(h_vec, d):
        n_local = len(h_vec)
        if d == 0:
            return float(np.sum(h_vec * (1.0 - h_vec)))
        elif d > 0:
            m = n_local - d
            if m <= 0:
                return 0.0
            return float(np.sum(h_vec[d:d+m] * (1.0 - h_vec[:m])))
        else:
            g = -d
            m = n_local - g
            if m <= 0:
                return 0.0
            return float(np.sum(h_vec[:m] * (1.0 - h_vec[g:g+m])))
    
    cons = [{'type': 'eq',
             'fun': lambda x: float(np.sum(x[:n])) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    
    for d in active_lags:
        cons.append({'type': 'ineq',
                     'fun': lambda x, dd=d: x[n] - _score_at(x[:n], dd),
                     'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])})
    
    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return h, _exact_c5(h)
    
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-6:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5
