    grad -= np.mean(grad)

def _simulated_annealing(h, deadline, rng, t_start=0.04, t_end=0.001):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t_start_time = time.time()
    total = max(deadline - t_start_time, 1.0)
    while time.time() < deadline:
        frac = min((time.time() - t_start_time) / total, 1.0)
        temp = t_start * (t_end / t_start) ** frac
        move_type = rng.random()
        if move_type < 0.4:
            scale = temp * 0.8
            h_new = h + rng.normal(0, scale, n)
            h_new = _project(h_new, n / 2.0)
        elif move_type < 0.7:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]
        elif move_type < 0.85:
            n_pert = max(2, int(0.2 * n))
            idx = rng.choice(n, size=n_pert, replace=False)
            h_new = h.copy()
            h_new[idx] += rng.normal(0, temp * 0.5, n_pert)
            h_new = _project(h_new, n / 2.0)
        else:
            h_new = h.copy()
            i = rng.integers(0, n - 1)
            seg_len = rng.integers(2, min(8, n - i))
            h_new[i:i + seg_len] = h[i:i + seg_len][::-1]
            h_new = _project(h_new, n / 2.0)
        c5_new = _exact_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _round_and_refine(h, deadline, rng):
    n = len(h)
    vals = np.sort(h)[::-1]
    threshold = vals[n // 2 - 1] if n % 2 == 0 else vals[n // 2]
    binary = (h >= threshold - 1e-9).astype(float)
    if float(np.sum(binary)) != n / 2:
       idxs = np.argsort(-h)[:n // 2]
        binary = np.zeros(n)
        binary[idxs] = 1.0
    h_bin = binary.copy()
    c5_bin = _exact_c5(h_bin)
    h_cont = h.copy()
    c5_cont = _exact_c5(h_cont)
    if c5_bin < c5_cont:
        h = h_bin
        c5_best = c5_bin
    else:
        c5_best = c5_cont
    # Thaw coordinates closest to 0.5
    dist_to_05 = np.abs(h - 0.5)
    n_thaw = max(n // 4, 8)
    thaw_idx = np.argsort(dist_to_05)[:n_thaw]
    fix_idx = np.setdiff1d(np.arange(n), thaw_idx)
    h_fix = h.copy()
    # Optimize thawed coordinates
    target_thaw = n / 2.0 - float(np.sum(h_fix[fix_idx]))
    h_thaw = h_thaw_init = h[thaw_idx].copy()
    for _it in range(2000):
        if time.time() > deadline:
            break
        if len(thaw_idx) < 2:
            break
        n_th = len(thaw_idx)
        vals = _scores(h_fix)
        top_k = min(5, 2 * n_th)
        top_idx = np.argsort(vals)[-top_k:]
        grad_full = np.zeros(n)
        for s_idx in top_idx:
            s = int(s_idx) - (n - 1)
            if s >= 0:
                m = n - s
                for i in range(m):
                    h_i = h_fix[i]
                    h_is = h_fix[i + s]
                    grad_full[i + s] += (1.0 - h_i)
                    grad_full[i] -= h_is
            else:
                g = -s
                m = n - g
                for i in range(m):
                    h_i = h_fix[i]
                    h_ig = h_fix[i + g]
                    grad_full[i] += (1.0 - h_ig)
                    grad_full[i + g] -= h_i
        grad_thaw = grad_full[thaw_idx] / float(top_k)
        grad_thaw -= np.mean(grad_thaw)
        gnorm = float(np.linalg.norm(grad_thaw))
        if gnorm < 1e-12:
            break
        direction = -grad_thaw / gnorm
        # Compute max step for thawed coords
        alphas = []
        for ii in range(n_th):
            d = direction[ii]
            if d > 1e-15:
                alphas.append((1.0 - h_thaw[ii]) / d)
            elif d < -1e-15:
                alphas.append(h_thaw[ii] / (-d))
        alpha_max = min(alphas) if alphas else 1.0
        alpha_max = max(alpha_max, 1e-8)
        # Simple line search
        best_a = 0.0
        best_c5 = c5_best
        for a in np.geomspace(1e-4, alpha_max, 12):
            h_try = h_fix.copy()
            h_try[thaw_idx] = h_thaw + a * direction
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try[fix_idx] = h_fix[fix_idx]
            h_try = _project(h_try, n / 2.0)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_a = a
        if best_a > 0:
            h_thaw = np.clip(h_thaw + best_a * direction, 0.0, 1.0)
            h_fix[thaw_idx] = h_thaw
            h_fix = _project(h_fix, n / 2.0)
            h_fix[fix_idx] = h[fix_idx]
            cur_c5 = _exact_c5(h_fix)
            if cur_c5 < c5_best:
                c5_best = cur_c5
                h = h_fix.copy()
        else:
            break
    return h, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Gradient descent from parent (35%)
    deadline1 = t0 + budget_s * 0.35
    h, c5 = _gradient_descent(h, deadline1, k=5, max_iters=80000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Binary hill climb (10%)
    deadline2 = t0 + budget_s * 0.45
    h, c5 = _binary_hill_climb(best_h.copy(), deadline2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    # Gradient descent again after binary
    if time.time() < deadline2:
        h, c5 = _gradient_descent(best_h.copy(), deadline2, k=3, max_iters=30000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Simulated annealing (20%)
    deadline3 = t0 + budget_s * 0.65
    h, c5 = _simulated_annealing(best_h.copy(), deadline3, rng, t_start=0.03, t_end=0.0005)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Restart gradient descent with perturbations (15%)
    deadline4 = t0 + budget_s * 0.80
    scale = 0.04
    for attempt in range(500):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        time_left = deadline4 - time.time()
        time_alloc = min(time_left / (500 - attempt + 1), budget_s * 0.012)
        h_try, c5_try = _gradient_descent(h_pert, time.time() + time_alloc, k=5, max_iters=5000)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)

    # Phase 5: Multi-resolution (12%)
    deadline5 = t0 + budget_s * 0.92
    n_candidates = [best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n + 8]
    n_candidates = [x for x in n_candidates if 32 <= x <= 100]
    for n_try in n_candidates:
        if time.time() > deadline5:
            break
        h_r = _resample_h(best_h, n_try)
        c5_r = _exact_c5(h_r)
        h_r, c5_r = _gradient_descent(h_r, min(deadline5, time.time() + budget_s * 0.015), k=5, max_iters=5000)
        h_r, c5_r = _binary_hill_climb(h_r, min(deadline5, time.time() + budget_s * 0.008))
        h_r, c5_r = _gradient_descent(h_r, min(deadline5, time.time() + budget_s * 0.008), k=3, max_iters=3000)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Round and refine (3%)
    if time.time() < t0 + budget_s * 0.95:
        h_rf, c5_rf = _round_and_refine(best_h.copy(), t0 + budget_s * 0.95, rng)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # Phase 7: Final polish with gradient + binary (5%)
    deadline7 = t0 + budget_s * 0.99
    h, c5 = _gradient_descent(best_h.copy(), min(deadline7, time.time() + budget_s * 0.02), k=3, max_iters=20000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < deadline7:
        h, c5 = _binary_hill_climb(best_h.copy(), deadline7)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < deadline7:
        h, c5 = _gradient_descent(best_h.copy(), deadline7, k=5, max_iters=10000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
