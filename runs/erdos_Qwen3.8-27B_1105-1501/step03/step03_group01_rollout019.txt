import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target} using bisection."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def soft_value_grad(h, tau):
    """Compute soft-max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    # Numerical stability: subtract max
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    value = max_s + tau * np.log(Z)
    
    # Gradient: sum_k w[k] * dx * d(corr[k])/d(h[j])
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                # corr[k] = sum_{i=0}^{m-1} h[i+d] * (1-h[i])
                # d/dh[j]:
                # From h[i+d]=h[j] when i=j-d: valid if 0<=j-d<m i.e. d<=j<=d+m-1=n-1
                #   contribution: (1-h[j-d]) * dk
                # From (1-h[i]) when i=j: valid if 0<=j<m
                #   contribution: -h[j+d] * dk
                if d <= n - 1:
                    # First term: j in [d, n-1], contribution (1-h[j-d])
                    grad[d:] += wk * dx * (1.0 - h[:m])  # h[j-d] for j in [d, d+m-1]
                if m > 0:
                    # Second term: j in [0, m-1], contribution -h[j+d]
                    grad[:m] += wk * dx * (-h[d:])
        else:
            g = -d
            m = n - g
            if m > 0:
                # corr[k] = sum_{i=g}^{n-1} h[i-g] * (1-h[i])
                # From h[i-g]=h[j] when i=j+g: valid if g<=j+g<=n-1 i.e. 0<=j<=n-1-g=m-1
                #   contribution: (1-h[j+g])
                # From (1-h[i]) when i=j: valid if g<=j<=n-1
                #   contribution: -h[j-g]
                # First term: j in [0, m-1], contribution (1-h[j+g])
                grad[:m] += wk * dx * (1.0 - h[g:])
                # Second term: j in [g, n-1], contribution -h[j-g]
                grad[g:] += wk * dx * (-h[:m])
    
    return value, grad

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _make_feasible(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    value = max_s + tau * float(np.log(Z))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            if d <= n - 1:
                grad[d:] += wk * dx * (1.0 - h[:m])
            grad[:m] += wk * dx * (-h[d:d + m])
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            grad[:m] += wk * dx * (1.0 - h[g:g + m])
            grad[g:] += wk * dx * (-h[:m])
    return value, grad

def _slsqp_obj(x, tau):
    val, _ = _soft_value_grad(x, tau)
    return val

def _slsqp_grad(x, tau):
    _, g = _soft_value_grad(x, tau)
    return g

def _sum_eq(x, target):

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _make_feasible(h_new, n_new / 2.0)

def _slsqp_run(h_start, target, deadline, tau_list, max_iter=150):
    n = len(h_start)
    h = h_start.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

            break

        if remaining < 2.0:
            break

                           bounds=bounds, constraints=cons, args=(tau,),
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
            if res.success and np.all(np.isfinite(res.x)):

                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_step(h, target, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    best_k = int(np.argmax(scores))
    d = best_k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:d + m]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g + m])
            grad[g:] += -h[:m]
    grad *= dx
    h_new = _make_feasible(h - lr * grad, target)
    return h_new

def _subgrad_loop(h, target, deadline, lr_init=0.02):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    while time.time() < deadline:

        c5_new = _exact_c5(h_new)

            if c5_new < best_c5:

            stall = 0
            lr = min(lr * 1.03, 0.1)
        else:

            lr *= 0.95
            if stall > 80:

def _pair_line_search(h, j, k):
    n = len(h)
    dx = 2.0 / n
    alpha_lo = max(-h[j], h[k] - 1.0)
    alpha_hi = min(1.0 - h[j], h[k])
    if alpha_hi - alpha_lo < 1e-12:
        return 0.0, _exact_c5(h)
    best_a = 0.0
    best_c5 = _exact_c5(h)
    n_samp = 12
    for i in range(1, n_samp + 1):
        a = alpha_lo + (alpha_hi - alpha_lo) * i / (n_samp + 1)
        h_t = h.copy()
        h_t[j] += a
        h_t[k] -= a
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, best_c5
    lo_a = max(alpha_lo, best_a * 0.5)

        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Perturbation + SLSQP restarts (25%)
    d3 = t0 + budget_s * 0.70
    scales = [0.03, 0.05, 0.08, 0.12, 0.18]
    scale_idx = 0
    tau_short = [0.15, 0.06, 0.025, 0.01]
    for attempt in range(60):
        if time.time() > d3:
            break
        if best_c5 < 0.3806:
            break
        scale = scales[min(scale_idx, len(scales) - 1)]
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _make_feasible(h_pert, best_n / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_try, c5_try = _slsqp_run(h_pert, best_n / 2.0, sub_dl, tau_short, max_iter=100)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale_idx = max(0, scale_idx - 1)
        else:
            scale_idx = min(len(scales) - 1, scale_idx + 1)

    # Phase 4: Diverse starts (10%)
    d4 = t0 + budget_s * 0.80
    diverse_starts = []
    for p in [3, 4, 5, 6, 7, 8, 10, 12]:
        if p <= best_n:
            diverse_starts.append((_gen_periodic(best_n, p, seed=seed + p), best_n / 2.0))
    for f in range(3):
        diverse_starts.append((_gen_fourier(best_n, seed=seed + 100 + f), best_n / 2.0))
    for h_div, tgt in diverse_starts:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_try, c5_try = _slsqp_run(h_div, tgt, sub_dl, [0.2, 0.08, 0.03], max_iter=80)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Intensive pair polish (15%)
    d5 = t0 + budget_s * 0.95
    h, c5 = _pair_polish(best_h, d5, max_passes=100)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Final micro-refinement
    if time.time() < hard_deadline:
        h, c5 = _pair_polish(best_h, hard_deadline, max_passes=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final validation
    best_h = _make_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_h = _make_feasible(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
