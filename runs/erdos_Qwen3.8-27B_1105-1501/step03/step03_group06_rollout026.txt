from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_active(h, active_rows):
    """
    Compute gradients for specific correlation rows.
    active_rows: list of row indices into the full correlation array.
    Returns: (G, scores) where G is (len(active_rows), n) and scores is the C values.
    """
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((len(active_rows), n))
    for idx, r in enumerate(active_rows):
        s = r - (n - 1)  # actual lag
        if s >= 0:
            for i in range(n - s):
                G[idx, i] += (1.0 - h[i + s])
                G[idx, i + s] += -h[i]
        else:
            g = -s
            for i in range(n - g):
                G[idx, i] += -h[i + g]
                G[idx, i + g] += (1.0 - h[i])
    return G, scores[active_rows]

def _grad_row(h, r):
    """Compute gradient of correlation row r with respect to h."""

        else:

So the Hessian of S_s is a banded matrix (bandwidth s).

We want to solve: maximize t s.t. S_{s_i}(h) >= t for i=1,...,K
