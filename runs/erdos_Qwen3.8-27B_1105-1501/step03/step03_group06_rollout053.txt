def _smooth_fun(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    # Gradient
    g = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * (1.0 - x[:m])
                g[:m] -= w[r] * x[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += w[r] * (1.0 - x[gg:])
                g[gg:] -= w[r] * x[:m]
    g -= np.mean(g)
    return f, g

def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha,), jac=True,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40, 'iprint': -1})
        x = _project(res.x, n/2.0)
    except Exception:
        x = _project(h, n/2.0)
    return x, _exact_c5(x)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
