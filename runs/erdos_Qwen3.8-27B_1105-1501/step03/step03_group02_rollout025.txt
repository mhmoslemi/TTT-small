    for t in candidates:
        vals = A + B * t + C * t * t
        v = float(np.max(vals))

            best_t = t

        best_t = 0.0

    return x

    return np.correlate(h, 1.0 - h, mode='full')

def _feasible_interval(h, d):
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):

        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))

def _line_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    # lag 0, index n-1
    A[n-1] = float(np.sum(h * (1.0 - h)))
    B[n-1] = float(np.sum(d * (1.0 - 2.0 * h)))
    C[n-1] = -float(np.sum(d * d))
    # positive lags
    for p in range(1, n):
        m = n - p
        r = n - 1 + p
        hp = h[p:p+m]
        hq = h[:m]
        dp = d[p:p+m]
        dq = d[:m]
        A[r] = float(np.sum(hp * (1.0 - hq)))
        B[r] = float(np.sum(dp * (1.0 - hq) - hp * dq))
        C[r] = -float(np.sum(dp * dq))
    # negative lags
    for p in range(1, n):
        m = n - p
        r = n - 1 - p
        hp = h[:m]
        hq = h[p:p+m]
        dp = d[:m]
        dq = d[p:p+m]
        A[r] = float(np.sum(hp * (1.0 - hq)))
        B[r] = float(np.sum(dp * (1.0 - hq) - hp * dq))
        C[r] = -float(np.sum(dp * dq))
    return A, B, C

def _exact_1d_solve(h, d, cur_raw):
    t_lo, t_hi = _feasible_interval(h, d)
    if t_hi - t_lo < 1e-14:
        return 0.0, cur_raw
    A, B, C = _line_coeffs(h, d)
    M = len(A)
    candidates = [t_lo, t_hi, 0.0]
    # Stationary points
    for r in range(M):
        if C[r] < -1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo - 1e-12 <= t0 <= t_hi + 1e-12:
                candidates.append(t0)
    # Pairwise roots
    eps = 1e-14
    for i in range(M):
        for j in range(i+1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < eps:
                if abs(b) > eps:
                    t0 = -c / b
                    if t_lo - 1e-12 <= t0 <= t_hi + 1e-12:
                        candidates.append(t0)
            else:
                disc = b*b - 4.0*a*c
                if disc >= -1e-14:
                    if disc < 0:
                        disc = 0.0
                    sq = np.sqrt(disc)
                    t1 = (-b + sq) / (2.0 * a)
                    t2 = (-b - sq) / (2.0 * a)
                    if t_lo - 1e-12 <= t1 <= t_hi + 1e-12:
                        candidates.append(t1)
                    if t_lo - 1e-12 <= t2 <= t_hi + 1e-12:
                        candidates.append(t2)
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_lo, t_hi)
    if len(candidates) > 1:
        candidates = np.sort(candidates)
        keep = [0]
        for k in range(1, len(candidates)):
            if candidates[k] - candidates[keep[-1]] > 1e-10:
                keep.append(k)
        candidates = candidates[keep]
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        vals = A + B * t + C * t * t
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_t = t
    if best_val >= cur_raw - 1e-12:
        return 0.0, cur_raw
    return best_t, best_val

def _score_grad_all(h):
    """Return matrix G where G[r] is the gradient of score r"""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    # lag 0
    G[n-1, :] = 1.0 - 2.0 * h
    # positive lags
    for p in range(1, n):
        m = n - p
        r = n - 1 + p
        G[r, p:p+m] += -h[:m]
        G[r, :m] += (1.0 - h[p:p+m])
    # negative lags
    for p in range(1, n):
        m = n - p
        r = n - 1 - p
        G[r, :m] += (1.0 - h[p:p+m])
        G[r, p:p+m] += -h[:m]
    return G

def _grad_direction(h, G, beta=50.0):
    """Smooth-max weighted gradient direction"""
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= wsum
    g = G.T @ w
    g -= float(np.mean(g))
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    return -g / norm

def _lp_direction(h, G):
    """LP-based steepest descent direction"""
    n = len(h)
    M = len(G)
    # min t s.t. G @ d - t <= 0, sum(d) = 0, -1 <= d <= 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(M)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            d -= float(np.mean(d))  # ensure zero sum (numerical)
            norm = float(np.max(np.abs(d)))
            if norm > 1e-14:
                return d / norm
    except Exception:
        pass
    return None

def _pair_direction(h, G, scores):
    """Generate a pair direction based on score gradients"""
    n = len(h)

                disc = b * b - 4.0 * a * c
                if disc >= -1e-14:
                    if disc < 0:
                        disc = 0.0
                    sq = np.sqrt(disc)
                    t1 = (-b + sq) / (2.0 * a)
                    t2 = (-b - sq) / (2.0 * a)
                    if t_lo - 1e-12 <= t1 <= t_hi + 1e-12:
                        candidates.append(t1)
                    if t_lo - 1e-12 <= t2 <= t_hi + 1e-12:
                        candidates.append(t2)
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_lo, t_hi)
    if len(candidates) > 1:
        candidates = np.sort(candidates)
        keep = [0]
        for k in range(1, len(candidates)):
            if candidates[k] - candidates[keep[-1]] > 1e-10:
                keep.append(k)
        candidates = candidates[keep]
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        vals = A + B * t + C * t * t
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_t = t
    v0 = float(np.max(A))
    if v0 < best_val:
        best_val = v0
        best_t = 0.0
    if best_val >= cur_raw - 1e-11:
        return 0.0, cur_raw
    return best_t, best_val


def _score_grad_all(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for p in range(1, n):
        m = n - p
        r = n - 1 + p
        G[r, p:p + m] += -h[:m]
        G[r, :m] += (1.0 - h[p:p + m])
    for p in range(1, n):
        m = n - p
        r = n - 1 - p
        G[r, :m] += (1.0 - h[p:p + m])
        G[r, p:p + m] += -h[:m]
    return G


def _grad_direction(h, G, beta):
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= wsum
    g = G.T @ w
    g -= float(np.mean(g))
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    return -g / norm


def _lp_direction(h, G):
    n = len(h)
    M = len(G)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(M)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            d -= float(np.mean(d))
            norm = float(np.max(np.abs(d)))
            if norm > 1e-14:
                return d / norm
    except Exception:
        pass
    return None


def _pair_directions(G, k=8):
    n = G.shape[1]
    D = G[:, :, None] - G[:, None, :]
    pair_max = np.max(D, axis=0)
    np.fill_diagonal(pair_max, np.inf)
    flat_idx = np.argsort(pair_max.max(axis=1) + pair_max.min(axis=1))
    idxs = []
    seen = set()
    for fi in flat_idx[:k * 3]:
        i, j = divmod(int(fi), n)
        if i != j and (i, j) not in seen:
            seen.add((i, j))
            idxs.append((i, j))
            if len(idxs) >= k:
                break
    return idxs


def _random_direction(rng, n):
    d = rng.standard_normal(n)
    d -= float(np.mean(d))
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None
    return d / norm


def _try_direction(h, d, cur_raw, target):
    d -= float(np.mean(d))
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None, cur_raw
    d = d / norm
    t_star, new_raw = _exact_1d_solve(h, d, cur_raw)
    if abs(t_star) <= 1e-12 or new_raw >= cur_raw - 1e-11:
        return None, cur_raw
    h_new = h + t_star * d
    h_new = _repair(h_new, target)
    new_c5 = _c5(h_new)
    cur_c5 = cur_raw * 2.0 / len(h)
    if new_c5 >= cur_c5 - 1e-11:
        return None, cur_raw
    return h_new, new_c5


def _local_descent(h_init, deadline, rng, max_iters=2000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_raw = float(np.max(_scores(h)))
    cur_c5 = cur_raw * 2.0 / n
    G = _score_grad_all(h)
    stall = 0
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        if cur_c5 < 0.3808 - 1e-9:
            break

        scores = _scores(h)
        cur_raw = float(np.max(scores))

        directions = []
        d1 = _grad_direction(h, G, 40.0)
        if d1 is not None:
            directions.append(d1)
        d2 = _grad_direction(h, G, 150.0)
        if d2 is not None:
            directions.append(d2)
        d3 = _lp_direction(h, G)
        if d3 is not None:
            directions.append(d3)

        pairs = _pair_directions(G, k=6)
        for (i, j) in pairs:
            dp = np.zeros(n)
            dp[i] = 1.0
            dp[j] = -1.0
            directions.append(dp)
            dp2 = np.zeros(n)
            dp2[i] = -1.0
            dp2[j] = 1.0
            directions.append(dp2)

        for _ in range(2):
            dr = _random_direction(rng, n)
            if dr is not None:
                directions.append(dr)

        improved = False
        for d in directions:
            if time.time() > deadline:
                break
            h_new, new_c5 = _try_direction(h, d, cur_raw, target)
            if h_new is not None:
                h = h_new
                cur_c5 = new_c5
                cur_raw = float(np.max(_scores(h)))
                G = _score_grad_all(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break

        if not improved:
            stall += 1
            if stall % 10 == 0:
                scale = 0.08 * (0.9 ** (stall // 10))
                h_pert = h + rng.normal(0, scale, n)
                h_pert = _repair(h_pert, target)
                pert_c5 = _c5(h_pert)
                if pert_c5 < cur_c5 - 1e-9:
                    h = h_pert
                    cur_c5 = pert_c5
                    cur_raw = float(np.max(_scores(h)))
                    G = _score_grad_all(h)
                    stall = 0
                else:
                    for _ in range(8):
                        dr = _random_direction(rng, n)
                        h_new2, new_c5_2 = _try_direction(h, dr, cur_raw, target)
                        if h_new2 is not None:
                            h = h_new2
                            cur_c5 = new_c5_2
                            cur_raw = float(np.max(_scores(h)))
                            G = _score_grad_all(h)
                            stall = 0
                            improved = True
                            break
                    if improved and cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(64, 0.5)
    n = len(h0)
    target = n / 2.0
    best_h = _repair(h0, target)
    best_c5 = _c5(best_h)

    # Phase 1: Main local descent from parent
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = _local_descent(best_h, dl1, rng, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Try transforms and population
    dl2 = t0 + budget_s * 0.60
    seeds = [best_h.copy()]
    seeds.append(_repair(1.0 - best_h.copy(), target))
    seeds.append(_repair(best_h[::-1].copy(), target))
    seeds.append(_repair((1.0 - best_h.copy())[::-1], target))
    # Smoothed versions
    w = np.array([0.5, 0.25, 0.125, 0.0625])
    w = w / w.sum()
    smooth = np.convolve(best_h, w, mode='same')
    seeds.append(_repair(smooth, target))
    # Random seeds near best
    for _ in range(3):
        h_r = best_h + rng.normal(0, 0.05, n)
        seeds.append(_repair(h_r, target))

    remaining = max(0, dl2 - time.time())
    per_seed = remaining / max(1, len(seeds))
    for si in range(len(seeds)):
        if time.time() >= dl2:
            break
        sh = seeds[si]
        sc5 = _c5(sh)
        if sc5 > best_c5 + 0.02:
            continue
        per_dl = min(dl2, time.time() + per_seed * 1.5)
        h_try, c5_try = _local_descent(sh, per_dl, rng, max_iters=800)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 3: Iterative multi-restart
    dl3 = t0 + budget_s * 0.90
    attempts = 0
    while time.time() < dl3:
        attempts += 1
        if best_c5 < 0.3808 - 1e-9:
            break
        scale = 0.05 * (0.93 ** min(attempts, 30))
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _repair(h_pert, target)
        per_dl = min(dl3, time.time() + max(5.0, (dl3 - time.time()) * 0.15))
        h_try, c5_try = _local_descent(h_pert, per_dl, rng, max_iters=600)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)

    # Phase 4: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.95:
        dl4 = t0 + budget_s * 0.96
        for dn in [-2, 2, -4, 4, 6, -6]:
            if time.time() > dl4:
                break
            n_try = n + dn
            if n_try < 20 or n_try > 200:
                continue
            h_r = _resample(best_h, n_try)
            per_dl = min(dl4, time.time() + 12.0)
            h_try, c5_try = _local_descent(h_r, per_dl, rng, max_iters=400)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                n = n_try

    # Phase 5: Final polish
    if time.time() < deadline:
        h_f, c5_f = _local_descent(best_h, deadline, rng, max_iters=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _repair(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    final_n = len(best_h)

    return best_h, final_c5, final_n
