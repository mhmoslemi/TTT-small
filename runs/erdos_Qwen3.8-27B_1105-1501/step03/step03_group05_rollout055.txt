def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Binary search for the water level
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_gradient(h):
    """Build gradient matrix G where G[r, m] = dScore[r]/dh[m] * dx"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            # S(d) = sum_{k=d}^{n-1} h[k]*(1-h[k-d])

                if m + d < n and d > 0:

                elif m + d < n and d == 0:

                G[r, m] += dx * grad
        else:

            for m in range(n):

                if m + g < n:
                    grad += (1.0 - h[m + g])
                if m - g >= 0:
                    grad -= h[m - g]
                G[r, m] += dx * grad

    dx = 2.0 / n

def _scores_dx(h):

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            for m in range(n):
                g = 0.0
                if m >= d:
                    g += 1.0 - h[m - d]
                if m + d < n:
                    g -= h[m + d]
                G[r, m] = dx * g
        else:
            g_abs = -d
            for m in range(n):
                g = 0.0
                if m + g_abs < n:
                    g += 1.0 - h[m + g_abs]
                if m - g_abs >= 0:
                    g -= h[m - g_abs]
                G[r, m] = dx * g
    return G

def _lp_direction(h, scores_scaled):
    n = len(h)
    dx = 2.0 / n
    G = _grad_matrix(h)
    # LP: min t s.t. G@x - t <= scores - G@h, 0<=x<=1, sum(x)=n/2
    # y = [x, t], c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    rhs = scores_scaled - G @ h
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 5})
    if res.success:
        return res.x[:n]
    return None

def _binary_swap_eval(h, i, j):

    h[j] = 1.0
