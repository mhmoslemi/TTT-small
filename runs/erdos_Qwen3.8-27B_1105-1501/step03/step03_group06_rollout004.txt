def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _grad_weighted(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += wk * (1.0 - h[d:])
                g[d:] -= wk * h[:m]
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                g[g2:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[g2:]
    return g

def _pair_sweep(h, deadline, max_passes=40):

        if len(interior) < 2:

                break

def _resample(h, n_new):
