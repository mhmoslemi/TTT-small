        p, _ = _lp_direction(h, scores, G, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            continue
        alpha, new_c5 = _line_search_exact(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n, 2):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _active_balance_opt(h, deadline, top_k=5, n_iters=500):
    """Try to equalize the top-k active lags using gradient flow."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        s = _scores(h)
        idx_top = np.argsort(s)[-top_k:]
        s_top = s[idx_top]
        mean_top = np.mean(s_top)
        # Want to reduce variance of top-k scores
        # Gradient direction: sum over top of (s_i - mean) * grad(s_i)
        G = _grad_matrix(h) * (2.0 / n)
        grad = np.zeros(n)
        for ii in range(top_k):
            weight = s_top[ii] - mean_top
            if abs(weight) > 1e-10:
                grad += weight * G[idx_top[ii]]
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        p_dir = -grad / gn
        v = momentum * v + (1.0 - momentum) * p_dir
        v -= np.mean(v)
        alpha = 0.15
        for _ in range(30):
            h_new = _project(h + alpha * v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                stall = 0
                break
            alpha *= 0.6
            if alpha < 1e-12:
                break
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick LP descent (10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_step(h, dl1, radius=0.2, top_k=None, max_iters=1500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Nesterov momentum with p-norm annealing (30%)
    if time.time() < t0 + budget_s * 0.42:
        dl2 = t0 + budget_s * 0.42
        h2, c5_2 = _nesterov_opt(best_h, dl2, p_start=8, p_end=300, n_steps=300, momentum=0.88)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = len(h2)

    # Phase 3: Active-set balance (15%)
    if time.time() < t0 + budget_s * 0.57:
        dl3 = t0 + budget_s * 0.57
        h3, c5_3 = _active_balance_opt(best_h, dl3, top_k=5, n_iters=400)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = len(h3)

    # Phase 4: Structured perturbation restarts (20%)
    if time.time() < t0 + budget_s * 0.77:
        dl4 = t0 + budget_s * 0.77
        # Different perturbation strategies
        strategies = [
            ('gauss', 0.03),
            ('gauss', 0.015),
            ('swap', 0),
            ('gauss', 0.008),
            ('swap', 0),
            ('gauss', 0.02),
        ]
        si = 0
        for attempt in range(300):
            if time.time() > dl4:
                break
            if best_c5 <= 0.3807:
                break
            strat = strategies[si % len(strategies)]
            si += 1
            if strat[0] == 'gauss':
                h_pert = best_h + rng.normal(0, strat[1], best_n)
            else:  # swap: move mass between random pairs
                h_pert = best_h.copy()
                for _ in range(4):
                    a, b = rng.integers(0, best_n, size=2)
                    if a != b:
                        amt = min(h_pert[a], 1.0 - h_pert[b]) * rng.uniform(0.3, 0.8)
                        h_pert[a] -= amt
                        h_pert[b] += amt
                h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.022)
            h_try, c5_try = _nesterov_opt(h_pert, sub_dl, p_start=15, p_end=200, n_steps=120, momentum=0.9)
            if c5_try < best_c5 - 1e-11:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 5: LP refinement on best (15%)
    if time.time() < t0 + budget_s * 0.92:
        dl5 = t0 + budget_s * 0.92
        h5, c5_5 = _lp_step(best_h, dl5, radius=0.04, top_k=8, max_iters=3000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = len(h5)

    # Phase 6: Resolution sweep (5%)
    if time.time() < t0 + budget_s * 0.96:
        dl6 = t0 + budget_s * 0.96
        for dn in (-3, -2, -1, 1, 2, 3, 4, -4):
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 120:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_step(h_r, sub_dl, radius=0.08, top_k=10, max_iters=200)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final pair polish + ultra-fine LP (3%)
    if time.time() < t0 + budget_s * 0.99:
        dl7 = t0 + budget_s * 0.99
        hp, c5p = _pair_polish(best_h, min(dl7, time.time() + budget_s * 0.012), max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
            best_n = len(hp)
        h_f, c5_f = _lp_step(best_h, dl7, radius=0.01, top_k=4, max_iters=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
