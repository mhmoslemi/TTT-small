def _c5_val(h):

def _all_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_transfer_direction(h, i, j, scores, top_k=30):
    """Compute a_d coefficients for pair transfer (i -> j)"""
    n = len(h)
    # Find top k lags
    top_indices = np.argsort(scores)[::-1][:top_k]
    
    a = np.zeros(len(top_indices))
    for idx, k in enumerate(top_indices):
        d = int(k - (n - 1))
        if d >= 0:
            # a_d = h[i+d]*[i<=n-1-d] - (1-h[i-d])*[d<=i<n] - h[j+d]*[j<=n-1-d] + (1-h[j-d])*[d<=j<n]
            val = 0.0
            if i <= n - 1 - d:
                val += h[i + d]
            if d <= i < n:
                val -= (1.0 - h[i - d])
            if j <= n - 1 - d:
                val -= h[j + d]
            if d <= j < n:
                val += (1.0 - h[j - d])
            a[idx] = val
        else:
            g = -d
            val = 0.0
            if i <= n - 1 - g:
                val -= (1.0 - h[i + g])
            if g <= i < n:
                val += h[i - g]
            if j <= n - 1 - g:
                val += (1.0 - h[j + g])
            if g <= j < n:
                val -= h[j - g]
            a[idx] = val
    
    top_scores = scores[top_indices]
    return top_scores, a, top_indices

def _optimal_delta(top_scores, a, delta_lo, delta_hi):
    """Find delta that minimizes max(top_scores + a * delta)"""
    if delta_lo >= delta_hi:
        return 0.0
    
    # Golden section search on the convex 1D function
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = delta_lo
    hi = delta_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(top_scores + a * c))
    fd = float(np.max(top_scores + a * d))
    
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(top_scores + a * c))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(top_scores + a * d))
    
    return (lo + hi) / 2.0

def _do_pair_transfer(h, i, j, scores, deadline, top_k=40):

    # Feasible range for delta

    delta_hi = min(h[i], 1.0 - h[j])

    opt_delta = _optimal_delta(top_scores, a, delta_lo, delta_hi)

    h_new[j] += opt_delta

    grad -= np.mean(grad)
