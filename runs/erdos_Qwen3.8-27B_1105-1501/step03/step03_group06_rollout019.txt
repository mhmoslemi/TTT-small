from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_row(h, r, n):
    k_lo = max(0, r - (n - 1))
    k_hi = min(n - 1, r)
    total = 0.0
    for k in range(k_lo, k_hi + 1):
        total += h[k] * (1.0 - h[r - k])
    return total

def _corr_row_grad(h, r, n):
    g = np.zeros(n)
    k_lo = max(0, r - (n - 1))
    k_hi = min(n - 1, r)
    for k in range(k_lo, k_hi + 1):
        j = r - k
        g[k] += (1.0 - h[j])
        g[j] -= h[k]
    return g

def _grad_matrix_all(h):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for r in range(K):
        k_lo = max(0, r - (n - 1))
        k_hi = min(n - 1, r)
        for k in range(k_lo, k_hi + 1):
            j = r - k
            G[r, k] += (1.0 - h[j])
            G[r, j] -= h[k]
    return G

def _smooth_obj(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -400.0, 400.0)
    ev = np.exp(vals)
    total_ev = float(np.sum(ev))
    f = ms + (1.0 / alpha) * np.log(total_ev)
    w = ev / total_ev
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        wi = w[r]
        k_lo = max(0, r - (n - 1))
        k_hi = min(n - 1, r)
        for k in range(k_lo, k_hi + 1):
            j = r - k
            g[k] += wi * (1.0 - x[j])
            g[j] -= wi * x[k]
    g -= np.mean(g)
    return f, g

def _lbfgs_step(h, alpha, maxiter):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(alpha,),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    target = n / 2.0
    alpha_max = 1e9
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 14):
        a = t * alpha_max
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    
    if best_a > 0:
        lo = max(0.0, best_a * 0.7)
        hi = min(alpha_max, best_a * 1.4)
        if hi > lo:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
            f2 = _exact_c5(_project(h + c2 * p, target))
            for _ in range(14):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(_project(h + c1 * p, target))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _exact_c5(_project(h + c2 * p, target))
            a = (lo + hi) / 2.0
            c5_t = _exact_c5(_project(h + a * p, target))
            if c5_t < best_c5:
                best_a, best_c5 = a, c5_t
    
    return best_a, best_c5

def _slsqp_ineq(x, args):
    dx, ks, r = args
    T = x[0]
    h_x = x[1:]
    c_r = 0.0
    for k in ks:
        c_r += h_x[k] * (1.0 - h_x[r - k])
    return T - dx * c_r

def _slsqp_ineq_jac(x, args):
    dx, ks, r = args
    h_x = x[1:]
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[0] = 1.0
    for k in ks:
        j = r - k
        g[1 + k] += -dx * (1.0 - h_x[j])
        g[1 + j] += -dx * h_x[k]
    return g

def _slsqp_eq(x, args):
    target = args[0]
    return np.sum(x[1:]) - target
