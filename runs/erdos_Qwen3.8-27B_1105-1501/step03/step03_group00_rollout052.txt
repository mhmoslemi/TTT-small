import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, target, c5_cur, max_alpha=0.5):
    n = len(h)
    a_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = max(a_max, 1e-12)
    n_samp = 16
    alphas = np.linspace(0, a_max, n_samp)
    vals = np.array([_exact_c5(_project(h + a * p, target)) for a in alphas])
    best_i = int(np.argmin(vals))
    best_a = alphas[best_i]
    best_c5 = vals[best_i]
    if best_i > 0 and best_i < n_samp - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h1 = _project(h + c1 * p, target)
        h2 = _project(h + c2 * p, target)
        f1 = _exact_c5(h1)
        f2 = _exact_c5(h2)
        for _ in range(32):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi, f2, c2 = c2, f1, c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo, f1, c1 = c1, f2, c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(_project(h + opt_a * p, target))
        if opt_c5 < best_c5:
            best_a, best_c5 = opt_a, opt_c5
    return best_a, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _pnorm_optimize(h, deadline, target, p_list=None, momentum=0.3, max_iters=5000):
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    vel = np.zeros(n)
    it_per_p = max(50, max_iters // len(p_list))
    for pi, p in enumerate(p_list):
        if time.time() > deadline:
            break
        for _ in range(it_per_p):
            if time.time() > deadline:
                break
            if stall > 100:
                break
            scores = _scores(h)
            sh = np.clip(scores, 1e-15, None)
            w = sh ** (p - 1)
            sw = float(np.sum(w))
            if sw < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= sw
            G = _grad_matrix(h)
            grad = G.T @ w
            grad -= np.mean(grad)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            grad = grad / gn * 0.1
            vel = momentum * vel + (1.0 - momentum) * grad
            a, new_c5 = _line_search(h, vel, target, cur_c5, max_alpha=0.3)
            if new_c5 < cur_c5 - 1e-14:
                h = _project(h + a * vel, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                vel *= 0.5
                stall += 1
    return best_h, best_c5

def _lp_optimize(h, deadline, target, top_k_list=None, radius_init=0.1, max_iters=5000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_restart(h_cont, n, rng, deadline):

    target = n / 2.0
    biased = h_cont + rng.normal(0, 0.15, n)
    order = np.argsort(biased)
    b = np.zeros(n)

    h_try, c5_try = _lp_optimize(b, deadline, target, top_k_list=[8, None], radius_init=0.12, max_iters=300)

def _pair_polish(h, deadline, max_passes=8):

    best_h = h.copy()

    target = n / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False

        top_lags = np.argsort(scores)[-6:]
        G = _grad_matrix(h)
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 0.05 or h[j] > 0.95:

            for k in range(max(0, j - 3), min(n, j + 4)):
                if k == j:
                    continue
                if h[k] < 0.05 or h[k] > 0.95:
                    continue
                for sign in [1, -1]:
                    if sign > 0:

                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_max * 0.5

def _active_equalize(h, deadline, target, max_iters=2000):
    """Use least-squares to equalize top-k active lags."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = float(np.max(scores))
        active = np.where(scores >= max_s - 0.005)[0]
        if len(active) < 2:
            active = np.argsort(scores)[-4:]
        elif len(active) > 12:
            active = np.argsort(scores)[-12:]
        G = _grad_matrix(h)
        Ga = G[active]
        sa = scores[active]
        # Find direction d that minimizes max Gaussian(sa + Ga@d) subject to sum(d)=0, ||d||<=r
        # Approximate: minimize ||(sa + Ga@d) - mean||^2 / r^2 + ||d||^2
        r = 0.05
        s_target = float(np.mean(sa))
        A = Ga.T
        b = s_target - sa
        # Solve: (Ga.T @ Ga + lambda*I + 1/n * 1@1) d = -Ga.T @ (sa - s_target)
        lam = 0.01
        I = np.eye(n)
        ones = np.ones(n)
        M = Ga.T @ Ga + lam * I + (1.0 / n) * np.outer(ones, ones)
        rhs = -Ga.T @ sa + s_target * (Ga.T @ ones)
        try:
            d = np.linalg.solve(M, rhs)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M, rhs, rcond=None)[0]
        d -= np.mean(d)
        dn = float(np.linalg.norm(d))
        if dn < 1e-12:
            break
        d = d / dn * r
        a, new_c5 = _line_search(h, d, target, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + a * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            r *= 0.7
            if r < 1e-6:
                break
    return best_h, best_c5
