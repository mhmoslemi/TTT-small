import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _gradient_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += -(h[d:d + m])
                G[r, d:d + m] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g + m])
                G[r, g:g + m] += -(h[:m])
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    # Final correction
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _smooth_max_obj(h, beta):
    scores = _scores(h)
    s_max = float(np.max(scores))
    return s_max + float(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _smooth_max_grad(h, beta):
    n = len(h)
    scores = _scores(h)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        k = n - 1 + d
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (-h[d:d + m])
                grad[d:d + m] += wk * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wk * (1.0 - h[g:g + m])
                grad[g:g + m] += wk * (-h[:m])
    return grad


def _momentum_optimize(h, target, deadline, beta=30.0, lr_init=0.008, momentum=0.85):

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    T = 0.003
    it = 0

        grad -= np.mean(grad)
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)

        delta = c5_new - best_c5
        if delta < -1e-14:

        it += 1

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] - 1e-12)
        hi_b = min(radius, 1.0 - h[j] + 1e-12)
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_residuals_tol': 1e-10})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur, max_evals=40):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    n_eval = min(max_evals, 30)
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-12:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo = max(0.0, best_a - alpha_max * 0.1)
    hi = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-12 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.1, max_iters=5000):

    target = n / 2.0

        else:
            stall += 1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _random_direction(h, rng):
    n = len(h)
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = np.sqrt(np.sum(d ** 2))
    if norm < 1e-12:
        return None
    return d / norm


def _escape_jump(h, target, rng, jump_sigma):
    n = len(h)
    noise = rng.standard_normal(n) * jump_sigma
    h_new = h + noise
    h_new = _project(h_new, target)
    return h_new
