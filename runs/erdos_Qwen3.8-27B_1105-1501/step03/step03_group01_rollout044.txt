def _line_search(h, p, c5_cur, n_rows):
    """
    1D line search along direction p (mean-zero) from point h.
    Returns (best_alpha, best_c5).
    """
    n = len(h)
    # Feasible interval
    t_min = 0.0
    t_max = 0.0
    for j in range(n):
        if p[j] > 1e-15:
            t_max = min(t_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-15:
            t_min = max(t_min, -h[j] / p[j])
    
    # Extend slightly for numerical safety? No, keep strict.
    if t_max - t_min < 1e-12:
        return 0.0, c5_cur
    
    # Actually t_min should be <= 0 <= t_max for feasibility of h itself
    # clamp to [t_min, t_max] but ensure 0 is included
    t_min = min(t_min, 0.0)
    t_max = max(t_max, 0.0)
    
    if t_max - t_min < 1e-12:
        return 0.0, c5_cur
    
    # Compute B, Q
    B, Q = _compute_BQ(h, p)
    A = np.correlate(h, 1.0 - h, mode='full')  # current row values
    
    dx = 2.0 / n
    
    # Evaluate objective at t: dx * max(A + B*t + Q*t^2)
    def obj(t):
        vals = A + B * t + Q * t * t
        return float(np.max(vals)) * dx
    
    # Dense sampling
    n_samples = 40
    ts = np.linspace(t_min, t_max, n_samples)
    vals = np.array([obj(t) for t in ts])
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Golden section around best
    if best_idx > 0 and best_idx < n_samples - 1:
        lo = ts[best_idx - 1]
        hi = ts[best_idx + 1]
    elif best_idx == 0:
        lo = t_min
        hi = ts[1]
    else:
        lo = ts[-2]
        hi = t_max
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_val = obj(opt_t)
    
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    if best_val < vals[best_idx]:
        best_t = ts[best_idx]  # this shouldn't happen since we started from there
    
    # Make sure best_t is within [t_min, t_max]
    best_t = max(t_min, min(t_max, best_t))
    
    if best_val < c5_cur:
        return best_t, best_val
    return 0.0, c5_cur

        val = 0.0
        jd = j - d
        if 0 <= jd < n:
            val += (1.0 - h[jd])
        jd2 = j + d
        if 0 <= jd2 < n:
            val -= h[jd2]
        grad[j] = val
    return grad

def _weighted_grad(h, scores, tau):
    """Compute weighted gradient (subgradient-like) of the max."""
    n = len(h)
    n_rows = 2 * n - 1
    max_s = float(np.max(scores))
    w = np.exp(tau * (scores - max_s))
    w = w / np.sum(w)
    grad = np.zeros(n)
    for k in range(n_rows):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        grad += w[k] * _grad_of_row(h, d, n)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline):
    """Try all pairs of coordinates for improvement."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_value(h)
    improved = True
    passes = 0
    while improved and passes < 20 and time.time() < deadline:
        improved = False
        passes += 1
        for j in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                # Try moving mass from j to k and vice versa
                for sign in [1.0, -1.0]:
                    # p[j] = sign, p[k] = -sign
                    am = min((1.0 - h[j]) / 1.0 if sign > 0 else h[j],
                             (1.0 - h[k]) / 1.0 if sign < 0 else h[k])
                    am = max(am, 0.0)
                    if am < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        t = am * frac
                        h_t = h.copy()
                        h_t[j] += sign * t
                        h_t[k] -= sign * t
                        if 0 <= h_t[j] <= 1 and 0 <= h_t[k] <= 1:
                            c5_t = _c5_value(h_t)
                            if c5_t < best_c5 - 1e-13:
                                best_c5 = c5_t
                                best_h = h_t.copy()
                                h = h_t.copy()
                                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5_value(h)
    best_n = n0
    
    def _time_left():
        return max(0.0, budget_s - (time.time() - t0))
    
    # Phase 1: Gradient-based local search with exact line search
    # Use softmax-weighted gradient of top active rows
    d_phase1 = t0 + budget_s * 0.30
    tau_list = [200.0, 100.0, 50.0, 25.0]
    lr_scale = 0.1
    stall = 0
    
    while time.time() < d_phase1:
        n = len(h)
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cur_c5 = float(np.max(scores))
        
        # Generate direction from weighted gradient
        grad = _weighted_grad(h, scores, tau_list[stall % len(tau_list)])
        # Make mean-zero
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            stall += 1
            if stall > 30:
                break
            continue
        
        # Try different step scales
        found_improvement = False
        for scale in [0.3, 0.5, 1.0, 0.7, 0.2]:
            p = -grad * scale / gnorm  # descent direction
            p -= np.mean(p)  # re-center (should already be ~0)
            
            t_best, c5_best = _line_search(h, p, cur_c5, 2*n-1)
            if t_best > 0 and c5_best < cur_c5 - 1e-12:
                h_new = h + t_best * p
                h_new = _project(h_new, n / 2.0)
                c5_check = _c5_value(h_new)
                if c5_check < cur_c5 - 1e-12:
                    h = h_new
                    if c5_check < best_c5:
                        best_c5 = c5_check
                        best_h = h.copy()
                    found_improvement = True
                    break
        
        if found_improvement:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                # Try random perturbation
                h = best_h + rng.normal(0, 0.03, best_n)
                h = _project(h, best_n / 2.0)
                stall = 0
    
    # Phase 2: Random direction search (mean-zero random vectors)
    d_phase2 = t0 + budget_s * 0.50
    n = best_n
    h = best_h.copy()
    stall = 0
    
    while time.time() < d_phase2:
        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Generate random mean-zero direction
        p = rng.normal(0, 1, n)
        p -= np.mean(p)
        pnorm = np.linalg.norm(p)
        if pnorm < 1e-12:
            continue
        p /= pnorm
        
        # Try both signs
        for sign in [1.0, -1.0]:
            p_try = sign * p
            t_best, c5_try = _line_search(h, p_try, cur_c5, 2*n-1)
            if t_best > 0 and c5_try < cur_c5 - 1e-12:
                h_new = h + t_best * p_try
                h_new = _project(h_new, n / 2.0)
                c5_check = _c5_value(h_new)
                if c5_check < cur_c5 - 1e-12:
                    h = h_new
                    stall = 0
                    break
        
        if stall > 50:
            h = best_h + rng.normal(0, 0.02, n)
            h = _project(h, n / 2.0)
            stall = 0
        else:
            stall += 1
    
    # Phase 3: Multi-resolution
    d_phase3 = t0 + budget_s * 0.70
    for dn in [4, -4, 8, -8, 12, -12, 20, -20, 30, -30, 50, -50]:
        if time.time() > d_phase3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 250:
            continue
        h_r = _resample(best_h, n_try)
        
        # Quick gradient search at new resolution
        sub_dl = min(d_phase3, time.time() + 15.0)
        for _ in range(50):
            if time.time() > sub_dl:
                break
            cur_c5 = _c5_value(h_r)
            scores = np.correlate(h_r, 1.0 - h_r, mode='full') * (2.0 / n_try)
            grad = _weighted_grad(h_r, scores, 50.0)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            p = -grad / gnorm
            t_best, c5_try = _line_search(h_r, p, cur_c5, 2*n_try-1)
            if t_best > 0 and c5_try < cur_c5 - 1e-12:
                h_r = h_r + t_best * p
                h_r = _project(h_r, n_try / 2.0)
        
        if _c5_value(h_r) < best_c5 - 1e-10:
            best_c5 = _c5_value(h_r)
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Focused improvement on top rows
    d_phase4 = t0 + budget_s * 0.85
    n = best_n
    h = best_h.copy()
    
    for _ in range(200):
        if time.time() > d_phase4:
            break
