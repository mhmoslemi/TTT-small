import numpy as np
from scipy.optimize import linprog
import time

def _scores(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:

def _grad_topk(h, k):
    """Compute gradient of the top-k mean of scores."""
    n = len(h)
    scores = _scores(h)
    # Get top k indices
    if k >= len(scores):
        k = len(scores)
    top_idx = np.argsort(scores)[::-1][:k]
    top_scores = scores[top_idx]
    # Softmax within top k for smoothness
    beta = 10.0
    m = float(np.max(top_scores))
    exps = np.exp(beta * (top_scores - m))
    weights = exps / float(np.sum(exps))
    
    # Compute gradient
    grad = np.zeros(n)
    for ri in range(k):
        w = weights[ri]
        r = top_idx[ri]
        d = r - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            # score = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
            # d/dh[j] = [d<=j<=n-1]*(1-h[j-d]) - [0<=j<=n-1-d]*h[j+d]
            j1 = np.arange(d, n)
            grad[j1] += w * (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            grad[j2] -= w * h[j2 + d]
        else:
            m2 = -d
            # score = sum_{i=0}^{n-1-m2} h[i]*(1-h[i+m2])
            # d/dh[j] = [0<=j<=n-1-m2]*(1-h[j+m2]) - [m2<=j<=n-1]*h[j-m2]
            j1 = np.arange(0, n - m2)
            grad[j1] += w * (1.0 - h[j1 + m2])
            j2 = np.arange(m2, n)
            grad[j2] -= w * h[j2 - m2]
    return grad, scores

def _grad_full_softmax(h, beta):
    """Compute gradient using full softmax weighting."""
    n = len(h)
    scores = _scores(h)
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    weights = exps / float(np.sum(exps))
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-15:

        d = r - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:

            grad[j1] += w * (1.0 - h[j1 - d])

def _topk_descent(h, deadline, k_init=8, k_min=1, max_iters=20000):
    """Gradient descent with top-k weighting and momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    beta_mom = 0.9
    k = k_init
    lr = 0.05
    stall = 0
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        # Adaptive k: decrease over time
        frac = min(it / max_iters, 1.0)
        k_cur = max(k_min, int(k_init * (1 - 0.7 * frac)))
        
        grad, scores = _grad_topk(h, k_cur)
        grad -= np.mean(grad)  # Remove mean to respect sum constraint
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        momentum = beta_mom * momentum + (1 - beta_mom) * direction
        direction = momentum / max(float(np.linalg.norm(momentum)), 1e-15)
        
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.3)
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.8
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    """Find LP direction that reduces all scores."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            j1 = np.arange(d, n)
            G[r, j1] += (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            G[r, j2] -= h[j2 + d]
        else:
            m2 = -d
            j1 = np.arange(0, n - m2)
            G[r, j1] += (1.0 - h[j1 + m2])
            j2 = np.arange(m2, n)
            G[r, j2] -= h[j2 - m2]
    
    # LP: minimize t s.t. G @ p <= -scores, |p_j| <= radius, sum(p) = 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]

def _line_search(h, p, c5_cur, n):
    """Line search along direction p."""
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Sample logarithmically
    alphas = np.geomspace(0.01, alpha_max, 20)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        h_t = h + a * p
        h_t = _project(h_t, n / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Refine with golden section
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(15):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_descent_v2(h, deadline, beta_start=20.0, beta_end=3000.0, max_iters=15000):
    """Improved soft descent with adaptive beta and momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.06
    stall = 0
    momentum = np.zeros(n)
    beta_mom = 0.85
    it = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        # Adaptive beta: increase more aggressively when stalled
        if stall > 20:
            beta = beta_start + (beta_end - beta_start) * min(1.0, frac + (stall - 20) * 0.02)
        else:
            beta = beta_start * (beta_end / beta_start) ** frac
        
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        
        # Compute gradient
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            w = weights[r]
            if w < 1e-15:
                continue
            d = r - (n - 1)
            if d == 0:
                grad += w * (1.0 - 2.0 * h)
            elif d > 0:
                j1 = np.arange(d, n)
                grad[j1] += w * (1.0 - h[j1 - d])
                j2 = np.arange(0, n - d)
                grad[j2] -= w * h[j2 + d]
            else:
                m2 = -d
                j1 = np.arange(0, n - m2)
                grad[j1] += w * (1.0 - h[j1 + m2])
                j2 = np.arange(m2, n)
                grad[j2] -= w * h[j2 - m2]
        
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        momentum = beta_mom * momentum + (1 - beta_mom) * direction
        mom_norm = float(np.linalg.norm(momentum))
        if mom_norm > 1e-15:
            direction = momentum / mom_norm
        
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.35)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.78
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def _thue_morse_init(n, amplitude=0.3):
    tm = np.zeros(n, dtype=int)

def _sine_init(n, freq=1, amplitude=0.3):

    h = 0.5 + amplitude * np.sin(2 * np.pi * freq * x)
    return _project(h, n / 2.0)

def _random_init(n, rng):

def _structured_perturb(h, rng, strength=0.1):
    """Apply a structured perturbation: flip a block."""
    n = len(h)
    h_p = h.copy()
    # Pick a random block and shift it

    block = h_p[start:start+block_len].copy()
    h_p[start:start+block_len] = h_p[start+block_len:start+2*block_len] if start + 2*block_len <= n else np.roll(h_p, block_len)

    return _project(h_p, n / 2.0)

def _coord_descent(h, deadline, max_passes=100):
    """2D coordinate descent: adjust pairs of coordinates."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.02, 0.05, 0.1, 0.15]
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline - 5:
                break
            if j + 1 >= n:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            if h[j+1] <= 1e-10 or h[j+1] >= 1.0 - 1e-10:
                continue
            for delta in deltas:
                for s1, s2 in [(1, -1), (-1, 1)]:
                    d = s1 * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    if h[j+1] - d < 0 or h[j+1] - d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    h_try[j+1] -= d
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            j1 = np.arange(d, n)
            grad[j1] += w * (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            grad[j2] -= w * h[j2 + d]
        else:
            m2 = -d
            j1 = np.arange(0, n - m2)
            grad[j1] += w * (1.0 - h[j1 + m2])
            j2 = np.arange(m2, n)
            grad[j2] -= w * h[j2 - m2]
    return grad

def _soft_weighted_grad(h, beta):
    scores = _scores(h)
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    total = float(np.sum(exps))
    weights = exps / total
    grad = _compute_gradient(h, weights)
    return grad, scores

def _topk_weighted_grad(h, k, beta_k=5.0):
    n = len(h)
    scores = _scores(h)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(scores)[::-1][:k]
    top_scores = scores[top_idx]
    m = float(np.max(top_scores))
    exps = np.exp(beta_k * (top_scores - m))
    top_weights = exps / float(np.sum(exps))
    weights = np.zeros(2 * n - 1)
    weights[top_idx] = top_weights
    grad = _compute_gradient(h, weights)
    return grad, scores

def _soft_descent_momentum(h, deadline, beta_start=15.0, beta_end=5000.0, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.06
    stall = 0
    momentum = np.zeros(n)
    beta_mom = 0.85
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        if stall > 30:
            beta = beta_start + (beta_end - beta_start) * min(1.0, frac + (stall - 30) * 0.03)
        else:
            beta = beta_start * (beta_end / beta_start) ** frac
        grad, scores = _soft_weighted_grad(h, beta)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        momentum = beta_mom * momentum + (1.0 - beta_mom) * direction
        mom_norm = float(np.linalg.norm(momentum))
        if mom_norm > 1e-15:
            direction = momentum / mom_norm
        alpha = lr
        improved = False
        for _ in range(22):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.35)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.78
            if lr < 1e-6 or stall > 120:
                break
    return best_h, best_c5

def _topk_descent(h, deadline, k_init=8, k_min=1, max_iters=15000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    beta_mom = 0.8
    lr = 0.05
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        k_cur = max(k_min, int(k_init * (1.0 - 0.7 * frac)))
        beta_k = 2.0 + 8.0 * frac
        grad, scores = _topk_weighted_grad(h, k_cur, beta_k)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        momentum = beta_mom * momentum + (1.0 - beta_mom) * direction
        mom_norm = float(np.linalg.norm(momentum))
        if mom_norm > 1e-15:
            direction = momentum / mom_norm
        alpha = lr
        improved = False
        for _ in range(22):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.3)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.77
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            j1 = np.arange(d, n)
            G[r, j1] += (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            G[r, j2] -= h[j2 + d]
        else:
            m2 = -d
            j1 = np.arange(0, n - m2)
            G[r, j1] += (1.0 - h[j1 + m2])
            j2 = np.arange(m2, n)
            G[r, j2] -= h[j2 - m2]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, n):
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    alphas = np.geomspace(0.01, alpha_max, 18)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        h_t = h + a * p
        h_t = _project(h_t, n / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(15):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.06, max_iters=4000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, n)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.8
            if stall % 15 == 0 and radius < 0.03:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse_init(n, amplitude=0.3):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)

def _sine_init(n, freq, amplitude):
    x = np.arange(n) / n
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq * x)
    return _project(h, n / 2.0)

def _random_init(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return _project(h, n / 2.0)

def _coord_descent(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.02, 0.05, 0.1]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n - 1, 2):
            if time.time() > deadline - 3:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            if h[j + 1] <= 1e-10 or h[j + 1] >= 1.0 - 1e-10:
                continue
            for delta in deltas:
                for s1, s2 in [(1, -1), (-1, 1)]:
                    d = s1 * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    if h[j + 1] - d < 0 or h[j + 1] - d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    h_try[j + 1] -= d
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Soft descent with momentum from parent
    deadline1 = t0 + budget_s * 0.22
    h, c5 = _soft_descent_momentum(h, deadline1, beta_start=12.0, beta_end=5000.0, max_iters=25000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Top-k descent for sharper focus
    deadline2 = t0 + budget_s * 0.32
    h_tk, c5_tk = _topk_descent(best_h.copy(), deadline2, k_init=6, k_min=1, max_iters=15000)
    if c5_tk < best_c5:
        best_c5 = c5_tk
        best_h = h_tk.copy()

    # Phase 3: LP polish
    deadline3 = t0 + budget_s * 0.48
    h_lp, c5_lp = _lp_polish(best_h.copy(), deadline3, radius_init=0.05, max_iters=4000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Diverse restarts
    deadline4 = t0 + budget_s * 0.75
    scale = 0.05
    for attempt in range(600):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3804:
            break
        r = rng.random()
        if r < 0.35:
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif r < 0.55:
            h_pert = best_h.copy()
            blen = int(rng.integers(2, 6))
            start = int(rng.integers(0, best_n - blen))
            h_pert[start:start + blen] = 1.0 - h_pert[start:start + blen]
        elif r < 0.75:
            h_pert = best_h.copy()
            i1 = int(rng.integers(0, best_n))
            i2 = int(rng.integers(0, best_n))
            if i1 != i2:
                h_pert[[i1, i2]] = h_pert[[i2, i1]]
            else:
                h_pert = best_h + rng.normal(0, scale, best_n)
        else:
            h_pert = best_h + rng.normal(0, scale * 0.5, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            j = int(rng.integers(0, best_n))
            h_pert[j] = np.clip(h_pert[j] + rng.choice([-1, 1]) * 0.3, 0, 1)
            h_pert = _project(h_pert, best_n / 2.0)

        h_pert = _project(h_pert, best_n / 2.0)
        time_left = deadline4 - time.time()
        n_remaining = max(600 - attempt, 1)
        time_per = min(max(0.3, time_left / n_remaining), budget_s * 0.012)
        h_try, c5_try = _topk_descent(h_pert, time.time() + time_per, k_init=5, k_min=1, max_iters=2500)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.35, scale * 1.08)

    # Phase 5: Multi-resolution
    deadline5 = t0 + budget_s * 0.87
    n_candidates = set()
    for dn in [-16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16]:
        nc = best_n + dn
        if 48 <= nc <= 128:
            n_candidates.add(nc)
    for n_try in sorted(n_candidates):
        if time.time() > deadline5:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _soft_descent_momentum(h_r, min(deadline5, time.time() + budget_s * 0.01),
                                            beta_start=20.0, beta_end=4000.0, max_iters=3000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Alternative initializations
    deadline6 = t0 + budget_s * 0.91
    inits = [
        _thue_morse_init(best_n, 0.25),
        _sine_init(best_n, 1, 0.25),
        _sine_init(best_n, 2, 0.20),
        _random_init(best_n, rng),
    ]
    for h_alt in inits:
        if time.time() > deadline6:
            break
        c5_alt = _exact_c5(h_alt)
        if c5_alt < best_c5 + 0.02:
            h_alt2, c5_alt2 = _topk_descent(h_alt, min(deadline6, time.time() + budget_s * 0.008),
                                             k_init=4, k_min=1, max_iters=2000)
            if c5_alt2 < best_c5 - 1e-10:
                best_c5 = c5_alt2
                best_h = h_alt2.copy()

    # Phase 7: Final LP polish
    deadline7 = t0 + budget_s * 0.955
    h_f, c5_f = _lp_polish(best_h.copy(), deadline7, radius_init=0.03, max_iters=5000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Phase 8: Coordinate descent
    deadline8 = t0 + budget_s * 0.995
    h_cd, c5_cd = _coord_descent(best_h.copy(), deadline8, max_passes=60)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
