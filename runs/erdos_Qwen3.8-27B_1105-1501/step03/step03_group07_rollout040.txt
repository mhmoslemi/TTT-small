import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    
    while time.time() < deadline:
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_corr = float(np.max(corr))
        
        # For each pair, try to reduce max_corr
        # Priority: pairs involving indices that appear in high-correlation lags
        # Compute "pressure" on each index
        pressure = np.zeros(n)
        top_k = 10
        top_lags = np.argsort(-corr)[:top_k]
        for r in top_lags:
            d = int(r) - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    pressure[:m] += corr[r] * h[d:]
                    pressure[d:] += corr[r] * (1.0 - h[:m])
            else:
                l = -d
                m = n - l
                if m > 0:
                    pressure[:m] += corr[r] * (1.0 - h[l:])
                    pressure[l:] += corr[r] * h[:m]
        
        # Sort pairs by pressure difference
        # Only consider "interior" pairs (both in (0,1))
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:

        # Try pairs in order of pressure difference

        pi = pressure[interior]

        for a in range(len(interior)):

                diff = abs(pi[a] - pi[b])
                pairs.append((diff, interior[a], interior[b]))
        pairs.sort(key=lambda x: -x[0])
        
        # Try top pairs
        tried = 0
        for diff, i, j in pairs[:200]:  # Top 200 pairs
            if time.time() > deadline:
                return best_h, best_c5
            tried += 1
            c = h[i] + h[j]  # conserved sum
            if c <= 1e-10 or c >= 2.0 - 1e-10:

            # Try different splits of c between i and j
            # h[i]_new ranges from max(0, c-1) to min(1, c)

            hi = min(1.0, c)
            if hi - lo < 1e-10:

        else:

    D = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        d = 0.0
        # h[i] contribution (increases by delta)
        ik = i - k
        if 0 <= ik < n:
            d += (1.0 - h[ik])
        ik2 = i + k
        if 0 <= ik2 < n:
            d -= h[ik2]
        # h[j] contribution (decreases by delta)
        jk = j - k
        if 0 <= jk < n:
            d -= (1.0 - h[jk])
        jk2 = j + k
        if 0 <= jk2 < n:
            d += h[jk2]
        D[r] = d
    
    # Valid range
    lo = max(-h[i], h[j] - 1.0) + 1e-12
    hi = min(1.0 - h[i], h[j]) - 1e-12
    if hi - lo < 1e-12:
        return h[i], h[j]
    
    # Golden section on f(delta) = max(corr + delta * D)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(corr + c1 * D))
    f2 = float(np.max(corr + c2 * D))
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(corr + c1 * D))
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(corr + c2 * D))
    
    delta = (a + b) * 0.5
    return h[i] + delta, h[j] - delta
