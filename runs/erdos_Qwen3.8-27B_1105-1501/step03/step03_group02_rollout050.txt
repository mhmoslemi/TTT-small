        elif k > 0:
            m = n - k
            B[r] = float(np.dot(d[k:k+m], 1.0 - h[:m])) - float(np.dot(h[k:k+m], d[:m]))
            C[r] = -float(np.dot(d[k:k+m], d[:m]))
        else:
            g = -k
            m = n - g
            B[r] = float(np.dot(d[:m], 1.0 - h[g:g+m])) - float(np.dot(h[:m], d[g:g+m]))
            C[r] = -float(np.dot(d[:m], d[g:g+m]))
    return A, B, C

def _exact_1d(h, d, cur_raw):
    n = len(h)
    M = 2 * n - 1
    
    # Feasible interval for t
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, cur_raw
    
    A, B, C = _quad_line(h, d)
    
    cands = np.array([t_lo, t_hi, 0.0])
    
    # Stationary points
    nz = np.abs(C) > 1e-14
    if np.any(nz):
        t0 = -B[nz] / (2.0 * C[nz])
        valid = (t0 >= t_lo) & (t0 <= t_hi)
        if np.any(valid):
            cands = np.concatenate([cands, t0[valid]])
    
    # Pairwise intersections
    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    t0 = -c / b
                    if t_lo <= t0 <= t_hi:
                        cands = np.append(cands, t0)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-14:
                    disc = max(disc, 0.0)
                    sq = np.sqrt(disc)
                    t1 = (-b + sq) / (2.0 * a)
                    t2 = (-b - sq) / (2.0 * a)
                    if t_lo <= t1 <= t_hi:
                        cands = np.append(cands, t1)
                    if t_lo <= t2 <= t_hi:
                        cands = np.append(cands, t2)
    
    # Clip, deduplicate
    cands = np.clip(cands, t_lo, t_hi)
    cands = np.unique(np.round(cands, 12))
    
    # Evaluate E(t) = max_r (A[r] + B[r]*t + C[r]*t^2)
    T = cands[:, None]
    vals = A[None, :] + B[None, :] * T + C[None, :] * T * T
    max_vals = vals.max(axis=1)
    
    best_idx = int(np.argmin(max_vals))
    best_raw = float(max_vals[best_idx])
    best_t = float(cands[best_idx])
    
    # Include current point
    cur = cur_raw
    if cur < best_raw:
        best_raw = cur
        best_t = 0.0
    
    return best_t, best_raw

def _smooth_max_dir(h, scores, beta):
    n = len(h)
    M = len(scores)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        # All weight on max
        idx = int(np.argmax(scores))
        w = np.zeros(M)
        w[idx] = 1.0
    else:
        w /= wsum
    
    # Compute weighted gradient
    G = np.zeros(n)
    for r in range(M):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)
        if k == 0:
            G += w[r] * (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            G[:m] += -w[r] * h[k:k+m]
            G[k:] += w[r] * (1.0 - h[:m])
        else:
            g = -k
            m = n - g
            G[:m] += w[r] * (1.0 - h[g:g+m])
            G[g:] += -w[r] * h[:m]
    
    # Make zero-sum
    G -= np.mean(G)
    norm = float(np.max(np.abs(G)))
    if norm < 1e-15:
        return None
    return G / norm

def _random_dir(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm
