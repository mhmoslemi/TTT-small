import numpy as np
from scipy.optimize import minimize
import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_simplex(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def score_gradient_row(h, d):
    """Compute gradient of score at lag d. Returns array of length n."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        gneg = -d
        m = n - gneg
        if m > 0:
            g[:m] += (1.0 - h[gneg:])
            g[gneg:] -= h[:m]
    return g


def compute_all_gradients(h):
    """Compute gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            gneg = -d
            m = n - gneg
            if m > 0:
                G[r, :m] += (1.0 - h[gneg:])
                G[r, gneg:] -= h[:m]
    return G


def subgradient_direction(h, scores, G, top_k=5, band_frac=0.15):
    """Compute weighted subgradient of top scores. Returns zero-mean direction."""
    n = len(h)
    s_max = float(np.max(scores))
    if s_max < 1e-12:
        return None
    band = band_frac * s_max
    active = np.where(scores >= s_max - band)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    # Weight by how close to max
    weights = scores[active] - (s_max - band)
    weights = np.maximum(weights, 1e-10)
    weights = weights / np.sum(weights)
    # Weighted sum of gradients
    d = G[active, :] @ weights  # shape (n,)
    d = d - np.mean(d)  # zero mean
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    return d / norm


def line_search_scores(h, direction, max_alpha, n_coarse=15, n_fine=20):
    """Line search: find best alpha in [0, max_alpha] minimizing max score."""
    n = len(h)
    # Find feasible alpha range
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
    
    # Coarse search
    alphas = np.linspace(0.0, alpha_max, n_coarse)
    for a in alphas:
        if a < 1e-12:
            continue
        h_t = h + a * direction
        val = np.max(np.correlate(h_t, 1.0 - h_t, mode='full'))
        if val < best_val - 1e-14:
            best_val = val
            best_a = a
    
    if best_a == 0.0:
        return 0.0, best_val
    
    # Fine golden section
    lo = max(0.0, best_a - alpha_max * 0.1)
    hi = min(alpha_max, best_a + alpha_max * 0.1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(np.correlate(h + c1 * direction, 1.0 - (h + c1 * direction), mode='full'))
    f2 = np.max(np.correlate(h + c2 * direction, 1.0 - (h + c2 * direction), mode='full'))
    for _ in range(n_fine):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(np.correlate(h + c1 * direction, 1.0 - (h + c1 * direction), mode='full'))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(np.correlate(h + c2 * direction, 1.0 - (h + c2 * direction), mode='full'))
    
    opt_a = (lo + hi) / 2.0
    opt_val = np.max(np.correlate(h + opt_a * direction, 1.0 - (h + opt_a * direction), mode='full'))
    if opt_val < best_val:
        return opt_a, opt_val
    return best_a, best_val


def pair_exchange_line(h, i, j, max_alpha, n_coarse=10, n_fine=15):
    """Try transferring mass from j to i (direction = e_i - e_j)."""
    direction = np.zeros(len(h))
    direction[i] = 1.0
    direction[j] = -1.0
    a, val = line_search_scores(h, direction, max_alpha, n_coarse, n_fine)
    return a, val


def pair_search_local(h, deadline, max_passes=20, top_radius=20):
    """Exhaustive pair search on interior variables."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_raw = np.max(np.correlate(h, 1.0 - h, mode='full'))
    
    # Identify interior variables
    interior = np.where((h > 0.01) & (h < 0.99))[0]
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(interior)):

            i = interior[ii]

def resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)

import numpy as np
from scipy.optimize import linprog, minimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_simplex(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G


def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

                    d[pj] = -1.0
                    a, new_raw = line_search(best_h, d, best_raw, n_coarse=8, n_fine=12)
                    if a > 1e-8 and new_raw < best_raw - 1e-11:
                        best_h = best_h + a * d
                        best_h = project_simplex(best_h, target)
                        best_raw = new_raw
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_h, exact_c5(best_h)


def nesterov_descent(h, target, deadline, beta=40.0, lr_init=0.008, momentum=0.8, max_iters=500):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_raw = float(np.max(np.correlate(h, 1.0 - h, mode='full')))

    for _it in range(max_iters):
        if time.time() > deadline:
            break
        y = h + momentum * v
        scores = np.correlate(y, 1.0 - y, mode='full')
        s_max = float(np.max(scores))
        if s_max < 1e-12:
            break
        shifted = scores - s_max
        exp_s = np.exp(beta * shifted)
        w = exp_s / np.sum(exp_s)
        G = compute_G(y)
        grad = G.T @ w
        grad = grad - np.mean(grad)

        v = momentum * v + lr * grad
        h_new = project_simplex(h - v, target)
        new_raw = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))

        if new_raw < best_raw:
            best_raw = new_raw
            best_h = h_new.copy()
        h = h_new
        lr = min(lr * 1.01, 0.25)

    return best_h, exact_c5(best_h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = project_simplex(h0, target0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Nesterov + subgradient (0-20%)
    d1 = t0 + budget_s * 0.20
    for beta in [20.0, 40.0, 80.0]:
        if time.time() > d1:
            break
        h1, c5_1 = nesterov_descent(best_h, best_n / 2.0, min(d1, time.time() + budget_s * 0.04),
                                    beta=beta, lr_init=0.006, momentum=0.78)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: LP optimization (20-40%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = lp_optimize(best_h, best_n / 2.0, d2, radius_init=0.12, max_iters=2000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Simulated annealing with LP (40-58%)
    d3 = t0 + budget_s * 0.58
    h3, c5_3 = anneal_lp(best_h, best_n / 2.0, d3, rng, T_init=0.004, max_iters=3000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: More LP + pair polish (58-72%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = lp_optimize(best_h, best_n / 2.0, d4, radius_init=0.08, max_iters=1500)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    d4b = t0 + budget_s * 0.72
    h4b, c5_4b = pair_polish(best_h, d4b, rng, max_passes=50)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()

    # Phase 5: Resolution sweep (72-88%)
    d5 = t0 + budget_s * 0.88
    for dn in [4, -4, 8, -8, 12, -12, 16, 24, -24, 32]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 192:
            continue
        h_r = resample(best_h, n_try)
        # Quick optimize
        if time.time() < d5:
            h_r, c5_r = lp_optimize(h_r, n_try / 2.0, min(d5, time.time() + budget_s * 0.02),
                                     radius_init=0.1, max_iters=500)
        else:
            c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final anneal + LP + pair (88-97%)
    d6 = t0 + budget_s * 0.92
    h6, c5_6 = anneal_lp(best_h, best_n / 2.0, d6, rng, T_init=0.001, T_min=1e-8, max_iters=2000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    d6b = t0 + budget_s * 0.95
    h6b, c5_6b = lp_optimize(best_h, best_n / 2.0, d6b, radius_init=0.03, max_iters=1500)
    if c5_6b < best_c5:
        best_c5 = c5_6b
        best_h = h6b.copy()

    d6c = t0 + budget_s * 0.97
    h6c, c5_6c = pair_polish(best_h, d6c, rng, max_passes=40)
    if c5_6c < best_c5:
        best_c5 = c5_6c
        best_h = h6c.copy()

    # Final validation
    best_h = project_simplex(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
