def _top_lag_optimize(h, top_k=3, max_iters=100):

    from scipy.optimize import minimize
    
    n = len(h)
    target = n / 2.0
    
    # Identify top k lags
    scores = _scores(h)
    top_idx = np.argsort(-scores)[:top_k]
    
    def objective(x):
        x = np.clip(x, 0, 1)
        s = _scores(x)
        # Max of top k
        return np.max([s[i] for i in top_idx])

        if 0 <= j-d < n:

    # Find the minimum.

    # The max is the upper envelope.

                f1 = f2

            new_c5 = c_t
            best_alpha = a_test

        return 0.0, f0
    
    return best_alpha, new_c5

```python
def _compute_pair_slopes(h, i, j, direction):
    n = len(h)
    di = -direction
    dj = direction
    slopes = np.zeros(2*n - 1)
    
    # Contribution from h[i] changing by di*α
    # For lag d: valid if 0≤i+d<n → contributes di*(1-h[i+d])
    #             valid if 0≤i-d<n → contributes -h[i-d]*di
    d_vals = np.arange(-(n-1), n)
    # i+d valid: 0 ≤ i+d ≤ n-1 → -i ≤ d ≤ n-1-i
    mask1 = (d_vals >= -i) & (d_vals <= n-1-i)
    # i-d valid: 0 ≤ i-d ≤ n-1 → 1-i ≤ -d ≤ n-1-i → -(n-1-i) ≤ d ≤ i-1
    mask2 = (d_vals >= -(n-1-i)) & (d_vals <= i-1)
    
    # For mask1: contribution is di*(1-h[i+d_vals])
    if np.any(mask1):
        idx = i + d_vals[mask1]
        slopes[mask1] += di * (1 - h[idx])
    
    # For mask2: contribution is -h[i-d_vals]*di
    if np.any(mask2):
        idx2 = i - d_vals[mask2]
        slopes[mask2] -= h[idx2] * di
    
    # Same for j
    mask3 = (d_vals >= -j) & (d_vals <= n-1-j)
    mask4 = (d_vals >= -(n-1-j)) & (d_vals <= j-1)
    
    if np.any(mask3):
        idx3 = j + d_vals[mask3]
        slopes[mask3] += dj * (1 - h[idx3])
    
    if np.any(mask4):
        idx4 = j - d_vals[mask4]
        slopes[mask4] -= h[idx4] * dj
    
    return slopes
```
