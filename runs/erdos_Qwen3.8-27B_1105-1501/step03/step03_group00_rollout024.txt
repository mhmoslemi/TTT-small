```python
import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def to_binary(h):
    """Convert h to binary with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # Sort by value descending, take top target
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def project(h, target):
    """Project h to [0,1]^n with sum = target."""
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0, 1)
```

```python
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    improved = True
    while improved and (deadline is None or time.time() < deadline):
        improved = False
        best_improvement = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                c5_new = exact_c5(h)
                h[i] = 1
                h[j] = 0
                if c5_new < best_c5 - best_improvement:
                    best_improvement = best_c5 - c5_new
                    best_swap = (i, j)
                    if best_improvement > 0.001:  # Good enough, don't search more
                        pass
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            best_c5 = best_c5 - best_improvement
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = True
    
    return best_h, best_c5
```

Now let me think about the continuous refinement. After finding a good binary solution, I want to try small continuous perturbations. The idea:

For each coordinate i where h[i] is 0 or 1, try setting h[i] to a small positive value (if 0) or 1-ε (if 1), and adjust another coordinate to maintain the sum.

Actually, a better approach: use the gradient of the active lag to determine which direction to move each coordinate.

For the active lag k*, the gradient is:
∇f_{k*}[i] = dx * ((1-h[i+k*]) - h[i-k*]) (with boundary handling)

At a binary solution, this is dx * (h[i-k*] - h[i+k*]) which is in {-dx, 0, dx}.

To minimize f_{k*}, we want to move h[i] in the direction of -∇f_{k*}. So:
- If h[i-k*]=1 and h[i+k*]=0: gradient is dx, we should decrease h[i] (but it's already 0 or 1)
- If h[i-k*]=0 and h[i+k*]=1: gradient is -dx, we should increase h[i]

The issue is that at a binary point, we're at a vertex, so the gradient points outside the feasible region. We need to move along an edge.

Let me try a different continuous approach: **pairwise continuous moves**.

For a pair (i, j) where h[i]=1 and h[j]=0, try:
h[i] = 1 - ε, h[j] = ε (for small ε > 0)
This maintains the sum constraint.

For each such pair, compute the new C₅ and accept if it improves.

This is O(n²) per sweep (for n=64, that's 1024 pairs), and each evaluation is O(n) for the correlate. So each sweep is O(n³) = O(262K) operations. Very fast.
