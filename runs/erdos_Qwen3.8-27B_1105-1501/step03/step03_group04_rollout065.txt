import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _focal_descent(h, deadline, max_iters=5000):
    """Focus on the argmax shift and try to reduce it directly."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        k_max = int(np.argmax(scores))
        # k_max is in [0, 2n-2], shift d = k_max - (n-1)
        d = k_max - (n - 1)
        
        # Compute contribution pairs for this shift
        if d >= 0:
            m = n - d if d > 0 else n
            if d == 0:
                # Central term: h_i * (1-h_i)
                contrib = h * (1.0 - h)
                idx_a = np.arange(m)
                idx_b = idx_a
            else:
                # h_i * (1-h_{i+d})
                contrib = h[:m] * (1.0 - h[d:d+m])
                idx_a = np.arange(m)
                idx_b = idx_a + d
        else:
            g = -d
            m = n - g
            # h_i * (1-h_{i-g})
            contrib = h[g:] * (1.0 - h[:m])
            idx_a = np.arange(g, n)
            idx_b = np.arange(m)
        
        # Try to reduce: find pairs where contrib is large
        # Strategy: for top-contributing pairs, try to swap or adjust
        top_k = min(20, m)
        top_idx = np.argsort(contrib)[-top_k:]
        
        improved = False
        best_local_c5 = best_c5
        best_local_h = best_h.copy()
        
        for ti in top_idx:
            if time.time() > deadline:
                break
            ia = idx_a[ti]
            ib = idx_b[ti]
            
            if d == 0:
                # Move h_ia toward 0.5 to reduce h*(1-h)
                for delta in [0.02, 0.05, 0.1, 0.15]:
                    for sign in [1.0, -1.0]:
                        h_try = h.copy()
                        direction = -sign * np.sign(h[ia] - 0.5) * delta
                        if abs(direction) < 1e-10:
                            direction = sign * delta
                        if h[ia] + direction < 0 or h[ia] + direction > 1:
                            continue
                        h_try[ia] += direction
                        # Compensate
                        mask = np.ones(n, dtype=bool)
                        mask[ia] = False
                        if direction > 0:
                            comp = int(np.argmax(np.where(mask, 1.0 - h_try, -1)))
                        else:
                            comp = int(np.argmax(np.where(mask, h_try, -1)))
                        h_try[comp] -= direction
                        if h_try[comp] < 0 or h_try[comp] > 1:
                            continue
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_local_c5 - 1e-12:
                            best_local_c5 = c5_try
                            best_local_h = h_try.copy()
                            improved = True
            else:
                # For d != 0: try to make h_ia and h_ib more similar (both 0 or both 1)
                for delta in [0.05, 0.1, 0.15, 0.2]:
                    # Decrease h_ia or increase h_ib
                    for move in [(ia, -delta), (ib, delta)]:
                        mi, di = move
                        if h[mi] + di < 0 or h[mi] + di > 1:
                            continue
                        h_try = h.copy()
                        h_try[mi] += di
                        # Compensate on the other (opposite direction)
                        if di < 0:
                            # We decreased h_ia, need to increase somewhere
                            mask = np.ones(n, dtype=bool)
                            mask[mi] = False
                            comp = int(np.argmax(np.where(mask, 1.0 - h_try, -1)))
                            h_try[comp] -= di  # increase
                        else:
                            mask = np.ones(n, dtype=bool)
                            mask[mi] = False
                            comp = int(np.argmax(np.where(mask, h_try, -1)))
                            h_try[comp] -= di  # decrease
                        if h_try[comp] < 0 or h_try[comp] > 1:
                            continue
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_local_c5 - 1e-12:
                            best_local_c5 = c5_try
                            best_local_h = h_try.copy()
                            improved = True
        
        if improved:
            h = best_local_h
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = best_local_h.copy()
    
    return best_h, best_c5

def _annealing(h, deadline, T_start=0.5, T_end=0.01, max_iters=20000):
    """Simulated annealing with exact evaluations."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Random perturbation
        n_moves = np.random.randint(1, 5)
        h_try = h.copy()
        for _ in range(n_moves):
            j = np.random.randint(n)
            delta = np.random.uniform(-0.15, 0.15)
            h_try[j] += delta
        
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < cur_c5:
            h = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        elif np.random.random() < np.exp(-(c5_try - cur_c5) / max(T, 1e-8)):
            h = h_try
            cur_c5 = c5_try
    
    return best_h, best_c5

def _swap_search(h, deadline):
    """Try pairwise swaps to reduce C5."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
        scores = _scores(h)
        k_max = int(np.argmax(scores))
        d = k_max - (n - 1)
        
        if d == 0:
            # Swap two values to reduce h_i*(1-h_i)
            # Best: move pair closer to (0,0) or (1,1) total
            i = int(np.argmax(h * (1.0 - h)))
            # Find j where swapping helps
            j = np.random.randint(n)
            if j == i:
                continue
            h_try = h.copy()
            h_try[i], h_try[j] = h_try[j], h_try[i]
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
        else:
            # Focus on shift d
            m = n - abs(d)
            if d > 0:
                contrib = h[:m] * (1.0 - h[d:d+m])
            else:
                g = -d
                contrib = h[g:] * (1.0 - h[:m])
            
            # Find the pair with max contribution, try to swap with a random other
            i_top = int(np.argmax(contrib))
            if d > 0:
                ia, ib = i_top, i_top + d
            else:
                ia, ib = i_top + (-d), i_top
            
            j1 = np.random.randint(n)
            j2 = np.random.randint(n)
            if j1 in (ia, ib) or j2 in (ia, ib) or j1 == j2:
                continue
            
            h_try = h.copy()
            # Swap values at j1 and j2
            h_try[j1], h_try[j2] = h_try[j2], h_try[j1]
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5

def _population_search(h_init, deadline, pop_size=8, generations=500):
    """Population-based search with genetic operators."""
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_pert = h_init + rng.normal(0, 0.1, n)
        population.append(_project(h_pert, target))
    
    scores_pop = [_exact_c5(h) for h in population]
    best_idx = int(np.argmin(scores_pop))
    best_h = population[best_idx].copy()
    best_c5 = scores_pop[best_idx]
    
    gen = 0
    while gen < generations and time.time() < deadline:
        gen += 1
        # Select parents (tournament)
        new_pop = [population[best_idx].copy()]  # elitism
        
        for _ in range(pop_size - 1):
            # Tournament selection
            cands = rng.choice(pop_size, 3, replace=False)
            cands_scores = [scores_pop[c] for c in cands]
            parent_idx = cands[int(np.argmin(cands_scores))]
            
            # Crossover + mutation
            if rng.random() < 0.5 and len(population) > 1:
                other_idx = int(np.argmin([scores_pop[i] for i in range(pop_size) if i != parent_idx]))
                alpha = rng.uniform(0.3, 0.7)
                h_child = alpha * population[parent_idx] + (1 - alpha) * population[other_idx]
            else:
                h_child = population[parent_idx].copy()
            
            # Mutation
            n_mut = rng.integers(1, 6)
            for _ in range(n_mut):
                j = rng.integers(n)
                h_child[j] += rng.normal(0, 0.08)
            
            h_child = _project(h_child, target)
            c5_child = _exact_c5(h_child)
            new_pop.append(h_child)
        
        # Update population (keep best pop_size)
        all_cands = new_pop
        all_scores = [_exact_c5(h) for h in all_cands]
        
        # Sort and keep best
        order = np.argsort(all_scores)
        population = [all_cands[i].copy() for i in order[:pop_size]]
        scores_pop = [all_scores[i] for i in order[:pop_size]]
        
        if scores_pop[0] < best_c5:
            best_c5 = scores_pop[0]
            best_h = population[0].copy()
    
    return best_h, best_c5

def _soft_descent(h, deadline, beta_start=30.0, beta_end=3000.0, max_iters=20000):
    """Soft-max based gradient descent (kept from before but refactored)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.08
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        
        # Gradient computation
        grad = np.zeros(n)
        # For each displacement d, the gradient contribution
        for d in range(-(n-1), n):
            # score index
            r = n - 1 + d
            w = weights[r]
            if w < 1e-10:
                continue
            if d == 0:
                grad += w * (1.0 - 2.0 * h)
            elif d > 0:
                m = n - d
                if m > 0:
                    grad[:m] += w * (1.0 - h[d:d+m])
                    grad[d:d+m] -= w * h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[g:] += w * (1.0 - h[:m])
                    grad[:m] -= w * h[g:]
        
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            if np.any(h_try < -0.1) or np.any(h_try > 1.1):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.4)
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-5 or stall > 60:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _two_point_local(h, deadline, max_iters=10000):
    """Two-point local search: pick two coordinates, try to swap or adjust them."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        k_max = int(np.argmax(scores))
        d = k_max - (n - 1)
        
        # Identify the most problematic pair for the max shift
        if d == 0:
            contrib = h * (1.0 - h)
            i1 = int(np.argmax(contrib))
            # Try to move h[i1] toward 0 or 1
            i2 = int(np.argmin(contrib))
            
            if h[i1] < h[i2]:
                # Swap
                h_try = h.copy()
                h_try[i1], h_try[i2] = h_try[i2], h_try[i1]
            else:
                # Average partially
                h_try = h.copy()
                avg = (h[i1] + h[i2]) / 2
                h_try[i1] = avg
                h_try[i2] = avg
        else:
            abs_d = abs(d)
            m = n - abs_d
            if d > 0:
                contrib = h[:m] * (1.0 - h[d:d+m])
            else:
                contrib = h[abs_d:] * (1.0 - h[:m])
            
            i_top = int(np.argmax(contrib))
            if d > 0:
                ia, ib = i_top, i_top + d
            else:
                ia, ib = i_top + abs_d, i_top
            
            # Try to make h[ia] and h[ib] more alike
            if h[ia] > 0.5:
                # Try decreasing ia or increasing ib
                h_try = h.copy()
                delta = min(0.1, h[ia] - 0.5, 1.0 - h[ib])
                if delta > 0.01:
                    h_try[ia] -= delta
                    h_try[ib] += delta
                else:
                    h_try = h.copy()
                    # Swap with random position
                    j = np.random.randint(n)
                    if j not in (ia, ib):
                        h_try[ia], h_try[j] = h_try[j], h_try[ia]
            else:
                h_try = h.copy()
                delta = min(0.1, 0.5 - h[ia], 1.0 - h[ib])
                if delta > 0.01:
                    h_try[ia] += delta
                    h_try[ib] -= delta
                else:
                    j = np.random.randint(n)
                    if j not in (ia, ib):
                        h_try[ia], h_try[j] = h_try[j], h_try[ia]
        
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
    
    return best_h, best_c5

    # Phase 8: Final two-point polish
