def _adam_soft_descent(h, deadline, beta_start=50.0, beta_end=500.0, 
                       lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
        t += 1
        frac = min(t / 5000.0, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute scores
        scores = _scores(h)
        m_score = float(np.max(scores))
        
        # Softmax weights
        exps = np.exp(beta * (scores - m_score))
        weights = exps / float(np.sum(exps))
        
        # Gradient
        G = _grad_matrix(h)
        grad = G.T @ weights * (2.0 / n)  # dx factor
        grad -= np.mean(grad)  # enforce sum constraint
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_try = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_try = _project(h_try, target)
        
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = h_try
    
    return best_h, best_c5
