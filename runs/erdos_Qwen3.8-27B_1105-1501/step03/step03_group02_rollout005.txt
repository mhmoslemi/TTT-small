import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, l):
    n = len(h)
    grad = np.zeros(n)
    if l == 0:
        grad[:] = 1.0 - 2.0 * h
    elif l > 0:
        m = n - l
        if m > 0:
            grad[:m] += -h[l:l+m]
            grad[l:l+m] += (1.0 - h[:m])
    else:
        g = -l
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:g+m] += -h[:m]
    return grad

def _quadratic_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for i in range(M):
        l = i - (n - 1)
        if l == 0:
            A[i] = np.dot(h, 1.0 - h)
            B[i] = np.dot(d, 1.0 - 2.0 * h)
            C[i] = -np.dot(d, d)
        elif l > 0:
            m = n - l
            A[i] = np.dot(h[l:], 1.0 - h[:m])
            B[i] = np.dot(d[l:], 1.0 - h[:m]) - np.dot(h[l:], d[:m])
            C[i] = -np.dot(d[l:], d[:m])
        else:
            g = -l
            m = n - g
            A[i] = np.dot(h[:m], 1.0 - h[g:])
            B[i] = np.dot(d[:m], 1.0 - h[g:]) - np.dot(h[:m], d[g:])
            C[i] = -np.dot(d[:m], d[g:])
    return A, B, C

def _feasible_interval(h, d):
    pos = d > 1e-15
    neg = d < -1e-15
    tlo = 0.0
    thi = 1e10
    if np.any(pos):
        tlo = max(tlo, float(np.max(-h[pos] / d[pos])))
        thi = min(thi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tlo = max(tlo, float(np.max((1.0 - h[neg]) / d[neg])))
        thi = min(thi, float(np.min(-h[neg] / d[neg])))
    return min(tlo, 0.0), max(thi, 0.0)

def _exact_1d_solve(h, d):
    """Solve min_{t in feasible} max_i f_i(t) where f_i(t) = A_i + B_i*t + C_i*t^2.
    Returns (best_t, best_max_val)."""
    n = len(h)
    M = 2 * n - 1
    tlo, thi = _feasible_interval(h, d)
    if thi - tlo < 1e-14:
        return 0.0, float(np.max(_scores(h)))
    
    A, B, C = _quadratic_coeffs(h, d)
    
    # Candidate t values
    candidates = [tlo, thi, 0.0]
    
    # Stationary points
    for i in range(M):
        if abs(C[i]) > 1e-14:
            t0 = -B[i] / (2.0 * C[i])
            if tlo - 1e-12 <= t0 <= thi + 1e-12:
                candidates.append(t0)
    
    # Pairwise intersections
    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) > 1e-14:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-14:
                    disc = max(disc, 0.0)
                    sq = np.sqrt(disc)
                    t1 = (-b + sq) / (2.0 * a)
                    t2 = (-b - sq) / (2.0 * a)
                    if tlo - 1e-12 <= t1 <= thi + 1e-12:

                    if tlo - 1e-12 <= t2 <= thi + 1e-12:

                if tlo - 1e-12 <= t1 <= thi + 1e-12:

    if len(candidates) == 0:
        return 0.0, float(np.max(A))

    candidates = np.sort(candidates)
    if len(candidates) > 1:
        diffs = np.diff(candidates)
        mask = np.concatenate(([True], diffs > 1e-9))
        candidates = candidates[mask]
    
    # Evaluate
    best_t = 0.0

def _smooth_max_dir(h, scores, top_k=30, beta=50.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        if idx >= n:
            l = idx - (n - 1)
        else:
            l = idx - (n - 1)
        d = _score_grad(h, l)
    else:
        w /= wsum
        d = np.zeros(n)
        for i in range(len(scores)):
            l = i - (n - 1)
            if w[i] > 1e-10:
                d += w[i] * _score_grad(h, l)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def _lp_dir(h, scores):
    n = len(h)
    M = 2 * n - 1
    # Build gradient matrix G (M x n)
    G = np.zeros((M, n))
    for i in range(M):
        l = i - (n - 1)
        G[i] = _score_grad(h, l)
    
    # LP: min z s.t. G @ d - z <= 0, sum(d) = 0, -1 <= d <= 1
    # Variables: d_0, ..., d_{n-1}, z
    # Total: n+1 variables
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize z
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(M)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    # Wait, d should be free (can be positive or negative), bounded by [-1, 1]
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            d = res.x[:n]
            d -= np.mean(d)
            norm = float(np.max(np.abs(d)))
            if norm > 1e-10:
                return d / norm
    except Exception:
        pass
    return None

def _random_dir(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-10:
        return None
    return d / norm

def _pair_dir(rng, n, h):
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0

        if best_raw < float(np.max(scores)) - 1e-10:
            h_new = h + best_t * best_d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
