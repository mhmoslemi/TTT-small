def _softmax_direction(G, scores, beta):
    """Compute softmax-weighted gradient direction."""
    s_max = float(np.max(scores))
    w = np.exp(beta * (scores - s_max))
    w /= np.sum(w)
    d = G.T @ w
    d -= np.mean(d)
    norm = float(np.sqrt(np.sum(d**2)))
    if norm < 1e-14:
        return np.zeros(len(d))
    return -d / norm  # descent direction

def _random_direction(n, rng):
