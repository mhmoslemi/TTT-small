import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _G_matrix(h):
    """Compute gradient matrix G where G[k,i] = d(score_k)/d(h[i])"""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] -= h[g:g+m]
                G[r, g:] += (1.0 - h[:m])
    return G

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_polish(h, target, deadline, radius_init=0.1):
    n = len(h)
    radius = radius_init
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline:
        if stall > 80:
            break
        
        scores = _scores(h)
        G = _G_matrix(h)
        
        # LP: minimize t s.t. G @ p + scores - t <= 0, sum(p) = 0, box constraints
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            stall += 5
            radius *= 0.5
            continue
        
        if not res.success or res.x is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        p = res.x[:n]
        t_pred = res.x[-1]  # predicted max new score
        
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        # Line search
        # Find max step size
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(30):
            if hi_a - lo_a < 1e-10 * alpha_max:
                break
            if f1 < f2:
                hi_a, c2, f2 = c2, c1, f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo_a, c1, f1 = c1, c2, f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(h + c2 * p)
        
        alpha = (lo_a + hi_a) / 2.0
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-13:
