def _smooth_max_grad(h, T, top_k=50):
    n = len(h)
    scores = _all_scores(h)
    # Use top-k for efficiency
    idx = np.argsort(scores)[::-1][:top_k]
    s_top = scores[idx]
    max_s = s_top[0]
    shifted = T * (s_top - max_s)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for i, r in enumerate(idx):
        if w[i] < 1e-10:
            continue
        d = int(r - (n - 1))
        grad += w[i] * _score_grad(h, d)
    return grad

def _smooth_descent(h, T_start, T_end, deadline):

    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    n_steps = 100
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Interpolate temperature
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        
        g = _smooth_max_grad(h, T, top_k=40)
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        g = -g / norm
        
        # Adaptive step size
        lr = 0.02 * (1.0 - 0.5 * frac)
        alphas = np.array([lr * f for f in [0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0]])
        best_a = 0.0
        best_val = c5
        for a in alphas:
            if a < 1e-8:
                continue
            h_t = _project(h + a * g, target)
            val = _exact_c5(h_t)
            if val < best_val - 1e-13:
                best_val = val
                best_a = a
        
        if best_a > 0:
            h = _project(h + best_a * g, target)
            c5 = best_val
            if c5 < best_c5:

def _slsqp_final(h, deadline, top_k=15, maxiter=200):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-6
    
    idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in idx]
    
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def score_tau(x, d):
        hh = x[:n]
        if d == 0:
            return float(np.sum(hh * (1.0 - hh)))
        elif d > 0:
            m = n - d
            return float(np.sum(hh[d:d+m] * (1.0 - hh[:m])))
        else:
            g2 = -d
            m = n - g2
            return float(np.sum(hh[:m] * (1.0 - hh[g2:g2+m])))
    
    def score_grad_tau(x, d):
        hh = x[:n]
        return _score_grad(hh, d)
    
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for d in active_lags:
        cons.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - score_tau(x, dd),
            'jac': lambda x, dd=d: np.concatenate([-score_grad_tau(x, dd), [1.0]])
        })
    
    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return h, _exact_c5(h)
    
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-6:
        h_out = _project(h_out, target)
    return h_out, _exact_c5(h_out)

def _resample(h, n_new):

    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    n_steps = 150
    lr = 0.03
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        g = _smooth_max_grad(h, T, top_k=40)
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        g = -g / norm
        cur_lr = lr * (1.0 - 0.6 * frac)
        alphas = np.array([cur_lr * f for f in [0.1, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]])
        best_a = 0.0
        best_val = c5
        for a in alphas:
            if a < 1e-10:
                continue
            h_t = _project(h + a * g, target)
            val = _exact_c5(h_t)
            if val < best_val - 1e-13:
                best_val = val
                best_a = a
        if best_a > 0:
            h = _project(h + best_a * g, target)
            c5 = best_val
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.2)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c5

def _subgradient_descent(h, deadline, top_k=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    lr = 0.01
    for it in range(500):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        idx = np.argsort(scores)[::-1][:top_k]
        max_s = scores[idx[0]]
        w = np.exp(50.0 * (scores[idx] - max_s))
        w /= np.sum(w)
        g = np.zeros(n)
        for i, r in enumerate(idx):
            d = int(r - (n - 1))
            g += w[i] * _score_grad(h, d)
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        g = -g / norm
        h_t = _project(h + lr * g, target)
        c5_t = _exact_c5(h_t)
        if c5_t < c5 - 1e-13:
            h = h_t
            c5 = c5_t
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
        else:
            lr *= 0.75
            if lr < 1e-6:
                break
    return best_h, best_c5

def _slsqp_final(h, deadline, top_k=12, maxiter=300):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-6
    idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in idx]
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 3.0)]

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def score_tau(hh, d):
        if d == 0:
            return float(np.sum(hh * (1.0 - hh)))
        elif d > 0:
            m = n - d
            return float(np.sum(hh[d:d+m] * (1.0 - hh[:m])))
        else:
            g2 = -d
            m = n - g2
            return float(np.sum(hh[:m] * (1.0 - hh[g2:g2+m])))

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for d in active_lags:
        cons.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - score_tau(x[:n], dd),
            'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        })

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return h, _exact_c5(h)
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-6:
        h_out = _project(h_out, target)
    return h_out, _exact_c5(h_out)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_restart_lp(h, deadline, n_restarts=50, scale_init=0.04):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    scale = scale_init
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if best_c5 < 0.38075:
            break
        per_time = (deadline - time.time()) / max(n_restarts - r, 1)
        if per_time < 1.0:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        h_try, c5_try = _lp_loop_fast(h_pert, time.time() + min(per_time * 0.8, 20.0),
                                       top_k=20, radius_init=0.04, max_iters=1500)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0.copy(), target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Fast top-k LP loop
    deadline1 = t0 + budget_s * 0.12
    h1, c5_1 = _lp_loop_fast(h, deadline1, top_k=25, radius_init=0.06, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smooth max gradient descent (adaptive T)
    deadline2 = t0 + budget_s * 0.24
    h2, c5_2 = _smooth_descent(best_h, T_start=30.0, T_end=300.0, deadline=deadline2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Another LP pass
    deadline3 = t0 + budget_s * 0.38
    h3, c5_3 = _lp_loop_fast(best_h, deadline3, top_k=20, radius_init=0.04, max_iters=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: SLSQP with tight active set
    deadline4 = t0 + budget_s * 0.50
    h4, c5_4 = _slsqp_final(best_h, deadline4, top_k=12, maxiter=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-restart LP (global search via perturbation)
    deadline5 = t0 + budget_s * 0.72
    h5, c5_5 = _multi_restart_lp(best_h, deadline5, n_restarts=60, scale_init=0.05)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Resolution exploration
    deadline6 = t0 + budget_s * 0.88
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12, 14, -14, 16, -16, 20, -20, 24]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 176 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        per_dl = min(deadline6, time.time() + 25.0)
        h_r, c5_r = _lp_loop_fast(h_r, per_dl, top_k=20, radius_init=0.05, max_iters=2000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final LP refinement at best resolution
    deadline7 = t0 + budget_s * 0.93
    h7, c5_7 = _lp_loop_fast(best_h, deadline7, top_k=15, radius_init=0.015, max_iters=5000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.97:
        h8, c5_8 = _slsqp_final(best_h, t0 + budget_s * 0.98, top_k=10, maxiter=200)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: One last micro LP pass
    if time.time() < t0 + budget_s * 0.995:
        h9, c5_9 = _lp_loop_fast(best_h, t0 + budget_s * 0.998, top_k=10,
                                  radius_init=0.005, max_iters=3000)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:

    return best_h, final_c5, best_n
