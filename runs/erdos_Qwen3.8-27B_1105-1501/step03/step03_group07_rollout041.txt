import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _bin_max_count(h01):
    """h01: int array of 0s and 1s. Returns max correlation count."""
    n = len(h01)
    corr = np.correlate(h01, 1 - h01, mode='full')
    return int(np.max(corr))

def _bin_c5(h01):
    n = len(h01)
    return 2.0 * _bin_max_count(h01) / n

def _hill_climb_binary(h01, deadline, max_stagnation=300):
    n = len(h01)
    h = h01.copy().astype(np.int8)
    cur_mc = _bin_max_count(h)
    best_mc = cur_mc
    best_h = h.copy()
    stag = 0
    
    while time.time() < deadline:
        if stag > max_stagnation:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_mc_try = cur_mc
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                mc = _bin_max_count(h)
                h[i] = 1
                h[j] = 0
                if mc < best_mc_try:
                    best_mc_try = mc
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            cur_mc = best_mc_try
            stag = 0
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_h = h.copy()
        else:
            stag += 1
    
    return best_h, best_mc

def _sa_binary(h01, deadline, T_init=10.0, T_min=0.5, cool=0.9997, seed=42):
    n = len(h01)
    h = h01.copy().astype(np.int8)
    cur_mc = _bin_max_count(h)
    best_mc = cur_mc
    best_h = h.copy()
    T = T_init
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline and T > T_min:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        
        h[i] = 0
        h[j] = 1
        new_mc = _bin_max_count(h)
        delta = new_mc - cur_mc
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_mc = new_mc
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T *= cool
    
    return best_h, best_mc

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _binarize(h, W):
    """Binarize h to have exactly W ones."""
    n = len(h)
    bits = (h >= 0.5).astype(np.int8)
    cur_W = int(np.sum(bits))
    if cur_W > W:
        # Need to flip (cur_W - W) ones to zeros
        # Flip the ones at positions where h is closest to 0.5
        one_positions = np.where(bits == 1)[0]
        # Sort by h value (ascending) - flip those with lowest h first
        order = np.argsort(h[one_positions])
        for idx in order[:cur_W - W]:
            bits[one_positions[idx]] = 0
    elif cur_W < W:
        # Need to flip (W - cur_W) zeros to ones
        zero_positions = np.where(bits == 0)[0]
        order = np.argsort(-h[zero_positions])
        for idx in order[:W - cur_W]:
            bits[zero_positions[idx]] = 1
    return bits

        wsum = float(np.sum(w))
        if wsum < 1e-30:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= wsum
        
        # Compute gradient
        grad = np.zeros(n)
        p = n - 1
        for r in range(2 * n - 1):
            if w[r] < 1e-10:

                    grad[:m] -= w[r] * h[d:]

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:

        grad /= gnorm

        v -= np.mean(v)
        
        # Line search
        best_a = 0.0
        best_c = cur_c5

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _bin_mc(h01):
    corr = np.correlate(h01, 1 - h01, mode='full')
    return int(np.max(corr))

def _hill_climb(hw, deadline, max_stag=200):
    """hw: int8 array. Returns (best_h, best_mc)."""
    n = len(hw)
    h = hw.copy()
    cur = _bin_mc(h)
    best_h = h.copy()
    best_mc = cur
    stag = 0
    while time.time() < deadline:
        if stag > max_stag:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_try = cur
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                mc = _bin_mc(h)
                h[i] = 1; h[j] = 0
                if mc < best_try:
                    best_try = mc
                    best_pair = (i, j)
        if best_pair:
            h[best_pair[0]] = 0; h[best_pair[1]] = 1
            cur = best_try
            stag = 0
            if cur < best_mc:
                best_mc = cur
                best_h = h.copy()
        else:
            stag += 1
    return best_h, best_mc

def _sa(hw, deadline, T0=8.0, Tmin=0.3, cool=0.9999, seed=42):
    n = len(hw)
    h = hw.copy()
    cur = _bin_mc(h)
    best_h = h.copy()
    best_mc = cur
    T = T0
    rng = np.random.default_rng(seed)
    while time.time() < deadline and T > Tmin:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i] = 0; h[j] = 1
        nm = _bin_mc(h)
        d = nm - cur
        if d <= 0 or rng.random() < np.exp(-d / T):
            cur = nm
            if cur < best_mc:
                best_mc = cur
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
        T *= cool
    return best_h, best_mc

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = float(np.sum(np.clip(v - mid, 0, 1)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)
    r = target - float(np.sum(x))
    if abs(r) > 1e-12:
        if r > 0: mask = x < 1 - 1e-10
        else: mask = x > 1e-10
        c = int(np.sum(mask))
        if c > 0: x[mask] += r / c
    return np.clip(x, 0, 1)

def _binarize(h, W):
    n = len(h)
    b = (h >= 0.5).astype(np.int8)
    cW = int(np.sum(b))
    if cW > W:
        op = np.where(b == 1)[0]
        ord_ = np.argsort(h[op])
        b[op[ord_[:cW-W]]] = 0
    elif cW < W:
        zp = np.where(b == 0)[0]
        ord_ = np.argsort(-h[zp])
        b[zp[ord_[:W-cW]]] = 1
    return b

def _resample(h, nn):
    if nn == len(h): return h.copy()
    xo = (np.arange(len(h)) + 0.5) / len(h)
    xn = (np.arange(nn) + 0.5) / nn
    return _project(np.interp(xn, xo, h), nn / 2)

def _cont_opt(hs, target, deadline, maxit=5000):
    n = len(hs)
    h = _project(hs, target)
    cur = _eval_c5(h)
    best_h = h.copy()
    best_c = cur
    T, alpha, v, beta = 0.3, 0.06, np.zeros(n), 0.5
    stag = 0
    for _ in range(maxit):
        if time.time() > deadline or stag > 100: break
        corr = np.correlate(h, 1-h, mode='full')
        sh = corr - float(np.max(corr))
        w = np.exp(sh / T)
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.zeros(len(corr)); w[int(np.argmax(corr))] = 1
        else: w /= ws
        grad = np.zeros(n)
        p = n - 1
        for r in range(2*n-1):
            if w[r] < 1e-10: continue
            d = r - p
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[d:] += w[r]*(1-h[:m]); grad[:m] -= w[r]*h[d:]
            else:
                l = -d; m = n - l
                if m > 0:
                    grad[:m] += w[r]*(1-h[l:]); grad[l:] -= w[r]*h[:m]
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12: break
        grad /= gn
        v = beta*v - alpha*grad; v -= np.mean(v)
        ba, bc = 0, cur
        for a in [0.01, 0.03, 0.1, 0.3, 0.8]:
            ht = _project(h + a*v, target)
            ct = _eval_c5(ht)
            if ct < bc - 1e-14: bc, ba = ct, a
        if ba > 0 and bc < cur - 1e-14:
            h = _project(h + ba*v, target)
            cur = _eval_c5(h)
            stag = 0; T = max(0.05, T*0.99); beta = min(0.9, beta+0.001)
            if cur < best_c: best_c, best_h = cur, h.copy()
        else:
            stag += 1; alpha *= 0.9; T = max(0.1, T*0.95)
    return best_h, best_c

import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _bin_mc(h):
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - float(np.sum(x))
    if abs(r) > 1e-12:
        if r > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        c = int(np.sum(mask))
        if c > 0:
            x[mask] += r / c
    return np.clip(x, 0.0, 1.0)

def _binarize(h, W):
    n = len(h)
    b = (h >= 0.5).astype(np.int8)
    cW = int(np.sum(b))
    if cW > W:
        op = np.where(b == 1)[0]
        ord_ = np.argsort(h[op])
        b[op[ord_[:cW - W]]] = 0
    elif cW < W:
        zp = np.where(b == 0)[0]
        ord_ = np.argsort(-h[zp])
        b[zp[ord_[:W - cW]]] = 1
    return b

def _resample(h, nn):
    if nn == len(h):
        return h.copy()
    xo = (np.arange(len(h)) + 0.5) / len(h)
    xn = (np.arange(nn) + 0.5) / nn
    return _project(np.interp(xn, xo, h), nn / 2.0)

def _hill_climb(hw, deadline, max_stag=200):
    n = len(hw)

    cur = _bin_mc(h)
    best_h = h.copy()
    best_mc = cur
    stag = 0
    while time.time() < deadline:
        if stag > max_stag:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_try = cur
        best_pair = None
        for i in ones:
            for j in zeros:

                h[j] = 1
                mc = _bin_mc(h)
                h[i] = 1
                h[j] = 0
                if mc < best_try:

        if best_pair is not None:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            cur = best_try
            stag = 0
            if cur < best_mc:
                best_mc = cur
                best_h = h.copy()
        else:
            stag += 1
    return best_h, best_mc
