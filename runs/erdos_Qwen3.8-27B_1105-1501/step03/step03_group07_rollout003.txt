def _lag_gradient(h, d):

        m = n - l

def _slsqp_refine(h, deadline, target, T=0.05):
    """Use SLSQP with smoothed max objective"""
    n = len(h)
    h_flat = h.astype(float)
    
    def objective(x):
        h_t = np.clip(x, 0, 1)
        scores = _scores(h_t)
        shifted = scores - float(np.max(scores))
        # Soft max
        return float(np.log(np.sum(np.exp(shifted / T))) * T)
    
    def constraint_sum(x):
        return float(np.sum(x)) - target
    
    # SLSQP
    try:
        res = minimize(objective, h_flat, method='SLSQP',
                      bounds=[(0, 1)] * n,
                      constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or (hasattr(res, 'x') and res.x is not None):
            h_new = _project(res.x, target)
            c = _exact_c5(h_new)
            return h_new, c
    except Exception:
        pass
    return h_flat.copy(), _exact_c5(h_flat)

                dj = 1.0 if rng.random() > 0.5 else -1.0
                dk = -dj
                a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                a_max = min(a_max_j, a_max_k)
                if a_max < 1e-10:
                    continue

                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-12:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_h, best_c5

def _ternary_search(h0, deadline, target):
    """Search over ternary (0, t, 1) assignments"""
    n = len(h0)
    h0 = h0.copy()
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # For each t in a grid, assign each coordinate to {0, t, 1-t, 1}
    # based on h0's value, then optimize
    t_values = np.linspace(0.1, 0.45, 15)
    for t in t_values:
        if time.time() > deadline:
            break
        # Assign levels based on h0
        levels = [0.0, t, 1-t, 1.0]
        h_assigned = np.zeros(n)
        for i in range(n):
            # Find closest level
            diffs = [abs(h0[i] - lv) for lv in levels]
            h_assigned[i] = levels[int(np.argmin(diffs))]
        h_assigned = _project(h_assigned, target)
        c = _exact_c5(h_assigned)
        if c < best_c5:
            best_c5 = c
            best_h = h_assigned.copy()
    
    return best_h, best_c5
