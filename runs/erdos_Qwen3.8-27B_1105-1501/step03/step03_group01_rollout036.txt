    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(15):
        a = alpha_max * (i + 1) / 15.0
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a = max(0.0, best_a * 0.6)
    hi_a = min(alpha_max, best_a * 1.4)
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = _corr_vec(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _slsqp_solve(h_init, deadline, top_k=None):
    """SLSQP solver on epigraph formulation."""
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init, target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    if top_k is None:
        top_k = 2 * n - 1  # use all constraints
    
    def _obj(x):
        return x[n]
    
    def _obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def _eq_fun(x):
        return np.array([np.sum(x[:n]) - target])
    
    def _eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J
    
    while time.time() < deadline:
        cv = _corr_vec(h)
        max_cv = float(np.max(cv))
        # Select active constraints
        if top_k < 2 * n - 1:
            # Get top-k, but also include some margin
            threshold = max_cv - 0.02
            active_mask = cv >= threshold
            if np.sum(active_mask) > top_k:
                top_idx = np.argsort(cv)[-top_k:]
                active_mask = np.zeros(2 * n - 1, dtype=bool)
                active_mask[top_idx] = True
            active_idx = np.where(active_mask)[0]
        else:
            active_idx = np.arange(2 * n - 1)
        
        k_act = len(active_idx)
        
        def _ineq_fun(x):
            h_x = np.clip(x[:n], 0, 1)
            t = x[n]
            cv_x = _corr_vec(h_x)
            return t - cv_x[active_idx]  # >= 0
        
        def _ineq_jac(x):
            h_x = np.clip(x[:n], 0, 1)
            J = np.zeros((k_act, n + 1))
            for i, r in enumerate(active_idx):
                d = r - (n - 1)
                if d >= 0:
                    m = n - d
                    if m > 0:
                        J[i, d:n] += (1.0 - h_x[:m])
                        J[i, :m] += -h_x[d:n]
                else:
                    g_val = -d
                    m = n - g_val
                    if m > 0:
                        J[i, :m] += (1.0 - h_x[g_val:n])
                        J[i, g_val:n] += -h_x[:m]
                J[i, n] = 1.0
            return J
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        x0 = np.concatenate([h, [max_cv + 1e-5]])
        
        constraints = [
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        ]
        
        try:
            res = minimize(_obj, x0, jac=_obj_grad, bounds=bounds,
                          constraints=constraints, method='SLSQP',
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.x is not None and len(res.x) == n + 1:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Only consider "interior" variables
        interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(interior) == 0:
            break
        for j in interior:
            if time.time() > deadline:
                break
            for k in interior:
                if k <= j:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = min((1.0 - h[j]) / max(dj, 0.01), h[j] / max(-dj, 0.01),
                             (1.0 - h[k]) / max(dk, 0.01), h[k] / max(-dk, 0.01))
                    if am < 1e-10:
                        continue
                    h_t = h.copy()
                    h_t[j] += dj * am * 0.5
                    h_t[k] += dk * am * 0.5
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _soft_grad(h, tau):
    n = len(h)
    cv = _corr_vec(h)
    max_cv = float(np.max(cv))
    exp_s = np.exp(tau * (cv - max_cv))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = G.T @ w
    return grad

def _grad_search(h, deadline, tau=80.0, lr=0.01, momentum_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad = _soft_grad(h, tau)
        mom = momentum_beta * mom + (1.0 - momentum_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.08)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
            if stall % 15 == 0:
                h = best_h + np.random.normal(0, 0.015, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _evolutionary(h_init, deadline, pop_size=10, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.03
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2) + rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.92)
            else:
                mut_scale = min(0.1, mut_scale * 1.05)
        else:
            mut_scale = min(0.1, mut_scale * 1.03)
    return best_h, best_f

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G; A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success: return res.x[:n]
    except Exception: pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14: alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14: alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for i in range(12):
        a = alpha_max * (i + 1) / 12.0
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-12: best_c5, best_a = c5_t, a
    if best_a == 0.0: return 0.0, c5_cur
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a, hi_a = max(0.0, best_a * 0.6), min(alpha_max, best_a * 1.4)
    c1, c2 = hi_a - gr*(hi_a-lo_a), lo_a + gr*(hi_a-lo_a)
    f1, f2 = _exact_c5(h+c1*p), _exact_c5(h+c2*p)
    for _ in range(25):
        if hi_a - lo_a < 1e-12: break
        if f1 < f2:

            c1 = hi_a - gr*(hi_a-lo_a)
            f1 = _exact_c5(h+c1*p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr*(hi_a-lo_a)
            f2 = _exact_c5(h+c2*p)
    opt_a = (lo_a+hi_a)/2.0
    opt_c5 = _exact_c5(h+opt_a*p)
    return (opt_a, opt_c5) if opt_c5 < best_c5 else (best_a, best_c5)

def _lp_loop(h, deadline, radius_init=0.08):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius, stall = radius_init, 0
    while time.time() < deadline:
        if stall > 60: break
        scores = _corr_vec(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1; radius *= 0.7
            if radius < 1e-7: break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = _project(h + alpha*p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5: best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.4)
        else:
            stall += 1; radius *= 0.72

    return best_h, best_c5

def _slsqp_solve(h_init, deadline, top_k=25, margin=0.015):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init, target)
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    
    def _obj(x): return x[n]
    def _obj_grad(x):
        g = np.zeros(n+1); g[n] = 1.0; return g
    def _eq_fun(x): return np.array([np.sum(x[:n]) - target])
    def _eq_jac(x):
        J = np.zeros((1, n+1)); J[0, :n] = 1.0; return J
    
    while time.time() < deadline:
        cv = _corr_vec(h)
        max_cv = float(np.max(cv))
        # Active set: constraints within margin of max, capped at top_k
        mask = cv >= (max_cv - margin)
        if np.sum(mask) > top_k:
            top_idx = np.argsort(cv)[-top_k:]
            mask = np.zeros(2*n-1, dtype=bool)
            mask[top_idx] = True
        active_idx = np.where(mask)[0]
        k_act = len(active_idx)
        
        def _ineq_fun(x, _act=active_idx):
            cv_x = _corr_vec(np.clip(x[:n], 0, 1))
            return x[n] - cv_x[_act]
        
        def _ineq_jac(x, _act=active_idx):
            G = _grad_matrix(np.clip(x[:n], 0, 1))
            J = np.zeros((k_act, n+1))
            J[:, :n] = -G[_act, :]
            J[:, n] = 1.0
            return J
        
        bounds = [(0.0, 1.0)] * (n+1)
        x0 = np.concatenate([h, [max_cv + 0.005]])
        
        constraints = [
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        ]
        
        try:
            res = minimize(_obj, x0, jac=_obj_grad, bounds=bounds,
                          constraints=constraints, method='SLSQP',
                          options={'maxiter': 150, 'ftol': 1e-9})
            if res.x is not None and len(res.x) == n+1:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5, best_h = c5_new, h_new.copy()
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new
