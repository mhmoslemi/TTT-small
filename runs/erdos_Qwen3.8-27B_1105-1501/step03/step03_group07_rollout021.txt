        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

        def fiq(x):
            h = x[:n]
            t = x[n]

    if d >= 0:
        L = n - d
        if L > 0:
            grad[:L] += (1.0 - h[d:])
            grad[d:] -= h[:L]
    else:
        k = -d
        L = n - k
        if L > 0:
            grad[k:] += (1.0 - h[:L])
            grad[:L] -= h[k:]
    return grad

def _project(v, target):
    n = len(v)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

class _LSE:
    def __init__(self, n, tau):
        self.n = n
        self.tau = tau
        self.dx = 2.0 / n
    
    def fun(self, x):
        h = x
        scores = np.correlate(h, 1.0 - h, mode='full') * self.dx
        m = float(np.max(scores))
        shifted = (scores - m) / self.tau
        exp_s = np.exp(shifted)
        return m + self.tau * np.log(float(np.sum(exp_s)))
    
    def jac(self, x):
        h = x
        scores = np.correlate(h, 1.0 - h, mode='full') * self.dx
        m = float(np.max(scores))
        shifted = (scores - m) / self.tau
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        g = np.zeros(self.n)
        for r in range(2 * self.n - 1):
            if w[r] < 1e-10:
                continue
            d = r - (self.n - 1)
            g += w[r] * self.dx * _lag_grad_unscaled(h, d)
        return g
    
    def eq_fun(self, x):
        return float(np.sum(x)) - self.n / 2.0
    
    def eq_jac(self, x):
        return np.ones(self.n)


class _Epigraph:
    def __init__(self, n, active_lags, t_ub):
        self.n = n
        self.active_lags = active_lags
        self.t_ub = t_ub
        self.dx = 2.0 / n
        # Precompute the lag offsets
        self.offsets = [r - (n - 1) for r in active_lags]
    
    def fun(self, x):
        return x[self.n]  # t
    
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    
    def ineq_fun(self, x):
        """Vector of inequality constraint values: t - c[r](h) >= 0"""
        h = x[:self.n]
        t = x[self.n]
        vals = np.zeros(len(self.offsets))
        for i, d in enumerate(self.offsets):
            if d >= 0:
                L = self.n - d
                vals[i] = t - self.dx * float(np.sum(h[:L] * (1.0 - h[d:])))
            else:
                k = -d
                L = self.n - k
                vals[i] = t - self.dx * float(np.sum(h[k:] * (1.0 - h[:L])))
        return vals
    
    def ineq_jac(self, x):
        h = x[:self.n]
        m = len(self.offsets)
        J = np.zeros((m, self.n + 1))
        for i, d in enumerate(self.offsets):
            J[i, self.n] = 1.0
            lg = _lag_grad_unscaled(h, d)
            J[i, :self.n] = -self.dx * lg
        return J
    
    def eq_fun(self, x):
        return float(np.sum(x[:self.n])) - self.n / 2.0
    
    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def _lse_anneal(h_init, deadline, maxiter_per_tau=200):
    n = len(h_init)
    h = h_init.copy()
    c5 = _exact_c5(h)
    
    # Temperature schedule
    c5_base = c5
    taus = [
        max(0.02, 0.05 * c5_base),
        max(0.01, 0.03 * c5_base),
        max(0.005, 0.02 * c5_base),
        max(0.002, 0.01 * c5_base),
        max(0.001, 0.005 * c5_base),
        max(0.0005, 0.002 * c5_base),
        max(0.0002, 0.001 * c5_base),
    ]
    
    for tau in taus:
        if time.time() > deadline:
            break
        lse = _LSE(n, tau)
        x0 = _project(h, n / 2.0)
        
        constraints = [{'type': 'eq', 'fun': lse.eq_fun, 'jac': lse.eq_jac}]
        bounds = [(0.0, 1.0)] * n
        
        res = minimize(lse.fun, x0, jac=lse.jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter_per_tau, 'ftol': 1e-12})
        
        if res.x is not None:
            h_new = _project(res.x, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
    
    return h, c5
