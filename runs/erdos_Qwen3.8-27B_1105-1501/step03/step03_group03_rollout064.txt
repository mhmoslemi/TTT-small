def _lp_descent(h, G, scores, radius, band=0.15):

    LP to find a direction p with sum(p)=0, |p|<=radius componentwise,
    such that scores + G@p are all reduced.
    We minimize the new maximum score.

    # s.t. scores + G@p - t <= 0  (each score reduced below t)
    #      sum(p) = 0
    #      -radius <= p_i <= radius
    #      t >= 0

    c[-1] = 1.0  # minimize t

    A_ub = np.zeros((L, n + 1))

    A_ub[:, -1] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, n + 1))

    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((0.0, None))  # t >= 0

def _random_direction(n, rng):
