def _pairwise_roots(A, B, C, a_lo, a_hi):
    m = len(A)
    roots = []
    # Vectorized pairwise
    i_idx, j_idx = np.triu_indices(m, k=1)
    dC = C[i_idx] - C[j_idx]
    dB = B[i_idx] - B[j_idx]
    dA = A[i_idx] - A[j_idx]
    
    # Quadratic case
    quad_mask = np.abs(dC) > 1e-14
    if np.any(quad_mask):
        dCq = dC[quad_mask]
        dBq = dB[quad_mask]
        dAq = dA[quad_mask]
        disc = dBq * dBq - 4.0 * dCq * dAq
        valid = disc >= 0
        if np.any(valid):
            disc_v = disc[valid]
            dC_v = dCq[valid]
            dB_v = dBq[valid]
            sq = np.sqrt(disc_v)
            r1 = (-dB_v + sq) / (2.0 * dC_v)
            r2 = (-dB_v - sq) / (2.0 * dC_v)
            roots.extend(r1[(r1 >= a_lo) & (r1 <= a_hi)].tolist())
            roots.extend(r2[(r2 >= a_lo) & (r2 <= a_hi)].tolist())
    
    # Linear case
    lin_mask = (np.abs(dC) <= 1e-14) & (np.abs(dB) > 1e-14)
    if np.any(lin_mask):
        r = -dA[lin_mask] / dB[lin_mask]
        roots.extend(r[(r >= a_lo) & (r <= a_hi)].tolist())
    
    return roots

def _build_G(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m_ = n - d
            # S_d = sum_{i=0}^{m_-1} h[i+d]*(1-h[i])

    n = len(v)

    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0

        if np.sum(np.clip(v - mid, 0, 1)) > target: lo = mid
        else: hi = mid

        c = m.sum()
        if c: x[m] = np.clip(x[m] + r/c, 0, 1)

def _aival(h, v):
    alo, ahi = -1e10, 1e10
    p = v > 1e-14
    if p.any():
        alo = max(alo, (-h[p]/v[p]).min())
        ahi = min(ahi, ((1-h[p])/v[p]).min())

    B = np.correlate(v, 1.0-h, 'full') - np.correlate(h, v, 'full')
    C = -np.correlate(v, v, 'full')
