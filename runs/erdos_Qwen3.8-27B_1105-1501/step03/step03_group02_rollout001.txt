Now, the direction generation. Let me think about what gives the best directions:

1. **Smooth-max gradient**: 
   - Compute scores = _scores(h)
   - max_score = scores.max()
   - weights = exp(beta * (scores - max_score) / max_score)
   - grad = sum_r weights[r] * _grad_all(h)[r]
   - Subtract mean: grad -= grad.mean()
   - Normalize: grad /= max(abs grad)
   - Direction is -grad (to reduce the max)

2. **Top-k gradient**: Average gradient of top k scores, subtract mean.

3. **Random**: Random zero-sum vector.

4. **Pair exchange**: e_i - e_j for specific pairs.

Let me think about the smooth-max gradient more carefully. The objective is:
F(h) = max_r Q_r(h)

The subgradient of F at h is the convex hull of {grad Q_r : Q_r = F(h)}. A smoothed version uses:
F_β(h) = (1/β) log(sum_r exp(β * Q_r(h) / F(h)))

The gradient is:
grad F_β = sum_r w_r * grad Q_r / F(h)
where w_r = exp(β * Q_r / F) / sum_s exp(β * Q_s / F)

For practical purposes, I'll use:
w_r = exp(β * (Q_r - Q_max)) 
grad = sum_r (w_r / sum(w)) * grad Q_r
Then subtract mean and negate.

The parameter β controls the "sharpness" - large β focuses on the top scores, small β spreads across all.

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_score(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        if m > 0:
            g[:m] += -h[lag:lag+m]
            g[lag:] += (1.0 - h[:m])
    else:
        gg = -lag
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:gg+m])
            g[gg:] += -h[:m]
    return g

def _quad_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for r in range(M):
        lag = r - (n - 1)
        if lag == 0:
            A[r] = np.dot(h, 1.0 - h)
            B[r] = np.dot(d, 1.0 - 2.0 * h)
            C[r] = -np.dot(d, d)
        elif lag > 0:
            m = n - lag
            A[r] = np.dot(h[lag:lag+m], 1.0 - h[:m])
            B[r] = np.dot(d[lag:lag+m], 1.0 - h[:m]) - np.dot(h[lag:lag+m], d[:m])
            C[r] = -np.dot(d[lag:lag+m], d[:m])
        else:
            gg = -lag
            m = n - gg
            A[r] = np.dot(h[:m], 1.0 - h[gg:gg+m])
            B[r] = np.dot(d[:m], 1.0 - h[gg:gg+m]) - np.dot(h[:m], d[gg:gg+m])
            C[r] = -np.dot(d[:m], d[gg:gg+m])
    return A, B, C

def _feasible_interval(h, d):
    tlo = 0.0
    thi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        tlo = max(tlo, float(np.max(-h[pos] / d[pos])))
        thi = min(thi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tlo = max(tlo, float(np.max((1.0 - h[neg]) / d[neg])))
        thi = min(thi, float(np.min(-h[neg] / d[neg])))
    return tlo, thi

def _exact_1d_solve(h, d, current_c5):
    """
    Solve min E(t) over feasible interval for line h+t*d.
    E(t) = max_r (A_r + B_r*t + C_r*t^2)
    Returns (t_best, c5_exact) if improvement found, else (None, current_c5).
    """
    n = len(h)
    dx = 2.0 / n
    A, B, C = _quad_coeffs(h, d)
    tlo, thi = _feasible_interval(h, d)
    if thi - tlo < 1e-14:
        return None, current_c5
    
    M = len(A)
    
    # Collect candidate t values
    cands = [tlo, thi, 0.0]
    
    # Stationary points
    nonzero_C = np.abs(C) > 1e-14
    if np.any(nonzero_C):
        t0 = -B[nonzero_C] / (2.0 * C[nonzero_C])
        t0 = t0[np.abs(t0) < 1e6]
        t0 = t0[(t0 >= tlo - 1e-9) & (t0 <= thi + 1e-9)]
        cands.extend([float(np.clip(t, tlo, thi)) for t in t0])
    
    # Pairwise crossings (vectorized)
    i_idx, j_idx = np.triu_indices(M, k=1)
    a_arr = C[i_idx] - C[j_idx]
    b_arr = B[i_idx] - B[j_idx]
    c_arr = A[i_idx] - A[j_idx]
    
    linear_mask = np.abs(a_arr) < 1e-14
    if np.any(linear_mask):
        lb = b_arr[linear_mask]
        lc = c_arr[linear_mask]
        valid = np.abs(lb) > 1e-14
        if np.any(valid):
            tl = -lc[valid] / lb[valid]
            tl = tl[np.abs(tl) < 1e6]
            tl = tl[(tl >= tlo - 1e-9) & (tl <= thi + 1e-9)]
            cands.extend([float(np.clip(t, tlo, thi)) for t in tl])
    
    quad_mask = ~linear_mask
    if np.any(quad_mask):
        qa = a_arr[quad_mask]
        qb = b_arr[quad_mask]
        qc = c_arr[quad_mask]
        disc = qb**2 - 4.0 * qa * qc
        real = disc >= -1e-12
        if np.any(real):
            disc_r = np.maximum(disc[real], 0.0)
            sq = np.sqrt(disc_r)
            t1 = (-qb[real] + sq) / (2.0 * qa[real])
            t2 = (-qb[real] - sq) / (2.0 * qa[real])
            for t_arr in [t1, t2]:
                t_arr = t_arr[np.abs(t_arr) < 1e6]
                t_arr = t_arr[(t_arr >= tlo - 1e-9) & (t_arr <= thi + 1e-9)]
                cands.extend([float(np.clip(t, tlo, thi)) for t in t_arr])
    
    cand_arr = np.clip(np.array(cands), tlo, thi)
    cand_arr = np.unique(np.round(cand_arr, 10))
    
    if len(cand_arr) == 0:
        return None, current_c5
    
    # Vectorized envelope evaluation
    # Shape: (K, M) -> max over M -> (K,)
    env = np.max(A[None, :] + B[None, :] * cand_arr[:, None] + C[None, :] * cand_arr[:, None]**2, axis=1)
    
    # Check best candidates
    sorted_idx = np.argsort(env)
    target_env = current_c5 / dx  # raw score equivalent
    
    for idx in sorted_idx[:10]:
        if env[idx] >= target_env - 1e-10:
            break
        t = cand_arr[idx]
        h_try = np.clip(h + t * d, 0.0, 1.0)
        s = np.sum(h_try)
        if abs(s - n/2.0) > 1e-8:
            h_try = _project(h_try, n/2.0)
        c5_val = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx)
        if
