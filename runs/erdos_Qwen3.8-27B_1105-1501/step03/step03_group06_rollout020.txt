import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        idx = np.argsort(x)
        if res > 0:
            for i in reversed(idx):
                add = min(res, 1.0 - x[i])
                x[i] += add
                res -= add
                if abs(res) < 1e-12:
                    break
        else:
            for i in idx:
                sub = min(-res, x[i])
                x[i] -= sub
                res += sub
                if abs(res) < 1e-12:
                    break
    return x

def _grad_matrix(h):
    """Compute gradient matrix for all lags. G[r, j] = dC_r/dh_j"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for s in range(-(n-1), n):
        r = n - 1 + s
        if s >= 0:
            m = n - s
            # h_i(1-h_{i+s}) for i=0..m-1
            # d/dh_j: for j < m, contribution from (1-h_{j+s}); for j >= s, contribution from -h_{j-s}
            G[r, :m] += one_minus_h[s:s+m]
            G[r, s:] += -h[:m]
        else:
            g = -s
            m = n - g
            # h_i(1-h_{i-g}) for i=g..n-1 (reindexing)
            # Actually for negative shift s = -g:
            # C = sum h_i(1-h_{i-g}) for i where i-g >= 0, i.e., i >= g
            # d/dh_j: for j >= g, contribution from (1-h_{j-g})
            #         for j < n-g, contribution from -h_{j+g} (where j+g is the "first" element)
            G[r, g:] += one_minus_h[0:m]  # h_i(1-h_{i-g}), i from g to n-1, h_{i-g} from 0 to n-g-1
            G[r, :m] += -h[g:g+m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if res > 1e-10:
        idx = np.argsort(-x)
        for i in idx:
            add = min(res, 1.0 - x[i])
            if add > 0:
                x[i] += add
                res -= add
                if res < 1e-12:
                    break
    elif res < -1e-10:
        idx = np.argsort(x)
        for i in idx:
            sub = min(-res, x[i])
            if sub > 0:
                x[i] -= sub
                res += sub
                if -res < 1e-12:
                    break
    return x

```python
def _active_lp_step(h, corr, G, k, radius):
    """Solve LP with top-k active lags, return new h."""
    n = len(h)
    # Find top-k lags
    top_idx = np.argsort(corr)[-k:]
    G_active = G[top_idx]
    scores_active = corr[top_idx]
    
    # LP: min t s.t. G_active @ p - t <= -scores_active, sum(p)=0, box
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -scores_active
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None
    p = res.x[:n]
    if np.max(np.abs(p)) < 1e-14:
        return None, None
    return p, res.x[-1]  # step direction and predicted max
```
