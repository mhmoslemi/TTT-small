import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    for _ in range(100):
        s = float(v.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-14
        else:
            mask = v > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        v[mask] += step
        v = np.clip(v, 0.0, 1.0)
    # Final exact correction
    diff = target - float(v.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            candidates = np.where(v < 0.9999999)[0]
        else:
            candidates = np.where(v > 1e-7)[0]
        if len(candidates) > 0:
            v[candidates[0]] += diff
            v[candidates[0]] = float(np.clip(v[candidates[0]], 0.0, 1.0))
    return v

def _build_gradient(h):
    """Build the Jacobian of all scores w.r.t. h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            for i in range(m):
                G[r, i] += dx * (1.0 - h[i + d])
                G[r, i + d] += -dx * h[i]
        else:
            g2 = -d
            m = n - g2
            for i in range(m):
                G[r, i + g2] += dx * (1.0 - h[i])
                G[r, i] += -dx * h[i + g2]
    return G

def _cvxpy_trust_step(h, rho, solver=None):
    """Solve one trust-region subproblem using CVXPY."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = _scores_raw(h) * dx
    G = _build_gradient(h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        cp.norm(x - h, 2) <= rho,
        G @ (x - h) + scores <= t
    ]
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
        if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
            x_val = np.array(x.value).flatten()
            x_val = _project_box_sum(x_val, n / 2.0)
            return x_val
    except Exception:
        pass
    return None

            if n_fails > 10:

            continue
        
        # Try different step sizes
        best_trial_c5 = cur_c5
        best_trial_h = h_work
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            trial = h_work + frac * (x_new - h_work)
            trial = _project_box_sum(trial, n / 2.0)
            c5_trial = _c5_exact(trial)
            if c5_trial < best_trial_c5:
                best_trial_c5 = c5_trial
                best_trial_h = trial
        
        if best_c5 - best_trial_c5 > 1e-12:
            h_work = best_trial_h
            cur_c5 = best_trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_work.copy()
            rho = min(rho * 1.2, max_rho)
            stall = 0
            n_fails = 0
        else:
            stall += 1
            rho *= 0.7
            if stall > 30:
                break
    
    return best_h, best_c5

def _binary_hill_climb(h, deadline, rng):
    """Binary hill-climbing with single and multi-swaps."""
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    cur_c5 = _c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    
    while time.time() < deadline and no_improve < 500:
        # Try single swaps
        best_swap = None
        best_val = cur_c5
        batch = min(50, len(ones) * len(zeros))
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _c5_exact(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            # Try multi-swap (2 pairs)
            improved = False
            for _ in range(10):
                i1 = ones[rng.integers(len(ones))]
                i2 = ones[rng.integers(len(ones))]
                j1 = zeros[rng.integers(len(zeros))]
                j2 = zeros[rng.integers(len(zeros))]
                if i1 == i2 or j1 == j2:
                    continue
                h_bin[i1] = 0.0
                h_bin[i2] = 0.0
                h_bin[j1] = 1.0
                h_bin[j2] = 1.0
                c5_new = _c5_exact(h_bin)
                h_bin[i1] = 1.0
                h_bin[i2] = 1.0
                h_bin[j1] = 0.0
                h_bin[j2] = 0.0
                if c5_new < cur_c5 - 1e-14:
                    # Accept
                    h_bin[i1] = 0.0
                    h_bin[i2] = 0.0
                    h_bin[j1] = 1.0
                    h_bin[j2] = 1.0
                    ones.remove(i1)
                    ones.remove(i2)
                    zeros.remove(j1)
                    zeros.remove(j2)
                    zeros.extend([i1, i2])
                    ones.extend([j1, j2])
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h_bin.copy()
                    improved = True
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Use trapezoidal rule for more accurate resampling
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _pair_optimize(h, deadline, max_passes=100):
    """Pairwise coordinate optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Optimize h[j] and h[k] with h[j]+h[k] = const
                s = h[j] + h[k]
                # h[j] = s/2 + t, h[k] = s/2 - t
                # t in [-min(s/2, 1-s/2), min(s/2, 1-s/2)]
                t_max = min(s / 2.0, 1.0 - s / 2.0)
                if t_max < 1e-10:
                    continue
                # Try a few values of t
                best_t = 0.0
                best_val = _c5_exact(h)
                for t in np.linspace(-t_max, t_max, 11):
                    if abs(t) < 1e-12:
                        continue
                    h[j] = s / 2.0 + t
                    h[k] = s / 2.0 - t
                    c5_val = _c5_exact(h)
                    if c5_val < best_val - 1e-14:
                        best_val = c5_val
                        best_t = t
                # Restore and apply best
                h[j] = s / 2.0 + best_t
                h[k] = s / 2.0 - best_t
                if best_t != 0.0:
                    improved = True
                    if _c5_exact(h) < best_c5:
                        best_c5 = _c5_exact(h)
                        best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c5
