        k_start2 = max(0, shift)
        k_end2 = min(n, n + shift)  # exclusive
        if k_end2 > k_start2:
            row[k_start2:k_end2] += -h[k_start2-shift:k_end2-shift]
        G[r] = row * dx
    return G
```

This is O(n²) total, which is much better.

def _lp_full(h, scores, G, radius):
    """Solve LP: min t s.t. G@d - t <= -s, sum(d)=0, bounds on d"""

    A_ub[:, n] = -1.0
    b_ub = -scores

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))

                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

Let me also think about a completely different approach: **interior point method for the non-smooth problem**.

The idea: minimize f(h) = max_d s_d(h) + (mu/2) * sum_d max(0, s_d(h) - t)² + (nu/2) * (sum(h) - T)² + barrier terms

As beta -> infinity, this approaches max_d s_d. And f_smooth is smooth (C-infinity), so we can use gradient-based optimization.

The gradient: ∇f_smooth = (sum_d s'_d * exp(beta * s_d)) / (sum_d exp(beta * s_d))
= weighted average of gradients, with weights proportional to exp(beta * s_d) / sum exp(beta * s_d)

This is exactly the "softmax-weighted subgradient" that the previous code's `_smooth_optimize` uses! So the previous code already has this idea.

**Let me just focus on making the LP approach better:**

Key insight: The LP direction with all lags gives the "best" feasible direction that decreases all active lags simultaneously. If the LP is infeasible (no direction can decrease all active lags), we're at a local optimum of the LP approximation. In that case, we should:
1. Reduce the radius (to allow smaller, more targeted moves)
2. Or add slack variables to allow some lags to increase slightly

Actually, the LP is:
min t
s.t. G @ d - t <= -s  (i.e., s + G@d <= t)
     sum(d) = 0
     |d_i| <= radius (and box constraints)

The optimal t tells us the best we can do: if t < 0, we can decrease the max score by |t| (approximately). If t >= 0, no feasible direction within the radius can decrease the max.

Wait, actually the constraint is s + G@d <= t, and we minimize t. At the current point, s + G@0 = s, so t >= max(s). The LP finds the direction d that minimizes the maximum score in the linearized model.

If the optimal t < max(s), we have a descent direction. If t = max(s), we're at a local optimum of the linear model.

The radius controls how far we can move. A smaller radius gives a more local approximation but might find directions that larger radius can't (because the feasible region is smaller in a different shape).

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for shift in range(-(n-1), n):
        r = shift + (n - 1)
        row = np.zeros(n)
        # h[k] as first factor: h[k]*(1-h[k+shift]), need 0<=k+shift<n
        k1s = max(0, -shift)
        k1e = min(n, n - shift)
        if k1e > k1s:
            row[k1s:k1e] += (1.0 - h[k1s+shift:k1e+shift])
        # h[k] as second factor: h[k-shift]*(1-h[k]), derivative is -h[k-shift]
        k2s = max(0, shift)
        k2e = min(n, n + shift)
        if k2e > k2s:
            row[k2s:k2e] += -h[k2s-shift:k2e-shift]
        G[r] = row * dx
    return G

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _lp_full(h, scores, G, radius):
    n = len(h)
    n_lags = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-12)
        hi = min(radius, 1.0 - h[j] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, radius_init=0.2, max_iters=10000, min_radius=1e-6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        if radius < min_radius:
            break
        
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_full(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Slightly increase radius on success
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

def _subgradient_optimize(h, deadline, beta_start=10.0, beta_end=500.0, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        scores = _scores(h)
        max_s = np.max(scores)
        # Softmax weights
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w = w / sw
        
        G = _grad_matrix(h)
        grad = G.T @ w
        # Center gradient (project onto sum=0 subspace)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step size
        alpha = 0.1
        for _ in range(30):
            h_try = _project(h + alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.6
            if alpha < 1e-12:
                break
    
    return best_h, best_c5

def _random_restart_polish(h_best, c5_best, deadline, rng, n_restarts=200):
    n = len(h_best)
    target = n / 2.0
    best_h = h_best.copy()
    best_c5 = c5_best
    
    for i in range(n_restarts):
        if time.time() > deadline:
            break
        
        # Generate perturbation
        strategy = i % 4
        if strategy == 0:
            # Gaussian
            h_pert = best_h + rng.normal(0, 0.05, n)
        elif strategy == 1:
            # Sinusoidal
            freq = rng.integers(1, n // 2)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.1)
            x = np.arange(n)
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif strategy == 2:
            # Block
            h_pert = best_h.copy()
            start = rng.integers(0, n - 10)
            length = rng.integers(5, 20)
            delta = rng.uniform(-0.1, 0.1)
            h_pert[start:start+length] += delta
        else:
            # Two-swap
            h_pert = best_h.copy()
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            delta = rng.uniform(-0.15, 0.15)
            h_pert[i1] += delta
            h_pert[i2] -= delta
        
        h_pert = _project(h_pert, target)
        
        # Quick polish
        sub_dl = min(deadline, time.time() + 3.0)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.15, max_iters=200)
        
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _two_coord_search(h, deadline, max_passes=20):
    """For each pair of adjacent coordinates, search for optimal (a, b) with a+b=const."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for k in range(n - 1):
            if time.time() > deadline:
                break
            s = h[k] + h[k+1]
            if s < 1e-10 or s > 1.0 - 1e-10:
                continue
            # Search for optimal h[k] in [max(0, s-1), min(1, s)]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            # Ternary search
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[k] = m1
                h[k+1] = s - m1
                f1 = _exact_c5(h)
                h[k] = m2
                h[k+1] = s - m2
                f2 = _exact_c5(h)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h[k] = (lo + hi) / 2.0
            h[k+1] = s - h[k]
            c5_new = _exact_c5(h)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()
        if not improved:
            break
    
    return best_h, best_c5
