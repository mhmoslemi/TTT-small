import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _corr_vector(h):
    """Compute the full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_corr(h):
    """Compute gradient of each correlation value w.r.t. h.
    Returns G of shape (2n-1, n) where G[r, j] = d(corr[r])/d(h[j]).
    
    For lag d (corresponding to index r = n-1+d):
    k = |d|
    G[r, j] = (1-h[j-k]) * [k <= j <= n-1] - h[j+k] * [0 <= j <= n-1-k]
    """
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        k = abs(d)
        if k == 0:
            # g(0) = sum_i h[i]*(1-h[i])
            # dg(0)/dh[j] = (1-2*h[j])
            G[r, :] = 1.0 - 2.0 * h
        else:
            # For j in [k, n-1]: contribution (1-h[j-k])
            # For j in [0, n-1-k]: contribution -h[j+k]
            # These ranges may overlap for j in [k, n-1-k]
            G[r, k:] += (1.0 - h[:n-k])
            G[r, :n-k] += -h[k:]
    return G

def _slsqp_epigraph(h_init, deadline, max_iter=500):
    """Solve the epigraph problem: minimize T s.t. T >= corr_i(h) for all i.
    Variable vector: [T, h_0, ..., h_{n-1}]
    """
    n = len(h_init)
    target = n / 2.0
    T_init = float(np.max(_corr_vector(h_init))) + 0.01
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[0] = 1.0
        return g
    
    def constr_func(x):
        h = x[1:]
        T = x[0]
        corrs = _corr_vector(h)
        # Constraints: T - corr_i >= 0 for all i
        return T - corrs
    
    def constr_jac(x):
        h = x[1:]
        T = x[0]
        G = _grad_corr(h)
        # Jacobian of (T - corr_i) w.r.t. [T, h]
        J = np.zeros((2*n-1, n+1))
        J[:, 0] = 1.0
        J[:, 1:] = -G
        return J
    
    def eq_func(x):
        return np.array([np.sum(x[1:]) - target])
    
    def eq_jac(x):
        J = np.zeros((n+1,))
        J[1:] = 1.0
        return J
    
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    x0 = np.concatenate([[T_init], h_init])
    
    constraints = [
        {'type': 'ineq', 'fun': constr_func, 'jac': constr_jac},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.fun < T_init:
            h_res = np.clip(res.x[1:], 0.0, 1.0)
            h_res = _project(h_res, target)
            c5 = _exact_c5(h_res)
            return h_res, c5
    except Exception:
        pass
    return h_init.copy(), _exact_c5(h_init)

def _active_set_direction(h, num_active=5):
    """Find a direction that reduces the top `num_active` correlation values.
    Uses LP: find p minimizing max_i (G[i,:] @ p) s.t. sum(p)=0, ||p||_inf <= 1
    """
    n = len(h)
    corrs = _corr_vector(h)
    G = _grad_corr(h)
    
    # Select the top num_active lags
    idx = np.argsort(corrs)[::-1][:num_active]
    
    # LP: minimize t s.t. G[idx] @ p <= t, sum(p) = 0, -1 <= p <= 1
    # Variables: [p_0, ..., p_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G[idx]
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(len(idx))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            # Project to maintain integral constraint
            p -= np.mean(p)
            return p
    except Exception:
        pass
    return None

def _smooth_minimax(h, deadline, alpha_smooth=20.0):
    """Use smoothed max (log-sum-exp) to find a good direction, then refine."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    # Compute smoothed gradient
    corrs = _corr_vector(h)
    ms = float(np.max(corrs))
    vals = np.clip(alpha_smooth * (corrs - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    
    G = _grad_corr(h)
    d = -(G.T @ w)
    d -= np.mean(d)
    
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    d = d / norm * 0.3
    
    cur_c5 = _exact_c5(h)
    alpha, new_c5 = _line_search_c5(h, d, cur_c5, alpha_max_override=0.5)
    if new_c5 < cur_c5:
        return _project(h + alpha * d, target), new_c5
    return h.copy(), cur_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coordinate_descent_polish(h, deadline, max_sweeps=100):
    """Try to improve by adjusting pairs of coordinates."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try adjusting individual coordinates
        for j in range(n):
            if time.time() > deadline:
                break
            # Try moving h[j] up or down by a small amount, compensating elsewhere
            for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                if h[j] + delta < 0 or h[j] + delta > 1:
                    continue
                h_t = h.copy()
                h_t[j] += delta
                # Compensate by adjusting a random other coordinate
                for k in range(j+1, n):
                    if h[k] - delta >= 0 and h[k] - delta <= 1:
                        h_t[k] -= delta
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            h = h_t
                            best_c5 = c5_t
                            improved = True
                        break
        if not improved:
            break
    
    return h, best_c5
