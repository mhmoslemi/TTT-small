import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

def _all_lag_data(h):
    """Compute all raw overlap values and their gradients"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    raw_vals = corr  # length 2n-1, index r corresponds to shift d = r - (n-1)
    grads = []
    for r in range(2 * n - 1):
        d = r - (n - 1)
        grads.append(_lag_gradient(h, d))
    return raw_vals, grads

def _lp_tangent_dir(h, active_indices, R, max_t):
    """Solve LP to find tangent descent direction.
    Returns direction d (length n) or None if infeasible."""
    n = len(h)
    raw_vals, grads = _all_lag_data(h)
    
    # Build LP
    # Variables: x = [t, p_0, q_0, p_1, q_1, ..., p_{n-1}, q_{n-1}]
    # Length: 1 + 2n
    nv = 1 + 2 * n
    
    # Objective: minimize t
    c = np.zeros(nv)
    c[0] = 1.0
    
    # Inequality constraints: -t + sum_j (g_j p_j - g_j q_j) <= -F_s
    # For active shifts
    A_ub_rows = []
    b_ub_vals = []
    
    for r in active_indices:
        row = np.zeros(nv)
        row[0] = -1.0
        g = grads[r]
        for j in range(n):
            row[1 + 2*j] = g[j]       # p_j coefficient
            row[2 + 2*j] = -g[j]      # q_j coefficient
        A_ub_rows.append(row)
        b_ub_vals.append(-float(raw_vals[r]))
    
    # L1 norm constraint: sum(p_j + q_j) <= R
    row = np.zeros(nv)
    row[1:] = 1.0
    A_ub_rows.append(row)
    b_ub_vals.append(R)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality constraint: sum(p_j - q_j) = 0
    A_eq = np.zeros((1, nv))
    for j in range(n):
        A_eq[0, 1 + 2*j] = 1.0
        A_eq[0, 2 + 2*j] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(None, max_t)]  # t
    for j in range(n):
        if h[j] < 1e-8:
            # At lower bound: p free, q = 0
            bounds.append((0, R))   # p_j
            bounds.append((0, 0))   # q_j
        elif h[j] > 1.0 - 1e-8:
            # At upper bound: p = 0, q free
            bounds.append((0, 0))   # p_j
            bounds.append((0, R))   # q_j
        else:
            # Interior
            bounds.append((0, R))   # p_j
            bounds.append((0, R))   # q_j
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return None
        
        # Extract direction
        x = res.x
        p = np.array([x[1 + 2*j] for j in range(n)])
        q = np.array([x[2 + 2*j] for j in range(n)])
        d = p - q
        d -= np.mean(d)  # Ensure zero sum
        
        if float(np.max(np.abs(d))) < 1e-10:
            return None
        return d
    except Exception:
        return None

def _line_search(h, direction, target, c5_cur, max_alpha=1.0, n_trials=15):
    """Line search along direction from h. Returns (best_alpha, best_c5)."""
    n = len(h)
    
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-8), 5.0)
    
    best_a = 0.0
    best_c = c5_cur
    
    # Geometric + linear search
    alphas = np.geomspace(0.001, alpha_max, n_trials)
    alphas = np.concatenate([[alpha_max * 0.1, alpha_max * 0.5, alpha_max], alphas])
    alphas = np.sort(alphas)
    alphas = alphas[alphas <= alpha_max + 1e-10]
    
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    span = best_a * 0.4
    lo = max(0.0, best_a - span)
    hi = min(alpha_max, best_a + span)
    if hi - lo < 1e-12:
        return best_a, best_c
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(15):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        best_a = opt_a
        best_c = opt_c
    
    return best_a, best_c

def _lp_descent(h, deadline, target, mode='full', max_iters=2000):
    """LP-based tangent descent"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    R = 0.1 * n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        raw_vals = np.correlate(h, 1.0 - h, mode='full')
        raw_max = float(np.max(raw_vals))
        
        # Select active set
        if mode == 'full':
            active = list(range(len(raw_vals)))
        else:
            gate = 0.01 * (raw_max + 1)
            active = [i for i in range(len(raw_vals)) if raw_vals[i] >= raw_max - gate]
            if len(active) < 3:
                top_idx = np.argsort(raw_vals)[-5:]
                active = list(top_idx)
        
        # Try different R values
        d = None
        for R_try in [R, 0.05 * n, 0.3 * n, 0.5 * n]:
            d = _lp_tangent_dir(h, active, R_try, max_t=100.0)
            if d is not None and float(np.linalg.norm(d)) > 1e-8:
                break
        
        if d is None:
            # Fallback: use negative max-lag gradient
            max_idx = int(np.argmax(raw_vals))
            d_max = max_idx - (n - 1)
            d = -_lag_gradient(h, d_max)
            d -= np.mean(d)
            gnorm = float(np.linalg.norm(d))
            if gnorm < 1e-10:
                stall += 10
                continue
            d /= gnorm
        
        # Try both signs
        improved = False
        for sign in [1.0, -1.0]:
            a, c_t = _line_search(h, sign * d, target, cur_c5, max_alpha=0.5)
            if a > 0 and c_t < best_c5 - 1e-13:
                h = _project(h + a * sign * d, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            R *= 0.7
    
    return best_h, best_c5

def _subgrad_descent(h, deadline, target, T_init=0.3, max_iters=5000):
    """Subgradient descent with momentum (fallback)"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.05
    beta = 0.7
    T = T_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        raw_vals = np.correlate(h, 1.0 - h, mode='full')
        shifted = raw_vals - float(np.max(raw_vals))
        w = np.exp(shifted / T)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            w = np.zeros(len(raw_vals))
            w[int(np.argmax(raw_vals))] = 1.0
        else:
            w /= wsum
        
        grad = np.zeros(n)
        for r in range(len(raw_vals)):
            if w[r] < 1e-8:
                continue
            d_shift = r - (n - 1)
            grad += w[r] * _lag_gradient(h, d_shift)
        
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        grad /= gnorm
        
        v = beta * v - alpha * grad
        v -= np.mean(v)
        
        a, new_c5 = _line_search(h, v, target, cur_c5, max_alpha=1.0)
        if a > 0 and new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * v, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            T = max(0.05, T * 0.99)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.9
            T = max(0.05, T * 0.95)
    
    return best_h, best_c5

        if remaining < 5.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 35.0)
        sub_dl = time.time() + alloc
        r = rng.random()
        if r < 0.15:
            h_try = h_best + rng.normal(0, 0.035, n)
        elif r < 0.30:
            h_try = h_best.copy()
            idx = rng.choice(n, size=max(4, n//6), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.45:
            h_try = (h_best + rng.normal(0, 0.06, n) > 0.45).astype(float)
        elif r < 0.55:
            shift = rng.integers(1, max(2, n//12))
            h_try = np.roll(h_best, shift)
        elif r < 0.65:
            h_try = 0.5 + (h_best - 0.5) * (0.7 + 0.5 * rng.random())
        elif r < 0.75:
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.4 * h_best + 0.6 * h_rand
        elif r < 0.85:
            half = n // 2
            h_try = np.concatenate([h_best[:half], 1.0 - h_best[half:]])
            if n > 2 * half:
                h_try = np.concatenate([h_try, h_best[2*half:]])
        else:
            h_try = np.zeros(n)
            h_try[::2] = 1.0
            h_try += rng.normal(0, 0.08, n)
        h_try = _project(h_try, target)
        h_opt, c5_opt = _lp_descent(h_try, sub_dl * 0.55, target, max_iters=1500)
        if time.time() < sub_dl:
            h_opt2, c5_opt2 = _subgrad_descent(h_opt, sub_dl, target, T_init=0.2, max_iters=3000)
            if c5_opt2 < c5_opt:
                h_opt = h_opt2
                c5_opt = c5_opt2
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def _fine_polish(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    stall = 0
    while time.time() < deadline and stall < 40:
        raw_vals = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(raw_vals)[-6:]
        improved = False
        for r in top_lags:
            if time.time() > deadline:
                break
            d_shift = int(r) - (n - 1)
            grad = -_lag_gradient(h, d_shift)
            grad -= np.mean(grad)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-10:
                continue
            grad /= gn
            for sign in [1.0, -1.0]:
                a, c_t = _line_search(h, sign * grad, target, cur_c5, max_alpha=0.3)
                if a > 0 and c_t < best_c5 - 1e-13:
                    h = _project(h + a * sign * grad, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            for _ in range(50):
                if time.time() > deadline:
                    break
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7 or h[k] <= 1e-7 or h[k] >= 1.0 - 1e-7:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = min((1-h[j])/abs(dj), (1-h[k])/abs(dk)) if dj == dk else \
                              min((1-h[j]) if dj>0 else h[j], (1-h[k]) if dk>0 else h[k])
                    if a_max <= 1e-10:
                        continue
                    for a in [a_max * 0.1, a_max * 0.3, a_max * 0.5, a_max * 0.7, a_max]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
        if not improved:
            stall += 1
            h_try = _project(h + rng.normal(0, 0.008, n), target)
            c_try = _exact_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h = h_try
                cur_c5 = c_try
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n_def = 64
        h0 = np.zeros(n_def)
        h0[::2] = 1.0
    
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP-based tangent descent (30%)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_descent(h, dl1, target0, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Subgradient polishing (15%)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = _subgrad_descent(best_h, dl2, best_n / 2.0, T_init=0.25, max_iters=15000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Multi-start (25%)
    dl3 = t0 + budget_s * 0.70
    h3, c5_3 = _multi_start(best_h, dl3, best_n, rng, n_starts=25)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Resolution sweep (15%)
    dl4 = t0 + budget_s * 0.85
    for dn in [8, -8, 16, -16, 4, -4, 12, -12, 20, -20, 24, -24, 32, -32]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 144:
            continue
        h_r = _resample(best_h, n_try)
        remaining = dl4 - time.time()
        if remaining < 3:
            break
        sub_dl = min(dl4, time.time() + min(12.0, remaining * 0.4))
        h_r, c5_r = _lp_descent(h_r, sub_dl, n_try / 2.0, max_iters=2000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Pair polish (8%)
    dl5 = t0 + budget_s * 0.93
    h5, c5_5 = _pair_polish(best_h, dl5, best_n / 2.0, max_passes=80)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Fine polish (7%)
    dl6 = deadline
    h6, c5_6 = _fine_polish(best_h, dl6, best_n / 2.0)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
