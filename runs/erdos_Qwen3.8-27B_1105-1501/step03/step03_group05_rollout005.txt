The strategy suggests using exact one-dimensional polynomial line descent with full coordinate directions, combined with basin-hopping/population search.

Key insight from the strategy: For a direction v, the objective along h + alpha*v is max of quadratics, and we can find the exact minimum by checking all pairwise intersections and stationary points.

Let me implement this approach:

1. Exact line search: Given direction v, compute A, B, C arrays (quadratic coefficients for each lag), then find the alpha that minimizes the maximum of all quadratics over the feasible interval.

2. Direction generation:
   - Subgradient directions (from active lags)
   - Random directions
   - Pair directions (e_i - e_j)

3. Basin hopping: Transform the current best (flip, roll, mix, threshold) and re-optimize.

4. Multi-resolution: Try different n values.

The key function is the exact line search:
- A = correlate(h, 1-h, 'full')
- B = correlate(v, 1-h, 'full') - correlate(h, v, 'full')
- C = -correlate(v, v, 'full')
- For each lag r: score_r(alpha) = A[r] + B[r]*alpha + C[r]*alpha^2
- Find alpha that minimizes max_r score_r(alpha) over feasible interval

For candidate alpha values:
- Endpoints of feasible interval
- Stationary points of each quadratic: -B[r]/(2*C[r])
- Pairwise intersections: solve (C[i]-C[j])*a^2 + (B[i]-B[j])*a + (A[i]-A[j]) = 0

For n=64, there are 127 lags. Pairwise intersections: ~127*126/2 = ~8000 pairs, each giving up to 2 roots. That's ~16000 candidates plus ~127 stationary points plus endpoints. Very manageable.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair_box_sum(h, target):
    n = len(h)
    x = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    for _ in range(200):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        if not mask.any():
            break
        frac = diff / mask.sum()
        x[mask] += frac
        x = np.clip(x, 0.0, 1.0)
    # Final exact fix
    residual = target - x.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_interval(h, v):
    lo = 0.0
    hi = 0.0
    for i in range(len(h)):
        if v[i] > 1e-15:
            lo = max(lo, -h[i] / v[i])
            hi = min(hi, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / v[i])
            hi = min(hi, -h[i] / v[i])
    return lo, hi

def _exact_line_alpha(h, v, cur_c5):
    n = len(h)
    dx = 2.0 / n
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    lo, hi = _feasible_interval(h, v)
    if lo >= hi:
        return 0.0, cur_c5
    
    # Ensure lo <= 0 <= hi
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if lo >= hi:
        return 0.0, cur_c5
    
    # Candidate alphas
    candidates = {0.0, lo + 1e-14, hi - 1e-14}
    
    # Stationary points
    for r in range(len(A)):
        if abs(C[r]) > 1e-15:
            a = -B[r] / (2.0 * C[r])
            if lo < a < hi:
                candidates.add(a)
    
    # Pairwise intersections (for n <= 96)
    if n <= 96:
        m = len(A)
        for i in range(m):
            for j in range(i + 1, m):
                dA = A[i] - A[j]
                dB = B[i] - B[j]
                dC = C[i] - C[j]
                if abs(dC) > 1e-14:
                    disc = dB * dB - 4.0 * dC * dA
                    if disc >= -1e-14:
                        if disc < 0:
                            disc = 0.0
                        sqrt_disc = np.sqrt(disc)
                        a1 = (-dB + sqrt_disc) / (2.0 * dC)
                        a2 = (-dB - sqrt_disc) / (2.0 * dC)
                        if lo < a1 < hi:
                            candidates.add(a1)
                        if lo < a2 < hi:
                            candidates.add(a2)
                elif abs(dB) > 1e-14:
                    a = -dA / dB
                    if lo < a < hi:
                        candidates.add(a)
    
    # Uniform grid as safety net
    n_grid = 61
    grid = np.linspace(lo, hi, n_grid)
    for g in grid:
        candidates.add(float(g))
    
    # Evaluate
    best_c5 = cur_c5
    best_a = 0.0
    for a in candidates:
        if a <= lo or a >= hi:
            continue
        scores = A + a * B + a * a * C
        c5 = float(np.max(scores) * dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_a = a
    
    return best_a, best_c5

def _line_search(h, v, cur_c5):
    # Normalize v to be zero-sum
    v = v.copy()
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return h, cur_c5
    v /= norm
    
    best_a, best_c5 = _exact_line_alpha(h, v, cur_c5)
    if best_a == 0.0 or best_c5 >= cur_c5 - 1e-13:
        return h, cur_c5
    
    h_new = h + best_a * v
    h_new = _repair_box_sum(h_new, len(h) / 2.0)
    actual_c5 = _exact_c5(h_new)
    
    if actual_c5 < cur_c5 - 1e-12:
        return h_new, actual_c5
    return h, cur_c5

def _subgradient_direction(h, scores, beta=10.0):
    m = len(scores)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(m)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m_count = n - d
            if m_count > 0:
                grad[:m_count] += wd * (1.0 - h[d:])
                grad[d:] += -wd * h[:m_count]
        else:
            g = -d
            m_count = n - g
            if m_count > 0:
                grad[g:] += wd * (1.0 - h[:m_count])
                grad[:m_count] += -wd * h[g:]
    return grad

def _local_line_descend(h, deadline, rng, max_rounds=200):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for rnd in range(max_rounds):

            g = _subgradient_direction(h, scores, beta=beta)
            v = -g
            h_new, c5_new = _line_search(h, v, cur_c5)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

            v = rng.standard_normal(n)
            h_new, c5_new = _line_search(h, v, cur_c5)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0

            i = rng.integers(n)

                    best_h = h.copy()
                improved = True

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_box_sum(h_new, n_new / 2.0)

def _binary_projection(h):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair_box_sum(h, target):
    n = len(h)
    x = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    for _ in range(200):
        diff = target - x.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        if not mask.any():
            break
        x[mask] += diff / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_interval(h, v):
    lo = 0.0
    hi = 0.0
    pos = v > 1e-15
    neg = v < -1e-15
    if pos.any():
        lo = max(lo, float(np.max(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if neg.any():
        lo = max(lo, float(np.max((1.0 - h[neg]) / v[neg])))
        hi = min(hi, float(np.max(-h[neg] / v[neg])))
    return lo, hi

def _line_search_exact(h, v, cur_c5):
    n = len(h)
    v = v.copy()
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return h, cur_c5
    v /= norm

    dx = 2.0 / n
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')

    lo, hi = _feasible_interval(h, v)
    if lo >= hi - 1e-14:
        return h, cur_c5
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if lo >= hi - 1e-14:
        return h, cur_c5

    m = len(A)
    candidates = [0.0, lo + 1e-10, hi - 1e-10]

    # Stationary points of each quadratic
    c_nonzero = np.abs(C) > 1e-15
    if c_nonzero.any():
        a_stat = -B[c_nonzero] / (2.0 * C[c_nonzero])
        valid = (a_stat > lo) & (a_stat < hi)
        candidates.extend(a_stat[valid].tolist())

    # Pairwise intersections - vectorized
    if n <= 128:
        idx_i, idx_j = np.triu_indices(m, k=1)
        if len(idx_i) <= 60000:
            dA = A[idx_i] - A[idx_j]
            dB = B[idx_i] - B[idx_j]
            dC = C[idx_i] - C[idx_j]
            quad_mask = np.abs(dC) > 1e-14
            if quad_mask.any():
                dq_C = dC[quad_mask]
                dq_B = dB[quad_mask]
                dq_A = dA[quad_mask]
                disc = dq_B * dq_B - 4.0 * dq_C * dq_A
                valid_d = disc >= -1e-14
                if valid_d.any():
                    disc_v = np.maximum(disc[valid_d], 0.0)
                    sq = np.sqrt(disc_v)
                    c_v = dq_C[valid_d]
                    b_v = dq_B[valid_d]
                    a1 = (-b_v + sq) / (2.0 * c_v)
                    a2 = (-b_v - sq) / (2.0 * c_v)
                    v1 = (a1 > lo) & (a1 < hi)
                    v2 = (a2 > lo) & (a2 < hi)
                    candidates.extend(a1[v1].tolist())
                    candidates.extend(a2[v2].tolist())
            lin_mask = (~quad_mask) & (np.abs(dB) > 1e-14)
            if lin_mask.any():
                a_lin = -dA[lin_mask] / dB[lin_mask]
                v_lin = (a_lin > lo) & (a_lin < hi)
                candidates.extend(a_lin[v_lin].tolist())

    # Uniform grid safety net
    n_grid = 51
    grid = np.linspace(lo, hi, n_grid)
    candidates.extend(grid.tolist())

    candidates = np.array(candidates)
    candidates = candidates[(candidates > lo) & (candidates < hi)]
    if len(candidates) == 0:
        return h, cur_c5

    # Evaluate all candidates vectorized
    scores_grid = A[None, :] + candidates[:, None] * B[None, :] + candidates[:, None] * candidates[:, None] * C[None, :]
    max_scores = np.max(scores_grid, axis=1) * dx
    best_idx = int(np.argmin(max_scores))
    best_a = float(candidates[best_idx])
    best_c5 = float(max_scores[best_idx])

    if best_a == 0.0 or best_c5 >= cur_c5 - 1e-13:
        return h, cur_c5

    h_new = h + best_a * v
    h_new = _repair_box_sum(h_new, n / 2.0)
    actual_c5 = _exact_c5(h_new)
    if actual_c5 < cur_c5 - 1e-12:
        return h_new, actual_c5
    return h, cur_c5

def _subgradient_direction(h, scores, beta=10.0):
    n = len(h)
    m = len(scores)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(m)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            cnt = n - d
            if cnt > 0:
                grad[:cnt] += -wd * h[d:d + cnt]
                grad[d:d + cnt] += wd * (1.0 - h[:cnt])
        else:
            g = -d
            cnt = n - g
            if cnt > 0:
                grad[g:g + cnt] += -wd * h[:cnt]
                grad[:cnt] += wd * (1.0 - h[g:g + cnt])
    return grad

def _local_line_descend(h, deadline, rng, max_rounds=300):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    betas = [5.0, 15.0, 50.0, 150.0]

    for _rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break

        scores = _raw_scores(h)
        improved = False

        # Subgradient directions
        for beta in betas:
            if time.time() > deadline:
                break
            g = _subgradient_direction(h, scores, beta=beta)
            h_new, c5_new = _line_search_exact(h, -g, cur_c5)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        # Random directions
        for _ in range(4):
            if time.time() > deadline:
                break
            v = rng.standard_normal(n)
            h_new, c5_new = _line_search_exact(h, v, cur_c5)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        # Pair directions
        for _ in range(6):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            v = np.zeros(n)
            v[i] = 1.0
            v[j] = -1.0
            h_new, c5_new = _line_search_exact(h, v, cur_c5)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_box_sum(h_new, n_new / 2.0)

def _binary_projection(h):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin
