        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coeff_for(h, idx, n):
    """Compute coefficient of h[idx] in each raw offset. Returns array of length 2n-1."""
    b = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        j = n - 1 + d  # index in correlate output
        c = 0.0
        if 0 <= idx + d < n:
            c += (1.0 - h[idx + d])
        if 0 <= idx - d < n:
            c -= h[idx - d]
        b[j] = c
    return b

def _exact_pair_delta(h, i, j, raw_arr, n):
    """Find best delta for h[i]+=delta, h[j]-=delta. Returns (delta, new_c5)."""
    ci = _coeff_for(h, i, n)
    cj = _coeff_for(h, j, n)
    b = ci - cj
    
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_max - delta_min < 1e-12:
        return 0.0, _c5(h)
    
    cur_max = float(np.max(raw_arr))
    
    # Candidate points: endpoints, 0, and where each line hits cur_max
    candidates = set()
    candidates.add(delta_min)
    candidates.add(delta_max)
    if delta_min <= 0.0 <= delta_max:
        candidates.add(0.0)
    
    for k in range(2 * n - 1):
        if abs(b[k]) > 1e-14:
            dk = (cur_max - raw_arr[k]) / b[k]
            if delta_min - 1e-10 <= dk <= delta_max + 1e-10:
                candidates.add(np.clip(dk, delta_min, delta_max))
    
    best_d = 0.0
    best_c = cur_max
    
    for dk in candidates:
        dk = float(np.clip(dk, delta_min, delta_max))
        new_raw = raw_arr + dk * b
        m = float(np.max(new_raw))
        if m < best_c - 1e-14:
            best_c = m
            best_d = dk
    
    # Refine with golden section around best_d
    if best_d > delta_min + 1e-10 and best_d < delta_max - 1e-10:
        span = (delta_max - delta_min) * 0.15
        lo = max(delta_min, best_d - span)
        hi = min(delta_max, best_d + span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw_arr + c1 * b))
        f2 = float(np.max(raw_arr + c2 * b))
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw_arr + c1 * b))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw_arr + c2 * b))
        opt_d = (lo + hi) / 2.0
        opt_c = float(np.max(raw_arr + opt_d * b))
        if opt_c < best_c:
            best_c = opt_c
            best_d = opt_d
    
    if best_d == 0.0:
        return 0.0, cur_max
    return best_d, best_c

def _pair_local_search(h, deadline, max_passes=200):
    """Local search using exact pair moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    raw_arr = _raw(h)
    cur_c5 = float(np.max(raw_arr) * 2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    # Score pairs by potential improvement
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Compute gradient to guide pair selection
        scores = _raw(h)
        shifted = scores - float(np.max(scores))
        T = 0.5
        w = np.exp(shifted / (T * max(1.0, float(np.max(scores)))))
        w /= float(np.sum(w))
        
        # Gradient of weighted sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-6:
                continue
            d = k - (n - 1)
            ci = _coeff_for(h, 0, n)  # placeholder, compute properly below
            # Actually compute gradient contribution
            for idx in range(n):
                c = 0.0
                if 0 <= idx + d < n:
                    c += (1.0 - h[idx + d])
                if 0 <= idx - d < n:
                    c -= h[idx - d]
                grad[idx] += w[k] * c
        grad -= np.mean(grad)
        
        # Select top coordinates
        top_count = min(n, 30)
        top = np.argsort(-np.abs(grad))[:top_count]
        
        # Try pairs from top set
        tried = set()
        order = np.argsort(-np.abs(grad))[:top_count]
        for ii in range(len(order)):
            if time.time() > deadline:
                return best_h, best_c5
            a = order[ii]
            if h[a] <= 1e-8 or h[a] >= 1.0 - 1e-8:
                continue
            for jj in range(ii + 1, len(order)):
                b_idx = order[jj]
                if h[b_idx] <= 1e-8 or h[b_idx] >= 1.0 - 1e-8:
                    continue
                # Try both directions
                delta, new_c5 = _exact_pair_delta(h, a, b_idx, raw_arr, n)
                if delta != 0.0 and new_c5 * 2.0 / n < best_c5 - 1e-13:
                    h[a] += delta
                    h[b_idx] -= delta
                    raw_arr = _raw(h)
                    cur_c5 = float(np.max(raw_arr) * 2.0 / n)
                    best_c5 = cur_c5
                    best_h = h.copy()
                    improved = True
                delta, new_c5 = _exact_pair_delta(h, b_idx, a, raw_arr, n)
                if delta != 0.0 and new_c5 * 2.0 / n < best_c5 - 1e-13:
                    h[b_idx] += delta
                    h[a] -= delta
                    raw_arr = _raw(h)
                    cur_c5 = float(np.max(raw_arr) * 2.0 / n)
                    best_c5 = cur_c5
                    best_h = h.copy()
                    improved = True
        
        # Random pairs
        for _ in range(300):
            if time.time() > deadline:
                return best_h, best_c5
            a = int(rng.integers(0, n))
            b_idx = int(rng.integers(0, n))
            if a == b_idx or h[a] <= 1e-8 or h[a] >= 1.0 - 1e-8 or h[b_idx] <= 1e-8 or h[b_idx] >= 1.0 - 1e-8:
                continue
            delta, new_c5 = _exact_pair_delta(h, a, b_idx, raw_arr, n)
            if delta != 0.0 and new_c5 * 2.0 / n < best_c5 - 1e-13:
                h[a] += delta
                h[b_idx] -= delta
                raw_arr = _raw(h)
                cur_c5 = float(np.max(raw_arr) * 2.0 / n)
                best_c5 = cur_c5
                best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def _subgrad_optimize(h, deadline, T_init=0.3, max_iters=10000):
    """Subgradient descent with exact line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    raw_arr = _raw(h)
    cur_c5 = float(np.max(raw_arr) * 2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.06
    beta = 0.5
    T = T_init
    stall = 0
    scale = max(1.0, float(np.max(raw_arr)))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = raw_arr
        shifted = (scores - float(np.max(scores))) / (T * scale)
        w = np.exp(shifted)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            d = k - (n - 1)
            for idx in range(n):
                c = 0.0
                if 0 <= idx + d < n:
                    c += (1.0 - h[idx + d])
                if 0 <= idx - d < n:
                    c -= h[idx - d]
                grad[idx] += w[k] * c
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        grad /= gnorm
        
        v = beta * v - alpha * grad
        v -= np.mean(v)
        
        # Line search along v
        best_a = 0.0
        best_val = cur_c5
        for a in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_t = _project(h + a * v, target)
            c_t = _c5(h_t)
            if c_t < best_val - 1e-13:
                best_val = c_t
                best_a = a
        
        if best_a > 0 and best_val < cur_c5 - 1e-13:
            h = _project(h + best_a * v, target)
            raw_arr = _raw(h)
            cur_c5 = float(np.max(raw_arr) * 2.0 / n)
            stall = 0
            T = max(0.03, T * 0.97)
            beta = min(0.9, beta + 0.002)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.8
            T = max(0.03, T * 0.9)
            if stall > 40:
                v *= 0.2
                alpha = 0.04
                T = 0.15
                stall = 0
    
    return best_h, best_c5

def _slsqp_optimize(h, deadline, max_iters=3):
    """SLSQP on smoothed objective."""
    n = len(h)
    target = n / 2.0
    
    def obj_and_grad(x):
        h_x = np.clip(x, 0.0, 1.0)
        raw = _raw(h_x)
        mx = float(np.max(raw))
        T = 0.3 * max(1.0, mx)
        shifted = (raw - mx) / T
        w = np.exp(shifted)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            val = mx
            grad = np.zeros(n)
        else:
            w /= wsum
            val = mx + T * np.log(wsum)
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-10:
                    continue
                d = k - (n - 1)
                for idx in range(n):
                    c = 0.0
                    if 0 <= idx + d < n:
                        c += (1.0 - h_x[idx + d])
                    if 0 <= idx - d < n:
                        c -= h_x[idx - d]
                    grad[idx] += w[k] * c
            grad *= (2.0 / n)
        return val, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    best_h = h.copy()
    best_c5 = _c5(h)
    
    seeds = [h.copy()]
    rng = np.random.default_rng(55)
    for _ in range(2):
        pert = h + rng.normal(0, 0.02, n)
        seeds.append(_project(pert, target))
    
    for seed_h in seeds:
        if time.time() > deadline:
            break
        try:
            res = scipy_minimize(obj_and_grad, seed_h, jac=None, method='SLSQP',
                                 bounds=bounds, constraints=constraints,
                                 options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            if res.success or res.x is not None:
                h_new = _project(np.clip(res.x, 0, 1), target)
                c5_new = _c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start(h0, deadline, rng, n_starts=20):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _c5(h0)
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 30.0)
        sub_dl = time.time() + alloc
        
        r = rng.random()
        if r < 0.25:
            h_try = h0 + rng.normal(0, 0.04, n)
        elif r < 0.45:
            h_try = h0.copy()
            idx = rng.choice(n, size=max(4, n // 6), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.6:
            h_try = (h0 + rng.normal(0, 0.08, n) > 0.5).astype(float)
        elif r < 0.75:
            shift = int(rng.integers(1, max(2, n // 10)))
            h_try = np.roll(h0, shift)
        elif r < 0.85:
            h_try = 0.5 + (h0 - 0.5) * (0.7 + 0.5 * rng.random())
        else:
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.5 * h0 + 0.5 * h_rand
        
        h_try = _project(h_try, target)
        c5_try = _c5(h_try)
        
        # Quick subgradient
        h_opt, c5_opt = _subgrad_optimize(h_try, sub_dl * 0.5, T_init=0.25, max_iters=2000)
        # Quick pair search
        h_opt, c5_opt = _pair_local_search(h_opt, sub_dl, max_passes=30)
        
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Initialize from pre-imported values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0
    
    # Phase 1: Subgradient optimization (20%)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _subgrad_optimize(h, dl1, T_init=0.35, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: SLSQP (10%)
    dl2 = t0 + budget_s * 0.30
    h2, c5_2 = _slsqp_optimize(best_h, dl2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Exact pair local search (25%)
    dl3 = t0 + budget_s * 0.55
    h3, c5_3 = _pair_local_search(best_h, dl3, max_passes=500)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Multi-start (25%)
    dl4 = t0 + budget_s * 0.80
    h4, c5_4 = _multi_start(best_h, dl4, rng, n_starts=15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Resolution changes (10%)
    dl5 = t0 + budget_s * 0.90
    for dn in [4, -4, 8, -8, 12, -12, 16, 20, -20]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 120 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + 15.0)
        h_r, c5_r = _subgrad_optimize(h_r, sub_dl * 0.6, T_init=0.25, max_iters=3000)
        h_r, c5_r = _pair_local_search(h_r, sub_dl, max_passes=40)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 6: Final polish (10%)
    dl6 = t0 + budget_s * 0.99
    h6, c5_6 = _pair_local_search(best_h, dl6, max_passes=300)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        # Something went wrong, fall back
        final_c5 = best_c5
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _c5(best_h)
    
    return best_h, final_c5, best_n
