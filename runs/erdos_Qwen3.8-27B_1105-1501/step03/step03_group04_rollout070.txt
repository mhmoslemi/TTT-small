def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k):
    """Gradient of f_k(h) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if 0 <= k < n:
        m = n - k
        g[:m] += (1.0 - h[k:])
        g[k:] -= h[:m]
    elif -n < k < 0:
        kk = -k
        m = n - kk
        g[kk:] += (1.0 - h[:m])
        g[:m] -= h[kk:]
    return g

def _subgrad(h, top_m=5, temperature=0.01):
    """Weighted subgradient using top-m lags with softmax weighting."""
    scores = _scores(h)
    n = len(h)
    # Softmax weights over all lags (temperature controls selectivity)
    shifted = scores - scores.max()
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    g = np.zeros(n)
    for r in range(2*n - 1):
        if weights[r] > 1e-10:
            k = r - (n - 1)
            g += weights[r] * _grad_lag(h, k)
    return g, scores

def _project_box_sum(v, target):
