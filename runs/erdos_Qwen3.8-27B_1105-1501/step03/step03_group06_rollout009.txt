import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Repair residual
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to increase
            order = np.argsort(x)[::-1]  # elements closest to 1 first
            for idx in order:
                room = 1.0 - x[idx]
                adj = min(room, residual)
                if adj > 0:
                    x[idx] += adj
                    residual -= adj
                    if residual < 1e-12:
                        break
        else:
            # Need to decrease
            order = np.argsort(x)  # elements closest to 0 first
            for idx in order:
                adj = min(x[idx], -residual)
                if adj > 0:
                    x[idx] -= adj
                    residual += adj
                    if residual > -1e-12:
                        break
    return x

def _binary_corr_max(b):
    """For binary b, compute max correlation."""
    n = len(b)
    corr = np.correlate(b.astype(float), 1.0 - b.astype(float), mode='full')
    return int(np.max(corr) + 0.5)

def _binary_energy(b):
    """For binary b, compute a measure of how bad the max corr is.
    Returns (max_corr, sum_of_squares) for tie-breaking."""
    n = len(b)
    corr = np.correlate(b.astype(float), 1.0 - b.astype(float), mode='full')
    mx = int(np.max(corr) + 0.5)
    ss = float(np.sum(corr**2))
    return mx, ss

def _sa_binary(n, budget, seed=42):
    """Simulated annealing for binary problem."""
    rng = np.random.default_rng(seed)
    target = n // 2
    # Initialize with random binary
    b = np.zeros(n)
    ones = rng.choice(n, size=target, replace=False)
    b[ones] = 1.0
    
    cur_mx, cur_ss = _binary_energy(b)
    best_b = b.copy()
    best_mx, best_ss = cur_mx, cur_ss
    
    t_start = time.time()
    T = 5.0
    T_min = 0.01
    cooling = 0.9997
    
    while time.time() - t_start < budget and T > T_min:
        # Choose a move
        move_type = rng.random()
        if move_type < 0.6:
            # Single swap: exchange a 1 and a 0
            ones_mask = np.where(b == 1)[0]
            zeros_mask = np.where(b == 0)[0]
            i = ones_mask[rng.integers(len(ones_mask))]
            j = zeros_mask[rng.integers(len(zeros_mask))]
            b[i] = 0
            b[j] = 1
            new_mx, new_ss = _binary_energy(b)
            
            # Energy: (mx, ss) lexicographic
            if new_mx < cur_mx or (new_mx == cur_mx and new_ss < cur_ss):
                accept = True
            else:

                b[j] = 0

def _slsqp_epigraph(h0, deadline, maxiter=500):
    """SLSQP on epigraph formulation: min t s.t. corr_r(h) <= t, sum(h)=n/2, 0<=h<=1."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def objective(v):
        h = v[:n]
        t = v[n]
        return t
    
    def jac(v):
        h = v[:n]
        t = v[n]
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Build constraints
    # corr_r(h) - t <= 0 for each r
    def corr_constraints(v):
        h = v[:n]
        t = v[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr  # must be >= 0
    
    def corr_jac(v):
        h = v[:n]
        n_ = len(h)
        K = 2 * n_ - 1
        J = np.zeros((K, n_ + 1))
        for r in range(K):
            s = r - (n_ - 1)
            lo = max(0, -s)
            hi = min(n_, n_ - s)
            idx = np.arange(lo, hi)
            # d corr_r / d h_i = sum over pairs (i, i+s): (1-h_{i+s}) - h_i contributes
            # Actually corr_r = sum_{i in range} h_i * (1 - h_{i+s})
            # d/dh_k = sum over pairs where k appears
            # If k = i: contribution (1 - h_{i+s})
            # If k = i+s: contribution -h_i
            if s == 0:
                # corr_0 = sum h_i(1-h_i) = sum(h_i - h_i^2)
                # d/dh_k = 1 - 2*h_k
                J[r, idx] = 1.0 - 2.0 * h[idx]
            else:
                for i in idx:
                    j = i + s
                    if 0 <= j < n_:
                        J[r, i] += (1.0 - h[j])
                        J[r, j] += (-h[i])
            J[r, n_] = -1.0  # derivative wrt t
        return J
    
    def sum_constraint(v):
        return float(np.sum(v[:n])) - target
    
    def sum_jac(v):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    constraints = [
        {'type': 'ineq', 'fun': corr_constraints, 'jac': corr_jac},
        {'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}
    ]
    
    v0 = np.concatenate([h0, [np.max(np.correlate(h0, 1-h0, mode='full')) + 0.1]])
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = minimize(objective, v0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        h_opt = res.x[:n]
        h_opt = _project(h_opt, target)
        return h_opt, _exact_c5(h_opt)
    except Exception:
        return h0.copy(), _exact_c5(h0)

def _lbfgs_smooth(h, alpha, target, maxiter=200):
    """L-BFGS-B on log-sum-exp smoothed objective."""
    n = len(h)
    
    def fun_and_grad(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -500.0, 500.0)
        exp_vals = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
        w = exp_vals / float(np.sum(exp_vals))
        # Gradient
        n_ = len(x)
        g = np.zeros(n_)
        for r in range(2 * n_ - 1):
            s_val = w[r]
            if s_val < 1e-15:
                continue
            shift = r - (n_ - 1)
            if shift == 0:
                idx = np.arange(n_)
                g[idx] += s_val * (1.0 - 2.0 * x[idx])
            else:
                lo = max(0, -shift)
                hi = min(n_, n_ - shift)
                for i in range(lo, hi):
                    j = i + shift
                    if 0 <= j < n_:
                        g[i] += s_val * (1.0 - x[j])
                        g[j] += s_val * (-x[i])
        g -= np.mean(g)
        return f, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)

def _grad_matrix(h):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for r in range(K):
        s = r - (n - 1)
        if s == 0:
            G[r, :] = 1.0 - 2.0 * h
        else:
            lo = max(0, -s)
            hi = min(n, n - s)
            for i in range(lo, hi):
                j = i + s
                if 0 <= j < n:
                    G[r, i] += (1.0 - h[j])
                    G[r, j] += (-h[i])
    return G

def _lp_direction(h, scores, radius):
    """LP to find descent direction."""
    n = len(h)
    G = _grad_matrix(h)
    K = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(_project(h + a * p, n / 2.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    return best_a, best_c5

    target = n / 2.0
    h = h.copy()

    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coarse_to_fine_binary(b_coarse, n_fine):
    """Upsample a binary vector to a finer resolution."""
    n_coarse = len(b_coarse)
    # Each coarse bin becomes n_fine/n_coarse fine bins
    scale = n_fine / n_coarse
    if scale != int(scale):
        # Just interpolate
        x_old = (np.arange(n_coarse) + 0.5) / n_coarse
        x_new = (np.arange(n_fine) + 0.5) / n_fine
        h_new = np.interp(x_new, x_old, b_coarse)
        return _project(h_new, n_fine / 2.0)
    s = int(scale)
    b_fine = np.repeat(b_coarse, s)
    return b_fine
