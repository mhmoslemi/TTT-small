```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_descent(h, deadline, p_values=None, max_steps=None):
    """
    p-norm gradient descent with adaptive p.
    """
    if p_values is None:
        p_values = [4, 8, 16, 32, 64, 128, 256, 512]
    if max_steps is None:
        max_steps = 2000
    
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for p_idx, p in enumerate(p_values):
        if time.time() > deadline:
            break
        steps_for_p = max_steps // len(p_values)
        for step in range(steps_for_p):
            if time.time() > deadline:
                break
            
            scores = _scores_vec(h)
            # p-norm weights
            eps = 1e-15
            s_pos = np.maximum(scores, eps)
            w = s_pos ** (p - 1)
            w_sum = np.sum(w)
            if w_sum < 1e-300:
                break
            w = w / w_sum  # normalize to sum = 1
            
            # Gradient: G^T @ w
            G = _grad_matrix(h)
            grad = G.T @ w  # shape (n,)
            
            # Project gradient onto constraint null space (sum(h) = target)
            grad = grad - np.mean(grad)
            
            # Normalize
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
            
            # Line search
            cur_c5 = _exact_c5(h)
            best_alpha = 0.0
            best_f = cur_c5
            alpha = 0.1
            for _ in range(30):
                h_try = h - alpha * grad
                h_try = _project(h_try, target)
                f_try = _exact_c5(h_try)
                if f_try < best_f - 1e-14:
                    best_f = f_try
                    best_alpha = alpha
                    h = h_try
                else:
                    alpha *= 0.7
                    if alpha < 1e-12:
                        break
            
            if best_alpha == 0.0:
                break
            if best_f < best_c5:
                best_c5 = best_f
                if time.time() > deadline:
                    break
                best_h = h.copy()
    
    return best_h, best_c5
```
