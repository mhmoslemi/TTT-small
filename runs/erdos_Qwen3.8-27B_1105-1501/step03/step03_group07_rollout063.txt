def _compute_lags_and_grads(h):
    """Compute all lag values and their gradients."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')  # 2n-1 values
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for idx in range(n_lags):
        d = idx - (n - 1)  # actual lag
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:d+m] += (1.0 - h[:m])
                grad[:m] -= h[d:d+m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grad[:m] += (1.0 - h[l:])
                grad[l:] -= h[:m]
        grads[idx] = grad
    return scores, grads

def _solve_lp(h, scores, grads, R, lam, top_k=None):
    """Solve the tangent plane LP. Returns direction delta or None."""
    n = len(h)
    n_lags = 2 * n - 1
    
    # Select lags
    if top_k is not None and top_k < n_lags:
        top_idx = np.argsort(-scores)[:top_k]
    else:
        top_idx = np.arange(n_lags)
    
    n_sel = len(top_idx)
    
    # Variables: t (1), p_0..p_{n-1} (n), q_0..q_{n-1} (n) → total 2n+1
    n_vars = 2 * n + 1
    
    # Objective: minimize t + lam * sum(p+q)
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0
    c_obj[1:n+1] = lam
    c_obj[n+1:] = lam
    
    # Inequality constraints: for each selected lag s:
    #   f_s + grad_s · (p - q) <= t
    #   → -t + grad_s · p - grad_s · q <= -f_s
    A_ub = np.zeros((n_sel, n_vars))
    b_ub = np.zeros(n_sel)
    for i, s in enumerate(top_idx):
        A_ub[i, 0] = -1.0
        A_ub[i, 1:n+1] = grads[s]      # grad_s · p
        A_ub[i, n+1:] = -grads[s]      # -grad_s · q
        b_ub[i] = -scores[s]
    
    # Equality constraints:
    # 1. sum(p) - sum(q) = 0
    A_eq = np.zeros((2, n_vars))
    b_eq = np.zeros(2)
    A_eq[0, 1:n+1] = 1.0
    A_eq[0, n+1:] = -1.0
    b_eq[0] = 0.0
    
    # 2. sum(p+q) <= R → this is inequality, add to A_ub
    A_ub = np.vstack([A_ub, np.zeros((1, n_vars))])
    b_ub = np.concatenate([b_ub, [R]])
    A_ub[-1, 1:n+1] = 1.0
    A_ub[-1, n+1:] = 1.0
    
    # Bounds
    bounds = [(-1000.0, 1000.0)]  # t is free
    for i in range(n):
        if h[i] < 1e-10:
            # Can only increase: p_i free, q_i = 0
            bounds.append((0, None))  # p_i
            bounds.append((0, 0))     # q_i
        elif h[i] > 1.0 - 1e-10:
            # Can only decrease: p_i = 0, q_i free
            bounds.append((0, 0))     # p_i
            bounds.append((0, None))  # q_i
        else:
            bounds.append((0, None))  # p_i
            bounds.append((0, None))  # q_i
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='revised simplex')
        except Exception:
            return None
    
    if res.success is False or res.x is None:
        return None
    
    x = res.x
    p = x[1:n+1]
    q = x[n+1:]
    delta = p - q
    delta -= np.mean(delta)  # ensure exact zero sum
    
    if np.max(np.abs(delta)) < 1e-15:
        return None
    return delta

    """Golden section line search. Returns (best_alpha, best_h, best_c5)."""
    n = len(h)
    # Find alpha_max
    alpha_max = 5.0

def _subgrad_direction(h, scores, T=0.3):
    """Fallback: softmax-weighted gradient direction."""
    n = len(h)
    n_lags = 2 * n - 1
    shifted = scores - np.max(scores)
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.zeros(n_lags)
        w[np.argmax(scores)] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    for r in range(n_lags):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:d+m] += w[r] * (1.0 - h[:m])
                grad[:m] -= w[r] * h[d:d+m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[l:])
                grad[l:] -= w[r] * h[:m]
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return None
    return grad / gnorm

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.where(np.isfinite(v), v, 0.5)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_lags_and_grads(h):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for idx in range(n_lags):
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grads[idx, d:d+m] += (1.0 - h[:m])
                grads[idx, :m] -= h[d:d+m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grads[idx, :m] += (1.0 - h[l:])
                grads[idx, l:] -= h[:m]
    return scores, grads

def _solve_lp(h, scores, grads, R, lam, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    if top_k is not None and top_k < n_lags:
        top_idx = np.argsort(-scores)[:top_k]
    else:
        top_idx = np.arange(n_lags)
    n_sel = len(top_idx)
    n_vars = 2 * n + 1

    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0
    c_obj[1:] = lam

    n_ineq = n_sel + 1
    A_ub = np.zeros((n_ineq, n_vars))
    b_ub = np.zeros(n_ineq)
    for i in range(n_sel):
        s = top_idx[i]
        A_ub[i, 0] = -1.0
        A_ub[i, 1:n+1] = grads[s]
        A_ub[i, n+1:] = -grads[s]
        b_ub[i] = -scores[s]
    A_ub[n_sel, 1:n+1] = 1.0
    A_ub[n_sel, n+1:] = 1.0
    b_ub[n_sel] = R

    A_eq = np.zeros((1, n_vars))
    b_eq = np.zeros(1)
    A_eq[0, 1:n+1] = 1.0
    A_eq[0, n+1:] = -1.0

    bounds = [(-1000.0, 1000.0)]
    for i in range(n):
        if h[i] < 1e-10:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif h[i] > 1.0 - 1e-10:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    x = res.x
    p = np.clip(x[1:n+1], 0, None)
    q = np.clip(x[n+1:], 0, None)
    delta = p - q
    delta -= np.mean(delta)
    if np.max(np.abs(delta)) < 1e-14:
        return None
    return delta

def _line_search(h, delta, target, c5_cur):
    n = len(h)
    alpha_max = 5.0
    for j in range(n):
        if delta[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / delta[j])
        elif delta[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-delta[j]))
    alpha_max = min(max(alpha_max, 0.001), 5.0)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * delta, target))
    f2 = _exact_c5(_project(h + c2 * delta, target))

    for _ in range(35):
        if hi - lo < 1e-9 * max(1.0, alpha_max):
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * delta, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * delta, target))

    opt_a = (lo + hi) / 2.0
    opt_h = _project(h + opt_a * delta, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 >= c5_cur:
        return 0.0, h, c5_cur
    return opt_a, opt_h, opt_c5

def _subgrad_dir(h, scores, T=0.2):
    n = len(h)
    n_lags = 2 * n - 1
    shifted = scores - np.max(scores)
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.zeros(n_lags)
        w[np.argmax(scores)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for r in range(n_lags):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:d+m] += w[r] * (1.0 - h[:m])
                grad[:m] -= w[r] * h[d:d+m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[l:])
                grad[l:] -= w[r] * h[:m]
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

def _resample(h, n_new):
    n = len(h)

    return _project(h_new, n_new / 2.0)
