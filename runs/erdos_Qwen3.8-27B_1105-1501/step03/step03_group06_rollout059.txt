```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    return x

def _correlate_all(h):
    """Compute all correlation values."""
    return np.correlate(h, 1.0 - h, mode='full')

def _slsqp_solve(h, maxiter=500, ftol=1e-12):
    """SLSQP on the epigraph problem."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    K = 2 * n - 1
    
    def objective(x):
        # x[0] = C (the max correlation in C5 scale)
        # x[1:] = h
        return x[0]
    
    def objective_grad(x):
        g = np.zeros(len(x))
        g[0] = 1.0
        return g
    
    def constraints_fun(x):
        """Inequality constraints: C - dx*corr_i >= 0"""
        h_x = x[1:]
        c_val = x[0]
        corr = _correlate_all(h_x)
        return c_val - dx * corr
    
    def equality_con(x):
        return np.array([np.sum(x[1:]) - target])
    
    # Initial C
    h0 = h.copy()
    c0 = _exact_c5(h0) + 0.05  # Start slightly above
    x0 = np.concatenate(([c0], h0))
    
    bounds = [(0.1, 1.0)] + [(0.0, 1.0)] * n
    
    cons = [
        {'type': 'ineq', 'fun': constraints_fun, 'jac': None},
        {'type': 'eq', 'fun': equality_con}
    ]
    
    try:
        res = minimize(objective, x0, jac=objective_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.success or res.fun < c0:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            c_new = _exact_c5(h_new)
            return h_new, c_new
    except Exception:
        pass
    
    return h.copy(), _exact_c5(h)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _slsqp_epigraph(h, maxiter=300, top_k=None):
    """SLSQP on epigraph: min C s.t. C >= dx*corr_i, sum(h)=n/2, 0<=h<=1"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    K = 2 * n - 1
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[0] = 1.0
        return g
    
    def ineq_constraints(x):
        h_x = x[1:]
        corr = np.correlate(h_x, 1.0 - h_x, mode='full')
        if top_k is not None:
            # Use only top-k constraints
            idx = np.argpartition(corr, -top_k)[-top_k:]
            return x[0] - dx * corr[idx]
        return x[0] - dx * corr
    
    def eq_constraints(x):
        return np.array([np.sum(x[1:]) - target])
    
    def eq_grad(x):
        g = np.zeros(len(x))
        g[1:] = 1.0
        return g
    
    c0 = _exact_c5(h)
    x0 = np.concatenate(([c0 + 0.01], h))
    
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * n
    
    ineq_F = ineq_constraints
    # If top_k, number of constraints changes
    n_ineq = top_k if top_k is not None else K
    
    cons = [
        {'type': 'ineq', 'fun': ineq_F},
        {'type': 'eq', 'fun': eq_constraints, 'jac': eq_grad}
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        c_new = _exact_c5(h_new)
        return h_new, c_new
    except Exception:
        pass
    return h.copy(), c0


def _lp_direction_active(h, k_active=15, radius=0.1):
    """LP direction using only top-k active constraints."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    K = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    
    # Select top-k most active constraints
    idx = np.argpartition(corr, -k_active)[-k_active:]
    
    # Build gradient submatrix for active constraints
    G_act = np.zeros((k_active, n))
    for ii, idx_i in enumerate(idx):
        d = idx_i - (n - 1)
        jlo = max(0, -d)
        jhi = min(n - 1 - d, n - 1)
        # dc/dh[m] for each m:
        # Term 1: (1-h[m-d]) if m-d in [jlo, jhi] and 0<=m-d<=n-1
        # Term 2: -h[m+d] if m in [jlo, jhi] and 0<=m+d<=n-1
        m_vals = np.arange(n)
        # Term 1
        md = m_vals - d
        mask1 = (md >= jlo) & (md <= jhi) & (md >= 0) & (md <= n - 1)
        # Term 2
        md2 = m_vals + d
        mask2 = (m_vals >= jlo) & (m_vals <= jhi) & (md2 >= 0) & (md2 <= n - 1)
        G_act[ii, :] = np.where(mask1, 1.0 - h[md[mask1]], 0.0) - np.where(mask2, h[md2[mask2]], 0.0)
    
    # LP: max t subject to G_act * d <= t * (1/dx) ... 
    # Actually let me reformulate. We want to minimize the max correlation.
    # Linearized: corr_i + G_i * alpha * p <= corr_i + t (for active i)
    # where p is the direction, alpha is step size, t is the reduction.
    # Equivalently: G_i * p <= t for all active i, and we want to maximize t (most negative = most reduction)
    # Subject to: sum(p) = 0, |p_i| <= radius
    
    # LP: maximize -t (i.e., minimize t)
    # subject to: G_act * p - t <= 0 (t >= G_act * p for all active)
    #   sum(p) = 0
    #   -radius <= p_i <= radius
    
    # Variables: [p_0, ..., p_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    A_ub = np.zeros((k_active, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(k_active)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))

    bounds_lp = [(-radius, radius)] * n + [(None, None)]
    
    # Also need to keep h + alpha*p in [0,1], but that's handled in line search

                     bounds=bounds_lp, method='highs')
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-10:
                return p
    except Exception:
        pass
    return None


def _line_search_exact(h, direction, radius=0.5):
    """Exact line search: find best alpha in [0, alpha_max] for h + alpha*direction"""
    n = len(h)
    target = n / 2.0
    
    alpha_max = radius
    for j in range(n):
        if direction[j] > 1e-10:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-10:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    c5_cur = _exact_c5(h)
    best_a = 0.0
    best_c5 = c5_cur
    
    # Coarse search
    for t in np.linspace(0, 1, 12):
        a = t * alpha_max
        h_t = _project(h + a * direction, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    # Golden section refinement
    if best_a > 0:
        lo = max(0.0, best_a - 0.2 * alpha_max)
        hi = min(alpha_max, best_a + 0.2 * alpha_max)
        if hi - lo > 1e-10:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
            f2 = _exact_c5(_project(h + c2 * direction, target))

                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(_project(h + c1 * direction, target))
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)

            a = (lo + hi) / 2.0

            c5_t = _exact_c5(h_t)
            if c5_t < best_c5:

def _smooth_step(h, alpha, target):
    """One L-BFGS-B step on smooth max surrogate."""
    n = len(h)
    
    def fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -400.0, 400.0)
        ev = np.exp(vals)
        return ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h, method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': 100, 'ftol': 1e-14, 'iprint': -1})
    x = _project(np.clip(res.x, 0.0, 1.0), target)
```

**Idea: Exploit the problem structure more directly.**

The correlation at lag 0 is: corr_0 = sum_j h[j](1-h[j]) = n/2 - sum_j h[j]^2 (using the constraint sum h = n/2).

To minimize corr_0, maximize sum h[j]^2. This is achieved by making h as "binary" as possible (entries at 0 or 1). But this necessarily increases other lags.

The problem is essentially: find a binary-ish sequence of length n with n/2 ones, such that the cross-correlation with its complement is uniformly low.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _corr_grad_submatrix(h, idx_list):
    """Compute gradient rows for given correlation indices."""
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((len(idx_list), n))
    for ii, idx_i in enumerate(idx_list):
        d = idx_i - (n - 1)
        jlo = max(0, -d)
        jhi = min(n - 1 - d, n - 1)
        m_vals = np.arange(n)
        md = m_vals - d
        mask1 = (md >= jlo) & (md <= jhi) & (md >= 0) & (md <= n - 1)
        md2 = m_vals + d
        mask2 = (m_vals >= jlo) & (m_vals <= jhi) & (md2 >= 0) & (md2 <= n - 1)
        row = np.zeros(n)
        if np.any(mask1):
            row[mask1] += (1.0 - h[md[mask1]])
        if np.any(mask2):
            row[mask2] -= h[md2[mask2]]
        G[ii, :] = row
    return G


def _slsqp_epigraph(h, maxiter=300, top_k=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    K = 2 * n - 1

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(len(x))
        g[0] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[1:]
        corr = np.correlate(h_x, 1.0 - h_x, mode='full')
        if top_k is not None and top_k < K:
            idx = np.argpartition(corr, -top_k)[-top_k:]
            return x[0] - dx * corr[idx]
        return x[0] - dx * corr

    def eq_fun(x):
        return np.array([np.sum(x[1:]) - target])

    def eq_grad(x):
        g = np.zeros(len(x))
        g[1:] = 1.0
        return g

    c0 = _exact_c5(h)
    x0 = np.concatenate(([c0 + 0.02], h))
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'ineq', 'fun': ineq_fun},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = _project(np.clip(res.x[1:], 0.0, 1.0), target)
        c_new = _exact_c5(h_new)
        return h_new, c_new
    except Exception:
        pass
    return h.copy(), c0


def _lp_direction(h, k_active, radius):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ka = min(k_active, 2 * n - 1)
    idx = np.argpartition(corr, -ka)[-ka:]
    G_act = _corr_grad_submatrix(h, idx)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((ka, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(ka)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds_lp = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-10:
                return p
    except Exception:
        pass
    return None


def _line_search(h, p, target):
    n = len(h)
    a_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = max(a_max, 1e-10)
    c5_cur = _exact_c5(h)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 14):
        a = t * a_max
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - 0.2 * a_max)
        hi = min(a_max, best_a + 0.2 * a_max)

            a = (lo + hi) / 2.0

            c5_t = _exact_c5(h_t)
            if c5_t < best_c5:

def _lp_loop(h, deadline, k_active=15, radius_init=0.12, max_iters=3000, stall_limit=45):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    rad = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:

def _smooth_step(h, alpha, target, maxiter=80):
    n = len(h)

    def fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -400.0, 400.0)
        ev = np.exp(vals)
        return ms + (1.0 / alpha) * np.log(float(np.sum(ev)))

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': -1})
        x = _project(np.clip(res.x, 0.0, 1.0), target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _binary_restart(h, n, target):
    """Create a binary-ish restart from current h."""
    h_sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:int(target)]] = 1.0
    # Add small noise
    h_bin += np.clip(np.random.default_rng(0).normal(0, 0.05, n), -0.1, 0.1)
    return _project(h_bin, target)

        best_c5, best_h = c5_f, h_f.copy()
    h_f2, c5_f2 = _lp_loop(best_h, d5, k_active=12, radius_init=0.05,
                           max_iters=3000, stall_limit=50)
    if c5_f2 < best_c5:
        best_c5, best_h = c5_f2, h_f2.copy()
    # One more SLSQP pass
    h_f3, c5_f3 = _slsqp_epigraph(best_h, maxiter=200, top_k=15)
    if c5_f3 < best_c5:
        best_c5, best_h = c5_f3, h_f3.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
