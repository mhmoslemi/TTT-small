def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

def _grad_matrix(h):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _smooth_fun(x, alpha):
    n = len(x)
    dx = 2.0 / n
    s = np.correlate(x, 1.0 - x, mode='full') * dx
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -400.0, 400.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = dx * (G.T @ w)
    g -= np.mean(g)
    return f, g
