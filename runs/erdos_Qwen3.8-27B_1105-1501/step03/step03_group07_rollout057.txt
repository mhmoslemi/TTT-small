    if best_a > 0:
        lo = max(0.0, best_a * 0.7)
        hi = min(alpha_max, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * direction, target))
        f2 = _exact_c5(_project(h + c2 * direction, target))
        for _ in range(15):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * direction, target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * direction, target))
        opt_a = (lo + hi) / 2.0
        opt_c = _exact_c5(_project(h + opt_a * direction, target))
        if opt_c < best_c:
            best_a, best_c = opt_a, opt_c
    return best_a, best_c

def _lp_optimize(h, deadline, target, rng, max_iters=20000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    T = 0.3
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        # Try full LP first, then top-k
        h_lp = _lp_tangent(h, target, max_lags=None)
        if h_lp is not None:
            direction = h_lp - h
            a, c_new = _line_search(h, direction, target, cur_c5)
            if a > 0 and c_new < cur_c5 - 1e-13:
                h = _project(h + a * direction, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
        # Fallback: top-10 LP
        h_lp = _lp_tangent(h, target, max_lags=10)
        if h_lp is not None:
            direction = h_lp - h
            a, c_new = _line_search(h, direction, target, cur_c5)
            if a > 0 and c_new < cur_c5 - 1e-13:
                h = _project(h + a * direction, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
        # Fallback: softmax gradient
        if stall > 5:
            direction = _softmax_dir(h, T)
            a, c_new = _line_search(h, direction, target, cur_c5)
            if a > 0 and c_new < cur_c5 - 1e-13:
                h = _project(h + a * direction, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
        # Stuck: add perturbation
        stall += 1
        T = max(0.05, T * 0.9)
        if stall % 10 == 0:
            pert = rng.normal(0, 0.05, n)
            h = _project(h + pert, target)
            cur_c5 = _exact_c5(h)
    return best_h, best_c5

def _binary_round(h, target):
    n = len(h)
    my_target = int(round(target))
    if my_target <= 0:
        return np.zeros(n)
    if my_target >= n:
        return np.ones(n)
    h_bin = np.zeros(n)
    top_idx = np.argsort(-h)[:my_target]
    h_bin[top_idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fine_polish(h, deadline, target, rng):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    while time.time() < deadline:
        improved = False
        # LP with top-5
        h_lp = _lp_tangent(h, target, max_lags=5)
        if h_lp is not None:
            direction = h_lp - h
            a, c_new = _line_search(h, direction, target, cur_c5, max_alpha=0.5)
            if a > 0 and c_new < cur_c5 - 1e-13:
                h = _project(h + a * direction, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if not improved:
            # Pairwise moves
            for _ in range(50):
                if time.time() > deadline:
                    return best_h, best_c5
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] < 1e-6 or h[j] > 1.0 - 1e-6 or h[k] < 1e-6 or h[k] > 1.0 - 1e-6:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = min((1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                                (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for frac in [0.1, 0.3, 0.5, 0.8, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
        if not improved:
            # Random micro-perturbation
            h_try = _project(h + rng.normal(0, 0.02, n), target)
            c_try = _exact_c5(h_try)
            if c_try < best_c5 - 1e-13:
                h = h_try
                cur_c5 = c_try
                best_c5 = c_try
                best_h = h.copy()
                improved = True
            elif cur_c5 > best_c5 + 0.01:
                h = best_h.copy()
                cur_c5 = best_c5
        if not improved:
            # Try binary push
            h_bin = _binary_round(h, target)
            c_bin = _exact_c5(h_bin)
            if c_bin < best_c5 - 1e-10:
                best_c5 = c_bin
                best_h = h_bin.copy()
                h = h_bin.copy()
                cur_c5 = c_bin
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        h0 = np.array([0.5] * 64)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP optimization from initial (45%)
    dl1 = t0 + budget_s * 0.45
    h1, c5_1 = _lp_optimize(h, dl1, target0, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-start (25%)
    dl2 = t0 + budget_s * 0.70
    n = best_n
    target = best_n / 2.0
    n_starts = 15
    for s in range(n_starts):
        if time.time() > dl2:
            break
        remaining = dl2 - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining * 0.8 / max(1, n_starts - s), 20.0)
        sub_dl = time.time() + alloc
        r = rng.random()
        if r < 0.25:
            h_try = best_h + rng.normal(0, 0.05, n)
        elif r < 0.4:
            h_try = best_h.copy()
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.55:
            h_try = _binary_round(best_h, target) + rng.normal(0, 0.02, n)
        elif r < 0.7:
            shift = rng.integers(1, max(2, n // 8))
            h_try = np.roll(best_h, int(shift))
        elif r < 0.85:
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.6 * best_h + 0.4 * h_rand
        else:
            h_try = 0.5 + (best_h - 0.5) * (0.7 + 0.5 * rng.random())
        h_try = _project(h_try, target)
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, target, rng, max_iters=5000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Resolution sweep (15%)
    dl3 = t0 + budget_s * 0.85
    for dn in [2, -2, 4, -4, 6, -6, 8, -
8, 12, -12, 16, -16, 32, -32]:
        if time.time() > dl3:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + min(15.0, (dl3 - time.time()) * 0.3))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, rng, max_iters=3000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Fine polish (12%)
    dl4 = t0 + budget_s * 0.97
    target = best_n / 2.0
    h4, c5_4 = _fine_polish(best_h, dl4, target, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Final validation
    target = best_n / 2.0
    best_h = _project(best_h, target)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        best_n = n0
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
