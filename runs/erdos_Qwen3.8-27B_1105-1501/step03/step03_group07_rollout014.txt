    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _max_corr_value(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def _init_alternating(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 4 == 0:
        # h[::2] gives n/2 ones, good
        pass
    else:
        # Adjust
        s = h.sum()
        if s > n // 2:
            # Remove excess
            excess = int(s) - n // 2
            ones = np.where(h == 1)[0]
            to_remove = rng.choice(ones, excess, replace=False)
            h[to_remove] = 0
        elif s < n // 2:
            deficit = n // 2 - int(s)
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, deficit, replace=False)
            h[to_add] = 1
    return h

def _greedy_binary(h, max_sweeps=5000):

    n = len(h)
    h = h.copy()
    cur_max = _max_corr_value(h)
    best_h = h.copy()
    best_max = cur_max
    
    for _ in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0.0
        best_i, best_j = -1, -1

            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nm = _max_corr_value(h)
                h[i] = 1.0
                h[j] = 0.0

                    best_delta = nm - best_max
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_max += best_delta
            best_h = h.copy()
        else:
            break

        h = h_init.copy()
    else:
        h = _init_alternating(n, rng)

    best_h = h.copy()
    best_max = cur_max
    
    T = 3.0
    T_min = 0.05
    cooling = 0.99997
    
    while time.time() < t_end:
        if T <= T_min:
            break
        
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_max = _max_corr_value(h)
        
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        elif rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
        else:
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    return best_h, best_max

def _continuous_refine(h, t_start, t_end, rng, target):

    n = len(h)

            cur_c5 = c_try
