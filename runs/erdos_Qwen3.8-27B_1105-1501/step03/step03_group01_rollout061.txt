def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        else:

            cur_c5 = new_c5
            improved = True
    return h, cur_c5, improved

def _subgrad_dir(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wk = w[r]
        if wk < 1e-14:
            continue
        d = r - (n - 1)
        for j in range(n):
            term = 0.0
            jm = j - d
            if 0 <= jm < n:
                term += (1.0 - h[jm])
            jp = j + d
            if 0 <= jp < n:
                term += -h[jp]
            grad[j] += wk * term
    return grad

def _subgrad_step(h, tau, lr):
    n = len(h)
    grad = _subgrad_dir(h, tau)
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return h, _exact_c5(h)
    h_new = h - lr * grad / gn
    h_new = _project(h_new, n / 2.0)
    return h_new, _exact_c5(h_new)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize(h, deadline, rng, n_passes_target=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    pass_count = 0
    while time.time() < deadline and pass_count < n_passes_target:
        pass_count += 1
        h, cur_c5, improved = _pair_descent_pass(h, deadline, rng)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                h, sc5 = _subgrad_step(h, 50.0, 0.02)
                if sc5 < cur_c5 - 1e-12:
                    cur_c5 = sc5
                    stall = 0
            if stall > 30:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick warmup
    d1 = t0 + budget_s * 0.05
    h, c5 = _optimize(h, d1, rng, n_passes_target=15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Deep pair descent with subgradient interplay
    d2 = t0 + budget_s * 0.40
    h = best_h.copy()
    cur_c5 = best_c5
    stall = 0
    pass_n = 0
    while time.time() < d2:
        pass_n += 1
        h, cur_c5, improved = _pair_descent_pass(h, d2, rng)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 8 == 0:
                h, sc5 = _subgrad_step(h, 80.0, 0.03)
                if sc5 < cur_c5 - 1e-12:
                    cur_c5 = sc5
                    if sc5 < best_c5:
                        best_c5 = sc5
                        best_h = h.copy()
                    stall = 0
            if stall > 40:
                h = best_h.copy()
                cur_c5 = best_c5
                stall = 0
                h += rng.normal(0, 0.02, best_n)
                h = _project(h, best_n / 2.0)

    # Phase 3: Multi-resolution search
    d3 = t0 + budget_s * 0.68
    n_list = [64, 72, 80, 84, 88, 96, 104, 112, 120, 128, 144]
    for n_try in n_list:
        if time.time() > d3:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _optimize(h_r, sub_dl, rng, n_passes_target=35)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Perturbation + re-optimization
    d4 = t0 + budget_s * 0.85
    scale = 0.04
    for attempt in range(60):
        if time.time() > d4:
            break
        if best_c5 < 0.38070:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_try, c5_try = _optimize(h_p, sub_dl, rng, n_passes_target=25)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.006, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.2)

    # Phase 5: Final deep polish
    d5 = t0 + budget_s * 0.97
    h = best_h.copy()
    cur_c5 = best_c5
    stall = 0
    while time.time() < d5:
        h, cur_c5, improved = _pair_descent_pass(h, d5, rng, n_grid=20, n_golden=25)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0:
                h, sc5 = _subgrad_step(h, 100.0, 0.015)
                if sc5 < cur_c5 - 1e-12:
                    cur_c5 = sc5
                    if sc5 < best_c5:
                        best_c5 = sc5
                        best_h = h.copy()
                    stall = 0
            if stall > 50:
                break

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
