Now for the optimization: for each pair (i,j), I want to minimize C5(α) over the feasible range. Since C5(α) is a max of quadratics (hence convex), I can use golden section search.

The feasible range for α:
- h[i] + α ≥ 0 → α ≥ -h[i]
- h[i] + α ≤ 1 → α ≤ 1 - h[i]
- h[j] - α ≥ 0 → α ≤ h[j]
- h[j] - α ≤ 1 → α ≥ h[j] - 1

So α_min = max(-h[i], h[j]-1) and α_max = min(1-h[i], h[j])

If α_max ≤ α_min, skip this pair.

Let me now think about the overall algorithm:

1. Initialize h from initial_h_values
2. Repeat:
   a. For each pair (i,j) (in some order):

      - Compute feasible range for α
      - Miniaturize C5(α) using golden section
      - If improvement > tolerance: accept, update h
   b. If no pair improved in a full pass: stop
3. Try different n values
4. Return best

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_at(h, i, j, alpha):
    """Compute C5 after h[i]+=alpha, h[j]-=alpha. Modifies h in place and restores."""
    old_i = h[i]
    old_j = h[j]
    h[i] = old_i + alpha
    h[j] = old_j - alpha
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr)) * (2.0 / n)
    h[i] = old_i
    h[j] = old_j
    return val

def _gold_pair(h, i, j, cur_c5):
    """Minimize C5 over alpha for pair (i,j). Returns (alpha, c5)."""
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    span = a_hi - a_lo
    if span < 1e-14:
        return 0.0, cur_c5
    
    # If span is very small, just try the endpoint
    if span < 1e-6:
        a_try = a_lo + span / 2.0
        c5_try = _c5_at(h, i, j, a_try)
        if c5_try < cur_c5 - 1e-13:
            return a_try, c5_try
        # Try endpoints too
        c5_lo = _c5_at(h, i, j, a_lo)
        if c5_lo < cur_c5 - 1e-13:
            return a_lo, c5_lo
        c5_hi = _c5_at(h, i, j, a_hi)
        if c5_hi < cur_c5 - 1e-13:
            return a_hi, c5_hi
        return 0.0, cur_c5
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_lo
    hi = a_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _c5_at(h, i, j, c)
    fd = _c5_at(h, i, j, d)
    
    for _ in range(45):
        if hi - lo < 1e-11 * max(1.0, span):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _c5_at(h, i, j, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _c5_at(h, i, j, d)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _c5_at(h, i, j, opt_a)
    
    # Check endpoints

    if c5_lo < opt_c5:

    c5_hi = _c5_at(h, i, j, a_hi)
    if c5_hi < opt_c5:
        opt_a, opt_c5 = a_hi, c5_hi
    
    if opt_c5 < cur_c5 - 1e-14:

    return 0.0, cur_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def _tent_h(n):
    """Triangle function: peak at center, 0 at edges."""
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = np.maximum(0, 1.0 - np.abs(x - 1.0))
    return _project_box_sum(h, n / 2.0)

def _two_block(n, rng):
    """Two blocks of 1s."""
    h = np.zeros(n)
    block_size = n // 4
    start1 = rng.integers(0, n // 2)
    start2 = rng.integers(n // 2, n - block_size)
    h[start1:start1 + block_size] = 1.0
    h[start2:start2 + block_size] = 1.0
    # Adjust to get exactly n/2 ones
    # Current sum = 2*block_size = n/2, so we're good
    return _project_box_sum(h, n / 2.0)
