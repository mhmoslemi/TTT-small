    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha0
    stall = 0
    
    for it in range(20000):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        scores = _all_scores(best_h)
        max_s = float(np.max(scores))
        
        # Softmax weights
        w = np.exp(beta * (scores - max_s))
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        
        # Weighted gradient
        g = np.zeros(n)
        for i in range(len(scores)):
            if w[i] > 1e-12:
                d = i - (n - 1)
                g += w[i] * _grad_score(best_h, d)
        
        # Remove mean (to preserve sum constraint naturally)
        g -= np.mean(g)
        
        # Normalize
        gn = float(np.sqrt(np.sum(g * g)))
        if gn < 1e-15:
            break
        g = g / gn
        
        # Try several step sizes
        best_alpha = 0.0
        best_c5_try = best_c5
        for a in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.1]:
            h_try = _project(best_h - a * g, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_alpha = a
        
        if best_alpha > 0:
            best_h = _project(best_h - best_alpha * g, target)
            best_c5 = _exact_c5(best_h)
            alpha = min(alpha * 1.02, 0.2)
            stall = 0
        else:
            alpha *= 0.7
            stall += 1
            if alpha < 1e-6:
                break
    
    return best_h, best_c5

def _slp_iteration(h, radius):
    """One SLP iteration: solve linearized problem, return direction"""
    from scipy.optimize import linprog
    n = len(h)
    scores = _all_scores(h)
    
    # Linearize: Q_k(h + delta) ≈ Q_k(h) + G_k · delta
    # where G_k = grad_score(h, k)
    
    # LP: min z
    # s.t. z - G_k · delta >= Q_k(h) for all k  (i.e., G_k · delta - z <= -Q_k(h))
    #      sum(delta) = 0
    #      -radius <= delta_i <= radius
    #      0 <= h_i + delta_i <= 1  (tighter bounds)
    
    # Variables: [delta_0, ..., delta_{n-1}, z]
    # c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: G_k · delta - z <= -Q_k(h) for all k
    # A_ub has 2n-1 rows
    A_ub = np.zeros((2 * n - 1, n + 1))
    for i in range(2 * n - 1):
        d = i - (n - 1)
        g = _grad_score(h, d)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds for delta
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # z is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_val(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _grad_score(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        grad[d:] += (1.0 - h[:m])
        grad[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        grad[g:] += (1.0 - h[:m])
        grad[:m] -= h[g:]
    return grad

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

            lo = mid

    n = len(h)

    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for i in range(2 * n - 1):
        d = i - (n - 1)
        g = _grad_score(h, d)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0

    b_eq = np.zeros(1)
    
    bounds = []

        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))

    return None
