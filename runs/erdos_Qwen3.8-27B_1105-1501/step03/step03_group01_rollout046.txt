    active = np.where(overlaps >= max_ov - tol / dx)[0]
    grad = G[active].mean(axis=0)
    grad -= np.mean(grad)  # make mean-free
    return grad

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    oh = 1.0 - h
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] = oh[d:]
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] = oh[:m]
                G[r, :m] += -h[g:]
    return G

def _t_max(h, p):
    """Maximum step size to stay in [0,1]"""
    tmax = 1e15
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        tmax = min(tmax, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        tmax = min(tmax, float(np.min(h[neg] / (-p[neg]))))
    return max(tmax, 1e-10)

def _line_search(h, p, c5_cur, dx, n_iter=40):
    """Exact line search along direction p (mean-free). Returns (best_t, best_c5)."""
    tmax = _t_max(h, p)
    if tmax < 1e-12:
        return 0.0, c5_cur
    # Grid search
    ts = np.linspace(0.0, tmax, 25)
    best_t = 0.0
    best_c5 = c5_cur
    for t in ts[1:]:
        h_t = h + t * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_t = t
    if best_t == 0.0:
        return 0.0, c5_cur
    # Golden section around best
    lo = max(0.0, best_t * 0.6)
    hi = min(tmax, best_t * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_iter):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_t = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_t * p)
    if opt_c5 < best_c5:
        return opt_t, opt_c5
    return best_t, best_c5

def _subgrad_optimize(h, deadline, tau_start=50.0, tau_end=500.0, max_iter=5000):
    """Subgradient descent with adaptive tau and line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5_cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    dx = 2.0 / n
    mom = np.zeros(n)
    beta = 0.7
    stall = 0
    
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # Build gradient matrix
        G = _build_G(h)
        # Compute overlaps
        overlaps = np.correlate(h, 1.0 - h, mode='full')
        max_ov = float(np.max(overlaps))
        
        # Adaptive tau: increase over iterations
        frac = min(1.0, it / max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Softmax weights
        exp_w = np.exp(tau * (overlaps - max_ov))
        w = exp_w / float(np.sum(exp_w))
        
        # Weighted subgradient
        grad = w @ G  # shape (n,)
        
        # Make mean-free
        grad -= np.mean(grad)
        
        # Apply box constraint to gradient (don't push into infeasible)

        grad -= np.mean(grad)

            stall += 1
            if stall > 50:
                break
            continue
        
        # Momentum
        p = -(mom + grad / norm)
        p -= np.mean(p)
        mom = beta * (mom + grad / norm)
        
        # Line search
