The problem is to find a step function h: [0,2] → [0,1] with ∫h = 1 that minimizes:
C₅ = max_k ∫ h(x)(1-h(x+k)) dx

The previous solution achieved C₅ ≤ 0.380917, and the target is 0.3808 (lower is better). Current record is 0.38085.

1. Use exact 1D line search along mean-preserving directions
2. For a direction D with sum(D)=0, the path h(t) = h + tD gives:

Actually let me be more careful. Let's expand:
corr(h+tD, 1-h-tD) = corr(h, 1-h) + t*corr(D, 1-h) + t*corr(h, -D) + t²*corr(D, -D)
= corr(h,1-h) + t*corr(D,1-h) - t*corr(h,D) - t²*corr(D,D)

So:
- S_r = corr(h,1-h)_r
- B_r = corr(D,1-h)_r - corr(h,D)_r
- Q_r = -corr(D,D)_r ≤ 0

And C₅(h(t)) = dx * max_r (S_r + t*B_r + t²*Q_r)

The feasible interval for t is determined by 0 ≤ h_i + t*D_i ≤ 1 for all i.

This is a great approach because:
1. Computing S, B, Q is O(n²) once per direction
2. Evaluating C₅ at any t is O(n) (just max of a vector)
3. We can sample many t values cheaply

Now let me think about what direction families to use:

1. **Soft gradient direction**: Weight the rows by softmax of scores, compute gradient
2. **Random block transfer**: Pick two random blocks of equal size, move mass between them
3. **Pair directions**: D[j] = 1, D[i] = -1 for random pairs
4. **Sine directions**: D = sin(freq * x + phase), mean-subtracted
5. **Random smooth directions**: Gaussian-filtered random, mean-subtracted

The key improvement over the previous approach is:
- The previous approach used LP subproblems for direction finding (expensive)
- The new approach just samples the exact objective along each direction (cheap)
- We can afford to try many more directions

Actually, let me reconsider the formula. Let me verify:

np.correlate(a, b, 'full') gives output of length 2n-1 where:
output[k] = sum_i a[i] * b[i + (k-(n-1))]
for valid indices.

More precisely, for np.correlate(a, b, 'full'):
output[k] = sum_i a[i] * b[i - (k - n + 1)]
where the index must be valid (0 ≤ i < n and 0 ≤ i - (k-n+1) < n).

Wait, let me look this up. numpy correlate:
c_k = sum_n a_{n+k} * conj(b_n)

For real vectors: c_k = sum_n a[n+k] * b[n]

With mode='full', k ranges from -(n-1) to n-1, output length 2n-1.
output index for lag k is k + n - 1.

So output[k + n - 1] = sum_n a[n+k] * b[n] where 0 ≤ n+k < n and 0 ≤ n < n.

Now, h(t) = h + tD, 1-h(t) = 1-h-tD

corr(h(t), 1-h(t)) = corr(h + tD, 1-h - tD)
= corr(h, 1-h) + t*corr(D, 1-h) + t*corr(h, -D) + t²*corr(D, -D)
= corr(h,1-h) + t*corr(D,1-h) - t*corr(h,D) - t²*corr(D,D)

So:
- S = correlate(h, 1-h, 'full')
- B = correlate(D, 1-h, 'full') - correlate(h, D, 'full')
- Q = -correlate(D, D, 'full')

C₅(h(t)) = dx * max_k (S[k] + t*B[k] + t²*Q[k])

This is correct.

Now for the feasible interval:
For each i: 0 ≤ h[i] + t*D[i] ≤ 1
If D[i] > 0: -h[i]/D[i] ≤ t ≤ (1-h[i])/D[i]
If D[i] < 0: (1-h[i])/D[i] ≤ t ≤ -h[i]/D[i]  (note D[i] < 0 so this flips)
If D[i] = 0: no constraint

So:
t_lo = max over i with D[i]>0 of (-h[i]/D[i]), and over i with D[i]<0 of ((1-h[i])/D[i])
t_hi = min over i with D[i]>0 of ((1-h[i])/D[i]), and over i with D[i]<0 of (-h[i]/D[i])

Note that t=0 is always feasible, so t_lo ≤ 0 ≤ t_hi.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
            if not np.any(mask):
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0)
            if not np.any(mask):
                mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, D, n_samples=40, golden_iters=30):
    """Exact line search along direction D (mean-preserving)."""
    n = len(h)
    dx = 2.0 / n
    
    # Ensure D is mean-zero
    D = D - np.mean(D)
    
    maxabs = float(np.max(np.abs(D)))
    if maxabs < 1e-14:
        return None, None
    
    # Feasible interval
    pos = D > 1e-14
    neg = D < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / D[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / D[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / D[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / D[neg])))
    
    # Ensure t=0 is in [t_lo, t_hi]
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    
    if t_hi - t_lo < 1e-12:
        return None, None
    
    # Precompute S, B, Q
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = -np.correlate(D, D, mode='full')
    
    def f_at(t):
        return float(np.max(S + t * B + t * t * Q) * dx)
    
    # Grid sampling
    ts = np.linspace(t_lo, t_hi, n_samples)
    # Add some focused samples around 0
    ts = np.concatenate([ts, np.linspace(max(t_lo, t_hi - 0.5 * (t_hi - t_lo)), min(t_hi, 0.0), min(n_samples, 10))])
    
    f0 = f_at(0.0)
    best_t = 0.0
    best_f = f0
    
    for t in ts:
        ft = f_at(t)
        if ft < best_f:
            best_f = ft
            best_t = t
    
    if best_t == 0.0 and best_f >= f0 - 1e-14:
        return None, f0
    
    # Golden section refinement around best_t
    width = (t_hi - t_lo) / n_samples
    lo_g = max(t_lo, best_t - 2 * width)
    hi_g = min(t_hi, best_t + 2 * width)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_g - gr * (hi_g - lo_g)
    c2 = lo_g + gr * (hi_g - lo_g)
    f1 = f_at(c1)
    f2 = f_at(c2)
    
    for _ in range(golden_iters):
        if hi_g - lo_g < 1e-10:
            break
        if f1 < f2:
            hi_g = c2
            c2 = c1
            f2 = f1
            c1 = hi_g - gr * (hi_g - lo_g)
            f1 = f_at(c1)
        else:
            lo_g = c1
            c1 = c2
            f1 = f2
            c2 = lo_g + gr * (hi_g - lo_g)
            f2 = f_at(c2)
    
    opt_t = (lo_g + hi_g) / 2.0
    opt_f = f_at(opt_t)
    
    # Also check endpoints of the golden bracket
    for tc in [lo_g, hi_g, c1, c2]:
        fc = f_at(tc)
        if fc < opt_f:
            opt_f = fc
            opt_t = tc
    
    if opt_f >= f0 - 1e-13:
        return None, f0
    
    return opt_t, opt_f

def _grad_direction(h, tau=50.0):
    """Soft gradient direction (negated to be a descent direction)."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    scores = S * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # Gradient of objective w.r.t. h
    # d/dh_j [S_k] where S_k = corr(h, 1-h)_k
    # From the expansion: B_j = sum of contributions
    # Actually, dS_k/dh_j: S_k = sum_i h[i]*(1-h[i-shift_k])
    # dS_k/dh_j = (1-h[j-shift_k]) - h[j] ... wait need to be careful
    
    # Let me use the formula from the line search:
    # If D = e_j (unit vector), then:
    # B_k = corr(e_j, 1-h)_k - corr(h, e_j)_k
    # corr(e_j, 1-h)_k: e_j has 1 at position j
    # corr(a,b)_k = sum_n a[n+k]*b[n] (with numpy convention)
    # Actually let me just compute the gradient directly.
    
    # The gradient of C5 w.r.t. h[i] is:
    # sum over active k where S_k = max of dS_k/dh[i] * dx
    # But we use the soft version with weights w
    
    # dS_k/dh[i] where S_k = sum_m h[m] * (1-h[m - (k-(n-1))])
    # Using numpy correlate convention:
    # correlate(h, 1-h, 'full')[k] = sum_m h[m] * (1-h)[m - (k-n+1)]
    # = sum_m h[m] * (1-h[m - (k-n+1)])
    # for valid m and m-(k-n+1)
    
    # dS_k/dh[i] = (1-h[i-(k-n+1)]) + h[i] * (-1) * delta when i-(k-n+1) is in range
    # Wait: S_k = sum_m h[m] * (1 - h[m - d_k]) where d_k = k - n + 1
    # dS_k/dh[i] = (1 - h[i - d_k]) [from first factor, m=i]
    #   - sum_m h[m] * delta(m - d_k = i) [from second factor]
    #   = (1 - h[i - d_k]) - h[i + d_k]  (if indices valid)
    
    # So gradient[i] = sum_k w[k] * dx * [(1 - h[i-d_k])_valid - h[i+d_k]_valid]
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        # First term: (1 - h[i - d]) for valid i-d
        # i ranges from max(0, d) to min(n-1, n-1+d)
        i_lo = max(0, d)
        i_hi = min(n - 1, n - 1 + d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - d])
        # Second term: -h[i + d] for valid i+d
        i_lo2 = max(0, -d)
        i_hi2 = min(n - 1, n - 1 - d)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] += wk * (-h[idx2 + d])
    
    # Make it a descent direction
    D = -grad
    D -= np.mean(D)
    maxabs = float(np.max(np.abs(D)))
    if maxabs > 1e-14:
        D /= maxabs
    return D

def _block_transfer_direction(n, block_size, rng):
    """Random equal-size block transfer direction."""
    D = np.zeros(n)
    # Pick two random blocks of same size
    max_start = n - block_size
    if max_start <= 0:
        return D
    s1 = int(rng.integers(0, max_start + 1))
    s2 = int(rng.integers(0, max_start + 1))
    if s1 == s2:
        s2 = (s2 + 1) % (max_start + 1)
    # Overlap check - if blocks overlap, the D values will cancel which is fine
    D[s1:s1+block_size] += 1.0
    D[s2:s2+block_size] -= 1.0
    D -= np.mean(D)  # Should already be ~0
    return D

def _sine_direction(n, rng):
    """Random sine direction, mean-subtracted."""
    x = np.arange(n, dtype=float)
    freq = rng.uniform(0.5, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    D = np.sin(freq * x / n * 2 * np.pi + phase)
    D -= np.mean(D)
    maxabs = float(np.max(np.abs(D)))
    if maxabs > 1e-14:
        D /= maxabs
    return D

def _random_smooth_direction(n, rng, smoothness=0.3):
    """Random direction with Gaussian smoothing."""
    D = rng.standard_normal(n)
    # Simple box smoothing
    k = max(1, int(smoothness * n / 4))
    kernel = np.ones(k) / k
    D = np.convolve(D, kernel, mode='same')
    D -= np.mean(D)
    maxabs = float(np.max(np.abs(D)))
    if maxabs > 1e-14:
        D /= maxabs
    return D

def _pair_direction(n, rng):
    """Random pair direction D[j]=1, D[i]=-1."""
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return None
    D = np.zeros(n)
    D[j] = 1.0
    D[i] = -1.0
    return D

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
