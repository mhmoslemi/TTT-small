def _active_gradients(h, active):
    n=len(h); G=np.zeros((len(active),n))
    for idx,r in enumerate(active):
       k=int(r)-(n-1)
       g=np.zeros(n)
       if k>=0:
           m=n-k
           if m>0:
              g[:m] -= h[k:k+m]
              g[k:k+m] += 1.0 - h[:m]
       else:
           l=-k; m=n-l
           if m>0:
              g[:m] += 1.0 - h[l:l+m]
              g[l:l+m] -= h[:m]
       G[idx]=g
    return G

def _lp_direction(h, active):
    n=len(h); K=len(active)
    if K==0: return None
    G = _active_gradients(h, active)
    norms = np.linalg.norm(G, axis=1)
    norms[norms < 1e-12] = 1.0
    Gn = G / norms[:, None]
    try:
       c = np.concatenate([np.zeros(n), [1.0]])
       A_ub = np.vstack([Gn, -np.ones((K,1))])
       b_ub = np.zeros(K)
       A_eq = np.concatenate([np.ones(n), [0.0]])
       bounds = [(-1.0,1.0) for _ in range(n)] + [(-10.0,10.0)]
       try:
          res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.reshape(1,-1), b_eq=0.0, bounds=bounds, method='highs', options={'presolve': True})
       except Exception:
          res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.reshape(1,-1), b_eq=0.0, bounds=bounds)

          p = res.x[:n]
          t = float(res.x[n])
          if p is not None and t < -1e-5:
             pmax = float(np.max(np.abs(p)))
             if pmax > 1e-8:
                p = p / pmax
                p -= np.mean(p)

                if pmax2 > 1e-8:
                   return p / pmax2

       pass

    dmax = float(np.max(np.abs(d)))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

def _active_gradients(h, active):
    n = len(h)
    G = np.zeros((len(active), n), dtype=float)
    for idx, r in enumerate(active):
        r = int(r)
        k = r - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            m = n - k
            if m > 0:
                g[:m] -= h[k:k + m]
                g[k:k + m] += 1.0 - h[:m]
        else:
            l = -k
            m = n - l
            if m > 0:
                g[:m] += 1.0 - h[l:l + m]
                g[l:l + m] -= h[:m]
        G[idx] = g
    return G

    n = _G_N
    G = _active_gradients(h, _G_ACTIVE)
    jac = np.empty((len(_G_ACTIVE), n + 1))
    jac[:, :n] = -G * _G_SCALE
    jac[:, n] = 1.0
    return jac


def _slsqp_polish(h, deadline, maxiter=80, max_active=32, max_rounds=3):
    h = _project(h, len(h) / 2.0)
    best_h = h.copy()
    best_M = _exact_M(h)
    n = len(h)
    ncons = 2 * n - 1
    max_active = max(1, min(int(max_active), ncons))

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        h_cur = _project(h, n / 2.0)
        M_cur = _exact_M(h_cur)
        if M_cur < best_M:
            best_M = M_cur
            best_h = h_cur.copy()

        scores = _scores(h_cur)
        maxs = float(np.max(scores))
        scale = 2.0 / n

        active = np.where(scores >= maxs - 0.05)[0]
        if active.size > max_active:
            active = np.argsort(-scores)[:max_active]
        elif active.size < 5 and ncons >= 5:
            active = np.argsort(-scores)[:5]

        _G_N = n
        _G_TARGET = n / 2.0
        _G_SCALE = scale
        _G_ACTIVE = active

        T0 = maxs * scale + 0.01
        z0 = np.concatenate([h_cur, np.array([T0], dtype=float)])
        bounds = [(0.0, 1.0) for _ in range(n)] + [(0.0, 2.0)]
        constraints = [
            {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac},
            {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac}
        ]

        try:
            res = minimize(
                _slsqp_obj, z0, method='SLSQP', jac=_slsqp_obj_jac,
                bounds=bounds, constraints=constraints,
                options={'maxiter': max(10, int(maxiter)), 'ftol': 1e-12}
            )
            if res is not None and getattr(res, 'x', None) is not None:
                h_new = _project(np.asarray(res.x[:n], dtype=float), n / 2.0)
                M_new = _exact_M(h_new)
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_M

def _pair_polish(h, deadline, target, top_pairs=300, max_passes=120):
    h = _project(h, target)
    n = len(h)
    best_h = h.copy()
    best_M = _exact_M(h)
    rng = np.random.default_rng(2048)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        scores = _scores(h)
        curM = float(np.max(scores))
        if curM < best_M - 1e-12:
            best_M = curM
            best_h = h.copy()

        threshold = curM - 0.08
        act = np.where(scores >= threshold)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(scores))])
        if act.size > 10:
            act = np.argsort(-scores)[:10]

        G = _active_gradients(h, act)
        norms = np.linalg.norm(G, axis=1)
        norms[norms < 1e-12] = 1.0
        g = np.sum(G / norms[:, None], axis=0)

        improved = False
        accepted = 0

        if n <= 160:
            diff = g[:, None] - g[None, :]
            neg_vals = np.where(diff < -1e-7, diff, 1e300)
            pos_vals = np.where(diff > 1e-7, -diff, 1e300)
            neg_order = np.argsort(neg_vals)
            pos_order = np.argsort(pos_vals)

            a_neg = min(len(neg_order), max(1, top_pairs // 2))
            a_pos = min(len(pos_order), max(1, top_pairs - a_neg))

            for t in range(a_neg):
                idx = int(neg_order[t])
                i = idx // n
                j = idx % n
                if i == j:
                    continue
                if h[i] >= 1.0 - 1e-10 or h[j] <= 1e-10:
                    continue
                a, Mnew = _line_search_pair(h, i, j, target, curM)
                if a > 0.0 and Mnew < curM - 1e-12:
                    h[i] += a
                    h[j] -= a
                    curM = Mnew
                    if curM < best_M - 1e-12:
                        best_M = curM
                        best_h = h.copy()
                    improved = True
                    accepted += 1
                    if accepted > 120:
                        break

            if accepted <= 120:
                for t in range(a_pos):
                    idx = int(pos_order[t])
                    i = idx // n
                    j = idx % n
                    if i == j:
                        continue
                    if h[j] >= 1.0 - 1e-10 or h[i] <= 1e-10:
                        continue
                    a, Mnew = _line_search_pair(h, j, i, target, curM)
                    if a > 0.0 and Mnew < curM - 1e-12:
                        h[j] += a
                        h[i] -= a
                        curM = Mnew
                        if curM < best_M - 1e-12:
                            best_M = curM
                            best_h = h.copy()
                        improved = True
                        accepted += 1
                        if accepted > 120:
                            break

        if not improved and n <= 2000:
            for _ in range(180):
                if time.time() > deadline:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                if rng.random() < 0.5:
                    if h[i] >= 1.0 - 1e-10 or h[j] <= 1e-10:
                        continue
                    a, Mnew = _line_search_pair(h, i, j, target, curM)
                    if a > 0.0 and Mnew < curM - 1e-12:
                        h[i] += a
                        h[j] -= a
                        curM = Mnew
                        if curM < best_M - 1e-12:
                            best_M = curM
                            best_h = h.copy()
                        improved = True
                else:
                    if h[j] >= 1.0 - 1e-10 or h[i] <= 1e-10:
                        continue
                    a, Mnew = _line_search_pair(h, j, i, target, curM)
                    if a > 0.0 and Mnew < curM - 1e-12:
                        h[j] += a
                        h[i] -= a
                        curM = Mnew
                        if curM < best_M - 1e-12:
                            best_M = curM
                            best_h = h.copy()
                        improved = True
                if improved and accepted > 25:
                    break

        if not improved:
            break

    return best_h, best_M


def _local_optimize(h, deadline, use_slsqp=True, slsqp_iter=80,
                    pair_top=300, pair_passes=120):
    h = _project(h, len(h) / 2.0)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_M = _exact_M(h)

    if deadline <= time.time():
        return best_h, best_M

    rem = deadline - time.time()
    if use_slsqp and n <= 110 and rem > 12.0:
        dl_s = time.time() + min(rem * 0.30, 20.0 if slsqp_iter > 50 else 10.0)
        h_s, M_s = _slsqp_polish(best_h, dl_s, maxiter=slsqp_iter,
                                 max_active=min(50, 2 * n - 1), max_rounds=2)
        if M_s < best_M - 1e-12:
            best_h = h_s.copy()
            best_M = M_s

    rem = deadline - time.time()
    if rem > 6.0:
        dl_a = time.time() + rem * 0.45
        h_a, M_a = _active_optimize(best_h, dl_a, target)
        if M_a < best_M - 1e-12:
            best_h = h_a.copy()
            best_M = M_a

    rem = deadline - time.time()
    if rem > 5.0:
        dl_p = time.time() + rem * 0.85
        h_p, M_p = _pair_polish(best_h, dl_p, target,
                                top_pairs=pair_top, max_passes=pair_passes)
        if M_p < best_M - 1e-12:
            best_h = h_p.copy()
            best_M = M_p

    rem = deadline - time.time()
    if use_slsqp and n <= 110 and rem > 18.0:
        h_s, M_s = _slsqp_polish(best_h, time.time() + rem * 0.55,
                                 maxiter=max(30, min(slsqp_iter, 120)),
                                 max_active=2 * n - 1, max_rounds=2)
        if M_s < best_M - 1e-12:
            best_h = h_s.copy()
            best_M = M_s

    rem = deadline - time.time()
    if rem > 3.0:
        dl_f = time.time() + rem * 0.85
        h_f, M_f = _pair_polish(best_h, dl_f, target,
                                top_pairs=min(int(pair_top), 800), max_passes=60)
        if M_f < best_M - 1e-12:
            best_h = h_f.copy()
            best_M = M_f

    return best_h, _exact_M(best_h) if best_h is not None else best_M


def _try_candidate(h, best_h, best_M, best_n):
    if h is None:
        return best_h, best_M, best_n
    h = _project(h, len(h) / 2.0)
    M = _exact_M(h)
    if M < best_M:
        return h.copy(), M, len(h)
    return best_h, best_M, best_n


def run(seed=42, budget_s=1100, **kwargs):
    global _DE_TARGET

    t0 = time.time()
    total = float(max(1.0, budget_s))
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = _pattern_sine(64, 2.0, 0.7)

    if h0.size == 0:
        h0 = _pattern_sine(64, 2.0, 0.7)
    if h0.size % 2 == 1:
        h0 = _resample(h0, h0.size + 1)
    if h0.size > 128:
        h0 = _resample(h0, 128)

    best_n = len(h0)
    best_h = _project(h0, best_n / 2.0)
    best_M = _exact_M(best_h)

    for s in range(best_n):
        best_h, best_M, best_n = _try_candidate(np.roll(best_h, s), best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(1.0 - best_h, best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(best_h[::-1], best_h, best_M, best_n)

    p1 = min(300.0, max(20.0, 0.30 * total))
    if p1 > 0.9 * total:
        p1 = 0.5 * total
    phase1_dl = t0 + p1

    add2 = min(450.0, max(60.0, 0.65 * (total - p1)), total - p1)
    phase2_dl = t0 + min(0.97 * total, p1 + max(0.0, add2))
    phase3_dl = t0 + 0.995 * total

    if time.time() < phase1_dl - 2.0:
        h1, M1 = _local_optimize(best_h, phase1_dl, use_slsqp=True,
                                 slsqp_iter=110, pair_top=650, pair_passes=220)
        best_h, best_M, best_n = _try_candidate(h1, best_h, best_M, best_n)

    cands = []
    base = best_n
    for delta in (0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12,
                  -16, 16, -20, 20, 24, -24, 32, -32, 40, -40):
        n_try = base + delta
        if 8 <= n_try <= 160 and n_try not in cands:
            cands.append(n_try)
    for n_try in (8, 10, 12, 14, 16, 18, 20, 24, 28, 32, 36, 40, 44, 48,
                  52, 56, 60, 64, 68, 72, 76, 80, 88, 96, 104, 112, 120, 128, 136, 144):
        if n_try not in cands and 8 <= n_try <= 160:
            cands.append(n_try)

    idx = 0
    while idx < len(cands) and time.time() < phase2_dl - 4.0:
        n_try = int(cands[idx])
        idx += 1
        rem = phase2_dl - time.time()
        if rem < 8.0:
            break
        left = max(1, len(cands) - idx + 1)
        alloc = rem * 0.9 / left
        alloc = min(35.0, max(5.0, alloc))
        if rem - alloc < 5.0:
            alloc = max(3.0, rem * 0.4)
        dl = time.time() + min(alloc, rem)

        inits = []
        if n_try == best_n:
            inits.append(best_h.copy())
        else:
            inits.append(_resample(best_h, n_try))
        if n_try <= 64:
            inits.append(_pattern_sine(n_try, 2.0, 0.7))
            inits.append(_pattern_sine(n_try, 3.0, 1.9))
            inits.append(_pattern_random_binary(n_try, rng))
            inits.append(_binarize_to_weight(inits[0], n_try / 2.0))

        if (differential_evolution is not None and n_try <= 20 and alloc > 15.0):
            _DE_TARGET = n_try / 2.0
            bounds = [(0.0, 1.0) for _ in range(n_try)]
            try:
                de_res = differential_evolution(
                    _de_objective, bounds,
                    maxiter=8, popsize=3, tol=1e-5,
                    seed=int(seed + 101 * n_try) % 1000000,
                    polish=False, init='latinhypercube'
                )
                if de_res is not None and getattr(de_res, 'x', None) is not None:
                    inits.append(_project(np.asarray(de_res.x, dtype=float), n_try / 2.0))
            except Exception:
                pass

        init_h = inits[0]
        init_h = _project(init_h, n_try / 2.0)
        init_M = _exact_M(init_h)
        for ht in inits[1:]:
            ht = _project(ht, n_try / 2.0)
            Mh = _exact_M(ht)
            if Mh < init_M:
                init_h = ht
                init_M = Mh

        if init_M < best_M:
            best_h = init_h.copy()
            best_M = init_M
            best_n = n_try

        if alloc > 4.0:
            slsqp_iter = 16 if alloc < 10.0 else (50 if alloc < 20.0 else 90)
            pair_top = 120 if alloc < 12.0 else (350 if alloc < 25.0 else 700)
            pair_passes = 50 if alloc < 20.0 else 120
            h_opt, M_opt = _local_optimize(
                init_h, dl,
                use_slsqp=(alloc > 8.0 and n_try <= 90),
                slsqp_iter=slsqp_iter,
                pair_top=pair_top,
                pair_passes=pair_passes
            )
            best_h, best_M, best_n = _try_candidate(h_opt, best_h, best_M, best_n)

    for s in range(best_n):
        best_h, best_M, best_n = _try_candidate(np.roll(best_h, s), best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(1.0 - best_h, best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(best_h[::-1], best_h, best_M, best_n)

    if time.time() < phase3_dl - 5.0:
        h3, M3 = _local_optimize(best_h, phase3_dl, use_slsqp=True,
                                 slsqp_iter=150, pair_top=1500, pair_passes=450)
        best_h, best_M, best_n = _try_candidate(h3, best_h, best_M, best_n)

    if time.time() < phase3_dl - 15.0:
        max_active = max(1, 2 * best_n - 1)
        h_f, M_f = _slsqp_polish(best_h, phase3_dl, maxiter=120,
                                 max_active=max_active, max_rounds=2)
        best_h, best_M, best_n = _try_candidate(h_f, best_h, best_M, best_n)

    if time.time() < phase3_dl - 10.0:
        h_p, M_p = _pair_polish(best_h, phase3_dl, best_n / 2.0,
                                top_pairs=max(1000, best_n * (best_n - 1)),
                                max_passes=600)
        best_h, best_M, best_n = _try_candidate(h_p, best_h, best_M, best_n)

    for s in range(best_n):
        best_h, best_M, best_n = _try_candidate(np.roll(best_h, s), best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(1.0 - best_h, best_h, best_M, best_n)
    best_h, best_M, best_n = _try_candidate(best_h[::-1], best_h, best_M, best_n)

    best_h = _project(best_h, best_n / 2.0)
    final_M = _exact_M(best_h)
    final_c5 = float(final_M * (2.0 / best_n))
    return best_h, final_c5, int(best_n)
