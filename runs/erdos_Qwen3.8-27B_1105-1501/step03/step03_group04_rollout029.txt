        m = n - g
        G[n - 1 + g, g:] += oh[:m]
        G[n - 1 + g, :m] -= h[g:]
        G[n - 1 - g, :m] += oh[g:]
        G[n - 1 - g, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, c5_cur, n):
    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] + 0.5)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) + 0.5)
    alpha_max = min(max(alpha_max, 0.01), 3.0)
    grid = np.geomspace(0.01, alpha_max, 16)
    best_a = 0.0
    best_c5 = c5_cur
    for a in grid:
        c5_t = _exact_c5(_project(h + a * p, target))
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        lo_a, hi_a = 0.0, alpha_max
    else:
        lo_a = max(0.0, best_a * 0.6)
        hi_a = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(18):
        if hi_a - lo_a < 1e-10:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo_a + hi_a) * 0.5
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        best_a, best_c5 = opt_a, opt_c5
    if best_a < grid[0]:
        best_a, best_c5 = 0.0, c5_cur
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.08, max_iters=10000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.zeros(1)
        bounds = tuple((max(-radius, -h[j] + 1e-9), min(radius, 1.0 - h[j] + 1e-9)) for j in range(n))
        bounds = bounds + ((None, None),)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            stall += 1
            radius *= 0.9
            continue
        if res.success:
            p = res.x[:n]
            p_mean = float(np.mean(p))
            p = p - p_mean
            pnorm = float(np.max(np.abs(p)))
            if pnorm < 1e-12:
                stall += 1
                radius *= 0.88
                if radius < 1e-6:
                    break
                continue
            p = p / pnorm * radius
            alpha, new_c5 = _line_search(h, p, cur_c5, n)
            if new_c5 < cur_c5 - 5e-13:
                h = _project(h + alpha * p, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.15, 0.5)
            else:
                stall += 1
                radius *= 0.82
                if stall % 20 == 0 and radius < 0.02:
                    radius = 0.05
        else:
            stall += 1
            radius *= 0.85
            if radius < 1e-6:
                break
    return best_h, best_c5

def _soft_descent(h, deadline, beta_start=20.0, beta_end=3000.0, max_iters=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.12
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        wsum = float(np.sum(exps))
        weights = exps / wsum
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.5)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.72
            if lr < 1e-5 or stall > 50:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_diverse(h, rng, scale, mode):
    n = len(h)
    target = n / 2.0
    h2 = h.copy()
    if mode == 0:
        h2 += rng.normal(0, scale, n)
    elif mode == 1:
        idx = rng.choice(n, size=2, replace=False)
        amt = scale * rng.uniform(0.3, 1.0)
        h2[idx[0]] += amt
        h2[idx[1]] -= amt
    elif mode == 2:
        mask = (h2 > 0.6) & (h2 < 0.95)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            j = idx[rng.integers(len(idx))]
            h2[j] -= scale
        mask2 = (h2 < 0.4) & (h2 > 0.05)
        idx2 = np.where(mask2)[0]
        if len(idx2) > 0:
            k = idx2[rng.integers(len(idx2))]
            h2[k] += scale
    else:
        perm = rng.permutation(n)
        h2 = h[perm]
    return _project(h2, target)

def _compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Soft descent to refine (20% budget)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _soft_descent(h, dl1, beta_start=15.0, beta_end=4000.0, max_iters=20000)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: LP polish (25% budget)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = _lp_polish(h1, dl2, radius_init=0.10, max_iters=15000)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Basin hopping (25% budget)
    dl3 = t0 + budget_s * 0.70
    scale = 0.04
    attempts = 0
    while time.time() < dl3:
        attempts += 1
        mode = int(rng.integers(0, 4))
        h_pert = _perturb_diverse(best_h, rng, scale, mode)
        rem = dl3 - time.time()
        sub_timeout = min(rem * 0.15, 8.0)
        h_t, c5_t = _soft_descent(h_pert, time.time() + sub_timeout * 0.5,
                                  beta_start=10.0, beta_end=2000.0, max_iters=2000)
        h_t, c5_t = _lp_polish(h_t, time.time() + sub_timeout * 0.5,
                                radius_init=0.06, max_iters=3000)
        if c5_t < best_c5 - 1e-11:
            best_c5, best_h = c5_t, h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.20, scale * 1.1)

    # Phase 4: Multi-resolution sweep (20% budget)
    dl4 = t0 + budget_s * 0.90
    n_candidates = sorted(set([best_n - 8, best_n - 4, best_n - 2, best_n + 2,
                               best_n + 4, best_n + 8, best_n + 12,
                               48, 56, 72, 80, 88, 96]))
    n_candidates = [x for x in n_candidates if 40 <= x <= 128]
    for n_try in n_candidates:
        if time.time() > dl4:
            break
        if n_try == best_n:
            continue
        rem = dl4 - time.time()
        if rem < 5:
            break
        h_r = _resample(best_h, n_try)
        t_sub = min(rem * 0.4, 20.0)
        h_r, c5_r = _lp_polish(h_r, time.time() + t_sub * 0.5,
                                radius_init=0.08, max_iters=3000)
        h_r, c5_r = _soft_descent(h_r, time.time() + t_sub * 0.3,
                                   beta_start=30.0, beta_end=2000.0, max_iters=2000)
        h_r, c5_r = _lp_polish(h_r, time.time() + t_sub * 0.2,
                                radius_init=0.05, max_iters=2000)
        if c5_r < best_c5 - 1e-11:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Final aggressive LP polish (10% budget)
    dl5 = t0 + budget_s * 0.99
    h_f, c5_f = _lp_polish(best_h, dl5, radius_init=0.04, max_iters=50000)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    # Final projection and cleanup
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
