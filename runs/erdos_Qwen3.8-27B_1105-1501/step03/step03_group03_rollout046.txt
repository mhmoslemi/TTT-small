def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    """Compute gradient matrix G of shape (2n-1, n) where G[k,:] is grad of score k w.r.t. h"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            # s_r = sum_{j=0}^{m-1} h[j+d] * (1-h[j])
            # d/dh[j] = -h[j+d] for j in [0,m)
            # d/dh[j+d] = (1-h[j]) for j in [0,m)
            G[r, d:d+m] += (1.0 - h[:m])
            G[r, :m] -= h[d:d+m]
        else:
            g = -d
            # s_r = sum_{j=0}^{m-1} h[j] * (1-h[j+g])

            G[r, :m] += (1.0 - h[g:g+m])
            G[r, g:g+m] -= h[:m]
    return G

def _lp_direction(h, scores, G, active_indices, radius):

    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize t
    A_ub = np.zeros((len(active_indices), n + 1))
    A_ub[:, :n] = G[active_indices, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[active_indices]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        pass
    return None

    return _project(h_new, n_new / 2.0)

def _sa_step(h, target, rng, temp):
    """One step of simulated annealing with structured moves"""
    n = len(h)
    cur_c5 = _exact_c5(h)
    
    move_type = rng.random()
    
    if move_type < 0.4:
        # Random zero-sum direction
        d = rng.standard_normal(n)
        d -= np.mean(d)
        norm = np.sqrt(np.sum(d**2))
        if norm < 1e-12:
            return h, cur_c5
        d /= norm
        scale = rng.uniform(0.01, 0.05)
        h_new = _project(h + scale * d, target)
    elif move_type < 0.7:
        # Pair exchange
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            return h, cur_c5
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        scale = rng.uniform(0.02, 0.1) * rng.choice([1.0, -1.0])
        h_new = _project(h + scale * d, target)
    else:
        # Random coordinate shift
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            j = (i + 1) % n
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        h_prev = h[i]
        h_new = h.copy()
        h_new[i] = min(1.0, max(0.0, h_prev + rng.uniform(-0.1, 0.1)))
        h_new = _project(h_new, target)
    
    new_c5 = _exact_c5(h_new)
    delta = new_c5 - cur_c5
    
    if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
        return h_new, new_c5
    return h, cur_c5

def _exact_c5(h):

    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:

        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)

            G[r, d:d+m] += (1.0 - h[:m])
            G[r, :m] -= h[d:d+m]
        else:
            g = -d

            G[r, g:g+m] -= h[:m]
    return G
