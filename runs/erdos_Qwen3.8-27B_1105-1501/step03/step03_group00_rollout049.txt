def _lp_step(h, scores, G, radius, top_k=None):
    """LP-based step: find direction δ that minimizes max active score within trust region"""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G
    
    # min t
    # s.t. G_k @ δ - t ≤ -s_k  (i.e., s_k + G_k @ δ ≤ t)
    #      sum(δ) = 0
    #      bounds on δ
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(s_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except:
        pass
    return None, None

def _pnorm_gradient(h, scores, G, p):
    """Compute gradient of p-norm of scores"""
    eps = 1e-15
    ss = np.maximum(scores, eps)
    w = ss ** (p - 1)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    grad = G.T @ w
    grad -= np.mean(grad)  # remove component along constraint
    return grad

    best_h = h.copy()

        G_act = G[idx]
        s_act = scores[idx]
        
        # Find direction δ that minimizes ||G_act @ δ||² (reduces all active scores)
        # subject to sum(δ) = 0, ||δ|| ≤ 1
        # This is equivalent to: min ||G_act @ δ||² s.t. 1^T δ = 0, ||δ|| ≤ 1
        # The solution is the smallest eigenvector of (G_act^T G_act) in the subspace 1^T δ = 0
        
        # Compute A = G_act^T G_act (n x n)
        A = G_act.T @ G_act
        # Project out the mean: P = I - (1/n) * 11^T
        P = np.eye(n) - np.ones((n, n)) / n
        A_proj = P @ A @ P
        
        # Find smallest eigenvalue/eigenvector
        eigvals, eigvecs = np.linalg.eigh(A_proj)
        # Get the eigenvector with smallest eigenvalue (most negative direction in A)
        # Actually we want to minimize δ^T A δ, so we want the smallest eigenvalue
        idx_min = np.argmin(eigvals)
        delta = eigvecs[:, idx_min]
        
        # Normalize
        dnorm = np.linalg.norm(delta)
        if dnorm < 1e-15:
            break
        delta /= dnorm
        
        # Line search
        alpha_max = 0.3
        for j in range(n):
            if delta[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / delta[j])
            elif delta[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-delta[j]))
        alpha_max = max(alpha_max, 1e-12)
        
        best_alpha = 0
        best_new_c5 = cur_c5
        for a in np.linspace(0, alpha_max, 10):
            h_new = _project(h + a * delta, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_new_c5:
                best_new_c5 = c5_new
                best_alpha = a
        
        if best_new_c5 < cur_c5 - 1e-13:
            h = _project(h + best_alpha * delta, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5
