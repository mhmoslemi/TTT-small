def _corr_lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
            g[k:] += -h[:m]
    else:
        m = n + k
        if m > 0:
            g[-k:] += (1.0 - h[:m])
            g[:m] += -h[-k:]
    return g


def _slsqp_run(h0, active_k, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    corr = _all_corr(h0)
    max_c = float(np.max(corr))
    idx = np.argsort(-corr)[:active_k]

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    constraints = []
    for d_idx in idx:
        lag = int(d_idx) - (n - 1)

        def _make_ineq(lag):
            def fun(x):
                return x[0] - _corr_lag(x[1:], lag)
            return fun

        def _make_ineq_jac(lag):
            def jac(x):
                g = np.zeros(n + 1)
                g[0] = 1.0
                g[1:] = -_corr_lag_grad(x[1:], lag)
                return g
            return jac

        constraints.append({'type': 'ineq', 'fun': _make_ineq(lag), 'jac': _make_ineq_jac(lag)})

    def _eq_fun(x):
        return float(np.sum(x[1:]) - target)

    def _eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    constraints.append({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac})

    x0 = np.zeros(n + 1)
    x0[0] = max_c + 0.05
    x0[1:] = h0
    bounds = [(-10.0, 20.0)] + [(0.0, 1.0)] * n

    best_h, best_c5 = h0.copy(), _exact_c5(h0)
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project_feasible(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    except Exception:
        pass

    if time.time() < deadline:
        x0[0] = _exact_c5(best_h) * n / 2.0 + 0.01
        x0[1:] = best_h
        try:
            res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter // 2, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = np.clip(res.x[1:], 0.0, 1.0)
                h_new = _project_feasible(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    return best_h, best_c5


def _subgradient_run(h0, deadline, n_iter=5000):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_raw = best_c5 * n / 2.0

    alpha_init = 0.02
    alpha = alpha_init
    gamma = 0.9997

    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = _all_corr(h)
        max_c = float(np.max(corr))
        if max_c < best_raw - 1e-12:
            best_raw = max_c
            best_c5 = max_c * 2.0 / n
            best_h = h.copy()

        # Find near-active set (within 2% of max)
        threshold = max_c * 0.98
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        # Compute subgradient: average gradient of active lags
        grad = np.zeros(n)
        for d_idx in active:
            lag = int(d_idx) - (n - 1)
            grad += _corr_lag_grad(h, lag)
        grad /= len(active)
        grad -= np.mean(grad)  # project out the equality constraint direction

        # Polyak-like step
        if best_raw < max_c:
            alpha = (max_c - best_raw) / (np.dot(grad, grad) + 1e-10)
            alpha = min(alpha, 0.1)
        alpha *= gamma

        h_new = h - alpha * grad
        h_new = _project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def _active_lag_cd(h0, deadline, max_passes=200):
    """Coordinate descent focusing on variables most affected by active lags."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        corr = _all_corr(h)
        max_c = float(np.max(corr))
        threshold = max_c * 0.97
        active = np.where(corr >= threshold)[0]

        # Find variables most involved in active lags
        involvement = np.zeros(n)
        for d_idx in active:
            lag = int(d_idx) - (n - 1)
            abs_lag = abs(lag)
            m = n - abs_lag
            if m > 0:
                involvement[:m] += 1.0
                involvement[abs_lag:] += 1.0

        # Sort by involvement, try top variables
        top_vars = np.argsort(-involvement)[:min(12, n)]

        improved = False
        for j in top_vars:
            if time.time() > deadline:
                break
            # Binary search for optimal h[j] value
            lo, hi_val = max(0.0, h[j] - 0.3), min(1.0, h[j] + 0.3)
            best_local = h[j]
            best_local_c5 = _exact_c5(h)
            for frac in np.linspace(0, 1, 15):
                trial = lo + frac * (hi_val - lo)
                h_old = h[j]
                h[j] = trial
                h = _project_feasible(h, target)
                c5_t = _exact_c5(h)
                if c5_t < best_local_c5 - 1e-13:
                    best_local_c5 = c5_t
                    best_local = h[j]
                h[j] = h_old
            if best_local != h[j]:
                h[j] = best_local
                h = _project_feasible(h, target)
                c5_new = _exact_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_feasible(h_new, n_new / 2.0)


def _pair_polish(h0, deadline, max_passes=80):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n - 1):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for (dj, dk) in ((1.0, -1.0), (-1.0, 1.0)):
                    if dj > 0:
                        a_max = 1.0 - h[j]
                    else:
                        a_max = h[j]
                    if dk > 0:
                        a_max = min(a_max, 1.0 - h[k])
                    else:
                        a_max = min(a_max, h[k])
                    if a_max < 1e-10:
                        continue
                    for frac in (0.5, 1.0, 0.25, 0.75):
                        a = frac * a_max
                        h[j] += dj * a
                        h[k] += dk * a
                        h = np.clip(h, 0.0, 1.0)
                        h = _project_feasible(h, target)
                        c5_t = _exact_c5(h)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h.copy()
                            improved = True
                            break
                        h[j] -= dj * a
                        h[k] -= dk * a
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _lp_direction(h, score, radius):
    n = len(h)
    K = 2 * n - 1
    # Build gradient matrix for all lags
    G = np.zeros((K, n))
    for d_idx in range(K):
        lag = d_idx - (n - 1)
        G[d_idx] = _corr_lag_grad(h, lag)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -score
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_polish(h0, deadline, radius_init=0.12, max_iters=5000, stall_limit=80):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        score = _all_corr(h)
        p = _lp_direction(h, score, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
            continue

        # Line search
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-9)

        cur_c5 = _exact_c5(h)
        best_a, best_c5_ls = 0.0, cur_c5
        for t in np.linspace(0, 1, 10):
            a = t * alpha_max
            if a < 1e-11:
                continue
            h_t = _project_feasible(h + a * p, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_ls:
                best_c5_ls, best_a = c5_t, a

        # Golden section refinement
        if best_a > 0:
            lo = max(0.0, best_a - alpha_max * 0.2)
            hi = best_a + alpha_max * 0.2
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(_project_feasible(h + c1 * p, target))
            f2 = _exact_c5(_project_feasible(h + c2 * p, target))
            for _ in range(12):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(_project_feasible(h + c1 * p, target))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _exact_c5(_project_feasible(h + c2 * p, target))
            a = (lo + hi) / 2.0
            h_t = _project_feasible(h + a * p, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_ls:
                best_c5_ls, best_a = c5_t, a

        if best_c5_ls < cur_c5 - 1e-13:
            h = _project_feasible(h + best_a * p, target)
            c5_new = _exact_c5(h)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.03:
                radius = 0.08

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project_feasible(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _remaining(frac):
        return t0 + budget_s * frac - time.time()

    # Phase 1: SLSQP from initial (10% budget)
    d1 = t0 + budget_s * 0.10
    for ak in [15, 25, 40]:
        if time.time() > d1:
            break
        h_s, c5_s = _slsqp_run(h, ak, maxiter=300, deadline=d1)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h = h_s

    # Phase 2: Subgradient (15% budget)
    d2 = t0 + budget_s * 0.25
    h_sg, c5_sg = _subgradient_run(h, d2, n_iter=8000)
    if c5_sg < best_c5:
        best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 3: LP polish (15% budget)
    d3 = t0 + budget_s * 0.40
    h_lp, c5_lp = _lp_polish(h, d3, radius_init=0.12, max_iters=4000, stall_limit=60)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: SLSQP again from improved point (10% budget)
    d4 = t0 + budget_s * 0.50
    h_s2, c5_s2 = _slsqp_run(best_h, 20, maxiter=400, deadline=d4)
    if c5_s2 < best_c5:
        best_c5, best_h = c5_s2, h_s2.copy()
    h_s3, c5_s3 = _slsqp_run(best_h, 35, maxiter=300, deadline=d4)
    if c5_s3 < best_c5:
        best_c5, best_h = c5_s3, h_s3.copy()

    # Phase 5: Multi-resolution search (25% budget)
    d5 = t0 + budget_s * 0.75
    resolutions = [n0 - 4, n0 - 2, n0 + 2, n0 + 4, n0 + 8, n0 - 8, n0 + 12, n0 + 16, n0 - 12, 48, 80, 96]
    for n_try in resolutions:
        if time.time() > d5:
            break
        if n_try < 24 or n_try > 200:
            continue
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        # Quick SLSQP
        sub_d = min(d5, time.time() + budget_s * 0.02)
        h_r, c5_r = _slsqp_run(h_r, 20, maxiter=200, deadline=sub_d)
        # Quick LP
        sub_d2 = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_polish(h_r, sub_d2, radius_init=0.1, max_iters=1000, stall_limit=40)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Random restarts with local optimization (15% budget)
    d6 = t0 + budget_s * 0.90
    scale = 0.03
    for _attempt in range(500):
        if time.time() > d6:
            break
        if best_c5 < 0.3805:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project_feasible(h_p, best_n / 2.0)
        # Quick subgradient
        sub_d = min(d6, time.time() + budget_s * 0.01)
        h_p, c5_p = _subgradient_run(h_p, sub_d, n_iter=300)
        if c5_p < best_c5 - 1e-11:
            # Follow with LP
            sub_d2 = min(d6, time.time() + budget_s * 0.015)
            h_p, c5_p = _lp_polish(h_p, sub_d2, radius_init=0.08, max_iters=800, stall_limit=50)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_p.copy()
                scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)

    # Phase 7: Active lag coordinate descent (5% budget)
    d7 = t0 + budget_s * 0.95
    h_cd, c5_cd = _active_lag_cd(best_h, d7, max_passes=100)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 8: Final pair polish + LP (5% budget)
    if time.time() < t0 + budget_s * 0.97:
        d8 = t0 + budget_s * 0.98
        h_pp, c5_pp = _pair_polish(best_h, d8, max_passes=60)
        if c5_pp < best_c5:
            best_c5, best_h = c5_pp, h_pp.copy()

    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_polish(best_h, t0 + budget_s * 0.995,
                                radius_init=0.03, max_iters=2000, stall_limit=100)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()
            best_n = len(best_h)

    best_h = _project_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
