    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx

    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    return tau * np.log(Z) + max_s

def _soft_obj_grad(h, tau):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx

    Z = float(np.sum(exp_s))
    w = exp_s / Z
    obj = tau * np.log(Z) + max_s
    grad = np.zeros(n)

        if wk < 1e-15:

        d = k - (n - 1)

        else:
            g = -d
            m = n - g
            grad[:m] += wk * (1.0 - h[g:])
            grad[g:] += wk * (-h[:m])
    return obj, grad

    _, g = _soft_obj_grad(x, _g_tau)

    # Phase 2: Gradient descent with momentum (10% budget)
    d2 = t0 + budget_s * 0.40
    h_p2, c_p2 = _grad_descent_phase(h_best, d2, lr_init=0.015, tau=0.08)
    if c_p2 < c_best:
        c_best = c_p2
        h_best = h_p2.copy()

    # Phase 3: Another SLSQP round on best (10% budget)
    d3 = t0 + budget_s * 0.50
    h_p3, c_p3 = _run_slqp_phase(h_best, d3, 200)
    if c_p3 < c_best:
        c_best = c_p3
        h_best = h_p3.copy()

    # Phase 4: Multi-resolution search (20% budget)
    d4 = t0 + budget_s * 0.70
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(h_best, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c_r = _run_slqp_phase(h_r, sub_dl, 150)
        if c_r < c_best - 1e-12:
            c_best = c_r
            h_best = h_r.copy()
            n_best = n_try

    # Phase 5: Pair coordinate descent (25% budget)
    d5 = t0 + budget_s * 0.95
    h_p5, c_p5 = _pair_descent(h_best, d5, max_passes=800)
    if c_p5 < c_best:
        c_best = c_p5
        h_best = h_p5.copy()

    # Phase 6: Perturbation + SLSQP restarts (remaining time)
    if time.time() < hard_deadline - 10:
        d6 = hard_deadline - 5
        scale = 0.03
        for _ in range(60):
            if time.time() > d6:
                break
            h_p = h_best + rng.normal(0, scale, n_best)
            h_p = _project(h_p, n_best / 2.0)
            sub_dl = min(d6, time.time() + budget_s * 0.015)
            h_try, c_try = _run_slqp_phase(h_p, sub_dl, 100)
            if c_try < c_best - 1e-11:
                c_best = c_try
                h_best = h_try.copy()
                scale = max(0.008, scale * 0.85)
            else:
                scale = min(0.15, scale * 1.2)

    # Phase 7: Final pair polish
    if time.time() < hard_deadline - 3:
        h_p7, c_p7 = _pair_descent(h_best, hard_deadline - 1, max_passes=200)
        if c_p7 < c_best:
            c_best = c_p7
            h_best = h_p7.copy()

    # Final validation
    h_best = _project(h_best, n_best / 2.0)
    final_c = _exact_c5(h_best)
    if final_c > c_best + 1e-9:
        h_best = _project(h0.copy(), target0)
        final_c = _exact_c5(h_best)
        c_best = final_c
        n_best = n0

    return h_best, final_c, n_best
