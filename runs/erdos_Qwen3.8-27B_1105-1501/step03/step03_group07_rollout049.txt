import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_values(h):
    """Compute all raw correlation values (before multiplying by dx)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_gradient(h, d):
    """Gradient of F_d w.r.t. h where d is signed offset"""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:d+m])
            grad[d:d+m] -= h[:m]
    else:
        k = -d
        m = n - k
        if m > 0:
            grad[k:k+m] += (1.0 - h[:m])
            grad[:m] -= h[k:k+m]
    return grad

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, target, R, lam, top_k=None, deadline=None):
    """Find LP tangent direction. Returns direction d or None."""
    n = len(h)
    raw = _raw_values(h)
    raw_max = float(np.max(raw))
    
    # Select which lags to constrain
    if top_k is not None and top_k < 2*n-1:
        # Use top_k lags (those closest to max)
        threshold = np.sort(raw)[-top_k]
        mask = raw >= threshold - 1e-12
    else:
        mask = np.ones(2*n-1, dtype=bool)
    
    active_indices = np.where(mask)[0]
    n_active = len(active_indices)
    
    if n_active == 0:
        return None
    
    # Build LP
    # Variables: [t, p_0, q_0, p_1, q_1, ..., p_{n-1}, q_{n-1}]
    nvars = 1 + 2*n
    
    # Objective: minimize t + lam * sum(p_i + q_i)
    c = np.zeros(nvars)
    c[0] = 1.0
    c[1:] = lam
    
    # Inequality constraints: A_ub @ x <= b_ub
    rows = []
    rhs = []
    
    for idx in active_indices:
        d = int(idx) - (n - 1)
        F_s = raw[idx]
        g = _lag_gradient(h, d)
        row = np.zeros(nvars)
        row[0] = -1.0  # -t
        for i in range(n):
            row[1 + 2*i] = g[i]       # +g_i * p_i
            row[2 + 2*i] = -g[i]      # -g_i * q_i
        rows.append(row)
        rhs.append(-F_s)
    
    # Norm constraint: sum(p_i + q_i) <= R
    row = np.zeros(nvars)
    row[1:] = 1.0
    rows.append(row)
    rhs.append(R)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(p_i - q_i) = 0
    A_eq = np.zeros((1, nvars))
    for i in range(n):
        A_eq[0, 1 + 2*i] = 1.0
        A_eq[0, 2 + 2*i] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds
    eps = 1e-10
    bounds = [(-1000.0, 1000.0)]  # t is free
    for i in range(n):
        if h[i] <= eps:
            bounds.append((0.0, None))   # p_i free (can increase)
            bounds.append((0.0, 0.0))    # q_i = 0 (can't decrease)
        elif h[i] >= 1.0 - eps:
            bounds.append((0.0, 0.0))    # p_i = 0 (can't increase)
            bounds.append((0.0, None))   # q_i free (can decrease)
        else:
            bounds.append((0.0, None))   # p_i free
            bounds.append((0.0, None))   # q_i free
    
    try:
        if deadline is not None and time.time() > deadline:
            return None
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if not res.success:
            return None
        x = res.x
        d = np.zeros(n)
        for i in range(n):
            d[i] = x[1 + 2*i] - x[2 + 2*i]
        d -= np.mean(d)  # ensure sum is zero (numerical correction)
        dnorm = float(np.max(np.abs(d)))
        if dnorm < 1e-12:
            return None
        return d / dnorm
    except Exception:
        return None

def _line_search(h, direction, target, current_c5, max_alpha=1.0):
    """Line search along direction. Returns (best_alpha, best_c5, best_h)."""
    n = len(h)
    # Find max alpha
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), 5.0)
    
    best_a = 0.0
    best_c = current_c5
    best_h = h.copy()
    
    # Coarse grid
    alphas = np.linspace(0.01, alpha_max, 15)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
            best_h = h_t
    
    if best_a == 0.0:
        return 0.0, current_c5, h
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.7)
    hi = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h_c1 = _project(h + c1 * direction, target)
    h_c2 = _project(h + c2 * direction, target)
    f1 = _exact_c5(h_c1)
    f2 = _exact_c5(h_c2)
    for _ in range(20):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            h_c1 = _project(h + c1 * direction, target)
            f1 = _exact_c5(h_c1)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            h_c2 = _project(h + c2 * direction, target)
            f2 = _exact_c5(h_c2)
    
    opt_a = (lo + hi) / 2.0
    h_opt = _project(h + opt_a * direction, target)
    opt_c = _exact_c5(h_opt)
    if opt_c < best_c:
        return opt_a, opt_c, h_opt
    return best_a, best_c, best_h

def _lp_optimize(h, deadline, target, max_iters=2000):

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += 1.0 - h[d:d+m]
            g[d:d+m] -= h[:m]
    else:
        k = -d
        m = n - k
        if m > 0:
            g[k:k+m] += 1.0 - h[:m]
            g[:m] -= h[k:k+m]
    return g

def _proj(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_dir(h, R, lam, topk, deadline):
    n = len(h)
    raw = _raw(h)
    nw = 2 * n - 1
    if topk is not None and topk < nw:
        thr = np.partition(raw, nw - topk)[nw - topk]
        idx = np.where(raw >= thr - 1e-12)[0]
    else:
        idx = np.arange(nw)
    if len(idx) == 0:
        return None
    nv = 1 + 2 * n
    c = np.full(nv, lam)
    c[0] = 1.0
    A_rows = []
    b_vals = []
    for r in idx:
        d = int(r) - (n - 1)
        g = _grad_lag(h, d)
        row = np.zeros(nv)
        row[0] = -1.0
        row[1::2] = g
        row[2::2] = -g
        A_rows.append(row)
        b_vals.append(-raw[r])
    row = np.zeros(nv)
    row[1:] = 1.0
    A_rows.append(row)
    b_vals.append(R)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, nv))
    A_eq[0, 1::2] = 1.0
    A_eq[0, 2::2] = -1.0
    b_eq = np.array([0.0])
    eps = 1e-10
    bounds = [(-1000.0, 1000.0)]
    for i in range(n):
        if h[i] <= eps:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif h[i] >= 1.0 - eps:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))
    try:
        if deadline is not None and time.time() > deadline:
            return None
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if not res.success:
            return None
        x = res.x
        d = x[1::2] - x[2::2]
        d -= np.mean(d)
        mx = float(np.max(np.abs(d)))
        if mx < 1e-12:
            return None
        return d / mx
    except Exception:
        return None

def _ls(h, d, target, c_cur, a_max_cap):
    n = len(h)
    amax = a_max_cap
    for j in range(n):
        if d[j] > 1e-14:
            amax = min(amax, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            amax = min(amax, h[j] / (-d[j]))
    amax = min(max(amax, 1e-7), 5.0)
    best_a = 0.0
    best_c = c_cur
    best_h = h.copy()
    alphas = np.concatenate([np.linspace(0.01, amax, 10),
                             np.logspace(np.log10(max(0.005, amax*0.1)),
                                         np.log10(amax), 8)])
    alphas = sorted(set(np.round(np.clip(alphas, 1e-6, amax), 10)))
    for a in alphas:
        ht = _proj(h + a * d, target)
        ct = _exact_c5(ht)
        if ct < best_c - 1e-14:
            best_c = ct
            best_a = a
            best_h = ht
    if best_a == 0.0:
        return 0.0, c_cur, h
    lo = max(0.0, best_a * 0.65)
    hi = min(amax, best_a * 1.35)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_proj(h + c1 * d, target))
    f2 = _exact_c5(_proj(h + c2 * d, target))
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_proj(h + c1 * d, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_proj(h + c2 * d, target))
    oa = (lo + hi) * 0.5
    oh = _proj(h + oa * d, target)
    oc = _exact_c5(oh)
    if oc < best_c:
        return oa, oc, oh
    return best_a, best_c, best_h

def _lp_opt(h, deadline, target, max_iters=5000):
    n = len(h)
    h = h.copy()
    ccur = _exact_c5(h)
    bh = h.copy()
    bc = ccur
    stall = 0
    R_list = [0.08 * n, 0.2 * n, 0.5 * n]
    lam_list = [1e-4, 5e-4, 2e-3, 0.01]
    topk_list = [None, None, max(15, (2*n-1)//4), max(25, (2*n-1)//3)]

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        improved = False
        for R in R_list:
            if time.time() > deadline:
                break
            for lam in lam_list:
                if time.time() > deadline:
                    break
                for tk in topk_list:
                    d = _lp_dir(h, R, lam, tk, deadline)
                    if d is None:
                        continue
                    a, nc, nh = _ls(h, d, target, ccur, R / (n + 1))
                    if a > 0 and nc < bc - 1e-14:
                        h, ccur = nh, nc
                        bh, bc = h.copy(), nc
                        stall = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall % 15 == 0:
                rng = np.random.default_rng(it * 31 + 7)
                pert = 0.015 * (stall // 15)
                h_try = _proj(h + rng.normal(0, pert, n), target)
                ct = _exact_c5(h_try)
                if ct < bc - 1e-14:
                    h, ccur, bh, bc = h_try, ct, h_try.copy(), ct
                    stall = 0

    return bh, bc

def _pair_search(h, deadline, target, max_passes=80):
    n = len(h)
    h = h.copy()
    ccur = _exact_c5(h)
    bh = h.copy()
    bc = ccur
    rng = np.random.default_rng(42)
    for _p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        raw = _raw(h)
        top_lags = np.argsort(-raw)[:8]
        for r in top_lags:
            d = int(r) - (n - 1)
            g = _grad_lag(h, d)
            g -= np.mean(g)
            gn = float(np.linalg.norm(g))
            if gn < 1e-10:
                continue
            g /= gn
            for sign in [1.0, -1.0]:
                a, nc, nh = _ls(h, sign * g, target, ccur, 0.3 * n)
                if a > 0 and nc < bc - 1e-14:
                    h, ccur = nh, nc
                    bh, bc = h.copy(), nc
                    improved = True
        if not improved:
            for _ in range(300):
                if time.time() > deadline:
                    return bh, bc
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] <= 1e-8 or h[j] >= 1 - 1e-8:
                    continue
                if h[k] <= 1e-8 or h[k] >= 1 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    aj = (1 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                    ak = (1 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                    amax = min(aj, ak)
                    if amax < 1e-10:
                        continue
                    for fr in [0.3, 0.6, 1.0]:
                        a = amax * fr
                        ht = h.copy()
                        ht[j] += dj * a
                        ht[k] += dk * a
                        ct = _exact_c5(ht)
                        if ct < bc - 1e-13:
                            bh, bc = ht.copy(), ct
                            h, ccur = ht, ct
                            improved = True
                            break
            if not improved:
                break
    return bh, bc

def _resample(h, n2):
    n = len(h)
    if n2 == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n2) + 0.5) / n2
    h2 = np.interp(x_new, x_old, h)
    return _proj(h2, n2 / 2.0)

def _multi_start(h0, deadline, target, rng, n_starts=20):
    n = len(h0)
    bh = h0.copy()
    bc = _exact_c5(h0)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 5.0:
            break
        alloc = min(rem * 0.85 / max(1, n_starts - s), 30.0)
        sub_dl = time.time() + alloc
        r = rng.random()
        if r < 0.25:
            ht = h0 + rng.normal(0, 0.04, n)
        elif r < 0.4:
            ht = h0.copy()
            idx = rng.choice(n, size=n // 6, replace=False)
            ht[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.55:
            ht = (h0 + rng.normal(0, 0.06, n) > 0.5).astype(float)
        elif r < 0.7:
            sh = rng.integers(1, max(2, n // 6))
            ht = np.roll(h0, sh).copy()
        elif r < 0.85:
            ht = 0.5 + (h0 - 0.5) * (0.7 + 0.6 * rng.random())
        else:
            hr = (rng.random(n) > 0.5).astype(float)
            ht = 0.5 * h0 + 0.5 * hr
        ht = _proj(ht, target)
        ho, co = _lp_opt(ht, sub_dl, target, max_iters=2000)
        if co < bc - 1e-14:
            bc, bh = co, ho
    return bh, bc
