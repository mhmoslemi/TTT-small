def smoothed_direction(h, beta=50.0):
    n = len(h)
    scores = all_scores(h)
    max_s = float(np.max(scores))
    # Softmax weights
    A = beta * (scores - max_s)
    A = A - np.max(A)  # shift for numerical stability (already 0 at max)
    W = np.exp(A)
    W = W / float(np.sum(W))
    
    # Compute weighted gradient
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if W[r] > 1e-15:
            d = r - (n - 1)
            g += W[r] * score_grad(h, d)
    
    # Project to zero-sum subspace
    g -= np.mean(g)
    
    # Normalize
    norm = float(np.sqrt(np.sum(g**2)))
    if norm < 1e-15:
        return None
    return -g / norm  # negative gradient = descent direction

def line_search(h, direction, c5_cur, deadline):
    n = len(h)
    # Find max step
    alpha_max = 1.0
    for i in range(n):
        if direction[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / direction[i])
        elif direction[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-direction[i]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Coarse search
    best_alpha = 0.0
    best_c5 = c5_cur
    n_evals = 20
    for i in range(n_evals):
        a = alpha_max * (i + 1) / n_evals
        h_t = h + a * direction
        c5_t = exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_alpha = a
        if time.time() > deadline:
            return best_alpha, best_c5
    
    # Golden section refinement around best_alpha
    if best_alpha == 0.0:
        return 0.0, c5_cur
    
    lo = max(0.0, best_alpha * 0.8)
    hi = min(alpha_max, best_alpha * 1.2)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * direction)
    f2 = exact_c5(h + c2 * direction)
    
    for _ in range(30):
        if hi - lo < 1e-10 or time.time() > deadline:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * direction)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * direction)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * direction)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_alpha, best_c5

def pair_transfer_search(h, target, deadline, max_pairs=100):

        if W[r] > 1e-15:
            d = r - (n - 1)
            g += W[r] * score_grad(h, d)
    
    # Coordinates where we want to increase (g < 0) and decrease (g > 0)
    inc_mask = g < -1e-10  # want to increase h
    dec_mask = g > 1e-10  # want to decrease h
    
    rng = np.random.default_rng(42)
    
    for it in range(max_pairs):
        if time.time() > deadline:
            break
        
        if np.sum(inc_mask) == 0 or np.sum(dec_mask) == 0:
            break
        
        # Pick random pairs
        inc_idx = np.where(inc_mask)[0]
        dec_idx = np.where(dec_mask)[0]
        
        i = dec_idx[rng.integers(len(dec_idx))]  # source (decrease)
        j = inc_idx[rng.integers(len(inc_idx))]  # target (increase)
        
        if h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
            continue
        
        # Direction: decrease h[i], increase h[j]
        direction = np.zeros(n)
        direction[i] = -1.0
        direction[j] = 1.0
        
        alpha_max = min(h[i], 1.0 - h[j])
        if alpha_max < 1e-12:
            continue
        
        # Line search
        best_a = 0.0
        best_c5_local = c5_cur
        for frac in [0.25, 0.5, 0.75, 1.0]:
            a = alpha_max * frac
            h_t = h + a * direction
            c5_t = exact_c5(h_t)
            if c5_t < best_c5_local:
                best_c5_local = c5_t
                best_a = a
            if time.time() > deadline:
                break
        
        if best_a > 0 and best_c5_local < best_c5 - 1e-14:
            h = h + best_a * direction
            best_h = h.copy()
            best_c5 = best_c5_local
            c5_cur = best_c5
            # Update gradient (expensive, do every few iterations)
            if it % 10 == 0:
                scores = all_scores(h)
                max_s = float(np.max(scores))
                A = beta * (scores - max_s)
                A = A - np.max(A)

                        d = r - (n - 1)
                        g += W[r] * score_grad(h, d)
                inc_mask = g < -1e-10
                dec_mask = g > 1e-10
        else:
            # Try a few random pairs
            pass
    
    return best_h, best_c5
