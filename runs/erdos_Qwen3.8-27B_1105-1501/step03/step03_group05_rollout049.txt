import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Water-filling algorithm
    for _ in range(100):
        resid = target - np.sum(v)
        if abs(resid) < 1e-12:
            break
        if resid > 0:
            mask = v < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = resid / cnt
            v[mask] = np.minimum(1.0, v[mask] + step)
        else:
            mask = v > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -resid / cnt
            v[mask] = np.maximum(0.0, v[mask] - step)
    # Final exact fix
    resid = target - np.sum(v)
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += resid / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _build_gradient_matrix(h):
    """Build the gradient matrix G where G[r, :] is d/dh of scaled score at lag index r."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    F = np.zeros(L)  # current scaled scores
    raw = np.correlate(h, 1.0 - h, mode='full')
    F = raw * dx
    
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return F, G

def _cvt_trust_step(h, rho, solver='CLARABEL', max_solve_time=30):
    """Solve the CVXPY trust-region SOCP step."""
    if not HAS_CVXPY:
        return None
    n = len(h)
    target = n / 2.0
    F, G = _build_gradient_matrix(h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h) <= rho,
    ]
    for r in range(2*n - 1):
        constraints.append(F[r] + G[r, :] @ (x - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    for s in ['CLARABEL', 'SCS', 'ECOS', 'OSQP', 'MOSEK', 'GUROBI', 'CBC', 'GLPK']:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    t_start = time.time()
    for s in solvers_to_try:
        if time.time() - t_start > max_solve_time:
            break
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
                return np.array(x.value, dtype=float)
        except Exception:
            continue
    return None

def _accept_step(h, candidate, cur_c5, tol=1e-10):
    """Try to accept a candidate step. Returns (accepted_h, accepted_c5) or (None, cur_c5)."""
    n = len(h)
    target = n / 2.0
    # Project candidate
    x = _project_box_sum(candidate, target)
    c5 = _exact_c5(x)
    if c5 < cur_c5 - tol:
        return x, c5
    # Try damped steps
    for frac in [0.7, 0.5, 0.3, 0.15, 0.05]:
        x_damped = _project_box_sum(h + frac * (candidate - h), target)
        c5_d = _exact_c5(x_damped)
        if c5_d < cur_c5 - tol:
            return x_damped, c5_d
    return None, cur_c5

def _cvx_main_loop(h, deadline, sigma_start=0.05, sigma_max=0.20, sigma_min=0.003, 
                   max_fails=15, max_iters=200):
    """Main CVXPY trust-region loop."""
    if not HAS_CVXPY:
        return h, _exact_c5(h)
    
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sigma = sigma_start
    fails = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if fails > max_fails:
            break
        
        rho = sigma * math.sqrt(n)
        candidate = _cvt_trust_step(h, rho)

            sigma *= 0.7

def _lp_minimax_step(h, radius=None):
    """LP-based minimax step (no trust region, just box+sum+optional radius via bounds)."""
    n = len(h)
    target = n / 2.0
    F, G = _build_gradient_matrix(h)
    L = 2 * n - 1
    
    # Variable: x (n) and t (1)
    # minimize t
    # s.t. G[r,:] @ x <= t - F[r] + G[r,:] @ h  =>  G[r,:] @ x - t <= -F[r] + G[r,:] @ h
    #      0 <= x <= 1
    #      sum(x) = target
    #      (optional: |x_i - h_i| <= radius for all i)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: G[r,:] @ x - t <= -F[r] + G[r,:] @ h
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -F + G @ h
    
    # Box constraints as inequalities
    # x_i <= 1: row of A_ub
    # x_i >= 0: handled by bounds
    
    # Equality: sum(x) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    if radius is not None:
        bounds = [(max(0.0, h[i] - radius), min(1.0, h[i] + radius)) for i in range(n)]
    else:
        bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 10000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_gradient_descent(h, deadline, beta_start=5.0, beta_end=200.0, n_beta=10):
    """Smooth gradient descent with continuation in beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    beta_list = np.geomspace(beta_start, beta_end, n_beta)
    for beta in beta_list:
        if time.time() > deadline:
            break
        stall = 0
        step_size = 0.05
        for step in range(500):
            if time.time() > deadline:
                break
            if stall > 100:
                break
            
            scores = _scores_raw(h)
            max_s = np.max(scores)
            exp_vals = np.exp(beta * (scores - max_s))
            w_sum = np.sum(exp_vals)
            if w_sum < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w = exp_vals / w_sum
            
            # Compute gradient of weighted score
            grad = np.zeros(n)
            for d in range(-(n-1), n):
                wd = w[n-1+d]
                if abs(wd) < 1e-14:
                    continue
                low = max(0, -d)
                high = min(n, n-d)
                for p in range(low, high):
                    q = p + d
                    grad[q] += wd * (1.0 - h[p])
                    grad[p] -= wd * h[q]
            
            grad -= np.mean(grad)
            norm = np.max(np.abs(grad))
            if norm < 1e-14:
                break
            
            d = -grad / norm * step_size
            # Line search
            best_a = 0.0
            best_val = cur_c5
            for a in [0.1, 0.3, 0.5, 0.7, 1.0]:
                h_t = _project_box_sum(h + a * d, target)
                c5_t = _exact_c5(h_t)
                if c5_t < best_val - 1e-12:
                    best_val = c5_t
                    best_a = a
            
            if best_a > 0:
                h = _project_box_sum(h + best_a * d, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step_size = min(step_size * 1.1, 0.3)
            else:
                stall += 1
                step_size *= 0.7
    
    return best_h, best_c5

def _resample_integral(h, n_new):
    """Resample h to n_new points using integral averaging."""
    n = len(h)
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        mass = 0.0
        for i in range(n):
            old_start = i * dx_old
            old_end = (i + 1) * dx_old
            overlap = max(0.0, min(x_end, old_end) - max(x_start, old_start))
            if overlap > 0:
                mass += h[i] * overlap
        h_new[j] = mass / dx_new
        h_new[j] = min(max(h_new[j], 0.0), 1.0)
    
    return _project_box_sum(h_new, n_new / 2.0)

def _random_transfer_search(h, deadline, rng, max_iters=20000):
    """Random pairwise transfer local search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        
        # Pick random pair and transfer
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Transfer from higher to lower
        if h[i] < h[j]:
            i, j = j, i
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer < 1e-8:
            continue
        delta = rng.uniform(0.01, max_transfer)
        
        h[i] -= delta
        h[j] += delta
        c5_new = _exact_c5(h)
        
        if c5_new < cur_c5 - 1e-12:
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            h[i] += delta
            h[j] -= delta
            no_improve += 1
    
    return best_h, best_c5

def _binary_search(h, deadline, rng, max_iters=50000):
    """Local search on binary patterns."""
    n = len(h)
    target = n / 2.0
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:int(target)]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    no_improve = 0
    
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 5000:
            break
        if not ones or not zeros:
            break
        
        best_swap = None
        best_val = cur_c5
        for _b in range(20):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5
