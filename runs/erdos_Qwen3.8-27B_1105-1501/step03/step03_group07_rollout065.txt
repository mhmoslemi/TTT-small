        k = rng.choice(tuple(zeros)) if len(zeros) > 1 else 0

        new_mc = _max_count_binary(new_b, n)
        evals += 1

            b = new_b
            cur_mc = new_mc
            # Update sets
            ones.discard(j)
            ones.add(k)
            zeros.discard(k)
            zeros.add(j)
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_b = b
        elif new_mc > cur_mc:
            # Reject (or accept with small probability for SA)
            if rng.random() < np.exp(-(new_mc - cur_mc) / max(T, 1e-10)):
                b = new_b
                cur_mc = new_mc
                ones.discard(j)
                ones.add(k)
                zeros.discard(k)
                zeros.add(j)
                if cur_mc < best_mc:
                    best_mc = cur_mc
                    best_b = b

            cur_mc = best_mc
            ones = set(i for i in range(n) if (b >> i) & 1)

    result = np.zeros(n, dtype=float)
    for i in range(n):

def _continuous_optimize(h, deadline, target, max_iters=10000):
    """Smooth max optimization with projected gradient"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.05
    beta = 0.5
    T = 0.3
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
            
        # Compute scores
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(corr))
        
        # Softmax weights
        shifted = corr - max_s
        w = np.exp(shifted / T)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            break
        w /= wsum
        
        # Gradient
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-8:
                continue
            d = r - (n - 1)
            m = n - abs(d)
            if m <= 0:
                continue
            if d >= 0:
                grad[d:] += w[r] * (1.0 - h[:m])
                grad[:m] -= w[r] * h[d:]
            else:
                l = -d
                grad[:m] += w[r] * (1.0 - h[l:])
                grad[l:] -= w[r] * h[:m]
        
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        grad /= gnorm
        
        # Momentum
        v = beta * v - alpha * grad
        v -= np.mean(v)
        
        # Line search
        best_a = 0.0
        best_c = cur_c5
        for a in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            h_t = _project(h + a * v, target)
            c_t = _exact_c5(h_t)
            if c_t < best_c:
                best_c = c_t
                best_a = a
        
        if best_a > 0 and best_c < cur_c5 - 1e-14:
            h = _project(h + best_a * v, target)
            cur_c5 = _exact_c5(h)
            T = max(0.05, T * 0.99)
            beta = min(0.9, beta + 0.002)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            alpha *= 0.8
            T = max(0.05, T * 0.9)
    
    return best_h, best_c5

def _fine_polish(h, deadline, target):

    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    for _ in range(2000):
        if time.time() > deadline:
            break
        improved = False
        
        # Try random pair moves
        for _ in range(50):
            if time.time() > deadline:
                return best_h, best_c5
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j == k:
                continue
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                a_max = min(
                    (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                    (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                )
                if a_max < 1e-10:
                    continue
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved = True
                        break
        
        if not improved:
            # Try single coordinate moves (with constraint repair)
            for _ in range(10):
                j = rng.integers(0, n)
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                    h_t = h.copy()
                    h_t[j] += delta
                    h_t = _project(h_t, target)
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def _thue_morse(n):
    """Generate Thue-Morse sequence of length n (padded if n not power of 2)"""
    # Build up to next power of 2
    k = 1
    while k < n:
        k *= 2
    tm = np.zeros(k, dtype=np.int32)
    # thue-morse: tm[2i] = tm[i], tm[2i+1] = 1 - tm[i]
    for i in range(1, k // 2 + 1):
        tm[2*i] = tm[i]
        tm[2*i+1] = 1 - tm[i]
    # Actually let me do it properly
    tm = np.zeros(k, dtype=np.int32)
    tm[0] = 0
    for i in range(1, k):
        tm[i] = bin(i).count('1') % 2
    result = tm[:n].copy()
    # Adjust to have exactly n//2 ones
    weight = n // 2
    cnt = int(np.sum(result))
    if cnt > weight:
        # Flip some 1s to 0s
        ones = np.where(result == 1)[0]
        for i in range(cnt - weight):
            result[ones[i]] = 0
    elif cnt < weight:
        zeros = np.where(result == 0)[0]
        for i in range(weight - cnt):
            result[zeros[i]] = 1
    return result.astype(float)
