def _min_max_lines(a, b, t_lo=0.0, t_hi=1.0):
    """Find t in [t_lo, t_hi] minimizing max_k (a[k]*t + b[k])"""
    m = len(a)
    # Collect candidates
    candidates = [t_lo, t_hi]
    
    # Pairwise crossings
    A_diff = a[:, None] - a[None, :]  # shape (m, m)
    B_diff = b[None, :] - b[:, None]  # shape (m, m)
    
    # Only consider upper triangle to avoid duplicates
    mask = np.triu(np.ones((m, m), dtype=bool), k=1)
    valid = mask & (np.abs(A_diff) > 1e-14)
    
    if np.any(valid):
        T = np.zeros((m, m))
        T[valid] = B_diff[valid] / A_diff[valid]
        T_in = valid & (T >= t_lo) & (T <= t_hi)
        if np.any(T_in):
            T_vals = T[T_in]
            # Evaluate max_k (a[k]*t + b[k]) for each t
            # Shape: (num_valid, m)
            vals = a[None, :] * T_vals[:, None] + b[None, :]
            f_vals = vals.max(axis=1)
            # Also need to include the t_lo and t_hi evaluations
            candidates.extend(T_vals[f_vals == f_vals.min()].tolist())
    
    # Evaluate at all candidates
    all_candidates = np.array(candidates)
    all_candidates = np.unique(all_candidates[(all_candidates >= t_lo) & (all_candidates <= t_hi)])
    
    best_t = t_lo
    best_f = np.inf
    for t in all_candidates:
        f = np.max(a * t + b)
        if f < best_f:
            best_f = f
            best_t = t
    return best_t, best_f

def _min_max_lines(a, b, t_lo=0.0, t_hi=1.0):
    m = len(a)
    # All pairwise crossings
    A_diff = a[:, None] - a[None, :]
    B_diff = b[None, :] - b[:, None]
    
    mask = np.triu(np.ones((m, m), dtype=bool), k=1) & (np.abs(A_diff) > 1e-14)
    
    T = np.full((m, m), np.inf)
    T[mask] = B_diff[mask] / A_diff[mask]
    T_in_mask = mask & (T >= t_lo) & (T <= t_hi)
    
    # Collect all candidate t values
    T_flat = T[T_in_mask]
    candidates = np.concatenate([[t_lo, t_hi], T_flat])
    candidates = candidates[(candidates >= t_lo) & (candidates <= t_hi)]
    
    # Evaluate f(t) = max_k (a[k]*t + b[k]) for all candidates at once
    # candidates shape: (C,), a shape: (m,)
    # result shape: (C, m)
    vals = a[None, :] * candidates[:, None] + b[None, :]
    f_vals = vals.max(axis=1)
    
    idx = np.argmin(f_vals)
    return candidates[idx], f_vals[idx]

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _coeffs_for_coord(h, i):
    n = len(h)
    a = np.zeros(2 * n - 1)
    
    # Positive and zero lags: index n-1+d for d = 0, ..., n-1
    d = np.arange(n)
    m1 = d <= i
    t1 = np.zeros(n)
    if np.any(m1):
        t1[m1] = 1.0 - h[i - d[m1]]
    m2 = (i + d) < n
    t2 = np.zeros(n)
    if np.any(m2):
        t2[m2] = -h[i + d[m2]]
    a[n-1:] = t1 + t2
    
    # Negative lags: index n-1-g for g = 1, ..., n-1
    g = np.arange(1, n)
    idx = n - 1 - g
    m1 = (i + g) < n
    t1 = np.zeros(n-1)
    if np.any(m1):
        t1[m1] = 1.0 - h[i + g[m1]]
    m2 = i >= g
    t2 = np.zeros(n-1)
    if np.any(m2):
        t2[m2] = -h[i - g[m2]]
    a[idx] = t1 + t2
    
    return a

def _min_max_lines(a, b, t_lo=0.0, t_hi=1.0):
    """Find t in [t_lo, t_hi] minimizing max_k (a[k]*t + b[k])"""
    m = len(a)
    
    # Compute all pairwise crossings
    # t_cross(i,j) = (b[j] - b[i]) / (a[i] - a[j])
    A_diff = a[:, None] - a[None, :]  # (m, m)
    B_diff = b[None, :] - b[:, None]  # (m, m)
    
    # Only upper triangle
    mask = np.triu(np.ones((m, m), dtype=bool), k=1)
    valid = mask & (np.abs(A_diff) > 1e-12)
    
    candidates = np.array([t_lo, t_hi])
    
    if np.any(valid):
        T = np.empty(np.sum(valid))
        T_vals = B_diff[valid] / A_diff[valid]
        T_in = (T_vals >= t_lo) & (T_vals <= t_hi)
        if np.any(T_in):
            candidates = np.concatenate([candidates, T_vals[T_in]])
    
    # Evaluate f(t) = max(a*t + b) at all candidates
    # candidates: (C,), a: (m,), b: (m,)
    # result: (C,)
    vals = a[None, :] * candidates[:, None] + b[None, :]  # (C, m)
    f_vals = vals.max(axis=1)  # (C,)
    
    idx = np.argmin(f_vals)
    return float(candidates[idx]), float(f_vals[idx])

def _cd_coordinate(h, i):
    """Optimize coordinate i of h exactly. Returns new value for h[i]."""
    n = len(h)
    a = _coeffs_for_coord(h, i)
    scores = _all_scores(h)
    b = scores - a * h[i]
    t_opt, _ = _min_max_lines(a, b, 0.0, 1.0)
    return t_opt

def _cd_pair(h, i, j):
    """Optimize pair (i,j) of h exactly (preserving h[i]+h[j]). Returns new (h[i], h[j])."""
    n = len(h)
    s = h[i] + h[j]
    a_i = _coeffs_for_coord(h, i)
    a_j = _coeffs_for_coord(h, j)
    a_prime = a_i - a_j
    scores = _all_scores(h)
    b_prime = scores - a_prime * h[i]
    
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    t_opt, _ = _min_max_lines(a_prime, b_prime, t_lo, t_hi)
    return t_opt, s - t_opt

def _cd_sweep(h, max_sweeps=10, rng=None):
    n = len(h)
    h = h.copy()
    target = n / 2.0
    
    for sweep in range(max_sweeps):
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        
        for i in order:
            t_opt = _cd_coordinate(h, int(i))
            if abs(t_opt - h[i]) > 1e-10:
                h[i] = t_opt
                improved = True
        
        if not improved:
            break
    
    # Project to ensure exact sum
    h = _project(h, target)
    return h, _exact_c5(h)

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coeffs_vector(h):
    """Compute coefficient matrix: shape (n, 2n-1), where coeffs[i,d] = d(Q_d)/d(h_i)"""
    n = len(h)
    coeffs = np.zeros((n, 2 * n - 1))
    for i in range(n):
        # d >= 0: index n-1+d, d=0,...,n-1
        d = np.arange(n)
        m1 = d <= i
        t1 = np.zeros(n)
        if np.any(m1):
            t1[m1] = 1.0 - h[i - d[m1]]
        m2 = (i + d) < n
        t2 = np.zeros(n)
        if np.any(m2):
            t2[m2] = -h[i + d[m2]]
        coeffs[i, n - 1:] = t1 + t2
        # d < 0: g=1,...,n-1, index n-1-g
        g = np.arange(1, n)
        idx = n - 1 - g
        m1 = (i + g) < n
        t1 = np.zeros(n - 1)
        if np.any(m1):
            t1[m1] = 1.0 - h[i + g[m1]]
        m2 = i >= g
        t2 = np.zeros(n - 1)
        if np.any(m2):
            t2[m2] = -h[i - g[m2]]
        coeffs[i, idx] = t1 + t2
    return coeffs

def _min_max_lines(a, b, t_lo=0.0, t_hi=1.0):
    m = len(a)
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(a * t_lo + b))
    A_diff = a[:, None] - a[None, :]
    B_diff = b[None, :] - b[:, None]
    mask = np.triu(np.ones((m, m), dtype=bool), k=1)
    valid = mask & (np.abs(A_diff) > 1e-12)
    candidates = [t_lo, t_hi]
    if np.any(valid):
        T_vals = B_diff[valid] / A_diff[valid]
        T_in = (T_vals >= t_lo) & (T_vals <= t_hi)
        if np.any(T_in):
            candidates.extend(T_vals[T_in].tolist())
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= t_lo) & (candidates <= t_hi)]
    vals = a[None, :] * candidates[:, None] + b[None, :]
    f_vals = vals.max(axis=1)
    idx = int(np.argmin(f_vals))
    return float(candidates[idx]), float(f_vals[idx])

def _cd_sweep(h, max_sweeps=20, rng=None, coeff_mat=None):
    n = len(h)
    h = h.copy()
    target = n / 2.0
    for sweep in range(max_sweeps):
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for i in order:
            i = int(i)
            a = _coeffs_vector(h)[i] if coeff_mat is None else _coeffs_vector(h)[i]
            # Recompute since h changed
            a = _coeffs_vector_row(h, i)
            scores = _all_scores(h)
            b = scores - a * h[i]
            t_opt, _ = _min_max_lines(a, b, 0.0, 1.0)
            if abs(t_opt - h[i]) > 1e-10:
                h[i] = t_opt
                improved = True
        if not improved:
            break
    h = _project(h, target)
    return h, _exact_c5(h)

def _coeffs_vector_row(h, i):
    n = len(h)
    a = np.zeros(2 * n - 1)
    d = np.arange(n)
    m1 = d <= i
    t1 = np.zeros(n)
    if np.any(m1):
        t1[m1] = 1.0 - h[i - d[m1]]
    m2 = (i + d) < n
    t2 = np.zeros(n)
    if np.any(m2):
        t2[m2] = -h[i + d[m2]]
    a[n - 1:] = t1 + t2
    g = np.arange(1, n)
    idx = n - 1 - g
    m1 = (i + g) < n
    t1 = np.zeros(n - 1)
    if np.any(m1):
        t1[m1] = 1.0 - h[i + g[m1]]
    m2 = i >= g
    t2 = np.zeros(n - 1)
    if np.any(m2):
        t2[m2] = -h[i - g[m2]]
    a[idx] = t1 + t2
    return a

def _pair_sweep(h, rng, n_pairs=500, deadline=None):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(n_pairs):
        if deadline is not None and time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        s = h[i] + h[j]
        t_lo = max(0.0, s - 1.0)
        t_hi = min(1.0, s)
        if t_hi - t_lo < 1e-12:
            continue
        a_i = _coeffs_vector_row(h, i)
        a_j = _coeffs_vector_row(h, j)
        a_p = a_i - a_j
        scores = _all_scores(h)
        b_p = scores - a_p * h[i]
        t_opt, _ = _min_max_lines(a_p, b_p, t_lo, t_hi)
        h_test = h.copy()
        h_test[i] = t_opt
        h_test[j] = s - t_opt
        c5_test = _exact_c5(h_test)
        if c5_test < cur_c5 - 1e-13:
            h = h_test
            cur_c5 = c5_test
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
