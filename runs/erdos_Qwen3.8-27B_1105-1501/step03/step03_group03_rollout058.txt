def _pair_pass(h, target, deadline, max_iter=35):
    n = len(h)
    h = h.copy()
    cur_c5 = _c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    pass_count = 0
    
    while time.time() < deadline:
        improved = False
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) < 2:
            break
        for ii in range(len(free)):
            for jj in range(ii+1, len(free)):
                if time.time() > deadline:
                    return best_h, best_c5
                i = free[ii]; j = free[jj]
                c = h[i] + h[j]
                _pair_opt(h, i, j, c, max_iter)
                new_c5 = float(np.max(_scores(h))) * 2.0 / n
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_h = h.copy()
                    cur_c5 = new_c5
                    improved = True
        pass_count += 1
        if not improved:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
