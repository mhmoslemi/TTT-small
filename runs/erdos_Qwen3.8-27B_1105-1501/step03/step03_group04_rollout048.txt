The Hessian of logsumexp(beta * scores(h)) / beta involves second derivatives of scores, which are -delta_{j-d} terms (since d²/dh[k]² of h[j]*(1-h[j-d]) = 0 for j≠k-d and = -1 for j=k-d). Actually the Hessian is:

H[k,l] = sum_r w_r * (δ_{k,l-d_r} * ... ) 

This is getting complex. Let me just use SLSQP which internally does BFGS.

Let me also think about the LP polish more carefully. The previous `_lp_direction` finds a direction p that minimizes the maximum of (scores + G@p) subject to ||p||_inf <= radius and sum(p) = 0. This is a valid first-order direction. But it only goes one step at a time.

A better LP approach: instead of finding a direction and doing line search, directly solve:
minimize t
subject to: scores_k + G_k @ (h + delta) <= t  (linearized)
            sum(h + delta) = n/2
            0 <= h + delta <= 1
            ||delta||_inf <= radius

This gives both the direction AND the step size simultaneously. Let me implement this:

```python
def _lp_step(h, radius=0.1):
    """Solve LP to find best h_new = h + delta within box around h."""
    n = len(h)
    target = n / 2.0
    G = _build_G(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Variables: delta[0..n-1], t
    # Objective: minimize t
    # Constraints:
    #   G @ delta - t <= -scores  (for each of 2n-1 rows)
    #   sum(delta) = 0
    #   -h[j] <= delta[j] <= 1-h[j]  (box)
    #   -radius <= delta[j] <= radius
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -scores
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            h_new = h + res.x[:n]
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h
```

This is cleaner than the previous approach and gives a better step.

The problem at n=64 has:

This is a non-convex bilinear optimization problem. The best approaches are:

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            if diff > 0:
                mask = x < 1.0
            else:
                mask = x > 0.0
            cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        k1s = max(0, d)
        k1e = min(n, n + d)
        if k1e > k1s:
            G[r, k1s:k1e] = 1.0 - h[k1s - d:k1e - d]
        k2s = max(0, -d)
        k2e = min(n, n - d)
        if k2e > k2s:
            G[r, k2s:k2e] -= h[k2s + d:k2e + d]
    return G

def _softmax_weights(scores, beta):
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    s = float(np.sum(exps))
    if s < 1e-300:
        return np.ones(len(scores)) / len(scores)
    return exps / s

def _soft_descent(h, deadline, beta_start=30.0, beta_end=8000.0, max_iters=15000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.1
    stall = 0
    G = _build_G(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max(1, max_iters), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = np.correlate(h, 1.0 - h, mode='full')
        weights = _softmax_weights(scores, beta)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = h + alpha * direction
            if np.any(h_try < -0.5) or np.any(h_try > 1.5):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                G = _build_G(h)
                stall = 0
                lr = min(lr * 1.06, 0.4)
                improved = True
                break
            alpha *= 0.55
            if alpha < 1e-7:
                break
        if not improved:
            stall += 1
            lr *= 0.82
            if lr < 1e-5 or stall > 60:
                break
    return best_h, best_c5

def _slsqp_step(h, beta=8000.0, maxiter=300):
    n = len(h)
    target = n / 2.0

    def obj(hv):
        sc = np.correlate(hv, 1.0 - hv, mode='full')
        return float(logsumexp(beta * sc) / beta)

    def grad_f(hv):
        sc = np.correlate(hv, 1.0 - hv, mode='full')
        w = _softmax_weights(sc, beta)
        Gv = _build_G(hv)
        return Gv.T @ w

    cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target,
             'jac': lambda hv: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad_f, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hv = _project(res.x, target)
            return hv, _exact_c5(hv)
    except Exception:
        pass
    return h, _exact_c5(h)

def _lp_step(h, radius=0.1):
    n = len(h)
    target = n / 2.0
    G = _build_G(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            h_new = h + res.x[:n]
            h_new = _project(h_new, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h, _exact_c5(h)

def _lp_polish(h, deadline, radius_init=0.08, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        h_new, c5_new = _lp_step(h, radius)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.3)
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for delta in deltas:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    comp = int(np.argmax(1.0 - h) if d > 0 else np.argmax(h))
                    if comp == j:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        if d > 0:
                            comp = int(np.argmax(np.where(mask, 1.0 - h, -1)))
                        else:
                            comp = int(np.argmax(np.where(mask, h, -1)))
                    if comp == j:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
```
