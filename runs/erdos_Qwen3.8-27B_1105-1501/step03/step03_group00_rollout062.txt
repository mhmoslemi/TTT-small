import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_pnorm(h, p):
    n = len(h)
    dx = 2.0 / n
    s = _scores(h)
    eps = 1e-15
    sc = np.maximum(s, eps)
    w = sc ** (p - 1.0)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1] * dx
        if wd < 1e-18:
            continue
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            grad[:m] += wd * (1.0 - h[d:])
            grad[d:] += -wd * h[:m]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            grad[g:] += wd * (1.0 - h[:m])
            grad[:m] += -wd * h[g:]
    return grad

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * dx

def _line_search(h, direction, max_alpha=None):

    alpha_max = 0.5
    if max_alpha is not None:
        alpha_max = min(alpha_max, max_alpha)
    for j in range(n):

            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j] * 0.99)

    alpha_max = max(alpha_max, 1e-12)
    n_samp = 20
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * direction) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])

    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * direction)
    f2 = _exact_c5(h + c2 * direction)
    for _ in range(28):

        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)

        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * direction)

    opt_c5 = _exact_c5(h + opt_a * direction)
    if opt_c5 < best_c5:

def _pnorm_descent(h, deadline, p_start=8.0, momentum=0.85, max_iters=50000):

    best_h = h.copy()

    p = float(p_start)
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if it > max_iters:

        grad = _grad_pnorm(h, p)
        grad -= np.mean(grad)
        at_lo = h < 1e-10
        at_hi = h > 1.0 - 1e-10
        grad[at_lo & (grad < 0)] = 0.0
        grad[at_hi & (grad > 0)] = 0.0
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            v = np.zeros(n)
            p = min(p * 2.0, 10000.0)

            continue
        v = momentum * v - grad / gn
        v -= np.mean(v)

def _lp_direction(h, scores, G, radius, active_mask=None):
    n = len(h)
    n_lags = len(scores)
    if active_mask is not None:
        idx = np.where(active_mask)[0]
    else:
        idx = np.arange(n_lags)
    if len(idx) == 0:
        return None
    G_k = G[idx]
    s_k = scores[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, active_mask=active)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            thresh *= 0.8
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=0.4)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
            thresh = min(thresh * 1.1, 5e-3)
        else:
            stall += 1
            radius *= 0.75
            thresh *= 1.1
    return best_h, best_c5

def _binary_c5(b):
    return _exact_c5(b.astype(float))

def _binary_anneal(n, rng, deadline, max_iters=200000):
    target = n // 2
    idx = rng.permutation(n)
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    cur_c5 = _binary_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    T = 0.008
    T_min = 1e-7
    cooling = 0.9997
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        if it > max_iters:
            break
        if stall > 5000:
            T = max(T, 0.003)
            stall = 0
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or b[i] == b[j]:
            continue
        b[i], b[j] = b[j], b[i]
        new_c5 = _binary_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_b = b.copy()
                stall = 0
        else:
            b[i], b[j] = b[j], b[i]
            stall += 1
        T = max(T * cooling, T_min)
    best_b = best_b.copy()
    return _project(best_b, n / 2.0), best_c5

def _subgradient_descent(h, deadline, max_iters=30000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    it = 0
    while time.time() < deadline:
        it += 1
        if it > max_iters:
            break
        scores = _scores(h)
        c5_cur = float(np.max(scores))
        active = scores >= (c5_cur - 2e-4)
        if np.sum(active) == 0:
            active[np.argmax(scores)] = True
        G = _grad_matrix(h)
        subgrad = G[active].mean(axis=0)
        subgrad -= np.mean(subgrad)
        sn = np.linalg.norm(subgrad)
        if sn < 1e-15:
            break
        v = 0.9 * v - subgrad / sn
        v -= np.mean(v)
        at_lo = h < 1e-10
        at_hi = h > 1.0 - 1e-10
        v[at_lo & (v < 0)] = 0.0
        v[at_hi & (v > 0)] = 0.0
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        alpha, new_c5 = _line_search(h, v / vn, max_alpha=0.2)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * (v / vn)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            v *= 0.5
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n, 2):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h[jj] = h[j]
                        h[kk] = h[k]
                        h[j] += sign * m1
                        h[k] -= sign * m1
                        f1 = _exact_c5(h)
                        h[j] += sign * (m2 - m1)
                        f2 = _exact_c5(h)
                        h[j] = h[jj]
                        h[k] = h[kk]
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_binary_pattern(seed, n, kind=0):
    rng = np.random.default_rng(seed)
    target = n // 2
    if kind == 0:
        b = rng.integers(0, 2, n).astype(float)
        ones = int(np.sum(b))
        if ones > target:
            idx = np.where(b == 1)[0]
            remove = rng.choice(idx, ones - target, replace=False)
            b[remove] = 0
        elif ones < target:
            idx = np.where(b == 0)[0]
            add = rng.choice(idx, target - ones, replace=False)
            b[add] = 1
    elif kind == 1:
        period = max(3, n // 8)
        pattern = np.zeros(period)
        n_ones_pat = max(1, period // 2)
        pattern[:n_ones_pat] = 1.0
        reps = n // period
        b = np.tile(pattern, reps)
        if len(b) < n:
            b = np.concatenate([b, pattern[:n - len(b)]])
        b = b[:n]
        ones = int(np.sum(b))
        if ones > target:
            idx = np.where(b == 1)[0]
            remove = rng.choice(idx, ones - target, replace=False)
            b[remove] = 0
        elif ones < target:
            idx = np.where(b == 0)[0]
            add = rng.choice(idx, target - ones, replace=False)
            b[add] = 1
    elif kind == 2:
        b = np.zeros(n)
        step = max(2, n // (2 * target))
        positions = np.arange(0, n, step)
        b[positions[:target]] = 1.0
        if int(np.sum(b)) < target:
            remaining = np.where(b == 0)[0]
            add = rng.choice(remaining, target - int(np.sum(b)), replace=False)
            b[add] = 1
    else:
        b = rng.integers(0, 2, n).astype(float)
        ones = int(np.sum(b))
        if ones > target:
            idx = np.where(b == 1)[0]
            remove = rng.choice(idx, ones - target, replace=False)
            b[remove] = 0
        elif ones < target:
            idx = np.where(b == 0)[0]
            add = rng.choice(idx, target - ones, replace=False)
            b[add] = 1
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: P-norm descent from initial
    dl1 = t0 + budget_s * 0.18
    h1, c5_1 = _pnorm_descent(best_h, dl1, p_start=8.0, momentum=0.85)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement
    dl2 = t0 + budget_s * 0.35
    h2, c5_2 = _lp_refine(best_h, dl2, radius_init=0.12, thresh=5e-4)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient descent
    dl3 = t0 + budget_s * 0.45
    h3, c5_3 = _subgradient_descent(best_h, dl3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary pattern search + anneal + continuous relax
    dl4 = t0 + budget_s * 0.65
    n_try = best_n
    for attempt in range(30):
        if time.time() > dl4:
            break
        if best_c5 < 0.3807:
            break
        kind = attempt % 4
        s_local = seed + 1000 + attempt
        b_pat = _make_binary_pattern(s_local, n_try, kind=kind)
        sub_dl = min(dl4, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _binary_anneal(n_try, rng, sub_dl, max_iters=80000)
        if c5_bin < best_c5 - 0.01:
            sub_dl2 = min(dl4, time.time() + budget_s * 0.04)
            h_rel, c5_rel = _pnorm_descent(h_bin, sub_dl2, p_start=4.0, momentum=0.8)
            if c5_rel < best_c5:
                best_c5 = c5_rel
                best_h = h_rel.copy()

    # Phase 5: Perturbation restarts from current best
    dl5 = t0 + budget_s * 0.78
    scale = 0.03
    for attempt in range(200):
        if time.time() > dl5:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.02)
        h_try, c5_try = _pnorm_descent(h_pert, sub_dl, p_start=16.0, momentum=0.85, max_iters=8000)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: LP refinement again
    if time.time() < t0 + budget_s * 0.88:
        dl6 = t0 + budget_s * 0.90
        h6, c5_6 = _lp_refine(best_h, dl6, radius_init=0.05, thresh=2e-4)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Resolution sweep
    if time.time() < t0 + budget_s * 0.94:
        dl7 = t0 + budget_s * 0.955
        for dn in (-3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10, -4, -5):
            if time.time() > dl7:
                break
            n_t = best_n + dn
            if n_t < 24 or n_t > 128:
                continue
            h_r = _resample(best_h, n_t)
            sub_dl = min(dl7, time.time() + budget_s * 0.018)
            h_r, c5_r = _pnorm_descent(h_r, sub_dl, p_start=16.0, momentum=0.85, max_iters=5000)
            if c5_r < best_c5 - 1e-12:
                sub_dl2 = min(dl7, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _lp_refine(h_r, sub_dl2, radius_init=0.05, thresh=3e-4)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()
                    best_n = n_t

    # Phase 8: Final pair polish
    if time.time() < t0 + budget_s * 0.98:
        dl8 = t0 + budget_s * 0.99
        hp, c5p = _pair_polish(best_h, dl8, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: ultra-fine LP
    if time.time() < t0 + budget_s * 0.995:
        hf, c5f = _lp_refine(best_h, t0 + budget_s * 0.998, radius_init=0.01, thresh=1e-5, max_iters=3000)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()
            best_n = len(hf)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
