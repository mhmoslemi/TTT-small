def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_interval(h, d, radius):

    t_max = radius

            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_max = min(t_max, h[i] / (-d[i]))
    return max(t_max, 0.0)

def _line_obj(h, d, t):

def _golden_1d(f, lo, hi, max_iter=50):
    """Golden section minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(max_iter):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    return (lo + hi) / 2.0, min(f1, f2)

def _smooth_direction(n, freq, phase, rng):
    """Generate a smooth random direction."""
    x = np.arange(n) / n
    d = np.zeros(n)
    for k in range(1, freq + 1):
        amp = rng.standard_normal()
        phi = rng.uniform(0, 2 * np.pi)
        d += amp * np.cos(2 * np.pi * k * x + phi)
    d -= np.mean(d)
    norm = float(np.linalg.norm(d))
    if norm < 1e-14:
        return None
    return d / norm

def _resample(h, n_new):

def _coordinate_descent(h, deadline, max_passes=20):
    """Pairwise coordinate descent with exact 1D search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Random order
        idx = np.arange(n)
        np.random.shuffle(idx)
        for i_pos in range(n):
            if time.time() > deadline:
                break
            i = idx[i_pos]
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j_pos in range(i_pos + 1, n):
                j = idx[j_pos]
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                # Try both directions
                for sign in [1, -1]:
                    d = np.zeros(n)
                    if sign > 0:
                        d[i] = 1.0
                        d[j] = -1.0
                    else:
                        d[i] = -1.0
                        d[j] = 1.0
                    h_new, c5_new = _try_direction(h, cur_c5, d, 0.3)
                    if h_new is not None:
                        h = h_new
                        cur_c5 = c5_new
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        if sign > 0:
                            break  # no need to try reverse
        if not improved:
            break
    
    return best_h, best_c5

def _anneal_restart(h, deadline, n_restarts=10, scale_init=0.06, seed=42):

    n = len(h)
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    current = h.copy()
    current_c5 = best_c5
    scale = scale_init
    t_cur = time.time()
    t_total = deadline - t_cur
    
    for attempt in range(n_restarts * 5):
        if time.time() > deadline:
            break
        frac = (time.time() - t_cur) / max(t_total, 1e-6)
        current_scale = scale * (1.0 - 0.7 * frac)
        
        d = rng.normal(0, 1, n)
        d -= d.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:

        d /= norm
        d *= current_scale
        
        h_try = current + d

            current_c5 = c5_try

                best_h = h_try.copy()
                scale = max(0.01, scale * 0.92)

                current = h_try
                current_c5 = c5_try
            scale = min(0.2, scale * 1.05)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_precompute(h, d):
    omh = 1.0 - h
    A = np.correlate(h, omh, mode='full')
    C = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

def _line_obj(A, C, Q, t, dx):
    scores = A + t * C - t * t * Q

            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:

    t_max = _feasible_t_max(h, d, radius)

def _grad_row(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    j_lo = max(0, -s)
    j_hi = min(n - 1, n - 1 - s)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        g[idx] += (1.0 - h[idx + s])
    j_lo2 = max(0, s)
    j_hi2 = min(n - 1, n - 1 + s)
    if j_lo2 <= j_hi2:
        idx2 = np.arange(j_lo2, j_hi2 + 1)
        g[idx2] += -h[idx2 - s]

def _subgradient_direction(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    if max_s < 1e-15:
        return None
    exp_s = np.exp(tau * (scores - max_s))
    total = float(np.sum(exp_s))
    if total < 1e-15:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = exp_s / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g_k = _grad_row(h, k, n)
        grad += w[k] * g_k
    grad -= np.mean(grad)
    norm = float(np.linalg.norm(grad))
    if norm < 1e-14:
        return None
    return -grad / norm

def _smooth_dir(n, freq_max, rng):
    x = np.arange(n) / n
    d = np.zeros(n)
    for k in range(1, freq_max + 1):
        amp = rng.standard_normal()
        phi = rng.uniform(0, 2.0 * np.pi)
        d += amp * np.cos(2.0 * np.pi * k * x + phi)
    d -= np.mean(d)
    norm = float(np.linalg.norm(d))
    if norm < 1e-14:
        return None
    return d / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize(h, deadline, radius_init=0.2, seed=42, mix_ratio=(0.4, 0.3, 0.2, 0.1)):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    tau_current = 50.0
    it = 0
    while time.time() < deadline and stall < 80:
        it += 1
        r = rng.random()
        if r < mix_ratio[0]:
            d = _subgradient_direction(h, tau=tau_current)
        elif r < mix_ratio[0] + mix_ratio[1]:
            freq = int(rng.integers(1, 6))
            d = _smooth_dir(n, freq, rng)
        elif r < mix_ratio[0] + mix_ratio[1] + mix_ratio[2]:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        else:
            n_block = int(rng.integers(3, min(8, n - 2)))
            start = int(rng.integers(0, n - n_block))
            d = np.zeros(n)
            block = rng.standard_normal(n_block)
            block -= block.mean()
            d[start:start + n_block] = block
        if d is None:
            stall += 1
            continue
        d = d - d.mean()
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            stall += 1
            continue
        d /= norm
        t, new_c5 = _line_search_fast(h, d, cur_c5, radius)
        if t > 1e-12:
            h = h + t * d
            h = _project(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            stall = 0
            radius = min(0.5, radius * 1.08)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            tau_current = max(20.0, tau_current * 0.98)
        else:
            stall += 1
            radius *= 0.85
            tau_current = min(150.0, tau_current * 1.15)
            if stall % 15 == 0:
                radius = 0.15
    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=30):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i_pos in range(n):
            if time.time() > deadline:
                break
            i = int(order[i_pos])
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j_pos in range(i_pos + 1, n):
                j = int(order[j_pos])
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                t, new_c5 = _line_search_fast(h, d, cur_c5, 0.4, n_gold=25)
                if t > 1e-12:
                    h = h + t * d
                    h = _project(h, n / 2.0)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    d2 = -d
                    t2, new_c5_2 = _line_search_fast(h, d2, cur_c5, 0.4, n_gold=25)
                    if t2 > 1e-12:
                        h = h + t2 * d2
                        h = _project(h, n / 2.0)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                else:
                    d_rev = -d
                    t_rev, new_c5_rev = _line_search_fast(h, d_rev, cur_c5, 0.4, n_gold=25)
                    if t_rev > 1e-12:
                        h = h + t_rev * d_rev
                        h = _project(h, n / 2.0)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _try_resolutions(h_best, c5_best, n_best, deadline, seed):
    best_h = h_best.copy()
    best_c5 = c5_best
    best_n = n_best
    rng = np.random.default_rng(seed)
    offsets = list(range(-30, 31, 2))
    rng.shuffle(offsets)
    offsets = sorted(offsets, key=lambda x: abs(x))
    time_per = max(4.0, (deadline - time.time()) / 15.0)
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 180:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.008:
            continue
        sub_dl = min(deadline, time.time() + time_per)
        h_r, c5_r = _optimize(h_r, sub_dl, radius_init=0.15, seed=n_try + seed)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    return best_h, best_c5, best_n

def _anneal(h, deadline, scale_init=0.05, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    current = h.copy()
    current_c5 = best_c5
    scale = scale_init
    t_start = time.time()
    t_total = max(deadline - t_start, 1.0)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, (time.time() - t_start) / t_total)
        cur_scale = scale * (1.0 - 0.6 * frac)
        d = rng.normal(0, cur_scale, n)
        d -= d.mean()
        h_try = current + d
        h_try = _project(h_try, n / 2.0)
        c5_try = _exact_c5(h_try)
        if c5_try < current_c5 - 1e-13:
            current = h_try
            current_c5 = c5_try
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_try.copy()
                scale = max(0.01, scale * 0.93)
        else:
            temp = 0.005 * (1.0 - frac)
            if temp > 1e-10 and rng.random() < np.exp(-(c5_try - current_c5) / temp):
                current = h_try
                current_c5 = c5_try
            scale = min(0.15, scale * 1.04)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _coord_descent(best_h, d1, max_passes=12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _optimize(best_h, d2, radius_init=0.2, seed=seed,
                         mix_ratio=(0.45, 0.25, 0.20, 0.10))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    d3 = t0 + budget_s * 0.62
    h3, c5_3, n3 = _try_resolutions(best_h, best_c5, best_n, d3, seed + 50)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = n3

    d4 = t0 + budget_s * 0.82
    h4, c5_4 = _anneal(best_h, d4, scale_init=0.04, seed=seed + 100)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    d5 = t0 + budget_s * 0.90
    h5, c5_5 = _optimize(best_h, d5, radius_init=0.1, seed=seed + 200,
                         mix_ratio=(0.5, 0.2, 0.2, 0.1))
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    d6 = t0 + budget_s * 0.97
    h6, c5_6 = _coord_descent(best_h, d6, max_passes=40)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    if time.time() < t0 + budget_s * 0.995:
        h7, c5_7 = _optimize(best_h, t0 + budget_s * 0.998, radius_init=0.05,
                             seed=seed + 300, mix_ratio=(0.6, 0.15, 0.15, 0.10))
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
