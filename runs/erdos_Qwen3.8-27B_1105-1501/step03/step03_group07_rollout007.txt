def _binary_milp(n, time_limit=30):
    """Solve binary minimax overlap problem via MILP"""
    # Variables: x[0..n-1] binary, M continuous
    # For each d=1..n-1, for each j=0..n-1-d: z[j,d] = x[j]*x[j+d]
    x = cp.Variable(n, boolean=True)
    M = cp.Variable()
    
    z_vars = {}
    for d in range(1, n):
        for j in range(n - d):
            z_vars[(j, d)] = cp.Variable()
    
    constraints = [
        cp.sum(x) == n // 2,
        M >= 0
    ]
    
    # Binary linearization
    for (j, d), z in z_vars.items():
        constraints.append(z <= x[j])
        constraints.append(z <= x[j + d])
        constraints.append(z >= x[j] + x[j + d] - 1)
        constraints.append(z >= 0)
    
    # Overlap constraints
    for d in range(1, n):
        # lag +d: sum_{j=0}^{n-1-d} x[j]*(1-x[j+d]) <= M
        expr_pos = 0
        expr_neg = 0
        for j in range(n - d):
            expr_pos += x[j] - z_vars[(j, d)]
            expr_neg += x[j + d] - z_vars[(j, d)]
        constraints.append(expr_pos <= M)
        constraints.append(expr_neg <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit)
    except:
        try:
            prob.solve(solver=cp.CPLEX)
        except:
            return None
    
    if x.value is None:
        return None
    
    h = np.round(x.value).astype(float)
    # Verify
    if int(np.sum(h)) != n // 2:
        return None
    return h
