import numpy as np

def _exact_c5(h):

    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search_exact(h, p):
    n = len(h)
    alpha_max = 0.8
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))

    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))

    best_a = float(alphas[best_i])
    if best_i == 0 or best_i == n_samp - 1:
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]

    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)

def _lp_direction(h, scores, G, radius, top_k):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j] + 1e-12), min(radius, 1.0 - h[j] - 1e-12)))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = 0.005
    t_end = 0.0001
    for it in range(n_iter):
        if time.time() > deadline:
            break
        temp = t_start * (t_end / t_start) ** (it / max(1, n_iter - 1))
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-6:
            max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-6:
                continue
            h[i] -= max_d * rng.random() * 0.3
            h[j] += max_d * rng.random() * 0.3
        else:
            h[i] -= max_d * rng.random() * 0.3
            h[j] += max_d * rng.random() * 0.3
        h = _project(h, target)
        c5_new = _exact_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            best_c5_local = best_c5
            if it % 500 == 0 and c5 > best_c5 + 0.01:
                h = best_h.copy()
                c5 = best_c5
    return best_h, best_c5

def _coord_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, rng, scale):
    n = len(h)
    target = n / 2.0
    x = np.arange(n) / n * 2.0
    op = rng.integers(4)
    if op == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        h_p = h + scale * np.sin(np.pi * f * x + phase)
    elif op == 1:
        shift = rng.uniform(-0.05, 0.05)
        x_shift = x + shift
        x_shift = np.clip(x_shift, 0, 2)
        h_p = np.interp(x_shift, x, h, left=h[0], right=h[-1])
    elif op == 2:
        sigma = rng.uniform(0.05, 0.15)
        ksize = int(4 * sigma * n) + 1
        if ksize > n:
            ksize = n
        if ksize % 2 == 0:
            ksize += 1
        kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (sigma * n / 2) ** 2)
        kernel /= kernel.sum()
        h_p = np.convolve(h, kernel, mode='same')
    else:
        h_p = h + rng.normal(0, scale, n)
    return _project(h_p, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: FISTA on smooth surrogate (15% budget)
    dl1 = t0 + budget_s * 0.15
    h, c5 = _fista_phase(h, dl1, beta_start=2.0, beta_end=30.0, n_iter=1500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP trust region (20% budget)
    dl2 = t0 + budget_s * 0.35
    h, c5 = _lp_phase(h, dl2, top_k_list=[None, 8, 5, 10], radius_init=0.12, max_iters=2000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Second FISTA with higher beta (10% budget)
    if time.time() < t0 + budget_s * 0.48:
        dl3 = t0 + budget_s * 0.45
        h, c5 = _fista_phase(best_h, dl3, beta_start=10.0, beta_end=80.0, n_iter=1000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Simulated annealing (20% budget)
    dl4 = t0 + budget_s * 0.65
    h, c5 = _anneal_phase(best_h, dl4, n_iter=40000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Multi-start with structured perturbations (20% budget)
    dl5 = t0 + budget_s * 0.85
    scale = 0.02
    for attempt in range(300):
        if time.time() > dl5:
            break
        if best_c5 < 0.3805:
            break
        h_pert = _structured_perturb(best_h, rng, scale)
        sub_dl = min(dl5, time.time() + budget_s * 0.02)
        h_try, c5_try = _lp_phase(h_pert, sub_dl, top_k_list=[8, 5, None], radius_init=0.08, max_iters=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.9)
        else:
            scale = min(0.12, scale * 1.08)

    # Phase 6: Coordinate polish (10% budget)
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.97
        hp, c5p = _coord_polish(best_h, dl6, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Resolution sweep (if time remains)
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.99
        for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10, 12]:
            if time.time() > dl7:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl7, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_phase(h_r, sub_dl, top_k_list=[8, None], radius_init=0.1, max_iters=200)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final FISTA polish
    if time.time() < t0 + budget_s * 0.995:
        h_f, c5_f = _fista_phase(best_h, t0 + budget_s * 0.998, beta_start=20.0, beta_end=100.0, n_iter=500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
