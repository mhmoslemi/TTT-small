```python
def _active_qp_step(h, n, K_active=5, step_max=0.1):
    """
    Active-set QP step.
    1. Find top-K correlation lags
    2. Linearize those K constraints
    3. Solve QP: min ||G_A @ d + (c_A - t)||^2 s.t. sum(d)=0, ||d||<=step_max
    4. Return h + d
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    # Active set: lags within 5% of max
    tol = 0.05 * mx
    active = np.where(corr >= mx - tol)[0]
    K = len(active)
    
    # Build G_A (K x n)
    G_A = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            G_A[idx, i] += (1.0 - h[i + d])
            G_A[idx, i + d] += (-h[i])
    
    # c_A = corr[active]
    c_A = corr[active]
    
    # QP: min (G_A @ d + c_A - mx)^2  [want to reduce all active to below mx]
    # = min d' G_A' G_A d + 2 (G_A' (c_A - mx))' d + const
    # s.t. 1' d = 0, ||d|| <= step_max
    # 
    # Without trust region: solve (G_A' G_A + eps*I) d = -G_A' (c_A - mx)
    # with constraint 1'd = 0
    
    ATA = G_A.T @ G_A
    Atb = G_A.T @ (c_A - mx)  # this should be <= 0 since c_A ≈ mx
    # Actually: we want to minimize (c_A + G_A @ d - mx)^2
    # Gradient: 2 G_A' (c_A + G_A @ d - mx)
    # At optimum (without constraint): G_A' G_A d = -G_A' (c_A - mx)
    
    # With sum constraint: use Lagrange multiplier
    # [ATA  1; 1'  0] [d; mu] = [-Atb; 0]
    
    # For small systems, use direct solve
    n1 = n + 1
    sys = np.zeros((n1, n1))
    sys[:n, :n] = ATA + 1e-10 * np.eye(n)
    sys[:n, n] = 1.0
    sys[n, :n] = 1.0
    rhs = np.zeros(n1)
    rhs[:n] = -Atb
    
    try:
        sol = np.linalg.solve(sys, rhs)
        d = sol[:n]
    except:
        d = -G_A.T @ (c_A - mx)
        d -= np.mean(d)
    
    # Trust region
    dn = np.linalg.norm(d)
    if dn > step_max:
        d = d / dn * step_max
    
    # Line search
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
        h_t = _project(h + alpha * d, n / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
    return best_h, best_c5
```
