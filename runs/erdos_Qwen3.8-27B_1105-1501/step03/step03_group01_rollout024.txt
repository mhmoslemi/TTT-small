def project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} using waterfill."""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Waterfill: find theta such that sum(clip(v - theta, 0, 1)) = target
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(v - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def soft_value_and_grad(h, tau):
    """Compute soft-max objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, 'full') * dx
    max_s = scores.max()
    shifted = (scores - max_s) / tau
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    w = exp_s / Z
    F = max_s + tau * np.log(Z)
    
    # Gradient: grad[j] = dx * sum_m w[m+n-1] * [(1-h[j-m])*[valid] - h[j+m]*[valid]]
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        wm = w[m + (n-1)]
        if wm < 1e-15:
            continue
        # First term: (1-h[j-m]) for j where j-m in [0,n-1]
        j_lo = max(0, m)
        j_hi = min(n-1, n-1 + m)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo-m:j_hi-m+1])
        # Second term: -h[j+m] for j where j+m in [0,n-1]
        j_lo2 = max(0, -m)
        j_hi2 = min(n-1, n-1 - m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wm * h[j_lo2+m:j_hi2+m+1]
    grad *= dx
    return F, grad

            A[freqs[i]] = amps[i] * np.cos(phs[i])
            A[freqs[i] + n] = amps[i] * np.sin(phs[i])
        h = 0.5 + np.real(np.fft.ifft(A))[:n]
        h = _proj(h, n / 2.0)
        c = _c5(h)
        if best is None or c < best[1]:
            best = (h, c)
    return best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + budget_s * 0.96

    init = np.asarray(initial_h_values, dtype=float)
    h0 = _proj(init.init_h_values, len(init) / 2.0) if hasattr(init, 'init_h_values') else _proj(init, len(init) / 2.0)
    if not np.all(np.isfinite(h0)):
        n0 = 84
        h0 = np.full(n0, 0.5)
    n0 = len(h0)
    best_h = h0.copy()
    best_c = _c5(h0)
    best_n = n0
    rng = np.random.default_rng(seed)

    def _update(h_try, c_try, n_try):
        nonlocal best_h, best_c, best_n
        if c_try < best_c - 1e-14:
            best_c, best_h, best_n = c_try, h_try.copy(), n_try

    # Phase 1: SLSQP from parent with tau annealing
    p1_dl = t0 + budget_s * 0.30
    tau_main = [0.6, 0.25, 0.1, 0.04, 0.015, 0.006, 0.0025]
    h_best1, c_best1 = _slsqp_phase(best_h, p1_dl, tau_main)
    _update(h_best1, c_best1, best_n)

    # Phase 2: Multi-start with perturbations
    p2_dl = t0 + budget_s * 0.52
    tau_med = [0.3, 0.1, 0.03, 0.01, 0.004]
    scales = [0.03, 0.06, 0.10, 0.15, 0.22, 0.30]
    for sc in scales:
        if time.time() > p2_dl:
            break
        tol = max(1.0, (best_c - 0.36))
        hd = best_h + rng.normal(0, sc, best_n)
        hd = _proj(hd, best_n / 2.0)
        cd = _c5(hd)
        if cd < best_c + tol:
            sub_dl = min(p2_dl, time.time() + budget_s * 0.035)
            ht, ct = _slsqp_phase(hd, sub_dl, tau_med)
            _update(ht, ct, best_n)

    # Phase 3: Fourier and structured starts
    if time.time() < t0 + budget_s * 0.62:
        p3_dl = t0 + budget_s * 0.64
        for fs in range(4):
            if time.time() > p3_dl:
                break
            h_f, c_f = _gen_fourier(best_n, seed + 100 + fs)
            if c_f < best_c + 0.04:
                sub_dl = min(p3_dl, time.time() + budget_s * 0.03)
                ht, ct = _slsqp_phase(h_f, sub_dl, tau_med)
                _update(ht, ct, best_n)

    # Phase 4: Multi-resolution
    if time.time() < t0 + budget_s * 0.78:
        p4_dl = t0 + budget_s * 0.78
        offs = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16,
                -20, 20, -24, 24, -30, 30]
        tau_short = [0.15, 0.05, 0.02, 0.008]
        tried = set()
        for dn in offs:
            if time.time() > p4_dl:
                break
            nt = best_n + dn
            if nt < 40 or nt > 220 or nt in tried:
                continue
            tried.add(nt)
            hr = _resample(best_h, nt)
            sub_dl = min(p4_dl, time.time() + budget_s * 0.025)
            ht, ct = _slsqp_phase(hr, sub_dl, tau_short)
            _update(ht, ct, nt)

    # Phase 5: Additional perturbations on best
    if time.time() < t0 + budget_s * 0.88:
        p5_dl = t0 + budget_s * 0.88
        tau_tight = [0.08, 0.03, 0.01, 0.004, 0.0015]
        sc = 0.015
        for _ in range(40):
            if time.time() > p5_dl:
                break
            hd = best_h + rng.normal(0, sc, best_n)
            hd = _proj(hd, best_n / 2.0)
            sub_dl = min(p5_dl, time.time() + budget_s * 0.018)
            ht, ct = _slsqp_phase(hd, sub_dl, tau_tight)
            if ct < best_c - 1e-12:
                sc = max(0.005, sc * 0.8)
            else:
                sc = min(0.5, sc * 1.2)
            _update(ht, ct, best_n)

    # Phase 6: Pair refinement
    if time.time() < hard_dl:
        p6_dl = min(hard_dl, time.time() + budget_s * 0.10)
        hp, cp = _pc_refine(best_h, p6_dl, max_passes=80)
        _update(hp, cp, best_n)

    # Final: one more SLSQP polish on the very best
    if time.time() < hard_dl - 2.0:
        tau_final = [0.01, 0.003, 0.001]
        hf, cf = _slsqp_phase(best_h, min(hard_dl, time.time() + budget_s * 0.03), tau_final)
        _update(hf, cf, best_n)
        if time.time() < hard_dl - 1.0:
            hp2, cp2 = _pc_refine(best_h, min(hard_dl, time.time() + budget_s * 0.02), max_passes=30)
            _update(hp2, cp2, best_n)

    # Ensure feasibility and return
    best_h = _proj(best_h, best_n / 2.0)
    final_c = _c5(best_h)
    # Fallback safety
    if final_c > 0.6:
        best_h = _proj(np.full(n0, 0.5), n0 / 2.0)
        final_c = _c5(best_h)
    return best_h, final_c, best_n
