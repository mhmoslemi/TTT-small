def _grad_matrix(h):
    # Returns matrix G of shape (2n-1, n) where G[k, i] is d(score_k)/d(h[i])
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                # score_k = sum_{i=0}^{m-1} h[i] * (1 - h[i+lag]) * dx
                # d/dh[i] = (1-h[i+lag])*dx for i < m

            m = n - g
            if m > 0:
                G[k, :m] += (1.0 - h[g:]) * dx
                G[k, g:] += -h[:m] * dx
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    lag = k - (n - 1)
    row = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            row[:m] += (1.0 - h[lag:]) * dx
            row[lag:] += -h[:m] * dx
    else:
        g = -lag
        m = n - g
        if m > 0:
            row[:m] += (1.0 - h[g:]) * dx
            row[g:] += -h[:m] * dx
    return row

def _grad_rows(h, indices):
    return np.array([_grad_row(h, k) for k in indices])

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _solve_qp(h, scores_active, G_active, rho):
    n = len(h)
    d = Variable(n)
    t = Variable()
    constraints = [
        sum(d) == 0,
        d >= -h + 1e-12,
        d <= 1.0 - h - 1e-12,

    for i in range(len(scores_active)):
        constraints.append(scores_active[i] + G_active[i] @ d <= t)
    obj = Minimize(quad_form(d, np.eye(n) * rho) + t)
    prob = Problem(obj, constraints)
    try:

        if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:

    return None

def _line_search_c5(h, d, c5_cur):

        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-10)

def _lp_direction(h, scores, G_active, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores), n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j] - 1e-12), min(radius, 1.0 - h[j] - 1e-12)))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Ternary search
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
