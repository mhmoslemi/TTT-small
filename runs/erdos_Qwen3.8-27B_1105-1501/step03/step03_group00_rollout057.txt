import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pnorm_grad(h, p):
    """Compute gradient of p-norm of scores."""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    eps = 1e-15
    s = np.maximum(scores, eps)
    w = s ** (p - 1)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # Fallback: uniform weight on max score
        w = np.zeros_like(s)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    # For lag l (0 to 2n-2), shift d = l - (n-1)
    # grad[j] += w[l] * dx * [(1-h[j+d])_valid - h[j-d]_valid]
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        d = l - (n - 1)
        wl = w[l] * dx
        if wl < 1e-18:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wl * (1.0 - h[d:])
                grad[d:] -= wl * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wl * (1.0 - h[:m])
                grad[:m] -= wl * h[g:]
    return grad

def _pnorm_optimize(h, deadline, p_start=100, p_end=5, n_steps=2000, momentum=0.9):
    """P-norm gradient descent with momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)  # momentum velocity
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Adaptive p
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        
        grad = _pnorm_grad(h, p)
        # Center the gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v - grad
        # Normalize step
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm  # normalize
        
        # Line search: find best alpha
        alpha_max = 0.1
        for j in range(n):
            if v[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / v[j])
            elif v[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-v[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        # Try a few step sizes
        best_alpha = 0.0
        best_val = cur_c5
        for trial in range(20):
            a = alpha_max * (1.0 - trial / 20.0)
            h_try = h + a * v
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = a
            if c5_try > cur_c5:
                break
        
        if best_alpha > 0 and best_val < cur_c5 - 1e-14:
            h = h + best_alpha * v
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Couldn't improve, reduce momentum and try again
            v *= 0.5
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_scores(h):
    """Compute gradient matrix: G[k, j] = ∂score_k / ∂h_j"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * dx

def _pnorm_direction(h, p):
    """Compute p-norm gradient direction."""
    n = len(h)
    scores = _scores(h)
    eps = 1e-15
    s = np.maximum(scores, eps)
    w = s ** (p - 1)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(s)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    G = _grad_scores(h)
    grad = G.T @ w
    return grad, w, scores

def _adam_step(h, grad, state, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with projection."""
    n = len(h)
    target = n / 2.0
    m = None
    v = None
    t = 1
    if state is not None:
        m, v, t = state
        t += 1
    else:
        m = np.zeros(n)
        v = np.zeros(n)
    
    # Center gradient
    grad_centered = grad - np.mean(grad)
    
    m = beta1 * m + (1 - beta1) * grad_centered
    v = beta2 * v + (1 - beta2) * grad_centered ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    
    h_new = _project(h + step, target)
    new_state = (m, v, t)
    return h_new, new_state

def _pnorm_adam_optimize(h, deadline, p=50, lr=0.1, n_steps=5000):
    """P-norm gradient descent with Adam optimizer."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    state = None
    stall = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        grad, w, scores = _pnorm_direction(h, p)
        h_new, state = _adam_step(h, grad, state, lr=lr)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                state = None  # reset momentum
                stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, rng, n_iters=50000):
    """Simulated annealing on binary patterns."""
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    # Start from thresholding
    h = (h_init >= np.median(h_init)).astype(float)
    # Ensure exactly target ones
    ones_idx = np.where(h == 1)[0]
    if len(ones_idx) > target:
        # Randomly set some to 0
        excess = len(ones_idx) - target
        to_zero = rng.choice(ones_idx, size=excess, replace=False)
        h[to_zero] = 0
    elif len(ones_idx) < target:
        zeros_idx = np.where(h == 0)[0]
        need = target - len(ones_idx)
        to_one = rng.choice(zeros_idx, size=need, replace=False)
        h[to_one] = 1
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_end = T * 1e-6
    cool_rate = (T_end / T) ** (1.0 / max(1, n_iters))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Choose a random 1 and a random 0, swap them
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i] = 0
        h[j] = 1
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T *= cool_rate
    
    return best_h, best_c5

def _coordinate_polish(h, deadline, max_passes=20):
    """Greedy coordinate descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try to optimally set h[i] given all others
            # score_k as function of x = h[i]:
            # score_k(x) = dx * (A_k + x * B_k)
            # where A_k and B_k depend on all other h[j]
            # We want to minimize max_k score_k(x) subject to:
            #   0 <= x <= 1
            #   sum constraint: x = target - sum(h[j] for j≠i)
            # Wait, if we fix the sum constraint, x is determined!
            # So we can't vary x independently.
            # Instead, let's do pair swaps: vary h[i] and h[j] with h[i]+h[j] = const.
            
            # For a quick polish, just try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                # Need to compensate in another coordinate
                # Find the coordinate with most "room"
                if delta > 0:
                    # h[i] increases, need to decrease someone else
                    candidates = [j for j in range(n) if j != i and h[j] > 1e-8]
                else:
                    candidates = [j for j in range(n) if j != i and h[j] < 1.0 - 1e-8]
                if not candidates:
                    continue
                # Try the coordinate that's furthest from the boundary
                if delta > 0:
                    j = max(candidates, key=lambda jj: h[jj])
                else:
                    j = min(candidates, key=lambda jj: h[jj])
                
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                    continue
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
        if not improved and _pass > 3:
            break
    
    return best_h, best_c5

def _fourier_start(n, mode=1):
    """Generate a Fourier-based starting point."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    if mode == 1:
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
    else:
        h = 0.5 + 0.2 * np.sin(np.pi * x) + 0.15 * np.sin(2*np.pi*x)
    return _project(np.clip(h, 0, 1), n / 2.0)
