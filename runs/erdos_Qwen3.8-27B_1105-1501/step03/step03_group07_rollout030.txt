def _lse_slsqp(h0, tau, target, deadline):
    n = len(h0)
    dx = 2.0 / n
    
    def obj(x):
        vals = np.correlate(x, 1.0-x, mode='full')
        m = np.max(vals)
        s = np.sum(np.exp((vals - m) / tau))
        return (m + tau * np.log(s)) * dx
    
    def obj_grad(x):
        vals = np.correlate(x, 1.0-x, mode='full')
        m = np.max(vals)
        w = np.exp((vals - m) / tau)
        wsum = np.sum(w)
        w /= wsum
        grad = np.zeros(n)
        for r in range(len(vals)):
            if w[r] < 1e-10:
                continue
            d = r - (n-1)
            s = abs(d)
            L = n - s
            if L <= 0:
                continue
            if d >= 0:

                grad[s:] += w[r] * (1.0 - x[0:L])
            else:
                grad[0:L] += w[r] * (1.0 - x[s:])
                grad[s:] += -w[r] * x[0:L]

        return grad

                h_new = _project(res.x, target)

def _solve_epigraph(h0, deadline, target, K=12):
    """Epigraph NLP with active constraint set"""
    n = len(h0)
    h = h0.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    dx = 2.0 / n

    for outer in range(4):
        if time.time() > deadline:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-raw)[:min(K, len(raw))]
        t0 = float(np.max(raw * dx)) + 0.003
        z0 = np.concatenate([h, [t0]])

        def _obj(z):
            return z[-1]

        def _obj_grad(z):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g

        def _eq_fun(z):
            return np.sum(z[:-1]) - target

        def _eq_jac(z):
            g = np.zeros(n + 1)
            g[:-1] = 1.0
            return g

        def _ineq_fun(z):
            hz = z[:-1]
            r = np.correlate(hz, 1.0 - hz, mode='full')
            return z[-1] - r[top_idx] * dx

        def _ineq_jac(z):
            hz = z[:-1]
            J = np.zeros((len(top_idx), n + 1))
            for i, idx in enumerate(top_idx):
                d = int(idx) - (n - 1)
                J[i, :n] = -_lag_grad(hz, d) * dx
                J[i, -1] = 1.0
            return J

        constraints = [
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
        ]
        bounds = [(0.0, 1.0)] * n + [(0.0, best_c5 + 0.02)]

        try:
            res = minimize(_obj, z0, jac=_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif c5_new < best_c5 + 0.001:
                h = h_new.copy()
        except Exception:
            pass
        K = min(K + 4, 20)
    return best_h, best_c5
