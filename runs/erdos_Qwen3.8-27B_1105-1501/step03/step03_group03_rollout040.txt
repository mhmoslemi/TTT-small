def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _lp_scores_grad(h):
    """Compute scores and the gradient matrix G where G[r, j] = d(score_r)/d(h_j)"""
    n = len(h)
    scores = _scores(h)
    # G[r, j] = d score(r) / d h[j]
    # score(r) = sum_n h[n+(r-n+1)] * (1-h[n]) for appropriate ranges
    # Actually, let me use the alpha formula:
    # d score(d) / d h[j] = 1 - h[j-d] - h[j+d] (with 0 for out of bounds)
    # where d is the lag (r - (n-1) in 0-indexed)
    lags = np.arange(-(n-1), n)
    h_pad = np.concatenate([np.zeros(n), h, np.zeros(n)])
    # For each j, the coefficient is 1 - h[j-lag] - h[j+lag] for each lag
    # G[lag_idx, j] = 1 - h_pad[j-lag+n] - h_pad[j+lag+n]
    # This is an (2n-1) x n matrix
    G = np.ones((2*n-1, n))
    for j in range(n):
        G[:, j] -= h_pad[j - lags + n] + h_pad[j + lags + n]
    return scores, G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    # Minimize t
    # subject to: G @ p + scores <= t (for all rows) -> G @ p - t*1 <= -scores
    #             sum(p) = 0 (preserve sum)
    #             |p_j| <= radius
    # Decision variables: p (n) + t (1) = n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    # bounds
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
