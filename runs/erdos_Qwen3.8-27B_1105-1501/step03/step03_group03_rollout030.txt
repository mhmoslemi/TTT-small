import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores(h):
    """Return G matrix where G[r, j] = dS(r-n+1)/dh[j]"""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = d + n - 1
        if d >= 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:d+m])  # (1-h[i+d]) for i=0..m-1
            G[r, d:] += (-h[:m])           # -h[i] for i=0..m-1, affecting position i+d
        else:
            g = -d
            m = n - g
            G[r, g:] += (1.0 - h[:m])      # (1-h[i]) for i=0..m-1, affecting position i+g
            G[r, :m] += (-h[g:g+m])        # -h[i+g] for i=0..m-1
    return G

def _smooth_max_obj_and_grad(h, beta):
    scores = _scores(h)
    s_max = float(np.max(scores))
    obj = s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    G = _grad_scores(h)
    grad = G.T @ w
    return obj, grad

def _project_sum(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, scores, G, radius):
    """Find LP direction to reduce max score"""
    n = len(h)
    # min t s.t. G@p - t <= -scores, sum(p) = 0, |p| <= radius (element-wise)
    # Variables: p[0..n-1], t
    # n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0-h[j])))
    bounds.append((0.0, None))  # t >= 0 (actually t can be anything, but we want to minimize)
    # Actually t should be unbounded below (we want to minimize it)
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_c5(h, direction, c5_cur):

    n = len(h)

            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * direction)

    c5_opt = _exact_c5(h + a_opt * direction)

def _gradient_descent(h, target, deadline, beta=20.0, lr=0.01, momentum=0.8):
    """Nesterov momentum gradient descent on smooth max"""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    T = 0.01
    
    while time.time() < deadline:
        # Nesterov look-ahead
        y = h + momentum * v
        obj, grad = _smooth_max_obj_and_grad(y, beta)
        grad = grad - np.mean(grad)  # Remove component along constraint direction
        
        v = momentum * v + lr * grad
        h_new = _project_sum(h - v, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)
        
        delta = c5_new - c5_cur
        if delta < -1e-14:
            h = h_new
            lr = min(lr * 1.01, 0.2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            lr *= 0.6
            v *= 0.3
            if delta > 0 and T > 1e-8:

                    h = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
        T *= 0.997
    
    return best_h, best_c5

def _subgradient_step(h, target, step_size, top_k=3):
    """Take a subgradient step using the top-k active shifts"""
    scores = _scores(h)
    s_max = float(np.max(scores))
    # Find top-k shifts
    top_idx = np.argsort(scores)[-top_k:]
    active = scores[top_idx]
    # Weight by proximity to max
    w = np.exp(50 * (active - s_max))
    w /= np.sum(w)
    
    G = _grad_scores(h)
    grad = np.zeros(len(h))
    for i, idx in enumerate(top_idx):
        grad += w[i] * G[idx]
    grad -= np.mean(grad)
    
    h_new = h - step_size * grad
    h_new = _project_sum(h_new, target)
    return h_new

def _pair_descent(h, target, deadline, max_passes=50):

    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:

        improved = False
        for j in range(n):
            if time.time() > deadline:

                a_max = min(h[j], 1.0 - h[k])

        if not improved:

    return h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_scores(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = d + n - 1
        if d >= 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += (-h[g:g + m])
    return G


def _project_sum(v, target):
    n = len(v)
    if target <= 0:

    if target >= n:

    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
