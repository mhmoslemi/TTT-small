    best_h = h.copy()
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try a subset of pairs (or all pairs for small n)
        pairs_to_try = _select_pairs(h, n)
        for (i, j) in pairs_to_try:
            if time.time() > deadline:
                break
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-14:

            # Find optimal t
            t_opt = _optimize_pair(h, i, j, s, t_lo, t_hi)

                h_new[i] = t_opt
                h_new[j] = s - t_opt
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

            # c_base[mq] + L[mq]*p + Q[mq]*p^2 = c_base[ma] + L[ma]*p
            # Q[mq]*p^2 + (L[mq]-L[ma])*p + (c_base[mq]-c_base[ma]) = 0
            a_coef = Q[mq]

                if p_lo <= p1 <= p_hi:
                    candidates.append(p1)
                if p_lo <= p2 <= p_hi:

            best_p = p

        return None, None
    
    return best_p, best_val * dx

def _project(v, target):
    n = len(v)

        return np.zeros(n)

        return np.ones(n)
