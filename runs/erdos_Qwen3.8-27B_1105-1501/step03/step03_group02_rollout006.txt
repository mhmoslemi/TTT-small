            b = B[i] - B[j]
            c = A[i] - A[j]
            scale = max(1.0, abs(A[i]), abs(B[i]), abs(C[i]), abs(A[j]), abs(B[j]), abs(C[j]))
            if abs(a) > 1e-12 * scale:
                disc = b*b - 4.0*a*c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b + sqrt_disc) / (2.0 * a)
                    t2 = (-b - sqrt_disc) / (2.0 * a)
                    for t in [t1, t2]:
                        if t_lo - 1e-12 <= t <= t_hi + 1e-12:
                            candidates.append(t)
            elif abs(b) > 1e-12 * scale:
                t = -c / b
                if t_lo - 1e-12 <= t <= t_hi + 1e-12:
                    candidates.append(t)
    
    # Deduplicate and clip
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_lo, t_hi)
    candidates = np.unique(np.round(candidates, 12))
    
    # Evaluate
    best_t = 0.0
    best_val = current_raw
    for t in candidates:
        vals = A + B*t + C*t*t
        mx = float(np.max(vals))
        if mx < best_val - 1e-14:
            best_val = mx
            best_t = t
    
    return best_t, best_val

            scores = _scores(h)
            current_raw = float(np.max(scores))
            dirs = []
            gm = _smooth_max_dir(h, scores, final_beta, final_topk)
            if gm is not None:
                dirs.append(gm)
            # Try all top 10 individual gradients
            ti = np.argsort(scores)[::-1][:10]
            for ki in ti:
                lag = int(ki) - (n - 1)
                g = _grad_for_lag(h, lag)
                g -= np.mean(g)
                mx = float(np.max(np.abs(g)))
                if mx > 1e-14:
                    dirs.append(g / mx)
            # Combinatorial: top 2 gradients summed
            if len(ti) >= 2:
                g1 = _grad_for_lag(h, int(ti[0]) - (n - 1))
                g2 = _grad_for_lag(h, int(ti[1]) - (n - 1))
                gc = g1 + g2
                gc -= np.mean(gc)
                mx = float(np.max(np.abs(gc)))
                if mx > 1e-14:
                    dirs.append(gc / mx)
            rd = _random_dir(n, rng)
            if rd is not None:
                dirs.append(rd)

            imp = False
            for d in dirs:
                if time.time() > deadline - 0.5:
                    break
                d = np.asarray(d, dtype=float).copy()
                d -= np.mean(d)
                mx = float(np.max(np.abs(d)))
                if mx < 1e-14:
                    continue
                d /= mx
                t_star, raw_star = _exact_1d(h, d, current_raw)
                if raw_star < current_raw - 1e-12:
                    h_new = np.clip(h + t_star * d, 0.0, 1.0)
                    h_new = _repair(h_new, target)
                    c5_new = _c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        h = h_new
                        best_h = h.copy()
                        best_c5 = c5_new
                        imp = True
                        final_stall = 0
                        if best_c5 <= 0.3808 + 1e-9:
                            return best_h, best_c5, n
                        break
            if not imp:
                final_stall += 1
                if final_stall % 5 == 0:
                    final_beta = max(20.0, final_beta * 0.8)
                    final_topk = max(10, final_topk - 5)
                if final_stall % 12 == 0:
                    h = _repair(best_h + rng.normal(0, 0.01, n), target)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair(best_h, target)
    final_c5 = _c5(best_h)

    return best_h, final_c5, n
