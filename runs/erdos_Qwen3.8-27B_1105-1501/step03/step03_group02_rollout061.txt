def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0: mask = x < 1.0 - 1e-12
        else: mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

One more optimization: instead of calling np.correlate for each coordinate (which is O(n²)), I can precompute the scores and update them incrementally. When I change h[p] from old to new, only the scores involving h[p] change. The change in score r is:
ΔS[r] = c_r * (new - old) + d_r * (new² - old²)

where c_r and d_r are the coefficients I already compute. So I can maintain a running score array and update it in O(n) per coordinate instead of O(n²).

```python
def _coord_sweep_fast(h, order, n_grid=200):
    n = len(h)
    # Precompute scores
    omh = 1.0 - h
    scores = np.correlate(h, omh, mode='full')
    
    for p in order:
        p = int(p)
        old = h[p]
        
        # Compute coefficients c, d for this coordinate
        c = np.zeros(2 * n - 1)
        d = np.zeros(2 * n - 1)
        c[n-1] = 1.0
        d[n-1] = -1.0
        for k in range(1, n):
            coeff = 0.0
            if p >= k:
                coeff += (1.0 - h[p - k])
            if p < n - k:
                coeff -= h[p + k]
            c[n - 1 + k] = coeff
            
            coeff = 0.0
            if p + k < n:
                coeff += (1.0 - h[p + k])
            if p >= k:
                coeff -= h[p - k]
            c[n - 1 - k] = coeff
        
        # Current scores as function of t = h[p]:
        # S[r](t) = scores[r] - c[r]*old - d[r]*old^2 + c[r]*t + d[r]*t^2
        #         = (scores[r] - c[r]*old - d[r]*old^2) + c[r]*t + d[r]*t^2
        
        base = scores - c * old - d * old * old
        
        # Minimize max_r (base[r] + c[r]*t + d[r]*t^2) over [0,1]
        ts = np.linspace(0.0, 1.0, n_grid)
        best_t = 0.0
        best_g = np.inf
        for t in ts:
            g_val = np.max(base + c * t + d * t * t)
            if g_val < best_g:
                best_g = g_val
                best_t = t
        
        # Golden section
        lo = max(0.0, best_t - 1.0/n_grid)
        hi = min(1.0, best_t + 1.0/n_grid)
        if hi > lo:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            gc1 = hi - gr * (hi - lo)
            gc2 = lo + gr * (hi - lo)
            f1 = np.max(base + c*gc1 + d*gc1*gc1)
            f2 = np.max(base + c*gc2 + d*gc2*gc2)
            for _ in range(60):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi = gc2; gc2 = gc1; f2 = f1
                    gc1 = hi - gr*(hi-lo)
                    f1 = np.max(base + c*gc1 + d*gc1*gc1)
                else:
                    lo = gc1; gc1 = gc2; f1 = f2
                    gc2 = lo + gr*(hi-lo)
                    f2 = np.max(base + c*gc2 + d*gc2*gc2)
            best_t = 0.5*(lo+hi)
        
        # Update scores incrementally
        new = best_t
        if abs(new - old) > 1e-15:
            scores += c * (new - old) + d * (new*new - old*old)
            h[p] = new
    
    return h, scores
```

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_coeffs(h, p):
    n = len(h)
    c = np.zeros(2 * n - 1)
    d = np.zeros(2 * n - 1)
    c[n - 1] = 1.0
    d[n - 1] = -1.0
    for k in range(1, n):

            coeff += 1.0 - h[p - k]

            coeff += 1.0 - h[p + k]
        if p >= k:
            coeff -= h[p - k]
        c[n - 1 - k] = coeff
    return c, d

    g = np.max(sm, axis=1)
    best_idx = int(np.argmin(g))
    best_t = ts[best_idx]

        return best_t
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(base + c * c1 + d * c1 * c1))

            c2 = c1

        else:

            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(base + c * c2 + d * c2 * c2))
    return 0.5 * (lo + hi)

def _coord_sweep(h, order, n_grid=200):
    n = len(h)
    omh = 1.0 - h
    scores = np.correlate(h, omh, mode='full').copy()
    
    for idx in range(len(order)):
        p = int(order[idx])
        old = h[p]
        c, d = _compute_coeffs(h, p)
        base = scores - c * old - d * old * old
        t_opt = _find_opt_t(base, c, d, n_grid)
        new = t_opt
        if abs(new - old) > 1e-14:
            scores += c * (new - old) + d * (new * new - old * old)
            h[p] = new
    
    return h, scores

def _smooth_grad_step(h, scores, step_size=0.05, beta=50.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    
    # Compute gradient of each score with respect to h
    # grad[r, i] = ∂S[r]/∂h[i]
    # This is expensive to build fully, so compute iteratively
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        gw = w[r]
        if gw < 1e-15:
            continue
        if k == 0:
            grad += gw * (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            if m > 0:
                grad[:m] += gw * (-h[k:k+m])
                grad[k:] += gw * (1.0 - h[:m])
        else:
            g = -k
            m = n - g
            if m > 0:
                grad[:m] += gw * (1.0 - h[g:g+m])
                grad[g:] += gw * (-h[:m])
    
    # Negative gradient direction (descent)
    grad -= np.mean(grad)  # zero-mean for feasibility
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return h
    direction = -grad / norm
    h_new = h + step_size * direction
    return _project(h_new, n / 2.0)

def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
