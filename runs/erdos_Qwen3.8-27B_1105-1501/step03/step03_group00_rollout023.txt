def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _softmax_dir(h, scores, G, beta):
    """Get direction from softmax-weighted gradient."""
    mx = scores.max()
    w = np.exp(beta * (scores - mx))
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    grad = G.T @ w
    grad -= np.mean(grad)  # ensure sum-zero direction
    return grad

def _active_lp_dir(h, scores, G, radius, k, tol=0.001):
    """LP direction minimizing max over active lags."""
    n = len(h)
    mx = scores.max()
    threshold = mx - tol
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(scores)[-k:]
    elif len(active) > k:
        active = np.argsort(scores)[-k:]

    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0, radius * 2))  # t >= 0 and bounded
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < radius:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anneal(h, deadline, t_init=0.003, t_min=0.0001, n_iter=None):
    """Simulated annealing with pair transfers."""
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    if n_iter is None:
        n_iter = 50000
    
    t = t_init
    cooling = (t_min / t_init) ** (1.0 / max(1, n_iter - 1))
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t *= cooling
        
        # Pick two random indices
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Random transfer amount
        delta = rng.uniform(0.01, 0.1)
        sign = 1 if rng.random() < 0.5 else -1
        
        h_new_val_i = h[i] + sign * delta
        h_new_val_j = h[j] - sign * delta
        
        if h_new_val_i < 0 or h_new_val_i > 1 or h_new_val_j < 0 or h_new_val_j > 1:
            continue
        
        h[i] = h_new_val_i
        h[j] = h_new_val_j
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Accept with probability exp(-(new_c5 - cur_c5) / t)
            if t > t_min and rng.random() < np.exp(-(new_c5 - cur_c5) / t):
                cur_c5 = new_c5
            else:
                # Reject: undo
                h[i] -= sign * delta
                h[j] += sign * delta
    
    return best_h, best_c5

def _equalize_top(k, h, deadline):

            break

        top_k = np.argsort(scores)[-k:]
        max_s = scores.max()
        min_top = scores[top_k].min()
        
        if max_s - min_top < 1e-6:
            break
        
        # Direction: decrease top scores, increase bottom of top-k
        # Simple approach: gradient of (max_s - min_top)^2
        G = _grad_matrix(h)
        # We want to reduce variance of top-k scores
        # Approximate: move in direction that reduces max and increases min of top-k
        grad_max = G[np.argmax(scores)]
        min_idx = top_k[np.argmin(scores[top_k])]
        grad_min = G[min_idx]
        p = grad_max - grad_min  # direction that reduces max and increases min
        p -= np.mean(p)

        if pn < 1e-15:
            break
        p = p / pn
        
        alpha, new_c5 = _line_search(h, p, best_c5)
        h_new = _project(h + alpha * p, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h_new.copy()
