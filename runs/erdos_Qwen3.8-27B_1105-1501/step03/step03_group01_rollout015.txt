import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    v = np.clip(v, -2.0, 2.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_direction(h, tau=50.0):
    """Compute subgradient descent direction (mean-zero)."""
    n = len(h)
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores_scaled = scores_raw * (2.0 / n)
    max_s = float(np.max(scores_scaled))
    exp_s = np.exp(tau * (scores_scaled - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # Compute weighted sum of gradients
    # G[r, j] = [0<=j-k<n]*(1-h[j-k]) + [0<=j+k<n]*(-h[j+k]) where k=r-(n-1)
    grad = np.zeros(n)
    j = np.arange(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-15:
            continue
        k = r - (n - 1)
        # term1: [0<=j-k<n] * (1-h[j-k])
        mask1 = (j >= k) & (j - k < n)
        if np.any(mask1):
            idx1 = j[mask1] - k
            grad += wr * (1.0 - h[idx1]) * mask1
        # term2: [0<=j+k<n] * (-h[j+k])
        mask2 = (j + k >= 0) & (j + k < n)
        if np.any(mask2):
            idx2 = j[mask2] + k
            grad += wr * (-h[idx2]) * mask2
    
    # Make mean-zero
    grad -= np.mean(grad)
    return -grad  # descent direction

def random_direction(n, rng):
    P = rng.standard_normal(n)
    P -= P.mean()
    return P

def feasible_interval(h, P):
    """Find [t_min, t_max] such that h + t*P stays in [0,1]."""
    t_min = 0.0
    t_max = 0.0
    for j in range(len(h)):
        if P[j] > 1e-15:
            t_min = max(t_min, -h[j] / P[j])
            t_max = min(t_max, (1.0 - h[j]) / P[j])
        elif P[j] < -1e-15:
            t_min = max(t_min, (h[j] - 1.0) / P[j])
            t_max = min(t_max, -h[j] / P[j])
    if t_max <= t_min:
        return 0.0, 0.0
    return t_min, t_max

def golden_search_1d(f, a, b, tol=1e-10, max_iter=50):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def fast_line_obj(a, b, c, dx, t):

    vals = a + t * b + t * t * c

def compute_coeffs(h, P):
    """Compute a, b, c for the quadratic line objective."""
    dx = 2.0 / len(h)
    a = np.correlate(h, 1.0 - h, mode='full')
    b = np.correlate(P, 1.0 - h, mode='full') - np.correlate(h, P, mode='full')
    c = -np.correlate(P, P, mode='full')

def get_feasible_interval(h, P):

    w = exp_s / np.sum(exp_s)

            D += wr * (1.0 - h[j[m1] - k]) * m1
        # Term 2: [0 <= j+k < n] * (-h[j+k])
        m2 = (j + k >= 0) & (j + k < n)
        if np.any(m2):
            D += wr * (-h[j[m2] + k]) * m2
    
    D -= D.mean()
    return -D  # descent direction

def smooth_dir(n, rng, freq=None, phase=None):
    x = np.arange(n) / n
    if freq is None:
        freq = rng.integers(1, 6)
    if phase is None:
        phase = rng.uniform(0, 2 * np.pi)
    P = np.sin(2 * np.pi * freq * x + phase)
    P += 0.3 * np.sin(2 * np.pi * (freq + 1) * x + phase * 1.3)
    P -= P.mean()
    return P
