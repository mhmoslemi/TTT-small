def _project(v, target):

    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)

        return np.ones(n)

        else:

def _active_lp(h, G_sub, s_sub, radius):

    K = len(s_sub)
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n+1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -s_sub
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j] + 1e-12)
        hi_j = min(radius, 1.0 - h[j] - 1e-12)
        if lo_j >= hi_j:
            lo_j, hi_j = 0.0, 0.0
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p, c5_cur, n_coarse=40):
    n = len(h)
    alpha_max = 1.0

        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.0, alpha_max, n_coarse)
    for a in alphas:
        if a < 1e-12:
            continue
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)

            c1 = hi - gr * (hi - lo)

        else:

            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)

    opt_c5 = _exact_c5(h + opt_a * p)

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_obj(h, beta):
    scores = _scores(h)
    s_max = float(np.max(scores))
    # Add small variance penalty for top scores
    top_k = min(len(scores), 10)
    s_top = np.sort(scores)[::-1][:top_k]
    s_mean = float(np.mean(s_top))
    s_var = float(np.var(s_top)) / max(s_mean, 1e-10)
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta + 0.1 * s_var

def _smooth_jac(h, beta):
    scores, G = _scores_and_grad(h)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    return G.T @ w

def _sum_constraint(h, target):
    return np.sum(h) - target

def _slsqp_pass(h, target, beta, max_iter=100):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': lambda x: np.ones(len(x)), 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, args=(beta,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and _exact_c5(res.x) < _exact_c5(h):
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()
