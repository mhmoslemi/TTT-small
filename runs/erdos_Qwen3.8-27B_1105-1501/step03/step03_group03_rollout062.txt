def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    """Compute the gradient matrix G where G[d, i] = ds_d/dh_i."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d  # row index in output
        if d >= 0:
            m = n - d
            if m > 0:
                # s_r = sum_{j=0}^{m-1} h[j+d]*(1-h[j])

                #          for i in [0, m-1] (as h[j]): += -h[j+d] where j=i
                G[r, d:] += (1.0 - h[:m])  # h[i] as h[j+d], j=i-d, so i from d to d+m-1=n-1
                G[r, :m] -= h[d:]           # h[i] as h[j], j=i, so i from 0 to m-1

                #          for i in [g, n-1] (as h[j+g]): += -h[j] where j=i-g
                G[r, :m] += (1.0 - h[g:])   # h[i] as h[j], j=i
                G[r, g:] -= h[:m]            # h[i] as h[j+g], j=i-g
    return G

from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _coord_descent(h, n_sweeps=50, grid_size=64, rng=None, deadline=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    t_grid = np.linspace(0.0, 1.0, grid_size)
    d0_idx = n - 1
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for i in order:
            old_val = h[i]
            h[i] = 0.0
            s0 = _scores(h)
            h[i] = 1.0
            s1 = _scores(h)
            h[i] = old_val
            
            # Linear interpolation for non-d0 scores
            delta = s1 - s0  # shape (2n-1,)
            f = s0 + t_grid[:, None] * delta  # shape (grid_size, 2n-1)
            f[:, d0_idx] = s0[d0_idx] + t_grid - t_grid**2
            fmax = np.max(f, axis=1)  # shape (grid_size,)
            
            idx = np.argmin(fmax)
            best_t = t_grid[idx]
            
            # Refine
            lo_i = max(0, idx - 1)
            hi_i = min(grid_size - 1, idx + 1)
            t_fine = np.linspace(t_grid[lo_i], t_grid[hi_i], 32)
            f_fine = s0 + t_fine[:, None] * delta
            f_fine[:, d0_idx] = s0[d0_idx] + t_fine - t_fine**2
            fmax_fine = np.max(f_fine, axis=1)
            best_t = t_fine[np.argmin(fmax_fine)]
            
            h[i] = best_t
        
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

    n = len(h)

        # Only use top-k scores

        G_sub = G[idx, :]
        s_sub = scores[idx]
    else:
        G_sub = G
        s_sub = scores

    c[-1] = 1.0
    A_ub = np.zeros((G_sub.shape[0], n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -s_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        pass
    return None
