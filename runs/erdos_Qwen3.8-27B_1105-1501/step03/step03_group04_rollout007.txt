from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores_weighted(h, w):
    """Compute sum_m w[m] * d(corr[m])/d(h) for given weights w."""
    n = len(h)
    grad = np.zeros(n)
    # k=0 (m = n-1): grad[j] += w[n-1] * (1-2*h[j])
    grad += w[n-1] * (1.0 - 2.0 * h)
    # k > 0 (m = n-1+k, k=1..n-1)
    for k in range(1, n):
        m = n - 1 + k
        wk = w[m]
        if wk == 0.0:
            continue
        # (1-h[j-k]) for j in [k, n-1]
        grad[k:] += wk * (1.0 - h[:n-k])
        # (-h[j+k]) for j in [0, n-1-k]
        grad[:n-k] -= wk * h[k:]
    # k < 0 (m = n-1-k, k=1..n-1, so g=k means m=n-1-g)
    for g in range(1, n):
        m = n - 1 - g
        wk = w[m]
        if wk == 0.0:
            continue
        # (1-h[j+g]) for j in [0, n-1-g]
        grad[:n-g] += wk * (1.0 - h[g:])
        # (-h[j-g]) for j in [g, n-1]
        grad[g:] -= wk * h[:n-g]
    return grad

def _logit_to_h(z, target):
    """Convert centered logit z to h with sum = target."""
    n = len(z)
    zc = z - np.mean(z)
    # Solve for c: sum sigmoid(zc_i + c) = target
    lo = -20.0
    hi = 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(1.0 / (1.0 + np.exp(-(zc + mid)))))
        if s > target:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    h = 1.0 / (1.0 + np.exp(-(zc + c)))
    # Fix any numerical issues
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 1e-10, 1.0 - 1e-10)
    return h

def _h_to_logit(h):
    """Convert h to centered logit representation."""
    eps = 1e-7
    h_safe = np.clip(h, eps, 1.0 - eps)
    z = np.log(h_safe) - np.log(1.0 - h_safe)
    return z - np.mean(z)

def _logsumexp_obj_grad(z, target, beta):
    """Compute log-sum-exp objective and gradient in z-space."""
    n = len(z)
    h = _logit_to_h(z, target)
    s = _scores(h)
    M = float(np.max(s))
    # Stable log-sum-exp
    shifted = beta * (s - M)
    exps = np.exp(shifted)
    Z = float(np.sum(exps))
    obj = M + np.log(Z) / beta
    
    # Weights
    w = exps / Z
    
    # Gradient wrt h
    gh = _grad_scores_weighted(h, w)
    
    # Chain rule through logit
    a = h * (1.0 - h)
    A = float(np.sum(a))
    S = float(np.dot(gh, a)) / A
    gz = a * (gh - S)
    
    return obj, gz

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize_z_lbfgs(z, n, target, deadline, betas, max_iters_list):
    """Run L-BFGS with multi-beta schedule starting from z."""
    best_z = z.copy()
    cur_c5 = _exact_c5(_logit_to_h(z, target))
    
    for beta, max_iter in zip(betas, max_iters_list):
        if time.time() > deadline:
            break
        # Run L-BFGS
        try:
            res = minimize(
                _lbfgs_fun,
                z,
                args=(n, target, beta),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'ftol': 1e-10, 'gtol': 1e-6, 'maxls': 20}
            )
            if res.success or res.fun < 1e10:
                h_try = _logit_to_h(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5:
                    cur_c5 = c5_try
                    best_z = res.x.copy()
                z = res.x.copy()
            else:
                z = best_z.copy()
        except Exception:
            pass
    
    return best_z, cur_c5
