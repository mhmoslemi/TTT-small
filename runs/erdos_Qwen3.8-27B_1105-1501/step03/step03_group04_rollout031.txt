def _exact_c5(h):

    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

def _subgrad_step(h, scores, G, top_k=5):
    """Take one subgradient step using top-k lags."""
    n = len(h)
    max_score = float(np.max(scores))
    # Find lags with scores close to max
    threshold = max_score - 1e-6  # active set
    active = np.where(scores >= threshold)[0]
    if len(active) > top_k:
        # Use only top-k
        top_idx = np.argsort(scores)[-top_k:]
        active = top_idx
    # Average gradient of active lags
    grad = G[active].mean(axis=0)
    # Remove mean (for sum constraint)
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return None
    return -grad / gnorm

    best_h = h.copy()
    best_c5 = cur_c5

        if time.time() > deadline:
            break
        # Cosine annealing
        frac = min(it / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * (1 + np.cos(np.pi * frac)) / 2

        weights = exps / float(np.sum(exps))

        # Backtracking line search
        alpha = lr
        improved = False

            G[r_neg, d:] -= h[:n - d]
    return G

def _soft_descent(h, deadline, beta_start=40.0, beta_end=8000.0, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.06
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * (1.0 + np.cos(np.pi * frac)) / 2.0
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        s_exp = float(np.sum(exps))
        if s_exp < 1e-300:
            break
        weights = exps / s_exp
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(22):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.35)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.78
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _hard_subgrad(h, deadline, top_k=3, max_iters=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.015
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        m = float(np.max(scores))
        active = np.where(scores >= m - 1e-8)[0]
        if len(active) > top_k:
            idx = np.argsort(scores)[::-1][:top_k]
            active = idx
        G = _grad_matrix(h)
        grad = G[active].mean(axis=0)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(18):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.01, 0.12)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.88
            if lr < 1e-6 or stall > 300:
                break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, p, c5_cur, target):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.1, alpha_max, 10)
    for a in alphas:
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.6)
    hi_a = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-11:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.06, max_iters=8000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.82
            if radius < 1e-5:
                break
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.05
            continue
        alpha, new_c5 = _line_search_exact(h, p, cur_c5, target)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.75
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.04
    return best_h, best_c5

def _pair_swap(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.02, 0.05, 0.1, 0.15]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                for delta in deltas:
                    for sign in [1.0, -1.0]:
                        d = sign * delta
                        if h[i] + d < 0 or h[i] + d > 1:
                            continue
                        if h[j] - d < 0 or h[j] - d > 1:
                            continue
                        h_try = h.copy()
                        h_try[i] += d
                        h_try[j] -= d
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _alternating_descent(h, deadline, max_rounds=100):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    soft_time_frac = 0.6
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        t_soft = time.time() + min((deadline - time.time()) * soft_time_frac, 15.0)
        h, c5 = _soft_descent(h, t_soft, beta_start=100.0, beta_end=5000.0, max_iters=3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        t_hard = min(deadline, time.time() + 10.0)
        h, c5 = _hard_subgrad(h, t_hard, top_k=3, max_iters=2000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Soft descent from parent with cosine annealing
    deadline1 = t0 + budget_s * 0.22
    h, c5 = _soft_descent(h, deadline1, beta_start=30.0, beta_end=6000.0, max_iters=15000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP polish
    deadline2 = t0 + budget_s * 0.38
    h, c5 = _lp_polish(h, deadline2, radius_init=0.08, max_iters=6000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Alternating soft-hard descent
    deadline3 = t0 + budget_s * 0.55
    h, c5 = _alternating_descent(h, deadline3, max_rounds=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Perturbation restarts
    deadline4 = t0 + budget_s * 0.72
    scale = 0.04
    for attempt in range(500):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        t_rem = deadline4 - time.time()
        t_alloc = min(t_rem / max(500 - attempt, 1), budget_s * 0.015)
        h_try, c5_try = _soft_descent(h_pert, time.time() + t_alloc,
                                      beta_start=80.0, beta_end=4000.0, max_iters=2000)
        if time.time() < deadline4 - 5:
            h_try, c5_try = _lp_polish(h_try, min(deadline4, time.time() + 5.0),
                                        radius_init=0.04, max_iters=500)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.012, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.08)

    # Phase 5: LP polish on best
    deadline5 = t0 + budget_s * 0.82
    h, c5 = _lp_polish(best_h, deadline5, radius_init=0.05, max_iters=5000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Try nearby n values
    deadline6 = t0 + budget_s * 0.92
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        t_alloc = min(budget_s * 0.012, (deadline6 - time.time()) / 2)
        h_r, c5_r = _soft_descent(h_r, time.time() + t_alloc,
                                  beta_start=60.0, beta_end=5000.0, max_iters=1500)
        t_alloc2 = min(budget_s * 0.008, (deadline6 - time.time()) / 2)
        h_r, c5_r = _lp_polish(h_r, time.time() + t_alloc2,
                                radius_init=0.04, max_iters=500)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Pair swap local search
    if time.time() < t0 + budget_s * 0.97:
        h_cd, c5_cd = _pair_swap(best_h, t0 + budget_s * 0.98, max_passes=30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.985:
        h_f, c5_f = _lp_polish(best_h, t0 + budget_s * 0.995,
                                radius_init=0.02, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
