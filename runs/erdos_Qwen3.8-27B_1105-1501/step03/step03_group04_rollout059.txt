def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)

    hi = float(v.max()) + 1.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))

            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _tm_init(n, amp=0.25):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amp * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)

def _sine_init(n, freq=1, amp=0.3):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
    return _project(h, n / 2.0)

def _binary_init(n, rng, pattern='tm'):
    if pattern == 'tm':
        tm = np.zeros(n, dtype=int)
        for i in range(n):
            tm[i] = bin(i).count('1') % 2
        h = np.where(tm == 0, 1.0, 0.0).astype(float)
    else:
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _smooth_descent(h, deadline, beta_start=30.0, beta_end=3000.0, max_iters=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.1
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        grad = np.zeros(n)
        for kk in range(2 * n - 1):
            if weights[kk] > 1e-10:
                d = kk - (n - 1)
                grad += weights[kk] * _grad_lag(h, d)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        improved = False
        for _ in range(20):
            h_try = _project(h + lr * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.04, 0.5)
                improved = True
                break
            else:
                lr *= 0.6
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-6 or stall > 60:
                break
    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for delta in deltas:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < -1e-10 or h[j] + d > 1.0 + 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    comp_mask = np.ones(n, dtype=bool)
                    comp_mask[j] = False
                    if d > 0:
                        comp = int(np.argmax(np.where(comp_mask, 1.0 - h_try, -2.0)))
                    else:
                        comp = int(np.argmax(np.where(comp_mask, h_try, -2.0)))
                    if comp == j:
                        continue
                    h_try[comp] -= d
                    if h_try[comp] < -1e-10 or h_try[comp] > 1.0 + 1e-10:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved and j % 16 == 0:
                pass
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Top-k descent with momentum (25% time)
    deadline1 = t0 + budget_s * 0.22
    h1, c5_1 = _topk_descent(h, deadline1, k_init=1, k_max=12, lr_init=0.12, momentum=0.88)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 1b: Smooth descent continuation
    deadline1b = t0 + budget_s * 0.30
    h1b, c5_1b = _smooth_descent(best_h, deadline1b, beta_start=20.0, beta_end=2000.0, max_iters=8000)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()

    # Phase 2: Multi-resolution (20% time)
    deadline2 = t0 + budget_s * 0.50
    n_list = sorted(set([best_n - 8, best_n - 4, best_n - 2, best_n,
                         best_n + 2, best_n + 4, best_n + 8,
                         best_n - 16, best_n + 16]))
    n_list = [n for n in n_list if 24 <= n <= 160]
    time_per_n = (deadline2 - time.time()) / max(len(n_list), 1)

    for n_try in n_list:
        if time.time() > deadline2:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.02:
            continue
        tl = max(1.0, min(time_per_n * 0.7, budget_s * 0.03))
        h_r, c5_r = _topk_descent(h_r, time.time() + tl, k_init=1, k_max=10, lr_init=0.1, momentum=0.85)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Basin hopping with perturbations (25% time)
    deadline3 = t0 + budget_s * 0.75
    scale = 0.05
    attempts = 0
    max_attempts = 200
    while attempts < max_attempts and time.time() < deadline3:
        attempts += 1
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        t_per = max(0.5, (deadline3 - time.time()) / max(max_attempts - attempts + 1, 1))
        t_per = min(t_per, budget_s * 0.015)
        h_try, c5_try = _topk_descent(h_pert, time.time() + t_per, k_init=1, k_max=8, lr_init=0.08, momentum=0.8, max_stall=30)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)

    # Phase 3b: Structured perturbations (shifts, flips)
    deadline3b = t0 + budget_s * 0.80
    for shift in range(1, min(6, best_n // 4)):
        if time.time() > deadline3b:
            break
        h_shift = np.roll(best_h, shift)
        h_shift = _project(h_shift, best_n / 2.0)
        c5_s = _exact_c5(h_shift)
        if c5_s < best_c5 - 1e-10:
            best_c5 = c5_s
            best_h = _project(h_shift, best_n / 2.0)
            h_s, c5_s2 = _topk_descent(h_shift, time.time() + budget_s * 0.008, k_init=1, k_max=6, lr_init=0.05, momentum=0.8, max_stall=20)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s.copy()

    # Phase 4: LP polish (12% time)
    deadline4 = t0 + budget_s * 0.92
    h4, c5_4 = _lp_polish(best_h, deadline4, radius_init=0.06, max_stall=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Coordinate descent final polish
    deadline5 = t0 + budget_s * 0.97
    h5, c5_5 = _coord_descent(best_h, deadline5, max_passes=30)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: One more short LP pass
    if time.time() < t0 + budget_s * 0.99:
        h6, c5_6 = _lp_polish(best_h, t0 + budget_s * 0.995, radius_init=0.03, max_stall=15)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final validation and cleanup
    best_h = _project(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
