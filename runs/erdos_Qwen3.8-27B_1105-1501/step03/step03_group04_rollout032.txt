                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    """Project v onto {x : 0 ≤ x ≤ 1, sum(x) = target}"""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_grad_descent(h, deadline, beta_start=20.0, beta_end=3000.0, 
                        max_iters=20000, lr_init=0.1, lr_end=0.01):
    """Softmax-smoothed gradient descent with adaptive learning rate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        elapsed_frac = min(1.0, it / max_iters)
        beta = beta_start * (beta_end / beta_start) ** elapsed_frac
        lr_cur = lr_init * (lr_end / lr_init) ** elapsed_frac
        
        scores = _scores(h)
        m = float(np.max(scores))
        # Numerical stability
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        
        G = _grad_matrix(h)
        grad = G.T @ weights
        # Remove the mean component (it's in the null space of the constraint)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        
        # Line search
        alpha = lr_cur
        improved = False
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr *= 1.03  # Slightly increase if improving
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _top_k_descent(h, deadline, top_k=3, max_iters=10000):
    """Focus on the top k lags and do subgradient-like steps."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        # Get top-k lags
        top_indices = np.argsort(scores)[-top_k:]
        top_scores = scores[top_indices]
        # Weights proportional to (score - min of top_k + epsilon)
        w = top_scores - top_scores.min() + 0.1
        w /= w.sum()
        
        # Average gradient over top-k lags
        G_top = _grad_matrix(h)[top_indices]  # (top_k, n)
        grad = G_top.T @ w  # (n,)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        
        alpha = lr
        improved = False
        for _ in range(15):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-6 or stall > 50:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, kind='sine'):
    """Generate structured initializations."""
    if kind == 'sine1':
        h = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n)
    elif kind == 'sine2':
        h = 0.5 + 0.25 * np.cos(4 * np.pi * np.arange(n) / n)
    elif kind == 'sine3':
        h = 0.5 + 0.2 * np.cos(6 * np.pi * np.arange(n) / n)
    elif kind == 'tm':
        # Thue-Morse
        tm = np.zeros(n, dtype=int)
        for i in range(n):
            tm[i] = bin(i).count('1') % 2
        h = 0.5 + 0.25 * (2.0 * tm - 1.0)
    elif kind == 'bump':
        # Concentrated in middle
        x = np.arange(n) / n
        h = 0.5 + 0.3 * np.exp(-10 * (x - 0.5)**2)
    else:
        h = np.full(n, 0.5)
    return _project(h, n / 2.0)

def _perturb_restart(h, rng, scale):
    """Create a perturbed version of h."""
    n = len(h)
    h_pert = h + rng.normal(0, scale, n)
    return _project(h_pert, n / 2.0)

def _local_refine(h, deadline):
    """Final exact local refinement using small perturbations."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Try small perturbations in random directions
    rng = np.random.default_rng(12345)
    for _ in range(200):
        if time.time() > deadline:
            break
        # Random direction in the constraint space (sum = 0)
        d = rng.normal(0, 1, n)
        d -= d.mean()
        d /= np.linalg.norm(d)
        
        # Line search in both directions
        for sign in [1.0, -1.0]:
            best_a = 0.0
            best_val = best_c5
            for a in [0.005, 0.01, 0.02, 0.05, 0.1]:
                h_try = _project(sign * a * d + h, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_val:
                    best_val = c5_try
                    best_a = a
            if best_a > 0 and best_val < best_c5 - 1e-12:
                h = _project(sign * best_a * d + h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
    
    # Coordinate-wise refinement
    for _pass in range(5):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            # Try small moves
            for delta in [0.005, 0.01, 0.02, 0.05]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    # Compensate
                    if d > 0:
                        comp = int(np.argmax(np.where(np.arange(n) != j, 1.0 - h_try, -1)))
                    else:
                        comp = int(np.argmax(np.where(np.arange(n) != j, h_try, -1)))
                    if comp == j:
                        continue
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break
        # Check if we improved
        if _exact_c5(h) >= best_c5 - 1e-12:
            break
    
    return best_h, best_c5

def _newton_like_step(h, scores, G, beta):
    """Compute a Newton-like step for the smoothed objective."""
    n = len(h)
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    weights = exps / float(np.sum(exps))
    
    grad = G.T @ weights
    grad -= np.mean(grad)
    
    # Hessian approximation: H = β * (G^T D G - G^T p p^T G) where D = diag(weights) - outer(p,p)
    # This is expensive for large n, so use a simplified version
    # For now, just return the gradient step with a heuristic step size
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-14:
        return None
    return -grad / gnorm
