    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_for_lag(d, n, h):
    """Gradient of the correlation at lag d with respect to h.
    d ranges from -(n-1) to (n-1).
    corr[d + n - 1] = sum of h[i] * (1-h[i+d]) for valid i (d>0)
    or sum of h[i] * (1-h[i+d]) for valid i (d<0)
    """
    g = np.zeros(n)
    if d == 0:
        # corr = sum h[i]*(1-h[i]), grad = 1-2h[i]
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        # corr = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        # d(corr)/d(h[j]) = (1-h[j+d]) for j=0..n-1-d  (from the h[i] term)
        #                  - h[j-d] for j=d..n-1       (from the (1-h[i]) term)
        m = n - d
        g[:m] += (1.0 - h[d:])  # contribution from h[i] factor
        g[d:] -= h[:m]          # contribution from (1-h[i+d]) factor
    else:
        e = -d
        # corr = sum_{i=e}^{n-1} h[i] * (1-h[i-e])
        # = sum_{i=0}^{n-1-e} h[i+e] * (1-h[i])
        m = n - e
        g[e:] += (1.0 - h[:m])  # from h[i+e] factor
        g[:m] -= h[e:]           # from (1-h[i]) factor
    return g

def _project(v, target):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - tau, 0.0, 1.0)

        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _corr_scores(h)
        # Find top-k lags
        top_idx = np.argsort(scores)[-top_k:]
        max_score = scores[top_idx[0]] if top_idx[0] == np.argmax(scores) else scores[np.argmax(scores)]
        max_score = float(np.max(scores))
        
        # Weighted combination of gradients for active lags
        weights = np.zeros(2*n-1)
        for idx in top_idx:
            w = (scores[idx] - max_score + 1.0) / top_k  # strongest for the max

        if wsum < 1e-12:
            break
        weights /= wsum

        grad = np.zeros(n)
        for idx in top_idx:

        # Normalize and take step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        
        # Line search
        alpha = lr
        improved = False

        if not improved:
            stall += 1
            lr *= 0.8
            if lr < 1e-6 or stall > 40:
                break
    
    return best_h, best_c5

def _lp_direction(h, scores, radius):

    G = np.zeros((2 * n - 1, n))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n + 1))
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:

                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def _binary_search_low_n(n, deadline, rng):
    """Try to find good binary sequences at given n."""
    target = n // 2
    best_h = None
    best_c5 = 1.0
    for trial in range(200):
        if time.time() > deadline:
            break
        # Generate random binary with exactly target ones
        idx = rng.permutation(n)
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
        c5_bin = _exact_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            # Try local improvements on the binary
            improved = True
            while improved and time.time() < deadline:
                improved = False
                for _swap in range(n * 10):
                    if time.time() > deadline:
                        break
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j or h_bin[i] == h_bin[j]:
                        continue
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                    c5_try = _exact_c5(h_bin)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_bin.copy()
                        improved = True
                        break
                    else:
                        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                # Also try fractional perturbations
                h_frac = h_bin + rng.normal(0, 0.1, n)
                h_frac = _project(h_frac, target)
                h_frac, c5_f = _active_descent(h_frac, min(deadline, time.time() + 5), top_k=3, max_iters=2000)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_frac.copy()
    return best_h, best_c5

def _resample_and_optimize(h, n_new, deadline, rng):
    n = len(h)
    if n_new == n:
        return h.copy(), _exact_c5(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = _project(h_new, n_new / 2.0)
    h_new, c5 = _active_descent(h_new, min(deadline, time.time() + 10), top_k=3, max_iters=5000)
    h_new, c5 = _lp_polish(h_new, min(deadline, time.time() + 8), radius_init=0.04, max_iters=1000)
    return h_new, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Strong active-lag descent from parent
    deadline1 = t0 + budget_s * 0.15
    h1, c5_1 = _active_descent(h, deadline1, top_k=3, max_iters=50000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Softmax descent for broader exploration
    deadline2 = t0 + budget_s * 0.30
    h2, c5_2 = _softmax_descent(h1, deadline2, beta_start=30.0, beta_end=5000.0, max_iters=15000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP polish
    deadline3 = t0 + budget_s * 0.42
    h3, c5_3 = _lp_polish(h2, deadline3, radius_init=0.05, max_iters=5000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Active descent again on LP result
    deadline4 = t0 + budget_s * 0.55
    h4, c5_4 = _active_descent(h3, deadline4, top_k=2, max_iters=50000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Random perturbation restarts
    deadline5 = t0 + budget_s * 0.75
    scale = 0.03
    for attempt in range(500):
        if time.time() > deadline5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        rem = deadline5 - time.time()
        if rem < 2:
            break
        t_sub = min(rem * 0.02, 8.0)
        h_try, c5_try = _active_descent(h_pert, time.time() + t_sub, top_k=3, max_iters=5000)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.15)

    # Phase 6: Try different n values
    deadline6 = t0 + budget_s * 0.85
    n_options = [best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n - 6, best_n + 6, best_n - 8, best_n + 8]
    n_options = [n for n in n_options if 32 <= n <= 128 and n != best_n]
    n_options = list(set(n_options))[:8]
    for n_try in n_options:
        if time.time() > deadline6:
            break
        rem = deadline6 - time.time()
        if rem < 5:
            break
        h_r, c5_r = _resample_and_optimize(best_h, n_try, time.time() + min(rem * 0.3, 20), rng)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Binary search at low n
    deadline7 = t0 + budget_s * 0.90
    for n_try in [best_n, best_n - 2, best_n - 4]:
        if time.time() > deadline7:
            break
        if n_try < 32:
            continue
        h_b, c5_b = _binary_search_low_n(n_try, min(deadline7, time.time() + 15), rng)
        if h_b is not None and c5_b < best_c5 - 1e-10:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_try

    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = _lp_polish(best_h, t0 + budget_s * 0.98, radius_init=0.03, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: Coordinate exchange polish
    if time.time() < t0 + budget_s * 0.995:
        h_cd, c5_cd = _coord_exchange(best_h, t0 + budget_s * 0.998, max_passes=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final: one more active descent push
    if time.time() < t0 + budget_s * 0.999:
        h_f, c5_f = _active_descent(best_h, t0 + budget_s * 0.9995, top_k=2, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
