    Variables: (T, h[0], ..., h[n-1])

        return x[0]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_for_shift(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] += (-h[:m])
    else:
        d2 = -d
        m = n - d2
        if m > 0:
            grad[d2:] += (1.0 - h[:m])
            grad[:m] += (-h[d2:])
    return grad

def _grad_matrix(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for i, k in enumerate(lags):

            m = n - d2
            if m > 0:
                G[i, d2:] += (1.0 - h[:m])
                G[i, :m] += (-h[d2:])
    return G
