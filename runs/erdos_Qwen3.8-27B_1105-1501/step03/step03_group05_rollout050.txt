import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    # Water-filling algorithm
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    # Final fix
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_gradient_matrix(h):
    """Build G where G[r, j] = dS(r)/dh[j] (scaled by dx)"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _cvx_trust_step(h, rho, solver=None):
    """Solve one CVXPY trust-region step. Returns candidate x or None."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    target = n / 2.0
    
    # Current scores and gradient
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _build_gradient_matrix(h)
    
    # CVXPY variables
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h) <= rho,
        G @ x + (F - G @ h) <= t,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        if solver is None:
            prob.solve()
        else:
            prob.solve(solver=solver)
        
        if x.value is not None:
            return x.value
    except Exception:
        pass
    return None

def _cvx_trust_optimize(h, deadline, rho_init=0.05, n_iters=500):
    """Main CVXPY trust-region optimization loop."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = rho_init
    stall = 0
    solvers = [None, 'CLARABEL', 'ECOS', 'SCS', 'OSQP']
    solver_idx = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        x = _cvx_trust_step(h, rho, solver=solvers[solver_idx] if solver_idx < len(solvers) else None)
        
        if x is None:
            stall += 1
            solver_idx = (solver_idx + 1) % len(solvers)
            rho *= 0.8
            if rho < 0.001:
                break
            continue
        
        # Try different step fractions
        best_trial_c5 = cur_c5
        best_trial_h = None
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            x_trial = _project_box_sum(h + frac * (x - h), target)
            c5_trial = _exact_c5(x_trial)
            if c5_trial < best_trial_c5 - 1e-12:
                best_trial_c5 = c5_trial
                best_trial_h = x_trial

            cur_c5 = best_trial_c5
            if cur_c5 < best_c5:

            stall = 0

def _smooth_grad_descent(h, deadline, beta=20.0):
    """Smoothed gradient descent as fallback."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.05
    
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_s = float(np.max(scores))
        exp_vals = np.exp(beta * (scores - max_s))
        w_sum = float(np.sum(exp_vals))
        if w_sum < 1e-300:
            break
        w = exp_vals / w_sum
        
        # Gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for d in range(-(n - 1), n):
            wd = w[n - 1 + d]
            if abs(wd) < 1e-14:
                continue
            low = max(0, -d)
            high = min(n, n - d)
            for p in range(low, high):
                q = p + d
                grad[q] += wd * dx * (1.0 - h[p])
                grad[p] -= wd * dx * h[q]
        
        grad -= np.mean(grad)
        norm = float(np.max(np.abs(grad)))
        if norm < 1e-14:
            break
        
        d_dir = -grad / norm * step
        # Line search
        best_a = 0.0
        best_val = cur_c5
        for a in np.linspace(0.01, 1.0, 10):
            h_t = _project_box_sum(h + a * d_dir, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_val:
                best_val = c5_t
                best_a = a
        
        if best_a > 0:
            h = _project_box_sum(h + best_a * d_dir, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            step *= 0.7
            if step < 1e-5:
                break
    
    return best_h, best_c5

def _binary_refine(h, deadline, rng, max_iters=20000):
    """Binary swap refinement."""
    n = len(h)
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:n // 2]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    ones = set(np.where(h_bin == 1.0)[0])
    zeros = set(np.where(h_bin == 0.0)[0])
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 1000:
            break
        
        batch = min(20, len(ones) * len(zeros))
        best_swap = None
        best_val = cur_c5
        ones_list = list(ones)
        zeros_list = list(zeros)
        
        for _b in range(batch):
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.discard(i)
            zeros.discard(j)
            ones.add(j)
            zeros.add(i)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def _phase_sweep(h, deadline):
    """Try circular shifts and keep the best."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()

        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_shifted.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    """Resample h to n_new points using integral averaging."""
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        # Find overlapping old cells
        i_start = int(a / dx_old)
        i_end = int(b / dx_old)
        if i_end >= n:
            i_end = n - 1
        total = 0.0
        for i in range(max(0, i_start), i_end + 1):
            ca = max(a, i * dx_old)
            cb = min(b, (i + 1) * dx_old)
            overlap = max(0.0, cb - ca)
            total += h[i] * overlap
        h_new[j] = total / dx_new
    return _project_box_sum(h_new, n_new / 2.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _cvx_step(h, rho, solver_list):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _grad_matrix(h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    consts = [
        x >= 0, x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h) <= rho,
        G @ x + (F - G @ h) <= t,

    prob = cp.Problem(cp.Minimize(t), consts)

            if solver is None:
                prob.solve()
            else:
                prob.solve(solver=solver)
            if x.value is not None:
                return x.value
        except Exception:
            continue
    return None
