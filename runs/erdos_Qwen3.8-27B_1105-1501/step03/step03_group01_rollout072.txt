import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_line_coeffs(h, d):
    n = len(h)
    num_lags = 2 * n - 1
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.zeros(num_lags)
    C = np.zeros(num_lags)
    for k in range(num_lags):
        delta = k - (n - 1)
        if delta >= 0:
            m = n - delta
            if m <= 0:
                continue
            B[k] = float(np.sum(d[:m] * (1.0 - h[delta:delta+m]) - h[:m] * d[delta:delta+m]))
            C[k] = -float(np.sum(d[:m] * d[delta:delta+m]))
        else:
            g = -delta
            m = n - g
            if m <= 0:
                continue
            B[k] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
            C[k] = -float(np.sum(d[g:] * d[:m]))
    return A, B, C

def _line_search_exact(h, d, t_max):
    n = len(h)
    A, B, C = _compute_line_coeffs(h, d)
    n_grid = 80
    t_vals = np.linspace(0.0, t_max, n_grid)
    M = A[None, :] + t_vals[:, None] * B[None, :] + (t_vals[:, None] ** 2) * C[None, :]
    obj_vals = np.max(M, axis=1)
    best_idx = int(np.argmin(obj_vals))
    best_t = float(t_vals[best_idx])
    best_obj = float(obj_vals[best_idx])
    if best_idx > 0 and best_idx < n_grid - 1:
        t_lo = float(t_vals[best_idx - 1])
        t_hi = float(t_vals[best_idx + 1])
    else:
        t_lo = 0.0
        t_hi = t_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        if t_hi - t_lo < 1e-13:
            break
        c1 = t_hi - gr * (t_hi - t_lo)
        c2 = t_lo + gr * (t_hi - t_lo)
        f1 = float(np.max(A + B * c1 + C * c1 * c1))
        f2 = float(np.max(A + B * c2 + C * c2 * c2))
        if f1 < f2:
            t_hi = c2
        else:
            t_lo = c1
    t_opt = (t_lo + t_hi) / 2.0
    obj_opt = float(np.max(A + B * t_opt + C * t_opt * t_opt))
    if obj_opt < best_obj:
        best_t = t_opt
        best_obj = obj_opt
    return best_t, best_obj

def _feasible_interval(h, d):
    amax = 1.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        amax = min(amax, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        amax = min(amax, float(np.min(h[neg] / (-d[neg]))))
    return max(amax, 1e-10)

def _softmax_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    exp_vals = np.exp(tau * (corr - max_c))
    w = exp_vals / float(np.sum(exp_vals))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            m = n - delta
            if m > 0:
                grad[:m] += w[k] * (1.0 - h[delta:delta+m])
                grad[delta:delta+m] += w[k] * (-h[:m])
        else:
            g = -delta
            m = n - g
            if m > 0:
                grad[g:] += w[k] * (1.0 - h[:m])
                grad[:m] += w[k] * (-h[g:])
    grad -= np.mean(grad)
    return grad

def _topk_lag_grad(h, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:topk]
    grad = np.zeros(n)
    for k in top_idx:
        delta = int(k) - (n - 1)
        if delta >= 0:
            m = n - delta
            if m > 0:
                grad[:m] += (1.0 - h[delta:delta+m])
                grad[delta:delta+m] += (-h[:m])
        else:
            g = -delta
            m = n - g
            if m > 0:
                grad[g:] += (1.0 - h[:m])
                grad[:m] += (-h[g:])
    grad -= np.mean(grad)
    return grad

def _try_step(h, d, target, t_max, min_gain=1e-12):

    t_max = min(t_max, _feasible_interval(h, d))
    if t_max < 1e-12:
        return None, cur_c5
    t_opt, _ = _line_search_exact(h, d, t_max)
    if t_opt < 1e-12:
        return None, cur_c5
    h_cand = h + t_opt * d
    h_cand = _project(h_cand, target)
    c5_cand = _exact_c5(h_cand)
    if c5_cand < cur_c5 - min_gain:
        return h_cand, c5_cand
    return None, cur_c5

def _lag_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            temp *= 0.97
            continue
