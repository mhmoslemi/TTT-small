import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):

        else:

    res = target - float(np.sum(x))

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

def _soft_grad_direction(h, tau, G):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * (2.0 / n)
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    w = exp_s / float(np.sum(exp_s))
    grad = G.T @ w
    grad = grad - np.mean(grad)
    nrm = float(np.max(np.abs(grad)))
    if nrm < 1e-14:
        return None
    return -grad / nrm

def _build_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mlen = n - d
            if mlen > 0:
                G[r, d:] += (1.0 - h[:mlen])
                G[r, :mlen] += -h[d:]
        else:
            g = -d
            mlen = n - g
            if mlen > 0:
                G[r, :mlen] += (1.0 - h[g:])
                G[r, g:] += -h[:mlen]
    return G

def _block_transfer_direction(rng, n):

    neg_starts = []
    used = np.zeros(n, dtype=bool)
    for _ in range(n_blocks):
        s = int(rng.integers(0, n - block_len + 1))
        if not np.any(used[s:s + block_len]):
            used[s:s + block_len] = True
            pos_starts.append(s)
        else:
            s = int(rng.integers(0, n - block_len + 1))
            used[s:s + block_len] = True
            pos_starts.append(s)
    for _ in range(n_blocks):
        s = int(rng.integers(0, n - block_len + 1))
        neg_starts.append(s)
    total_pos = 0
    for s in pos_starts:
        pos_mask = np.zeros(n)
        pos_mask[s:s + block_len] = 1.0
        D += pos_mask / (n_blocks * block_len)
    total_neg = 0
    for s in neg_starts:
        neg_mask = np.zeros(n)
        neg_mask[s:s + block_len] = 1.0
        D -= neg_mask / (n_blocks * block_len)
    D -= np.mean(D)
    return D

def _sine_direction(rng, n):
    freq = float(rng.uniform(0.5, 4.0))
    phase = float(rng.uniform(0, 2 * np.pi))
    t = np.arange(n, dtype=float)
    D = np.sin(2.0 * np.pi * freq * t / n + phase)
    if rng.random() < 0.3:
        freq2 = float(rng.uniform(2.0, 6.0))
        phase2 = float(rng.uniform(0, 2 * np.pi))
        D += 0.5 * np.sin(2.0 * np.pi * freq2 * t / n + phase2)
    D -= np.mean(D)
    nrm = float(np.max(np.abs(D)))
    if nrm < 1e-14:
        return None
    return D / nrm

def _pair_direction(rng, n, h):
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    D = np.zeros(n)
    if h[i] < h[j]:
        D[i] = 1.0
        D[j] = -1.0
    else:
        D[i] = -1.0
        D[j] = 1.0
    return D

def _resample(h, n_new):

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)

        best_n = len(h1)

    # Phase 2: Extended search with more diversity
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _search_loop(best_h, d2, rng, max_iters=500)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Multi-resolution
    d3 = t0 + budget_s * 0.72
    offsets = list(range(-10, 11)) + [12, -12, 14, -14, 16, -16, 20, -20, 24, -24, 30, -30]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _search_loop(h_r, sub_dl, rng, max_iters=100)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Perturbation escapes
    d4 = t0 + budget_s * 0.85
    scale = 0.04
    for attempt in range(60):
        if time.time() > d4:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_try, c5_try = _search_loop(h_pert, sub_dl, rng, max_iters=80)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.15)

    # Phase 5: Final polish
    if time.time() < hard_stop:
        d5 = min(hard_stop, time.time() + budget_s * 0.05)
        h_f, c5_f = _final_polish(best_h, d5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, float(best_c5), int(best_n)
