- minimize max_r score_r(h) subject to 0 ≤ h ≤ 1, sum(h) = n/2

This is a non-convex problem (the scores are bilinear in h). But at a given point h, the scores are linear in δ = x - h:
score_r(h + δ) ≈ score_r(h) + G[r,:] · δ

So the trust-region LP is:
- min t
- s.t. score_r(h) + G[r,:] · δ ≤ t for all r
- 0 ≤ h + δ ≤ 1
- sum(δ) = 0
- ||δ||₂ ≤ ρ

This is a second-order cone program (SOC) because of the ||δ||₂ ≤ ρ constraint. CVXPY can handle this.

Let me also think about the binary approach more carefully. If h is binary (values 0 or 1), then:
score_r = sum over valid p of h[p+d] * (1-h[p]) for lag d

For binary h with exactly n/2 ones, this counts the number of "1 at position p+d and 0 at position p" for each lag d. This is essentially counting "transitions" at each lag.

For a binary sequence of length n with n/2 ones arranged to minimize the maximum difference count, this is related to the problem of finding a binary sequence that "spreads out" the differences.

def _trust_region_step(h, rho):

    dx = 2.0 / n
    L = 2 * n - 1
    F = _scores_raw(h) * dx  # scaled scores
    G = _build_gradient(h)
    
    # Variables: t (scalar), delta (vector of length n)
    t = cp.Variable()
    delta = cp.Variable(n)
    
    constraints = [
        G @ delta <= t - F,  # linearized scores <= t
        h + delta >= 0,
        h + delta <= 1,
        cp.sum(delta) == 0,
        cp.norm(delta, 2) <= rho,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.OSQP)
            except Exception:
                return None
    
    if t.value is None or delta.value is None:
        return None
    
    x = h + delta.value
    return x
```

def _smoothed_max_grad(h, beta):
    """Gradient of smooth approximation to max of scores."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(scores)
    # Softmax weights
    exp_vals = np.exp(beta * (scores - max_s))
    w = exp_vals / np.sum(exp_vals)
    G = _build_gradient(h)
    return G.T @ w
```

def _soc_trust_step(h, rho, solver='CLARABEL'):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = _scores_scaled(h)
    G = _build_gradient(h)
    
    t = cp.Variable()
    delta = cp.Variable(n)
    
    constraints = [
        G @ delta <= t - F,
        h + delta >= 0,
        h + delta <= 1,
        cp.sum(delta) == 0,
        cp.norm(delta, 2) <= rho,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = ['CLARABEL', 'ECOS', 'OSQP', 'SCS']
    if solver in solvers:

            prob.solve(solver=s)
            if t.value is not None and delta.value is not None:
                x = h + np.array(delta.value).flatten()
                return x
        except Exception:
            continue
    return None

            break
        if no_improve > 5000 and temperature < 1e-8:
            break
        if not ones or not zeros:
            break
        
        # Batch evaluation
        batch = min(50, len(ones) * len(zeros))
        best_swap = None
        best_val = cur_c5
        
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
            temperature *= 0.999

def _resample_exact(h_old, n_new):
    """Transfer between resolutions using integral averaging."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # Interval-based transfer
    # Old cell i covers [i*dx_old, (i+1)*dx_old]
    # New cell j covers [j*dx_new, (j+1)*dx_new]
    h_new = np.zeros(n_new)
    mass_weights = np.zeros(n_new)
    
    for i in range(n_old):
        a_max = max(i * dx_old, 0.0)
        a_min = min((i + 1) * dx_old, 2.0)
        for j in range(n_new):
            b_max = min((j + 1) * dx_new, 2.0)
            b_min = j * dx_new
            overlap = min(a_max, b_max) - max(a_min, b_min)
            if overlap > 0:
                h_new[j] += h_old[i] * overlap
                mass_weights[j] += overlap
    
    nonzero = mass_weights > 1e-14
    h_new[nonzero] /= mass_weights[nonzero]
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_feasible_perturbation(h, rng, n_moves=5, max_delta=0.1):

    n = len(h)

    h = h.copy()
    for _ in range(n_moves):
        i = rng.integers(n)
        j = rng.integers(n)

            continue
        if h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
    return _project_box_sum(h, target)

def _build_gradient(h):
    n = len(h)

            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
