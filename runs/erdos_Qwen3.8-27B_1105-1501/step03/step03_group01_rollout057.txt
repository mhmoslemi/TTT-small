import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

        if res > 0:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:

def _soft_grad(h, tau):
    """Compute softmax-weighted gradient of the max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # For each lag k (0 to 2n-2), s_k = k - (n-1)
    # grad_k[j] = (1-h[j+s_k]) - h[j-s_k] (with boundary handling)
    # soft_grad = sum_k w_k * grad_k
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        # Term 1: (1 - h[j+s]) for valid j where j+s is in [0, n-1]
        # j ranges from max(0, -s) to min(n-1, n-1-s)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wk * (1.0 - h[j_lo+s:j_hi+1+s])
        # Term 2: -h[j-s] for valid j where j-s is in [0, n-1]
        # j ranges from max(0, s) to min(n-1, n-1+s)  [wait, j-s >= 0 means j >= s, j-s <= n-1 means j <= n-1+s]
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, n - 1 + s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += -wk * h[j_lo2-s:j_hi2+1-s]
    
    return grad

def _line_search(h, d, c5_cur, n_eig=60):
    """Exact line search along direction d (sum zero) from point h.
    Returns (best_alpha, best_c5)."""
    n = len(h)
    
    # Find feasible range for alpha: h + alpha*d must be in [0,1]^n
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Also search in negative direction
    alpha_min = -1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_min = max(alpha_min, -h[j] / d[j])
        elif d[j] < -1e-14:
            alpha_min = max(alpha_min, -(1.0 - h[j]) / d[j])
    alpha_min = min(alpha_min, -1e-10)
    
    # Grid search over [alpha_min, alpha_max]
    best_alpha = 0.0
    best_c5 = c5_cur
    
    # Sample on a grid
    n_grid = max(20, min(n_eig, int(alpha_max - alpha_min) * 50 + 10))
    grid = np.linspace(alpha_min, alpha_max, n_grid)
    vals = []
    for a in grid:
        h_t = h + a * d
        c5_t = _exact_c5(h_t)
        vals.append(c5_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_alpha = a
    
    # Refine around the best grid point
    if best_alpha > alpha_min and best_alpha < alpha_max and len(grid) > 2:
        idx = int(np.argmin(vals))
        lo = grid[max(0, idx-1)]
        hi = grid[min(len(grid)-1, idx+1)]
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * d)
        f2 = _exact_c5(h + c2 * d)
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * d)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * d)
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(h + opt_a * d)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_alpha = opt_a
    
    return best_alpha, best_c5

def _sign_transfer_direction(h, grad, tau, rng, k_size=None):
    """Generate a sign-transfer direction.
    Move mass from high-h positions (where grad suggests decreasing) to low-h positions."""
    n = len(h)
    if k_size is None:
        k_size = max(4, n // 10)
    
    d = np.zeros(n)
    
    # Score each position: we want to decrease h where grad is large (positive grad means
    # increasing h there increases the objective, so we want to decrease h there)
    # Actually, the direction to decrease the objective is -grad (projected to sum zero)
    # For sign transfer: pick positions to decrease (where -grad is most negative, i.e., grad positive)
    # and positions to increase (where grad is negative)
    
    # Score = -grad (direction to decrease objective)
    score = -grad
    
    # Pick top k positions to increase
    inc_candidates = np.argsort(score)[::-1][:k_size*2]
    # Pick top k positions to decrease
    dec_candidates = np.argsort(score)[:k_size*2]
    
    # Ensure we don't pick the same position
    inc_set = set(inc_candidates)
    dec_filtered = [i for i in dec_candidates if i not in inc_set]
    
    n_inc = min(k_size, len(inc_set))
    n_dec = min(k_size, len(dec_filtered))
    
    if n_inc == 0 or n_dec == 0:
        return None
    
    inc_idx = list(inc_set)[:n_inc]
    dec_idx = dec_filtered[:n_dec]
    
    total_inc = 0.0
    total_dec = 0.0
    for i in inc_idx:
        total_inc += (1.0 - h[i])
    for i in dec_idx:
        total_dec += h[i]
    
    if total_inc < 1e-10 or total_dec < 1e-10:
        return None
    
    scale = min(total_inc, total_dec) / max(n_inc, 1)
    
    for i in inc_idx:
        d[i] = scale * (1.0 - h[i]) / max(total_inc, 1e-10)
    for i in dec_idx:
        d[i] = -scale * h[i] / max(total_dec, 1e-10)
    
    # Fix sum to zero
    d_sum = float(np.sum(d))
    d -= d_sum / n
    
    # Normalize
    d_norm = float(np.max(np.abs(d)))
    if d_norm < 1e-10:
        return None
    d /= d_norm
    
    return d

def _smooth_random_direction(n, rng, order=3):
    """Generate a random smooth direction (sum zero, bounded)."""
    x = np.linspace(0, 1, n, endpoint=False)
    d = np.zeros(n)
    for k in range(1, order + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        d += amp * np.sin(2 * np.pi * k * x + phase)
    
    # Make sum zero
    d -= np.mean(d)
    
    # Normalize
    d_norm = float(np.max(np.abs(d)))
    if d_norm < 1e-10:
        return None
    d /= d_norm
    return d

def _neighbor_swap_direction(h, rng):
    """Generate a direction that swaps mass between nearby positions."""
    n = len(h)
    d = np.zeros(n)
    
    # Pick a random pair of nearby positions
    j = rng.integers(0, n)
    dist = rng.integers(1, max(2, n // 4))
    k = (j + dist) % n  # or just j+dist if in range
    
    if k >= n:
        k = (j - dist) % n
    
    # Transfer from higher to lower
    if h[j] > h[k]:
        d[j] = -1.0
        d[k] = 1.0
    else:
        d[j] = 1.0
        d[k] = -1.0
    
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):

    A_eq = np.zeros((1, n + 1))

    for k in range(1, n_modes + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick warmup on initial (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_opt(h, d1, use_lp=True, use_grad=True, max_iter=200, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Intensive optimization (25%)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _local_opt(best_h, d2, use_lp=True, use_grad=True, max_iter=600, seed=seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (30%)
    d3 = t0 + budget_s * 0.63
    offsets = list(range(-12, 13, 2)) + list(range(-20, 21, 4)) + [16, -16, 24, -24, 30, -30, 36, -36]
    offsets = sorted(set(o for o in offsets if 20 <= best_n + o <= 160))
    tried = set()
    per_res = budget_s * 0.018
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + per_res)
        h_r, c5_r = _local_opt(h_r, sub_dl, use_lp=True, use_grad=True, max_iter=200, seed=seed + n_try)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Random smooth restarts (20%)
    d4 = t0 + budget_s * 0.83
    for attempt in range(40):
        if time.time() > d4:
            break
        if best_c5 < 0.3806:
            break
        # Mix of random smooth and perturbation
        if attempt % 3 == 0:
            h_try = _random_smooth(best_n, rng, n_modes=rng.integers(2, 6))
        else:
            scale = 0.02 + 0.04 * (attempt % 5)
            h_try = best_h + rng.normal(0, scale, best_n)
            h_try = _project(h_try, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_opt, c5_opt = _local_opt(h_try, sub_dl, use_lp=True, use_grad=True, max_iter=120, seed=seed + 100 + attempt)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Final pair refinement (10%)
    if time.time() < t0 + budget_s * 0.95:
        d5 = t0 + budget_s * 0.95
        h_p, c5_p = _pair_refine(best_h, d5, max_passes=80)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 6: Final LP polish (5%)
    if time.time() < t0 + budget_s * 0.995:
        d6 = t0 + budget_s * 0.995
        h_f, c5_f = _local_opt(best_h, d6, use_lp=True, use_grad=True, max_iter=300, seed=seed + 999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
