import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : 0 ≤ x_i ≤ 1, sum(x) = target}
    n = len(v)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build gradient matrix G where G[r, j] = dS_{r-(n-1)}/dh[j]"""
    n = len(h)
    k = n - 1
    G = np.zeros((2 * n - 1, n))

            G[r, :m] += (-h[d:])         # -h[i] part for i+d = j
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += (-h[:m])
    return G

    n = len(h)
    dx = 2.0 / n
    # a_k = scores
    a = np.correlate(h, 1.0 - h, mode='full') * dx
    # b_k = G[k] @ d = dS_k/dh · d (the linear coefficient)
    # b_k = Σ_i d[i]*(1-h[i+k]) - h[i]*d[i+k] (for valid i)
    # This is the same as _build_G(h) @ d
    # But I can compute it more directly:
    b = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            b[r] = np.sum(d[k:] * (1.0 - h[:m])) + np.sum(-h[k:] * d[:m])
        else:

            m = n - g
            b[r] = np.sum(d[:m] * (1.0 - h[g:])) + np.sum(-h[:m] * d[g:])

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
