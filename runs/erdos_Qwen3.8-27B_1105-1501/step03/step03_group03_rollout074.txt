import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_and_grad(h):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d+m])
            G[r, d:d+m] -= h[:m]
        else:
            g = -d
            G[r, g:g+m] += (1.0 - h[:m])
            G[r, :m] -= h[g:g+m]
    return scores, G

def _lp_direction(h, scores, G, radius, idx=None):
    n = len(h)
    if idx is not None:
        G_sub = G[idx, :]
        scores_sub = scores[idx]
    else:
        G_sub = G
        scores_sub = scores
    n_con = len(scores_sub)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_con, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -scores_sub
    A_eq = np.array([[np.ones(n), 0.0]])
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Coarse grid
    best_a = 0.0
    best_c5 = c5_cur
    n_eval = 20
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slaqp_optimize(h, target, beta, max_iter=100):
    n = len(h)
    def obj(x):
        scores = np.correlate(x, 1.0 - x, mode='full')
        s_max = float(np.max(scores))
        return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta
    def jac(x):
        scores, G = _scores_and_grad(x)
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        w = exp_s / np.sum(exp_s)
        return G.T @ w
    def sum_fun(x):
        return np.sum(x) - target
    def sum_jac(x):
        return np.ones(len(x))
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()
