def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, d, n):
    """Gradient of correlation at lag d (relative to center) wrt h."""
    g = np.zeros(n)
    # d > 0 means output index n-1+d: sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
    # d < 0 means output index n-1+d (d negative): sum_{i=|d|}^{n-1} h[i]*(1-h[i-|d|])
    # d = 0: sum h[i]*(1-h[i])
    if d > 0:

        g[:m] += (1.0 - h[d:d+m])
        g[d:d+m] -= h[:m]
    elif d < 0:
        g = -d
        m = n - g
        g[:m] += (1.0 - h[g:g+m])
        g[g:g+m] -= h[:m]
    else:

    return g

        frac = min((it + 1) / max_iters, 1.0)

But here, h is defined as n samples over [0,2]. Each sample represents the value on an interval of width 2/n. So the "function" is:
h_cont(x) = h[i] for x in [2i/n, 2(i+1)/n)

The integral ∫ h(x)(1-h(x+k)) dx is then:
sum_i ∫_{2i/n}^{2(i+1)/n} h[i] * (1-h(x+k)) dx

For the discrete correlate, we're approximating this as:
sum_i h[i] * (1-h[i+k]) * (2/n)

This is valid when k is a multiple of 2/n (i.e., the shift aligns with the grid). For non-aligned shifts, we'd need a more careful calculation. But the problem statement says to use `np.correlate(h, 1-h, mode="full") * dx`, so we're only considering grid-aligned shifts.

def _grad_for_lag(h, d, n):
    """Gradient of correlation score at relative lag d (d=idx-(n-1)) wrt h."""
    g = np.zeros(n)
    if d > 0:

        # grad[i] += (1-h[i-d]) for i in [d, n-1]
        g[d:] += (1.0 - h[:m])
        # grad[i] += -h[i+d] for i in [0, n-1-d]
        g[:m] -= h[d:]
    elif d < 0:

        m = n - dd
        # grad[i] += (1-h[i+dd]) for i in [0, n-1-dd]
        g[:m] += (1.0 - h[dd:])
        # grad[i] += -h[i-dd] for i in [dd, n-1]

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, d, n):
    g = np.zeros(n)
    if d > 0:
        m = n - d
        g[d:] += (1.0 - h[:m])
        g[:m] -= h[d:]
    elif d < 0:
        dd = -d
        m = n - dd
        g[:m] += (1.0 - h[dd:])
        g[dd:] -= h[:m]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def _soft_descent(h, deadline, beta_start=50.0, beta_end=100000.0, max_iters=30000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    top_k = min(6, 2 * n - 1)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        top_idx = np.argsort(scores)[-top_k:]
        top_scores = scores[top_idx]
        w = np.exp(beta * (top_scores - top_scores.max()))
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            break
        w /= wsum
        grad = np.zeros(n)
        for i, idx in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            d = idx - (n - 1)
            grad += w[i] * _grad_for_lag(h, d, n)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.01, 0.3)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.85
            if lr < 1e-9 or stall > 300:
                break
    return best_h, best_c5

def _alternating_descent(h, deadline, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.04
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        worst_idx = int(np.argmax(scores))
        d = worst_idx - (n - 1)
        g = _grad_for_lag(h, d, n)
        g -= np.mean(g)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        direction = -g / gnorm
        alpha = lr
        improved = False
        for _ in range(20):

            c5_try = _exact_c5(h_try)

                    best_h = h.copy()
                improved = True
                break

                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _balanced_binary_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)
    h[idx[:half]] = 1.0
    h = h + rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _sine_init(n, freq, amplitude):
    x = np.arange(n) / n
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq * x)
    return _project(h, n / 2.0)
