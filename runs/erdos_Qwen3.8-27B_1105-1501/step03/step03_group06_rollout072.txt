            best_t = (lo + hi_t) / 2.0
            new_max = np.max(c + best_t * a) * dx
            if new_max < best_c5 - 1e-13:
                h[i] += best_t
                h[j] -= best_t
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, target)
                c = _corr_vec(h)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved_this_sweep = True
        if not improved_this_sweep:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5


def _line_search_direction(h, d, target, n_scan=150):
    """Line search along direction d from h. Returns (best_h, best_c5)."""
    n = len(h)
    d = d - np.mean(d)
    norm = np.max(np.abs(d))
    if norm < 1e-14:
        return h.copy(), _exact_c5(h)
    d = d / norm
    t_min, t_max = 0.0, 1.0
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_min = max(t_min, float(np.max(-h[pos] / d[pos])))
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((1.0 - h[neg]) / d[neg])))
        t_max = min(t_max, float(np.min(-h[neg] / d[neg])))
    if t_max - t_min < 1e-10:
        return h.copy(), _exact_c5(h)
    ts = np.linspace(t_min, t_max, n_scan)
    best_t, best_c5 = 0.0, _exact_c5(h)
    best_h = h.copy()
    for t in ts:
        if abs(t) < 1e-12:
            continue
        h_t = _project(np.clip(h + t * d, 0.0, 1.0), target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5, best_t, best_h = c5_t, t, h_t
    if best_t != 0.0:
        span = t_max - t_min
        lo = max(t_min, best_t - span / n_scan)
        hi = min(t_max, best_t + span / n_scan)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(_project(np.clip(h + c1 * d, 0.0, 1.0), target))
            f2 = _exact_c5(_project(np.clip(h + c2 * d, 0.0, 1.0), target))
            if f1 < f2:
                hi, c2 = c2, c1
            else:
                lo, c1 = c1, c2
        t_star = (lo + hi) / 2.0
        h_star = _project(np.clip(h + t_star * d, 0.0, 1.0), target)
        c5_star = _exact_c5(h_star)
        if c5_star < best_c5 - 1e-14:
            best_c5, best_h = c5_star, h_star
    return best_h, best_c5


def _active_gradient(h, c, n_active=3):
    """Return gradient direction targeting the top n_active lags."""
    n = len(h)
    idx = np.argsort(c)[-n_active:]
    g = np.zeros(n)
    for k_idx in idx:
        k = k_idx - (n - 1)
        for m in range(n):
            if 0 <= m - k < n:
                g[m] += (1.0 - h[m - k])
            if 0 <= m + k < n:
                g[m] -= h[m + k]
    g -= np.mean(g)
    return g


def _smooth_direction(h, c, alpha=20.0):
    """Log-sum-exp weighted gradient direction."""
    n = len(h)
    ms = float(np.max(c))
    vals = np.clip(alpha * (c - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    g = np.zeros(n)
    for m in range(n):
        for k in range(-(n - 1), n):
            r = n - 1 + k
            if 0 <= m - k < n:
                g[m] += w[r] * (1.0 - h[m - k])
            if 0 <= m + k < n:
                g[m] -= w[r] * h[m + k]
    g -= np.mean(g)
    return g


def _fourier_direction(n, mode, phase):
    """Generate a mean-free Fourier direction."""
    d = np.cos(2.0 * np.pi * mode * np.arange(n) / n + phase)
    d -= np.mean(d)
    return d


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Pair coordinate descent (50% budget)
    d1 = t0 + budget_s * 0.50
    h, c5 = _pair_sweep(h, best_c5, d1, max_sweeps=2000, n_scan=100)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 2: Direction-based line search (20% budget)
    d2 = t0 + budget_s * 0.70
    target = best_n / 2.0
    h_cur = best_h.copy()
    for it in range(5000):
        if time.time() > d2:
            break
        c = _corr_vec(h_cur)
        directions = []
        # Active gradient
        directions.append(_active_gradient(h_cur, c, n_active=2))
        directions.append(_active_gradient(h_cur, c, n_active=4))
        # Smooth gradient
        directions.append(_smooth_direction(h_cur, c, alpha=15.0))
        directions.append(_smooth_direction(h_cur, c, alpha=40.0))
        # Fourier directions
        for mode in [1, 2, 3, 5]:
            directions.append(_fourier_direction(best_n, mode, rng.uniform(0, 2 * np.pi)))
        # Random
        directions.append(rng.standard_normal(best_n))
        # Pair-like: find max/min gap
        imax = int(np.argmax(h_cur))
        imin = int(np.argmin(h_cur))
        d_pair = np.zeros(best_n)
        d_pair[imax] = -1.0
        d_pair[imin] = 1.0
        directions.append(d_pair)

        best_improve = 1e-10
        for d in directions:
            if time.time() > d2:
                break
            h_try, c5_try = _line_search_direction(h_cur, d, target, n_scan=80)
            if c5_try < best_c5 - best_improve:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try
                best_improve = _exact_c5(h_cur) - c5_try
                break
            elif c5_try < _exact_c5(h_cur) - 1e-12:
                h_cur = h_try

    # Phase 3: More pair sweeps from improved point (15% budget)
    d3 = t0 + budget_s * 0.85
    h_cur = best_h.copy()
    h, c5 = _pair_sweep(h_cur, best_c5, d3, max_sweeps=3000, n_scan=120)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 4: Resolution exploration (8% budget)
    d4 = t0 + budget_s * 0.93
    for dn in [1, -1, 2, -2, 3, 4, -3]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _pair_sweep(h_r, _exact_c5(h_r), sub_d, max_sweeps=300, n_scan=80)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Final intensive pair polish (7% budget)
    d5 = t0 + budget_s * 0.99
    h_cur = best_h.copy()
    h, c5 = _pair_sweep(h_cur, best_c5, d5, max_sweeps=5000, n_scan=150)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
