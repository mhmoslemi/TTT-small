    G = np.zeros((m, n))
    for k in range(m):
        d = k - (n - 1)
        for j in range(n):
            g = 0.0
            jp = j + d
            jm = j - d
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            if 0 <= jm < n:
                g += -h[jm]
            G[k, j] = g
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def gen_random_dirs(h, rng, n_dirs=5):
    n = len(h)
    dirs = []
    for _ in range(n_dirs):
        typ = rng.integers(0, 3)
        if typ == 0:
            p = np.zeros(n)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                p[i] = 1.0
                p[j] = -1.0
        elif typ == 1:
            freqs = rng.integers(1, 5, size=3)
            phases = rng.uniform(0, 2 * np.pi, 3)
            p = np.zeros(n)
            xs = (np.arange(n) + 0.5) / n
            for f, ph in zip(freqs, phases):
                p += np.cos(2 * np.pi * f * xs + ph)
        else:
            p = rng.normal(0, 1, n)
            p -= p.mean()
        p -= p.mean()
        norm = np.max(np.abs(p))
        if norm > 1e-12:
            p = p / norm
            dirs.append(p)
    return dirs

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)

def local_search(h, deadline, tau_list=None, topk_list=None, use_lp=True, rng=None):
    n = len(h)
    target = n / 2.0
    if tau_list is None:
        tau_list = [30.0, 60.0, 120.0]
    if topk_list is None:
        topk_list = [3, 5, 10, None]
    if rng is None:
        rng = np.random.default_rng(123)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.15
    stall = 0
    lp_cooldown = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 80:
            break
        direction = None
        r = rng.random()
        if r < 0.35 or (use_lp and lp_cooldown <= 0):
            scores = row_scores(h)
            tau = tau_list[int(rng.integers(0, len(tau_list)))]
            tk = topk_list[int(rng.integers(0, len(topk_list)))]
            direction = compute_soft_grad(h, tau=tau, top_k=tk)
        elif r < 0.55:
            if use_lp and lp_cooldown <= 0:
                scores = row_scores(h)
                direction = lp_descent_dir(h, scores, radius)
                if direction is not None:
                    lp_cooldown = 4
        else:
            rds = gen_random_dirs(h, rng, n_dirs=3)
            direction = rds[int(rng.integers(0, len(rds)))]
        lp_cooldown = max(0, lp_cooldown - 1)
        if direction is None:
            stall += 1
            continue
        if np.max(np.abs(direction)) < 1e-13:
            stall += 1
            continue
        direction -= direction.mean()
        t_best, c5_new = exact_line_search(h, direction)
        if c5_new < cur_c5 - 1e-13:
            h = h + t_best * direction
            h = project_h(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            if stall % 15 == 0:
                radius *= 0.75
                if radius < 0.03:
                    break
            if stall % 25 == 0:
                h = best_h + rng.normal(0, 0.02, n)
                h = project_h(h, target)
                cur_c5 = exact_c5(h)
        if it % 50 == 0 and time.time() > deadline * 0.7:
            if best_c5 < 0.3809:
                break
    return best_h, best_c5

def anneal_search(h, deadline, rng=None, t_start=0.005, t_end=0.0001):
    n = len(h)
    target = n / 2.0
    if rng is None:
        rng = np.random.default_rng(456)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t0 = time.time()
    total_time = max(deadline - t0, 1.0)
    step = 0
    total_steps = max(int(total_time / 0.05), 10)
    while time.time() < deadline:
        step += 1
        temp = t_start * (t_end / t_start) ** (step / total_steps)
        direction = None
        r = rng.random()
        if r < 0.4:
            tau = 40.0 + 80.0 * rng.random()
            tk = int(rng.integers(2, 12))
            direction = compute_soft_grad(h, tau=tau, top_k=tk)
        elif r < 0.6:
            rds = gen_random_dirs(h, rng, n_dirs=2)
            direction = rds[0]
        else:
            direction = rng.normal(0, 1, n)
            direction -= direction.mean()
        if direction is None or np.max(np.abs(direction)) < 1e-13:
            continue
        direction -= direction.mean()
        norm = np.max(np.abs(direction))
        if norm > 1e-13:
            direction = direction / norm
        t_best, c5_new = exact_line_search(h, direction, n_samples=15, refine_iters=15)
        delta = c5_new - cur_c5
        if delta < -1e-14:
            h = h + t_best * direction
            h = project_h(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            accept_prob = np.exp(-delta / max(temp, 1e-8))
            if rng.random() < accept_prob:
                h = h + t_best * direction
                h = project_h(h, target)
                cur_c5 = exact_c5(h)
    return best_h, best_c5

def multi_res_search(h_best, n_best, deadline, rng):
    search_ns = []
    for dn in range(-20, 21, 2):
        nt = n_best + dn
        if 40 <= nt <= 140 and nt != n_best:
            search_ns.append(nt)
    rng.shuffle(search_ns)
    best_h = h_best.copy()
    best_c5 = exact_c5(h_best)
    per_n_budget = max(0.5, (deadline - time.time()) / max(len(search_ns), 1))
    for nt in search_ns:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + per_n_budget)
        h_r = resample_h(h_best, nt)
        c5_r = exact_c5(h_r)
        if time.time() < sub_dl:
            h_r, c5_r = local_search(h_r, min(sub_dl, time.time() + per_n_budget * 0.7),
                                     tau_list=[40.0, 80.0], topk_list=[4, 8, None],
                                     use_lp=True, rng=np.random.default_rng(nt * 7 + 1))
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def pair_local_refine(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(60):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, min(j + 6, n)):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / dj
                    else:
                        am = h[j] / (-dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / dk)
                    else:
                        am = min(am, h[k] / (-dk))
                    if am < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        nt = (t_lo2, t_hi2) = (-1.0, 1.0)
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = project_h(h0, target0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Quick gradient descent warmup (5% budget)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = local_search(h, d1, tau_list=[30.0, 80.0, 150.0],
                            topk_list=[3, 5, 10, None], use_lp=False,
                            rng=np.random.default_rng(seed + 1))
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Main diverse local search (30% budget)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = local_search(best_h, d2, tau_list=[20.0, 50.0, 100.0, 200.0],
                            topk_list=[2, 4, 8, 15, None], use_lp=True,
                            rng=np.random.default_rng(seed + 10))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Simulated annealing (20% budget)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = anneal_search(best_h, d3, rng=np.random.default_rng(seed + 20),
                             t_start=0.003, t_end=0.00005)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (25% budget)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = multi_res_search(best_h, best_n, d4, rng=np.random.default_rng(seed + 30))
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(h4)

    # Phase 5: Final local polish (10% budget)
    d5 = t0 + budget_s * 0.92
    h5, c5_5 = local_search(best_h, d5, tau_list=[50.0, 100.0],
                            topk_list=[3, 5, 10, None], use_lp=True,
                            rng=np.random.default_rng(seed + 40))
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair refinement (5% budget)
    d6 = t0 + budget_s * 0.97
    h6, c5_6 = pair_local_refine(best_h, d6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final projection and evaluation
    best_h = project_h(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5:
        best_h = project_h(best_h, best_n / 2.0)
        final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
