    for k in range(n):
        w = weights[n - 1 + k]
        if abs(w) < 1e-15:
            continue
        m = n - k
        if m > 0:
            grad[k:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[k:]
    for g in range(1, n):
        w = weights[n - 1 - g]
        if abs(w) < 1e-15:
            continue
        m = n - g
        if m > 0:
            grad[:m] += w * (1.0 - h[g:])
            grad[g:] -= w * h[:m]
    return grad

def _softmax_weights(scores, beta):
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    s = float(np.sum(exps))
    if s < 1e-300:
        return np.full(len(scores), 1.0 / len(scores))
    return exps / s

def _topk_smooth_weights(scores, k, margin):
    n = len(scores)
    s = np.sort(scores)[::-1]
    threshold = s[min(k, n) - 1] - margin
    w = np.maximum(0.0, scores - threshold)
    total = float(np.sum(w))
    if total < 1e-15:
        return np.full(n, 1.0 / n)
    return w / total

def _soft_descent(h, deadline, beta_start=20.0, beta_end=3000.0,
                  max_iters=20000, momentum=0.15, lr_start=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    vel = np.zeros(n)
    lr = lr_start
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max(1, max_iters), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        weights = _softmax_weights(scores, beta)
        grad = _vectorized_grad(h, weights)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        vel = momentum * vel + (1.0 - momentum) * direction
        vnorm = float(np.linalg.norm(vel))
        if vnorm < 1e-14:
            break
        vel = vel / vnorm
        alpha = lr
        improved = False
        for _ in range(30):
            h_try = h + alpha * vel
            if np.any(h_try < -0.5) or np.any(h_try > 1.5):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.5)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.8
            vel *= 0.5
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _fw_step(h, grad, target):
    n = len(h)
    k = int(target)
    if k <= 0 or k >= n:
        return h.copy()
    idx = np.argpartition(grad, k)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def _frank_wolfe(h, deadline, max_iters=5000, beta=50.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        scores = _scores(h)
        m = float(np.max(scores))
        weights = _softmax_weights(scores, beta)
        grad = _vectorized_grad(h, weights)
        v = _fw_step(h, grad, target)
        diff = v - h
        dnorm = float(np.linalg.norm(diff))
        if dnorm < 1e-12:
            break
        best_a = 0.0
        best_val = cur_c5
        for a in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            h_t = _project(h + a * diff, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_val - 1e-14:
                best_val = c5_t
                best_a = a
        if best_a == 0.0:
            stall += 1
            if stall > 100:
                break
            beta *= 1.05
            continue
        h = _project(h + best_a * diff, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        stall = 0
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    G = np.zeros((2 * n - 1, n))
    for k in range(n):
        m = n - k
        if m > 0:
            G[n - 1 + k, k:] += (1.0 - h[:m])
            G[n - 1 + k, :m] -= h[k:]
    for g in range(1, n):
        m = n - g
        if m > 0:
            G[n - 1 - g, :m] += (1.0 - h[g:])
            G[n - 1 - g, g:] -= h[:m]
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, target):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    alphas = np.linspace(0.02, alpha_max, 14)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.6)
    hi_a = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_polish(h, deadline, radius_init=0.08, max_iters=2000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, target)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    deltas = [0.01, 0.02, 0.04, 0.08, 0.12, 0.2]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                continue
            for delta in deltas:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    if d > 0:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        comp = int(np.argmax(np.where(mask, 1.0 - h_try, -2.0)))
                    else:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        comp = int(np.argmax(np.where(mask, h_try, -2.0)))
                    if comp == j or h_try[comp] + (-d) < 0 or h_try[comp] + (-d) > 1:
                        continue
                    h_try[comp] += (-d)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, rng, scale):
    n = len(h)
    h_p = h.copy()
    mode = rng.integers(0, 4)
    if mode == 0:
        h_p += rng.normal(0, scale, n)
    elif mode == 1:
        f = rng.uniform(0.5, 3.0)
        x = np.arange(n) / n
        h_p += scale * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        k = rng.integers(4, max(5, n // 4))
        perm = rng.permutation(n)
        for _ in range(k):
            j = perm[_]
            d = rng.normal(0, scale * 2)
            if 0 <= h_p[j] + d <= 1:
                h_p[j] += d
    else:
        half = n // 2
        shift = rng.integers(1, max(2, half))
        h_p = np.roll(h_p, shift)
        h_p += rng.normal(0, scale * 0.3, n)
    return _project(h_p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Soft gradient descent with momentum (fast, many iterations)
    deadline1 = t0 + budget_s * 0.25
    h, c5 = _soft_descent(h, deadline1, beta_start=15.0, beta_end=4000.0,
                          max_iters=30000, momentum=0.2, lr_start=0.1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Frank-Wolfe to push toward vertices
    deadline2 = t0 + budget_s * 0.35
    h2, c5_2 = _frank_wolfe(best_h, deadline2, max_iters=5000, beta=80.0)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP polish
    deadline3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_polish(best_h, deadline3, radius_init=0.06, max_iters=1500)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Basin hopping with structured perturbations
    deadline4 = t0 + budget_s * 0.72
    scale = 0.05
    for attempt in range(500):
        if time.time() > deadline4:
            break
        h_pert = _structured_perturb(best_h, rng, scale)
        time_left = deadline4 - time.time()
        alloc = min(time_left / max(500 - attempt, 1), budget_s * 0.012)
        h_try, c5_try = _soft_descent(h_pert, time.time() + alloc,
                                      beta_start=10.0, beta_end=3000.0,
                                      max_iters=2000, momentum=0.1, lr_start=0.08)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.9)
        else:
            scale = min(0.2, scale * 1.08)

    # Phase 5: Multi-resolution search
    deadline5 = t0 + budget_s * 0.85
    n_list = sorted(set([best_n + 2*d for d in range(-8, 9, 2)
                        if 40 <= best_n + 2*d <= 128]))
    for n_try in n_list:
        if time.time() > deadline5:
            break
        n_try = int(n_try)
        if n_try == best_n or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        alloc = min(deadline5 - time.time(), budget_s * 0.008)
        h_r, c5_r = _soft_descent(h_r, time.time() + alloc,
                                  beta_start=20.0, beta_end=5000.0,
                                  max_iters=1500, momentum=0.1, lr_start=0.08)
        alloc2 = min(deadline5 - time.time(), budget_s * 0.005)
        h_r, c5_r = _lp_polish(h_r, time.time() + alloc2, radius_init=0.04, max_iters=500)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final LP polish (most important for squeezing out last bits)
    if time.time() < t0 + budget_s * 0.95:
        hf, c5f = _lp_polish(best_h, t0 + budget_s * 0.97,
                             radius_init=0.03, max_iters=3000)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    # Phase 7: Coordinate descent final polish
    if time.time() < t0 + budget_s * 0.99:
        hcd, c5cd = _coord_descent(best_h, t0 + budget_s * 0.998, max_passes=120)
        if c5cd < best_c5:
            best_c5 = c5cd
            best_h = hcd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
