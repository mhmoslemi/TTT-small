def _exact_c5(h):
    n = len(h)

def _compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d + m] += (1.0 - h[:m])
                G[r, :m] -= h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g + m])
                G[r, g:g + m] -= h[:m]
    return G


def _project(v, target):
