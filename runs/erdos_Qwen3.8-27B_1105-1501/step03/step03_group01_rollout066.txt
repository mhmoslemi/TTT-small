    scaled = S / float(np.max(S)) if np.max(S) > 0 else S
    w = np.exp(tau * (scaled - scaled.max()))
    w /= w.sum()
    # Direction = -weighted sum of row gradients (mean-centered)
    dir = -(w @ G)  # shape (n,)
    dir -= dir.mean()  # ensure sum = 0
    norm = np.linalg.norm(dir)
    if norm < 1e-12:
        return None
    return dir / norm  # normalize

def _gen_smooth_dir(n, rng, n_harm=3):
    """Generate a random smooth direction (sum of sinusoids), mean-zero."""
    x = np.arange(n) / n
    dir = np.zeros(n)
    for _ in range(n_harm):
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        dir += amp * np.sin(2 * np.pi * freq * x + phase)
    dir -= dir.mean()
    norm = np.linalg.norm(dir)
    if norm < 1e-12:
        return None
    return dir / norm

def _gen_pair_dir(n, rng):
    """Generate a random pair direction."""
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    P = np.zeros(n)
    P[i] = 1.0
    P[j] = -1.0
    return P

def _gen_block_dir(n, rng):
    """Generate a random block direction."""
    block_len = rng.integers(2, min(8, n // 2))
    start = rng.integers(0, n - block_len)
    P = np.zeros(n)
    P[start:start+block_len] = 1.0
    # Subtract mean to make sum zero
    P -= P.mean()
    norm = np.linalg.norm(P)
    if norm < 1e-12:
        return None
    return P / norm

def _compute_line_data(h, P):
    """Precompute S, A, B for line search along direction P."""
    one_minus_h = 1.0 - h
    S = np.correlate(h, one_minus_h, mode='full')
    C1 = np.correlate(P, one_minus_h, mode='full')
    C2 = np.correlate(h, P, mode='full')
    A = C1 - C2
    C3 = np.correlate(P, P, mode='full')
    B = -C3
    return S, A, B

    #   If P[i] < 0: t*P[i] <= 1-h[i], so t >= (1-h[i])/P[i] (dividing by negative)
    
    # So:
    # Lower bounds on t: max of {-h[i]/P[i] for P[i]>0} and {(1-h[i])/P[i] for P[i]<0}

def _optimize_at_fixed_n(h_init, deadline, rng, max_iter=5000):
    """Main optimization loop at fixed n."""
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _c5_eval(h)
    cur_c5 = best_c5
    
    stall = 0
    it = 0
    
    while time.time() < deadline and it < max_iter:
        it += 1
        if stall > 100:
            break
        
        S = np.correlate(h, 1.0 - h, mode='full')
        max_S = float(np.max(S))
        
        # Generate and try several directions
        best_improvement = 0.0
        best_new_h = None
        best_new_c5 = cur_c5
        
        n_dirs = 30
        for di in range(n_dirs):
            # Mix of direction types
            r = di % 4
            if r == 0:
                P = _gen_subgrad_dir(h, S, tau=40.0, n_dirs=1)
            elif r == 1:
                P = _gen_smooth_dir(n, rng)
            elif r == 2:
                P = _gen_pair_dir(n, rng)
            else:
                P = _gen_block_dir(n, rng)
            
            if P is None:
                continue
            
            # Scale P to a reasonable magnitude
            # We want the step to be meaningful but not too large
            norm_P = np.linalg.norm(P)
            # Scale so that max |h + t*P| stays reasonable
            # Just normalize
            P = P / max(norm_P, 1e-12)
            
            # Compute feasible interval
            t_lo, t_hi = _feasible_interval(h, P)
            if t_hi - t_lo < 1e-10:
                continue
            
            # Try both P and -P (both directions)
            for sign in [1.0, -1.0]:
                P_s = sign * P
                t_lo_s, t_hi_s = _feasible_interval(h, P_s)
                if t_hi_s - t_lo_s < 1e-10:
                    continue
                
                S_l, A_l, B_l = _compute_line_data(h, P_s)
                t_opt, val_opt = _line_search_exact(h, P_s, S_l, A_l, B_l, t_lo_s, t_hi_s)
                
                if abs(t_opt) < 1e-12:
                    continue
                
                h_new = h + t_opt * P_s
                # Project to fix any numerical issues
                h_new = _project(h_new, target)
                c5_new = _c5_eval(h_new)
                
                if c5_new < best_new_c5 - 1e-12:
                    best_new_c5 = c5_new
                    best_new_h = h_new.copy()
        
        if best_new_h is not None:
            h = best_new_h
            cur_c5 = best_new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
