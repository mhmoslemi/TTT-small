def _optimal_coord_fast(h, i):
    """Find optimal t = h[i] that minimizes C5, given all other coords fixed."""
    n = len(h)
    h_i = h[i]
    scores = np.correlate(h, 1.0 - h, mode='full')
    n_full = 2 * n - 1
    base = n - 1
    
    # Compute derivative of each score w.r.t. h[i]
    deriv = np.zeros(n_full)
    for d in range(-(n-1), n):
        r = base + d
        if 0 <= i + d < n:
            deriv[r] += (1.0 - h[i + d])
        if 0 <= i - d < n:
            deriv[r] -= h[i - d]
    
    # For the d=0 lag (r = base), the dependence is t - t², not linear
    # c_base(t) = scores[base] + (t - h_i) - (t² - h_i²)
    #           = scores[base] + (t - h_i)(1 - t - h_i)
    
    # For other lags: c_r(t) = scores[r] + deriv[r] * (t - h_i)
    
    # Candidates for optimal t
    candidates = [0.0, 1.0, h_i, 0.5]
    
    # Intersections of quadratic with each linear
    for r in range(n_full):
        if r == base:
            continue
        # scores[r] + deriv[r]*u = scores[base] + u*(1-2*h_i-u) where u = t - h_i
        # u² + (deriv[r] - (1-2*h_i))*u + (scores[r] - scores[base]) = 0
        a = 1.0
        b = deriv[r] - (1.0 - 2.0*h_i)
        c = scores[r] - scores[base]
        disc = b*b - 4.0*a*c
        if disc >= 0:
            sq = np.sqrt(disc)
            for u_c in [(-b + sq)/(2*a), (-b - sq)/(2*a)]:
                t_c = h_i + u_c
                if -1e-10 <= t_c <= 1.0 + 1e-10:
                    candidates.append(np.clip(t_c, 0.0, 1.0))
    
    # Also add a fine grid to be safe

        candidates.append(k / 20.0)
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, 0.0, 1.0)
    
    # Evaluate C5(t) = max_r c_r(t) for each candidate
