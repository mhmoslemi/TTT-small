def _lp_direction(h, scores, G, radius):
    """Find LP direction to reduce max(scores)."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize t
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores  # G @ delta - t <= -scores, i.e., new_scores <= scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _soft_subgrad(h, tau):

    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx

    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = G.T @ w  # shape (n,)
    return grad / dx  # normalize

    stall = 0

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        P = _lp_direction(h, scores, G, radius)
        
        if P is None or float(np.max(np.abs(P))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        # Ensure mean-preserving
        P = P - np.mean(P)
        
        # Get feasible interval
        t_lo, t_hi = _feasible_interval(h, P)
        if t_hi - t_lo < 1e-10:
            stall += 1
            radius *= 0.7
            continue
        
        # Exact line search
        S0, B, Q = _quadratic_line(h, P)
        t_opt, obj_opt = _line_search_exact(S0, B, Q, t_lo, t_hi)
        
        # Verify with actual evaluation
        h_cand = h + t_opt * P
        h_cand = _project(h_cand, target)
        c5_cand = _exact_c5(h_cand)
        
        if c5_cand < cur_c5 - 1e-12:
            h = h_cand
            cur_c5 = c5_cand
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            # Try a smaller step
            t_try = t_opt * 0.5
            h_cand2 = h + t_try * P
            h_cand2 = _project(h_cand2, target)
            c5_cand2 = _exact_c5(h_cand2)
            if c5_cand2 < cur_c5 - 1e-12:
                h = h_cand2
                cur_c5 = c5_cand2
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
    
    return best_h, best_c5

def _cycling_optimize(h, deadline, top_k=5):
    """Cycle through active lags and try to reduce each one specifically."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline and stall < 60:
        scores = np.correlate(h, 1.0 - h, mode='full')
        active = _active_lags(h, top_k)
        G = _grad_matrix(h)
        
        improved = False
        for r in active:
            if time.time() > deadline:
                break
            # Find direction that reduces lag r while not increasing others too much
            # Use the gradient of lag r as direction
            grad_r = G[r].copy()
            grad_r = grad_r - np.mean(grad_r)  # mean-preserving
            
            # Also try negative (in case we need to increase the lag to reduce others)
            for sign in [1.0, -1.0]:
                P = sign * grad_r
                if float(np.max(np.abs(P))) < 1e-12:
                    continue
                
                t_lo, t_hi = _feasible_interval(h, P)
                if t_hi - t_lo < 1e-10:
                    continue
                
                S0, B, Q = _quadratic_line(h, P)
                t_opt, obj_opt = _line_search_exact(S0, B, Q, t_lo, t_hi)
                
                h_cand = _project(h + t_opt * P, target)
                c5_cand = _exact_c5(h_cand)
                
                if c5_cand < cur_c5 - 1e-12:
                    h = h_cand
                    cur_c5 = c5_cand
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _gradient_polish(h, deadline, tau=100.0, lr=0.01):
    """Gradient-based polynomial using soft subgradient."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    momentum = np.zeros(n)
    
    while time.time() < deadline and stall < 50:
        grad = _soft_subgrad(h, tau)
        momentum = 0.9 * momentum + 0.1 * grad
        h_new = _project(h - lr * momentum, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            lr *= 0.9
    
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=50):
    """Try swapping small amounts between pairs of coordinates."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Random pairs for speed
        rng = np.random.default_rng(_pass * 12345)
        for _ in range(200):
            if time.time() > deadline:
                break
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j == k:
                continue
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-10:
                    continue
                for frac in [0.1, 0.3, 0.6, 1.0]:
                    h_t = h.copy()
                    h_t[j] += dj * am * frac
                    h_t[k] += dk * am * frac
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def _evolutionary_refine(h_init, deadline, pop_size=10, seed=42):
    """Simple evolutionary search."""
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = _project(h_init + rng.normal(0, 0.05, n), target)
        pop.append(h_i)
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.04
    
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.6 * w1 + 0.4 * w2
        child += rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst] - 1e-14:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f - 1e-14:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.01, mut_scale * 0.95)
            else:
                mut_scale = min(0.12, mut_scale * 1.05)
        else:
            mut_scale = min(0.12, mut_scale * 1.03)
    
    return best_h, best_f
