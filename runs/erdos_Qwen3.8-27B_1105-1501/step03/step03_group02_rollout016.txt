def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _all_scores(h):

def _score_grad(h, d):
    """Gradient of S_d with respect to h, where d is the signed lag."""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            # S_d = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # dS/dh[j] = (1-h[j-d]) for j>=d (from h[i+d] term, i=j-d)
            #           - h[j+d] for j < n-d (from (1-h[i]) term, i=j)
            grad[d:] += (1.0 - h[:m])  # j = d, ..., n-1: 1-h[j-d] = 1-h[0],...,1-h[n-d-1]

    n = len(h)
    scores = _all_scores(h)
    # Soft-max weights
    max_s = float(np.max(scores))

        w[int(np.argmax(scores))] = 1.0

    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))

    return grad / norm

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g

def _score_tau(h, d):

    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:

        return np.sum(x[:n]) - target

    def _eq_jac(x):
        return np.concatenate([np.ones(n), [0.0]])

    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for d in active_lags:
        cons.append({'type': 'ineq', 'fun': _make_ineq_fun(d), 'jac': _make_ineq_jac(d)})

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return None
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _make_ineq_fun(d):
    return _ineq_fun_cache(d)

def _ineq_fun_cache(d):
    def f(x):
        return x[len(x)-1] - _score_tau(x[:len(x)-1], d)
    return f

def _make_ineq_jac(d):
    return _ineq_jac_cache(d)

def _ineq_jac_cache(d):
    def j(x):
        n = len(x) - 1
        return np.concatenate([-_score_grad(x[:n], d), [1.0]])
    return j

def _slsqp_refine(h_init, deadline, top_k=8, maxiter=200, ftol=1e-13):
    n = len(h_init)
    target = n / 2.0
    h_cur = h_init.copy()
    for _ref in range(3):
        if time.time() > deadline:
            break
        scores = _all_scores(h_cur)
        z0 = float(np.max(scores)) + 1e-6
        all_idx = np.argsort(scores)[::-1][:top_k]
        active_lags = [int(k - (n - 1)) for k in all_idx]
        x0 = np.concatenate([h_cur, [z0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        for d in active_lags:
            cons.append({'type': 'ineq', 'fun': _make_ineq_fun(d), 'jac': _make_ineq_jac(d)})
        def _cb2(xk):
            if time.time() > deadline:
                raise RuntimeError("timeout")
        def _obj2(x):
            return x[n]
        def _obj2_jac(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        try:
            result = minimize(_obj2, x0, method='SLSQP', jac=_obj2_jac,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                              callback=_cb2)
        except (RuntimeError, Exception):
            return h_cur, _exact_c5(h_cur)
        h_new = np.clip(result.x[:n], 0.0, 1.0)
        if abs(np.sum(h_new) - target) > 1e-4:
            h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < _exact_c5(h_cur) - 1e-14:
            h_cur = h_new
        else:
            break
        top_k = max(5, top_k - 3)
    return h_cur, _exact_c5(h_cur)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_restart_softmax(h_best, deadline, rng, n_trials=20):
    n = len(h_best)
    best_c5 = _exact_c5(h_best)
    best_h = h_best.copy()
    scale = 0.05
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        if best_c5 < 0.3808:
            break
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        per_dl = min(deadline, time.time() + max(5.0, (deadline - time.time()) * 0.4))
        h_try, c5_try = _nesterov_softmax_optimize(h_pert, per_dl,
                                                     beta_start=30, beta_end=300,
                                                     n_iter=150, lr_start=0.02,
                                                     lr_end=0.005, momentum=0.85)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Nesterov soft-max (broad)
    deadline1 = t0 + budget_s * 0.22
    h_sm, c5_sm = _nesterov_softmax_optimize(h, deadline1,
                                              beta_start=10, beta_end=150,
                                              n_iter=400, lr_start=0.03,
                                              lr_end=0.005, momentum=0.9)
    if c5_sm < best_c5:
        best_c5 = c5_sm
        best_h = h_sm.copy()

    # Phase 2: SLSQP with moderate active set
    deadline2 = t0 + budget_s * 0.32
    result = _slsqp_solve(best_h, deadline2, top_k=40, maxiter=150, ftol=1e-11)
    if result is not None:
        h_try, c5_try = result
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 3: SLSQP refinement (tight)
    deadline3 = t0 + budget_s * 0.42
    h_ref, c5_ref = _slsqp_refine(best_h, deadline3, top_k=10, maxiter=250, ftol=1e-12)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()

    # Phase 4: Nesterov soft-max (focused, high beta)
    deadline4 = t0 + budget_s * 0.55
    h_sm2, c5_sm2 = _nesterov_softmax_optimize(best_h, deadline4,
                                                beta_start=80, beta_end=500,
                                                n_iter=400, lr_start=0.015,
                                                lr_end=0.003, momentum=0.85)
    if c5_sm2 < best_c5:
        best_c5 = c5_sm2
        best_h = h_sm2.copy()

    # Phase 5: Pair search
    deadline5 = t0 + budget_s * 0.72
    h_pair, c5_pair = _pair_search(best_h, deadline5, n_rounds=10, rng=rng)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair.copy()

    # Phase 6: Multi-restart with smaller scale
    deadline6 = t0 + budget_s * 0.85
    h_mr, c5_mr = _multi_restart_softmax(best_h, deadline6, rng, n_trials=15)
    if c5_mr < best_c5:
        best_c5 = c5_mr
        best_h = h_mr.copy()

    # Phase 7: SLSQP final polish
    if time.time() < t0 + budget_s * 0.92:
        deadline7 = t0 + budget_s * 0.93
        h_f, c5_f = _slsqp_refine(best_h, deadline7, top_k=6, maxiter=300, ftol=1e-13)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.96:
        dl8 = t0 + budget_s * 0.97
        for dn in [-2, 2, -4, 4, -8, 8, -16, 16]:
            if time.time() > dl8:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 160 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _nesterov_softmax_optimize(h_r, min(dl8, time.time() + 12),
                                                    beta_start=50, beta_end=300,
                                                    n_iter=100, lr_start=0.015,
                                                    lr_end=0.005, momentum=0.85)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 9: Final pair polish
    if time.time() < t0 + budget_s * 0.99:
        h_fp, c5_fp = _pair_search(best_h, t0 + budget_s * 0.995, n_rounds=4, rng=rng)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
