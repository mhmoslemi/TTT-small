def _slsqp_epigraph(h, maxiter=200, ftol=1e-12):
    """Solve the epigraph problem using SLSQP."""
    n = len(h)
    K = 2 * n - 1
    target = n / 2.0
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def constr_func(x, ci):
        """Constr ci: T - g_{ci}(h) >= 0"""
        h = x[1:]
        T = x[0]
        corr = _corr(h)
        return T - corr[ci]
    
    # Actually, SLSQP requires all constraints in one function
    # Let's use a different approach: define the constraint as a vector
    
    def constr(x):
        """Inequality constraints: T - g_i(h) >= 0 for all i"""
        h = x[1:]
        T = x[0]
        corr = _corr(h)
        return T - corr
    
    def constr_jac(x):
        h = x[1:]
        n = len(h)
        K = 2 * n - 1
        J = np.zeros((K, n + 1))
        J[:, 0] = 1.0
        # Compute -d g_d / d h[j] for each lag d and coordinate j
        for j in range(n):
            for k in range(K):
                d = k - (n - 1)
                val = 0.0
                if 0 <= j - d < n:
                    val += (1.0 - h[j - d])
                if 0 <= j + d < n:
                    val -= h[j + d]
                J[k, j + 1] = -val
        return J
    
    def eq_constr(x):
        return np.sum(x[1:]) - target
    
    def eq_constr_jac(x):
        J = np.zeros(n + 1)
        J[1:] = 1.0
        return J
    
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    
    x0 = np.zeros(n + 1)
    x0[0] = float(np.max(_corr(h))) + 0.01
    x0[1:] = h
    
    cons = [

        {'type': 'eq', 'fun': eq_constr, 'jac': eq_constr_jac},

    res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,

    return np.correlate(h, 1.0 - h, mode='full')

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

    """Compute gradient matrix: G[k, j] = d(g_k)/d(h[j]) for all k, j."""
    n = len(h)

    w[active] = 1.0 / len(active)
    sg = G.T @ w
    sg -= np.mean(sg)  # Maintain sum constraint
    return sg, mx, active

        return np.sum(x[1:]) - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[1:] = 1.0
        return J
    
    bounds = [(None, None)] + [(0.0, 1.0)] * n

    cons = [

        {'type': 'eq', 'fun': eq, 'jac': eq_jac},

g_k = sum_i h[i] * (1 - h[i - d_k]) where d_k = k - (n-1)

The sum is over i where both i and i-d_k are in [0, n).

∂g_k / ∂h[j]:
- j appears as i (first factor): when i=j, need 0≤j<n (always true) and 0≤j-d_k<n
  Contribution: (1 - h[j-d_k])
- j appears as i-d_k (second factor): when i-d_k=j, i.e., i=j+d_k, need 0≤j+d_k<n and 0≤j<n (always true)
  Contribution: -h[j+d_k]

So ∂g_k/∂h[j] = (1-h[j-d_k])*I(0≤j-d_k<n) - h[j+d_k]*I(0≤j+d_k<n)

def _coord_descent(h, deadline, max_passes=100, direction=1):
    n = len(h)
    K = 2 * n - 1
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    order = np.arange(n) if direction == 1 else np.arange(n - 1, -1, -1)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        for j in order:
            if time.time() > deadline:
                break
            corr = _corr(h)
            hj = h[j]
            C = np.zeros(K)
            D = np.zeros(K)
            for k in range(K):
                d = k - (n - 1)
                term1 = 0.0
                if 0 <= j - d < n:
                    term1 = hj * (1.0 - h[j - d])
                term2 = 0.0
                if 0 <= j + d < n:
                    term2 = h[j + d] * (1.0 - hj)
                C[k] = corr[k] - term1 - term2
                D[k] = 0.0
                if 0 <= j - d < n:
                    D[k] += (1.0 - h[j - d])
                if 0 <= j + d < n:
                    D[k] -= h[j + d]

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(C + c1 * D))
            f2 = float(np.max(C + c2 * D))
            for _ in range(45):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(C + c1 * D))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(C + c2 * D))
            h[j] = (lo + hi) / 2.0

        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    from scipy.optimize import linprog
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return None

    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(14):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.12, max_iters=2000, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _corr(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _block_descent(h, deadline, block_size=4, max_passes=50):
    n = len(h)
    K = 2 * n - 1
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_blocks = (n + block_size - 1) // block_size

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for b in range(n_blocks):
            if time.time() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            bsize = end - start
            h_block = h[start:end].copy()
            h_fixed = h.copy()
            h_fixed[start:end] = 0.0
            sum_fixed = float(np.sum(h_fixed))
            sum_block = float(np.sum(h_block))
            target_block = sum_block

            def block_obj(x):
                h_full = h_fixed.copy()
                h_full[start:end] = x
                return _exact_c5(h_full)

            bounds = [(0.0, 1.0)] * bsize
            A_eq_b = np.ones((1, bsize))
            b_eq_b = np.array([target_block])
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_block}]
            try:
                res = minimize(block_obj, h_block, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-12, 'iprint': 0})
                x_new = np.clip(res.x, 0.0, 1.0)
                x_new = x_new * target_block / max(float(np.sum(x_new)), 1e-15)
                x_new = np.clip(x_new, 0.0, 1.0)
                h_new = h.copy()
                h_new[start:end] = x_new
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _check_best(h_try, c5_try):
        nonlocal best_h, best_c5, best_n
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)

    # Phase 1: Smooth optimization (18%)
    d1 = t0 + budget_s * 0.18
    h, c5 = _smooth_loop(h, d1, alphas=(5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 250.0))
    _check_best(h, c5)

    # Phase 2: SLSQP full epigraph (22%)
    d2 = t0 + budget_s * 0.40
    for attempt in range(20):
        if time.time() > d2:
            break
        h_s, c5_s = _slsqp_full(best_h, maxiter=400, ftol=1e-14)
        _check_best(h_s, c5_s)
        if time.time() < d2:
            h_s2, c5_s2 = _slsqp_reduced(h_s, maxiter=300, n_keep=15, ftol=1e-14)
            _check_best(h_s2, c5_s2)

    # Phase 3: LP polish (12%)
    d3 = t0 + budget_s * 0.52
    h_l, c5_l = _lp_loop(best_h, d3, radius_init=0.12, max_iters=3000, stall_limit=60)
    _check_best(h_l, c5_l)

    # Phase 4: SLSQP from multiple restarts (18%)
    d4 = t0 + budget_s * 0.70
    scale = 0.03
    for attempt in range(200):
        if time.time() > d4:
            break
        if best_c5 < 0.3806:
            break
        # Generate diverse starting points
        if attempt % 4 == 0:
            h_start = best_h + rng.normal(0, scale, best_n)
        elif attempt % 4 == 1:
            h_start = 1.0 - best_h + rng.normal(0, scale * 0.5, best_n)
        elif attempt % 4 == 2:
            h_start = best_h[::-1] + rng.normal(0, scale * 0.5, best_n)
        else:
            h_start = best_h + rng.normal(0, scale * 2, best_n)
        h_start = _project(h_start, best_n / 2.0)

        h_s, c5_s = _slsqp_reduced(h_start, maxiter=200, n_keep=12, ftol=1e-13)
        if c5_s < best_c5 - 1e-10:
            _check_best(h_s, c5_s)
            scale = max(0.005, scale * 0.85)
            # Follow up with full SLSQP
            if time.time() < d4:
                h_f, c5_f = _slsqp_full(h_s, maxiter=300, ftol=1e-14)
                _check_best(h_f, c5_f)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 5: Coordinate descent (10%)
    d5 = t0 + budget_s * 0.80
    h_cd, c5_cd = _coord_descent(best_h, min(d5, time.time() + budget_s * 0.05), max_passes=30, direction=1)
    _check_best(h_cd, c5_cd)
    if time.time() < d5:
        h_cd2, c5_cd2 = _coord_descent(best_h, d5, max_passes=30, direction=-1)
        _check_best(h_cd2, c5_cd2)

    # Phase 6: Subgradient polish (5%)
    d6 = t0 + budget_s * 0.85
    h_sg, c5_sg = _subgrad_polish(best_h, d6, n_steps=2000, alpha0=0.004)
    _check_best(h_sg, c5_sg)

    # Phase 7: Resolution search (10%)
    d7 = t0 + budget_s * 0.95
    for dn in [0, 1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 8, -8, 12, -12, 16, -16]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 320:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d7, time.time() + budget_s * 0.02)
        h_r, c5_r = _smooth_loop(h_r, sub_d, alphas=(20.0, 50.0, 100.0))
        if time.time() < d7 and c5_r <= best_c5 + 0.005:
            sub_d2 = min(d7, time.time() + budget_s * 0.015)
            h_r, c5_r = _slsqp_reduced(h_r, maxiter=200, n_keep=15, ftol=1e-13)
        if c5_r < best_c5 - 1e-10:
            _check_best(h_r, c5_r)

    # Phase 8: Final polish (5%)
    d8 = t0 + budget_s * 0.99
    if time.time() < d8:
        h_f, c5_f = _slsqp_full(best_h, maxiter=500, ftol=1e-15)
        _check_best(h_f, c5_f)
    if time.time() < d8:
        h_f2, c5_f2 = _lp_loop(best_h, d8, radius_init=0.03, max_iters=2000, stall_limit=100)
        _check_best(h_f2, c5_f2)
    if time.time() < d8:
        h_f3, c5_f3 = _coord_descent(best_h, d8, max_passes=20, direction=1)
        _check_best(h_f3, c5_f3)

    # Final projection and validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 0.001:
        # Something went wrong, use the original
        final_c5 = best_c5
    return best_h, final_c5, best_n
