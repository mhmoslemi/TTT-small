def _smooth_jac(x, beta):
    scores = _scores(x)
    G = _grad_matrix(x)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / float(np.sum(exp_s))
    grad = G.T @ w
    return grad

def _sum_constraint(x, target):
    return np.sum(x) - target

def _slsqp_optimize(h, target, beta, max_iter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': lambda x: np.ones(len(x)), 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
