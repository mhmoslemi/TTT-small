    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    downtown = 0
    k = k_active

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if downtown > 100:
            break

        scores, G = _scores_G(h)
        p = _lp_dir(h, scores, G, radius, k)

        if p is None:
            downtown += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            downtown = 0
            radius = min(radius * 1.04, 0.4)
            if it % 200 == 0:
                k = min(k + 1, 2 * n - 1)
        else:
            downtown += 1
            radius *= 0.72
            if downtown % 15 == 0:
                k = max(5, k - 2)
                if radius < 0.008:
                    radius = 0.04

    return best_h, best_c5


def _smooth_obj(x, beta):
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _smooth_jac(x, beta):
    n = len(x)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - x[:m])
                G[r, :m] -= x[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - x[g:])
                G[r, g:] -= x[:m]
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    return G.T @ w


def _sum_fun(x, target):
    return np.sum(x) - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _slsqp_polish(h, target, beta, max_iter=100):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    from scipy.optimize import minimize
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _fourier_perturb(h, rng, n_modes=3, amp_range=(0.008, 0.035)):
    n = len(h)
    x = (np.arange(n) + 0.5) / n
    h_p = h.copy()
    for _ in range(n_modes):
        freq = int(rng.integers(1, 10))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(2 * np.pi * freq * x + phase)
    return _project(h_p, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Main LP optimization (45%)
    deadline1 = t0 + budget_s * 0.45
    h1, c5_1 = _lp_loop(h, deadline1, radius_init=0.12, k_active=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP polish with high beta (8%)
    if time.time() < t0 + budget_s * 0.53:
        h_sp = _slsqp_polish(best_h, best_n / 2.0, beta=80.0, max_iter=120)
        c5_sp = _exact_c5(h_sp)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()
        h_sp2 = _slsqp_polish(best_h, best_n / 2.0, beta=200.0, max_iter=80)
        c5_sp2 = _exact_c5(h_sp2)
        if c5_sp2 < best_c5:
            best_c5 = c5_sp2
            best_h = h_sp2.copy()

    # Phase 3: LP with different parameters (10%)
    if time.time() < t0 + budget_s * 0.63:
        deadline3 = t0 + budget_s * 0.63
        for (r_init, k_act) in [(0.06, 8), (0.15, 25), (0.04, 5), (0.08, 30)]:
            if time.time() > deadline3:
                break
            h_t, c5_t = _lp_loop(best_h, min(deadline3, time.time() + budget_s * 0.025),
                                  radius_init=r_init, k_active=k_act)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 4: Resolution sweep (20%)
    if time.time() < t0 + budget_s * 0.83:
        deadline4 = t0 + budget_s * 0.83
        resolutions = []
        base = best_n
        for dn in [-16, -12, -8, -4, 4, 8, 12, 16, -6, -2, 2, 6, -1, 1, -3, 3, -10, 10, 20, -20]:
            nr = base + dn
            if 24 <= nr <= 200 and nr not in resolutions:
                resolutions.append(nr)
        for nr in resolutions:
            if time.time() > deadline4:
                break
            h_r = _resample(best_h, nr)
            h_r, c5_r = _lp_loop(h_r, min(deadline4, time.time() + budget_s * 0.012),
                                  radius_init=0.08, k_active=12)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = nr

    # Phase 5: Escape restarts (8%)
    deadline5 = t0 + budget_s * 0.91
    for attempt in range(40):
        if time.time() > deadline5:
            break
        if best_c5 < 0.3807:
            break
        h_pert = _fourier_perturb(best_h, rng, n_modes=3, amp_range=(0.005, 0.03))
        h_try, c5_try = _lp_loop(h_pert, min(deadline5, time.time() + budget_s * 0.01),
                                  radius_init=0.06, k_active=12)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(best_h)

    # Phase 6: Final polish (9%)
    if time.time() < t0 + budget_s * 0.96:
        deadline6 = t0 + budget_s * 0.96
        h_f, c5_f = _lp_loop(best_h, deadline6, radius_init=0.025, k_active=20, max_iters=8000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # SLSQP final micro-polish
    if time.time() < t0 + budget_s * 0.985:
        h_mp = _slsqp_polish(best_h, best_n / 2.0, beta=150.0, max_iter=60)
        c5_mp = _exact_c5(h_mp)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
