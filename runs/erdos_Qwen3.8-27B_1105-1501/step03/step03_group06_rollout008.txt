    """Find a in [a_lo, a_hi] that minimizes max_m (corr[m] + a * s[m])"""
    # Ternary search on convex piecewise-linear function
    for _ in range(40):
        m1 = a_lo + (a_hi - a_lo) / 3.0
        m2 = a_hi - (a_hi - a_lo) / 3.0
        f1 = np.max(corr + m1 * s)
        f2 = np.max(corr + m2 * s)
        if f1 < f2:
            a_hi = m2
        else:
            a_lo = m1
    return (a_lo + a_hi) / 2.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_matrix(h):
