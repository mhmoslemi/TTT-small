import time

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _bin_maxcount(b, n):

    """Return dict of lag -> max(N01, N10) for all lags."""
    M_all = (1 << n) - 1
    result = {}
    for d in range(1, n):

        n01 = ((M_all ^ b) & (b >> d) & mask_d).bit_count()

def _bin_to_array(b, n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = (b >> i) & 1
    return h

def _array_to_bin(h, n):
    b = 0
    for i in range(n):
        if h[i] >= 0.5:
            b |= (1 << i)
    return b

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:

        cnt = int(np.sum(mask))

    """Simulated annealing over binary states with exactly W ones."""
    cur_mc = _bin_maxcount(b, n)
    best_b = b
    best_mc = cur_mc

    masks = [(1 << (n - d)) - 1 for d in range(1, n)]
    M_all = (1 << n) - 1

            ones.append(i_0)
            zeros.remove(i_0)
            zeros.append(i_1)
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_b = b

    return best_b, best_mc

def _exact_c5(h):

    mc = 0
    for d in range(1, n):
        md = masks[d-1]
        v = (b & ~(b >> d) & md).bit_count()
        if v > mc: mc = v
        v = ((M_all ^ b) & (b >> d) & md).bit_count()
        if v > mc: mc = v

def _bin_to_arr(b, n):

def _sa_binary(b, n, W, deadline, rng, T0=20.0, T1=0.5, iters=200000):

    masks = [(1 << (n - d)) - 1 for d in range(1, n)]
    
    cur_mc = _bin_maxcount(b, n, masks, M_all)
    best_b = b
    best_mc = cur_mc
    
    ones = [i for i in range(n) if (b >> i) & 1]
    zeros = [i for i in range(n) if not ((b >> i) & 1)]

    for it in range(iters):

            break
        i1 = ones[rng.integers(len(ones))]
        i0 = zeros[rng.integers(len(zeros))]
        new_b = b ^ (1 << i1) ^ (1 << i0)
        new_mc = _bin_maxcount(new_b, n, masks, M_all)

            cur_mc = new_mc

            ones.append(i0)
            zeros.remove(i0)
            zeros.append(i1)
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_b = b
        T *= cool
    
    return best_b, best_mc
