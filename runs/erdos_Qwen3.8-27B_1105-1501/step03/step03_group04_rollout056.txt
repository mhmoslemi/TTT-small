def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores_d(h, d):
    """Gradient of score at lag d w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[:m] += (1.0 - h[d:d+m])
        g[d:d+m] -= h[:m]
    else:
        g2 = -d
        m = n - g2
        g[:m] += (1.0 - h[g2:g2+m])
        g[g2:g2+m] -= h[:m]
    return g

def _grad_max_scores(h):

    s = _scores(h)

    return g / len(idx)

        else:

    x = np.clip(v - tau, 0.0, 1.0)

    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)

**Key new idea: Use the structure of the problem to define a better parameterization.**

The problem is equivalent to: given a sequence h[0..n-1] in [0,1] with mean 0.5, minimize the maximum cross-correlation of h with (1-h).

Let me define q = 2h - 1, so q[i] ∈ [-1, 1] and mean(q) = 0. Then:
h = (1+q)/2
1-h = (1-q)/2
h*(1-h) = (1-q²)/4

The score at lag d:
score[d] = sum_{valid i} h[i]*(1-h[i+d]) = sum (1-q[i])*(1-q[i+d])/4
= (1/4) * sum (1 - q[i] - q[i+d] + q[i]*q[i+d])
= (1/4) * (sum_valid 1 - sum_valid q[i] - sum_valid q[i+d] + sum_valid q[i]*q[i+d])

Since mean(q) = 0, the sums of q[i] and q[i+d] are approximately 0 (exactly 0 if we had circular correlation, but not exactly for linear correlation).

So score[d] ≈ (1/4) * (m + sum_valid q[i]*q[i+d]) where m is the number of valid pairs.

To minimize max score[d], we want the autocorrelation of q to be as flat as possible (close to 0 for d≠0, and as negative as possible for d=0... wait no, for d=0, q[i]*q[i+d] = q[i]², so sum q[i]² is fixed by the variance of q).

Actually, for d=0: score[0] = (1/4) * sum (1-q[i]²) = (1/4) * (n - sum q[i]²)
To minimize score[0], maximize sum q[i]², i.e., make q as "bipartite" as possible (±1).

For d≠0: score[d] ≈ (1/4) * (m_d + sum q[i]*q[i+d])
To minimize, make the autocorrelation sum q[i]*q[i+d] negative (anti-correlated at those lags).

def _coord_refine(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            # Try moving h[j] and compensate on j2
            for delta in [0.02, 0.05, 0.1]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < -0.01 or h[j] + d > 1.01:
                        continue
                    # Find best compensating coordinate
                    h_try = h.copy()
                    h_try[j] += d
                    if d > 0:
                        comp_mask = (h_try < 1.0)
                    else:
                        comp_mask = (h_try > 0.0)
                    comp_mask[j] = False
                    if not comp_mask.any():
                        continue
                    if d > 0:
                        j2 = int(np.argmax(np.where(comp_mask, 1.0 - h_try, -1e9)))
                    else:
                        j2 = int(np.argmax(np.where(comp_mask, h_try, -1e9)))
                    h_try[j2] -= d
                    if h_try[j2] < -0.01 or h_try[j2] > 1.01:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, _exact_c5(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    def _time_left():
        return max(budget_s - (time.time() - t0), 0.0)

    # Phase 1: Local optimize from parent (20% of budget)
    dl = t0 + budget_s * 0.18
    h, c5 = _full_local_optimize(best_h, dl)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP polish (5%)
    dl = t0 + budget_s * 0.23
    h, c5 = _lp_polish(best_h, dl, radius=0.1, max_iters=1000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Multi-start from diverse initializations (45%)
    dl3 = t0 + budget_s * 0.70
    inits = []

    # Perturbed parent at same n
    for sc in [0.03, 0.06, 0.10, 0.15, 0.20]:
        hp = best_h + rng.standard_normal(best_n) * sc
        inits.append((hp, best_n))

    # Thue-Morse variants at various n
    for n_try in [best_n - 6, best_n - 3, best_n, best_n + 3, best_n + 6]:
        if n_try < 24 or n_try > 160:
            continue
        for amp in [0.2, 0.3, 0.4, 0.45]:
            inits.append((_thue_morse_h(n_try, amp), n_try))
        for sh in [1, 2, 3, 5, 7, 11]:
            inits.append((_tm_variant_h(n_try, 0.35, sh), n_try))

    # Sine and mixed
    for n_try in [best_n - 4, best_n + 4]:
        if n_try < 24 or n_try > 160:
            continue
        for freq in [1, 2, 3, 5]:
            for ph in [0.0, 0.5, 1.0, 1.5]:
                for amp in [0.2, 0.3, 0.4]:
                    inits.append((_sine_h(n_try, freq, amp, ph), n_try))
        inits.append((_m_sequence_like_h(n_try, 0.35, rng), n_try))
        inits.append((_blended_init(n_try, rng), n_try))
        inits.append((_binary_random_h(n_try, rng), n_try))

    # Other n values
    for n_try in [32, 40, 48, 56, 72, 80, 96, 112]:
        if abs(n_try - best_n) <= 4:
            continue
        inits.append((_thue_morse_h(n_try, 0.35), n_try))
        inits.append((_sine_h(n_try, 1, 0.3, 0.7), n_try))
        inits.append((_blended_init(n_try, rng), n_try))

    # Pre-filter: compute initial C5, sort by quality
    inits_scored = []
    for hi, ni in inits:
        c5i = _exact_c5(hi)
        inits_scored.append((c5i, hi, ni))
    inits_scored.sort(key=lambda x: x[0])
    inits_filtered = [(hi, ni) for c5i, hi, ni in inits_scored if c5i < best_c5 + 0.12]

    # Shuffle for diversity
    idx = rng.permutation(len(inits_filtered))
    inits_filtered = [inits_filtered[i] for i in idx]

    for hi, ni in inits_filtered:
        if time.time() > dl3:
            break
        remaining = dl3 - time.time()
        if remaining < 2.0:
            break
        n_left = len(inits_filtered)
        alloc = min(remaining / max(n_left, 1) * 1.5, budget_s * 0.018)
        alloc = max(alloc, 1.5)
        t_end = time.time() + alloc
        h_try, c5_try = _full_local_optimize(hi, t_end)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = ni

    # Phase 4: Resample to different n and optimize (10%)
    dl4 = t0 + budget_s * 0.82
    for dn in range(-12, 13, 2):
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 192 or n_try == best_n:
            continue
        hr = _resample_h(best_h, n_try)
        alloc = min(dl4 - time.time(), budget_s * 0.012)
        alloc = max(alloc, 1.0)
        h_try, c5_try = _full_local_optimize(hr, time.time() + alloc)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 5: Final intensive optimization on best (12%)
    dl5 = t0 + budget_s * 0.94
    h, c5 = _full_local_optimize(best_h, dl5)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: LP polish + coordinate refine (5%)
    dl6 = t0 + budget_s * 0.975
    h, c5 = _lp_polish(best_h, dl6, radius=0.05, max_iters=2000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    if time.time() < t0 + budget_s * 0.995:
        h, c5 = _coord_refine(best_h, t0 + budget_s * 0.998, max_passes=20)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
