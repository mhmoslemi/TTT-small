```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            x[free] += res / cnt
    return np.clip(x, 0.0, 1.0)

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    scores = corr * dx
    M = np.max(scores)
    exp_s = np.exp((scores - M) / tau)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = M + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * one_h[d:d + m]
                grad[d:] += -wk * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * one_h[:m]
                grad[:m] += -wk * h[g:g + m]
    return val, dx * grad
```

```python
            break
        lo_a, hi_a = 0.0, alpha_max
        for _ in range(25):
            mid = (lo_a + hi_a) / 2.0
            h_t = h + mid * p
            if exact_c5(h_t) < cur_c5:
                lo_a = mid
            else:
                hi_a = mid
        opt_a = hi_a
        h_t = h + opt_a * p
        h_t = project(h_t, target)
        c5_t = exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            h = h_t.copy()
            cur_c5 = c5_t
            stall = 0
        else:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = project(h0, target0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    rng = np.random.default_rng(seed)

    # Phase 1: SLSQP tau annealing on parent resolution
    d1 = t0 + budget_s * 0.28
    tau_list = [0.4, 0.15, 0.06, 0.02, 0.008, 0.003, 0.001]
    for tau in tau_list:
        if time.time() > d1:
            break
        max_it = 250 if n0 <= 100 else 120
        ftol = 1e-12
        x_new = slsqp_solve(h, tau, max_it, ftol)
        x_new = project(x_new, target0)
        c5_new = exact_c5(x_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = x_new.copy()
        h = x_new

    # Phase 2: Multi-resolution SLSQP
    d2 = t0 + budget_s * 0.58
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -12, 12, 16, -16, 20, -20, 24]
    tried_n = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 200 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_mid(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.022)
        for tau in [0.1, 0.03, 0.01, 0.003]:
            if time.time() > sub_dl:
                break
            max_it = 120 if n_try <= 100 else 60
            h_r = slsqp_solve(h_r, tau, max_it, 1e-11)
            h_r = project(h_r, n_try / 2.0)
        c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Perturbation + re-optimize
    d3 = t0 + budget_s * 0.80
    scale = 0.03
    for attempt in range(200):
        if time.time() > d3:
            break
        if best_c5 < 0.38075:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = project(h_p, best_n / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        for tau in [0.02, 0.005]:
            if time.time() > sub_dl:
                break
            max_it = 80
            h_p = slsqp_solve(h_p, tau, max_it, 1e-11)
            h_p = project(h_p, best_n / 2.0)
        c5_p = exact_c5(h_p)
        if c5_p < best_c5 - 1e-14:
            best_c5 = c5_p
            best_h = h_p.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.15, scale * 1.25)

    # Phase 4: LP refine + pair polish
    d4 = t0 + budget_s * 0.90
    if time.time() < d4:
        h_lp, c5_lp = _lp_refine(best_h, min(d4, time.time() + budget_s * 0.03), radius=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
        h_pp, c5_pp = pair_polish(best_h, d4, max_passes=80)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp.copy()

    # Phase 5: Final SLSQP polish at very small tau
    if time.time() < t0 + budget_s * 0.94:
        d5 = t0 + budget_s * 0.94
        x_f = slsqp_solve(best_h, 0.001, 100, 1e-13)
        x_f = project(x_f, best_n / 2.0)
        c5_f = exact_c5(x_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = x_f.copy()

    # Final pair polish
    if time.time() < t0 + budget_s * 0.96:
        h_fp, c5_fp = pair_polish(best_h, t0 + budget_s * 0.96, max_passes=100)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, best_n
```
