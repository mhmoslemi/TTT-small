```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_score(h, d):
    """Gradient of score at lag d with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[d:d+m] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] += -h[:m]
    return g

def _smooth_max_grad(h, beta=100.0):
    """Compute gradient of smooth max of all scores."""
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    # Shift for numerical stability
    shifted = beta * (scores - max_s)
    exp_s = np.exp(shifted)
    w = exp_s / float(np.sum(exp_s))
    
    # Composite gradient
    g = np.zeros(n)
    for idx in range(len(scores)):
        lag = idx - (n - 1)
        g += w[idx] * _grad_score(h, lag)
    return g

def _project(v, target):
    """Project v onto {x : sum(x) = target, 0 <= x <= 1}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Correct residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

1. Compute g = smooth_max_grad(h, β) (already computed)
2. i* = argmax(g), j* = argmin(g)
3. Transfer: h[i*] += t, h[j*] -= t
4. Find optimal t by 1D search (golden section or bisection on the derivative)

- t must satisfy: 0 <= h[i*] + t <= 1 and 0 <= h[j*] - t <= 1
- So t ∈ [max(0, -h[i*]), min(1-h[i*], h[j*])]
- Within this interval, find t that minimizes C5(h)

    """Project v onto {x in [0,1]^n : sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(res) > 1e-10:
        if res > 0:

    n = len(h)

    max_s = float(np.max(scores))
    shifted = beta * (scores - max_s)
    # Clip for numerical stability

    try:
        res = minimize(partial(_obj_fun, n_pts=n), x0, method='SLSQP',
                      jac=partial(_obj_jac, n_pts=n),
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-13},
                      callback=_timeout_cb)
    except RuntimeError:
        return h.copy(), c5_before
    except Exception:
        return h.copy(), c5_before
    
    h_out = np.clip(res.x[:n], 0.0, 1.0)
    h_out = _project(h_out, target)
    c5_out = _exact_c5(h_out)
    return h_out, c5_out

def _select_active_lags(scores, n, gap_thresh=0.1):
    """Select lags with score within gap_thresh of the maximum."""
    max_s = float(np.max(scores))
    active = []
    for idx, s in enumerate(scores):
        lag = idx - (n - 1)
        if s >= max_s - gap_thresh:
            active.append(lag)
    if 0 not in active:
        active.append(0)
    return active

def _slsqp_adaptive(h, deadline, max_rounds=10, gap_thresh=0.1):
    """Adaptive SLSQP: re-identify active set after each round."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        scores = _all_scores(best_h)
        max_s = float(np.max(scores))
        active = _select_active_lags(scores, n, gap_thresh)
        
        # Adaptive gap: tighten over rounds
        cur_gap = gap_thresh * (0.7 ** rnd)
        if cur_gap < 0.01:
            cur_gap = 0.01
        active = _select_active_lags(scores, n, cur_gap)
        
        per_dl = min(deadline, time.time() + (deadline - time.time()) / (max_rounds - rnd))
        h_sol, c5_sol = _slsqp_one(best_h, active, per_dl)
        
        if c5_sol < best_c5 - 1e-14:
            best_h = h_sol
            best_c5 = c5_sol
        else:
            # Try with slightly larger gap
            if rnd < max_rounds - 2:
                active2 = _select_active_lags(scores, n, cur_gap * 2)
                h_sol2, c5_sol2 = _slsqp_one(best_h, active2, per_dl)
                if c5_sol2 < best_c5 - 1e-14:
                    best_h = h_sol2
                    best_c5 = c5_sol2
                else:
                    break
    
    return best_h, best_c5

def _smooth_descent(h, deadline, beta_start=20.0, beta_end=300.0):
    """Smooth max gradient descent with adaptive beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_iter = 0
    
    while time.time() < deadline:
        n_iter += 1
        # Adaptive beta
        frac = min(1.0, n_iter / 200.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        g = _smooth_grad(h, beta)
        g -= np.mean(g)  # Ensure sum(g) = 0
        
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-12:
            break
        
        g = g / gnorm  # Normalize
        
        # Backtracking line search
        alpha = 0.05
        improved = False
        for _ls in range(30):
            h_try = h - alpha * g
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                h = h_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        if not improved:
            # Try with smaller step and different beta
            beta *= 0.7
            if beta < 5.0:
                break
    
    return best_h, best_c5

def _golden_section(f, a, b, n_iter=40, deadline=None):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d
    
    for _ in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    x_opt = (a + b) / 2.0
    return x_opt, f(x_opt)
