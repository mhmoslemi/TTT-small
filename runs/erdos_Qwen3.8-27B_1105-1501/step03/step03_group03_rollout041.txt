    """Project v onto {x: 0 <= x <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = float(n)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)

        cnt = np.sum(mask)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_rows(h, rows):
    """Compute gradient rows for specific lag indices."""
    n = len(h)
    G = np.zeros((len(rows), n))
    for idx, s in enumerate(rows):
        d = s - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] -= h[:m]
    return G


def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = float(n)

            x = np.clip(x, 0.0, 1.0)
    return x


def _slp_iter(h, target, scores, active_idx, radius):
    """One SLP iteration. Returns (new_h, improved) or (None, False)."""
    n = len(h)
    G_act = _grad_rows(h, active_idx)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = G_act @ h - scores[active_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = []
    for j in range(n):
        lo = max(0.0, h[j] - radius)
        hi = min(1.0, h[j] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, False
    
    h_lp = res.x[:n]
    d = h_lp - h
    
    if np.max(np.abs(d)) < 1e-14:
        return None, False
    
    # Line search
    c5_cur = _exact_c5(h)
    best_a = 0.0
    best_c5 = c5_cur
    
    # Coarse search
    for a in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        h_try = _project(h + a * d, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_a = a
    
    if best_a == 0.0:
        return None, False
    
    # Refine with golden section
    step = 0.2
    lo = max(0.0, best_a - step)
    hi = min(1.0, best_a + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * d, target))
    if opt_c5 < best_c5:
        best_a, best_c5 = opt_a, opt_c5
    
    if best_c5 < c5_cur - 1e-14:
        return _project(h + best_a * d, target), True
    return None, False

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_rows(h, rows):
    n = len(h)
    G = np.zeros((len(rows), n))
    for idx, s in enumerate(rows):
        d = s - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] -= h[:m]
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = float(n)
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _slp_subproblem(h, target, scores, active_idx, radius):
    n = len(h)
    G_act = _grad_rows(h, active_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = G_act @ h - scores[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = []
    for j in range(n):
        lo = max(0.0, h[j] - radius)
        hi = min(1.0, h[j] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x[:n]


def _line_search(h, d, target, c5_cur, n_coarse=10):
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.05, 1.0, n_coarse)
    for a in alphas:
        h_try = _project(h + a * d, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    step = 1.0 / n_coarse
    lo = max(0.0, best_a - step)
    hi = min(1.0, best_a + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * d, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _slp_loop(h, target, deadline, radius_init=0.12, max_iters=20000, active_k=16):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break

        scores = _scores(h)
        if active_k < len(scores):
            top_idx = np.argsort(scores)[-active_k:]
        else:
            top_idx = np.arange(len(scores))

        h_lp = _slp_subproblem(h, target, scores, top_idx, radius)
        if h_lp is None:
            radius *= 0.5
            stall += 1
            if radius < 1e-7:
                break
            continue

        d = h_lp - h
        d_max = float(np.max(np.abs(d)))
        if d_max < 1e-14:
            radius *= 0.5
            stall += 1
            if radius < 1e-7:
                break
            continue

        best_a, new_c5 = _line_search(h, d, target, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + best_a * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.06

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _active_gradient_step(h, target, deadline, n_top=12, max_iters=5000):
    """Refined SLP focusing on top scores with tighter radius."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.03
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break

        scores = _scores(h)
        top_idx = np.argsort(scores)[-n_top:]
        s_max = scores[top_idx].max()
        # Expand active set if scores are close
        threshold = s_max - 0.01 * abs(s_max)
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) > 40:
            active_idx = np.argsort(scores)[-40:]

        h_lp = _slp_subproblem(h, target, scores, active_idx, radius)
        if h_lp is None:
            radius *= 0.6
            stall += 1
            if radius < 1e-8:
                break
            continue

        d = h_lp - h
        if float(np.max(np.abs(d))) < 1e-14:
            radius *= 0.6
            stall += 1
            if radius < 1e-8:
                break
            continue

        best_a, new_c5 = _line_search(h, d, target, cur_c5, n_coarse=8)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + best_a * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.1)
        else:
            stall += 1
            radius *= 0.8

    return best_h, best_c5


def _pair_polish(h, target, deadline, max_passes=50):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)
        s_max = float(np.max(scores))
        # Find coordinates with large gradient contribution to max scores
        active = np.where(scores >= s_max - 0.02 * s_max)[0]
        G_act = _grad_rows(h, active)
        grad = G_act.mean(axis=0)
        grad -= np.mean(grad)

        # Try moving pairs in opposite gradient directions
        sorted_idx = np.argsort(np.abs(grad))[::-1]
        tried = 0
        for ii in range(min(15, n)):
            if time.time() > deadline:
                break
            for jj in range(ii + 1, min(ii + 20, n)):
                i, j = sorted_idx[ii % n], sorted_idx[jj % n]
                if i == j:
                    continue
                if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
                    continue
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                    continue
                # Move: decrease h at high-grad, increase at low-grad
                sg_i = 1.0 if grad[i] > 0 else -1.0
                sg_j = -sg_i
                a_max = 1e9
                if sg_i > 0:
                    a_max = min(a_max, (1.0 - h[i]) / sg_i)
                else:
                    a_max = min(a_max, h[i] / (-sg_i))
                if sg_j > 0:
                    a_max = min(a_max, (1.0 - h[j]) / sg_j)
                else:
                    a_max = min(a_max, h[j] / (-sg_j))
                if a_max < 1e-10:
                    continue
                best_a = 0.0
                best_c5_try = best_c5
                for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                    a = a_max * t
                    h_try = h.copy()
                    h_try[i] += sg_i * a
                    h_try[j] += sg_j * a
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5_try - 1e-14:
                        best_c5_try = c5_try
                        best_a = a
                if best_a > 0:
                    best_c5_try_h = h.copy()
                    best_c5_try_h[sorted_idx[ii % n]] += sg_i * best_a
                    best_c5_try_h[sorted_idx[jj % n]] += sg_j * best_a
                    best_c5 = _exact_c5(best_c5_try_h)
                    h = best_c5_try_h
                    best_h = h.copy()
                    improved = True
                    tried += 1
        if not improved:
            break

    return best_h, best_c5


def _random_perturb(h, target, rng, scale=0.03):
    n = len(h)
    d = rng.standard_normal(n)
    d -= np.mean(d)
    d = d / (np.max(np.abs(d)) + 1e-15)
    h_new = _project(h + scale * d, target)
    return h_new
