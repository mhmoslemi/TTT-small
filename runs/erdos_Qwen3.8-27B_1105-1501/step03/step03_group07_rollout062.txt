def _lp_solve(h, R, top_k=None, lam=0.0):
    """Solve the tangent plane LP. Returns direction d or None."""
    n = len(h)
    raw = _all_raw(h)
    grads = _all_gradients(h)
    
    # Select which constraints to include
    if top_k is not None and top_k < 2 * n - 1:
        sorted_idx = np.argsort(-raw)[:top_k]
    else:
        sorted_idx = np.arange(2 * n - 1)
    
    m = len(sorted_idx)
    
    # Variable ordering: [t, p_0..p_{n-1}, q_0..q_{n-1}]
    # Total: 1 + 2n
    nv = 1 + 2 * n
    
    # Objective: minimize t + lam * (sum p + sum q)
    c = np.zeros(nv)
    c[0] = 1.0
    if lam > 0:

        c[1+n:1+2*n] = lam

    # For each selected offset s: -t + grad_s^T p - grad_s^T q <= -F_s

    A_ub = np.zeros((m + 1, nv))
    b_ub = np.zeros(m + 1)
    
    for i, s in enumerate(sorted_idx):
        A_ub[i, 0] = -1.0
        A_ub[i, 1:1+n] = grads[s]
        A_ub[i, 1+n:1+2*n] = -grads[s]
        b_ub[i] = -raw[s]

def _all_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))

    for r in range(2 * n - 1):
        d = r - (n - 1)

            if m > 0:
                grads[r, d:d+m] += omh[:m]
                grads[r, :m] -= h[d:d+m]
        else:

            m = n - l
            if m > 0:
                grads[r, :m] += omh[l:l+m]
                grads[r, l:l+m] -= h[:m]

import numpy as np
import time
from scipy.optimize import linprog


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _raw_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _all_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grads[r, d:d + m] += omh[:m]
                grads[r, :m] -= h[d:d + m]
        else:
            l = -d
            m = n - l
            if m > 0:
                grads[r, :m] += omh[l:l + m]
                grads[r, l:l + m] -= h[:m]
    return grads


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lp_direction(h, R):
    n = len(h)
    raw = _raw_vals(h)
    grads = _all_grads(h)
    m = 2 * n - 1
    nv = 1 + 2 * n

    c = np.zeros(nv)
    c[0] = 1.0

    A_ub = np.zeros((m + 1, nv))
    b_ub = np.zeros(m + 1)
    for s in range(m):
        A_ub[s, 0] = -1.0
        A_ub[s, 1:1 + n] = grads[s]
        A_ub[s, 1 + n:] = -grads[s]
        b_ub[s] = -raw[s]
    A_ub[m, 1:] = 1.0
    b_ub[m] = R

    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:] = -1.0
    b_eq = np.zeros(1)

    bounds = [(-1e4, 1e4)]
    for i in range(n):
        if h[i] < 1e-8:
            bounds.append((0.0, R))
            bounds.append((0.0, 0.0))
        elif h[i] > 1.0 - 1e-8:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, R))
        else:
            bounds.append((0.0, R))
            bounds.append((0.0, R))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None

    p = res.x[1:1 + n]
    q = res.x[1 + n:]
    d = p - q
    d -= np.mean(d)
    maxmag = float(np.max(np.abs(d)))
    if maxmag < 1e-10:
        return None
    d /= maxmag
    return d


def _evaluate_alpha(h, d, a, target):
    ht = np.clip(h + a * d, 0.0, 1.0)
    ht = _project(ht, target)
    return _exact_c5(ht)


def _line_search(h, d, target, c5_cur):
    n = len(h)
    alpha_max = 2.0
    for i in range(n):
        if d[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    alpha_max = min(max(alpha_max, 1e-6), 2.0)

    best_a = 0.0
    best_c = c5_cur
    best_h = h.copy()

    alphas = np.linspace(0.02, alpha_max, 10)
    for a in alphas:
        c_t = _evaluate_alpha(h, d, a, target)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
            ht = np.clip(h + a * d, 0.0, 1.0)
            best_h = _project(ht, target)

    if best_a == 0.0:
        return h, c5_cur

    lo = max(0.0, best_a * 0.5)
    hi = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _evaluate_alpha(h, d, c1, target)
    f2 = _evaluate_alpha(h, d, c2, target)
    for _ in range(22):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _evaluate_alpha(h, d, c1, target)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _evaluate_alpha(h, d, c2, target)

    opt_a = (lo + hi) / 2.0
    c_opt = _evaluate_alpha(h, d, opt_a, target)
    if c_opt < best_c:
        ht = np.clip(h + opt_a * d, 0.0, 1.0)
        return _project(ht, target), c_opt
    return best_h, best_c


def _softmax_direction(h, T):
    n = len(h)
    raw = _raw_vals(h)
    grads = _all_grads(h)
    shifted = raw - float(np.max(raw))
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.zeros(len(raw))
        w[int(np.argmax(raw))] = 1.0
    else:
        w /= wsum
    g = w @ grads
    g -= np.mean(g)
    gnorm = float(np.linalg.norm(g))
    if gnorm < 1e-12:
        return None
    g /= gnorm
    return g


def _tangent_optimize(h, deadline, target, max_iters):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    R = n * 0.5
    fb_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break

        d = None
        d = _lp_direction(h, R)
        if d is None or fb_count > 8:
            g = _softmax_direction(h, T=0.15 + 0.05 * (stall % 8))
            if g is not None:
                d = g
                d -= np.mean(d)
                gnorm = float(np.linalg.norm(d))
                if gnorm > 1e-12:
                    d /= gnorm
            fb_count += 1
        else:
            fb_count = 0

        if d is None:
            break

        new_h, new_c5 = _line_search(h, d, target, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = new_h
            cur_c5 = new_c5
            stall = 0
            R = min(R * 1.03, 3.0 * n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            R = max(R * 0.85, 2.0)
            if stall % 30 == 0:
                h = _project(h + rng.normal(0, 0.02 + 0.01 * (stall // 30), n), target)
                cur_c5 = _exact_c5(h)

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _gen_binary(n, rng):
    v = (rng.random(n) > 0.5).astype(float)
    return _project(v, n / 2.0)


def _gen_perturbed(h, rng, sigma):
    n = len(h)
    return _project(h + rng.normal(0, sigma, n), n / 2.0)


def _gen_shifted(h, rng):
    n = len(h)
    shift = rng.integers(1, max(2, n // 6))
    return _project(np.roll(h, int(shift)), n / 2.0)


def _gen_block(n, rng):
    half = n // 2
    v = np.zeros(n)
    v[:half] = 1.0
    return _project(v + rng.normal(0, 0.1, n), n / 2.0)


def _pair_polish(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)

    for _ in range(5000):
        if time.time() > deadline:
            break
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
            continue
        if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
            continue
        for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
            a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
            a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
            a_max = min(a_max_j, a_max_k)
            if a_max < 1e-10:
                continue
            for frac in [0.3, 0.6, 1.0]:
                a = a_max * frac
                h_t = h.copy()
                h_t[j] += dj * a
                h_t[k] += dk * a
                c_t = _exact_c5(h_t)
                if c_t < best_c5 - 1e-12:
                    best_c5 = c_t
                    best_h = h_t.copy()
                    h = h_t.copy()
                    cur_c5 = c_t
                    break

    return best_h, best_c5
