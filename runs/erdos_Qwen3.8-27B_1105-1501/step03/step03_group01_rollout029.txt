            m = n - gp
            if m > 0:
                G[r, gp:] += (1.0 - h[:m])
                if gp > 0:
                    G[r, :m] += -h[gp:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores * dx
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j] + 1e-15), min(radius, 1.0 - h[j] - 1e-15)))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(15):
        a = alpha_max * (i + 1) / 15.0
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h0, deadline, radius_init=0.15):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        p = _lp_dir(h, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_refine(h0, deadline, max_passes=50):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = (1.0 - h[j]) / abs(dj) if dj != 0 else 1.0
                    if dk != 0:
                        am = min(am, (1.0 - h[k]) / abs(dk))
                    if am < 1e-12:
                        continue
                    for frac in [0.5, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5


def _evolutionary(h_init, deadline, pop_size=12, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.035
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2) + rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.92)
            else:
                mut_scale = min(0.1, mut_scale * 1.06)
        else:
            mut_scale = min(0.1, mut_scale * 1.04)
    return best_h, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_end = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP with increasing K (35% budget)
    d1 = t0 + budget_s * 0.35
    K_seq = [30, 60, 100, 150]
    for K in K_seq:
        sub_dl = min(d1, time.time() + budget_s * 0.07)
        h_temp, c5_temp = _slsqp_opt(best_h, K, sub_dl, maxiter=300)
        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h = h_temp.copy()
        if time.time() > d1:
            break

    # Phase 2: Projected gradient (10% budget)
    d2 = t0 + budget_s * 0.45
    h_temp, c5_temp = _proj_grad(best_h, 80, d2, lr=0.004, beta=0.88)
    if c5_temp < best_c5:
        best_c5 = c5_temp
        best_h = h_temp.copy()
    h_temp, c5_temp = _slsqp_opt(best_h, 120, min(d2, time.time() + budget_s * 0.04), maxiter=200)
    if c5_temp < best_c5:
        best_c5 = c5_temp
        best_h = h_temp.copy()

    # Phase 3: LP-based local search (15% budget)
    d3 = t0 + budget_s * 0.60
    h_temp, c5_temp = _lp_loop(best_h, d3, radius_init=0.12)
    if c5_temp < best_c5:
        best_c5 = c5_temp
        best_h = h_temp.copy()

    # Phase 4: Multi-resolution search (20% budget)
    d4 = t0 + budget_s * 0.80
    n_candidates = [48, 54, 60, 66, 72, 78, 84, 90, 96, 102, 108, 120]
    tried_n = set([best_n])
    for n_try in n_candidates:
        if time.time() > d4:
            break
        if n_try in tried_n or n_try < 24:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl1 = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _slsqp_opt(h_r, 80, sub_dl1, maxiter=250)
        if time.time() < d4 - 2:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.08)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Perturbation + re-optimization (12% budget)
    d5 = t0 + budget_s * 0.92
    scale = 0.03
    for attempt in range(100):
        if time.time() > d5:
            break
        if best_c5 < 0.38075:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_try, c5_try = _slsqp_opt(h_pert, 100, sub_dl, maxiter=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.006, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.15)

    # Phase 6: Evolutionary (5% budget)
    if time.time() < t0 + budget_s * 0.96:
        d6 = t0 + budget_s * 0.96
        h_evo, c5_evo = _evolutionary(best_h, d6, pop_size=10, seed=seed + 13)
        if c5_evo < best_c5 - 1e-12:
            best_c5 = c5_evo
            best_h = h_evo.copy()

    # Phase 7: Final LP polish + pair refinement
    if time.time() < t0 + budget_s * 0.98:
        d7 = t0 + budget_s * 0.98
        h_f, c5_f = _lp_loop(best_h, min(d7, time.time() + budget_s * 0.015), radius_init=0.05)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if time.time() < t0 + budget_s * 0.995:
        d8 = t0 + budget_s * 0.995
        h_p, c5_p = _pair_refine(best_h, d8, max_passes=60)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final validation and return
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
