from scipy.optimize import minimize, linprog

# Global state for objective function (no closures allowed)
_G = {'z': None, 'n': 0, 'beta': 1.0}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -30, 30)))

def _solve_c(zc, n):
    target = n * 0.5
    # Initial bracket
    spread = max(1.0, float(np.max(np.abs(zc))))
    lo = -spread - 10.0
    hi = spread + 10.0
    # Ensure bracket
    for _ in range(30):
        if float(np.sum(_sigmoid(zc + lo))) < target:
            break
        lo -= 5.0
    for _ in range(30):
        if float(np.sum(_sigmoid(zc + hi))) > target:
            break
        hi += 5.0
    # Bisection
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(_sigmoid(zc + mid))) > target:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def _z_to_h(z, n):
    zc = z - np.mean(z)
    c = _solve_c(zc, n)
    return _sigmoid(zc + c)

def _h_to_z(h, n):
    eps = 1e-8
    hc = np.clip(h, eps, 1.0 - eps)
    z = np.log(hc / (1.0 - hc))
    return z - np.mean(z)

def _obj_grad(z):
    """Objective and gradient for L-BFGS. Uses global _G state."""
    n = _G['n']
    beta = _G['beta']
    h = _z_to_h(z, n)
    scores = _scores(h)
    M = float(np.max(scores))
    shifted = beta * (scores - M)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    sum_exps = float(np.sum(exps))
    if sum_exps < 1e-300:
        weights = np.zeros(len(scores))
        weights[int(np.argmax(scores))] = 1.0
    else:
        weights = exps / sum_exps
    LSE = M + np.log(sum_exps) / beta
    
    # Regularizer
    reg = 1e-4
    val = LSE + reg * float(np.sum(z * z))
    
    # Gradient
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:

        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:

                grad_h[:m] -= w * h[d:]

                grad_h[g:] -= w * h[:m]

    else:
        S = float(np.dot(a, grad_h)) / A
        grad_z = a * (grad_h - S)
    grad_z += 2.0 * reg * z
    return val, grad_z

def _multi_beta_lbfgs(z_init, n, betas, maxiter_per_stage, tol=1e-8):
    """Run L-BFGS with multiple beta stages."""
    z = z_init.copy()
    best_z = z.copy()
    best_c5 = _exact_c5(_z_to_h(z, n))
    
    for beta in betas:
        _G['n'] = n
        _G['beta'] = beta
        res = minimize(_obj_grad, z, method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage, 'ftol': tol, 'gtol': 1e-6})
        if res.x is not None:

            h = _z_to_h(z, n)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_z = z.copy()

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    n = len(v)
    lo = float(np.min(v)) - 5.0
    hi = float(np.max(v)) + 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    # Compensate on another coordinate
                    if d > 0:
                        # Need to decrease another coordinate by d
                        comp_mask = np.ones(n, dtype=bool)
                        comp_mask[j] = False
                        candidates = np.where(comp_mask & (h_try > d + 1e-10))[0]
                        if len(candidates) == 0:
                            continue
                        # Choose the one that gives best improvement (approximately)
                        comp = candidates[np.argmin(h_try[candidates])]
                    else:
                        # Need to increase another coordinate by |d|
                        comp_mask = np.ones(n, dtype=bool)
                        comp_mask[j] = False
                        candidates = np.where(comp_mask & (h_try < 1.0 - (-d) - 1e-10))[0]
                        if len(candidates) == 0:
                            continue
                        comp = candidates[np.argmax(h_try[candidates])]
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def _lp_polish(h, deadline, radius_init=0.05, max_iters=5000):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < 80:
        scores = _scores(h)
        m = float(np.max(scores))
        
        # Build LP: find direction p that decreases all scores
        # Variables: p[0..n-1], t (improvement amount)
        # Maximize t
        # s.t. G @ p <= -scores + m * 1 (normalize: G@p <= 0 means all scores decrease)
        #      Actually: new_score_i = scores_i + G[i,:] @ p <= scores_i - t
        #      => G[i,:] @ p + t <= -scores_i ... hmm
        # Let me reformulate:
        # We want: for all i, R_i(h + alpha*p) <= R_i(h) - alpha*t for some t > 0
        # Linearized: R_i(h) + G[i,:] @ (alpha*p) <= R_i(h) - alpha*t
        # => G[i,:] @ p <= -t
        # Maximize t, s.t. G[i,:] @ p <= -t for all i, sum p = 0, |p_j| <= radius
        
        # But this forces ALL scores to decrease, which might be too restrictive.
        # Better: just force the max score to decrease.
        # Actually, let's use: G[i,:] @ p <= 0 for all i (non-increasing)
        # and minimize the max: t = -min_i (G[i,:] @ p) ... 
        # Hmm, let me use the original formulation from the previous code.
        
        # Original: maximize t s.t. G[i,:] @ p - t <= -scores[i] for all i
        # This means: scores[i] + G[i,:] @ p - t <= 0, i.e., new score - t <= 0
        # Which means new score_i <= t... no that's not right either.
        
        # Let me just use: find p that minimizes max_i (scores_i + G[i,:] @ p)
        # subject to sum p = 0, |p_j| <= radius
        # This is: minimize t s.t. G[i,:] @ p + scores[i] - t <= 0 for all i, sum p = 0, |p_j| <= radius
        
        # Variables: x = [p_0, ..., p_{n-1}, t], length n+1
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize t
        
        A_ub = np.zeros((2 * n - 1, n + 1))
        for i in range(2 * n - 1):
            d = i - (n - 1)
            for j in range(n):
                if d >= 0:
                    m_ = n - d
                    if d <= j < n:
                        A_ub[i, j] += (1.0 - h[j - d])
                    if 0 <= j < m_:
                        A_ub[i, j] -= h[j + d]
                else:
                    g = -d
                    m_ = n - g
                    if 0 <= j < m_:
                        A_ub[i, j] += (1.0 - h[j + g])
                    if g <= j < n:
                        A_ub[i, j] -= h[j - g]
        A_ub[:, n] = -1.0  # -t
        b_ub = -scores
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo_j = max(-radius, -h[j])
            hi_j = min(radius, 1.0 - h[j])
            bounds.append((lo_j, hi_j))
        bounds.append((0, None))  # t >= 0
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] > 1e-10:
                p = res.x[:n]
                # Line search
                alpha_max = 1.0
                for j in range(n):
                    if p[j] > 1e-14:
                        alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                    elif p[j] < -1e-14:
                        alpha_max = min(alpha_max, h[j] / (-p[j]))
                alpha_max = max(alpha_max, 1e-8)
                
                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo_a = 0.0
                hi_a = alpha_max
                c1 = hi_a - gr * (hi_a - lo_a)
                c2 = lo_a + gr * (hi_a - lo_a)
                f1 = _exact_c5(_project(h + c1 * p, target))
                f2 = _exact_c5(_project(h + c2 * p, target))
                for _ in range(25):
                    if hi_a - lo_a < 1e-10:
                        break
                    if f1 < f2:
                        hi_a = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi_a - gr * (hi_a - lo_a)
                        f1 = _exact_c5(_project(h + c1 * p, target))
                    else:
                        lo_a = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo_a + gr * (hi_a - lo_a)
                        f2 = _exact_c5(_project(h + c2 * p, target))
                opt_a = 0.5 * (lo_a + hi_a)
                new_h = _project(h + opt_a * p, target)
                new_c5 = _exact_c5(new_h)
                if new_c5 < cur_c5 - 1e-12:
                    h = new_h
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.1, 0.3)
                else:
                    stall += 1
                    radius *= 0.8
                    if radius < 1e-4:
                        break
            else:
                stall += 1
                radius *= 0.85
                if radius < 1e-4:
                    break
        except Exception:
            stall += 10
            radius *= 0.7
            if radius < 1e-4:
                break
    
    return best_h, best_c5
