    n = len(h)
    # Sort scores, keep top_k
    idx = np.argsort(-scores)[:top_k]
    L = len(idx)
    
    G = np.zeros((L, n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        if d > 0:
            m = n - d
            G[ii, d:] += (1.0 - h[:m])
            G[ii, :m] += -h[d:]
        elif d < 0:
            e = -d
            m = n - e
            G[ii, :m] += (1.0 - h[e:])
            G[ii, e:] += -h[:m]
        else:
            G[ii, :] += (1.0 - 2.0 * h)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 1000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _correct_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    scaled = scores * dx
    max_s = float(np.max(scaled))
    exp_vals = np.exp(beta * (scaled - max_s))
    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scaled)] = 1.0
    else:
        w = exp_vals / w_sum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d] * dx
        if abs(wd) < 1e-15:
            continue
        if d > 0:
            m = n - d
            grad[d:] += wd * (1.0 - h[:m])
            grad[:m] += -wd * h[d:]
        elif d < 0:
            e = -d
            m = n - e
            grad[:m] += wd * (1.0 - h[e:])
            grad[e:] += -wd * h[:m]
        else:
            grad += wd * (1.0 - 2.0 * h)
    return grad

def _line_search(h, d, cur_c5, n_steps=20):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = cur_c5
    alphas = np.linspace(0.0, alpha_max, n_steps)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5
    lo = max(0.0, best_a - alphas[1])
    hi = min(alpha_max, best_a + alphas[1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(14):
        if hi - lo < 1e-10 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _build_G_row(h, d):
    n = len(h)
    row = np.zeros(n)
    if d > 0:
        m = n - d
        row[d:] += (1.0 - h[:m])
        row[:m] += -h[d:]
    elif d < 0:
        e = -d
        m = n - e
        row[:m] += (1.0 - h[e:])
        row[e:] += -h[:m]
    else:
        row[:] = (1.0 - 2.0 * h)
    return row

def _lp_step(h, scores, radius, top_k=None):
    n = len(h)
    if top_k is None:
        top_k = min(len(scores), 30)
    idx = np.argsort(-scores)[:top_k]
    L = len(idx)
    G = np.zeros((L, n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        G[ii] = _build_G_row(h, d)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx].astype(float)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 2000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, max_iters=5000, top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        p = _lp_step(h, scores, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:

            continue
        a, new_c5 = _line_search(h, p, cur_c5, n_steps=15)
        if new_c5 < cur_c5 - 1e-12:

            h = _project_box_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _binary_local_search(h, deadline, max_iters=60000, rng=None):

    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:

        best_swap = None
        best_val = cur_c5
        batch = min(25, len(ones) * len(zeros))

            h_bin[i] = 0.0

            zeros.append(i)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)
