            # Compute valid delta range
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-14:

            best_delta, new_Fmax = _pair_exact_delta(F, coeffs, h[i], h[j], d_lo, d_hi, n)

                F = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(F)) * (2.0 / n)

                    cur_c5 = new_c5
                    any_improve = True
        
        if any_improve:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5
