    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

            lo = mid
        else:

            break

        best_i = -1
        best_j = -1
        best_new_C = None

            # shift s_actual = k - (n-1) where k is the index in C
            A_i = np.zeros(2*n-1)

            # i-s_actual in [0, n-1] => s_actual in [i-(n-1), i]

                if 0 <= ns < n:

                ps = i + s
                if 0 <= ps < n:
                    A_i[k] += b[ps]

            base = C + A_i  # vector of length 2n-1

        ones_idx = np.where(b > 0.5)[0]
        zeros_idx = np.where(b < 0.5)[0]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _binary_correlations(b):
    return np.correlate(b.astype(float), 1.0 - b, mode='full')


def _binary_local_search(b, deadline, max_rounds=100000, stall_limit=500):
    n = len(b)
    b = b.astype(float).copy()
    offset = n - 1
    K = 2 * n - 1
    
    # Build index/validation matrices (computed once)
    j_range = np.arange(n)[:, None]
    k_range = np.arange(K)[None, :]
    s_mat = k_range - offset
    ns_mat = j_range - s_mat
    ps_mat = j_range + s_mat
    ns_valid = (ns_mat >= 0) & (ns_mat < n)
    ps_valid = (ps_mat >= 0) & (ps_mat < n)
    ns_safe = np.clip(ns_mat, 0, n - 1)
    ps_safe = np.clip(ps_mat, 0, n - 1)
    
    C = _binary_correlations(b)
    cur_max = float(C.max())
    
    # Precompute B_matrix column indices for fast updates
    # B[j, k] = ns_valid[j,k]*(1-b[ns_safe[j,k]]) - ps_valid[j,k]*b[ps_safe[j,k])
    
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
