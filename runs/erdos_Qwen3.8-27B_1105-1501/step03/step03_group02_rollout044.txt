    t0 = (h[i] - h[j]) / 2.0
    tl = max(-s / 2.0, s / 2.0 - 1.0)
    tu = min(s / 2.0, 1.0 - s / 2.0)
    if tu - tl < 1e-12:
        return h, cur_c5
    t0 = max(tl + 1e-10, min(tu - 1e-10, t0))
    
    def obj(t):
        h_trial = h.copy()
        h_trial[i] = s / 2.0 + t
        h_trial[j] = s / 2.0 - t
        return _exact_c5(h_trial)
    
    try:
        res = minimize_scalar(obj, bounds=(tl, tu), method='bounded',
                              options={'xatol': 1e-12, 'maxiter': 60})
        if time.time() > deadline:
            return h, cur_c5
        if res.fun < cur_c5 - 1e-14:
            t_opt = float(res.x)
            h_new = h.copy()
            h_new[i] = s / 2.0 + t_opt
            h_new[j] = s / 2.0 - t_opt
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5:
                return h_new, c5_new
    except Exception:
        pass
    return h, cur_c5

def _pairwise_pass(h, deadline, max_pairs=400, top_frac=1.0):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    grad, _ = _smooth_max_grad(h, 50.0)
    
    # Score pairs by |grad[i] - grad[j]|
    pair_scores = []
    for i in range(n):
        for j in range(i + 1, n):
            ds = abs(grad[i] - grad[j])
            pair_scores.append((ds, i, j))
    pair_scores.sort(key=lambda x: -x[0])
    
    n_pairs = int(len(pair_scores) * top_frac)
    n_pairs = min(n_pairs, max_pairs)
    
    improved = 0
    for idx in range(n_pairs):
        if time.time() > deadline:
            break
        _, i, j = pair_scores[idx]
        if abs(h[i] - h[j]) < 1e-8 and abs(grad[i] - grad[j]) < 1e-6:
            continue
        h, cur_c5 = _pair_optimize(h, i, j, cur_c5, deadline)
        if h is not None and cur_c5 > 0:
            improved += 1
    return h, cur_c5, improved

def _gradient_phase(h, deadline, beta_start=20, beta_end=200, max_iters=2000):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.02
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = float(it) / max(float(max_iters), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        grad, smooth_val = _smooth_max_grad(h, beta)
        # Remove mean component (for sum constraint)
        grad = grad - np.mean(grad)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-10:
            break
        grad = grad / gnorm
        
        # Backtracking line search
        alpha = step
        found = False
        for _ in range(25):
            h_try = h - alpha * grad
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                found = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.7
        if not found:
            step *= 0.85
            if step < 1e-8:
                break
        else:
            step = min(step * 1.05, 0.05)
    return best_h, best_c5

def _slsqp_active(h_init, deadline, top_k=30, maxiter=100):
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    z0 = float(np.max(scores)) + 1e-5
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]
    
    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    }]
    for d in active_lags:
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
            'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        })
    
    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return None
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    def update_best(h_new, c5_new, n_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_new
    
    # Phase 1: SLSQP with broad active set
    dl1 = t0 + budget_s * 0.12
    res = _slsqp_active(h, dl1, top_k=40, maxiter=100)
    if res is not None:
        update_best(res[0], res[1], best_n)
    h = best_h.copy()
    
    # Phase 2: Smooth max gradient descent
    dl2 = t0 + budget_s * 0.30
    h, c5 = _gradient_phase(h, dl2, beta_start=15, beta_end=150, max_iters=3000)
    update_best(h, c5, best_n)
    h = best_h.copy()
    
    # Phase 3: Pairwise optimization (key new component)
    dl3 = t0 + budget_s * 0.55
    pass_count = 0
    while time.time() < dl3:
        sub_dl = min(dl3, time.time() + 30.0)
        h, c5, improved = _pairwise_pass(h, sub_dl, max_pairs=300, top_frac=0.8)
        if c5 < best_c5 - 1e-14:
            update_best(h, c5, best_n)
            h = best_h.copy()
            pass_count += 1
        else:
            if improved == 0:
                # Try with more pairs and lower top_frac
                h, c5, improved = _pairwise_pass(h, min(dl3, time.time() + 20.0),
                                                  max_pairs=500, top_frac=1.0)
                if c5 < best_c5 - 1e-14:
                    update_best(h, c5, best_n)
                    h = best_h.copy()
                else:
                    break
        pass_count += 1
    
    # Phase 4: SLSQP refinement on best
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.70
        res = _slsqp_active(best_h, dl4, top_k=25, maxiter=150)
        if res is not None:
            update_best(res[0], res[1], best_n)
        # Second SLSQP with different active set
        if time.time() < t0 + budget_s * 0.72:
            h2 = best_h + rng.normal(0, 0.02, best_n)
            h2 = _project(h2, best_n / 2.0)
            res = _slsqp_active(h2, t0 + budget_s * 0.72, top_k=35, maxiter=80)
            if res is not None:
                update_best(res[0], res[1], best_n)
    
    # Phase 5: More pairwise passes
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.85
        while time.time() < dl5:
            sub_dl = min(dl5, time.time() + 20.0)
            h, c5, improved = _pairwise_pass(best_h, sub_dl, max_pairs=200, top_frac=0.5)
            if c5 < best_c5 - 1e-14:
                update_best(h, c5, best_n)
            else:
                # Try gradient polish
                h_g, c5_g = _gradient_phase(best_h, min(dl5, time.time() + 15.0),
                                            beta_start=100, beta_end=300, max_iters=500)
                if c5_g < best_c5 - 1e-14:
                    update_best(h_g, c5_g, best_n)
                else:
                    break
    
    # Phase 6: Multi-resolution sweep
    if time.time() < t0 + budget_s * 0.93:
        dl6 = t0 + budget_s * 0.93
        for dn in [-4, 4, -8, 8, -16, 16, -2, 2]:
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 140 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            c5_r = _exact_c5(h_r)
            if c5_r > best_c5 + 0.002:
                continue
            # Quick gradient polish at new resolution
            h_r, c5_r = _gradient_phase(h_r, min(dl6, time.time() + 15.0),
                                        beta_start=30, beta_end=100, max_iters=500)
            if c5_r < best_c5 - 1e-14:
                update_best(h_r, c5_r, n_try)
    
    # Phase 7: Final pairwise + gradient polish
    if time.time() < t0 + budget_s * 0.98:
        dl7 = t0 + budget_s * 0.98
        # Gradient with very high beta (near-exact max)
        h, c5 = _gradient_phase(best_h, min(dl7, time.time() + 20.0),
                                beta_start=200, beta_end=500, max_iters=2000)
        update_best(h, c5, best_n)
        # Final pairwise
        if time.time() < dl7:
            h, c5, _ = _pairwise_pass(best_h, dl7, max_pairs=300, top_frac=1.0)
            update_best(h, c5, best_n)
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
