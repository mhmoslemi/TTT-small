        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / w_sum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:

                grad[d:] += -wd * h[:m]

            g = -d

import numpy as np

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):

    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _smooth_grad(h, beta):

    ev = np.exp(beta * (sc - mx))
    ws = float(np.sum(ev))

        w[np.argmax(sc)] = 1.0

def _cvx_trust_solve(h, rho):
    n = len(h)
    dx = 2.0 / n
    F = _scores(h) * dx
    G = _build_G(h)
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [
        F + G @ delta <= t,
        t >= 0,
        cp.sum(delta) == 0,
        cp.norm(delta) <= rho,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None:
                return np.array(delta.value).flatten()
        except Exception:
            continue
    return None

def _cvx_lp_solve(h, radius):

    dx = 2.0 / n
    F = _scores(h) * dx
    G = _build_G(h)
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [
        F + G @ delta <= t,
        t >= 0,
        cp.sum(delta) == 0,
        delta >= -radius,
        delta <= radius,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None:
                return np.array(delta.value).flatten()
        except Exception:
            continue
    return None

                break
            d = -g / nm * ss
            a, nc = _line_search(h, d, cur, n_coarse=10, n_fine=18)
            if nc < cur - 1e-13:
                h = _project_box_sum(h + a * d, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5, best_h = cur, h.copy()
                stall = 0
                ss = min(ss * 1.04, 0.4)
            else:
                stall += 1
                ss *= 0.72
    return best_h, best_c5

def _nesterov_smooth(h, deadline, beta=50.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    y = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    momentum = 0.9
    lr = 0.05
    for it in range(3000):
        if time.time() > deadline:
            break
        g = _smooth_grad(y, beta)
        g -= np.mean(g)
        nm = np.max(np.abs(g))
        if nm < 1e-14:
            break
        d = -g / nm
        new_h = _project_box_sum(y + lr * d, target)
        t_new = (1.0 + momentum) / (1.0 + momentum + 2e-4)
        y = _project_box_sum(new_h + (momentum / (1.0 + momentum + 2e-4) - 1.0) * (new_h - h), target)
        y = _project_box_sum(y, target)
        c5_new = _exact_c5(y)
        if c5_new < cur - 1e-13:
            h = y.copy()
            cur = c5_new
            if cur < best_c5:
                best_c5, best_h = cur, h.copy()
        else:
            lr *= 0.8
    return best_h, best_c5

def _binary_ls(h, deadline, rng, max_iters=40000):
    n = len(h)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    _ = _project_box_sum(hb, n / 2.0)
    cur = _exact_c5(hb)
    best_h, best_c5 = hb.copy(), cur
    ones = list(np.where(hb == 1.0)[0])
    zeros = list(np.where(hb == 0.0)[0])
    ni = 0
    for _ in range(max_iters):
        if time.time() > deadline or ni > 1500 or not ones or not zeros:
            break
        bs, bv = None, cur
        batch = min(15, len(ones) * len(zeros))
        for _b in range(batch):
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            hb[i], hb[j] = 0.0, 1.0
            cn = _exact_c5(hb)
            hb[i], hb[j] = 1.0, 0.0
            if cn < bv - 1e-14:
                bv, bs = cn, (i, j)
        if bs:
            i, j = bs
            hb[i], hb[j] = 0.0, 1.0
            ones.remove(i); zeros.remove(j)
            zeros.append(i); ones.append(j)
            cur = bv
            if cur < best_c5:
                best_c5, best_h = cur, hb.copy()
            ni = 0
        else:
            ni += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old, dx_new = 2.0 / n, 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        xs, xe = j * dx_new, (j + 1) * dx_new
        mass = 0.0
        for i in range(n):
            os_, oe = i * dx_old, (i + 1) * dx_old
            ov = max(0.0, min(xe, oe) - max(xs, os_))
            if ov > 0:
                mass += ov * h[i]
        h_new[j] = mass / dx_new
    return _project_box_sum(h_new, n_new / 2.0)

def _random_moves(h, deadline, rng, n_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    ni = 0
    for _ in range(n_iters):
        if time.time() > deadline or ni > 800:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        while j == i:
            j = int(rng.integers(n))
        improved = False
        for frac in [0.4, 0.2, 0.1, 0.05, 0.02]:
            md = min(h[i], 1.0 - h[j]) * frac
            if md < 1e-12:
                md = -min(1.0 - h[i], h[j]) * frac
            if abs(md) < 1e-12:
                continue
            ht = h.copy()
            ht[i] += md
            ht[j] -= md
            ht = _project_box_sum(ht, target)
            c5t = _exact_c5(ht)
            if c5t < cur - 1e-12:
                h = ht
                cur = c5t
                if cur < best_c5:
                    best_c5, best_h = cur, h.copy()
                ni = 0
                improved = True
                break
        if not improved:
            ni += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h0 = None
    if h0 is not None and len(h0) > 0 and np.all(np.isfinite(h0)):
        n0 = len(h0)
        h = _project_box_sum(h0, n0 / 2.0)
    else:
        n0 = 64
        h = np.zeros(n0)
        h[::2] = 1.0
        h = _project_box_sum(h, n0 / 2.0)
    best_h, best_c5 = h.copy(), _exact_c5(h)
    best_n = len(h)
    try:
        # Phase 1: CVXPY trust-region
        dl = t0 + budget_s * 0.18
        h1, c1 = _cvx_optimize(best_h, dl, rho_init=0.4)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        # Phase 2: Gradient continuation
        if time.time() < t0 + budget_s * 0.32:
            dl = t0 + budget_s * 0.32
            h2, c2 = _grad_continuation(best_h, dl, b0=3.0, b1=300.0, nsteps=10)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
        # Phase 3: Nesterov
        if time.time() < t0 + budget_s * 0.40:
            dl = t0 + budget_s * 0.40
            h3, c3 = _nesterov_smooth(best_h, dl, beta=60.0)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
        # Phase 4: CVXPY again
        if time.time() < t0 + budget_s * 0.50:
            dl = t0 + budget_s * 0.50
            h4, c4 = _cvx_optimize(best_h, dl, rho_init=0.3)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
        # Phase 5: Random moves for diversification
        if time.time() < t0 + budget_s * 0.58:
            dl = t0 + budget_s * 0.58
            h5, c5 = _random_moves(best_h, dl, rng, n_iters=15000)
            if c5 < best_c5:
                best_c5, best_h = c5, h5.copy()
        # Phase 6: Binary search + refinement
        if time.time() < t0 + budget_s * 0.70:
            dl_b = t0 + budget_s * 0.64
            hb, cb = _binary_ls(best_h, dl_b, rng, max_iters=40000)
            h6 = hb if cb < best_c5 else best_h.copy()
            c6 = min(cb, best_c5)
            if time.time() < t0 + budget_s * 0.70:
                dl_r = t0 + budget_s * 0.70
                h6, c6r = _grad_continuation(h6, dl_r, b0=10.0, b1=500.0, nsteps=8)
                c6 = min(c6, c6r)
                if c6r < c6:
                    h6 = h6.copy()
            if c6 < best_c5:
                best_c5, best_h = c6, h6.copy()
        # Phase 7: CVXPY polish
        if time.time() < t0 + budget_s * 0.78:
            dl = t0 + budget_s * 0.78
            h7, c7 = _cvx_optimize(best_h, dl, rho_init=0.25)
            if c7 < best_c5:
                best_c5, best_h = c7, h7.copy()
        # Phase 8: Multi-resolution
        if time.time() < t0 + budget_s * 0.90:
            dl = t0 + budget_s * 0.90
            nc = len(best_h)
            tries = sorted(set(x for x in [nc-8, nc-4, nc+4, nc+8, nc+16, 48, 56, 72, 80, 96, 40, 36]
                               if 24 <= x <= 160 and x % 2 == 0))
            rem = max(10.0, dl - time.time())
            per = rem / max(len(tries), 1)
            for nt in tries:
                if time.time() > dl - 15:
                    break
                hr = _resample(best_h, nt)
                dlr = min(dl - 15, time.time() + per)
                hr, cr = _grad_continuation(hr, dlr, b0=5.0, b1=200.0, nsteps=8)
                if time.time() < dlr:
                    hr2, cr2 = _cvx_optimize(hr, dlr, rho_init=0.2)
                    if cr2 < cr:
                        hr, cr = hr2, cr2
                if cr < best_c5 - 5e-5:
                    best_c5, best_h, best_n = cr, hr.copy(), nt
        # Phase 9: Final polish
        if time.time() < t0 + budget_s * 0.96:
            dl = t0 + budget_s * 0.96
            h9, c9 = _cvx_optimize(best_h, dl, rho_init=0.15)
            if c9 < best_c5:
                best_c5, best_h = c9, h9.copy()
        # Phase 10: Final gradient
        if time.time() < t0 + budget_s * 0.99:
            dl = t0 + budget_s * 0.995
            h10, c10 = _grad_continuation(best_h, dl, b0=80.0, b1=600.0, nsteps=6)
            if c10 < best_c5:
                best_c5, best_h = c10, h10.copy()
    except Exception:
        pass
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
