    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 25
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(40):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.5)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _active_set_optimize(h, deadline, k_list=None, max_iters=3000):
    """Active-set equalization: minimize ||G_A p - t*1||^2 + lam*||p||^2 s.t. sum(p)=0, |p_j|<=radius"""
    if k_list is None:
        k_list = [4, 6, 8, 10, 15]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = 0.1
    ti = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        k = k_list[ti % len(k_list)]
        idx = np.argsort(scores)[-k:]
        G_A = G[idx]
        s_A = scores[idx]
        t_target = s_A.min()
        b_vec = G_A @ h - t_target
        lam = 0.01
        A_mat = G_A.T @ G_A + lam * np.eye(n)
        A_mat += (1.0 / n) * np.ones((n, n))
        rhs = -G_A.T @ b_vec
        pf = np.linalg.solve(A_mat, rhs)
        pf -= np.mean(pf)
        p = np.clip(pf, -radius, radius)
        p = np.minimum(p, 1.0 - h)
        p = np.maximum(p, -h)
        p -= np.mean(p)
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.75
            ti += 1
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 15:
                radius = 0.05
    return best_h, best_c5

def _sim_anneal(h, deadline, rng, n_max_iters=50000):
    """Simulated annealing with pair transfers and structured moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    T = 0.008
    T_min = 0.0003
    cooling = (T_min / T) ** (1.0 / max(1, n_max_iters))
    for it in range(n_max_iters):
        if time.time() > deadline:
            break
        if T < T_min + 1e-7:
            break
        T *= cooling
        h_try = h.copy()
        move_type = rng.integers(0, 5)
        if move_type == 0:
            # Pair transfer: move mass from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amt = rng.uniform(0.02, 0.15) * (1.0 + T / T_min * 0.3)
            amt = min(amt, h[i], 1.0 - h[j])
            if amt < 1e-6:
                continue
            h_try[i] -= amt
            h_try[j] += amt
        elif move_type == 1:
            # Small shift (circular)
            shift = int(rng.integers(1, 3))
            h_try = np.roll(h, shift)
        elif move_type == 2:
            # Local smoothing: blend a small block with neighbors
            i = rng.integers(2, n - 2)
            block = h_try[i-1:i+2].copy()
            h_try[i-1] = 0.5 * h_try[i-1] + 0.5 * 0.5 * (h_try[i-2] + h_try[i])
            h_try[i] = 0.5 * h_try[i] + 0.5 * 0.5 * (h_try[i-1] + h_try[i+1])
            h_try[i+1] = 0.5 * h_try[i+1] + 0.5 * 0.5 * (h_try[i] + h_try[i+2])
        elif move_type == 3:
            # Gaussian perturbation (small)
            sigma = 0.03 * (1.0 + 2.0 * T / T_min)
            mask = rng.random(n) < 0.3
            h_try[mask] += rng.normal(0, sigma, mask.sum())
        else:
            # Block reversal
            blen = int(rng.integers(3, min(8, n - 2)))
            i = rng.integers(0, n - blen)
            h_try[i:i+blen] = h_try[i:i+blen][::-1]
        h_try = _project(h_try, target)
        new_c5 = _exact_c5(h_try)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _smooth_gradient(h, deadline, n_steps=200):
    """Softmax-weighted gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    h_prev = h.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 8.0 * (50.0 / 8.0) ** frac
        scores = _scores(h)
        mx = np.max(scores)
        w = np.exp(beta * (scores - mx))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        # Nesterov momentum
        v = 0.8 * (h - h_prev) - grad * 0.02
        v = np.clip(v, -0.3, 0.3)
        v -= np.mean(v)
        alpha = 0.1
        for _ in range(35):
            h_new = _project(h + alpha * v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h_prev = h.copy()
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.65
            if alpha < 1e-10:
                break
        else:
            h_prev = h.copy()
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=15):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0.0, 1.0)
                        h1 = _project(h1, n / 2.0)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0.0, 1.0)
                        h2 = _project(h2, n / 2.0)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = _project(h_try, n / 2.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_starts(h, n, rng):
    """Generate diverse starting points from h using structured transformations."""
    starts = []
    # Mirror
    starts.append(h[::-1].copy())
    # Complement (preserves sum = n/2)
    starts.append(1.0 - h.copy())
    # Shift by 1
    starts.append(np.roll(h, 1).copy())
    # Shift by 2
    starts.append(np.roll(h, 2).copy())
    # Smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    h_sm = np.convolve(h, kernel, mode='same')
    starts.append(_project(h_sm, n / 2.0))
    # Random structured perturbation
    for _ in range(3):
        pert = h.copy()
        mask = rng.random(n) < 0.25
        pert[mask] += rng.normal(0, 0.08, mask.sum())
        starts.append(_project(pert, n / 2.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization from initial (25% budget)
    dl = t0 + budget_s * 0.25
    h, c5 = _lp_optimize(h, dl, top_k_list=[None, 15, 8, 5], radius_init=0.25, max_iters=4000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Smooth gradient with momentum (10%)
    if time.time() < t0 + budget_s * 0.32:
        dl2 = t0 + budget_s * 0.35
        h, c5 = _smooth_gradient(best_h, dl2, n_steps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Active-set equalization (15%)
    if time.time() < t0 + budget_s * 0.42:
        dl3 = t0 + budget_s * 0.48
        h, c5 = _active_set_optimize(best_h, dl3, k_list=[4, 6, 8, 12, 15], max_iters=3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Simulated annealing (20%)
    if time.time() < t0 + budget_s * 0.55:
        dl4 = t0 + budget_s * 0.68
        h, c5 = _sim_anneal(best_h, dl4, rng, n_max_iters=40000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Quick LP after annealing
        if c5 < best_c5 + 0.001 and time.time() < dl4:
            h2, c5_2 = _lp_optimize(h, dl4, top_k_list=[8, 5, None], radius_init=0.1, max_iters=1000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 5: Structured restarts with LP (15%)
    if time.time() < t0 + budget_s * 0.70:
        dl5 = t0 + budget_s * 0.80
        starts = _structured_starts(best_h, best_n, rng)
        for hs in starts:
            if time.time() > dl5:
                break
            if best_c5 < 0.3806:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_try, c5_try = _lp_optimize(hs, sub_dl, top_k_list=[10, 5, None], radius_init=0.15, max_iters=800)
            if c5_try < best_c5 - 1e-11:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Multi-resolution sweep (10%)
    if time.time() < t0 + budget_s * 0.82:
        dl6 = t0 + budget_s * 0.90
        for dn in [-8, -6, -4, -3, -2, -1, 0, 1, 2, 3, 4, 6, 8, 12]:
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 96 or n_try == best_n:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[10, 5, None], radius_init=0.12, max_iters=600)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final LP polish with very small radius (5%)
    if time.time() < t0 + budget_s * 0.93:
        dl7 = t0 + budget_s * 0.97
        h_f, c5_f = _lp_optimize(best_h, dl7, top_k_list=[3, 4, 5, 6], radius_init=0.02, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.96:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.99, max_passes=12)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
            best_n = len(best_h)

    final_proj = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(final_proj)
    if final_c5 > best_c5:
        final_proj = best_h
        final_c5 = best_c5
    return final_proj, final_c5, len(final_proj)
