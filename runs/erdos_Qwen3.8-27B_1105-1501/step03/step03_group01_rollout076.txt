def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    A_ub[:, n] = -1.0

    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    if n_new % 2 == 0:
        h_new = 0.5 * (h_new + 1.0 - h_new[::-1])
    return _project(h_new, n_new / 2.0)

def _coord_descent(h, deadline, n_passes_max=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(n_passes_max):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
                continue
            best_j = -1
            best_delta = 0.0
            best_val = best_c5
            for j in range(i + 1, n):
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                    continue
                for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if di > 0:
                        am = (1.0 - h[i]) / di
                    else:
                        am = h[i] / (-di)
                    if dj > 0:
                        am = min(am, (1.0 - h[j]) / dj)
                    else:
                        am = min(am, h[j] / (-dj))
                    if am < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        h_t = h.copy()
                        h_t[i] = np.clip(h[i] + di * am * frac, 0.0, 1.0)
                        h_t[j] = np.clip(h[j] + dj * am * frac, 0.0, 1.0)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_val - 1e-13:
                            best_val = c5_t
                            best_i = i
                            best_j = j
                            best_di = di * am * frac
                            best_dj = dj * am * frac
            if best_j >= 0:
                h[best_i] = np.clip(h[best_i] + best_di, 0.0, 1.0)
                h[best_j] = np.clip(h[best_j] + best_dj, 0.0, 1.0)
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(corr))
    exp_s = np.exp(tau * (corr - max_s))
    s_sum = float(np.sum(exp_s))
    if s_sum < 1e-300:
        idx = int(np.argmax(corr))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        w = exp_s / s_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                idx1 = np.arange(m)
                idx2 = idx1 + d
                grad[idx1] += wk * (1.0 - h[idx2]) * dx
                grad[idx2] += wk * (-h[idx1]) * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                idx1 = np.arange(m)
                idx2 = idx1 + g
                grad[idx2] += wk * (1.0 - h[idx1]) * dx
                grad[idx1] += wk * (-h[idx2]) * dx
    return grad

def _grad_search(h, deadline, tau=100.0, lr=0.02, momentum_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad = _soft_grad(h, tau)
        mom = momentum_beta * mom + (1.0 - momentum_beta) * grad
        h_new = _project(h - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.08)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
            if stall % 15 == 0:
                h = best_h + np.random.default_rng(stall * 1000 + n).normal(0, 0.03, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _evolutionary(h_init, deadline, pop_size=12, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(h) for h in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.04
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=2)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2)
        child += rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.93)
            else:
                mut_scale = min(0.15, mut_scale * 1.06)
        else:
            mut_scale = min(0.15, mut_scale * 1.04)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick anti-symmetric polish (8%)
    d1 = t0 + budget_s * 0.08
    if best_n % 2 == 0:
        g0 = _h_to_g(best_h, best_n)
        h_anti, c5_anti = _lp_loop_anti(g0, best_n, d1, radius_init=0.15)
        if c5_anti < best_c5:
            best_c5 = c5_anti
            best_h = h_anti.copy()

    # Phase 2: Full LP polish on current best (12%)
    d2 = t0 + budget_s * 0.20
    h_lp, c5_lp = _lp_loop_full(best_h, d2, radius_init=0.1)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Multi-resolution search (35%)
    d3 = t0 + budget_s * 0.55
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12,
               -14, 14, -16, 16, -20, 20, -24, 24, -30, 30, -40, 40]
    tried_n = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 200:
            continue
        if n_try < 20:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        if n_try % 2 == 0 and time.time() < sub_dl:
            g_r = _h_to_g(h_r, n_try)
            h_r2, c5_r2 = _lp_loop_anti(g_r, n_try, sub_dl, radius_init=0.12)
            if c5_r2 < best_c5 - 1e-12:
                c5_r = c5_r2
                h_r = h_r2
            else:
                c5_r = _exact_c5(h_r)
        else:
            h_r, c5_r = _lp_loop_full(h_r, sub_dl, radius_init=0.1)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            # Continue trying nearby values from new best
            extra_offsets = [-2, 2, -4, 4, -6, 6]
            for edn in extra_offsets:
                if time.time() > d3:
                    break
                en = best_n + edn
                if en < 30 or en > 200 or en in tried_n:
                    continue
                tried_n.add(en)
                h_er = _resample(best_h, en)
                sub_dl2 = min(d3, time.time() + budget_s * 0.018)
                if en % 2 == 0 and time.time() < sub_dl2:
                    g_er = _h_to_g(h_er, en)
                    h_er, c5_er = _lp_loop_anti(g_er, en, sub_dl2, radius_init=0.1)
                else:
                    h_er, c5_er = _lp_loop_full(h_er, sub_dl2, radius_init=0.1)
                if c5_er < best_c5 - 1e-12:
                    best_c5 = c5_er
                    best_h = h_er.copy()
                    best_n = en

    # Phase 4: Perturbation + re-optimization (20%)
    d4 = t0 + budget_s * 0.75
    scale = 0.04
    for attempt in range(120):
        if time.time() > d4:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        if best_n % 2 == 0 and time.time() < sub_dl:
            g_pert = _h_to_g(h_pert, best_n)
            h_try, c5_try = _lp_loop_anti(g_pert, best_n, sub_dl, radius_init=0.08)
        else:
            h_try, c5_try = _lp_loop_full(h_pert, sub_dl, radius_init=0.08)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.88)
        else:
            scale = min(0.25, scale * 1.15)

    # Phase 5: Evolutionary search (12%)
    d5 = t0 + budget_s * 0.87
    if time.time() < d5:
        h_evo, c5_evo = _evolutionary(best_h, d5, pop_size=14, seed=seed + 13)
        if c5_evo < best_c5 - 1e-12:
            best_c5 = c5_evo
            best_h = h_evo.copy()

    # Phase 6: Gradient search for fine-tuning (5%)
    d6 = t0 + budget_s * 0.92
    if time.time() < d6:
        h_gr, c5_gr = _grad_search(best_h, d6, tau=150.0, lr=0.015, momentum_beta=0.88)
        if c5_gr < best_c5 - 1e-12:
            best_c5 = c5_gr
            best_h = h_gr.copy()

    # Phase 7: Final LP polish + coordinate descent (8%)
    d7 = t0 + budget_s * 0.96
    if time.time() < d7:
        h_f, c5_f = _lp_loop_full(best_h, min(d7, time.time() + budget_s * 0.025), radius_init=0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < t0 + budget_s * 0.99:
        d8 = t0 + budget_s * 0.99
        h_cd, c5_cd = _coord_descent(best_h, d8, n_passes_max=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, best_n
