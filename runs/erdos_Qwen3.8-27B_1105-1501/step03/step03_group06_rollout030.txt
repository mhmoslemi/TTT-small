def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    err = target - np.sum(x)
    if abs(err) > 1e-12:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(interior):
            adj = err / np.sum(interior)
            x[interior] = np.clip(x[interior] + adj, 0.0, 1.0)

            w /= total
        val = float(np.sum(w * s))
        g = np.zeros(n)
        for k in range(len(s)):
            if w[k] > 1e-10:
                g += w[k] * _grad_at_lag(x, k)
        g -= np.mean(g)
        return val, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x


def _sin_dir(n, freq, phase):
    x = np.arange(n) * 2.0 / n
    d = np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    d -= np.mean(d)
    return d


def _cos_dir(n, freq, phase):
    x = np.arange(n) * 2.0 / n
    d = np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
    d -= np.mean(d)
    return d


def _block_dir(n, rng, block_size):
    d = np.zeros(n)
    start = int(rng.integers(0, max(1, n - block_size)))
    d[start:start + block_size] = rng.normal(0, 1, block_size)
    d -= np.mean(d)
    return d


def _binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b


def _intensive_local(h, deadline, max_rounds=200, seed=777):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    scale = 1.0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        improved = False

        # Smooth gradient directions
        for alpha in [10.0, 30.0, 80.0]:
            if time.time() > deadline:
                break
            d = -_smooth_grad(h, alpha) * scale
            nh, nc5 = _try_direction(h, d, target)
            if nc5 < cur_c5 - 1e-13:
                h, cur_c5 = nh, nc5
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        # Active set gradient
        if time.time() < deadline:
            d = -_active_grad(h, 0.05) * scale
            nh, nc5 = _try_direction(h, d, target)
            if nc5 < cur_c5 - 1e-13:
                h, cur_c5 = nh, nc5
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        # Sinusoidal directions
        for freq in [1, 2, 3, 5]:
            if time.time() > deadline:
                break
            for _ in range(2):
                phase = float(rng.uniform(0, 2 * np.pi))
                d = _sin_dir(n, freq, phase)
                nh, nc5 = _try_direction(h, d, target)
                if nc5 < cur_c5 - 1e-13:
                    h, cur_c5 = nh, nc5
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()

        # Cosine directions
        for freq in [1, 2, 4]:
            if time.time() > deadline:
                break
            phase = float(rng.uniform(0, 2 * np.pi))
            d = _cos_dir(n, freq, phase)
            nh, nc5 = _try_direction(h, d, target)
            if nc5 < cur_c5 - 1e-13:
                h, cur_c5 = nh, nc5
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        # Block directions
        for bs in [4, 8, n // 4]:
            if time.time() > deadline:
                break
            for _ in range(2):
                d = _block_dir(n, rng, bs)
                nh, nc5 = _try_direction(h, d, target)
                if nc5 < cur_c5 - 1e-13:
                    h, cur_c5 = nh, nc5
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()

        # Pair directions
        for _ in range(4):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sig = 1.0 if h[j] > h[i] else -1.0
            d = np.zeros(n)
            d[i] = sig
            d[j] = -sig
            nh, nc5 = _try_direction(h, d, target, scale=1.0)
            if nc5 < cur_c5 - 1e-13:
                h, cur_c5 = nh, nc5
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        # Quadratic active set optimization
        if time.time() < deadline and rnd % 3 == 0:
            try:
                x_q = _active_set_opt(h, epsilon_frac=0.08, maxiter=100)
                c5_q = _exact_c5(x_q)
                if c5_q < cur_c5 - 1e-13:
                    h = x_q
                    cur_c5 = c5_q
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()
            except Exception:
                pass

        if not improved:
            stall += 1
            scale *= 0.85
            if stall % 8 == 0:
                h = _project(h + rng.normal(0, 0.03, n), target)
                cur_c5 = _exact_c5(h)
        else:
            scale = min(scale * 1.05, 1.5)

    return best_h, best_c5


def _pair_sweep(h, deadline, max_passes=10, seed=999):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)

    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if n <= 64:
            indices = np.arange(n)
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        else:
            pairs = [(int(rng.integers(0, n)), int(rng.integers(0, n)))
                     for _ in range(400)]

        for i, j in pairs:
            if time.time() > deadline:
                break
            if i == j:
                continue
            for sig in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = sig
                d[j] = -sig
                nh, nc5 = _try_direction(h, d, target, scale=1.0)
                if nc5 < cur_c5 - 1e-13:
                    h, cur_c5 = nh, nc5
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _evolve(pop, deadline, seed=555):
    rng = np.random.default_rng(seed)
    n = pop[0][0].shape[0]
    scores = [(h, _exact_c5(h)) for h, _ in pop]
    scores.sort(key=lambda x: x[1])
    best_h, best_c5 = scores[0]
    pop_list = scores[:5]
    evals = 0

    while time.time() < deadline:
        if evals > 3000:
            break
        k = min(3, len(pop_list))
        p1 = pop_list[int(rng.integers(0, k))][0]
        p2 = pop_list[int(rng.integers(0, k))][0]
        alpha = float(rng.uniform(0.3, 0.7))
        child = alpha * p1 + (1.0 - alpha) * p2
        ms = float(rng.uniform(0.02, 0.15))
        child += ms * rng.normal(0, 1, n)
        child = _project(child, n / 2.0)
        c5_child = _exact_c5(child)
        evals += 1

        if c5_child < best_c5 + 0.005:
            for _ in range(3):
                if time.time() > deadline:
                    break
                d = -_smooth_grad(child, 40.0)
                nh, nc5 = _try_direction(child, d, n / 2.0)
                if nc5 < c5_child - 1e-13:
                    child, c5_child = nh, nc5
                else:
                    break
            if c5_child < best_c5:
                best_h, best_c5 = child.copy(), c5_child

        pop_list.append((child, c5_child))
        pop_list.sort(key=lambda x: x[1])
        pop_list = pop_list[:5]

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        n0 = 64
        x = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        h0 = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
        h0 = np.clip(h0, 0, 1)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Intensive local optimization (45% budget)
    d1 = t0 + budget_s * 0.45
    h_opt, c5_opt = _intensive_local(h, d1, max_rounds=250, seed=12345)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 1b: Active set refinement
    if time.time() < d1:
        try:
            x_q = _active_set_opt(best_h, epsilon_frac=0.05, maxiter=200)
            c5_q = _exact_c5(x_q)
            if c5_q < best_c5:
                best_c5, best_h = c5_q, x_q
        except Exception:
            pass

    # Phase 2: Pair sweep (12% budget)
    d2 = t0 + budget_s * 0.57
    if time.time() < d2:
        hp, c5p = _pair_sweep(best_h, d2, max_passes=12, seed=7777)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 3: Evolutionary + transforms (20% budget)
    d3 = t0 + budget_s * 0.77
    if time.time() < d3:
        pop = []
        pop.append((best_h.copy(), best_c5))
        # Mirror
        h_m = best_h[::-1].copy()
        h_m = _project(h_m, best_n / 2.0)
        pop.append((h_m, _exact_c5(h_m)))
        # Complement
        h_c = 1.0 - best_h
        h_c = _project(h_c, best_n / 2.0)
        pop.append((h_c, _exact_c5(h_c)))
        # Binarized
        h_b = _binarize(best_h)
        pop.append((h_b, _exact_c5(h_b)))
        # Smooth random
        for _ in range(2):
            h_r = best_h + rng.normal(0, 0.08, best_n)
            h_r = _project(h_r, best_n / 2.0)
            pop.append((h_r, _exact_c5(h_r)))
        # Shifted
        for shift in [1, 2, best_n // 4]:
            h_s = np.roll(best_h, shift)
            h_s = _project(h_s, best_n / 2.0)
            pop.append((h_s, _exact_c5(h_s)))

        for hh, cc in pop:
            if cc < best_c5:
                best_c5, best_h = cc, hh.copy()

        h_evo, c5_evo = _evolve(pop, d3, seed=4242)
        if c5_evo < best_c5:
            best_c5, best_h = c5_evo, h_evo.copy()

    # Phase 4: Resolution adaptation (13% budget)
    d4 = t0 + budget_s * 0.90
    if time.time() < d4:
        for dn in [2, -2, 4, -4, 6, -6, 8, -8, 3, -3]:
            if time.time() > d4:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            sub_d = min(d4, time.time() + budget_s * 0.018)
            h_r, c5_r = _intensive_local(h_r, sub_d, max_rounds=60, seed=100 + dn)
            if c5_r < best_c5 - 1e-10:
                best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Final intensive polish (9% budget)
    d5 = t0 + budget_s * 0.98
    if time.time() < d5:
        h_f, c5_f = _intensive_local(best_h, d5, max_rounds=100, seed=99999)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Final pair polish
    if time.time() < d5:
        hp, c5p = _pair_sweep(best_h, d5, max_passes=5, seed=31337)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
