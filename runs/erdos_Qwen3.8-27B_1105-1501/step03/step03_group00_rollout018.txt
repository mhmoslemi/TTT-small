import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_weighted(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        d = r - (n - 1)
        g = abs(d)
        if g == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - g] += w * dx * (1.0 - h[g:])
            grad[g:] += -w * dx * h[:n - g]
        else:
            grad[g:] += w * dx * (1.0 - h[:n - g])
            grad[:n - g] += -w * dx * h[g:]
    return grad

def _max_alpha(h, p):
    n = len(h)
    a_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    return max(a_max, 1e-12)

def _line_search(h, p, c5_cur):
    a_max = _max_alpha(h, p)
    n_samp = 20
    alphas = np.linspace(0, a_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)

    return best_a, best_c5

def _softmax_weights(scores, tau):
    s = scores - np.max(scores)
    w = np.exp(s / tau)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    return w

def _lp_direction(h, scores, radius, top_k=None):
    from scipy.optimize import linprog
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
    else:
        idx = np.arange(n_lags)
        scores_k = scores
    G = np.zeros((len(idx), n))
    for i, r in enumerate(idx):
        d = r - (n - 1)
        g = abs(d)
        dx = 2.0 / n
        if g == 0:
            G[i, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[i, :n - g] += dx * (1.0 - h[g:])
            G[i, g:] += -dx * h[:n - g]
        else:
            G[i, g:] += dx * (1.0 - h[:n - g])
            G[i, :n - g] += -dx * h[g:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _resample(h, n_new):

                best_n = n_try

    # Phase 8: Final subgradient polish
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = _subgrad_optimize(best_h, t0 + budget_s * 0.995, max_iters=20000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_l, c5_l = _lp_optimize(best_h, t0 + budget_s * 0.998, top_k_list=[3, 5], radius_init=0.02, max_iters=1000)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
