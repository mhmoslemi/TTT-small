def _project(v, target):
    n = len(v)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        r = n - 1 + d
        m = n - d
        if m > 0:
            G[r, :m] += (1.0 - h[d:d+m])
            G[r, d:] += (-h[:m])
    for g in range(1, n):
        r = n - 1 - g
        m = n - g
        G[r, g:] += (1.0 - h[:m])
        G[r, :m] += (-h[g:g+m])
    return G

def _project(v, target):
    n = len(v)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_descent(h, deadline, beta_start=30.0, beta_end=10000.0, max_iters=20000):

    target = n / 2.0
    h = h.copy()

    best_c5 = cur_c5
    lr = 0.06
    stall = 0

        else:
            beta = max(beta_start, beta * 0.95)  # decrease beta when stuck
        scores = _scores(h)
        m = float(np.max(scores))

        if wsum < 1e-300:

        b_eq = np.array([target])
        
        lo_b = np.maximum(0.0, h - radius)
        hi_b = np.minimum(1.0, h + radius)
        bounds = list(zip(lo_b, hi_b)) + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_a, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.85
                continue
            h_new = res.x[:n]
            p = h_new - h
            pnorm = float(np.linalg.norm(p))
            if pnorm < 1e-13:
                stall += 1
                radius *= 0
