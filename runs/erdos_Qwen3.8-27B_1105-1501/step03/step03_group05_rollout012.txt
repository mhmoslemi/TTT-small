def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    # Bisection-based projection onto {0 ≤ x ≤ 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_score(h, d, n):
    """Gradient of score at lag d with respect to h."""
    dx = 2.0 / n
    r = np.zeros(n)
    low = max(0, -d)
    high = min(n, n - d)
    for p in range(low, high):
        q = p + d
        r[q] += dx * (1.0 - h[p])
        r[p] -= dx * h[q]
    return r

def _grad_all_scores(h):
    """Gradient matrix: shape (2n-1, n), row r is gradient of score r."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _active_set_lp(h, active_idx, scores, rho):
    """
    Solve: min t
    s.t. scores[r] + G[r,:]·δ ≤ t for r in active_idx
         ||δ||₂ ≤ rho
         sum(δ) = 0
         -h[i] ≤ δ[i] ≤ 1-h[i] for all i
    
    Approximate ||δ||₂ ≤ rho by using L∞ norm (componentwise) for LP solvability:
         |δ[i]| ≤ rho for all i
    """
    n = len(h)
    G = _grad_all_scores(h)
    La = G[active_idx, :]  # (k, n)
    sa = scores[active_idx]  # (k,)
    
    # Variables: δ (n) + t (1) = n+1
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: La @ δ - t ≤ -sa  =>  La @ δ + (-1)*t ≤ -sa
    A_ub = np.zeros((len(active_idx), n+1))
    A_ub[:, :n] = La
    A_ub[:, n] = -1.0
    b_ub = -sa
    
    # Box constraints on δ: -min(h[i], rho) ≤ δ[i] ≤ min(1-h[i], rho)
    # These go into bounds
    
    # Sum constraint: sum(δ) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_i = max(-h[i], -rho)
        hi_i = min(1-h[i], rho)
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'maxiter': 2000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample_feasible(h, n_new):
    """Resample h to n_new points using integral averaging, then project."""
    n = len(h)
    if n_new == n:
        return h.copy()
    target_new = n_new / 2.0
    # Use simple interpolation
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, target_new)

def _make_periodic(n, period, ones_in_period):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    for i in range(period):
        if ones_in_period[i]:
            for j in range(i, n, period):
                h[j] = 1.0
    return h

    return best_h, final_c5, best_n
