def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_coeffs(h, D):
    """Compute coefficients for f(t) = max_r (S_r + B_r*t + Q_r*t^2) * dx"""
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = -np.correlate(D, D, mode='full')
    return S, B, Q

def _line_eval(S, B, Q, t, dx):
    """Evaluate f(t) = max_r (S_r + B_r*t + Q_r*t^2) * dx"""
    vals = S + B * t + Q * t * t
    return float(np.max(vals) * dx)

def _feasible_interval(h, D):
    """Find the interval of t where h + t*D is in [0,1]"""
    lo = 0.0
    hi = 0.0
    for i in range(len(h)):
        if D[i] > 1e-14:
            lo = max(lo, -(h[i]) / D[i])
            hi = min(hi, (1.0 - h[i]) / D[i])
        elif D[i] < -1e-14:
            lo = max(lo, (1.0 - h[i]) / D[i])
            hi = min(hi, -h[i] / D[i])
    return lo, hi

def _golden_search(S, B, Q, dx, a, b, iters=50):
    """Golden section search for min of f(t) on [a,b] using line formula"""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _line_eval(S, B, Q, c, dx)
    fd = _line_eval(S, B, Q, d, dx)
    for _ in range(iters):
        if b - a < 1e-14:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = _line_eval(S, B, Q, c, dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = _line_eval(S, B, Q, d, dx)
    t_opt = (a + b) / 2.0
    return t_opt, _line_eval(S, B, Q, t_opt, dx)

def _line_search(h, D, c5_cur, dx, deadline):
    """Exact line search along direction D"""
    maxabs = float(np.max(np.abs(D)))
    if maxabs < 1e-13:
        return None, c5_cur

    D = D / maxabs
    
    # Mean correction
    D = D - float(np.mean(D))

    if abs(float(np.sum(D))) > 1e-10:

    S, B, Q = _line_coeffs(h, D)

    t_vals = np.concatenate([[0.0], t_vals])

    best_idx = int(np.argmin(f_vals))

        b = t_vals[best_idx + 1]
    else:

def _soft_grad_direction(h, tau):
    """Generate direction from soft-max weighted gradient of active lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        # Gradient of corr_k with respect to h
        # corr_k = sum_i h[i] * (1-h[i+d]) for d>=0
        # or sum_i h[i] * (1-h[i-d]) for d<0
        # d(h[i] - h[i]*(1-h[i+d]))/dh[j]
        if d >= 0:
            m = n - d
            if m > 0:
                # h[i] * (1-h[i+d]) for i=0..m-1
                # d/dh[j]: if j in [0,m): (1-h[j+d]); if j in [d, n): -h[j-d]
                idx1 = np.arange(m)
                grad[idx1] += wk * (1.0 - h[idx1 + d])
                idx2 = np.arange(d, n)
                grad[idx2] += wk * (-h[idx2 - d])
        else:
            g = -d
            m = n - g
            if m > 0:
                # h[i] * (1-h[i-g]) for i=g..n-1
                # d/dh[j]: if j in [g, n): (1-h[j-g]); if j in [0, m): -h[j+g]
                idx1 = np.arange(g, n)
                grad[idx1] += wk * (1.0 - h[idx1 - g])
                idx2 = np.arange(m)
                grad[idx2] += wk * (-h[idx2 + g])
    
    # Descending direction, remove mean
    D = -grad
    D = D - float(np.mean(D))
    return D

def _block_transfer_direction(rng, n, block_sizes=None):
    """Generate a block transfer direction with equal +1/-1 cardinality"""
    if block_sizes is None:
        block_sizes = [2, 3, 4, 5, 6, 8, 10, 14, 20, 30]
    
    k = int(rng.integers(0, len(block_sizes)))
    size = block_sizes[k]
    
    if size >= n:
        size = max(1, n // 4)
    
    # Choose two non-overlapping (or minimally overlapping) blocks
    perm = rng.permutation(n)
    pos_block = perm[:size]
    neg_block = perm[size:2*size]
    
    # Remove overlap
    neg_set = set(neg_block.tolist())
    pos_block = np.array([x for x in pos_block if x not in neg_set])
    if len(pos_block) < size:
        # Fill from remaining
        remaining = [x for x in perm[2*size:] if x not in set(pos_block.tolist())]
        if len(remaining) >= size - len(pos_block):
            fill = rng.choice(remaining, size - len(pos_block), replace=False)
            pos_block = np.concatenate([pos_block, fill])
    
    size_actual = len(pos_block)
    if size_actual == 0:
        return np.zeros(n)
    
    # Adjust neg_block to same size
    neg_candidates = [x for x in perm[size:2*size+size_actual] if x not in set(pos_block.tolist())]
    if len(neg_candidates) < size_actual:
        remaining = [x for x in perm[2*size:] if x not in set(pos_block.tolist())]
        for x in remaining:
            if len(neg_candidates) >= size_actual:
                break
            if x not in set(neg_candidates):
                neg_candidates.append(x)
    neg_block = np.array(neg_candidates[:size_actual])
    
    D = np.zeros(n)
    D[pos_block] = 1.0
    D[neg_block] = -1.0
    
    if float(np.sum(D)) != 0:
        # Shouldn't happen if sizes match
        D = D - float(np.mean(D))
    
    return D

def _sine_direction(rng, n, max_freq=6):
    """Generate a random smooth sine direction"""
    f1 = float(rng.uniform(1, max_freq))
    f2 = float(rng.uniform(1, max_freq))
    p1 = float(rng.uniform(0, 2 * np.pi))
    p2 = float(rng.uniform(0, 2 * np.pi))
    a1 = float(rng.uniform(0.5, 1.0))
    a2 = float(rng.uniform(0.3, 0.8))
    
    x = np.arange(n, dtype=float) / n
    D = a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
    D = D - float(np.mean(D))
    return D

def _random_smooth_direction(rng, n, bandwidth=5):
    """Generate a random smoothed direction"""
    D = rng.normal(0, 1, n)
    # Simple box filter smoothing
    for _ in range(3):
        D = np.convolve(D, np.ones(bandwidth) / bandwidth, mode='same')
    D = D - float(np.mean(D))
    return D

def _pair_direction(rng, n):
    """Generate a pair direction e_i - e_j"""
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        j = (i + 1) % n
    D = np.zeros(n)
    D[i] = 1.0
    D[j] = -1.0
    return D

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_coeffs(h, D):
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = -np.correlate(D, D, mode='full')
    return S, B, Q

def _line_eval(S, B, Q, t, dx):
    vals = S + B * t + Q * t * t
    return float(np.max(vals) * dx)

def _feasible_interval(h, D):
    n = len(h)
    lo = -1e18
    hi = 1e18
    for i in range(n):
        if D[i] > 1e-14:
            lo = max(lo, -h[i] / D[i])
            hi = min(hi, (1.0 - h[i]) / D[i])
        elif D[i] < -1e-14:
            lo = max(lo, (1.0 - h[i]) / D[i])
            hi = min(hi, -h[i] / D[i])
    return max(lo, -1e10), min(hi, 1e10)

def _golden_search(S, B, Q, dx, a, b, iters=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _line_eval(S, B, Q, c, dx)
    fd = _line_eval(S, B, Q, d, dx)
    for _ in range(iters):
        if b - a < 1e-15:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = _line_eval(S, B, Q, c, dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = _line_eval(S, B, Q, d, dx)
    t_opt = (a + b) / 2.0
    return t_opt, _line_eval(S, B, Q, t_opt, dx)

def _line_search(h, D, c5_cur, dx, deadline):
    maxabs = float(np.max(np.abs(D)))
    if maxabs < 1e-13:
        return None, c5_cur
    D = D / maxabs
    D = D - float(np.sum(D)) / len(D)
    lo, hi = _feasible_interval(h, D)
    if lo >= hi - 1e-12:
        return None, c5_cur
    S, B, Q = _line_coeffs(h, D)
    n_samp = 35
    t_vals = np.linspace(lo, hi, n_samp)
    f_vals = np.empty(n_samp)
    for i in range(n_samp):
        f_vals[i] = _line_eval(S, B, Q, t_vals[i], dx)
    best_idx = int(np.argmin(f_vals))
    best_t = t_vals[best_idx]
    best_f = f_vals[best_idx]
    if best_f >= c5_cur - 1e-12:
        return None, c5_cur
    if best_idx > 0:
        a = t_vals[best_idx - 1]
    else:
        a = lo
    if best_idx < n_samp - 1:
        b = t_vals[best_idx + 1]
    else:
        b = hi
    t_opt, f_opt = _golden_search(S, B, Q, dx, a, b, iters=45)
    if f_opt > best_f:
        t_opt, f_opt = best_t, best_f
    if f_opt >= c5_cur - 1e-12:
        return None, c5_cur
    h_new = h + t_opt * D
    if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
        h_new = _project(h_new, len(h) / 2.0)
    c5_new = _exact_c5(h_new)
    if c5_new < c5_cur - 1e-11:
        return h_new, c5_new
    return None, c5_cur

def _soft_grad_direction(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                idx1 = np.arange(m)
                grad[idx1] += wk * (1.0 - h[idx1 + d])
                idx2 = np.arange(d, n)
                grad[idx2] += wk * (-h[idx2 - d])
        else:
            g = -d
            m = n - g
            if m > 0:
                idx1 = np.arange(g, n)
                grad[idx1] += wk * (1.0 - h[idx1 - g])
                idx2 = np.arange(m)
                grad[idx2] += wk * (-h[idx2 + g])
    D = -grad
    D = D - float(np.mean(D))
    return D

def _block_transfer_direction(rng, n, block_sizes=None):
    if block_sizes is None:
        block_sizes = [2, 3, 4, 5, 6, 8, 10, 14, 20, 30, 40]
    k = int(rng.integers(0, len(block_sizes)))
    size = block_sizes[k]
    if size >= n // 2:
        size = max(1, n // 6)
    perm = rng.permutation(n)
    pos_block = perm[:size].copy()
    neg_block = perm[size:2 * size].copy()
    pos_set = set(pos_block.tolist())
    neg_mask = np.array([x not in pos_set for x in neg_block])
    neg_block = neg_block[neg_mask]
    if len(neg_block) < size:
        remaining = perm[2 * size:]
        for x in remaining:
            if len(neg_block) >= size:
                break
            if x not in pos_set:
                neg_block = np.append(neg_block, x)
    size_actual = min(len(pos_block), len(neg_block))
    if size_actual == 0:
        return np.zeros(n)
    D = np.zeros(n)
    D[pos_block[:size_actual]] = 1.0
    D[neg_block[:size_actual]] = -1.0
    return D

def _sine_direction(rng, n, max_freq=6):
    f1 = float(rng.uniform(1, max_freq))
    f2 = float(rng.uniform(1, max_freq))
    p1 = float(rng.uniform(0, 2 * np.pi))
    p2 = float(rng.uniform(0, 2 * np.pi))
    a1 = float(rng.uniform(0.5, 1.0))
    a2 = float(rng.uniform(0.3, 0.8))
    x = np.arange(n, dtype=float) / n
    D = a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
    D = D - float(np.mean(D))
    return D

def _random_smooth_direction(rng, n, bandwidth=4):
    D = rng.normal(0, 1, n)
    kernel = np.ones(bandwidth) / bandwidth
    for _ in range(3):
        D = np.convolve(D, kernel, mode='same')
    D = D - float(np.mean(D))
    return D

def _pair_direction(rng, n, interior_mask=None):
    if interior_mask is not None and np.sum(interior_mask) >= 2:
        idxs = np.where(interior_mask)[0]
        i = idxs[int(rng.integers(0, len(idxs)))]
        j = idxs[int(rng.integers(0, len(idxs)))]
    else:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
    if i == j:
        j = (i + 1) % n
    D = np.zeros(n)
    D[i] = 1.0
    D[j] = -1.0
    return D

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize_directions(h, c5_cur, n, dx, deadline, rng):
    h_local = h.copy()
    c5_local = c5_cur
    block_sizes = [2, 3, 4, 5, 6, 8, 10, 14, 20, 28, 40]
    stall = 0
    n_iter = 0
    while time.time() < deadline:
        n_iter += 1
        if stall > 120:
            break
        r = rng.random()
        if r < 0.25:
            tau = float(rng.uniform(15, 300))
            D = _soft_grad_direction(h_local, tau)
        elif r < 0.50:
            D = _block_transfer_direction(rng, n, block_sizes)
        elif r < 0.65:
            D = _sine_direction(rng, n, max_freq=5)
        elif r < 0.80:
            D = _random_smooth_direction(rng, n, bandwidth=int(rng.integers(2, 8)))
        elif r < 0.92:
            interior = (h_local > 0.05) & (h_local < 0.95)
            D = _pair_direction(rng, n, interior)
        else:
            # Combined: soft grad + random
            tau = float(rng.uniform(30, 150))
            D1 = _soft_grad_direction(h_local, tau)
            D2 = _random_smooth_direction(rng, n, bandwidth=3)
            D = 0.7 * D1 + 0.3 * D2
        if float(np.max(np.abs(D))) < 1e-12:
            stall += 1
            continue
        h_new, c5_new = _line_search(h_local, D, c5_local, dx, deadline)
        if h_new is not None:
            h_local = h_new
            c5_local = c5_new
            stall = 0
        else:
            # Try negated direction
            h_new2, c5_new2 = _line_search(h_local, -D, c5_local, dx, deadline)
            if h_new2 is not None:
                h_local = h_new2
                c5_local = c5_new2
                stall = 0
            else:

        if stall > 30 and stall % 30 == 0:
            scale = 0.015 + 0.01 * (stall // 30)
            h_local = h_local + rng.normal(0, scale, n)
            h_local = _project(h_local, n / 2.0)
            c5_local = _exact_c5(h_local)
            if c5_local > c5_cur + 0.01:

                c5_local = c5_cur
    return h_local, c5_local
