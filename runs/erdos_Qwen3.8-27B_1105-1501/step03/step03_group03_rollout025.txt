def _pair_line_search(scores, G, p, q, h):
    # Transfer from q to p
    delta = G[:, p] - G[:, q]
    # Feasible: 0 <= alpha <= min(h[q], 1-h[p])
    alpha_max = min(h[q], 1.0 - h[p])
    if alpha_max < 1e-12:
        return None, None
    # Golden section to minimize max(scores + alpha * delta)
    # But we need to handle both directions: +alpha (p up, q down) and -alpha (p down, q up)
    # For p up, q down: alpha in [0, min(h[q], 1-h[p])]
    # For p down, q up: alpha in [0, min(h[p], 1-h[q])]
    # Actually, let's just search alpha in the full feasible range for direction d = e_p - e_q
    # That's alpha in [-min(h[p], 1-h[q]), min(h[q], 1-h[p])]
    a_lo = -min(h[p], 1.0 - h[q])
    a_hi = min(h[q], 1.0 - h[p])
    # Golden section on [a_lo, a_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_lo + gr * (a_hi - a_lo)
    f1 = float(np.max(scores + c1 * delta))
    f2 = float(np.max(scores + c2 * delta))
    for _ in range(40):
        if a_hi - a_lo < 1e-10:
            break
        if f1 < f2:
            a_hi = c2
            c2 = c1
            f2 = f1
            c1 = a_hi - gr * (a_hi - a_lo)
            f1 = float(np.max(scores + c1 * delta))
        else:
            a_lo = c1
            c1 = c2
            f1 = f2
            c2 = a_lo + gr * (a_hi - a_lo)
            f2 = float(np.max(scores + c2 * delta))
    best_a = (a_lo + a_hi) / 2.0
    best_f = float(np.max(scores + best_a * delta))
    return best_a, best_f
```

def _compute_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for k in range(0, n):
        r = k + n - 1  # output index for lag k
        m = n - k
        # h[j+k] * (1-h[j]) for j = 0..m-1
        # dc/dh[j+k] += (1-h[j]) for j=0..m-1
        G[r, k:k+m] += one_minus_h[:m]
        # dc/dh[j] += -h[j+k] for j=0..m-1
        G[r, :m] -= h[k:k+m]
    for g in range(1, n):
        r = n - 1 - g  # output index for lag -g
        m = n - g
        # h[j] * (1-h[j+g]) for j = 0..m-1
        # dc/dh[j] += (1-h[j+g]) for j=0..m-1
        G[r, :m] += one_minus_h[g:g+m]
        # dc/dh[j+g] += -h[j] for j=0..m-1
        G[r, g:g+m] -= h[:m]
    return G
```

    v = np.array(v, dtype=float, copy=True)
    v[~np.isfinite(v)] = 0.5
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-11:
        if res > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for k in range(n):
        r = k + n - 1
        m = n - k
        if m > 0:
            G[r, k:k+m] += omh[:m]
            G[r, :m] -= h[k:k+m]
    for g in range(1, n):
        r = n - 1 - g
        m = n - g
        if m > 0:
            G[r, :m] += omh[g:g+m]
            G[r, g:g+m] -= h[:m]
    return G

def _f_2coord(h, p, q, a):
    hc = h.copy()
    hc[p] += a
    hc[q] -= a
    return float(np.max(np.correlate(hc, 1.0 - hc, mode='full')))

def _golden_2coord(h, p, q, n_iter=35):
    a_lo = -min(h[p], 1.0 - h[q])
    a_hi = min(h[q], 1.0 - h[p])
    if a_hi - a_lo < 1e-14:
        return 0.0, _c5_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_lo + gr * (a_hi - a_lo)
    f1 = _f_2coord(h, p, q, c1)
    f2 = _f_2coord(h, p, q, c2)
    for _ in range(n_iter):
        if a_hi - a_lo < 1e-13:
            break
        if f1 < f2:
            a_hi = c2
            c2 = c1
            f2 = f1
            c1 = a_hi - gr * (a_hi - a_lo)
            f1 = _f_2coord(h, p, q, c1)
        else:
            a_lo = c1
            c1 = c2
            f1 = f2
            c2 = a_lo + gr * (a_hi - a_lo)
            f2 = _f_2coord(h, p, q, c2)
    best_a = (a_lo + a_hi) / 2.0
    best_f = _f_2coord(h, p, q, best_a)
    # Check endpoints
    eps = 1e-13
    f_lo = _f_2coord(h, p, q, a_lo + eps)
    if f_lo < best_f:
        best_a = a_lo + eps
        best_f = f_lo
    f_hi = _f_2coord(h, p, q, a_hi - eps)
    if f_hi < best_f:
        best_a = a_hi - eps
        best_f = f_hi
    return best_a, best_f

def _pass_2coord(h, target, deadline, cur_max_raw):
    """One full pass of two-coordinate search. Returns updated h, max_raw, improved."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    improved = False
    stall = 0
    for p in range(n):
        if time.time() > deadline:
            break
        for q in range(p + 1, n):
            if time.time() > deadline:
                break
            best_a, best_f = _golden_2coord(h, p, q, n_iter=30)
            if best_f < cur_max_raw - 1e-12:
                h[p] += best_a
                h[q] -= best_a
                ## Project to fix any drift
                h = _project(h, target)
                cur_max_raw = _c5_exact(h) * n / 2.0  # raw max
                # Wait, _c5_exact returns C5 = raw * dx = raw * 2/n
                # So raw = C5 * n / 2
                improved = True
                stall = 0
            else:
                stall += 1
    return h, cur_max_raw, improved

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_perturb(h, target, rng, sigma):
    n = len(h)
    noise = rng.standard_normal(n) * sigma
    noise -= np.mean(noise)
    h_new = h + noise
    return _project(h_new, target)

def _direction_line_search(h, d, n_iter=25):
    """Line search along direction d (must have sum(d)~0)."""
    n = len(h)
    d = d - np.mean(d)  # ensure zero sum
    norm = float(np.sqrt(np.sum(d**2)))
    if norm < 1e-14:
        return 0.0
    d = d / norm  # normalize
    
    # Find feasible interval
    a_lo = -1e15
    a_hi = 1e15
    for i in range(n):
        if d[i] > 1e-14:
            a_lo = max(a_lo, -h[i] / d[i])
            a_hi = min(a_hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            a_lo = max(a_lo, (1.0 - h[i]) / d[i])
            a_hi = min(a_hi, -h[i] / d[i])
    if a_hi - a_lo < 1e-12:
        return 0.0
    
    # Clip to reasonable range
    a_lo = max(a_lo, -1.0)
    a_hi = min(a_hi, 1.0)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_lo + gr * (a_hi - a_lo)
    
    def _eval(a):
        return float(np.max(np.correlate(h + a * d, 1.0 - (h + a * d), mode='full')))
    
    # Can't use closure! Let me inline.
    hc1 = h + c1 * d
    f1 = float(np.max(np.correlate(hc1, 1.0 - hc1, mode='full')))
    hc2 = h + c2 * d
    f2 = float(np.max(np.correlate(hc2, 1.0 - hc2, mode='full')))
    for _ in range(n_iter):
        if a_hi - a_lo < 1e-13:
            break
        if f1 < f2:
            a_hi = c2
            c2 = c1
            f2 = f1
