    c[n] = 1.0  # minimize mu

                      options={'maxiter': 10000})

            d = res.x[:n]
            return _zero_sum(d)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time

def _repair(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _score_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        m = n - k
        if m > 0:
            g[:m] -= h[k:k+m]
            g[k:] += 1.0 - h[:m]
    else:
        ga = -k
        m = n - ga
        if m > 0:
            g[:m] += 1.0 - h[ga:ga+m]
            g[ga:] -= h[:m]
    return g

def _line_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for r in range(M):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                A[r] = np.dot(h[k:k+m], 1.0 - h[:m])
                B[r] = np.dot(d[k:k+m], 1.0 - h[:m]) - np.dot(h[k:k+m], d[:m])
                C[r] = -np.dot(d[k:k+m], d[:m])
        else:
            ga = -k
            m = n - ga
            if m > 0:
                A[r] = np.dot(h[:m], 1.0 - h[ga:ga+m])
                B[r] = np.dot(d[:m], 1.0 - h[ga:ga+m]) - np.dot(h[:m], d[ga:ga+m])
                C[r] = -np.dot(d[:m], d[ga:ga+m])
    return A, B, C

def _exact_1d(h, d):
    n = len(h)
    A, B, C = _line_coeffs(h, d)
    M = 2 * n - 1
    lo, hi = -1e10, 1e10
    for i in range(n):
        di = d[i]
        if di > 1e-14:
            lo = max(lo, -h[i] / di)
            hi = min(hi, (1.0 - h[i]) / di)
        elif di < -1e-14:
            lo = max(lo, (1.0 - h[i]) / di)
            hi = min(hi, -h[i] / di)
    if lo > hi:
        return 0.0, float(np.max(A))
    if hi - lo < 1e-14:
        return 0.0, float(np.max(A))
    cands = [lo, hi, 0.0]
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if lo - 1e-10 <= t0 <= hi + 1e-10:
                cands.append(t0)
    for i in range(M):
        Ci, Bi, Ai = C[i], B[i], A[i]
        for j in range(i + 1, M):
            a = Ci - C[j]
            b = Bi - B[j]
            c = Ai - A[j]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    t0 = -c / b
                    if lo - 1e-10 <= t0 <= hi + 1e-10:
                        cands.append(t0)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-12:
                    if disc < 0:
                        disc = 0.0
                    sd = np.sqrt(disc)
                    t1 = (-b + sd) / (2.0 * a)
                    t2 = (-b - sd) / (2.0 * a)
                    if lo - 1e-10 <= t1 <= hi + 1e-10:
                        cands.append(t1)
                    if lo - 1e-10 <= t2 <= hi + 1e-10:
                        cands.append(t2)
    cands = np.array(cands)
    cands = np.clip(cands, lo, hi)
    cands = np.sort(cands)
    if len(cands) > 2:
        mask = np.concatenate([[True], np.diff(cands) > 1e-11])
        cands = cands[mask]
    T = cands[:, None]
    S_mat = A[None, :] + B[None, :] * T + C[None, :] * T * T
    env = S_mat.max(axis=1)
    bi = int(env.argmin())
    return float(cands[bi]), float(env[bi])

def _zero_sum(d):
    d = d - d.mean()
    mx = float(np.max(np.abs(d)))
    if mx < 1e-15:
        return None
    return d / mx

def _smooth_grad_direction(h, scores, top_k, beta):
    n = len(h)
    idx = np.argsort(scores)[::-1][:top_k]
    max_s = float(scores[idx[0]])
    w = np.exp(beta * (scores[idx] - max_s))
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros(len(idx))
        w[0] = 1.0
    else:
        w /= ws
    d = np.zeros(n)
    for wi, ri in zip(w, idx):
        k = int(ri) - (n - 1)
        d += wi * _score_grad(h, k)
    return _zero_sum(d)

def _lp_direction(h, scores, top_k):
    n = len(h)
    idx = np.argsort(scores)[::-1][:top_k]
    nf = len(idx)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((nf, n + 1))
    b_ub = np.zeros(nf)
    for ii, ri in enumerate(idx):
        k = int(ri) - (n - 1)
        g = _score_grad(h, k)
        A_ub[ii, :n] = -g
        A_ub[ii, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            return _zero_sum(d)
    except Exception:
        pass
    return None

def _random_direction(rng, n):
    d = rng.standard_normal(n)
    return _zero_sum(d)

def _pair_direction(rng, n):
    i, j = rng.choice(n, 2, replace=False)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new / 2.0)

    if time.time() < deadline:
        h_f, c5_f = _local_descent(best_h, deadline, rng, top_k=25, max_iters=10000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _repair(best_h, best_n / 2.0)
    final_c5 = _c5_val(best_h)
    if final_c5 > best_c5 + 1e-8:
        # Repair degraded quality; use un-repaired if it was feasible
        if float(np.sum(best_h)) >= best_n / 2.0 - 1e-6 and float(np.sum(best_h)) <= best_n / 2.0 + 1e-6:
            pass
        else:
            best_h = _repair(best_h, best_n / 2.0)
            final_c5 = _c5_val(best_h)

    return best_h, final_c5, best_n
