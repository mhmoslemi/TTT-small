import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_descent(h, deadline, p_start=8.0, p_end=200.0, n_steps=200, step_scale=0.03):
    """p-norm annealing: minimize (sum(s^p))^(1/p) which approaches max(s) as p->inf"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        # Interpolate p from p_start to p_end
        p = p_start * (p_end / p_start) ** frac
        
        scores = _scores(h)
        # Compute softmax weights: w_r = exp(p * (s_r - max_s)) / sum(exp(p * (s_j - max_s)))
        max_s = np.max(scores)
        exps = np.exp(p * (scores - max_s))
        w = exps / np.sum(exps)
        
        # Gradient of p-norm: proportional to G^T @ w (for the weighted sum version)
        # More precisely for (sum(s^p))^(1/p), grad = p * G^T @ (s^(p-1)) / (sum(s^p))^(1 - 1/p)
        # But the direction is the same as G^T @ w where w_r proportional to s_r^(p-1)
        G = _grad_matrix(h)
        grad = G.T @ w
        
        # Remove mean to satisfy sum constraint
        grad -= np.mean(grad)
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Line search
        # Find max alpha that keeps h in [0,1]
        alpha_max = step_scale
        for j in range(n):
            if grad[j] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h[j]) / grad[j])
            elif grad[j] < -1e-12:
                alpha_max = min(alpha_max, h[j] / (-grad[j]))
        alpha_max = max(alpha_max * 0.9, 0.001)
        
        # Try a few step sizes
        best_alpha = 0
        best_val = cur_c5
        for frac_a in [1.0, 0.5, 0.25, 0.1]:
            a = alpha_max * frac_a
            h_new = h + a * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_val:
                best_val = c5_new
                best_alpha = a
            if c5_new < cur_c5 - 1e-12:
                break
        
        if best_alpha > 0:
            h = h + best_alpha * grad
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Try smaller step
            for a in [0.01, 0.005, 0.001]:
                h_new = h + a * grad
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
    
    return best_h, best_c5

def _active_lsq(h, deadline, k_active=8, n_iters=200):
    """Active-set least squares: solve LSQ on top-k constraints"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        # Get top-k active constraints
        idx = np.argsort(scores)[-k_active:]
        G = _grad_matrix(h)
        G_active = G[idx]  # (k_active, n)
        s_active = scores[idx]
        
        # We want to solve: minimize ||G_active @ d + s_active||^2
        # subject to sum(d) = 0, |d| <= radius
        # This is: min ||G_active @ d||^2 + 2 * s_active^T G_active @ d + const
        # Using Lagrange for sum=0: (G_active^T G_active - lambda * 1) d = -G_active^T s_active
        
        # Add regularization for stability
        A = G_active.T @ G_active + 1e-6 * np.eye(n)
        A[np.diag_indices(n)] += 1e-6
        b_vec = -G_active.T @ s_active
        
        # Solve with sum constraint: use null space of [1,1,...,1]
        # Project A and b to remove mean component
        # d = particular + null * t where null is the null space of sum
        # Simpler: solve (A + alpha * 11^T) d = b_vec - alpha * sum(b) * 1 / n
        # This enforces sum(d) = 0 in the limit as alpha -> inf
        # For practical purposes, just solve and subtract mean
        try:
            d = np.linalg.solve(A, b_vec)
            d -= np.mean(d)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(A, b_vec, rcond=None)[0]
            d -= np.mean(d)
        
        # Scale direction
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            break
        d /= d_norm
        
        # Line search
        alpha_max = 0.5
        for j in range(n):
            if d[j] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
            elif d[j] < -1e-12:
                alpha_max = min(alpha_max, h[j] / (-d[j]))
        alpha_max = max(alpha_max * 0.8, 0.01)
        
        best_alpha = 0
        best_val = cur_c5
        for frac_a in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a = alpha_max * frac_a
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_val:
                best_val = c5_new
                best_alpha = a
            if c5_new < cur_c5 - 1e-12:
                break
        
        if best_alpha > 0:
            h = h + best_alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Try even smaller
            improved = False
            for a in [0.005, 0.001, 0.0001]:
                h_new = h + a * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                # Try different k
                k_active = max(3, k_active - 1)
                if k_active < 3:
                    break
    
    return best_h, best_c5

def _binary_optimize(h_binary, deadline, n_neighbors=200, max_rounds=50):
    """Local search on binary vectors (0/1 with exactly n/2 ones)"""
    n = len(h_binary)
    target = n // 2  # exactly n/2 ones for even n
    assert n % 2 == 0, "Need even n for binary"
    
    h_b = h_binary.copy()
    cur_c5 = _exact_c5(h_b.astype(float))
    best_c5 = cur_c5
    best_h = h_b.copy()
    
    ones = set(np.where(h_b == 1)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        
        # Generate random neighbor moves: swap one 1 with one 0
        zeros = sorted(ones.__class__({i for i in range(n) if i not in ones}))
        
        # Try top candidates first
        tried = 0
        for _ in range(n_neighbors):
            if time.time() > deadline:
                break
            a = list(ones)[np.random.randint(len(ones))]
            b = list(zeros)[np.random.randint(len(zeros))]
            
            # Try swap: set a to 0, b to 1
            h_try = h_b.copy()
            h_try[a] = 0
            h_try[b] = 1
            c5_try = _exact_c5(h_try.astype(float))
            
            if c5_try < cur_c5 - 1e-14:
                h_b = h_try
                ones.discard(a)
                ones.add(b)
                zeros.remove(b) if b in zeros else None
                zeros = sorted({i for i in range(n) if i not in ones})
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_b.copy()
                improved = True
            tried += 1
        
        if not improved:
            break
    
    return best_h.astype(float), best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, step_limit=0.5):
    n = len(h)
    alpha_max = step_limit
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    best_a = 0
    best_c5 = _exact_c5(h)
    n_samp = 15
    alphas = np.linspace(0, alpha_max, n_samp)
    for a in alphas[1:]:
        h_new = h + a * p
        h_new = _project(h_new, n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_a = a
    return best_a, best_c5
