import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    """Gradient of Q_d(h) = sum h[i+d]*(1-h[i]) with respect to h.
    d is the signed lag: d = r - (n-1) where r is the correlate index."""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _weighted_grad(h, scores, beta=20.0, top_k=20):
    """Compute weighted gradient of top-k scores."""
    n = len(h)
    # Get top-k score indices
    idx = np.argsort(scores)[::-1][:top_k]
    # Softmax weights among top-k
    s_top = scores[idx]
    max_s = float(np.max(s_top))
    w = np.exp(beta * (s_top - max_s))
    w /= np.sum(w)
    
    # Compute weighted gradient
    g = np.zeros(n)
    for i, r in enumerate(idx):
        d = int(r - (n - 1))
        g += w[i] * _score_grad(h, d)
    
    # Project to zero-mean
    g -= np.mean(g)
    return g

def _line_search_c5(h, direction, max_alpha=None):

            d_ = c
            f_d = f_c

            c = d_

def _pair_transfer_c5(h, i, j):
    """Try to improve C5 by transferring mass from h[i] to h[j] (or vice versa).
    We move t: h[i] += t, h[j] -= t (if t > 0 moves from j to i).
    Find optimal t in [-h[j], 1-h[i]] (for moving from j to i)

def _project(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection on threshold."""
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_gradient_descent(h, deadline, beta=20.0, top_k=15, max_iters=5000):
    """Main optimization loop using smoothed gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        scores = _all_scores(h)
        g = _weighted_grad(h, scores, beta=beta, top_k=top_k)
        
        # Normalize direction
        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-15:
            stall += 1
            continue
        g = -g / g_norm  # descent direction

        alpha, c5_new = _line_search_c5(h, g)

            if abs(np.sum(h) - target) > 1e-10:
                h = _project(h, target)
            best_h = h.copy()
            best_c5 = c5_new
            stall = 0
        else:
            # Try reducing beta (focus more on the max)
            stall += 1
            if stall % 50 == 0:
                beta = max(5.0, beta * 0.9)
                top_k = max(5, top_k - 1)
    
    return best_h, best_c5

def _pair_transfer_loop(h, deadline, max_rounds=50):

    n = len(h)
    target = n / 2.0
    h = h.copy()

                        h[i] += t_opt
                        h[j] -= t_opt
                        best_c5 = c5_new
                        improved = True
                    
            # Direction 2: increase h[j], decrease h[i]
            t_max2 = min(h[i], 1.0 - h[j])
            if t_max2 > 1e-15:
                h_mid = h.copy()
                h_mid[j] += t_max2 / 2

                    h_tmp = h.copy()
                    # Swap i and j
                    _, c5_new = _pair_transfer_c5(h_tmp, j, i)
                    if c5_new < best_c5 - 1e-14:
                        t_opt, _ = _pair_transfer_c5(h, j, i)
                        h[j] += t_opt
                        h[i] -= t_opt

        else:

def _antipodal_seed(n, seed):
    """Generate an antipodal half-complement seed."""
    rng = np.random.default_rng(seed)
    m = n // 2
    y = rng.random(m)
    # Try a few permutations
    best_h = None
    best_c5 = float('inf')
    
    for variant in range(5):
        if variant == 0:
            h = np.concatenate([y, 1.0 - y])
        elif variant == 1:
            h = np.concatenate([y, 1.0 - y[::-1]])
        elif variant == 2:
            perm = rng.permutation(m)
            h = np.concatenate([y, 1.0 - y[perm]])
        elif variant == 3:
            # Shift
            shift = rng.integers(1, m)
            h = np.concatenate([y, 1.0 - np.roll(y, shift)])
        else:
            # Interleaved
            h = np.zeros(n)
            h[0::2] = y[:n//2]
            h[1::2] = 1.0 - y[:(n-1)//2] if n % 2 else 1.0 - y
            h = h[:n]
        
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, n / 2.0)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

    f_d = _exact_c5(hd)
    for _ in range(20):
        if b - a < 1e-12:
            break
        if f_c < f_d:
            b = d_; d_ = c; f_d = f_c
            c = b - gr2 * (b - a)
            hc = h.copy(); hc[i] += c; hc[j] -= c
            f_c = _exact_c5(hc)
        else:
            a = c; c = d_; f_c = f_d
            d_ = a + gr2 * (b - a)
            hd = h.copy(); hd[i] += d_; hd[j] -= d_
            f_d = _exact_c5(hd)
    t = (a + b) / 2.0
    h_new = h.copy(); h_new[i] += t; h_new[j] -= t
    return t, _exact_c5(h_new)

def _top1_descent(h, deadline, max_iters=20000):
    """Focus on only the single maximum score."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        scores = _all_scores(h)
        r_max = int(np.argmax(scores))
        d = r_max - (n - 1)
        g = _score_grad(h, d)
        g -= np.mean(g)
        norm = float(np.max(np.abs(g)))
        if norm < 1e-15:
            stall += 1
            continue
        direction = -g / norm
        alpha, c5_new = _line_search(h, direction)
        if c5_new < best_c5 - 1e-14:
            h_new = np.clip(h + alpha * direction, 0.0, 1.0)
            if abs(float(np.sum(h_new)) - target) > 1e-10:
                h_new = _project(h_new, target)
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _top3_balance(h, deadline, max_iters=20000):
    """Equalize the top 3 scores."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        scores = _all_scores(h)
        idx3 = np.argsort(scores)[::-1][:3]
        s3 = scores[idx3]
        max_s = float(np.max(s3))
        # Equal weights on top 3
        g = np.zeros(n)
        for i in range(3):
            r = int(idx3[i])
            d = r - (n - 1)
            g += _score_grad(h, d)
        g -= np.mean(g)
        norm = float(np.max(np.abs(g)))
        if norm < 1e-15:
            stall += 1
            continue
        direction = -g / (3.0 * norm)
        alpha, c5_new = _line_search(h, direction)
        if c5_new < best_c5 - 1e-14:
            h_new = np.clip(h + alpha * direction, 0.0, 1.0)
            if abs(float(np.sum(h_new)) - target) > 1e-10:
                h_new = _project(h_new, target)
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Top-1 focused descent (aggressive on max score)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _top1_descent(best_h, dl1, max_iters=30000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Smoothed gradient with moderate beta
    dl2 = t0 + budget_s * 0.28
    h2, c5_2 = _gradient_descent(best_h, dl2, beta=25.0, top_k=8, max_iters=30000, momentum=0.2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Top-3 balance
    dl3 = t0 + budget_s * 0.40
    h3, c5_3 = _top3_balance(best_h, dl3, max_iters=20000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Pair transfer polish
    dl4 = t0 + budget_s * 0.50
    h4, c5_4 = _pair_loop(best_h, dl4, max_rounds=60, n_trials_per_round=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: More gradient descent with different params
    dl5 = t0 + budget_s * 0.60
    h5, c5_5 = _gradient_descent(best_h, dl5, beta=60.0, top_k=5, max_iters=20000, momentum=0.1)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Random restarts
    dl6 = t0 + budget_s * 0.82
    attempt = 0
    while time.time() < dl6:
        attempt += 1
        scale = 0.04 * (0.93 ** min(attempt, 25))
        if scale < 0.004:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        per_dl = time.time() + min(20.0, (dl6 - time.time()) * 0.35)
        if per_dl - time.time() < 2.0:
            break
        h_try, c5_try = _gradient_descent(h_pert, per_dl, beta=30.0, top_k=8, max_iters=5000, momentum=0.2)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 7: Try different resolutions
    dl7 = t0 + budget_s * 0.92
    if time.time() < dl7:
        for n_try in [best_n + 16, best_n + 8, best_n + 32, best_n - 8]:
            if time.time() > dl7:
                break
            if n_try < 32 or n_try > 256 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            per_dl = min(dl7, time.time() + 25.0)
            if per_dl - time.time() < 3.0:
                continue
            h_opt1, c5_opt1 = _top1_descent(h_r, per_dl, max_iters=10000)
            if c5_opt1 < best_c5:
                h_opt2, c5_opt2 = _gradient_descent(h_opt1, min(dl7, time.time() + 10.0),
                                                     beta=30, top_k=8, max_iters=5000, momentum=0.2)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n_try
                elif c5_opt1 < best_c5:
                    best_c5 = c5_opt1
                    best_h = h_opt1.copy()
                    best_n = n_try
    
    # Phase 8: Final alternating polish
    if time.time() < t0 + budget_s * 0.99:
        dl8 = t0 + budget_s * 0.995
        for _ in range(5):
            if time.time() > dl8:
                break
            h_a, c5_a = _top1_descent(best_h, min(dl8, time.time() + 8.0), max_iters=5000)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
            h_b, c5_b = _pair_loop(best_h, min(dl8, time.time() + 5.0), max_rounds=20, n_trials_per_round=30)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
