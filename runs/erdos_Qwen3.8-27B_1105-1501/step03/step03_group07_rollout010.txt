import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for threshold
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_tangent_step(h, top_k=None, delta=0.01, reg=1e-4):
    """
    Solve the tangent LP and return the optimal direction d.
    top_k: if set, only use top K constraints
    delta: threshold for top-k selection
    reg: regularization parameter on |d|
    """
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    n_scores = len(scores)  # 2n-1
    
    # Build gradient matrix
    # For shift s (index i in scores, shift = i - (n-1)):
    # grad[i, j] = dJ_i/dh[j]
    
    # Instead of building full matrix, let's compute constraints on the fly
    # Actually for linprog we need the matrix, so let's build it
    
    # For shift d >= 0 (index i = d + n - 1):
    #   J = sum_{m=0}^{n-1-d} h[m+d]*(1-h[m])
    #   grad[j] = 1 if d<=j<=n-1, minus h[j-d] if d<=j<=n-1, minus h[j+d] if j<=n-1-d
    
    # For shift -d (index i = n-1-d):
    #   J = sum_{m=0}^{n-1-d} h[m]*(1-h[m+d])
    #   grad[j] = 1 if 0<=j<=n-1-d, minus h[j+d] if 0<=j<=n-1-d, minus h[j-d] if j>=d
    
    # Let me build the gradient matrix
    # G[i, j] = d(scores[i])/d(h[j])
    
    G = np.zeros((2*n-1, n))
    
    for d in range(0, n):
        m = n - d  # number of terms
        if m <= 0:
            continue
        # Positive shift d: index = d + n - 1
        idx_pos = d + n - 1
        # grad[j] for j in [d, n-1]: contribution from h[j] as the "first" term
        #   +1 from h[m+d] term
        #   -h[j-d] from product term
        # grad[j] for j in [0, m-1]: contribution from (1-h[m]) term  
        #   -h[j+d] from product term
        
        # For j in [d, n-1]:
        G[idx_pos, d:n] += 1.0 - h[0:m]  # 1 - h[j-d] for j=d..n-1, h[j-d]=h[0..m-1]
        # For j in [0, m-1]:
        G[idx_pos, 0:m] += -h[d:n]  # -h[j+d] for j=0..m-1, h[j+d]=h[d..n-1]
        
        # Negative shift -d: index = n-1-d
        # (Skip d=0 since it's the same)
        if d > 0:
            idx_neg = n - 1 - d
            # J = sum_{m=0}^{m-1} h[m]*(1-h[m+d])
            # grad[j] for j in [0, m-1]:
            #   +1 from h[m] term
            #   -h[j+d] from product term
            # grad[j] for j in [d, n-1]:
            #   -h[j-d] from (1-h[m+d]) term
            
            G[idx_neg, 0:m] += 1.0 - h[d:n]  # 1 - h[j+d]
            G[idx_neg, d:n] += -h[0:m]  # -h[j-d]
    
    # Select constraints
    if top_k is not None and top_k < n_scores:
        sorted_idx = np.argsort(-scores)[:top_k]
    else:
        # Use all constraints above (J_max - delta)
        threshold = float(np.max(scores)) - delta
        sorted_idx = np.where(scores >= threshold)[0]
        if len(sorted_idx) == 0:
            sorted_idx = np.argsort(-scores)[:5]
    
    n_constr = len(sorted_idx)
    
    # Build LP
    # Variables: d[0..n-1], t
    # Minimize: reg * sum(|d_j|) + t
    # To handle |d_j|, split d_j = d_j^+ - d_j^-
    # Variables: d_0^+, d_0^-, ..., d_{n-1}^+, d_{n-1}^-, t
    # n_vars = 2n + 1
    
    # Actually, since we have bounds on d_j (-h_j <= d_j <= 1-h_j),
    # and we want to minimize reg*sum|d_j| + t, we can use the split.
    # But bounds: 0 <= d_j^+ <= 1-h_j, 0 <= d_j^- <= h_j
    # Sum constraint: sum(d_j^+ - d_j^-) = 0
    
    # Let's use a simpler approach: just use d directly with bounds,
    # and add reg * sum(d_j) + reg * sum(-d_j) as a proxy... 
    # Actually, let's just minimize t and rely on bounds to limit step size.
    # Or use a small L1 penalty via split variables.
    
    # Let me use the split variable approach for the L1 penalty.
    # Variables: [d0+, d0-, d1+, d1-, ..., d[n-1]+, d[n-1]-, t]
    # Total: 2n + 1 variables
    
    n_vars = 2 * n + 1
    c = np.zeros(n_vars)
    c[:2*n:2] = reg  # penalty on d^+
    c[1:2*n:2] = reg  # penalty on d^-
    c[2*n] = 1.0  # minimize t
    
    # Bounds
    bounds = []
    for j in range(n):
        bounds.append((0.0, 1.0 - h[j]))  # d_j^+
        bounds.append((0.0, h[j]))  # d_j^-
    bounds.append((0.0, None))  # t >= 0
    
    # Constraints:
    # 1. Sum constraint: sum(d_j^+ - d_j^-) = 0
    A_eq = np.zeros((1, n_vars))
    for j in range(n):
        A_eq[0, 2*j] = 1.0
        A_eq[0, 2*j+1] = -1.0
    b_eq = [0.0]
    
    # 2. Tangent constraints: for each selected shift i:
    #    G[i, :] @ d - t <= -scores[i]
    #    where d_j = d_j^+ - d_j^-
    
    A_ub = np.zeros((n_constr, n_vars))
    b_ub = np.zeros(n_constr)
    
    for idx, i in enumerate(sorted_idx):
        for j in range(n):
            A_ub[idx, 2*j] = G[i, j]       # coefficient of d_j^+
            A_ub[idx, 2*j+1] = -G[i, j]     # coefficient of d_j^-
        A_ub[idx, 2*n] = -1.0  # coefficient of t
        b_ub[idx] = -scores[i]
    
    # Solve
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                        bounds=bounds, method='highs')
    except Exception:
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='simplex')
        except Exception:
            return None
    
    if not result.success:
        return None
    
    # Extract d
    d = np.zeros(n)
    for j in range(n):
        d[j] = result.x[2*j] - result.x[2*j+1]
    
    # Normalize small noise
    if np.max(np.abs(d)) < 1e-12:
        return None
    
    return d

def _line_search(h, direction, target, cur_c5, max_alpha=1.0):
    """Line search along direction from h"""
    n = len(h)
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-6)
    alpha_max = min(alpha_max, 5.0)
    
    best_a = 0.0
    best_c = cur_c5
    
    # Try a range of alphas
    alphas = np.linspace(0.01, alpha_max, 15)
    
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-13:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, cur_c5
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.6)
    hi = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _lp_descent(h, deadline, target, max_iters=5000):
    """LP-based tangent descent"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    # Adaptively choose top_k and delta
    top_k_seq = [None, None, 20, 15, 10, 8, 6, 5, 4, 3]
    delta_seq = [0.0, 0.02, 0.01, 0.05, 0.005, 0.1, 0.01, 0.03]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        
        # Choose constraints
        tk = top_k_seq[min(it // 30, len(top_k_seq)-1)]
        dl = delta_seq[min(it // 30, len(delta_seq)-1)]
        
        d = _lp_tangent_step(h, top_k=tk, delta=dl, reg=1e-4)
        
        if d is None:
            stall += 5
            continue
        
        # Try both directions
        improved = False
        for sign in [1.0, -1.0]:
            a, new_c5 = _line_search(h, sign * d, target, cur_c5, max_alpha=1.0)
            if a > 0 and new_c5 < cur_c5 - 1e-12:
                h = _project(h + a * sign * d, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        
        if not improved:
            stall += 1
    
    return best_h, best_c5

def _pair_polish(h, deadline, target, max_moves=50000):
    """Fine pair-based polish"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    moves = 0
    while time.time() < deadline and moves < max_moves:
        if stall > 2000:
            break
        
        # Random pair
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        if j == k:
            moves += 1
            continue
        if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
            moves += 1
            continue
        
        dj = 1.0 if rng.random() > 0.5 else -1.0
        dk = -dj
        a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
        a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
        a_max = min(a_max_j, a_max_k)
        if a_max < 1e-10:
            moves += 1
            continue
        
        for frac in [0.3, 0.6, 1.0]:
            a = a_max * frac
            h_t = h.copy()
            h_t[j] += dj * a
            h_t[k] += dk * a
            c_t = _exact_c5(h_t)
            if c_t < best_c5 - 1e-13:
                best_c5 = c_t
                best_h = h_t.copy()
                h = h_t.copy()
                stall = 0
                break
        else:
            stall += 1
        moves += 1
    
    return best_h, best_c5

def _subgrad_fallback(h, deadline, target, max_iters=3000):

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _build_grad_matrix(h):
    """Build the gradient matrix G where G[i,j] = d(scores[i])/d(h[j])"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        m = n - d
        if m <= 0:
            continue
        idx_pos = d + n - 1
        G[idx_pos, d:n] += 1.0 - h[0:m]
        G[idx_pos, 0:m] += -h[d:n]
        if d > 0:
            idx_neg = n - 1 - d
            G[idx_neg, 0:m] += 1.0 - h[d:n]
            G[idx_neg, d:n] += -h[0:m]
    return G

    """Solve tangent LP, return direction d or None"""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    n_s = len(scores)
    
    # Select constraints
    if mode == 'top_k':
        k = max(3, int(delta))  # delta doubles as k
        idx_sel = np.argsort(-scores)[:k]
    elif mode == 'threshold':
        thr = float(np.max(scores)) - delta
        idx_sel = np.where(scores >= thr)[0]
        if len(idx_sel) == 0:
            idx_sel = np.argsort(-scores)[:5]
    else:  # 'full'
        idx_sel = np.arange(n_s)
    
    n_c = len(idx_sel)
    G = _build_grad_matrix(h)
    
    # LP: min reg*sum(|d|) + t
    # vars: d0+, d0-, ..., d[n-1]+, d[n-1]-, t
    n_vars = 2 * n + 1
    c = np.zeros(n_vars)
    c[:2*n:2] = reg
