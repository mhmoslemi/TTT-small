            # If ds[d] < 0: t >= (scores0[d] - M) / (-ds[d])
            # If ds[d] = 0: need M >= scores0[d]
            
            # Binary search on M:
            lo_t = 0.0
            hi_t = 1.0
            
            # Find optimal t by ternary search on the convex function
            def objective(t):
                return float(np.max(scores0 + t * ds))
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - gr * (hi_t - lo_t)
            c2 = lo_t + gr * (hi_t - lo_t)

            f2 = objective(c2)

                    hi_t = c2
                    f2 = f1
                    c2 = c1
                    c1 = hi_t - gr * (hi_t - lo_t)
                    f1 = objective(c1)
                else:
                    lo_t = c1
                    f1 = f2
                    c1 = c2
                    c2 = lo_t + gr * (hi_t - lo_t)
                    f2 = objective(c2)
            
            t_opt = (lo_t + hi_t) / 2.0
            new_score = objective(t_opt)
            
            # But we also need to maintain the sum constraint!
            # If we change h[j] from h_save to t_opt, the sum changes by (t_opt - h_save)
            # We need to adjust...

            # Option 2: adjust one other coordinate to compensate
            # Option 3: project after the change
            
            if new_score < best_c5 - 1e-14:

                h = _project(h, target)  # rebalance

                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True

    return best_h, best_c5
