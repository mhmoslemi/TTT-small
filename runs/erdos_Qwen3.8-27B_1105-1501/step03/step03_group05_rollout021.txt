```python
def _slsqp_minimax(h, deadline):
    """SLSQP on the minimax formulation: min t s.t. t >= s_k for all k, sum(h)=n/2, 0<=h<=1"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    target = n / 2.0
    
    def objective(x):
        return x[0]  # minimize t
    
    def constraint_scores(x):
        t = x[0]
        h_vec = x[1:]
        s = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
        return t - s  # must be >= 0
    
    def constraint_sum(x):
        return np.sum(x[1:]) - target
    
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': constraint_scores},
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    x0 = np.concatenate([[float(_exact_c5(h))], h])
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project_box_sum(h_new, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)
```

```python
        if delta_max - delta_min < 1e-14:
            continue
        
        Lc = dx * (G[:, i] - G[:, j])
        di = j - i
        r1 = n - 1 + di
        r2 = n - 1 - di  # = n-1+(i-j)
        
        def obj(delta):
            s = base + delta * Lc
            s[r1] += delta * delta * dx
            s[r2] += delta * delta * dx
            return float(np.max(s))
        
        # Golden section
        gr = 0.6180339887498949
        lo = delta_min
        hi = delta_max
        c = lo + (1.0 - gr) * (hi - lo)
        dpt = lo + gr * (hi - lo)
        fc = obj(c)
        fd = obj(dpt)
        for _ in range(45):
            if hi - lo < 1e-13:
                break
            if fc < fd:
                hi = dpt; dpt = c; fd = fc
                c = lo + (1.0 - gr) * (hi - lo); fc = obj(c)
            else:
                lo = c; c = dpt; fc = fd
                dpt = lo + gr * (hi - lo); fd = obj(dpt)
        
        best_delta = (lo + hi) / 2.0
        best_val = obj(best_delta)
        
        if best_val < cur_c5 - 1e-13:
            h[i] += best_delta
            h[j] -= best_delta
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            base = _scores_scaled(h)
            G = _build_G(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5
```

This is much faster because:
1. G is precomputed (O(n²) per accepted step)
2. The per-pair work is O(n) for computing Lc and O(n) for each obj evaluation
3. Golden section does ~45 evaluations

The main cost per accepted step is rebuilding base (O(n²) for correlate) and G (O(n²)). Between accepted steps, each rejected pair costs O(n) for Lc + 45*O(n) for golden section ≈ 46n operations.

For n=64: 46*64 ≈ 3000 operations per rejected pair. With, say, 100 rejects between accepts: 300,000 ops. That's very fast.
