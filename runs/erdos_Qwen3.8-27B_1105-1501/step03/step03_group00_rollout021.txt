    dl = t0 + budget_s * 0.72
    n_values = [50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 80, 84, 88, 96]
    # Prioritize n near current best
    n_values.sort(key=lambda x: abs(x - best_n))
    for n_try in n_values:
        if time.time() > dl:
            break
        if n_try == best_n:
            continue
        if n_try < 30 or n_try > 120:
            continue
        h_r = _resample(best_h, n_try)
        c5_r0 = _exact_c5(h_r)
        if c5_r0 > best_c5 + 0.005:
            continue  # too far off, skip
        sub_dl = min(dl, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[10, 5, None, 3], radius_init=0.1, max_iters=400)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Vertex proposals + LP from vertices (10% time)
    tl = t0 + budget_s * 0.72
    dl = t0 + budget_s * 0.82
    for attempt in range(30):
        if time.time() > dl or time.time() < tl:
            if time.time() > dl:
                break
        h_v, c5_v = _vertex_propose(best_h, _scores(best_h), 5, rng)
        if c5_v > best_c5 + 0.002:
            continue
        sub_dl = min(dl, time.time() + budget_s * 0.012)
        h_v, c5_v = _lp_optimize(h_v, sub_dl, top_k_list=[5, None, 3], radius_init=0.08, max_iters=300)
        if c5_v < best_c5 - 1e-12:
            best_c5 = c5_v
            best_h = h_v.copy()

    # Phase 7: Perturbation restarts (10% time)
    dl = t0 + budget_s * 0.92
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[8, 5, 3, None], radius_init=0.06, max_iters=300)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.1)

    # Phase 8: Pair polish (5% time)
    if time.time() < t0 + budget_s * 0.96:
        dl = t0 + budget_s * 0.96
        hp, c5p = _pair_polish(best_h, dl, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final ultra-fine LP (remaining time)
    dl = t0 + budget_s * 0.995
    h_f, c5_f = _lp_optimize(best_h, dl, top_k_list=[3, 2, 5, 4, None], radius_init=0.012, max_iters=8000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
