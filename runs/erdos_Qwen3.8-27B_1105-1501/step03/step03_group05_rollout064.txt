import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    diff = target - float(np.sum(v))
    if abs(diff) < 1e-14:
        return v
    # Water-filling approach
    if diff > 0:
        # Need to add mass: prioritize coordinates with most room
        room = 1.0 - v
        idx = np.argsort(-room)  # most room first
        per = diff / n
        for i in idx:
            add = min(per, room[i])
            v[i] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
        # If still have diff (shouldn't happen if target <= n)
        if diff > 1e-14:
            # Distribute remaining proportionally
            room2 = 1.0 - v
            mask2 = room2 > 1e-14
            if np.sum(mask2) > 0:
                v[mask2] += diff / np.sum(mask2)
                v = np.clip(v, 0.0, 1.0)
    else:
        # Need to remove mass: prioritize coordinates with least value
        diff = -diff
        idx = np.argsort(v)  # least value first
        per = diff / n
        for i in idx:
            rem = min(per, v[i])
            v[i] -= rem
            diff -= rem
            if abs(diff) < 1e-14:
                break
        if diff > 1e-14:
            mask2 = v > 1e-14
            if np.sum(mask2) > 0:
                v[mask2] -= diff / np.sum(mask2)
                v = np.clip(v, 0.0, 1.0)
    # Final exact fix
    residual = target - float(np.sum(v))
    if abs(residual) > 1e-15:
        if residual > 0:
            mask = v < 1.0 - 1e-14
        else:
            mask = v > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += residual / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _build_gradient(h):
    """Build gradient matrix G and scores F for the objective."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return F, G

def _trust_region_step(h, sigma, deadline):
    """Solve the convex trust-region minimax subproblem."""
    n = len(h)
    target = n / 2.0
    F, G = _build_gradient(h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    rho = sigma * np.sqrt(n)
    
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h, 2) <= rho,
        t >= 0,
        G @ (x - h) + F <= t
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try solvers in order
    solvers = ['CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'OSQP', 'GLPK', 'CBC']
    result = None
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if x.value is not None:
                    result = np.array(x.value).flatten()
                    break
        except Exception:
            continue
    
    if result is None:
        return None
    
    return result

def _refine_trust(h, deadline, sigma_start=0.04, min_sigma=0.003, max_sigma=0.25):
    """Iteratively apply trust-region steps."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sigma = sigma_start
    stall = 0
    
    while time.time() < deadline:
        if stall > 40:
            break
        result = _trust_region_step(h, sigma, deadline)
        if result is None:
            stall += 5
            sigma *= 0.6
            if sigma < min_sigma:
                break
            continue
        
        # Try different step sizes
        best_trial_c5 = cur_c5
        best_trial_h = None
        for factor in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
            trial = h + factor * (result - h)
            trial = _project_box_sum(trial, target)
            c5_t = _exact_c5(trial)
            if c5_t < best_trial_c5 - 1e-12:
                best_trial_c5 = c5_t
                best_trial_h = trial.copy()
        
        if best_trial_h is not None:
            h = best_trial_h
            cur_c5 = best_trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            sigma = min(sigma * 1.15, max_sigma)
        else:
            stall += 1
            sigma = max(sigma * 0.6, min_sigma)
    
    return best_h, best_c5

def _lp_refine(h, deadline, radius_init=0.1, max_iters=2000):
    """LP-based direction finding and line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        scores = _scores(h) * (2.0/n)
        # Build the LP for a descent direction
        # minimize t s.t. G*(x-h) <= t - scores, ||x-h||_inf <= radius
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            low = max(0, -d)
            high = min(n, n - d)
            for p in range(low, high):
                q = p + d
                G[r, q] += (1.0 - h[p])
                G[r, p] -= h[q]
        
        # LP: minimize t s.t. G*(x-h) - t <= -scores, |x_i - h_i| <= radius
        # Variables: x[0..n-1], t
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.zeros((4*(2*n-1) + 2*n, n+1))
        b_ub = np.zeros(4*(2*n-1) + 2*n)
        row = 0
        # G*(x-h) - t <= -scores
        A_ub[:2*n-1, :n] = G
        A_ub[:2*n-1, n] = -1.0
        b_ub[:2*n-1] = -scores
        row = 2*n-1
        for i in range(n):
            # x_i - h_i <= radius
            A_ub[row, i] = 1.0
            b_ub[row] = radius + h[i]
            row += 1
            # -(x_i - h_i) <= radius
            A_ub[row, i] = -1.0
            b_ub[row] = radius + h[i]
            row += 1
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        bounds = [(h[i] - radius, h[i] + radius) for i in range(n)]
        bounds.append((None, None))
        # Clip bounds
        bounds = [((max(0, lo)), min(1, hi)) for lo, hi in bounds[:-1]] + [bounds[-1]]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'maxiter': 3000})
            if res.success:
                direction = res.x[:n]
            else:
                stall += 1
                radius *= 0.7
                if radius < 1e-5:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        
        if float(np.max(np.abs(direction))) < 1e-10:
            stall += 1
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        
        # Line search along direction
        best_a = 0.0
        best_c5_trial = cur_c5
        alpha_max = 1.0
        for j in range(n):
            if direction[j] > 1e-10:
                alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
            elif direction[j] < -1e-10:
                alpha_max = min(alpha_max, h[j] / (-direction[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        for a in np.linspace(0.05, min(1.0, alpha_max), 10):
            trial = h + a * direction
            trial = _project_box_sum(trial, target)
            c5_t = _exact_c5(trial)
            if c5_t < best_c5_trial - 1e-12:
                best_c5_trial = c5_t
                best_a = a

            h = h + best_a * direction
            h = _project_box_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _binary_search_local(h, deadline, max_iters=50000, rng=None):
    """Binary local search: start from binary approximation, then swap optimally."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    target = n // 2
    # Start from best binary approximation
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        if not ones or not zeros:
            break
        best_swap = None
        best_val = cur_c5
        batch = min(50, len(ones) * len(zeros))
        for _b in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    """Resample h from n to n_new using piecewise constant interpolation."""
    n = len(h)
    if n_new == n:
        return h.copy()
    # Cell-centered interpolation
    x_old = (np.arange(n) + 0.5) * (2.0/n)
    x_new = (np.arange(n_new) + 0.5) * (2.0/n_new)
    h_new = np.interp(x_new, x_old, h, left=0, right=0)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_transfer(h, n_moves, rng):
    """Apply random mass-preserving transfers."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _ in range(n_moves):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        h[i] -= delta
        h[j] += delta
    return _project_box_sum(h, target)

def _rotate(h, shift):
    """Circularly rotate h."""
    return np.roll(h, shift)

def _reflect(h):
    """Reverse h."""
    return h[::-1].copy()

def _mix(h1, h2, alpha):
    """Mix two feasible solutions (preserves mass if both have same sum)."""
    m = alpha * h1 + (1.0 - alpha) * h2
    target = len(m) / 2.0
    return _project_box_sum(m, target)
