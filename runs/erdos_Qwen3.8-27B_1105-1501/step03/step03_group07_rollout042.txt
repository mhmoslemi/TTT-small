                need -= amt

        total = float(np.sum(y))
        new = target - (total - y[idx])
        if 0.0 <= new <= 1.0:
            y[idx] = new
    return y

def _lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

def _softmax_grad(h, scores, T):
    n=len(h)

    ws=float(np.sum(w))

        return _lag_gradient(h, r-(n-1))
    w = w/ws
    grad=np.zeros(n)
    for r in range(len(scores)):
        wr=float(w[r])
        if wr > 1e-12:
            d=r-(n-1)
            grad += wr * _lag_gradient(h,d)
    return grad

def _pair_polish(h, deadline, target, rng, passes=8, top_count=10, random_pairs=300):
    h=np.asarray(h,dtype=float).copy()
    best=_exact_c5(h)

    for _p in range(passes):
       if time.time() > deadline: break
       improved=False

def _lp_tangent(h, scores, selected, R, lam):

    sel=np.asarray(selected,dtype=int)
    m=len(sel)
    if m <= 0: return None
    cols=1+2*n
    A_ub=np.zeros((m+1, cols))

       d=r-(n-1)
       g=_lag_gradient(h,d)
       A_ub[idx,0] = -1.0

          bounds.append((0.0,0.0)); bounds.append((0.0,None))

_PAIR_FRACS = (0.25, 0.50, 0.75, 1.00)


def _exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def _scores(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full')


def _lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad


def _softmax_grad(h, scores, T):
    n = len(h)
    if T <= 0.0:
        T = 0.1
    mx = float(np.max(scores))
    w = np.exp((scores - mx) / T)
    ws = float(np.sum(w))
    if ws < 1e-300:
        r = int(np.argmax(scores))
        return _lag_gradient(h, r - (n - 1))
    w = w / ws
    grad = np.zeros(n, dtype=float)
    for r in range(len(scores)):
        wr = float(w[r])
        if wr > 1e-12:
            d = r - (n - 1)
            grad += wr * _lag_gradient(h, d)
    return grad


def _adjust_sum(y, target):
    n = len(y)
    if n == 0:
        return y
    diff = target - float(np.sum(y))
    if abs(diff) <= 1e-13:
        return y

    if diff > 0:
        slack = 1.0 - y
        total = float(np.sum(slack))
        if total > 1e-15:
            y = y + (diff / total) * slack
            np.clip(y, 0.0, 1.0, out=y)
    else:
        total = float(np.sum(y))
        if total > 1e-15:
            y = y + (diff / total) * y

        return np.ones(n, dtype=float)

def _soft_optimize(h, deadline, target, iters, T, rng):

    best_h = h.copy()

    v = np.zeros(n, dtype=float)

    alpha = 0.05
    stall = 0

        T = max(0.1, float(np.max(_scores(h))) * 0.05)

        if time.time() > deadline:
            break
        if stall > 120:
            break

        scores = _scores(h)
        grad = _softmax_grad(h, scores, T)
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12:
            break

        d = v - alpha * (grad / gn)
        d -= np.mean(d)

        a, c, vnew = _line_search(h, d, target, best, max_alpha=1.0)
        if c < best - 1e-12:
            h = vnew
            best = c
            best_h = h.copy()
            stall = 0
            beta = min(0.9, beta + 0.002)
            T = max(0.03, T * 0.99)
        else:
            stall += 1
            if stall > 20:
                v *= 0.5
                alpha *= 0.7
                T = max(0.02, T * 0.9)
            if stall > 80:
                h_try = _project(h + rng.normal(0.0, 0.02, n), target)
                c_try = _exact_c5(h_try)
                if c_try < best - 1e-12:
                    h = h_try
                    best = c_try
                    best_h = h.copy()
                    stall = 0
                    T = max(0.1, T * 1.2)
                else:

    return best_h, best


def _lp_configs(n, scores, rng):
    M = len(scores)
    rawmax = float(np.max(scores))
    cfgs = []

    all_sel = np.arange(M, dtype=int)
    for rf in (0.06, 0.15, 0.35):
        R = rf * float(n)
        for lf in (1e-8, 1e-6):
            lam = max(rawmax * lf, 1e-14)
            cfgs.append((all_sel, R, lam))

    for gf in (0.03, 0.10):
        thr = rawmax - rawmax * gf
        sel = np.where(scores >= thr)[0]
        if len(sel) > 4:
            cfgs.append((sel, 0.12 * float(n), max(rawmax * 1e-6, 1e-14)))

    for tk in (10, 25, 50):
        if tk < M:
            sel = np.argsort(-scores)[:tk]
            cfgs.append((sel, 0.18 * float(n), max(rawmax * 1e-6, 1e-14)))

    if M > 30:
        sel = rng.choice(M, size=min(M, 24), replace=False)
        cfgs.append((sel, 0.12 * float(n), max(rawmax * 1e-6, 1e-14)))

    return cfgs

        A_ub[idx, 0] = -1.0
        A_ub[idx, 1:1 + n] = g
        A_ub[idx, 1 + n:1 + 2 * n] = -g
        b_ub[idx] = -float(scores[r])

    rr = m
    A_ub[rr, 1:1 + 2 * n] = 1.0
    b_ub[rr] = R

    c = np.zeros(cols, dtype=float)
    c[0] = 1.0
    c[1:1 + 2 * n] = lam

    A_eq = np.zeros((1, cols), dtype=float)
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:1 + 2 * n] = -1.0
    b_eq = np.zeros(1, dtype=float)

    bounds = [(-1e6, 1e6)]
    eps = 1e-9
    for hv in h:
        hv = float(hv)
        if hv <= eps:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif hv >= 1.0 - eps:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))

    res = None
    for method in ('highs', 'revised simplex', 'interior-point'):
        try:
            r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method=method)
            if r is not None and r.success:
                res = r
                break
        except Exception:
            pass

    if res is None:
        return None

    x = res.x
    if x is None or len(x) != cols:
        return None
    if not np.all(np.isfinite(x)):
        return None

    p = x[1:1 + n]
    q = x[1 + n:1 + 2 * n]
    d = p - q
    d -= np.mean(d)
    maxabs = float(np.max(np.abs(d)))
    if maxabs < 1e-10:
        return None
    return d / maxabs


def _lp_optimize(h, deadline, target, cur_c5, rng, max_rounds=20):
    n = len(h)
    if n > 128:
        return np.asarray(h, dtype=float).copy(), cur_c5

    h = np.asarray(h, dtype=float).copy()
    best = cur_c5
    best_h = h.copy()
    rounds = 0

    while time.time() < deadline and rounds < max_rounds:
        scores = _scores(h)
        cfgs = _lp_configs(n, scores, rng)
        improved = False

        for selected, R, lam in cfgs:
            if time.time() > deadline:
                break
            d = _lp_tangent(h, scores, selected, R, lam)
            if d is None:
                continue
            for sgn in (1.0, -1.0):
                if time.time() > deadline:
                    break
                a, c, vnew = _line_search(h, sgn * d, target, best, max_alpha=1.5)
                if c < best - 1e-12:
                    h = vnew
                    best = c
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            grad = _softmax_grad(h, scores, max(0.05, float(np.max(scores)) * 0.03))
            grad -= np.mean(grad)
            if float(np.linalg.norm(grad)) > 1e-12:
                a, c, vnew = _line_search(h, -grad, target, best, max_alpha=0.5)
                if c < best - 1e-12:
                    h = vnew
                    best = c
                    best_h = h.copy()
                    improved = True

        if not improved:
            break
        rounds += 1

    return best_h, best


def _make_perturb(base, kind, rng, strength):
    base = np.asarray(base, dtype=float)
    n = len(base)
    target = n / 2.0

    if kind == 'gauss':
        v = base + rng.normal(0.0, strength, n)
    elif kind == 'sparse':
        v = base.copy()
        cnt = max(1, n // 4)
        idx = rng.choice(n, cnt, replace=False)
        v[idx] += rng.normal(0.0, 2.0 * strength, cnt)
    elif kind == 'binary':
        v = (base + rng.normal(0.0, strength, n) > 0.5).astype(float)
    elif kind == 'roll':
        sh = int(rng.integers(1, max(2, n // 4 + 1)))
        v = np.roll(base, sh)
    elif kind == 'mix':
        v = 0.5 * base + 0.5 * (rng.random(n) > 0.5).astype(float)
    else:
        v = base.copy()
        if n % 2 == 0 and n >= 4:
            half = n // 2
            u = 0.5 * (base[:half] + base[half:])
            if kind == 'periodic':
                v = np.concatenate((u, u))
            elif kind == 'antiperiodic':
                v = np.concatenate((u, 1.0 - u))

    return _project(v, target)


def _multi_start(base, base_c, deadline, rng, max_starts=20):
    base = np.asarray(base, dtype=float)
    n = len(base)
    target = n / 2.0

    best_h = base.copy()
    best_c = base_c
    kinds = ('gauss', 'sparse', 'binary', 'roll', 'mix', 'periodic', 'antiperiodic')

    for s in range(max_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        alloc = min(remaining * 0.9 / max(1.0, float(max_starts - s)), 20.0)
        if alloc < 4.0:
            break
        sub = time.time() + alloc

        kind = kinds[s % len(kinds)]
        strength = 0.02 + 0.03 * float(rng.random())
        h_try = _make_perturb(best_h, kind, rng, strength)

        h_opt, c_opt = _soft_optimize(h_try, sub, target, 400, None, rng)

        if c_opt < best_c + 5e-5 and len(h_opt) <= 96:
            h_opt, c_opt = _lp_optimize(h_opt, min(sub, time.time() + 4.0),
                                        target, c_opt, rng, max_rounds=2)

        h_opt, c_opt = _pair_polish(h_opt, min(sub, time.time() + 3.0),
                                    target, rng, passes=2, top_count=6, random_pairs=80)

        if c_opt < best_c - 1e-12:
            best_h = h_opt.copy()
            best_c = c_opt

    return best_h, best_c


def _resample(h, n_new):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == n_new:
        return _project(h, n_new / 2.0)

    x_old = (np.arange(n, dtype=float) + 0.5) / float(n)
    x_new = (np.arange(n_new, dtype=float) + 0.5) / float(n_new)
    v = np.interp(x_new, x_old, h)
    return _project(v, n_new / 2.0)


def _resolution_sweep(base, base_n, best_c, deadline, rng):
    base = np.asarray(base, dtype=float)
    best_h = base.copy()
    best_n = base_n
    best = best_c

    all_m = (16, 24, 32, 40, 48, 56, 64, 72, 80, 88, 96, 104, 112, 120, 128)
    offs = (-24, -16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32)

    cand_list = []
    for m in all_m:
        if m != best_n and 14 <= m <= 160 and m % 2 == 0:
            cand_list.append((abs(m - best_n), m, m))

    for off in offs:
        m = best_n + off
        if m != best_n and 14 <= m <= 160 and m % 2 == 0:
            cand_list.append((abs(m - best_n), m, m))

    cand_list.sort()
    seen = set()
    tested = 0

    for _dist, m, _mn in cand_list:
        if m in seen:
            continue
        seen.add(m)

        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        alloc = min(remaining * 0.18, 18.0)
        if alloc < 6.0:
            break
        sub = time.time() + alloc

        h_m = _resample(best_h, m)
        c_m = _exact_c5(h_m)

        h_m, c_m = _soft_optimize(h_m, sub, m / 2.0, 700, None, rng)

        if len(h_m) <= 100:
            h_m, c_m = _lp_optimize(h_m, min(sub, time.time() + 4.0),
                                    m / 2.0, c_m, rng, max_rounds=2)

        h_m, c_m = _pair_polish(h_m, min(sub, time.time() + 3.0),
                                m / 2.0, rng, passes=2, top_count=6, random_pairs=80)

        h_m = _box_adjust(h_m, m / 2.0)
        c_m = _exact_c5(h_m)

        if c_m < best - 1e-12:
            best_h = h_m.copy()
            best_n = m
            best = c_m
        elif m < best_n and abs(c_m - best) <= 1e-12:
            best_h = h_m.copy()
            best_n = m
            best = c_m

        tested += 1
        if tested >= 10:
            break

    return best_h, best_n, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(seed)

    try:
        h = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h = None

    if h is None or h.size < 4:
        n0 = 64
        h = np.full(n0, 0.5, dtype=float)
    else:
        n0 = int(h.size)

    target = n0 / 2.0
    h = _project(h, target)

    fallback_h = h.copy()
    fallback_c = _exact_c5(h)

    best_h = h.copy()
    best_n = n0
    best_c = fallback_c

    if budget_s > 20.0:
        if budget_s > 60.0:
            dl = min(deadline, t0 + max(3.0, budget_s * 0.02))
            hq, cq = _pair_polish(best_h, dl, target, rng,
                                  passes=2, top_count=8, random_pairs=120)
            if cq < best_c - 1e-12:
                best_h = hq.copy()
                best_c = cq

        dl = min(deadline, t0 + budget_s * 0.12)
        best_h, best_c = _soft_optimize(best_h, dl, target, 3000, None, rng)

        if best_n <= 128:
            dl = min(deadline, t0 + budget_s * 0.40)
            best_h, best_c = _lp_optimize(best_h, dl, target, best_c, rng, max_rounds=40)

        dl = min(deadline, t0 + budget_s * 0.50)
        best_h, best_c = _pair_polish(best_h, dl, target, rng,
                                      passes=5, top_count=10, random_pairs=250)

        if deadline - time.time() > 30.0:
            dl = min(deadline, t0 + budget_s * 0.75)
            best_h, best_c = _multi_start(best_h, best_c, dl, rng, max_starts=18)

        if deadline - time.time() > 30.0:
            dl = min(deadline, t0 + budget_s * 0.92)
            best_h, best_n, best_c = _resolution_sweep(best_h, best_n, best_c, dl, rng)

        if deadline - time.time() > 20.0 and best_n <= 100:
            dl = min(deadline, t0 + budget_s * 0.97)
            best_h, best_c = _lp_optimize(best_h, dl, best_n / 2.0, best_c, rng, max_rounds=5)

        if deadline - time.time() > 10.0:
            dl = min(deadline, t0 + budget_s * 0.995)
            best_h, best_c = _pair_polish(best_h, dl, best_n / 2.0, rng,
                                          passes=3, top_count=8, random_pairs=150)

    saved_h = best_h.copy()
    saved_c = best_c

    cand = _project(best_h, best_n / 2.0)
    cand_c = _exact_c5(cand)

    if np.isfinite(cand_c) and cand_c <= saved_c + 1e-9:
        best_h = cand
        final_c = cand_c
    else:
        best_h = saved_h
        final_c = saved_c

    if (not np.isfinite(final_c)) or final_c > fallback_c + 1e-7:
        best_h = fallback_h
        final_c = fallback_c

    return best_h, float(final_c), int(best_n)
