import numpy as np
import time

_MIN_TOL = 1e-12


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(np.asarray(h, dtype=float), 1.0 - np.asarray(h, dtype=float), mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    c = np.clip(v, 0.0, 1.0)
    s = float(np.sum(c))
    if s <= target:
        # need more mass
        rem = target - s
        mask = c < 1.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            add = min(rem / cnt, 1.0 - c[mask])
            c[mask] = np.minimum(1.0, c[mask] + add)
    else:
        rem = s - target
        mask = c > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            sub = min(rem / cnt, c[mask])
            c[mask] = np.maximum(0.0, c[mask] - sub)
    # bisection-finish for exactness
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(x)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-11:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        mlo = max(0, -k)
        mhi = min(n - 1, n - 1 - k)
        if mlo > mhi:
            continue
        m = np.arange(mlo, mhi + 1)
        aidx = m + k
        G[k, aidx] += (1.0 - h[m])
        G[k, m] += -h[aidx]
    return G


def _line_f(h, D, dx):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = np.correlate(D, D, mode='full')
    return S, B, Q, dx


def _f_scalar(t, S, B, Q, dx):
    return float(dx * np.max(S + t * B - (t * t) * Q))


def _f_vec(T, S, B, Q, dx):
    T = np.asarray(T, dtype=float)
    val = S[None, :] + T[:, None] * B[None, :] - (T[None, :] ** 2) * Q[None, :]
    return dx * val.max(axis=1)


def _feasible_interval(h, D):
    tlo = 0.0
    thi = 0.0
    for i in range(len(h)):
        d = D[i]
        if d > 1e-12:
            tlo = max(tlo, -h[i] / d)
            thi = min(thi, (1.0 - h[i]) / d)
        elif d < -1e-12:
            tlo = max(tlo, (1.0 - h[i]) / d)
            thi = min(thi, -h[i] / d)
    if not (np.isfinite(tlo) and np.isfinite(thi)) or tlo >= thi:
        return None
    return tlo, thi


def _golden_min(f, a, b, iters=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    m = 0.5 * (a + b)
    return m, f(m)

def _dir_soft(h, G, tau):
    scores = np.correlate(h, 1.0 - h, mode='full')
    sms = scores - float(scores.max())
    w = np.exp(sms / max(tau, 1e-6))
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        return None
    w = w / ws
    g = G.T @ w
    D = -(g - float(np.mean(g)))
    if float(np.max(np.abs(D))) < 1e-13:
        return None
    return D


def _dir_block(rng, n):
    if n < 4:
        return None
    m = int(rng.integers(2, max(3, n // 4)))
    m = min(m, n // 2 - 1)
    idx = rng.permutation(n)
    A = idx[:m]
    Bset = idx[m:2 * m]
    D = np.zeros(n)
    D[A] = 1.0
    D[Bset] = -1.0
    return D


def _dir_pair(rng, n):
    if n < 2:
        return None
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return None
    D = np.zeros(n)
    D[i] = 1.0
    D[j] = -1.0
    return D


def _dir_sine(rng, n):
    x = np.linspace(0.0, 1.0, n)
    D = np.zeros(n)
    for _ in range(int(rng.integers(1, 3))):
        f = int(rng.integers(1, 5))
        ph = float(rng.uniform(-np.pi, np.pi))
        D = D + np.sin(2.0 * np.pi * f * x + ph)
    D = D - float(np.mean(D))
    if float(np.max(np.abs(D))) < 1e-12:
        return None
    return D

    return _project(np.interp(xn, xo, h), n_new / 2.0)

            G = _grad_matrix(h)
            if c5n < best_c5 - 1e-13:
                best_c5 = c5n
                best_h = h.copy()
                best_n = n
        else:
            break

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
