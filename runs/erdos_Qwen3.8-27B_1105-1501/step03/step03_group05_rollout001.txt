def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    # Water-filling to adjust sum to target
    if abs(v.sum() - target) < 1e-12:
        return v
    # Use iterative bisection on threshold
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        alpha_lo = max(alpha_lo, float(np.max(-h[pos] / v[pos])))
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max((1.0 - h[neg]) / v[neg])))
        alpha_hi = min(alpha_hi, float(np.min(-h[neg] / v[neg])))
    return alpha_lo, alpha_hi

def _line_min_max_quad(A, B, C, a_lo, a_hi, n):
    """Find alpha in [a_lo, a_hi] minimizing max_r (A[r] + B[r]*a + C[r]*a²)"""
    candidates = [a_lo, a_hi, 0.0]
    # Stationary points
    for r in range(len(C)):
        if C[r] < -1e-15:
            a = -B[r] / (2.0 * C[r])
            if a_lo - 1e-10 <= a <= a_hi + 1e-10:
                candidates.append(max(a_lo, min(a_hi, a)))
    # Pairwise intersections (for n <= 96, do all pairs)
    m = len(A)
    if m <= 130:  # 2*64-1 = 127
        for i in range(m):
            for j in range(i + 1, m):
                dC = C[i] - C[j]
                dB = B[i] - B[j]
                dA = A[i] - A[j]
                if abs(dC) > 1e-15:
                    disc = dB * dB - 4.0 * dC * dA
                    if disc >= -1e-12:
                        if disc < 0:
                            disc = 0.0
                        sqrt_disc = np.sqrt(disc)
                        a1 = (-dB + sqrt_disc) / (2.0 * dC)
                        a2 = (-dB - sqrt_disc) / (2.0 * dC)
                        for a in [a1, a2]:
                            if a_lo - 1e-10 <= a <= a_hi + 1e-10:
                                candidates.append(max(a_lo, min(a_hi, a)))
                elif abs(dB) > 1e-15:
                    a = -dA / dB
                    if a_lo - 1e-10 <= a <= a_hi + 1e-10:
                        candidates.append(max(a_lo, min(a_hi, a)))
    # Add a grid
    n_grid = 32
    grid = np.linspace(a_lo, a_hi, n_grid)
    candidates.extend(grid.tolist())
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= a_lo) & (candidates <= a_hi)]
    if len(candidates) == 0:
        return 0.0, float(np.max(A))
    
    # Evaluate max of quadratics at each candidate
    # Vectorized: for each candidate a, compute A + B*a + C*a² for all r, take max
    best_val = float('inf')
    best_a = 0.0
    n_cand = len(candidates)
    # Process in chunks to avoid memory issues
    chunk = 500
    for start in range(0, n_cand, chunk):
        end = min(start + chunk, n_cand)
        a_chunk = candidates[start:end]
        # vals shape: (chunk_size, m)
        vals = A[None, :] + a_chunk[:, None] * B[None, :] + a_chunk[:, None]**2 * C[None, :]
        max_vals = vals.max(axis=1)
        idx = int(np.argmin(max_vals))
        if max_vals[idx] < best_val:
            best_val = float(max_vals[idx])
            best_a = float(a_chunk[idx])
    return best_a, best_val

def _exact_line_search(h, v, cur_c5):
    """Exact line search along direction v from point h"""
    n = len(h)
    # Make v zero-sum
    v = v - np.mean(v)
    v_norm = float(np.max(np.abs(v)))
    if v_norm < 1e-14:
        return None, cur_c5
    v = v / v_norm
    
    A = np.correlate(h, 1.0 - h, mode='full').astype(float)
    B = np.correlate(v, 1.0 - h, mode='full').astype(float) - np.correlate(h, v, mode='full').astype(float)
    C = -np.correlate(v, v, mode='full').astype(float)
    
    a_lo, a_hi = _feasible_interval(h, v)
    if a_hi - a_lo < 1e-12:
        return None, cur_c5
    
    best_a, best_raw = _line_min_max_quad(A, B, C, a_lo, a_hi, n)
    
    if best_a == 0.0:
        return None, cur_c5
    
    h_new = h + best_a * v
    h_new = np.clip(h_new, 0.0, 1.0)
    new_c5 = _exact_c5(h_new)
    
    if new_c5 < cur_c5 - 1e-14:
        return h_new, new_c5
    return None, cur_c5

def _active_gradient_direction(h, G, beta=20.0):
    """Generate a descent direction based on active scores"""
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    g = G.T @ w
    v = -g
    v -= np.mean(v)
    norm = float(np.max(np.abs(v)))
    if norm < 1e-14:
        return None
    return v / norm

def _pair_direction(rng, n):
    """Generate a random pair direction e_i - e_j"""
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    while j == i:
        j = int(rng.integers(n))
    v = np.zeros(n)
    v[i] = 1.0
    v[j] = -1.0
    return v

def _random_direction(rng, n):
    v = rng.standard_normal(n)
    v -= np.mean(v)
    norm = float(np.max(np.abs(v)))
    if norm < 1e-14:
        return None
    return v / norm

def _local_line_descent(h, deadline, rng, n_directions_per_round=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    max_stall = 300
    
    while time.time() < deadline and stall < max_stall:
        G = _build_G(h)
        improved = False
        
        # Try active gradient directions with different betas
        for beta in [5.0, 20.0, 80.0, 200.0]:
            if time.time() > deadline:
                break
            v = _active_gradient_direction(h, G, beta)
            if v is not None:
                h_new, c5_new = _exact_line_search(h, v, cur_c5)
                if h_new is not None:
                    h_new = _project_box_sum(h_new, target)
                    h = h_new
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        
        # Try random directions
        if not improved:
            for _ in range(n_directions_per_round):
                if time.time() > deadline:
                    break
                v = _random_direction(rng, n)
                if v is None:
                    continue
                h_new, c5_new = _exact_line_search(h, v, cur_c5)
                if h_new is not None:
                    h_new = _project_box_sum(h_new, target)
                    h = h_new
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        
        # Try pair directions
        if not improved:
            for _ in range(n_directions_per_round):
                if time.time() > deadline:
                    break
                v = _pair_direction(rng, n)
                h_new, c5_new = _exact_line_search(h, v, cur_c5)
                if h_new is not None:
                    h_new = _project_box_sum(h_new, target)
                    h = h_new
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _flip(h):
    return h[::-1].copy()

def _roll(h, k):
    return np.roll(h, k).copy()

def _threshold_bin(h, n):
    """Create a binary version with exactly n//2 ones"""
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin
