def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]

def _smooth_max_grad(h, beta):
    """Compute gradient of smooth max of correlation scores"""
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    # Shift for numerical stability
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    # Gradient: sum_k w_k * grad(s_k)/grad_h
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if w[r] > 1e-15:
            grad += w[r] * _score_grad(h, d)
    return grad

def _smooth_max_val(h, beta):
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    val = max_s + np.log(float(np.sum(np.exp(beta * (scores - max_s))))) / beta
    return val

def _antiperiodic_project(h):
    """Project h onto anti-periodic subspace: h[i] + h[i+n/2] = 1"""
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    half = n // 2
    y = (h[:half] + 1.0 - h[half:]) / 2.0
    # y is the "first half" of the anti-periodic vector
    out = np.empty(n)
    out[:half] = y
    out[half:] = 1.0 - y
    return out

def _antiperiodic_from_y(y):
    """Construct anti-periodic vector from first half"""
    half = len(y)
    out = np.empty(2 * half)
    out[:half] = y
    out[half:] = 1.0 - y
    return out

def _slsqp_solve(h_init, deadline, top_k=30, maxiter=100, ftol=1e-11):

        return x[n]

        g = np.zeros(n + 1)

        return g

        'type': 'eq',

        })

            raise RuntimeError("timeout")

    except RuntimeError:
        return None

        return None

    c5 = _exact_c5(h_out)
    return h_out, c5

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _smooth_max_grad(h, beta):
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if w[r] > 1e-15:
            grad += w[r] * _score_grad(h, d)
    return grad

    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _slsqp_refine(h_init, deadline, top_k=20, maxiter=200, ftol=1e-13):
    n = len(h_init)
    target = n / 2.0
    for _refine in range(5):
        if time.time() > deadline:
            break
        scores = _all_scores(h_init)
        z0 = float(np.max(scores)) + 1e-6
        all_idx = np.argsort(scores)[::-1][:top_k]
        active_lags = [int(k - (n - 1)) for k in all_idx]
        x0 = np.concatenate([h_init, [z0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
        def obj_s(x):
            return x[n]
        def obj_jac_s(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        cons = [{'type': 'eq',
                 'fun': lambda x: np.sum(x[:n]) - target,
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        for d in active_lags:
            cons.append({
                'type': 'ineq',
                'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
                'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
            })
        def _cb2(xk):
            if time.time() > deadline:
                raise RuntimeError("timeout")
        try:
            result = minimize(obj_s, x0, method='SLSQP', jac=obj_jac_s,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                              callback=_cb2)
        except RuntimeError:
            return h_init, _exact_c5(h_init)
        except Exception:
            return h_init, _exact_c5(h_init)
        h_new = np.clip(result.x[:n], 0.0, 1.0)
        if abs(np.sum(h_new) - target) > 1e-4:
            h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h_init)
        if c5_new < c5_cur - 1e-15:
            h_init = h_new
        else:
            break
        top_k = max(5, top_k - 4)
    return h_init, _exact_c5(h_init)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _two_coord_search(h, deadline, n_trials=400):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_val = best_c5
        di = np.linspace(-0.04, 0.04, 9)
        dj = np.linspace(-0.04, 0.04, 9)
        for a in di:
            if time.time() > deadline:
                break
            for b in dj:
                h_try = best_h.copy()
                h_try[i] = float(np.clip(best_h[i] + a, 0.0, 1.0))
                h_try[j] = float(np.clip(best_h[j] + b, 0.0, 1.0))
                h_try = _project(h_try, target)
                val = _exact_c5(h_try)
                if val < cur_val - 1e-14:
                    cur_val = val
                    best_h = h_try.copy()
    return best_h, _exact_c5(best_h)

def _multistart_optimize(h_cur, c5_cur, n, deadline, rng):
    """Try multiple perturbations and return best"""
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for attempt in range(20):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 8.0:
            break
        # Vary perturbation scale
        scale = 0.02 + 0.06 * rng.random()
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        # Smooth max descent
        dm = time.time() + min(10.0, rem * 0.4)
        h_try, c5_try = _smooth_max_descent(h_pert, dm, n_steps=200, beta_start=10.0, beta_end=80.0)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            # Follow with SLSQP
            if time.time() < deadline - 8.0:
                result = _slsqp_solve(best_h, min(deadline - 4.0, time.time() + 15.0),
                                      top_k=25, maxiter=80, ftol=1e-11)
                if result is not None and result[1] < best_c5:
                    best_c5 = result[1]
                    best_h = result[0].copy()
        # Also try trust-constr from perturbed
        if time.time() < deadline - 10.0 and c5_try >= best_c5 - 1e-12:
            result_tc = _trust_constr_solve(h_pert, min(deadline - 4.0, time.time() + 12.0),
                                            top_k=25, maxiter=100)
            if result_tc is not None and result_tc[1] < best_c5 - 1e-12:
                best_c5 = result_tc[1]
                best_h = result_tc[0].copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial optimization (15% budget)
    tl1 = t0 + budget_s * 0.15
    h1, c5_1 = _smooth_max_descent(best_h, tl1, n_steps=500, beta_start=5.0, beta_end=150.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    if time.time() < t0 + budget_s * 0.18:
        result = _slsqp_solve(best_h, t0 + budget_s * 0.20, top_k=40, maxiter=100, ftol=1e-11)
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0].copy()

    # Phase 2: Trust-constr attempt (10% budget)
    if time.time() < t0 + budget_s * 0.30:
        result_tc = _trust_constr_solve(best_h, t0 + budget_s * 0.30, top_k=35, maxiter=200)
        if result_tc is not None and result_tc[1] < best_c5:
            best_c5 = result_tc[1]
            best_h = result_tc[0].copy()

    # Phase 3: Multi-restart (45% budget)
    tl3 = t0 + budget_s * 0.75
    best_h, best_c5 = _multistart_optimize(best_h, best_c5, best_n, tl3, rng)

    # Phase 4: Two-coordinate local search (10% budget)
    tl4 = t0 + budget_s * 0.85
    if time.time() < tl4:
        h_pt, c5_pt = _two_coord_search(best_h, tl4, n_trials=400)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()

    # Phase 5: SLSQP refine with tight active set (8% budget)
    if time.time() < t0 + budget_s * 0.93:
        h_ref, c5_ref = _slsqp_refine(best_h, t0 + budget_s * 0.93, top_k=12, maxiter=250, ftol=1e-14)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Resolution changes (5% budget)
    if time.time() < t0 + budget_s * 0.97:
        dl6 = t0 + budget_s * 0.975
        for dn in [0, -2, 2, -4, 4, -6, 6]:
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 200 or n_try % 2 != 0:
                continue
            if dn == 0:
                h_r = best_h.copy()
            else:
                h_r = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + 12.0)
            h_r2, c5_r2 = _smooth_max_descent(h_r, sub_dl, n_steps=80, beta_start=20.0, beta_end=80.0)
            result = _slsqp_solve(h_r2, min(dl6, time.time() + 8.0), top_k=20, maxiter=50, ftol=1e-11)
            if result is not None:
                if result[1] < best_c5 - 1e-12:
                    best_c5 = result[1]
                    best_h = result[0].copy()
                    best_n = n_try
                h_r = result[0]
                c5_r = result[1]
            else:
                h_r = h_r2
                c5_r = c5_r2
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final polish (3% budget)
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _smooth_max_descent(best_h, t0 + budget_s * 0.995,
                                             n_steps=150, beta_start=80.0, beta_end=500.0)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if time.time() < t0 + budget_s * 0.998:
            result = _slsqp_solve(best_h, t0 + budget_s * 0.999, top_k=8, maxiter=150, ftol=1e-14)
            if result is not None and result[1] < best_c5:
                best_c5 = result[1]
                best_h = result[0].copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
