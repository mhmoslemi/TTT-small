    # G: gradient matrix (n_lags x n)

- `G[r, d:] += (1.0 - h[:m])`: This adds (1-h[0]), (1-h[1]), ..., (1-h[m-1]) to G[r, d], G[r, d+1], ..., G[r, d+m-1]. Since d+m-1 = d+n-d-1 = n-1, this covers j from d to n-1. ✓
- `G[r, :m] += -h[d:]`: This adds -h[d], -h[d+1], ..., -h[d+m-1] = -h[n-1] to G[r, 0], G[r, 1], ..., G[r, m-1]. This covers j from 0 to m-1. ✓

Great, so the gradient matrix is correct.

Now, for the vertex direction approach:
Given weights w on the scores, the composite gradient is g = G.T @ w (shape n).
The vertex that minimizes g^T v subject to sum(v) = n/2, 0 <= v <= 1 is:
- Sort g in ascending order
- Set v_i = 1 for the n/2 coordinates with smallest g_i
- (If n/2 is not integer, set one coordinate to the fractional part)

Since n/2 might not be integer for odd n, I'll handle that.

For the direction p = v - h:
- sum(p) = sum(v) - sum(h) = n/2 - n/2 = 0 ✓
- |p_i| <= 1 for all i ✓
- h + alpha * p is feasible for alpha in [0, 1] (convex combination) ✓

For the line search, I'll evaluate f(alpha) = _exact_c5(h + alpha * p) for alpha in [0, 1].

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_vertex(h, p, alpha_lo=0.0, alpha_hi=1.0):
    """Line search along h + alpha*p. p is a direction (vertex - h)."""
    n = len(h)
    # Determine valid alpha range
    a_lo, a_hi = alpha_lo, alpha_hi
    for j in range(n):
        if p[j] > 1e-14:
            a_hi = min(a_hi, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_lo = max(a_lo, -h[j] / p[j])
    a_lo = max(a_lo, 0.0)
    a_hi = min(a_hi, 1.0)
    if a_hi - a_lo < 1e-12:
        return a_lo, _exact_c5(h)
    # Dense grid
    n_samp = 20
    alphas = np.linspace(a_lo, a_hi, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = np.argmin(vals)
    best_a = alphas[best_i]
    best_c5 = vals[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    # Golden section refinement
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _vertex_from_gradient(g, n):
    """Given gradient g, find the vertex minimizing g^T v subject to sum(v)=n/2, 0<=v<=1."""
    T = n / 2.0
    T_int = int(T)
    T_frac = T - T_int
    order = np.argsort(g)
    v = np.zeros(n)
    v[order[:T_int]] = 1.0
    if T_frac > 1e-12:
        v[order[T_int]] = T_frac
    return v

def _make_weights(scores, k, temperature=None, rng=None):
    """Make convex combination weights for top-k scores."""
    s = scores.copy()
    M = np.max(s)
    if temperature is not None and temperature > 0:
        w = np.exp((s - M) / temperature)
        w[w < 1e-15] = 0
        sw = np.sum(w)
        if sw < 1e-30:
            w = np.zeros(len(s))
            w[np.argmax(s)] = 1.0
        else:
            w /= sw
        return w
    # Top-k uniform
    idx = np.argsort(s)[-k:]
    w = np.zeros(len(s))
    w[idx] = 1.0 / k
    return w

def _vertex_explore(h, deadline, n_vertices=500, k_list=None):
    """Explore vertex directions."""
    if k_list is None:
        k_list = [1, 2, 3, 5, 8, 15, 30]
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    rng = np.random.default_rng(12345)
    scores = _scores(cur_h)
    G = _grad_matrix(cur_h)
    n_done = 0
    while time.time() < deadline and n_done < n_vertices:
        scores = _scores(cur_h)
        G = _grad_matrix(cur_h)
        k = k_list[n_done % len(k_list)]
        temp = 0.01 + 0.1 * (n_done % 5) / 4.0
        w = _make_weights(scores, k, temperature=temp, rng=rng)
        g = G.T @ w
        v = _vertex_from_gradient(g, n)
        p = v - cur_h
        if np.max(np.abs(p)) < 1e-14:
            n_done += 1
            continue
        a, c5_try = _line_search_vertex(cur_h, p)
        if c5_try < cur_c5 - 1e-13:
            cur_h = cur_h + a * p
            cur_h = _project(cur_h, target)
            cur_c5 = _exact_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        n_done += 1
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _subgradient_optimize(h, deadline, n_steps=200):
    """Subgradient method with momentum and adaptive step size."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    momentum = np.zeros(n)
    beta_m = 0.7  # momentum coefficient
    step = 0.05
    rng = np.random.default_rng(777)
    for step_idx in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        # Choose active set with some randomness
        M = np.max(scores)
        # Softmax weights with temperature decreasing over time
        frac = step_idx / max(1, n_steps - 1)
        tau = 0.05 * (1.0 - frac) + 0.005
        w = np.exp((scores - M) / tau)
        w[w < 1e-15] = 0
        sw = np.sum(w)
        if sw < 1e-30:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        g = G.T @ w
        g -= np.mean(g)  # Ensure sum = 0
        # Momentum
        momentum = beta_m * momentum + g
        g_norm = np.linalg.norm(momentum)
        if g_norm < 1e-15:
            break
        g_dir = momentum / g_norm
        # Adaptive step size with backtracking
        alpha = step
        improved = False
        for _ in range(30):
            h_new = _project(h + alpha * g_dir, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                step = min(step * 1.1, 0.15)
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
        if not improved:
            step *= 0.8
        # Occasionally add small random perturbation
        if step_idx % 50 == 49:
            h = _project(h + 0.01 * rng.normal(0, 1, n), target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5
