def _pair_line_search(h, i, j, scores, n):
    """Find optimal transfer a from j to i (h[i]+=a, h[j]-=a) minimizing max score."""
    dx = 2.0 / n
    L = np.zeros(2*n-1)  # linear coefficients for each lag
    Q = np.zeros(2*n-1)  # quadratic coefficients for each lag
    
    # For each lag d (index n-1+d), compute L_d and Q_d
    # Lag d score = sum_p h[p]*(1-h[p+d]) for valid p
    # Change from h[i]+=a, h[j]-=a:
    #   +a*(1-h[p+d]) if p==i and p+d != j
    #   -a*h[p] if p+d==i and p != j
    #   -a*(1-h[p+d]) if p==j and p+d != j  (wait, p==j and p+d != i)
    #   +a*h[p] if p+d==j and p != i
    #   Quadratic: +a*(1-h[j])+a*h[i]+a² if p==i and p+d==j (i.e., j=i+d)
    #              -a*(1-h[i])+a*h[j]+a² if p==j and p+d==i (i.e., i=j+d, i.e., j=i-d)
    
    for d in range(-(n-1), n):
        r = n - 1 + d  # index in scores array
        # Find all valid p: max(0,-d) <= p < min(n, n-d)
        p_lo = max(0, -d)
        p_hi = min(n, n - d)
        
        Ld = 0.0
        Qd = 0.0
        
        for p in range(p_lo, p_hi):
            p2 = p + d  # partner index
            if p2 < 0 or p2 >= n:
                continue
            
            # Check if i or j is involved
            i_in = (p == i) or (p2 == i)
            j_in = (p == j) or (p2 == j)
            
            if not i_in and not j_in:
                continue
            
            if p == i and p2 == j:
                # Both: (h[i]+a)(1-(h[j]-a)) = h[i](1-h[j]) + a(1-h[j]) + a*h[i] + a²
                Ld += (1 - h[j]) + h[i]
                Qd += 1
            elif p == j and p2 == i:
                # Both: (h[j]-a)(1-(h[i]+a)) = h[j](1-h[i]) - a(1-h[i]) - a*h[j] + a²
                Ld += -(1 - h[i]) - h[j]
                Qd += 1
            elif p == i:
                # Only i as p: (h[i]+a)(1-h[p2]) = h[i](1-h[p2]) + a(1-h[p2])
                Ld += (1 - h[p2])
            elif p2 == i:
                # Only i as p2: h[p](1-(h[i]+a)) = h[p](1-h[i]) - a*h[p]
                Ld += -h[p]
            elif p == j:
                # Only j as p: (h[j]-a)(1-h[p2]) = h[j](1-h[p2]) - a(1-h[p2])
                Ld += -(1 - h[p2])
            elif p2 == j:
                # Only j as p2: h[p](1-(h[j]-a)) = h[p](1-h[j]) + a*h[p]
                Ld += h[p]
        
        L[r] = Ld
        Q[r] = Qd
    
    # Now find min of max_r (scores[r] + a*L[r] + a²*Q[r]) * dx over feasible a
    # Feasible: 0 <= h[i]+a, h[j]-a <= 1
    a_min = max(-h[i], 0.0)  # h[i]+a >= 0 => a >= -h[i]; h[j]-a >= 0 => a <= h[j]
    a_max = min(1.0 - h[i], h[j])  # h[i]+a <= 1 => a <= 1-h[i]; h[j]-a <= 1 => a >= h[j]-1
    
    # Actually: h[i]+a >= 0 => a >= -h[i]; h[i]+a <= 1 => a <= 1-h[i]
    #           h[j]-a >= 0 => a <= h[j];    h[j]-a <= 1 => a >= h[j]-1
    a_min = max(-h[i], h[j] - 1.0)
    a_max = min(1.0 - h[i], h[j])
    
    if a_min >= a_max:
        return 0.0, scores  # no feasible transfer
    
    # Golden section search for min of f(a) = max_r (scores[r] + a*L[r] + a²*Q[r])
    # f is convex (max of convex quadratics with Q >= 0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f(a):
        vals = scores + a * L + a * a * Q
        return np.max(vals)
    
    best_a = 0.0
    best_f = f(0.0)
    
    # Check endpoints
    f_lo = f(a_min)
    f_hi = f(a_max)
    if f_lo < best_f:
        best_a = a_min
        best_f = f_lo
    if f_hi < best_f:
        best_a = a_max
        best_f = f_hi
    
    # Golden section
    lo = a_min
    hi = a_max
    if hi - lo < 1e-12:
        return best_a, scores + best_a * L + best_a * best_a * Q
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(60):
        if hi - lo < 1e-10 * max(1.0, abs(hi - lo) + 1e-300):
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    opt_a = (lo + hi) / 2.0
    f_opt = f(opt_a)
    if f_opt < best_f:
        best_a = opt_a
        best_f = f_opt
    
    new_scores = scores + best_a * L + best_a * best_a * Q
    return best_a, new_scores
