                G[r, :m] += oim[g:]
                G[r, g:] += -h[:m]
    return corr, G


def _slsqp_epigraph(h, deadline, maxiter=300):
    n = len(h)
    target = n / 2.0
    K = 2 * n - 1
    arr = np.zeros(n + 1)
    arr[0] = _exact_c5(h) / (2.0 / n)  # raw max
    arr[1:] = h

    def obj(x, *args):
        return x[0]

    def obj_grad(x, *args):
        g = np.zeros(len(x))
        g[0] = 1.0
        return g

    def ineq(x, *args):
        h_v = x[1:]
        c = np.correlate(h_v, 1.0 - h_v, mode='full')
        return x[0] - c

    def ineq_jac(x, *args):
        h_v = x[1:]
        _, G = _corr_n_grad(h_v)
        J = np.concatenate([np.ones((K, 1)), -G], axis=1)
        return J

    def eqc(x, *args):
        return float(np.sum(x[1:])) - target

    def eqc_jac(x, *args):
        J = np.zeros(1 + n)
        J[1:] = 1.0
        return J

    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eqc, 'jac': eqc_jac}
    ]
    bounds = [(0.0, 10.0)] + [(0.0, 1.0)] * n

    try:
        res = minimize(obj, arr, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'iprint': 0})
        if time.time() > deadline:
            return _project(h, target), _exact_c5(h)
        if res.success or res.x[0] < arr[0]:
            h_new = _project(res.x[1:], target)
            c5_new = _exact_c5(h_new)
            # Also try the raw solution before projection
            c5_raw = _exact_c5(np.clip(res.x[1:], 0, 1))
            if c5_raw < c5_new:
                h_new = _project(np.clip(res.x[1:], 0, 1), target)
                c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h.copy(), _exact_c5(h)


def _active_subgrad(h, deadline, tol=0.01, maxiter=500, alpha0=8.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.03

    for it in range(maxiter):
        if time.time() > deadline:
            break
        corr, G = _corr_n_grad(h)
        mx = float(np.max(corr))
        active = corr >= mx - tol * mx
        if not np.any(active):
            active = np.zeros(K_ := len(corr), dtype=bool)
            active[int(np.argmax(corr))] = True
        w = np.zeros(len(corr))
        w[active] = 1.0 / float(np.sum(active))
        grad = G.T @ w
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        # Backtracking line search
        cur_c5 = _exact_c5(h)
        best_a = 0.0
        best_c5_new = cur_c5
        a = step
        for trial in range(25):
            h_t = _project(h + a * grad, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_new:
                best_c5_new = c5_t
                best_a = a
            a *= 0.7
            if a < 1e-8:
                break
        if best_a > 0 and best_c5_new < cur_c5 - 1e-14:
            h = _project(h + best_a * grad, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        else:
            step *= 0.7
            if step < 1e-10:
                break
    return best_h, best_c5


def _topk_weighted_subgrad(h, deadline, k=8, maxiter=200, alpha_start=20.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_start
    step = 0.05

    for it in range(maxiter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(corr))
        # Top-k weighting with softmax
        shifts = corr - mx
        vals = np.clip(alpha * shifts, -200.0, 200.0)
        ew = np.exp(vals)
        w = ew / float(np.sum(ew))
        _, G = _corr_n_grad(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        grad = -grad / gnorm  # negate: we want to decrease the weighted sum
        cur_c5 = _exact_c5(h)
        best_a = 0.0
        best_c5_new = cur_c5
        a = step
        for trial in range(20):
            h_t = _project(h + a * grad, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_new:
                best_c5_new = c5_t
                best_a = a
            a *= 0.6
            if a < 1e-9:
                break
        if best_a > 0 and best_c5_new < cur_c5 - 1e-14:
            h = _project(h + best_a * grad, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            alpha *= 0.98
            step = min(step * 1.05, 0.15)
        else:
            step *= 0.6
            alpha *= 1.2
            if step < 1e-10:
                break
    return best_h, best_c5


def _pair_polish_h(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = [j for j in range(n) if 1e-8 < h[j] < 1.0 - 1e-8]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            j = interior[ii]
            if not (1e-8 < h[j] < 1.0 - 1e-8):
                continue
            for kk in range(ii + 1, len(interior)):
                k = interior[kk]
                if not (1e-8 < h[k] < 1.0 - 1e-8):
                    continue
                for dj in [1.0, -1.0]:
                    dk = -dj
                    a_max = min(h[j] / 1.0 if dj < 0 else (1.0 - h[j]),
                                h[k] / 1.0 if dk < 0 else (1.0 - h[k]))
                    if a_max < 1e-12:
                        continue
                    # Try multiple fractions
                    found = False
                    for frac in [0.5, 1.0, 0.25, 0.75, 0.125, 0.375, 0.625]:
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_lbfgs(h, alpha, maxiter=150):
    n = len(h)
    target = n / 2.0

    def smooth_fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -300.0, 300.0)
        ev = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
        w = ev / float(np.sum(ev))
        _, G = _corr_n_grad(x)
        g = G.T @ w
        g -= np.mean(g)
        return f, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_fun, h, jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_exact(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(12):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.12, max_iters=3000, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        _, G = _corr_n_grad(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP epigraph optimization (30% budget)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _slsqp_epigraph(h, d1, maxiter=200)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Try SLSQP again from a perturbed start
    if time.time() < d1 - 10:
        h_p = best_h + rng.normal(0, 0.02, best_n)
        h_p = _project(h_p, best_n / 2.0)
        h1b, c5_1b = _slsqp_epigraph(h_p, min(d1, time.time() + 60), maxiter=150)
        if c5_1b < best_c5:
            best_c5, best_h = c5_1b, h1b.copy()

    # Phase 2: Top-k weighted subgradient (15% budget)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _topk_weighted_subgrad(best_h, d2, k=6, maxiter=300, alpha_start=15.0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Active-set subgradient
    if time.time() < d2 - 5:
        h2b, c5_2b = _active_subgrad(best_h, min(d2, time.time() + 40),
                                     tol=0.005, maxiter=300, alpha0=5.0)
        if c5_2b < best_c5:
            best_c5, best_h = c5_2b, h2b.copy()

    # Phase 3: L-BFGS smooth from current best (10% budget)
    d3 = t0 + budget_s * 0.55
    for alpha in [20.0, 50.0, 100.0]:
        if time.time() > d3:
            break
        h3, c5_3 = _smooth_lbfgs(best_h, alpha, maxiter=150)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: LP polishing (15% budget)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.12, max_iters=3000, stall_limit=50)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Pair polish (15% budget)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _pair_polish_h(best_h, d5, max_passes=100)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Re-SLSQP from pair-polished result
    if time.time() < t0 + budget_s * 0.90:
        h6, c5_6 = _slsqp_epigraph(best_h, t0 + budget_s * 0.92, maxiter=100)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Resolution search (5% budget)
    d7 = t0 + budget_s * 0.94
    for dn in [1, -1, 2, -2, 3, -3, 4, -4, 8, -8, 16, -16]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        # Quick optimize
        if time.time() < d7 - 10:
            h_r, c5_r = _smooth_lbfgs(h_r, 30.0, maxiter=80)
        if time.time() < d7 - 5:
            h_r, c5_r = _lp_loop(h_r, min(d7, time.time() + 20),
                                  radius_init=0.08, max_iters=500, stall_limit=30)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 8: Final LP + pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.97:
        hf, c5f = _lp_loop(best_h, t0 + budget_s * 0.98,
                           radius_init=0.04, max_iters=2000, stall_limit=60)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()
        if time.time() < t0 + budget_s * 0.995:
            hp, c5p = _pair_polish_h(best_h, t0 + budget_s * 0.998, max_passes=30)
            if c5p < best_c5:
                best_c5, best_h = c5p, hp.copy()

    # Final projection and validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    # Verify feasibility
    if final_c5 > _exact_c5(h0) + 0.01:
        # Fallback to original if something went wrong
        return h0, _exact_c5(h0), n0
    return best_h, final_c5, len(best_h)
