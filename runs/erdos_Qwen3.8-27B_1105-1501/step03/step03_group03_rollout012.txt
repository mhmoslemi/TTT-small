def _exact_c5(h):

    dx = 2.0 / n

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scores_and_grad(h):
    n = len(h)
    L = 2 * n - 1
    scores = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                scores[r] = np.sum(h[d:] * (1.0 - h[:m]))
                # dS/dh[i] for i in [0, m-1]: -(h[i+d])
                # dS/dh[i] for i in [d, n-1]: +(1-h[i-d])
                G[r, :m] += -h[d:d+m]
                G[r, d:n] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                scores[r] = np.sum(h[:m] * (1.0 - h[g:]))
                G[r, :m] += (1.0 - h[g:])
                G[r, g:n] += -h[:m]
    return scores, G

def _project(v, target):

        for j in range(n):

                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j+1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try moving mass from j to k
                a_max = min(h[j], 1.0 - h[k])

                    a = a_max * frac

                        improved = True

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_row(h, d):
    """Gradient of score at lag d with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += -h[d:d+m]
            g[d:n] += (1.0 - h[:m])
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:n] += -h[:m]
    return g


def _scores_and_grad(h, active_idx=None):
    n = len(h)
    L = 2 * n - 1
    if active_idx is None:
        active_idx = range(L)
    G = np.zeros((len(list(active_idx) if isinstance(active_idx, (list, tuple, np.ndarray)) else active_idx), n))
    k = 0
    for d in active_idx:
        if isinstance(d, int) and d >= 0:
            lag = d - (n - 1)
        else:
            lag = d - (n - 1)
        G[k] = _grad_row(h, lag)
        k += 1
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lp_direction(h, scores, radius):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        G[r] = _grad_row(h, d)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_exact(h, p, target):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / max(p[j], 1e-14))
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / max(-p[j], 1e-14))
    alpha_max = max(alpha_max, 1e-8)

    c5_cur = _exact_c5(h)
    best_a = 0.0
    best_c5 = c5_cur

    n_eval = 30
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        h_t = _project(h + a * p, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(30):

        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)

        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))

    opt_h = _project(h + opt_a * p, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 < best_c5:

    return best_a, best_c5

    radius = radius_init
    stall = 0

        if time.time() > deadline:
            break
        if stall > 80:
            break

        scores = _scores(h)
        p = _lp_direction(h, scores, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.85
            if radius < 1e-7:
                break
            continue

        alpha, new_c5 = _line_search_exact(h, p, target)
        if new_c5 < cur_c5 - 5e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.05

    return best_h, best_c5


def _subgrad_descent(h, target, deadline, top_k=8, lr_init=0.02):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0

    while time.time() < deadline:
        if stall > 80:
            break

        scores = _scores(h)
        max_s = float(np.max(scores))
        idx = np.argsort(scores)[-top_k:]
        w = scores[idx]
        w = w / w.sum() if w.sum() > 0 else np.ones(len(idx)) / len(idx)

        grad = np.zeros(n)
        for i, d_idx in enumerate(idx):
            lag = d_idx - (n - 1)
            grad += w[i] * _grad_row(h, lag)
        grad -= grad.mean()

        h_new = _project(h - lr * grad, target)
        c5_new = _exact_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.03, 0.15)
            stall = 0
        else:
            h = h_new.copy()
            lr *= 0.85
            stall += 1

    return best_h, best_c5


def _frank_wolfe(h, target, deadline, max_iter=1000, top_k=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        scores = _scores(h)
        max_s = float(np.max(scores))
        active = np.where(scores >= max_s - 0.02 * abs(max_s))[0]
        if len(active) > top_k:
            active = np.argsort(scores)[-top_k:]

        grad = np.zeros(n)
        for d_idx in active:
            lag = d_idx - (n - 1)
            grad += _grad_row(h, lag)
        grad /= len(active)
        grad -= grad.mean()

        order = np.argsort(grad)
        v = np.zeros(n)
        v[order[:n // 2]] = 1.0

        best_t = 0.0
        best_f = _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _random_perturb(h, target, rng, scale=0.03):
    n = len(h)
    v = np.zeros(n)
    for freq in range(1, 6):
        phase = rng.uniform(0, 2 * np.pi)
        amp = scale / freq
        v += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    v -= v.mean()
    return _project(h + v, target)


def _pair_polish(h, target, deadline, max_passes=30):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 5e-9 or h[j] >= 1.0 - 5e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 5e-9 or h[k] >= 1.0 - 5e-9:
                    continue
                a_max = min(h[j], 1.0 - h[k])
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    a = a_max * frac
                    h[j] -= a
                    h[k] += a
                    c5_t = _exact_c5(h)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        improved = True
                    else:
                        h[j] += a
                        h[k] -= a
        if not improved:
            break
    return h, best_c5
