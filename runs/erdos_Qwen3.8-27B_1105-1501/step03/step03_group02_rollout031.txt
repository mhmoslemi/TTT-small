    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += res / mask.sum()
    return np.clip(x, 0.0, 1.0)

def _feible_interval(h, d):
    """Find t range where h + t*d stays in [0,1]^n"""
    n = len(h)
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.min((1.0 - h[neg]) / d[neg]))
    return t_lo, t_hi

def _exact_1d_solve(h, d, target):

    n = len(h)
    t_lo, t_hi = _feible_interval(h, d)
    if t_hi - t_lo < 1e-14:
        return h.copy(), _exact_c5(h)
    
    # Compute A, B, C using 3 evaluations
    s0 = np.correlate(h, 1.0-h, mode='full')
    s1 = np.correlate(h+d, 1.0-h-d, mode='full')
    s2 = np.correlate(h+2.0*d, 1.0-2.0*d, mode='full')
    A = s0
    C = (s2 - 2.0*s1 + s0) / 2.0
    B = (s1 - s0) - C
    
    M = len(A)  # 2n-1
    
    # Generate candidate t values
    candidates = [t_lo, t_hi, 0.0]
    
    # Vertices

        if abs(C[r]) > 1e-14:
            tv = -B[r] / (2.0 * C[r])
            if t_lo - 1e-12 <= tv <= t_hi + 1e-12:
                candidates.append(tv)
    
    # Pairwise crossings
    for i in range(M):
        for j in range(i+1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) > 1e-14:
                disc = b*b - 4.0*a*c
                if disc >= -1e-14:
                    disc = max(disc, 0.0)
                    sq = np.sqrt(disc)
                    for t_cand in [(-b+sq)/(2.0*a), (-b-sq)/(2.0*a)]:
                        if t_lo - 1e-12 <= t_cand <= t_hi + 1e-12:
                            candidates.append(t_cand)
            elif abs(b) > 1e-14:

                if t_lo - 1e-12 <= t_cand <= t_hi + 1e-12:
                    candidates.append(t_cand)
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_lo, t_hi)
    
    # Evaluate E(t) = max_r (A + B*t + C*t²) for all candidates

    h_new = _project(h_new, target)

def _smooth_max_direction(h, beta=50.0, top_k=20):
    """Generate direction from smooth max of active lags"""
    n = len(h)
    scores = np.correlate(h, 1.0-h, mode='full')
    max_s = np.max(scores)
    # Weights concentrated on top lags
    w = np.exp(beta * (scores - max_s))
    # Keep only top_k for efficiency
    top_idx = np.argsort(scores)[::-1][:top_k]
    w_full = np.zeros(len(scores))
    w_full[top_idx] = w[top_idx]
    w_full /= w_full.sum()
    
    # Compute gradients for top lags
    eps = 1e-7
    grad = np.zeros(n)
    # For each top lag, compute gradient numerically
    # Actually, let's compute all gradients at once
    for i in range(n):
        hp = h.copy(); hp[i] += eps
        hm = h.copy(); hm[i] -= eps
        sp = np.correlate(hp, 1.0-hp, mode='full')
        sm = np.correlate(hm, 1.0-hm, mode='full')
        grad += w_full * (sp - sm) / (2.0 * eps)
        # This is expensive: n * 2 correlates
    # Make zero-sum
    grad -= np.mean(grad)
    norm = np.max(np.abs(grad))
    if norm < 1e-15:
        return None
    return -grad / norm  # Negative = descent direction

def _compute_all_grads(h):
    """Compute gradient matrix: grads[r, i] = d score_r / d h[i]"""
    n = len(h)
    M = 2*n - 1
    eps = 1e-7
    grads = np.zeros((M, n))
    for i in range(n):
        hp = h.copy(); hp[i] += eps
        hm = h.copy(); hm[i] -= eps
        sp = np.correlate(hp, 1.0-hp, mode='full')
        sm = np.correlate(hm, 1.0-hm, mode='full')
        grads[:, i] = (sp - sm) / (2.0 * eps)
    return grads

def _smooth_max_direction(h, beta=50.0, top_k=30):
    n = len(h)
    scores = np.correlate(h, 1.0-h, mode='full')
    max_s = np.max(scores)
    # Softmax weights, restricted to top_k
    top_idx = np.argsort(scores)[::-1][:top_k]
    w = np.zeros(len(scores))
    w[top_idx] = np.exp(beta * (scores[top_idx] - max_s))
    w /= w.sum()
    
    # Compute gradients
    grads = _compute_all_grads(h)  # (M, n)
    direction = grads.T @ w  # (n,)
    direction -= np.mean(direction)
    norm = np.max(np.abs(direction))
    if norm < 1e-15:
        return None
    return -direction / norm  # Negative = descent

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _feasible_interval(h, d):
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        ub = np.min((1.0 - h[pos]) / d[pos])
        t_hi = min(t_hi, ub)
    if np.any(neg):
        lb = np.min((1.0 - h[neg]) / d[neg])
        t_lo = max(t_lo, lb)
    return t_lo, t_hi

def _exact_1d_solve(h, d, target):
    n = len(h)
    t_lo, t_hi = _feasible_interval(h, d)
    if t_hi - t_lo < 1e-13:
        return h.copy(), _exact_c5(h)

    s0 = np.correlate(h, 1.0 - h, mode='full')
    s1 = np.correlate(h + d, 1.0 - h - d, mode='full')
    s2 = np.correlate(h + 2.0 * d, 1.0 - 2.0 * h - 2.0 * d, mode='full')
    A = s0
    C = (s2 - 2.0 * s1 + s0) / 2.0
    B = (s1 - s0) - C
    M = len(A)

    candidates = [t_lo, t_hi, 0.0]

    # Vertices
    c_mask = np.abs(C) > 1e-14
    if np.any(c_mask):
        tv = -B[c_mask] / (2.0 * C[c_mask])
        tv = tv[(tv >= t_lo - 1e-12) & (tv <= t_hi + 1e-12)]
        candidates.extend(tv.tolist())

    # Pairwise crossings (vectorized)
    pi, pj = np.triu_indices(M, k=1)
    a = C[pi] - C[pj]
    b = B[pi] - B[pj]
    c = A[pi] - A[pj]

    quad_mask = np.abs(a) > 1e-14
    if np.any(quad_mask):
        disc = b[quad_mask]**2 - 4.0 * a[quad_mask] * c[quad_mask]
        valid = disc >= 0
        if np.any(valid):
            sq = np.sqrt(np.maximum(disc[valid], 0.0))
            r1 = (-b[quad_mask][valid] + sq) / (2.0 * a[quad_mask][valid])
            r2 = (-b[quad_mask][valid] - sq) / (2.0 * a[quad_mask][valid])
            r1 = r1[(r1 >= t_lo - 1e-12) & (r1 <= t_hi + 1e-12)]
            r2 = r2[(r2 >= t_lo - 1e-12) & (r2 <= t_hi + 1e-12)]
            candidates.extend(r1.tolist())
            candidates.extend(r2.tolist())

    lin_mask = (~quad_mask) & (np.abs(b) > 1e-14)
    if np.any(lin_mask):
        rl = -c[lin_mask] / b[lin_mask]
        rl = rl[(rl >= t_lo - 1e-12) & (rl <= t_hi + 1e-12)]
        candidates.extend(rl.tolist())

    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_lo, t_hi)

    if len(candidates) > 50000:
        candidates = candidates[np.random.default_rng(0).choice(len(candidates), 50000, replace=False)]

    t_mat = candidates[:, np.newaxis]
    Q = A[np.newaxis, :] + B[np.newaxis, :] * t_mat + C[np.newaxis, :] * t_mat * t_mat
    E = np.max(Q, axis=1)
    best_idx = int(np.argmin(E))
    best_t = candidates[best_idx]

    if abs(best_t) < 1e-14:
        return h.copy(), _exact_c5(h)

    h_new = np.clip(h + best_t * d, 0.0, 1.0)
    h_new = _project(h_new, target)
    c5_new = _exact_c5(h_new)
    c5_cur = _exact_c5(h)
    if c5_new > c5_cur + 1e-10:
        return h.copy(), c5_cur
    return h_new, c5_new

def _compute_grads(h):
    n = len(h)
    M = 2 * n - 1
    eps = 1e-6
    grads = np.zeros((M, n))
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        hm = h.copy()
        hm[i] -= eps
        sp = np.correlate(hp, 1.0 - hp, mode='full')
        sm = np.correlate(hm, 1.0 - hm, mode='full')
        grads[:, i] = (sp - sm) / (2.0 * eps)
    return grads

def _smooth_max_dir(h, beta, top_k, grads):
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    top_idx = np.argsort(scores)[::-1][:top_k]
    w = np.zeros(len(scores))
    w[top_idx] = np.exp(beta * (scores[top_idx] - max_s))
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= ws
    direction = grads.T @ w
    direction -= np.mean(direction)
    norm = float(np.max(np.abs(direction)))
    if norm < 1e-14:
        return None
    return -direction / norm

def _rebalance_dir(h, n):
    d = 0.5 - h
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None
    return d / norm

def _laplacian_dir(h, n):
    d = np.zeros(n)
    d[1:-1] = h[2:] - 2.0 * h[1:-1] + h[:-2]
    d[0] = h[1] - h[0]
    d[-1] = h[-2] - h[-1]
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None
    return d / norm

def _gen_directions(h, n, rng, scores, grads):

    # Smooth max with various betas and top_k

    # Single worst lag

    if d is not None:

    # Rebalance

        dirs.append(d)
    # Laplacian

    if d is not None:

        dr -= np.mean(dr)
        nr = float(np.max(np.abs(dr)))
        if nr > 1e-14:

    active = np.argsort(scores)[::-1][:10]

        if i != j:

            d[i2] = 1.0
            d[j2] = -1.0

def _try_directions(h, directions, target, deadline):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for d in directions:
        if time.time() > deadline:
            break
        h_new, c5_new = _exact_1d_solve(h, d, target)
        if c5_new < best_c5 - 5e-11:
            best_h = h_new
            best_c5 = c5_new
            h = h_new
    return best_h, best_c5

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
