def _solve_lp(h, raw_vals, grads, top_idx, target, R, lam):
    n = len(h)
    # Variables: [t, p_0, q_0, p_1, q_1, ..., p_{n-1}, q_{n-1}]
    # Index: 0=t, 2i+1=p_i, 2i+2=q_i
    
    n_vars = 2*n + 1
    
    # Objective: minimize t + lam * sum(p_i + q_i)
    c = np.zeros(n_vars)
    c[0] = 1.0
    for i in range(n):
        c[2*i+1] = lam
        c[2*i+2] = lam
    
    # Constraints:
    # 1) For each selected lag s: -t + sum_j grad_{s,j}*(p_j - q_j) <= -F_s
    # 2) sum_j (p_j - q_j) = 0
    # 3) sum_j (p_j + q_j) <= R
    
    A_ub = []
    b_ub = []
    
    for s_idx in top_idx:
        row = np.zeros(n_vars)
        row[0] = -1.0
        g = grads[s_idx]
        for j in range(n):
            row[2*j+1] = g[j]   # coefficient of p_j
            row[2*j+2] = -g[j]  # coefficient of q_j
        A_ub.append(row)
        b_ub.append(-raw_vals[s_idx])
    
    # Norm constraint: sum(p + q) <= R
    row = np.zeros(n_vars)
    for j in range(n):
        row[2*j+1] = 1.0
        row[2*j+2] = 1.0
    A_ub.append(row)
    b_ub.append(R)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(p - q) = 0
    A_eq = np.zeros((1, n_vars))
    for j in range(n):
        A_eq[0, 2*j+1] = 1.0
        A_eq[0, 2*j+2] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for j in range(n):
        if h[j] < 1e-6:
            bounds.append((0, None))  # p_j free
            bounds.append((0, 0))     # q_j = 0 (can't decrease)
        elif h[j] > 1 - 1e-6:
            bounds.append((0, 0))     # p_j = 0 (can't increase)
            bounds.append((0, None))  # q_j free
        else:
            bounds.append((0, None))  # p_j free
            bounds.append((0, None))  # q_j free
    
    # Solve LP
    from scipy.optimize import linprog
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='revisal')
        except Exception:
            return None
    
    if not res.success or res.x is None:
        return None
    
    # Extract d
    p = np.array([res.x[2*j+1] for j in range(n)])
    q = np.array([res.x[2*j+2] for j in range(n)])
    d = p - q
    d -= np.mean(d)  # enforce exact zero sum
    
    dnorm = np.max(np.abs(d))
    if dnorm < 1e-10:
        return None
    
    return d / dnorm  # normalize to unit max

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _offset_data(h):

    n = len(h)
    n_off = 2 * n - 1
    raw = np.zeros(n_off)
    grads = np.zeros((n_off, n))

import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _offset_data(h):
    n = len(h)
    n_off = 2 * n - 1
    raw = np.zeros(n_off)
    grads = np.zeros((n_off, n))

        if d >= 0:
            m = n - d
            if m > 0:
                raw[idx] = float(np.dot(h[:m], 1.0 - h[d:d+m]))
                grads[idx, :m] += (1.0 - h[d:d+m])

def _solve_tangent_lp(h, raw, grads, top_idx, R):
    n = len(h)
    n_sel = len(top_idx)
    n_vars = 2 * n + 1
    c = np.zeros(n_vars)
    c[0] = 1.0
    A_ub_rows = []
    b_ub = []
    for s_idx in top_idx:
        row = np.zeros(n_vars)
        row[0] = -1.0
        g = grads[s_idx]
        row[1::2] = g
        row[2::2] = -g
        A_ub_rows.append(row)
        b_ub.append(-raw[s_idx])
    norm_row = np.zeros(n_vars)
    norm_row[1:] = 1.0
    A_ub_rows.append(norm_row)
    b_ub.append(R)
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1::2] = 1.0
    A_eq[0, 2::2] = -1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)]
    for j in range(n):
        if h[j] < 1e-8:
            bounds.append((0, None))
            bounds.append((0, 0))
        elif h[j] > 1.0 - 1e-8:
            bounds.append((0, 0))
            bounds.append((0, None))
        else:
            bounds.append((0, None))
            bounds.append((0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        return None, None
    if not res.success or res.x is None:
        return None, None
    t_opt = float(res.x[0])
    p = res.x[1::2].copy()
    q = res.x[2::2].copy()
    d = p - q
    d -= np.mean(d)
    dnorm = float(np.max(np.abs(d)))

    return d / dnorm, t_opt

def _line_search(h, d, target, best_c5, max_alpha=3.0):
    n = len(h)
    dnorm = float(np.max(np.abs(d)))
    if dnorm < 1e-14:
        return h.copy(), best_c5
    d = d / dnorm
    best_h = h.copy()
    best_c = best_c5
    alphas = [0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 2.5]
    alphas = [a for a in alphas if a <= max_alpha]
    if not alphas:
        alphas = [0.1]
    for a in alphas:
        h_try = _project(h + a * d, target)
        c_try = _exact_c5(h_try)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if float(np.sum(best_h) - target) != 0.0:
        best_h = _project(best_h, target)
    return best_h, best_c

def _subgrad_dir(h, raw, grads, T=0.4):
    n = len(h)
    rmax = float(np.max(raw))
    shifted = raw - rmax
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.zeros(len(raw))
        w[int(np.argmax(raw))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(len(h))
    for i in range(len(raw)):
        if w[i] > 1e-10:
            grad += w[i] * grads[i]
    grad -= np.mean(grad)
    gnorm = float(np.linalg.norm(grad))
    if gnorm < 1e-12:
        return None
    return grad / gnorm

                            best_c5 = c_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                            cur_c5 = c_try
                            improved = True
                            break
                    if improved:
                        break
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    target_best = target0
    
    # Phase 1: LP optimization from initial (35%)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_optimize(h, dl1, target0, rng, max_iters=80000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-start (30%)
    dl2 = t0 + budget_s * 0.65
    n_starts = 15
    for s in range(n_starts):
        if time.time() > dl2:
            break
        remaining = dl2 - time.time()
        if remaining < 5.0:
            break
        alloc = min(remaining * 0.8 / max(1, n_starts - s), 30.0)
        sub_dl = time.time() + alloc
        h_try = _perturb_start(best_h, rng, target_best)
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, target_best, rng, max_iters=15000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Resolution sweep (20%)
    dl3 = t0 + budget_s * 0.85
    n_start = best_n
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, -12, 12, 16, -16, 20, -20]:
        if time.time() > dl3:
            break
        n_try = n_start + dn
        if n_try < 16 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + min(15.0, (dl3 - time.time()) * 0.3))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, rng, max_iters=8000)
        h_r, c5_r = _fine_polish(h_r, min(dl3, time.time() + 5.0), n_try / 2.0, rng)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            target_best = n_try / 2.0
    
    # Phase 4: Fine polish (remaining)
    dl4 = t0 + budget_s * 0.98
    h4, c5_4 = _fine_polish(best_h, dl4, target_best, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Final validation
    best_h = _project(best_h, target_best)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
        best_n = n0
    
    s_check = float(np.sum(best_h))
    if abs(s_check - best_n / 2.0) > 1e-8:
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
