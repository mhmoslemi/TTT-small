import numpy as np
from scipy.optimize import linprog
import time

def compute_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def lp_direction(h, G, scores, delta_bound):
    n = len(h)
    n_scores = 2*n - 1
    # Variables: [delta_0, ..., delta_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub: G @ delta - t <= -scores
    # Shape: (n_scores, n+1)
    A_ub = np.zeros((n_scores, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # A_eq: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -delta_bound)
        hi = min(1.0 - h[i], delta_bound)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is unbounded (but will be bounded by constraints)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def line_search(h, direction, scores_current, max_c5):
    n = len(h)
    dx = 2.0 / n
    # Find max step
    alpha_max = 1.0
    for i in range(n):
        if direction[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / direction[i])
        elif direction[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-direction[i]))
    alpha_max = max(alpha_max, 0.0)
    
    if alpha_max < 1e-10:
        return 0.0, max_c5
    
    # Golden section search
    best_alpha = 0.0
    best_val = max_c5
    
    # First try a few points
    candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
    for frac in candidates:
        a = alpha_max * frac
        h_new = h + a * direction
        s = np.max(compute_scores(h_new)) * dx
        if s < best_val:
            best_val = s
            best_alpha = a
    
    # Golden section refinement around best
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.1 * alpha_max)
        hi = min(alpha_max, best_alpha + 0.1 * alpha_max)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = np.max(compute_scores(h + c1 * direction)) * dx
        f2 = np.max(compute_scores(h + c2 * direction)) * dx
        for _ in range(20):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = np.max(compute_scores(h + c1 * direction)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = np.max(compute_scores(h + c2 * direction)) * dx
        opt_a = (lo + hi) / 2.0
        opt_val = np.max(compute_scores(h + opt_a * direction)) * dx
        if opt_val < best_val:
            best_val = opt_val
            best_alpha = opt_a
    
    return best_alpha, best_val

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(compute_scores(h)) * (2.0 / n))

def compute_grad(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def project_onto_simplex(v, target):
    """Project v onto {x : 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on lambda: sum(clip(v - lambda, 0, 1)) = target

        else:

    return None
