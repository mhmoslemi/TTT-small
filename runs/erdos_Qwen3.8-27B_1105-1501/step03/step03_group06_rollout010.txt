                    best_c5 = c5
                    best_h = h_new.copy()
                    h = h_new
                else:

            else:

        except Exception:
            break
    
    return best_h, best_c5

Wait, there's an issue. The SLSQP with only top-k constraints might find a solution where a non-top-k constraint becomes the new maximum. That's fine - we'll catch it in the next outer iteration.

But there's another issue: the SLSQP is minimizing T, and the constraints are T ≥ dx*C_r(h) for r in top-k. If the current T_cur is already the max of ALL constraints, then the SLSQP solution will try to reduce T while keeping T ≥ dx*C_r for the top-k rows. This means it will try to reduce those specific correlation values. Good.

But what if the solution wants to move h in a direction that increases some other correlation row? The SLSQP doesn't know about it. That's the whole point of the active set method - we check after each iteration.

Let me also think about the L-BFGS-B smooth approach. The previous code used log-sum-exp which is a smooth approximation. The issue is that the log-sum-exp gives too much weight to non-bottleneck rows when alpha is large, and not enough differentiation when alpha is small.

A better approach might be: **Minimize a weighted sum of the top-k rows** (not all rows). This is like a "top-k mean" objective.

f(h) = (1/k) * sum_{r in top-k(h)} dx * C_r(h)

This is non-smooth (the top-k changes with h), but locally it's smooth. We can use a subgradient or just recompute the top-k at each step.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo >= hi:
            continue
        idx1 = np.arange(lo, hi)
        idx2 = idx1 + s
        G[r, idx1] += (1.0 - h[idx2])
        G[r, idx2] -= h[idx1]
    return G


def _lse_fun(x, alpha, target):
    h = x
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * c
    mx = float(np.max(vals))
    shifted = vals - mx
    exp_vals = np.exp(np.clip(shifted, -500.0, 500.0))
    f = mx + np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(h)
    g = G.T @ w
    g /= alpha
    g -= np.mean(g)
    return f, g


def _slsqp_active(h, dx, target, top_k=5, maxiter=150, tol=1e-14):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()

    for outer in range(40):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        T_cur = float(np.max(corr))
        top_idx = np.argsort(corr)[-top_k:]

        def _obj(x):
            return x[0]

        def _obj_g(x):
            g = np.zeros(n + 1)
            g[0] = 1.0
            return g

        cons = []
        for r in top_idx:
            s = r - (n - 1)
            lo = max(0, -s)
            hi = min(n, n - s)

            def _cfun(x, r=r, s=s, lo=lo, hi=hi):
                h_ = x[1:]
                cval = float(np.sum(h_[lo:hi] * (1.0 - h_[lo + s:hi + s]))) * dx
                return x[0] - cval

            def _cjac(x, r=r, s=s, lo=lo, hi=hi):
                h_ = x[1:]
                g = np.zeros(n + 1)
                g[0] = 1.0
                for j in range(lo, hi):
                    g[1 + j] += dx * (1.0 - h_[j + s])
                    g[1 + j + s] -= dx * h_[j]
                return g

            cons.append({'fun': _cfun, 'jac': _cjac, 'type': 'ineq'})

        def _eqf(x):
            return float(np.sum(x[1:])) - target

        def _eqj(x):
            g = np.zeros(n + 1)
            g[1:] = 1.0
            return g

        cons.append({'fun': _eqf, 'jac': _eqj, 'type': 'eq'})

        bounds = [(0.0, 4.0 / n)] + [(0.0, 1.0)] * n
        x0 = np.zeros(n + 1)
        x0[0] = T_cur
        x0[1:] = h

        try:
            res = minimize(_obj, x0, jac=_obj_g, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': tol, 'iprint': 0})
            if res.success:
                h_new = np.clip(res.x[1:], 0.0, 1.0)
                h_new = _project(h_new, target)
                c5 = _exact_c5(h_new)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h = h_new
                else:
                    break
            else:
                break
        except Exception:
            break

    return best_h, best_c5


def _bundle_descent(h, target, max_iter=500, deadline=None):
    """Subgradient method for the non-smooth max with line search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    c = best_c5
    best_c = best_c5

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        T_cur = float(np.max(corr))
        # Average gradient of active rows (within tolerance of max)
        tol_act = max(1e-6, (T_cur - best_c) * 0.1)
        active_mask = corr >= T_cur - tol_act
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            break
        G = _grad_matrix(h)
        active_G = G[active_mask, :]
        # Gradient of the max is average of active gradients (subgradient)
        d = active_G.mean(axis=0)
        d -= np.mean(d)  # Remove component along equality constraint
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-14:
            break
        d = -d / d_norm

        # Line search
        best_a = 0.0
        best_c5_local = T_cur
        for t in np.linspace(0.0, 1.0, 16):
            a = t * 0.3
            h_t = _project(h + a * d, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_local:
                best_c5_local = c5_t
                best_a = a

        if best_a > 0 and best_c5_local < T_cur - 1e-10:
            h = _project(h + best_a * d, target)
            T_new = _exact_c5(h)
            if T_new < best_c5:
                best_c5 = T_new
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def _lbfgs_smooth(h, alpha, maxiter=200, target=0):
    n = len(h)
    if target == 0:
        target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_fun, h, args=(alpha, target), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'maxls': 50, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for sign in [1.0, -1.0]:
                    a_max_j = (1.0 - h[j]) if sign > 0 else h[j]
                    a_max_k = h[k] if sign > 0 else (1.0 - h[k])
                    a_max = min(a_max_j, a_max_k) * 0.8
                    if a_max < 1e-10:
                        continue
                    for frac in [1.0, 0.5, 0.25, 0.75]:
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break

    return best_h, best_c5


def _lp_polish(h, deadline, radius_init=0.12, max_iters=3000, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break

        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        # LP: max t s.t. G @ p + scores <= scores + t, sum(p)=0, |p| <= radius
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] > 1e-10:
                p = res.x[:n]
            else:
                stall += 1
                radius *= 0.8
                continue
        except Exception:
            stall += 1
            radius *= 0.8
            continue

        if float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            continue

        # Line search
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)

        best_a, best_c5_ls = 0.0, cur_c5
        for t in np.linspace(0.0, 1.0, 12):
            a = t * alpha_max
            if a < 1e-12:
                continue
            c5_t = _exact_c5(_project(h + a * p, target))
            if c5_t < best_c5_ls:
                best_c5_ls, best_a = c5_t, a

        if best_a > 0 and best_c5_ls < cur_c5 - 1e-12:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.03:
                radius = 0.06

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Smooth L-BFGS-B optimization (20% budget)
    d1 = t0 + budget_s * 0.20
    for alpha in [8.0, 15.0, 30.0, 60.0, 100.0]:
        if time.time() > d1:
            break
        h_try, c5_try = _lbfgs_smooth(h, alpha, maxiter=250, target=target)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = h_try
        if time.time() < d1:
            h_try2, c5_try2 = _lbfgs_smooth(h, alpha, maxiter=100, target=target)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
            h = h_try2

    # Phase 2: SLSQP active set optimization (20% budget)
    d2 = t0 + budget_s * 0.40
    current = best_h.copy()
    for top_k in [3, 5, 7, 10]:
        if time.time() > d2:
            break
        current, c5_s = _slsqp_active(current, 2.0 / len(current),
                                      target, top_k=top_k, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = current.copy()

    # Second SLSQP pass with more iterations
    if time.time() < d2:
        current, c5_s2 = _slsqp_active(best_h, 2.0 / best_n,
                                       target, top_k=5, maxiter=300)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = current.copy()

    # Phase 3: LP polishing (15% budget)
    d3 = t0 + budget_s * 0.55
    h_tp, c5_tp = _lp_polish(best_h, d3, radius_init=0.12, max_iters=4000, stall_limit=50)
    if c5_tp < best_c5:
        best_c5 = c5_tp
        best_h = h_tp.copy()
        best_n = len(best_h)

    # Phase 4: Bundle descent (10% budget)
    d4 = t0 + budget_s * 0.65
    h_bd, c5_bd = _bundle_descent(best_h, target, max_iter=1000, deadline=d4)
    if c5_bd < best_c5:
        best_c5 = c5_bd
        best_h = h_bd.copy()

    # Phase 5: Resolution search (15% budget)
    d5 = t0 + budget_s * 0.80
    for dn in [-4, -2, 2, 4, 6, 8, -8, -12, 12, 16]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        tgt_r = n_try / 2.0
        # Quick smooth
        h_r, c5_r = _lbfgs_smooth(h_r, 30.0, maxiter=100, target=tgt_r)
        # SLSQP
        if time.time() < d5:
            h_r, c5_r = _slsqp_active(h_r, 2.0 / n_try, tgt_r,
                                       top_k=5, maxiter=150)
        # LP
        if time.time() < d5:
            sub_d = min(d5, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_polish(h_r, sub_d, radius_init=0.08,
                                    max_iters=500, stall_limit=30)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Multiple SLSQP + LP from perturbations (10% budget)
    d6 = t0 + budget_s * 0.90
    scale = 0.03
    for attempt in range(200):
        if time.time() > d6:
            break
        if best_c5 < 0.3807:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        # Quick SLSQP
        sub_d = min(d6, time.time() + budget_s * 0.015)
        h_try, c5_try = _slsqp_active(h_p, 2.0 / best_n, best_n / 2.0,
                                       top_k=5, maxiter=100)
        if c5_try < best_c5 - 1e-12:
            sub_d2 = min(d6, time.time() + budget_s * 0.02)
            h_try, c5_try = _lp_polish(h_try, sub_d2, radius_init=0.06,
                                        max_iters=300, stall_limit=30)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.15, scale * 1.2)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 7: Final SLSQP polish (5% budget)
    if time.time() < t0 + budget_s * 0.95:
        d7 = t0 + budget_s * 0.95
        for tk in [3, 5]:
            if time.time() >= d7:
                break
            h_fin, c5_fin = _slsqp_active(best_h, 2.0 / best_n,
                                           best_n / 2.0, top_k=tk, maxiter=200)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 8: Final LP + pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.98:
        d8 = t0 + budget_s * 0.98
        h_fin, c5_fin = _lp_polish(best_h, d8, radius_init=0.04,
                                    max_iters=2000, stall_limit=60)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if time.time() < t0 + budget_s * 0.99:
            hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=40)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
