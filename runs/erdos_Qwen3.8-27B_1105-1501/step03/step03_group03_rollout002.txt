        if d >= 0:
            m = n - d
            G[r, :m] -= h[d:]
            G[r, d:] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0 - h[g:]
            G[r, g:] -= h[:m]
    return G


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lp_direction(h, G, scores, radius):
    n = len(h)
    L = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.95)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.95)
    if alpha_max <= 1e-14:
        return 0.0, c5_cur
    best_a, best_c5 = 0.0, c5_cur
    for i in range(60):
        a = alpha_max * (0.6 ** i)
        if a < 1e-13:
            break
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_a = a
        elif best_a > 0 and a > best_a:
            break
    if best_a > 0:
        lo = best_a * 0.5
        hi = alpha_max if best_a == alpha_max else best_a * 1.5
        hi = min(hi, alpha_max)
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = _exact_c5(h + m1 * p)
            f2 = _exact_c5(h + m2 * p)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        a_opt = (lo + hi) / 2.0
        c5_opt = _exact_c5(h + a_opt * p)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_a = a_opt
    return best_a, best_c5


def _subgradient_step(h, scores, G, lr):
    n = len(h)
    S_max = scores.max()
    thresh = S_max - 0.02 * max(S_max, 1.0)
    active = np.where(scores >= thresh)[0]
    if len(active) == 0:
        active = np.where(scores == S_max)[0]
    if len(active) == 0:
        return h.copy()
    g = G[active].mean(axis=0)
    g = g - g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return h.copy()
    h_new = h - lr * g
    h_new = _project(h_new, n / 2.0)
    return h_new


def _lp_optimize(h, deadline, radius_init=0.1, use_subgrad=True):
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    sg_lr = 0.02
    subgrad_count = 0

    while time.time() < deadline:
        scores = _scores(h)
        G = _grad_matrix(h)

        p = _lp_direction(h, G, scores, radius)

        if p is not None and np.max(np.abs(p)) > 1e-13:
            alpha, new_c5 = _line_search(h, p, cur_c5)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.4)
                continue
        stall += 1

        if stall > 8 and use_subgrad:
            h_new = _subgradient_step(h, scores, G, sg_lr)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 + 0.001:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                subgrad_count += 1
                sg_lr = min(sg_lr * 1.1, 0.15)
                radius = min(radius * 1.5, 0.3)
                stall = max(stall - 3, 1)
            else:
                sg_lr *= 0.7

        radius *= 0.78
        if radius < 1e-7:
            break
        if stall > 60:
            break
        if stall % 20 == 0 and radius < 0.03:
            radius = 0.06

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slsqp_quick(h, target, beta=30.0, maxiter=80):
    n = len(h)

    def obj(x):
        sc = np.correlate(x, 1.0 - x, mode='full')
        sm = sc.max()
        return sm + np.log(np.sum(np.exp(beta * (sc - sm)))) / beta

    def jac(x):
        G = _grad_matrix(x)
        sc = np.correlate(x, 1.0 - x, mode='full')
        sm = sc.max()
        w = np.exp(beta * (sc - sm))
        w /= w.sum()
        g = G.T @ w
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _pair_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(40):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-10 or h[k] >= 1.0 - 1e-10:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-12:
                        continue
                    for t in [0.3, 0.6, 1.0]:
                        a = a_max * t
                        h[j] += dj * a
                        h[k] += dk * a
                        c5_t = _exact_c5(h)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            improved = True
                        else:
                            h[j] -= dj * a
                            h[k] -= dk * a
                            break
                        break
        if not improved:
            break
    return h, best_c5


def _kick(h, rng, size=0.03):
    n = len(h)
    direction = rng.standard_normal(n)
    direction -= direction.mean()
    norm = np.linalg.norm(direction)
    if norm < 1e-12:
        return h.copy()
    direction /= norm
    h_new = h + size * direction
    return _project(h_new, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.985

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP warm start
    deadline1 = t0 + budget_s * 0.12
    h_s = _slsqp_quick(h, target0, beta=25.0, maxiter=60)
    c5_s = _exact_c5(h_s)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
        h = h_s.copy()

    # Phase 2: Main LP optimization
    deadline2 = t0 + budget_s * 0.50
    h_lp, c5_lp = _lp_optimize(best_h, deadline2, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: LP with random kicks to escape local optima
    deadline3 = t0 + budget_s * 0.72
    for attempt in range(300):
        if time.time() > deadline3:
            break
        if best_c5 < 0.38075:
            break
        h_kick = _kick(best_h, rng, size=rng.uniform(0.01, 0.04))
        h_try, c5_try = _lp_optimize(h_kick, min(deadline3, time.time() + budget_s * 0.012),
                                     radius_init=0.08)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Resolution exploration
    deadline4 = t0 + budget_s * 0.88
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 12]:
        if time.time() > deadline4:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 160:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_optimize(h_r, min(deadline4, time.time() + budget_s * 0.018),
                                 radius_init=0.10)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Pair polish
    deadline5 = t0 + budget_s * 0.94
    h_p, c5_p = _pair_polish(best_h, deadline5)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 6: Final LP polish
    deadline6 = hard_deadline
    h_f, c5_f = _lp_optimize(best_h, deadline6, radius_init=0.03, use_subgrad=False)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
