```python
def _fast_grad(h, weights):
    """Compute gradient of weighted sum of scores, only for significant weights."""
    n = len(h)
    grad = np.zeros(n)
    for r in range(2*n-1):
        if abs(weights[r]) < 1e-10:
            continue
        w = weights[r]
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    return grad
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
```

```python
    b[idx] = 1.0
    return b


def _swiftlight(h, deadline, beta_start=100.0, beta_end=20000.0, top_k=4, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    it = 0
    rng_local = np.random.default_rng(12345)
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / max_iters)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        # Randomized top-k selection for diversity
        top_k_actual = min(top_k, 2 * n - 1)
        if it % 50 == 0:
            # Occasionally use random subset of top lags
            top_idx = np.argsort(scores)[-top_k_actual * 2:]
            sel = rng_local.choice(top_idx, size=top_k_actual, replace=False)
        else:
            sel = np.argsort(scores)[-top_k_actual:]
        exps = np.zeros(2 * n - 1)
        for s in sel:
            exps[s] = np.exp(-beta * (m - scores[s]))
        total = float(np.sum(exps))
        if total < 1e-15:
            weights = np.zeros(2 * n - 1)
            weights[int(np.argmax(scores))] = 1.0
        else:
            weights = exps / total
        grad = _fast_grad(h, weights)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            if np.any(h_try < -0.3) or np.any(h_try > 1.3):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.02, 0.3)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-5 or stall > 60:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial soft descent to ensure good local optimum
    deadline1 = t0 + budget_s * 0.12
    h, c5 = _swiftlight(h, deadline1, beta_start=100.0, beta_end=20000.0, top_k=4, max_iters=20000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP polish (fine tuning)
    deadline2 = t0 + budget_s * 0.30
    h, c5 = _lp_polish(h, deadline2, radius_init=0.02, max_iters=10000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Frank-Wolfe descent (explores binary-like directions)
    deadline3 = t0 + budget_s * 0.38
    h, c5 = _fw_descent(best_h, deadline3, max_iters=5000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Basin hopping with diverse small perturbations
    deadline4 = t0 + budget_s * 0.72
    scale = 0.03
    attempts = 0
    max_attempts = 200
    while time.time() < deadline4 and attempts < max_attempts:
        attempts += 1
        rem_time = deadline4 - time.time()
        if rem_time < 3:
            break
        time_per = min(rem_time / max(1, max_attempts - attempts), budget_s * 0.012)
        mut_type = attempts % 6
        if mut_type == 0:
            # Gaussian noise
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif mut_type == 1:
            # Single coordinate perturbation
            h_pert = best_h.copy()
            j = int(rng.integers(0, best_n))
            delta = rng.choice([0.02, 0.05, 0.1]) * rng.choice([1.0, -1.0])
            if 0 <= h_pert[j] + delta <= 1:
                h_pert[j] += delta
        elif mut_type == 2:
            # Pair swap
            h_pert = best_h.copy()
            j1, j2 = rng.choice(best_n, size=2, replace=False)
            h_pert[j1], h_pert[j2] = h_pert[j2], h_pert[j1]
        elif mut_type == 3:
            # Mix with random binary
            b = _binary_snapped(best_h + rng.normal(0, 0.3, best_n))
            mix = rng.uniform(0.1, 0.4)
            h_pert = (1.0 - mix) * best_h + mix * b
        elif mut_type == 4:
            # Small cyclic shift
            shift = int(rng.integers(1, 4)) * rng.choice([1, -1])
            h_pert = np.roll(best_h, shift)
        else:
            # Larger Gaussian
            h_pert = best_h + rng.normal(0, scale * 2, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        h_try, c5_try = _swiftlight(h_pert, time.time() + time_per,
                                    beta_start=50.0, beta_end=5000.0, top_k=4, max_iters=3000)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.9)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 5: LP polish on best
    deadline5 = t0 + budget_s * 0.82
    h, c5 = _lp_polish(best_h, deadline5, radius_init=0.015, max_iters=8000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Multi-n exploration
    deadline6 = t0 + budget_s * 0.93
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10]:
        if time.time() > deadline6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        time_left = deadline6 - time.time()
        if time_left < 5:
            break
        t_sub = min(time_left * 0.4, budget_s * 0.04)
        h_r, c5_r = _swiftlight(h_r, time.time() + t_sub,
                                beta_start=50.0, beta_end=5000.0, top_k=4, max_iters=5000)
        t_sub2 = min(deadline6 - time.time(), budget_s * 0.02)
        if t_sub2 > 3:
            h_r, c5_r = _lp_polish(h_r, time.time() + t_sub2, radius_init=0.015, max_iters=3000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final fine polish
    deadline7 = t0 + budget_s * 0.995
    h, c5 = _lp_polish(best_h, deadline7, radius_init=0.008, max_iters=10000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
