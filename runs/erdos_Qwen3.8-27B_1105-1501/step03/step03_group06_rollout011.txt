import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = float(np.sum(x)) - target
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(-x)
            remaining = res
            for i in idx:
                reduce = min(remaining, x[i])
                x[i] -= reduce
                remaining -= reduce
                if remaining <= 1e-12:
                    break
        else:
            idx = np.argsort(x)
            remaining = -res
            for i in idx:
                add = min(remaining, 1.0 - x[i])
                x[i] += add
                remaining -= add
                if remaining <= 1e-12:
                    break
    return x


def _grad_matrix(h):

        if d == 0:

        elif d > 0:

        else:

            m = n - g
            G[r, :m] += (1.0 - h[g:g + m])

def _topk_smooth_fun(x, alpha, K, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    nc = len(corr)
    if K < nc:
        top_idx = np.argpartition(corr, -K)[-K:]

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):

    alpha_max = 0.5

def _binary_sa(h_init, iterations, temperature, seed=42):
    """Simulated annealing on binary vectors with fixed sum."""
    rng = np.random.default_rng(seed)
    n = len(h_init)
    target = n // 2
    b = (h_init > 0.5).astype(float)
    while int(b.sum()) > target:
        idx = np.where(b == 1)[0]
        b[idx[rng.integers(len(idx))]] = 0
    while int(b.sum()) < target:
        idx = np.where(b == 0)[0]
        b[idx[rng.integers(len(idx))]] = 1

    def _energy(bin_vec):
        corr = np.correlate(bin_vec, 1.0 - bin_vec, mode='full')
        return float(np.max(corr))

    cur_e = _energy(b)
    best_b = b.copy()
    best_e = cur_e
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]

    for it in range(iterations):
        T = temperature * (0.97 ** it)
        if T < 1e-6:
            break
        ni = rng.integers(len(ones))
        nz = rng.integers(len(zeros))
        oi = ones[ni]
        oz = zeros[nz]
        if oi == oz:
            continue
        b[oi] = 0
        b[oz] = 1
        new_e = _energy(b)
        delta = new_e - cur_e
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_e = new_e
            ones[ni] = oz
            zeros[nz] = oi
            if cur_e < best_e:
                best_e = cur_e
                best_b = b.copy()
        else:
            b[oi] = 1
            b[oz] = 0
    return best_b, best_e


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _directed_perturb(h, scale=0.06):
    """Perturb h in direction that reduces the top correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_star = int(np.argmax(corr))
    G = _grad_matrix(h)
    d = -G[r_star]
    d -= np.mean(d)
    norm = float(np.linalg.norm(d))
    if norm < 1e-15:
        return h.copy()
    d = d / norm
    h_new = h + scale * d
    return _project(h_new, n / 2.0)
