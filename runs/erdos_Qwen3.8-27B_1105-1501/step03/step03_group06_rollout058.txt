def _project(v, target):

    """Try all swaps, return the best one that reduces max_corr."""
    n = len(b)
    corr = np.correlate(b, 1-b, mode='full')
    cur_max = float(np.max(corr))
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    best_delta = 0
    best_i, best_j = -1, -1
    
    for i in ones:
        for j in zeros:
            # Compute change in correlation if we swap i and j
            # At each lag s, corr_s changes by: -(1-b[i+s]) at position i, +(1-b[j+s]) at position j
            # But we need to be careful about the lag indexing.
            # np.correlate(b, 1-b, mode='full')[k] = sum_i b[i] * (1-b[i - (k-(n-1))])
            # So for lag shift d = k-(n-1): corr_d = sum_{valid i} b[i]*(1-b[i+d])

            #   For each d where i+d is valid: corr_d decreases by (1-b[i+d])
            # If we flip b[j] from 0 to 1:
            #   For each d where j+d is valid: corr_d increases by (1-b[j+d])
            
            # Net change at lag d:
            #   change_d = -(1-b[i+d]) if i+d valid, else 0
            #             + (1-b[j+d]) if j+d valid, else 0
            
            delta = np.zeros(2*n-1)
            # Effect of removing 1 at position i
            for d in range(-(n-1), n):
                if 0 <= i+d < n:
                    idx = n-1+d
                    delta[idx] -= (1 - b[i+d])
            # Effect of adding 1 at position j
            for d in range(-(n-1), n):
                if 0 <= j+d < n:
                    idx = n-1+d
                    delta[idx] += (1 - b[j+d])
            
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            if new_max < cur_max + best_delta:
                best_delta = new_max - cur_max
                best_i, best_j = i, j
    
    if best_i >= 0:
        return best_i, best_j, cur_max + best_delta
    return -1, -1, cur_max

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _max_corr_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(-x)
            for idx in order:
                if x[idx] >= 1.0 - 1e-12:
                    continue
                add = min(residual, 1.0 - x[idx])
                x[idx] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            order = np.argsort(x)
            for idx in order:
                if x[idx] <= 1e-12:
                    continue
                rem = min(-residual, x[idx])
                x[idx] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    return x

def _compute_effect(b):
    """Compute effect matrix E where E[p, n-1+d] = 1-b[p+d] if valid."""
    n = len(b)
    E = np.zeros((n, 2*n-1))
    for p in range(n):
        d_lo = max(0, -(p))  # p+d >= 0 => d >= -p
        d_hi = min(n-1, n-1-p)  # p+d <= n-1 => d <= n-1-p
        for d in range(d_lo, d_hi+1):
            E[p, n-1+d] = 1.0 - b[p+d]
    return E

def _binary_sa(b, deadline, seed=42):
    """Simulated annealing on binary vector."""
    rng = np.random.default_rng(seed)
    n = len(b)
    b = b.copy().astype(float)
    corr = np.correlate(b, 1-b, mode='full')
    cur_max = float(np.max(corr))
    best_b = b.copy()
    best_max = cur_max
    
    n_iter = 50000
    T_start = 3.0
    T_end = 0.05
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if best_max <= 12.0 + 1e-9:
            break
        
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_corr = np.correlate(b, 1-b, mode='full')
        new_max = float(np.max(new_corr))
        
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_max

def _binary_best_swap(b, deadline, max_passes=2000):
    """Best-improvement local search on binary vector."""
    n = len(b)
    b = b.copy().astype(float)
    cur_max = _max_corr_raw(b)
    best_b = b.copy()
    best_max = cur_max
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if best_max <= 12.0 + 1e-9:
            break
        
        corr = np.correlate(b, 1-b, mode='full')
        E = _compute_effect(b)
        
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        
        best_val = cur_max
        best_i, best_j = -1, -1
        
        for i in ones:
            base = corr - E[i]
            for j in zeros:
                nm = float(np.max(base + E[j]))
                if nm < best_val - 1e-12:
                    best_val = nm
                    best_i, best_j = i, j
        
        if best_i < 0:
            break
        
        b[best_i] = 0.0
        b[best_j] = 1.0
        cur_max = best_val
        if cur_max < best_max:
            best_max = cur_max
            best_b = b.copy()
    
    return best_b, best_max

def _binary_search(b_init, deadline, rng, n_tries=5):
    """Run binary search from multiple restarts."""
    n = len(b_init)
    target = n // 2
    best_b = b_init.copy().astype(float)
    best_max = _max_corr_raw(best_b)
    
    for attempt in range(n_tries):
        if time.time() > deadline:
            break
        
        # SA phase
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.5 / (n_tries - attempt))
        b_sa, max_sa = _binary_sa(b_init, sub_dl, seed=rng.integers(100000))
        if max_sa < best_max:
            best_max = max_sa
            best_b = b_sa.copy()
        
        # Best-swap phase
        b_bs, max_bs = _binary_best_swap(b_sa, deadline, max_passes=500)
        if max_bs < best_max:
            best_max = max_bs
            best_b = b_bs.copy()
        
        # SA from best found
        b_sa2, max_sa2 = _binary_sa(best_b, deadline, seed=rng.integers(100000))
        if max_sa2 < best_max:
            best_max = max_sa2
            best_b = b_sa2.copy()

        b_bs2, max_bs2 = _binary_best_swap(best_b, deadline, max_passes=300)
        if max_bs2 < best_max:
            best_max = max_bs2
            best_b = b_bs2.copy()
        
        if best_max <= 12.0 + 1e-9:
            break
    
    return best_b, best_max

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 10):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.06
    
    return best_h, best_c5

def _smooth_fun(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g

def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha,),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _max_corr_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(-x)
            for idx in order:
                if x[idx] >= 1.0 - 1e-12:
                    continue
                add = min(residual, 1.0 - x[idx])
                x[idx] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            order = np.argsort(x)
            for idx in order:
                if x[idx] <= 1e-12:
                    continue
                rem = min(-residual, x[idx])
                x[idx] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    return x


def _compute_effect(b):
    n = len(b)
    E = np.zeros((n, 2 * n - 1))
    for p in range(n):
        d_lo = max(0, -p)
        d_hi = min(n - 1, n - 1 - p)
        for d in range(d_lo, d_hi + 1):
            E[p, n - 1 + d] = 1.0 - b[p + d]
    return E


def _binary_sa(b, deadline, seed=42, n_iter=60000):
    rng = np.random.default_rng(seed)
    n = len(b)
    b = b.copy().astype(float)
    cur_max = _max_corr_raw(b)
    best_b = b.copy()
    best_max = cur_max
    T_start = 3.0
    T_end = 0.03
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if best_max <= 12.0 + 1e-9:
            break
        T = T_start * (T_end / T_start) ** (it / n_iter)
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        new_max = _max_corr_raw(b)
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_max


def _binary_best_swap(b, deadline, max_passes=3000):
    n = len(b)
    b = b.copy().astype(float)
    cur_max = _max_corr_raw(b)
    best_b = b.copy()
    best_max = cur_max
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if best_max <= 12.0 + 1e-9:
            break
        corr = np.correlate(b, 1.0 - b, mode='full')
        E = _compute_effect(b)
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        best_val = cur_max
        best_i, best_j = -1, -1
        for i in ones:
            base = corr - E[i]
            for j in zeros:
                nm = float(np.max(base + E[j]))
                if nm < best_val - 1e-12:
                    best_val = nm
                    best_i, best_j = i, j
        if best_i < 0:
            break
        b[best_i] = 0.0
        b[best_j] = 1.0
        cur_max = best_val
        if cur_max < best_max:
            best_max = cur_max
            best_b = b.copy()
    return best_b, best_max


def _binary_search_from(b_init, deadline, rng, n_rounds=3):
    n = len(b_init)
    best_b = b_init.copy().astype(float)
    best_max = _max_corr_raw(best_b)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        if best_max <= 12.0 + 1e-9:
            break
        sub_dl = min(deadline, time.time() + max(5.0, (deadline - time.time()) * 0.4))
        b_sa, m_sa = _binary_sa(best_b, sub_dl, seed=int(rng.integers(2**31)))
        if m_sa < best_max:
            best_max, best_b = m_sa, b_sa.copy()
        if time.time() > deadline:
            break
        b_bs, m_bs = _binary_best_swap(best_b, deadline, max_passes=800)
        if m_bs < best_max:
            best_max, best_b = m_bs, b_bs.copy()
    return best_b, best_max


def _make_binary_init(h0, n, rng):
    density = n // 2
    b = (h0 > 0.5).astype(float)
    if len(b) != n:
        b = np.zeros(n)
        idx = rng.permutation(n)
        b[idx[:density]] = 1.0
    while b.sum() > density:
        idx = np.where(b == 1)[0]
        b[idx[rng.integers(len(idx))]] = 0
    while b.sum() < density:
        idx = np.where(b == 0)[0]
        b[idx[rng.integers(len(idx))]] = 1
    return b


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.2)
        hi = best_a + alpha_max * 0.2
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(12):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.72
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5


def _smooth_fun(x, alpha):
    n = len(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    exp_vals = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
    w = exp_vals / float(np.sum(exp_vals))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_step(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_fun, h, args=(alpha,),
                       jac=True, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                                'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
