def _trust_region_step(h, rho, solver=None):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = dx * np.correlate(h, 1.0 - h, mode='full')
    
    # Build gradient matrix G
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d  # q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    
    # Solve trust region
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        F + G @ (x - h) <= t,
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        cp.norm(x - h) <= rho,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, max_time=10)
    except:
        return None
    if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
        return np.array(x.value).flatten()
    return None

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        for p in range(max(0, -d), min(n, n - d)):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])

def _trust_lp_direction(h, rho):
    """Solve LP: min t s.t. F + G(x-h) <= t, bounds on x, sum(x)=n/2.
    Returns direction (x - h) or None."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = dx * _scores(h)
    G = _build_G(h)
    
    # Variables: x (n) and t (1), total n+1
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: F_r + G[r,:]@(x-h) <= t for each r

    # => G[r,:]@x - t <= -F_r + G[r,:]@(h)... wait
    # F + G@(x-h) <= t
    # F + G@x - G@h <= t

    # But G@h should approximately equal scores(h)/dx... let me check.
    # Actually G@h = gradient of (dx*scores) at h evaluated at h = dx * (derivative of scores at h) * ... 
    # No, G is the Jacobian of (dx*scores) with respect to h. So G@h is not scores.
    # Let me just compute the RHS directly.
    # F + G@(x-h) <= t
    # G@x - t <= G@h - F
    # G@x - t <= G@h - dx*correlate(h, 1-h, 'full')
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F  # = G@h - F
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []

        hi_i = min(1.0, h[i] + rho)
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))  # t is free (but implicitly >= 0 from constraints)

        if c5_t < best_c5 - 1e-13:

            best_a = a

        return 0.0, cur_c5
    lo = max(0.0, best_a - alpha_max / n_steps)
    hi = min(alpha_max, best_a + alpha_max / n_steps)

        return best_a, best_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0

def _smooth_grad(h, beta):
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    exp_vals = np.exp(np.clip(beta * (scores - max_s), -500, 0))
    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / w_sum
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] += -wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] += -wd * h[g:]
    return grad

        if no_improve > 1500:

        if not ones or not zeros:

        best_swap = None
        best_val = cur_c5
        batch = min(20, len(ones_list) * len(zeros_list))
        for _b in range(batch):

            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
