import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            mc = n - d
            if mc > 0:
                G[r, :mc] += (1.0 - h[d:d + mc])
                G[r, d:] += -h[:mc]
        else:
            g = -d
            mc = n - g
            if mc > 0:
                G[r, g:] += (1.0 - h[:mc])
                G[r, :mc] += -h[g:g + mc]
    return G * dx


def _line_search(h, p, max_alpha=None):
    n = len(h)
    if max_alpha is None:
        alpha_max = 0.8
    else:
        alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.999)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.999)
    alpha_max = max(alpha_max, 1e-14)
    n_samp = 25
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = float(alphas[max(0, best_i - 1)])
    hi = float(alphas[min(n_samp - 1, best_i + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(38):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_step(h, G, scores, idx, radius):
    n = len(h)
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = -1.0
    A_ub = np.zeros((len(idx), n_vars))
    A_ub[:, :n] = G[idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_optimize(h, deadline, radius_init=0.1, max_iters=5000, top_k_mode='cycle'):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_lags = 2 * n - 1
    top_k_cycle = [None, 12, 6, None, 8, 4] if top_k_mode == 'cycle' else [top_k_mode]

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_cycle[it % len(top_k_cycle)]
        scores = _scores(h)
        if tk is not None and tk < n_lags:
            idx = np.argsort(scores)[-tk:]
        else:
            idx = np.arange(n_lags)
        G = _grad_matrix(h)
        d = _lp_step(h, G, scores, idx, radius)
        if d is None or float(np.max(np.abs(d))) < 1e-14:
            stall += 1
            radius *= 0.75
            continue
        alpha, new_c5 = _line_search(h, d)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.4)
        else:
            stall += 1
            radius *= 0.8
            if stall > 30 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5


def _pnorm_descent(h, deadline, p_start=4.0, p_end=300.0, n_steps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p_val = p_start * (p_end / p_start) ** frac
        G = _grad_matrix(h)
        scores = _scores(h)
        eps = 1e-14
        sc = scores + eps
        w = sc ** (p_val - 1.0)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(sc))
            w[np.argmax(sc)] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        alpha = 0.08
        improved = False
        for _ in range(50):
            h_new = _project(h + alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
    return best_h, best_c5


def _active_equalize(h, deadline, max_iters=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        n_act = np.sum(scores >= max_s - 1e-8)
        n_act = max(n_act, 3)
        n_act = min(int(n_act), 2 * n - 1)
        G = _grad_matrix(h)
        idx = np.argsort(scores)[-n_act:]
        G_a = G[idx]
        d = -G_a.sum(axis=0)
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        alpha, c5_try = _line_search(h, d, max_alpha=0.3)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def _structured_perturb(h, rng):
    n = len(h)
    x = np.arange(n, dtype=float) / n * 2.0
    n_freqs = int(rng.integers(1, 4))
    freqs = [float(rng.choice([0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 6.0, 8.0]))
             for _ in range(n_freqs)]
    pert = np.zeros(n)
    for f in freqs:
        amp = rng.uniform(0.01, 0.06)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        pert += amp * np.sin(2.0 * np.pi * f * x + phase)
    h_new = _project(h + pert, n / 2.0)
    return h_new


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                    if a_max < 1e-10:

                    lo, hi = 0.0, a_max
