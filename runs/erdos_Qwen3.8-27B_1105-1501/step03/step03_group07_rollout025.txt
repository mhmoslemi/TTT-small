def _binary_sa(b, deadline, n_restarts=5, T_init=0.01):
    n = len(b)
    W = n // 2
    best_b = b.copy()
    best_c = _binary_c5(b)
    rng = np.random.default_rng(123)
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        T = T_init
        cur_b = b.copy()
        cur_c = best_c
        # Maybe start from a perturbed version
        if restart > 0:
            for _ in range(n // 4):
                i = rng.integers(len(cur_b))
                j = rng.integers(len(cur_b))
                # swap if different values
                if cur_b[i] != cur_b[j]:
                    cur_b[i], cur_b[j] = cur_b[j], cur_b[i]

            zeros = np.where(cur_b == 0)[0].copy()

            oi = rng.integers(len(ones))
            oz = rng.integers(len(zeros))

            j = zeros[oz]

            cur_b[j] = 1

                ones[oi] = j
                zeros[oz] = i
                if cur_c < best_c:

                    best_b = cur_b.copy()
                    stuck = 0
            else:

                cur_b[j] = 0

            T *= 0.9995

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binarize(h, n):
    W = n // 2
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:W]] = 1.0
    return b

    t = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    return t

def _binary_c5(b):
    n = len(b)
    dx = 2.0 / n
    max_c = 0
    for d in range(1, n):
        m = n - d
        c1 = int(np.sum((b[:m] == 0) & (b[d:] == 1)))
        c2 = int(np.sum((b[:m] == 1) & (b[d:] == 0)))
        if c1 > max_c: max_c = c1
        if c2 > max_c: max_c = c2
    return max_c * dx

def _binary_lag_max(b):
    """Return (max_count, list of lags achieving max)"""
    n = len(b)
    max_c = 0
    lags = []
    for d in range(1, n):
        m = n - d
        c1 = int(np.sum((b[:m] == 0) & (b[d:] == 1)))
        c2 = int(np.sum((b[:m] == 1) & (b[d:] == 0)))
        if c1 > max_c:
            max_c = c1; lags = [d]
        elif c1 == max_c:
            lags.append(d)
        if c2 > max_c:
            max_c = c2; lags = [-d]
        elif c2 == max_c:
            lags.append(-d)
    return max_c, lags

def _greedy_binary(b, deadline, max_iters=500):
    """Greedy: find best single swap, apply, repeat"""
    n = len(b)
    b = b.copy()
    cur_c = _binary_c5(b)
    best_b = b.copy()
    best_c = cur_c
    ones = np.where(b == 1)[0].copy()
    zeros = np.where(b == 0)[0].copy()
    rng = np.random.default_rng(7)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        best_swap = None
        best_swap_c = cur_c
        # Try a subset of swaps
        tried = 0
        max_tries = min(200, len(ones) * len(zeros))
        while tried < max_tries:
            oi = rng.integers(len(ones))
            oz = rng.integers(len(zeros))
            i = ones[oi]; j = zeros[oz]
            b[i] = 0; b[j] = 1
            c = _binary_c5(b)
            b[i] = 1; b[j] = 0
            tried += 1
            if c < best_swap_c:
                best_swap_c = c
                best_swap = (oi, oz)
        if best_swap is None:
            break
        oi, oz = best_swap
        i = ones[oi]; j = zeros[oz]
        b[i] = 0; b[j] = 1
        ones[oi] = j; zeros[oz] = i
        cur_c = best_swap_c
        if cur_c < best_c:
            best_c = cur_c
            best_b = b.copy()
    
    return best_b, best_c

def _binary_sa(b, deadline, rng, T_init=0.03, cool=0.9997, max_stall=5000):
    n = len(b)
    b = b.copy()
    cur_c = _binary_c5(b)
    best_b = b.copy()
    best_c = cur_c
    ones = np.where(b == 1)[0].copy()
    zeros = np.where(b == 0)[0].copy()
    T = T_init
    stall = 0
    
    while time.time() < deadline and T > 0.002:
        oi = rng.integers(len(ones))
        oz = rng.integers(len(zeros))
        i = ones[oi]; j = zeros[oz]
        b[i] = 0; b[j] = 1
        new_c = _binary_c5(b)
        if new_c < cur_c or (new_c > cur_c and rng.random() < np.exp(-(new_c - cur_c) / T)):
            cur_c = new_c
            ones[oi] = j; zeros[oz] = i
            if cur_c < best_c:
                best_c = cur_c
                best_b = b.copy()
            stall = 0
        else:
            b[i] = 1; b[j] = 0
            stall += 1
        T *= cool
        if stall > max_stall:
            break
    return best_b, best_c

def _lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad
