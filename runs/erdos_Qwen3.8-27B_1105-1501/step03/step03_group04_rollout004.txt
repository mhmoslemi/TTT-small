import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = 0.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for d in range(-(n - 1), n):
        w = weights[n - 1 + d]
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * omh[:m]
                grad[:m] -= w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * omh[g:]
                grad[g:] -= w * h[:m]
    return grad

def _soft_descent(h, deadline, beta_start=15.0, beta_end=4000.0,
                  max_iters=50000, lr_init=0.15, momentum=0.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        mx = float(np.max(scores))
        exps = np.exp(beta * (scores - mx))
        ws = float(np.sum(exps))
        if ws < 1e-300:
            break
        weights = exps / ws
        grad = _weighted_grad(h, weights)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        grad /= gnorm
        v = momentum * v - lr * grad
        improved = False
        for a_scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = a_scale
            h_try = _project(h + v * alpha, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                if stall < 20:
                    lr = min(lr * 1.02, 0.6)
                break
        if not improved:
            stall += 1
            lr *= 0.65
            v = 0.5 * v
            if lr < 1e-7 or stall > 100:
                break
            lr = max(lr, 5e-5)
    return best_h, best_c5

def _lp_polish(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        # Build G efficiently for this h
        G = np.zeros((2 * n - 1, n))
        omh = 1.0 - h
        for d in range(-(n - 1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                if m > 0:
                    G[r, d:] = omh[:m]
                    G[r, :m] -= h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, :m] = omh[g:]
                    G[r, g:] -= h[:m]
        # LP: minimize t s.t. G@p - t <= -scores, |p_j| <= radius, sum(p)=0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success:
                stall += 1
                radius *= 0.85
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.85
            continue
        if float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        best_a = 0.0
        best_c5_try = cur_c5
        for a in np.linspace(0.1, alpha_max, 10):
            h_t = _project(h + a * p, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_try:
                best_c5_try = c5_t
                best_a = a
        if best_a > 0 and best_c5_try < cur_c5 - 1e-12:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.8
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _mutate(h, rng, mode):
    n = len(h)
    target = n / 2.0
    if mode == 0:
        return _project(h + rng.normal(0, 0.08, n), target)
    elif mode == 1:
        shift = rng.randint(1, n // 2)
        return _project(np.roll(h, shift), target)
    elif mode == 2:
        shift = rng.randint(1, n // 2)
        t = rng.uniform(0.3, 0.7)
        h2 = np.roll(h, shift)
        return _project(t * h + (1 - t) * h2, target)
    elif mode == 3:
        r = rng.standard_normal(n)
        r -= r.mean()
        scale = 0.1 * n / max(np.linalg.norm(r), 1e-10)
        return _project(h + scale * r, target)
    elif mode == 4:
        # Sparse perturbation
        idx = rng.choice(n, size=n // 4, replace=False)
        pert = np.zeros(n)
        pert[idx] = rng.normal(0, 0.3, len(idx))
        return _project(h + pert, target)
    elif mode == 5:
        # Threshold-based: push values toward 0 or 1
        h_new = h.copy()
        h_new[h_new < 0.3] *= 0.5
        h_new[h_new > 0.7] = 1.0 - (1.0 - h_new[h_new > 0.7]) * 0.5
        return _project(h_new, target)
    else:
        return _project(h + rng.normal(0, 0.05, n), target)

def _thue_morse_init(n, amplitude=0.3):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amplitude * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)
