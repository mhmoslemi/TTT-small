    g = 1.0 - h

    for r in range(2*n - 1):
        k = r - (n - 1)
        if k >= 0:

            # d/d(h[p]) = (1-h[p-k]) for p in [k, n-1] - h[p+k] for p in [0, m-1]
            # First term: for p in [k, n-1], grad = g[p-k]
            # Second term: for p in [0, m-1], grad -= h[p+k]
            # Combined: for p in [0, m-1]: grad[p] += g[p-k] - h[p+k]  (but p-k might be negative for p < k)

                idx = rng.choice(n_cur, size=n_cur // 6, replace=False)
                h_try[idx] += rng.normal(0, 0.06, len(idx))
            elif r < 0.55:
                h_try = (best_h + rng.normal(0, 0.04, n_cur) > 0.5).astype(float)
            elif r < 0.7:
                shift = rng.integers(1, max(2, n_cur // 10))
                h_try = np.roll(best_h, int(shift))
            elif r < 0.85:
                h_try = 0.5 + (best_h - 0.5) * (0.7 + 0.6 * rng.random())
            else:
                h_rand = (rng.random(n_cur) > 0.5).astype(float)
                h_try = 0.6 * best_h + 0.4 * h_rand

            h_try = _project(h_try, n_cur / 2.0)
            # Quick LSE
            sub_dl = min(dl4, time.time() + max(5.0, rem * 0.15))
            if time.time() < sub_dl:
                h_try = _solve_lse(h_try, 0.08, n_cur / 2.0, maxiter=100)
                c_try = _exact_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                if time.time() < sub_dl - 2:
                    h_try2 = _solve_lse(h_try, 0.03, n_cur / 2.0, maxiter=100)
                    c_try2 = _exact_c5(h_try2)
                    if c_try2 < best_c5:
                        best_c5 = c_try2
                        best_h = h_try2.copy()

    # Phase 5: Resolution sweep (200s)
    dl5 = t0 + min(800.0, budget_s * 0.73)
    if remaining() > 20:
        n_best = len(best_h)
        for dn in [4, -4, 8, -8, 12, -12, 16, -16, 20, -20, 24, -24]:
            if time.time() > dl5:
                break
            n_try = n_best + dn
            if n_try < 32 or n_try > 200 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl5, time.time() + min(25.0, (dl5 - time.time()) * 0.3))
            if time.time() < sub_dl:
                h_r = _solve_lse(h_r, 0.1, n_try / 2.0, maxiter=150)
                h_r = _solve_lse(h_r, 0.04, n_try / 2.0, maxiter=150)
                c_r = _exact_c5(h_r)
                if c_r < best_c5 - 1e-10:
                    best_c5 = c_r
                    best_h = h_r.copy()
                    best_n = n_try

    # Phase 6: Fine LSE polish (100s)
    dl6 = t0 + min(950.0, budget_s * 0.86)
    if remaining() > 10:
        n_cur = len(best_h)
        for tau in [0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]:
            if time.time() > dl6:
                break
            h_f = _solve_lse(best_h, tau, n_cur / 2.0, maxiter=300)
            c_f = _exact_c5(h_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()

    # Phase 7: Final epigraph full (80s)
    dl7 = t0 + min(1040.0, budget_s * 0.945)
    if remaining() > 10:
        n_cur = len(best_h)
        vals = _values(best_h) * (2.0 / n_cur)
        top_k = min(20, 2 * n_cur - 1)
        active = np.argsort(-vals)[:top_k]
        t_ub = best_c5 + 0.02
        h_fin = _solve_epigraph(best_h, active, n_cur / 2.0, t_ub, maxiter=200)
        c_fin = _exact_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()
        # Another round
        if time.time() < dl7:
            vals2 = _values(best_h) * (2.0 / n_cur)
            active2 = np.argsort(-vals2)[:top_k]
            h_fin2 = _solve_epigraph(best_h, active2, n_cur / 2.0, best_c5 + 0.01, maxiter=200)
            c_fin2 = _exact_c5(h_fin2)
            if c_fin2 < best_c5:
                best_c5 = c_fin2
                best_h = h_fin2.copy()

    # Final: short subgradient polish
    dl8 = t0 + budget_s * 0.99
    if remaining() > 5:
        h_fin, c_fin = _subgrad_refine(best_h, dl8, best_n / 2.0, T_init=0.15)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        # Something went wrong, use initial
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
