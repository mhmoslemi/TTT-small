def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    mu = 0.85
    lr_base = 0.08
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, n_iters - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores_vec(h)
        sw = scores + 1e-15
        w = sw ** (p - 1)
        G = _grad_matrix(h / 2.0 + 0.5 - 0.5)  # just _grad_matrix(h)
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        v = mu * v + lr_base * grad
        alpha = 1.0
        for _ in range(30):
            h_try = _project(h - alpha * v, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
            alpha *= 0.6
            if alpha < 1e-10:
                break
        if alpha < 1e-10:
            v *= 0.5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _population_search(h_init, deadline, pop_size=12, n_gens=2000, rng=None):
    n = len(h_init)
    target = n / 2.0
    if rng is None:
        rng = np.random.default_rng(0)
    pop = [h_init.copy()]
    cur_c5 = _exact_c5(h_init)
    for _ in range(pop_size - 1):
        hp = _project(h_init + rng.normal(0, 0.05, n), target)
        pop.append(hp)
    pop_c5 = [_exact_c5(p) for p in pop]
    best_idx = int(np.argmin(pop_c5))
    best_h = pop[best_idx].copy()
    best_c5 = pop_c5[best_idx]
    scale = 0.03
    gen = 0
    while gen < n_gens:
        if time.time() > deadline:
            break
        gen += 1
        scale = max(0.002, min(0.15, scale))
        for i in range(pop_size):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, pop_size))
            k = int(rng.integers(0, pop_size))
            if j == i or k == i:
                j = (j + 1) % pop_size
                k = (k + 2) % pop_size
            mutant = pop[i] + scale * (pop[j] - pop[k]) + rng.normal(0, scale * 0.3, n)
            mutant = _project(mutant, target)
            c5_m = _exact_c5(mutant)
            if c5_m < pop_c5[i] or rng.random() < 0.05:
                pop[i] = mutant
                pop_c5[i] = c5_m
            if c5_m < best_c5:
                best_c5 = c5_m
                best_h = mutant.copy()
                scale *= 0.9
            else:
                scale = min(0.15, scale * 1.02)
        if gen % 50 == 0 and best_c5 > 0.3805:
            refined, ref_c5 = _pair_cd(best_h, min(deadline, time.time() + 30), max_rounds=15)
            if ref_c5 < best_c5:
                best_c5 = ref_c5
                best_h = refined.copy()
                pop[0] = best_h.copy()
                pop_c5[0] = best_c5
    return best_h, best_c5

def _lp_refine(h, deadline, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _scores_vec(h)
        G = _grad_matrix(h)
        top_k = 5
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(scores_k), n + 1))
        A_ub[:, :n] = -G_k
        A_ub[:, n] = 1.0
        b_ub = np.zeros(len(scores_k))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        alphas = np.linspace(0, alpha_max, 20)
        vals = np.array([_exact_c5(h + a * p) for a in alphas])
        best_i = int(np.argmin(vals))
        if best_i > 0 and vals[best_i] < best_c5 - 1e-14:
            lo = alphas[max(0, best_i - 1)]
            hi = alphas[min(19, best_i + 1)]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
            f2 = _exact_c5(h + c2 * p)
            for _ in range(25):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(h + c1 * p)
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _exact_c5(h + c2 * diff if False else h + c2 * p)
            opt_a = (lo + hi) / 2.0
            h_try = _project(h + opt_a * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.3)
            else:
                stall += 1
                radius *= 0.75
        else:
            stall += 1
            radius *= 0.75
            if stall > 20 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    T = budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: p-norm gradient (fast, many iterations)
    dl1 = t0 + T * 0.12
    h1, c5_1 = _pnorm_descent(h, dl1, p_start=4, p_end=200, n_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refine
    dl2 = t0 + T * 0.30
    h2, c5_2 = _lp_refine(best_h, dl2, max_iters=3000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair coordinate descent (thorough local search)
    dl3 = t0 + T * 0.52
    h3, c5_3 = _pair_cd(best_h, dl3, max_rounds=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Population-based global search
    dl4 = t0 + T * 0.72
    h4, c5_4 = _population_search(best_h, dl4, pop_size=10, n_gens=3000, rng=rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(best_h)

    # Phase 5: Resolution sweep
    dl5 = t0 + T * 0.88
    for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 100 or n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + T * 0.03)
        h_r, c5_r = _pnorm_descent(h_r, sub_dl, p_start=8, p_end=100, n_iters=500)
        sub_dl2 = min(dl5, time.time() + T * 0.02)
        h_r, c5_r = _pair_cd(h_r, sub_dl2, max_rounds=30)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final LP + pair polish
    dl6 = t0 + T * 0.96
    h6, c5_6 = _lp_refine(best_h, min(dl6, time.time() + T * 0.06), max_iters=2000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    dl6b = t0 + T * 0.99
    h6b, c5_6b = _pair_cd(best_h, dl6b, max_rounds=50)
    if c5_6b < best_c5:
        best_c5 = c5_6b
        best_h = h6b.copy()

    # Final cleanup
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection increased C5 slightly, try without re-projection
        pass
    return best_h, final_c5, best_n
