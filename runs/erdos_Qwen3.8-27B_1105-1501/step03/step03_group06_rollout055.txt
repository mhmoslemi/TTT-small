    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:

    lo = float(v.min()) - 2.0

    cur_max = _binary_max_corr(b)
    best_b = b.copy()
    best_max = cur_max

    zeros = np.where(b == 0)[0]

def _slsqp_epigraph(h0, n, G, maxiter=500):
    """SLSQP optimization of the epigraph formulation."""
    x0 = np.append(h0, _raw_max(h0))
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constr_fun(x):
        h = x[:n]
        t = x[-1]
        c = G @ h
        return np.concatenate([t - c, h, 1.0 - h])
    
    def constr_jac(x):
        h = x[:n]
        # Recompute G (since h changed)
        G_cur = _grad_matrix(h)
        # Rows: -(G_cur) with +1 in last col, then identity for h, then -identity for (1-h)
        J1 = -G_cur
        J1 = np.column_stack([J1, np.ones(2*n-1)])
        J2 = np.column_stack([np.eye(n), np.zeros((n, 1))])
        J3 = np.column_stack([-np.eye(n), np.zeros((n, 1))])
        return np.vstack([J1, J2, J3])
    
    def eq_fun(x):
        return np.array([np.sum(x[:n]) - n/2.0])
    
    def eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J
    
    constraints = [
        {'type': 'ineq', 'fun': constr_fun, 'jac': constr_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

def _raw_max(h):

def _project(v, target):

        return np.zeros(n)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))

        else:
            g = -s
            m = n - g
            G[k, g:] += (1.0 - h[:m])

    return G
