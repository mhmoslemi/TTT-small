    elif d > 0:

        if m > 0:

            grad[g:] += (1.0 - h[:m])

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:

        else:

    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] += -h[:m]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += -h[g:]
            grad[g:] += (1.0 - h[:m])
    return grad

def _weighted_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if abs(w[idx]) > 1e-15:
            d = idx - (n - 1)
            grad += w[idx] * _score_grad(h, d)
    return grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, direction, deadline, n_iter=35):
    n = len(h)
    alpha_max = 0.5
    pos = direction > 1e-14
    neg = direction < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-direction[neg]))))
    alpha_max = max(alpha_max, 1e-12)
    if alpha_max < 1e-12:
        return h, _exact_c5(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, alpha_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _exact_c5(np.clip(h + c1 * direction, 0, 1))
    f2 = _exact_c5(np.clip(h + c2 * direction, 0, 1))
    for _ in range(n_iter):
        if time.time() > deadline or (b - a) < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _exact_c5(np.clip(h + c1 * direction, 0, 1))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _exact_c5(np.clip(h + c2 * direction, 0, 1))
    alpha = (a + b) / 2.0
    h_new = np.clip(h + alpha * direction, 0, 1)
    tgt = n / 2.0
    if abs(float(np.sum(h_new)) - tgt) > 1e-10:
        h_new = _project(h_new, tgt)
    return h_new, _exact_c5(h_new)

def _compute_direction(h, beta=50.0, top_k=5):
    """Compute descent direction using combination of smooth max and top-k lags."""
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    
    # Smooth weights
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -400, 0)
    w_smooth = np.exp(shifted)
    wsum = float(np.sum(w_smooth))
    if wsum < 1e-300:
        w_smooth = np.zeros(len(scores))
        w_smooth[int(np.argmax(scores))] = 1.0
    else:
        w_smooth /= wsum
    
    # Top-k weights
    idx_sorted = np.argsort(scores)[::-1]
    w_top = np.zeros(len(scores))
    k = min(top_k, len(scores))
    w_top[idx_sorted[:k]] = 1.0 / k
    
    # Blend
    w = 0.5 * w_smooth + 0.5 * w_top
    grad = _weighted_grad(h, w)
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return None
    return -grad / norm

def _local_opt(h, deadline, max_iters=200, beta_start=30.0):

def _pairwise_opt(h, deadline, passes=3):
    """Pairwise transfer optimization."""
    n = len(h)
    cur = _exact_c5(h)
    
    for _pass in range(passes):
        if time.time() > deadline:
            break
        any_improve = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                # Determine valid range
                t_hi = min(1.0 - h[i], h[j])
                t_lo = max(-h[i], h[j] - 1.0)
                if t_hi <= t_lo + 1e-12:
                    continue
                # Quick descent check at midpoint
                t_mid = (t_hi + t_lo) / 2.0
                h_try = h.copy()
                h_try[i] += t_mid
                h_try[j] -= t_mid
                c5_try = _exact_c5(h_try)
                if c5_try < cur - 1e-13:
                    # Full line search
                    direction = np.zeros(n)
                    direction[i] = 1.0
                    direction[j] = -1.0
                    h_new, c5_new = _line_search(h, direction, deadline, n_iter=25)
                    if c5_new < cur - 1e-13:
                        h, cur = h_new, c5_new
                        any_improve = True
                    else:
                        # Try reverse
                        direction[i] = -1.0
                        direction[j] = 1.0

                        if c5_new < cur - 1e-13:
                            h, cur = h_new, c5_new

            break

def _resample(h, n_new):

    x_new = (np.arange(n_new) + 0.5) / n_new

Actually, I just realized something. Let me re-examine the `_pairwise_opt` function. The issue is that for each pair (i,j), I'm doing a "quick check" at t_mid. But t_mid might not be the direction of descent. A better quick check would be to compute the directional derivative at t=0:

∂C/∂t at t=0 in the direction (e_i - e_j)

This is the difference between the gradient components at i and j. If this is negative, moving mass from j to i (i.e., t>0) decreases C. If positive, moving from i to j decreases C.

But computing the gradient requires knowing which lags are active. For a single lag d, the gradient at position k is _score_grad(h, d)[k]. The directional derivative in direction (e_i - e_j) is grad[i] - grad[j].

So a quick check would be:
1. Identify the active lag(s)
2. Compute the gradient
3. For each pair (i,j), check if grad[i] - grad[j] < 0 (move mass from j to i) or > 0 (move from i to j)
4. Only do full line search for pairs with significant descent

This is much faster than evaluating C at t_mid for each pair.

        # We want this to be negative for descent, so we want grad[i] - grad[j] < 0
        # i.e., grad[j] > grad[i], meaning move mass from i to j

        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _exact_c5(np.clip(h + c2 * direction, 0, 1))
    alpha = (a + b) / 2.0
    h_new = np.clip(h + alpha * direction, 0, 1)
    tgt = n / 2.0
    if abs(float(np.sum(h_new)) - tgt) > 1e-10:
        h_new = _project(h_new, tgt)
    return h_new, _exact_c5(h_new)

def _compute_direction(h, beta=50.0, top_k=5):
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    shifted = np.clip(beta * (scores - max_s), -400, 0)
    w_s = np.exp(shifted)
    ws = float(np.sum(w_s))
    if ws < 1e-300:
        w_s = np.zeros(len(scores))
        w_s[int(np.argmax(scores))] = 1.0
    else:
        w_s /= ws
    idx_sorted = np.argsort(scores)[::-1]
    w_t = np.zeros(len(scores))
    k = min(top_k, len(scores))
    w_t[idx_sorted[:k]] = 1.0 / k
    w = 0.5 * w_s + 0.5 * w_t
    grad = _weighted_grad(h, w)
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return None
    return -grad / norm

def _local_opt(h, deadline, max_iters=200, beta_start=30.0):
    n = len(h)
    cur = _exact_c5(h)
    beta = beta_start
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline or stall > 50:
            break
        improved = False
        for b in [10.0, beta, beta * 2.0, min(400.0, beta * 4.0)]:
            d = _compute_direction(h, beta=b, top_k=5)
            if d is None:
                continue
            h_new, c5_new = _line_search(h, d, deadline, n_iter=30)
            if c5_new < cur - 1e-13:
                h, cur, improved = h_new, c5_new, True
                beta = b
                break
        if not improved:
            scores = _all_scores(h)
            max_s = float(np.max(scores))
            active = scores >= max_s - 1e-5
            w = np.zeros(len(scores))
            w[active] = 1.0 / max(1, int(np.sum(active)))
            grad = _weighted_grad(h, w)
            grad -= np.mean(grad)
            norm = float(np.max(np.abs(grad)))
            if norm > 1e-15:
                d = -grad / norm
                h_new, c5_new = _line_search(h, d, deadline, n_iter=30)
                if c5_new < cur - 1e-13:
                    h, cur, improved = h_new, c5_new, True
        if not improved:
            for b in [5.0, 8.0, 150.0]:
                d = _compute_direction(h, beta=b, top_k=10)
                if d is not None:
                    h_new, c5_new = _line_search(h, d, deadline, n_iter=20)
                    if c5_new < cur - 1e-13:
                        h, cur, improved = h_new, c5_new, True
                        break
        if improved:
            stall = 0
        else:
            stall += 1
            beta = max(10.0, beta * 0.85)
    return h, cur

def _pairwise_opt(h, deadline, passes=3):
    n = len(h)
    cur = _exact_c5(h)
    top_k = 5
    for _pass in range(passes):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        idx_sorted = np.argsort(scores)[::-1]
        w = np.zeros(len(scores))
        for idx in idx_sorted[:top_k]:
            if scores[idx] >= max_s - 1e-3 * max(1.0, max_s):
                w[idx] = 1.0
        ws = float(np.sum(w))
        if ws < 1e-15:
            w[idx_sorted[0]] = 1.0
            ws = 1.0
        w /= ws
        grad = _weighted_grad(h, w)
        any_improve = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                dd = grad[i] - grad[j]
                if abs(dd) < 1e-8:
                    continue
                direction = np.zeros(n)
                if dd > 0:
                    direction[i] = -1.0
                    direction[j] = 1.0
                else:
                    direction[i] = 1.0
                    direction[j] = -1.0
                h_new, c5_new = _line_search(h, direction, deadline, n_iter=20)
                if c5_new < cur - 1e-13:
                    h, cur = h_new, c5_new
                    any_improve = True
                    scores = _all_scores(h)
                    max_s = float(np.max(scores))
                    idx_sorted = np.argsort(scores)[::-1]
                    w = np.zeros(len(scores))
                    for idx in idx_sorted[:top_k]:
                        if scores[idx] >= max_s - 1e-3 * max(1.0, max_s):
                            w[idx] = 1.0
                    ws = float(np.sum(w))
                    if ws < 1e-15:
                        w[idx_sorted[0]] = 1.0
                        ws = 1.0
                    w /= ws
                    grad = _weighted_grad(h, w)
        if not any_improve:
            break
    return h, cur

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)

def _structured_perturb(h, rng, scale):
    n = len(h)
    n_harms = int(rng.integers(2, 7))
    pert = np.zeros(n)
    for _ in range(n_harms):
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        pert += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    mx = float(np.max(np.abs(pert)))
    if mx > 1e-15:
        pert /= mx
    return h + scale * pert

def _slsqqp_polish(h, deadline, top_k=12, maxiter=200, ftol=1e-12):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    z0 = float(np.max(scores)) + 1e-5
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
    cons_list = [{'type': 'eq', 'fun': _sls_eq_fun, 'jac': _sls_eq_jac}]
    for d in active_lags:
        cons_list.append({'type': 'ineq', 'fun': _sls_ineq_fun, 'jac': _sls_ineq_jac, 'args': (d,)})
    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    try:
        result = minimize(_sls_obj, x0, method='SLSQP', jac=_sls_jac,
                          bounds=bounds, constraints=cons_list,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return h, _exact_c5(h)
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _sls_obj(x):
    return x[len(x) - 1]

def _sls_jac(x):
    n = len(x)
    g = np.zeros(n)
    g[n - 1] = 1.0
    return g

def _sls_eq_fun(x):
    n = len(x) - 1
    return float(np.sum(x[:n])) - n / 2.0

def _sls_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _sls_ineq_fun
