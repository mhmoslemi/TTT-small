def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(v, -10, 10)  # prevent overflow in binary search
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    res = target - np.sum(x)

        else:
            mask = x > 0.0
        cnt = np.sum(mask)

            x = np.clip(x, 0.0, 1.0)
    return x

def exact_c5(h):

def project(v, target):

            hi = mid

    res = target - np.sum(x)

        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def scores_raw(h):

def soft_gradient(h, tau):

            if P is not None:
                dirs.append(P)
        for _ in range(6):
            P = gen_rand_dir(rng, n_cur)
            if P is not None:
                dirs.append(P)
        for _ in range(3):
            P = gen_struct_dir(h_cur, rng, n_cur)
            if P is not None:
                dirs.append(P)
        improved = False
        for P in dirs:
            if time.time() > d2:
                break
            P = P - np.mean(P)
            t_best, v_best = line_search(h_cur, P)
            if t_best != 0.0 and v_best is not None:
                h_new = project(h_cur + t_best * P, n_cur / 2.0)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_cur = h_new.copy()
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
            if stall % 25 == 0:
                # Perturbation escape
                h_pert = h_cur + rng.normal(0, 0.04, n_cur)
                h_pert = project(h_pert, n_cur / 2.0)
                h_pert, c5_p = optimize_block(h_pert, min(d2, time.time() + budget_s * 0.015), rng, max_iters=15)
                if c5_p < best_c5 - 1e-12:
                    best_h = h_pert.copy()
                    best_c5 = c5_p
                    h_cur = h_pert.copy()
                    stall = 0

    # Phase 3: Multi-resolution refinement (next 10%)
    d3 = t0 + budget_s * 0.80
    tried = set()
    for dn in list(range(-8, 9, 2)):
        if time.time() > d3:
            break
        nt = best_n + dn
        if nt < 40 or nt > 160 or nt in tried:
            continue
        tried.add(nt)
        h_r = resample_fn(best_h, nt)
        sub_dl = min(d3, time.time() + budget_s * 0.01)
        h_r, c5_r = optimize_block(h_r, sub_dl, rng, max_iters=40)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt

    # Phase 4: Final polish (last 20%)
    d4 = t0 + budget_s * 0.98
    h_cur = best_h.copy()
    n_cur = best_n
    h_cur, c5_cur = optimize_block(h_cur, d4, rng, max_iters=100)
    if c5_cur < best_c5 - 1e-14:
        best_h = h_cur.copy()
        best_c5 = c5_cur

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
