def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair_box_sum(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using bisection water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _feasible_interval(h, v):
    """Compute feasible alpha interval for h + alpha*v with 0<=x<=1."""
    n = len(h)
    lo = -1e10
    hi = 1e10

            hi = min(hi, (1.0 - h[i]) / v[i])

            hi = min(hi, -h[i] / v[i])

def _grad_direction(h, beta):
    """Compute smoothed max gradient direction."""
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    exp_vals = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / w_sum
    
    # Build gradient: dS_d/dh_i
    # S_d = sum_p h[p+d]*(1-h[p]) for appropriate range
    # dS_d/dh_i: for term h[p+d]*(1-h[p]), if i=p+d: coefficient (1-h[p])
    #             if i=p: coefficient -h[p+d]
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            grad[q] += wd * (1.0 - h[p])
            grad[p] -= wd * h[q]
    return grad

def _grad_direction(h, beta):
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    ev = np.exp(beta * (scores - max_s))
    ws = float(np.sum(ev))
    if ws < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = ev / ws
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            grad[q] += wd * (1.0 - h[p])
            grad[p] -= wd * h[q]
    return grad

    # Phase 1: Heavy local line descent
    d1 = min(t0 + budget_s * 0.28, hard_dead)
    h1, c5_1 = _local_descend(h, d1, rng, max_iters=20000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary local search from current best
    d2 = min(t0 + budget_s * 0.40, hard_dead)
    hb, c5b = _binary_swap_search(_binary_snap(best_h), d2, rng, max_iters=60000)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 3: Refine binary result fractionally
    if c5b < _exact_c5(h0):
        d3 = min(t0 + budget_s * 0.50, hard_dead)
        h3, c5_3 = _local_descend(best_h, d3, rng, max_iters=10000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Transformations + local descent
    d4 = min(t0 + budget_s * 0.65, hard_dead)
    transforms = []
    transforms.append(best_h[::-1].copy())
    for k in [1, 2, 3, -1, -2, n0 // 4]:
        transforms.append(np.roll(best_h, k).copy())
    transforms.append(_binary_snap(best_h))
    for t_h in transforms:
        if time.time() > d4:
            break
        t_h = _repair_box_sum(t_h, len(t_h) / 2.0)
        per_t = max(5.0, (d4 - time.time()) / max(1, len(transforms)))
        d_t = min(d4, time.time() + per_t)
        h_t, c5_t = _local_descend(t_h, d_t, rng, max_iters=3000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 5: Multi-resolution search
    d5 = min(t0 + budget_s * 0.82, hard_dead)
    n_tries = []
    for dn in [-8, -4, 4, 8, 16, -16, -24, 24, 32, -32]:
        nn = best_n + dn
        if 32 <= nn <= 192:
            n_tries.append(nn)
    n_tries = sorted(set(n_tries))
    if time.time() < d5:
        per_n = max(8.0, (d5 - time.time()) / max(1, len(n_tries)))
        for n_try in n_tries:
            if time.time() > d5:
                break
            d_r = min(d5, time.time() + per_n)
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _local_descend(h_r, d_r, rng, max_iters=2000)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 6: Random restarts
    d6 = min(t0 + budget_s * 0.90, hard_dead)
    n_restart = 5
    for r in range(n_restart):
        if time.time() > d6:
            break
        scale = 0.2 + 0.15 * r
        pert = best_h + scale * (rng.standard_normal(best_n) - 0.5)
        pert = _repair_box_sum(pert, best_n / 2.0)
        per_r = max(5.0, (d6 - time.time()) / max(1, n_restart - r))
        d_r = min(d6, time.time() + per_r)
        h_p, c5_p = _local_descend(pert, d_r, rng, max_iters=1500)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 7: Final binary search + fractional polish
    d7 = min(t0 + budget_s * 0.95, hard_dead)
    hb2, c5b2 = _binary_swap_search(_binary_snap(best_h), d7, rng, max_iters=40000)
    if c5b2 < best_c5:
        best_c5 = c5b2
        best_h = hb2.copy()

    # Phase 8: Final local descent
    if time.time() < hard_dead:
        h8, c5_8 = _local_descend(best_h, hard_dead, rng, max_iters=5000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _repair_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
