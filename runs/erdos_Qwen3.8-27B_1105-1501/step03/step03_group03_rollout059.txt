def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _active_direction(h, band_frac=0.1):
    """Compute subgradient direction from active scores"""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')

    if s_max < 1e-10:
        return None

    active = np.where(scores >= threshold)[0]
    
    if len(active) == 0:
        active = np.where(scores == s_max)[0]
    
    # Compute gradient of each active score
    # Score at position r corresponds to lag d = r - (n-1)
    # For lag d >= 0: s = sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
    #   d s / d h[j] = (1-h[j-d]) if j >= d and j-d < n-d, i.e., j in [d, n-1] and j-d in [0, n-d-1]

    #   More carefully:
    #   s_d = sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
    #   d s_d / d h[j]:
    #     If j appears as h[i+d] for some i: i = j-d, need 0 <= j-d < n-d, i.e., d <= j <= n-1
    #       contribution: (1-h[j-d])
    #     If j appears as h[i] for some i: i = j, need 0 <= j < n-d
    #       contribution: -h[j+d]
    #     Both can happen simultaneously.
    
    grad = np.zeros(n)
    n_scores = len(scores)  # 2n-1
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            # For i in [0, m-1]: term is h[i+d] * (1-h[i])
            # d s / d h[j]:
            #   j = i+d for i in [0,m-1] => j in [d, d+m-1] = [d, n-1], contribution (1-h[i]) = (1-h[j-d])
            #   j = i for i in [0,m-1] => j in [0, m-1] = [0, n-d-1], contribution -h[i+d] = -h[j+d]
            for j in range(d, n):
                grad[j] += (1.0 - h[j - d])
            for j in range(m):
                grad[j] -= h[j + d]
        else:

            m = n - g
            if m <= 0:
                continue
            # s_{-g} = sum_{i=0}^{m-1} h[i]*(1-h[i+g])
            # d s / d h[j]:
            #   j = i for i in [0,m-1] => j in [0, m-1] = [0, n-g-1], contribution (1-h[j+g])
            #   j = i+g for i in [0,m-1] => j in [g, g+m-1] = [g, n-1], contribution -h[j-g]
            for j in range(m):
                grad[j] += (1.0 - h[j + g])
            for j in range(g, n):
                grad[j] -= h[j - g]
    
    # Normalize: subtract mean to make zero-sum, normalize
    grad -= np.mean(grad)
    norm = float(np.sqrt(np.sum(grad**2)))
    if norm < 1e-12:
        return None
    return -grad / norm  # negative gradient = descent direction

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < (1.0 - 1e-12)
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _feasible_interval(h, d):
    t_min = -1e10
    t_max = 1e10
    eps = 1e-14
    for i in range(len(h)):

            t_min = max(t_min, -h[i] / d[i])
            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -eps:
            t_min = max(t_min, (1.0 - h[i]) / d[i])
            t_max = min(t_max, -h[i] / d[i])

def _active_gradient(h, band_frac=0.12):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    s_max = float(np.max(scores))
    if s_max < 1e-10:
        return None
    threshold = s_max * (1.0 - band_frac)
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    grad = np.zeros(n)
    for r in active:
        d = int(r) - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            for j in range(d, n):
                grad[j] += (1.0 - h[j - d])
            for j in range(m):
                grad[j] -= h[j + d]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            for j in range(m):
                grad[j] += (1.0 - h[j + g])
            for j in range(g, n):
                grad[j] -= h[j - g]
    grad -= np.mean(grad)
    norm = float(np.sqrt(np.sum(grad ** 2)))
    if norm < 1e-12:
        return None
    return -grad / norm


def _random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.sqrt(np.sum(d ** 2)))
    if norm < 1e-12:
        return None
    return d / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quick_pair_scan(h, c5_cur, n_samples=6):

        h_new, c5_new, accepted = _line_search_general(h, d, best_c5, n_samples=25, golden_iters=22)
        if accepted:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            d2 = _random_direction(n, rng)
            if d2 is not None:
                h_new2, c5_new2, acc2 = _line_search_general(h, d2, best_c5, n_samples=20, golden_iters=18)
                if acc2:
                    h = h_new2
                    best_c5 = c5_new2
                    best_h = h.copy()
                    stall = 0
                    continue
            # Try with different band
            d3 = _active_gradient(h, band_frac=0.3)
            if d3 is not None:
                h_new3, c5_new3, acc3 = _line_search_general(h, d3, best_c5, n_samples=20, golden_iters=18)
                if acc3:
                    h = h_new3
                    best_c5 = c5_new3
                    best_h = h.copy()
                    stall = 0

    # Phase 2: Systematic pair exchange (35% time)
    deadline2 = t0 + budget_s * 0.55
    no_improve = 0
    while time.time() < deadline2 and no_improve < 8:
        no_improve += 1
        cands = _quick_pair_scan(h, best_c5, n_samples=6)
        if not cands:
            break
        top_k = min(20, len(cands))
        improved_this_round = False
        for i in range(top_k):
            if time.time() > deadline2:
                break
            _, p, q = cands[i]
            h_new, c5_new, accepted = _line_search_pair(h, p, q, best_c5, n_samples=20, golden_iters=25)
            if accepted:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved_this_round = True
                no_improve = 0
        if not improved_this_round and no_improve >= 3:
            break

    # Phase 3: Random + active directions with jumps (15% time)
    deadline3 = t0 + budget_s * 0.70
    stall3 = 0
    while time.time() < deadline3 and stall3 < 60:
        # Random jump
        jump = rng.uniform(0.01, 0.06)
        d = _random_direction(n, rng)
        if d is None:
            break
        h_try = _project(h + jump * d, n / 2.0)
        # Line search from jumped position
        d2 = _active_gradient(h_try, band_frac=0.15)
        if d2 is not None:
            h_new, c5_new, accepted = _line_search_general(h_try, d2, best_c5, n_samples=20, golden_iters=20)
            if accepted:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stall3 = 0
                continue
        # Try directly from current with active direction
        d3 = _active_gradient(h, band_frac=0.08)
        if d3 is not None:
            h_new3, c5_new3, acc3 = _line_search_general(h, d3, best_c5, n_samples=25, golden_iters=22)
            if acc3:
                h = h_new3
                best_c5 = c5_new3
                best_h = h.copy()
                stall3 = 0
                continue
        stall3 += 1

    # Phase 4: Resolution changes (15% time)
    deadline4 = t0 + budget_s * 0.85
    for dn in [1, -1, 2, -2, 3, -3, 4, -4, -6, 6]:
        if time.time() > deadline4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        # Quick local optimization on new resolution
        d_act = _active_gradient(h_r, band_frac=0.12)
        if d_act is not None:
            h_r, c5_r, _ = _line_search_general(h_r, d_act, _exact_c5(h_r), n_samples=20, golden_iters=18)
        # Quick pair scan
        cands_r = _quick_pair_scan(h_r, _exact_c5(h_r), n_samples=5)
        for i in range(min(10, len(cands_r))):
            if time.time() > deadline4:
                break
            _, p, q = cands_r[i]
            h_r, c5_r, _ = _line_search_pair(h_r, p, q, _exact_c5(h_r), n_samples=15, golden_iters=18)
        c5_final = _exact_c5(h_r)
        if c5_final < best_c5 - 1e-9:
            best_c5 = c5_final
            best_h = h_r.copy()
            best_n = n_try
            h = h_r.copy()

    # Phase 5: Final polish (remaining time)
    deadline5 = hard_deadline
    stall5 = 0
    while time.time() < deadline5 and stall5 < 15:
        # Pair exchange on most "active" variables
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        # Find which variables most influence active scores
        influence = np.zeros(n)
        for r in range(len(scores)):
            if scores[r] >= s_max * 0.85:
                d = r - (n - 1)
                if d >= 0:
                    m = n - d
                    influence[d:] += 1.0
                    influence[:m] += 1.0
                else:
                    g = -d
                    m = n - g
                    influence[:m] += 1.0
                    influence[g:] += 1.0
        top_vars = np.argsort(-np.abs(influence))[:min(24, n)]
        top_vars_list = top_vars.tolist()
        improved = False
        for i_len in range(len(top_vars_list)):
            if time.time() > deadline5:
                break
            p = top_vars_list[i_len]
            for j_len in range(i_len + 1, len(top_vars_list)):
                if time.time() > deadline5:
                    break
                q = top_vars_list[j_len]
                if h[p] < 1e-8 or h[p] > 1.0 - 1e-8:
                    continue
                if h[q] < 1e-8 or h[q] > 1.0 - 1e-8:
                    continue
                h_new, c5_new, accepted = _line_search_pair(h, p, q, best_c5, n_samples=15, golden_iters=20)
                if accepted:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        # Active direction
        d_act = _active_gradient(h, band_frac=0.10)
        if d_act is not None:
            h_new, c5_new, accepted = _line_search_general(h, d_act, best_c5, n_samples=25, golden_iters=22)
            if accepted:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
        if not improved:
            stall5 += 1

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5
    return best_h, best_c5, best_n
