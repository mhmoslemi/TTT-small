    quad_j = np.zeros(2 * n - 1)
    quad_ij = np.zeros(2 * n - 1)

    for m in range(2 * n - 1):
        total_const = 0.0
        total_ci = 0.0
        total_cj = 0.0
        total_qi = 0.0
        total_qj = 0.0
        total_qij = 0.0
        ji_m = m - i
        jj_m = m - j
        if valid_i[m]:
            if self_i[m]:
                total_const += (corr[m] - hi + hi * hi) - corr[m]
                total_ci += 1.0
                total_qi += -1.0
            else:
                total_const += (-hi + 2.0 * hi * h[ji_m])
                total_ci += (1.0 - 2.0 * h[ji_m])
        if valid_j[m]:
            if self_j[m]:
                total_const += (corr[m] - hj + hj * hj) - corr[m]
                total_cj += 1.0
                total_qj += -1.0
            else:
                total_const += (-hj + 2.0 * hj * h[jj_m])
                total_cj += (1.0 - 2.0 * h[jj_m])
        # Cross term: if j appears as the "other" for i and i appears as "other" for j
        if valid_i[m] and not self_i[m] and ji_m == j:
            # h[i]*(1-h[j]) and h[j]*(1-h[i]) both present, but they're the same terms
            # Already counted above, so no additional cross term needed
            pass
        if valid_j[m] and not self_j[m] and jj_m == i:
            pass

    const_adj = const + total_const
    ci_total = coef_i + total_ci
    cj_total = coef_j + total_cj
    qi_total = quad_i + total_qi
    qj_total = quad_j + total_qj

    def f2(ai, aj):
        vals = const_adj + ci_total * ai + cj_total * aj + qi_total * ai * ai + qj_total * aj * aj
        return float(np.max(vals))

    # 2D grid search
    alphas = np.linspace(0.0, 1.0, 15)
    best_a, best_b = alphas[0], alphas[0]
    best_v = f2(best_a, best_b)
    for ai in alphas:
        for aj in alphas:
            v = f2(ai, aj)
            if v < best_v:
                best_v = v
                best_a, best_b = ai, aj
    return best_a, best_b, best_v


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _scores_and_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    n_eval = 20
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores, G = _scores_and_grad(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_obj(x, beta):
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _smooth_jac(x, beta):
    scores, G = _scores_and_grad(x)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    return G.T @ w


def _sum_fun(x, target):
    return np.sum(x) - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _slsqp_optimize(h, target, beta, max_iter=100):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _structured_jump(h, rng, size=0.03):
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    for _ in range(3):
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        d = rng.uniform(-size, size)
        h_new[i] += d
        h_new[j] -= d
    return _project(h_new, target)


def _circular_shift(h, shift):
    return np.roll(h, shift)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Coordinate descent (new key method)
    deadline1 = t0 + budget_s * 0.15
    h_cd, c5_cd = _coord_descent(h, n_sweeps=200, rng=rng)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 2: SLSQP smooth optimization
    deadline2 = t0 + budget_s * 0.30
    for beta in [10.0, 30.0, 80.0]:
        if time.time() > deadline2:
            break
        h_s = _slsqp_optimize(best_h, best_n / 2.0, beta=beta, max_iter=120)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        # Follow with coord descent
        h_cd2, c5_cd2 = _coord_descent(h_s, n_sweeps=50, rng=rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 3: LP-based refinement
    deadline3 = t0 + budget_s * 0.45
    h_lp, c5_lp = _lp_loop(best_h, deadline3, radius_init=0.12, max_iters=3000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Random restarts with structured jumps
    deadline4 = t0 + budget_s * 0.70
    n_restarts = 30
    for attempt in range(n_restarts):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3805:
            break
        # Generate perturbed starting point
        h_jump = _structured_jump(best_h, rng, size=rng.uniform(0.02, 0.08))
        # Try circular shifts too
        if attempt % 3 == 0:
            h_jump = _circular_shift(best_h, rng.integers(1, best_n))
            h_jump = _project(h_jump, best_n / 2.0)
        elif attempt % 3 == 1:
            h_jump = _structured_jump(best_h, rng, size=rng.uniform(0.05, 0.15))
        else:
            # Add smooth perturbation
            freq = rng.uniform(0.5, 3.0)
            x = np.arange(best_n) / best_n
            pert = rng.uniform(0.01, 0.05) * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            h_jump = _project(best_h + pert, best_n / 2.0)

        # Coordinate descent from jump
        h_try, c5_try = _coord_descent(h_jump, n_sweeps=80, rng=rng)
        # LP refinement
        if time.time() < deadline4:
            h_try2, c5_try2 = _lp_loop(h_try, min(deadline4, time.time() + budget_s * 0.01),
                                       radius_init=0.06, max_iters=500)
            if c5_try2 < c5_try:
                h_try, c5_try = h_try2, c5_try2
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Aggressive coordinate descent with pair updates
    deadline5 = t0 + budget_s * 0.82
    h = best_h.copy()
    for sweep in range(200):
        if time.time() > deadline5:
            break
        n = len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = False
        for idx in range(n):
            if idx + 1 < n:
                i, j = order[idx], order[idx + 1] * 0 + (order[idx + 1] if idx + 1 < n else 0)
                j = order[idx + 1] if (idx + 1) < n else (idx + 1) % n
                if i == j:
                    j = (i + 1) % n
                # Pair update
                best_a, best_b, best_v = _pair_coord_update(h, i, j, corr)
                if abs(best_a - h[i]) > 1e-10 or abs(best_b - h[j]) > 1e-10:
                    h[i] = best_a
                    h[j] = best_b
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    idx += 1
                    improved = True
                    continue
            # Single update
            i = order[idx]
            new_val, _ = _coord_update(h, i, corr)
            if abs(new_val - h[i]) > 1e-12:
                h[i] = new_val
                corr = np.correlate(h, 1.0 - h, mode='full')
                improved = True
        h = _project(h, n / 2.0)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break

    # Phase 6: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.92:
        deadline6 = t0 + budget_s * 0.94
        for dn in [2, -2, 1, -1, 4, -4, 6, -6]:
            if time.time() > deadline6:
                break
            n_try = best_n + dn
            if n_try < 20 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _coord_descent(h_r, n_sweeps=60, rng=rng)
            if time.time() < deadline6:
                h_r2, c5_r2 = _lp_loop(h_r, min(deadline6, time.time() + budget_s * 0.015),
                                       radius_init=0.08, max_iters=800)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _coord_descent(best_h, n_sweeps=100, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        h_f2, c5_f2 = _lp_loop(best_h, t0 + budget_s * 0.995,
                               radius_init=0.02, max_iters=2000)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
