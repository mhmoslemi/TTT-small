def _line_search_fast(h, D, dx):
    """Exact line search for minimizing C5 along direction D."""
    n = len(h)
    S, B, Q = _line_coeffs(h, D)
    t_lo, t_hi = _feasible_interval(h, D)
    if t_hi <= t_lo + 1e-12:
        return 0.0, float(np.max(S) * dx)
    
    # Grid search
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Evaluate g at all points vectorized
    # Shape: (n_grid, 2n-1)
    mat = S[None, :] + B[None, :] * ts[:, None] + Q[None, :] * ts[:, None] ** 2
    vals = mat.max(axis=1) * dx
    
    # Find best region
    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    
    # Refine with golden section on a neighborhood
    if 0 < best_idx < n_grid - 1:
        lo_t = ts[max(0, best_idx - 2)]
        hi_t = ts[min(n_grid - 1, best_idx + 2)]
    else:
        lo_t = t_lo
        hi_t = t_hi
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = _eval_g(S, B, Q, c1) * dx
    f2 = _eval_g(S, B, Q, c2) * dx
    
    for _ in range(30):
        if hi_t - lo_t < 1e-10:
            break
        if f1 < f2:
            hi_t = c2
            c2 = c1
            f2 = f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _eval_g(S, B, Q, c1) * dx

            lo_t = c1
            c1 = c2
            f1 = f2
            c2 = lo_t + gr * (hi_t - lo_t)

    opt_t = (lo_t + hi_t) / 2.0
    opt_val = _eval_g(S, B, Q, opt_t) * dx
