import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _active_indices(h, tol):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(scores))
    return list(np.where(scores >= mx - tol)[0])


def _lp_active_direction(h, active, G, radius):

    k = len(active)
    if k == 0:

    c = np.zeros(n + 1)

    A_ub = np.zeros((k, n + 1))

    A_eq = np.zeros((1, n + 1))

def _slsqp_solve(h, active, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    k = len(active)
    if k == 0:
        return h.copy(), _exact_c5(h)

    def obj(v):
        return v[0]

    def obj_grad(v):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq_fun(v):
        T = v[0]
        h_t = np.clip(v[1:], 0.0, 1.0)
        scores = np.correlate(h_t, 1.0 - h_t, mode='full') * dx
        return T - scores[active]

    def ineq_jac(v):
        h_t = np.clip(v[1:], 0.0, 1.0)
        G = _grad_matrix(h_t)
        J = np.zeros((k, n + 1))
        J[:, 0] = 1.0
        for i, r in enumerate(active):
            J[i, 1:] = -dx * G[r]
        return J

    def eq_fun(v):
        return np.sum(v[1:]) - target

    def eq_jac(v):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n
    x0 = np.zeros(n + 1)
    x0[0] = _exact_c5(h) + 0.003
    x0[1:] = h
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
                           {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                       ],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    except Exception:
        return h.copy(), _exact_c5(h)


def _smooth_step(h, alpha):
    n = len(h)
    target = n / 2.0
    scores = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(scores))
    vals = np.clip(alpha * (scores - ms), -400.0, 400.0)
    w = np.exp(vals)

    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))

        return h.copy(), _exact_c5(h)

        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, t
    if best_a > 0:
        return _project(h + best_a * d, target), best_c5
    return h.copy(), _exact_c5(h)


def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
