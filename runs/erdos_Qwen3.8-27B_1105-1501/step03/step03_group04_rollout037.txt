def _lp_polish(h, deadline, radius_seq=None, max_total_iters=5000):
    """LP-based polish with adaptive radius sequence."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

        if time.time() > deadline:
            break
        if it_total >= max_total_iters:
            break
        while time.time() < deadline and it_total < max_total_iters:

            G = _grad_matrix(h)
            p = _lp_direction(h, scores, G, radius)

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Golden section search
    lo_a = 0.0
    hi_a = alpha_max
    if hi_a < 1e-10:
        return 0.0, c5_cur
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    
    for _ in range(30):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    return opt_a, opt_c5

def _binary_local_search(h, deadline, max_flips=5000):
    """Try flipping individual coordinates (h[i] -> 1-h[i]) with compensation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for _ in range(max_flips):
        if time.time() > deadline:
            break
        # Pick a random coordinate to flip
        i = rng.integers(n)
        # Flip: h[i] -> 1-h[i], compensate on another coordinate
        delta = 1.0 - 2.0 * h[i]  # change to h[i]
        if abs(delta) < 1e-10:
            continue
        # Find compensation coordinate
        # We need to add -delta to some other coordinate
        if delta > 0:
            # h[i] increases, need to decrease another
            candidates = np.where(h > 1e-6)[0]
            candidates = candidates[candidates != i]
        else:
            # h[i] decreases, need to increase another
            candidates = np.where(h < 1.0 - 1e-6)[0]
            candidates = candidates[candidates != i]
        
        if len(candidates) == 0:
            continue
        j = candidates[rng.integers(len(candidates))]
        
        h_try = h.copy()
        h_try[i] = 1.0 - h[i]
        h_try[j] -= delta
        if h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
            continue
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    
    return best_h, best_c5

def _mirror_perturb(h, rng, frac=0.1):
    """Flip a random fraction of coordinates to their mirror."""
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    n_flip = max(1, int(frac * n))
    idx = rng.choice(n, size=n_flip, replace=False)
    h_new[idx] = 1.0 - h_new[idx]
    return _project(h_new, target)

def _crossover(h1, h2, rng, alpha=None):
    """Create a new candidate by mixing two parents."""
    if alpha is None:
        alpha = rng.uniform(0.3, 0.7)
    h_new = alpha * h1 + (1 - alpha) * h2 + rng.normal(0, 0.02, len(h1))
    return _project(h_new, len(h1) / 2.0)

def _greedy_coord_descent(h, deadline, max_passes=100):
    """Try all single-coordinate moves with optimal compensation."""
    n = len(h)
    target = n / 2.0

    best_h = h.copy()

            if time.time() > deadline:
                break

                    continue

                    if h_try[j] < -1e-12 or h_try[j] > 1.0 + 1e-12:
                        continue
                    h_try = np.clip(h_try, 0.0, 1.0)

                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:])
            G[r, d:] -= h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgrad_descent(h, deadline, K=5, max_iters=20000, lr_init=0.10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        m = float(np.max(scores))
        idx = np.argsort(scores)[-K:]
        G = _grad_matrix(h)
        grad = np.mean(G[idx], axis=0)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(18):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.5)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.82
            if stall % 15 == 0:
                K = max(1, K // 2) if K > 4 else min(2 * n - 1, K * 2)
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    lo_a, hi_a = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(28):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    return opt_a, opt_c5

def _lp_polish(h, deadline, radius_seq=None, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if radius_seq is None:
        radius_seq = [0.12, 0.08, 0.05, 0.03, 0.02, 0.01]
    it_total = 0
    for radius in radius_seq:
        if time.time() > deadline or it_total >= max_iters:
            break
        while time.time() < deadline and it_total < max_iters:
            it_total += 1
            scores = _scores(h)
            G = _grad_matrix(h)
            p = _lp_direction(h, scores, G, radius)
            if p is None or float(np.max(np.abs(p))) < 1e-12:
                break
            alpha, new_c5 = _line_search(h, p, cur_c5)
            if new_c5 < cur_c5 - 1e-13:
                h = _project(h + alpha * p, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
    return best_h, best_c5

def _binary_sa(h_start, deadline, n_steps=3000000, T_init=0.002, T_end=0.0001):
    """Simulated annealing on binary arrangements."""
    n = len(h_start)
    n_ones = int(round(np.sum(h_start > 0.5)))
    # Start from binary projection of h_start
    h_bin = (h_start > 0.5).astype(float)
    # Ensure exactly n_ones ones
    if int(np.sum(h_bin)) != n_ones:
        idx_ones = np.where(h_bin == 1)[0]
        idx_zeros = np.where(h_bin == 0)[0]
        if len(idx_ones) > n_ones:
            for _ in range(len(idx_ones) - n_ones):
                i = idx_ones[np.random.randint(len(idx_ones))]
                h_bin[i] = 0
                idx_ones = np.where(h_bin == 1)[0]
        elif len(idx_ones) < n_ones:
            for _ in range(n_ones - len(idx_ones)):
                i = idx_zeros[np.random.randint(len(idx_zeros))]
                h_bin[i] = 1
                idx_zeros = np.where(h_bin == 0)[0]
    
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    t_start = time.time()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(n_steps, 1)
        T = T_init * (T_end / T_init) ** frac
        
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]
        
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = _exact_c5(h_bin)
        
        if new_c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return best_h, best_c5

def _refine_binary(h_bin, deadline):
    """Refine a binary solution to fractional."""
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    h, c5 = _subgrad_descent(h, deadline, K=4, max_iters=5000, lr_init=0.06)
    return h, c5

def _fast_coord(h, deadline, max_passes=200, rng=None):
    """Fast coordinate descent with random pairs."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if rng is None:
        rng = np.random.default_rng(777)
    deltas = [0.003, 0.008, 0.015, 0.03, -0.003, -0.008, -0.015, -0.03]
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        attempts = max(4 * n, 512)
        for _ in range(attempts):
            if time.time() > deadline:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            d = deltas[rng.integers(len(deltas))]
            if h[i] + d < 0 or h[i] + d > 1.0:
                continue
            if h[j] - d < 0 or h[j] - d > 1.0:
                continue
            h[i] += d
            h[j] -= d
            c5_try = _exact_c5(h)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved:
            break
    return best_h, best_c5

def _mirror_crossover(h1, h2, rng):
    n = len(h1)
    mask = rng.random(n) < 0.5
    h_new = np.where(mask, h1, h2)
    h_new += rng.normal(0, 0.015, n)
    return _project(h_new, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    rng2 = np.random.default_rng(seed + 1)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Subgradient descent with varying K (12% time)
    d1 = t0 + budget_s * 0.12
    for K in [3, 5, 1, 8]:
        if time.time() > d1:
            break
        t_sub = min(d1, time.time() + budget_s * 0.03)
        h, c5 = _subgrad_descent(h, t_sub, K=K, max_iters=10000, lr_init=0.10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: LP polish (13% time)
    d2 = t0 + budget_s * 0.25
    h, c5 = _lp_polish(h, d2, radius_seq=[0.12, 0.08, 0.05, 0.03, 0.02])
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 3: Binary simulated annealing (22% time)
    d3 = t0 + budget_s * 0.47
    # Run SA from the current best (binary projection)
    h_bin_best = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin_best))
    if n_ones > best_n // 2:
        extra = n_ones - best_n // 2
        idx = np.where(h_bin_best == 1)[0]
        order = rng2.permutation(len(idx))
        for k in range(extra):
            h_bin_best[idx[order[k]]] = 0
    elif n_ones < best_n // 2:
        deficit = best_n // 2 - n_ones
        idx = np.where(h_bin_best == 0)[0]
        order = rng2.permutation(len(idx))
        for k in range(deficit):
            h_bin_best[idx[order[k]]] = 1
    
    n_sa_steps = int(2000000 * (d3 - time.time()) / max(budget_s * 0.22, 1))
    n_sa_steps = max(n_sa_steps, 200000)
    h_sa, c5_sa = _binary_sa(best_h, d3, n_steps=n_sa_steps, T_init=0.0015, T_end=0.00005)
    
    # Refine the binary SA result
    if c5_sa < best_c5 + 0.005:
        t_ref = min(d3 + budget_s * 0.03, t0 + budget_s * 0.52)
        h_ref, c5_ref = _refine_binary(h_sa, t_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
    
    # Phase 4: Population-based exploration (20% time)
    d4 = t0 + budget_s * 0.72
    population = [best_h.copy()]
    scale = 0.035
    for attempt in range(300):
        if time.time() > d4:
            break
        if best_c5 < 0.38080:
            break
        
        r = rng.random()
        if r < 0.4:
            h_cand = best_h + rng.normal(0, scale, best_n)
        elif r < 0.7:
            # Mirror perturbation
            h_cand = best_h.copy()
            n_flip = max(1, int(0.06 * best_n))
            idx = rng.choice(best_n, size=n_flip, replace=False)
            h_cand[idx] = 1.0 - h_cand[idx]
        elif r < 0.85:
            # Crossover
            if len(population) > 1:
                h_par = population[rng.integers(len(population))]
                h_cand = _mirror_crossover(best_h, h_par, rng)
            else:
                h_cand = best_h + rng.normal(0, scale, best_n)
        else:
            # Binary SA short run as perturbation
            h_cand = (best_h > 0.5).astype(float)
            ones_i = np.where(h_cand == 1)[0]
            zeros_i = np.where(h_cand == 0)[0]
            for _ in range(5000):
                ii = rng.integers(len(ones_i))
                jj = rng.integers(len(zeros_i))
                h_cand[ones_i[ii]] = 0
                h_cand[zeros_i[jj]] = 1
                ones_i[ii] = zeros_i[jj]
                zeros_i[jj] = ones_i[ii]
        
        h_cand = _project(h_cand, best_n / 2.0)
        
        # Quick refinement
        t_sub = min(d4, time.time() + budget_s * 0.010)
        h_try, c5_try = _subgrad_descent(h_cand, t_sub, K=5, max_iters=4000, lr_init=0.07)
        
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            population.append(h_try.copy())
            if len(population) > 6:
                population.pop(0)
            scale = max(0.01, scale * 0.92)
        else:
            scale = min(0.18, scale * 1.06)
    
    # Phase 5: Multi-n exploration (8% time)
    d5 = t0 + budget_s * 0.80
    for dn in [-8, -4, 4, 8, -16, 16]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        t_sub = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _subgrad_descent(h_r, t_sub, K=4, max_iters=3000, lr_init=0.08)
        if time.time() < d5:
            h_r, c5_r = _lp_polish(h_r, min(d5, time.time() + budget_s * 0.008),
                                    radius_seq=[0.05, 0.03, 0.015])
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 6: Intensive LP polish (8% time)
    d6 = t0 + budget_s * 0.88
    h, c5 = _lp_polish(best_h, d6, radius_seq=[0.06, 0.04, 0.025, 0.015, 0.008, 0.004])
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 7: Coordinate descent (5% time)
    d7 = t0 + budget_s * 0.93
    h, c5 = _fast_coord(best_h, d7, max_passes=400, rng=rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 8: Final fine polish (5% time)
    d8 = t0 + budget_s * 0.99
    h, c5 = _lp_polish(best_h, d8, radius_seq=[0.008, 0.004, 0.002])
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Final: one more coordinate pass
    if time.time() < t0 + budget_s * 0.995:
        h, c5 = _fast_coord(best_h, t0 + budget_s * 0.998, max_passes=100, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
