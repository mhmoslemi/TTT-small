import numpy as np
from scipy.optimize import linprog, minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_and_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G

def _lp_dir(h, scores, G, radius):
    n = len(h)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun > -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    a_max = 1e9
    for j in range(n):
        if p[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            a_max = min(a_max, h[j] / (-p[j]))
    a_max = min(a_max, 1.0)
    if a_max < 1e-12:
        return 0.0, c5_cur
    best_a = 0.0
    best_c5 = c5_cur
    ng = 35
    alphas = np.linspace(0.0, a_max, ng)
    for a in alphas[1:]:
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo = max(0.0, best_a - a_max * 0.15)
    hi = min(a_max, best_a + a_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(35):
        if hi - lo < 1e-13:

            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)

            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _slsqp_smooth_obj(x, beta, target):
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta

def _slsqp_smooth_jac(x, beta, target):
    scores, G = _scores_G(x)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    return G.T @ w

def _sum_fun(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _run_slsqp(h, target, beta, max_iter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_slsqp_smooth_obj, h, jac=_slsqp_smooth_jac, args=(beta, target),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            return _proj(res.x, target)
    except Exception:
        pass
    return h.copy()

def _subgrad_dir(h, scores, G, band_frac=0.1):
    n = len(h)
    s_max = float(np.max(scores))
    band = band_frac * s_max
    active = np.where(scores >= s_max - band)[0]
    if len(active) == 0:
        active = np.where(scores == s_max)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    grad = G[active].mean(axis=0)
    grad -= np.mean(grad)
    norm = float(np.sqrt(np.sum(grad * grad)))
    if norm < 1e-12:
        return None
    return grad / norm

def _random_dir(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.sqrt(np.sum(d * d)))
    if norm < 1e-12:
        return None
    return d / norm

def _resample(h, n_new):

        best_h = h_lp.copy()
        h = h_lp.copy()

    # Phase 3: Diverse direction search (15% time)
    deadline3 = t0 + budget_s * 0.65
    h_div, c5_div = _diverse_search(h, rng, deadline3, n_max=300)
    if c5_div < best_c5:
        best_c5 = c5_div
        best_h = h_div.copy()
        h = h_div.copy()

    # Phase 4: More LP (15% time)
    deadline4 = t0 + budget_s * 0.80
    h_lp2, c5_lp2 = _lp_search(h, deadline4, radius_init=0.10, max_iters=100000)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()
        h = h_lp2.copy()

    # Phase 5: Pair search (10% time)
    deadline5 = t0 + budget_s * 0.90
    h_pair, c5_pair = _pair_search_grad(h, deadline5, top_k=16)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair.copy()
        h = h_pair.copy()

    # Phase 6: Resolution exploration (5% time)
    deadline6 = t0 + budget_s * 0.95
    n_list = [best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n - 8, best_n + 8]
    n_list = sorted(set(x for x in n_list if 32 <= x <= 128 and x != best_n))
    for n_try in n_list:
        if time.time() > deadline6:
            break
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-10:
            h_r2, c5_r2 = _lp_search(h_r, min(deadline6, time.time() + 3.0),
                                      radius_init=0.10, max_iters=5000)
            if c5_r2 < best_c5 - 1e-10:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_try

    # Phase 7: Final LP polish
    deadline7 = hard_deadline
    h_final, c5_final = _lp_search(best_h, deadline7, radius_init=0.05, max_iters=50000)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    best_h = _proj(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
