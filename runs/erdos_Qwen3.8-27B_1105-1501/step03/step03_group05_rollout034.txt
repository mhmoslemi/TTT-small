    # Since f is convex, we can use bisection on the subgradient
    if best_alpha == a_min or best_alpha == a_max:

    lo = best_alpha - (a_max - a_min) * 0.1
    hi = best_alpha + (a_max - a_min) * 0.1
    lo = max(lo, a_min)
    hi = min(hi, a_max)

            hi = mid  # need to go left

        else:
            lo = hi = mid

    opt_val = np.max(scores + opt_alpha * L)
