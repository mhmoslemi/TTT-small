def _swap_minimax(h, i, j):
    """Find optimal swap amount t for swapping between i and j.
    h[i] -= t, h[j] += t. Returns optimal t and new h."""
    n = len(h)
    # Compute current scores
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute a[r] = dS[r]/dt for each score r
    a = np.zeros(2*n - 1)
    for r in range(2*n - 1):
        d = r - (n - 1)  # lag
        if d == 0:
            # dC/dh[m] = 1 - 2h[m]
            da_j = 1.0 - 2.0 * h[j]
            da_i = 1.0 - 2.0 * h[i]
        elif d > 0:
            # dC/dh[m] = (1-h[m-d])*I(m>=d) - h[m+d]*I(m<=n-d-1)
            da_j = 0.0
            if j >= d:
                da_j += 1.0 - h[j - d]
            if j <= n - d - 1:
                da_j -= h[j + d]
            da_i = 0.0
            if i >= d:
                da_i += 1.0 - h[i - d]
            if i <= n - d - 1:
                da_i -= h[i + d]
        else:
            g = -d  # g > 0
            # dC/dh[m] = (1-h[m+g])*I(m<=n-g-1) - h[m-g]*I(m>=g)
            da_j = 0.0
            if j <= n - g - 1:
                da_j += 1.0 - h[j + g]
            if j >= g:
                da_j -= h[j - g]
            da_i = 0.0
            if i <= n - g - 1:
                da_i += 1.0 - h[i + g]
            if i >= g:
                da_i -= h[i - g]
        a[r] = da_j - da_i
    
    # t bounds
    t_min = max(-h[j], h[i] - 1.0)
    t_max = min(h[i], 1.0 - h[j])
    
    if t_max - t_min < 1e-15:
        return 0.0
    
    # Find t* = argmin max_r (scores[r] + a[r]*t) for t in [t_min, t_max]
    # This is a 1D convex minimization
    # Use the fact that the minimum is at a kink point or boundary
    
    # Evaluate at boundaries
    f_min = float(np.max(scores + a * t_min))
    f_max = float(np.max(scores + a * t_max))
    best_t = t_min if f_min < f_max else t_max
    best_f = min(f_min, f_max)
    
    # Find the point where the derivative of the max changes sign
    # At the optimum, the slope of the max should change from negative to positive
    # We can binary search on t
    
    lo = t_min
    hi = t_max
    for _ in range(60):
        mid = (lo + hi) / 2.0
        # Compute subgradient at mid: the derivative of max is a[r*] where r* = argmax
        vals = scores + a * mid
        r_star = int(np.argmax(vals))
        slope = a[r_star]
        if slope < 0:
            lo = mid
        elif slope > 0:
            hi = mid
        else:
            break
    
    t_star = (lo + hi) / 2.0
    f_star = float(np.max(scores + a * t_star))
    if f_star < best_f:
        best_f = f_star
        best_t = t_star
    
    return best_t

    """Find t in [t_min, t_max] that minimizes max_r (scores[r] + a[r]*t).
    Returns the optimal t."""
    n = len(scores)
    
    # Evaluate at boundaries
    f_at_min = float(np.max(scores + a * t_min))
    f_at_max = float(np.max(scores + a * t_max))
    best_t = t_min
    best_f = f_at_min
    if f_at_max < best_f:
        best_f = f_at_max
        best_t = t_max
    
    # Sweep method: start at t_min, follow the upper envelope
    t = t_min
    for _ in range(n + 5):  # at most n segments
        vals = scores + a * t
        # Find all indices achieving (near) the max
        max_val = float(np.max(vals))
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]

        slopes_active = a[active]
        if np.max(slopes_active) >= -1e-15:
            # The minimum is at or before this point
            # If there's a line with slope exactly 0, we're at the min
            # If all active slopes are > 0, we've passed the min
            # The minimum is at this kink (or slightly before)
            f_here = float(np.max(vals))
            if f_here < best_f:
                best_f = f_here
                best_t = t
            break
        
        # All active slopes are < 0, so the function is decreasing at t

        # For each line r not in active, find where scores[r] + a[r]*t' = max_val

            if r in active:

            if a[r] <= a[active[0]] - 1e-15:  # same or smaller slope, won't catch up from above

            # t = (max_val - scores[r]) / a[r]

    """Minimize max(scores + a*t) over t in [t_min, t_max]."""
    if t_max - t_min < 1e-15:
        return t_min
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_max - gr * (t_max - t_min)

    fc = float(np.max(scores + a * c))

            c = d
            fc = fd
            d = t_min + gr * (t_max - t_min)

    """Compute grad[m, r] = dS_r/dh[m] for all m and r."""
    n = len(h)
    grad = np.zeros((n, 2*n - 1))

        elif d > 0:

    """Try all pairs, apply the best swap. Returns new h and improvement."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    grad = _compute_grad_matrix(h)
    
    best_improvement = 0.0
    best_pair = None
    best_t = 0.0
    
    for i in range(n):
        for j in range(i+1, n):
            if abs(h[i] - h[j]) < 1e-10:
                continue  # no point swapping equal values
            a = grad[j] - grad[i]
            t_min = max(-h[j], h[i] - 1.0)
            t_max = min(h[i], 1.0 - h[j])
            if t_max - t_min < 1e-12:

                best_pair = (i, j)
                best_t = t_star

        h[i] -= best_t
