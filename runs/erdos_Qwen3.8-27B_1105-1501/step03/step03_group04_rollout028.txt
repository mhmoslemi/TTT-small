import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    one_h = 1.0 - h
    for r in range(2 * n - 1):
        lag = r - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[r, :m] += one_h[lag:lag + m]
                G[r, lag:] -= h[:m]
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[r, g:] += one_h[:m]
                G[r, :m] -= h[g:]
    return G


def _project(v, target):
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)


def _softmax_weights(scores, beta):
    m = scores.max()
    exps = np.exp(beta * (scores - m))
    s = exps.sum()
    if s < 1e-300:
        return np.ones(len(scores)) / len(scores)
    return exps / s


def _smooth_gradient(h, beta):
    scores = _all_scores(h)
    weights = _softmax_weights(scores, beta)
    G = _grad_matrix(h)
    dx = 2.0 / len(h)
    return dx * (G.T @ weights)


def _smooth_value(h, beta):
    scores = _all_scores(h)
    m = scores.max()
    exps = np.exp(beta * (scores - m))
    return m + np.log(exps.sum()) / beta


def _line_search(h, direction, cur_c5, alpha_init=0.5):
    n = len(h)
    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j] * 2.0)
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]) * 2.0)
    alpha_max = max(alpha_max, 0.1)
    alphas = np.concatenate([np.array([0.01, 0.05, 0.1]), np.geomspace(0.2, alpha_max, 10)])
    alphas = np.unique(np.sort(alphas))
    best_a = 0.0
    best_c5 = cur_c5
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max * 1.5, best_a * 2.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(25):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_h = _project(h + opt_a * direction, target)
    opt_c5 = _exact_c5(opt_h)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _soft_descent(h, deadline, beta_start=20.0, beta_end=3000.0, max_iters=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.15
    stall = 0
    momentum = np.zeros(n)
    mu = 0.9
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad = _smooth_gradient(h, beta)
        mean_grad = np.mean(grad)
        grad -= mean_grad
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        momentum = mu * momentum + (1 - mu) * direction
        mnorm = np.linalg.norm(momentum)
        if mnorm > 1e-14:
            direction = -momentum / mnorm
        alpha, new_c5 = _line_search(h, direction, cur_c5, alpha_init=lr)
        if alpha > 0 and new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * direction, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.08, 1.5)
            mu = min(mu * 1.005, 0.95)
        else:
            stall += 1
            lr *= 0.7
            mu *= 0.95
            if lr < 1e-6 or stall > 80:
                break
        if it % 200 == 0 and stall > 30:
            break
    return best_h, best_c5


def _lp_active_direction(h, active_lags, radius=0.15):
    n = len(h)
    scores = _all_scores(h)
    dx = 2.0 / n
    G = _grad_matrix(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(active_lags), n + 1))
    b_ub = np.zeros(len(active_lags))
    for idx, r in enumerate(active_lags):
        A_ub[idx, :n] = G[r] * dx
        A_ub[idx, n] = -1.0
        b_ub[idx] = -scores[r]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_polish(h, deadline, radius_init=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _all_scores(h)
        max_val = scores.max()
        threshold = max_val - 0.005
        active = list(np.where(scores >= threshold)[0])
        if len(active) == 0:
            active = [int(np.argmax(scores))]
        if len(active) > 20:
            top_idx = np.argsort(scores)[-20:]
            active = list(top_idx)
        p = _lp_active_direction(h, active, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_init=0.5)
        if alpha > 0 and new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _perturb_descend(h, deadline, rng, scale=0.08):
    n = len(h)
    h_pert = h + rng.normal(0, scale, n)
    h_pert = _project(h_pert, n / 2.0)
    h_out, c5_out = _soft_descent(h_pert, deadline, beta_start=15.0, beta_end=2000.0, max_iters=2000)
    return h_out, c5_out


def _try_n_values(h_best, c5_best, n_current, deadline, rng):
    n_candidates = set()
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
        nc = n_current + dn
        if 32 <= nc <= 128:
            n_candidates.add(nc)
    for nc in sorted(n_candidates):
        if time.time() > deadline:
            break
        h_r = _resample(h_best, nc)
        c5_r = _exact_c5(h_r)
        if c5_r > c5_best + 0.02:
            continue
        time_left = deadline - time.time()
        if time_left < 5:
            continue
        h_t, c5_t = _soft_descent(h_r, min(time.time() + time_left * 0.4, deadline),
                                  beta_start=20.0, beta_end=3000.0, max_iters=3000)
        if time.time() < deadline - 3:
            h_t, c5_t = _lp_polish(h_t, deadline, radius_init=0.06, max_iters=500)
        if c5_t < c5_best - 1e-10:
            c5_best = c5_t
            h_best = h_t.copy()
            n_current = nc
    return h_best, c5_best, n_current
