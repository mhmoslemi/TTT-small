def _exact_line_min(h, d, n_evals=300):
    """Find min of max_k corr_k(t)*dx for t in feasible interval."""
    n = len(h)
    dx = 2.0 / n
    
    # Find feasible interval
    t_min = 0.0
    t_max = 0.0
    # Actually, t can be negative too
    t_lo = 0.0
    t_hi = 0.0
    
    # h + t*d in [0,1] for all i
    # For d[i] > 0: t >= -h[i]/d[i]
    # For d[i] < 0: t <= (1-h[i])/(-d[i])
    
    pos = d > 1e-15
    neg = d < -1e-15
    
    if np.any(pos):
        t_lo = max(t_lo, np.max(-h[pos] / d[pos]))
    if np.any(neg):
        t_hi = min(t_hi, np.min((1.0 - h[neg]) / (-d[neg])))
    
    if t_hi <= t_lo:
        return None
    
    # Evaluate at many points
    t_vals = np.linspace(t_lo, t_hi, n_evals)
    # For each t, compute max corr * dx
    # Use the quadratic formula: corr(t) = corr0 + t*B + t^2*C
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    # corr_d_1h = correlate(d, 1-h)
    corr_d1h = np.correlate(d, 1.0 - h, mode='full')
    corr_hmd = np.correlate(h, -d, mode='full')
    corr_dmd = np.correlate(d, -d, mode='full')
    
    # corr(t) = corr0 + t*(corr_d1h + corr_hmd) + t^2*corr_dmd
    B = corr_d1h + corr_hmd
    C = corr_dmd
    
    # For each t, max of (corr0 + t*B + t^2*C) over all components
    # vectorized: for each t, compute all components, take max
    # Shape: t_vals is (M,), B and C are (2n-1,)
    # corr at t: corr0 + t*B + t^2*C -> shape (M, 2n-1)
    M = len(t_vals)
    K = len(B)
    # Reshape for broadcasting
    tt = t_vals[:, None]  # (M, 1)
    corr_mat = corr0[None, :] + tt * B[None, :] + tt**2 * C[None, :]  # (M, K)
    max_vals = np.max(corr_mat, axis=1) * dx  # (M,)
    
    best_idx = np.argmin(max_vals)
    best_t = t_vals[best_idx]
    best_val = max_vals[best_idx]
    
    # Refine with golden section in neighborhood
    span = t_vals[1] - t_vals[0]
    lo = max(t_lo, best_t - 2*span)
    hi = min(t_hi, best_t + 2*span)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def obj(t):
        c = corr0 + t * B + t**2 * C
        return float(np.max(c)) * dx
    
    f1 = obj(c1)
    f2 = obj(c2)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    t_star = (lo + hi) / 2.0
    obj_star = obj(t_star)
    if obj_star < best_val:
        best_t, best_val = t_star, obj_star
    
    return best_t, best_val

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _mean_free(d):
    return d - np.mean(d)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _smooth_direction(h, alpha):
    n = len(h)
    s = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    G = _grad_matrix(h)
    g = G.T @ w
    return _mean_free(g)


def _exact_line_min(h, d):
    n = len(h)
    dx = 2.0 / n
    d = _mean_free(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    d = d / norm
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_cand = float(np.max(-h[pos] / d[pos]))
        t_lo = max(t_lo, t_cand)
    if np.any(neg):
        t_cand = float(np.min((1.0 - h[neg]) / (-d[neg])))
        t_hi = min(t_hi, t_cand)
    if t_hi <= t_lo + 1e-14:
        return None
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    corr_d1h = np.correlate(d, 1.0 - h, mode='full')
    corr_hmd = np.correlate(h, -d, mode='full')
    corr_dmd = np.correlate(d, -d, mode='full')
    B = corr_d1h + corr_hmd
    C = corr_dmd
    M = 256
    t_vals = np.linspace(t_lo, t_hi, M)
    tt = t_vals[:, None]
    corr_mat = corr0[None, :] + tt * B[None, :] + tt * tt * C[None, :]
    max_vals = np.max(corr_mat, axis=1)
    best_idx = int(np.argmin(max_vals))
    best_t = t_vals[best_idx]
    best_val = max_vals[best_idx]
    span = (t_hi - t_lo) / M
    lo = max(t_lo, best_t - 4.0 * span)
    hi = min(t_hi, best_t + 4.0 * span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    def _obj(t):
        return float(np.max(corr0 + t * B + t * t * C)) * dx

    f1 = _obj(c1)
    f2 = _obj(c2)
    for _ in range(45):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _obj(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _obj(c2)
    t_star = (lo + hi) / 2.0
    obj_star = _obj(t_star)
    if obj_star < best_val * dx:
        best_t, best_val = t_star, obj_star
    else:
        best_val = float(np.max(corr0 + best_t * B + best_t * best_t * C)) * dx
    h_new = np.clip(h + best_t * d, 0.0, 1.0)
    return h_new, best_val


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _try_direction(h, d, n):
    res = _exact_line_min(h, d)
    if res is None:
        return h, _exact_c5(h)
    h_new, c5_est = res
    h_new = _project(h_new, n / 2.0)
    c5_actual = _exact_c5(h_new)
    return h_new, c5_actual


def _systematic_pair_sweep(h, n, deadline, max_rounds=5):
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                h_new, c5_new = _try_direction(h, d, n)
                if c5_new < cur_c5 - 1e-13:
                    h = h_new
                    cur_c5 = c5_new
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5
