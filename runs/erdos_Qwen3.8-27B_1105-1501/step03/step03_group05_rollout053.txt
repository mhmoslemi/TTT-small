import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    return G

def _lp_trust_step(h, radius):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _build_G(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 10000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.12, max_iters=30000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_start
    stall = 0
    factors = [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        x_new = _lp_trust_step(h, radius)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        x_new = _project_box_sum(x_new, n / 2.0)
        best_step = 0.0
        best_trial_c5 = cur_c5
        for factor in factors:
            trial = _project_box_sum(h + factor * (x_new - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < best_trial_c5 - 1e-13:
                best_trial_c5 = tc5
                best_step = factor
        if best_step > 0:

                best_h = h.copy()
            stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

        best_c5 = final_c5

    # Safety check
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h, best_c5, best_n
