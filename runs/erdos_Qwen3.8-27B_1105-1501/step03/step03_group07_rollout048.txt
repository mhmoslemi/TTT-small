from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection to find threshold
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, d):
    """Gradient of J_d with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] -= h[:m]
    else:
        k = -d
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    return grad

    grads = []
    for r in range(2 * n - 1):
        d = r - (n - 1)
        grads.append(_lag_grad(h, d))
    return raw_vals, grads

    """Find descent direction via LP tangent plane"""
    n = len(h)
    if R is None:
        R = 0.3 * n
    raw_vals, grads = _all_lag_data(h)
    
    # Select top k lags

        c[1 + n + i] = lam  # q[i]

    b_ub = []

        row[0] = -1.0
        row[1:1+n] = g        # +g · p
        row[1+n:1+2*n] = -g   # -g · q
        A_ub.append(row)
        b_ub.append(-float(raw_vals[s_idx]))

    row = np.zeros(nvar)
    for i in range(n):
        row[1 + i] = 1.0
        row[1 + n + i] = 1.0
    A_ub.append(row)
    b_ub.append(R)

    b_ub = np.array(b_ub)

        A_eq[0, 1 + n + i] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    eps = 1e-10
    for i in range(n):

            bounds.append((0.0, 0.0))    # p[i] = 0
            bounds.append((0.0, None))    # q[i] free

            bounds.append((0.0, None))
            bounds.append((0.0, None))

    if opt_c < best_c:

    return best_a, best_c

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] -= h[:m]
    else:
        k = -d
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    return grad

def _lp_dir(h, top_k, R, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    order = np.argsort(-corr)
    sel = order[:top_k]
    nvar = 1 + 2 * n
    c = np.zeros(nvar)
    c[0] = 1.0
    c[1:n+1] = lam
    c[n+1:n*2+1] = lam
    rows = []
    rhs = []
    for si in sel:
        d = int(si) - (n - 1)
        g = _lag_grad(h, d)
        row = np.zeros(nvar)
        row[0] = -1.0
        row[1:n+1] = g
        row[n+1:n*2+1] = -g
        rows.append(row)
        rhs.append(-float(corr[si]))
    row = np.zeros(nvar)
    row[1:n+1] = 1.0
    row[n+1:n*2+1] = 1.0
    rows.append(row)
    rhs.append(R)
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:n+1] = 1.0
    A_eq[0, n+1:n*2+1] = -1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)]
    eps = 1e-10
    for i in range(n):
        if h[i] >= 1.0 - eps:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        elif h[i] <= eps:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
    except Exception:
        return None
    x = res.x
    d = x[1:n+1] - x[n+1:n*2+1]
    d -= np.mean(d)
    no = np.linalg.norm(d)
    if no < 1e-12:
        return None
    return d / no

def _softmax_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - float(np.max(corr))
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.zeros(2*n-1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for r in range(2*n-1):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        grad += w[r] * _lag_grad(h, d)
    return grad

def _line_search(h, direction, target, cur_c5, max_alpha=2.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            a = (1.0 - h[j]) / direction[j]
            if a < alpha_max:
                alpha_max = a
        elif direction[j] < -1e-14:
            a = h[j] / (-direction[j])
            if a < alpha_max:

    alpha_max = min(max(alpha_max, 1e-6), 5.0)

        if a <= 0:

        c_t = _exact_c5(h_t)

            best_a = a

        return 0.0, cur_c5
    lo = max(0.0, best_a * 0.65)

        return best_a, best_c
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    oa = (lo + hi) / 2.0
    oc = _exact_c5(_project(h + oa * direction, target))
    if oc < best_c:

    return best_a, best_c
