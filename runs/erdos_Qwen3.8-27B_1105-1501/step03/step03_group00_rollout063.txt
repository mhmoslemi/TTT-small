def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _pnorm_grad(h, p, G, scores):
    s = np.maximum(scores, 1e-15)
    sp = s ** (p - 1)
    grad = G.T @ sp
    # Normalize
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad

def _lp_step(h, scores, G, radius, top_k, deadline):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _binary_sa(h_start, n, deadline, rng, t0=0.001, t_end=1e-5, n_iters=None):
    # Binary simulated annealing
    if n % 2 != 0:
        return h_start, _exact_c5(h_start)
    target = n // 2
    # Initialize binary
    h_bin = np.zeros(n)
    h_bin[np.argsort(h_start)[-target:]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    if n_iters is None:
        n_iters = max(1000, min(50000, int(30 / (time.time() + 1e9 - deadline + time.time()))))
    
    n_sa = n_iters
    for it in range(n_sa):
        if time.time() > deadline:
            break
        frac = it / max(1, n_sa - 1)
        T = t0 * (t_end / t0) ** frac
        if T < 1e-8:
            break
        # Pick random 1 and random 0 to swap
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i_val = ones[i_pos]
        j_val = zeros[j_pos]
        if i_val == j_val:
            continue
        h_bin[i_val] = 0.0
        h_bin[j_val] = 1.0
        new_c5 = _exact_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            # Update sets
            ones[i_pos] = j_val
            zeros[j_pos] = i_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i_val] = 1.0
            h_bin[j_val] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

= M^(-p+1) * M * (Σ u_r^p)^(1/p - 1) * M^(p-1) * Σ u_r^(p-1) * ∂s_r/∂h_i / M

Hmm, this is getting complicated. Let me just use:
weights = s^（p-1) / max(s)^(p-1) = (s/max(s))^(p-1)

Then grad ∝ G^T @ weights

This avoids underflow since weights are in [0, 1] (for s ≥ 0).

Actually, for very large p, even (s/M)^(p-1) will underflow for s << M. But that's actually what we want - it makes the gradient focus on the maximum.

Let me just use np.power with clipping:
s_clipped = np.maximum(s, 1e-30)
weights = s_clipped ** (p-1)
# Normalize to avoid overflow
weights = weights / weights.max() if weights.max() > 0 else weights

This should work in practice for p up to a few hundred.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _pnorm_descent(h, deadline, p_list=None, steps_per_p=60):
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128, 256, 512]
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

    for p_val in p_list:
        if time.time() > deadline:
            break
        stall = 0
        for step in range(steps_per_p):
            if time.time() > deadline or stall > 30:
                break
            scores = _scores_vec(h)
            G = _grad_matrix(h)
            s_max = np.max(scores)
            if s_max < 1e-15:
                break
            u = np.maximum(scores, 0.0) / s_max
            weights = u ** (p_val - 1)
            w_max = np.max(weights)
            if w_max < 1e-300:
                weights = np.zeros(len(scores))
                weights[np.argmax(scores)] = 1.0
            else:
                weights = weights / w_max
            grad = G.T @ weights
            grad = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad = grad / gnorm
            # Find max step
            alpha_max = 1.0
            for j in range(n):
                if grad[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / grad[j])
                elif grad[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-grad[j]))
            alpha_max = max(alpha_max, 1e-10)
            # Try descent direction (negative gradient)
            best_a = 0.0
            best_c = cur_c5
            alphas = np.geomspace(0.01, alpha_max, 12)
            for a in alphas:
                if time.time() > deadline:
                    break
                h_try = h - a * grad
                h_try = np.clip(h_try, 0.0, 1.0)
                c_try = _exact_c5(h_try)
                if c_try < best_c:
                    best_c = c_try
                    best_a = a
            if best_a > 0 and best_c < cur_c5 - 1e-14:
                h = np.clip(h - best_a * grad, 0.0, 1.0)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                # Try smaller step or different p
                if best_a == 0:
                    stall += 1
                else:
                    # Try half step
                    a2 = best_a * 0.5
                    h_try2 = _project(np.clip(h - a2 * grad, 0, 1), target)
                    c_try2 = _exact_c5(h_try2)
                    if c_try2 < cur_c5 - 1e-14:
                        h = h_try2
                        cur_c5 = c_try2
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
        # Update best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _lp_optimize(h_init, deadline, top_k_list=None, radius_init=0.12, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 12, 6, 4]
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores_vec(h)
        G = _grad_matrix(h)
        p_dir, t_val = _lp_direction(h, scores, G, radius, tk)
        if p_dir is None or float(np.max(np.abs(p_dir))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p_dir[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p_dir[j])
            elif p_dir[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p_dir[j]))
        alpha_max = max(alpha_max, 1e-12)
        alphas = np.linspace(0, alpha_max, 18)
        best_a = 0.0
        best_c = cur_c5
        for a in alphas[1:]:
            if time.time() > deadline:
                break
            h_try = np.clip(h + a * p_dir, 0, 1)
            c_try = _exact_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_a = a
        if best_c < cur_c5 - 1e-13:
            h = np.clip(h + best_a * p_dir, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.5)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 25 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _binary_sa(h_start, n, deadline, rng, t0=0.0008, t_floor=5e-6, n_iters=15000):
    if n % 2 != 0:
        return h_start, _exact_c5(h_start)
    target = n // 2
    h_bin = np.zeros(n)
    h_bin[np.argsort(h_start)[-target:]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin == 1.0)[0].copy()
    zeros = np.where(h_bin == 0.0)[0].copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, n_iters - 1)
        T = t0 * (t_floor / t0) ** frac
        if T < 1e-10:
            break
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i_val = ones[i_pos]
        j_val = zeros[j_pos]
        if i_val == j_val:
            continue
        h_bin[i_val] = 0.0
        h_bin[j_val] = 1.0
        new_c5 = _exact_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or (delta < T * 5 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones[i_pos] = j_val
            zeros[j_pos] = i_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i_val] = 1.0
            h_bin[j_val] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_rounds=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    target = n / 2.0
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Focus on interior points
        interior = [j for j in range(n) if 0.05 < h[j] < 0.95]
        if not interior:
            break
        rng_local = np.random.default_rng(42 + _round)
        for _trial in range(200):
            if time.time() > deadline:
                break
            if len(interior) < 2:
                break
            i1 = interior[rng_local.integers(len(interior))]
            i2 = interior[rng_local.integers(len(interior))]
            if i1 == i2:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[i1], h[i2])
                else:
                    a_max = min(h[i1], 1.0 - h[i2])
                if a_max < 1e-10:
                    continue
                # Golden section search
                lo, hi = 0.0, a_max
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                h1 = h.copy()
                h1[i1] += sign * c1
                h1[i2] -= sign * c1
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[i1] += sign * c2
                h2[i2] -= sign * c2
                f2 = _exact_c5(h2)
                for _ in range(14):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2
                        f2 = f1
                        c2 = c1
                        c1 = hi - gr * (hi - lo)
                        h1 = h.copy()
                        h1[i1] += sign * c1
                        h1[i2] -= sign * c1
                        f1 = _exact_c5(h1)
                    else:
                        lo = c1
                        f1 = f2
                        c1 = c2
                        c2 = lo + gr * (hi - lo)
                        h2 = h.copy()
                        h2[i1] += sign * c2
                        h2[i2] -= sign * c2
                        f2 = _exact_c5(h2)
                a_opt = (lo + hi) / 2.0
                h_try = h.copy()
                h_try[i1] += sign * a_opt
                h_try[i2] -= sign * a_opt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _unfreeze_polish(h_bin, deadline, rng):
    n = len(h_bin)
    target = n / 2.0
    # Add small noise to unfreeze binary
    h_cont = h_bin + rng.normal(0, 0.08, n)
    h_cont = _project(h_cont, target)
    # LP polish
    h_pol, c5_pol = _lp_optimize(h_cont, deadline, top_k_list=[10, 6, None],
                                  radius_init=0.1, max_iters=500)
    return h_pol, c5_pol

def run(seed=42, budget_s=1100, **kwargs):

                                  radius_init=0.15, max_iters=4000)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()

    # Phase 3: Binary SA exploration (20%)
    if time.time() < t0 + budget_s * 0.65:
        dl3 = t0 + budget_s * 0.65
        for attempt in range(8):
            if time.time() > dl3:
                break
            if best_c5 < 0.3804:
                break
            # Perturb for diverse basin
            scale = 0.08 + 0.04 * attempt
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl3, time.time() + budget_s * 0.035)
            h_sa, c5_sa = _binary_sa(h_pert, best_n, sub_dl, rng,

                h_unf, c5_unf = _unfreeze_polish(h_sa, unf_dl, rng)
                if c5_unf < best_c5:
                    best_c5 = c5_unf
                    best_h = h_unf.copy()
                else:
                    best_h = _project(h_sa + rng.normal(0, 0.03, best_n), best_n / 2.0)
            elif time.time() < dl3 - budget_s * 0.02:
                # Try p-norm on perturbed
                sub_dl2 = min(dl3, time.time() + budget_s * 0.03)
                h_pn, c5_pn = _pnorm_descent(h_pert, sub_dl2,
                                              p_list=[16, 32, 64, 128], steps_per_p=25)

                    best_h = h_pn.copy()

            h_r = _resample(best_h, n_try)
            sub_dl = min(dl6, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None, 4],
                                      radius_init=0.1, max_iters=400)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final ultra-fine LP (5%)
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.995,
                                   top_k_list=[3, 4, 5, 2],
                                   radius_init=0.008, max_iters=6000)

    # Final cleanup
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
