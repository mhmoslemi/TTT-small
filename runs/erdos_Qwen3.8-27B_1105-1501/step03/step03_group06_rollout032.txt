    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0

def _active_grad(h, d):
    """Gradient of corr[n-1+d] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        # corr[n-1+d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # d/dh[j]: 
        #   if j in [d, n-1]: contributes (1-h[j-d])
        #   if j in [0, n-1-d]: contributes -h[j+d]
        m = n - d
        g[d:] += (1.0 - h[:m])
        g[:m] += -h[d:]
    else:
        g = -d
        # corr[n-1-g] = sum_{i=g}^{n-1} h[i-g]*(1-h[i])
        # d/dh[j]:
        #   if j in [0, n-1-g]: contributes (1-h[j+g])
        #   if j in [g, n-1]: contributes -h[j-g]
        m = n - g
        g[:m] += (1.0 - h[g:])
        g[g:] += -h[:m]
    return g

def _max_subgradient(h, alpha=50.0):
    """Weighted subgradient of max(corr) using softmax weights."""
    corr = _corr_vector(h)
    n = len(h)
    # Softmax weights
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    # Weighted subgradient
    g = np.zeros(n)
    for i, d in enumerate(range(-(n-1), n)):
        if w[i] > 1e-10:
            g += w[i] * _active_grad(h, d)
    return g, ms, w

def _slsqp_active(h, n_act=10, maxiter=300):
    """SLSQP on epigraph with only active constraints."""
    n = len(h)
    target = n / 2.0
    corr = _corr_vector(h)
    ms = float(np.max(corr))
    
    # Select active constraints (closest to max)
    sorted_idx = np.argsort(-corr)
    active_idx = sorted_idx[:n_act]
    
    # Variables: x[0] = T (raw), x[1:] = h
    x0 = np.zeros(n + 1)
    x0[0] = ms
    x0[1:] = h
    
    # Objective: minimize T
    def obj(x):
        return x[0]
    
    # Inequality constraints: T - corr_k >= 0 for active k
    def make_incon(k_idx):
        def incon(x):
            h_x = x[1:]
            c = _corr_vector(h_x)
            return x[0] - c[k_idx]
        return incon
    
    # Equality constraint: sum(h) = target
    def eq_con(x):
        return float(np.sum(x[1:])) - target
    
    # Bounds
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    
    # Collect inequality constraints
    incons = []
    for k_idx in active_idx:
        incons.append({'type': 'ineq', 'fun': make_incon(k_idx)})
    incons.append({'type': 'eq', 'fun': eq_con})
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=incons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            h_new = _project(res.x[1:], target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)

        ms = float(np.max(corr))

        s = float(np.sum(exp_vals))

def _coord_descent(h, deadline, max_passes=100):
    """Iterate over each coordinate, finding the best value for that coordinate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            # Try to optimize h[j] while keeping sum fixed
            # If we change h[j] by delta, we need to compensate
            # Option 1: change only h[j] and project (changes other values slightly)
            # Option 2: change h[j] and h[k] for some k to maintain sum exactly
            
            # Simple approach: try a few values of h[j], project, check
            old_val = h[j]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.001, -0.001]:

from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_at_lag(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        return float(np.sum(h[d:] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        return float(np.sum(h[:m] * (1.0 - h[g:])))

def _grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[d:] += (1.0 - h[:m])
        g[:m] += -h[d:]
    else:
        g2 = -d
        m = n - g2
        g[:m] += (1.0 - h[g2:])
        g[g2:] += -h[:m]
    return g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_direction(h, alpha=80.0):
    """Compute weighted subgradient of max(corr)."""
    n = len(h)
    corr = _corr_full(h)
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    g = np.zeros(n)
    for i in range(2*n-1):
        if w[i] > 1e-10:
            d = i - (n - 1)
            g += w[i] * _grad_lag(h, d)
    return g, ms, w

def _lbfsgs(h, alpha, maxiter=300):
    """L-BFGS-B on smooth surrogate."""
    n = len(h)
    target = n / 2.0
    
    def fun(x):
        corr = _corr_full(x)
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        ev = np.exp(vals)
        s = float(np.sum(ev))
        f = ms + (1.0/alpha) * np.log(s)
        w = ev / s
        g = np.zeros(n)
        for i in range(2*n-1):
            if w[i] > 1e-10:
                d = i - (n-1)
                g += w[i] * _grad_lag(x, d)
        g -= np.mean(g)
        return f, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50, 'iprint': -1})
        return _project(res.x, target), _exact_c5(_project(res.x, target))
    except Exception:
        return h.copy(), _exact_c5(h)

def _slsqp_epigraph(h, n_active=12, maxiter=400):
    """SLSQP on epigraph form with active constraints."""
    n = len(h)
    target = n / 2.0
    corr = _corr_full(h)
    ms = float(np.max(corr))
    top = np.argsort(-corr)[:n_active]
    
    x0 = np.concatenate([[ms], h])
    
    def obj(x):
        return x[0]
    
    cons = []
    for k in top:
        d = k - (n - 1)
        def make_ic(d_val):
            def ic(x):
                h_x = x[1:]
                return x[0] - _corr_at_lag(h_x, d_val)
            return ic
        cons.append({'type': 'ineq', 'fun': make_ic(d)})
    
    def eq(x):
        return float(np.sum(x[1:])) - target
    cons.append({'type': 'eq', 'fun': eq})
    
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = _project(res.x[1:], target)
        return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)

def _pair_golden(h, j, k, t_max, c5_best):
    """Golden section search for pair (j, k)."""
    n = len(h)
    target = n / 2.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    e_j = np.zeros(n); e_j[j] = 1.0
    e_k = np.zeros(n); e_k[k] = 1.0
    d = e_j - e_k  # direction: increase j, decrease k
    
    lo, hi = 0.0, t_max
    best_t, best_c5 = 0.0, c5_best
    
    for _ in range(16):
        if hi - lo < 1e-10:
            break
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h1 = _project(h + c1 * d, target)
        h2 = _project(h + c2 * d, target)
        f1 = _exact_c5(h1)
        f2 = _exact_c5(h2)
        if f1 < best_c5:
            best_c5, best_t = f1, c1
        if f2 < best_c5:
            best_c5, best_t = f2, c2
        if f1 < f2:
            hi = c2
        else:
            lo = c1
    
    if best_t > 1e-12:
        h_new = _project(h + best_t * d, target)
        return h_new, _exact_c5(h_new)
    return h, c5_best

def _pair_polish(h, deadline, max_passes=80):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        
        # Compute subgradient to guide pair selection
        g, ms, w = _subgrad_direction(h, alpha=100.0)
        
        # Priority: pairs where g[j] - g[k] is large (one wants to increase, other decrease)
        # Direction: if g[j] > g[k], try decreasing h[j] and increasing h[k]
        # (move mass from high-gradient to low-gradient coordinate)
        
        improved = False
        # Sample a subset of promising pairs
        idx = np.argsort(-np.abs(g))
        candidates = idx[:min(30, n)]
        
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            j = candidates[ii]
            for kk in range(ii+1, len(candidates)):
                if time.time() > deadline:
                    break
                k = candidates[kk]
                
                if g[j] > g[k]:
                    # Try: decrease h[j], increase h[k]
                    t_max = min(h[j], 1.0 - h[k])
                    if t_max > 1e-8:
                        h_new, c5_new = _pair_golden(h, j, k, t_max, best_c5)
                        if c5_new < best_c5 - 1e-13:
                            h = h_new
                            best_c5 = c5_new
                            best_h = h.copy()
                            improved = True
                    # Also try reverse: increase h[j], decrease h[k]
                    t_max2 = min(1.0 - h[j], h[k])
                    if t_max2 > 1e-8:
                        h_new, c5_new = _pair_golden(h, k, j, t_max2, best_c5)
                        # Note: _pair_golden(h, k, j, ...) means increase k, decrease j
                        # which is the same as decrease j, increase k... 
                        # Wait no, _pair_golden(h, j, k, t_max) does direction e_j - e_k
                        # So _pair_golden(h, k, j, t_max2) does direction e_k - e_j = increase k, decrease j
                        if c5_new < best_c5 - 1e-13:
                            h = h_new
                            best_c5 = c5_new
                            best_h = h.copy()
                            improved = True
        
        if not improved:
            # Try all pairs if no improvement from guided pairs
            for j in range(n):
                if time.time() > deadline:
                    break
                for k in range(j+1, n):
                    if time.time() > deadline:
                        break
                    t_max1 = min(h[j], 1.0 - h[k])
                    t_max2 = min(1.0 - h[j], h[k])
                    if t_max1 <= 1e-8 and t_max2 <= 1e-8:
                        continue
                    if t_max1 > 1e-8:
                        h_new, c5_new = _pair_golden(h, j, k, t_max1, best_c5)
                        if c5_new < best_c5 - 1e-13:
                            h = h_new

                            improved = True
                    if t_max2 > 1e-8:

                        if c5_new < best_c5 - 1e-13:

                            best_c5 = c5_new
                            best_h = h.copy()
                            improved = True
                if time.time() > deadline:
                    break
            if not improved:
                break
    
    return best_h, best_c5

Actually, let me reconsider. The direction that should help is:
- If g[j] > g[k] (the subgradient wants to decrease h[j] more than h[k]), then the descent direction has a component -g, which means we want to decrease h[j] and increase h[k].
- The direction vector is -e_j + e_k = e_k - e_j.
- So we want to search along e_k - e_j, i.e., h + t*(e_k - e_j) for t > 0.
- This means h[k] increases and h[j] decreases.
- Constraint: h[j] - t >= 0 → t <= h[j], and h[k] + t <= 1 → t <= 1 - h[k].
- So t_max = min(h[j], 1 - h[k]).

def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        # Compute gradient for guidance
        g, _, _ = _subgrad_direction(h, alpha=80.0)
        
        # Try all ordered pairs (j, k) where j != k
        # Direction: e_k - e_j means decrease j, increase k
        # We want g[j] > g[k] for this to be a descent direction
        
        # For efficiency, only try pairs where the gradient suggests improvement
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(n):
                if k == j:
                    continue
                if time.time() > deadline:
                    break
                if g[j] <= g[k]:
                    continue  # Not a descent direction
                t_max = min(h[j], 1.0 - h[k])
                if t_max < 1e-8:
                    continue
                h_new, c5_new = _pair_search_1d(h, j, k, best_c5, deadline)
                if c5_new < best_c5 - 1e-13:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        
        if not improved:
            # Fallback: try all pairs without gradient guidance
            for j in range(n):
                if time.time() > deadline:
                    break
                for k in range(n):
                    if k == j:
                        continue
                    if time.time() > deadline:
                        break
                    t_max = min(h[j], 1.0 - h[k])
