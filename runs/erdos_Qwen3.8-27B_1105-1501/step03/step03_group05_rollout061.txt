```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # Need to add more; distribute to elements < 1
        slack = 1.0 - v
        mask = slack > 1e-10
        if not np.any(mask):
            return v
        step = diff / np.sum(mask)
        if step < slack[mask].max():
            v[mask] += step
        else:
            # Some will saturate; use water-filling
            v = np.clip(v + diff / np.sum(mask), 0, 1)
            # Repeat
            for _ in range(20):
                diff = target - np.sum(v)
                if abs(diff) < 1e-12:
                    break
                mask = (v < 1.0 - 1e-10) if diff > 0 else (v > 1e-10)
                if not np.any(mask):
                    break
                v[mask] += diff / np.sum(mask)
                v = np.clip(v, 0, 1)
    else:
        # Need to remove; distribute from elements > 0
        v = np.clip(v, 0, 1)
        for _ in range(20):
            diff = target - np.sum(v)
            if abs(diff) < 1e-12:
                break
            mask = (v > 1e-10) if diff < 0 else (v < 1.0 - 1e-10)
            if not np.any(mask):
                break
            v[mask] += diff / np.sum(mask)
            v = np.clip(v, 0, 1)
    # Final exact fix
    diff = target - np.sum(v)
    if abs(diff) > 1e-12:
        mask = (v < 1.0) if diff > 0 else (v > 0.0)
        v[mask] += diff / np.sum(mask)
        v = np.clip(v, 0, 1)
    return v

def _build_G(h):
    """Build gradient matrix G where G[r, j] = d(score_r * dx)/d(h_j)"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _cvx_trust_step(h, rho, solver=None):
    """Solve one trust-region step using CVXPY."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    F = _scores_raw(h) * dx  # scaled scores
    G = _build_G(h)
    b = F - G @ h  # so that G @ x <= t + b is equivalent to F + G@(x-h) <= t
    
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        G @ x - t <= -b,
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        cp.norm(x - h, 2) <= rho,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    for s in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK', 'GUROBI', 'OSQP']:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                x_val = x.value
                if x_val is not None and np.all(np.isfinite(x_val)):
                    return x_val
        except Exception:
            continue
    return None

def _cvx_trust_optimize(h, deadline, rho_start=0.3, rho_min=0.01):
    """Iterative trust-region optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = rho_start
    stall = 0
    
    while time.time() < deadline:
        x_cand = _cvx_trust_step(h, rho)
        if x_cand is None:
            stall += 1
            rho *= 0.7
            if rho < rho_min:
                break
            continue
        
        # Try different step sizes
        best_step_c5 = cur_c5
        best_x = h.copy()
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15]:
            x_trial = _project_box_sum(h + factor * (x_cand - h), target)
            c5_trial = _exact_c5(x_trial)
            if c5_trial < best_step_c5 - 1e-12:
                best_step_c5 = c5_trial
                best_x = x_trial
        
        if best_step_c5 < cur_c5 - 1e-12:
            h = best_x.copy()
            cur_c5 = best_step_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            rho = min(rho * 1.15, 1.0)
            stall = 0
        else:
            stall += 1
            rho *= 0.7
            if stall > 30 or rho < rho_min:
                break
    
    return best_h, best_c5
```

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_box_sum(v, target):
    n = len(v)

    v = np.clip(v, 0.0, 1.0)

def _lp_minimax_step(h, rho):
    """Solve LP: min t s.t. linearized scores <= t, box, sum, trust."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n

    # Variables: [x_0, ..., x_{n-1}, t]

            return x_star

    return None
