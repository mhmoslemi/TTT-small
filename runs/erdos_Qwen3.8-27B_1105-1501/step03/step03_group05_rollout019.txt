def _smooth_grad(h, beta):
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    exp_vals = np.exp(beta * (scores - max_s))
    ws = float(np.sum(exp_vals))
    if ws < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / ws
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] += -wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] += -wd * h[g:]

        order = rng.permutation(n)
        for i in order:

                break
            k2 = int(rng.integers(n))
            if k2 == i:
                k2 = (k2 + 1) % n
            t, new_c5 = _solve_pair(h, i, k2, cur_c5, 2.0 / n)

def _binary_local_search(h, deadline, rng, max_iters=100000):

        best_swap = None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

        scores_grad[:, 0] = -1.0
        for d in range(-(n-1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                scores_grad[r, 1:m+1] = -dx * (1.0 - h_x[d:])
                scores_grad[r, 1+d:n+1] = -dx * h_x[:m]
            else:
                g = -d
                m = n - g
                scores_grad[r, 1:n-g+1] = -dx * (1.0 - h_x[:m])
                scores_grad[r, 1+g:n+1] = -dx * h_x[m:]
        return np.vstack([jac_obj, scores_grad.T]).T
    cons = [{'type': 'ineq', 'fun': constraint, 'jac': jac}]
    con_eq = [{'type': 'eq', 'fun': lambda x: x[1:].sum() - n/2.0,
               'jac': lambda x: np.zeros(n+1) | 0 + np.concatenate([[0.0], np.ones(n)])}]
    def constraint_eq(x):
        return x[1:].sum() - n / 2.0
    def jac_eq(x):
        j = np.zeros(n + 1)
        j[1:] = 1.0
        return j
    cons.append({'type': 'eq', 'fun': constraint_eq, 'jac': jac_eq})
    bounds = [(0, 1)] * (n + 1)
    h0 = h.copy()
    c5Init = _exact_c5(h0)
    x0 = np.concatenate([[c5Init * 1.01], h0])
    try:
        res = minimize(objective, x0, method='SLSQP', jac=jac, constraints=cons,
                       bounds=bounds, options={'maxiter': 100, 'ftol': 1e-12})
        if res.x is not None:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project_box_sum(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < c5Init:
                return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Pair descent from parent
    deadline1 = t0 + budget_s * 0.20
    h1, c5_1 = _pair_descent(h, deadline1, rng, max_passes=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Continuation optimization
    deadline2 = t0 + budget_s * 0.35
    h2, c5_2 = _continuation_opt(best_h, deadline2, beta_start=5.0, beta_end=200.0, n_beta_steps=12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More pair descent
    deadline3 = t0 + budget_s * 0.50
    h3, c5_3 = _pair_descent(best_h, deadline3, rng, max_passes=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary local search
    deadline4 = t0 + budget_s * 0.62
    hb, c5b = _binary_local_search(best_h, deadline4, rng, max_iters=80000)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 5: Continuation from binary
    deadline5 = t0 + budget_s * 0.72
    h5, c5_5 = _continuation_opt(best_h, deadline5, beta_start=5.0, beta_end=300.0, n_beta_steps=10)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair descent final
    deadline6 = t0 + budget_s * 0.82
    h6, c5_6 = _pair_descent(best_h, deadline6, rng, max_passes=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: SLSQP polish
    if time.time() < t0 + budget_s * 0.88:
        h7, c5_7 = _slsqp_optimize(best_h, t0 + budget_s * 0.90)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Multi-resolution
    if time.time() < deadline:
        remaining = deadline - time.time()
        n_tries = sorted(set(x for x in [best_n-4, best_n+4, best_n+8, best_n-8, 48, 72, 96, 112]
                             if 24 <= x <= 160))
        per_n_time = remaining / (len(n_tries) + 1)
        for n_try in n_tries:
            if time.time() > deadline:
                break
            h_r = _resample(best_h, n_try)
            dl_r = min(deadline, time.time() + per_n_time * 0.7)
            h_r, c5_r = _continuation_opt(h_r, dl_r, beta_start=5.0, beta_end=200.0, n_beta_steps=6)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
            if time.time() > deadline:
                break
            dl_r2 = min(deadline, time.time() + per_n_time * 0.3)
            h_r2, c5_r2 = _pair_descent(best_h, dl_r2, rng, max_passes=50)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_try

    # Phase 9: Final pair polish
    if time.time() < deadline - 5:
        h9, c5_9 = _pair_descent(best_h, deadline - 2, rng, max_passes=100)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
            best_n = len(best_h)

    # Final projection and verification
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, final_c5, best_n
