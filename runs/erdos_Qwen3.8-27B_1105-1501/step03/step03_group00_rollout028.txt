∂Φ_p/∂h_j = (p/m) * Σ_r (s_r + ε)^(p-1) * ∂s_r/∂h_j

This is computable from the gradient matrix G (same as in the parent).

- This is equivalent to finding a direction that pushes all active lags down simultaneously
- A simple approach: d = -G_active^T @ w where w are weights proportional to (s_r - target) for active lags

    best_c5 = cur_c5

        n_iterations = 50000

    cooling = (T_min / T) / max(1, n_iterations)
    
    rng = np.random.default_rng(42)
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]

        if time.time() > deadline: break

        i = int(ones_idx[rng.integers(len(ones_idx))])

        if i == j: continue

                best_h = h_bin.copy()

            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]  # revert

                    lo, hi = 0.0, a_max

                        h1[j] += sign * m1

                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)

        else:

    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial LP refinement (8%)
    dl1 = t0 + budget_s * 0.08
    hdr, c5r = _lp_optimize(h, dl1, top_k=None, radius=0.2, max_iters=1500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 2: P-norm annealing - main engine (28%)
    dl2 = t0 + budget_s * 0.36
    hdr, c5r = _pnorm_descent(best_h, dl2, p_start=6.0, p_end=2048.0, n_steps=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: Active set equalization (10%)
    dl3 = t0 + budget_s * 0.46
    hdr, c5r = _active_equalize(best_h, dl3, n_steps=600)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 4: LP refinement (15%)
    dl4 = t0 + budget_s * 0.61
    hdr, c5r = _lp_optimize(best_h, dl4, top_k=12, radius=0.06, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: P-norm with higher p (12%)
    dl5 = t0 + budget_s * 0.73
    hdr, c5r = _pnorm_descent(best_h, dl5, p_start=128.0, p_end=16384.0, n_steps=800)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 6: Multi-start perturbation (12%)
    dl6 = t0 + budget_s * 0.85
    scale = 0.025
    for attempt in range(300):
        if time.time() > dl6:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.018)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k=10, radius=0.07, max_iters=250)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 7: Resolution sweep (6%)
    dl7 = t0 + budget_s * 0.91
    for dn in [-4, -3, -2, -1, 2, 3, 4, 6, 8, 12]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 144:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k=10, radius=0.1, max_iters=150)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final ultra-fine LP (4%)
    dl8 = t0 + budget_s * 0.95
    hdr, c5r = _lp_optimize(best_h, dl8, top_k=4, radius=0.012, max_iters=4000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 9: Pair polish (3%)
    if time.time() < t0 + budget_s * 0.98:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
