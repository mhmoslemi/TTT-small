def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):

def _gradient_matrix(h):
    """Build the (2n-1) x n gradient matrix for all scores."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(1, n):
        # Positive lag d: score = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # d(h[i+d](1-h[i]))/dh[i] = -h[i+d]
        # d(h[i+d](1-h[i]))/dh[i+d] = (1-h[i])
        m = n - d
        G[n-1+d, :m] += -h[d:]      # derivative w.r.t. h[i] for i in 0..m-1
        G[n-1+d, d:] += (1.0 - h[:m])  # derivative w.r.t. h[i+d] for i in 0..m-1
        # Negative lag -d: score = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # d(h[i](1-h[i+d]))/dh[i] = (1-h[i+d])
        # d(h[i](1-h[i+d]))/dh[i+d] = -h[i]
        G[n-1-d, :m] += (1.0 - h[d:])   # derivative w.r.t. h[i]
        G[n-1-d, d:] += -h[:m]           # derivative w.r.t. h[i+d]
    # Lag 0: score = sum_i h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
    # derivative = 1 - 2*h[i]
    G[n-1, :] = 1.0 - 2.0 * h
    return G

    n = len(scores)

    c[-1] = 1.0

    A_ub[:, :n_h] = G
    A_ub[:, n_h] = -1.0
    b_ub = np.zeros(len(scores))

        if active_mask[i]:
            b_ub[i] = -scores[i]  # scores[i] + G[i]@p - t <= 0 => G[i]@p - t <= -scores[i]
        else:
            b_ub[i] = -scores[i] + radius  # relaxed: allow scores to increase by up to radius

    b_eq = np.zeros(1)
    
    bounds = []

    bounds.append((None, None))

def _feasible_interval(h, d):
    """Find the interval [a_min, a_max] such that h + a*d is in [0,1]^n."""
    n = len(h)
    a_min = 0.0
    a_max = 0.0
    for i in range(n):

            a_min = max(a_min, -h[i] / d[i])
            a_max = min(a_max, (1.0 - h[i]) / d[i])

            a_min = max(a_min, (1.0 - h[i]) / d[i])
            a_max = min(a_max, -h[i] / d[i])
    if a_max < a_min:
        return None
    return a_min, a_max

def _line_search_exact(h, d, c5_cur):

    interval = _feasible_interval(h, d)
    if interval is None:
        return 0.0, c5_cur
    a_min, a_max = interval
    if a_max - a_min < 1e-14:
        return 0.0, c5_cur
    
    # Shrink slightly for safety
    margin = (a_max - a_min) * 0.01
    a_min_s = a_min + margin
    a_max_s = a_max - margin
    if a_max_s <= a_min_s:
        a_min_s = a_min
        a_max_s = a_max
    
    best_a = 0.0
    best_c5 = c5_cur

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, target, deadline):

    n = len(h)

            for k in range(j + 1, n):

                # Try transferring from j to k and k to j

                    if sign > 0:

                    else:

                    if a_max < 1e-12:

                    # Try a few fractions of max
                    for frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
                        a = a_max * frac
                        h_try = h.copy()
                        if sign > 0:

                            h = h_try.copy()

        if not improved:

    return best_h, best_c5

def _structured_init(n, pattern='sine', freq=1.0):
    """Generate structured initial conditions."""
    x = np.arange(n) / n
    if pattern == 'sine':
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x)
    elif pattern == 'cosine':
        h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x)
    elif pattern == 'ramp':
        h = 0.5 + 0.3 * (2 * x - 1)
    elif pattern == 'triangle':
        h = 0.5 + 0.3 * (1 - 4 * np.abs(x - 0.5))
    elif pattern == 'sine2':
        h = 0.5 + 0.2 * np.sin(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 4 * x)
    else:
        h = np.full(n, 0.5)
    return _project(h, n / 2.0)

def _momentum_refine(h, target, deadline, beta=20.0):

    n = len(h)

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = 0.01
    momentum = 0.85
    T = 0.003

        # Smoothed max approximation
        scores = _scores(h)
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        w = exp_s / float(np.sum(exp_s))
        
        G = _gradient_matrix(h)
        grad = G.T @ w
        grad = grad - np.mean(grad)
        
        v = momentum * v + lr * grad

        c5_new = _exact_c5(h_new)

            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.2)
        else:
            lr *= 0.6
            v *= 0.3

def _exact_c5(h):

    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _scores(h):

def _gradient_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(1, n):
        m = n - d
        G[n - 1 + d, :m] -= h[d:]
        G[n - 1 + d, d:] += (1.0 - h[:m])
        G[n - 1 - d, :m] += (1.0 - h[d:])
        G[n - 1 - d, d:] -= h[:m]
    G[n - 1, :] = 1.0 - 2.0 * h
    return G

                        else:
                            h_try[j] += a
                            h_try[k] -= a
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_c5 = c5_try
                            h = h_try.copy()
                            best_h = h.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5


def _momentum_refine(h, target, deadline, beta=30.0):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = 0.012
    mom = 0.84
    while time.time() < deadline:
        scores = _scores(h)
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        w = exp_s / float(np.sum(exp_s))
        G = _gradient_matrix(h)
        grad = G.T @ w
        grad = grad - np.mean(grad)
        v = mom * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.6
            v *= 0.25
            h = h_new
    return best_h, best_c5


def _random_restart(h, target, n, rng, scale=0.05):
    noise = rng.standard_normal(n)
    noise = noise - np.mean(noise)
    h_new = h + scale * noise
    h_new = _project(h_new, target)
    return h_new


def _structured_h(n, mode, rng):
    x = np.arange(n) / n
    if mode == 0:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.25 * np.cos(4 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.2 * np.sin(2 * np.pi * x) + 0.1 * np.sin(6 * np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.25 * (2 * x - 1)
    elif mode == 4:
        h = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * x)
    else:
        h = 0.5 + 0.15 * np.sin(2 * np.pi * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.97

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP from parent (20%)
    dl1 = t0 + budget_s * 0.20
    h1, c51 = _lp_optimize(h, target0, dl1, radius_init=0.12, active_frac=0.03)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Momentum (12%)
    dl2 = t0 + budget_s * 0.32
    h2, c52 = _momentum_refine(best_h, best_n / 2.0, dl2, beta=35.0)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    # Phase 3: LP again (15%)
    dl3 = t0 + budget_s * 0.47
    h3, c53 = _lp_optimize(best_h, best_n / 2.0, dl3, radius_init=0.08, active_frac=0.02)
    if c53 < best_c5:
        best_c5 = c53
        best_h = h3.copy()

    # Phase 4: Resolution exploration (30%)
    dl4 = t0 + budget_s * 0.77
    n_candidates = [best_n + dn for dn in [1, -1, 2, -2, 4, -4, 8, -8, 16, 32, -16, 3, -3, 6, 12, 24]]
    n_candidates = [x for x in n_candidates if 24 <= x <= 200]
    n_candidates = sorted(set(n_candidates))
    for n_try in n_candidates:
        if time.time() > dl4:
            break
        h_r = _resample(best_h, n_try)
        d_sub = min(dl4, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_optimize(h_r, n_try / 2.0, d_sub, radius_init=0.10, active_frac=0.03)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Structured restarts + random restarts (10%)
    dl5 = t0 + budget_s * 0.87
    for mode in range(6):
        if time.time() > dl5:
            break
        h_init = _structured_h(best_n, mode, rng)
        h_mix = 0.4 * h_init + 0.6 * best_h
        h_mix = _project(h_mix, best_n / 2.0)
        d_sub = min(dl5, time.time() + budget_s * 0.012)
        h_mix, c5_mix = _lp_optimize(h_mix, best_n / 2.0, d_sub, radius_init=0.1)
        if c5_mix < best_c5 - 1e-11:
            best_c5 = c5_mix
            best_h = h_mix.copy()
    for _ in range(6):
        if time.time() > dl5:
            break
        for scale in [0.03, 0.06, 0.1]:
            h_rr = _random_restart(best_h, best_n / 2.0, best_n, rng, scale)
            d_sub = min(dl5, time.time() + budget_s * 0.008)
            h_rr, c5_rr = _lp_optimize(h_rr, best_n / 2.0, d_sub, radius_init=0.08)
            if c5_rr < best_c5 - 1e-11:
                best_c5 = c5_rr
                best_h = h_rr.copy()

    # Phase 6: Pair polish + final LP (8%)
    dl6 = hard_deadline
    hp, c5p = _pair_polish(best_h, best_n / 2.0, min(dl6, t0 + budget_s * 0.94))
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    if time.time() < hard_deadline - 10:
        hf, c5f = _lp_optimize(best_h, best_n / 2.0, hard_deadline, radius_init=0.03, active_frac=0.015)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
