def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)

    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

            lo = mid
        else:

def _grad_matrix(h):
    """Compute the Jacobian of all 2n-1 correlation rows w.r.t. h."""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _correlation_vector(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _grad_rows(h, lag_indices):
    n = len(h)
    rows = len(lag_indices)
    G = np.zeros((rows, n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] = (1.0 - h[d:d + m])
                G[idx, d:d + m] += (-h[0:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] = (-h[g:n])
                G[idx, g:g + m] += (1.0 - h[0:m])
    return G


def _grad_all(h):
    n = len(h)
    return _grad_rows(h, np.arange(2 * n - 1))


def _lp_direction(h, G_sub, scores_sub, radius):
    n = len(h)
    k = len(scores_sub)
    if k == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -scores_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        if lo_j > hi_j:
            lo_j = hi_j = 0.0
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, d, c5_cur):
    n = len(h)
    alpha_max = 5.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 0.0)
    if alpha_max < 1e-14:
        return 0.0, c5_cur

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-12:
            continue
        c5_t = _exact_c5(np.clip(h + a * d, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a <= 0:
        return 0.0, c5_cur

    lo = max(0.0, best_a - alpha_max * 0.2)
    hi = best_a + alpha_max * 0.2
    if hi > alpha_max:
        hi = alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * d, 0.0, 1.0))
    f2 = _exact_c5(np.clip(h + c2 * d, 0.0, 1.0))
    for _ in range(22):
        if hi - lo < 1e-13 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * d, 0.0, 1.0))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * d, 0.0, 1.0))
    a = (lo + hi) * 0.5
    c5_t = _exact_c5(np.clip(h + a * d, 0.0, 1.0))
    if c5_t < best_c5:
        best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _active_lp_loop(h, deadline, k_list, radius_init, max_iters, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    it = 0

    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        scores = _correlation_vector(h)
        k = k_list[it % len(k_list)]
        k = min(k, 2 * n - 1)
        active_idx = np.argsort(scores)[-k:]
        active_scores = scores[active_idx]

        G_sub = _grad_rows(h, active_idx)
        d = _lp_direction(h, G_sub, active_scores, radius)

        if d is None or float(np.max(np.abs(d))) < 1e-13:
            stall += 1
            radius *= 0.82
            if radius < 1e-7:
                break
            continue

        alpha, new_c5 = _line_search(h, d, cur_c5)

        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.35)
        else:
            stall += 1
            radius *= 0.72
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.08

    return best_h, best_c5


def _smooth_obj(x, alpha):
    s = _correlation_vector(x)
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -400.0, 400.0)
    ev = np.exp(vals)
    total = float(np.sum(ev))
    f = ms + (1.0 / alpha) * np.log(total)
    w = ev / total
    G = _grad_all(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g


def _smooth_loop(h, deadline, alphas):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h, args=(alpha,), jac=True,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 250, 'ftol': 1e-15, 'maxls': 40, 'iprint': -1})
            h_new = _project(np.clip(res.x, 0.0, 1.0), target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h, args=(alpha,), jac=True,
                           method='L-BFGS-B', bounds=bounds,

            h_new = _project(np.clip(res.x, 0.0, 1.0), target)

            if c5_new < best_c5:

                best_h = h_new.copy()
            h = h_new

def _subgradient_step(h, scores, active_idx, radius):
    n = len(h)
    G_sub = _grad_rows(h, active_idx)
    w = np.ones(len(active_idx)) / len(active_idx)
    d = -(G_sub.T @ w)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    d = d / norm * radius
    cur_c5 = _exact_c5(h)
    a, c5_new = _line_search(h, d, cur_c5)
    if a > 1e-12:
        return _project(h + a * d, n / 2.0), c5_new
    return h.copy(), cur_c5


def _pair_polish(h, deadline, max_passes=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 1):
            if time.time() > deadline:
                break
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            for k_j in range(j + 1, n, 1):
                if h[k_j] <= 1e-6 or h[k_j] >= 1.0 - 1e-6:
                    continue
                for (dj, dk) in ((1.0, -1.0), (-1.0, 1.0)):
                    a_max = 0.0
                    if dj > 0:
                        a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max if a_max > 0 else 1e9, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[k_j]) / dk)
                    else:
                        a_max = min(a_max if a_max > 0 else 1e9, h[k_j] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for frac in (0.5, 1.0, 0.25, 0.75):
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k_j] += dk * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_h, best_c5
