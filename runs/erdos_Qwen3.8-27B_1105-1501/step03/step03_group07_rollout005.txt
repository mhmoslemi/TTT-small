def _exact_c5(h):

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_corrs_and_grads(h):
    """Compute all correlation values and their gradients."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2n-1, index r corresponds to lag d = r - (n-1)
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n - 1)
        if d >= 0:

def _lp_direction(h, corr, grads, R, lam):
    """Solve LP to find best descent direction."""
    n = len(h)
    m = 2 * n - 1
    # Variables: t (index 0), p_0..p_{n-1} (indices 1..n), q_0..q_{n-1} (indices n+1..2n)
    nvar = 1 + 2 * n
    # Objective: minimize t + lam * sum(p + q)
    c_obj = np.zeros(nvar)
    c_obj[0] = 1.0
    c_obj[1:1+n] = lam
    c_obj[1+n:1+2*n] = lam
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag r: grad_r · (p-q) - t <= -corr[r]
    # i.e., -t + grad_r·p - grad_r·q <= -corr[r]
    A_ub = np.zeros((m + 1, nvar))
    b_ub = np.zeros(m + 1)
    for r in range(m):
        A_ub[r, 0] = -1.0
        A_ub[r, 1:1+n] = grads[r]
        A_ub[r, 1+n:1+2*n] = -grads[r]
        b_ub[r] = -corr[r]
    # L1 constraint: sum(p+q) <= R
    A_ub[m, 1:1+n] = 1.0
    A_ub[m, 1+n:1+2*n] = 1.0
    b_ub[m] = R
    
    # Equality constraint: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)

    for i in range(n):

            bounds.append((0, 0))     # q_i

            bounds.append((0, 0))     # p_i
            bounds.append((0, 1.0))   # q_i
        else:
            bounds.append((0, 1.0 - h[i]))
            bounds.append((0, h[i]))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    
    x = res.x
    p = x[1:1+n]
    q = x[1+n:1+2*n]
    d = p - q
    # Subtract mean to ensure sum = 0 (should already be 0 from equality constraint)
    d -= np.mean(d)
    
    dnorm = float(np.linalg.norm(d))
    if dnorm < 1e-12:

    return d / dnorm  # Normalize to unit length

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_step(h, R=1.0, lam=0.0):
    """Find best descent direction via LP. Returns (direction, predicted_reduction)."""
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    
    # Compute gradients
    # For efficiency, precompute gradient matrix
    # grad[r, j] = ∂corr[r]/∂h[j]
    grads = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                grads[r, d:] += (1.0 - h[:mm])
                grads[r, :mm] -= h[d:]
        else:
            l = -d
            mm = n - l
            if mm > 0:
                grads[r, :mm] += (1.0 - h[l:])
                grads[r, l:] -= h[:mm]
    
    # LP: variables t, p_0..p_{n-1}, q_0..q_{n-1}
    nv = 1 + 2 * n
    c = np.zeros(nv)
    c[0] = 1.0
    c[1:1+n] = lam
    c[1+n:] = lam
    
    A_ub = np.zeros((m + 1, nv))
    b_ub = np.zeros(m + 1)
    for r in range(m):
        A_ub[r, 0] = -1.0
        A_ub[r, 1:1+n] = grads[r]
        A_ub[r, 1+n:] = -grads[r]
        b_ub[r] = -corr[r]
    # L1 bound
    A_ub[m, 1:] = 1.0
    b_ub[m] = R
    
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1000.0, 1000.0)]
    for i in range(n):
        hi_i = 1.0 - h[i]
        lo_i = h[i]
        if h[i] < 1e-10:
            bounds.append((0, R))
            bounds.append((0, 0))
        elif h[i] > 1.0 - 1e-10:
            bounds.append((0, 0))
            bounds.append((0, R))
        else:
            bounds.append((0, min(hi_i, R)))
            bounds.append((0, min(lo_i, R)))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, 0.0
    
    d = res.x[1:1+n] - res.x[1+n:]
    dnorm = float(np.linalg.norm(d))
    if dnorm < 1e-12:
        return None, 0.0
    predicted = max_corr - res.x[0]
    return d / dnorm, predicted

        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:

            best_a = a

                best_a = mid
                hi = mid
            else:

    return best_a, best_c
