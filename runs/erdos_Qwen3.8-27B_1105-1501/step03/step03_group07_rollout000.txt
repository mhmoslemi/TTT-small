                dj = 1.0 if rng.random() > 0.5 else -1.0
                dk = -dj
                a_max = min((1.0-h[j])/dj if dj>0 else h[j],
                            (1.0-h[k])/dk if dk>0 else h[k])
                if a_max < 1e-8:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_h, best_c5

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_constrained(h0, deadline, max_iter=200):
    """SLSQP with explicit max constraints (t >= f_k for all k)"""
    n = len(h0)
    dx = 2.0 / n
    
    def objective(x):
        return x[n]  # minimize t
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    # Constraints: t - f_k(h)*dx >= 0 for all k
    h_start = h0.copy()
    t_start = _exact_c5(h_start) + 0.005
    
    # For efficiency, only use top lags as constraints
    corr = np.correlate(h_start, 1.0-h_start, mode='full')
    max_corr = np.max(corr)
    # Lags within 95% of max
    active = np.where(corr >= 0.9 * max_corr)[0]
    
    cons = [{'type': 'eq', 'fun': lambda x, ns=n: np.sum(x[:ns]) - ns/2.0,
             'jac': lambda x, ns=n: np.concatenate([np.ones(ns), [0]])}]
    
    for r in active:
        s = r - (n-1)
        def make_con(r, s):
            def fun(x, _r=r, _s=s):
                h = np.clip(x[:n], 0, 1)
                return x[n] - np.correlate(h, 1.0-h, mode='full')[_r] * dx
            def jac(x, _r=r, _s=s):
                h = np.clip(x[:n], 1e-10, 1.0-1e-10)
                g = np.zeros(n+1)
                g[n] = 1.0
                sg = _lag_grad(h, _s) * dx
                g[:n] = -sg
                return g
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        cons.append(make_con(r, s))
    
    bounds = [(0, 1)] * n + [(0, 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    res = minimize(objective, x0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    
    if res.x is not None:
        h_res = _project(res.x[:n], n/2.0)
        return h_res, _exact_c5(h_res)
    return h0.copy(), _exact_c5(h0)
