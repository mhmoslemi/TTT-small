    A_eq = np.zeros((1, n + 1))

def _binary_anneal(h_start, deadline, rng, n_iters=500000):
    """Simulated annealing on binary vectors near h_start"""
    n = len(h_start)
    target = n / 2.0
    # Start from thresholding
    h = h_start.copy()
    # Make it binary
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:int(target)]] = 1.0
    cur_c5 = _exact_c5(b)
    best_h = b.copy()
    best_c5 = cur_c5
    
    T_init = 0.02
    T_final = 0.001
    it = 0
    while it < n_iters and time.time() < deadline:
        # Batch of moves
        n_batch = min(1000, n_iters - it)
        for _ in range(n_batch):
            if time.time() > deadline:
                break
            # Random swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if b[i] == b[j]:
                continue
            # Swap
            b[i], b[j] = b[j], b[i]
            new_c5 = _exact_c5(b)
            T = T_init * (T_final / T_init) ** (it / n_iters)
            if new_c5 < cur_c5 or (rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10))):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = b.copy()
            else:
                # Reject
                b[i], b[j] = b[j], b[i]
            it += 1
    return best_h, best_c5

        pop = np.vstack([pop, pop[0:n_add] + rng_base.normal(0, 0.03, (n_add, n_params))])
        scores = np.concatenate([scores, np.array([_eval(pop[i]) for i in range(len(pop))])[pop_size//2:]])

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lse_lipschitz(h, beta, deadline, max_iters=200):
    """Log-sum-exp optimization with L-BFGS"""
    n = len(h)
    target = n / 2.0

        scores = _scores(h)
        m = np.max(scores)
        exp_s = np.exp(beta * (scores - m))
        w = exp_s / np.sum(exp_s)

        G = _grad_matrix(h)
        grad = G.T @ w
        return val, grad

                   options={'maxiter': max_iters, 'gtol': 1e-10, 'maxfun': max_iters * 10})
