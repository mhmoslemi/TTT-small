    def obj_grad(z_cur):
        h = _z_to_h(z_cur, n)
        h = _project_feasible(h, n/2.0)  # Should already be feasible
        scores = _scores(h)
        M = np.max(scores)
        exps = np.exp(beta * (scores - M))
        weights = exps / np.sum(exps)
        F = M + np.log(np.mean(exps)) / beta
        g_h = _grad_weighted_scores(h, weights)

        S = np.dot(a, g_h)
        g_z = a * (g_h - S / A)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(h, target):
    n = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(h - tau, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _h_to_z(h):
    eps = 1e-12
    hc = np.clip(h, eps, 1.0 - eps)
    z = np.log(hc / (1.0 - hc))
    return z - np.mean(z)

def _z_to_h(z, n):
    z = z - np.mean(z)
    target = n / 2.0
    # Solve sum sigmoid(z_i + c) = target
    lo = -float(np.max(np.abs(z))) - 50.0
    hi = float(np.max(np.abs(z))) + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(1.0 / (1.0 + np.exp(-(z + mid))))
        if s > target:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    u = np.clip(z + c, -500, 500)
    h = 1.0 / (1.0 + np.exp(-u))
    return h

def _grad_weighted_scores(h, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for d in range(-(n-1), n):
        w = weights[n-1+d]
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            grad[:m] += w * omh[d:]
            grad[d:] -= w * h[:m]
        else:
            g = -d
            m = n - g
            grad[g:] += w * omh[:m]
            grad[:m] -= w * h[g:]
    return grad

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    omh = 1.0 - h
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] = omh[d:].copy()
            G[r, d:] -= h[:m]
        else:
            g = -d
            m = n - g
            G[r, g:] = omh[:m].copy()
            G[r, :m] -= h[g:]
    return G

def _zspace_objgrad(z, n, beta):
    h = _z_to_h(z, n)
    scores = _scores(h)
    M = np.max(scores)
    shifted = np.clip(beta * (scores - M), -700.0, 700.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    if total < 1e-300:
        weights = np.zeros(len(scores))
        weights[np.argmax(scores)] = 1.0
    else:
        weights = exps / total
    F = M + np.log(max(total, 1e-300)) / beta
    g_h = _grad_weighted_scores(h, weights)
    a = h * (1.0 - h)
    A = np.sum(a)
    if A < 1e-15:
        return F, np.zeros(n)
    S = np.dot(a, g_h)
    g_z = a * (g_h - S / A)
    return F, g_z

def _lbfgs_z(z, n, beta, maxiter, deadline):

        if time.time() > deadline:
            raise Exception("deadline exceeded")
        return _zspace_objgrad(z_cur, n, beta)
    z_cur = z.copy()
    try:
        res = minimize(objgrad, z_cur, method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15,
                                'maxls': 50})
        if hasattr(res, 'x') and res.x is not None:
            return res.x
    except Exception:
        pass
    return z_cur

        total = np.sum(exps)
