```python
        # Find all 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        best_gain = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                # Compute new f after swapping i (1->0) and j (0->1)
                new_f = f.copy()
                # Update all affected lags
```

For simplicity, let me just recompute f and deltas from scratch each iteration. For n=64, this is fast enough.

Let me rewrite:

```python
def _compute_deltas(h):
    n = len(h)
    d1 = np.zeros((n, 2*n-1))
    d0 = np.zeros((n, 2*n-1))
    for s_idx in range(2*n-1):
        s = s_idx - (n-1)
        # For each i where i+s is valid: direct term
        valid_direct = np.arange(max(0, -s), min(n, n-s))
        # h[i] * (1 - h[i+s])
        # d1[i] -= (1-h[i+s]) for i in valid_direct
        # d0[i] += (1-h[i+s]) for i in valid_direct
        d1[valid_direct, s_idx] = -(1.0 - h[valid_direct + s])
        d0[valid_direct, s_idx] = (1.0 - h[valid_direct + s])
        
        # For each i where i-s is valid: indirect term (h[i-s] * (1-h[i]))
        valid_indirect = np.arange(max(0, s), min(n, n+s))
        # h[i-s] * (1-h[i]) where i is in valid_indirect
        # d1[i] += h[i-s] (h[i] goes 1->0, so (1-h[i]) goes 0->1)
        # d0[i] -= h[i-s] (h[i] goes 0->1, so (1-h[i]) goes 1->0)
        d1[valid_indirect, s_idx] += h[valid_indirect - s]
        d0[valid_indirect, s_idx] -= h[valid_indirect - s]
    return d1, d0
```

This is vectorized and should be fast.

```python
def _binary_local_search(h, max_iters=10000, deadline=None):

        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        d1, d0 = _compute_deltas(h)
        
        d1_sub = d1[ones]  # (n1, 2n-1)
        d0_sub = d0[zeros]  # (n0, 2n-1)
        
        # new_max[i,j] = max_s (f[s] + d1_sub[i,s] + d0_sub[j,s])
        # Compute efficiently
        new_f = f[None, None, :] + d1_sub[:, None, :] + d0_sub[None, :, :]
        new_max = new_f.max(axis=2)
        
        best_val = float(new_max.min())
        if best_val >= cur_max - 1e-12:
            break
        
        flat_idx = int(new_max.argmin())
        i_idx = flat_idx // len(zeros)
        j_idx = flat_idx % len(zeros)
        
        h[ones[i_idx]] = 0.0
        h[zeros[j_idx]] = 1.0
    
    f = _all_lags(h)
    return h, float(f.max())
```

The binary local search avoids this issue because it evaluates the exact objective at each step. The question is whether the optimum is binary or not.

For the Erdős problem, I believe the optimal continuous solution is NOT exactly binary (it has some fractional values), but it's close to binary. So the binary local search should get close, and then a continuous refinement can squeeze out the last bit.

```python
def _all_lags(h):

def _project(v, target):
    n = len(v)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt

        if neg_mask.any():

        a_max = max(a_max, 1e-8)

        for a in alphas:

            c_t = _exact_c5(h_t)
            if c_t < best_c - 1e-13:
                best_c = c_t
                best_a = a
        
        if best_a > 0 and best_c < cur_c5 - 1e-13:
            h = _project(h + best_a * v, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            T = max(0.05, T * 0.97)
            beta_mom = min(0.9, beta_mom + 0.002)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.8
            T = max(0.05, T * 0.9)
            if stall > 40:
                v *= 0.2
                alpha = 0.03
                T = 0.15
    
    return best_h, best_c5

        j = interior[rng.integers(0, len(interior))]

            a_max = min(a_max_i, a_max_j)
            if a_max < 1e-8:

            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:

                    h = h_t.copy()

                    interior = np.where((h > 1e-4) & (h < 1.0 - 1e-4))[0]
                    break
    
    return best_h, best_c5
```

```python
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()
        best_n = len(h_c)
    
    # Phase 3: SLSQP optimization (15%)
    dl3 = t0 + budget_s * 0.75
    h_s, c5_s = _slsqp_optimize(best_h, dl3, max_iters=30)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
    
    # Phase 4: Binary search again on refined solution (10%)
    dl4 = t0 + budget_s * 0.85
    h_bin_final = np.zeros(best_n)
    idx_f = np.argsort(-best_h)
    h_bin_final[idx_f[:best_n//2]] = 1.0
    h_bf, fbf = _binary_local_search(h_bin_final, max_iters=5000, deadline=dl4, seed=seed+200)
    c5_bf = fbf * (2.0 / best_n)
    if c5_bf < best_c5 - 1e-9:
        best_c5 = c5_bf
        best_h = h_bf.copy()
    
    # Phase 5: Final polish (15%)
    dl5 = deadline
    h_p, c5_p = _pair_micro(best_h, min(dl5, time.time()+30), n_tries=10000, seed=seed+50)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()
    
    remaining = dl5 - time.time()
    if remaining > 8:
        h_f, c5_f = _continuous_refine(best_h, dl5, max_iters=20000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    if final_c5 > _exact_c5(h_init) + 0.01:
        best_h = h_init.copy()
        final_c5 = _exact_c5(h_init)
        best_n = n0
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
```
