def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_dir(h, scores, G, radius, K):

    L = 2 * n - 1

    A_ub[:, -1] = -1.0

    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi2 = min(radius, 1.0 - h[j])
        if lo < hi2:
            bounds.append((lo, hi2))
        else:
            bounds.append((0.0, 0.0))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')

            return res.x[:n]
    except Exception:
        pass
    return None
