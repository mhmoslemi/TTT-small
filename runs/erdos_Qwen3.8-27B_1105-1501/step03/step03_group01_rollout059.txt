    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    # Gradient of sum_k w_k * G_k
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        # First term: (1-h[j+d]) for valid j
        if d >= 0:
            if n - d > 0:
                grad[:n-d] += wk * (1.0 - h[d:])
        else:
            gd = -d
            if n - gd > 0:
                grad[gd:] += wk * (1.0 - h[:n-gd])
        # Second term: -h[j-d] for valid j
        if d >= 0:
            if n - d > 0:
                grad[d:] += -wk * h[:n-d]
        else:
            gd = -d
            if n - gd > 0:
                grad[:n-gd] += -wk * h[gd:]
    return grad

def _nesterov_search(h, deadline, tau0=50.0, tau1=200.0, lr0=0.01):
    """Nesterov-accelerated smoothed gradient search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    y = h.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr0
    beta = 0.9
    stall = 0
    
    while time.time() < deadline:
        tau = tau0 + (tau1 - tau0) * min(1.0, stall / 30.0)
        grad = _softmax_grad(y, tau)
        # Make gradient mean-zero
        grad -= np.mean(grad)
        v = beta * v + (1 - beta) * grad
        y_new = y + v
        y_new = _project(y_new, target)
        
        # Line search
        h_trial = y_new
        c5_trial = _exact_c5(h_trial)
        if c5_trial < cur_c5 - 1e-12:
            h = h_trial
            cur_c5 = c5_trial
            y = y_new
            stall = 0
            lr = min(lr * 1.02, 0.05)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.95
            if stall > 50:
                break
            # Reset momentum periodically
            if stall % 20 == 0:
                v = np.zeros(n)
                y = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_perturb(h, rng, n_modes=6, amp=0.05):
    """Add small Fourier mode perturbations."""
    n = len(h)
    x = (np.arange(n) + 0.5) / n
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp_k = amp / k
        pert += amp_k * np.sin(2 * np.pi * k * x + phase)
    h_new = h + pert
    return _project(h_new, n / 2.0)

def _pair_descent(h, deadline):
    """Optimize pairs of coordinates while maintaining the sum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
        improved = False
        # Try all pairs (or a subset for speed)
        step = max(1, n // 50)  # For large n, skip some
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, min(j + 6, n)):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try moving mass from j to k (and vice versa)
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / abs(dj)
                    else:
                        am = h[j] / abs(dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / abs(dk))
                    else:
                        am = min(am, h[k] / abs(dk))
                    if am < 1e-10:
                        continue
                    # Binary search for best alpha in [0, am]
                    lo_a, hi_a = 0.0, am
                    best_a = 0.0
                    for _ in range(15):
                        mid_a = (lo_a + hi_a) / 2.0
                        h_t = h.copy()

                        h[k] += dk * best_a
                        h = _project(h, target)
        
        if not improved:
            break
    
    return best_h, best_c5

            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.1
                top_m = max(8, int(0.2 * (2 * n - 1)))
            if stall % 30 == 0:
                h = _project(best_h + np.random.default_rng(it * 7).normal(0, 0.01, n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_pert(h, rng, amp=0.05, n_modes=8):
    n = len(h)
    x = (np.arange(n) + 0.5) / n
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        ph = rng.uniform(0, 2 * np.pi)
        pert += (amp / k) * np.sin(2.0 * np.pi * k * x + ph)
    return _project(h + pert, n / 2.0)

def _pair_refine(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        step = max(1, n // 40)
        for j in range(0, n, step):
            if time.time() > deadline:
                return best_h, best_c5
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, min(j + 8, n)):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am1 = (1.0 - h[j]) / abs(dj) if dj != 0 else 1e10
                    am2 = (1.0 - h[k]) / abs(dk) if dk != 0 else 1e10
                    am = min(am1, am2)
                    if am < 1e-10:
                        continue
                    best_a = 0.0
                    best_c = best_c5
                    lo_a, hi_a = 0.0, am
                    for _ in range(18):
                        mid_a = 0.5 * (lo_a + hi_a)
                        h_t = h.copy()
                        h_t[j] = h[j] + dj * mid_a
                        h_t[k] = h[k] + dk * mid_a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c - 1e-13:
                            best_c = c5_t
                            best_a = mid_a
                            lo_a = mid_a
                        else:
                            hi_a = mid_a
                    if best_a > 0:
                        h[j] += dj * best_a
                        h[k] += dk * best_a
                        h = _project(h, target)
                        if best_c < best_c5:
                            best_c5 = best_c
                            best_h = h.copy()
                        improved_any = True
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Nesterov warmup
    d1 = t0 + budget_s * 0.06
    hh, cc = _nesterov(h, d1, tau0=60.0, tau1=300.0, lr0=0.015)
    if cc < best_c5:
        best_c5 = cc
        best_h = hh.copy()

    # Phase 2: LP trust region (primary)
    d2 = t0 + budget_s * 0.38
    hh, cc = _lp_loop(best_h, d2, radius_init=0.18, top_init=10)
    if cc < best_c5:
        best_c5 = cc
        best_h = hh.copy()

    # Phase 3: Multi-resolution
    d3 = t0 + budget_s * 0.60
    n_candidates = list(range(max(40, best_n - 24), best_n + 60, 3))
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try < 32 or n_try > 220:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.022)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, top_init=10)
        if time.time() < d3 - budget_s * 0.005:
            h_r, c5_r = _nesterov(h_r, min(d3, time.time() + budget_s * 0.008),
                                  tau0=80.0, tau1=300.0, lr0=0.008)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Diverse perturbations + re-optimize
    d4 = t0 + budget_s * 0.82
    scale = 0.04
    for att in range(300):
        if time.time() > d4:
            break
        if best_c5 < 0.3800:
            break
        mode = att % 4
        if mode == 0:
            h_p = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
        elif mode == 1:
            h_p = _fourier_pert(best_h, rng, amp=scale, n_modes=10)
        elif mode == 2:
            h_p = _fourier_pert(best_h, rng, amp=scale * 0.6, n_modes=4)
        else:
            h_p = _project(best_h + rng.normal(0, scale * 0.5, best_n), best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_t, c5_t = _lp_loop(h_p, sub_dl, radius_init=0.10, top_init=8)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.18, scale * 1.12)

    # Phase 5: Pair refinement
    d5 = t0 + budget_s * 0.90
    hh, cc = _pair_refine(best_h, d5)
    if cc < best_c5:
        best_c5 = cc
        best_h = hh.copy()

    # Phase 6: Final LP polish (tight radius, more constraints)
    d6 = t0 + budget_s * 0.96
    hh, cc = _lp_loop(best_h, d6, radius_init=0.06, top_init=max(10, int(0.4 * (2 * best_n - 1))))
    if cc < best_c5:
        best_c5 = cc
        best_h = hh.copy()

    # Phase 7: Tiny Nesterov finish
    if time.time() < t0 + budget_s * 0.99:
        hh, cc = _nesterov(best_h, t0 + budget_s * 0.995, tau0=120.0, tau1=600.0, lr0=0.004)
        if cc < best_c5:
            best_c5 = cc
            best_h = hh.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
