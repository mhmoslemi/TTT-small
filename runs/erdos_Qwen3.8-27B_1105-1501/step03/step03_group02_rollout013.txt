    elif d > 0:

def _project(v, target):

    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

             'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    
    for d in lags:
        dd = d
        cons.append({'type': 'ineq',
                     'fun': lambda x, dd=dd: x[n] - _score_tau(x[:n], dd),
                     'jac': lambda x, dd=dd: np.concatenate([-_score_grad(x[:n], dd), [1.0]])})
    
    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                       callback=_cb)
    except RuntimeError:
        return h_init, _c5(h_init)
    except Exception:
        return h_init, _c5(h_init)
    
    h_out = np.clip(res.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-6:
        h_out = _project(h_out, target)
    return h_out, _c5(h_out)

```python
def _subgrad(h_init, deadline, max_iter=5000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        max_score = float(np.max(scores))
        # Adaptive beta: start at 30, end at 500
        progress = min(1.0, it / max_iter)
        beta = 30.0 * (500.0 / 30.0) ** progress
        
        # Weights
        shifted = beta * (scores - max_score)
        shifted = np.clip(shifted, -500.0, 0.0)
        w = np.exp(shifted)
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        
        # Smooth gradient
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            G[n-1+d] = _score_grad(h, d)
        grad = G.T @ w
        grad -= np.mean(grad)  # keep sum unchanged
        
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break
        
        # Step size (decreasing)
        alpha = 1.0 / (norm_g * (1.0 + 0.01 * it))
        
        # Try the step
        h_try = _project(h - alpha * grad, target)
        c5_try = _c5(h_try)
        
        if c5_try < cur_c5:
            h = h_try
            cur_c5 = c5_try
            scores = _scores(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Backtrack
            for bt in range(5):
                alpha *= 0.5
                h_try = _project(h - alpha * grad, target)
                c5_try = _c5(h_try)
                if c5_try < cur_c5:
                    h = h_try
                    cur_c5 = c5_try
                    scores = _scores(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
    
    return best_h, best_c5
```

Wait, this subgradient approach might be too slow because of the _score_grad loop (O(n^2) per iteration). For n=64, that's 127 * 64 = 8128 operations per gradient computation. With 5000 iterations, that's 40M operations. At ~10M ops/sec, that's 4 seconds. Plus the _scores and _c5 calls. Should be fine.

But actually, I realize the subgradient approach with the mean-removal trick might not be the best. The mean removal ensures the sum is approximately preserved, but it might prevent the algorithm from finding the right direction. Let me remove the mean removal and rely on the projection to fix the sum.

Actually, the mean removal is important because without it, the gradient step might change the sum significantly, and the projection would have to make a large correction. With the mean removal, the step is "tangent" to the sum constraint, and the projection only needs to make a small correction for the box constraints.

Let me keep the mean removal.
