def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _overlap_data(h):
    """Compute all overlap values and gradients"""
    n = len(h)
    n_lags = 2 * n - 1
    J = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for idx in range(n_lags):
        d = idx - (n - 1)  # offset
        if d >= 0:
            m = n - d
            if m > 0:
                # J = sum_{i=0}^{m-1} h[i] * (1-h[i+d])
                J[idx] = np.sum(h[:m] * (1.0 - h[d:]))
                # Gradient
                # For i in [0, m-1]: contrib (1-h[i+d])
                # For i in [d, m-1+d]=n-1: contrib -h[i-d]
                g = np.zeros(n)
                g[:m] += (1.0 - h[d:d+m])
                g[d:d+m] -= h[:m]
                grads[idx] = g
        else:
            l = -d
            m = n - l
            if m > 0:
                # J = sum_{i=0}^{m-1} h[i+l] * (1-h[i])
                J[idx] = np.sum(h[l:l+m] * (1.0 - h[:m]))
                # Gradient
                # For i in [l, l+m-1]=n-1: contrib (1-h[i-l])  [first factor]
                # For i in [0, m-1]: contrib -h[i+l]  [second factor]
                g = np.zeros(n)
                g[l:l+m] += (1.0 - h[:m])
                g[:m] -= h[l:l+m]
                grads[idx] = g
    
    return J, grads

def _solve_tangent_lp(h, target):

    n = len(h)
    J, grads = _overlap_data(h)
    
    # LP: minimize t
    # subject to: t - grads[s] @ h' >= J[s] - grads[s] @ h  for all s
    #             0 <= h'[i] <= 1
    #             sum h' = target
    #
    # In standard form for linprog (minimize c^T x subject to A_ub x <= b_ub, A_eq x = b_eq, bounds):
    # x = [h'_0, ..., h'_{n-1}, t]
    # c = [0, ..., 0, 1]
    #
    # Constraint: t - grads[s] @ h' >= c_s
    #   => -t + grads[s] @ h' <= -c_s
    #   => [grads[s], -1] @ x <= -(J[s] - grads[s] @ h)
    
    c_s = J - grads @ h  # c_s = J[s] - grads[s] @ h
    
    # A_ub: for each lag s, row is [grads[s, :], -1]
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -c_s
    
    # A_eq: sum h' = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # bounds: h'[i] in [0,1], t free (or bounded)
    bounds = [(0.0, 1.0)] * n + [(-1000.0, 1000.0)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None
    return res.x[:n]

def _project(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    current_sum = float(np.sum(v))
    if abs(current_sum - target) < 1e-12:
        return v
    
    # Binary search for threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, direction, target):

    # First find valid alpha range
    alpha_max = 1.0
    for j in range(len(h)):

    for a in alphas[1:]:
        h_try = _project(h + a * direction, target)
        c_try = _exact_c5(h_try)
        if c_try < best_c:

            best_a = a
    
    # Golden section refinement
    if best_a > 0:
        lo = max(0, best_a * 0.7)
        hi = min(1.0, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * direction, target))
        f2 = _exact_c5(_project(h + c2 * direction, target))
        for _ in range(30):
            if hi - lo < 1e-12:

            if f1 < f2:
                hi = c2; c2 = c1; f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * direction, target))
            else:
                lo = c1; c1 = c2; f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * direction, target))

        opt_c = _exact_c5(_project(h + opt_a * direction, target))
        if opt_c < best_c:
            best_a = opt_a
            best_c = opt_c
    
    return best_a, best_c

        else:
            stall += 1
            alpha *= 0.85
            T = max(0.03, T * 0.9)
    
    return best_h, best_c5

def _pair_polish(h, deadline, target, max_passes=100):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        J, grads = _overlap_and_grads(h)
        grad_max = w @ grads if False else np.zeros(n)
        
        # Use gradient to guide
        shifted = J - float(np.max(J))
        w = np.exp(shifted / max(0.1, T_init_w)) if (T_init_w := 0.2) else np.zeros(len(J))
        w = np.exp(J / 0.2 - np.max(J) / 0.2)
        w /= float(np.sum(w))
        grad = w @ grads
        grad -= np.mean(grad)
        
        top = np.argsort(-np.abs(grad))[:min(20, n)]
        for ii in range(len(top)):
            if time.time() > deadline:
                return best_h, best_c5
            j = top[ii]
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for jj in range(len(top)):
                if jj == ii:
                    continue
                k = top[jj]
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    amax = min((1-h[j])/dj if dj > 0 else h[j]/(-dj),
                               (1-h[k])/dk if dk > 0 else h[k]/(-dk))
                    if amax < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 1.0]:
                        a = amax * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-12:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
        # Random pairs
        for _ in range(300):
            if time.time() > deadline:
                return best_h, best_c5
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            dj = 1.0 if rng.random() > 0.5 else -1.0
            dk = -dj
            amax = min((1-h[j])/dj if dj > 0 else h[j]/(-dj),
                       (1-h[k])/dk if dk > 0 else h[k]/(-dk))
            if amax < 1e-10:
                continue
            for frac in [0.3, 0.7, 1.0]:
                a = amax * frac
                h_t = h.copy()
                h_t[j] += dj * a
                h_t[k] += dk * a
                c_t = _exact_c5(h_t)
                if c_t < best_c5 - 1e-12:
                    best_c5 = c_t
                    best_h = h_t.copy()
                    h = h_t
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start(h0, deadline, target, rng, n_starts=20):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 30.0)
        sub_dl = time.time() + alloc
        
        r = rng.random()
        if r < 0.2:
            h_try = h0 + rng.normal(0, 0.04, n)
        elif r < 0.35:
            h_try = h0.copy()
            idx = rng.choice(n, size=n // 6, replace=False)
            h_try[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.45:
            shift = int(rng.integers(1, max(2, n // 6)))
            h_try = np.roll(h0, shift)
        elif r < 0.55:
            h_try = 0.5 + (h0 - 0.5) * (0.7 + 0.6 * rng.random())
        elif r < 0.65:
            h_try = (h0 + rng.normal(0, 0.06, n) > 0.5).astype(float)
        elif r < 0.75:
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_try = 0.4 * h0 + 0.6 * h_rand
        elif r < 0.85:
            # Anti-periodic structure
            half = n // 2
            h_try = h0[:half].copy()
            h_try[half:] = 1.0 - h0[:half]
        else:
            # Blip perturbation
            h_try = h0.copy()
            for _ in range(3):
                c = int(rng.integers(0, n))
                w = int(rng.integers(1, n // 4))
                h_try[max(0, c-w):min(n, c+w)] += rng.choice([-0.1, 0.1])
        
        h_try = _project(h_try, target)
        
        # LP optimize from this start
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, target, max_iters=2000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

def _fine_polish(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    
    while time.time() < deadline:
        improved = False
        
        # Focus on top lags
        J, grads = _overlap_and_grads(h)
        top_lags = np.argsort(-J)[:8]
        
        for r in top_lags:
            if time.time() > deadline:
                return best_h, best_c5
            g = grads[r, :].copy()
            g -= np.mean(g)
            gn = float(np.linalg.norm(g))
            if gn < 1e-10:
                continue
            g /= gn
            for sign in [1.0, -1.0]:
                a, c_t = _line_search(h, sign * g, target, max_alpha=0.4)
                if a > 0 and c_t < best_c5 - 1e-13:
                    h = _project(h + a * sign * g, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        
        # Micro perturbations
        if not improved:
            for _ in range(30):
                if time.time() > deadline:
                    return best_h, best_c5
                h_try = _project(h + rng.normal(0, 0.008, n), target)
                c_try = _exact_c5(h_try)
                if c_try < best_c5 - 1e-13:
                    a, c_t = _line_search(h, h_try - h, target, max_alpha=0.3)
                    if c_t < best_c5:
                        h = _project(h + a * (h_try - h), target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
        
        # Pair fine moves
        if not improved:
            for _ in range(200):
                if time.time() > deadline:
                    return best_h, best_c5
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j == k:
                    continue
                if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6 or h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    amax = min((1-h[j])/dj if dj > 0 else h[j]/(-dj),
                               (1-h[k])/dk if dk > 0 else h[k]/(-dk))
                    if amax <= 1e-12:
                        continue
                    for a in [amax * 0.1, amax * 0.3, amax * 0.5, amax]:
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t
                            cur_c5 = c_t
                            improved = True
                            break
        
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Load initial
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP optimization from initial (30% budget)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_optimize(h, dl1, target0, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Weighted gradient fallback (15%)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = _weighted_grad_opt(best_h, dl2, target0, T_init=0.3)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Multi-start (20%)
    dl3 = t0 + budget_s * 0.65
    h3, c5_3 = _multi_start(best_h, dl3, best_n / 2.0, rng, n_starts=20)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Resolution sweep (15%)
    dl4 = t0 + budget_s * 0.80
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, 12, -12, 16, -16, 20, -20]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + min(15.0, max(1.0, (dl4 - time.time()) * 0.3)))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, max_iters=3000)
        h_r, c5_r = _pair_polish(h_r, min(dl4, time.time() + 5.0), n_try / 2.0, max_passes=30)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Pair polish (7%)
    dl5 = t0 + budget_s * 0.87
    h5, c5_5 = _pair_polish(best_h, dl5, best_n / 2.0, max_passes=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Fine polish (12%)
    dl6 = t0 + budget_s * 0.99
    h6, c5_6 = _fine_polish(best_h, dl6, best_n / 2.0)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
