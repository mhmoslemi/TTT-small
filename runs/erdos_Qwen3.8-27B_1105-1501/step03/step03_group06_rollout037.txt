import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_matrix(h):
    """Compute the gradient matrix G where G[r, j] = dC_r/dh_j."""
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for r in range(K):
        d = r - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j + d < n:
                val += (1.0 - h[j + d])
            if 0 <= j - d < n:
                val -= h[j - d]
            G[r, j] = val
    return G


class _EpiProblem:
    """Encapsulates the epigraph NLP for SLSQP."""
    
    def __init__(self, n, active_r):
        self.n = n
        self.dx = 2.0 / n
        self.active_r = active_r
        self.K = len(active_r)
    
    def obj(self, x):
        return x[self.n]
    
    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    
    def ineq_fun(self, x):
        h = x[:self.n]
        T = x[self.n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return T - corr[self.active_r] * self.dx
    
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        J = np.zeros((self.K, n + 1))
        for idx, r in enumerate(self.active_r):
            J[idx, n] = 1.0
            d = r - (n - 1)
            for j in range(n):
                val = 0.0
                if 0 <= j + d < n:
                    val += (1.0 - h[j + d])
                if 0 <= j - d < n:
                    val -= h[j - d]
                J[idx, j] = -self.dx * val
        return J
    
    def eq_fun(self, x):
        return float(np.sum(x[:self.n])) - self.n / 2.0
    
    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J


def _slsqp_solve(h0, n, maxiter=500, ftol=1e-12, top_k=None):
    """Solve epigraph problem via SLSQP."""
    dx = 2.0 / n
    K_total = 2 * n - 1
    T0 = _exact_c5(h0)
    
    if top_k is not None and top_k < K_total:
        corr = _corr_values(h0)
        active_r = np.argsort(corr)[-top_k:]
        active_r = np.sort(active_r)
    else:
        active_r = np.arange(K_total)
    
    prob = _EpiProblem(n, active_r)
    
    x0 = np.concatenate([h0, [T0]])
    bounds = [(0.0, 1.0)] * n + [(T0 * 0.5, 2.0)]
    
    cons = [
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}
    ]
    
    try:
        res = minimize(prob.obj, x0, jac=prob.obj_grad, constraints=cons,
                      bounds=bounds, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < T0:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = _project(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < T0:
                return h_new, c5_new
    except Exception:
        pass
    
    return h0, T0

  - So G[r, :n-d] += (1-h[d:]) and G[r, d:] += -h[:n-d]

- For d < 0 (let g = -d > 0):
  - (1-h[j-d]) = (1-h[j+g]) is (1-h[g:]) for j=0..n-g-1
  - h[j-d] = h[j+g]... wait, that's not right.

Let me be more careful. d = r - (n-1).

C_r = Σ_i h[i] * (1-h[i+d]) where i+d must be in [0, n-1].

∂C_r/∂h[j]:
- j appears as the first index: i=j, so we need i+d = j+d to be valid. Contribution: (1-h[j+d]) if 0≤j+d<n.
- j appears as the second index: i+d=j, so i=j-d. We need j-d to be valid (0≤j-d<n). Contribution: -h[j-d].

So G[r,j] = (1-h[j+d])*[0≤j+d<n] - h[j-d]*[0≤j-d<n]

For d=0: G[r,j] = (1-h[j]) - h[j] = 1-2h[j].

For d>0:
- First term: j+d < n means j < n-d. So for j=0,...,n-d-1: (1-h[j+d])
- Second term: j-d ≥ 0 means j ≥ d. So for j=d,...,n-1: -h[j-d]

For d<0 (let g=-d>0):
- First term: j+d = j-g ≥ 0 means j ≥ g. So for j=g,...,n-1: (1-h[j-g])
- Second term: j-d = j+g < n means j < n-g. So for j=0,...,n-g-1: -h[j+g]

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_matrix_vec(h):
    n = len(h)
    K = 2 * n - 1
    G = np.zeros((K, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            G[r, :m] += (1.0 - h[d:])
            G[r, d:] += -h[:m]
        else:
            g = -d
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += -h[g:]
    return G


class _EpiProblem:
    def __init__(self, n, active_r):
        self.n = n
        self.dx = 2.0 / n
        self.active_r = active_r
        self.K = len(active_r)

    def obj(self, x):
        return x[self.n]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq_fun(self, x):
        h = x[:self.n]
        T = x[self.n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return T - corr[self.active_r] * self.dx

    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        J = np.zeros((self.K, n + 1))
        for idx, r in enumerate(self.active_r):
            J[idx, n] = 1.0
            d = r - (n - 1)
            m = n - abs(d)
            if d == 0:
                J[idx, :n] = -self.dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[idx, :m] += -self.dx * (1.0 - h[d:])
                J[idx, d:] += self.dx * h[:m]
            else:
                g = -d
                J[idx, g:] += -self.dx * (1.0 - h[:m])
                J[idx, :m] += self.dx * h[g:]
        return J

    def eq_fun(self, x):
        return float(np.sum(x[:self.n])) - self.n / 2.0

    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J


def _slsqp_solve(h0, n, maxiter=400, ftol=1e-12, top_k=None):
    dx = 2.0 / n
    K_total = 2 * n - 1
    T0 = _exact_c5(h0)

    if top_k is not None and top_k < K_total:
        corr = _corr_values(h0)
        active_r = np.sort(np.argsort(corr)[-top_k:])
    else:
        active_r = np.arange(K_total)

    prob = _EpiProblem(n, active_r)
    x0 = np.concatenate([h0, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}
    ]

    try:
        res = minimize(prob.obj, x0, jac=prob.obj_grad, constraints=cons,
                      bounds=bounds, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = _project(h_new, n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < T0:
            return h_new, c5_new
    except Exception:
        pass
    return h0, T0


def _slsqp_iterative(h0, n, deadline, max_rounds=8, top_k_start=20, top_k_end=60):
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n_rounds = max_rounds

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        frac = (rnd + 1) / n_rounds
        top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        top_k = min(top_k, 2 * n - 1)
        maxiter = max(100, 500 - rnd * 50)
        h_new, c5_new = _slsqp_solve(h, n, maxiter=maxiter, top_k=top_k)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            # Try with all constraints
            h_new2, c5_new2 = _slsqp_solve(h, n, maxiter=200, top_k=None)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
            else:
                break
    return best_h, best_c5


def _lp_direction(h, scores, G, radius):
    n = len(h)
    K = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(12):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.15, max_iters=5000, stall_limit=50):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        scores = _corr_values(h)
        G = _grad_matrix_vec(h)
        p = _lp_direction(h, scores, G, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.06

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _basin_hop_step(h, n, rng, scale):
    """Generate a perturbed copy and run a quick SLSQP."""
    h_p = h + rng.normal(0, scale, n)
    h_p = np.clip(h_p, 0.0, 1.0)
    h_p = _project(h_p, n / 2.0)
    h_new, c5_new = _slsqp_solve(h_p, n, maxiter=200, top_k=30)
    if c5_new < _exact_c5(h_p):
        return h_new, c5_new
    return h_p, _exact_c5(h_p)


def _pair_polish(h, deadline, max_passes=30):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng_p = np.random.default_rng(_pass * 1000 + 7)
        for _trial in range(30):
            if time.time() > deadline:
                break
            j = int(rng_p.integers(0, n))
            k = int(rng_p.integers(0, n))
            if j == k:
                continue
            if not (1e-8 < h[j] < 1.0 - 1e-8 and 1e-8 < h[k] < 1.0 - 1e-8):
                continue
            for sign in [1.0, -1.0]:
                a_max = min(h[j] if sign < 0 else (1.0 - h[j]),
                           h[k] if sign > 0 else (1.0 - h[k]))
                if a_max < 1e-10:
                    continue
                for frac in [0.5,
                             1.0, 0.25]:
                    a = frac * a_max
                    h_t = h.copy()
                    h_t[j] += sign * a
                    h_t[k] -= sign * a
                    h_t = np.clip(h_t, 0.0, 1.0)
                    h_t = _project(h_t, n / 2.0)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5
