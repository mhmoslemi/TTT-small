def _lp_direction(h, scores, radius):
    """Find LP direction: minimize max_r (scores_r + G_r . p) subject to ||p||_inf <= radius, sum(p)=0."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, d:] += 1.0
            G[r, d:] -= h[:m]
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0
            G[r, :m] -= h[g:]
            G[r, g:] -= h[:m]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9):
    """Gradient descent with softmax approximation of max, adaptive tau."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = _weighted_grad(h, w) * dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)

                best_h = h.copy()

def _resample(h, n_new):

        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + time_per)
        h_try, c5_try = _lp_local(h_pert, sub_dl, radius_init=0.1)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.2)
    return best_h, best_c5

def _diverse_directions(h, n_dirs=8, radius=0.2, seed=0):
    """Generate diverse mean-zero directions."""
    n = len(h)
    rng = np.random.default_rng(seed)
    dirs = []
    # Gradient-like directions from top scores
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    # Top scoring rows
    top_idx = np.argsort(scores)[::-1][:4]
    for r in top_idx[:n_dirs // 2]:
        d = r - (n - 1)
        p = np.zeros(n)
        if d >= 0:
            m = n - d
            p[d:] += 1.0
            p[d:] -= h[:m]
            p[:m] -= h[d:]
        else:
            g = -d
            m = n - g
            p[:m] += 1.0
            p[:m] -= h[g:]
            p[g:] -= h[:m]
        p -= np.mean(p)
        norm = np.max(np.abs(p))
        if norm > 1e-10:
            p = p / norm * radius
        dirs.append(p)
    # Random directions
    for _ in range(n_dirs - len(dirs)):
        p = rng.normal(0, 1, n)
        p -= np.mean(p)
        norm = np.max(np.abs(p))
        if norm > 1e-10:
            p = p / norm * radius
        dirs.append(p)
    # Anti-gradient direction (move opposite to top score gradient)
    if len(dirs) < n_dirs:
        r_top = int(np.argmax(scores))
        d = r_top - (n - 1)
        p = np.zeros(n)
        if d >= 0:
            m = n - d
            p[d:] += 1.0
            p[d:] -= h[:m]
            p[:m] -= h[d:]
        else:
            g = -d
            m = n - g
            p[:m] += 1.0
            p[:m] -= h[g:]
            p[g:] -= h[:m]
        p -= np.mean(p)
        norm = np.max(np.abs(p))
        if norm > 1e-10:
            p = -p / norm * radius
        dirs.append(p)
    return dirs[:n_dirs]

def _direction_search(h, deadline, n_dirs=10, radius=0.25, seed=0):
    """Search along diverse directions with exact line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline and stall < 40:
        dirs = _diverse_directions(h, n_dirs=n_dirs, radius=radius, seed=seed + stall)
        improved = False
        for p in dirs:
            if time.time() > deadline:
                break
            # Compute feasible alpha_max
            alpha_max = radius
            for j in range(n):
                if p[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            if alpha_max < 1e-10:
                continue
            alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
            radius *= 0.8
            if radius < 0.01:
                break
        else:
            stall = 0
            radius = min(radius * 1.1, 0.4)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick gradient warmup (5% budget)
    d1 = t0 + budget_s * 0.05
    h, c5 = _grad_optimize(h, d1, tau_init=12.0, lr_init=0.02, mom_beta=0.88)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP local search (20% budget)
    d2 = t0 + budget_s * 0.25
    h, c5 = _lp_local(best_h, d2, radius_init=0.15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Direction search (10% budget)
    d3 = t0 + budget_s * 0.35
    h, c5 = _direction_search(best_h, d3, n_dirs=12, radius=0.3, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Multi-resolution (20% budget)
    d4 = t0 + budget_s * 0.55
    best_h, best_c5, best_n = _multi_resolution(best_h, best_n, d4, budget_s)

    # Phase 5: Intensive LP + gradient at best resolution (20% budget)
    d5 = t0 + budget_s * 0.75
    h, c5 = _lp_local(best_h, d5, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d5:
        h, c5 = _grad_optimize(best_h, d5, tau_init=50.0, lr_init=0.008, mom_beta=0.9)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d5:
        h, c5 = _lp_local(best_h, d5, radius_init=0.05)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Perturbation + refine (15% budget)
    d6 = t0 + budget_s * 0.90
    best_h, best_c5 = _perturb_refine(best_h, d6, budget_s, seed=seed + 100)

    # Phase 7: Final polishing (10% budget)
    d7 = t0 + budget_s * 0.96
    h, c5 = _lp_local(best_h, min(d7, time.time() + budget_s * 0.02), radius_init=0.04)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    if time.time() < d7:
        h, c5 = _pair_refine(best_h, d7, max_passes=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h, c5 = _lp_local(best_h, t0 + budget_s * 0.995, radius_init=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
