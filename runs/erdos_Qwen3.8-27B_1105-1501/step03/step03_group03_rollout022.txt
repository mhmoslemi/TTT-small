import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scores_and_grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subgradient_step(h, target, scores, G, lr, active_frac=0.1):
    """Step using subgradient of the max with hard active set."""
    s_max = float(np.max(scores))
    threshold = s_max - active_frac * max(s_max, 1.0)
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.where(scores == s_max)[0]
    
    # Weighted average gradient of active constraints
    # Use softmax weighting for stability
    weights = np.zeros(len(active))
    if len(active) > 1:
        sp = scores[active]
        sp = sp - sp.max()
        weights = np.exp(sp / 0.1)
        weights /= weights.sum()
    else:
        weights[0] = 1.0
    
    grad = G[active, :].T @ weights
    grad -= np.mean(grad)  # Remove component orthogonal to constraint
    
    h_new = _project(h - lr * grad, target)
    return h_new

def _slsqp_smooth(h, target, beta, max_iter=100):
    n = len(h)
    
    def obj(x):
        sc = np.correlate(x, 1.0 - x, mode='full')
        sm = float(np.max(sc))
        return sm + np.log(np.sum(np.exp(beta * (sc - sm)))) / beta
    
    def jac(x):
        sc, G = _scores_and_grad(x)
        sm = float(np.max(sc))
        exp_s = np.exp(beta * (sc - sm))
        w = exp_s / np.sum(exp_s)
        return G.T @ w
    
    def sum_fun(x):
        return np.sum(x) - target
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()

def _lp_direction(h, scores, G, radius):
    n = len(h)
    L = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = c5_cur
    
    # Coarse search
    n_eval = 20
    alphas = np.linspace(0.0, alpha_max, n_eval)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

            break
        if stall > 50:
            break

        p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.6)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.05
    
    return best_h, best_c5

def _subgradient_descent(h, target, deadline, lr_init=0.005, active_frac=0.1,
                         decay=0.999, max_iter=10000):
    """Subgradient descent with hard active set."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    h = h.copy()

        h_new = _subgradient_step(h, target, scores, G, lr, active_frac)

            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

            if c5_new < _exact_c5(h):
                h = h_new.copy()
            lr *= 0.9
            if lr < 1e-10:
                break
        
        lr = max(lr * decay, 1e-10)
    
    return best_h, best_c5

def _pair_polish(h, target, deadline, max_passes=20):
    """Pair coordinate descent."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):  # Step by 2 for speed
            if time.time() > deadline:
                break
            for k in range(j+1, n, 2):
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                    continue
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-12:
                        continue
                    a = a_max * 0.5
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        h = h_t.copy()
                        best_h = h.copy()
                        improved = True
        if not improved:

def _epigraph_slsqp(h, deadline, max_iter=200):

    n = len(h)
    target = n / 2.0
    
    # Get initial scores
    scores = _scores(h)
    s_max = float(np.max(scores))

    for d in range(-(n-1), n):

        if d >= 0:

                G_full[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G_full[r, :m] += (1.0 - h[g:])
                G_full[r, g:] -= h[:m]
    
    G_sub = G_full[idx, :]
    scores_sub = scores[idx]
    
    # Variable: [h_0, ..., h_{n-1}, tau]
    def obj(x):
        return -x[-1]  # Minimize -tau = maximize tau... wait no
        # Actually we want to MINIMIZE the max, so we want to minimize tau
        # subject to tau >= S_r(h) for all r
        # So objective is tau (minimize), constraints are S_r(h) - tau <= 0
    
    # Let me reformulate: minimize tau subject to S_r(h) <= tau, 0<=h<=1, sum(h)=n/2
    # In scipy minimize: minimize f, subject to c(x) >= 0 (ineq) or c(x) = 0 (eq)
    # So constraints: tau - S_r(h) >= 0 for all r
    
    x0 = np.concatenate([h, [s_max]])
    
    def obj2(x):
        return x[-1]

def _momentum_subgrad(h, target, deadline, lr_init=0.01, momentum=0.8, 
                      active_frac=0.15, max_iter=5000):
    """Momentum subgradient descent."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        scores, G = _scores_and_grad(h)
        s_max = float(np.max(scores))
        threshold = s_max - active_frac * max(abs(s_max), 1.0)
        active = np.where(scores >= threshold)[0]
        
        if len(active) > 1:
            sp = scores[active] - scores[active].max()
            weights = np.exp(sp / 0.5)
            weights /= weights.sum()
        else:
            weights = np.ones(max(len(active), 1))
        
        grad = G[active, :].T @ weights
        grad -= np.mean(grad)
        
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= _exact_c5(h):
            h = h_new.copy()
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.8
            v *= 0.5
        
        if lr < 1e-9:
            break
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _scores_and_grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = _scores(h)
    return scores, G


def _project(v, target):

        return np.zeros(n)

        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _topk_smooth_obj(x, k, beta):
    sc = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(sc)[-k:]
    sc_k = sc[idx]
    sm = float(np.max(sc_k))
    return sm + np.log(np.sum(np.exp(beta * (sc_k - sm)))) / beta


def _topk_smooth_jac(x, k, beta):
    n = len(x)
    scores, G = _scores_and_grad(x)
    idx = np.argsort(scores)[-k:]
    sc_k = scores[idx]
    sm = float(np.max(sc_k))
    exp_s = np.exp(beta * (sc_k - sm))
    w = exp_s / np.sum(exp_s)
    return G[idx, :].T @ w

        return _topk_smooth_jac(x, k, beta)

    def sum_jac(x):
        return np.ones(n)

def _subgrad_momentum(h, target, deadline, lr_init=0.006, momentum=0.8,
                      active_frac=0.12, max_iter=5000):

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    h = h.copy()

        if time.time() > deadline:
            break

        scores, G = _scores_and_grad(h)
        s_max = float(np.max(scores))
        threshold = s_max - active_frac * max(abs(s_max), 0.5)
        active = np.where(scores >= threshold)[0]

        if len(active) > 1:
            sp = scores[active] - scores[active].max()
            weights = np.exp(sp / 0.3)
            weights /= weights.sum()
        else:
            weights = np.ones(max(len(active), 1))

        grad -= np.mean(grad)

                        continue
                    a = a_max * 0.5
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        h = h_t.copy()
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break

    return h, best_c5


def _epigraph_slsqp(h, deadline, max_iter=250):
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    s_max = float(np.max(scores))
    L = 2 * n - 1

    if L <= 100:
        idx = np.arange(L)
    else:
        top_k = min(80, L)
        idx = np.argsort(scores)[-top_k:]

    G_full = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G_full[r, d:] += (1.0 - h[:m])
                G_full[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G_full[r, :m] += (1.0 - h[g:])
                G_full[r, g:] -= h[:m]

    G_sub = G_full[idx, :]
    scores_sub = scores[idx]
    n_con = len(idx)

    x0 = np.concatenate([h, [s_max]])

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    constraints = []
    for i in range(n_con):
        gi = G_sub[i]
        sc_i = scores_sub[i]

        def make_con(gi, sc_i):
            def con(x):
                return x[-1] - (sc_i + gi @ (x[:n] - h))
            def con_jac(x):
                gj = np.zeros(n + 1)
                gj[:n] = -gi
                gj[-1] = 1.0
                return gj
            return {'type': 'ineq', 'fun': con, 'jac': con_jac}
        constraints.append(make_con(gi, sc_i))

    def sum_con(x):
        return np.sum(x[:n]) - target
    def sum_con_jac(x):
        gj = np.zeros(n + 1)
        gj[:n] = 1.0
        return gj
    constraints.append({'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            h_try = _project(res.x[:n], target)
            return h_try
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(64, 0.5)

    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    hard_deadline = t0 + budget_s * 0.97

    # Phase 1: SLSQP with top-k smooth max (various k and beta)
    deadline1 = min(t0 + budget_s * 0.18, hard_deadline)
    configs = [
        (n0, 10.0, 50), (n0, 20.0, 50), (n0, 40.0, 30),
        (n0, 5.0, 40), (n0, 15.0, 30), (n0, 30.0, 20),
    ]
    for k, beta, mx in configs:
        if time.time() > deadline1:
            break
        k_use = min(k, 2 * best_n - 1)
        h_s = _slsqp_topk(best_h, best_n / 2.0, k_use, beta, max_iter=mx)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: LP steepest descent loop
    deadline2 = min(t0 + budget_s * 0.42, hard_deadline)
    h_lp, c5_lp = _lp_loop(best_h, deadline2, radius_init=0.12, max_iters=6000)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Subgradient momentum (try several active fractions)
    deadline3 = min(t0 + budget_s * 0.55, hard_deadline)
    for af in [0.05, 0.08, 0.12, 0.2]:
        if time.time() > deadline3:
            break
        sub_dl = min(deadline3, time.time() + budget_s * 0.03)
        h_m, c5_m = _subgrad_momentum(best_h, best_n / 2.0, sub_dl,
                                      lr_init=0.005, momentum=0.8,
                                      active_frac=af, max_iter=3000)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase 4: Epigraph SLSQP + LP follow-up
    deadline4 = min(t0 + budget_s * 0.70, hard_deadline)
    for attempt in range(6):
        if time.time() > deadline4:
            break
        h_epi = _epigraph_slsqp(best_h, deadline4, max_iter=200)
        c5_epi = _exact_c5(h_epi)
        if c5_epi < best_c5:
            best_c5 = c5_epi
            best_h = h_epi.copy()
        sub_dl = min(deadline4, time.time() + budget_s * 0.02)
        h_lp2, c5_lp2 = _lp_loop(best_h, sub_dl, radius_init=0.06, max_iters=800)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Phase 5: Resolution exploration
    deadline5 = min(t0 + budget_s * 0.84, hard_deadline)
    steps = [1, -1, 2, -2, 4, -4, 8, -8, 12, -12]
    for dn in steps:
        if time.time() > deadline5:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        h_r = _slsqp_topk(h_r, n_try / 2.0, min(20, 2 * n_try - 1), 30.0, max_iter=80)
        sub_dl = min(deadline5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.08, max_iters=600)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Pair polish
    deadline6 = min(t0 + budget_s * 0.90, hard_deadline)
    hp, c5p = _pair_polish(best_h, best_n / 2.0, deadline6, max_passes=12)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 7: Final LP + SLSQP polish
    deadline7 = min(t0 + budget_s * 0.95, hard_deadline)
    h_f, c5_f = _lp_loop(best_h, deadline7, radius_init=0.02, max_iters=4000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    if time.time() < hard_deadline:
        h_fin = _slsqp_topk(best_h, best_n / 2.0, min(10, 2 * best_n - 1),
                            100.0, max_iter=80)
        c5_fin = _exact_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
