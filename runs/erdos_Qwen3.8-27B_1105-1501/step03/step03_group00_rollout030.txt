The issue is that the LP might not find the exact KKT direction due to numerical issues or the linearization being poor for large steps.

Let me try a different approach: **Directly solve the KKT system.**

At the optimum, we need:
1. Σ_{k active} λ_k ∇score_k = 0 (on the tangent space, i.e., sum of components = 0)
2. Σ λ_k = 1, λ_k ≥ 0
3. Only active lags have λ_k > 0

This is a system of equations in the λ_k's. If I know which lags are active (say, the top m lags), I can solve:
G_active^T λ = 0 (where G_active is the submatrix of active lag gradients)
Σ λ_k = 1
λ ≥ 0

This is a linear system (or LP if we need the inequality). If m = n (all lags active), it's overdetermined. But typically only a few lags are active.

The parent's LP does something similar but in a different way (it finds a direction, not the multipliers directly).

Let me try a **direct KKT solver** approach:
1. Identify active lags (those within ε of the max)
2. Solve for λ that minimizes the norm of G_active^T λ subject to Σλ=1, λ≥0
3. Use this to determine if we're at a KKT point
4. If not, take a step in the direction -G_active^T λ (which is in the "descent" direction for the max)

This is essentially the same as the parent's approach but potentially more stable.

OK let me just write a practical solution. I'll combine:
1. A modified version of the parent's LP approach with different hyperparameters
2. A power-mean subgradient approach for global exploration
3. A binary search phase
4. Resolution sweeping
5. Active-set focused polishing

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, c5_cur, max_alpha=0.5):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):

        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.999)
    alpha_max = max(alpha_max, 1e-12)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))

    return None, None

def _power_mean_subgrad(h, deadline, p=10, beta0=10.0, max_iters=2000):
    """Subgradient descent on the power mean of scores."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    beta = beta0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h_cur)
        G = _grad_matrix(h_cur)
        # Power mean gradient
        sp = scores ** p
        denom = np.sum(sp)
        if denom < 1e-300:
            break
        factor = denom ** (1.0/p - 1.0)
        weights = scores ** (p-1) * factor
        grad = G.T @ weights
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        # Backtracking line search
        alpha = beta / gn
        alpha = min(alpha, 0.3)
        found = False
        for _ls in range(50):
            h_try = _project(h_cur - alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h_cur = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_cur.copy()
                found = True
                break
            alpha *= 0.6
            if alpha < 1e-12:
                break
        if not found:
            beta *= 0.8
            if beta < 0.01:
                break
        else:
            beta = min(beta * 1.05, 50.0)
        # Slowly increase p
        p = p * 1.01 + 0.1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _symmetric_init(n, freq=1.0):

    x = (np.arange(n) + 0.5) / n

        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        alpha = beta / gn
        alpha = min(alpha, 0.25)
        found = False
        for _ in range(40):
            h_try = _project(h_cur - alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h_cur = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h_cur.copy()
                found = True
                break
            alpha *= 0.6
            if alpha < 1e-12:
                break
        if not found:
            stall += 1
            beta *= 0.85
        else:
            stall = 0
            beta = min(beta * 1.03, 40.0)
        p = p * 1.005 + 0.05
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n or n_new < 16:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _active_set_polish(h, deadline, max_iters=3000):
    """Focus LP on the top active lags with adaptive selection."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.03
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        cur_max = np.max(scores)
        tol = 1e-4
        active = np.where(scores > cur_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        if len(active) <= 3:
            top_k = None
        else:
            top_k = min(len(active), 8)
        p, _ = _lp_direction(h, scores, G, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=0.15)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.08)
        else:
            stall += 1
            radius *= 0.65
    return best_h, best_c5


def _fourier_basis_search(h_init, deadline, n_basis=8, max_gens=200, pop_size=20):
    """Evolutionary search in Fourier coefficient space."""
    n = len(h_init)
    target = n / 2.0
    x = (np.arange(n) + 0.5) / n
    # Build basis: cos and sin for frequencies 1..n_basis
    cols = []
    for f in range(1, n_basis + 1):
        cols.append(np.cos(2.0 * np.pi * f * x))
        cols.append(np.sin(2.0 * np.pi * f * x))
    B = np.column_stack(cols)
    K = B.shape[1]
    # Fit initial coefficients
    res = np.linalg.lstsq(B, h_init - 0.5, rcond=None)
    coeffs_init = res[0]
    rng = np.random.default_rng(12345)
    pop = [coeffs_init.copy() + rng.normal(0, 0.05, K) for _ in range(pop_size)]
    pop[0] = coeffs_init.copy()
    best_c5 = _exact_c5(h_init)
    best_h = h_init.copy()
    best_coeffs = coeffs_init.copy()
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        evaluated = [(c, _exact_c5(_project(0.5 + B @ c, target))) for c in pop[:5]]
        evaluated.sort(key=lambda t: t[1])
        for c, c5_val in evaluated:
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = _project(0.5 + B @ c, target)
                best_coeffs = c.copy()
        # Create next generation
        new_pop = []
        # Keep top
        new_pop.append(evaluated[0][0].copy())
        new_pop.append(evaluated[1][0].copy())
        # Crossover and mutation
        while len(new_pop) < pop_size:
            if rng.random() < 0.3:
                new_pop.append(best_coeffs.copy() + rng.normal(0, 0.03, K))
            else:
                i1 = rng.integers(0, 2)
                i2 = rng.integers(0, 2)
                cross = 0.5 * (evaluated[i1][0] + evaluated[i2][0])
                cross += rng.normal(0, 0.05, K)
                new_pop.append(cross)
        pop = new_pop
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Power-mean subgradient (broader exploration)
    dl = t0 + budget_s * 0.12
    h_p, c5_p = _power_mean_subgrad(best_h, dl, p_init=8, beta0=15, max_iters=800)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 2: Aggressive LP (large radius)
    if time.time() < t0 + budget_s * 0.30:
        dl = t0 + budget_s * 0.30
        h_l, c5_l = _lp_optimize(best_h, dl, top_k_list=[None, 15, 10, 5],
                                 radius_init=0.25, max_iters=3000)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()

    # Phase 3: Active-set focused polish
    if time.time() < t0 + budget_s * 0.45:
        dl = t0 + budget_s * 0.45
        h_a, c5_a = _active_set_polish(best_h, dl, max_iters=3000)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 4: Refined LP (smaller radius, varied top_k)
    if time.time() < t0 + budget_s * 0.58:
        dl = t0 + budget_s * 0.58
        h_l2, c5_l2 = _lp_optimize(best_h, dl, top_k_list=[5, 8, 6, 10, None, 4, 7],
                                   radius_init=0.06, max_iters=4000)
        if c5_l2 < best_c5:
            best_c5 = c5_l2
            best_h = h_l2.copy()

    # Phase 5: Fourier basis evolutionary search (different basin)
    if time.time() < t0 + budget_s * 0.68:
        dl = t0 + budget_s * 0.70
        h_f, c5_f = _fourier_basis_search(best_h, dl, n_basis=6, max_gens=150, pop_size=15)
        if c5_f < best_c5 - 1e-12:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Also try from a different starting point
        h_alt = _project(best_h + rng.normal(0, 0.05, best_n), best_n / 2.0)
        if time.time() < dl:
            h_f2, c5_f2 = _fourier_basis_search(h_alt, dl, n_basis=5, max_gens=80, pop_size=10)
            if c5_f2 < best_c5 - 1e-12:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Phase 6: Perturbation restarts
    dl = t0 + budget_s * 0.80
    scale = 0.015
    for attempt in range(400):
        if time.time() > dl:
            break
        if best_c5 < 0.38065:
            break
        if attempt % 4 == 0:
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif attempt % 4 == 1:
            x = (np.arange(best_n) + 0.5) / best_n
            f = rng.uniform(0.5, 3.5)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = best_h + scale * 0.6 * np.cos(2 * np.pi * f * x + phase)
        elif attempt % 4 == 2:
            # Binary-ish perturbation on random subset
            h_pert = best_h.copy()
            n_flip = max(2, best_n // 10)
            idx = rng.choice(best_n, size=n_flip, replace=False)
            h_pert[idx] = rng.random(n_flip)
        else:
            # Mirror perturbation
            h_pert = best_h.copy()
            h_pert[best_n // 2:] = best_h[:best_n // 2][::-1] + rng.normal(0, scale * 0.5, best_n - best_n // 2)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.025)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[8, 5, None],
                                     radius_init=0.08, max_iters=400)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.87:
        dl = t0 + budget_s * 0.89
        hp, c5p = _pair_polish(best_h, dl, max_passes=15)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Resolution sweep
    if time.time() < t0 + budget_s * 0.93:
        dl = t0 + budget_s * 0.95
        candidates = list(range(max(24, best_n - 8), best_n + 9))
        candidates.sort(key=lambda x: abs(x - best_n))
        for n_try in candidates:
            if time.time() > dl:
                break
            if n_try == best_n or n_try < 20:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None],
                                     radius_init=0.1, max_iters=300)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 9: Final ultra-fine active-set polish
    if time.time() < t0 + budget_s * 0.97:
        h_fin, c5_fin = _active_set_polish(best_h, t0 + budget_s * 0.995, max_iters=5000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final: one more LP pass with tiny radius
    if time.time() < t0 + budget_s * 0.995:
        h_fin2, c5_fin2 = _lp_optimize(best_h, t0 + budget_s * 0.998,
                                       top_k_list=[3, 4, 5], radius_init=0.01, max_iters=2000)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
