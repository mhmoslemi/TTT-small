import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _active_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    lag = int(np.argmax(corr)) - (n - 1)
    s = abs(lag)
    if s == 0:
        return np.zeros(n)
    g = np.zeros(n)
    g[s:] += (1.0 - h[:n-s])
    g[:n-s] -= h[s:]
    return g

def _smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    # Softmax weights
    vals = np.clip(alpha * (corr - ms), -500.0, 0.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    # Weighted sum of gradients
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        s = abs(lag)
        if s == 0:
            continue
        g[s:] += w[k] * (1.0 - h[:n-s])
        g[:n-s] -= w[k] * h[s:]
    return g

def _mean_free(d):
    return d - np.mean(d)

def _feasible_interval(h, d):
    pos = d > 1e-15
    neg = d < -1e-15
    t_min = -1e10
    t_max = 1e10
    if np.any(pos):
        t_min = max(t_min, float(np.max(-h[pos] / d[pos])))
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((1.0 - h[neg]) / d[neg])))
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    return t_min, t_max

def _try_direction(h, d):
    """Try moving along direction d to the best feasible endpoint.
    Returns (best_h, best_c5) or (None, None) if no improvement."""
    d = _mean_free(d)
    if np.max(np.abs(d)) < 1e-14:
        return None, None
    t_min, t_max = _feasible_interval(h, d)
    cur_c5 = _exact_c5(h)
    best_h = None
    best_c5 = cur_c5
    for t in [t_min, t_max]:
        if abs(t) < 1e-12:
            continue
        h_t = h + t * d
        h_t = np.clip(h_t, 0.0, 1.0)
        # Verify sum
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = h_t
    if best_h is not None:
        # Re-project to ensure exact sum
        best_h = _project(best_h, len(h) / 2.0)
        best_c5 = _exact_c5(best_h)
    return best_h, best_c5

The only issue is floating point error.

So I should NOT re-project after the move (it might change the value). Instead, I should just clip with a tiny tolerance and verify the sum is close to n/2.

Actually, let me think about this differently. The feasible interval is computed such that at t_min and t_max, all coordinates are in [0,1]. The sum is exactly n/2 (mathematically). In floating point, it's off by at most O(ε*n) ≈ O(1e-14). This is negligible for the objective function.

So I'll skip the re-projection after endpoint evaluation, and only project when I do other operations (like resampling).
