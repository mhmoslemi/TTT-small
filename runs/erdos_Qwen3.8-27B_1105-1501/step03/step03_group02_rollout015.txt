def _exact_1d_solve(h, d, tlo, thi):
    """Find min of max_r (A_r + B_r*t + C_r*t²) for t in [tlo, thi]"""
    n = len(h)
    M = 2*n - 1
    
    # Compute A, B, C arrays
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    
    for r in range(M):
        l = r - (n - 1)
        if l == 0:
            A[r] = np.sum(h * (1-h))
            B[r] = np.sum(d * (1 - 2*h))
            C[r] = -np.sum(d * d)
        elif l > 0:
            m = n - l
            A[r] = np.sum(h[l:l+m] * (1 - h[:m]))
            B[r] = np.sum(d[l:l+m] * (1 - h[:m]) - h[l:l+m] * d[:m])
            C[r] = -np.sum(d[l:l+m] * d[:m])
        else:
            g = -l
            m = n - g
            A[r] = np.sum(h[:m] * (1 - h[g:g+m]))
            B[r] = np.sum(d[:m] * (1 - h[g:g+m]) - h[:m] * d[g:g+m])
            C[r] = -np.sum(d[:m] * d[g:g+m])
    
    # Generate candidates
    candidates = [tlo, thi, 0.0]
    
    # Stationary points
    for r in range(M):
        if abs(C[r]) > 1e-15:
            t0 = -B[r] / (2 * C[r])
            if tlo <= t0 <= thi:
                candidates.append(t0)
    
    # Pairwise crossings
    for i in range(M):
        for j in range(i+1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:
                    t = -c / b
                    if tlo <= t <= thi:
                        candidates.append(t)
            else:
                disc = b*b - 4*a*c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b + sqrt_disc) / (2*a)
                    t2 = (-b - sqrt_disc) / (2*a)
                    for t in [t1, t2]:
                        if tlo <= t <= thi:
                            candidates.append(t)
    
    # Evaluate envelope at unique candidates
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= tlo) & (candidates <= thi)]
    # Deduplicate
    order = np.argsort(candidates)
    candidates = candidates[order]
    # Keep unique
    unique = [candidates[0]]
    for c in candidates[1:]:
        if c - unique[-1] > 1e-12:
            unique.append(c)
    candidates = np.array(unique)
    
    # Evaluate
    best_t = tlo
    best_val = np.inf
    for t in candidates:
        vals = A + B*t + C*t*t
        mx = np.max(vals)
        if mx < best_val:
            best_val = mx
            best_t = t
    
    return best_t, best_val

def _gradient_for_score(h, r):
    """Compute gradient of score r with respect to h."""
    n = len(h)
    l = r - (n - 1)
    g = np.zeros(n)
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    elif l > 0:
        m = n - l
        g[:m] += -h[l:l+m]
        g[l:] += (1.0 - h[:m])
    else:
        g_abs = -l
        m = n - g_abs
        g[:m] += (1.0 - h[g_abs:g_abs+m])
        g[g_abs:] += -h[:m]
    return g

def _softmax_direction(h, scores, beta=50.0, K=None):
    """Generate a zero-sum direction from weighted score gradients."""
    n = len(h)
    M = len(scores)
    if K is not None:
        # Only use top K scores
        top_idx = np.argsort(scores)[::-1][:K]
        w = np.zeros(M)
        for idx in top_idx:
            w[idx] = 1.0
        w /= np.sum(w)
    else:
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(M)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
    
    # Weighted gradient
    G = np.zeros(n)
    for r in range(M):
        if w[r] > 1e-10:
            G += w[r] * _gradient_for_score(h, r)
    
    # Make zero-sum
    G -= np.mean(G)
    norm = np.max(np.abs(G))
    if norm < 1e-15:
        return None
    G /= norm
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    # Bisection to find alpha such that sum(clip(v - alpha, 0, 1)) = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
