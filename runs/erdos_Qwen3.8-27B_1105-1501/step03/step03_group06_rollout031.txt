```python
def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _corr_and_grad(h):
    """Compute correlation vector and its Jacobian (J[k, j] = dcorr[k]/dh[j])."""
    n = len(h)
    K = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    J = np.zeros((K, n))
```

```python
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = len(corr)
    
    # Select top-k active lags
    idx = np.argsort(corr)[::-1][:top_k]
    T0 = float(corr[idx[0]])
    
    def objective(x):
        return x[0]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    # Inequality constraints: x[0] - corr[r](h) >= 0
    def constraints(x):
        h_x = x[1:]
        c = np.correlate(h_x, 1.0 - h_x, mode='full')
        return x[0] - c[idx]
    
    def constraints_jac(x):
        h_x = x[1:]
        K2 = 2 * len(h_x) - 1
        J = np.zeros((len(idx), n + 1))
        for i, r in enumerate(idx):
            d = r - (n - 1)
            dp = abs(d)
            J[i, 0] = 1.0
            for j in range(n):
                val = 0.0
                if j <= n - 1 - dp:
                    val += (1.0 - h_x[j + dp])
                if dp <= j <= n - 1:
                    val -= h_x[j - dp]
                J[i, j + 1] = -val
        return J
    
    # Equality constraint: sum(h) = target
    def eq_constraint(x):
        return float(np.sum(x[1:])) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    x0 = np.concatenate([[T0], h])
    bounds = [(T0 - 1.0, T0 + 1.0)] + [(0.0, 1.0)] * n
    
    res = minimize(objective, x0, jac=objective_grad,
                   method='SLSQP',
                   constraints=[
                       {'type': 'ineq', 'fun': constraints, 'jac': constraints_jac},
                       {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac}
                   ],
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
```

```python
import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # bisection to find threshold
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

def _corr_jacobian_row(h, k_idx):
    """Compute d(output[k_idx])/dh for a single lag index k_idx."""
    n = len(h)
    k = k_idx - (n - 1)  # the lag
    dp = abs(k)
    grad = np.zeros(n)
    if k >= 0:
        # grad[m] = (1-h[m-k]) * [k <= m <= n-1] - h[m+k] * [0 <= m <= n-1-k]
        for m in range(k, n):
            grad[m] += (1.0 - h[m-k])
        for m in range(0, n-k):
            grad[m] -= h[m+k]
    else:
        # grad[m] = (1-h[m+dp]) * [0 <= m <= n-1-dp] - h[m-dp] * [dp <= m <= n-1]
        for m in range(0, n-dp):
            grad[m] += (1.0 - h[m+dp])
        for m in range(dp, n):
            grad[m] -= h[m-dp]
    return grad
```

```python
def _corr_jacobian(h):
    """Full Jacobian of np.correlate(h, 1-h, 'full') with respect to h."""
    n = len(h)
    K = 2 * n - 1
    J = np.zeros((K, n))
    one_minus_h = 1.0 - h
    for r in range(K):
        k = r - (n - 1)
        dp = abs(k)
```

```python
def _slsqp_optimize(h, top_k=30, maxiter=300, ftol=1e-14):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = len(corr)
    
    # Select active lags
    active = np.argsort(corr)[::-1][:top_k]
    T0 = float(corr[active[0]])
    
    # Precompute Jacobian rows for active lags
    J_active = np.zeros((top_k, n))
    one_minus_h = 1.0 - h
    for i, r in enumerate(active):
        k = r - (n - 1)
        dp = abs(k)
        if k >= 0:
            if k < n:
                J_active[i, k:] += one_minus_h[:n-k]
            if n - k > 0:
                J_active[i, :n-k] -= h[k:]
        else:
            if n - dp > 0:
                J_active[i, :n-dp] += one_minus_h[dp:]
            if dp < n:
                J_active[i, dp:] -= h[:n-dp]
    
    # SLSQP setup
    x0 = np.concatenate([[T0], h])
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def ineq(x):
        h_x = x[1:]
        c = np.correlate(h_x, 1.0 - h_x, mode='full')
        return x[0] - c[active]
    
    def ineq_jac(x):
        h_x = x[1:]
        one_mh = 1.0 - h_x
        J = np.zeros((top_k, n + 1))
        J[:, 0] = 1.0
        for i, r in enumerate(active):
            k = r - (n - 1)
            dp = abs(k)
            if k >= 0:
                if k < n:
                    J[i, 1+k:] += one_mh[:n-k]
                if n - k > 0:
                    J[i, 1:1+n-k] -= h_x[k:]
            else:
                if n - dp > 0:
                    J[i, 1:1+n-dp] += one_mh[dp:]
                if dp < n:
                    J[i, 1+dp:] -= h_x[:n-dp]
        return J
    
    def eq(x):
        return float(np.sum(x[1:])) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    bounds = [(-1.0, 5.0)] + [(0.0, 1.0)] * n
    
    res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                   constraints=[
                       {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                       {'type': 'eq', 'fun': eq, 'jac': eq_jac}
                   ],
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol})
    
    if res.success:
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    return h.copy(), _exact_c5(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _slsqp_step(h, top_k=30, maxiter=200, ftol=1e-13):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(corr)[::-1][:top_k]
    T0 = float(corr[active[0]])

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq(x):
        h_x = x[1:]
        c = np.correlate(h_x, 1.0 - h_x, mode='full')
        return x[0] - c[active]

    def ineq_jac(x):
        h_x = x[1:]
        omh = 1.0 - h_x
        J = np.zeros((top_k, n + 1))
        J[:, 0] = 1.0
        for i in range(top_k):
            r = active[i]
            k = r - (n - 1)
            dp = abs(k)
            if k >= 0:
                if k < n:
                    J[i, 1 + k:1 + n] += omh[:n - k]
                if n - k > 0:
                    J[i, 1:1 + n - k] -= h_x[k:]
            else:
                if n - dp > 0:
                    J[i, 1:1 + n - dp] += omh[dp:]
                if dp < n:
                    J[i, 1 + dp:1 + n] -= h_x[:n - dp]
        return J

    def eq(x):
        return float(np.sum(x[1:])) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.concatenate([[T0], h])
    bounds = [(T0 - 2.0, T0 + 2.0)] + [(0.0, 1.0)] * n

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       constraints=[
                           {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                           {'type': 'eq', 'fun': eq, 'jac': eq_jac}
                       ],
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h.copy()


def _slsqp_loop(h, deadline, top_k=30, max_rounds=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur = h.copy()

    for rnd in range(max_rounds):
        if time.time() > deadline:

        cur = _slsqp_step(cur, top_k=top_k, maxiter=200, ftol=1e-13)

            c5 = _exact_c5(cur)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = cur.copy()

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _project(h_new, n_new / 2.0)


def _shift_h(h, s):
    return np.roll(h, s)


def _reflect_h(h):
    return h[::-1].copy()


def _smooth_max_step(h, beta, step=0.02, n_iter=200):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    best = _exact_c5(cur)
    best_h = cur.copy()

    for _ in range(n_iter):
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        ms = float(np.max(corr))
        vals = np.clip(beta * (corr - ms), -50.0, 50.0)
        w = np.exp(vals)
        w /= float(np.sum(w))

        # Gradient of smooth max w.r.t. h
        g = np.zeros(n)
        for i in range(len(corr)):
            if w[i] < 1e-15:
                continue
            r = i
            k = r - (n - 1)
            dp = abs(k)
            if k >= 0:
                if k < n:
                    g[k:] += w[i] * (1.0 - cur[:n - k])
                if n - k > 0:
                    g[:n - k] -= w[i] * cur[k:]
            else:
                if n - dp > 0:
                    g[:n - dp] += w[i] * (1.0 - cur[dp:])
                if dp < n:
                    g[dp:] -= w[i] * cur[:n - dp]

        g -= np.mean(g)  # maintain sum constraint approximately
        cur_new = _project(cur - step * g, target)
        c5_new = _exact_c5(cur_new)
        if c5_new < best:
            best = c5_new
            best_h = cur_new.copy()
            cur = cur_new
        else:
            step *= 0.7

    return best_h, best


def _coordinate_polish(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    best_c5 = _exact_c5(cur)
    best_h = cur.copy()

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            for val in [0.0, 1.0]:
                if abs(cur[j] - val) < 1e-8:
                    continue
                h_try = cur.copy()
                # Adjust to maintain sum
                diff = val - cur[j]
                h_try[j] = val
                # Redistribute to other non-bound coords
                others = [i for i in range(n) if i != j and 1e-6 < h_try[i] < 1.0 - 1e-6]
                if not others:
                    continue
                adj = -diff / len(others)
                for i in others:
                    h_try[i] += adj
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    cur = h_try
                    best_c5 = c5_try
                    best_h = cur.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_h, best_c5
```
