    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(12):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.12, max_iters=5000, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _corr(h)
        p = _lp_dir(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 8 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5


def _slsqp_epigraph(h, active_idx, deadline, max_iter=500):
    n = len(h)
    target = n / 2.0
    K = len(active_idx)

    def _epi_obj(x):
        return x[0]

    def _epi_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def _eq_fun(x):
        return x[1:].sum() - target

    def _eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

    for i, r in enumerate(active_idx):
        d = r - (n - 1)

        def _ineq(x, d=d, r=r, i=i):
            T = x[0]
            hvec = x[1:]
            if d >= 0:
                m_max = n - d
                val = np.sum(hvec[:m_max] * (1.0 - hvec[d:d + m_max]))
            else:
                g = -d
                m_max = n - g
                val = np.sum(hvec[g:] * (1.0 - hvec[:m_max]))
            return T - val

        def _ineq_jac(x, d=d):
            hvec = x[1:]
            jg = np.zeros(n + 1)
            jg[0] = 1.0
            if d >= 0:
                m_max = n - d
                jg[1:1 + m_max] += (1.0 - hvec[d:d + m_max])
                jg[1 + d:n + 1] += -hvec[:m_max]
            else:
                g = -d
                m_max = n - g
                jg[1 + g:n + 1] += (1.0 - hvec[:m_max])
                jg[1:1 + m_max] += -hvec[g:g + m_max]
            return jg

        cons.append({'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac})

    x0 = np.zeros(n + 1)
    x0[0] = float(np.max(_corr(h)))
    x0[1:] = h
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    try:
        res = minimize(_epi_obj, x0, jac=_epi_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
        if res.success or res.fun < x0[0]:
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)


def _constraint_generation(h, deadline, max_rounds=30, k_init=8, k_max=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    k = k_init

    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        idx = _active_indices(h, k)
        # Smooth optimization focused on active set
        for alpha in [30.0, 60.0, 120.0]:
            if time.time() > deadline:
                break
            h_new, c5_new = _lbfgs_smooth(h, alpha, idx, maxiter=80, deadline=deadline)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        # Check for new violations
        g = _corr(h)
        g_max = float(np.max(g))
        top_k = np.sort(g)[::-1][:k]
        if len(top_k) > 0:
            threshold = g_max - 0.02 * g_max
        else:
            threshold = 0
        violating = np.where(g > threshold)[0]
        if len(violating) > k:
            k = min(k + 5, k_max)
        elif k > k_init:
            k = max(k - 1, k_init)
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = [j for j in range(n) if 1e-5 < h[j] < 1.0 - 1e-5]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in interior:
            if time.time() > deadline:
                break
            if not (1e-5 <= h[j] <= 1.0 - 1e-5):
                continue
            for k_idx in range(j + 1, n):
                if not (1e-5 <= h[k_idx] <= 1.0 - 1e-5):
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 0.5
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k_idx]) / dk)
                    else:
                        a_max = min(a_max, h[k_idx] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for frac in [0.5, 1.0, 0.25, 0.75]:
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] = np.clip(h_t[j] + dj * a, 0.0, 1.0)
                        h_t[k_idx] = np.clip(h_t[k_idx] + dk * a, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial smooth optimization (10% budget)
    d1 = t0 + budget_s * 0.10
    for alpha in [10.0, 25.0, 50.0, 100.0]:
        if time.time() > d1:
            break
        idx = _active_indices(h, 20)
        h, c5 = _lbfgs_smooth(h, alpha, idx, maxiter=100, deadline=d1)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Constraint generation with focused optimization (25% budget)
    d2 = t0 + budget_s * 0.35
    h_cg, c5_cg = _constraint_generation(h, d2, max_rounds=20, k_init=6, k_max=35)
    if c5_cg < best_c5:
        best_c5, best_h = c5_cg, h_cg.copy()

    # Phase 3: SLSQP epigraph on active constraints (20% budget)
    d3 = t0 + budget_s * 0.55
    for attempt in range(5):
        if time.time() > d3:
            break
        idx = _active_indices(best_h, 15)
        h_sl, c5_sl = _slsqp_epigraph(best_h, idx, d3, max_iter=400)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl.copy()
        # Try with slightly different active set
        if time.time() < d3:
            idx2 = _active_indices(best_h, 25)
            h_sl2, c5_sl2 = _slsqp_epigraph(best_h, idx2, d3, max_iter=300)
            if c5_sl2 < best_c5:
                best_c5, best_h = c5_sl2, h_sl2.copy()

    # Phase 4: LP polishing (20% budget)
    d4 = t0 + budget_s * 0.75
    h_lp, c5_lp = _lp_loop(best_h, d4, radius_init=0.12, max_iters=5000, stall_limit=45)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 5: Interleaved smooth + LP with restarts (15% budget)
    d5 = t0 + budget_s * 0.90
    scale = 0.03
    for attempt in range(2000):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        idx_p = _active_indices(h_p, 15)
        h_try, c5_try = _lbfgs_smooth(h_p, 80.0, idx_p, maxiter=40, deadline=d5)
        if c5_try < best_c5 - 1e-11:
            sub_d = min(d5, time.time() + budget_s * 0.03)
            h_try, c5_try = _lp_loop(h_try, sub_d, radius_init=0.08, max_iters=500, stall_limit=30)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.15)

    # Phase 6: Resolution search (7% budget)
    d6 = t0 + budget_s * 0.97
    for dn in [4, 8, 16, -4, -8, 32, -16, 12, -12, 6, -6]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _constraint_generation(h_r, sub_d, max_rounds=5, k_init=6, k_max=20)
        if time.time() < d6:
            sub_d2 = min(d6, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_d2, radius_init=0.08, max_iters=400, stall_limit=30)
        if c5_r < best_c5 - 1e-11:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Final polish (3% budget)
    if time.time() < t0 + budget_s * 0.99:
        d7 = t0 + budget_s * 0.995
        hp, c5p = _pair_polish(best_h, d7, max_passes=30)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()
        if time.time() < d7:
            hf, c5f = _lp_loop(best_h, d7, radius_init=0.03, max_iters=2000, stall_limit=60)
            if c5f < best_c5:
                best_c5, best_h = c5f, hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
