def _compute_gradients(h, n):
    """Compute gradients of all correlation terms."""
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for r in range(n_corr):
        s = r - (n - 1)  # lag
        if s == 0:
            grads[r] = 1.0 - 2.0 * h
        elif s > 0:
            m = n - s
            # First index: i ranges 0 to m-1
            for j in range(m):

                grads[r, j] += (1.0 - h[j + s])

            for i in range(m):

    dx = 2.0 / n

def _project(v, target):
    """Project v onto {x : 0 ≤ x ≤ 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 1e-10

        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _compute_F_and_grads(h):

def _lp_tangent_step(h, target, R, top_k=None):
    """
    Solve tangent plane LP to find descent direction.
    Returns d (direction vector) or None.
    """
    n = len(h)
    n_corr = 2 * n - 1
    F, grads = _compute_F_and_grads(h)
    
    if top_k is not None and top_k < n_corr:
        # Select top-k lags
        idx = np.argsort(-F)[:top_k]
    else:
        idx = np.arange(n_corr)
    
    k = len(idx)
    
    # Variables: [t, p_0, ..., p_{n-1}, q_0, ..., q_{n-1}]
    # Size: 1 + n + n = 2n+1
    
    # Objective: minimize t
    c = np.zeros(2 * n + 1)
    c[0] = 1.0
    
    # Inequality constraints: F_i + grad_i · d ≤ t
    # → -t + grad_i · p - grad_i · q ≤ -F_i
    A_ub_rows = []
    b_ub_rows = []
    for i in idx:
        row = np.zeros(2 * n + 1)
        row[0] = -1.0
        row[1:1+n] = grads[i]      # grad · p
        row[1+n:1+2*n] = -grads[i]  # -grad · q
        A_ub_rows.append(row)
        b_ub_rows.append(-F[i])
    
    # L1 norm: sum(p) + sum(q) ≤ R
    row = np.zeros(2 * n + 1)
    row[1:1+n] = 1.0
    row[1+n:1+2*n] = 1.0
    A_ub_rows.append(row)
    b_ub_rows.append(R)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality: sum(p) - sum(q) = 0
    A_eq_row = np.zeros(2 * n + 1)
    A_eq_row[1:1+n] = 1.0
    A_eq_row[1+n:1+2*n] = -1.0
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for j in range(n):
        if h[j] <= 1e-8:
            # Can only increase: p free, q = 0
            bounds.append((0, None))  # p_j
            bounds.append((0, 0))     # q_j = 0
        elif h[j] >= 1.0 - 1e-8:
            # Can only decrease: p = 0, q free
            bounds.append((0, 0))     # p_j = 0
            bounds.append((0, None))  # q_j
        else:
            bounds.append((0, None))  # p_j
            bounds.append((0, None))  # q_j
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
    except Exception:
        return None
    
    d = np.zeros(n)
    p = res.x[1:1+n]
    q = res.x[1+n:1+2*n]
    d = p - q
    d -= np.mean(d)  # Ensure zero sum
    
    # Normalize
    norm = float(np.max(np.abs(d)))
    if norm < 1e-10:
        return None
    d /= norm
    
    return d

def _line_search(h, direction, target, c5_cur, max_alpha=2.0):
    """Find best step size along direction."""
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-6)
    
    best_a = 0.0
    best_c = c5_cur
    alphas = np.linspace(0.05, alpha_max, 10)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.5)
    hi = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(15):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c
