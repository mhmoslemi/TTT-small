def _lp_grad_matrix(h):

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_c5(h, p, c5_cur, n_trial=12):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, n_trial):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(12):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    
    return best_a, best_c5


def _lp_loop_active(h, deadline, radius_init=0.1, max_iters=3000, stall_limit=50):
    """LP loop using only active constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    n_active = 15
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        scores = _correlations(h)
        max_s = float(np.max(scores))
        # Select active lags: top n_active or those within tolerance
        tol = max(0.01, 0.05 * max_s)
        active = np.where(scores > max_s - tol)[0]
        if len(active) > n_active:
            active = np.argsort(scores)[-n_active:]
        
        G = _lp_grad_matrix(h)
        p = _lp_direction(h, scores, G, active, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search_c5(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.06
    
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _binary_local_search(h, deadline, max_iters=50000):
    """Local search on binary assignments (n/2 ones, n/2 zeros)."""
    n = len(h)
    n_ones = n // 2
    target = n / 2.0
    
    # Convert to binary
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:n_ones]] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        
        # Random pair flip
        i = int(np.random.randint(len(ones)))
        j = int(np.random.randint(len(zeros)))
        pi, pj = int(ones[i]), int(zeros[j])
        
        h_try = h_bin.copy()
        h_try[pi] = 0.0
        h_try[pj] = 1.0
        
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            h_bin = h_try
            best_c5 = c5_try
            best_h = h_bin.copy()
            # Update sets
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
    
    # Also try continuous relaxation around the best binary
    h_cont = best_h.astype(float)
    # Small continuous perturbation
    noise = np.random.normal(0, 0.05, n)
    h_cont = _project(h_cont + noise, target)
    h_cont, c5_cont = _smooth_step(h_cont, 50.0, maxiter=80)
    h_cont, c5_cont = _active_set_solve(h_cont, min(deadline, time.time() + 5.0), init_tol=0.03, max_rounds=15)
    
    if c5_cont < best_c5:
        best_h, best_c5 = h_cont, c5_cont
    
    return best_h, best_c5


def _structural_transforms(h, rng):
    """Generate diverse structural transforms of h."""
    n = len(h)
    target = n / 2.0
    results = []
    
    # Reflection
    h_ref = _project(h[::-1].copy(), target)
    results.append(h_ref)
    
    # Complement reflection
    h_comp = _project(1.0 - h[::-1].copy(), target)
    results.append(h_comp)
    
    # Circular shift
    shift = int(rng.integers(1, max(2, n // 4)))
    h_shift = _project(np.roll(h, shift), target)
    results.append(h_shift)
    
    # Block swap
    bw = max(2, n // 8)
    start1 = int(rng.integers(0, n - 2 * bw))
    start2 = int(rng.integers(0, n - 2 * bw))
    if abs(start1 - start2) > bw:
        h_bs = h.copy()
        h_bs[start1:start1+bw], h_bs[start2:start2+bw] = h[start2:start2+bw].copy(), h[start1:start1+bw].copy()
        h_bs = _project(h_bs, target)
        results.append(h_bs)
    
    # Local inversion
    wl = max(2, n // 6)
    ws = int(rng.integers(0, n - wl))
    h_inv = h.copy()
    h_inv[ws:ws+wl] = 1.0 - h_inv[ws:ws+wl]
    h_inv = _project(h_inv, target)
    results.append(h_inv)
    
    return results

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_corr(x, idx):
    n = len(x)
    d = int(idx) - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - x[:m])
            g[:m] += -x[d:]
    else:
        gd = -d
        m = n - gd
        if m > 0:
            g[:m] += (1.0 - x[gd:])
            g[gd:] += -x[:m]
    return g


def _active_obj(x, active):
    c = _correlations(x)
    return float(np.sum(c[active]))


def _active_obj_grad(x, active):
    n = len(x)
    g = np.zeros(n)
    for idx in active:
        g += _grad_corr(x, int(idx))
    g -= np.mean(g)
    return g


def _smooth_obj(x, alpha):
    c = _correlations(x)
    ms = float(np.max(c))
    vals = np.clip(alpha * (c - ms), -500.0, 500.0)
    ev = np.exp(vals)
    return ms + (1.0 / alpha) * np.log(float(np.sum(ev)))


def _smooth_obj_grad(x, alpha):
    c = _correlations(x)
    ms = float(np.max(c))
    vals = np.clip(alpha * (c - ms), -500.0, 500.0)
    ev = np.exp(vals)
    w = ev / float(np.sum(ev))
    n = len(x)
    g = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] > 1e-10:
            g += w[i] * _grad_corr(x, i)
    g -= np.mean(g)
    return g


def _smooth_step(h, alpha, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), jac=_smooth_obj_grad,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40,
                               'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _active_set_solve(h, deadline, init_tol=0.05, max_rounds=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tol = init_tol
    bounds = [(0.0, 1.0)] * n

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = _correlations(h)
        max_corr = float(np.max(corr))
        active = np.where(corr > max_corr - tol)[0]
        if len(active) == 0:

        if len(active) > 35:

def _lp_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            gd = -d
            m = n - gd
            if m > 0:
                G[idx, :m] += (1.0 - h[gd:])
                G[idx, gd:] += -h[:m]
    return G


def _lp_direction(h, scores, G, active, radius):
    n = len(h)
    na = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((na, n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_c5(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a, best_c5 = 0.0, c5_cur
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a

    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        for _ in range(14):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(np.clip(h + a * p, 0.0, 1.0))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t

    return best_a, best_c5


def _lp_loop_active(h, deadline, radius_init=0.1, max_iters=3000, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    n_active = 12

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        scores = _correlations(h)
        max_s = float(np.max(scores))
        tol = max(0.01, 0.04 * max_s)
        active = np.where(scores > max_s - tol)[0]
        if len(active) > n_active:
            active = np.argsort(scores)[-n_active:]

        G = _lp_grad_matrix(h)
        p = _lp_direction(h, scores, G, active, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue

        alpha, new_c5 = _line_search_c5(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.08

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _binary_local_search(h, deadline, rng, max_iters=30000):
    n = len(h)
    n_ones = n // 2
    target = n / 2.0

    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:n_ones]] = 1.0

    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        pi = int(rng.integers(0, n_ones))
        pj = int(rng.integers(n_ones, n))
        # Find pi-th one and pj-th zero
        ones_pos = np.where(h_bin == 1.0)[0]
        zeros_pos = np.where(h_bin == 0.0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        a_idx = int(ones_pos[pi % len(ones_pos)])
        b_idx = int(zeros_pos[pj % len(zeros_pos)])

        h_try = h_bin.copy()
        h_try[a_idx] = 0.0
        h_try[b_idx] = 1.0
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            h_bin = h_try
            best_c5 = c5_try
            best_h = h_bin.copy()

    # Continuous relaxation
    h_cont = best_h.astype(float) + rng.normal(0, 0.04, n)
    h_cont = _project(h_cont, target)
    h_cont, c5_cont = _smooth_step(h_cont, 50.0, maxiter=60)
    sub_d = min(deadline, time.time() + 3.0)
    h_cont, c5_cont = _active_set_solve(h_cont, sub_d, init_tol=0.03, max_rounds=10)
    if c5_cont < best_c5:
        best_h, best_c5 = h_cont, c5_cont
    return best_h, best_c5


def _structural_transforms(h, rng):
    n = len(h)
    target = n / 2.0
    results = []

    h_ref = _project(h[::-1].copy(), target)
    results.append(h_ref)

    h_comp = _project(1.0 - h[::-1].copy(), target)
    results.append(h_comp)

    shift = int(rng.integers(1, max(2, n // 3)))
    h_shift = _project(np.roll(h, shift), target)
    results.append(h_shift)

    bw = max(2, n // 6)
    s1 = int(rng.integers(0, max(1, n - 2 * bw)))
    s2 = int(rng.integers(0, max(1, n - 2 * bw)))
    if abs(s1 - s2) > bw:
        h_bs = h.copy()
        tmp = h_bs[s1:s1 + bw].copy()
        h_bs[s1:s1 + bw] = h_bs[s2:s2 + bw]
        h_bs[s2:s2 + bw] = tmp
        h_bs = _project(h_bs, target)
        results.append(h_bs)

    wl = max(2, n // 5)
    ws = int(rng.integers(0, max(1, n - wl)))
    h_inv = h.copy()
    h_inv[ws:ws + wl] = 1.0 - h_inv[ws:ws + wl]
    h_inv = _project(h_inv, target)
    results.append(h_inv)

    return results

                        continue
                    a = min(a_max, 0.5)
                    h_t = h.copy()
                    h_t[int(j)] += dj * a
                    h_t[k] += dk * a
                    h_t = np.clip(h_t, 0.0, 1.0)
                    h_t = _project(h_t, target)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick smooth init (10%)
    d1 = t0 + budget_s * 0.10
    h, c5 = _smooth_step(h, 15.0, maxiter=200)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    h, c5 = _smooth_step(h, 50.0, maxiter=150)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Active set + LP (30%)
    d2 = t0 + budget_s * 0.40
    h, c5 = _active_set_solve(h, d2, init_tol=0.06, max_rounds=25)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    h, c5 = _lp_loop_active(h, d2, radius_init=0.12, max_iters=2500, stall_limit=45)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    if time.time() < d2:
        h, c5 = _active_set_solve(h, d2, init_tol=0.015, max_rounds=15)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: Diverse restarts (30%)
    d3 = t0 + budget_s * 0.70
    scale = 0.035
    for attempt in range(3000):
        if time.time() > d3:
            break
        if best_c5 < 0.3805:
            break
        rtype = rng.random()
        if rtype < 0.35:
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _project(h_p, best_n / 2.0)
            h_try, c5_try = _smooth_step(h_p, 35.0, maxiter=80)
            if c5_try < best_c5 - 1e-10:
                sub_d = min(d3, time.time() + budget_s * 0.025)
                h_try, c5_try = _active_set_solve(h_try, sub_d, init_tol=0.03, max_rounds=12)
                if time.time() < d3:
                    sub_d2 = min(d3, time.time() + budget_s * 0.015)
                    h_try, c5_try = _lp_loop_active(h_try, sub_d2, radius_init=0.08,
                                                    max_iters=800, stall_limit=30)
                if c5_try < best_c5:
                    best_c5, best_h = c5_try, h_try.copy()
                    scale = max(0.005, scale * 0.85)
                else:
                    scale = min(0.3, scale * 1.1)
            else:
                scale = min(0.3, scale * 1.1)
        elif rtype < 0.50:
            # Binary local search
            sub_d = min(d3, time.time() + budget_s * 0.04)
            h_b, c5_b = _binary_local_search(best_h, sub_d, rng, max_iters=25000)
            if c5_b < best_c5:
                best_c5, best_h = c5_b, h_b.copy()
        elif rtype < 0.80:
            # Structural transform
            transforms = _structural_transforms(best_h, rng)
            h_p = transforms[int(rng.integers(0, len(transforms)))]
            h_try, c5_try = _smooth_step(h_p, 40.0, maxiter=80)
            if c5_try < best_c5 - 1e-10:
                sub_d = min(d3, time.time() + budget_s * 0.02)
                h_try, c5_try = _active_set_solve(h_try, sub_d, init_tol=0.03, max_rounds=12)
                if c5_try < best_c5:
                    best_c5, best_h = c5_try, h_try.copy()
        else:
            # Very small perturbation + LP
            h_p = best_h + rng.normal(0, scale * 0.3, best_n)
            h_p = _project(h_p, best_n / 2.0)
            sub_d = min(d3, time.time() + budget_s * 0.02)
            h_try, c5_try = _lp_loop_active(h_p, sub_d, radius_init=0.06,
                                            max_iters=500, stall_limit=30)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.004, scale * 0.9)

    # Phase 4: Resolution sweep (10%)
    d4 = t0 + budget_s * 0.80
    for dn in [1, 2, 3, 4, -1, -2, -3, -4, 5, -5, 6, 8, -8]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _smooth_step(h_r, 30.0, maxiter=100)
        if time.time() < d4:
            h_r, c5_r = _active_set_solve(h_r, min(d4, time.time() + budget_s * 0.02),
                                          init_tol=0.04, max_rounds=10)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Pair polish (8%)
    if time.time() < t0 + budget_s * 0.88:
        d5 = t0 + budget_s * 0.88
        hp, c5p = _pair_polish(best_h, d5, max_passes=25)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 6: Final intensive polish (12%)
    d6 = t0 + budget_s * 0.97
    h, c5 = _active_set_solve(best_h, d6, init_tol=0.008, max_rounds=20)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    if time.time() < d6:
        h, c5 = _lp_loop_active(h, d6, radius_init=0.04, max_iters=5000, stall_limit=80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
