import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient of each lag-score with respect to h."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    # For lag index r (0 to 2n-2), the shift is d = r - (n-1)
    # S_r = dx * sum over valid (i) of h[i] * (1 - h[i-d]) for d>=0
    #      = dx * sum over valid (i) of h[i-d] * (1 - h[i]) for d<0
    # dScore_r / dh[j] = dx * (contribution when j appears)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            # S_r = dx * sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # dS/dh[j]: j appears as h[j] with (1-h[j-d]) if j>=d, and as (1-h[j]) multiplied by -h[j+d] if j+d < n
            # First term: j >= d: contribution = dx * (1 - h[j-d])
            # Second term: j+d < n: contribution = dx * (-h[j+d])
            if d == 0:
                # S = dx * sum h[i]^2... wait no
                # S = dx * sum h[i]*(1-h[i]) for d=0
                # dS/dh[j] = dx*(1-2*h[j])
                G[r, :] = dx * (1.0 - 2.0 * h)
            else:
                # j >= d: h[j]*(1-h[j-d])
                G[r, d:] += dx * (1.0 - h[:m])
                # j+d < n: h[j+d]*(1-h[j]), so dS/dh[j] includes -dx*h[j+d]
                G[r, :m] += dx * (-h[d:])
        else:
            g = -d
            m = n - g
            # S_r = dx * sum_{i=0}^{n-1-g} h[i+g]*(1-h[i])
            # = same as lag g but reversed
            # dS/dh[j]: j appears as h[j] with (1-h[j-g]) if j>=g, and as (1-h[j]) with -h[j+g] if j+g<n
            # Wait, let me re-derive. For shift d<0 (i.e., g=-d>0):
            # S = dx * sum_{i=0}^{n-1-g} h[i+g] * (1-h[i])
            # dS/dh[j]: 
            #   If j = i+g for some valid i (i.e., j >= g): contributes dx*(1-h[j-g])
            #   If j = i for some valid i (i.e., j < n-g): contributes dx*(-h[j+g])
            G[r, g:] += dx * (1.0 - h[:m])  # j = i+g, i from 0 to n-1-g
            G[r, :m] += dx * (-h[g:])  # j = i, i from 0 to n-1-g, so h[j+g] = h[i+g]
    return G

def _smooth_descent(h, deadline, tau_schedule=None, max_iters=2000):
    """Smooth softmax-based gradient descent with adaptive temperature."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    if tau_schedule is None:
        tau_schedule = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    n_tau = len(tau_schedule)
    iters_per_tau = max(50, max_iters // n_tau)
    
    for tau_idx, tau in enumerate(tau_schedule):
        for it in range(iters_per_tau):
            if time.time() > deadline:
                return best_h, best_c5
            scores = _scores(h)
            mx = scores.max()
            w = np.exp((scores - mx) / tau)
            sw = w.sum()
            if sw < 1e-300:
                w = np.zeros(len(scores))
                w[np.argmax(scores)] = 1.0
            else:
                w /= sw
            
            G = _grad_matrix(h)
            grad = G.T @ w
            grad -= np.mean(grad)
            
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            
            # Adaptive step size
            alpha = 0.02 * (1.0 + 0.5 * tau)
            for attempt in range(30):
                h_new = _project(h - alpha * grad, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
                alpha *= 0.6
                if alpha < 1e-12:
                    break
    
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    # First do a coarse grid to find a good bracket
    n_grid = 15
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = np.argmin(vals)
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i == 0 or best_i == n_grid - 1:
        return best_a, best_c5
    
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=20):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            # Only try a few partners for efficiency
            partners = [(j + k) % n for k in range(1, min(n, 12))]
            for k in partners:
                if k == j or h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _anneal(h, deadline, n_steps=5000, T_init=0.01, T_end=0.0005):
    """Simulated annealing with pair-transfer moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        T = T_init * (T_end / T_init) ** frac
        
        # Choose two random coordinates
        j = rng.integers(0, n)
        k = rng.integers(0, n)
        if j == k:
            continue
        
        # Transfer amount
        delta = rng.uniform(0.01, 0.08)
        sign = 1 if rng.random() < 0.5 else -1
        amount = sign * delta
        
        # Check bounds
        new_j = h[j] + amount
        new_k = h[k] - amount
        if new_j < 0 or new_j > 1 or new_k < 0 or new_k > 1:
            continue
        
        h[j] += amount
        h[k] -= amount
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            delta_f = new_c5 - cur_c5
            if rng.random() < np.exp(-delta_f / T):
                cur_c5 = new_c5
            else:
                h[j] -= amount
                h[k] += amount
    
    return best_h, best_c5

def _active_equalize(h, deadline, max_iters=500):
    """Try to equalize the top active lags using least squares."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        mx = scores.max()
        # Active set: lags within 5% of max
        threshold = mx - 0.05 * (mx - scores.min())
        active = np.where(scores >= threshold)[0]
        
        if len(active) == 0:
            break
        
        G = _grad_matrix(h)
        G_active = G[active]
        s_active = scores[active]
        
        # Solve: minimize ||G_active @ p + s_active - c||^2 + reg*||p||^2
        # subject to sum(p) = 0
        # This is a least-squares problem. Use the normal equations with constraint.
        # Let's use a simple approach: minimize max reduction over active lags
        # p direction = -G_active^T @ (s_active - mean(s_active)) normalized
        
        target_val = np.mean(s_active)
        residual = s_active - target_val
        p = -G_active.T @ residual
        p -= np.mean(p)  # ensure sum = 0
        
        pn = np.linalg.norm(p)
        if pn < 1e-15:
            break
        p = p / pn * 0.02  # small step
        
        # Check bounds
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 0.01)
        
        # Try a few step sizes
        for scale in [0.3, 0.5, 0.7, 1.0]:
            h_try = _project(h + alpha_max * scale * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
    
    return best_h, best_c5
