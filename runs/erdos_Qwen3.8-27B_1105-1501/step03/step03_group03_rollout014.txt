def _penalty_direction(h, scores, G, radius, K=12, mu=1.0, gamma=5.0,
                       n_iters=50):

    M = float(np.max(scores))
    # Select active set
    threshold = M - 0.05 * M  # top 5% of range
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        idx = np.argsort(scores)[-K:]
        active = idx
    else:
        # Limit to K elements
        if len(active) > K:
            order = np.argsort(scores[active])[::-1]
            active = active[order[:K]]

    G_active = G[active, :]
    
    p = np.zeros(n)
    step = 0.1 / (mu + gamma * np.sum(weights))
    
    for _ in range(n_iters):
        # Compute gradient
        inner = G_active @ p - allowed
        pos = np.maximum(inner, 0.0)

        if p_new is None:
