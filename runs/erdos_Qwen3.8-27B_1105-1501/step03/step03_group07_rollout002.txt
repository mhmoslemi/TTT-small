        return h0, _exact_c5(h0)

def _epigraph_iterative(h0, t0, deadline, target, max_rounds=5):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = t0
    for _round in range(max_rounds):
        if time.time() > deadline - 2.0:
            break
        vals = _values(best_h)
        cur_max = float(np.max(vals))
        K = min(40, 2 * n - 1)
        offsets = list(np.argsort(-vals)[:K].astype(int))
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (max_rounds - _round + 1) * 0.8)
        h_new, c_new = _epigraph_optimize(best_h, cur_max, sub_dl, target, offsets, reg=0.003)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
        else:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_start(n, rng):
    h = (rng.random(n) > 0.5).astype(float)
    return _project(h, n / 2.0)

def _smooth_start(n, rng):
    freq = rng.integers(1, 6)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(freq * x + phase) + 0.1 * np.sin(2 * freq * x + phase * 1.7)
    return _project(h, n / 2.0)

def _perturb_start(h0, rng, scale=0.05):
    return _project(h0 + rng.normal(0, scale, len(h0)), len(h0) / 2.0)

def _line_search(h, direction, target, c_cur, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-7), 3.0)
    best_a = 0.0
    best_c = c_cur
    alphas = np.linspace(0.05, alpha_max, 8)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c_cur
    lo = max(0.0, best_a * 0.6)
    hi = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(15):
        if hi - lo < 1e-9:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _gradient_polish(h, deadline, target, max_iters=5000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        vals = _values(h)
        max_idx = int(np.argmax(vals))
        s = max_idx - (n - 1)
        grad = _vals_grad(h, s)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        grad /= gnorm
        a, c_new = _line_search(h, -grad, target, cur_c5, max_alpha=0.8)
        if a > 0 and c_new < best_c5 - 1e-14:
            h = _project(h - a * grad, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                h_try = _perturb_start(best_h, rng, 0.02)
                a2, c2 = _line_search(h, h_try - h, target, cur_c5, max_alpha=0.5)
                if a2 > 0 and c2 < best_c5 - 1e-14:
                    h = _project(h + a2 * (h_try - h), target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
    return best_h, best_c5

def _pair_polish(h, deadline, target, max_rounds=30):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(456)
    for _r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        vals = _values(h)
        top_lags = np.argsort(-vals)[:5]
        for r in top_lags:
            if time.time() > deadline:
                break
            s = r - (n - 1)
            grad = _vals_grad(h, s)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                continue
            grad /= gnorm
            for sign in [1.0, -1.0]:
                a, c_new = _line_search(h, sign * grad, target, cur_c5, max_alpha=0.4)
                if a > 0 and c_new < best_c5 - 1e-14:
                    h = _project(h + a * sign * grad, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            for _ in range(200):
                if time.time() > deadline:
                    break
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                    continue
                if h[k] <= 1e-7 or h[k] >= 1.0 - 1e-7:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                    a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                    a_max = min(a_max_j, a_max_k)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LSE optimization from initial (30%)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _lse_optimize(h, dl1, target0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE with different tau schedule (15%)
    dl2 = t0 + budget_s * 0.45
    tau_alt = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008]
    h2, c5_2 = _lse_optimize(best_h, dl2, best_n / 2.0, tau_list=tau_alt)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Epigraph iterative (20%)
    dl3 = t0 + budget_s * 0.65
    h3, c5_3 = _epigraph_iterative(best_h, _exact_c5(best_h), dl3, best_n / 2.0, max_rounds=6)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Gradient polish (10%)
    dl4 = t0 + budget_s * 0.75
    h4, c5_4 = _gradient_polish(best_h, dl4, best_n / 2.0, max_iters=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution search (15%)
    dl5 = t0 + budget_s * 0.90
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8, -10, 10, 12, -12, 16, -16, 20, -20, 24, -24]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        remaining = dl5 - time.time()
        if remaining < 4.0:
            break
        alloc = min(remaining * 0.35, 30.0)
        sub_dl = time.time() + alloc
        h_r, c5_r = _lse_optimize(h_r, sub_dl, n_try / 2.0,
                                   tau_list=[0.04, 0.02, 0.01, 0.005, 0.002])
        if time.time() < dl5 - 2.0:
            sub_dl2 = min(dl5, time.time() + 8.0)
            h_r, c5_r = _gradient_polish(h_r, sub_dl2, n_try / 2.0, max_iters=500)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Diverse starts (5%)
    dl6 = t0 + budget_s * 0.95
    for _s in range(8):
        if time.time() > dl6 - 3.0:
            break
        r = rng.random()
        if r < 0.4:
            h_try = _perturb_start(best_h, rng, 0.04)
        elif r < 0.7:
            h_try = _smooth_start(best_n, rng)
        else:
            h_try = _binary_start(best_n, rng)
        alloc = min(dl6 - time.time() - 1.0, 15.0)
        if alloc < 3.0:
            break
        sub_dl = time.time() + alloc
        h_try, c_try = _lse_optimize(h_try, sub_dl, best_n / 2.0,
                                      tau_list=[0.03, 0.015, 0.008, 0.004])
        if c_try < best_c5 - 1e-10:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 7: Final pair polish (5%)
    dl7 = t0 + budget_s * 0.99
    h7, c5_7 = _pair_polish(best_h, dl7, best_n / 2.0, max_rounds=50)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    s = float(np.sum(best_h))
    diff = best_n / 2.0 - s
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
