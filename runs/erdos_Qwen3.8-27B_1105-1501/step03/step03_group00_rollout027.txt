def exact_step(h, p, scores, G):
    """Find exact optimal step size along direction p."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Compute grad for each lag
    grad = G @ p  # shape (n_lags,)
    grad *= dx  # scale by dx
    
    # Box constraint on alpha
    alpha_max = np.inf
    for i in range(n):
        if p[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-15:
            alpha_max = min(alpha_max, -h[i] / p[i])
    if not np.isfinite(alpha_max) or alpha_max <= 0:
        return 0.0, _exact_c5(h)
    alpha_max = min(alpha_max, 2.0)  # cap for safety
    
    # Solve: min t s.t. scores + alpha*grad <= t, 0 <= alpha <= alpha_max
    # Variables: [alpha, t]
    # Objective: [0, 1]
    c = np.array([0.0, 1.0])
    # Constraints: scores[d] + alpha*grad[d] - t <= 0 for all d
    # => grad[d]*alpha - t <= -scores[d]
    A = np.column_stack([grad, -np.ones(n_lags)])
    b = -scores
    # Bounds: 0 <= alpha <= alpha_max, t free (but t >= 0 practically)
    bounds = [(0, alpha_max), (0, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        alpha_opt = res.x[0]
        t_opt = res.x[1]
    else:
        # Fallback: evaluate at a few points
        alphas = np.linspace(0, alpha_max, 10)
        vals = np.array([np.max(scores + a * grad) for a in alphas])
        alpha_opt = alphas[np.argmin(vals)]
        t_opt = vals.min()
    
    h_new = h + alpha_opt * p
    h_new = np.clip(h_new, 0, 1)
    c5 = _exact_c5(h_new)
    return alpha_opt, c5

        return np.zeros(n)

    lo = float(v.min()) - 2.0

    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _proj(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute G@p where G is the gradient matrix of scores."""
    n = len(h)
    dx = 2.0 / n
    return dx * (np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full'))

def _exact_step(h, p, scores):

    n = len(h)
    n_lags = 2 * n - 1
    grad = _grad_scores(h, p)
    
    # Alpha bounds from box constraints
    alpha_max = 10.0
    for i in range(n):
        if p[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-15:
            alpha_max = min(alpha_max, -h[i] / p[i])
    if alpha_max <= 1e-12:
        return 0.0, _exact_c5(h)
    
    # LP: min t s.t. scores + alpha*grad <= t, 0 <= alpha <= alpha_max
    # Variables: [alpha, t]
    c = np.array([0.0, 1.0])
    A = np.column_stack([grad, -np.ones(n_lags)])
    b = -scores
    bounds = [(0.0, alpha_max), (0.0, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})

            alpha_opt = max(0.0, res.x[0])
            h_new = np.clip(h + alpha_opt * p, 0.0, 1.0)
            # Re-project to maintain sum exactly
            h_new = _proj(h_new, n / 2.0)
            c5 = _exact_c5(h_new)
            return alpha_opt, c5
    except Exception:
        pass
    
    # Fallback: golden section
    alpha_max = min(alpha_max, 2.0)
    a, b_ = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        c1 = b_ - gr * (b_ - a)
        c2 = a + gr * (b_ - a)
        f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
        f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
        if f1 < f2:

        else:
            a = c1
    alpha_opt = (a + b_) / 2.0
    h_new = _proj(np.clip(h + alpha_opt * p, 0, 1), n / 2.0)
    return alpha_opt, _exact_c5(h_new)

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _proj(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt

    return x

    n = len(h)
    dx = 2.0 / n
    return dx * (np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full'))

def _exact_step(h, p, scores):

    n_lags = 2 * n - 1
    grad = _grad_scores(h, p)
    
    alpha_max = 10.0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b_ - gr*(b_ - a); c2 = a + gr*(b_ - a)
    f1 = _exact_c5(np.clip(h + c1*p, 0, 1)); f2 = _exact_c5(np.clip(h + c2*p, 0, 1))
    for _ in range(40):
        if f1 < f2: b_ = c2; c2 = c1; f2 = f1; c1 = b_ - gr*(b_ - a); f1 = _exact_c5(np.clip(h + c1*p, 0, 1))
        else: a = c1; c1 = c2; f1 = f2; c2 = a + gr*(b_ - a); f2 = _exact_c5(np.clip(h + c2*p, 0, 1))
    alpha_opt = (a + b_) / 2.0
    h_new = _proj(np.clip(h + alpha_opt*p, 0, 1), n/2.0)
    return alpha_opt, _exact_c5(h_new)

def _build_G_rows(h, idx):
    """Build gradient matrix rows for specified lag indices."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(idx), n))
    for row, r in enumerate(idx):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[row, k:k+m] += (1.0 - h[:m])
                G[row, :m] += -h[k:k+m]
        else:
            m = -k
            if m < n:
                G[row, :n-m] += (1.0 - h[m:])
                G[row, m:] += -h[:n-m]
    G *= dx
    return G

def _lp_direction(h, scores, radius, top_k=None):
    n = len(h)
    n_lags = 2 * n - 1
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    s_k = scores[idx]
    G_k = _build_G_rows(h, idx)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _vertex_directions(h, n_vertices=10, seed=0):
    """Generate vertex-relation directions."""
    n = len(h)
    target = n // 2
    rng = np.random.default_rng(seed)
    directions = []
    
    # Greedy vertex: top n/2 entries
    order = np.argsort(h)[::-1]
    v = np.zeros(n)
    v[order[:target]] = 1.0
    directions.append(v - h)
    
    # Random vertices
    for _ in range(n_vertices - 1):
        idx = rng.choice(n, size=target, replace=False)
        v = np.zeros(n)
        v[idx] = 1.0
        directions.append(v - h)
    
    return directions

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _local_optimize(h, deadline, top_k_list=None, radius_init=0.1, max_iters=3000):
    """Main local optimizer using LP directions + exact step sizes."""
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        scores = _scores(h)
        tk = top_k_list[ti % n_tk]
        p = _lp_direction(h, scores, radius, top_k=tk)
        
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        
        # Ensure sum p = 0
        p -= np.mean(p)
        
        alpha, new_c5 = _exact_step(h, p, scores)
        
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = np.clip(h_new, 0.0, 1.0)
            if abs(np.sum(h_new) - target) > 1e-10:
                h_new = _proj(h_new, target)
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.05
    
    return best_h, best_c5

def _coord_descent(h, deadline):
    """Pair coordinate descent polish."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(5):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Try to optimize h[i] and h[j] jointly while keeping sum fixed
                # h[i] + h[j] = s (constant)
                s = h[i] + h[j]
                if s < 1e-12 or s > 2.0 - 1e-12:
                    continue
                # h[i] in [max(0, s-1), min(1, s)]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                # Try a few values
                for frac in [0.25, 0.5, 0.75]:
                    h_try_i = lo + frac * (hi - lo)
                    if abs(h_try_i - h[i]) < 1e-10:
                        continue
                    h_try_j = s - h_try_i
                    if h_try_j < -1e-10 or h_try_j > 1.0 + 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[i] = h_try_i
                    h_try[j] = h_try_j
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5
