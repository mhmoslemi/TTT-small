def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    # Bisection for exact projection onto {0<=x<=1, sum=x=target}
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    # Fix rounding
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:

def _resample(h, n_new):
    n = len(h)

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _block_perturb(h, rng, n_blocks=3, block_size=None):
    """Structured perturbation: flip blocks of values"""
    n = len(h)
    h_p = h.copy()
    if block_size is None:
        block_size = max(2, n // (2 * n_blocks))
    for _ in range(n_blocks):
        start = rng.integers(0, n - block_size)
        end = start + block_size
        # Invert the block (around 0.5)
        h_p[start:end] = 0.5 + 0.5 * (1.0 - 2.0 * h[start:end])  # = 1 - h
    return _project(h_p, n / 2.0)

def _shift_perturb(h, rng, max_shift=3):
    """Cyclic shift perturbation"""
    n = len(h)
    shift = rng.integers(1, max_shift + 1)

    blend = 0.3
    h_p[:shift] = (1-blend) * h_p[:shift] + blend * 0.5
    h_p[-shift:] = (1-blend) * h_p[-shift:] + blend * 0.5
    return _project(h_p, n / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (x > 1e-15) & (x < 1.0 - 1e-15)
        else:
            mask = (x > 1e-15) & (x < 1.0 - 1e-15)
        if not np.any(mask):
            mask = x < 1.0 if diff > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _adaptive_descent(h, deadline, beta_start=20.0, beta_end=50000.0,
                      max_iters=50000, use_momentum=True, use_anneal=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.04
    stall = 0
    it = 0
    mom = 0.85
    vel = np.zeros(n)
    T_start = 0.008
    T_end = 0.00005

    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        T = T_start * (T_end / T_start) ** frac

        scores = _scores(h)
        max_s = float(np.max(scores))
        eps = max_s * 0.05 + 0.01
        near_idx = np.where(scores > max_s - eps)[0]

        if len(near_idx) <= 3:
            K = len(near_idx)
            weights_full = np.zeros(len(scores))
            weights_full[near_idx] = 1.0 / K
        else:
            K = min(8, len(near_idx))
            top_idx = np.argpartition(scores, -K)[-K:]
            top_scores = scores[top_idx]
            m = float(np.max(top_scores))
            exps = np.exp(beta * (top_scores - m))
            weights_full = np.zeros(len(scores))
            weights_full[top_idx] = exps / float(np.sum(exps))

        sel = weights_full > 1e-15
        G = _grad_matrix(h)
        grad = G[:, sel].T @ weights_full[sel]
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm

        if use_momentum:
            vel = mom * vel + (1.0 - mom) * direction
            vnorm = float(np.linalg.norm(vel))
            if vnorm > 1e-14:
                direction = vel / vnorm

        alpha = lr
        improved = False
        cur_scores_cost = cur_c5
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.25)
                improved = True
                break
            elif use_anneal and (c5_try - cur_c5) < T:
                if np.random.random() < np.exp(-(c5_try - cur_c5) / T):
                    h = h_try
                    cur_c5 = c5_try
                    stall = 0
                    improved = True
                    break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.82
            mom = min(mom + 0.005, 0.95)
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _balance_descent(h, deadline, K=5):
    """Directly balance the top K lags by minimizing their variance + max."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.03
    stall = 0

    while time.time() < deadline:
        scores = _scores(h)
        top_idx = np.argpartition(scores, -K)[-K:]

        max_s = float(np.max(top_scores))
        # Weight: emphasize those closest to max
        w_rel = top_scores / (max_s + 1e-10)
        w_full = np.zeros(len(scores))
        w_full[top_idx] = w_rel / float(np.sum(w_rel))

        G = _grad_matrix(h)
        sel = w_full > 1e-15
        grad = G[:, sel].T @ w_full[sel]
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False

                if cur_c5 < best_c5:

                improved = True
                break

        if improved:

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline, radius_init=0.06, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()

    best_h = h.copy()

            break

            break

                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:

            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = _exact_c5(_project(h + c1 * p, target))

        for _ in range(25):
            if hi_a - lo_a < 1e-10:
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1
                f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo_a = c1
                c1 = c2
                f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _exact_c5(_project(h + c2 * p, target))
        opt_a = (lo_a + hi_a) / 2.0

def _block_perturb(h, rng, n_blocks=3):
    n = len(h)
    h_p = h.copy()
    block_size = max(2, n // (2 * n_blocks + 1))
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = start + block_size
        h_p[start:end] = 1.0 - h[start:end]
    return _project(h_p, n / 2.0)

def _shift_perturb(h, rng, max_shift=4):
    n = len(h)
    shift = int(rng.integers(1, max_shift + 1))
    d = 1 if rng.random() > 0.5 else -1
    h_p = np.roll(h, shift * d)
    blend = 0.3
    h_p[:shift] = (1.0 - blend) * h_p[:shift] + blend * 0.5
    h_p[-shift:] = (1.0 - blend) * h_p[-shift:] + blend * 0.5
    return _project(h_p, n / 2.0)

def _gauss_perturb(h, rng, scale):
    return _project(h + rng.normal(0, scale, len(h)), len(h) / 2.0)

def _sine_init(n, freq, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
    return _project(h, n / 2.0)

def _thue_morse_init(n, amp):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amp * (2.0 * tm - 1.0)
    return _project(h, n / 2.0)
