def _line_min(h, d, C, A, Q, use_model, dx):
    n = len(h)
    tmin, tmax = _box_interval(h, d)
    if not (tmax - tmin > 1e-12):
        return 0.0, h.copy(), _exact_c5(h)
    def fline(t):
        if use_model:
            return float(np.max(C + A*t - Q*t*t)) * dx
        return _exact_c5(h + t*d)
    gr = (np.sqrt(5.0)-1.0)/2.0
    lo, hi = tmin, tmax
    c = hi - gr*(hi-lo)
    dd = lo + gr*(hi-lo)
    fc = fline(c); fd = fline(dd)
    for _ in range(48):
        if hi-lo < 1e-10:
            break
        if fc < fd:
            hi = dd; dd = c; fd = fc; c = hi - gr*(hi-lo); fc = fline(c)
        else:
            lo = c; c = dd; fc = fd; dd = lo + gr*(hi-lo); fd = fline(dd)
    tbest = 0.5*(lo+hi)
    h_new = h + tbest*d
    h_new = _project(h_new, n/2.0)
    c5_new = _exact_c5(h_new)
    return tbest, h_new, c5_new

def _subgrad_dir(h, tau, rng, noise):
    n = len(h)
    corr = np.correlate(h,1-h,'full')
    s = corr - corr.max()
    if tau <= 0:
        w = np.zeros(2*n-1); w[np.argmax(corr)] = 1.0
    else:
        ew = np.exp(tau*s); w = ew/np.sum(ew)
    d = np.zeros(n)
    for r in range(2*n-1):
        wk = w[r]
        if wk < 1e-15: continue
        lag = r-(n-1)
        i_lo = max(0,-lag); i_hi=min(n-1,n-1-lag)
        if i_lo>i_hi: continue
        i1=np.arange(i_lo,i_hi+1); i2=i1+lag
        # gradient of corr_r wrt h_j = 1 - h_{j+lag} - h_{j-lag} (valid neighbors)
        # We accumulate over r: d = - sum wk * grad_r. So compute grad contributions.
        # For each j in i1 (as the 'i' role): grad_j += (1 - h[j+lag]) ; as 'i+lag' role at j=i2: grad_{i2} += (1 - h[i2-lag])? 
        # Let's use the earlier gradient formula gradient_j f_r = 1 - h_{j+lag} - h_{j-lag} with OOR dropped.
    # Simpler: compute per-j by two accumulations:
    grad = np.zeros(n)
    for r in range(2*n-1):
        wk = w[r]
        if wk < 1e-15: continue
        lag = r-(n-1)
        # term (1 - h_{j+lag}) for j in [0,n-1] with 0<=j+lag<n
        a_lo=max(0,-lag); a_hi=min(n-1,n-1-lag)
        j1=np.arange(a_lo,a_hi+1)
        grad[j1] += wk*(1.0 - h[j1+lag])
        # term - h_{j-lag} for j with 0<=j-lag<n
        b_lo=max(0,lag); b_hi=min(n-1,n-1+lag)
        j2=np.arange(b_lo,b_hi+1)
        grad[j2] += wk*(-h[j2-lag])
    d = -grad
    if noise>0:
        d = d + noise*rng.normal(0,1,n)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    d /= nrm
    return d

import numpy as np
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _random_start(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    hn = np.interp(xn, xo, h)
    return _project(hn, n_new / 2.0)


def _box_interval(h, d):
    tmin = 0.0
    tmax = 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        tmin = max(tmin, float(np.max(-h[pos] / d[pos])))
        tmax = min(tmax, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tmin = max(tmin, float(np.max((1.0 - h[neg]) / d[neg])))
        tmax = min(tmax, float(np.min(-h[neg] / d[neg])))
    tmin = min(tmin, 0.0)
    tmax = max(tmax, 0.0)
    return tmin, tmax


def _coeffs(h, d):
    n = len(h)
    m = 2 * n - 1
    C = np.zeros(m)
    A = np.zeros(m)
    Q = np.zeros(m)
    for r in range(m):
        lag = r - (n - 1)
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo > i_hi:
            continue
        i1 = np.arange(i_lo, i_hi + 1)
        i2 = i1 + lag
        hi = h[i1]
        h2 = h[i2]
        di = d[i1]
        d2 = d[i2]
        C[r] = float(np.sum(hi) - np.sum(hi * h2))
        A[r] = float(np.sum(di) - np.sum(hi * d2 + di * h2))
        Q[r] = float(np.sum(di * d2))
    return C, A, Q


def _eval_line(t, h, d, C, A, Q, dx, use_model):
    if use_model:
        return float(np.max(C + A * t - Q * t * t)) * dx
    return _exact_c5(h + t * d)


def _validate(h, d, C, A, Q, dx, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if float(np.max(np.abs(C - corr))) > 1e-6:
        return False
    tp = 0.07 * (1.0 + np.float64(rng.uniform(0, 1)))
    model = float(np.max(C + A * tp - Q * tp * tp)) * dx
    direct = _exact_c5(h + tp * d)
    return abs(model - direct) < 1e-5


def _line_min(h, d, C, A, Q, use_model, dx):
    n = len(h)
    tmin, tmax = _box_interval(h, d)
    if not (tmax - tmin > 1e-12):
        return 0.0, h.copy(), _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = tmin
    hi = tmax
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc = _eval_line(c, h, d, C, A, Q, dx, use_model)
    fd = _eval_line(dd, h, d, C, A, Q, dx, use_model)
    for _ in range(46):
        if hi - lo < 1e-10:
            break
        if fc < fd:
            hi = dd
            dd = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _eval_line(c, h, d, C, A, Q, dx, use_model)
        else:
            lo = c
            c = dd
            fc = fd
            dd = lo + gr * (hi - lo)
            fd = _eval_line(dd, h, d, C, A, Q, dx, use_model)
    tbest = 0.5 * (lo + hi)
    h_new = h + tbest * d
    h_new = _project(h_new, n / 2.0)
    c5_new = _exact_c5(h_new)
    return tbest, h_new, c5_new


def _subgrad_dir(h, tau, rng, noise):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if tau <= 0.0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        s = corr - float(np.max(corr))
        ew = np.exp(tau * s)
        w = ew / float(np.sum(ew))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wk = w[r]
        if wk < 1e-15:
            continue
        lag = r - (n - 1)
        a_lo = max(0, -lag)
        a_hi = min(n - 1, n - 1 - lag)
        if a_lo <= a_hi:
            j1 = np.arange(a_lo, a_hi + 1)
            grad[j1] += wk * (1.0 - h[j1 + lag])
        b_lo = max(0, lag)
        b_hi = min(n - 1, n - 1 + lag)
        if b_lo <= b_hi:
            j2 = np.arange(b_lo, b_hi + 1)
            grad[j2] += wk * (-h[j2 - lag])
    d = -grad
    if noise > 0.0:
        d = d + noise * rng.normal(0.0, 1.0, n)
    d = d - float(np.mean(d))
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _descent_pass(h, deadline, tau_list, noise, random_frac, rng):

    target = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    best_c5 = _exact_c5(h)
    cur = h.copy()
    cur_c5 = best_c5
    ntau = len(tau_list)
    rsi = 0
    stall = 0
    while time.time() < deadline:
        if stall > 140:
            break
        rsi += 1
        rf = int(round(10.0 * random_frac))
        if (rsi % 10) < rf:
            d = rng.normal(0.0, 1.0, n)
        else:
            d = _subgrad_dir(cur, tau_list[rsi % ntau], rng, noise)
        if d is None:
            stall += 1
            continue
        d = d - float(np.mean(d))
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            continue
        d = d / nrm
        C, A, Q = _coeffs(cur, d)
        use_model = _validate(cur, d, C, A, Q, dx, rng)
        _tb, hn, c5n = _line_min(cur, d, C, A, Q, use_model, dx)
        if c5n < cur_c5 - 1e-13:
            cur = hn
            cur_c5 = c5n
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        else:
            stall += 1
            if stall % 30 == 0:
                kn = 0.02 * rng.normal(0.0, 1.0, n)

                kn = _project(cur + kn, target)
                kc = _exact_c5(kn)
                if kc < cur_c5 + 0.02:
                    cur = kn
                    cur_c5 = kc
    return best, best_c5

    best = h.copy()

    hh = h.copy()
    for _p in range(400):
        if time.time() > deadline:

            if time.time() > deadline:

        h0 = _random_start(84, rng)
    n = len(h0)
    h = _project(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    tau_list = [0.0, 25.0, 80.0, 200.0, 500.0]

    # Phase 0: coarse-resolution basin scan
    d0 = t0 + budget_s * 0.06
    for nn in (60, 64, 68, 72, 84, 96, 108, 120):
        if time.time() > d0:
            break
        hr = _resample(h, nn)
        hS, cS = _descent_pass(hr, min(d0, time.time() + budget_s * 0.004), tau_list, 0.02, 0.10, rng)
        if cS < best_c5:
            best_h, best_c5, best_n = hS, cS, nn

    # Phase 1: subgradient-driven exact convex line descent
    d1 = t0 + budget_s * 0.42
    h1, c1 = _descent_pass(best_h, d1, tau_list, 0.02, 0.10, rng)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: random directional diversity to escape local optima
    d2 = t0 + budget_s * 0.66
    h2, c2 = _descent_pass(best_h, d2, tau_list, 0.05, 0.60, rng)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: multi-resolution resampling
    d3 = t0 + budget_s * 0.84
    for dn in (-8, -4, -2, -1, 1, 2, 4, 8, -16, 16, -12, 12, -6, 6):
        if time.time() > d3:
            break
        nn = best_n + dn
        if nn < 44 or nn > 170:
            continue
        hr = _resample(best_h, nn)
        h3, c3 = _descent_pass(hr, min(d3, time.time() + budget_s * 0.012), tau_list, 0.02, 0.15, rng)
        if c3 < best_c5 - 1e-12:
            best_h, best_c5, best_n = h3, c3, nn

    # Phase 4: pair coordinate polish
    d4 = t0 + budget_s * 0.94
    hp, ca = _pair_polish(best_h, d4)
    if ca < best_c5:
        best_h, best_c5 = hp, ca

    # Phase 5: final micro descent
    d5 = deadline - budget_s * 0.005
    hm, cm = _descent_pass(best_h, d5, tau_list, 0.005, 0.20, rng)
    if cm < best_c5:
        best_h, best_c5 = hm, cm

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
