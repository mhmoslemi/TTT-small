    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bin_fwd_bwd(b):
    n = len(b)
    fwd = np.zeros(n, dtype=int)
    bwd = np.zeros(n, dtype=int)
    for d in range(1, n):
        m = n - d
        fwd[d] = int(np.sum(b[:m] * (1 - b[d:])))
        bwd[d] = int(np.sum((1 - b[:m]) * b[d:]))
    return fwd, bwd

def _bin_max_count(b):
    fwd, bwd = _bin_fwd_bwd(b)
    return max(int(np.max(fwd[1:])), int(np.max(bwd[1:])))

def _swap_delta(b, i, j, d):
    """Delta for fwd and bwd at distance d when b[i] 1->0, b[j] 0->1."""
    df = 0
    db = 0
    n = len(b)
    if i + d < n:
        if b[i + d] == 0:
            df -= 1
        else:
            db += 1
    if i - d >= 0:
        if b[i - d] == 0:
            db -= 1
        else:
            df += 1
    if j + d < n:
        if b[j + d] == 1:
            db -= 1
        else:
            df += 1
    if j - d >= 0:
        if b[j - d] == 1:
            df -= 1
        else:
            db += 1
    return df, db

def _binary_local_search(b, deadline, max_passes=200000, target_max=12):
    n = len(b)
    b = b.astype(int).copy()
    fwd, bwd = _bin_fwd_bwd(b)
    max_c = max(int(np.max(fwd[1:])), int(np.max(bwd[1:])))
    best_b = b.copy()
    best_mc = max_c
    rng = np.random.default_rng(7)
    stuck = 0

    for p in range(max_passes):
        if time.time() > deadline:
            break
        if max_c <= target_max:
            break
        if stuck > 500:
            break

        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Critical distances: those with count >= max_c - 1
        crit = []
        for d in range(1, n):
            if fwd[d] >= max_c - 1 or bwd[d] >= max_c - 1:
                crit.append(d)
        if not crit:
            crit = list(range(1, n))

        best_delta = 0
        best_swap = None

        for i in ones:
            for j in zeros:
                new_max = 0
                for d in crit:
                    df, db = _swap_delta(b, i, j, d)
                    nc = max(fwd[d] + df, bwd[d] + db)
                    if nc > new_max:
                        new_max = nc
                        if new_max >= max_c:
                            break
                if new_max < max_c:
                    delta = max_c - new_max
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                        if delta >= 2:
                            break
            if best_delta >= 2:
                break

        if best_swap is not None:
            i, j = best_swap
            b[i] = 0
            b[j] = 1
            max_c -= best_delta
            stuck = 0
            if max_c < best_mc:
                best_mc = max_c
                best_b = b.copy()
            fwd, bwd = _bin_fwd_bwd(b)
        else:
            stuck += 1
            if stuck % 20 == 0:
                nf = min(3, len(ones), len(zeros))
                fo = rng.choice(len(ones), nf, replace=False)
                fz = rng.choice(len(zeros), nf, replace=False)
                for a in fo:
                    b[ones[a]] = 0
                for a in fz:
                    b[zeros[a]] = 1
                fwd, bwd = _bin_fwd_bwd(b)
                max_c = max(int(np.max(fwd[1:])), int(np.max(bwd[1:])))
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]

    return best_b, best_mc

def _gen_mseq63():
    """Generate m-sequence of length 63 (degree 6 LFSR)."""
    # Primitive polynomial x^6 + x + 1, taps at bit 0 and bit 5

        fb = state & 1
        state >>= 1

            # Actually for x^6+x+1: feedback = bit0, taps at positions 0,5
            # Standard form: new_bit = bit0 XOR bit5, shift right
            state &= 0x3F
            # Let me redo this properly
    # Redo with proper LFSR
    state = 0x3F  # all ones
    seq = []
    for _ in range(63):
        seq.append(state & 1)
        fb = (state & 1) ^ ((state >> 5) & 1)
        state = (state >> 1) | (fb << 5)

def _gen_candidates(n, target, h0, rng):
    """Generate multiple initial binary candidates."""
    cands = []
    # 1. Binarize parent
    b = (h0 >= 0.5).astype(int)
    cnt = int(np.sum(b))
    if cnt > target:
        excess = cnt - target
        scores = np.abs(h0 - 0.5)
        idx = np.where(b == 1)[0]
        order = idx[np.argsort(scores[idx], kind='stable')]
        for k in range(excess):
            b[order[k]] = 0
    elif cnt < target:
        deficit = target - cnt
        scores = np.abs(h0 - 0.5)
        idx = np.where(b == 0)[0]
        order = idx[np.argsort(scores[idx], kind='stable')]
        for k in range(deficit):
            b[order[k]] = 1
    cands.append(b.copy())

    # 2. M-sequence based
    if n == 64:
        ms = _gen_mseq63()
        # Pad to 64: try different insertion points
        for pos in [0, 31, 32, 63]:
            b2 = np.empty(64, dtype=int)
            b2[:pos] = ms[:pos]
            b2[pos] = 1 - int(ms.sum() % 2)  # add minority bit
            b2[pos:] = ms[pos:63]
            # Fix balance
            cnt2 = int(np.sum(b2))
            if cnt2 > 32:
                b2[np.argmin(np.abs(np.arange(64) - 32.0).astype(float))] = 0
            elif cnt2 < 32:
                b2[np.argmin(np.abs(np.arange(64) - 32.0).astype(float))] = 1
            # Properly fix
            cnt2 = int(np.sum(b2))
            if cnt2 != 32:
                if cnt2 > 32:
                    for k in range(cnt2 - 32):
                        z = np.where(b2 == 1)[0]
                        b2[z[rng.integers(len(z))]] = 0
                else:
                    for k in range(32 - cnt2):
                        z = np.where(b2 == 0)[0]
                        b2[z[rng.integers(len(z))]] = 1
            cands.append(b2.copy())

    # 3. Random balanced
    for _ in range(8):
        b3 = np.zeros(n, dtype=int)
        ones_pos = rng.choice(n, size=target, replace=False)
        b3[ones_pos] = 1
        cands.append(b3.copy())

    # 4. Structured: semi-regular patterns
    for pattern_seed in range(4):
        b4 = np.zeros(n, dtype=int)
        period = n // (4 + pattern_seed)
        for i in range(n):
            b4[i] = 1 if (i % (period * 2)) < period else 0
        # Shuffle slightly
        for _ in range(20):
            j1, j2 = rng.integers(0, n, 2)
            if b4[j1] != b4[j2]:
                b4[j1], b4[j2] = b4[j2], b4[j1]
        cands.append(b4.copy())

    return cands

def _subgrad_optimize(h, deadline, target, T_init=0.3, max_iters=20000):
    n = len(h)
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    v = np.zeros(n)
    alpha = 0.06
    beta = 0.5
    T = T_init
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        shifted = corr - float(np.max(corr))
        w = np.exp(shifted / T)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= wsum

        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-8:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[d:] += w[r] * (1.0 - h[:m])
                    grad[:m] -= w[r] * h[d:]
            else:
                l = -d
                m = n - l
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[l:])
                    grad[l:] -= w[r] * h[:m]

        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        grad /= gnorm

        v = beta * v - alpha * grad
        v -= np.mean(v)

        # Line search
        alpha_max = 2.0
        for j in range(n):
            if v[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / v[j])
            elif v[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-v[j]))
        alpha_max = min(max(alpha_max, 1e-6), 3.0)

        best_a = 0.0
        best_c = cur_c5
        for a in np.linspace(0.01, alpha_max, 8):
            h_t = _project(h + a * v, target)
            c_t = _c5_exact(h_t)
            if c_t < best_c - 1e-14:
                best_c = c_t
                best_a = a

        if best_a > 0 and best_c < cur_c5 - 1e-14:
            h = _project(h + best_a * v, target)
            cur_c5 = _c5_exact(h)
            stall = 0
            T = max(0.05, T * 0.98)
            beta = min(0.9, beta + 0.001)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            alpha *= 0.85
            T = max(0.05, T * 0.93)
            if stall > 25:
                v *= 0.3
                alpha = 0.04
                T = 0.15

    return best_h, best_c5

def _fine_polish(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55)

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(-corr)[:8]
        improved = False

        for r in top_lags:
            if time.time() > deadline:
                break
            d = int(r) - (n - 1)
            grad = np.zeros(n)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[d:] += (1.0 - h[:m])
                    grad[:m] -= h[d:]
            else:
                l = -d
                m = n - l
                if m > 0:
                    grad[:m] += (1.0 - h[l:])
                    grad[l:] -= h[:m]
            grad -= np.mean(grad)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-10:
                continue
            grad /= gnorm

            for sign in [1.0, -1.0]:
                for a in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    h_t = _project(h + sign * a * grad, target)
                    c_t = _c5_exact(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved =
                        break

        if not improved:
            # Random micro-moves
            for _ in range(50):
                if time.time() > deadline:
                    break
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k:
                    continue
                if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6 or h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = min((1 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                                (1 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                    if a_max <= 0:
                        continue
                    for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _c5_exact(h_t)
                        if c_t < best_c5 - 1e-14:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            cur_c5 = c_t
                            improved = True
                            break
        if not improved:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    best_h = _project(h0, target0)
    best_c5 = _c5_exact(best_h)
    best_n = n0

    # Phase 1: Binary local search (50% of budget)
    dl1 = t0 + budget_s * 0.50
    cands = _gen_candidates(n0, int(target0), best_h, rng)

    best_bin = None
    best_bin_mc = 999
    per_cand_time = min(20.0, (dl1 - time.time()) / len(cands))

    for ci, cand in enumerate(cands):
        if time.time() > dl1:
            break
        sub_dl = time.time() + per_cand_time
        b_opt, mc = _binary_local_search(cand, sub_dl, target_max=12)
        if mc < best_bin_mc:
            best_bin_mc = mc
            best_bin = b_opt.copy()
        # If we hit target, stop early
        if best_bin_mc <= 12:
            break

    # If we found binary with max_count <= 12, that's C5 = 12/32 = 0.375
    if best_bin is not None and best_bin_mc <= int(target0 * 0.3808):
        # C5 for this binary solution
        h_bin = best_bin.astype(float)
        c5_bin = _c5_exact(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 2: More binary search with different strategies (20%)
    dl2 = t0 + budget_s * 0.70
    if best_bin is not None and best_bin_mc > 12 and time.time() < dl2:
        # Try 2-swap moves to escape local optima
        b = best_bin.copy()
        fwd, bwd = _bin_fwd_bwd(b)
        max_c = max(int(np.max(fwd[1:])), int(np.max(bwd[1:])))
        for _ in range(500):
            if time.time() > dl2:
                break
            if max_c <= 12:
                break
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            # Try 2-swap: swap two 1s with two 0s
            for _trial in range(50):
                if len(ones) < 2 or len(zeros) < 2:
                    break
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                b[ones[i1]] = 0
                b[ones[i2]] = 0
                b[zeros[j1]] = 1
                b[zeros[j2]] = 1
                mc = _bin_max_count(b)
                if mc < max_c:
                    max_c = mc
                    fwd, bwd = _bin_fwd_bwd(b)
                    if max_c < best_bin_mc:
                        best_bin_mc = max_c
                        best_bin = b.copy()
                else:
                    b[ones[i1]] = 1
                    b[ones[i2]] = 1
                    b[zeros[j1]] = 0
                    b[zeros[j2]] = 0

        # Continue single-swap search from improved point
        if best_bin is not None:
            b_opt, mc = _binary_local_search(best_bin, dl2, target_max=12)
            if mc < best_bin_mc:
                best_bin_mc = mc
                best_bin = b_opt.copy()

    # Phase 3: Continuous optimization from best binary (25%)
    dl3 = t0 + budget_s * 0.95
    if best_bin is not None:
        h_start = best_bin.astype(float)
        # Suggest continuous: nudge slightly off binary to allow fractional improvement
        for nudge_scale in [0.0, 0.02, 0.05]:
            if time.time() > dl3:
                break
            if nudge_scale > 0:
                h_try = _project(h_start + nudge_scale * rng.normal(0, 1, n0), target0)
            else:
                h_try = h_start.copy()
            sub_dl = min(dl3, time.time() + (dl3 - time.time()) * 0.7)
            h_opt, c5_opt = _subgrad_optimize(h_try, sub_dl, target0, T_init=0.25, max_iters=30000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Also run continuous from original parent
    if time.time() < dl3:
        sub_dl = min(dl3, time.time() + (dl3 - time.time()) * 0.5)
        h_opt, c5_opt = _subgrad_optimize(best_h, sub_dl, target0, T_init=0.3, max_iters=20000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Fine polish (5%)
    dl4 = t0 + budget_s * 0.99
    if time.time() < dl4:
        h_pol, c5_pol = _fine_polish(best_h, dl4, target0)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    s = float(np.sum(best_h))
    diff = best_n / 2.0 - s
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5_exact(best_h)
    if final_c5 > best_c5 + 1e-8:
        # Something went wrong, use parent
        best_h = _project(h0, target0)
        final_c5 = _c5_exact(best_h)

    return best_h, final_c5, best_n
