```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    """Compute gradient row for correlation lag k (index in full correlation)."""

            val += (1.0 - h[jd])
        jd2 = j - d
        if 0 <= jd2 < n:
            val += (-h[jd2])
```

```python
def _active_lp_direction(h, scores, K, radius):
    n = len(h)
    top_k = np.argsort(scores)[::-1][:K]
    A = _grad_rows(h, top_k)
    
    # LP: max s
    # s.t. A @ p + s <= 0  (i.e., each active correlation decreases by at least s)
    #      sum(p) = 0
    #      |p_i| <= radius, box constraints
    
    c = np.zeros(n + 1)
    c[n] = -1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-10:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, 0.0
```

```python
def _line_search(h, p, c5_cur, n_grid=80):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-14:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-10)
    
    best_alpha = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.0, alpha_max, n_grid)
    for a in alphas:
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_alpha = a
    
    # Golden section refinement around best
    if best_alpha > 0:
        lo = max(0.0, best_alpha - alpha_max / n_grid)
        hi = min(alpha_max, best_alpha + alpha_max / n_grid)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        best_lo = c1 if f1 < f2 else c2
        best_f = min(f1, f2)
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
            best_f = min(f1, f2)
            best_lo = c1 if f1 < f2 else c2
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_alpha = opt_a
    
    return best_alpha, best_c5
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _score(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_rows(h, ks):
    n = len(h)
    rows = []
    for k in ks:
        d = k - (n - 1)
        row = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                row[d:] += (1.0 - h[:m])
                row[:m] += -h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                row[:m] += (1.0 - h[g:])
                row[g:] += -h[:m]
        rows.append(row)
    return np.array(rows)


def _active_lp(h, scores, K, radius):
    n = len(h)
    top_k = np.argsort(scores)[::-1][:K]
    A = _grad_rows(h, top_k)
    c = np.zeros(n + 1)
    c[n] = -1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-10:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, 0.0


def _line_search(h, p, c5_cur, n_grid=100):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            a = (1.0 - h[j]) / p[j]
            if a < alpha_max:
                alpha_max = a
        elif p[j] < -1e-14:
            a = h[j] / (-p[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-10)
    best_alpha = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.0, alpha_max, n_grid)
    for a in alphas:
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_alpha = a
    if best_alpha > 0 and alpha_max > 1e-8:
        span = alpha_max / n_grid
        lo = max(0.0, best_alpha - span)
        hi = min(alpha_max, best_alpha + span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(35):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_alpha = opt_a
    return best_alpha, best_c5


def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _score(h) * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        wk = w[k]
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wk * (1.0 - h[:m])
                grad[:m] += -wk * h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wk * (1.0 - h[g:])
                grad[g:] += -wk * h[:m]
    return grad


def _subgrad_optimize(h, deadline, tau=60.0, lr=0.02):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    beta = 0.85
    stall = 0
    while time.time() < deadline:
        grad = _soft_grad(h, tau)
        mom = beta * mom + (1.0 - beta) * grad
        h_new = _project(h - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.12)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
            if stall % 15 == 0:
                rng = np.random.default_rng(int(time.time() * 1000) % 100000)
                h = best_h + rng.normal(0, 0.03, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5


def _lp_optimize(h, deadline, K=45, radius_init=0.15, n_grid=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    stall2 = 0
    while time.time() < deadline:
        if stall > 200:
            break
        scores = _score(h)
        p, s_val = _active_lp(h, scores, K, radius)
        if p is None or s_val < 1e-11:
            stall += 1
            radius *= 0.65
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, n_grid)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            stall2 = 0
            radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            stall2 += 1
            radius *= 0.7
            if stall2 % 20 == 0 and radius < 0.015:
                radius = 0.06
    return best_h, best_c5


def _pair_descent(h, deadline, max_passes=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n - 1):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-10 or h[k] >= 1.0 - 1e-10:
                    continue
                c = h[j] + h[k]
                lo = max(0.0, c - 1.0)
                hi = min(1.0, c)
                if hi - lo < 1e-12:
                    continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                h1 = h.copy()
                h1[j] = x1
                h1[k] = c - x1
                f1 = _exact_c5(h1)
                h2 = h.copy()
                h2[j] = x2
                h2[k] = c - x2
                f2 = _exact_c5(h2)
                for _ in range(25):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = hi - gr * (hi - lo)
                        ht = h.copy()
                        ht[j] = x1
                        ht[k] = c - x1
                        f1 = _exact_c5(ht)
                    else:
                        lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = lo + gr * (hi - lo)
                        ht = h.copy()
                        ht[j] = x2
                        ht[k] = c - x2
                        f2 = _exact_c5(ht)
                opt_x = (lo + hi) / 2.0
                h_o = h.copy()
                h_o[j] = opt_x
                h_o[k] = c - opt_x
                f_o = _exact_c5(h_o)
                best_x = x1 if f1 < f2 else x2
                best_f = min(f1, f2)
                if f_o < best_f:
                    best_x = opt_x
                    best_f = f_o
                if best_f < best_c5 - 1e-13:
                    h = h.copy()
                    h[j] = best_x
                    h[k] = c - best_x
                    best_c5 = best_f
                    best_h = h.copy()
                    improved = True
                elif best_f < f1 and best_f < f2:
                    h[j] = best_x
                    h[k] = c - best_x
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _evolutionary(h_init, deadline, pop_size=14, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = _project(h_init + rng.normal(0, 0.05, n), target)
        pop.append(h_i)
    fits = [_exact_c5(h) for h in pop]
    best_idx = int(np.argmin(fits))
    best_h = pop[best_idx].copy()
    best_f = fits[best_idx]
    mut_scale = 0.04
    while time.time() < deadline:
        i1 = int(rng.integers(0, pop_size))
        i2 = int(rng.integers(0, pop_size))
        if i1 == i2:
            i2 = (i2 + 1) % pop_size
        child = 0.5 * (pop[i1] + pop[i2]) + rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.95)
            else:
                mut_scale = min(0.10, mut_scale * 1.05)
        else:
            mut_scale = min(0.10, mut_scale * 1.03)
    return best_h, best_f


def _random_dir_search(h, deadline, n_tries=50, scale=0.1):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(int(time.time()) % 100003)
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        p = rng.normal(0, 1, n)
        p -= p.mean()
        pn = float(np.max(np.abs(p)))
        if pn < 1e-10:
            continue
        p = p / pn * scale
        alpha, new_c5 = _line_search(h, p, cur_c5, 60)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 80
        x = (np.arange(n0) + 0.5) / n0
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Subgradient warmup
    d1 = t0 + budget_s * 0.04
    h1, c51 = _subgrad_optimize(best_h, d1, tau=80.0, lr=0.015)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Main LP optimization
    d2 = t0 + budget_s * 0.38
    h2, c52 = _lp_optimize(best_h, d2, K=45, radius_init=0.15, n_grid=100)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    # Phase 3: Continue LP with different K
    if time.time() < t0 + budget_s * 0.50:
        d3 = t0 + budget_s * 0.50
        h3, c53 = _lp_optimize(best_h, d3, K=60, radius_init=0.10, n_grid=80)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

    # Phase 4: Multi-resolution
    d4 = t0 + budget_s * 0.68
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, 18, -18, 24, -24]
    tried_n = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.022)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, K=40, radius_init=0.12, n_grid=80)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Perturbation + re-optimize
    d5 = t0 + budget_s * 0.82
    scale = 0.04
    for attempt in range(100):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, K=40, radius_init=0.08, n_grid=70)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.006, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.2)

    # Phase 6: Evolutionary
    if time.time() < t0 + budget_s * 0.90:
        d6 = t0 + budget_s * 0.90
        h_evo, c5_evo = _evolutionary(best_h, d6, pop_size=12, seed=seed + 13)
        if c5_evo < best_c5 - 1e-11:
            best_c5 = c5_evo
            best_h = h_evo.copy()

    # Phase 7: Pair descent
    if time.time() < t0 + budget_s * 0.95:
        d7 = t0 + budget_s * 0.95
        h_pd, c5_pd = _pair_descent(best_h, d7, max_passes=80)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()

    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        d8 = t0 + budget_s * 0.99
        h_f, c5_f = _lp_optimize(best_h, d8, K=80, radius_init=0.03, n_grid=120)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
