    return np.correlate(h, 1.0 - h, mode='full')

def compute_score_gradient(h, weights):

    Compute gradient of weighted sum of scores wrt h.
    weights: vector of length 2n-1, corresponding to np.correlate output order.
    Output r corresponds to shift k = r - (n-1).

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

_GS = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-15:
            continue
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[k:]
        else:
            g = -k
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    return grad

def _solve_c(zc, n):
    target = n / 2.0
    lo = -15.0
    hi = 15.0
    # Ensure f(lo) < 0 and f(hi) > 0
    # f(c) = sum(expit(zc + c)) - target
    # Since zc is centered (mean 0), c=0 gives sum ≈ n/2
    # So we just need a wide enough bracket
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(expit(zc + mid))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def _h_from_z(z, n):
    zc = z - np.mean(z)
    c = _solve_c(zc, n)
    h = expit(zc + c)
    return h

def _z_from_h(h, n):
    h_clip = np.clip(h, 1e-8, 1.0 - 1e-8)
    z = np.log(h_clip) - np.log(1.0 - h_clip)
    return z - np.mean(z)

def _smooth_eval(z):
    n = _GS['n']
    beta = _GS['beta']
    zc = z - np.mean(z)
    c = _solve_c(zc, n)
    h = expit(zc + c)
    scores = _scores(h)
    L = len(scores)
    M = np.max(scores)
    shifted = beta * (scores - M)
    max_sh = np.max(shifted)
    log_sum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    val = M + (log_sum - np.log(L)) / beta
    # Gradient
    exps = np.exp(shifted - max_sh)
    weights = exps / np.sum(exps)
    G = _score_grad(h, weights)
    a = h * (1.0 - h)
    A = np.sum(a)
    grad_z = a * (G - np.sum(G * a) / A)
    return val, grad_z

def _lbfgs_run(z0, n, beta, maxiter=200, time_limit=None):
    _GS['n'] = n
    _GS['beta'] = beta
    try:
        res = minimize(_smooth_eval, z0, method='L-BFGS-B',
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                                          'maxfun': maxiter * 2})
        return res.x
    except Exception:
        return z0

def _beta_schedule(z, n, check_time):
    """Run the full beta schedule on z."""
    betas = [20, 80, 300, 1000, 3000, 8000]
    z_cur = z.copy()
    for beta in betas:
        if check_time():
            break
        z_cur = _lbfgs_run(z_cur, n, beta, maxiter=150)
    return z_cur

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new // 2)

def _thue_morse_z(n, amp, rng):
    """Generate z from Thue-Morse pattern."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = 0.5 + amp * (2.0 * tm - 1.0)
    h = _project(h, n // 2)
