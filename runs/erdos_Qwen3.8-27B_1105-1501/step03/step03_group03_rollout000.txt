def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scores_and_grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] -= h[d:]
                G[r, d:] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    scores = np.correlate(h, 1.0 - h, mode='full')
    return scores, G

def _project(v, target):
    # Water-filling projection to {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, target, deadline, max_passes=50):

    n = len(h)

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):

        improved = False

        # Try pairs that are both interior
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        
        for j in interior:
            if time.time() > deadline:
                break
            for k in interior:
                if k <= j:
                    continue
                # Try transferring mass from j to k and vice versa
                for (dj, dk) in [(0.01, -0.01), (-0.01, 0.01), (0.02, -0.02), (-0.02, 0.02)]:
                    if h[j] + dj < 0 or h[j] + dj > 1:

                    h_t = h.copy()
                    h_t[j] += dj

                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        h = h_t.copy()
                        improved = True
                        break
    return h, best_c5

def _fourier_perturb(h, target, rng, n_modes=5, amplitude=0.03):
