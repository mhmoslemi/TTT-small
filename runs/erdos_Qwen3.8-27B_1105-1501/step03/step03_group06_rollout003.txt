def _slsqp_epigraph(h_init, target, deadline, active_tol=0.01):
    """Solve the epigraph problem using SLSQP."""
    n = len(h_init)
    T_init = float(np.max(_correlate_full(h_init)))
    x0 = np.concatenate([[T_init], h_init])
    
    # Precompute gradient structure
    # g_d for d in -(n-1), ..., n-1
    lags = np.arange(-(n-1), n)
    K = 2*n - 1
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[0] = 1.0
        return g
    
    def make_constraint(d):
        def cons(x):
            h = x[1:]
            return x[0] - _g_i(h, d)
        def cons_grad(x):
            h = x[1:]
            g = np.zeros(n+1)
            g[0] = 1.0
            g[1:] = -_gradient_g_i(h, d)
            return g
        return (cons, cons_grad)
    
    # Identify active constraints
    g_vals = _correlate_full(h_init)
    T_max = float(np.max(g_vals))
    active = [i for i in range(K) if g_vals[i] >= T_max - active_tol]
    
    # Build constraint list
    constraints = []
    for idx in active:
        d = int(lags[idx])
        pairs = make_constraint(d)
        constraints.append({'type': 'ineq', 'fun': pairs[0], 'jac': pairs[1]})
    
    # Equality constraint
    def eq_cons(x):
        return x[1:].sum() - target
    def eq_grad(x):
        g = np.zeros(n+1)
        g[1:] = 1.0
        return g
    constraints.append({'type': 'eq', 'fun': eq_cons, 'jac': eq_grad})
    
    bounds = [(T_init - 0.1, None)] + [(0.0, 1.0)] * n
    
    # Solve
    res = minimize(obj, x0, jac=obj_grad, constraints=constraints,
                   bounds=bounds, method='SLSQP',
                   options={'maxiter': 200, 'ftol': 1e-10})
    
    if res.success or res.fun < T_init:
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    return h_init, _exact_c5(h_init)
