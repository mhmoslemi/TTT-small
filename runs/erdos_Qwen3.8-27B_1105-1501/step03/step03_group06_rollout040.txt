def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = float(np.sum(x)) - target
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.argsort(-x)
            rem = res
            for i in idx:
                dec = min(x[i], rem)
                x[i] -= dec
                rem -= dec
                if rem < 1e-12:
                    break
        else:
            idx = np.argsort(x)
            rem = -res
            for i in idx:
                inc = min(1.0 - x[i], rem)
                x[i] += inc
                rem -= inc
                if rem < 1e-12:
                    break
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        # dC_s/dh[j] = (1-h[j*s]) * 1{0<=j+s<n} - h[j-s] * 1{0<=j-s<n}
        # First term: j+s in [0, n-1], i.e., j in [-s, n-1-s] intersect [0, n-1]
        j1_lo = max(0, -s)
        j1_hi = min(n, n - s)
        if j1_hi > j1_lo:
            G[r, j1_lo:j1_hi] += (1.0 - h[j1_lo:j1_hi + s])
        # Second term: j-s in [0, n-1], i.e., j in [s, n-1+s] intersect [0, n-1]
        j2_lo = max(0, s)
        j2_hi = min(n, n + s)
        if j2_hi > j2_lo:
            G[r, j2_lo:j2_hi] += -h[j2_lo - s : j2_hi - s]
    return G

    x = _project(x, target)
    return x, _exact_c5(x)

```python
def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        # First term
        j1_lo = max(0, -s)
        j1_hi = min(n, n - s)
        if j1_hi > j1_lo:
            G[r, j1_lo:j1_hi] += (1.0 - h[j1_lo:j1_hi + s])
        # Second term
        j2_lo = max(0, s)
        j2_hi = min(n, n + s)
        if j2_hi > j2_lo:
            G[r, j2_lo:j2_hi] += -h[j2_lo - s : j2_hi - s]
    return G
```

This is a loop over 2n-1 = 127 iterations, each doing O(n) work. Total: O(n^2) ≈ 4096 operations. Very fast.

Alternative vectorized version:
For a fixed s, the first term is a shifted version of (1-h), and the second term is a shifted version of -h. I could use FFT or direct shifting, but the loop is already fast enough for n=64.

def _slsqp_reduced(h0, deadline, maxiter=200, k_active=20):
    n = len(h0)
    target = n / 2.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    max0 = float(np.max(corr0))
    
    # Select the k_active constraints closest to max
    active = np.argsort(corr0)[-k_active:]
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr[active]
    
    def ineq_jac(x):
        h = x[:-1]
        G = _build_G(h)
        G_active = G[active, :]
        J = np.zeros((k_active, n+1))
        J[:, :n] = -G_active
        J[:, n] = 1.0
        return J
    
    def eq(x):
        return np.sum(x[:-1]) - target
    
    def eq_jac(x):
        J = np.zeros(n+1)
        J[:n] = 1.0
        return J
    
    x0 = np.concatenate([h0, [max0 + 0.1]])
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
    ]
    bounds = [(0, 1)] * n + [(0, None)]
    
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    for _ in range(3):  # Try 3 times with expanding active set
        if time.time() > deadline:
            break
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': 0})
            if res.success:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    x0 = res.x.copy()
                else:
                    break
            else:
                break
        except Exception:
            break
        
        # Update active set
        corr_new = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_active = min(len(corr_new), k_active + 10)
        active = np.argsort(corr_new)[-k_active:]
    
    return best_h, best_c5
