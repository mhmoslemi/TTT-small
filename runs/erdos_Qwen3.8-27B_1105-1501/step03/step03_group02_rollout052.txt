import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0: return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0: return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_epigraph(h_init, deadline, top_k=40, maxiter=150):
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    z0 = float(np.max(scores)) + 1e-4
    idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in idx]

    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 5.0)]

    def obj(x):
        return x[n]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for d in active_lags:
        cons.append({'type': 'ineq',
                     'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
                     'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])})

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return None, None
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-4:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _smooth_grad_step(h, scores, beta=60.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    d = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi > 1e-15:
            d += wi * _score_grad(h, s)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return -d / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)

                best_c5 = min(best_c5, c5_new)
                best_h = h.copy()
            break

        # Check for new active constraints
        new_scores = _all_scores(h)
        new_max = float(np.max(new_scores))
        old_constraints = set(active_lags)
        for r in np.argsort(new_scores)[::-1][:top_k]:
            d = int(r - (n - 1))
            if d not in old_constraints and new_scores[r] > new_max - 0.005:
                active_lags.append(d)
                old_constraints.add(d)

    return best_h, best_c5

def _gradient_descent(h, deadline, lr_init=0.02, n_max=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0

    for it in range(n_max):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        scores = _all_scores(h)
        direction = _smooth_grad_step(h, scores, beta=80.0)
        if direction is None:
            break
        # Line search
        best_alpha = 0.0
        best_try_c5 = cur_c5
        for alpha in [lr * 0.1, lr * 0.3, lr * 0.5, lr * 0.7, lr, lr * 1.5]:
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_try_c5:
                best_try_c5 = c5_try
                best_alpha = alpha
        if best_alpha > 0 and best_try_c5 < cur_c5 - 1e-13:
            h = h + best_alpha * direction
            h = _project(h, target)
            cur_c5 = best_try_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if lr < 1e-6:
                break
    return best_h, best_c5

def _binary_search_short(n, deadline, rng, max_iters=50000):
    """Quick binary sequence search for a good seed"""
    target = n // 2
    h = np.zeros(n)
    ones = list(range(target))
    zeros = list(range(target, n))
    h[ones] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, len(ones))
        j = rng.integers(0, len(zeros))
        a = ones[i]
        b = zeros[j]
        # Swap: h[a]=1->0, h[b]=0->1
        h[a] = 0.0
        h[b] = 1.0
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            # Accept
        else:
            # Revert
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP iterative with active set
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _slsqp_iterate(h, dl1, top_k=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient descent
    dl2 = t0 + budget_s * 0.30
    h2, c5_2 = _gradient_descent(best_h, dl2, lr_init=0.015, n_max=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP again on improved point
    dl3 = t0 + budget_s * 0.42
    h3, c5_3 = _slsqp_iterate(best_h, dl3, top_k=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pairwise optimization (gradient-guided)
    dl4 = t0 + budget_s * 0.70
    h4, c5_4 = _pairwise_round(best_h, dl4, max_rounds=100, n_grid=25, n_golden=15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: SLSQP polish
    dl5 = t0 + budget_s * 0.78
    h5, c5_5 = _slsqp_iterate(best_h, dl5, top_k=25)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pairwise all-pairs (more thorough)
    dl6 = t0 + budget_s * 0.88
    h6, c5_6 = _pairwise_round_all_pairs(best_h, dl6, n_grid=20, n_golden=12)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Try nearby resolutions
    dl7 = t0 + budget_s * 0.95
    for dn in [0, -2, 2, -4, 4, -6, 6]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.003:
            continue
        # Quick SLSQP on new resolution
        h_r2, c5_r2 = _slsqp_iterate(h_r, min(dl7, time.time() + 30), top_k=30)
        if c5_r2 < best_c5 - 1e-8:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try
        elif c5_r < best_c5 - 1e-8:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final pairwise polish
    dl8 = t0 + budget_s * 0.99
    h8, c5_8 = _pairwise_round(best_h, dl8, max_rounds=30, n_grid=30, n_golden=20)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
        best_n = len(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_target = len(best_h) / 2.0
    best_h = _project(best_h, final_target)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-5:
        # Projection slightly degraded; use pre-projection if feasible enough
        best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
