import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_opt(h, i, j):
    """Find best t for h[i]+=t, h[j]-=t minimizing max correlation score."""
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi - 1e-13:
        return 0.0, _max_corr(h)
    
    # Coarse grid search
    n_c = 24
    ts = np.linspace(t_lo, t_hi, n_c)
    best_val = _max_corr(h)
    best_t = 0.0
    hc = h.copy()
    for t in ts:
        hc[i] = h[i] + t
        hc[j] = h[j] - t
        val = _max_corr(hc)
        if val < best_val:
            best_val = val
            best_t = t
    
    if abs(best_t) < 1e-14:
        return 0.0, best_val
    
    # Golden section refinement
    span = (t_hi - t_lo) / n_c * 2.5
    lo = max(t_lo, best_t - span)
    hi = min(t_hi, best_t + span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    hc[i] = h[i] + c1; hc[j] = h[j] - c1
    f1 = _max_corr(hc)
    hc[i] = h[i] + c2; hc[j] = h[j] - c2
    f2 = _max_corr(hc)
    
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            hc[i] = h[i] + c1; hc[j] = h[j] - c1
            f1 = _max_corr(hc)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            hc[i] = h[i] + c2; hc[j] = h[j] - c2
            f2 = _max_corr(hc)
    
    t_opt = (lo + hi) / 2.0
    hc[i] = h[i] + t_opt; hc[j] = h[j] - t_opt
    f_opt = _max_corr(hc)
    
    if f_opt < best_val:
        return t_opt, f_opt
    return best_t, best_val

def _two_pair_opt(h, i, j, k, l):
    """Optimize h[i]+=t1, h[j]-=t1, h[k]+=t2, h[l]-=t2."""

    ts2 = np.linspace(t2_lo, t2_hi, n2)

    best_t1, best_t2 = 0.0, 0.0
    hc = h.copy()
    
    for t1 in ts1:
        for t2 in ts2:
            hc[i] = h[i] + t1
            hc[j] = h[j] - t1
            hc[k] = h[k] + t2
            hc[l] = h[l] - t2
            val = _max_corr(hc)
            if val < best_val:
                best_val = val
                best_t1, best_t2 = t1, t2
    
    if abs(best_t1) < 1e-14 and abs(best_t2) < 1e-14:
        return 0.0, 0.0, best_val
    
    # Coordinate descent refinement
    for _cd in range(8):
        # Optimize t1 with t2 fixed
        t1_lo2 = max(t1_lo, -h[i], h[j] - 1.0 - best_t2 + best_t2)  # just use original bounds

        hc[l] = h[l]

    return best_t1, best_t2, best_val

def _smooth_gradient_direction(h, beta=50.0):
    """Compute smoothed max gradient direction (negative, sum-zero)."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    z_max = float(np.max(scores))
    A = beta * (scores - z_max) / (z_max + 1e-15)
    A = A - np.max(A)
    W = np.exp(A)
    W = W / np.sum(W)
    
    # Compute gradient of each score wrt h
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if W[r] < 1e-10:
            continue
        d = r - (n - 1)
        if d == 0:
            grad[:] += W[r] * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += W[r] * (-h[d:d+m])
            grad[d:] += W[r] * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            grad[:m] += W[r] * (1.0 - h[g:g+m])
            grad[g:] += W[r] * (-h[:m])
    
    grad -= np.mean(grad)
    norm = np.max(np.abs(grad))
    if norm < 1e-15:
        return None
    return -grad / norm

def _direction_search(h, v, max_steps=50):
    """Line search along direction v (sum-zero) to minimize max corr."""
    n = len(h)
    # Determine feasible alpha range
    alpha_max = 1.0
    for i in range(n):
        if v[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-v[i]))
    
    # Also check negative direction
    alpha_min = -alpha_max  # by symmetry of the feasibility (not exact, but good enough)
    for i in range(n):
        if v[i] > 1e-14:
            alpha_min = max(alpha_min, -h[i] / v[i])
        elif v[i] < -1e-14:
            alpha_min = max(alpha_min, (1.0 - h[i]) / v[i])
    
    best_val = _max_corr(h)
    best_a = 0.0
    hc = h.copy()
    
    # Grid search
    n_grid = 20
    alphas = np.linspace(alpha_min, alpha_max, n_grid)
    for a in alphas:
        if abs(a) < 1e-12:
            continue
        hc[:] = h
        hc += a * v
        hc = np.clip(hc, 0.0, 1.0)
        val = _max_corr(hc)
        if val < best_val:
            best_val = val
            best_a = a
    
    if abs(best_a) < 1e-12:
        return 0.0, best_val
    
    # Golden section around best
    span = (alpha_max - alpha_min) / n_grid * 2.0
    lo = max(alpha_min, best_a - span)
    hi = min(alpha_max, best_a + span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    hc[:] = h; hc += c1 * v; hc = np.clip(hc, 0.0, 1.0)
    f1 = _max_corr(hc)
    hc[:] = h; hc += c2 * v; hc = np.clip(hc, 0.0, 1.0)
    f2 = _max_corr(hc)
    
    for _ in range(35):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            hc[:] = h; hc += c1 * v; hc = np.clip(hc, 0.0, 1.0)
            f1 = _max_corr(hc)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            hc[:] = h; hc += c2 * v; hc = np.clip(hc, 0.0, 1.0)
            f2 = _max_corr(hc)
    
    a_opt = (lo + hi) / 2.0
    hc[:] = h; hc += a_opt * v; hc = np.clip(hc, 0.0, 1.0)
    f_opt = _max_corr(hc)
    
    if f_opt < best_val:
        return a_opt, f_opt
    return best_a, best_val

def _greedy_sweep(h, deadline, use_two_pair=False):
    """One greedy sweep: try all pairs, apply best single (or double) move."""
    n = len(h)
    base_val = _max_corr(h)
    best_improve = 0.0
    best_move = None
    
    # Build pair list, shuffle for diversity
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    np.random.shuffle(pairs)
    
    for (i, j) in pairs:
        if time.time() > deadline:
            break
        t, val = _pair_opt(h, i, j)
        if val < base_val - 1e-11:
            improve = base_val - val
            if improve > best_improve:
                best_improve = improve
                best_move = ('single', i, j, t)
    
    if use_two_pair and best_move is not None:
        # Try two-pair moves with the best pair and another
        if time.time() < deadline:
            _, i1, j1, _ = best_move
            candidates = [(i1, j1, i, j) for i in range(n) for j in range(n)
                         if 0 < abs(i-j) and (i,j) != (i1,j1) and (i,j) != (j1,i1)]
            np.random.shuffle(candidates)
            for (i1, j1, i, j) in candidates[:32]:
                if time.time() > deadline:
                    break
                t1, t2, val = _two_pair_opt(h, i1, j1, i, j)
                if val < base_val - 1e-11:
                    improve = base_val - val
                    if improve > best_improve:
                        best_improve = improve
                        best_move = ('double', i1, j1, i, j, t1, t2)
    
    if best_move is not None:
        if best_move[0] == 'single':
            _, i, j, t = best_move
            h[i] += t
            h[j] -= t
        else:
            _, i1, j1, i, j, t1, t2 = best_move
            h[i1] += t1
            h[j1] -= t1
            h[i] += t2
            h[j] -= t2
    
    return h, best_improve

def _local_search(h, deadline, max_sweeps=200):
    """Iterative greedy pair sweeps with occasional direction search."""
    n = len(h)
    cur_val = _max_corr(h)
    best_val = cur_val
    best_h = h.copy()
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        # Every 5th sweep, try a direction search
        if sweep % 5 == 3:
            v = _smooth_gradient_direction(h, beta=30.0)
            if v is not None:
                a, val = _direction_search(h, v)
                if val < cur_val - 1e-11:
                    h = h + a * v
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, n / 2.0)
                    cur_val = _max_corr(h)
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()
                    stall = 0
                    continue
        
        h, improve = _greedy_sweep(h, deadline, use_two_pair=(sweep % 3 == 0))
        if improve > 1e-11:
            h = _project(h, n / 2.0)
            cur_val = _max_corr(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_val * (2.0 / n)

def _antipodal_seed(n, rng):
    """Generate an antipodal seed: h = [y, 1-y] with random y."""
    m = n // 2
    y = rng.uniform(0, 1, m)
    # Add some structure
    x = np.arange(m) / m
    y = 0.5 + 0.3 * np.sin(2 * np.pi * x * (1 + rng.integers(1, 4))) + rng.normal(0, 0.1, m)
    y = np.clip(y, 0, 1)
    h = np.concatenate([y, 1.0 - y])
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
