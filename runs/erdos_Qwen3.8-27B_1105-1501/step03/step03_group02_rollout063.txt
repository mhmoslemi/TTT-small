    return float(np.max(corr) * dx)

def _all_scores(h):

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:

def _linearize_lp(h, top_k, radius):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    
    # Build LP: min z
    # Variables: h[0..n-1], z (total n+1 variables)
    # Constraints:
    #   For each active lag r: z - sum_j grad_r[j] * (h[j] - h0[j]) >= scores[r]
    #   => z - sum_j grad_r[j] * h[j] >= scores[r] - sum_j grad_r[j] * h0[j]
    #   => sum_j grad_r[j] * h[j] - z <= -scores[r] + sum_j grad_r[j] * h0[j]
    #   sum(h) = n/2
    #   0 <= h[j] <= 1
    #   |h[j] - h0[j]| <= radius (box constraint around h0)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize z
    
    A_ub = np.zeros((top_k, n + 1))

    bounds = []

    bounds.append((None, None))  # z is free

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:d+m] += 0.0
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _all_score_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        G[r] = _score_grad(h, d)
    return G

def _smooth_max_obj_grad(h, beta):
    n = len(h)
    S = _all_scores(h)
    max_S = float(np.max(S))
    A = beta * (S - max_S)
    A = A - float(np.max(A))
    W = np.exp(A)
    Wsum = float(np.sum(W))
    W = W / Wsum
    G = _all_score_grads(h)
    g = G.T @ W
    obj = max_S + float(np.log(Wsum)) / beta
    return obj, g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_antipodal(y, n, perm_type):
    m = n // 2
    h = np.empty(n)
    h[:m] = y
    if perm_type == 0:
        h[m:] = 1.0 - y
    elif perm_type == 1:
        h[m:] = 1.0 - y[::-1]
    elif perm_type == 2:
        h[m:] = 1.0 - np.roll(y, 1)
    elif perm_type == 3:
        h[m:] = 1.0 - np.roll(y, m // 2)
    return h

def _extract_y(h, perm_type, n):
    m = n // 2
    if perm_type == 0:
        return h[:m].copy()
    elif perm_type == 1:
        return (1.0 - h[m:][::-1]).copy()
    elif perm_type == 2:
        return (1.0 - np.roll(h[m:], -1)).copy()
    elif perm_type == 3:
        return (1.0 - np.roll(h[m:], -m // 2)).copy()
    return h[:m].copy()

def _antipodal_obj_func(y, beta, perm_type, n):
    h = _make_antipodal(y, n, perm_type)
    obj, _ = _smooth_max_obj_grad(h, beta)
    return obj

def _antipodal_jac_func(y, beta, perm_type, n):
    h = _make_antipodal(y, n, perm_type)
    _, g_h = _smooth_max_obj_grad(h, beta)
    m = n // 2
    if perm_type == 0:
        g_y = g_h[:m] - g_h[m:]
    elif perm_type == 1:
        g_y = g_h[:m] - g_h[m:][::-1]
    elif perm_type == 2:
        g_y = g_h[:m] - np.roll(g_h[m:], -1)
    elif perm_type == 3:
        g_y = g_h[:m] - np.roll(g_h[m:], -m // 2)
    return g_y

    m = n // 2
    y_init = np.clip(y_init, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * m
    try:
        result = minimize(_antipodal_obj_func, y_init,
                          args=(beta, perm_type, n),
                          jac=_antipodal_jac_func, bounds=bounds,
                          method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxfun': 500, 'ftol': 1e-14})
        y_out = np.clip(result.x, 0.0, 1.0)
        h_out = _make_antipodal(y_out, n, perm_type)
        c5 = _exact_c5(h_out)
        return h_out, c5, y_out
    except Exception:
        return None, None, y_init

def _proj_gradient(h_init, beta, deadline, lr_init=0.05):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 100:
            break
        obj, g = _smooth_max_obj_grad(h, beta)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = h - alpha * g
            h_try = _project(h_try, target)
            obj_try, _ = _smooth_max_obj_grad(h_try, beta)
            if obj_try < obj - 1e-14:
                improved = True
                break
            alpha *= 0.6
        
        if improved:
            h = _project(h - alpha * g, target)
            c5_try = _exact_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def _lp_direction_corrected(h, top_k, radius):
    n = len(h)
    target = n / 2.0
    scores = _all_scores(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    G = _all_score_grads(h)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(top_idx), n + 1))
    b_ub = np.zeros(len(top_idx))
    for i, r in enumerate(top_idx):
        A_ub[i, :n] = G[r]
        A_ub[i, n] = -1.0
        b_ub[i] = float(G[r] @ h) - scores[r]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = []
    for j in range(n):
        lo_b = max(0.0, h[j] - radius)
        hi_b = min(1.0, h[j] + radius)
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_trust_region(h_init, deadline, radius_init=0.1, top_k=25):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 80:
            break
        p = _lp_direction_corrected(h, top_k, radius)
        if p is None:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        
        diff = p - h
        if float(np.max(np.abs(diff))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        
        best_alpha = 0.0
        best_val = cur_c5
        for a in [0.3, 0.6, 1.0, 0.15, 0.8, 1.5, 2.0]:
            h_try = h + a * diff
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = a
        
        if best_alpha > 0 and best_val < cur_c5 - 1e-13:
            h = _project(h + best_alpha * diff, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_smooth(h_init, deadline, top_k=15, beta=50):
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    z0 = float(np.max(scores)) + 1e-4
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]
    
    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 3.0)]
    
    cons = [{'type': 'eq', 'fun': _slsqp_sum_eq, 'jac': _slsqp_sum_jac, 'args': (n, target)}]
    for d in active_lags:
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac, 'args': (d, x0[n])})
    
    _slsqp_state = (n, target, deadline)
    
    try:
        result = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None, None
    
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _slsqp_obj(x):
    return x[len(x) - 1]

def _slsqp_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def _slsqp_sum_eq(x, n, target):
    return float(np.sum(x[:n]) - target)

def _slsqp_sum_jac(x, n, target):
    g = np.zeros(len(x))
    g[:n] = 1.0
    return g

def _slsqp_ineq(x, d, z_ref):
    h = x[:len(x)-1]
    n = len(h)
    scores = _all_scores(h)
    r = n - 1 + d
    if 0 <= r < len(scores):
        return x[n:]
    return z_ref

def _slsqp_ineq_jac(x, d, z_ref):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _pair_swap_search(h_init, deadline, max_rounds=5000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rng = np.random.default_rng(123)
    
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        delta = h[j] - h[i]
        if abs(delta) < 1e-10:
            continue
        for t in [0.5, 1.0, 0.25]:
            new_di = t * delta
            new_dj = -t * delta
            hi_new = h[i] + new_di
            hj_new = h[j] + new_dj
            if hi_new < -1e-12 or hi_new > 1.0 + 1e-12:
                continue
            if hj_new < -1e-12 or hj_new > 1.0 + 1e-12:
                continue
            h_try = h.copy()
            h_try[i] = np.clip(hi_new, 0.0, 1.0)
            h_try[j] = np.clip(hj_new, 0.0, 1.0)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
    
    return best_h, best_c5

def _multi_beta_projected_gradient(h_init, deadline):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    betas = [10, 25, 50, 100, 200, 400]
    
    for beta in betas:
        if time.time() > deadline:
            break
        phase_time = min(30.0, (deadline - time.time()) * 0.4)
        h_try, c5_try = _proj_gradient(best_h, beta, time.time() + phase_time, lr_init=0.1)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
