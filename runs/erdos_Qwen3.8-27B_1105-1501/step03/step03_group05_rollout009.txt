Given h (n-dimensional) and direction v (n-dimensional, sum=0):
- A[r] = correlate(h, 1-h)[r] for r = 0..2n-2
- B[r] = correlate(v, 1-h)[r] - correlate(h, v)[r]
- C[r] = -correlate(v, v)[r]

Then score_r(alpha) = A[r] + alpha*B[r] + alpha²*C[r]

We want to minimize f(alpha) = max_r score_r(alpha) over feasible alpha.

Feasible alpha: for each i with v[i] ≠ 0:
- If v[i] > 0: alpha ∈ [-h[i]/v[i], (1-h[i])/v[i]]
- If v[i] < 0: alpha ∈ [(1-h[i])/v[i], -h[i]/v[i]]

Take intersection of all intervals, also include 0.

Candidate alpha values:
1. 0
2. Left and right endpoints
3. For each r with C[r] ≠ 0: alpha* = -B[r]/(2*C[r]) if in interval
4. For each pair (i,j): solve (C[i]-C[j])α² + (B[i]-B[j])α + (A[i]-A[j]) = 0

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
            if not mask.any():
                break
            caps = (1.0 - v)[mask]
            total = caps.sum()
            if total <= 0:
                break
            v[mask] += caps * (diff / total)
        else:
            mask = v > 1e-15
            if not mask.any():
                break
            excess = v[mask]
            total = excess.sum()
            if total <= 0:
                break
            v[mask] -= excess * (-diff / total)
    v = np.clip(v, 0.0, 1.0)
    # Final correction
    residual = target - v.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            m = v < 1.0
        else:
            m = v > 0.0
        if m.any():
            v[m] += residual / m.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _build_G(h):
    """Build the gradient matrix G where G[r, i] = d(S_r)/d(h[i])."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        lo = max(0, -d)
        hi = min(n, n - d)
        for p in range(lo, hi):
            q = p + d
            G[r, q] += 1.0 - h[p]
            G[r, p] -= h[q]
    return G

def _line_search_exact(h, v, raw_A):
    """Exact 1D line search. Returns (alpha, c5) or None."""
    n = len(h)
    v = v - v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:
        return None
    v = v / vn  # Normalize
    
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    # Feasible interval
    a_lo = 0.0
    a_hi = 0.0
    pos = v > 1e-14
    neg = v < -1e-14
    if pos.any():
        a_lo = max(a_lo, float(-np.min(h[pos] / v[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if neg.any():
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / v[neg])))  # (1-h)/v < 0 since v<0
        a_hi = min(a_hi, float(np.min(-h[neg] / v[neg])))  # -h/v > 0 since v<0
    a_lo = min(a_lo, 0.0)
    a_hi = max(a_hi, 0.0)
    if a_hi - a_lo < 1e-12:
        return None
    
    cur_max_raw = float(np.max(raw_A))
    
    # Collect candidates
    cands = [a_lo, a_hi, 0.0]
    
    # Stationary points
    mask_c = np.abs(C) > 1e-10
    if mask_c.any():
        a_stars = -B[mask_c] / (2.0 * C[mask_c])
        valid = (a_stars >= a_lo) & (a_stars <= a_hi)
        cands.extend(a_stars[valid].tolist())
    
    # Pairwise intersections
    m = len(raw_A)
    if m <= 130:
        # Use vectorized approach for speed
        # Only consider pairs where both C are non-negligible or at least one B is non-zero
        for i in range(m):
            dC = C[i] - C
            dB = B[i] - B
            dA = raw_A[i] - raw_A
            # Skip i itself
            dC[i] = 0.0  # Will be skipped by j > i filter
            for j in range(i + 1, m):
                if abs(dC[j]) < 1e-12 and abs(dB[j]) < 1e-12:
                    continue
                if abs(dC[j]) > 1e-12:
                    disc = dB[j] ** 2 - 4.0 * dC[j] * dA[j]
                    if disc < -1e-10:
                        continue
                    if disc < 0:
                        disc = 0.0
                    sq = np.sqrt(disc)
                    a1 = (-dB[j] + sq) / (2.0 * dC[j])
                    a2 = (-dB[j] - sq) / (2.0 * dC[j])
                    if a_lo <= a1 <= a_hi:
                        cands.append(float(a1))
                    if a_lo <= a2 <= a_hi:
                        cands.append(float(a2))
                else:
                    a1 = -dA[j] / dB[j]
                    if a_lo <= a1 <= a_hi:
                        cands.append(float(a1))
    
    # Add grid
    n_grid = max(30, int(20 * (a_hi - a_lo)))
    n_grid = min(n_grid, 200)
    cands.extend(np.linspace(a_lo, a_hi, n_grid).tolist())

    # Vectorized evaluation
    scores_mat = raw_A + cands_arr[:, None] * B + (cands_arr ** 2)[:, None] * C
    max_scores = scores_mat.max(axis=1)
    best_idx = int(np.argmin(max_scores))
    best_a = float(cands_arr[best_idx])

    if c5_new < c5_cur - 1e-12:
        return (best_a, c5_new)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
            if not mask.any():
                break
            caps = (1.0 - v)[mask]
            total = caps.sum()
            if total <= 0:
                break
            v[mask] += caps * (diff / total)
        else:
            mask = v > 1e-15
            if not mask.any():
                break
            excess = v[mask]
            total = excess.sum()
            if total <= 0:
                break
            v[mask] -= excess * (-diff / total)
    v = np.clip(v, 0.0, 1.0)
    residual = target - v.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            m = v < 1.0
        else:
            m = v > 0.0
        if m.any():
            v[m] += residual / m.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        lo = max(0, -d)
        hi = min(n, n - d)
        for p in range(lo, hi):
            q = p + d
            G[r, q] += 1.0 - h[p]
            G[r, p] -= h[q]
    return G

def _line_search_exact(h, v, raw_A):
    n = len(h)
    v = v - v.mean()
    vn = float(np.max(np.abs(v)))
    if vn < 1e-14:
        return None
    v = v / vn
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    a_lo = 0.0
    a_hi = 0.0
    pos = v > 1e-14
    neg = v < -1e-14
    if pos.any():
        a_lo = max(a_lo, float(-np.min(h[pos] / v[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if neg.any():
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / v[neg])))
        a_hi = min(a_hi, float(np.min(-h[neg] / v[neg])))
    a_lo = min(a_lo, 0.0)
    a_hi = max(a_hi, 0.0)
    if a_hi - a_lo < 1e-12:
        return None
    cur_max_raw = float(np.max(raw_A))
    cands = [a_lo, a_hi, 0.0]
    mask_c = np.abs(C) > 1e-10
    if mask_c.any():
        a_stars = -B[mask_c] / (2.0 * C[mask_c])
        valid = (a_stars >= a_lo) & (a_stars <= a_hi)
        cands.extend(a_stars[valid].tolist())
    m = len(raw_A)
    if m <= 140:
        idx_i, idx_j = np.triu_indices(m, 1)
        dC = C[idx_i] - C[idx_j]
        dB = B[idx_i] - B[idx_j]
        dA = raw_A[idx_i] - raw_A[idx_j]
        valid_quad = np.abs(dC) > 1e-12
        valid_lin = (~valid_quad) & (np.abs(dB) > 1e-12)
        roots = []
        if valid_quad.any():
            disc = dB[valid_quad] ** 2 - 4.0 * dC[valid_quad] * dA[valid_quad]
            ok = disc >= -1e-10
            if ok.any():
                dcl = np.maximum(disc[ok], 0.0)
                sq = np.sqrt(dcl)
                dv = dB[valid_quad][ok]
                dc = dC[valid_quad][ok]
                roots.append((-dv + sq) / (2.0 * dc))
                roots.append((-dv - sq) / (2.0 * dc))
        if valid_lin.any():
            roots.append(-dA[valid_lin] / dB[valid_lin])
        if roots:
            all_r = np.concatenate(roots)
            all_r = all_r[(all_r >= a_lo) & (all_r <= a_hi)]
            cands.extend(all_r.tolist())
    n_grid = min(80, max(20, int(5 * (a_hi - a_lo) + 40)))
    cands.extend(np.linspace(a_lo, a_hi, n_grid).tolist())
    cands_arr = np.array(cands, dtype=float)
    cands_arr = np.clip(cands_arr, a_lo + 1e-11, a_hi - 1e-11)
    cands_arr = np.unique(np.round(cands_arr, 9))
    if len(cands_arr) > 5000:
        cands_arr = cands_arr[np.random.default_rng(0).choice(len(cands_arr), 5000, replace=False)]
        cands_arr.sort()
    scores_mat = raw_A + cands_arr[:, None] * B + (cands_arr ** 2)[:, None] * C
    max_scores = scores_mat.max(axis=1)
    best_idx = int(np.argmin(max_scores))
    best_a = float(cands_arr[best_idx])
    best_raw = float(max_scores[best_idx])
    if best_raw >= cur_max_raw - 1e-13:
        return None
    h_new = h + best_a * v
    h_new = _project_box_sum(h_new, n / 2.0)
    c5_new = _exact_c5(h_new)
    c5_cur = cur_max_raw * (2.0 / n)
    if c5_new < c5_cur - 1e-12:
        return (best_a, c5_new)
    return None

def _smooth_dir(G, raw_A, beta):
    z = float(np.max(raw_A))
    w = np.exp(beta * (raw_A - z))
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(raw_A))
        w[np.argmax(raw_A)] = 1.0
    else:
        w = w / wsum
    g = G.T @ w
    v = -g
    v = v - v.mean()
    return v

def _topk_dir(G, raw_A, k):
    n_scores = len(raw_A)
    top_idx = np.argsort(raw_A)[-k:]
    w = np.zeros(n_scores)
    w[top_idx] = 1.0 / k
    g = G.T @ w
    v = -g
    v = v - v.mean()
    return v

def _random_dir(rng, n):
    v = rng.standard_normal(n)
    v = v - v.mean()
    return v

def _pair_dir(n, i, j):
    v = np.zeros(n)
    v[i] = 1.0
    v[j] = -1.0
    return v

def _pair_polish(h, deadline, max_passes=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for.pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        raw_A = _raw_scores(h)
        # Try all pairs, prioritizing those with non-extreme values
        order = np.argsort(-np.abs(h - 0.5))
        for idx in order:
            if time.time() > deadline:
                break
            i = int(idx)
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for j in range(n):
                if j == i:
                    continue
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                v = _pair_dir(n, i, j)
                res = _line_search_exact(h, v, raw_A)
                if res is not None:
                    a, c5_new = res
                    h = h + a * v
                    h = _project_box_sum(h, target)
                    c5 = _exact_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    improved = True
                    raw_A = _raw_scores(h)
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _transform(h, kind, rng):
    n = len(h)
    if kind == 0:
        return h[::-1].copy()
    elif kind == 1:
        roll = rng.integers(1, n)
        return np.roll(h, int(roll)).copy()
    elif kind == 2:
        # Binary threshold
        idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[idx[:n // 2]] = 1.0
        return h_bin
    elif kind == 3:
        # Perturb
        v = rng.standard_normal(n) * 0.1
        v = v - v.mean()
        h_p = h + v
        return _project_box_sum(h_p, n / 2.0)
    else:
        return h.copy()
