    m2a = (d >= (j - (n - 1))) & (d <= j)
    if m2a.any():
        c[m2a] -= (1.0 - h[j - d[m2a]])
    m2b = (d >= -j) & (d <= n - 1 - j)
    if m2b.any():
        c[m2b] += h[j + d[m2b]]
    return c


def _pair_optimal_delta(S, c, lo, hi):
    if hi - lo < 1e-12:
        return 0.0
    eps_c = 1e-14
    pos_mask = c > eps_c
    neg_mask = c < -eps_c
    zero_mask = ~(pos_mask | neg_mask)
    max_abs_c = float(np.max(np.abs(c)))
    span = max(abs(lo), abs(hi))
    v_lo = float(np.min(S)) - max_abs_c * span - 0.1
    v_hi = float(np.max(S)) + max_abs_c * span + 0.1
    S_pos = S[pos_mask]
    S_neg = S[neg_mask]
    c_pos = c[pos_mask]
    c_neg = c[neg_mask]
    for _ in range(60):
        if v_hi - v_lo < 1e-11:
            break
        v_mid = (v_lo + v_hi) / 2.0
        if zero_mask.any() and float(np.max(S[zero_mask])) > v_mid + 1e-12:
            v_lo = v_mid
            continue
        fe_lo = lo
        fe_hi = hi
        if pos_mask.any():
            fe_hi = min(fe_hi, float(np.min((v_mid - S_pos) / c_pos)))
        if neg_mask.any():
            fe_lo = max(fe_lo, float(np.max((v_mid - S_neg) / c_neg)))
        if fe_lo <= fe_hi + 1e-12:
            v_hi = v_mid
        else:
            v_lo = v_mid
    v = v_hi
    fe_lo = lo
    fe_hi = hi
    if pos_mask.any():
        fe_hi = min(fe_hi, float(np.min((v - S_pos) / c_pos)))
    if neg_mask.any():
        fe_lo = max(fe_lo, float(np.max((v - S_neg) / c_neg)))
    delta_opt = (fe_lo + fe_hi) / 2.0
    delta_opt = max(lo, min(hi, delta_opt))
    return delta_opt


def _pair_optimize(h, deadline, max_passes=500):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    no_improve_streak = 0

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if no_improve_streak > 3:
            break
        pass_improved = False
        S = _scores(h)

        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                if h[i] < 1e-10 or h[j] < 1e-10:
                    continue
                c = _pair_coeff_vec(h, i, j)
                lo = -h[i]
                hi = h[j]
                if hi - lo < 1e-10:
                    continue
                delta = _pair_optimal_delta(S, c, lo, hi)
                if abs(delta) < 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                new_c5 = _exact_c5(h)
                if new_c5 < cur_c5 - 5e-14:
                    cur_c5 = new_c5
                    S = _scores(h)
                    pass_improved = True
                else:
                    h[i] -= delta
                    h[j] += delta

        if pass_improved:
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve_streak = 0
        else:
            no_improve_streak += 1

    return best_h, best_c5


def _smooth_obj(x, beta):
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _smooth_jac(x, beta):
    n = len(x)
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    d = np.arange(-(n - 1), n)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        dd = r - (n - 1)
        if dd >= 0:
            m = n - dd
            if m > 0:
                G[r, dd:] += (1.0 - x[:m])
                G[r, :m] -= x[dd:]
        else:
            g = -dd
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - x[g:])
                G[r, g:] -= x[:m]
    return G.T @ w


def _sum_fun(x, target):
    return np.sum(x) - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _slsqp_phase(h, target, deadline, betas=None):
    if betas is None:
        betas = [10.0, 30.0, 80.0]
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    for beta in betas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, best_h, jac=_smooth_jac, args=(beta,),
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 150, 'ftol': 1e-12})
            if res.x is not None:
                h_try = _project(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        except Exception:
            pass
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _penalty_direction(h, S, K=8, mu=1.0, gamma=5.0, n_iter=50):
    n = len(h)
    s_max = float(np.max(S))
    slacks = s_max - S
    order = np.argsort(slacks)
    active = order[:K]
    s_min = float(np.min(S))
    width = max(s_max - s_min, 1e-6)
    weights = np.zeros(2 * n - 1)
    for r in active:
        weights[r] = max(0.1, 1.0 - slacks[r] / (0.5 * width))
    d = np.arange(-(n - 1), n)
    G = np.zeros((2 * n - 1, n))
    for r in active:
        dd = r - (n - 1)
        if dd >= 0:
            m = n - dd
            if m > 0:
                G[r, dd:] += (1.0 - h[:m])
                G[r, :m] -= h[dd:]
        else:
            g = -dd
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    G_a = G[active, :]
    w = weights[active]
    allowed = (s_max - slacks[active]) - 0.02 * width
    p = np.zeros(n)
    for _ in range(n_iter):
        Ap = G_a @ p
        v = np.maximum(0.0, Ap - allowed)
        grad = mu * p + gamma * (G_a.T @ (w * v))
        grad = grad - np.mean(grad)
        nrm = np.linalg.norm(grad)
        if nrm < 1e-12:
            break
        step = 1.0 / (mu + gamma * np.max(w) * np.max(np.sum(G_a ** 2, axis=1)))
        step = min(step * 3.0, 0.5 / max(nrm, 1e-8))
        p_new = p - step * grad
        p_new = p_new - np.mean(p_new)
        if np.linalg.norm(p_new) > 1.0:
            p_new = p_new / np.linalg.norm(p_new)
        p = p_new
    p = p - np.mean(p)
    return p
