def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    """Gradient of Q_d with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Fix rounding
    res = target - sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    # Softmax weights
    max_s = float(np.max(scores))
    exp_s = np.exp(beta * (scores - max_s))

    for i, d in enumerate(range(-(n-1), n)):
        if w[i] > 1e-15:
            g += w[i] * _score_grad(h, d)
    return g

    target = n / 2.0

    best_h = h.copy()

    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:

def _slsqp_solve(h_init, deadline, top_k=30, maxiter=150, ftol=1e-11):
    """SLSQP epigraph optimization with active set constraints."""
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    z0 = float(np.max(scores)) + 1e-4

    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]

    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 3.0)]

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    }]
    for d in active_lags:
        cons.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
            'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        })

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                          callback=_cb)
    except RuntimeError:
        return None
    except Exception:
        return None

    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(np.sum(h_out) - target) > 1e-4:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    for top_k in [40, 25, 15, 10, 6]:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / 4.0)
        result = _slsqp_solve(best_h, sub_dl, top_k=top_k, maxiter=120, ftol=1e-11)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction(h, scores, radius):
    """Find LP descent direction that reduces all active scores."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        G[r] = _score_grad(h, d)

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_line_search(h, p, cur_c5, deadline):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    best_a = 0.0
    best_c5 = cur_c5
    for a in np.linspace(0.01, alpha_max, 15):
        if time.time() > deadline:
            break
        h_t = np.clip(h + a * p, 0.0, 1.0)
        h_t = _project(h_t, n / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, cur_c5

    # Golden section refine
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    h1 = _project(np.clip(h + c1 * p, 0, 1), n / 2.0)
    h2 = _project(np.clip(h + c2 * p, 0, 1), n / 2.0)
    f1 = _exact_c5(h1)
    f2 = _exact_c5(h2)
    for _ in range(15):
        if hi_a - lo_a < 1e-11 or time.time() > deadline:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(np.clip(h + c1 * p, 0, 1), n / 2.0))
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(np.clip(h + c2 * p, 0, 1), n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(np.clip(h + opt_a * p, 0, 1), n / 2.0))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop(h_init, deadline, radius_init=0.1, max_iters=2000):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break

        scores = _all_scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue

        alpha, new_c5 = _lp_line_search(h, p, cur_c5, deadline)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(np.clip(h + alpha * p, 0.0, 1.0), n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth gradient descent (primary new method)
    dl1 = t0 + budget_s * 0.22
    h1, c5_1 = _pgd_smooth(h, dl1, beta_start=5.0, beta_end=60.0, max_iters=6000, momentum=0.6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pairwise coordinate descent
    dl2 = t0 + budget_s * 0.35
    h2, c5_2 = _pair_cd(best_h, dl2, max_rounds=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP multi-width
    dl3 = t0 + budget_s * 0.50
    h3, c5_3 = _slsqp_multi(best_h, dl3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP loop
    dl4 = t0 + budget_s * 0.62
    h4, c5_4 = _lp_loop(best_h, dl4, radius_init=0.08, max_iters=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-restart with perturbation
    dl5 = t0 + budget_s * 0.82
    scale = 0.04
    attempts = 0
    while time.time() < dl5:
        attempts += 1
        if best_c5 < 0.38075:
            break
        scale = 0.04 * (0.93 ** min(attempts, 40))
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)

        per_dl = min(dl5, time.time() + max(4.0, (dl5 - time.time()) * 0.25))

        # Short PGD on perturbed
        h_try, c5_try = _pgd_smooth(h_pert, per_dl * 0.4, beta_start=10, beta_end=50,
                                     max_iters=2000, momentum=0.5)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.8)
        else:
            # Try SLSQP
            h_s, c5_s = _slsqp_solve(h_pert, per_dl * 0.7, top_k=25, maxiter=60, ftol=1e-10)
            if h_s is not None and c5_s < best_c5 - 1e-12:
                best_c5 = c5_s
                best_h = h_s.copy()
                scale = max(0.01, scale * 0.8)
            else:
                scale = min(0.12, scale * 1.1)

    # Phase 6: Final refinement - alternating PGD and SLSQP
    dl6 = t0 + budget_s * 0.94
    for _round in range(10):
        if time.time() > dl6:
            break
        sub_dl = min(dl6, time.time() + 15.0)
        h_a, c5_a = _pgd_smooth(best_h, sub_dl, beta_start=20, beta_end=100,
                                 max_iters=2000, momentum=0.3)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
        sub_dl2 = min(dl6, time.time() + 15.0)
        h_b, c5_b = _slsqp_solve(best_h, sub_dl2, top_k=12, maxiter=200, ftol=1e-12)
        if h_b is not None and c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Pair CD polish
        sub_dl3 = min(dl6, time.time() + 10.0)
        h_c, c5_c = _pair_cd(best_h, sub_dl3, max_rounds=80)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 7: Resolution sweep
    dl7 = t0 + budget_s * 0.98
    for dn in [2, -2, 4, -4, 6, -6, 8, -8]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 140 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + 20.0)
        h_r, c5_r = _slsqp_solve(h_r, sub_dl, top_k=25, maxiter=80, ftol=1e-10)
        if h_r is not None and c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Final LP polish
    if time.time() < t0 + budget_s * 0.995:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.998, radius_init=0.02, max_iters=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
