def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _project(v, target):
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(free))
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

    rng = np.random.default_rng(123)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
                continue
            candidates = [j for j in range(n) if j != i and 1e-9 < h[j] < 1.0 - 1e-9]
            if not candidates:
                break
            if len(candidates) > 8:
                candidates = list(rng.choice(candidates, 8, replace=False))
            for j in candidates:
                for delta in [0.01, 0.03, 0.08, 0.15]:
                    for sign in [1.0, -1.0]:
                        d = sign * delta
                        if h[i] + d < 0 or h[i] + d > 1:
                            continue
                        if h[j] - d < 0 or h[j] - d > 1:
                            continue
                        h_try = h.copy()
                        h_try[i] += d
                        h_try[j] -= d
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    elapsed = lambda: time.time() - t0
    remaining = lambda: max(0.0, budget_s - elapsed())

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Soft descent from parent (adaptive time)
    deadline1 = t0 + budget_s * 0.22
    h, c5 = _soft_descent(h, deadline1, beta_start=20.0, beta_end=4000.0,
                          max_iters=80000, momentum=0.85, mode='softmax')
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 1b: Switch to top-k mode if still improving
    if elapsed() < budget_s * 0.30:
        deadline1b = t0 + budget_s * 0.30
        h, c5 = _soft_descent(h, deadline1b, beta_start=50.0, beta_end=5000.0,
                              max_iters=50000, momentum=0.8, mode='topk')
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: LP polish
    deadline2 = t0 + budget_s * 0.40
    h, c5 = _lp_polish(h, deadline2, radius_init=0.08, max_iters=4000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Multi-n exploration
    deadline3 = t0 + budget_s * 0.72
    n_list = sorted(set([max(32, best_n + dn) for dn in range(-12, 13, 4)]))
    n_list = [n for n in n_list if 32 <= n <= 128]
    for n_try in n_list:
        if time.time() > deadline3 or remaining() < 30:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        t_sub_descent = min(deadline3, time.time() + budget_s * 0.025)
        h_r, c5_r = _soft_descent(h_r, t_sub_descent, beta_start=30.0,
                                  beta_end=5000.0, max_iters=30000, momentum=0.85)
        t_sub_lp = min(deadline3, time.time() + max(5.0, budget_s * 0.012))
        h_r, c5_r = _lp_polish(h_r, t_sub_lp, radius_init=0.05, max_iters=1500)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Restart cycles with structured perturbations
    deadline4 = t0 + budget_s * 0.88
    scale = 0.04
    for attempt in range(200):
        if time.time() > deadline4 or remaining() < 50:
            break
        if best_c5 < 0.3806:
            break
        h_p = _perturb(best_h, rng, scale)
        time_per = max(8.0, (deadline4 - time.time()) / max(200 - attempt, 1))
        time_per = min(time_per, budget_s * 0.03)
        h_t, c5_t = _soft_descent(h_p, time.time() + time_per * 0.6,
                                  beta_start=15.0, beta_end=3000.0,
                                  max_iters=20000, momentum=0.85)
        if time.time() < deadline4:
            h_t, c5_t = _lp_polish(h_t, time.time() + time_per * 0.35,
                                    radius_init=0.04, max_iters=800)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.15)

    # Phase 5: Final LP polish (small radius for fine-tuning)
    if remaining() > 10:
        deadline5 = t0 + budget_s * 0.96
        h_f, c5_f = _lp_polish(best_h, deadline5, radius_init=0.03, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Fine coordinate descent
    if remaining() > 8:
        deadline6 = t0 + budget_s * 0.995
        h_cd, c5_cd = _finer_coord(best_h, deadline6, max_passes=200)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
