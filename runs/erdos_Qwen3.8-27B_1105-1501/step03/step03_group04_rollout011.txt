def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _logit_from_h(h):

    h_safe = np.clip(h, eps, 1.0 - eps)
    z = np.log(h_safe) - np.log(1.0 - h_safe)
    return z - z.mean()

def _compute_grad_h(h, weights):
    """Compute gradient of weighted sum of scores w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for r in range(len(weights)):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:d+m])
                grad[d:d+m] -= w * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[g:]
    return grad

    # Stable log-sum-exp
    shifted = scores - m
    lse = m + (1.0/beta) * np.log(np.sum(np.exp(beta * shifted)))
    return lse

def _objective_z_grad(z, beta):
    """Log-sum-exp smoothed objective and gradient in z-space."""

    # Gradient of weighted scores w.r.t. h
    grad_h = _compute_grad_h(h, weights)

    grad_z = grad_z - grad_z.mean()
    
    return lse, grad_z

    # Phase 7: Binary flip + coordinate descent (7%)
    t7 = t0 + budget_s * 0.95
    h7, c5_7 = _binary_flip_search(best_h, t0 + min(t7, time.time() + (t7 - time.time()) * 0.5), rng, max_iters=5000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7b, c5_7b = _coord_descent_r(best_h, t7, rng, max_iters=40000)
    if c5_7b < best_c5:
        best_c5 = c5_7b
        best_h = h7b.copy()

    # Final validation and projection
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    # Fallback if something went wrong
    if not np.isfinite(final_c5) or final_c5 > 0.45:
        best_h = _project(h0.copy(), n0 / 2.0)
        final_c5 = _exact_c5(best_h)
        best_n = n0

    return best_h, final_c5, best_n
