        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Phase 8: Final micro-polish
    if time.time() < t0 + budget_s * 0.995:
        df = t0 + budget_s * 0.995
        hf, cf = _pnorm_descent(best_h, df, beta_start=200.0, beta_end=10000.0,
                                 n_steps=500, momentum=0.70, alpha_init=0.005)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
