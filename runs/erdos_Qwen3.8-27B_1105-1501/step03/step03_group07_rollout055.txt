import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _offset_grads(h):
    """Compute gradients of all lag correlations. Returns array of shape (2n-1, n)"""
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(-(n-1), n):
        r = n - 1 + k
        if k == 0:
            grads[r] = 1.0 - 2.0 * h
        elif k > 0:
            m = n - k
            grads[r, :m] += (1.0 - h[k:])  # h[j+k] term
            grads[r, k:] -= h[:m]            # h[j] term
        else:
            k2 = -k
            m = n - k2
            grads[r, k2:] += (1.0 - h[:m])   # h[j] term
            grads[r, :m] -= h[k2:]           # h[j+k2] term
    return grads

def _lp_direction(h, target, R, top_k=None, lam=0.0):
    """Solve LP to find descent direction. Returns direction d or None."""
    n = len(h)
    raw = _raw_corrs(h)
    M = 2 * n - 1
    
    # Select which lags to include
    if top_k is not None and top_k < M:
        idx_sel = np.argsort(-raw)[:top_k]
    else:
        idx_sel = np.arange(M)
    
    # Get gradients for selected lags
    # Instead of computing all, compute per selected lag
    # Variable structure: [t, p_0, ..., p_{n-1}, q_0, ..., q_{n-1}]
    # Total: 1 + 2n
    nvar = 1 + 2 * n
    
    # Objective: min t + lam * sum(p+q)
    c = np.zeros(nvar)
    c[0] = 1.0
    c[1:] = lam
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each selected lag r: -t + sum_j grad[r,j]*p_j - sum_j grad[r,j]*q_j <= -F_r
    # sum(p) + sum(q) <= R
    n_constr = len(idx_sel) + 1
    A_ub = np.zeros((n_constr, nvar))
    b_ub = np.zeros(n_constr)
    
    for i, r in enumerate(idx_sel):
        A_ub[i, 0] = -1.0
        # grad[r, j] for p_j
        k = r - (n - 1)
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            m = n - k
            g = np.zeros(n)
            g[:m] = (1.0 - h[k:])
            g[k:] -= h[:m]
        else:
            k2 = -k
            m = n - k2
            g = np.zeros(n)
            g[k2:] = (1.0 - h[:m])
            g[:m] -= h[k2:]
        A_ub[i, 1:1+n] = g       # p coefficients
        A_ub[i, 1+n:] = -g       # q coefficients
        b_ub[i] = -raw[r]
    
    # L1 norm constraint: sum(p) + sum(q) <= R
    A_ub[-1, 1:] = 1.0
    b_ub[-1] = R
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for j in range(n):
        if h[j] <= 1e-10:
            bounds.append((0, None))  # p_j free
            bounds.append((0, 0))     # q_j = 0
        elif h[j] >= 1.0 - 1e-10:
            bounds.append((0, 0))     # p_j = 0
            bounds.append((0, None))  # q_j free
        else:
            bounds.append((0, None))
            bounds.append((0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='revised_simplex')
        except Exception:
            return None
    
    if not res.success:
        return None
    
    x = res.x
    p = x[1:1+n]
    q = x[1+n:1+2*n]
    d = p - q
    d -= np.mean(d)  # ensure zero sum
    
    # Check if direction is meaningful
    if np.max(np.abs(d)) < 1e-12:
        return None
    
    return d

def _line_search(h, d, target, c5_cur, max_alpha=2.0, n_coarse=15):
    """Line search along direction d from h"""
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(len(h)):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = min(max(alpha_max, 1e-8), 5.0)
    
    if alpha_max < 1e-8:
        return 0.0, c5_cur
    
    best_a = 0.0
    best_c = c5_cur
    
    # Coarse search
    alphas = np.linspace(0.01 * alpha_max, alpha_max, n_coarse)
    for a in alphas:
        h_t = _project(h + a * d, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_a - best_a * 0.4)
    hi = min(alpha_max, best_a + best_a * 0.4)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, target))
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * d, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

def _optimize_lp(h, deadline, target, R_init=0.1, max_iters=10000):
    """Main LP-based optimization"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    R = R_init * n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        # Get LP direction
        d = _lp_direction(h, target, R, top_k=None, lam=0.01 * n)
        
        if d is None:
            stall += 5
            continue
        
        # Normalize
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            stall += 1
            continue
        d = d / d_norm * 0.5  # scale direction
        
        # Line search
        a, new_c5 = _line_search(h, d, target, cur_c5, max_alpha=1.0)
        
        if a > 0 and new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * d, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Adapt R
            R = min(R * 1.1, 2.0 * n)
        else:
            stall += 1
            R *= 0.8
            # Try both signs
            if stall % 10 == 0:
                a2, c2 = _line_search(h, -d, target, cur_c5, max_alpha=0.5)
                if a2 > 0 and c2 < cur_c5 - 1e-13:
                    h = _project(h + a2 * (-d), target)
                    cur_c5 = _exact_c5(h)
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
    
    return best_h, best_c5

def _pair_descent(h, deadline, target, max_passes=30):
    """Pairwise coordinate descent"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs (or random subset for large n)
        if n <= 80:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            pairs = [(rng.integers(0, n), rng.integers(0, n)) for _ in range(n*5)]
        
        for (i, j) in pairs:
            if time.time() > deadline:
                return best_h, best_c5
            if i == j:
                continue
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                a_max_i = (1.0 - h[i]) / di if di > 0 else h[i] / (-di)
                a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                a_max = min(a_max_i, a_max_j)
                if a_max < 1e-10:
                    continue
                # Try a few fractions
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[i] += di * a
                    h_t[j] += dj * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-12:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_lp(h0, deadline, target, rng, n_starts=10):

    n = len(h0)

        if remaining < 5.0:

        sub_dl = time.time() + alloc
        
        r = rng.random()
        if r < 0.3:
            h_try = h0 + rng.normal(0, 0.05, n)
        elif r < 0.5:
            h_try = h0.copy()

        h_try = _project(h_try, target)

        if c5_opt < best_c5 - 1e-12:

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(np.nan_to_num(v), 0.0, 1.0)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, k):
    """Gradient of correlation at signed offset k."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        m = n - k
        g[:m] = (1.0 - h[k:])
        g[k:] -= h[:m]
    else:
        m2 = -k
        m = n - m2
        g[m2:] = (1.0 - h[:m])
        g[:m] -= h[m2:]
    return g

def _lp_direction(h, R, top_k=None):
    """Solve tangent LP. Returns normalized direction or None."""
    n = len(h)
    raw = _raw_corrs(h)
    M = 2 * n - 1

    if top_k is not None and top_k < M:
        idx_sel = np.argsort(-raw)[:top_k]
    else:
        idx_sel = np.arange(M)

    nvar = 1 + 2 * n
    c = np.zeros(nvar)
    c[0] = 1.0

    n_ineq = len(idx_sel) + 1
    A_ub = np.zeros((n_ineq, nvar))
    b_ub = np.zeros(n_ineq)

    for i, r in enumerate(idx_sel):
        k = int(r) - (n - 1)
        g = _grad_for_lag(h, k)
        A_ub[i, 0] = -1.0
        A_ub[i, 1:1 + n] = g
        A_ub[i, 1 + n:] = -g
        b_ub[i] = -raw[r]

    A_ub[-1, 1:] = 1.0
    b_ub[-1] = R

    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:] = -1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)]
    for j in range(n):
        if h[j] <= 1e-10:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif h[j] >= 1.0 - 1e-10:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    except Exception:
        return None

        return None

    x = np.nan_to_num(res.x)

    d -= np.mean(d)
    dn = float(np.linalg.norm(d, 1))

def _line_search(h, d, target, c5_cur, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = min(max(alpha_max, 1e-8), 3.0)
    if alpha_max < 1e-8:

    best_a, best_c = 0.0, c5_cur

        if c_t < best_c - 1e-14:

    if best_a == 0.0:

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_a - best_a * 0.4)
    hi = min(alpha_max, best_a + best_a * 0.4)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))

            f2 = _exact_c5(_project(h + c2 * d, target))

    opt_c = _exact_c5(_project(h + opt_a * d, target))

            break
        if stall > 60:
            break

        d = None
        for tk in [None, 8, 3, 1]:
            d = _lp_direction(h, R, top_k=tk)
            if d is not None:
                break
        if d is None:
            stall += 5
            R *= 0.7
            continue

        a, new_c5 = _line_search(h, d, target, cur_c5, max_alpha=1.5)
        if a > 0 and new_c5 < cur_c5 - 1e-13:
            h = _project(h + a * d, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            R = min(R * 1.05, 2.0 * n)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
        else:
            stall += 1
            R *= 0.85
            if stall > 20:
                a2, c2 = _line_search(h, -d, target, cur_c5, max_alpha=0.5)
                if a2 > 0 and c2 < cur_c5 - 1e-13:
                    h = _project(h + a2 * (-d), target)
                    cur_c5 = _exact_c5(h)
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()

    return best_h, best_c5

def _pair_cd(h, deadline, target, max_passes=40):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    rng = np.random.default_rng(77)

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if n <= 72:
            idx = list(range(n))
        else:
            idx = rng.choice(n, size=min(n, 36), replace=False).tolist()

        for i in idx:
            if time.time() > deadline:
                return best_h, best_c5
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for j in idx:
                if j == i or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = min((1.0 - h[i]) / di if di > 0 else h[i] / (-di),
                             (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj))
                    if am < 1e-10:
                        continue
                    for frac in [0.33, 0.66, 1.0]:
                        a = am * frac
                        ht = h.copy()
                        ht[i] += di * a
                        ht[j] += dj * a
                        ct = _exact_c5(ht)
                        if ct < best_c5 - 1e-12:
                            best_c5, best_h, h, cur_c5 = ct, ht.copy(), ht.copy(), ct
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def _binary_polish(h, deadline, target):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    rng = np.random.default_rng(42)

    while time.time() < deadline:
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) == 0:
            break
        order = interior[np.argsort(np.abs(h[interior] - 0.5))[::-1]][:8]

        for j in order:
            if time.time() > deadline:
                return best_h, best_c5
            for tv in [0.0, 1.0]:
                delta = tv - h[j]
                if abs(delta) < 1e-10:
                    continue
                cands = np.where((h > 1e-8) if delta > 0 else (h < 1.0 - 1e-8))[0]
                cands = cands[cands != j]
                if len(cands) == 0:
                    continue
                for k in cands[:4]:
                    hk_new = h[k] - delta
                    if hk_new < -1e-10 or hk_new > 1.0 + 1e-10:
                        continue
                    ht = h.copy()
                    ht[j] = tv
                    ht[k] = hk_new
                    ct = _exact_c5(ht)
                    if ct < best_c5 - 1e-12:
                        best_c5, best_h, h, cur_c5 = ct, ht.copy(), ht.copy(), ct
                        improved = True
                        break

        if not improved:
            for _ in range(30):
                if time.time() > deadline:
                    return best_h, best_c5
                j = rng.integers(0, n)
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                tv = 0.0 if h[j] < 0.5 else 1.0
                delta = tv - h[j]
                if abs(delta) < 1e-10:
                    continue
                k = rng.integers(0, n)
                if k == j:
                    continue
                if h[k] - delta < -1e-10 or h[k] - delta > 1.0 + 1e-10:
                    continue
                ht = h.copy()
                ht[j] = tv
                ht[k] -= delta
                ct = _exact_c5(ht)
                if ct < best_c5 - 1e-12:
                    best_c5, best_h, h, cur_c5 = ct, ht.copy(), ht.copy(), ct
                    improved = True
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _multi_start(h0, deadline, target, rng, n_starts=10):
    n = len(h0)
    best_h, best_c5 = h0.copy(), _exact_c5(h0)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 5.0:
            break
        alloc = min(rem / max(1, n_starts - s) * 0.8, 25.0)
        sub_dl = time.time() + alloc
        r = rng.random()
        if r < 0.25:
            ht = h0 + rng.normal(0, 0.06, n)
        elif r < 0.45:
            ht = h0.copy()
            idx = rng.choice(n, size=max(4, n // 6), replace=False)
            ht[idx] += rng.normal(0, 0.1, len(idx))
        elif r < 0.65:
            ht = (h0 + rng.normal(0, 0.07, n) > 0.5).astype(float)
        elif r < 0.8:
            ht = np.roll(h0, rng.integers(1, max(2, n // 6)))
        else:
            hr = (rng.random(n) > 0.5).astype(float)
            ht = 0.5 * h0 + 0.5 * hr
        ht = _project(ht, target)
        ho, co = _lp_optimize(ht, sub_dl, target, R_frac=0.6, max_iters=3000)
        if co < best_c5 - 1e-12:
            best_c5, best_h = co, ho.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h, best_c5 = h.copy(), _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization (35%)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_optimize(h, dl1, target0, R_frac=1.0, max_iters=30000)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Pair descent (15%)
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _pair_cd(best_h, dl2, best_n / 2.0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-start (20%)
    dl3 = t0 + budget_s * 0.70
    h3, c5_3 = _multi_start(best_h, dl3, best_n / 2.0, rng, n_starts=12)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary polish (10%)
    dl4 = t0 + budget_s * 0.80
    h4, c5_4 = _binary_polish(best_h, dl4, best_n / 2.0)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Resolution sweep (10%)
    dl5 = t0 + budget_s * 0.90
    for dn in [-4, 4, -8, 8, -2, 2, -12, 12, -16, 16]:
        if time.time() > dl5:
            break
        nt = best_n + dn
        if nt < 16 or nt > 128 or nt % 2 != 0:
            continue
        hr = _resample(best_h, nt)
        sub_dl = min(dl5, time.time() + 12.0)
        hr, cr = _lp_optimize(hr, sub_dl, nt / 2.0, R_frac=0.7, max_iters=2000)
        hr, cr = _pair_cd(hr, min(dl5, time.time() + 4.0), nt / 2.0)
        if cr < best_c5 - 1e-9:
            best_c5, best_h, best_n = cr, hr.copy(), nt

    # Phase 6: Final LP polish (10%)
    dl6 = t0 + budget_s * 0.99
    h6, c5_6 = _lp_optimize(best_h, dl6, best_n / 2.0, R_frac=0.5, max_iters=10000)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
