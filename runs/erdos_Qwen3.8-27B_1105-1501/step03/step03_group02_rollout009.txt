**Computing A, B, C for each score r:**

The score at lag l (where l = r - (n-1)) is:
- l = 0: Q(h) = Σ h[i](1-h[i])
- l > 0: Q(h) = Σ_{i=0}^{n-l-1} h[i+l](1-h[i])
- l < 0: let g = -l, Q(h) = Σ_{i=0}^{n-g-1} h[i](1-h[i+g])

Along the line h(t) = h + t*d:
- A_r = Q(h) (current value)
- B_r = dQ/dt at t=0
- C_r = (1/2) * d²Q/dt² ... actually since Q is quadratic in h component-wise, let me just compute directly.

For l = 0:
Q(h+td) = Σ (h[i]+td[i])(1-h[i]-td[i]) = Σ h[i](1-h[i]) + t Σ d[i](1-2h[i]) - t² Σ d[i]²
So A = Σ h(i)(1-h[i]), B = Σ d[i](1-2h[i]), C = -Σ d[i]²

For l > 0, m = n-l:
Q(h+td) = Σ_{i=0}^{m-1} (h[i+l]+td[i+l])(1-h[i]-td[i])
= Σ h[i+l](1-h[i]) + t Σ (d[i+l](1-h[i]) - h[i+l]d[i]) - t² Σ d[i+l]d[i]

So A = Σ h[i+l](1-h[i]), B = Σ (d[i+l](1-h[i]) - h[i+l]d[i]), C = -Σ d[i+l]d[i]

For l < 0, g = -l, m = n-g:
Q(h+td) = Σ_{i=0}^{m-1} (h[i]+td[i])(1-h[i+g]-td[i+g])
= Σ h[i](1-h[i+g]) + t Σ (d[i](1-h[i+g]) - h[i]d[i+g]) - t² Σ d[i]d[i+g]

So A = Σ h[i](1-h[i+g]), B = Σ (d[i](1-h[i+g]) - h[i]d[i+g]), C = -Σ d[i]d[i+g]

**Finding the minimum of max of quadratics:**

The candidate t values are:
1. Endpoints: t_min, t_max
2. For each r: stationary point of Q_r(t) = -B[r]/(2*C[r]) if C[r] < 0 (which it always is since C = -Σ products of d components, and for non-zero d, at least one term is non-zero making C < 0... actually wait, C could be 0 if the relevant d components are all zero)
3. For each pair (r1, r2): roots of Q_r1(t) = Q_r2(t), i.e., (C1-C2)t² + (B1-B2)t + (A1-A2) = 0

This is O(M²) pairs where M = 2n-1 = 127 for n=64, so ~8000 pairs. For each pair, solve a quadratic. This is very fast.

Then evaluate the max at each candidate t (clipped to feasible interval) and pick the best.

**Direction generation:**

1. Smooth max gradient: weighted sum of gradients of top-k scores
2. LP direction: minimize max of linearized scores
3. Random zero-sum directions
4. Pair exchange directions

def _repair_feasible(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-12:
        return v
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _line_coeffs(h, d):
    """Compute A, B, C arrays for all scores along line h + t*d."""
    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    
    for r in range(M):
        l = r - (n - 1)  # lag
        if l == 0:
            A[r] = float(np.sum(h * (1.0 - h)))
            B[r] = float(np.sum(d * (1.0 - 2.0 * h)))
            C[r] = -float(np.sum(d * d))

            B[r] = float(np.sum(dl * (1.0 - h0) - hl * d0))
            C[r] = -float(np.sum(dl * d0))

            B[r] = float(np.sum(d0 * (1.0 - hg) - h0 * dg))

def _exact_line_search(h, d, deadline):
    """Exact 1D line search: minimize max_r (A_r + B_r*t + C_r*t^2) over feasible t."""
    n = len(h)
    t_lo, t_hi = _feasible_interval(h, d)
    if t_hi - t_lo < 1e-15:
        return 0.0, _exact_c5(h)
    
    A, B, C = _line_coeffs(h, d)
    M = len(A)
    
    # Collect candidate t values
    candidates = [t_lo, t_hi, 0.0]
    
    # Stationary points of each quadratic
    for r in range(M):
        if abs(C[r]) > 1e-15:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo <= t0 <= t_hi:
                candidates.append(t0)
    
    # Pairwise intersections
    # To limit computation, only consider pairs where both have significant C difference or B difference
    eps = 1e-12
    for i in range(M):
        if time.time() > deadline:
            break
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) > eps:
                disc = b * b - 4.0 * a * c
                if disc >= 0:
                    sq = np.sqrt(disc)
                    t1 = (-b + sq) / (2.0 * a)
                    t2 = (-b - sq) / (2.0 * a)
                    if t_lo <= t1 <= t_hi:
                        candidates.append(t1)
                    if t_lo <= t2 <= t_hi:
                        candidates.append(t2)
            elif abs(b) > eps:
                t1 = -c / b
                if t_lo <= t1 <= t_hi:
                    candidates.append(t1)
    
    # Deduplicate and clip
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= t_lo - 1e-12) & (candidates <= t_hi + 1e-12)]
    if len(candidates) == 0:
        return 0.0, _exact_c5(h)
    candidates = np.clip(candidates, t_lo, t_hi)
    candidates = np.unique(np.round(candidates, 14))
    
    # Cap candidates for performance
    if len(candidates) > 5000:
        # Keep endpoints and sample evenly
        idx = np.linspace(0, len(candidates) - 1, 5000).astype(int)
        candidates = np.concatenate([[t_lo, t_hi], candidates[idx]])
    
    # Evaluate max of quadratics at each candidate
    best_t = 0.0
    best_val = float(np.max(A))  # value at t=0
    # Evaluate in chunks to avoid memory issues
    chunk = 500
    for start in range(0, len(candidates), chunk):
        if time.time() > deadline:
            break
        t_batch = candidates[start:start+chunk]
        # Compute all quadratics at these t values
        # shape: (len(t_batch), M)
        vals = A[None, :] + B[None, :] * t_batch[:, None] + C[None, :] * t_batch[:, None] ** 2
        maxvals = vals.max(axis=1)
        min_idx = int(np.argmin(maxvals))
        if maxvals[min_idx] < best_val:
            best_val = float(maxvals[min_idx])
            best_t = float(t_batch[min_idx])
    
    return best_t, best_val  # best_val is raw (without dx)

def _smooth_max_direction(h, scores, top_k=20, beta=50.0):
    """Generate direction from smooth max of top scores."""
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    
    # Compute gradient of weighted sum of scores
    G = np.zeros(n)
    M = len(scores)
    for r in range(M):
        if w[r] < 1e-15:
            continue
        l = r - (n - 1)
        if l == 0:
            G += w[r] * (1.0 - 2.0 * h)
        elif l > 0:
            m = n - l
            if m > 0:
                G[:m] += w[r] * (-h[l:l+m])
                G[l:] += w[r] * (1.0 - h[:m])
        else:
            g = -l
            m = n - g
            if m > 0:
                G[:m] += w[r] * (1.0 - h[g:g+m])
                G[g:] += w[r] * (-h[:m])
    
    # Zero-sum
    G -= np.mean(G)
    norm = float(np.max(np.abs(G)))
    if norm < 1e-15:
        return None
    return -G / norm  # negative gradient direction

def _random_direction(rng, n):
    """Generate random zero-sum unit direction."""
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _pair_direction(rng, h, scores, n, num_pairs=10):
    """Generate pair exchange directions."""
    # Score each coordinate by its "importance" in the max scores
    # Top coordinates to move
    importance = np.zeros(n)
    M = len(scores)
    top_k = min(20, M)
    top_idx = np.argsort(scores)[::-1][:top_k]
    for r in top_idx:
        l = r - (n - 1)
        if l == 0:
            importance += np.abs(1.0 - 2.0 * h) * scores[r]
        elif l > 0:
            m = n - l
            if m > 0:
                importance[:m] += np.abs(h[l:l+m]) * scores[r]
                importance[l:] += np.abs(1.0 - h[:m]) * scores[r]
        else:
            g = -l
            m = n - g
            if m > 0:
                importance[:m] += np.abs(1.0 - h[g:g+m]) * scores[r]
                importance[g:] += np.abs(h[:m]) * scores[r]
    
    directions = []
    for _ in range(num_pairs):
        # Weighted random selection
        w = importance + 0.1
        w /= float(np.sum(w))
        i = int(rng.choice(n, p=w))
        j = int(rng.choice(n, p=w))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        # Normalize
        d /= 2.0  # so max abs is 0.5, keeping steps reasonable
        directions.append(d)
    
    return directions

def _lp_direction(h, scores, n, active_k=30):
    """LP-based direction: minimize max of linearized scores over unit ball."""
    M = len(scores)
    top_idx = np.argsort(scores)[::-1][:active_k]
    
    # Build gradient matrix for active constraints
    G = np.zeros((active_k, n))
    for idx, r in enumerate(top_idx):
        l = r - (n - 1)
        if l == 0:
            G[idx, :] = (1.0 - 2.0 * h)
        elif l > 0:
            m = n - l
            if m > 0:
                G[idx, :m] += -h[l:l+m]
                G[idx, l:] += (1.0 - h[:m])
        else:
            g = -l
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g+m])
                G[idx, g:] += -h[:m]
    
    # LP: min t s.t. G[idx,:] @ d <= t for all idx, sum(d)=0, -1<=d_i<=1
    # Variables: d[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((active_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(active_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [( -1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            d -= np.mean(d)  # ensure zero sum (LP should enforce but numerical)
            norm = float(np.max(np.abs(d)))
            if norm > 1e-12:
                return d / norm
    except Exception:
        pass
    return None

def _local_descent(h, deadline, rng, max_stall=30):
    """Local descent using exact 1D line search with diverse directions."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    iter_count = 0
    
    while time.time() < deadline:
        iter_count += 1
        if stall > max_stall:
            break
        
        scores = _scores(h)
        raw_max = float(np.max(scores))
        
        # Generate candidate directions
        directions = []
        
        # 1. Smooth max gradient
        d = _smooth_max_direction(h, scores, top_k=20, beta=50.0)
        if d is not None:
            directions.append(d)
        
        # 2. LP direction
        d = _lp_direction(h, scores, n, active_k=30)
        if d is not None:
            directions.append(d)
        
        # 3. Random directions
        for _ in range(3):
            d = _random_direction(rng, n)
            if d is not None:
                directions.append(d)
        
        # 4. Pair directions
        pairs = _pair_direction(rng, h, scores, n, num_pairs=5)
        directions.extend(pairs)
        
        improved = False
        for d in directions:
            if time.time() > deadline:
                break
            
            # Try both d and -d (though the line search should handle it)
            for dd in [d]:
                t_star, raw_new = _exact_line_search(h, dd, deadline)
                if raw_new < raw_max - 1e-10:
                    h_new = h + t_star * dd
                    h_new = _repair_feasible(h_new, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < cur_c5 - 1e-12:
                        h = h_new
                        cur_c5 = c5_new
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        stall = 0
                        break
                # Try small perturbation if exact line search didn't help
                break
        
        if not improved:
            stall += 1
            # Try a random restart with small perturbation
            if stall % 5 == 0:
                h_pert = h + rng.normal(0, 0.02, n)
                h_pert = _repair_feasible(h_pert, target)
                c5_pert = _exact_c5(h_pert)
                if c5_pert < cur_c5:
                    h = h_pert
                    cur_c5 = c5_pert
                    stall = 0
    
    return best_h, best_c5

def _smooth_max_direction(h, scores, beta=50.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))

def _local_descent(h, deadline, rng, max_stall=35, min_improve=1e-11):

            break

        improved = False

def _resample(h, n_new):

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(64, 0.5)
    n = len(h0)
    target = n / 2.0
    h = _repair_feasible(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    hard_deadline = t0 + budget_s * 0.97

    # Phase 1: Intensive local descent from parent
    dl1 = min(t0 + budget_s * 0.35, hard_deadline)
    h1, c5_1 = _local_descent(h, dl1, rng, max_stall=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    if best_c5 < 0.3808:
        return best_h, best_c5, best_n

    # Phase 2: Symmetry + perturbation exploration
    dl2 = min(t0 + budget_s * 0.55, hard_deadline)
    transforms = [1.0 - best_h, best_h[::-1], 1.0 - best_h[::-1]]
    for tr in transforms:
        if time.time() > dl2:
            break
        tr = _repair_feasible(tr, target)
        # Slight perturbation to break degeneracy
        tr_pert = tr + rng.normal(0, 0.03, n)
        tr_pert = _repair_feasible(tr_pert, target)
        per_dl = min(15.0, dl2 - time.time())
        if per_dl < 3:
            break
        h_tr, c5_tr = _local_descent(tr_pert, time.time() + per_dl, rng, max_stall=15)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
        if best_c5 < 0.3808:
            break

    # Phase 3: Diverse seeds
    dl3 = min(t0 + budget_s * 0.82, hard_deadline)
    seeds = [best_h.copy()]
    # Binary seeds from different thresholds
    for thresh in [0.4, 0.5, 0.6]:
        h_bin = (best_h > thresh).astype(float)
        h_bin = _repair_feasible(h_bin, target)
        seeds.append(h_bin)
    # Gaussian perturbations at different scales
    for sc in [0.08, 0.12, 0.20]:
        h_p = best_h + rng.normal(0, sc, n)
        h_p = _repair_feasible(h_p, target)
        seeds.append(h_p)
    # Mixed: half parent half random
    h_mix = 0.5 * best_h + 0.5 * rng.uniform(0, 1, n)
    h_mix = _repair_feasible(h_mix, target)
    seeds.append(h_mix)

    for s in seeds[1:]:
        if time.time() > dl3:
            break
        c5_s = _exact_c5(s)
        per_time = min(25.0, (dl3 - time.time()) * 0.4)
        if per_time < 3:
            break
        h_s, c5_s2 = _local_descent(s, time.time() + per_time, rng, max_stall=20)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s.copy()
        if best_c5 < 0.3808:
            break

    # Phase 4: Resolution sweep (try smaller n)
    if time.time() < hard_deadline:
        dl4 = min(hard_deadline, t0 + budget_s * 0.92)
        for n_try in [48, 32, 56, 40]:
            if time.time() > dl4:
                break
            h_r = _resample(best_h, n_try)
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5 + 0.001:
                # Worth trying to optimize at this resolution
                per_time = min(20.0, dl4 - time.time())
                if per_time < 3:
                    break
                h_ro, c5_ro = _local_descent(h_r, time.time() + per_time, rng, max_stall=25)
                if c5_ro < best_c5:
                    best_c5 = c5_ro
                    best_h = h_ro.copy()
                    best_n = n_try
            if best_c5 < 0.3808:
                break

    # Phase 5: Final intensive polish
    if time.time() < hard_deadline:
        dl5 = hard_deadline
        h5, c5_5 = _local_descent(best_h, dl5, rng, max_stall=50)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    # If repair degraded, try the pre-repair version if it was feasible
    if final_c5 > best_c5 + 1e-8:
        # Re-check: use best_c5 if the array is still valid
        s = float(np.sum(best_h))
        if abs(s - best_n / 2.0) < 1e-6:
            pass
    return best_h, final_c5, best_n
