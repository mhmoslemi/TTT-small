    if s == 0:
        return 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] += -h[s:]
    else:
        ga = -s
        m = n - ga
        if m > 0:
            g[:m] += (1.0 - h[ga:])
            g[ga:] += -h[:m]
    return g

def _smooth_max_grad(h, beta=60.0):
    scores = _scores(h)
    n = len(h)
    mx = float(np.max(scores))
    w = np.exp(beta * (scores - mx))
    ws = float(np.sum(w))
    if ws < 1e-300:
        idx = int(np.argmax(scores))
        s0 = idx - (n - 1)
        return _score_grad(h, s0)
    w /= ws
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        s = r - (n - 1)
        g += w[r] * _score_grad(h, s)
    return g

def _pair_transfer(h, i, j, deadline, max_iters=40):
    t_max = min(1.0 - h[i], h[j])
    if t_max < 1e-11:
        return 0.0
    scores0 = _scores(h)
    max0 = float(np.max(scores0))
    
    def eval_f(t):
        h2 = h.copy()
        h2[i] = h[i] + t
        h2[j] = h[j] - t
        return float(np.max(_scores(h2)))
    
    lo, hi = 0.0, t_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = eval_f(c1), eval_f(c2)
    for _ in range(max_iters):
        if hi - lo < 1e-13 or time.time() > deadline:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_f(c2)
    t_opt = (lo + hi) * 0.5
    if t_opt < 0:
        t_opt = 0.0
    if t_opt > t_max:
        t_opt = t_max
    return t_opt

def _select_pairs(h, g, n_pairs=50):
    n = len(h)
    scores = np.zeros(n * n)
    for i in range(n):
        for j in range(n):
            if i != j:
                scores[i * n + j] = g[j] - g[i]
    idx = np.argsort(scores)[::-1][:n_pairs]
    pairs = []
    for k in idx[:n_pairs]:
        i = int(k) // n
        j = int(k) % n
        if i != j:
            pairs.append((i, j))
    return pairs

def _pair_phase(h, deadline, max_rounds=200, n_pairs=40, beta=60.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > 15:
            break
        g = _smooth_max_grad(h, beta)
        pairs = _select_pairs(h, g, n_pairs)
        best_delta = 0.0
        best_pair = None
        best_t = 0.0
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            t = _pair_transfer(h, i, j, deadline, max_iters=30)
            if t < 1e-10:
                continue
            h2 = h.copy()
            h2[i] += t
            h2[j] -= t
            c5_new = _exact_c5(h2)
            if c5_new < best_c5 - 1e-13:
                best_delta = c5_new - best_c5
                best_pair = (i, j)
                best_t = t
        if best_pair is not None and abs(best_delta) > 1e-13:
            h[best_pair[0]] += best_t
            h[best_pair[1]] -= best_t
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            beta = max(beta * 0.8, 10.0)
    return best_h, best_c5

def _slsqp_solve(h_init, deadline, top_k=30, maxiter=80, ftol=1e-10):
    n = len(h_init)
    target = n / 2.0
    scores = _scores(h_init)
    z0 = float(np.max(scores)) + 1e-5
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]
    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]
    
    def obj_fn(x):
        return x[n]
    
    def obj_jac_fn(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - target,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    }]
    for d in active_lags:
        cons.append({
            'type': 'ineq',
            'fun': lambda x, dd=d: x[n] - _score_tau(x[:n], dd),
            'jac': lambda x, dd=d: np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        })
    
    def _cb_fn(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")
    
    try:
        result = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol},
                          callback=_cb_fn)
    except Exception:
        return None
    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
