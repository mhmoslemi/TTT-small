    h_t = h + t * d
    corr = np.correlate(h_t, 1.0 - h_t, mode='full')
    return float(np.max(corr) * dx)

def _line_search(h, d, cur_c5):
    """Exact 1D minimization of C5 along h + t*d for t >= 0."""
    n = len(h)
    dx = 2.0 / n
    
    # Find t_max
    t_max = np.inf
    for i in range(n):
        if d[i] > 1e-15:
            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            t_max = min(t_max, h[i] / (-d[i]))
    
    if t_max <= 1e-14:
        return 0.0, cur_c5
    
    t_max = min(t_max, 2.0)  # Safety cap
    
    # Precompute correlation coefficients
    a = np.correlate(h, 1.0 - h, mode='full')
    b = np.correlate(d, 1.0 - h, mode='full') + np.correlate(h, -d, mode='full')
    c = np.correlate(d, -d, mode='full')
    
    # Candidate points: 0, t_max, vertices of each quadratic in (0, t_max)
    candidates = [0.0, t_max]
    for k in range(len(a)):
        if abs(c[k]) > 1e-30:
            t_v = -b[k] / (2.0 * c[k])
            if 1e-12 < t_v < t_max:
                candidates.append(t_v)
    
    # Add a grid
    grid = np.linspace(0.0, t_max, 33)
    candidates.extend(grid.tolist())
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= -1e-12) & (candidates <= t_max + 1e-12)]
    candidates = np.clip(candidates, 0.0, t_max)
    
    # Evaluate at all candidates
    best_t = 0.0
    best_c5 = cur_c5
    for t in candidates:
        h_t = h + t * d
        # Verify feasibility (should be fine by construction)
        if h_t.min() < -1e-9 or h_t.max() > 1.0 + 1e-9:
            continue
        c5_t = _eval_line(h_t, d, 0.0, dx)  # Actually just evaluate directly
        if c5_t < best_c5:
            best_c5 = c5_t
            best_t = t
    
    # Golden section refinement around best_t
    if 0 < best_t < t_max:
        span = max(t_max * 0.05, 1e-6)
        lo = max(0.0, best_t - span)
        hi = min(t_max, best_t + span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _eval_line(h, d, c1, dx)
        f2 = _eval_line(h, d, c2, dx)
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _eval_line(h, d, c1, dx)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _eval_line(h, d, c2, dx)
        t_gold = (lo + hi) / 2.0
        f_gold = _eval_line(h, d, t_gold, dx)
        if f_gold < best_c5:
            best_c5 = f_gold
            best_t = t_gold
    
    return best_t, best_c5

def _smooth_gradient(h, alpha):
    """Compute smooth approximation of the subgradient of C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    w = np.exp(vals)
    w /= float(np.sum(w))
    # Gradient of each corr[k] with respect to h
    # corr[k] = sum_i h[i] * (1-h[i+lag]) where lag = k - (n-1)
    # d(corr[k])/d(h[j]) = (1-h[j+lag]) - h[j-lag] (with bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        # Contribution to each j
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) if j <= n-1-lag, minus h[j-lag] if j >= lag
            m = n - lag
            if m > 0:
                grad[:m] += w[k] * (1.0 - h[lag:m+lag])  # (1-h[j+lag]) for j in 0..m-1
                grad[lag:] += w[k] * (-h[:m])  # -h[j-lag] for j in lag..n-1
        else:
            g = -lag
            m = n - g
            if m > 0:
                grad[:m] += w[k] * (1.0 - h[g:])  # (1-h[j+lag]) = (1-h[j-g]) for j in 0..m-1
                grad[g:] += w[k] * (-h[:m])  # -h[j-lag] = -h[j+g] for j in g..n-1
    return grad

def _random_direction(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    return d

def _fourier_direction(n, mode, phase):
    x = np.arange(n) / n
    if mode % 2 == 0:
        d = np.cos(2 * np.pi * (mode // 2) * x + phase)
    else:
        d = np.sin(2 * np.pi * (mode // 2) * x + phase)
    d -= np.mean(d)
    return d

def _coordinate_pair_direction(n, i, j):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _mirror(h):
    return h[::-1].copy()

def _complement(h):
    n = len(h)
    return _project(1.0 - h, n / 2.0)

def _rotate(h, shift):
    return np.roll(h, shift)

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G


def _line_coeffs(h, d):
    a = np.correlate(h, 1.0 - h, mode='full')
    b = np.correlate(h, -d, mode='full') + np.correlate(d, 1.0 - h, mode='full')
    c = np.correlate(d, -d, mode='full')
    return a, b, c


def _line_search(h, d, cur_c5):
    n = len(h)
    dx = 2.0 / n

    t_max = np.inf
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if np.any(pos_mask):
        t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        t_max = min(t_max, float(np.min(h[neg_mask] / (-d[neg_mask]))))

    if t_max <= 1e-14:
        return 0.0, cur_c5
    t_max = min(t_max, 2.0)

    a, b, c = _line_coeffs(h, d)

    candidates = np.array([0.0, t_max])

    mask_c = np.abs(c) > 1e-20
    if np.any(mask_c):
        t_v = -b[mask_c] / (2.0 * c[mask_c])
        valid = (t_v > 1e-12) & (t_v < t_max)
        if np.any(valid):
            candidates = np.concatenate([candidates, t_v[valid]])

    n_grid = 32
    candidates = np.concatenate([candidates, np.linspace(0.0, t_max, n_grid)])
    candidates = np.clip(candidates, 0.0, t_max)
    candidates = np.unique(np.round(candidates, 14))

    vals = a[None, :] + b[None, :] * candidates[:, None] + c[None, :] * candidates[:, None] ** 2
    c5_vals = np.max(vals, axis=1) * dx

    best_idx = int(np.argmin(c5_vals))
    best_t = float(candidates[best_idx])
    best_c5 = float(c5_vals[best_idx])

    if best_t > 0 and best_t < t_max and best_c5 < cur_c5:
        span = max(t_max * 0.04, 1e-7)
        lo_f = max(0.0, best_t - span)
        hi_f = min(t_max, best_t + span)
        fine = np.linspace(lo_f, hi_f, 50)
        vals_f = a[None, :] + b[None, :] * fine[:, None] + c[None, :] * fine[:, None] ** 2
        c5_f = np.max(vals_f, axis=1) * dx
        fi = int(np.argmin(c5_f))
        if c5_f[fi] < best_c5:
            best_t = float(fine[fi])
            best_c5 = float(c5_f[fi])

    return best_t, best_c5


def _smooth_direction(h, G, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    w = np.exp(vals)
    s = float(np.sum(w))
    if s < 1e-300:
        return None
    w /= s
    g = G.T @ w
    g -= np.mean(g)
    nm = float(np.max(np.abs(g)))
    if nm < 1e-14:
        return None
    return g / nm


def _active_subgrad_dir(h, G):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    active = corr >= mx - 1e-10 * max(1.0, abs(mx))
    if not np.any(active):
        active = np.zeros_like(corr)
        active[int(np.argmax(corr))] = 1.0
    w = active / float(np.sum(active))
    g = G.T @ w
    g -= np.mean(g)
    nm = float(np.max(np.abs(g)))
    if nm < 1e-14:
        return None
    return g / nm


def _random_dir(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    nm = float(np.max(np.abs(d)))
    if nm < 1e-14:
        return None
    return d / nm


def _fourier_dir(n, mode, phase):
    x = np.arange(n, dtype=float) / n
    if mode == 0:
        return None
    d = np.cos(2.0 * np.pi * mode * x + phase)
    d -= np.mean(d)
    nm = float(np.max(np.abs(d)))
    if nm < 1e-14:
        return None
    return d / nm


def _pair_dir(n, i, j):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d


def _full_optimize(h, deadline, rng, max_rounds=5000, verbose=False):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    G = _grad_matrix(h)
    stall = 0
    best_local = h.copy()
    best_local_c5 = cur_c5
    scale = 0.02

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 100:
            break

        directions = []

        for alpha in [30.0, 80.0, 200.0, 500.0]:
            d = _smooth_direction(h, G, alpha)
            if d is not None:
                directions.append(d)

        d = _active_subgrad_dir(h, G)
        if d is not None:
            directions.append(d)

        for _ in range(4):
            d = _random_dir(rng, n)
            if d is not None:
                directions.append(d)

        for mode in [1, 2, 3, 5]:
            d = _fourier_dir(n, mode, rng.uniform(0, 2 * np.pi))
            if d is not None:
                directions.append(d)

        for _ in range(8):
            i, j = rng.choice(n, 2, replace=False)
            d = _pair_dir(n, int(i), int(j))
            directions.append(d)

        best_improvement = None
        best_new_c5 = cur_c5
        best_new_h = None

        for d in directions:
            if time.time() > deadline:
                break
            t_star, c5_new = _line_search(h, d, cur_c5)
            if c5_new < best_new_c5 - 1e-14 and t_star > 1e-14:
                new_h = h + t_star * d
                new_h = _project(new_h, n / 2.0)
                actual_c5 = _exact_c5(new_h)
                if actual_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = actual_c5
                    best_new_h = new_h
                    best_improvement = True

        if best_new_h is not None:
            h = best_new_h
            cur_c5 = best_new_c5
            G = _grad_matrix(h)
            stall = 0
            if cur_c5 < best_local_c5:
                best_local_c5 = cur_c5
                best_local = h.copy()
        else:
            stall += 1
            if stall % 15 == 0:
                h = _project(h + rng.normal(0, scale, n), n / 2.0)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                if cur_c5 < best_local_c5:
                    best_local_c5 = cur_c5
                    best_local = h.copy()
                scale = max(0.005, scale * 0.8)
            elif stall % 5 == 0:
                scale = min(0.15, scale * 1.3)

    return best_local, best_local_c5


def _pair_sweep(h, deadline, max_pairs=None):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    if max_pairs is None:
        if n <= 80:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            idx = np.arange(n * (n - 1) // 2)
            np.random.shuffle(idx)
            pairs = []
            count = 0
            for k in idx[:max_pairs]:
                i = int(k // (n - 1))
                j = int(k % (n - 1))
                if j >= i:
                    j += 1
                pairs.append((i, j))
                count += 1
    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)][:max_pairs]

    for i, j in pairs:
        if time.time() > deadline:
            break
        d = _pair_dir(n, i, j)
        t_star, c5_new = _line_search(h, d, cur_c5)
        if c5_new < cur_c5 - 1e-13 and t_star > 1e-14:
            new_h = h + t_star * d
            new_h = _project(new_h, n / 2.0)
            actual_c5 = _exact_c5(new_h)
            if actual_c5 < cur_c5 - 1e-14:
                h = new_h
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
