def _subgrad_descent(h, target, deadline, K=8, lr_init=0.008, momentum=0.8,
                     nesterov=True, cool_rate=0.997, T_init=0.003):

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)

    T = T_init

    while time.time() < deadline:

        s_max = float(np.max(scores))
        active = _active_indices(scores, K)
        G = _grad_matrix(h)
        g = G[active].mean(axis=0)
        g -= np.mean(g)

        if nesterov:
            y = h + momentum * v
            scores_y = _scores(y)
            active_y = _active_indices(scores_y, K)
            G_y = _grad_matrix(y)
            gy = G_y[active_y].mean(axis=0)
            gy -= np.mean(gy)
            g = gy

        v = momentum * v + lr * g
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)

        delta = c5_new - c5_cur

                best_h = h_new.copy()
            lr = min(lr * 1.01, 0.15)
        else:

            v *= 0.3

def _lp_direction(h, G, active, radius):
    n = len(h)
    scores = _scores(h)
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n_vars))
    b_ub = np.zeros(len(active))
    for i, ai in enumerate(active):
        A_ub[i, :n] = G[ai]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[ai]
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)

    n_eval = 30
    alphas = np.linspace(0.0, alpha_max, n_eval)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        if a < 1e-12:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)

    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.05

    return best_h, best_c5


def _pair_search(h, target, deadline, max_passes=20):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-12:
                        continue
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        a = a_max * t
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _spectral_perturb(h, target, rng, max_freq=6, amplitude=0.04):
    n = len(h)
    k = rng.integers(1, max_freq + 1)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.01, amplitude)
    t = np.arange(n)
    pert = amp * np.cos(2 * np.pi * k * t / n + phase)
    pert -= np.mean(pert)
    h_new = _project(h + pert, target)
    return h_new


def _random_perturb(h, target, rng, scale=0.03):
    n = len(h)
    d = rng.standard_normal(n)
    d -= np.mean(d)
    d *= scale / (np.std(d) + 1e-12)
    return _project(h + d, target)


def _anneal_slsqp(h, target, deadline, betas):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for K in [8, 12, 16, 20, 25, 30]:
        if time.time() > deadline:
            break
        for r in [1e-7, 1e-6, 1e-5]:
            if time.time() > deadline:
                break
            h_try = _epi_solve(h, target, K=K, max_iter=100, r=r, deadline=deadline)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
    return best_h, best_c5


def _anneal_trust(h, target, deadline, betas):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for K in [8, 12, 16]:
        if time.time() > deadline:
            break
        for r in [1e-7, 1e-6]:
            if time.time() > deadline:
                break
            h_try = _epi_solve_trust(h, target, K=K, max_iter=80, r=r, deadline=deadline)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Epigraph SLSQP annealing (30%)
    deadline1 = t0 + budget_s * 0.30
    betas = [8, 12, 16, 20, 25, 30, 40]
    h_s, c5_s = _anneal_slsqp(h, target0, deadline1, betas)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
        h = h_s.copy()

    # Phase 2: Trust-constr epigraph (10%)
    deadline2 = t0 + budget_s * 0.40
    h_t, c5_t = _anneal_trust(h, target0, deadline2, betas)
    if c5_t < best_c5:
        best_c5 = c5_t
        best_h = h_t.copy()
        h = h_t.copy()

    # Phase 3: Subgradient with momentum (15%)
    deadline3 = t0 + budget_s * 0.55
    for K in [6, 8, 10, 12, 15]:
        if time.time() > deadline3:
            break
        h_sub, c5_sub = _subgrad_descent(best_h, best_n / 2.0,
                                         min(deadline3, time.time() + budget_s * 0.03),
                                         K=K, lr_init=0.006, momentum=0.78,
                                         nesterov=True)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            h = h_sub.copy()

    # Phase 4: LP-based local search (20%)
    deadline4 = t0 + budget_s * 0.75
    for radius in [0.15, 0.08, 0.04, 0.02]:
        if time.time() > deadline4:
            break
        h_lp, c5_lp = _lp_loop(best_h, best_n / 2.0,
                               min(deadline4, time.time() + budget_s * 0.05),
                               radius_init=radius, max_iters=4000, K=12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 5: Diversification - try perturbations and re-optimize (10%)
    deadline5 = t0 + budget_s * 0.85
    n_attempts = 0
    while time.time() < deadline5 and n_attempts < 50:
        n_attempts += 1
        if n_attempts % 3 == 0:
            h_j = _spectral_perturb(best_h, best_n / 2.0, rng)
        elif n_attempts % 3 == 1:
            h_j = _random_perturb(best_h, best_n / 2.0, rng, scale=rng.uniform(0.02, 0.06))
        else:
            h_j = _project(best_h[::-1], best_n / 2.0)

        for method in ['lp', 'epi']:
            if time.time() > deadline5:
                break
            sub_deadline = min(deadline5, time.time() + budget_s * 0.012)
            if method == 'lp':
                h_try, c5_try = _lp_loop(h_j, best_n / 2.0, sub_deadline,
                                         radius_init=0.08, max_iters=500, K=10)
            else:
                h_try = _epi_solve(h_j, best_n / 2.0, K=12, max_iter=60, r=1e-6, deadline=sub_deadline)
                c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Pair search polish (5%)
    deadline6 = t0 + budget_s * 0.90
    if time.time() < deadline6:
        hp, c5p = _pair_search(best_h, best_n / 2.0, deadline6, max_passes=30)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Resolution exploration (5%)
    deadline7 = t0 + budget_s * 0.95
    if time.time() < deadline7:
        for dn in [1, -1, 2, -2, 4, -4]:
            if time.time() > deadline7:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            c5_r_init = _exact_c5(h_r)
            if c5_r_init < best_c5 + 0.005:
                sub_dl = min(deadline7, time.time() + budget_s * 0.02)
                h_r2, c5_r2 = _lp_loop(h_r, n_try / 2.0, sub_dl,
                                       radius_init=0.06, max_iters=800, K=10)
                if c5_r2 < best_c5 - 1e-12:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()
                    best_n = n_try

    # Phase 8: Final LP polish (5%)
    deadline8 = t0 + budget_s * 0.995
    if time.time() < deadline8:
        h_f, c5_f = _lp_loop(best_h, best_n / 2.0, deadline8,
                             radius_init=0.015, max_iters=5000, K=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
