        w = np.exp(beta * shifted)
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum

        G = np.zeros((2 * n - 1, n))
        for d in range(-(n - 1), n):
            r = n - 1 + d
            G[r] = _score_grad(h, d)

        d = G.T @ w
        d -= np.mean(d)
        norm = float(np.max(np.abs(d)))
        if norm < 1e-15:
            break
        d /= norm

        best_step = 0.0
        best_c5_try = cur_c5
        best_dir = None

        for sign in [1.0, -1.0]:
            p = sign * d
            alpha_max = 1.0
            for j in range(n):
                if p[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            alpha_max = max(alpha_max, 1e-10)

            lo_a, hi_a = 0.0, alpha_max
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)

            h_c1 = _project(np.clip(h + c1 * p, 0.0, 1.0), target)
            h_c2 = _project(np.clip(h + c2 * p, 0.0, 1.0), target)
            f1 = _exact_c5(h_c1)
            f2 = _exact_c5(h_c2)

            for _ in range(35):
                if hi_a - lo_a < 1e-11 or time.time() > deadline:
                    break
                if f1 < f2:
                    hi_a = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    h_c1 = _project(np.clip(h + c1 * p, 0.0, 1.0), target)
                    f1 = _exact_c5(h_c1)
                else:
                    lo_a = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    h_c2 = _project(np.clip(h + c2 * p, 0.0, 1.0), target)
                    f2 = _exact_c5(h_c2)

            opt_a = (lo_a + hi_a) / 2.0
            h_opt = _project(np.clip(h + opt_a * p, 0.0, 1.0), target)
            f_opt = _exact_c5(h_opt)
            if f_opt < best_c5_try:
                best_c5_try = f_opt
                best_step = opt_a
                best_dir = p.copy()

        if best_step == 0.0 or best_c5_try >= cur_c5 - 1e-14:
            stall += 1
            beta_init = max(10.0, beta_init * 0.8)
            continue

        h = _project(np.clip(h + best_step * best_dir, 0.0, 1.0), target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        stall = 0

    return best_h, best_c5


def _pair_transfer(h, deadline, max_rounds=200):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 20:
            break
        improved = False
        for trial in range(80):
            i = (rnd * 31 + trial * 17) % n
            j = (rnd * 47 + trial * 23 + 7) % n
            if i == j:
                j = (i + 1) % n
            t_max = min(1.0 - h[i], h[j])
            if t_max < 1e-10:
                continue
            for frac in [0.2, 0.5, 0.8]:
                t = frac * t_max
                h_try = h.copy()
                h_try[i] += t
                h_try[j] -= t
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-13:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def _nelder_mead_anti_periodic(y0, deadline, maxiter=300):
    n_half = len(y0)
    bounds = [(0.0, 1.0)] * n_half

    def obj_wrapper(y):
        y_c = np.clip(y, 0.0, 1.0)
        return _obj_from_y(y_c)

    options = {'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12}
    try:
        result = minimize(obj_wrapper, y0, method='Nelder-Mead',
                          options=options, bounds=bounds)
        y_opt = np.clip(result.x, 0.0, 1.0)
        h = _construct_from_y(y_opt)
        return h, _exact_c5(h)
    except Exception:
        return None, None


def _de_anti_periodic(seed, n, deadline, init_y=None):
    n_half = n // 2
    bounds = [(0.0, 1.0)] * n_half
    remaining = deadline - time.time()
    if remaining < 3.0:
        return None, None

    try:
        result = differential_evolution(
            _obj_from_y, bounds,
            seed=seed,
            maxiter=min(30, int(remaining / 2)),
            popsize=10,
            tol=1e-7,
            mutation=(0.4, 1.0),
            recombination=0.6,
            polish=True,
            init='soo'
        )
        if result.x is not None:
            y_opt = np.clip(result.x, 0.0, 1.0)
            h = _construct_from_y(y_opt)
            return h, _exact_c5(h)
    except Exception:
        pass
    return None, None


def _slsqp_refine(h_init, deadline, top_k=12, maxiter=150):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()

    for _round in range(4):
        if time.time() > deadline:
            break
        scores = _all_scores(h)
        z0 = float(np.max(scores)) + 1e-6
        all_idx = np.argsort(scores)[::-1][:top_k]
        active_lags = [int(k - (n - 1)) for k in all_idx]

        x0 = np.concatenate([h, [z0]])
        bnds = [(0.0, 1.0)] * n + [(0.0, z0 + 2.0)]

        def obj_fn(x):
            return x[n]

        def obj_jac_fn(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        cons_list = []

        def eq_fun(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            return np.concatenate([np.ones(n), [0.0]])

        cons_list.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})

        def make_ineq_fun(dd):
            def f(x):
                n_x = len(x) - 1
                h_x = x[:n_x]
                if dd == 0:
                    s = float(np.sum(h_x * (1.0 - h_x)))
                elif dd > 0:
                    m = n_x - dd
                    if m <= 0:
                        return x[n_x]
                    return x[n_x] - float(np.sum(h_x[dd:dd + m] * (1.0 - h_x[:m])))
                else:
                    g = -dd
                    m = n_x - g
                    if m <= 0:
                        return x[n_x]
                    return x[n_x] - float(np.sum(h_x[:m] * (1.0 - h_x[g:g + m])))
            return f

        def make_ineq_jac(dd):
            def j(x):
                n_x = len(x) - 1
                h_x = x[:n_x]
                sg = _score_grad(h_x, dd)
                return np.concatenate([-sg, [1.0]])
            return j

        for d in active_lags:
            cons_list.append({
                'type': 'ineq',
                'fun': make_ineq_fun(d),
                'jac': make_ineq_jac(d)
            })

        def _timeout_cb(xk):
            if time.time() > deadline:
                raise RuntimeError("timeout")

        try:
            result = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                              bounds=bnds, constraints=cons_list,
                              options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False},
                              callback=_timeout_cb)
        except (RuntimeError, Exception):
            return h, _exact_c5(h)

        h_new = np.clip(result.x[:n], 0.0, 1.0)
        if abs(np.sum(h_new) - target) > 1e-4:
            h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < _exact_c5(h) - 1e-14:
            h = h_new
        else:
            break
        top_k = max(6, top_k - 3)

    return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick local refinement of initial construction
    dl1 = t0 + budget_s * 0.12
    h_ref, c5_ref = _local_refine(h, dl1, max_iters=80)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()

    # Phase 2: SLSQP refinement
    dl2 = t0 + budget_s * 0.22
    if time.time() < dl2:
        h_sl, c5_sl = _slsqp_refine(best_h, dl2, top_k=15, maxiter=100)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 3: Anti-periodic global search (DE + Nelder-Mead)
    dl3 = t0 + budget_s * 0.50
    if time.time() < dl3:
        y0 = _project_to_anti_periodic(best_h)

        # Nelder-Mead on anti-periodic subspace
        h_nm, c5_nm = _nelder_mead_anti_periodic(y0, time.time() + 20, maxiter=200)
        if h_nm is not None and c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()
        elif h_nm is not None and c5_nm < best_c5 + 0.002:
            h_nm_ref, c5_nm_ref = _local_refine(h_nm, time.time() + 10, max_iters=40)
            if c5_nm_ref < best_c5:
                best_c5 = c5_nm_ref
                best_h = h_nm_ref.copy()

        # DE runs with different seeds
        for de_i in range(4):
            if time.time() > dl3:
                break
            rem = dl3 - time.time()
            if rem < 5:
                break
            h_de, c5_de = _de_anti_periodic(seed + de_i * 100, best_n, time.time() + rem * 0.6)
            if h_de is not None:
                h_de_ref, c5_de_ref = _local_refine(h_de, time.time() + min(10, (dl3 - time.time()) * 0.3), max_iters=30)
                if c5_de_ref < best_c5:
                    best_c5 = c5_de_ref
                    best_h = h_de_ref.copy()

    # Phase 4: Pairwise transfer
    dl4 = t0 + budget_s * 0.62
    if time.time() < dl4:
        h_pw, c5_pw = _pair_transfer(best_h, dl4, max_rounds=150)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Phase 5: Multi-restart local optimization
    dl5 = t0 + budget_s * 0.82
    if time.time() < dl5:
        scale = 0.04
        for attempt in range(30):
            if time.time() > dl5:
                break
            if best_c5 < 0.3807:
                break
            rem = dl5 - time.time()
            if rem < 5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            per_dl = time.time() + min(12.0, rem * 0.35)
            h_try, c5_try = _local_refine(h_pert, per_dl, max_iters=50)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.01, scale * 0.7)
            else:
                scale = min(0.12, scale * 1.15)

    # Phase 6: Try nearby resolutions
    dl6 = t0 + budget_s * 0.93
    if time.time() < dl6:
        for dn in [2, -2, 4, -4, 6, -6]:
            if time.time() > dl6:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            h_r_ref, c5_r = _local_refine(h_r, min(dl6, time.time() + 15), max_iters=40)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r_ref.copy()
                best_n = n_try

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.995:
        h_f1, c5_f1 = _slsqp_refine(best_h, t0 + budget_s * 0.97, top_k=8, maxiter=80)
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            best_h = h_f1.copy()

        h_f2, c5_f2 = _local_refine(best_h, t0 + budget_s * 0.985, max_iters=30)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

        h_f3, c5_f3 = _pair_transfer(best_h, t0 + budget_s * 0.995, max_rounds=80)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    # If projection degraded result, use pre-projection if feasible
    if final_c5 > best_c5 + 1e-7:
        pre_h = np.clip(best_h, 0.0, 1.0)
        if abs(float(np.sum(pre_h)) - best_n / 2.0) < 0.5:
            pre_c5 = _exact_c5(pre_h)
            if pre_c5 < final_c5:
                best_h = pre_h
                final_c5 = pre_c5

    return best_h, final_c5, best_n
