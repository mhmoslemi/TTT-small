    best_c = _exact_c5(h)

    def obj_wrap(x):
        return _soft_obj(x, tau, n)

    def grad_wrap(x):
        return _soft_grad(x, tau, n)

    bounds = [(0.0, 1.0)] * n
    cur_c = _exact_c5(h)
    for round_i in range(10):
        if time.time() > deadline:
            break
        x0 = h.copy()
        try:
            res = minimize(obj_wrap, x0, method='L-BFGS-B',
                           jac=grad_wrap, bounds=bounds,
                           options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c - 1e-13:
                best_c = c5_new
                best_h = h_new.copy()
                h = h_new
                tau = min(tau * 1.5, 2000.0)
            else:
                h = best_h.copy()
                tau = min(tau * 1.5, 2000.0)
        except Exception:
            break
    return best_h, best_c


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_c5(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(15):
        a = alpha_max * (i + 1) / 15.0
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    return best_a, best_c5


def _lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        _, G = _corr_and_J(h)
        p = _lp_direction(h, scores * (n / 2.0) / (n / 2.0), G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_c5(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_refine(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = int(np.argmax(corr))
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / dj
                    else:
                        am = h[j] / (-dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / dk)
                    else:
                        am = min(am, h[k] / (-dk))
                    if am < 1e-12:
                        continue
                    for frac in [0.5, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _evolutionary(h_init, deadline, pop_size=12, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.05, n)
        pop.append(_project(h_i, target))
    fits = [_exact_c5(x) for x in pop]
    best_h = pop[int(np.argmin(fits))].copy()
    best_f = float(np.min(fits))
    mut_scale = 0.03
    while time.time() < deadline:
        idxs = rng.integers(0, pop_size, size=3)
        w1 = pop[idxs[int(np.argmin([fits[i] for i in idxs]))]].copy()
        idxs2 = rng.integers(0, pop_size, size=3)
        w2 = pop[idxs2[int(np.argmin([fits[i] for i in idxs2]))]].copy()
        child = 0.5 * (w1 + w2)
        child += rng.normal(0, mut_scale, n)
        child = _project(child, target)
        cf = _exact_c5(child)
        worst = int(np.argmax(fits))
        if cf < fits[worst]:
            pop[worst] = child
            fits[worst] = cf
            if cf < best_f:
                best_f = cf
                best_h = child.copy()
                mut_scale = max(0.008, mut_scale * 0.92)
            else:
                mut_scale = min(0.1, mut_scale * 1.08)
        else:
            mut_scale = min(0.1, mut_scale * 1.04)
    return best_h, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_end = t0 + budget_s * 0.97

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP active set on current resolution
    d1 = t0 + budget_s * 0.12
    h, c5 = _slsqp_active(h, k_active=15, max_iter=80, deadline=d1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Soft BFGS
    d2 = t0 + budget_s * 0.22
    h, c5 = _soft_bfgs(best_h, d2, tau=150.0, max_iter=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: LP local search
    d3 = t0 + budget_s * 0.32
    h, c5 = _lp_loop(best_h, d3, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: SLSQP again after LP
    d4 = t0 + budget_s * 0.42
    h, c5 = _slsqp_active(best_h, k_active=12, max_iter=60, deadline=d4)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Multi-resolution search
    d5 = t0 + budget_s * 0.65
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -30, 30, -40, 40]
    tried_n = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 180:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        h_r, c5_r = _slsqp_active(h_r, k_active=12, max_iter=50, deadline=sub_dl)
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.08)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Perturbation + re-optimization
    d6 = t0 + budget_s * 0.82
    scale = 0.03
    for attempt in range(100):
        if time.time() > d6:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_try, c5_try = _slsqp_active(h_pert, k_active=10, max_iter=40, deadline=sub_dl)
        if time.time() < d6:
            sub_dl2 = min(d6, time.time() + budget_s * 0.005)
            h_try, c5_try = _lp_loop(h_try, sub_dl2, radius_init=0.05)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.006, scale * 0.88)
        else:
            scale = min(0.15, scale * 1.2)

    # Phase 7: Evolutionary search
    if time.time() < t0 + budget_s * 0.88:
        d7 = t0 + budget_s * 0.88
        h_evo, c5_evo = _evolutionary(best_h, d7, pop_size=10, seed=seed + 13)
        if c5_evo < best_c5 - 1e-12:
            best_c5 = c5_evo
            best_h = h_evo.copy()

    # Phase 8: Final SLSQP + LP polish
    if time.time() < hard_end:
        sub_dl = min(hard_end, time.time() + budget_s * 0.03)
        h_f, c5_f = _slsqp_active(best_h, k_active=20, max_iter=80, deadline=sub_dl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < hard_end:
        sub_dl = min(hard_end, time.time() + budget_s * 0.02)
        h_f, c5_f = _lp_loop(best_h, sub_dl, radius_init=0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < hard_end:
        sub_dl = min(hard_end, time.time() + budget_s * 0.015)
        h_p, c5_p = _pair_refine(best_h, sub_dl, max_passes=50)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    if time.time() < hard_end:
        sub_dl = min(hard_end, time.time() + budget_s * 0.01)
        h_f, c5_f = _slsqp_active(best_h, k_active=25, max_iter=50, deadline=sub_dl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
