def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_jac(h):

    w = expv / s
    if idx is not None:
        w_full = np.zeros(len(corr) + 2 * (len(hvec) - 1) - (len(corr)))
        w_full = np.zeros(2 * len(hvec) - 1)
        w_full[idx] = w
        w = w_full
    J = _corr_jac(hvec)
    g = J.T @ w
    g -= np.mean(g)
    return f, g


def _lbfgs_smooth(h, alpha, maxiter=300, topM=None):
    n = len(h)
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    idx = None
    if topM is not None:
        idx = np.argsort(corr0)[::-1][:topM]
    try:
        res = minimize(_smooth_fun, h, args=(alpha, idx), jac=True,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': 0})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, n / 2.0)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_refine(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 0.0
                    if dj > 0:
                        a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max if a_max > 0 else 1e9, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max if a_max > 0 else 1e9, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max if a_max > 0 else 1e9, h[k] / (-dk))
                    if a_max < 1e-10:
                        continue
                    for frac in (0.5, 1.0, 0.25, 0.75, 0.125):
                        a = frac * a_max
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        h_t = np.clip(h_t, 0.0, 1.0)
                        h_t = _project(h_t, target)
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _block_refine(h, deadline, block_size=8, max_rounds=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    nblocks = (n + block_size - 1) // block_size
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for b in range(nblocks):
            lo = b * block_size
            hi = min(lo + block_size, n)
            if hi - lo < 1:
                continue
            jb = h[lo:hi].copy()
            res = _slsqp_solve(h, maxiter=50, ftol=1e-10, T_margin=0.005)
            if res[1] < best_c5 - 1e-13:
                h = res[0]
                best_c5 = res[1]
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP and trust-constr from initial (35% budget)
    d1 = t0 + budget_s * 0.35
    if time.time() < d1:
        h_try, c5_try = _slsqp_solve(h, maxiter=500, ftol=1e-13, T_margin=0.01)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
    if time.time() < d1:
        h_try, c5_try = _tc_solve(h, maxiter=300, T_margin=0.015)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
    # Multiple SLSQP restarts with perturbation
    for attempt in range(30):
        if time.time() > d1:
            break
        scale = 0.02 + 0.01 * (attempt % 3)
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        h_try, c5_try = _slsqp_solve(h_p, maxiter=200, ftol=1e-12, T_margin=0.008)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
        else:
            h_try2, c5_try2 = _tc_solve(h_p, maxiter=150, T_margin=0.01)
            if c5_try2 < best_c5:
                best_c5, best_h = c5_try2, h_try2.copy()

    # Phase 2: Top-M SLSQP with adaptive active set (25% budget)
    d2 = t0 + budget_s * 0.60
    for topM_val in (20, 15, 10):
        if time.time() > d2:
            break
        h_try, c5_try = _slsqp_solve(best_h, maxiter=400, ftol=1e-13,
                                     topM=topM_val, T_margin=0.005)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
    # Re-identify active set and refine
    for _ in range(5):
        if time.time() > d2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        ms = float(np.max(corr))
        active = np.where(corr > ms - 0.05)[0]
        M = min(len(active), 20)
        if M < 3:
            M = 20
        h_try, c5_try = _slsqp_solve(best_h, maxiter=300, ftol=1e-14,
                                     topM=M, T_margin=0.003)
        if c5_try < best_c5 - 1e-14:
            best_c5, best_h = c5_try, h_try.copy()
        else:
            break
    # trust-constr with top-M
    if time.time() < d2:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        ms = float(np.max(corr))
        active = np.where(corr > ms - 0.03)[0]
        M = min(len(active), 12)
        h_try, c5_try = _tc_solve(best_h, maxiter=400, topM=max(M, 8), T_margin=0.005)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    # Phase 3: Smooth L-BFGS refinement (15% budget)
    d3 = t0 + budget_s * 0.75
    for alpha in (30.0, 60.0, 120.0):
        if time.time() > d3:
            break
        h_try, c5_try = _lbfgs_smooth(best_h, alpha, maxiter=300)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
    # More SLSQP after smooth
    if time.time() < d3:
        h_try, c5_try = _slsqp_solve(best_h, maxiter=300, ftol=1e-13, T_margin=0.005)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    # Phase 4: Resolution exploration (15% budget)
    d4 = t0 + budget_s * 0.90
    for dn in (1, -1, 2, -2, 4, -4):
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        # Quick optimization at new resolution
        sub_d = min(d4, time.time() + budget_s * 0.03)
        h_r, c5_r = _slsqp_solve(h_r, maxiter=200, ftol=1e-12, T_margin=0.008)
        if time.time() < d4:
            h_r, c5_r = _lbfgs_smooth(h_r, 40.0, maxiter=200)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Pair refinement (7% budget)
    if time.time() < t0 + budget_s * 0.97:
        d5 = t0 + budget_s * 0.97
        hp, c5p = _pair_refine(best_h, d5, max_passes=100)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 6: Final SLSQP polish (3% budget)
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _slsqp_solve(best_h, maxiter=300, ftol=1e-14, T_margin=0.002)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
