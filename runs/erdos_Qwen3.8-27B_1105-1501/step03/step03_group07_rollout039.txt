    raw = np.zeros(n_r)
    grads = np.zeros((n_r, n))

    return raw, grads

    dx = 2.0 / n

    return float(np.max(corr) * dx)

        return np.zeros(n)

        return np.ones(n)

    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new

    Returns (h_new, improved) where h_new is the proposed next iterate."""

            raw[r] = float(np.sum(h[idx + k] * (1.0 - h[idx])))

        n_start = max(0, -k)
        n_end = min(n, n - k)
        if n_end <= n_start:

        # Or: skip this and rely on the natural bounds [0,1]
        pass  # We'll skip explicit L1 constraint; the [0,1] bounds already limit steps

    A_eq = np.zeros((1, n + 1))

    """Iterate LP cutting plane steps with line search."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

            break
        if stall > 50:
            break
        
        h_cand, ok = _lp_step(h, R=R, top_k=top_k)
        if not ok:
            stall += 5
            continue
        
        # Line search
        direction = h_cand - h
        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-12:
            stall += 1
            continue
        
        best_a = 0.0
        best_c = cur_c5
        for a in [1.0, 0.5, 0.25, 0.75, 0.125, 0.375, 0.625]:
            h_try = _project(h + a * direction, target)
            c_try = _exact_c5(h_try)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_a = a
        
        if best_a > 0:
            h = _project(h + best_a * direction, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
    
    return best_h, best_c5

            best_h = h_opt.copy()
    
    return best_h, best_c5, best_n

def _fine_polish(h, target, deadline):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    
    while time.time() < deadline:
        h_cand = _lp_solve(h, top_k=40, max_step=0.5)
        if h_cand is not None:
            direction = h_cand - h
            dnorm = float(np.sum(np.abs(direction)))
            if dnorm > 1e-13:
                a, c_new = _line_search(h, direction, target, cur_c5, max_alpha=0.8)
                if a > 0 and c_new < best_c5 - 1e-15:
                    h = _project(h + a * direction, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    continue
        
        improved = False
        for _ in range(30):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            if h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                a_max_j = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                a_max_k = (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk)
                a_max = min(a_max_j, a_max_k)
                if a_max < 1e-10:
                    continue
                for frac in [0.3, 0.6, 1.0, 0.1, 0.5]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[j] += dj * a
                    h_t[k] += dk * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    
    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        target0 = n0 / 2.0
        h = _project(h0, target0)
    else:
        n0 = 64
        target0 = 32.0
        h = _gen_binary(n0, rng)
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP optimization with varying parameters (35%)
    dl1 = t0 + budget_s * 0.35
    configs = [
        (None, None), (127, None), (100, None), (80, 3.0),
        (60, 2.5), (40, 2.0), (127, 3.0), (80, None), (60, None),
    ]
    for top_k, ms in configs:
        if time.time() > dl1:
            break
        sub_dl = min(dl1, time.time() + (dl1 - time.time()) * 0.35)
        h_opt, c5_opt = _lp_optimize(best_h, best_n / 2.0, sub_dl,
                                      top_k=top_k, max_step=ms, max_iters=5000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Subgradient from best (10%)
    dl2 = t0 + budget_s * 0.45
    if time.time() < dl2:
        h_opt, c5_opt = _subgrad_optimize(best_h, best_n / 2.0, dl2, max_iters=10000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Multi-start with LP (25%)
    dl3 = t0 + budget_s * 0.70
    if time.time() < dl3:
        best_h, best_c5, best_n = _multi_start_lp(best_h, best_n, best_c5, dl3, rng)
    
    # Phase 4: Resolution sweep (15%)
    dl4 = t0 + budget_s * 0.85
    for dn in [-16, -8, -4, 4, 8, 16, -12, 12, -20, 20]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(dl4, time.time() + min(15.0, (dl4 - time.time()) * 0.3))
        h_r, c5_r = _lp_optimize(h_r, n_try / 2.0, sub_dl, top_k=60, max_step=2.5, max_iters=3000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Final LP + polish (10%)
    dl5 = t0 + budget_s * 0.97
    if time.time() < dl5:
        h_opt, c5_opt = _lp_optimize(best_h, best_n / 2.0, min(dl5, time.time() + 20),
                                      top_k=100, max_step=1.5, max_iters=3000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final fine polish
    dl6 = t0 + budget_s * 0.995
    if time.time() < dl6:
        h_opt, c5_opt = _fine_polish(best_h, best_n / 2.0, dl6)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
        best_n = n0
    
    return best_h, final_c5, best_n
