import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores(h):
    """Compute all correlation scores S_d."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def score_grad_coeff(h, i):
    """For coordinate i, compute coefficients a_d and b_d for each lag d.
    
    S_d = a_d * h[i] + b_d for all d.
    Returns (a, b) arrays of length 2n-1.
    """
    n = len(h)
    s = scores(h)
    # a_d = ∂S_d/∂h[i]
    a = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d == 0:
            a[r] = 1.0 - 2.0 * h[i]
        elif d > 0:
            m = n - d
            if i >= d and i < n:
                a[r] += (1.0 - h[i - d])
            if i < m:
                a[r] -= h[i + d]
        else:
            g = -d
            m = n - g
            if i < m:
                a[r] += (1.0 - h[i + g])
            if i >= g:
                a[r] -= h[i - g]
    b = s - a * h[i]
    return a, b

def coord_optimize_1d(a, b, h_i):
    """Find t in [0,1] that minimizes max_d(a_d * t + b_d).
    
    Returns the optimal t.
    """
    # The function f(t) = max(a_d * t + b_d) is convex in t.
    # Use golden section search.
    def f(t):
        return np.max(a * t + b)
    
    # Check boundaries
    f0 = f(0.0)
    f1 = f(1.0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d_pt)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi, d_pt, fd = d_pt, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d_pt, fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    
    t_opt = (lo + hi) / 2.0

def coord_descent(h, n_iterations=20):
    """Coordinate descent for the min-max problem."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    for _ in range(n_iterations):
        for i in range(n):
            a, b = score_grad_coeff(h, i)
            t_opt = coord_optimize_1d(a, b, h[i])
            h[i] = t_opt

        h = project(h, target)
        h = np.clip(h, 0, 1)

    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        cnt = int(np.sum(mask))
        if cnt > 0:

For a single coordinate update:

- The function t → max_d S_d(t) is the max of quadratic functions (each S_d is at most quadratic in t)

Wait, is each S_d convex in h[i]? Let's check:
S_d = sum_p h[p] * (1 - h[p+d])

For a fixed d, S_d as a function of h[i]:
- Terms involving h[i]: 
  - h[i] * (1 - h[i+d]) if i+d is valid
  - h[i-d] * (1 - h[i]) if i-d is valid
  
So S_d = h[i]*(1-h[i+d]) + h[i-d]*(1-h[i]) + (other terms not involving h[i])
= h[i] - h[i]*h[i+d] + h[i-d] - h[i-d]*h[i] + const
= h[i]*(1 - h[i+d] - h[i-d]) + h[i-d] + const

This is LINEAR in h[i]! (Because the only terms with h[i] are linear: h[i] and h[i]*h[i+d] where h[i+d] is fixed, and h[i]*h[i-d] where h[i-d] is fixed.)

= h[i] - h[i+d]*h[i] + h[i-d] - h[i-d]*h[i]
= h[i] * (1 - h[i+d] - h[i-d]) + h[i-d]

Yes, it's linear in h[i]! So the 1D problem is:
minimize max_d (a_d * t + b_d) over t ∈ [0, 1] (and t = target - sum_{j≠i} h[j] for the sum constraint)

But with the sum constraint, t is FIXED: t = target - sum_{j≠i} h[j]. So I can't freely choose t!

This means single-coordinate descent with the sum constraint is trivial (t is determined). I need to either:
1. Ignore the sum constraint during the coordinate update and project afterward
2. Use pair updates (h[i] + h[j] = constant)
3. Use a Lagrange multiplier for the sum constraint

Option 1 is the simplest: update h[i] to minimize max_d S_d, then project. The projection changes all coordinates slightly, but the net effect is still a descent step (approximately).

Option 2 is cleaner: update h[i] and h[j] jointly with h[i] + h[j] = s (fixed). Then:
S_d = a_i_d * t + a_j_d * (s - t) + c_d = (a_i_d - a_j_d) * t + a_j_d * s + c_d

where c_d = S_d - a_i_d * h[i] - a_j_d * h[j] (constant).

This is again linear in t! So the 1D problem is:
minimize max_d (alpha_d * t + beta_d) over t ∈ [max(0, s-1), min(1, s)]

where alpha_d = a_i_d - a_j_d and beta_d = a_j_d * s + c_d.

This is a 1D minimax of linear functions, which is a linear program in 1D. The optimal t is at a breakpoint or boundary.

To solve it:
1. Compute all the lines f_d(t) = alpha_d * t + beta_d
2. Find the upper envelope
3. Find the minimum of the upper envelope on [lo, hi]

The minimum of the upper envelope of linear functions occurs at a point where two (or more) lines are simultaneously maximal (a "vertex" of the upper envelope), or at a boundary.

An efficient way: evaluate f(t) = max_d f_d(t) at t = lo, hi, and at all pairwise intersection points within [lo, hi]. The pairwise intersections are at t = (beta_d - beta_e) / (alpha_e - alpha_d) for alpha_d ≠ alpha_e.

With 127 lines, there are at most 127*126/2 ≈ 8000 intersections, but we only care about those in [lo, hi]. In practice, most won't be in the range.

A simpler approach: since f(t) is convex (max of affine functions), use ternary search or golden section. 50 iterations of golden section gives ~1e-10 precision, which is more than enough.

def pair_coord_update(h, i, j):
    """Update h[i] and h[j] with h[i]+h[j] fixed. Returns new h[i]."""
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    
    if lo >= hi - 1e-12:
        return h[i]
    
    # Compute S_d as linear function of t = h[i] (with h[j] = s - t)
    # S_d = (a_i_d - a_j_d) * t + a_j_d * s + c_d
    # where c_d = S_d_current - a_i_d * h[i] - a_j_d * h[j]
    
    s_cur = all_scores(h)
    n_lags = 2 * n - 1
    
    alpha = np.zeros(n_lags)  # a_i_d - a_j_d
    beta = np.zeros(n_lags)   # a_j_d * s + c_d
    
    for d in range(-(n-1), n):
        r = n - 1 + d
        ai = score_coeff(h, d, i)  # coefficient of h[i] in S_d
        aj = score_coeff(h, d, j)  # coefficient of h[j] in S_d
        alpha[r] = ai - aj
        beta[r] = aj * s + (s_cur[r] - ai * h[i] - aj * h[j])
    
    def f(t):
        return np.max(alpha * t + beta)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(60):
        if b - a < 1e-12:
            break
        if fc < fd:
            b, d_pt, fd = d_pt, c_pt, fc
            c_pt = b - gr * (b - a)
            fc = f(c_pt)
        else:
            a, c_pt, fc = c_pt, d_pt, fd
            d_pt = a + gr * (b - a)
            fd = f(d_pt)
    
    t_opt = (a + b) / 2.0
    return t_opt

def score_coeff(h, d, i):
    """Coefficient of h[i] in S_d."""
    n = len(h)
    if d == 0:
        return 1.0 - 2.0 * h[i]
    elif d > 0:
        m = n - d
        c = 0.0
        if i >= d: c += (1.0 - h[i - d])
        if i < m: c -= h[i + d]
        return c
    else:
        g = -d
        m = n - g
        c = 0.0
        if i < m: c += (1.0 - h[i + g])
        if i >= g: c -= h[i - g]
        return c

```python
import numpy as np
from scipy.optimize import minimize
import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-9:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def score_coeff(h, d, i):
    """Coefficient of h[i] in the score at lag d."""
    n = len(h)
    if d == 0:
        return 1.0 - 2.0 * h[i]
    elif d > 0:
        m = n - d
        c = 0.0
        if d <= i < n:
            c += (1.0 - h[i - d])
        if 0 <= i < m:
            c -= h[i + d]
        return c
    else:
        g = -d
        m = n - g
        c = 0.0
        if 0 <= i < m:
            c += (1.0 - h[i + g])
        if g <= i < n:
            c -= h[i - g]
        return c


def score_grad_vec(h, d):
    """Full gradient vector of score at lag d."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[d:] += (1.0 - h[:m])
        g[:m] -= h[d:]
    else:
        g_ = -d
        m = n - g_
        g[:m] += (1.0 - h[g_:])
        g[g_:] -= h[:m]
    return g


def _epi_obj(x, n):
    """Epigraph objective: minimize tau (last element)."""
    return x[-1]


def _epi_obj_grad(x, n):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def _epi_ineq_fun(x, active, n):
    """Inequality constraints: tau - S_d >= 0 for active d's."""
    h_part = x[:-1]
    tau = x[-1]
    sp = all_scores(h_part)
    return tau - sp[active]


def _epi_ineq_jac(x, active, n):
    h_part = x[:-1]
    jac = np.zeros((len(active), len(x)))
    for k, r in enumerate(active):
        d = r - (n - 1)
        jac[k, :-1] = -score_grad_vec(h_part, d)
        jac[k, -1] = 1.0
    return jac


def _sum_eq_fun(x, target):
    return np.sum(x[:-1]) - target


def _sum_eq_jac(x, target):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def epigraph_slsqp(h, target, deadline, K=24, max_rounds=25):
    """Epigraph SLSQP optimization with active set management."""
    n = len(h)
    h_work = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        s = all_scores(h_work)
        s_max = float(np.max(s))
        
        # Active set: indices of top K scores
        idx_sorted = np.argsort(s)[::-1]
        active = idx_sorted[:K]
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        constraints = [
            {'type': 'ineq', 'fun': _epi_ineq_fun, 'jac': _epi_ineq_jac, 'args': (active, n)},
            {'type': 'eq', 'fun': _sum_eq_fun, 'jac': _sum_eq_jac, 'args': (target,)}
        ]
        
        # Initial point
        h_init = np.clip(h_work, 1e-8, 1.0 - 1e-8)
        h_init = project(h_init, target)
        tau_init = float(np.max(all_scores(h_init))) + 0.05
        x0 = np.concatenate([h_init, [tau_init]])
        
        try:
            res = minimize(_epi_obj, x0, jac=_epi_obj_grad, args=(n,),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = np.clip(res.x[:-1], 0.0, 1.0)
                h_new = project(h_new, target)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_work = h_new.copy()
                else:
                    # Even if not better, use the solution for next round
                    h_work = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5

                    return 1e10
                return float(np.max(alpha_k * t + beta_k))

            best_h = h.copy()
        
        if not improved and sweep > 5:
            break
    
    return best_h, best_c5
