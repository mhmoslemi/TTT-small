def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_all(h, scores, beta=50.0):
    """Compute gradient of softmax-smoothed max of scores"""
    n = len(h)

    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad_single(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        ga = -d
        m = n - ga
        if m > 0:
            g[ga:] += (1.0 - h[:m])
            g[:m] -= h[ga:]
    return g

def _grad_smooth(h, scores, beta):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        g = np.zeros(n)
        r_star = int(np.argmax(scores))
        d_star = r_star - (n - 1)
        g = _score_grad_single(h, d_star)
        return g
    w = w / wsum
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        d = r - (n - 1)
        if d == 0:
            g += w[r] * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            g[d:] += w[r] * (1.0 - h[:m])
            g[:m] -= w[r] * h[d:]
        else:
            ga = -d
            m = n - ga
            g[ga:] += w[r] * (1.0 - h[:m])
            g[:m] -= w[r] * h[ga:]
    return g

    active = np.where(scores >= max_s - tol)[0]
    g = np.zeros(n)
    for r in active:

        g += _score_grad_single(h, d)
    g /= float(len(active))
    return g

    n = len(h)
    target = float(n) / 2.0
    h = h.copy()
    v = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    beta = beta_init
    no_improve = 0
    mu = momentum

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 500:
            break

        scores = _all_scores(h)
        g = _grad_smooth(h, scores, beta)
        g -= np.mean(g)
        gnorm = float(np.sqrt(np.sum(g * g)))
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm

        # Nesterov extrapolation
        if mu > 0 and it > 0:
            y = h + mu * v
            y = np.clip(y, 0.0, 1.0)
            y = _project(y, target)
            y_scores = _all_scores(y)
            g2 = _grad_smooth(y, y_scores, beta)
            g2 -= np.mean(g2)
            g2norm = float(np.sqrt(np.sum(g2 * g2)))

                gnorm = g2norm
            else:
                v *= 0.5
                continue

        # Line search with backtracking
        best_trial_c5 = cur_c5
        best_step = 0.0
        best_trial_h = h

                best_step = s
                best_trial_h = h_trial

            h = best_trial_h
            cur_c5 = best_trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0

            lr *= 0.85

    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

    c1 = _exact_c5(_project(h1, n / 2.0))
    c2 = _exact_c5(_project(h2, n / 2.0))
    if c1 <= c2:
        return _project(h1, n / 2.0)
    else:
        return _project(h2, n / 2.0)

            h_pert += rng.normal(0, scale * 0.5, best_n)

        h_pert = _project(h_pert, best_n / 2.0)
        per_dl = min(dl4, time.time() + max(2.0, (dl4 - time.time()) * 0.12))
        ht, ct = _gd_main(h_pert, per_dl, 50, 500, 0.03, 0.2, 300)
        if ct < best_c5 - 1e-12:
            best_c5 = ct
            best_h = ht.copy()
            scale = max(0.01, scale * 0.88)
        else:
            scale = min(0.12, scale * 1.08)

    # Phase 5: Resolution sweep (12% budget)
    dl5 = t0 + budget_s * 0.92
    for dn in [-4, 4, -8, 8, -2, 2, 12, -12, 16, -16]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 256 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        per_dl = min(dl5, time.time() + 20.0)
        ht, ct = _gd_main(h_r, per_dl, 30, 300, 0.04, 0.2, 3000)
        if ct < best_c5 - 1e-10:
            best_c5 = ct
            best_h = ht.copy()
            best_n = n_try

    # Phase 6: Final high-beta polish (7% budget)
    dl6 = t0 + budget_s * 0.99
    hf, cf = _gd_main(best_h, dl6, 300, 2000, 0.01, 0.1, 50000)
    if cf < best_c5:
        best_c5 = cf
        best_h = hf.copy()

    # Final pair swap polish
    if time.time() < dl6 + 5:
        hp2, cp2 = _pair_swap_opt(best_h, min(dl6 + 5, t0 + budget_s * 0.998), 100)
        if cp2 < best_c5:
            best_c5 = cp2
            best_h = hp2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-5:
        final_c5 = best_c5

    return best_h, final_c5, best_n
