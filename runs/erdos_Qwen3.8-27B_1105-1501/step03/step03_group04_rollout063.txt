```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_scores(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] -= h[:m]
    return G

def _soft_descent(h, deadline, beta_start=30.0, beta_end=5000.0,
                  max_iters=20000, lr_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        s_max = float(np.max(scores))
        exps = np.exp(beta * (scores - s_max))
        w_sum = float(np.sum(exps))
        if w_sum < 1e-300:
            break
        weights = exps / w_sum
        G = _grad_scores(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-12:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.5)
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-5 or stall > 100:
                break
    return best_h, best_c5

def _slsqp_polish(h, deadline, beta=200.0, max_iters=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    def objective(x):
        scores = _scores(x)
        s_max = float(np.max(scores))
        log_sum = np.log(np.sum(np.exp(beta * (scores - s_max)))) + beta * s_max
        return log_sum / beta
    
    def grad_objective(x):
        scores = _scores(x)
        s_max = float(np.max(scores))
        exps = np.exp(beta * (scores - s_max))
        weights = exps / float(np.sum(exps))
        G = _grad_scores(x)
        return G.T @ weights
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h, jac=grad_objective, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iters, 'epslin': 1e-9})
        if res.success or res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(best_h, best_c5, best_n, rng, deadline):
    # Adaptive scale
    scale = 0.05
    n = best_n
    target = n / 2.0
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # Choose perturbation type
        ptype = rng.integers(5)
        h_try = best_h.copy()
        
        if ptype == 0:
            # Gaussian perturbation
            h_try = h_try + rng.normal(0, scale, n)
        elif ptype == 1:
            # Random pair swaps
            h_try = h_try.copy()
            for _ in range(rng.integers(3, 12)):
                i = rng.integers(n)
                j = rng.integers(n)
                h_try[i], h_try[j] = h_try[j], h_try[i]
        elif ptype == 2:
            # Pair blending
            h_try = h_try.copy()
            for _ in range(rng.integers(3, 15)):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                delta = rng.uniform(-0.2, 0.2)
                h_try[i] += delta
                h_try[j] -= delta
        elif ptype == 3:
            # Reversal
            h_try = best_h[::-1].copy()
        else:
            # Complement + small perturbation
            h_try = 1.0 - best_h + rng.normal(0, scale * 0.5, n)
        
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < best_c5:
            # Quick local descent
            h_try, c5_try = _soft_descent(h_try, min(time.time() + 5, deadline - 10),
                                          beta_start=100, beta_end=2000, max_iters=3000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.01, scale * 0.85)
            else:
                scale = min(0.3, scale * 1.1)
        else:
            # Accept with some probability ( simulated annealing-like)
            prob = 0.05 if c5_try - best_c5 < 0.001 else 0.01
            if rng.random() < prob and c5_try < best_c5 + 0.01:
                h_try, c5_try = _soft_descent(h_try, min(time.time() + 3, deadline - 10),
                                              beta_start=50, beta_end=1000, max_iters=2000)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    scale = max(0.01, scale * 0.9)
            else:
                scale = min(0.3, scale * 1.05)
    
    return best_h, best_c5, best_n

def _coord_descent(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = h.copy()
    best_c5 = _exact_c5(h)
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                continue
            # Try moving h[j] and compensating on another coordinate
            for delta in [0.005, 0.01, 0.02, 0.05]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    # Find compensation coordinate
                    if d > 0:
                        # Need to decrease somewhere else
                        comp = int(np.argmin(h_try))
                    else:
                        comp = int(np.argmax(h_try))
                    if comp == j:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        if d > 0:
                            comp = int(np.argmin(np.where(mask, h_try, 1e9)))
                        else:
                            comp = int(np.argmax(np.where(mask, h_try, -1e9)))
                    if comp == j:
                        continue
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_scores(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] -= h[:m]
    return G

def _slsqp_polish(h, deadline, beta=100.0, max_iters=80):
    n = len(h)
    target = n / 2.0
    h0 = h.copy()

    def obj(x):
        sc = _scores(x)
        sm = float(np.max(sc))
        return (1.0 / beta) * (np.log(np.sum(np.exp(beta * (sc - sm)))) + beta * sm)

    def obj_grad(x):
        sc = _scores(x)
        sm = float(np.max(sc))
        ex = np.exp(beta * (sc - sm))
        w = ex / float(np.sum(ex))
        G = _grad_scores(x)
        return G.T @ w

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iters, 'ftol': 1e-12})
        h_new = _project(res.x, target)
        return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h0, _exact_c5(h0)

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try == best_n:
            continue
        h_r = _resample_h(best_h, n_try)
        time_left = deadline4 - time.time()
        if time_left < 5:
            break
        h_r, c5_r = _soft_descent_adaptive(h_r, min(deadline4, time.time() + time_left * 0.3),
                                            beta_min=30, beta_max=5000,
                                            max_iters=4000, lr_init=0.1)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Final high-beta polish (10% time)
    deadline5 = t0 + budget_s * 0.99
    h5, c5_5 = _soft_descent_adaptive(best_h, deadline5, beta_min=2000, beta_max=80000,
                                       max_iters=20000, lr_init=0.03)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Coordinate descent final
    if time.time() < t0 + budget_s * 0.998:
        h6, c5_6 = _coord_descent(best_h, t0 + budget_s * 0.998, max_passes=25)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
