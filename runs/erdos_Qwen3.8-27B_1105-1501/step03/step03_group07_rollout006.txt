Now, for the LP-based approach:

At a current point h, the objective is C(h) = dx * max_s c_s (where c_s are the correlation values).

The tangent plane approximation at h in direction d (with sum(d) = 0) is:
c_s + (∇c_s)^T d

We want to minimize the maximum of these tangent approximations:
minimize t
subject to: c_s + (∇c_s)^T d <= t for all s
sum(d) = 0
||d||_1 <= R (bounded step)
h_i = 0: d_i >= 0 (can only increase)
h_i = 1: d_i <= 0 (can only decrease)

This is a linear program! The variables are (t, d_0, ..., d_{n-1}).

After solving, we do a line search from h in direction d:
h(α) = h + α * d
C(h(α)) and find the optimum.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlate_values(h):
    """Return the raw correlation values (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if np.any(~np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    
    # Bisection for threshold
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            frac = res / cnt
            x[mask] = np.clip(x[mask] + frac, 0.0, 1.0)
    
    # Final check
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        # Distribute over interior points
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    
    return x

def _gradient_offset(h, k):
    """Compute gradient of c_k with respect to h.
    k is the offset: c_k = sum_i h[i+k] * (1-h[i]) for valid i.
    """
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            # c_k = sum_{i=0}^{m-1} h[i+k] * (1-h[i])
            # d/dh_j: 
            #   if j in [k, n-1]: j = i+k for i in [0, m-1], contrib = (1-h[i]) = (1-h[j-k])
            #   if j in [0, m-1]: j = i for i in [0, m-1], contrib = -h[i+k] = -h[j+k]
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    else:
        m = n + k  # k is negative, m = n - |k|
        if m > 0:
            # c_k = sum_{i=-k}^{n-1} h[i+k] * (1-h[i]) = sum_{j=0}^{m-1} h[j] * (1-h[j-k])
            # where j = i+k, i = j-k, j-k in [0, n-1-k] = [0, m-1+k]... 
            # Let me redo: k = -p where p > 0
            # c_{-p} = sum_{i=p}^{n-1} h[i-p] * (1-h[i])
            #        = sum_{j=0}^{n-1-p} h[j] * (1-h[j+p])
            # p = -k
            p = -k
            m2 = n - p
            # d/dh_j:
            #   if j in [0, m2-1]: contrib from h[j] term = (1-h[j+p])
            #   if j in [p, n-1]: j = j2 + p for j2 in [0, m2-1], contrib = -h[j2] = -h[j-p]
            grad[:m2] += (1.0 - h[p:])
            grad[p:] -= h[:m2]
    return grad

def _lp_tangent_step(h, target, r_step, top_k=None, T=None):
    """Solve LP to find best tangent direction.
    
    Variables: [t, d_0, d_1, ..., d_{n-1}]
    Minimize: t + lambda * sum(|d_i|)  [approximated with split vars or just L1 penalty]
    
    Actually, let's use split variables for L1:
    Variables: [t, p_0, q_0, p_1, q_1, ..., p_{n-1}, q_{n-1}]
    where d_i = p_i - q_i, p_i >= 0, q_i >= 0
    
    Minimize: t + eps * sum(p_i + q_i)
    Subject to:
      c_s + (grad_s)^T d <= t  for selected s
      sum(d_i) = 0, i.e., sum(p_i) - sum(q_i) = 0
      sum(p_i + q_i) <= r_step
      h_i = 0: q_i = 0
      h_i = 1: p_i = 0
      p_i >= 0, q_i >= 0
    """
    n = len(h)
    corr = _correlate_values(h)
    raw_vals = corr  # length 2n-1
    
    # Select which offsets to constrain
    if top_k is not None:
        sorted_idx = np.argsort(-raw_vals)
        selected = sorted_idx[:top_k]
    elif T is not None:
        thresh = float(raw_vals.max()) - T
        selected = np.where(raw_vals >= thresh)[0]
    else:
        selected = np.arange(len(raw_vals))
    
    # Build constraint matrix
    m_cons = len(selected)
    n_vars = 1 + 2 * n  # t, p_0, q_0, ..., p_{n-1}, q_{n-1}
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[0] = 1.0  # t
    
    # Inequality constraints: c_s + (grad_s)^T d <= t
    # => -t + (grad_s)^T (p-q) <= -c_s
    # => -t + grad_s @ p - grad_s @ q <= -c_s
    A_ub = np.zeros((m_cons, n_vars))
    b_ub = np.zeros(m_cons)
    
    for row, s in enumerate(selected):
        k = s - (n - 1)  # offset
        grad = _gradient_offset(h, k)
        A_ub[row, 0] = -1.0  # coefficient of t
        A_ub[row, 1:n+1] = grad  # coefficients of p
        A_ub[row, n+1:2*n+1] = -grad  # coefficients of q
        b_ub[row] = -raw_vals[s]
    
    # Equality constraints:
    # sum(p) - sum(q) = 0
    n_eq = 1
    A_eq = np.zeros((n_eq, n_vars))
    b_eq = np.zeros(n_eq)
    A_eq[0, 1:n+1] = 1.0
    A_eq[0, n+1:2*n+1] = -1.0
    b_eq[0] = 0.0
    
    # L1 norm constraint: sum(p + q) <= r_step (as inequality)
    # Add to A_ub
    A_ub_full = np.vstack([A_ub, np.zeros((1, n_vars))])
    b_ub_full = np.concatenate([b_ub, [r_step]])
    A_ub_full[-1, 1:n+1] = 1.0
    A_ub_full[-1, n+1:2*n+1] = 1.0
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for i in range(n):
        if h[i] <= 1e-10:
            # Can only increase: q_i = 0, p_i >= 0
            bounds.append((0, None))  # p_i
            bounds.append((0, 0))    # q_i
        elif h[i] >= 1.0 - 1e-10:
            # Can only decrease: p_i = 0, q_i >= 0
            bounds.append((0, 0))    # p_i
            bounds.append((0, None))  # q_i
        else:
            bounds.append((0, None))  # p_i
            bounds.append((0, None))  # q_i
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                     method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='interior-point')
        except Exception:
            return None
    
    if not res.success:
        return None
    
    # Extract direction
    d = res.x[1:n+1] - res.x[n+1:2*n+1]
    d -= np.mean(d)  # ensure zero mean
    
    # Check if direction has meaningful magnitude
    if np.linalg.norm(d) < 1e-10:
        return None
    
    return d
