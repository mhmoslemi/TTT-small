Actually, let me reconsider the approach. The key insight is:

The LP tangent plane method finds the direction that most reduces the maximum of all tangent planes. This is equivalent to finding the solution of:

min_d max_m {F_m + grad_m · d}
s.t. sum(d) = 0, ||d||_1 <= R, h + d ∈ [0,1]^n

By duality, this LP has an optimal dual solution (weights) that gives a convex combination of the active gradients. But we don't need the dual - we just need the primal direction.

The main advantage over subgradient methods:
1. It considers ALL lags simultaneously, not just a weighted average.
2. It gives the exact direction to reduce the maximum.
3. The line search then verifies actual improvement.

- R (the step size bound) needs to be tuned. Too small = slow progress. Too large = the tangent approximation is poor.
- The LP is small enough to solve quickly.

For the line search, I'll use a combination of:

2. For each, project to the feasible set.

import time
from scipy.optimize import linprog

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0

        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()

        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lp_direction(h, R, top_k=None, lam=0.0):
    """Solve LP to find tangent descent direction."""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    n_lags = 2 * n - 1
    
    # Select which lags to include
    if top_k is not None and top_k < n_lags:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_lags)
    
    # Build constraint matrix
    # Variables: [t, p_0,...,p_{n-1}, q_0,...,q_{n-1}]
    nvar = 1 + 2 * n
    ncon = len(top_idx) + 2  # lag constraints + sum + L1
    
    A_ub = np.zeros((ncon, nvar))
    b_ub = np.zeros(ncon)
    
    for idx, lag_i in enumerate(top_idx):
        d = lag_i - (n - 1)  # actual lag
        F = corr[lag_i]
        
        # Compute gradient for this lag
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                # F = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
                # grad[j] = (1-h[j-d]) for j >= d, j < n
                #         - h[j+d] for j < m

                # Wait, let me be more careful
                # F_d = sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
                # dF/dh[j]:
                #   j appears in h[i+d] when i = j-d, valid: 0 <= j-d <= n-1-d → d <= j <= n-1
                #     contributes (1-h[j-d])
                #   j appears in (1-h[i]) when i = j, valid: 0 <= j <= n-1-d → 0 <= j <= n-1-d
                #     contributes -h[j+d]
                
                # So: grad[j] += (1-h[j-d]) for j in [d, n-1]
                #           grad[j] -= h[j+d] for j in [0, n-1-d]
                
                # First part: for j = d, d+1, ..., n-1: grad[j] += (1-h[j-d])
                #   j-d ranges from 0 to n-1-d
                grad[d:] += one_minus_h[:n-d]
                # Second part: for j = 0, 1, ..., n-1-d: grad[j] -= h[j+d]
                #   j+d ranges from d to n-1
                grad[:n-d] -= h[d:]
        else:
            k = -d  # k > 0
            m = n - k
            if m > 0:
                # F_{-k} = sum_{j=0}^{n-1-k} h[j] * (1-h[j+k])
                # dF/dh[j]:
                #   j appears as first factor h[j] when 0 <= j <= n-1-k
                #     contributes (1-h[j+k])
                #   j appears in (1-h[j+k]) when j+k = j' → j = j'-k, valid: k <= j <= n-1
                #     Wait, (1-h[j+k]) depends on h[j+k], so if we vary h[k']:
                #     j+k = k' → j = k'-k, valid when 0 <= k'-k <= n-1-k → k <= k' <= n-1
                #     contributes -h[k'-k] = -h[j+k-j] ... 
                
                # Let me redo: F_{-k} = sum_{j=0}^{m-1} h[j] * (1-h[j+k]) where m = n-k
                # dF/dh[k']:
                #   k' = j for some j in [0, m-1]: contributes (1-h[j+k]) = (1-h[k'+k])
                #   k' = j+k for some j in [0, m-1]: j = k'-k, valid when 0 <= k'-k <= m-1, i.e., k <= k' <= n-1
                #     contributes -h[k'-k] = -h[j]
                
                # So: grad[j] += (1-h[j+k]) for j in [0, m-1] = [0, n-k-1]
                #       grad[j] -= h[j-k] for j in [k, n-1]
                #   where h[j-k] with j-k in [0, m-1]
                
                grad[:m] += one_minus_h[k:n]  # (1-h[k], ..., (1-h[n-1]))
                grad[k:] -= h[:m]  # h[0], ..., h[m-1]
        
        # LP constraint: F + grad · d <= t
        # → -t + grad·p - grad·q <= -F
        A_ub[idx, 0] = -1.0
        A_ub[idx, 1:1+n] = grad
        A_ub[idx, 1+n:1+2*n] = -grad
        b_ub[idx] = -F
    
    # Sum constraint: sum(p) - sum(q) = 0
    sum_row = ncon - 2
    A_ub[sum_row, 1:1+n] = 1.0
    A_ub[sum_row, 1+n:1+2*n] = -1.0
    b_ub[sum_row] = 0.0  # This is equality, handle separately
    
    # L1 constraint: sum(p) + sum(q) <= R
    l1_row = ncon - 1
    A_ub[l1_row, 1:1+n] = 1.0
    A_ub[l1_row, 1+n:1+2*n] = 1.0
    b_ub[l1_row] = R
    
    # Move sum constraint to A_eq
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)
    
    # Remove sum from A_ub
    A_ub = A_ub[:ncon-2, :]
    b_ub = b_ub[:ncon-2]
    
    # Objective: minimize t + lam * sum(p + q)
    c_obj = np.zeros(nvar)
    c_obj[0] = 1.0
    if lam > 0:
        c_obj[1:] = lam
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for j in range(n):
        bounds.append((0.0, max(0.0, 1.0 - h[j])))  # p_j
    for j in range(n):
        bounds.append((0.0, max(0.0, h[j])))  # q_j
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    p = res.x[1:1+n]
    q = res.x[1+n:1+2*n]
    d = p - q
    
    # Remove mean to ensure exact sum zero

c = [1, lam, lam, ..., lam, lam, lam, ..., lam]

Inequality constraints (A_ub @ x <= b_ub):

L1 constraint:

Equality constraint (A_eq @ x = b_eq):
  sum_j p_j - sum_j q_j = 0

Bounds:
  t: free → (None, None) or (-inf, inf)
  p_j: [0, 1-h_j]
  q_j: [0, h_j]

Actually, I want to also add a "direct" coordinate descent as a fallback/polish. The idea:

For each coordinate j (or pair j,k):
- Compute the objective as a function of h[j] (with sum constraint handled by adjusting h[k]).
- Find the optimal value.

For a single coordinate j with compensation at coordinate k:
- Let s = h[j] + h[k] (fixed).
- For x = h[j], we have h[k] = s - x.
- The objective is: C(x) = dx * max_m (a_m * x + b_m * (s-x) + c_m) = dx * max_m ((a_m - b_m)*x + b_m*s + c_m)
- This is convex in x (max of linear functions).
- Find x* that minimizes C(x) over x ∈ [max(0, s-1), min(1, s)].

For the 1D convex piecewise-linear minimization, I'll use a simple approach:
- Evaluate at several points (e.g., 20 evenly spaced + the current value).
- Find the best.
- Optionally refine with golden section.

This is fast and effective for the final polish.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-9:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _corr_and_grads(h):
    """Compute correlation values and all lag gradients."""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    
    for m in range(n_lags):
        d = m - (n - 1)
        if d >= 0:
            m_len = n - d
            if m_len > 0:
                grads[m, d:] += one_minus_h[:m_len]
                grads[m, :m_len] -= h[d:]
        else:
            k = -d
            m_len = n - k
            if m_len > 0:
                grads[m, :m_len] += one_minus_h[k:]
                grads[m, k:] -= h[:m_len]
    
    return corr, grads

    nvar = 1 + 2 * n

    A_ub = np.zeros((n_ineq, nvar))
    b_ub = np.zeros(n_ineq)
    
    for idx, m in enumerate(top_idx):

        g = grads[m]
        A_ub[idx, 0] = -1.0
        A_ub[idx, 1:1+n] = g
        A_ub[idx, 1+n:1+2*n] = -g
        b_ub[idx] = -F
    
    # L1 constraint
    A_ub[-1, 1:1+n] = 1.0
    A_ub[-1, 1+n:1+2*n] = 1.0
    b_ub[-1] = R

    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1+n] = 1.0
