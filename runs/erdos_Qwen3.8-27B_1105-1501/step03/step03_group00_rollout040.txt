def _fw_vertex(g, target):
    """Frank-Wolfe: find min g^T y subject to 0<=y_i<=1, sum(y)=target.
    Solution: sort by g, fill from smallest g to reach target."""
    n = len(g)
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0 - 1e-12:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
    return v

def _line_search_exact(h, p, c5_cur, n_coarse=20, n_fine=30):
    """Line search for h + alpha*p, exact evaluation."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-14)
    
    # Coarse grid
    alphas = np.linspace(0, alpha_max, n_coarse)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i in (0, n_coarse - 1):
        return best_a, best_c5
    
    # Fine golden section
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_coarse - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)

        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _fw_optimize(h, deadline, n_iters=5000, momentum_beta=0.5):
    """Frank-Wolfe with momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)  # momentum velocity
    
    for t in range(n_iters):
        if time.time() > deadline:
            break
        # Adaptive step size for FW: gamma = 2/(t+2)
        gamma = 2.0 / (t + 2)
        
        scores = _scores(h)
        G = _grad_matrix(h)
        
        # Use softmax-weighted gradient to handle the max
        max_s = np.max(scores)
        tau = 0.05  # temperature
        w = np.exp((scores - max_s) / tau)
        w /= np.sum(w)
        grad = G.T @ w
        grad -= np.mean(grad)  # ensure sum(grad) = 0 for sum constraint
        
        # Frank-Wolfe vertex
        v_f = _fw_vertex(-grad, target)  # minimize grad^T y → use -grad
        s = v_f - h  # direction toward vertex
        
        # Step with momentum
        v = momentum_beta * v + (1 - momentum_beta) * s
        h_new = _project(h + gamma * v, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-14:
            # Line search to find best step
            alpha, c5_line = _line_search_exact(h, v, cur_c5, n_coarse=15, n_fine=25)
            if c5_line < c5_new:

                c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    return best_h, best_c5

        # Compute subgradient at y
        scores_y = _scores(y)
        max_s = np.max(scores_y)
        w = np.exp((scores_y - max_s) / tau)
        w /= np.sum(w)
        G_y = _grad_matrix(y)
        grad = G_y.T @ w
        grad -= np.mean(grad)
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        # Step size: alpha = 2 / (L * (t+1)) where L is Lipschitz constant
        # Approximate L: max norm of gradient
        L = np.linalg.norm(grad) * 10 + 1.0
        alpha = 2.0 / (L * (t_new))
        
        h_new = _project(y - alpha * grad, target)
        # Update ghost
        beta = (t - 1.0) / t_new if t_new > 0 else 0.0
        y_new = h_new + beta * (h_new - y)
        y_new = _project(y_new, target)
        
        cur_c5 = _exact_c5(h_new)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()
        
        h = h_new
        y = y_new
        t = t_new
    
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p, cur_c5, n_coarse=25, n_fine=35)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1

    return best_h, best_c5

def _smooth_optimize(h, deadline, n_steps=400):
    """Asymmetric softmax-weighted gradient descent with adaptive temperature."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    tau_start = 0.02
    tau_end = 0.15
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        tau = tau_start + (tau_end - tau_start) * (1.0 - frac) ** 2
        
        scores = _scores(h)
        max_s = np.max(scores)
        # Use a "sharp" weighting that emphasizes the top lags
        w = np.exp((scores - max_s) / tau)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step size with backtracking
        alpha = 0.1 / (gn + 0.01)
        h_new = _project(h - alpha * grad, target)
        c5_new = _exact_c5(h_new)
        
        # Backtracking if needed
        while c5_new >= cur_c5 - 1e-15 and alpha > 1e-12:
            alpha *= 0.5
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=15):
    """Pairwise coordinate descent with exact line search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Ternary search
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _vertex_probe(h, scores, G, deadline, n_probes=50):
    """Generate vertex directions and probe them."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    rng = np.random.default_rng(12345)
    
    for probe_i in range(n_probes):
        if time.time() > deadline:
            break
        # Random weighting of lags
        temp = 0.05 + 0.1 * rng.random()
        max_s = np.max(scores)
        w = np.exp((scores - max_s) / temp)
        w /= np.sum(w)
        
        # Gradient
        grad = G.T @ w
        grad -= np.mean(grad)
        
        # Vertex in direction of -grad
        v = _fw_vertex(-grad, target)
        p = v - h
        
        # Line search
        alpha, c5_new = _line_search_exact(h, p, _exact_c5(h), n_coarse=12, n_fine=20)
        if c5_new < best_c5 - 1e-13:
            h_new = _project(h + alpha * p, target)
            best_c5 = _exact_c5(h_new)
            best_h = h_new.copy()
            # Update scores and G for next probe
            scores = _scores(best_h)
            G = _grad_matrix(best_h)
    
    return best_h, best_c5

Wait, I realize I should double-check the `_fw_vertex` function. The Frank-Wolfe vertex for the problem:
minimize g^T y
subject to 0 ≤ y_i ≤ 1, ∑ y_i = T

The solution is: sort g in ascending order, fill the smallest g entries with 1 until reaching T. If T is not an integer, the last partially-filled entry gets the fractional value.

```python
def _fw_vertex(g, target):
    n = len(g)
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0 - 1e-12:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
    return v
```

This is correct. The vertices of the polytope {0≤y≤1, ∑y=T} are exactly the 0/1 vectors with exactly floor(T) ones, plus fractional extensions when T is not integer.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _fw_vertex(g, target):
    n = len(g)
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = float(target)
    for idx in order:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0 - 1e-12:
            v[idx] = 1.0
            remaining -= 1.0
        else:
            v[idx] = remaining
            remaining = 0.0
    return v

def _line_search_exact(h, p, n_coarse=20, n_fine=30):
    n = len(h)
    alpha_max = 2.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-14)
    alphas = np.linspace(0, alpha_max, n_coarse)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_coarse - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_coarse - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_fine):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _fw_optimize(h, deadline, n_iters=3000, beta=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    vel = np.zeros(n)
    stall = 0
    for t in range(n_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        gamma = 2.0 / (t + 2)
        scores = _scores(h)
        G = _grad_matrix(h)
        max_s = np.max(scores)
        tau = 0.06
        w = np.exp((scores - max_s) / tau)
        w /= np.sum(w)
        grad = G.T @ w
        grad -= np.mean(grad)
        v_f = _fw_vertex(-grad, target)
        s = v_f - h
        vel = beta * vel + (1.0 - beta) * s
        alpha, c5_new = _line_search_exact(h, vel, n_coarse=12, n_fine=25)
        if c5_new < cur_c5 - 1e-13:

                cur_c5 = c5_check
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _smooth_optimize(h, deadline, n_steps=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        tau = 0.015 + 0.12 * (1.0 - frac) ** 2
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp((scores - max_s) / tau)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        alpha = 0.15 / (gn * gn + 0.005)
        h_new = _project(h - alpha * grad, target)
        c5_new = _exact_c5(h_new)
        while c5_new >= cur_c5 - 1e-15 and alpha > 1e-14:
            alpha *= 0.4
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and 0 < top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
