import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    # Gradient of Q_d with respect to h
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _smooth_grad(h, beta=10.0):
    """Compute smooth min-max gradient direction (for descent)"""
    n = len(h)
    S = _scores(h)
    z = float(np.max(S))
    if z < 1e-30:
        return np.zeros(n)
    A = beta * (S - z) / (z + 1e-12)
    A = A - A.max()
    W = np.exp(A)
    W = W / W.sum()
    G = np.zeros(n)
    for r in range(2*n-1):
        d = r - (n - 1)
        G += W[r] * _score_grad(h, d)
    G -= G.mean()  # zero-sum
    return -G  # negated for descent

def _max_step(h, d):
    """Largest t >= 0 such that 0 <= h[i] + t*d[i] <= 1 for all i"""
    t_max = np.inf
    for i in range(len(h)):

            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            t_max = min(t_max, h[i] / (-d[i]))
    return min(t_max, 1.0)

            continue
        h_t = h + a * d
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section refinement
    lo = max(0, best_a * 0.8 if best_a > 0 else 0)
    hi = min(t_max, best_a * 1.2 if best_a < t_max else t_max)
    if hi - lo < 1e-8:
        return best_a, best_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(30):
        if hi - lo < 1e-10 or time.time() > deadline:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_descent(h, deadline, n_iter=500, beta_start=10.0, beta_end=5.0):
    """Smooth gradient descent with exact line search"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        beta = beta_start + (beta_end - beta_start) * (it / max(n_iter-1, 1))
        d = _smooth_grad(h, beta)
        if np.max(np.abs(d)) < 1e-14:

        d = d / np.max(np.abs(d))

            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Try smaller beta or different direction
            if it > 50:
                break
    return best_h, best_c5

        # For j=2: y[::-1][1] = y[m-2]
        # So h[m+j] = 1 - y[m-1-(j-1)%m]
        idx = (m - 1 - (np.arange(m) - 1) % m) % m
        for j in range(m):
            gy[j] = Gh[j] - Gh[m + idx[j]]
    return _exact_c5(h), gy, h

def _antipodal_optimize(rng, n, deadline):
    m = n // 2
    best_c5 = 1.0
    best_y = None
    best_perm = 'rev'
    
    for perm in ['rev', 'id', 'shift1']:
        if time.time() > deadline:
            break
        # Multiple starts
        n_starts = 4
        for s in range(n_starts):
            if time.time() > deadline:
                break
            if s == 0:
                y = np.full(m, 0.5)
            elif s == 1:
                # Bipolar start
                y = np.where(np.arange(m) % 2 == 0, 0.8, 0.2)
            elif s == 2:
                # Sine-based
                y = 0.5 + 0.4 * np.sin(2 * np.pi * np.arange(m) / m)
            else:
                y = rng.uniform(0, 1, m)
            y = np.clip(y, 0, 1)
            
            c5 = _exact_c5(_make_antipodal(y, perm))
            
            # Gradient descent on y
            for it in range(200):
                if time.time() > deadline:
                    break
                beta = 5.0 + 5.0 * (it / 200.0)
                _, gy, _ = _antipodal_grad(y, perm, beta)
                norm = float(np.max(np.abs(gy)))
                if norm < 1e-13:
                    break
                d = -gy / norm
                # Max step in [0,1] for y
                t_max = 1.0
                pos = d > 1e-15
                neg = d < -1e-15
                if np.any(pos):
                    t_max = min(t_max, float(np.min((1.0 - y[pos]) / d[pos])))
                if np.any(neg):
                    t_max = min(t_max, float(np.min(y[neg] / (-d[neg]))))
                if t_max < 1e-10:
                    break
                # Line search
                best_a = 0.0
                best_c = c5
                for a in np.linspace(0.02 * t_max, t_max, 10):
                    y_t = np.clip(y + a * d, 0, 1)
                    c_t = _exact_c5(_make_antipodal(y_t, perm))
                    if c_t < best_c:
                        best_c = c_t
                        best_a = a
                if best_a < 1e-10:
                    break
                y = np.clip(y + best_a * d, 0, 1)
                c5 = _exact_c5(_make_antipodal(y, perm))
            
            if c5 < best_c5:
                best_c5 = c5
                best_y = y.copy()
                best_perm = perm
    
    if best_y is not None:
        return _make_antipodal(best_y, best_perm), best_c5
    return None, 1.0

def _pair_polish(h, deadline, max_rounds=50):
    """Pair exchange polish: move mass from i to j"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Focus on pairs involving high-overlap regions
        S = _scores(h)
        # Find which coordinates are in high-overlap lags
        # Simple: try random pairs biased toward extremes
        n_trials = min(n * 3, 500)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = int(rng_pair_trial())
            j = int(rng_pair_trial())
            if i == j or abs(h[i] - h[j]) < 1e-4:
                continue
            # Try transferring mass from higher to lower (or vice versa)
            if h[i] > h[j]:
                t_max = min(h[i] - h[j], 0.5)
            else:
                t_max = min(h[j] - h[i], 0.5)
            if t_max < 1e-4:
                continue
            # 1D search: h[i] += t*sign, h[j] -= t*sign where sign moves from high to low
            sign = 1.0 if h[i] > h[j] else -1.0
            best_t = 0.0
            best_c = _exact_c5(h)
            for t in np.linspace(0.05 * t_max, t_max, 8):
                h_t = h.copy()
                h_t[i] = h[i] + sign * t
                h_t[j] = h[j] - sign * t
                if h_t[i] < -1e-12 or h_t[i] > 1 + 1e-12 or h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12:
                    continue
                c_t = _exact_c5(h_t)
                if c_t < best_c:
                    best_c = c_t
                    best_t = t
            if best_t > 1e-6 and best_c < best_c5 - 1e-12:
                h[i] += sign * best_t
                h[j] -= sign * best_t
                h = np.clip(h, 0, 1)
                best_c5 = _exact_c5(h)
                if best_c5 < float(np.max(np.max(h) - 1.0) if False else best_c5):
                    pass
                improved = True
        if not improved:
            break
    
    return h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    dl2 = t0 + budget_s * 0.55
    h_ant, c5_ant = _antipodal_optimize(rng, best_n, dl2)
    if h_ant is not None and c5_ant < best_c5:
        best_c5 = c5_ant

    for dn in [-8, 8, -4, 4, -16, 16]:
        n_try = best_n + dn
        if n_try < 24 or n_try > 160 or n_try % 2 != 0:

        if time.time() > dl2:
            break
        h_ant2, c5_ant2 = _antipodal_optimize(rng, n_try, t0 + budget_s * 0.55)
        if h_ant2 is not None and c5_ant2 < best_c5:
            best_c5 = c5_ant2
            best_h = h_ant2.copy()
            best_n = n_try
    
    # Phase 3: Refine best in full space (25% budget)
    dl3 = t0 + budget_s * 0.80
    if best_n != n0:
        h = _resample(best_h, n0)
    else:
        h = best_h.copy()
    # Actually work with best_n
    h = best_h.copy()
    hn = len(h)
    for attempt in range(5):

        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        per_dl = min(dl3, time.time() + max(5.0, (dl3 - time.time()) * 0.4))
        h_try, c5_try = _gd_full(h_pert, per_dl, max_iter=200)
        if c5_try < best_c5:

        best_h = _project(best_h, final_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, final_n
