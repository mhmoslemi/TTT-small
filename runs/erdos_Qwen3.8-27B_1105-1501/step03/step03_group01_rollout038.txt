                vals[i] = t - self.dx * np.dot(hv[gg:], 1.0 - hv[:m])
        return vals
    
    def constr_jac(self, x):
        hv = x[:self.n]
        J = np.zeros((len(self.active), self.n + 1))
        for i in range(len(self.active)):
            r = self.active[i]
            d = r - (self.n - 1)
            J[i, :self.n] = -self.dx * _corr_grad(hv, d)
            J[i, self.n] = 1.0
        return J
    
    def eq(self, x):
        return x[:self.n].sum() - self.target
    
    def eq_jac(self, x):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J

def _slsqp_opt(h_init, deadline, n_active=10, maxiter=300, ftol=1e-12):
    """SLSQP optimization on epigraph with top-n_active constraints."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    c0 = _exact_c5(h_init)
    corr_all = _corr_values(h_init)
    max_c = float(np.max(corr_all))
    
    # Get top n_active lags
    sorted_idx = np.argsort(corr_all)[::-1]
    active = sorted_idx[:min(n_active, len(sorted_idx))]
    
    epi = _Epi(n, active, dx, target)
    x0 = np.concatenate([h_init, [max_c + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'fun': epi.constr, 'jac': epi.constr_jac, 'type': 'ineq'},
        {'fun': epi.eq, 'jac': epi.eq_jac, 'type': 'eq'}
    ]
    
    try:
        res = minimize(epi.obj, x0, jac=epi.obj_grad, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_new = _project(res.x[:n], target)
        c_new = _exact_c5(h_new)
        return h_new, c_new
    except Exception:
        return h_init.copy(), c0

def _slsqp_full(h_init, deadline, maxiter=150):
    """SLSQP with all constraints."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    c0 = _exact_c5(h_init)
    corr_all = _corr_values(h_init)

    except Exception:
        return h_init.copy(), c0

def _lp_active_dir(h, scores, n_active, radius):
    """LP direction using only top-n_active constraints."""
    n = len(h)
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:n_active]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    for i, r in enumerate(active):
        d = r - (n - 1)
        g = _corr_grad(h, d)
        # Linearized constraint: scores[r] + g . (h_new - h) <= scores[r] + t_new
        # i.e., g . h_new - t_new <= 0 (approximately, at the linearization point)
        # Direction: G @ p - p_t <= -scores[r] + scores[r] = ... 
        # Actually: we want g . (h + alpha*p - h) - (p_t)*alpha <= 0
        # i.e., g . p - p_t <= 0 (per unit step)
        # But we also want to ensure we don't increase above current: g . p - p_t <= 0
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
    b_ub = np.zeros(len(active))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(15):
        a = alpha_max * (i + 1) / 15.0
        h_t = h + a * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_loop_active(h, deadline, radius_init=0.1, n_active=12):
    """LP direction search using only active constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 40:
            break
        scores = _corr_values(h)
        p = _lp_active_dir(h, scores, n_active, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-11:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_max_grad(h, tau):
    """Gradient of soft-max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    scores = _corr_values(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        g = _corr_grad(h, d)
        grad += wk * dx * g
    return grad

def _grad_search(h, deadline, tau=100.0, lr=0.01):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 30:
            break
        grad = _soft_max_grad(h, tau)
        mom = 0.85 * mom + 0.15 * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.05)
        else:
            stall += 1
            lr *= 0.92
            if stall % 15 == 0:
                h = best_h + np.random.default_rng(stall * 7).normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=20):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = min((1.0 - h[j]) / max(dj, 1e-12) if dj > 0 else h[j] / max(-dj, 1e-12),
                             (1.0 - h[k]) / max(dk, 1e-12) if dk > 0 else h[k] / max(-dk, 1e-12))
                    if am < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _corr_grad(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] += (-h[:m])
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] += (-h[:m])
    return g

class _Epi:
    def __init__(self, n, active, dx, target):
        self.n = n
        self.active = active
        self.dx = dx
        self.target = target

    def obj(self, x):
        return x[self.n]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def constr(self, x):
        hv = x[:self.n]
        t = x[self.n]
        vals = np.zeros(len(self.active))
        for i in range(len(self.active)):
            r = self.active[i]
            d = r - (self.n - 1)
            if d >= 0:
                m = self.n - d
                vals[i] = t - self.dx * np.dot(hv[:m], 1.0 - hv[d:])
            else:
                gg = -d
                m = self.n - gg
                vals[i] = t - self.dx * np.dot(hv[gg:], 1.0 - hv[:m])
        return vals

    def constr_jac(self, x):
        hv = x[:self.n]
        J = np.zeros((len(self.active), self.n + 1))
        for i in range(len(self.active)):
            r = self.active[i]
            d = r - (self.n - 1)
            J[i, :self.n] = -self.dx * _corr_grad(hv, d)
            J[i, self.n] = 1.0
        return J

    def eq(self, x):
        return x[:self.n].sum() - self.target

    def eq_jac(self, x):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J

def _slsqp_opt(h_init, deadline, n_active=10, maxiter=300, ftol=1e-12):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    c0 = _exact_c5(h_init)
    corr_all = _corr_values(h_init)
    max_c = float(np.max(corr_all))
    sorted_idx = np.argsort(corr_all)[::-1]
    active = sorted_idx[:min(n_active, 2 * n - 1)]
    epi = _Epi(n, active, dx, target)
    x0 = np.concatenate([h_init, [max_c + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'fun': epi.constr, 'jac': epi.constr_jac, 'type': 'ineq'},
        {'fun': epi.eq, 'jac': epi.eq_jac, 'type': 'eq'}

                      constraints=constraints, method='SLSQP',

def _soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _corr_values(h)
    max_s = float(np.max(scores))

        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final: tight LP polish
    if time.time() < t0 + budget_s * 0.995:
        h_f2, c5_f2 = _lp_loop_active(best_h, t0 + budget_s * 0.998, radius_init=0.02, n_active=8)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
