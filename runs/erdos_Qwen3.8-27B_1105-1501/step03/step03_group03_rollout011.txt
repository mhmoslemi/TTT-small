def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x


def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d + m] += (1.0 - h[:m])
                G[r, :m] -= h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g + m])
                G[r, g:g + m] -= h[:m]
    return G


def _smooth_obj(h, beta):
    scores = _scores(h)
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _smooth_jac(h, beta):
    scores = _scores(h)
    s_max = float(np.max(scores))
    exp_s = np.exp(beta * (scores - s_max))
    w = exp_s / np.sum(exp_s)
    G = _build_G(h)
    return G.T @ w


def _sum_fun(h, target):
    return np.sum(h) - target


def _sum_jac(h, target):
    return np.ones(len(h))

    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()

def _lp_direction(h, scores, G, radius):

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

    b_eq = np.zeros(1)

        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:

                      bounds=bounds, method='highs')

def _line_search(c5_func, p, h, c5_cur, n_evals=30):
    """Line search along direction p from h."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):

            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Coarse grid
    best_a = 0.0
    best_f = c5_cur
    alphas = np.linspace(0, alpha_max, n_evals)
    for a in alphas:
        if a < 1e-12:
            continue
        f = c5_func(h + a * np.sign(1) * p)  # Wait, we need h + a*p
        f = c5_func(h + a * p)
        if f < best_f:
            best_f = f
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = c5_func(h + c1 * p)
    f2 = c5_func(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = c5_func(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = c5_func(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_f = c5_func(h + opt_a * p)
    if opt_f < best_f:
        return opt_a, opt_f
    return best_a, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _pair_descent(h, deadline, max_passes=30):
    """Coordinate descent on pairs of variables."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / dj)
                    else:
                        a_max = min(a_max, h[j] / (-dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / dk)
                    else:
                        a_max = min(a_max, h[k] / (-dk))
                    if a_max < 1e-12:
                        continue
                    # Try a few fractions
                    for t in [0.3, 0.7, 1.0]:
                        a = a_max * t
                        h[j] += dj * a
                        h[k] += dk * a
                        c5_t = _exact_c5(h)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h.copy()
                            improved = True
                            break
                        # Revert
                        h[j] -= dj * a
                        h[k] -= dk * a
                    else:
                        # No improvement, revert already done
                        continue
                    # If we broke out of t-loop, h is already updated
                    # If not, h was reverted
        if not improved:
            break
    
    return h, best_c5


def _gradient_descent(h, target, deadline, beta=20.0, lr=0.01, momentum=0.8):
    """Nesterov momentum gradient descent with soft-max surrogate."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    
    while time.time() < deadline:
        y = h + momentum * v
        grad = _smooth_jac(y, beta)
        grad = grad - np.mean(grad)
        
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.2)
        else:
            lr *= 0.85
            v *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5


def _random_block_init(n, rng, n_blocks=4):

    h = np.zeros(n)
    pos = 0
    for i, bs in enumerate(block_sizes):
        if i % 2 == 0:
            h[pos:pos+bs] = 1.0
        pos += bs
    return _project(h, n / 2.0)


def _fourier_init(n, rng, n_harm=8, amp=0.4):
