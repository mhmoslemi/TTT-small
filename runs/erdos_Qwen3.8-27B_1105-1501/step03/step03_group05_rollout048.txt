import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        m = high - low
        if m <= 0:
            continue
        p = low + np.arange(m)
        q = p + d
        G[r, q] += (1.0 - h[p]) * dx
        G[r, p] -= h[q] * dx
    return G

def _project_box_sum(v, target):
    n = len(v)
    x = np.clip(v, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) < 1e-12:
        return x
    # Water-filling
    if res > 0:
        # Need to add 'res' total, respecting upper bound
        # Add uniformly to coordinates below 1
        for _ in range(n + 2):
            mask = x < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = res / cnt
            if step >= 1.0 - 1e-15:
                # Would exceed bound for some
                # Add what we can
                x[mask] = 1.0
            else:
                x[mask] += step
            res = target - float(np.sum(x))
            if abs(res) < 1e-12:
                break
    else:
        # Need to subtract (-res) total, respecting lower bound
        for _ in range(n + 2):
            mask = x > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = (-res) / cnt
            if step >= 1e-15:
                x[mask] = 0.0
            else:
                x[mask] -= step
            res = target - float(np.sum(x))
            if abs(res) < 1e-12:
                break
    # Final correction
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _opt_2coord(h, j, k, scores0, G):
    """Find optimal t for h[j]+=t, h[k]-=t using linearized model."""
    c = G[:, j] - G[:, k]
    t_lo = max(-h[j], h[k] - 1.0)
    t_hi = min(1.0 - h[j], h[k])
    if t_lo >= t_hi - 1e-14:
        return 0.0
    
    # Bisection on subgradient
    lo, hi = t_lo, t_hi
    for _ in range(45):
        mid = (lo + hi) / 2.0
        vals = scores0 + c * mid
        v_max = vals.max()
        # Active set: scores that achieve (or are very close to) the max
        active = vals >= v_max - 1e-12
        s_min = c[active].min()
        s_max = c[active].max()
        if s_min > 1e-14:
            hi = mid
        elif s_max < -1e-14:
            lo = mid
        else:
            lo = hi = mid
            break
    
    return (lo + hi) / 2.0

def _lp_refine(h, deadline, max_iters=5000, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        G = _build_G(h)
        scores0 = _scores_raw(h)
        
        # LP: find x minimizing max linearized score
        # minimize t
        # s.t. G @ (x - h) + scores0 <= t  =>  G @ x - t <= scores0 - G @ h  => G @ x - t <= scores0 (since G@h is already in scores0)
        # Actually: linearized score = scores0 + G @ (x - h) = scores0 + G@x - G@h
        # We want: scores0 + G@(x-h) <= t for all rows
        # i.e., G@x - t <= -scores0 + G@h
        # But G@h is the derivative of scores at h, not the scores themselves.
        # The linearization is: score_r(x) ≈ score_r(h) + G[r,:] @ (x - h)
        # So: score_r(h) + G[r,:] @ (x-h) <= t
        # => G[r,:] @ x <= t - score_r(h) + G[r,:] @ h
        # Let me use the penalty variable formulation:
        # minimize eta
        # s.t. G@x - G@h + scores0 - eta <= 0  (i.e., linearized score <= eta)
        #      sum(x) = n/2
        #      |x - h|_inf <= radius
        #      eta >= 0
        
        L = 2 * n - 1
        # Variables: x (n vars) + eta (1 var) = n+1 vars
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize eta
        
        # Inequality: G@x - G@h + scores0 - eta <= 0
        # => G@x - eta <= G@h - scores0
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = G @ h - scores0
        
        # Sum constraint
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Box + radius constraints
        bounds = []
        for j in range(n):
            lo = max(0.0, h[j] - radius)
            hi = min(1.0, h[j] + radius)
            bounds.append((lo, hi))
        bounds.append((0.0, None))  # eta >= 0
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'maxiter': 3000})
            if not res.success:
                stall += 1
                radius *= 0.8
                continue
            x_new = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.8
            continue
        
        if float(np.max(np.abs(x_new - h))) < 1e-12:
            stall += 1
            radius *= 0.85
            continue
        
        # Try the solution with damped steps
        best_step = 0.0
        best_c5_step = cur_c5
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15]:
            x_try = h + factor * (x_new - h)
            x_try = _project_box_sum(x_try, target)
            c5_try = _exact_c5(x_try)
            if c5_try < best_c5_step - 1e-12:
                best_c5_step = c5_try
                best_step = factor
        
        if best_step > 0:
            h_new = h + best_step * (x_new - h)
            h_new = _project_box_sum(h_new, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            stall = 0
            radius = min(radius * 1.3, 0.5)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Use integral averaging: each old cell has unit width 1/n, map to new grid
    x_old = (np.arange(n) + 0.5) / n  # cell centers in [0, 1)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)
