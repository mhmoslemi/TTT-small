def _anneal(h, deadline, rng, max_moves=50000):

    target = n / 2.0

    for mv in range(max_moves):

        if j == k:

        delta = rng.uniform(-sigma, sigma)

        h[j] += d

                no_improve = 0
            else:

    # Phase 6: Resolution sweep (12%)
    if rem() > budget_s * 0.10:
        dl = t0 + budget_s * 0.90
        n_cur = len(best_h)
        for dn in [-8, -6, -4, -2, -1, 1, 2, 4, 6, 8, 10, 12]:
            if time.time() > dl:
                break
            n_try = n_cur + dn
            if n_try < 32 or n_try > 100:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_ks=[8, None, 5], radius_init=0.1, max_iters=500)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_cur = n_try

    # Phase 7: Pair polish (5%)
    if rem() > budget_s * 0.04:
        dl = t0 + budget_s * 0.955
        hp, c5p = _pair_polish(best_h, dl)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 8: Final ultra-fine LP (3%)
    if rem() > budget_s * 0.01:
        dl = t0 + budget_s * 0.995
        hf, c5f = _lp_optimize(best_h, dl, top_ks=[3, 4, 5], radius_init=0.01, max_iters=2000)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, len(best_h)
