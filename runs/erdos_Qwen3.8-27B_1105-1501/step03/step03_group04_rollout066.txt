import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """G[lag, i] = df/dh[i] where f = sum_j h[j]*(1-h[j+lag])"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:d+m] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:g+m] += (1.0 - h[:m])
                G[r, :m] -= h[g:g+m]
    return G

def _soft_max_descent(h, deadline, beta_start, beta_end, max_iters, lr_start, lr_end):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_start
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        # Use a tempered softmax over top half of scores
        # This avoids the "exponential explosion" issue
        mask = scores > m - 5.0 / beta
        if mask.sum() == 0:
            mask = scores > m - 10.0 / beta
        if mask.sum() == 0:
            mask = np.ones(len(scores), dtype=bool)
        exps = np.zeros(len(scores))
        exps[mask] = np.exp(beta * (scores[mask] - m))
        wsum = float(np.sum(exps))
        if wsum < 1e-30:
            weights = np.zeros(len(scores))
            weights[np.argmax(scores)] = 1.0
        else:
            weights = exps / wsum
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(30):
            h_try = h + alpha * direction
            if np.any(h_try < -0.5) or np.any(h_try > 1.5):
                alpha *= 0.5
                continue
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.8
            if stall > 80:
                break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, t_offset=0.0):
    """Find direction p that reduces max of (G @ (h + t*p)) subject to box and sum constraints."""
    n = len(h)
    # Minimize t subject to:
    # G @ (h + t*p)[lag] <= 0 for all lags (i.e., G@h + t*G@p <= 0)
    # sum(p) = 0
    # -radius <= p[i] <= radius, adjusted by current bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores - t_offset
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_golden(h, p, c5_cur):
    n = len(h)
    # Find max alpha such that h + alpha*p stays in [0,1]
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a = 0.0
    hi_a = alpha_max
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    for _ in range(30):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a = c2
            c2 = c1
            f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, n / 2.0))
        else:
            lo_a = c1
            c1 = c2
            f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(_project(h + c2 * p, n / 2.0))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, n / 2.0))
    return opt_a, opt_c5

def _lp_polish(h, deadline, radius_init=0.05, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)

        scores = _scores(h)

        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1

            continue
        alpha, new_c5 = _line_search_golden(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:

            if cur_c5 < best_c5:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new
