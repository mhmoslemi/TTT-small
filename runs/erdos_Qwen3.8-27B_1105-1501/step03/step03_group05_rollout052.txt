def _resample_integral(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Integral-average method
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New cell [j*dx_new, (j+1)*dx_new]
        a = j * dx_new
        b = (j + 1) * dx_new
        # Find overlap with old cells
        total_weight = 0.0
        total_value = 0.0
        for i in range(n):
            old_a = i * dx_old
            old_b = (i + 1) * dx_old
            overlap = max(0.0, min(b, old_b) - max(a, old_a))
            if overlap > 0:
                total_weight += overlap
                total_value += overlap * h[i]
        if total_weight > 0:
            h_new[j] = total_value / total_weight
    return _project_box_sum(h_new, n_new / 2.0)

import numpy as np
import cvxpy as cp
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    diff = target - float(np.sum(v))
    if abs(diff) < 1e-13:
        return v
    for _ in range(500):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-14
        else:
            mask = v > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
        diff = target - float(np.sum(v))
    diff = target - float(np.sum(v))
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _build_grad_scaled(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        lo = max(0, -d)
        hi = min(n, n - d)
        for p in range(lo, hi):
            q = p + d
            G[r, q] += (1.0 - h[p])
            G[r, p] -= h[q]
    G *= dx
    return G

def _solve_trust(h, rho):
    n = len(h)
    scores = _scores_scaled(h)
    G = _build_grad_scaled(h)
    c = G @ h - scores
    x = cp.Variable(n, name='x')
    t = cp.Variable(name='t')
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        cp.norm(x - h) <= rho,
        G @ x - t <= c,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = ['CLARABEL', 'SCS', 'ECOS', 'MOSEK']
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if x.value is not None:
                return np.asarray(x.value, dtype=float)
        except Exception:
            continue
    return None

def _line_search(h, step, cur_c5, n_trials=12):
    n = len(h)
    target = n / 2.0
    alpha_max = 1.0
    for j in range(n):
        d = step[j]
        if d > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d)
        elif d < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d))
    alpha_max = max(alpha_max, 1e-8)
    best_a = 0.0
    best_c5 = cur_c5
    for i in range(1, n_trials + 1):
        a = alpha_max * i / n_trials
        h_new = _project_box_sum(h + a * step, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_a = a
    if best_a <= 0:
        return 0.0, cur_c5
    return best_a, best_c5

def _trust_loop(h_init, deadline, sigma_start=0.05, max_iters=400, seed=0):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sigma = sigma_start
    stall = 0
    rng = np.random.default_rng(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if sigma < 0.003:
            break
        rho = sigma * math.sqrt(n)
        x_star = _solve_trust(h, rho)
        if x_star is None:
            sigma *= 0.5
            stall += 1
            continue
        step = x_star - h
        a, new_c5 = _line_search(h, step, cur_c5)
        if a > 0 and new_c5 < cur_c5 - 1e-12:
            h = _project_box_sum(h + a * step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            sigma = min(sigma * 1.25, 0.25)
        else:
            stall += 1
            sigma *= 0.6
            if stall > 20:
                h_pert = h.copy()
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    delta = rng.uniform(-0.08, 0.08)
                    h_pert[i] += delta
                    h_pert[j] -= delta
                h_pert = _project_box_sum(h_pert, target)
                if _exact_c5(h_pert) < cur_c5:
                    h = h_pert
                    cur_c5 = _exact_c5(h)
                    stall = 0
                    sigma = sigma_start
                else:
                    break
    return best_h, best_c5

def _smooth_grad_descent(h_init, deadline, beta_start=5.0, beta_end=200.0, n_beta=10, max_steps=3000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for beta in np.geomspace(beta_start, beta_end, n_beta):
        if time.time() > deadline:
            break
        step_size = 0.1
        stall = 0
        for _s in range(max_steps):
            if time.time() > deadline:
                break
            if stall > 80:
                break
            scores = _scores_scaled(h)
            max_s = float(np.max(scores))
            exps = np.exp(beta * (scores - max_s))
            w_sum = float(np.sum(exps))
            if w_sum < 1e-300:
                w = np.zeros(len(scores))
                w[np.argmax(scores)] = 1.0
            else:
                w = exps / w_sum
            grad = np.zeros(n)
            for d in range(-(n - 1), n):
                wd = w[n - 1 + d]
                if abs(wd) < 1e-14:
                    continue
                lo = max(0, -d)
                hi = min(n, n - d)
                for p in range(lo, hi):
                    q = p + d
                    grad[q] += wd * (1.0 - h[p])
                    grad[p] -= wd * h[q]
            grad -= np.mean(grad)
            norm = float(np.max(np.abs(grad)))
            if norm < 1e-14:
                break
            d = -grad / norm * step_size
            a, new_c5 = _line_search(h, d, cur_c5, n_trials=10)
            if new_c5 < cur_c5 - 1e-12:
                h = _project_box_sum(h + a * d, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step_size = min(step_size * 1.1, 0.4)
            else:
                stall += 1
                step_size *= 0.7
    return best_h, best_c5

def _binary_search(h, deadline, rng, max_iters=50000):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 1500:
            break
        best_swap = None
        best_val = cur_c5
        batch = min(25, max(1, len(ones) * len(zeros)))
        for _b in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def _resample_integral(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        tw = 0.0
        tv = 0.0
        i_start = max(0, int(a / dx_old) - 1)
        i_end = min(n, int(b / dx_old) + 2)
        for i in range(i_start, i_end):
            oa = i * dx_old
            ob = (i + 1) * dx_old
            overlap = max(0.0, min(b, ob) - max(a, oa))
            if overlap > 0:
                tw += overlap
                tv += overlap * h[i]
        if tw > 0:
            h_new[j] = tv / tw
    return _project_box_sum(h_new, n_new / 2.0)

def _diversify(h_best, deadline, rng, sigma_start=0.04):
    n = len(h_best)
    targets = n / 2.0
    pool = []
    pool.append(h_best.copy())
    for shift in [1, 2, 3, n // 4, n // 3]:
        pool.append(np.roll(h_best, shift))
    pool.append(1.0 - h_best)
    pool.append(np.roll(1.0 - h_best, 1))
    pool.append(np.roll(1.0 - h_best, 2))
    for _ in range(5):
        hp = h_best.copy()
        for _m in range(8):
            i = rng.integers(n)
            j = rng.integers(n)
            delta = rng.uniform(-0.12, 0.12)
            hp[i] += delta
            hp[j] -= delta
        pool.append(_project_box_sum(hp, targets))
    for i in range(min(4, len(pool) - 1)):
        for j in range(i + 1, min(len(pool), i + 3)):
            alpha = rng.uniform(0.25, 0.75)
            pool.append(_project_box_sum(alpha * pool[i] + (1 - alpha) * pool[j], targets))
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    per_time = max(5.0, (deadline - time.time()) / max(1, len(pool)))
    for h_start in pool:
        if time.time() > deadline:
            break
        h_start = _project_box_sum(h_start, targets)
        c5_s = _exact_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()
        sub_dl = min(deadline, time.time() + per_time)
        h_r, c5_r = _trust_loop(h_start, sub_dl, sigma_start=sigma_start, max_iters=120)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _iterated_local_search(h_best, deadline, rng, n_trials=15, pert_size=0.08):
    n = len(h_best)
    targets = n / 2.0
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    per_time = max(5.0, (deadline - time.time()) / max(1, n_trials))
    for _t in range(n_trials):
        if time.time() > deadline:
            break
        hp = h_best.copy()
        for _m in range(10):
            i = rng.integers(n)
            j = rng.integers(n)
            delta = rng.uniform(-pert_size, pert_size)
            hp[i] += delta
            hp[j] -= delta
        hp = _project_box_sum(hp, targets)
        sub_dl = min(deadline, time.time() + per_time)
        h_r, c5_r = _trust_loop(hp, sub_dl, sigma_start=0.04, max_iters=100,
                                 seed=rng.integers(10000))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5
