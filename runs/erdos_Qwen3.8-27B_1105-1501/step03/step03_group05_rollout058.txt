def _project_box_sum(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Binary search for the threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    F = scores * dx
    G = np.zeros((2*n-1, n))

        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return F, G

def _lp_minimax_step(h, trust_radius=None, solver='ECOS'):
    """Solve the LP (or SOCP with trust region) to find best x."""

        if i != j and h_pert[i] > 0.05 and h_pert[j] < 0.95:
            eps = min(h_pert[i], 1.0 - h_pert[j]) * 0.4
            h_pert[i] -= eps
            h_pert[j] += eps
    candidates.append(_project_box_sum(h_pert, target))
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: LP-based optimization from parent (35% time)
    deadline1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_refine(h, deadline1, max_iters=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Gradient refinement (10% time)
    deadline2 = t0 + budget_s * 0.40
    h2, c5_2 = _gradient_refine(best_h, deadline2, max_iters=3000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Multi-start diversification (20% time)
    deadline3 = t0 + budget_s * 0.60
    starts = _make_diverse_starts(best_h, rng)
    per_start_time = max(5.0, (deadline3 - time.time()) / max(1, len(starts)))
    for s in starts:
        if time.time() > deadline3:
            break
        d_s = min(deadline3, time.time() + per_start_time)
        h_s, c5_s = _lp_refine(s, d_s, max_iters=1000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 4: More LP from best (10% time)
    deadline4 = t0 + budget_s * 0.70
    h4, c5_4 = _lp_refine(best_h, deadline4, max_iters=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Binary local search (12% time)
    deadline5 = t0 + budget_s * 0.82
    hb, c5b = _binary_local_search(best_h, deadline5, max_iters=80000, rng=rng)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()
        best_n = len(best_h)
        # Refine the binary result
        if time.time() < deadline5:
            h_br, c5_br = _lp_refine(best_h, deadline5, max_iters=500)
            if c5_br < best_c5:
                best_c5 = c5_br
                best_h = h_br.copy()
    
    # Phase 6: Resolution exploration (8% time)
    deadline6 = t0 + budget_s * 0.90
    n_candidates = sorted(set([best_n - 8, best_n - 4, best_n + 4, best_n + 8,
                                best_n + 16, 48, 52, 56, 60, 64, 68, 72, 76, 80, 88, 96]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 128 and x != best_n]
    time_per_n = max(3.0, (deadline6 - time.time()) / max(1, len(n_candidates)))
    for n_try in n_candidates:
        if time.time() > deadline6:
            break
        h_r = _integral_resample(best_h, n_try)
        d_r = min(deadline6, time.time() + time_per_n)
        h_r, c5_r = _lp_refine(h_r, d_r, max_iters=500)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 7: Final LP polish (5% time)
    deadline7 = t0 + budget_s * 0.95
    h7, c5_7 = _lp_refine(best_h, deadline7, max_iters=2000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Phase 8: Pair transfer polish (3% time)
    deadline8 = t0 + budget_s * 0.97
    h8, c5_8 = _pair_transfer_refine(best_h, deadline8, max_iters=20000, rng=np.random.default_rng(seed + 7))
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    
    # Final projection and evaluation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
