def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _corr_and_max(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _subgrad_from_corr(h, corr, max_s, eps=1e-5):
    n = len(h)
    dx = 2.0 / n
    active = np.where(corr >= max_s - eps)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    w = 1.0 / len(active)
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad[js] += -h[js + d]
        lo1 = max(d, 0)
        hi1 = min(n - 1 + d, n - 1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += (1.0 - h[js - d])

def _nesterov_loop(h, deadline, lr_init=0.01, beta=0.9, active_eps=1e-5, restart_every=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    y = h.copy()
    momentum = np.zeros(n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    it = 0
    
    while time.time() < deadline:
        it += 1
        if it % restart_every == 0:
            y = h.copy()
            momentum = np.zeros(n)
        
        corr, max_s = _corr_and_max(h)
        grad = _subgrad_from_corr(h, corr, max_s, active_eps)
        
        # Nesterov step: y = h + beta * (h - h_prev) -- simplified as momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_new = y - lr * momentum
        h_new = _project(h_new, target)

        if c5_new < cur_c5 - 1e-12:
            y = h.copy()
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            stall += 1
            lr *= 0.95
            if stall > 100:
                # Restart with perturbation
                h = best_h + np.random.normal(0, 0.01, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                y = h.copy()
                momentum = np.zeros(n)
                lr = lr_init
                stall = 0
            if lr < lr_init * 0.01:
                # Reset learning rate
                lr = lr_init
                y = h.copy()
                momentum = np.zeros(n)
    
    return best_h, best_c5

def _pair_coord_descent(h, deadline, n_sweeps=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Precompute gradient matrix G
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        lo2 = max(0, -d)
        hi2 = min(n-1, n-1-d)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2+1)
            G[k, js] += -h[js+d]
        lo1 = max(d, 0)
        hi1 = min(n-1+d, n-1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1+1)
            G[k, js] += (1.0 - h[js-d])
    
    improved_total = True
    sweep = 0
    while time.time() < deadline and improved_total and sweep < n_sweeps:
        sweep += 1
        improved_total = False
        corr = np.correlate(h, 1.0-h, mode='full')
        
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                # Try h[i] -= alpha, h[j] += alpha
                # alpha range: [0, min(h[i], 1-h[j])]
                alpha_max = min(h[i], 1.0 - h[j])
                if alpha_max < 1e-10:
                    continue
                
                # G_k = G[k,j] - G[k,i]
                Gk = G[:, j] - G[:, i]
                
                # Find alpha in [0, alpha_max] minimizing max(corr + alpha*Gk)
                # Use ternary search (convex function)
                lo, hi = 0.0, alpha_max
                for _ in range(30):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = np.max(corr + m1 * Gk)
                    f2 = np.max(corr + m2 * Gk)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                
                alpha = (lo + hi) / 2.0
                if alpha < 1e-10:
                    continue
                
                h_new = h.copy()
                h_new[i] -= alpha
                h_new[j] += alpha
                h_new = np.clip(h_new, 0, 1)
                c5_new = _exact_c5(h_new)
                
                if c5_new < best_c5 - 1e-11:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved_total = True
                    # Update corr and G
                    corr = np.correlate(h, 1.0-h, mode='full')
                    # Recompute G row for affected columns
                    # (For efficiency, recompute all G)
                    for k in range(2*n-1):
                        d = k - (n-1)
                        lo2 = max(0, -d)
                        hi2 = min(n-1, n-1-d)

                        lo1 = max(d, 0)

                        if lo1 <= hi1:
                            js = np.arange(lo1, hi1+1)
                            G[k, js] += (1.0 - h[js-d])
    
    return best_h, best_c5
