import numpy as np
import time
import math
from scipy.optimize import linprog, minimize, differential_evolution

_USE_ANALYTIC_GRAD = True
_OPT_N = 0
_OPT_TARGET = 0.0
_OPT_BETA = 1.0
_PARAM_N = 0


def _scores(h):
    n = len(h)
    if n == 0:
        return np.zeros(0, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _exact_c5(h):
    s = _scores(h)
    if s.size == 0:
        return 0.0
    return float(np.max(s))


def _is_better(c5, n, bc5, bn):
    if c5 < bc5 - 1e-12:
        return True
    if abs(c5 - bc5) <= 1e-12 and n < bn:
        return True
    return False


def _project_box_sum(v, target):
    n = len(v)
    if n == 0:
        return v.copy()
    target = float(np.clip(target, 0.0, float(n)))
    x = np.clip(v, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x

    for _ in range(2 * n + 20):
        diff = target - x.sum()
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            add = diff / cnt
            if add <= (1.0 - np.max(x[mask])) + 1e-15:
                x[mask] += add
            else:
                x[mask] = 1.0
        else:
            mask = x > 1e-12
            if not np.any(mask):
                break
            cnt = int(np.sum(mask))
            sub = -diff / cnt
            if sub <= np.min(x[mask]) + 1e-15:
                x[mask] -= sub
            else:
                x[mask] = 0.0

    if abs(x.sum() - target) <= 1e-12:
        return x

    lo = float(min(np.min(v), 0.0)) - 1.0
    hi = float(max(np.max(v), 1.0)) + 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        xs = np.clip(v - mid, 0.0, 1.0)
        if xs.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if diff > 0.0:
        mask = x < 1.0 - 1e-14
    elif diff < 0.0:
        mask = x > 1e-14
    else:
        return x
    if np.any(mask):
        x[mask] += diff / float(np.sum(mask))
        x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if np.any(mask):
        x[mask] += diff / float(np.sum(mask))
        x = np.clip(x, 0.0, 1.0)
    return x


def _gradient_matrix_analytic(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=float)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:]) * dx
                G[r, d:] += -h[:m] * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m]) * dx
                G[r, :m] += -h[g:] * dx
    return G


def _gradient_matrix_fd(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    eps = 1e-6
    G = np.zeros((L, n), dtype=float)
    for j in range(n):
        hj = float(h[j])
        hp = h.copy()
        hm = h.copy()
        if hj <= eps:
            hp[j] = min(1.0, hj + eps)
            hm[j] = 0.0
            den = max(eps, hp[j])
        elif hj >= 1.0 - eps:
            hp[j] = 1.0
            hm[j] = max(0.0, hj - eps)
            den = max(eps, 1.0 - hm[j])
        else:
            hp[j] = hj + eps
            hm[j] = hj - eps
            den = 2.0 * eps
        sp = np.correlate(hp, 1.0 - hp, mode="full") * dx
        sm = np.correlate(hm, 1.0 - hm, mode="full") * dx
        G[:, j] = (sp - sm) / den
    return G


def _gradient_matrix(h):
    if _USE_ANALYTIC_GRAD:
        return _gradient_matrix_analytic(h)
    return _gradient_matrix_fd(h)


def _test_gradient():
    rng = np.random.default_rng(987654321)
    n = 32
    h = 0.15 + 0.70 * rng.random(n)
    h = _project_box_sum(h, n / 2.0)
    try:
        ga = _gradient_matrix_analytic(h)
        gf = _gradient_matrix_fd(h)
        if ga.shape != gf.shape:
            return False
        return bool(np.max(np.abs(ga - gf)) < 1e-5)
    except Exception:
        return False


def _smooth_value(h, beta):
    s = _scores(h)
    m = float(np.max(s))
    e = np.exp(beta * (s - m))
    ss = float(e.sum())
    if ss <= 1e-300:
        return m
    return m + math.log(ss) / beta


def _smooth_grad(h, beta):
    s = _scores(h)
    m = float(np.max(s))
    e = np.exp(beta * (s - m))
    ss = float(e.sum())
    if ss <= 1e-300:
        w = np.zeros_like(s)
        w[int(np.argmax(s))] = 1.0
    else:
        w = e / ss
    return _gradient_matrix(h).dot(w)


def _eval_alpha(h, d, a, target):
    if a <= 0.0:
        x = h.copy()
        return _exact_c5(x), x
    x = h + a * d
    if x.min() < -1e-12 or x.max() > 1.0 + 1e-12 or abs(x.sum() - target) > 1e-8:
        x = _project_box_sum(x, target)
    else:
        x = np.clip(x, 0.0, 1.0)
    return _exact_c5(x), x


def _line_search_feasible(h, d, cur, target, alpha_hi=1.0, n_grid=32):
    if alpha_hi <= 1e-14:
        return 0.0, cur, None
    d = np.asarray(d, dtype=float)
    best_a = 0.0
    best_c = cur
    best_x = None
    best_idx = 0
    alphas = np.linspace(0.0, alpha_hi, n_grid)
    for idx in range(1, n_grid):
        a = float(alphas[idx])
        c, x = _eval_alpha(h, d, a, target)
        if c < best_c - 1e-13:
            best_c = c
            best_a = a
            best_x = x.copy()
            best_idx = idx
    if best_x is None:
        return 0.0, cur, None

    lo = float(alphas[best_idx - 1]) if best_idx > 0 else 0.0
    hi = float(alphas[best_idx + 1]) if best_idx + 1 < n_grid else float(alpha_hi)
    if hi - lo <= 1e-12:
        return best_a, best_c, best_x

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, _ = _eval_alpha(h, d, c1, target)
    f2, _ = _eval_alpha(h, d, c2, target)

    for _ in range(25):
        if hi - lo < 1e-8 * max(1.0, alpha_hi):
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1, _ = _eval_alpha(h, d, c1, target)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2, _ = _eval_alpha(h, d, c2, target)

    opt_a = 0.5 * (lo + hi)
    c, x = _eval_alpha(h, d, opt_a, target)
    if c < best_c - 1e-13:
        best_c = c
        best_a = opt_a
        best_x = x.copy()
    return best_a, best_c, best_x


def _smooth_refine(h, deadline, beta_start=5.0, beta_end=300.0, n_beta=8, max_steps=220):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    if time.time() >= deadline or n <= 4:
        return best_h, best_c

    betas = np.geomspace(float(beta_start), float(beta_end), max(1, int(n_beta)))
    for beta in betas:
        if time.time() >= deadline:
            break
        fail = 0
        while time.time() < deadline and fail < 22:
            g = _smooth_grad(h, float(beta))
            d = -g
            d -= float(np.mean(d))
            m = float(np.max(np.abs(d)))
            if m < 1e-13:
                break
            d = d / m
            a, c, x = _line_search_feasible(h, d, cur, target, 1.0, 32)
            if x is not None and c < cur - 1e-12:
                h = x
                cur = c
                if _is_better(cur, n, best_c, n):
                    best_c = cur
                    best_h = h.copy()
                fail = 0
            else:
                fail += 1
    return best_h, best_c


def _lp_d_lininf(h, r):
    n = len(h)
    if n <= 0 or r <= 0.0:
        return None
    L = 2 * n - 1
    G = _gradient_matrix(h)
    Gh = _scores(h)
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    A_ub = np.zeros((L, n + 1), dtype=float)
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -Gh

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=float)

    bounds = []
    for val in h:
        lo = float(max(-val, -r))
        hi = float(min(1.0 - val, r))
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, 2.5))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs",
                      options={"maxiter": 30000})
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="revised simplex")
        except Exception:
            return None

    if res is not None and getattr(res, "success", False):
        d = np.asarray(res.x[:n], dtype=float).copy()
        if np.all(np.isfinite(d)):
            return d
    return None


def _lp_refine(h, deadline, r_start=0.15, r_max=0.5, fail_limit=70):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    if time.time() >= deadline or n <= 4:
        return best_h, best_c

    r = float(r_start)
    fail = 0
    zero = 0
    while time.time() < deadline and r > 1e-5 and fail < fail_limit and zero < 45:
        d = _lp_d_lininf(h, r)
        if d is None:
            fail += 1
            r *= 0.70
            continue
        s = float(np.sum(d))
        if abs(s) > 1e-10:
            d = d - s / n
        m = float(np.max(np.abs(d)))
        if m < 1e-9:
            if r < r_max:
                r = min(r_max, r * 1.35)
                zero += 1
            else:
                fail += 1
                r *= 0.70
            continue
        a, c, x = _line_search_feasible(h, d, cur, target, 1.6, 40)
        if x is not None and c < cur - 1e-11:
            h = x
            cur = c
            if _is_better(cur, n, best_c, n):
                best_c = cur
                best_h = h.copy()
            r = min(r_max, max(r, r * 1.06))
            fail = 0
            zero = 0
        else:
            fail += 1
            r *= 0.70
    return best_h, best_c


def _opt_obj(h):
    return _smooth_value(h, _OPT_BETA)


def _opt_grad(h):
    return _smooth_grad(h, _OPT_BETA)


def _opt_con_fun(h):
    return np.array([np.sum(h) - _OPT_TARGET], dtype=float)


def _opt_con_jac(h):
    return np.ones(len(h), dtype=float)


def _slsqp_refine(h, deadline, betas=(5.0, 20.0, 80.0, 300.0)):
    global _OPT_N, _OPT_TARGET, _OPT_BETA
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    if time.time() >= deadline or n <= 4:
        return best_h, best_c

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _opt_con_fun, "jac": _opt_con_jac}]

    for beta in betas:
        if time.time() >= deadline:
            break
        _OPT_N = n
        _OPT_TARGET = float(target)
        _OPT_BETA = float(beta)
        try:
            res = minimize(_opt_obj, h, method="SLSQP", jac=_opt_grad,
                           bounds=bounds, constraints=constraints,
                           options={"maxiter": 55, "ftol": 1e-12, "disp": False})
            x = np.asarray(res.x, dtype=float)
            x = _project_box_sum(x, target)
            c = _exact_c5(x)
            if c < cur - 1e-12:
                h = x
                cur = c
                if _is_better(cur, n, best_c, n):
                    best_c = cur
                    best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c


def _best_alignment(h):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    if n <= 4:
        return h.copy(), _exact_c5(h)
    best_h = h.copy()
    best_c = _exact_c5(h)
    rev = h[::-1].copy()
    c = _exact_c5(rev)
    if _is_better(c, n, best_c, n):
        best_c = c
        best_h = rev.copy()
    for s in range(1, n):
        x = np.roll(h, s)
        c = _exact_c5(x)
        if _is_better(c, n, best_c, n):
            best_c = c
            best_h = x.copy()
        x = np.roll(rev, s)
        c = _exact_c5(x)
        if _is_better(c, n, best_c, n):
            best_c = c
            best_h = x.copy()
    return best_h, best_c


def _binary_refine(h, deadline, rng):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    if n % 2 != 0 or n < 10:
        return h.copy(), _exact_c5(h)
    target = n // 2
    b = np.zeros(n, dtype=float)
    score = -h + 1e-9 * rng.uniform(0.0, 1.0, n)
    idx = np.argsort(score)
    b[idx[:target]] = 1.0
    cur = _exact_c5(b)
    best_h = b.copy()
    best_c = cur
    ones = [int(i) for i in np.where(b == 1.0)[0]]
    zeros = [int(i) for i in np.where(b == 0.0)[0]]

    for _pass in range(40):
        if time.time() >= deadline:
            break
        improved = False
        order_i = list(ones)
        order_j = list(zeros)
        rng.shuffle(order_i)
        rng.shuffle(order_j)
        for i in order_i:
            if time.time() >= deadline:
                break
            if b[i] != 1.0:
                continue
            for j in order_j:
                if i == j or b[j] != 0.0:
                    continue
                b[i] = 0.0
                b[j] = 1.0
                c = _exact_c5(b)
                if c < cur - 1e-13:
                    cur = c
                    if _is_better(c, n, best_c, n):
                        best_c = c
                        best_h = b.copy()
                    ones = [int(x) for x in np.where(b == 1.0)[0]]
                    zeros = [int(x) for x in np.where(b == 0.0)[0]]
                    improved = True
                    break
                b[i] = 1.0
                b[j] = 0.0
            if improved:
                break
        if not improved:
            break

    T = 0.01
    min_T = 1e-5
    no_improve = 0
    it = 0
    while time.time() < deadline and no_improve < 5000 and ones and zeros:
        it += 1
        u = float(rng.random())
        accepted = False

        if u < 0.75:
            i = int(ones[int(rng.integers(len(ones)))])
            j = int(zeros[int(rng.integers(len(zeros)))])
            if i == j or b[i] != 1.0 or b[j] != 0.0:
                continue
            b[i] = 0.0
            b[j] = 1.0
            c = _exact_c5(b)
            delta = c - cur
            if delta < 0.0 or (delta < 0.05 and float(rng.random()) < math.exp(-delta / T)):
                accepted = True
                cur = c
                if _is_better(c, n, best_c, n):
                    best_c = c
                    best_h = b.copy()
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
            else:
                b[i] = 1.0
                b[j] = 0.0

        elif u < 0.90:
            i = int(ones[int(rng.integers(len(ones)))])
            cand = []
            if i > 0 and b[i - 1] == 0.0:
                cand.append(i - 1)
            if i + 1 < n and b[i + 1] == 0.0:
                cand.append(i + 1)
            if not cand:
                no_improve += 1
                continue
            j = int(cand[int(rng.integers(len(cand)))])
            b[i] = 0.0
            b[j] = 1.0
            c = _exact_c5(b)
            delta = c - cur
            if delta < 0.0 or (delta < 0.05 and float(rng.random()) < math.exp(-delta / T)):
                accepted = True
                cur = c
                if _is_better(c, n, best_c, n):
                    best_c = c
                    best_h = b.copy()
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
            else:
                b[i] = 1.0
                b[j] = 0.0

        else:
            L = 2 + int(rng.integers(0, max(2, n // 4)))
            if L > n:
                L = n
            if L < 2:
                no_improve += 1
                continue
            a = int(rng.integers(0, n - L + 1))
            old = b[a:a + L].copy()
            b[a:a + L] = old[::-1]
            c = _exact_c5(b)
            delta = c - cur
            if delta < 0.0 or (delta < 0.05 and float(rng.random()) < math.exp(-delta / T)):
                accepted = True
                cur = c
                if _is_better(c, n, best_c, n):
                    best_c = c
                    best_h = b.copy()
                ones = [int(x) for x in np.where(b == 1.0)[0]]
                zeros = [int(x) for x in np.where(b == 0.0)[0]]
            else:
                b[a:a + L] = old

        if accepted:
            no_improve = 0
        else:
            no_improve += 1
        if it % 1000 == 0:
            T = max(min_T, T * 0.95)
    return best_h, best_c


def _resample(h, n_new):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n_new == n:
        return h.copy()
    if n <= 0:
        return np.zeros(n_new, dtype=float)
    x_old = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h)
    return _project_box_sum(y, n_new / 2.0)


def _random_binary(n, rng):
    if n % 2 != 0 or n <= 0:
        return np.zeros(n, dtype=float)
    idx = rng.permutation(n)
    b = np.zeros(n, dtype=float)
    b[idx[:n // 2]] = 1.0
    return b


def _thue_binary(n, rng=None):
    if n <= 0:
        return np.zeros(0, dtype=float)
    a = np.array([1.0])
    while len(a) < n:
        a = np.concatenate((a, 1.0 - a))
    a = a[:n].astype(float)
    target = n // 2 if n % 2 == 0 else (n - 1) // 2
    if target <= 0:
        return np.zeros(n, dtype=float)
    score = -a
    if rng is not None:
        score = score + 1e-8 * rng.uniform(0.0, 1.0, n)
    idx = np.argsort(score)
    b = np.zeros(n, dtype=float)
    b[idx[:target]] = 1.0
    return b


def _default_start(n=64):
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = 0.5 + 0.35 * np.cos(np.pi * x)
    return _project_box_sum(h, n / 2.0)


def _param_to_h(params):
    n = int(_PARAM_N)
    if n <= 0:
        return np.zeros(0, dtype=float)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    p = np.asarray(params, dtype=float)
    h = 0.5
    for i in range(min(len(p), 5)):
        h = h + p[i] * np.cos((i + 1) * np.pi * x)
    return _project_box_sum(h, n / 2.0)


def _param_obj(params):
    h = _param_to_h(params)
    return _exact_c5(h)


def _param_search(n, local_s, rng):
    global _PARAM_N
    if local_s <= 2 or n <= 16:
        return None
    best_h = None
    best_c = float("inf")

    for _ in range(max(8, min(40, int(local_s)))):
        p = rng.uniform(-0.55, 0.55, 5)
        h = _param_to_h(p)
        c = _exact_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    maxfev = max(80, min(500, int(local_s * 25)))
    _PARAM_N = n
    bound = [(-0.60, 0.60) for _ in range(5)]
    seed = int(rng.integers(0, 1000000))
    try:
        res = differential_evolution(_param_obj, bound, seed=seed, maxfev=maxfev,
                                     maxiter=max(10, maxfev // 20),
                                     popsize=8, tol=1e-8, polish=True)
        if res is not None:
            h = _param_to_h(res.x)
            c = _exact_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    except Exception:
        pass
    return best_h


def _n_start_list(n, parent, rng):
    starts = []
    parent = np.asarray(parent, dtype=float)
    if n == len(parent):
        starts.append(parent.copy())
    else:
        starts.append(_resample(parent, n))

    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    for a in (0.25, 0.35, 0.45):
        starts.append(_project_box_sum(0.5 + a * np.cos(np.pi * x), n / 2.0))
    for a in (0.45, 0.75):
        starts.append(_project_box_sum(a * (1.0 - np.abs(x - 1.0)), n / 2.0))

    if n % 2 == 0 and n > 0:
        starts.append(_random_binary(n, rng))
        starts.append(_thue_binary(n, rng))
        p = np.tile([1.0, 0.0], n // 2)
        starts.append(p.astype(float).copy())
        if n % 4 == 0:
            p = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
            starts.append(p.astype(float).copy())
    return starts[:14]


def _quick_polish(h, deadline):
    h = np.asarray(h, dtype=float).copy()
    t = time.time()
    total = max(0.0, deadline - t)
    if total <= 3:
        return h, _exact_c5(h)
    t1 = t + 0.45 * total
    h1, c1 = _smooth_refine(h, t1, 5.0, 160.0, 5, 90)
    h2, c2 = _lp_refine(h1, deadline, 0.07, 0.35, 35)
    return h2, c2


def run(seed=42, budget_s=1100, **kwargs):
    global _USE_ANALYTIC_GRAD
    t0 = time.time()
    budget = max(1.0, float(budget_s))
    deadline = t0 + 0.96 * budget
    rng = np.random.default_rng(seed)

    if _USE_ANALYTIC_GRAD is None:
        _USE_ANALYTIC_GRAD = _test_gradient()

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) == 0 or not np.all(np.isfinite(h0)):
        h0 = _default_start(64)
    h0 = _project_box_sum(h0, len(h0) / 2.0)

    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)
    best_n = len(best_h)

    if time.time() < deadline:
        best_h, best_c5 = _best_alignment(best_h)
        best_n = len(best_h)

    param_deadline = min(deadline, t0 + min(35.0, 0.04 * budget))
    if time.time() < param_deadline:
        ph = _param_search(best_n, max(0.0, param_deadline - time.time()), rng)
        if ph is not None:
            ph, pc = _best_alignment(ph)
            if _is_better(pc, len(ph), best_c5, best_n):
                best_h = ph.copy()
                best_c5 = pc
                best_n = len(best_h)

    d1 = min(deadline, t0 + 0.18 * budget)
    h1, c1 = _smooth_refine(best_h, d1, 4.0, 220.0, 10, 260)
    if _is_better(c1, len(h1), best_c5, best_n):
        best_h = h1.copy()
        best_c5 = c1
        best_n = len(best_h)

    best_h, best_c5 = _best_alignment(best_h)
    best_n = len(best_h)

    d2 = min(deadline, t0 + 0.38 * budget)
    h2, c2 = _lp_refine(best_h, d2, 0.18, 0.55, 80)
    if _is_better(c2, len(h2), best_c5, best_n):
        best_h = h2.copy()
        best_c5 = c2
        best_n = len(best_h)

    best_h, best_c5 = _best_alignment(best_h)
    best_n = len(best_h)

    d3 = min(deadline, t0 + 0.52 * budget)
    h3, c3 = _slsqp_refine(best_h, d3, (5.0, 25.0, 100.0, 350.0))
    if _is_better(c3, len(h3), best_c5, best_n):
        best_h = h3.copy()
        best_c5 = c3
        best_n = len(best_h)

    best_h, best_c5 = _best_alignment(best_h)
    best_n = len(best_h)

    d4 = min(deadline, t0 + 0.62 * budget)
    h4, c4 = _lp_refine(best_h, d4, 0.08, 0.40, 80)
    if _is_better(c4, len(h4), best_c5, best_n):
        best_h = h4.copy()
        best_c5 = c4
        best_n = len(best_h)

    best_h, best_c5 = _best_alignment(best_h)
    best_n = len(best_h)

    d5 = min(deadline, t0 + 0.80 * budget)
    hb, cb = _binary_refine(best_h, d5, rng)
    if _is_better(cb, len(hb), best_c5, best_n):
        best_h = hb.copy()
        best_c5 = cb
        best_n = len(best_h)

    d6 = min(deadline, t0 + 0.88 * budget)
    h6, c6 = _smooth_refine(best_h, d6, 10.0, 600.0, 7, 120)
    if _is_better(c6, len(h6), best_c5, best_n):
        best_h = h6.copy()
        best_c5 = c6
        best_n = len(best_h)

    best_h, best_c5 = _best_alignment(best_h)
    best_n = len(best_h)

    h7, c7 = _lp_refine(best_h, min(deadline, t0 + 0.94 * budget), 0.05, 0.30, 50)
    if _is_better(c7, len(h7), best_c5, best_n):
        best_h = h7.copy()
        best_c5 = c7
        best_n = len(best_h)

    if time.time() < deadline - 20.0:
        candidates = set()
        base_list = (24, 32, 40, 48, 56, 64, 72, 80, 96, 112, 128)
        for x in base_list:
            candidates.add(int(x))
        for x in (best_n - 16, best_n - 8, best_n - 4, best_n + 4, best_n + 8, best_n + 16):
            xi = int(x)
            if 24 <= xi <= 128 and xi % 2 == 0:
                candidates.add(xi)
        order = sorted(candidates)
        if best_n in order:
            order.remove(best_n)
        order = order[:9]

        polish_count = 0
        max_polish = 7
        for n_try in order:
            if time.time() >= deadline - 20.0 or polish_count >= max_polish:
                break
            starts = _n_start_list(n_try, best_h, rng)
            if n_try in (32, 48, 64, 80) and time.time() < deadline - 60.0:
                ph = _param_search(n_try, max(0.0, min(25.0, deadline - time.time() - 30.0)), rng)
                if ph is not None:
                    starts.append(ph)
            for hs in starts:
                if time.time() >= deadline - 15.0 or polish_count >= max_polish:
                    break
                hs = _project_box_sum(np.asarray(hs, dtype=float), n_try / 2.0)
                if time.time() < deadline - 8.0:
                    hs, sc = _best_alignment(hs)
                else:
                    sc = _exact_c5(hs)
                if _is_better(sc, n_try, best_c5, best_n):
                    best_h = hs.copy()
                    best_c5 = sc
                    best_n = n_try
                    polish_count += 1
                    continue
                if sc > best_c5 + 0.02 and float(rng.random()) > 0.25:
                    continue
                subd = min(deadline - 5.0, time.time() + min(25.0, 0.018 * budget))
                if subd <= time.time() + 3.0:
                    continue
                hp, cp = _quick_polish(hs, subd)
                polish_count += 1
                if hp is not None:
                    hp, cp = _best_alignment(hp)
                    if _is_better(cp, len(hp), best_c5, best_n):
                        best_h = hp.copy()
                        best_c5 = cp
                        best_n = len(best_h)

    if time.time() < deadline - 2.0:
        best_h, best_c5 = _best_alignment(best_h)
        best_n = len(best_h)
        hf, cf = _lp_refine(best_h, deadline - 0.2, 0.035, 0.18, 60)
        if _is_better(cf, len(hf), best_c5, best_n):
            best_h = hf.copy()
            best_c5 = cf
            best_n = len(best_h)
        best_h, best_c5 = _best_alignment(best_h)
        best_n = len(best_h)

    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, int(len(best_h))
