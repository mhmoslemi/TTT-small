def _penalty_direction(h, G, scores, radius=0.1, k_active=10, mu=1.0, gamma=5.0):
    """Convex penalty subproblem to find descent direction"""
    n = len(h)
    s_max = float(np.max(scores))
    
    # Select active rows
    idx = np.argsort(scores)[::-1][:k_active]
    G_act = G[idx]  # (k_active, n)
    s_act = scores[idx]
    
    # Allowed reduction for each active row
    target_reduction = 0.02 * s_max  # want to reduce max by this amount
    allowed = s_max - s_act - target_reduction  # slack - required reduction
    allowed = np.maximum(allowed, -target_reduction)  # can be negative (requiring reduction)
    
    # Weights: more weight on closer-to-max rows
    slacks = s_max - s_act
    w = np.maximum(1.0 - slacks / max(0.1 * s_max, 1e-9), 0.1)
    
    # Solve: min 0.5*mu*||p||^2 + 0.5*gamma*sum w_r * max(0, G_r·p - allowed_r)^2
    # subject to sum(p) = 0, |p_i| <= radius
    
    # Use projected gradient on the subproblem
    p = np.zeros(n)
    lr_sub = 1.0 / max(mu + gamma * np.sum(w) * np.max(np.sum(G_act**2, axis=1)), 1e-6)
    
    for it in range(50):
        # Gradient of subproblem
        viol = G_act @ p - allowed  # (k_active,)
        pos_viol = np.maximum(viol, 0.0)
        grad = mu * p + gamma * (G_act.T @ (w * pos_viol))
        
        # Project gradient onto zero-sum subspace
        grad = grad - np.mean(grad)
        
        # Gradient step
        p_new = p - lr_sub * grad
        
        # Project onto box and zero-sum
        p_new = np.clip(p_new, -radius, radius)
        p_new = p_new - np.mean(p_new)  # approximate zero-sum
        p_new = np.clip(p_new, -radius, radius)
        
        p = p_new
    
    # Ensure exact zero sum
    p = p - np.mean(p)
    
    if np.max(np.abs(p)) < 1e-13:
        return None
    return p

def _grad_direction(h, G, scores, radius=0.1, k_active=8):

    p = p - np.mean(p)  # re-zero after clip

    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = 0.1
    
    while time.time() < deadline and stall < 80:
        scores = _scores(h)
        G = _grad_matrix(h)
        
        # Try multiple direction methods
        directions = []
        
        p1 = _lp_direction(h, G, scores, radius, active_frac=0.2)
        if p1 is not None:
            directions.append(p1)
        
        p2 = _penalty_direction(h, G, scores, radius=radius, k_active=10)
        if p2 is not None:
            directions.append(p2)
        
        p3 = _grad_direction(h, G, scores, radius=radius, k_active=8)
        if p3 is not None:
            directions.append(p3)
        
        improved = False
        for p in directions:
            if time.time() > deadline:
                break
            alpha, new_c5 = _line_search_exact(h, p, cur_c5)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                radius = min(radius * 1.1, 0.4)
        
        if improved:
            continue
        
        # No improvement: reduce radius and add perturbation
        stall += 1
        radius *= 0.8
        if radius < 0.005:
            # Escape: small random perturbation
            rng = np.random.default_rng(int(time.time() * 1000) % 100000)
            h_pert = h + rng.standard_normal(n) * 0.02
            h_pert = _project(h_pert, target)
            if _exact_c5(h_pert) < cur_c5 + 0.01:  # allow small increase
                h = h_pert
                cur_c5 = _exact_c5(h)
        if radius < 1e-5:
            break
    
    return best_h, best_c5

def _momentum_descent(h, target, deadline, beta=30.0, lr_init=0.02, momentum=0.85):
    """Nesterov momentum on smooth max"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    lr = lr_init
    T = 0.003
    
    while time.time() < deadline:
        y = h + momentum * v
        scores = _scores(y)
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        wts = exp_s / np.sum(exp_s)
        G = _grad_matrix(y)
        grad = G.T @ wts
        grad = grad - np.mean(grad)
        
        v = momentum * v + lr * grad
        h_new = _project(h - v, target)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h)
        
        delta = c5_new - c5_cur
        accept = delta < 0
        if not accept and T > 1e-7:
            if np.random.random() < np.exp(-min(delta / T, 100)):
                accept = True
        
        if accept:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.65
            v *= 0.3
        
        T *= 0.997
        if lr < 1e-8:
            break
    
    return best_h, best_c5

def _fourier_start(n, seed, k_range=(2, 12)):
    """Generate a diverse starting point using Fourier modes"""
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    h = np.zeros(n)
    for _ in range(3):
        k = rng.integers(k_range[0], k_range[1])
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.cos(2 * np.pi * k * x + phase)
    h = 0.5 + h
    return _project(h, n / 2.0)

def _box_start(n, seed):
    """Generate a box-like starting point"""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    # Random box positions and widths
    n_boxes = rng.integers(2, 5)
    for _ in range(n_boxes):
        start = rng.integers(0, n)
        width = rng.integers(n // 4, n // 2)
        h[start:start+width] += rng.uniform(0.3, 0.8)
    h = np.clip(h, 0, 1)
    return _project(h, n / 2.0)
