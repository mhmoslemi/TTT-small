    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        # G[r, i] = (1-h[i+d]) - h[i-d] with h[x]=0 for x outside [0,n-1]
        # First term: (1-h[i+d]) for valid i+d

        idx2 = np.arange(n) - d
