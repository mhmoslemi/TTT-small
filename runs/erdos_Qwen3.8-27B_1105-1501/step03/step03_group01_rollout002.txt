        if time.time() > deadline:
            raise TimeoutError()
        val, grad = soft_value_grad(x, tau)
        return val, grad
    
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_wrap, h0.copy(), jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x, target)
            return h_new
    except (TimeoutError, Exception):
        pass
    return h0

def smooth_phase(h, deadline, tau_list=None):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    if tau_list is None:
        if n <= 100:
            tau_list = [0.5, 0.2, 0.1, 0.05, 0.025, 0.01, 0.005]
        elif n <= 200:
            tau_list = [0.4, 0.15, 0.08, 0.04, 0.02, 0.01]
        else:
            tau_list = [0.3, 0.12, 0.06, 0.03, 0.015]
    for tau in tau_list:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        sub_dl = time.time() + min(remaining, budget_per_tau(remaining, len(tau_list)))
        h = smooth_solve(h, tau, sub_dl, maxiter=slsqp_maxiter(n))
        h = make_feasible(h, n / 2.0)
        c5 = exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_maxiter(n):
    if n <= 90:
        return 350
    elif n <= 150:
        return 200
    else:
        return 100

def budget_per_tau(remaining, n_taus):
    return max(remaining / max(n_taus, 1) * 1.2, 5.0)

def pair_polish(h, deadline, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(seed)
    
    for passes in range(15):
        if time.time() > deadline:
            break
        improved = False
        # Adjacent pairs
        for i in range(n - 1):
            if time.time() > deadline:
                break
            if h[i] < 1e-10 or h[i] > 1.0 - 1e-10:
                continue
            if h[i+1] < 1e-10 or h[i+1] > 1.0 - 1e-10:
                continue
            for sign in [1.0, -1.0]:
                amax = min((1.0 - h[i]) if sign > 0 else h[i],
                           h[i+1] if sign > 0 else (1.0 - h[i+1]))
                if amax < 1e-12:
                    continue
                best_a = 0.0
                best_f = best_c5
                for frac in [0.4, 1.0]:
                    a = amax * frac
                    ht = h.copy()
                    ht[i] += sign * a
                    ht[i+1] -= sign * a
                    f = exact_c5(ht)
                    if f < best_f:
                        best_f = f
                        best_a = a
                if best_a > 1e-10:
                    ht = h.copy()
                    ht[i] += sign * best_a
                    ht[i+1] -= sign * best_a
                    ht = make_feasible(ht, target)
                    c5 = exact_c5(ht)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        best_h = ht.copy()
                        h = ht.copy()
                        improved = True
        # Random pairs
        n_rand = min(300, n * 4)
        for _ in range(n_rand):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if h[i] < 1e-10 or h[i] > 1.0 - 1e-10:
                continue
            if h[j] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            for sign in [1.0, -1.0]:
                amax = min((1.0 - h[i]) if sign > 0 else h[i],
                           h[j] if sign <= 0 else (1.0 - h[j]))
                if amax < 1e-12:
                    continue
                for frac in [0.5, 1.0]:
                    a = amax * frac
                    ht = h.copy()
                    ht[i] += sign * a
                    ht[j] -= sign * a
                    f = exact_c5(ht)
                    if f < best_c5 - 1e-12:
                        ht = make_feasible(ht, target)
                        f2 = exact_c5(ht)
                        if f2 < best_c5:
                            best_c5 = f2
                            best_h = ht.copy()
                            h = ht.copy()
                            improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new / 2.0)

def grad_step(h, deadline, tau=0.1, lr=0.02, beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        _, grad = soft_value_grad(h, tau)
        mom = beta * mom + (1.0 - beta) * grad
        h_new = make_feasible(h - lr * mom, target)
        c5_new = exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.08, 0.15)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
            if stall % 20 == 0:
                mom = np.zeros(n)
                h = best_h + np.random.default_rng(int(time.time()*1000) % 10000).normal(0, 0.02, n)
                h = make_feasible(h, target)
                best_c5 = exact_c5(h)
                best_h = h.copy()
                stall = 0
    return best_h, best_c5

def gen_periodic(n, seed, period):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase + rng.uniform(0, 2*np.pi))
    return make_feasible(h, n / 2.0)

def gen_fourier(n, seed, n_modes=5):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    h = np.full(n, 0.5)
    for m in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.25) / np.sqrt(m)
        freq = m * rng.integers(1, 4)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x + ph)
    return make_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.95
    
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = gen_periodic(84, 42, 7)
    
    n_p = len(h_parent)
    h_parent = make_feasible(h_parent, n_p / 2.0)
    best_h = h_parent.copy()
    best_c5 = exact_c5(h_parent)
    best_n = n_p
    rng = np.random.default_rng(seed)
    
    def update_best(h_new, c5_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = len(h_new)
    
    # Phase 1: Smooth SLSQP on parent (30% budget)
    d1 = t0 + budget_s * 0.30
    h, c5 = smooth_phase(best_h, d1)
    update_best(h, c5)
    
    # Phase 2: Gradient refinement (10%)
    if time.time() < t0 + budget_s * 0.42:
        d2 = t0 + budget_s * 0.42
        h, c5 = grad_step(best_h, d2, tau=0.08, lr=0.02, beta=0.85)
        update_best(h, c5)
        h, c5 = smooth_phase(best_h, min(d2, time.time() + budget_s * 0.04),
                            tau_list=[0.03, 0.015, 0.008])
        update_best(h, c5)
    
    # Phase 3: Multi-resolution (25%)
    d3 = t0 + budget_s * 0.67
    offsets = [0, -2, 2, -4, 4, -8, 8, -12, 12, -16, 16, -24, 24, 32, -32, 40]
    tried_n = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = smooth_phase(h_r, sub_dl)
        if time.time() < d3 and c5_r < best_c5 + 0.01:
            h_r, c5_r = grad_step(h_r, min(d3, time.time() + budget_s * 0.01),
                                   tau=0.05, lr=0.015)
        update_best(h_r, c5_r)
    
    # Phase 4: Diverse restarts (18%)
    d4 = t0 + budget_s * 0.85
    attempts = 0
    scale = 0.04
    while time.time() < d4 and attempts < 40:
        attempts += 1
        if best_c5 < 0.38060:
            break
        # Mix perturbation with fresh generation
        if attempts % 3 == 0 and best_n > 50:
            h_start = gen_fourier(best_n, seed + attempts * 13, n_modes=min(8, best_n // 12))
        elif attempts % 5 == 0 and best_n > 50:
            period = int(rng.integers(4, max(5, best_n // 8)))
            h_start = gen_periodic(best_n, seed + attempts * 7, period)
        else:
            h_start = best_h + rng.normal(0, scale, best_n)
        h_start = make_feasible(h_start, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_try, c5_try = smooth_phase(h_start, sub_dl,
                                     tau_list=[0.2, 0.08, 0.03, 0.01])
        if c5_try < best_c5 - 1e-11:
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.3, scale * 1.2)
        update_best(h_try, c5_try)
    
    # Phase 5: Pair polish (10%)
    d5 = t0 + budget_s * 0.94
    h_p, c5_p = pair_polish(best_h, d5, seed=seed + 99)
    update_best(h_p, c5_p)
    
    # Phase 6: Final smooth refinement on best (5%)
    if time.time() < hard_deadline:
        h_f, c5_f = smooth_phase(best_h, hard_deadline,
                                 tau_list=[0.01, 0.005])
        update_best(h_f, c5_f)
        if time.time() < hard_deadline - 5:
            h_pp, c5_pp = pair_polish(best_h, hard_deadline, seed=seed + 200)
            update_best(h_pp, c5_pp)
    
    # Final validation
    best_h = make_feasible(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = make_feasible(h_parent, n_p / 2.0)
        final_c5 = exact_c5(best_h)
        best_n = n_p
    
    return best_h, final_c5, best_n
