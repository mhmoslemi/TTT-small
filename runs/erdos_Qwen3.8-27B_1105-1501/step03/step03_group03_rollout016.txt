                #   if i in [0, m-1]: appears in term i as h[i+d]*(1-h[i]), coeff = -h[i+d]

        return h.copy()

    h_new = h + step_size * direction
    h_new = _project(h_new, n / 2.0)
    return h_new

            best_val = h_i

            for cv in candidates:

                    others = np.arange(n)

                if c5_t < best_c - 1e-12:

            if c5_new < best_c5 - 1e-12:

                best_h = h.copy()
                cur_c5 = c5_new
                improved_any = True
            else:
                h = best_h.copy()
                cur_c5 = best_c5
        
        if not improved_any and _round > 2:
            break
    
    return best_h, best_c5
