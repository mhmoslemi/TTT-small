import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def feasible_interval(h, D):
    """Compute [t_lo, t_hi] such that h + t*D stays in [0,1]^n."""
    n = len(h)
    t_lo = 0.0
    t_hi = 0.0
    mask_pos = D > 1e-14
    mask_neg = D < -1e-14
    if np.any(mask_pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[mask_pos]) / D[mask_pos])))
        t_lo = max(t_lo, float(np.max(-h[mask_pos] / D[mask_pos])))
    if np.any(mask_neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[mask_neg]) / D[mask_neg])))
        t_hi = min(t_hi, float(np.min(-h[mask_neg] / D[mask_neg])))
    # Ensure t=0 is included (it should be since h is feasible)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    if t_lo >= t_hi:
        return None
    return (t_lo, t_hi)

def line_objective(S, B, Q, t):
    """Compute f(t) = max_r (S[r] + t*B[r] + t^2*Q[r]) for scalar t."""
    vals = S + t * B + t * t * Q
    return float(np.max(vals))

def line_search(h, D, S, B, Q, dx, deadline, n_samples=25):
    """Exact line search along direction D. Returns (best_t, best_val) or (0, current)."""
    n = len(h)
    max_abs = float(np.max(np.abs(D)))
    if max_abs < 1e-14:
        return 0.0, float(np.max(S) * dx)
    
    D = D / max_abs  # normalize
    
    # Recompute B and Q with normalized D
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = -np.correlate(D, D, mode='full')
    
    interval = feasible_interval(h, D * max_abs)  # use original scale for feasibility
    if interval is None:
        return 0.0, float(np.max(S) * dx)
    
    t_lo, t_hi = interval
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(S) * dx)
    
    # Grid sampling
    ts = np.linspace(t_lo, t_hi, n_samples)
    # Include 0 if not already there
    ts = np.append(ts, 0.0)
    
    best_t = 0.0
    best_val = float(np.max(S) * dx)
    
    for t in ts:
        if t < t_lo - 1e-10 or t > t_hi + 1e-10:
            continue
        val = line_objective(S, B, Q, t) * dx
        if val < best_val - 1e-12:
            best_val = val
            best_t = t
    
    if best_t == 0.0 and best_val >= float(np.max(S) * dx) - 1e-12:
        # No improvement found in grid
        return 0.0, float(np.max(S) * dx)
    
    # Golden section refinement around best grid point
    # Find the neighborhood
    best_idx = int(np.argmin([abs(t - best_t) for t in ts]))
    if best_idx > 0:
        lo = ts[best_idx - 1]
    else:
        lo = t_lo
    if best_idx < len(ts) - 1:
        hi = ts[best_idx + 1]
    else:
        hi = t_hi
    lo = max(lo, t_lo)
    hi = min(hi, t_hi)
    
    if hi - lo < 1e-12:
        return best_t, best_val
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = line_objective(S, B, Q, c1) * dx
    f2 = line_objective(S, B, Q, c2) * dx
    
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = line_objective(S, B, Q, c1) * dx
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = line_objective(S, B, Q, c2) * dx
    
    opt_t = (lo + hi) / 2.0
    opt_val = line_objective(S, B, Q, opt_t) * dx
    
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    
    # Verify with exact evaluation
    h_new = h + best_t * (D * max_abs)
    h_new = np.clip(h_new, 0.0, 1.0)
    exact_val = exact_c5(h_new)
    
    # Use the exact value if it's better (accounts for any numerical issues)
    if exact_val < best_val:
        best_val = exact_val
    
    return best_t, best_val

def soft_gradient(h, tau):
    """Compute soft-grad direction (negative, mean-zero)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    total = float(np.sum(exp_s))
    if total < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_s / total
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        # grad_r[i] = (1-h[i+d]) if 0<=i+d<n minus h[i-d] if 0<=i-d<n
        # For lag d:
        # term 1: (1-h[i+d]) for i in [0, n-1-d] if d>=0, or i in [0, n-1+min(d,0)... ]
        # Let me be more careful.
        # c[k] = sum_j h[j] * (1-h[j+d]) where d = k-(n-1)
        # d/dh[i]: from h[j] with j=i: (1-h[i+d]) if 0<=i+d<n
        #           from (1-h[j+d]) with j+d=i, i.e., j=i-d: -h[i-d] if 0<=i-d<n and 0<=i-d<n
        
        if d >= 0:
            m = n - d
            if m > 0:
                # (1-h[i+d]) for i = 0..m-1
                idx1 = np.arange(m)
                g[idx1] += wk * (1.0 - h[idx1 + d])
                # -h[i-d] for i = d..n-1, i.e., i-d = 0..n-1-d
                # Wait: j = i-d, need 0 <= j < n, so 0 <= i-d < n, i.e., d <= i <= d+n-1
                # But also need 0 <= i < n, so i in [d, n-1]
                idx2 = np.arange(d, n)
                g[idx2] += wk * (-h[idx2 - d])
        else:
            g_neg = -d
            m = n - g_neg
            if m > 0:
                # (1-h[i+d]) where d = -g_neg, so i+d = i-g_neg
                # Need 0 <= i-g_neg < n, so g_neg <= i <= g_neg+n-1, intersect [0,n-1]
                # i in [g_neg, n-1]
                idx1 = np.arange(g_neg, n)
                g[idx1] += wk * (1.0 - h[idx1 + d])  # h[i-g_neg]
                # -h[i-d] = -h[i+g_neg], need 0 <= i+g_neg < n, so i in [0, n-1-g_neg]
                idx2 = np.arange(m)
                g[idx2] += wk * (-h[idx2 + g_neg])
    
    # Make mean-zero
    g = g - np.mean(g)
    # Negative gradient direction
    D = -g
    max_abs = float(np.max(np.abs(D)))
    if max_abs < 1e-15:
        return np.zeros(n)

def random_block_direction(rng, n, h):
    """Generate a random block transfer direction."""
    # Pick block length
    block_len = int(rng.integers(2, min(20, n // 4) + 1))
    
    # Pick source and destination blocks (non-overlapping)
    # Source: a contiguous block
    src_start = int(rng.integers(0, n - block_len))
    src = np.zeros(n)
    src[src_start:src_start + block_len] = 1.0
    
    # Destination: a contiguous block (allow overlap with source, the overlapping parts cancel)
    dst_start = int(rng.integers(0, n - block_len))
    dst = np.zeros(n)
    dst[dst_start:dst_start + block_len] = 1.0
    
    D = dst - src
    
    # Ensure non-trivial
    if np.max(np.abs(D)) < 1e-10:
        return None
    
    # Ensure mean-zero (it should be if block lengths are equal)
    D = D - np.mean(D)
    
    # Optionally weight by current h values to prefer moving from high to low
    # This is a heuristic: scale positive parts by (1-h) and negative by h
    # Actually, let's keep it simple for now
    
    max_abs = float(np.max(np.abs(D)))
    if max_abs < 1e-10:
        return None
    return D / max_abs

def random_sine_direction(rng, n):
    """Generate a random sinusoidal direction."""
    freq = rng.uniform(1.0, 6.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    D = np.sin(2 * np.pi * freq * x / n + phase)
    D = D - np.mean(D)
    max_abs = float(np.max(np.abs(D)))
    if max_abs < 1e-10:
        return None
    return D / max_abs

def random_pair_direction(rng, n, h):
    """Generate a random pair transfer direction."""
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return None
    D = np.zeros(n)
    # Transfer from high to low
    if h[i] > h[j]:
        D[i] = -1.0
        D[j] = 1.0
    else:
        D[i] = 1.0
        D[j] = -1.0
    D -= np.mean(D)
    return D

def resample(h, n_new):
    """Resample h to a new length."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new / 2.0)

            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = float(np.max(S + c2 * B + c2 * c2 * Q) * dx)
    opt_t = (lo_r + hi_r) / 2.0
    opt_f = float(np.max(S + opt_t * B + opt_t * opt_t * Q) * dx)
    if opt_f < best_f:
        best_t = opt_t
        best_f = opt_f
    if best_f >= f0 - 1e-12:
        return 0.0
    h_new = h + best_t * (D * ma)
    h_new = np.clip(h_new, 0.0, 1.0)
    exact_f = exact_c5(h_new)
    if exact_f < f0 - 1e-12:
        return best_t
    return 0.0

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new / 2.0)

def optimize_at_resolution(h, target, deadline, n_iters_max=500):
    n = len(h)
    rng_local = np.random.default_rng(int(n * 7 + target))
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    stall = 0
    for it in range(n_iters_max):
        if time.time() > deadline:
            break
        r = rng_local.random()
        if r < 0.40:
            tau = float(rng_local.choice([30.0, 60.0, 100.0, 200.0, 80.0]))
            D = soft_gradient(h, tau)
        elif r < 0.65:
            D = random_block_direction(rng_local, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        elif r < 0.85:
            D = random_sine_direction(rng_local, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        else:
            D = random_gaussian_direction(rng_local, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        if np.max(np.abs(D)) < 1e-13:
            stall += 1
            continue
        t_b = line_search_exact(h, D)
        if t_b != 0.0:
            h_new = h + t_b * D
            h_new = make_feasible(h_new, target)
            c5_new = exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            h = best_h + rng_local.normal(0.0, 0.02, n)
            h = make_feasible(h, target)
            stall = 0
    return best_h, best_c5

def pair_refine_pass(h, target, deadline):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(100):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n - 1):
            if time.time() > deadline:
                break
            if abs(h[j] - h[j + 1]) < 1e-8:
                continue
            D = np.zeros(n)
            if h[j] > h[j + 1]:
                D[j] = -1.0
                D[j + 1] = 1.0
            else:
                D[j] = 1.0
                D[j + 1] = -1.0
            D -= np.mean(D)
            t_b = line_search_exact(h, D, n_samples=20)
            if t_b != 0.0:
                h_new = h + t_b * D
                h_new = make_feasible(h_new, target)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_stop = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h = np.zeros(100)
        h[:50] = 1.0
    n = len(h)
    target = n / 2.0
    h = make_feasible(h, target)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n

    d1 = t0 + budget_s * 0.12
    h, c5 = optimize_at_resolution(h, target, d1, n_iters_max=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    d2 = t0 + budget_s * 0.50
    h = best_h.copy()
    n = len(h)
    target = n / 2.0
    stall = 0
    it2 = 0
    while time.time() < d2:
        r = rng.random()
        if r < 0.35:
            tau = float(rng.choice([25.0, 50.0, 80.0, 120.0, 200.0, 350.0]))
            D = soft_gradient(h, tau)
        elif r < 0.60:
            D = random_block_direction(rng, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        elif r < 0.78:
            D = random_sine_direction(rng, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        elif r < 0.90:
            D = random_gaussian_direction(rng, n)
            if D is None:
                D = soft_gradient(h, 80.0)
        else:
            j1 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            if j1 == j2:
                D = soft_gradient(h, 80.0)
            else:
                D = np.zeros(n)
                if h[j1] > h[j2]:
                    D[j1] = -1.0
                    D[j2] = 1.0
                else:
                    D[j1] = 1.0
                    D[j2] = -1.0
                D -= np.mean(D)
                if np.max(np.abs(D)) < 1e-10:
                    D = soft_gradient(h, 80.0)
        if np.max(np.abs(D)) < 1e-13:
            stall += 1
            continue
        t_b = line_search_exact(h, D)
        if t_b != 0.0:
            h_new = h + t_b * D
            h_new = make_feasible(h_new, target)
            c5_new = exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
        it2 += 1
        if stall > 150:
            h = best_h + rng.normal(0.0, 0.025, n)
            h = make_feasible(h, target)
            stall = 0

    d3 = t0 + budget_s * 0.68
    tried_n = set([best_n])
    offsets = [0, -2, 2, -4, 4, 6, -6, 8, -8, 10, -10, 12, -12, 14, -14, 16, -16, 20, -20, 24, -24, 28, -28, 32, -32, 40, -40]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 280:
            continue
        if n_try % 2 != 0:
            n_try += 1
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        tl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = optimize_at_resolution(h_r, n_try / 2.0, tl, n_iters_max=80)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    d4 = t0 + budget_s * 0.88
    h = best_h.copy()
    n = best_n
    target = n / 2.0
    h, c5 = pair_refine_pass(h, target, min(d4, hard_stop))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    if time.time() < t0 + budget_s * 0.93:
        d5 = t0 + budget_s * 0.93
        for attempt in range(40):
            if time.time() > d5:
                break
            scale = 0.02 + 0.03 * rng.random()
            h_p = best_h + rng.normal(0.0, scale, best_n)
            h_p = make_feasible(h_p, best_n / 2.0)
            tl = min(d5, time.time() + budget_s * 0.008)
            h_try, c5_try = optimize_at_resolution(h_p, best_n / 2.0, tl, n_iters_max=50)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()

    if time.time() < t0 + budget_s * 0.97:
        h = best_h.copy()
        n = best_n
        h, c5 = pair_refine_pass(h, n / 2.0, min(t0 + budget_s * 0.97, hard_stop))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = make_feasible(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
