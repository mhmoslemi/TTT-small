```python
def _project(v, target):
    # Project v onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
```

```python
def _slsqp_solve(h0, maxiter=500, ftol=1e-12):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    def constraints_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0-h, mode='full')
        return t - corr
    
    def constraints_jac(x):
        h = x[:n]
        G = _grad_matrix(h)
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    def eq_fun(x):
        return x[:n].sum() - target
    
    def eq_jac(x):
        J = np.zeros(n+1)
        J[:n] = 1.0
        return J
    
    x0 = np.zeros(n+1)
    x0[:n] = h0
    x0[n] = float(np.max(np.correlate(h0, 1.0-h0, mode='full'))) + 0.1
    
    cons = [
        {'type': 'ineq', 'fun': constraints_fun, 'jac': constraints_jac},

            h_new = _project(h_new, target)

    except Exception:
        pass
    return h0, _exact_c5(h0)
```

```python
def _slsqp_reduced(h0, k_active=20, maxiter=300):
    """SLSQP with only top-k active constraints."""
    n = len(h0)
    target = n / 2.0
    h = _project(h0.copy(), target)
    
    for _outer in range(5):
        # Identify active constraints
        corr = np.correlate(h, 1.0-h, mode='full')
        cmax = float(np.max(corr))
        # Get indices of top-k
        topk = np.argsort(corr)[-k_active:][::-1]
        
        # Solve SLSQP with only these constraints
        def obj(x):
            return x[-1]
        
        def obj_grad(x):
            g = np.zeros(n+1)
            g[-1] = 1.0
            return g
        
        idx = topk
        def con_fun(x):
            h_x = x[:n]
            t = x[-1]
            c = np.correlate(h_x, 1.0-h_x, mode='full')
            return t - c[idx]
        
        def con_jac(x):
            h_x = x[:n]
            G = _grad_matrix(h_x)
            J = np.zeros((len(idx), n+1))
            J[:, :n] = -G[idx, :]
            J[:, n] = 1.0
            return J
        
        def eq_fun(x):
            return x[:n].sum() - target
        
        def eq_jac(x):
            J = np.zeros(n+1)
            J[:n] = 1.0
            return J
        
        x0 = np.zeros(n+1)
        x0[:n] = h
        x0[n] = cmax + 0.01
        
        cons = [
            {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

        bounds = [(0.0, 1.0)]*n + [(0.0, None)]
```

```python
def _binary_local_search(n, max_iters=100000, seed=42):
    """Find a good binary sequence with n/2 ones minimizing max correlation count."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    
    def max_count(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        return int(np.max(corr))
    
    best_mc = max_count(h)
    curr_mc = best_mc
    
    for it in range(max_iters):
        # Try a random swap: pick a 1 and a 0, swap them
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_mc = max_count(h)
        
        if new_mc <= curr_mc:
            curr_mc = new_mc
            if new_mc < best_mc:
                best_mc = new_mc
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
    
    return h, best_mc
```

Let me optimize: instead of computing the full correlation at each step, I can compute the change in correlation due to a single swap.

If I swap h[i] from 1 to 0 and h[j] from 0 to 1, the change in C_s is:
- For each s where pair (i, i+s) or (i+s, i) was counted: the count decreases by 1
- For each s where pair (j, j+s) or (j+s, j) was counted: the count increases by 1

Specifically:
- C_s was counting h[i](1-h[i+s]). If h[i]=1, h[i+s]=0, this contributed 1. After swap, h[i]=0, so contribution becomes 0. Change: -1.
- Similarly for all shifts involving i.
- And for all shifts involving j (now h[j]=1, was 0): if h[j+s]=0, C_s increases by 1 (new contribution h[j](1-h[j+s])=1).

This is O(n) per swap to compute the exact change, and O(1) to update the max (if we maintain a count of how many shifts have each value).

But implementing this efficiently is complex. For n=64, let me just use the full correlate and limit iterations.

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G
