def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        n_lo = max(0, -s)
        n_hi = min(n, n - s)
        n_idx = np.arange(n_lo, n_hi)
        i_idx = n_idx + s
        G[r, i_idx] += (1.0 - h[n_idx])
        G[r, n_idx] += (-h[i_idx])
    return G


def _slsqp_solve(h_init, n, deadline):
    """SLSQP on the epigraph formulation."""
    h_init = h_init.copy()
    
    def obj(z):
        return z[n]
    
    def obj_jac(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_fun(z):
        return float(np.sum(z[:n]) - n / 2.0)
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def ineq_fun(z):
        h = z[:n]
        t = z[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr
    
    def ineq_jac(z):
        h = z[:n]
        G = _grad_matrix(h)
        J = np.empty((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    x0 = np.zeros(n + 1)
    x0[:n] = h_init
    x0[n] = _raw_max_corr(h_init) + 0.01
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-12, 'iprint': 0})
        if res.x is not None:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = _project(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    except Exception:
        pass
    
    if time.time() > deadline:
        return best_h, best_c5
    
    # Second SLSQP call with tighter tolerance from improved point
    try:
        x0 = np.zeros(n + 1)
        x0[:n] = best_h
        x0[n] = _raw_max_corr(best_h) + 0.005
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = _project(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new

    except Exception:
        pass

def _active_subgrad(h, deadline, tol=0.01, max_iter=500):
    """Active-set subgradient method with line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m = len(h)
    
    step_base = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr))
        active = corr > cmax - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(corr))] = True
            n_active = 1
        
        G = _grad_matrix(h)
        # Subgradient: average over active lags
        g = G[active, :].mean(axis=0)
        # Project onto tangent space of sum constraint
        g = g - np.mean(g)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        
        # Line search: decrease step size
        step = step_base / (1.0 + 0.05 * it)
        improved = False
        for _ in range(10):
            h_new = _project(h - step * g / g_norm, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            step *= 0.6
        if not improved:
            step_base *= 0.9
            if step_base < 1e-6:
                break
    
    return best_h, best_c5


def _lp_trust(h, deadline, radius_init=0.1, max_iters=3000, stall_limit=50):
    """LP trust region method."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        
        # LP: find direction p that decreases all corr_r
        # minimize t subject to G @ p + corr <= t, |p| <= radius, sum(p)=0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.8
                if radius < 1e-6:
                    break
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        if float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        # Line search
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        best_a, best_c5_local = 0.0, cur_c5
        for t in np.linspace(0.0, 1.0, 12):
            a = t * alpha_max
            if a < 1e-10:
                continue
            h_t = _project(h + a * p, target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_local:
                best_c5_local, best_a = c5_t, a
        
        if best_a > 0 and best_c5_local < cur_c5 - 1e-13:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 8 == 0 and radius < 0.02:
                radius = 0.06
    
    return best_h, best_c5


def _smooth_lbfgs(h, alpha, target, maxiter=150):
    """Single L-BFGS step on log-sum-exp smoothed objective."""
    n = len(h)
    
    def fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -500.0, 500.0)
        exp_vals = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_vals)))
        w = exp_vals / float(np.sum(exp_vals))
        G = _grad_matrix(x)
        g = G.T @ w
        g -= np.mean(g)
        return f, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                                               'ftol': 1e-14, 'maxls': 30,
                                               'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _raw_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.argsort(-x)
            for i in idx:
                a = min(res, 1.0 - x[i])
                if a > 1e-15:
                    x[i] += a
                    res -= a
                    if res < 1e-12:
                        break
        else:
            idx = np.argsort(x)
            for i in idx:
                a = min(-res, x[i])
                if a > 1e-15:
                    x[i] -= a
                    res += a
                    if res > -1e-12:
                        break
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        r = n - 1 + s
        lo = max(0, -s)
        hi = min(n, n - s)
        idx = np.arange(lo, hi)
        G[r, idx + s] += (1.0 - h[idx])
        G[r, idx] += (-h[idx + s])
    return G


def _slsqp_solve(h_init, n, deadline, maxiter=400, ftol=1e-12):

        return z[n]

    def ineq_jac(z):

def _lp_trust(h, deadline, radius_init=0.12, max_iters=5000, stall_limit=50):
    """LP trust region descent."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds_list = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds_list.append((None, None))

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds_list, method='highs')
        except Exception:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue

        if not res.success or float(np.max(np.abs(res.x[:n]))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue

        p = res.x[:n]
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)

        best_a, best_local = 0.0, cur_c5
        for t in np.linspace(0.0, 1.0, 12):
            a = t * alpha_max
            if a < 1e-10:

            c5_t = _exact_c5(h_t)

            cur_c5 = _exact_c5(h)

            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.72

                radius = 0.05

    return best_h, best_c5


def _smooth_lbfgs(h, alpha, target, maxiter=150):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(h)

    def fun(x):
        s = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(s))
        vals = np.clip(alpha * (s - ms), -500.0, 500.0)
        ev = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
        w = ev / float(np.sum(ev))
        G = _grad_matrix(x)
        g = G.T @ w
        g -= np.mean(g)
        return f, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                               'maxls': 30, 'iprint': -1})
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = h.copy()
    x = _project(x, target)
    return x, _exact_c5(x)
