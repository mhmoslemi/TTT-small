def _project(v, target):

    n = len(v)

    if target >= n:

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

    max_s = float(np.max(scores))

    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    
    # Gradient: grad_i = dx * sum_k w_k * [(1-h_{i+d}) - h_{i-d}]

        wk = w[k]
        if wk < 1e-15:

        d = k - (n - 1)
        # Term from (1 - h_{i+d}) for valid i
        # partial corr_k / partial h_i = (1 - h_{i+d}) - h_{i-d}
        # For the (1-h) term: i ranges where i+d in [0, n-1]

        i_hi = min(n - 1, n - 1 - d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + d])
        # For the -h term: i ranges where i-d in [0, n-1]
        i_lo2 = max(0, d)
        i_hi2 = min(n - 1, n - 1 + d)
        if i_lo2 <= i_hi2:

            grad[idx2] -= wk * h[idx2 - d]
    grad *= dx
    return val, grad

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10.0, 10.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    shifted = (scores - max_s) / tau
    exp_s = np.exp(shifted)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + d])
        i_lo2 = max(0, d)
        i_hi2 = min(n - 1, n - 1 + d)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] -= wk * h[idx2 - d]
    grad *= dx
    return val, grad

def _slsqp(h0, tau, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    call_count = [0]
    max_calls = maxiter * 4 + 50
    
    def obj(x):
        call_count[0] += 1
        if time.time() > deadline or call_count[0] > max_calls:
            return 1e10
        v, _ = _soft_grad(x, tau)
        return v
    
    def obj_grad(x):
        call_count[0] += 1
        if time.time() > deadline or call_count[0] > max_calls:
            return np.zeros(n)
        _, g = _soft_grad(x, tau)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, np.clip(h0, 0.0, 1.0), jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            x = _project(np.clip(res.x, 0.0, 1.0), target)
            return x
    except Exception:
        pass
    return None
