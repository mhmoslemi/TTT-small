def _line_search(h, D, hard_stop):
    n = len(h)
    dx = 2.0 / n
    
    # Normalize D to avoid numerical issues
    maxD = float(np.max(np.abs(D)))
    if maxD < 1e-15:
        return None
    D = D / maxD
    
    # Ensure sum(D) ≈ 0
    D -= float(np.mean(D))
    
    # Recompute maxD after mean subtraction (should be close)
    maxD = float(np.max(np.abs(D)))
    if maxD < 1e-15:
        return None
    D = D / maxD
    
    # Feasible interval
    t_lo, t_hi = _feasible_interval(h, D)
    if t_hi - t_lo < 1e-12:
        return None
    
    # Precompute S, B, Q
    S = np.correlate(h, 1.0 - h, mode='full')  # base
    # B = corr(D, 1-h) - corr(h, D)
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    # Q = -corr(D, D)
    Q = -np.correlate(D, D, mode='full')
    
    # F(t) = max(S + t*B + t²*Q) * dx
    # Sample on a grid
    n_samples = 32
    t_grid = np.linspace(t_lo, t_hi, n_samples)
    
    best_f = float('inf')
    best_t = 0.0
    
    for t in t_grid:
        vals = S + t * B + t * t * Q
        f = float(np.max(vals) * dx)
        if f < best_f:
            best_f = f
            best_t = t
    
    # Golden section refinement around best_t
    # Determine neighborhood
    idx = int(np.argmin(np.abs(t_grid - best_t)))
    # Widen bracket by 1-2 grid cells
    half_width = (t_hi - t_lo) / n_samples * 2.0
    lo = max(t_lo, best_t - half_width)
    hi = min(t_hi, best_t + half_width)
    
    if hi - lo < 1e-14:
        return (best_t, best_f, h + best_t * D)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * B + c1 * c1 * Q) * dx)
    f2 = float(np.max(S + c2 * B + c2 * c2 * Q) * dx)
    
    for _ in range(40):
        if hi - lo < 1e-14:
            break
        if time.time() > hard_stop:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * B + c1 * c1 * Q) * dx)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * B + c2 * c2 * Q) * dx)
    
    opt_t = (lo + hi) / 2.0
    opt_f = float(np.max(S + opt_t * B + opt_t * opt_t * Q) * dx)
    
    # Pick the best
    if opt_f < best_f:
        best_t = opt_t
        best_f = opt_f
    
    new_h = h + best_t * D
    # Clip to be safe (should be within [0,1] by construction)
    new_h = np.clip(new_h, 0.0, 1.0)
    
    # Verify with exact evaluation
    exact_f = _exact_c5(new_h)
    
    return (best_t, exact_f, new_h)

def _grad_direction(h, tau):
    n = len(h)
    dx = 2.0 / n
    S_raw = np.correlate(h, 1.0 - h, mode='full')
    S = S_raw * dx
    max_S = float(np.max(S))
    
    # Soft max weights
    w = np.exp(tau * (S - max_S))
    w /= float(np.sum(w))
    
    # Compute gradient
    # grad[i] = dx * sum_d w[n-1+d] * [(1-h[i+d])*1[0<=i+d<n] - h[i-d]*1[0<=i-d<n]]
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        wk = w[r] * dx
        if wk < 1e-15:
            continue
        # (1-h[i+d]) for i+d in [0,n-1]
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * (1.0 - h[:m])
        # -h[i-d] for i-d in [0,n-1]
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += -wk * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += -wk * h[g:]
    
    # Subtract mean to ensure sum(grad) = 0 (approximately)
    grad -= float(np.mean(grad))
    
    return grad

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_interval(h, D):
    eps = 1e-14
    pos = D > eps
    neg = D < -eps
    t_lo = 0.0
    t_hi = 1e30
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / D[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / D[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / D[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / D[neg])))
    return t_lo, t_hi

def _line_search(h, D, n_samples=32, golden_iters=35):
    n = len(h)
    dx = 2.0 / n
    maxD = float(np.max(np.abs(D)))
    if maxD < 1e-15:
        return None
    D = D / maxD
    D -= float(np.mean(D))
    maxD = float(np.max(np.abs(D)))
    if maxD < 1e-15:
        return None
    D = D / maxD
    t_lo, t_hi = _feasible_interval(h, D)
    if t_hi - t_lo < 1e-12:
        return None
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = -np.correlate(D, D, mode='full')
    t_grid = np.linspace(t_lo, t_hi, n_samples)
    vals = S[None, :] + t_grid[:, None] * B[None, :] + (t_grid * t_grid)[:, None] * Q[None, :]
    f_grid = np.max(vals, axis=1) * dx
    best_idx = int(np.argmin(f_grid))
    best_f = float(f_grid[best_idx])
    best_t = float(t_grid[best_idx])
    half_w = (t_hi - t_lo) / n_samples * 2.5
    lo = max(t_lo, best_t - half_w)
    hi = min(t_hi, best_t + half_w)
    if hi - lo > 1e-14:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(S + c1 * B + c1 * c1 * Q) * dx)
        f2 = float(np.max(S + c2 * B + c2 * c2 * Q) * dx)
        for _ in range(golden_iters):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(S + c1 * B + c1 * c1 * Q) * dx)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(S + c2 * B + c2 * c2 * Q) * dx)
        opt_t = (lo + hi) / 2.0
        opt_f = float(np.max(S + opt_t * B + opt_t * opt_t * Q) * dx)
        if opt_f < best_f:
            best_f = opt_f
            best_t = opt_t
    new_h = h + best_t * D
    new_h = np.clip(new_h, 0.0, 1.0)
    exact_f = _exact_c5(new_h)
    if exact_f < best_f - 1e-10:
        best_f = exact_f
    return (best_t, best_f, new_h)

def _grad_direction(h, tau):
    n = len(h)
    dx = 2.0 / n
    S_raw = np.correlate(h, 1.0 - h, mode='full')
    S = S_raw * dx
    max_S = float(np.max(S))
    w = np.exp(tau * (S - max_S))
    wsum = float(np.sum(w))
    if wsum > 0:
        w /= wsum
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        wk = w[r] * dx
        if wk < 1e-16:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:])
                grad[d:] += -wk * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * (1.0 - h[:m])
                grad[:m] += -wk * h[g:]
    grad -= float(np.mean(grad))
    return grad

def _grad_direction_hard(h, k_top=3):
    n = len(h)
    dx = 2.0 / n
    S_raw = np.correlate(h, 1.0 - h, mode='full')
    S = S_raw * dx
    idx = np.argsort(S)[::-1][:k_top]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / k_top
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        wk = w[r] * dx
        if wk < 1e-16:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:])
                grad[d:] += -wk * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * (1.0 - h[:m])
                grad[:m] += -wk * h[g:]
    grad -= float(np.mean(grad))
    return grad

def _block_transfer_direction(rng, n, h):
    L = int(rng.integers(2, min(16, n // 4) + 1))
    a = int(rng.integers(0, n - L + 1))
    b = int(rng.integers(0, n - L + 1))
    for _ in range(20):
        if not (b + L <= a or a + L <= b):
            break
        b = int(rng.integers(0, n - L + 1))
    D = np.zeros(n)
    D[a:a+L] -= 1.0
    D[b:b+L] += 1.0
    D -= float(np.mean(D))
    return D

def _smooth_direction(rng, n):
    freq = float(rng.uniform(1.0, 5.0))
    phase = float(rng.uniform(0, 2 * np.pi))
    x = np.arange(n)
    D = np.sin(2 * np.pi * freq * x / n + phase)
    D -= float(np.mean(D))
    D += rng.normal(0, 0.3, n)
    D -= float(np.mean(D))
    return D

def _pair_direction(rng, n, h):
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    for _ in range(50):
        if h[i] > 0.01 and h[j] < 0.99 and i != j:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
    D = np.zeros(n)
    D[i] = 1.0
    D[j] = -1.0
    return D

def _random_direction(rng, n):
    D = rng.normal(0, 1, n)
    D -= float(np.mean(D))
    return D

def _resample(h, n_new):
    n = len(h)
    if n_new <= 0:
        return None
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _direction_loop(h, cur_c5, deadline, rng, n_max_iter=5000):

    best_h = h.copy()
    best_c5 = cur_c5
    cur_h = h.copy()
    cur_c5v = cur_c5
    stall = 0
    tau_cycle = [30.0, 50.0, 80.0, 100.0, 150.0]
    tau_idx = 0
    it = 0
    while it < n_max_iter and time.time() < deadline:
        it += 1
        if time.time() > deadline:
            break
        r = rng.random()
        if r < 0.35:
            D = _grad_direction(cur_h, tau_cycle[tau_idx % len(tau_cycle)])
        elif r < 0.50:
            D = _grad_direction_hard(cur_h, k_top=2)
        elif r < 0.65:
            D = _block_transfer_direction(rng, n, cur_h)
        elif r < 0.78:
            D = _smooth_direction(rng, n)
        elif r < 0.88:
            D = _pair_direction(rng, n, cur_h)
        else:
            D = _random_direction(rng, n)
        D -= float(np.mean(D))
        res = _line_search(cur_h, D)
        if res is not None:

            if new_c5 < cur_c5v - 1e-12:
                impt = cur_c5v - new_c5
                cur_h = new_h
                cur_c5v = new_c5

            else:

                    tau_idx = (tau_idx + 1) % len(tau_cycle)
                    if stall > 60:
                        cur_h = best_h + rng.normal(0, 0.02, n)
                        cur_h = _project(cur_h, n / 2.0)
                        cur_c5v = _exact_c5(cur_h)
                        stall = 0
        if best_c5 < 0.3806:
            break
    return best_h, best_c5

def _multi_resolution(best_h, best_c5, best_n, deadline, rng):
    offsets = [0, -6, -4, -2, 2, 4, 6, 8, 10, 14, 18, 22, 28, 34, 42, 50, 60, 72]
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        if h_r is None:
            continue
        c5_r = _exact_c5(h_r)
        sub_dl = min(deadline, time.time() + 12.0)
        h_r, c5_r = _direction_loop(h_r, c5_r, sub_dl, rng, n_max_iter=800)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_stop = t0 + budget_s * 0.97

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Intensive line search at original resolution (25% budget)
    d1 = t0 + budget_s * 0.25
    best_h, best_c5 = _direction_loop(best_h, best_c5, d1, rng, n_max_iter=20000)

    # Phase 2: Continue with more diverse search (25% budget)
    d2 = t0 + budget_s * 0.50
    best_h, best_c5 = _direction_loop(best_h, best_c5, d2, rng, n_max_iter=20000)

    # Phase 3: Multi-resolution exploration (30% budget)
    d3 = t0 + budget_s * 0.80
    best_h, best_c5, best_n = _multi_resolution(best_h, best_c5, best_n, d3, rng)

    # Phase 4: Intensive search at best resolution (15% budget)
    d4 = hard_stop
    best_h, best_c5 = _direction_loop(best_h, best_c5, d4, rng, n_max_iter=30000)

    # Final safety: ensure feasibility
    best_n = len(best_h)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # projection slightly hurt; try to recover
        h_alt = np.array(initial_h_values, dtype=float)
        h_alt = _project(h_alt, len(h_alt) / 2.0)
        c5_alt = _exact_c5(h_alt)
        if c5_alt < final_c5:
            best_h = h_alt
            best_n = len(h_alt)
            final_c5 = c5_alt
    return best_h, final_c5, best_n
