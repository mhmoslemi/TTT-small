            continue
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[idx] * (1.0 - h[d:d+m])
                grad[d:] += w[idx] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[idx] * (1.0 - h[g:g+m])
                grad[g:] += w[idx] * (-h[:m])
    return grad

def _line_search(h, direction, c5_cur, deadline, max_iter=50):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    if alpha_max < 1e-12:
        return 0.0, c5_cur
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = _exact_c5(h + c * direction)
    fd = _exact_c5(h + d_pt * direction)
    
    for _ in range(max_iter):
        if time.time() > deadline or (hi - lo) < 1e-13:
            break
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _exact_c5(h + c * direction)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _exact_c5(h + d_pt * direction)
    
    opt = (lo + hi) / 2.0
    return opt, _exact_c5(h + opt * direction)

def _gradient_descent(h, deadline, max_iters=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta = 50.0
    
    for it in range(max_iters):
        if time.time() > deadline or stall > 80:
            break
        
        scores = _all_scores(h)
        grad = _smooth_max_grad(h, scores, beta)
        grad -= np.mean(grad)  # preserve sum
        norm = np.linalg.norm(grad)
        if norm < 1e-15:
            stall += 10
            break
        grad /= norm
        
        direction = -grad
        alpha, new_c5 = _line_search(h, direction, cur_c5, deadline)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * direction
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Gradually increase beta

                beta = min(beta * 1.05, 500.0)
        else:
            stall += 1
            if stall % 20 == 0:
                beta = max(20.0, beta * 0.9)
    
    return best_h, best_c5

def _pair_transfer(h, deadline, max_iters=5000, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline or stall > 100:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        
        alpha_max = min(1.0 - h[i], h[j])
        if alpha_max < 1e-12:
            stall += 1
            continue
        
        alpha, new_c5 = _line_search(h, d, cur_c5, deadline, max_iter=25)
        
        if new_c5 < cur_c5 - 1e-14:
            h[i] += alpha
            h[j] -= alpha
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            # Try reverse
            d[i], d[j] = -1.0, 1.0
            alpha_max_r = min(1.0 - h[i], h[j])
            # Wait, after the first attempt, h hasn't changed
            alpha_max_r = min(1.0 - h[j], h[i])  # swap i,j roles
            if alpha_max_r > 1e-12:
                d2 = np.zeros(n)
                d2[i] = -1.0
                d2[j] = 1.0
                alpha, new_c5 = _line_search(h, d2, cur_c5, deadline, max_iter=25)
                if new_c5 < cur_c5 - 1e-14:
                    h[i] -= alpha
                    h[j] += alpha
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    continue
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):

    pop = np.zeros((popsize, n))
    pop[0] = h_init

        pop[i] = _project(h_init + rng.normal(0, 0.15, n), target)

    best_f = f[best_idx]

            break

            break
        
        for i in range(popsize):
            if time.time() > deadline:
                break
            
            # Mutation
            idxs = list(range(popsize))
            idxs.remove(i)
            idxs.remove(best_idx) if best_idx != i else None
            if len(idxs) < 2:

            r1 = int(rng.integers(0, len(idxs)))
            r2 = int(rng.integers(0, len(idxs)))
            while r2 == r1:
                r2 = int(rng.integers(0, len(idxs)))
            r1 = idxs[r1]
            r2 = idxs[r2]

            v = _project(v, target)

            trial = v.copy()
            jrand = int(rng.integers(0, n))

                    best_f = f_trial
                    best_h = trial.copy()
                    best_idx = i

    corr = np.correlate(h, 1.0 - h, mode='full')

        return np.zeros(n)

        return np.ones(n)

        else:

            beta = min(beta * 1.03, 300.0)
        else:
            stall += 1
            if stall % 15 == 0:
                beta = max(15.0, beta * 0.85)
                lr = max(0.1, lr * 0.9)
    
    return best_h, best_c5

def _pair_transfer(h, deadline, max_iters=20000, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline or stall > 80:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        # Try both directions
        for sign in [1.0, -1.0]:
            d = np.zeros(n)
            d[i] = sign
            d[j] = -sign
            alpha_max = 0.0
            if sign > 0:
                alpha_max = min(1.0 - h[i], h[j])
            else:
                alpha_max = min(1.0 - h[j], h[i])
            if alpha_max < 1e-12:
                continue
            
            alpha, new_c5 = _line_search(h, d, deadline, max_iter=20)
            if new_c5 < cur_c5 - 1e-14:
                h[i] += alpha * sign
                h[j] -= alpha * sign
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                break
        else:
            stall += 1
    
    return best_h, best_c5

def _de_full(h_init, deadline, popsize=40, max_iter=200, seed=42):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    
    pop = np.zeros((popsize, n))
    pop[0] = h_init
    for i in range(1, popsize):
        pop[i] = _project(h_init + rng.normal(0, 0.1, n), target)
    
    f = np.array([_exact_c5(pop[i]) for i in range(popsize)])
    best_idx = int(np.argmin(f))
    best_h = pop[best_idx].copy()
    best_f = f[best_idx]
    
    F = 0.7
    CR = 0.9
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if no_improve > 30:
            break
        
        improved = False
        for i in range(popsize):
            if time.time() > deadline:
                break
            
            candidates = list(range(popsize))
            candidates.remove(i)
            if best_idx != i:
                candidates.remove(best_idx)
            if len(candidates) < 2:
                continue
            r1 = int(rng.integers(0, len(candidates)))
            r2 = int(rng.integers(0, len(candidates)))
            while r2 == r1:
                r2 = int(rng.integers(0, len(candidates)))
            r1 = candidates[r1]
            r2 = candidates[r2]
            
            v = pop[best_idx] + F * (pop[r1] - pop[r2])
            v = _project(v, target)
            
            trial = pop[i].copy()
            jrand = int(rng.integers(0, n))
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = v[mask]
            trial = _project(trial, target)
            
            f_trial = _exact_c5(trial)
            if f_trial <= f[i]:
                pop[i] = trial
                f[i] = f_trial
                improved = True
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()
                    best_idx = i
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_h, best_f

def _de_anti_complement(h_init, deadline, pairing='adj', popsize=30, max_iter=300, seed=42):
    n = len(h_init)
    half = n // 2
    
    def _obj(y):
        if pairing == 'adj':
            return _exact_c5(_expand_adj(y))
        else:
            return _exact_c5(_expand_half(y))
    
    _de_dl[0] = deadline
    bounds = [(0.0, 1.0)] * half
    try:
        res = differential_evolution(
            _obj, bounds, seed=seed, maxiter=max_iter,
            popsize=popsize, tol=1e-12, mutation=(0.5, 1.0),
            recombination=0.8, polish=True,
            callback=_de_cb
        )
        if res.fun < _exact_c5(h_init):
            if pairing == 'adj':
                h_out = _expand_adj(res.x)
            else:
                h_out = _expand_half(res.x)
            return h_out, float(res.fun)
    except Exception:
        pass
    return h_init, _exact_c5(h_init)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
