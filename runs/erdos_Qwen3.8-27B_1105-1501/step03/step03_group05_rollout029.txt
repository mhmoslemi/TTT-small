def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

        diff = target - np.sum(v)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-14
        else:
            mask = v > 0.0 + 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        v[mask] += step
        v = np.clip(v, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1

            if 0 <= q + d < n:
                coeff += (1.0 - h[q + d])
            if 0 <= q - d < n:
                coeff -= h[q - d]
            G[r, q] = dx * coeff
    return G

    scores = _scores(h)
    F = scores * (2.0 / n)
    G = _build_grad(h)
    rhs = G @ h - F
    x = cp.Variable(n)
    t = cp.Variable()
    cons = [G @ x - t <= rhs, x >= 0, x <= 1,
            cp.sum(x) == n/2, cp.norm(x - h, 2) <= rho]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'ECOS', 'OSQP']:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.array(x.value).flatten()
        except Exception:
            continue
    return None

def _binary_search(h, deadline, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    k = n // 2
    # Start from binary version
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    best = hb.copy()
    best_c5 = _exact_c5(hb)
    ones = list(np.where(hb == 1.0)[0])
    zeros = list(np.where(hb == 0.0)[0])
    no_imp = 0
    while time.time() < deadline:
        if no_imp > 3000 or not ones or not zeros:
            break
        best_swap = None
        best_val = best_c5
        batch = min(50, len(ones) * len(zeros))
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            hb[i], hb[j] = 0.0, 1.0
            c5t = _exact_c5(hb)
            hb[i], hb[j] = 1.0, 0.0
            if c5t < best_val - 1e-14:
                best_val = c5t
                best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            hb[i], hb[j] = 0.0, 1.0
            ones.remove(i); zeros.remove(j)
            zeros.append(i); ones.append(j)
            best_c5 = best_val
            best = hb.copy()
            no_imp = 0
        else:
            no_imp += 1
    return best, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _gradient_descent(h, deadline, beta=50.0):
    """Smoothed max gradient descent as fallback."""
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.1
    stall = 0
    while time.time() < deadline and stall < 50:
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        w /= np.sum(w)
        # Gradient of smoothed max
        g = np.zeros(n)
        for d in range(-(n-1), n):
            wd = w[n-1+d]
            if abs(wd) < 1e-14:
                continue
            if d >= 0:
                m = n - d
                g[:m] += wd * (1.0 - h[d:])
                g[d:] -= wd * h[:m]
            else:
                gd = -d
                m = n - gd
                g[gd:] += wd * (1.0 - h[:m])
                g[:m] -= wd * h[gd:]
        g -= np.mean(g)
        norm = np.max(np.abs(g))
        if norm < 1e-14:
            break
        d = -g / norm * step
        improved = False
        for f in [1.0, 0.7, 0.5, 0.3, 0.1]:
            trial = _project_box_sum(h + f * step * d, target)
            c5t = _exact_c5(trial)
            if c5t < best_c5 - 1e-12:
                h = trial
                best = trial.copy()
                best_c5 = c5t
                improved = True
                break
        if improved:
            stall = 0
            step = min(step * 1.05, 0.4)
        else:
            stall += 1
            step *= 0.7
    return best, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    for _ in range(120):
        diff = target - np.sum(v)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
        else:
            mask = v > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v


def _build_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        row = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                row[:m] += 1.0 - h[d:]
                row[d:] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                row[g:] += 1.0 - h[:m]
                row[:m] -= h[g:]
        G[r] = dx * row
    return G


def _solve_trust_lp(h, rho):
    n = len(h)
    dx = 2.0 / n
    F = _scores(h) * dx
    G = _build_grad(h)
    rhs = G @ h - F
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - rho)
    hi = np.minimum(1.0, h + rho)
    bounds = list(zip(lo, hi)) + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 2000})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _solve_trust_socp(h, rho):
    n = len(h)
    dx = 2.0 / n
    F = _scores(h) * dx
    G = _build_grad(h)
    rhs = G @ h - F
    x = cp.Variable(n)
    t = cp.Variable()
    cons = [G @ x - t <= rhs, x >= 0, x <= 1,
            cp.sum(x) == n / 2.0, cp.norm(x - h, 2) <= rho]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['CLARABEL', 'ECOS', 'OSQP']:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.array(x.value).flatten()
        except Exception:
            continue
    return None


def _accept_step(h, direction, target, best_c5):
    n = len(h)
    best_trial = None
    best_c5_t = best_c5
    for f in [1.0, 0.8, 0.6, 0.45, 0.3, 0.15, 0.08]:
        trial = _project_box_sum(h + f * direction, target)
        c5t = _exact_c5(trial)
        if c5t < best_c5_t - 1e-11:
            best_c5_t = c5t
            best_trial = trial
    if best_trial is not None:
        return best_trial, best_c5_t
    return None, best_c5


def _trust_lp_loop(h, deadline, rho_init=0.10):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _exact_c5(h)
    rho = rho_init
    stall = 0
    while time.time() < deadline:
        if stall > 30:
            break
        xc = _solve_trust_lp(h, rho)
        if xc is None:
            xc = _solve_trust_socp(h, rho)
        if xc is None:
            stall += 1
            rho *= 0.7
            if rho < 1e-4:
                break
            continue
        xc = _project_box_sum(xc, target)
        trial, c5t = _accept_step(h, xc - h, target, best_c5)
        if trial is not None:
            h = trial
            best = trial.copy()
            best_c5 = c5t
            stall = 0
            rho = min(rho * 1.12, 0.45)
        else:
            stall += 1
            rho *= 0.65
    return best, best_c5


def _wide_lp_loop(h, deadline, rho_init=0.5):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _exact_c5(h)
    rho = rho_init
    stall = 0
    while time.time() < deadline:
        if stall > 15:
            break
        xc = _solve_trust_lp(h, rho)
        if xc is None:
            stall += 1
            rho *= 0.7
            continue
        xc = _project_box_sum(xc, target)
        trial, c5t = _accept_step(h, xc - h, target, best_c5)
        if trial is not None:
            h = trial
            best = trial.copy()
            best_c5 = c5t
            stall = 0
        else:
            stall += 1
            rho *= 0.8
    return best, best_c5


def _smooth_grad(h, beta):
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = w / ws
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        if d >= 0:
            m = n - d
            g[:m] += wd * (1.0 - h[d:])
            g[d:] -= wd * h[:m]
        else:
            gd = -d
            m = n - gd
            g[gd:] += wd * (1.0 - h[:m])
            g[:m] -= wd * h[gd:]
    return g


def _grad_loop(h, deadline, beta_start=20.0, beta_end=500.0):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _exact_c5(h)
    n_beta = 10
    beta_list = np.geomspace(beta_start, beta_end, n_beta)
    for beta in beta_list:
        if time.time() >= deadline:
            break
        stall = 0
        step = 0.08
        while time.time() < deadline and stall < 60:
            g = _smooth_grad(h, beta)
            g -= np.mean(g)
            norm = np.max(np.abs(g))
            if norm < 1e-14:
                break
            d = -g / norm * step
            trial, c5t = _accept_step(h, d, target, best_c5)
            if trial is not None:
                h = trial
                best = trial.copy()
                best_c5 = c5t
                stall = 0
                step = min(step * 1.04, 0.35)
            else:
                stall += 1
                step *= 0.7
    return best, best_c5


def _binary_init_from_fractional(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    return hb


def _binary_init_sin(n, freq):
    hb = np.zeros(n)
    t = np.arange(n)
    vals = np.sin(2.0 * np.pi * freq * t / n)
    top = np.argsort(-vals)
    hb[top[:n // 2]] = 1.0
    return hb


def _swap_search(hb, deadline, rng):
    n = len(hb)
    k = n // 2
    best = hb.copy()
    best_c5 = _exact_c5(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb < 0.5)[0])
    no_imp = 0
    while time.time() < deadline:
        if no_imp > 2500 or not ones or not zeros:
            break
        best_swap = None
        best_val = best_c5
        batch = min(40, len(ones) * len(zeros))
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            hb[i], hb[j] = 0.0, 1.0
            c5t = _exact_c5(hb)
            hb[i], hb[j] = 1.0, 0.0
            if c5t < best_val - 1e-14:
                best_val = c5t
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            hb[i], hb[j] = 0.0, 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            best_c5 = best_val
            best = hb.copy()
            no_imp = 0
        else:
            no_imp += 1
    return best, best_c5


def _try_binary(h, deadline, rng, n_freqs=5):
    n = len(h)
    best_bin = None
    best_c5_bin = _exact_c5(h)
    inits = [_binary_init_from_fractional(h)]
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(_binary_init_sin(n, freq))
    for hb_init in inits:
        if time.time() >= deadline:
            break
        per_time = max(2.0, (deadline - time.time()) / max(1, n_freqs + 1))
        dl = min(deadline, time.time() + per_time)
        hb_c, c5_c = _swap_search(hb_init.copy(), dl, rng)
        if c5_c < best_c5_bin:
            best_c5_bin = c5_c
            best_bin = hb_c.copy()
    if best_bin is not None:
        return best_bin, best_c5_bin
    return _binary_init_from_fractional(h), _exact_c5(_binary_init_from_fractional(h))


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)


def _multi_start_trust(h, deadline, rng, n_starts=4):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _exact_c5(h)
    starts = [h.copy()]
    for _ in range(n_starts - 1):
        noise = rng.normal(0, 0.05, n)
        pert = _project_box_sum(h + noise, target)
        starts.append(pert)
    per_time = max(3.0, (deadline - time.time()) / max(1, n_starts))
    for s in starts:
        if time.time() >= deadline:
            break
        dl = min(deadline, time.time() + per_time)
        h_c, c5_c = _trust_lp_loop(s, dl, rho_init=0.08)
        if c5_c < best_c5:
            best_c5 = c5_c
            best = h_c.copy()
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is None or not np.all(np.isfinite(h0)):
        n0 = 64
        h0 = (np.arange(n0) % 2 == 0).astype(float)

    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Trust-region LP (main new method) - 30% budget
    dl1 = t0 + budget_s * 0.28
    h1, c5_1 = _trust_lp_loop(h, dl1, rho_init=0.12)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1, len(h1)

    # Phase 2: Wide LP exploration - 10%
    dl2 = t0 + budget_s * 0.38
    h2, c5_2 = _wide_lp_loop(best_h, dl2, rho_init=0.4)
    if c5_2 < best_c5:
        best_c5, best_h, best_n = c5_2, h2, len(h2)

    # Phase 3: Trust-region LP again with smaller rho - 12%
    dl3 = t0 + budget_s * 0.50
    h3, c5_3 = _trust_lp_loop(best_h, dl3, rho_init=0.06)
    if c5_3 < best_c5:
        best_c5, best_h, best_n = c5_3, h3, len(h3)

    # Phase 4: Binary local search - 18%
    dl4 = t0 + budget_s * 0.68
    hb, c5b = _try_binary(best_h, dl4, rng)
    if c5b < best_c5:
        best_c5, best_h, best_n = c5b, hb, len(hb)
        # Refine binary result fractionally
        dl4b = t0 + budget_s * 0.78
        h4, c5_4 = _trust_lp_loop(best_h, dl4b, rho_init=0.15)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4
    else:
        # If binary didn't help, do gradient descent
        dl4b = t0 + budget_s * 0.78
        h4, c5_4 = _grad_loop(best_h, dl4b, beta_start=30.0, beta_end=600.0)
        if c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4, len(h4)

    # Phase 5: Multi-start trust from current best - 10%
    dl5 = t0 + budget_s * 0.88
    h5, c5_5 = _multi_start_trust(best_h, dl5, rng, n_starts=3)
    if c5_5 < best_c5:
        best_c5, best_h, best_n = c5_5, h5, len(h5)

    # Phase 6: Multi-resolution - 6%
    dl6 = t0 + budget_s * 0.94
    n_candidates = sorted(set(x for x in
                              [best_n - 4, best_n - 8, best_n + 4, best_n + 8,
                               best_n + 16, best_n - 16, 48, 56, 72, 80, 96, 112]
                              if 32 <= x <= 128))
    per_n_time = max(2.0, (dl6 - time.time()) / max(1, len(n_candidates)))
    for n_try in n_candidates:
        if time.time() > dl6:
            break
        h_r = _resample(best_h, n_try)
        c5r = _exact_c5(h_r)
        dl_r = min(dl6, time.time() + per_n_time)
        h_r, c5r = _trust_lp_loop(h_r, dl_r, rho_init=0.08)
        if c5r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5r, h_r, n_try

    # Phase 7: Final tight trust polish
    if time.time() < deadline:
        h_f, c5_f = _trust_lp_loop(best_h, deadline, rho_init=0.04)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Final projection and evaluation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
