                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.3)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.8
            if lr < 1e-6 or stall > 50:
                break
    
    return best_h, best_c5

def _softmax_descent(h, deadline, beta_start=20.0, beta_end=3000.0, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    lr = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        S = _scores(h)
        m = float(np.max(S))
        exps = np.exp(beta * (S - m))
        weights = exps / float(np.sum(exps))
        
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.3)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-6 or stall > 50:
                break
    
    return best_h, best_c5

def _binary_local_search(h, deadline, max_stall=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    # Binarize
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    n_half = n // 2
    if s > n_half:
        ones = np.where(h == 1.0)[0]
        for idx in ones[:s - n_half]:
            h[idx] = 0.0
    elif s < n_half:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:n_half - s]:
            h[idx] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S = _scores(h)
    G = _grad_matrix(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    stall = 0
    step = 0
    while time.time() < deadline:
        step += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute C[i,j] = max_k (S_k - G[k,i] + G[k,j]) for all (i,j)
        # A[i,k] = S_k - G[k, ones[i]]
        # B[j,k] = G[k, zeros[j]]
        A = S[None, :] - G[:, ones].T  # (len(ones), n_corr)
        B = G[:, zeros].T  # (len(zeros), n_corr)
        C = np.max(A[:, None, :] + B[None, :, :], axis=2)  # (len(ones), len(zeros))
        
        min_val = float(np.min(C))
        cur_max = float(np.max(S))
        
        if min_val < cur_max - 1e-12:
            min_idx = np.unravel_index(np.argmin(C), C.shape)
            i = ones[min_idx[0]]
            j = zeros[min_idx[1]]
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            S = _scores(h)
            G = _grad_matrix(h)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
    
    return best_h, best_c5

def _binary_sa(h, deadline, T_start=0.5, T_end=0.01, max_iters=50000):
    """Simulated annealing on binary sequences."""
    n = len(h)
    target = n / 2.0
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    n_half = n // 2
    if s > n_half:
        ones = np.where(h == 1.0)[0]
        for idx in ones[:s - n_half]:
            h[idx] = 0.0
    elif s < n_half:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:n_half - s]:
            h[idx] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S = _scores(h)
    G = _grad_matrix(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        if T < 0.005:
            T = 0.005
        
        # Random flip
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        i = int(i)
        j = int(j)
        
        new_S = S - G[:, i] + G[:, j]
        new_c5 = float(np.max(new_S) * 2.0 / n)
        old_c5 = float(np.max(S) * 2.0 / n)
        delta = new_c5 - old_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            S = new_S
            G = _grad_matrix(h)
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
