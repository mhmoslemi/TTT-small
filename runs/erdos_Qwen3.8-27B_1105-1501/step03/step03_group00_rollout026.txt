def _scores(h):

def _project(v, target):

def _subgrad_step(h, scores, G, alpha):
    """Move in negative subgradient direction of the max score."""
    n = len(h)
    # Find active lags (top 5% of scores)
    max_s = np.max(scores)
    threshold = max_s - 0.01 * max_s
    active = scores >= threshold
    if not np.any(active):
        active = np.zeros(len(scores), dtype=bool)
        active[np.argmax(scores)] = True
    # Weighted negative subgradient
    weights = scores[active] - (max_s - 0.01 * max_s + 1e-10)
    weights = weights ** 2  # Emphasize the most active lags
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        grad = G[active].mean(axis=0)
    else:
        grad = (G[active].T @ weights) / w_sum
    # Subtract mean to preserve sum constraint
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_init(n, n_harmonics, seed=42):
    """Generate a starting point using Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    # Base: 0.5

        h += a * np.cos(2 * np.pi * m * x) + b * np.sin(2 * np.pi * m * x)

def _pair_polish(h, deadline, max_passes=8, subset=None):
    """Coordinate descent on pairs of coordinates."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    if subset is None:
        # Use all pairs but in a smart order
        indices = list(range(n))
    else:
        indices = subset
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(len(indices)):
            if time.time() > deadline:
                break
            ji = indices[j]
            if h[ji] < 1e-8 or h[ji] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, len(indices)):
                if time.time() > deadline:
                    break
                ki = indices[k]
                if h[ki] < 1e-8 or h[ki] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[ji], h[ki])
                    else:
                        a_max = min(h[ji], 1.0 - h[ki])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[ji] += sign * m1
                        h1[ki] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[ji] += sign * m2
                        h2[ki] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[ji] += sign * a_opt
                    h_try[ki] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _coordinate_descent(h, deadline, max_passes=20):
    """Single-coordinate descent: adjust one coordinate at a time."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            # Binary search for optimal h[j] given others fixed
            # The objective as a function of h[j] is a max of linear functions
            # (since h[j](1-h[j+k]) is linear in h[j] for fixed other h values)
            # So we can find where the max transitions
            lo, hi = 0.0, 1.0
            # Actually this is piecewise linear (max of linear functions), so
            # the minimum is at a breakpoint or at the boundary
            # Let's just do a fine search
            best_local = h[j]
            best_local_c5 = best_c5
            candidates = [0.0, 1.0, h[j]]
            # Add some candidates around current
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                c = h[j] + delta
                if 0.0 <= c <= 1.0:
                    candidates.append(c)
            # Evaluate all candidates
            for c in candidates:
                h_try = h.copy()
                h_try[j] = c
                # Don't adjust sum constraint here (handle separately)
                c5_try = _exact_c5(h_try)
                if c5_try < best_local_c5 - 1e-13:
                    best_local_c5 = c5_try
                    best_local = c
            if best_local != h[j]:
                h[j] = best_local
                # Re-project to maintain sum constraint
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 - 1e-13:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _evolutionary_fourier(h_init, n, deadline, max_gens=200, pop_size=16, n_harm=8):
    """Evolutionary search in Fourier coefficient space."""
    rng = np.random.default_rng(42)
    target = n / 2.0
    x = np.arange(n) / n
    
    # Build Fourier basis
    basis = []
    for m in range(1, n_harm + 1):
        basis.append(np.cos(2 * np.pi * m * x))
        basis.append(np.sin(2 * np.pi * m * x))
    B = np.column_stack(basis)  # n x (2*n_harm)
    n_params = B.shape[1]
    
    # Initialize population around h_init
    # Project h_init onto Fourier space
    residual = h_init - 0.5
    coeff_init = np.linalg.lstsq(B, residual, rcond=None)[0]
    
    # Population
    pop = [coeff_init.copy() + rng.normal(0, 0.05, n_params)]
    for _ in range(pop_size - 1):
        pop.append(coeff_init.copy() + rng.normal(0, 0.15, n_params))
    
    def eval_coeff(coeff):
        h_raw = 0.5 + B @ coeff
        h = _project(h_raw, target)
        return _exact_c5(h), h
    
    # Evaluate initial population
    pop_c5 = []
    for coeff in pop:
        c5, _ = eval_coeff(coeff)
        pop_c5.append(c5)
    
    best_c5 = min(pop_c5)
    best_h = None
    for i, c5 in enumerate(pop_c5):
        if c5 == best_c5:
            _, best_h = eval_coeff(pop[i])
            break
    
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        # Sort by fitness
        order = np.argsort(pop_c5)
        pop = [pop[i] for i in order]
        pop_c5 = [pop_c5[i] for i in order]
        
        # Elitism: keep top half
        n_elite = max(2, pop_size // 2)
        new_pop = [pop[i].copy() for i in range(n_elite)]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if time.time() > deadline:
                break
            # Crossover + mutation
            i = rng.integers(n_elite)
            j = rng.integers(n_elite)
            child = 0.5 * pop[i] + 0.5 * pop[j]
            # Adaptive mutation
            scale = 0.05 * (1.0 - gen / max_gens) + 0.01
            child += rng.normal(0, scale, n_params)
            child = np.clip(child, -0.5, 0.5)
            new_pop.append(child)
        
        # Evaluate new population
        pop_c5_new = []
        for coeff in new_pop:
            c5, _ = eval_coeff(coeff)
            pop_c5_new.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                _, best_h = eval_coeff(coeff)
        
        pop = new_pop
        pop_c5 = pop_c5_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    x = np.clip(v - thresh, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None
