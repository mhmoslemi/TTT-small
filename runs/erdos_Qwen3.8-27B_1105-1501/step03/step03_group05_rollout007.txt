The feasible interval for alpha is determined by box constraints:

- Plus alpha must keep sum(h + alpha*v) = sum(h) (automatic if sum(v) = 0)

2. Compute all pairwise intersections of the quadratics
3. Include endpoints and alpha=0
4. Evaluate the max at all candidate points, pick the best

    corr = np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target):
    """Project h onto box [0,1] with sum = target using water-filling."""
    n = len(h)
    x = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_direction(h, beta):
    """Generate a descent direction using smooth active gradient."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    exp_vals = np.exp(beta * (scores - max_s))
    w = exp_vals / exp_vals.sum()
    
    # Build gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[n-1+d]
        if abs(wd) < 1e-14:
            continue
        lo = max(0, -d)
        hi = min(n, n-d)
        for p in range(lo, hi):
            q = p + d
            grad[q] += wd * (1.0 - h[p])
            grad[p] += -wd * h[q]
    
    # Make it a descent direction with zero sum
    v = -grad
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return None
    return v / norm

def _random_direction(rng, n, h):
    """Generate a random zero-sum feasible direction."""
    v = rng.standard_normal(n)
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-10:
        return None
    v /= norm
    return v

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, target):
    n = len(h)
    x = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(80):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1.0 + 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, v, cur_c5):
    n = len(h)
    dx = 2.0 / n
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    alpha_lo = 0.0
    alpha_hi = 0.0
    vp = v > 1e-15
    vn = v < -1e-15
    if np.any(vp):
        alpha_lo = max(alpha_lo, float(np.max(-h[vp] / v[vp])))
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[vp]) / v[vp])))
    if np.any(vn):
        alpha_lo = max(alpha_lo, float(np.max((1.0 - h[vn]) / v[vn])))
        alpha_hi = min(alpha_hi, float(np.min(-h[vn] / v[vn])))
    
    if alpha_lo > alpha_hi - 1e-14:
        return h, cur_c5
    alpha_lo = min(alpha_lo, 0.0)
    alpha_hi = max(alpha_hi, 0.0)
    if alpha_hi - alpha_lo < 1e-13:
        return h, cur_c5
    
    candidates = [0.0, alpha_lo + 1e-13, alpha_hi - 1e-13]
    
    m = len(A)
    # Stationary points
    for r in range(m):
        if abs(C[r]) > 1e-15:
            ast = -B[r] / (2.0 * C[r])
            if alpha_lo < ast < alpha_hi:
                candidates.append(ast)
    
    # Pairwise intersections
    if m <= 255:
        for i in range(m):
            for j in range(i + 1, m):
                dC = C[i] - C[j]
                dB = B[i] - B[j]
                dA = A[i] - A[j]
                if abs(dC) > 1e-15:
                    disc = dB * dB - 4.0 * dC * dA
                    if disc >= -1e-12:
                        disc = max(disc, 0.0)
                        sd = np.sqrt(disc)
                        r1 = (-dB + sd) / (2.0 * dC)
                        r2 = (-dB - sd) / (2.0 * dC)
                        if alpha_lo < r1 < alpha_hi:
                            candidates.append(r1)
                        if alpha_lo < r2 < alpha_hi:
                            candidates.append(r2)
                elif abs(dB) > 1e-15:
                    root = -dA / dB
                    if alpha_lo < root < alpha_hi:
                        candidates.append(root)
    
    # Grid for safety
    for k in range(41):
        candidates.append(alpha_lo + (alpha_hi - alpha_lo) * k / 40.0)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= alpha_lo - 1e-12) & (candidates <= alpha_hi + 1e-12)]
    if len(candidates) == 0:
        return h, cur_c5
    
    # Evaluate all candidates vectorized
    scores = A[None, :] + B[None, :] * candidates[:, None] + C[None, :] * candidates[:, None] ** 2
    costs = scores.max(axis=1)
    
    best_idx = int(np.argmin(costs))
    best_cost = costs[best_idx]
    cur_cost = A.max()
    
    if best_cost >= cur_cost - 1e-14:
        return h, cur_c5
    
    best_alpha = candidates[best_idx]
    if abs(best_alpha) < 1e-13:
        return h, cur_c5
    
    h_new = h + best_alpha * v
    h_new = _repair(h_new, n / 2.0)
    new_c5 = _exact_c5(h_new)
    
    if new_c5 < cur_c5 - 1e-12:
        return h_new, new_c5
    return h, cur_c5

def _grad_dir(h, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    exp_vals = np.exp(beta * (scores - max_s))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        p = np.arange(lo, hi)
        q = p + d
        grad[q] += wd * (1.0 - h[p])
        grad[p] += -wd * h[q]
    
    v = -grad
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return None
    return v / norm

def _rand_dir(rng, n):
    v = rng.standard_normal(n)
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-10:
        return None
    return v / norm

def _subset_dir(rng, n, frac=0.3):
    v = np.zeros(n)
    k = max(2, int(n * frac))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.standard_normal(k)
    vals -= vals.mean()
    v[idx] = vals
    v -= v.mean()
    norm = np.max(np.abs(v))
    if norm < 1e-12:
        return None
    return v / norm

def _descent(h, deadline, rng, n_betas=8, max_stall=30):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    stall = 0
    betas = np.geomspace(3.0, 150.0, n_betas)
    
    while time.time() < deadline:
        directions = []
        for beta in betas:
            v = _grad_dir(cur, beta)
            if v is not None:
                directions.append(v)
        for _ in range(4):
            v = _rand_dir(rng, n)
            if v is not None:
                directions.append(v)
        for _ in range(2):
            v = _subset_dir(rng, n)
            if v is not None:
                directions.append(v)
        
        improved = False
        for v in directions:
            if time.time() > deadline:
                break
            new_h, new_c5 = _line_search(cur, v, cur_c5)
            if new_c5 < cur_c5 - 1e-12:
                cur = new_h
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best = cur.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > max_stall:
                break
    
    return best, best_c5

def _pair_polish(h, deadline):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    
    improved_any = True
    pass_count = 0
    while improved_any and time.time() < deadline and pass_count < 20:
        improved_any = False
        pass_count += 1
        for i in range(n):
            if time.time() > deadline:
                return best, best_c5
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best, best_c5
                if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                    if abs(h[j] - 0.5) < 1e-6:
                        continue
                v = np.zeros(n)
                v[i] = 1.0
                v[j] = -1.0
                new_h, new_c5 = _line_search(cur, v, cur_c5)
                if new_c5 < cur_c5 - 1e-13:
                    cur = new_h
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best = cur.copy()
                    improved_any = True
                v[i] = -1.0
                v[j] = 1.0
                new_h, new_c5 = _line_search(cur, v, cur_c5)
                if new_c5 < cur_c5 - 1e-13:
                    cur = new_h
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best = cur.copy()
                    improved_any = True
    return best, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n // 2]
    h[idx] = 1.0
    return h

def _mix(a, b, t):
    return _repair(t * a + (1.0 - t) * b, len(a) / 2.0)
