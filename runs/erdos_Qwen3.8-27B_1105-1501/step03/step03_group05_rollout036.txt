def _pair_line_search(h, i, j):
    """Find best alpha for transfer h[i]+=alpha, h[j]-=alpha."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Feasible range for alpha
    alpha_lo = max(-h[i], h[j] - 1.0)
    alpha_hi = min(1.0 - h[i], h[j])
    if alpha_hi - alpha_lo < 1e-15:
        return 0.0, scores.max() * dx
    
    # Compute L_d for all d
    # d ranges from -(n-1) to n-1, indexed as 0 to 2n-2 in the scores array
    # L_d = (1-h[i+d])*I(0<=i+d<n) - h[i-d]*I(0<=i-d<n) - (1-h[j+d])*I(0<=j+d<n) + h[j-d]*I(0<=j-d<n)
    
    L = np.zeros(2*n-1)
    # For d = 0 to n-1 (i+d valid when d <= n-1-i)
    # Let's use offset: index = d + (n-1)
    offset = n - 1
    
    # (1-h[i+d]) for d where 0<=i+d<n, i.e., -i <= d < n-i
    d_start = max(-(n-1), -i)
    d_end = min(n-1, n-1-i)
    if d_start <= d_end:
        L[d_start+offset:d_end+1+offset] += (1.0 - h[i+d_start:d_end+1])
    
    # -h[i-d] for d where 0<=i-d<n, i.e., i-(n-1) <= d <= i
    d_start = max(-(n-1), i-(n-1))
    d_end = min(n-1, i)
    if d_start <= d_end:
        L[d_start+offset:d_end+1+offset] -= h[i-d_start:d_end+1]
    
    # -(1-h[j+d]) for d where 0<=j+d<n
    d_start = max(-(n-1), -j)
    d_end = min(n-1, n-1-j)
    if d_start <= d_end:
        L[d_start+offset:d_end+1+offset] -= (1.0 - h[j+d_start:d_end+1])
    
    # +h[j-d] for d where 0<=j-d<n
    d_start = max(-(n-1), j-(n-1))
    d_end = min(n-1, j)
    if d_start <= d_end:
        L[d_start+offset:d_end+1+offset] += h[j-d_start:d_end+1]

    # Q[(j-i) + offset] = +1 if j != i
    # Q[(i-j) + offset] = +1 if i != j
    Q = np.zeros(2*n-1)
    Q[offset] = -2.0  # d=0
    if i != j:
        Q[(j-i)+offset] = 1.0
        Q[(i-j)+offset] = 1.0

    # over alpha in [alpha_lo, alpha_hi]

import time

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _get_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_LQ(h, i, j):
    n = len(h)
    off = n - 1
    L = np.zeros(2*n-1)
    Q = np.zeros(2*n-1)
    
    # (1-h[i+d]) where 0<=i+d<n => -i <= d < n-i => idx in [off-i, off+n-1-i]
    a = max(0, off - i)
    b = min(2*n-2, off + n - 1 - i)
    if a <= b:
        dr = np.arange(a-off, b-off+1)
        L[a:b+1] += 1.0 - h[i+dr]
    
    # -h[i-d] where 0<=i-d<n => i-n+1 <= d <= i => idx in [i, off+i]
    a = max(0, i)
    b = min(2*n-2, off + i)
    if a <= b:
        dr = np.arange(a-off, b-off+1)
        L[a:b+1] -= h[i-dr]
    
    # -(1-h[j+d]) where 0<=j+d<n
    a = max(0, off - j)
    b = min(2*n-2, off + n - 1 - j)
    if a <= b:
        dr = np.arange(a-off, b-off+1)
        L[a:b+1] -= 1.0 - h[j+dr]
    
    # +h[j-d] where 0<=j-d<n
    a = max(0, j)
    b = min(2*n-2, off + j)
    if a <= b:
        dr = np.arange(a-off, b-off+1)
        L[a:b+1] += h[j-dr]
    
    Q[off] = -2.0
    d = j - i
    if d != 0:
        Q[off + d] = 1.0
        Q[off - d] = 1.0
    return L, Q

def _ls_pair(h, i, j, scores, L, Q, n_coarse=80):
    a_lo = max(-h[i], h[j]-1.0)
    a_hi = min(1.0-h[i], h[j])
    if a_hi - a_lo < 1e-14:
        return 0.0, np.inf
    dx = 2.0/len(h)
    
    # Coarse
    alphas = np.linspace(a_lo, a_hi, n_coarse)
    mat = scores[None,:] + alphas[:,None]*L[None,:] + (alphas**2)[:,None]*Q[None,:]
    mv = mat.max(1) * dx
    bi = int(np.argmin(mv))
    ba, bv = alphas[bi], mv[bi]
    
    # Golden section refine
    lo = alphas[max(0,bi-3)]
    hi = alphas[min(n_coarse-1,bi+3)]
    gr = (np.sqrt(5.)-1.)/2.
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1 = (scores + c1*L + c1*c1*Q).max()*dx
    f2 = (scores + c2*L + c2*c2*Q).max()*dx

        if f1 < f2:
            hi,c2,f2 = c2,c1,f1
            c1 = hi-gr*(hi-lo)
            f1 = (scores+c1*L+c1*c1*Q).max()*dx

            lo,c1,f1 = c1,c2,f2
            c2 = lo+gr*(hi-lo)
            f2 = (scores+c2*L+c2*c2*Q).max()*dx
    oa = (lo+hi)/2.
    ov = (scores+oa*L+oa*oa*Q).max()*dx
    if ov < bv:
        ba, bv = oa, ov
    return ba, bv

def _binary_sa(h_init, deadline, rng, n_iters=500000, T_start=0.01, T_end=0.001):
    """Simulated annealing on binary subproblem."""
    n = len(h_init)
    half = n // 2
    # Initialize from h_init: top half values are 1
    idx = np.argsort(-h_init)
    h_bin = np.zeros(n)
    h_bin[idx[:half]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    cur = cur_c5
    
    ones = set(idx[:half].tolist())
    zeros = set(idx[half:].tolist())
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    T = T_start
    T_factor = (T_end/T_start)**(1.0/n_iters)
    
    i = 0
    while i < n_iters and time.time() < deadline:
        i += 1
        T *= T_factor
        # Random swap
        oi = ones_list[rng.integers(len(ones_list))]
        zi = zeros_list[rng.integers(len(zeros_list))]
        h_bin[oi] = 0.0
        h_bin[zi] = 1.0
        new_c5 = _exact_c5(h_bin)
        
        if new_c5 < cur or rng.random() < np.exp((cur-new_c5)/max(T, 1e-10)):
            # Accept
            cur = new_c5
            ones_list.remove(oi)
            ones_list.append(zi)
            zeros_list.remove(zi)
            zeros_list.append(oi)
            if cur < best_c5:
                best_c5 = cur
                best_h = h_bin.copy()
        else:
            # Reject
            h_bin[oi] = 1.0
            h_bin[zi] = 0.0
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new/2.0)
