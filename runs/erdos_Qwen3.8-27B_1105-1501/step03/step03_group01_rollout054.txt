def _line_direction(h, scores, k_top=5, tau=50.0):
    """
    Generate a descent direction targeting top-k active correlations.
    Returns direction d with sum(d)=0.
    """
    n = len(h)
    # Softmax weights on top correlations
    scaled_scores = scores * (2.0 / n)  # same as dx scaling
    max_s = np.max(scaled_scores)
    weights = np.exp(tau * (scaled_scores - max_s))
    weights /= weights.sum()
    
    # Compute weighted gradient of F
    # grad of S_r w.r.t. h_i:
    # S_r = sum_i h_i * (1 - h_{i+shift})
    # dS_r/dh_j = (1 - h_{j+shift}) - h_{j-shift}  (for valid indices)

            continue
        shift = r - (n-1)  # this is the lag
        w = weights[r]
        # dS_r/dh_j = (1 - h_{j+shift}) for j where j+shift < n
        #             - h_{j-shift} for j where j-shift >= 0

            # Second term: from h_{j-shift} * (1 - h_j)
            # d/dh_j = -h_{j-shift} if j-shift valid
            jm = j - shift
            if 0 <= jm < n:

    # Simplest: d = -grad + mean(-grad) ... but we need sum(d)=0
    # Actually sum(grad) might not be zero. Let's project to mean-zero.
    d = -grad
    d -= d.mean()
    
    # Scale
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    d /= norm  # unit norm direction
    return d
