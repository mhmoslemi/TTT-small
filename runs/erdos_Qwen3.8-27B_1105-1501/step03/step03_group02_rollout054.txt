def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    """Gradient of Q_d with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] -= h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] -= h[:m]
    return grad

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        grad -= np.mean(grad)

        # Adaptive step size with line search
        step_size = 0.5 / max(1.0, float(np.max(np.abs(grad))))
        cur_f = _exact_c5(h)
        improved = False

        for ls in range(40):
            if time.time() > deadline:
                break
            h_try = _project(h - step_size * grad, target)
            f_try = _exact_c5(h_try)
            if f_try < cur_f - 1e-14:
                h = h_try
                cur_f = f_try
                improved = True
                step_size *= 1.15
            else:
                step_size *= 0.6
                if step_size < 1e-10:
                    break

        if improved:
            if cur_f < best_c5:
                best_c5 = cur_f
                best_h = h.copy()
            # Increase beta slowly to focus more on max
            beta *= 1.02
        else:
            # Try larger perturbation
            beta *= 0.8
            if beta < 2.0:
                break

    return best_h, best_c5

def _slsqp_weighted(h_init, deadline, beta=50.0, top_k=40, maxiter=150):
    """SLSQP on the weighted quadratic objective"""
    n = len(h_init)
    target = n / 2.0
    scores = _all_scores(h_init)
    n_lags = 2 * n - 1

    # Compute weights
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum

    # Restrict to top_k
    if top_k < n_lags:
        idx = np.argsort(scores)[::-1][:top_k]
        w2 = np.zeros(n_lags)
        for i in idx:
            w2[i] = w[i]
        s2 = float(np.sum(w2))
        if s2 > 0:
            w2 /= s2
            w = w2

    # Build c and H matrices
    c = np.zeros(n)
    H = np.zeros((n, n))
    for r in range(n_lags):
        if w[r] < 1e-10:
            continue
        d = r - (n - 1)
        if d == 0:
            c[:] += w[r]
            H[np.diag_indices(n)] += w[r]
        elif d > 0:
            m = n - d
            c[d:] += w[r]
            H[:m, d:d+m] += w[r] * 0.5
            H[d:d+m, :m] += w[r] * 0.5
        else:
            g = -d
            m = n - g
            c[:m] += w[r]
            H[:m, g:g+m] += w[r] * 0.5
            H[g:g+m, :m] += w[r] * 0.5

    def obj_func(x):
        return c @ x - x @ H @ x

    def grad_func(x):
        return c - 2.0 * H @ x

    def hess_func(x, v):
        return (-2.0 * H @ v).reshape(-1)

    # Constraints: sum(x) = target (equality), 0<=x<=1 (bounds)
    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    x0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n

    from scipy.optimize import LinearConstraint, Bounds
    A = np.ones((1, n))
    lin_cons = LinearConstraint(A, [target], [target])
    bnds = Bounds(np.zeros(n), np.ones(n))

    try:
        res = minimize(obj_func, x0, jac=grad_func, hessp=hess_func,
                       method='trust-constr', constraints=[lin_cons], bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-10})
    except Exception:
        return None

    if res is None or not hasattr(res, 'x'):
        return None

    h_out = np.clip(res.x, 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _exact_c5(h_out)
    return h_out, c5

def _pair_exchange_polish(h, deadline, max_rounds=500):
    """Polish using pair exchanges (transfer t from i to j)"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > 50:
            break

        scores = _all_scores(h)
        max_score = float(np.max(scores))
        top_lags = np.argsort(scores)[::-1][:8]

        best_delta = 0.0
        best_move = None

        # Try swapping between high and low values
        sorted_idx = np.argsort(h)
        lows = sorted_idx[:n//4]
        highs = sorted_idx[-n//4:]

        for i in highs:
            for j in lows:
                if abs(h[i] - h[j]) < 0.05:
                    continue
                t = min(h[i], 1.0 - h[j]) * 0.5
                if t < 0.01:
                    continue
                h[i] -= t
                h[j] += t
                sc_new = _all_scores(h)
                new_max = float(np.max(sc_new))
                h[i] += t
                h[j] -= t
                delta = new_max - max_score
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j, t)
                if best_delta < -0.01:
                    break
            if best_delta < -0.01:
                break

        if best_move is None or best_delta >= -1e-10:
            no_improve += 1
            continue

        i, j, t = best_move
        h[i] -= t
        h[j] += t
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            h = best_h.copy()

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_and_optimize(h_best, n, rng, deadline, scale=0.04):
    """Perturb and run reweighted optimization"""
    h_p = h_best + rng.normal(0, scale, n)
    h_p = _project(h_p, n / 2.0)
    h_opt, c5_opt = _rwq_optimize(h_p, deadline, beta_init=30.0, max_iters=80)
    return h_opt, c5_opt

def _multi_beta_slq(h_init, deadline):
    """Try multiple beta values for SLSQP weighted optimization"""
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    betas = [10.0, 25.0, 50.0, 100.0, 200.0]
    per_time = max(5.0, (deadline - time.time()) / len(betas))

    for beta in betas:
        if time.time() > deadline:
            break
        dl = time.time() + per_time
        result = _slsqp_weighted(best_h, dl, beta=beta, top_k=min(40, 2*n-1), maxiter=100)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary local search (novel approach)
    dl1 = t0 + budget_s * 0.12
    bits = _binary_init_from_h(h)
    bits_best, bin_c5 = _binary_local_search(bits, dl1, max_rounds=150)
    h_bin = bits_best.astype(float)
    h_bin = _project(h_bin, best_n / 2.0)
    bin_c5_exact = _exact_c5(h_bin)
    if bin_c5_exact < best_c5:
        best_c5 = bin_c5_exact
        best_h = h_bin.copy()

    # Phase 2: Reweighted quadratic optimization (different from SLSQP active-set)
    dl2 = t0 + budget_s * 0.30
    h_rwq, c5_rwq = _rwq_optimize(best_h, dl2, beta_init=25.0, max_iters=300)
    if c5_rwq < best_c5:
        best_c5 = c5_rwq
        best_h = h_rwq.copy()

    # Phase 3: Multi-beta SLSQP with weighted quadratic
    dl3 = t0 + budget_s * 0.50
    h_slq, c5_slq = _multi_beta_slq(best_h, dl3)
    if c5_slq < best_c5:
        best_c5 = c5_slq
        best_h = h_slq.copy()

    # Phase 4: Pair exchange polish
    dl4 = t0 + budget_s * 0.60
    h_pe, c5_pe = _pair_exchange_polish(best_h, dl4, max_rounds=400)
    if c5_pe < best_c5:
        best_c5 = c5_pe
        best_h = h_pe.copy()

    # Phase 5: Multi-restart with perturbation
    dl5 = t0 + budget_s * 0.82
    scale = 0.05
    attempts = 0
    while time.time() < dl5:
        attempts += 1
        if best_c5 < 0.38075:
            break
        scale = 0.05 * (0.93 ** min(attempts, 40))
        per_budget = max(4.0, min(30.0, (dl5 - time.time()) * 0.25))
        per_dl = time.time() + per_budget
        h_try, c5_try = _perturb_and_optimize(best_h, best_n, rng, per_dl, scale)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.05)

    # Phase 6: Try different resolutions
    dl6 = t0 + budget_s * 0.92
    for dn in [-8, 8, -4, 4, -16, 16, -12, 12]:
        if time.time() > dl6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 160 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        # Quick check before expensive optimization
        if c5_r > best_c5 + 0.001:
            continue
        per_dl = min(dl6, time.time() + 25.0)
        h_opt, c5_opt = _rwq_optimize(h_r, per_dl, beta_init=30.0, max_iters=100)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 7: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.96:
        dl7 = t0 + budget_s * 0.97
        result = _slsqp_weighted(best_h, dl7, beta=200.0, top_k=25, maxiter=200)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 8: Final pair polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _pair_exchange_polish(best_h, t0 + budget_s * 0.995, max_rounds=200)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    # If projection degraded, try the unprojected version
    if final_c5 > best_c5 + 1e-8:
        h_alt = np.clip(best_h, 0.0, 1.0)
        if abs(float(np.sum(h_alt)) - best_n / 2.0) < 1e-6:
            final_c5 = _exact_c5(h_alt)
            best_h = h_alt

    return best_h, final_c5, best_n
