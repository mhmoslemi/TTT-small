If d = i-j < 0 (let g = j-i > 0, so d = -g): the term is -h[i]*h[i+g] = -h[i]*h[j] → same thing, coefficient -1.

So for the lag d = ±(j-i), the t^2 coefficient in s_d is -1.
For all other lags, the t^2 coefficient is 0.

So s_d(h + t*(e_i - e_j)) is:
- For d = j-i or d = i-j: a quadratic with t^2 coefficient -1 (CONCAVE!)
- For all other d: linear in t

So the MAX of these is: max(linear functions, one concave quadratic). This is NOT necessarily convex!

The concave quadratic opens downward, so the max of a concave quadratic and linear functions could have a maximum in the interior. This means the 1D problem is non-convex and we need to be careful.

However, in practice, the change over a small interval (constrained by 0 ≤ h_i + t ≤ 1 and 0 ≤ h_j - t ≤ 1) will be small, and the function will be approximately linear. So a grid search should work well.

```python
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_scores(h):
    """Compute gradient of all scores with respect to h.
    Returns array G of shape (2n-1, n) where G[d, i] = d(score[d])/d(h[i])
    """
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = d + n - 1  # row index in G
        if d >= 0:
            m = n - d
            # score[d] = sum_{j=0}^{m-1} h[j+d]*(1-h[j])
            # d(score)/d(h[i]):
            #   if d <= i <= n-1: (1 - h[i-d])
            #   if 0 <= i <= m-1: -h[i+d]
            idx_pos = np.arange(d, n)  # i from d to n-1
            G[r, idx_pos] += (1.0 - h[idx_pos - d])
            idx_neg = np.arange(0, m)  # i from 0 to m-1
            G[r, idx_neg] -= h[idx_neg + d]
        else:
            g = -d
            m = n - g
            # score[d] = sum_{j=0}^{m-1} h[j]*(1-h[j+g])
            # d(score)/d(h[i]):
            #   if 0 <= i <= m-1: (1 - h[i+g])
            #   if g <= i <= n-1: -h[i-g]
            idx_pos = np.arange(0, m)
            G[r, idx_pos] += (1.0 - h[idx_pos + g])
            idx_neg = np.arange(g, n)
            G[r, idx_neg] -= h[idx_neg - g]
    return G

def _project(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Bisection on lambda
    s = np.sum(v)

    lo = -2.0

    for _ in range(60):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def _pair_cd(h, target, deadline, n_passes=100):
    """Coordinate descent on pairs (i,j): transfer mass between h_i and h_j."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h = h.copy()
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                # Try transferring from j to i (and vice versa)
                # h_i + t, h_j - t
                # Feasible: -h_j <= t <= 1 - h_i, AND for the reverse: -h_i <= -t <= 1-h_j, i.e., h_i - 1 <= t <= h_j
                # Combined: max(-h_j, h_i-1) <= t <= min(1-h_i, h_j)
                t_min = max(-h[j], h[i] - 1.0)
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-12:
                    continue
                
                # Grid search
                n_grid = 16
                for t in np.linspace(t_min, t_max, n_grid):
                    if abs(t) < 1e-12:
                        continue
                    h[i], h[j] = h[i] + t, h[j] - t
                    c5_val = _exact_c5(h)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = h[i] - t, h[j] + t
                        break
                    # Keep the change and try more
                # Reset if we improved (we're now at a better point, continue from there)
        if not improved:
            break
    
    return best_h, best_c5
```
