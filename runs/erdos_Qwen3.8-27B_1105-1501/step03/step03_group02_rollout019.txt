    z0 = float(np.max(scores)) + 1e-5
    all_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in all_idx]

    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 5.0)]

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    constraints = [{'type': 'eq',
                    'fun': lambda x: np.sum(x[:n]) - target,
                    'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for d in active_lags:
        def fun(x, dd=d):
            return x[n] - _score_tau_local(x[:n], dd)
        def jac(x, dd=d):
            return np.concatenate([-_score_grad(x[:n], dd), [1.0]])
        constraints.append({'type': 'ineq', 'fun': fun, 'jac': jac})

    def _cb(xk):
        if time.time() > deadline:
            raise RuntimeError("timeout")

    try:
        result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol, 'disp': False},
                          callback=_cb)
    except (RuntimeError, Exception):
        return None

    h_out = np.clip(result.x[:n], 0.0, 1.0)
    if abs(float(np.sum(h_out)) - target) > 1e-3:
        h_out = _project(h_out, target)
    c5 = _c5(h_out)
    return h_out, c5

def _score_tau_local(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d + m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g + m])))

def _slsqp_refine(h_init, deadline, top_k=20):
    n = len(h_init)
    for _ in range(4):
        if time.time() > deadline:
            break
        scores = _all_scores(h_init)
        max_s = float(np.max(scores))
        active_mask = scores >= max_s - 0.05
        active_lags = [int(k - (n - 1)) for k in range(len(scores)) if active_mask[k]]
        active_lags = active_lags[:top_k]

        z0 = max_s + 1e-6
        x0 = np.concatenate([h_init, [z0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 5.0)]

        def obj(x):
            return x[n]
        def obj_jac(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        cons = [{'type': 'eq',
                 'fun': lambda x: np.sum(x[:n]) - n / 2.0,
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        for d in active_lags:
            def fun2(x, dd=d):
                return x[n] - _score_tau_local(x[:n], dd)
            def jac2(x, dd=d):
                return np.concatenate([-_score_grad(x[:n], dd), [1.0]])
            cons.append({'type': 'ineq', 'fun': fun2, 'jac': jac2})

        def _cb(xk):
            if time.time() > deadline:
                raise RuntimeError("t/o")

        try:
            result = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-12, 'disp': False},
                              callback=_cb)
        except (RuntimeError, Exception):
            break

        h_out = np.clip(result.x[:n], 0.0, 1.0)
        if abs(float(np.sum(h_out)) - n / 2.0) > 1e-4:
            h_out = _project(h_out, n / 2.0)
        c5_new = _c5(h_out)
        if c5_new < _c5(h_init) - 1e-14:
            h_init = h_out
        else:
            break
    return h_init, _c5(h_init)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_restart(h_best, deadline, rng, n):
    best_h = h_best.copy()
    best_c5 = _c5(h_best)
    scale = 0.04

    while time.time() < deadline:
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)

        per_deadline = time.time() + min(15.0, (deadline - time.time()) * 0.3)
        result = _slsqp_solve(h_pert, per_deadline, top_k=25, maxiter=80, ftol=1e-10)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.01, scale * 0.85)
            else:
                scale = min(0.12, scale * 1.1)
        else:
            h_lp, c5_lp = _targeted_pair_descent(h_pert, time.time() + min(8.0, (deadline - time.time()) * 0.2), max_sweeps=100)
            if c5_lp < best_c5 - 1e-12:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                scale = max(0.01, scale * 0.9)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Nesterov gradient (20%)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _nesterov_phase(h, dl1, beta_init=35.0, alpha_init=0.02)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Targeted pair descent (30%)
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _targeted_pair_descent(best_h, dl2, max_sweeps=500)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP with active set (15%)
    dl3 = t0 + budget_s * 0.65
    result = _slsqp_solve(best_h, dl3, top_k=40, maxiter=120, ftol=1e-11)
    if result is not None and result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0].copy()

    # Phase 4: Multi-restart (20%)
    dl4 = t0 + budget_s * 0.85
    h4, c5_4 = _multi_restart(best_h, dl4, rng, best_n)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: SLSQP refine (10%)
    dl5 = t0 + budget_s * 0.95
    h5, c5_5 = _slsqp_refine(best_h, dl5, top_k=15)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair descent final polish
    if time.time() < t0 + budget_s * 0.99:
        h6, c5_6 = _targeted_pair_descent(best_h, t0 + budget_s * 0.995, max_sweeps=100)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Try nearby resolutions if time permits
    if time.time() < t0 + budget_s * 0.99:
        for dn in [2, -2, 4, -4]:
            if time.time() > t0 + budget_s * 0.99:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _targeted_pair_descent(h_r, time.time() + 5.0, max_sweeps=50)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(final_h)

    return final_h, final_c5, best_n
