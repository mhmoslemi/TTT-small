                contribs[i, r] = 1.0 - 2.0*h[i]
            elif d > 0:
                if i >= d:
                    contribs[i, r] += 1.0 - h[i-d]
                if i + d < n:
                    contribs[i, r] += -h[i+d]
            else:
                g = -d
                if i < n - g:
                    contribs[i, r] += 1.0 - h[i+g]
                if i - g >= 0:
                    contribs[i, r] += -h[i-g]
    return contribs

def find_best_delta(scores, linear, quad_vec, dl, dh, n_grid=200):
    """Find delta in [dl, dh] minimizing max(scores + linear*delta + quad_vec*delta^2)"""
    if dh - dl < 1e-12:
        return 0.0, float(np.max(scores))
    
    deltas = np.linspace(dl, dh, n_grid)
    vals = scores[None,:] + linear[None,:]*deltas[:,None] + quad_vec[None,:]*(deltas[:,None])**2
    f = vals.max(axis=1)
    
    best_idx = int(np.argmin(f))
    best_delta = deltas[best_idx]
    best_f = f[best_idx]
    
    # Golden section refinement
    step = (dh - dl) / max(n_grid - 1, 1)
    lo = max(dl, best_delta - 2.0*step)
    hi = min(dh, best_delta + 2.0*step)
    
    if hi - lo > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr*(hi-lo)
        c2 = lo + gr*(hi-lo)
        f1 = float(np.max(scores + linear*c1 + quad_vec*c1*c1))
        f2 = float(np.max(scores + linear*c2 + quad_vec*c2*c2))
        for _ in range(40):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr*(hi-lo)
                f1 = float(np.max(scores + linear*c1 + quad_vec*c1*c1))
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr*(hi-lo)
                f2 = float(np.max(scores + linear*c2 + quad_vec*c2*c2))
        best_delta = (lo+hi)/2.0
        best_f = float(np.max(scores + linear*best_delta + quad_vec*best_delta*best_delta))
    
    return best_delta, best_f

def pair_exchange_sweep(h, order, deadline=None, n_grid=200, tol=1e-13):
    """One sweep of pair exchanges. Returns (h, n_improvements)."""
    n = len(h)
    scores = compute_scores(h)
    contribs = precompute_contribs(h)
    quad_vec = np.zeros(2*n-1)
    quad_vec[n-1] = -2.0
    
    n_imps = 0
    
    for (i, j) in order:
        if deadline is not None and time.time() > deadline:
            break
        
        linear = contribs[i] - contribs[j]
        dl = max(-h[i], h[j] - 1.0)
        dh = min(1.0 - h[i], h[j])
        
        if dh - dl < 1e-12:
            continue
        
        current_max = float(np.max(scores))
        best_delta, best_f = find_best_delta(scores, linear, quad_vec, dl, dh, n_grid)
        
        if best_f < current_max - tol:
            h[i] += best_delta
            h[j] -= best_delta
            h[i] = max(0.0, min(1.0, h[i]))
            h[j] = max(0.0, min(1.0, h[j]))
            scores = compute_scores(h)
            contribs = precompute_contribs(h)
            n_imps += 1
    
    return h, n_imps

def std_cyclic_order(n):
    return [(i, j) for i in range(n) for j in range(i+1, n)]

def random_order(n, rng):
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    # Project to satisfy sum constraint
    target = n_new / 2.0
    current_sum = float(np.sum(h_new))
    if current_sum > 0 and current_sum < n_new:
        # Simple projection: scale and clip
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust to meet sum
        diff = target - float(np.sum(h_new))
        if abs(diff) > 1e-10:
            # Distribute the difference
            free = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
            cnt = int(np.sum(free))
            if cnt > 0:
                h_new[free] += diff / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                # Crude redistribution
                h_new = np.clip(h_new + diff/n_new, 0.0, 1.0)
    return h_new

def antipodal_candidate(m, rng, seed_offset=0):
    """Generate a random antipodal candidate h = [y, 1-y_rev] or [y, 1-y]"""
    rng_local = np.random.default_rng(seed_offset)
    variant = rng_local.integers(0, 3)
    if variant == 0:
        y = rng_local.uniform(0.2, 0.8, m)
        h = np.concatenate([y, 1-y])
    elif variant == 1:
        y = rng_local.uniform(0.2, 0.8, m)
        h = np.concatenate([y, 1-y[::-1]])
    else:
        # Smooth random

        y = 0.5 + 0.3*rng_local.normal(0, 1, m)
        y = 0.5 + 0.3*np.sin(t + rng_local.uniform(0, 2*np.pi))
        y = np.clip(y, 0.05, 0.95)
        h = np.concatenate([y, 1-y])
    return np.clip(h, 0.0, 1.0)

                f2 = float(np.max(scores + linear * c2 + quad_vec * c2 * c2))
        best_delta = (lo + hi) / 2.0
        best_f = float(np.max(scores + linear * best_delta + quad_vec * best_delta * best_delta))
    return best_delta, best_f

def pair_exchange_sweep(h, order, deadline, n_grid, tol):
    n = len(h)
    scores = compute_scores(h)
    contribs = precompute_contribs(h)
    quad_vec = np.zeros(2 * n - 1)
    quad_vec[n - 1] = -2.0
    n_imps = 0
    for idx in range(len(order)):
        i, j = order[idx]
        if time.time() > deadline:
            break
        linear = contribs[i] - contribs[j]
        dl = max(-h[i], h[j] - 1.0)
        dh = min(1.0 - h[i], h[j])
        if dh - dl < 1e-12:
            continue
        current_max = float(np.max(scores))
        best_delta, best_f = find_best_delta(scores, linear, quad_vec, dl, dh, n_grid)
        if best_f < current_max - tol:
            h[i] += best_delta
            h[j] -= best_delta
            h[i] = max(0.0, min(1.0, h[i]))
            h[j] = max(0.0, min(1.0, h[j]))
            scores = compute_scores(h)
            contribs = precompute_contribs(h)
            n_imps += 1
    return h, n_imps

def cyclic_order(n):
    return [(i, j) for i in range(n) for j in range(i + 1, n)]

def random_order(n, rng):
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    rng.shuffle(pairs)
    return pairs

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - float(np.sum(h_new))
    if abs(diff) > 1e-10:
        free = (h_new > 1e-8) & (h_new < 1.0 - 1e-8)
        cnt = int(np.sum(free))
        if cnt > 0:
            h_new[free] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def make_sine_start(n, freq, amp, phase):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def make_antipodal_start(n, rng):
    m = n // 2
    variant = int(rng.integers(0, 4))
    if variant == 0:
        y = rng.uniform(0.1, 0.9, m)
        h = np.concatenate([y, 1.0 - y])
    elif variant == 1:
        y = rng.uniform(0.1, 0.9, m)
        h = np.concatenate([y, 1.0 - y[::-1]])
    elif variant == 2:
        t = np.linspace(0, 2.0 * np.pi, m, endpoint=False)
        y = 0.5 + 0.35 * np.sin(t + rng.uniform(0, 2 * np.pi))
        y = np.clip(y, 0.05, 0.95)
        h = np.concatenate([y, 1.0 - y])
    else:
        y = 0.5 + 0.3 * rng.normal(0, 1, m)
        y = np.clip(y, 0.05, 0.95)
        h = np.concatenate([y, 1.0 - y[::-1]])
    return np.clip(h, 0.0, 1.0)

def optimize_from_start(h, deadline, rng, max_sweeps=500):
    n = len(h)
    order = cyclic_order(n)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        h, n_imps = pair_exchange_sweep(h, order, deadline, 150, 1e-13)
        if n_imps == 0:
            break
        if sweep > 20 and n_imps < 3 and sweep % 10 == 0:
            order = random_order(n, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: Optimize from parent (cyclic, converge)
    dl1 = t0 + budget_s * 0.20
    h = optimize_from_start(h0, dl1, rng, max_sweeps=300)
    c5 = exact_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Random order sweeps for escape
    dl2 = t0 + budget_s * 0.38
    h = best_h.copy()
    n = len(h)
    stall = 0
    for sweep in range(500):
        if time.time() > dl2:
            break
        order_r = random_order(n, rng)
        h, n_imps = pair_exchange_sweep(h, order_r, dl2, 150, 1e-13)
        c5 = exact_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if n_imps == 0:
                stall += 5
        if stall > 40:
            h = best_h + rng.normal(0, 0.03, n)
            h = np.clip(h, 0.0, 1.0)
            target = n / 2.0
            diff = target - float(np.sum(h))
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            cnt = int(np.sum(free))
            if cnt > 0:
                h[free] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
            stall = 0

    # Phase 3: Antipodal family exploration
    dl3 = t0 + budget_s * 0.62
    n_use = len(best_h)
    for trial in range(40):
        if time.time() > dl3:
            break
        h_ant = make_antipodal_start(n_use, rng)
        sub_dl = min(dl3, t0 + elapsed() + budget_s * 0.015)
        h_ant = optimize_from_start(h_ant, sub_dl, rng, max_sweeps=150)
        c5_ant = exact_c5(h_ant)
        if c5_ant < best_c5 - 1e-14:
            best_c5 = c5_ant
            best_h = h_ant.copy()
            best_n = n_use

    # Phase 4: Sine wave starts
    dl4 = t0 + budget_s * 0.74
    freqs = [1, 2, 3, 4, 5, 6, 7, 8, 1.5, 2.5, 3.5]
    for fi, freq in enumerate(freqs):
        if time.time() > dl4:
            break
        for amp in [0.2, 0.3, 0.4, 0.45]:
            if time.time() > dl4:
                break
            h_sine = make_sine_start(n_use, freq, amp, rng.uniform(0, 2 * np.pi))
            sub_dl = min(dl4, t0 + elapsed() + budget_s * 0.008)
            h_sine = optimize_from_start(h_sine, sub_dl, rng, max_sweeps=100)
            c5_s = exact_c5(h_sine)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = h_sine.copy()

    # Phase 5: Resolution adaptation
    dl5 = t0 + budget_s * 0.88
    for dn in [2, -2, 4, -4, 6, -6, 8, -8]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200 or n_try % 2 != 0:
            continue
        h_try = resample_h(best_h, n_try)
        sub_dl = min(dl5, t0 + elapsed() + budget_s * 0.02)
        h_try = optimize_from_start(h_try, sub_dl, rng, max_sweeps=200)
        c5_try = exact_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 6: More random restarts from best
    dl6 = t0 + budget_s * 0.95
    n = len(best_h)
    for restart in range(30):
        if time.time() > dl6:
            break
        scale = 0.02 + 0.04 * (restart / 30.0)
        h_p = best_h + rng.normal(0, scale, n)
        h_p = np.clip(h_p, 0.0, 1.0)
        target = n / 2.0
        diff = target - float(np.sum(h_p))
        free = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        cnt = int(np.sum(free))
        if cnt > 0:
            h_p[free] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
        sub_dl = min(dl6, t0 + elapsed() + budget_s * 0.012)
        h_p = optimize_from_start(h_p, sub_dl, rng, max_sweeps=200)
        c5_p = exact_c5(h_p)
        if c5_p < best_c5 - 1e-14:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n

    # Phase 7: Final high-precision polish
    if elapsed() < budget_s * 0.98:
        dl7 = t0 + budget_s * 0.995
        h = best_h.copy()
        n = len(h)
        order_f = cyclic_order(n)
        for sweep in range(1000):
            if time.time() > dl7:
                break
            h, n_imps = pair_exchange_sweep(h, order_f, dl7, 300, 1e-14)
            c5 = exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if n_imps == 0:
                break
        # One more pass with random order
        if time.time() < dl7:
            order_rf = random_order(n, rng)
            h = best_h.copy()
            for sweep in range(500):
                if time.time() > dl7:
                    break
                h, n_imps = pair_exchange_sweep(h, order_rf, dl7, 300, 1e-14)
                c5 = exact_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                if n_imps == 0:
                    break

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target = n_final / 2.0
    cur_sum = float(np.sum(best_h))
    if abs(cur_sum - target) > 1e-8:
        diff = target - cur_sum
        free = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, n_final
