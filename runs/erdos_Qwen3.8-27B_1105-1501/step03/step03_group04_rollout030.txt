def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    # Bisection on threshold
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    # Fix sum to be exactly target
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    G = np.zeros((2 * n - 1, n))

            m = n - g

    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:

                G[r, g:] -= h[:m]
    return G

def _soft_descent(h, deadline, beta_start=20.0, beta_end=5000.0, max_iters=20000):

    target = n / 2.0

    best_h = h.copy()

    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:

def _lp_direction(h, scores, G, radius):

        alpha = lr
        improved = False
        for _ls in range(15):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-7 or stall > 60:
                break
        else:
            lr = min(lr * 1.04, 0.25)
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] * 0.999)
        hi = min(radius, (1.0 - h[j]) * 0.999)
        if lo >= hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.82
            if radius < 1e-7:
                break
            continue
        new_c5 = None
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_try = _project(h + frac * p, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                new_c5 = c5_try
                h = h_try
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                break
        if new_c5 is None:
            stall += 1
            radius *= 0.8
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.04
        else:
            radius = min(radius * 1.08, 0.4)
    return best_h, best_c5

def _steepest_local_search(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step_sizes = [0.001, 0.003, 0.005, 0.01, 0.02]
    max_passes = 50000
    passes = 0
    while time.time() < deadline and passes < max_passes:
        passes += 1
        improved = False
        scores = _scores(h)
        r_star = int(np.argmax(scores))
        G = _grad_matrix(h)
        row = G[r_star, :]
        # Find top candidates: most negative row values (increase h[i]) 
        # and most positive row values (decrease h[i])
        inc_candidates = np.argsort(row)[:8]
        dec_candidates = np.argsort(-row)[:8]
        tried = set()
        for i in list(inc_candidates) + list(dec_candidates):
            if i in tried:
                continue
            tried.add(i)
            if h[i] < 1e-9 or h[i] > 1.0 - 1e-9:
                continue
            for delta in step_sizes:
                for sign in [1.0, -1.0]:
                    val = h[i] + sign * delta
                    if val < 0 or val > 1:
                        continue
                    h_try = h.copy()
                    h_try[i] = val
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < cur_c5 - 1e-13:
                        h = h_try
                        cur_c5 = c5_try
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

def _simulated_annealing(h, deadline, rng, n_pop=6):
    n = len(h)
    target = n / 2.0
    # Initial temperature: estimate typical C5 change
    h_test = _project(h + rng.normal(0, 0.05, n), target)
    dT_est = abs(_exact_c5(h_test) - _exact_c5(h))
    T0 = max(dT_est * 3.0, 0.001)
    T1 = 1e-7
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Population for diversity
    pop = [h.copy()]
    for _ in range(n_pop - 1):
        p = _project(h + rng.normal(0, 0.08, n), target)
        pop.append(p)
    total_time = max(deadline - time.time(), 1.0)
    it = 0
    max_iters = 300000
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        T = T0 * (T1 / T0) ** frac
        roll = rng.random()
        h_try = None
        if roll < 0.40:
            # Point mutation
            i = int(rng.integers(n))
            delta = rng.uniform(0.002, 0.05) * rng.choice([1, -1])
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_try = _project(h_try, target)
        elif roll < 0.70:
            # Pair transfer
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.002, 0.04) * rng.choice([1, -1])
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_try[j] = np.clip(h[j] - delta, 0.0, 1.0)
            h_try = _project(h_try, target)
        elif roll < 0.85:
            # Sinusoidal perturbation
            freq = int(rng.integers(1, 8))
            amp = rng.uniform(0.005, 0.03)
            phase = rng.uniform(0, 2 * np.pi)
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            h_try = _project(h + pert, target)
        else:
            # Crossover with population member
            pi = int(rng.integers(n_pop))
            if pi != 0:
                pt = rng.uniform(0.3, 0.7)
                h_try = _project(pt * h + (1 - pt) * pop[pi], target)
            else:
                h_try = _project(h + rng.normal(0, 0.03, n), target)
        c5_try = _exact_c5(h_try)
        delta_e = c5_try - cur_c5
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(T, 1e-15)):
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                pop[0] = h.copy()
            # Update population: replace worst with current if better
            if it % 200 == 0:
                pi = int(rng.integers(1, n_pop))
                c_pop = _exact_c5(pop[pi])
                if cur_c5 < c_pop:
                    pop[pi] = h.copy()
        # Occasionally refresh one population member with a kick
        if it % 500 == 0:
            pi = int(rng.integers(1, n_pop))
            pop[pi] = _project(h + rng.normal(0, 0.1, n), target)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h, deadline, rng):
    """Simulated annealing in near-binary space."""
    n = len(h)
    target = n / 2.0
    # Start from thresholded version of h
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = _project(h_bin, target)
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    T = 0.005
    it = 0
    while time.time() < deadline and it < 200000:
        it += 1
        if it % 1000 == 0:
            T *= 0.97
        # Flip one or two bits
            k = int(rng.integers(1, 4))
            idx = rng.choice(n, size=k, replace=False)
            h_try = h_bin.copy()
            for i in idx:
                h_try[i] = 1.0 - h_try[i]
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            de = c5_try - cur_c5
            if de < 0 or rng.random() < np.exp(-de / max(T, 1e-15)):
                h_bin = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
    # Also try ternary: {0, 0.5, 1}
    h_tern = np.where(h_bin > 0.75, 1.0, np.where(h_bin < 0.25, 0.0, 0.5))
    h_tern = _project(h_tern, target)
    c5_tern = _exact_c5(h_tern)
    if c5_tern < best_c5:
        best_c5 = c5_tern
        best_h = h_tern.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    final_deadline = t0 + budget_s * 0.995
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick soft descent (20% budget)
    d1 = t0 + budget_s * 0.12
    h1, c51 = _soft_descent(h, d1, beta_start=10.0, beta_end=3000.0, max_iters=15000)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Steepest local search (20%)
    d2 = t0 + budget_s * 0.32
    h2, c52 = _steepest_local_search(best_h, d2)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    # Phase 3: Simulated annealing (25%)
    d3 = t0 + budget_s * 0.57
    h3, c53 = _simulated_annealing(best_h, d3, rng, n_pop=6)
    if c53 < best_c5:
        best_c5 = c53
        best_h = h3.copy()

    # Phase 4: Local search after SA
    d4 = t0 + budget_s * 0.70
    h4, c54 = _steepest_local_search(best_h, d4)
    if c54 < best_c5:
        best_c5 = c54
        best_h = h4.copy()

    # Phase 5: LP polish (15%)
    d5 = t0 + budget_s * 0.85
    h5, c55 = _lp_polish(best_h, d5, radius_init=0.1, max_iters=8000)
    if c55 < best_c5:
        best_c5 = c55
        best_h = h5.copy()

    # Phase 6: Try other n values (8%)
    d6 = t0 + budget_s * 0.93
    for dn in [-16, -8, 8, 16, -4, 4]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        c5_r0 = _exact_c5(h_r)
        if c5_r0 > best_c5 + 0.02:
            continue
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _soft_descent(h_r, sub_dl, beta_start=20.0, beta_end=3000.0, max_iters=3000)
        sub_dl2 = min(d6, time.time() + budget_s * 0.01)
        h_r, c5_r = _lp_polish(h_r, sub_dl2, radius_init=0.08, max_iters=2000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final combined polish (7%)
    d7 = final_deadline
    h_f, c5_f = _steepest_local_search(best_h, min(d7, time.time() + budget_s * 0.02))
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d7 - 5:
        h_f, c5_f = _lp_polish(best_h, d7, radius_init=0.05, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final verification and projection
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        # Shouldn't happen, but safety
        final_c5 = best_c5
    return best_h, final_c5, best_n
