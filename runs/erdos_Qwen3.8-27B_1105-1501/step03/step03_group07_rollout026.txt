import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Bisection for threshold
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_offsets(h):
    """Compute all J_s values (raw, unscaled by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _gradient_for_offset(h, s):
    """Compute gradient of J_s with respect to h. s is offset from -(n-1) to n-1."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[s:s+m] += (1.0 - h[:m])
            grad[:m] -= h[s:s+m]

        m = n + s  # s is negative, so m = n - |s|
        if m > 0:
            grad[:m] += (1.0 - h[-m:])
            grad[-m:] -= h[:m]
    return grad

    c_obj[1:] = lam

    if top_k is not None and top_k < n_off:
        idx = np.argsort(-offsets)[:top_k]
    else:
        idx = np.arange(n_off)

    n_cons = len(idx)
    A_ub = np.zeros((n_cons, nvars))
    b_ub = np.zeros(n_cons)
    for ii, s_idx in enumerate(idx):
        d_off = int(s_idx) - (n - 1)
        grad = _grad_offset(h, d_off)
        A_ub[ii, 0] = -1.0
        A_ub[ii, 1::2] = grad
        A_ub[ii, 2::2] = -grad
        b_ub[ii] = -offsets[s_idx]

    A_eq = np.zeros((1, nvars))
    A_eq[0, 1::2] = 1.0
    A_eq[0, 2::2] = -1.0
    b_eq = np.zeros(1)

    bounds = [(-np.inf, np.inf)]
    for i in range(n):
        if h[i] <= 1e-10:
            bounds.append((0, d_bound))
            bounds.append((0, 0))
        elif h[i] >= 1.0 - 1e-10:
            bounds.append((0, 0))
            bounds.append((0, d_bound))
        else:
            bounds.append((0, d_bound))
            bounds.append((0, d_bound))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success:
        return None

    d = res.x[1::2] - res.x[2::2]
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    return d / norm

def _line_search(h, direction, target, c5_cur, max_alpha=1.5):
    n = len(h)
    direction = np.asarray(direction, dtype=float)
    norm_d = np.linalg.norm(direction)
    if norm_d < 1e-15:
        return h.copy(), c5_cur
    direction = direction / norm_d

    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            a_u = (1.0 - h[j]) / direction[j]
            if a_u < alpha_max:
                alpha_max = a_u
        elif direction[j] < -1e-14:
            a_u = h[j] / (-direction[j])
            if a_u < alpha_max:
                alpha_max = a_u
    alpha_max = max(min(alpha_max, 3.0), 1e-8)

    best_a = 0.0
    best_c = c5_cur
    alphas = np.linspace(0.01, alpha_max, 12)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a

    if best_a == 0.0:
        return h.copy(), c5_cur

    lo = max(0.0, best_a * 0.5)
    hi = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(22):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))

    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        best_a, best_c = opt_a, opt_c
    return _project(h + best_a * direction, target), best_c

def _softmax_dir(h, offsets, T=0.08):
    n = len(h)
    mx = float(np.max(offsets))
    w = np.exp((offsets - mx) / T)
    wsum = np.sum(w)
    if wsum < 1e-30:
        return None
    w /= wsum
    grad = np.zeros(n)
    for s_idx in range(len(offsets)):
        if w[s_idx] < 1e-7:
            continue
        d_off = int(s_idx) - (n - 1)
        grad += w[s_idx] * _grad_offset(h, d_off)
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize_lp(h, deadline, target, max_iters=10000, use_topk=False):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    lam = 0.005

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break

        offsets = np.correlate(h, 1.0 - h, mode='full')

        d = None
        if use_topk:
            d = _tangent_lp(h, offsets, lam=lam, d_bound=1.0, top_k=20)
        if d is None:
            d = _tangent_lp(h, offsets, lam=lam, d_bound=1.0)

        if d is None:
            d = _softmax_dir(h, offsets, T=0.05)
            if d is None:
                d = _softmax_dir(h, offsets, T=0.15)
            if d is None:
                break

        h_new, c5_new = _line_search(h, d, target, cur_c5, max_alpha=1.5)

        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            lam = max(0.001, lam * 0.97)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            lam = min(0.5, lam * 1.3)
            if stall % 10 == 5:
                d2 = _softmax_dir(h, offsets, T=0.03)
                if d2 is not None:
                    h_new2, c5_new2 = _line_search(h, d2, target, cur_c5, max_alpha=0.8)
                    if c5_new2 < cur_c5 - 1e-14:
                        h = h_new2
                        cur_c5 = c5_new2
                        stall = 0
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
            if stall % 10 == 8:
                rng_p = np.random.default_rng(it + 1)
                h_pert = _project(h + rng_p.normal(0, 0.02, n), target)
                h_new3, c5_new3 = _line_search(h, h_pert - h, target, cur_c5, max_alpha=0.5)
                if c5_new3 < cur_c5 - 1e-14:
                    h = h_new3
                    cur_c5 = c5_new3
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()

    return best_h, best_c5

def _pair_descent(h, deadline, target, max_passes=30):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(123)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        offsets = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(offsets))
        grad_w = np.zeros(n)
        for s_idx in range(len(offsets)):
            w = offsets[s_idx] / mx
            if w > 0.95:
                d_off = int(s_idx) - (n - 1)
                grad_w += w * _grad_offset(h, d_off)
        grad_w -= np.mean(grad_w)

        top_n = min(n, 16)
        top = np.argsort(-np.abs(grad_w))[:top_n]
        improved = False

        for i in range(len(top)):
            if time.time() > deadline:
                return best_h, best_c5
            j = top[i]
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k_idx in range(len(top)):
                if k_idx == i:
                    continue
                k = top[k_idx]
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    am = min((1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                             (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                    if am < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        a = am * frac
                        h[j] += dj * a
                        h[k] += dk * a
                        c_t = _exact_c5(h)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[j] -= dj * a
                            h[k] -= dk * a
        # Random pairs
        if not improved:
            for _ in range(100):
                if time.time() > deadline:
                    return best_h, best_c5
                j = rng.integers(0, n)
                k = rng.integers(0, n)
                if j == k or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                dj = 1.0 if rng.random() > 0.5 else -1.0
                dk = -dj
                am = min((1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj),
                         (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                if am < 1e-10:
                    continue
                a = am * (0.3 + 0.7 * rng.random())
                h[j] += dj * a
                h[k] += dk * a
                c_t = _exact_c5(h)
                if c_t < best_c5 - 1e-13:
                    best_c5 = c_t
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] -= dj * a
                    h[k] -= dk * a
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization (35%)
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _optimize_lp(h, dl1, target0, max_iters=20000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair descent (15%)
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _pair_descent(best_h, dl2, best_n / 2.0, max_passes=50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More LP (15%)
    dl3 = t0 + budget_s * 0.65
    h3, c5_3 = _optimize_lp(best_h, dl3, best_n / 2.0, max_iters=15000, use_topk=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-start (20%)
    dl4 = t0 + budget_s * 0.85
    n_starts = 12
    for s in range(n_starts):
        if time.time() > dl4:
            break
        rem = dl4 - time.time()
        if rem < 8.0:
            break
        alloc = min(rem / (n_starts - s), 15.0)
        sub_dl = time.time() + alloc

        r = rng.random()
        if r < 0.3:
            h_try = _project(best_h + rng.normal(0, 0.04, best_n), best_n / 2.0)
        elif r < 0.5:
            idx = rng.choice(best_n, size=best_n // 3, replace=False)
            h_try = best_h.copy()
            h_try[idx] = 1.0 - h_try[idx]
            h_try = _project(h_try, best_n / 2.0)
        elif r < 0.7:
            bin_h = (best_h > 0.5).astype(float)
            h_try = _project(0.5 * best_h + 0.5 * bin_h + rng.normal(0, 0.02, best_n), best_n / 2.0)
        else:
            sh = rng.integers(1, best_n // 5 + 1)
            h_try = _project(np.roll(best_h, sh), best_n / 2.0)

        h_try, c5_try = _optimize_lp(h_try, sub_dl, best_n / 2.0, max_iters=5000)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Resolution sweep (10%)
    dl5 = t0 + budget_s * 0.97
    for dn in [-4, 4, -2, 2, -8, 8, -6, 6, 12, -12]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 120 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + min(12.0, (dl5 - time.time()) * 0.25))
        h_r, c5_r = _optimize_lp(h_r, sub_dl, n_try / 2.0, max_iters=3000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final polish
    h6, c5_6 = _optimize_lp(best_h, deadline, best_n / 2.0, max_iters=20000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
