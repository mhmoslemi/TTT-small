import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_of_score(h, d):
    """Gradient of correlation score at lag d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] -= h[:m]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[g:] += (1.0 - h[:m])
            grad[:m] -= h[g:]
    return grad

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _topk_descent(h, deadline, topk=5, max_iters=10000, momentum_gamma=0.9):
    """Gradient descent focusing on top-k active correlations with momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    velocity = np.zeros(n)
    lr = 0.15
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute scores and find top-k
        scores = _scores(h)
        # Get indices of top-k scores
        k_vals = np.argsort(scores)[-topk:]
        
        # Average gradient over top-k
        grad = np.zeros(n)
        for kv in k_vals:
            d = int(kv) - (n - 1)
            grad += _grad_of_score(h, d)
        grad /= topk
        
        # Remove mean to stay on sum constraint
        grad -= np.mean(grad)
        
        # Momentum update
        velocity = momentum_gamma * velocity - lr * grad
        h_new = h + velocity
        h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall_count = 0
            lr = min(lr * 1.02, 0.5)
        else:
            h = h  # revert
            stall_count += 1
            lr *= 0.85
            velocity *= 0.5
            if lr < 1e-6 or stall_count > 100:
                break
    
    return best_h, best_c5
