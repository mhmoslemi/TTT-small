import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_grad(h):
    """Compute gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _line_search(h, d, c5_cur, n_evals=20):
    """1D line search: minimize exact_c5(h + alpha * d) over feasible alpha."""
    n = len(h)
    # Find max feasible alpha
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    # Coarse search
    best_a = 0.0
    best_c5 = c5_cur
    alphas = np.linspace(0.0, alpha_max, n_evals)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 <= best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _subgradient_step(h, target, G, scores, band_frac=0.05):
    """Compute subgradient descent direction."""
    n = len(h)
    s_max = float(np.max(scores))
    # Active set: lags within band of max
    active = np.where(scores >= s_max - band_frac * s_max)[0]
    if len(active) == 0:
        active = np.where(scores == s_max)[0]
    # Average gradient over active set
    d = -G[active].mean(axis=0)
    # Ensure zero sum
    d -= np.mean(d)
    # Normalize
    norm = np.sqrt(np.sum(d**2))
    if norm < 1e-12:
        return None
    return d / norm

def _lp_step(h, scores, G, radius=0.1):
    """LP-based direction finding (from parent)."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sinusoidal_perturb(h, rng, freq_range=(0.5, 5.0), amp_range=(0.01, 0.08)):
    """Add a sinusoidal perturbation and project."""
    n = len(h)
    target = n / 2.0
    x = np.arange(n)
    f = rng.uniform(*freq_range)
    a = rng.uniform(*amp_range)
    phase = rng.uniform(0, 2 * np.pi)
    pert = a * np.sin(2 * np.pi * f * x / n + phase)
    h_new = h + pert
    return _project(h_new, target)

def _random_perturb(h, rng, magnitude=0.03):
    """Add random perturbation and project."""
    n = len(h)
    target = n / 2.0
    pert = rng.standard_normal(n) * magnitude
    pert -= np.mean(pert)
    h_new = h + pert
    return _project(h_new, target)
