def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    # Project v onto {x: 0 <= x_i <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    A, B, C = _quadratic_coeffs(h, P)

    t_lo = 0.0

    vals = np.array([float(np.max(A + B * tg - C * tg * tg)) * dx for tg in t_grid])
    
    best_i = int(np.argmin(vals))
    best_t = t_grid[best_i]
    best_val = vals[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_samp:
        lo = t_grid[best_i - 1] if vals[best_i - 1] >= best_val else best_t
        hi = t_grid[best_i + 1] if vals[best_i + 1] >= best_val else best_t
        if hi - lo > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(A + B * c1 - C * c1 * c1)) * dx
            f2 = float(np.max(A + B * c2 - C * c2 * c2)) * dx
            for _ in range(35):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(A + B * c1 - C * c1 * c1)) * dx
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(A + B * c2 - C * c2 * c2)) * dx
            opt_t = (lo + hi) / 2.0
            opt_val = float(np.max(A + B * opt_t - C * opt_t * opt_t)) * dx
            if opt_val < best_val:
                best_t = opt_t
                best_val = opt_val
    
    return best_t, best_val
