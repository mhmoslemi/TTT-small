import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _subgrad_dir(h, alpha):
    n = len(h)
    C = _corr_raw(h)
    Cmax = float(np.max(C))
    exps = np.exp(np.clip(alpha * (C - Cmax), -500, 0))
    w = exps / float(np.sum(exps))
    G = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k]
        if wk < 1e-15:
            continue
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            G[js:je] += wk * (1.0 - h[js-k:je-k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            G[js2:je2] -= wk * h[js2+k:je2+k]
    G -= np.mean(G)
    norm = float(np.max(np.abs(G)))
    if norm < 1e-15:
        return None
    return G / norm

def _line_search(h, d):
    """Find t >= 0 that minimizes C5(h + t*d), respecting bounds."""
    n = len(h)
    # Find t_max
    t_max = 1e10
    pos_mask = d > 1e-14
    if np.any(pos_mask):
        t_max = float(np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    t_max = max(t_max, 0.0)
    
    if t_max < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Precompute C0, A, B
    C0 = _corr_raw(h)
    A = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    B = -np.correlate(d, d, mode='full')
    
    def f(t):
        return float(np.max(C0 + t * A + t * t * B))
    
    # Coarse grid
    n_grid = 30
    ts = np.linspace(0, t_max, n_grid)
    fs = np.array([f(t) for t in ts])
    i_best = int(np.argmin(fs))
    
    # Refine: dense grid around best
    if i_best > 0 and i_best < n_grid - 1:
        t_lo = ts[max(0, i_best-2)]
        t_hi = ts[min(n_grid-1, i_best+2)]
    else:
        t_lo, t_hi = 0.0, t_max
    
    n_fine = 80
    ts_fine = np.linspace(t_lo, t_hi, n_fine)
    fs_fine = np.array([f(t) for t in ts_fine])
    i_fine = int(np.argmin(fs_fine))
    
    # Final: golden section
    if i_fine > 0 and i_fine < n_fine - 1:
        a = ts_fine[max(0, i_fine-1)]
        b = ts_fine[min(n_fine-1, i_fine+1)]
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        c = b - phi * (b - a)
        dd = a + phi * (b - a)
        fc = f(c)
        fd = f(dd)
        for _ in range(50):
            if b - a < 1e-13:
                break
            if fc < fd:
                b = dd; dd = c; fd = fc
                c = b - phi * (b - a); fc = f(c)
            else:
                a = c; c = dd; fc = fd
                dd = a + phi * (b - a); fd = f(dd)
        t_opt = (a + b) / 2.0
    else:
        t_opt = ts_fine[i_fine]
    
    return t_opt, f(t_opt)

def _sgd_step(h, alpha, max_steps=200):
    """Multiple subgradient steps with line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for step in range(max_steps):
        d = _subgrad_dir(h, alpha)
        if d is None:
            break
        # Go in negative gradient direction
        d = -d
        t, new_c5 = _line_search(h, d)
        if t < 1e-12:
            break
        h_new = h + t * d
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

Wait, this pair optimization is very basic. Let me think of a better approach.

For a pair (j, k), the objective as a function of δ is:
f(δ) = max_lag C_lag(δ)

where C_0(δ) = C_0(0) + 2(h[k]-h[j])δ - 2δ² (quadratic)
and C_lag(δ) = C_lag(0) + a_lag * δ for lag ≠ 0 (linear)

where a_lag = (1-h[j-lag])*I(valid) - h[j+lag]*I(valid) - (1-h[k-lag])*I(valid) + h[k+lag]*I(valid)

So f(δ) is the max of 1 quadratic and (2n-2) linear functions.

To find the minimum of f on [d_lo, d_hi]:
- The minimum occurs either at a breakpoint (where two pieces intersect) or at an endpoint
- Breakpoints: where C_0(δ) = C_lag(δ) for some lag, or where C_lag1(δ) = C_lag2(δ)

The C_0 = C_lag intersection: C_0(0) + 2(h[k]-h[j])δ - 2δ² = C_lag(0) + a_lag*δ
→ 2δ² - (2(h[k]-h[j]) - a_lag)δ + (C_0(0) - C_lag(0)) = 0

This is a quadratic equation with at most 2 solutions per lag. Total: at most 2*(2n-2) ≈ 4n solutions from C_0 vs C_lag intersections.

C_lag1 = C_lag2: (a_lag1 - a_lag2)*δ = C_lag2(0) - C_lag1(0)
→ δ = (C_lag2(0) - C_lag1(0)) / (a_lag1 - a_lag2) if a_lag1 ≠ a_lag2

This gives at most O(n²) intersections, but most will be outside [d_lo, d_hi].
