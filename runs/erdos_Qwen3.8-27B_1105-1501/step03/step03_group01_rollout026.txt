```python
def c5_eval(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum(v, target):
    """Project v onto [0,1]^n with sum = target using binary search on offset."""
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi)/2.0, 0.0, 1.0)

            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

    """Exact line search along direction D (assumed mean-zero).
    Returns (best_t, best_c5, new_h) or (0, c5(h), h) if no improvement."""

    dx = 2.0 / n
    c5_cur = c5_eval(h)
    
    # Scale D
    max_d = np.max(np.abs(D))
    if max_d < 1e-14:
        return 0.0, c5_cur, h

    pos_mask = D > 1e-12
    neg_mask = D < -1e-12
    t_hi_candidates = []
    t_lo_candidates = []

    if np.any(neg_mask):
        t_lo_candidates.append(np.min(h[neg_mask] / (-D[neg_mask])))
    
    if not t_hi_candidates or not t_lo_candidates:
        return 0.0, c5_cur, h

    t_hi = t_hi_candidates[0]
    
    if t_hi <= t_lo + 1e-12:
        return 0.0, c5_cur, h

    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
    Q = np.correlate(D, D, mode='full')

    span = (t_hi - t_lo) / n_samples

    b = min(t_hi, best_t + span)

            d = c

            c = d
            f_c = f_d
            d = a + gr * (b - a)
            f_d = np.max(S + d * B - d**2 * Q) * dx

    # Verify and project if needed
```

```python
def compute_soft_grad(h, weights):
    """weights is the array of softmax weights for each lag (length 2n-1).
    Returns the gradient of the weighted sum of correlations."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if abs(wk) < 1e-15:
            continue
        d = k - (n - 1)  # lag
        # dc_k/dh[i] = (1-h[i-d]) for i-d in [0,n), minus h[i+d] for i+d in [0,n)
        # First term: (1-h[i-d]) where 0 <= i-d < n => d <= i < d+n
        # Second term: -h[i+d] where 0 <= i+d < n => -d <= i < n-d
        if d >= 0:
            # First term: i in [d, n) => h[i-d] is valid
            g[d:] += wk * (1.0 - h[:n-d])
            # Second term: i in [0, n-d) => h[i+d] is valid
            g[:n-d] += -wk * h[d:]
        else:
            dd = -d
            # First term: i in [0, n-dd) => h[i-d] = h[i+dd] valid
            g[:n-dd] += wk * (1.0 - h[dd:])
            # Second term: i in [dd, n) => h[i+d] = h[i-dd] valid
            g[dd:] += -wk * h[:n-dd]
    return g
```

```python
def c5_eval(h):

def project_sum(v, target):

    if target <= 0:
        return np.zeros(n)

        return np.ones(n)
    lo = float(v.min()) - 2.0

    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid

def gen_block_dir(rng, n):

    mx = float(np.max(np.abs(D)))

def gen_sine_dir(rng, n):

        S = np.correlate(best_h, 1.0 - best_h, mode='full')
        dx = 2.0 / n
        one_minus_h = 1.0 - best_h
        improvement = 0.0
        best_D = None
        best_t = 0.0
        best_f = best_c5
        for di in range(n_dir):
            r = rng.random()
            if r < 0.40:
                tau = tau_list[rng.integers(0, len(tau_list))]
                D = gen_soft_dir(best_h, tau)
            elif r < 0.55:
                D = gen_block_dir(rng, n)
            elif r < 0.70:
                D = gen_sine_dir(rng, n)
            elif r < 0.85:
                D = gen_pair_dir(rng, n)
            else:
                D = gen_noise_dir(rng, n)
            max_d = float(np.max(np.abs(D)))
            if max_d < 1e-14:
                continue
            B = np.correlate(D, one_minus_h, mode='full') - np.correlate(best_h, D, mode='full')
            Q = np.correlate(D, D, mode='full')
            pos = D > 1e-12
            neg = D < -1e-12
            t_lo = 0.0
            t_hi = np.inf
            if np.any(pos):
                t_hi = float(np.min((1.0 - best_h[pos]) / D[pos]))
            if np.any(neg):
                t_lo = -float(np.min(best_h[neg] / (-D[neg])))
            if not np.isfinite(t_hi) or t_hi - t_lo < 1e-12:
                continue
            n_samp = min(50, max(15, int(30 * (t_hi - t_lo))))
            ts = np.linspace(t_lo, t_hi, n_samp)
            f_best_local = best_c5
            t_best_local = 0.0
            for i in range(n_samp):
                f_t = eval_line(S, B, Q, ts[i], dx)
                if f_t < f_best_local:
                    f_best_local = f_t
                    t_best_local = ts[i]
            if f_best_local < best_f - 1e-13:
                best_f = f_best_local
                best_D = D.copy()
                best_t = t_best_local
        if best_D is not None:
            h_try, c5_try, _ = line_search(best_h, best_D, n_samples=40, gold_iters=40)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try
                best_c5 = c5_try
                improvement = 1
            else:
                improvement = 0
        else:
            if it % 100 == 0:
                h_pert = best_h + rng.normal(0, 0.02, n)
                h_pert = project_sum(h_pert, target)
                c5_pert = c5_eval(h_pert)
                if c5_pert < best_c5 + 0.005:
                    best_h = h_pert
                    best_c5 = c5_eval(best_h)
        if best_c5 < 0.38083:
            break
    return best_h, best_c5

def polish_loop(h, deadline, rng, n_dir, tau_list):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h)
    stall = 0
    while time.time() < deadline:
        S = np.correlate(best_h, 1.0 - best_h, mode='full')
        dx = 2.0 / n
        one_minus_h = 1.0 - best_h
        improved = False
        for tau in tau_list:
            D = gen_soft_dir(best_h, tau)
            max_d = float(np.max(np.abs(D)))
            if max_d < 1e-14:
                continue
            _, c5_try, h_try = line_search(best_h, D, n_samples=50, gold_iters=50)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for _ in range(3):
                D = gen_pair_dir(rng, n)
                _, c5_try, h_try = line_search(best_h, D, n_samples=30, gold_iters=40)
                if c5_try < best_c5 - 1e-13:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
            if stall % 20 == 0:
                h_pert = best_h + rng.normal(0, 0.01, n)
                h_pert = project_sum(h_pert, n / 2.0)
                if c5_eval(h_pert) < best_c5 + 0.003:
                    best_h = h_pert
                    best_c5 = c5_eval(best_h)
                    stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None
    if h0 is not None and len(h0) > 0:
        best_h = project_sum(h0, len(h0) / 2.0)
    else:
        n_init = 84
        best_h = np.zeros(n_init)
        best_h[::2] = 1.0
        best_h = project_sum(best_h, n_init / 2.0)
    best_n = len(best_h)
    best_c5 = c5_eval(best_h)
    tau_list = [20.0, 50.0, 100.0, 200.0, 400.0]
    d1 = t0 + budget_s * 0.15
    best_h, best_c5 = batch_search(best_h, d1, rng, 15, tau_list, 5)
    best_n = len(best_h)
    d2 = t0 + budget_s * 0.55
    best_h, best_c5 = batch_search(best_h, d2, rng, 12, tau_list, 5)
    best_n = len(best_h)
    d3 = t0 + budget_s * 0.72
    n_base = best_n
    offsets = [0, -2, 2, -4, 4, -6, 6, 8, -8, 10, -10, 12, -12, 16, -16, 20, -20, 24, -24, 30, -30]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_base + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = batch_search(h_r, sub_dl, rng, 8, tau_list, 3)
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_r, c5_r = polish_loop(h_r, sub_dl2, rng, 8, tau_list[:3])
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r
            best_n = n_try
            n_base = best_n
    d4 = t0 + budget_s * 0.88
    best_h, best_c5 = polish_loop(best_h, d4, rng, 10, tau_list)
    best_n = len(best_h)
    d5 = t0 + budget_s * 0.95
    for attempt in range(200):
        if time.time() > d5:
            break
        scale = 0.015 + 0.02 * rng.random()
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = project_sum(h_pert, best_n / 2.0)
        c5_pert = c5_eval(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
    d6 = t0 + budget_s * 0.99
    best_h, best_c5 = polish_loop(best_h, d6, rng, 6, [50.0, 150.0, 300.0])
    best_n = len(best_h)
    best_h = project_sum(best_h, best_n / 2.0)
    final_c5 = c5_eval(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = project_sum(h0 if h0 is not None else best_h, (len(h0) if h0 is not None else best_n) / 2.0)
        final_c5 = c5_eval(best_h)
    return best_h, final_c5, best_n
```
