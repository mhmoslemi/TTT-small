    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    omh = 1.0 - h
    G = np.zeros((2*n-1, n))
    # m = 0
    G[n-1, :] = omh - h
    # m > 0
    for m in range(1, n):
        G[n-1+m, :n-m] += omh[m:]
        G[n-1+m, m:] -= h[:n-m]
    # m < 0
    for k in range(1, n):
        G[n-1-k, k:] += omh[:n-k]
        G[n-1-k, :n-k] -= h[k:]
    return G

def _pair_line_search(c, s, q, h_i, h_j, n_search=40):
    """Minimize f(a) = max_m(c[m] + a*s[m] + a*a*q[m]) over feasible a.
    Returns (best_a, best_val)."""
    a_lo = max(-h_i, h_j - 1.0)
    a_hi = min(1.0 - h_i, h_j)
    if a_lo >= a_hi:
        return 0.0, float(np.max(c))
    
    # Grid search
    a_grid = np.linspace(a_lo, a_hi, n_search)
    vals = np.empty(n_search)
    for idx, a in enumerate(a_grid):
        vals[idx] = float(np.max(c + a * s + a * a * q))
    
    idx_best = int(np.argmin(vals))
    best_val = vals[idx_best]
    best_a = a_grid[idx_best]
    
    # Golden section refinement
    step = (a_hi - a_lo) / n_search
    lo = max(a_lo, best_a - 2*step)
    hi = min(a_hi, best_a + 2*step)
    if hi - lo < 1e-12:
        return best_a, best_val
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(c + c1 * s + c1 * c1 * q))
    f2 = float(np.max(c + c2 * s + c2 * c2 * q))
    
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(c + c1 * s + c1 * c1 * q))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(c + c2 * s + c2 * c2 * q))
    
    a_star = (lo + hi) / 2.0
    f_star = float(np.max(c + a_star * s + a_star * a_star * q))
    if f_star < best_val:
        best_a, best_val = a_star, f_star
    
    return best_a, best_val
