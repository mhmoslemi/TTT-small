    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_offsets(h):
    n = len(h)
    omh = 1.0 - h
    raw = np.zeros(2 * n - 1)
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        off = abs(d)
        m = n - off
        if m > 0:
            raw[k] = float(np.dot(h[off:off+m], omh[:m]))
            grads[k, off:] += omh[:m]
            grads[k, :m] -= h[off:off+m]
    return raw, grads

def _solve_lp_dir(h, raw, grads, lam, R, top_k):
    n = len(h)
    # Select top-k offsets
    if top_k >= 2 * n - 1:
        sel = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(-raw)
        sel = top_idx[:top_k]
    
    n_sel = len(sel)
    # Build LP
    # Variables: [t, p_0..p_{n-1}, q_0..q_{n-1}], total 2n+1
    nvar = 2 * n + 1
    
    # Objective: min t + lam * (sum p + sum q)
    c = np.zeros(nvar)
    c[0] = 1.0
    c[1:1+n] = lam
    c[1+n:2*n+1] = lam
    
    # A_ub: inequality constraints
    # For each selected offset: -t + grad·p - grad·q <= -raw_s
    # Plus: sum(p) + sum(q) <= R
    A_ub = np.zeros((n_sel + 1, nvar))
    b_ub = np.zeros(n_sel + 1)
    
    for i, s in enumerate(sel):
        A_ub[i, 0] = -1.0
        A_ub[i, 1:1+n] = grads[s]
        A_ub[i, 1+n:2*n+1] = -grads[s]
        b_ub[i] = -raw[s]
    
    # Step bound
    A_ub[n_sel, 1:2*n+1] = 1.0
    b_ub[n_sel] = R
    
    # A_eq: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:2*n+1] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    eps = 1e-9
    for i in range(n):
        # p_i
        if h[i] >= 1.0 - eps:
            bounds.append((0, 0))
        else:
            bounds.append((0, None))
    for i in range(n):
        # q_i
        if h[i] <= eps:
            bounds.append((0, 0))
        else:
            bounds.append((0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    p = res.x[1:1+n]
    q = res.x[1+n:2*n+1]
    d = p - q
    d -= np.mean(d)  # ensure sum zero
    
    maxmag = float(np.max(np.abs(d)))
    if maxmag < 1e-12:
        return None
    
    return d / maxmag  # normalize

def _line_search(h, direction, target):
    """Find best alpha along direction, maintaining [0,1] bounds."""
    n = len(h)
    cur_c5 = _exact_c5(h)
    
    # Compute max valid alpha
    alpha_max = 10.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), 5.0)
    
    best_a = 0.0
    best_c = cur_c5
    
    # Coarse grid
    alphas = np.linspace(0.0, alpha_max, 20)
    for a in alphas[1:]:
        h_t = h + a * direction
        if np.any(h_t < -1e-9) or np.any(h_t > 1.0 + 1e-9):
            h_t = _project(h_t, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c:
            best_c = c_t
            best_a = a
    
    if best_a == 0.0:
        return h.copy(), cur_c5
    
    # Golden section refinement
    lo = max(0.0, best_a * 0.7)
    hi = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def _eval_alpha(a):
        h_t = h + a * direction
        if np.any(h_t < -1e-9) or np.any(h_t > 1.0 + 1e-9):
            h_t = _project(h_t, target)
        return _exact_c5(h_t)
    
    f1 = _eval_alpha(c1)
    f2 = _eval_alpha(c2)
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_alpha(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_alpha(c2)
    
    opt_a = (lo + hi) / 2.0
    opt_c = _eval_alpha(opt_a)
    if opt_c < best_c:
        best_c = opt_c
        best_a = opt_a
    
    h_new = h + best_a * direction
    if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
        h_new = _project(h_new, target)
    return h_new, best_c

def _lp_optimize(h, deadline, target, lam=0.01, R_frac=0.3, top_k=10):
    """LP-based optimization loop."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = R_frac * n
    stall = 0
    
    while time.time() < deadline:
        if stall > 30:
            break
        
        raw, grads = _compute_offsets(h)
        d = _solve_lp_dir(h, raw, grads, lam, R, top_k)
        
        if d is None:
            # Fallback: negative weighted gradient
            shifted = raw - float(np.max(raw))
            T = max(0.05, float(np.max(raw)) * 0.1)
            w = np.exp(shifted / T)
            wsum = float(np.sum(w))
            if wsum > 1e-30:
                w /= wsum
            else:
                w = np.zeros(len(raw))
                w[np.argmax(raw)] = 1.0
            d = np.zeros(n)
            for r in range(len(raw)):
                if w[r] > 1e-8:
                    d += w[r] * grads[r]
            d -= np.mean(d)
            gnorm = float(np.linalg.norm(d))
            if gnorm < 1e-10:
                break
            d /= gnorm

        h_new, c_new = _line_search(h, d, target)
        if c_new < cur_c5 - 1e-14:

            cur_c5 = c_new

                best_h = h.copy()
        else:

            if stall % 5 == 0:

                R *= 0.7
                top_k = min(2*n-1, top_k + 3)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        return None
    return d / maxmag

def _eval_alpha(h, direction, alpha, target):
    h_t = h + alpha * direction
    if np.any(h_t < -1e-9) or np.any(h_t > 1.0 + 1e-9):
        h_t = _project(h_t, target)
    return _exact_c5(h_t)

def _line_search(h, direction, target):
    n = len(h)
    cur_c5 = _exact_c5(h)
    alpha_max = 10.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), 5.0)
    best_a = 0.0
    best_c = cur_c5
    alphas = np.linspace(0.0, alpha_max, 25)
    for a in alphas[1:]:
        c_t = _eval_alpha(h, direction, a, target)
        if c_t < best_c:
            best_c = c_t
            best_a = a
    if best_a == 0.0:
        return h.copy(), cur_c5
    lo = max(0.0, best_a * 0.65)
    hi = min(alpha_max, best_a * 1.35)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _eval_alpha(h, direction, c1, target)
    f2 = _eval_alpha(h, direction, c2, target)
    for _ in range(30):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_alpha(h, direction, c1, target)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_alpha(h, direction, c2, target)
    opt_a = (lo + hi) / 2.0
    opt_c = _eval_alpha(h, direction, opt_a, target)
    if opt_c < best_c:
        best_c = opt_c
        best_a = opt_a
    h_new = h + best_a * direction
    if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
        h_new = _project(h_new, target)
    return h_new, best_c

def _fallback_dir(h, raw, grads):
    n = len(h)
    shifted = raw - float(np.max(raw))
    T = max(0.05, float(np.max(raw)) * 0.1)
    w = np.exp(shifted / T)
    wsum = float(np.sum(w))
    if wsum > 1e-30:
        w /= wsum
    else:
        w = np.zeros(len(raw))
        w[np.argmax(raw)] = 1.0
    d = np.zeros(n)
    for r in range(len(raw)):
        if w[r] > 1e-8:
            d += w[r] * grads[r]
    d -= np.mean(d)
    gnorm = float(np.linalg.norm(d))
    if gnorm < 1e-10:
        return None
    return d / gnorm

def _lp_optimize(h, deadline, target, lam=0.01, R_frac=0.3, top_k=10, max_stall=40):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = R_frac * n
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > max_stall:
            break
        it += 1
        raw, grads = _compute_offsets(h)
        d = _solve_lp_dir(h, raw, grads, lam, R, top_k)
        if d is None:
            d = _fallback_dir(h, raw, grads)
            if d is None:
                break
        h_new, c_new = _line_search(h, d, target)
        if c_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall % 5 == 0:
                lam *= 0.6
                R *= 0.7
                top_k = min(2 * n - 1, top_k + 2)
                if stall > 20:
                    nsmall = len(h)
                    h = _project(h + np.random.default_rng(it).normal(0, 0.02, nsmall), target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 > best_c5 + 0.001:
                        h = best_h.copy()
                        cur_c5 = best_c5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, rng, strength=0.05):
    n = len(h)
    target = n / 2.0
    r = rng.random()
    if r < 0.25:
        h_try = h + rng.normal(0, strength, n)
    elif r < 0.45:
        h_try = h.copy()
        cnt = max(1, n // 8)
        idx = rng.choice(n, size=cnt, replace=False)
        h_try[idx] += rng.normal(0, strength * 2.5, cnt)
    elif r < 0.6:
        h_try = (h + rng.normal(0, 0.08, n) > 0.5).astype(float)
    elif r < 0.75:
        shift = int(rng.integers(1, max(2, n // 6)))
        h_try = np.roll(h, shift)
    elif r < 0.9:
        h_try = 0.5 + (h - 0.5) * (0.7 + 0.6 * rng.random())
    else:
        h_rand = (rng.random(n) > 0.5).astype(float)
        h_try = 0.5 * h + 0.5 * h_rand
    return _project(h_try, target)

def _pair_polish(h, deadline, target, max_iters=500):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        raw, grads = _compute_offsets(h)
        max_raw = float(np.max(raw))
        active = np.where(raw > max_raw - 0.05 * max_raw)[0]
        improved = False
        n_active = len(active)
        if n_active > 0 and n_active <= 30:
            for ri in active:
                if time.time() >= deadline:
                    return best_h, best_c5
                g = grads[ri]
                for i in range(n):
                    if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                        continue
                    best_j = -1
                    best_val = 0.0
                    for j in range(n):
                        if j == i or h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                            continue
                        val = g[i] - g[j]
                        if val > best_val:
                            best_val = val
                            best_j = j
                    if best_j >= 0 and best_val > 1e-6:
                        a_max = min((1 - h[i]) / best_val, h[best_j] / best_val)
                        for frac in [0.4, 0.7, 1.0]:
                            a = a_max * frac
                            h_t = h.copy()
                            h_t[i] += a
                            h_t[best_j] -= a
                            c_t = _exact_c5(h_t)
                            if c_t < best_c5 - 1e-14:
                                best_c5 = c_t
                                best_h = h_t.copy()
                                h = h_t.copy()
                                cur_c5 = c_t
                                improved = True
                                break
        if not improved:
            for _ in range(30):
                if time.time() >= deadline:
                    return best_h, best_c5
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                if h[i] <= 1e-8 or h[i] >= 1 - 1e-8 or h[j] <= 1e-8 or h[j] >= 1 - 1e-8:
                    continue
                di = 1.0 if rng.random() > 0.5 else -1.0
                a_i = (1 - h[i]) / di if di > 0 else h[i] / (-di)
                a_j = (1 - h[j]) / (-di) if -di > 0 else h[j] / di
                a_max = min(a_i, a_j)
                if a_max <= 1e-10:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    a = a_max * frac
                    h_t = h.copy()
                    h_t[i] += di * a
                    h_t[j] -= di * a
                    c_t = _exact_c5(h_t)
                    if c_t < best_c5 - 1e-14:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        cur_c5 = c_t
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization with various parameters (30%)
    dl1 = t0 + budget_s * 0.30
    params = [
        (0.01, 0.3, 10),
        (0.0, 0.5, 20),
        (0.05, 0.2, 8),
        (0.001, 0.8, 30),
        (0.02, 0.4, 15),
    ]
    for (lam, Rf, tk) in params:
        if time.time() >= dl1:
            break
        sub_dl = min(dl1, time.time() + max(5.0, (dl1 - time.time()) * 0.45))
        h_new, c5_new = _lp_optimize(h, sub_dl, target0, lam=lam, R_frac=Rf, top_k=tk)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Multi-start with diverse perturbations (25%)
    dl2 = t0 + budget_s * 0.55
    n_starts = 20
    for s in range(n_starts):
        if time.time() >= dl2:
            break
        remaining = dl2 - time.time()
        if remaining < 4.0:
            break
        alloc = min(remaining * 0.85 / max(1, n_starts - s), 20.0)
        sub_dl = time.time() + alloc
        h_try = _perturb(best_h, rng, strength=0.04)
        h_opt, c5_opt = _lp_optimize(h_try, sub_dl, best_n / 2.0, lam=0.01, R_frac=0.3, top_k=10)
        h_opt, c5_opt = _pair_polish(h_opt, min(sub_dl, time.time() + 3.0), best_n / 2.0, max_iters=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Resolution sweep (25%)
    dl3 = t0 + budget_s * 0.80
    deltas = [-4, 4, -8, 8, -16, 16, -32, 32, -64, 64]
    for dn in deltas:
        if time.time() >= dl3:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + min(12.0, (dl3 - time.time()) * 0.35))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_try / 2.0, lam=0.01, R_frac=0.3, top_k=10)
        h_r, c5_r = _pair_polish(h_r, min(dl3, time.time() + 4.0), n_try / 2.0, max_iters=80)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Final fine polish (15%)
    dl4 = t0 + budget_s * 0.985
    h4, c5_4 = _pair_polish(best_h, dl4, best_n / 2.0, max_iters=2000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # One more LP pass with very fine parameters if time remains
    if time.time() < dl4 - 2.0:
        sub_dl = min(dl4, time.time() + max(2.0, (dl4 - time.time()) * 0.5))
        h5, c5_5 = _lp_optimize(best_h, sub_dl, best_n / 2.0, lam=0.001, R_frac=0.1, top_k=20, max_stall=20)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
