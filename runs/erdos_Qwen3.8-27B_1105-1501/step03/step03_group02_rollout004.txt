import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _smooth_direction(h, scores, beta):
    n = len(h)
    max_s = float(np.max(scores))
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for idx in range(len(scores)):
        if w[idx] > 1e-12:
            d = int(idx - (n - 1))
            g += w[idx] * _score_grad(h, d)
    g -= np.mean(g)
    norm = float(np.max(np.abs(g)))
    if norm < 1e-15:
        return None
    return -g / norm

def _antiperiodic_search(h, deadline):

    half = n // 2
    if n % 2 != 0:
        return h, _exact_c5(h)
    
    def _obj(y):

        return _exact_c5(h_vec)

    try:

                                         tol=1e-8, mutation=(0.3, 1.0),

            if c5_new < best_c5:

    except Exception:

    return best_h, best_c5
