import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for d in range(1, n):
        r = n - 1 + d
        m = n - d
        G[r, :m] += dx * (1.0 - h[d:d+m])
        G[r, d:n] -= dx * h[:m]
        r2 = n - 1 - d
        G[r2, d:n] += dx * (1.0 - h[:m])
        G[r2, :m] -= dx * h[d:d+m]
    r0 = n - 1
    G[r0, :] = dx * (1.0 - 2.0 * h)
    return G

def _socab_step(h, rho, solver_list):
    n = len(h)
    G = _build_G(h)
    F = _scores(h) * (2.0 / n)
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.norm(x - h) <= rho,
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        G @ (x - h) + F <= t
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for s in solver_list:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if x.value is not None:
                    return np.array(x.value).flatten(), float(t.value)
        except Exception:
            continue
    return None, None

def _lp_direction(h, scores, radius):
    n = len(h)
    dx = 2.0 / n
    G = _build_G(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -(scores * dx)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _socab_optimize(h, deadline, solver_list, rho_init=None, max_iters=5000):
    n = len(h)
    if rho_init is None:
        rho_init = 0.15 * np.sqrt(n)
    rho_max = 0.6 * np.sqrt(n)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = rho_init
    stall = 0
    solvers_tried = list(solver_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        x_sol, t_val = _socab_step(h, rho, solvers_tried)
        if x_sol is None:
            stall += 5
            rho *= 0.6
            if rho < 0.01:
                break
            continue
        best_trial_c5 = cur_c5
        best_x = None
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
            x_trial = h + alpha * (x_sol - h)
            x_trial = _project_box_sum(x_trial, n / 2.0)
            c5_t = _exact_c5(x_trial)
            if c5_t < best_trial_c5 - 1e-12:
                best_trial_c5 = c5_t
                best_x = x_trial
        if best_x is not None:
            h = best_x.copy()
            cur_c5 = best_trial_c5
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rho = min(rho * 1.25, rho_max)
        else:
            stall += 1
            rho *= 0.7
            if rho < 0.005:
                break
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.15, max_iters=3000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        best_a = 0.0
        best_c5_t = cur_c5
        for a in [1.0, 0.6, 0.35, 0.2]:
            x_t = _project_box_sum(h + a * p, n / 2.0)
            c5_t = _exact_c5(x_t)
            if c5_t < best_c5_t - 1e-12:
                best_c5_t = c5_t
                best_a = a
        if best_a > 0:
            h = _project_box_sum(h + best_a * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.2, 0.5)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _binary_greedy(h, deadline, max_iters=50000):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 100:
            break
        if not ones or not zeros:
            break
        best_swap = None
        best_val = cur_c5
        for i in ones:
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c5_new = _exact_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _rotate(h, k):
    return np.roll(h, k)

def _reflect(h):
    return h[::-1].copy()

def _mix(h1, h2, alpha):
    return alpha * h1 + (1.0 - alpha) * h2

def _grad_line_search(h, deadline, beta=20.0, max_iters=2000, step_init=0.05):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        max_s = float(np.max(scores))
        exp_vals = np.exp(beta * (scores - max_s))
        w_sum = float(np.sum(exp_vals))
        if w_sum < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w = exp_vals / w_sum
        grad = np.zeros(n)
        for d in range(-(n - 1), n):
            wd = w[n - 1 + d]
            if abs(wd) < 1e-14:
                continue
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += wd * (1.0 - h[d:])
                    grad[d:] += -wd * h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[g:] += wd * (1.0 - h[:m])
                    grad[:m] += -wd * h[g:]
        grad -= np.mean(grad)
        norm = float(np.max(np.abs(grad)))
        if norm < 1e-14:
            break
        d_dir = -grad / norm * step
        best_a = 0.0
        best_c5_t = cur_c5
        for a in [1.0, 0.6, 0.35, 0.2]:
            x_t = _project_box_sum(h + a * d_dir, n / 2.0)
            c5_t = _exact_c5(x_t)
            if c5_t < best_c5_t - 1e-12:
                best_c5_t = c5_t
                best_a = a
        if best_a > 0:
            h = _project_box_sum(h + best_a * d_dir, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.1, 0.3)
        else:
            stall += 1
            step *= 0.6
    return best_h, best_c5
