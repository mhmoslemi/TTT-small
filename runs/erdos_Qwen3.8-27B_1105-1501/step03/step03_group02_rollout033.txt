def _line_search(h, d, deadline):
    n = len(h)
    M = 2*n - 1
    # Compute A, B, C arrays
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    
    # Lag 0: r = n-1
    A[n-1] = np.sum(h*(1-h))
    B[n-1] = np.sum(d*(1-2*h))
    C[n-1] = -np.sum(d*d)
    
    # Positive lags k=1..n-1: r = n-1+k
    for k in range(1, n):
        m = n - k
        A[n-1+k] = np.sum(h[k:k+m]*(1-h[:m]))
        B[n-1+k] = np.sum(d[k:k+m]*(1-h[:m]) - h[k:k+m]*d[:m])
        C[n-1+k] = -np.sum(d[k:k+m]*d[:m])
    
    # Negative lags k=-1..-(n-1): r = n-1+k = n-1-k
    for k in range(1, n):
        m = n - k
        r = n-1-k
        A[r] = np.sum(h[:m]*(1-h[k:k+m]))
        B[r] = np.sum(d[:m]*(1-h[k:k+m]) - h[:m]*d[k:k+m])
        C[r] = -np.sum(d[:m]*d[k:k+m])
    
    # Feasible interval
    t_lo, t_hi = _feasible_interval(h, d)
    if t_hi - t_lo < 1e-14:
        return h, _exact_c5(h)
    
    # Candidates
    candidates = [t_lo, t_hi, 0.0]
    
    # Stationary points
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t0 = -B[r] / (2*C[r])
            if t_lo - 1e-12 <= t0 <= t_hi + 1e-12:
                candidates.append(t0)
    
    # Pair crossings (vectorized)
    # For pairs (i,j) with i < j:
    # (A[i]-A[j]) + (B[i]-B[j])*t + (C[i]-C[j])*t² = 0
    # a*t² + b*t + c = 0
    for i in range(M):
        a = C[i] - C[i+1:]
        b = B[i] - B[i+1:]
        c = A[i] - A[i+1:]
        # Where |a| is large: quadratic
        mask_quad = np.abs(a) > 1e-12
        mask_lin = ~mask_quad & (np.abs(b) > 1e-12)
        
        if np.any(mask_quad):
            disc = b[mask_quad]**2 - 4*a[mask_quad]*c[mask_quad]
            mask_real = disc >= -1e-12
            if np.any(mask_real):
                disc_real = np.maximum(disc[mask_real], 0)
                t1 = (-b[mask_quad][mask_real] - np.sqrt(disc_real)) / (2*a[mask_quad][mask_real])
                t2 = (-b[mask_quad][mask_real] + np.sqrt(disc_real)) / (2*a[mask_quad][mask_real])
                candidates.extend(t1.tolist())
                candidates.extend(t2.tolist())
        
        if np.any(mask_lin):
            t_lin = -c[mask_lin] / b[mask_lin]
            candidates.extend(t_lin.tolist())
    
    # Filter and deduplicate
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= t_lo - 1e-12) & (candidates <= t_hi + 1e-12)]
    candidates = np.clip(candidates, t_lo, t_hi)
    if len(candidates) > 0:
        candidates = np.unique(np.round(candidates, 12))
    
    # Evaluate E(t) at all candidates
    if len(candidates) > 0:
        # Vectorized: values[r, i] = A[r] + B[r]*t_i + C[r]*t_i²
        # Shape: (M, len(candidates))
        vals = A[:, None] + B[:, None] * candidates[None, :] + C[:, None] * candidates[None, :]**2
        E_vals = np.max(vals, axis=0)
        best_idx = np.argmin(E_vals)
        best_t = candidates[best_idx]
        best_E = E_vals[best_idx]
    else:
        best_t = 0.0
        best_E = float(np.max(A))
    
    # Try the best t
    if best_t != 0.0:
        h_new = h + best_t * d
        h_new = _repair(h_new, n/2.0)
        c5_new = _exact_c5(h_new)
        c5_old = _exact_c5(h)
        if c5_new < c5_old - 1e-14:
            return h_new, c5_new

def _line_coeffs(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.empty(M)
    B = np.empty(M)
    C = np.empty(M)
    
    # r = n-1 is lag 0
    A[n-1] = np.sum(h * (1.0 - h))
    B[n-1] = np.sum(d * (1.0 - 2.0 * h))
    C[n-1] = -np.sum(d * d)
    
    for k in range(1, n):
        m = n - k
        # Positive lag k: r = n-1+k
        A[n-1+k] = np.sum(h[k:k+m] * (1.0 - h[:m]))
        B[n-1+k] = np.sum(d[k:k+m] * (1.0 - h[:m]) - h[k:k+m] * d[:m])
        C[n-1+k] = -np.sum(d[k:k+m] * d[:m])

        C[n-1-k] = -np.sum(d[:m] * d[k:k+m])

def _exact_1d_min(A, B, C, t_lo, t_hi):
    M = len(A)
    candidates = [t_lo, t_hi, 0.0]
    
    # Stationary points of individual quadratics
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                candidates.append(t0)
    
    # Pair crossings
    for i in range(M - 1):
        a = C[i] - C[i+1:]
        b = B[i] - B[i+1:]
        c = A[i] - A[i+1:]
        
        # Quadratic case
        mask_q = np.abs(a) > 1e-12
        if np.any(mask_q):
            aq = a[mask_q]
            bq = b[mask_q]
            cq = c[mask_q]
            disc = bq*bq - 4.0*aq*cq
            mask_real = disc >= -1e-14
            if np.any(mask_real):
                aq2 = aq[mask_real]
                bq2 = bq[mask_real]
                disc2 = np.maximum(disc[mask_real], 0.0)
                sqrt_disc = np.sqrt(disc2)
                t1 = (-bq2 - sqrt_disc) / (2.0 * aq2)
                t2 = (-bq2 + sqrt_disc) / (2.0 * aq2)
                candidates.extend(t1.tolist())
                candidates.extend(t2.tolist())
        
        # Linear case
        mask_l = (~mask_q) & (np.abs(b) > 1e-12)
        if np.any(mask_l):
            t_lin = -c[mask_l] / b[mask_l]
            candidates.extend(t_lin.tolist())
    
    # Filter to interval
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= t_lo - 1e-10) & (candidates <= t_hi + 1e-10)]
    if len(candidates) == 0:
        return 0.0, float(np.max(A))
    candidates = np.clip(candidates, t_lo, t_hi)
    
    # Deduplicate
    candidates = np.sort(candidates)
    # Remove duplicates within tolerance
    keep = [0]
    for i in range(1, len(candidates)):
        if candidates[i] - candidates[i-1] > 1e-10:
            keep.append(i)
    candidates = candidates[keep]
    
    # Cap number of candidates
    if len(candidates) > 20000:
        # Keep endpoints and evenly spaced ones
        idx = np.linspace(0, len(candidates)-1, 20000).astype(int)
        candidates = candidates[idx]
    
    # Evaluate E(t) = max_r (A_r + B_r*t + C_r*t²)
    # Vectorized
    t_col = candidates[None, :]  # shape (1, K)
    vals = A[:, None] + B[:, None] * t_col + C[:, None] * t_col**2  # shape (M, K)
    E_vals = np.max(vals, axis=0)  # shape (K,)
    best_idx = np.argmin(E_vals)
    return float(candidates[best_idx]), float(E_vals[best_idx])

    candidates = [t_lo, t_hi, 0.0]
    
    # Stationary points
    for r in range(M):
        cr = C[r]
        if abs(cr) > 1e-14:
            t0 = -B[r] / (2.0 * cr)
            if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                candidates.append(t0)
    
    # Pair crossings
    for i in range(M - 1):
        a = C[i] - C[i+1:]
        b = B[i] - B[i+1:]
        c = A[i] - A[i+1:]
        
        mask_q = np.abs(a) > 1e-13
        if np.any(mask_q):
            aq = a[mask_q]
            bq = b[mask_q]
            cq = c[mask_q]
            disc = bq * bq - 4.0 * aq * cq
            mask_real = disc >= -1e-12
            if np.any(mask_real):
                aq2 = aq[mask_real]
                bq2 = bq[mask_real]
                d2 = np.maximum(disc[mask_real], 0.0)
                sd = np.sqrt(d2)
                candidates.extend(((-bq2 - sd) / (2.0 * aq2)).tolist())
                candidates.extend(((-bq2 + sd) / (2.0 * aq2)).tolist())

        if np.any(mask_l):
            candidates.extend((-c[mask_l] / b[mask_l]).tolist())

    candidates = candidates[np.isfinite(candidates)]
    candidates = candidates[(candidates >= t_lo - 1e-10) & (candidates <= t_hi + 1e-10)]
    if len(candidates) == 0:
        return 0.0
    candidates = np.clip(candidates, t_lo, t_hi)
    candidates = np.sort(candidates)
    
    # Deduplicate
    if len(candidates) > 1:
        diffs = np.diff(candidates)
        keep = np.concatenate([[True], diffs > 1e-10])
        candidates = candidates[keep]
    
    if len(candidates) > 25000:
        idx = np.linspace(0, len(candidates) - 1, 25000).astype(int)
        candidates = candidates[idx]
    
    # Evaluate
    tc = candidates[None, :]
    vals = A[:, None] + B[:, None] * tc + C[:, None] * tc * tc
    E_vals = np.max(vals, axis=0)
    best_idx = int(np.argmin(E_vals))
    return float(candidates[best_idx])

def _smooth_max_grad(h, beta=50.0):
    n = len(h)
    M = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(M)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for r in range(M):
        wr = w[r]
        if wr < 1e-14:
            continue
        k = r - (n - 1)
        if k == 0:
            grad += wr * (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            grad[:m] += wr * (-h[k:k+m])
            grad[k:] += wr * (1.0 - h[:m])
        else:
            g = -k
            m = n - g
            grad[:m] += wr * (1.0 - h[g:g+m])
            grad[g:] += wr * (-h[:m])
    
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return None
    return grad / norm
