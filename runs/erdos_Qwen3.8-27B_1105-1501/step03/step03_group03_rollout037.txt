def lp_direction(scores, G, active_indices, R, target_sum_zero=True):
    """
    Solve LP to find direction p that reduces active scores.
    max epsilon
    s.t. G[r,:] @ p <= (M - S[r]) - epsilon  for r in active
         sum(p) = 0
         -R <= p[i] <= R
    """
    n = len(scores)
    M = float(np.max(scores))
    m_active = len(active_indices)
    
    # Variables: p[0..n-1], epsilon
    # c = [0, ..., 0, -1] (maximize epsilon = minimize -epsilon)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: G[r,:] @ p + epsilon <= M - S[r]
    # => G[r,:] @ p + epsilon * 1 <= M - S[r]
    A_ub = np.zeros((m_active, n + 1))
    A_ub[:, :n] = G[active_indices, :]
    A_ub[:, -1] = 1.0
    b_ub = M - scores[active_indices]
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-R, R)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def lp_minimax_step(h, scores, G, active_set, R):
    """
    Solve: min M
           s.t. S[r] + G[r,:] @ (x-h) <= M  for r in active_set
                0 <= x_i <= 1
                sum(x) = n/2
                h_i - R <= x_i <= h_i + R
    Returns x* or None
    """
    n = len(h)
    target = n / 2.0
    m = len(active_set)
    
    # Variables: x[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: G[r,:] @ x - M <= G[r,:] @ h - S[r]
    # => G[r,:] @ x + (-1) * M <= G[r,:] @ h - S[r]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active_set, :]
    A_ub[:, -1] = -1.0
    b_ub = G[active_set, :] @ h - scores[active_set]
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: max(0, h_i - R) <= x_i <= min(1, h_i + R)
    bounds = [(max(0.0, h[i] - R), min(1.0, h[i] + R))] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n].copy()
    return None

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else: hi = mid

        else: mask = x > 0.0

import numpy as np
from scipy.optimize import linprog, minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)


def _lp_minimax(h, scores, G, active, R):
    n = len(h)
    m = len(active)
    target = n / 2.0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active, :]
    A_ub[:, -1] = -1.0
    b_ub = G[active, :] @ h - scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    lb = np.maximum(0.0, h - R)
    ub = np.minimum(1.0, h + R)
    # Check feasibility of sum constraint
    if lb.sum() > target + 1e-12 or ub.sum() < target - 1e-12:
        return None
    bounds = list(zip(lb, ub)) + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x = res.x[:n].copy()
            x = np.clip(x, 0.0, 1.0)
            if abs(float(x.sum()) - target) > 1e-8:
                x = _project(x, target)
            return x
    except Exception:
        pass
    return None


def _smooth_obj(x, beta):
    s = np.correlate(x, 1.0 - x, mode='full')
    smax = float(np.max(s))
    return smax + float(np.sum(np.log(np.exp(beta * (s - smax))))) / beta


def _smooth_jac(x, beta):
    G = _grad(x)
    s = np.correlate(x, 1.0 - x, mode='full')
    smax = float(np.max(s))
    e = np.exp(beta * (s - smax))
    w = e / e.sum()
    return G.T @ w


def _slsqp_pass(h, target, beta, max_iter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_jac, args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _sum_fun(x, target):
    return float(np.sum(x)) - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _line_search(h, p, cur_c5):

    pos = p > 1e-14

            best_c5 = c5t
            best_a = a
    if best_a == 0.0:

    lo = max(0.0, best_a - amax * 0.15)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
