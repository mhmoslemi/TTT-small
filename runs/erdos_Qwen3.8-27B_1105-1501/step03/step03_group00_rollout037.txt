max(-h[i], h[j]-1) <= a <= min(1-h[i], h[j])

For each such (i,j), I need to find the a that minimizes C5(h_new). This is a 1D optimization problem.

The key insight is that C5 as a function of a is piecewise quadratic (since s_r is bilinear in h, and we're only changing two components). So C5(a) = max_r s_r(a) where each s_r(a) is quadratic in a. The maximum of quadratics is convex! So we can use ternary search or golden section.

Wait, is that right? s_r(a) = ∑_i h'_i * (1 - h'_{i+k}) where h' differs from h only at positions i and j. Each term in the sum is either:
- Unchanged (if neither i nor j is involved)
- Linear in a (if exactly one of i,j is involved)
- Quadratic in a (if both i and j are involved, i.e., if j = i+k or i = j+k for some k)

So s_r(a) is at most quadratic in a, and C5(a) = max_r s_r(a) is the max of quadratics, which is convex. Great, so golden section will work.

Actually wait, I need to be more careful. The score s_r involves products h[i]*(1-h[i+r]). If I change h[i] and h[j], then:
- Terms involving h[i] but not h[j]: linear in a
- Terms involving h[j] but not h[i]: linear in a
- Terms involving both h[i] and h[j]: these are terms where {i, i+r} = {i, j} or {i, i+r} = {j, i}, i.e., r = j-i or r = i-j

So only the lags r = j-i and r = i-j are quadratic. All other lags are linear. The max of a set of linear and quadratic functions is still convex. Good.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_direction(h, p, eps=1e-12):
    """Compute negative p-norm gradient direction (to decrease the surrogate)."""
    s = _scores(h)
    s = np.maximum(s, eps)
    w = s ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(s))
        w[np.argmax(s)] = 1.0
    else:
        w /= w_sum
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[g:])
                grad[g:] += w[r] * (-h[:m])
    # This is the gradient of the p-norm; we want to go in -grad direction
    # But we need to preserve sum, so subtract mean
    grad -= np.mean(grad)
    return -grad  # negative gradient = descent direction

    alpha_max = max(alpha_max, 1e-12)

    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_a = alphas[best_i]
    best_c5 = vals[best_i]
    
    if best_i in (0, n_coarse - 1):
        return best_a, best_c5

    hi = alphas[best_i + 1]

    c2 = lo + gr * (hi - lo)

            f2 = f1
            c2 = c1

            lo = c1
            f1 = f2
            c1 = c2

            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:

    return best_a, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    m = len(scores_k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):

        improved = False
        for j in range(n):

                break
            if h[j] < 0.01 or h[j] > 0.99:

            for k in range(j + 1, n):
                if h[k] < 0.01 or h[k] > 0.99:
                    continue
                for sign in (1, -1):
                    a_max = min(1.0 - h[j] - 1e-10, h[k] - 1e-10) if sign > 0 \

                    if a_max < 1e-8:

                    # Ternary search
                    lo, hi = 0.0, a_max

                        h2[j] += sign * m2

                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)

                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _pnorm_direction(h, p, eps=1e-12):
    s = _scores(h)
    s_safe = np.maximum(s, eps)
    w = s_safe ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(s))
        w[np.argmax(s_safe)] = 1.0
    else:
        w /= w_sum
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        wr = w[r]
        if wr < 1e-18:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wr * (1.0 - h[d:])
                grad[d:] += wr * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += wr * (-h[:m])
    grad -= np.mean(grad)
    return -grad

def _line_search_c5(h, p, c5_cur, n_coarse=25):

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    m = len(scores_k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _pnorm_anneal(h, deadline, n_steps=300, p_start=4.0, p_end=300.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

                    if a_max < 1e-9:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
