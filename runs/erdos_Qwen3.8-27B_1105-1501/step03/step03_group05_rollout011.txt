import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair_box_sum(h, target):
    """Water-fill projection onto [0,1]^n with sum = target."""
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    for _ in range(100):
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-15
            count = mask.sum()
            if count == 0:
                break
            add = residual / count
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-15
            count = mask.sum()
            if count == 0:
                break
            sub = (-residual) / count
            h[mask] = np.maximum(h[mask] - sub, 0.0)
        residual = target - h.sum()
    # Final fixup
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] += residual / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _alpha_interval(h, v):
    """Compute feasible alpha interval for h + alpha*v staying in [0,1]."""
    n = len(h)
    lo = -np.inf
    hi = np.inf
    for i in range(n):
        vi = v[i]
        if vi > 1e-15:
            lo = max(lo, -h[i] / vi)
            hi = min(hi, (1.0 - h[i]) / vi)
        elif vi < -1e-15:
            lo = max(lo, (1.0 - h[i]) / vi)
            hi = min(hi, -h[i] / vi)
    lo = max(lo, -10.0)
    hi = min(hi, 10.0)
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if lo > hi:
        return None
    return (lo, hi)

def _line_search(h, v, cur_raw):
    """Exact line search: minimize max of quadratics over feasible alpha.
    
    Returns (best_alpha, best_raw_value) or (0, cur_raw) if no improvement.
    """
    n = len(h)
    interval = _alpha_interval(h, v)
    if interval is None:
        return 0.0, cur_raw
    lo, hi = interval
    if hi - lo < 1e-14:
        return 0.0, cur_raw
    
    # Compute A, B, C for the quadratics
    A = _raw_scores(h)  # length 2n-1
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    m = len(A)  # 2n-1
    
    # Generate candidate alphas
    candidates = [0.0, lo + 1e-14, hi - 1e-14]
    
    # Uniform grid
    n_grid = min(121, max(61, int(m)))
    grid = np.linspace(lo, hi, n_grid)
    candidates.extend(grid.tolist())
    
    # Stationary points
    for r in range(m):
        if abs(C[r]) > 1e-15:
            a = -B[r] / (2.0 * C[r])
            if lo - 1e-12 <= a <= hi + 1e-12:
                candidates.append(a)
    
    # Pairwise intersections (for n <= 96, i.e., m <= 191)
    if m <= 191:
        for i in range(m):
            for j in range(i+1, m):
                dC = C[j] - C[i]
                dB = B[j] - B[i]
                dA = A[j] - A[i]

                    if disc >= -1e-12:
                        disc = max(disc, 0.0)

    if len(candidates) == 0:
        return 0.0, cur_raw
    candidates = np.clip(candidates, lo, hi)

    # Evaluate each candidate

        if cost < best_cost - 1e-14:
            best_cost = cost
            best_alpha = a

    return best_alpha, best_cost

def _build_gradient(h):
    """Build the gradient matrix G of shape (2n-1, n) for raw scores."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            # dS_d/dh[q] gets (1-h[p]) from the h[p+d]*(1-h[p]) term where q=p+d
            G[r, q] += (1.0 - h[p])
            # dS_d/dh[p] gets -h[q] from the same term
            G[r, p] += (-h[q])
    return G

def _active_direction(h, beta):
    """Generate a descent direction using softmax-weighted active gradient."""
    n = len(h)
    scores = _raw_scores(h)
    max_s = np.max(scores)
    exp_vals = np.exp(beta * (scores - max_s))
    w = exp_vals / np.sum(exp_vals)
    G = _build_gradient(h)
    g = G.T @ w
    v = -g
    v -= np.mean(v)
    norm = np.max(np.abs(v))
    if norm < 1e-15:
        return None
    v /= norm
    return v

def _random_direction(rng, n):
    """Generate a random zero-sum direction."""
    v = rng.normal(size=n)
    v -= np.mean(v)
    norm = np.max(np.abs(v))
    if norm < 1e-15:
        return None
    v /= norm
    return v

def _top_k_direction(h, k):
    """Direction from top-k scores only."""
    n = len(h)
    scores = _raw_scores(h)
    idx = np.argsort(-scores)[:k]
    w = np.zeros(2*n-1)
    w[idx] = 1.0 / k
    G = _build_gradient(h)
    g = G.T @ w
    v = -g
    v -= np.mean(v)
    norm = np.max(np.abs(v))
    if norm < 1e-15:
        return None
    v /= norm
    return v

def _local_line_descent(h, deadline, rng, max_iters=5000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_box_sum(h_new, n_new / 2.0)

def _binarize(h):
    n = len(h)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def _flip(h):
    return h[::-1].copy()

def _roll(h, k):
    return np.roll(h, k).copy()

def _pair_polish(h, deadline, n_trials=20000, rng=None):

    if rng is None:

    cur_c5 = _exact_c5(h)

                best_h = h.copy()
                best_c5 = actual_c5

def _population_descent(pool, deadline, rng, per_item_time=30.0):
    """Local descent on each pool member."""
    for i in range(len(pool)):
        if time.time() > deadline:
            break
        h_i, c5_i = pool[i]
        sub_deadline = min(deadline, time.time() + per_item_time)
        h_new, c5_new = _local_line_descent(h_i, sub_deadline, rng, max_iters=2000)
        if c5_new < c5_i:
            pool[i] = (h_new, c5_new)
    # Sort by c5
    pool.sort(key=lambda x: x[1])
    return pool

def _build_G(h):
