def _project_to_feasible(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _logit_map(z, n):

def _grad_scores(h, lag):
    """Compute gradient of the score at a given lag position in np.correlate output."""
    n = len(h)
    # np.correlate(h, 1-h, 'full') output index r corresponds to shift k = r - (n-1)
    k = lag - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # c[k] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        m = n - k
        # d/dh[j]: from h[i+k] when i+k=j, i=j-k, valid j in [k, n-1]
        #   contrib: (1-h[j-k])
        if k <= n - 1:

            grad[idx] += (1.0 - h[:m])
        # from (1-h[i]) when i=j, valid j in [0, m-1]
        #   contrib: -h[j+k]
        idx = np.arange(0, m)
        grad[idx] -= h[k:k+m]
    else:
        g = -k
        # c[-g] = sum_{i=g}^{n-1} h[i-g] * (1-h[i])
        m = n - g

        #   contrib: (1-h[j+g])
        idx = np.arange(0, m)
        grad[idx] += (1.0 - h[g:g+m])
        # from (1-h[i]) when i=j, valid j in [g, n-1]
        #   contrib: -h[j-g]
        idx = np.arange(g, n)
        grad[idx] -= h[:m]
    return grad

def _c5_logsumexp(z, beta):
    """Log-sum-exp of scores, smooth surrogate for C5 (without dx factor)."""
    n = _g_n
    h = _logit_map(z, n)
    scores = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(scores))
    if beta < 1e-6:
        return M, h, scores
    exps = np.exp(beta * (scores - M))
    val = M + np.log(np.sum(exps)) / beta
    return val, h, scores

def _c5_logsumexp_grad(z, beta):
    """Value and gradient of log-sum-exp surrogate."""
    n = _g_n
    h = _logit_map(z, n)
    scores = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(scores))
    L = len(scores)
    if beta < 1e-6:
        val = M
        # Just use gradient of the max score
        kmax = int(np.argmax(scores))
        grad_h = _grad_scores(h, kmax)
    else:
        exps = np.exp(beta * (scores - M))
        weights = exps / float(np.sum(exps))
        val = M + np.log(float(np.sum(exps))) / beta
        # Compute grad_h = sum_k weights_k * grad(scores_k)
        grad_h = np.zeros(n)
        # Only compute for lags with significant weight
        threshold = float(np.max(weights)) * 1e-6
        for k in range(L):
            if weights[k] > threshold:
                grad_h += weights[k] * _grad_scores(h, k)
    
    # Convert grad_h to grad_z via logistic chain rule
    h_safe = np.clip(h, 1e-15, 1.0 - 1e-15)
    a = h_safe * (1.0 - h_safe)
    S = float(np.sum(a))
    if S < 1e-14:
        grad_z = np.zeros(n)
    else:
        S_term = float(np.dot(a, grad_h)) / S
        grad_z = a * (grad_h - S_term)
    
    return val, grad_z

        z_opt = z_init

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_to_feasible(h_new, n_new / 2.0)

def _project_to_feasible(v, target):

    eps = 1e-8
    h_safe = np.clip(h, eps, 1.0 - eps)
    z = np.log(h_safe) - np.log(1.0 - h_safe)

def _grad_score_k(h, k):

    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:

            grad[idx] += (1.0 - h[:m])
            idx2 = np.arange(0, m)
            grad[idx2] -= h[k:k + m]
    else:

            grad[idx] += (1.0 - h[g:g + m])
            idx2 = np.arange(g, n)
            grad[idx2] -= h[:m]

                    h_try[j] += d
                    if d > 0:
                        comp = int(np.argmax(np.where(np.arange(n) != j, 1.0 - h_try, -1)))
                    else:
                        comp = int(np.argmax(np.where(np.arange(n) != j, h_try, -1)))
                    if comp == j:
                        continue
                    h_try[comp] -= d
                    if h_try[comp] < 0 or h_try[comp] > 1:
                        continue
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _h_space_soft_descent(h, deadline, beta_start=20.0, beta_end=2000.0, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min((it + 1) / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(scores))
        exps = np.exp(np.clip(beta * (scores - M), -700, 0))
        weights = exps / float(np.sum(exps))
        grad = np.zeros(n)
        max_w = float(np.max(weights))
        threshold = max_w * 1e-6
        L = len(scores)
        for kk in range(L):
            if weights[kk] > threshold:
                grad += weights[kk] * _grad_score_k(h, kk - (n - 1))
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            if np.any(h_try < -0.2) or np.any(h_try > 1.2):
                alpha *= 0.5
                continue
            h_try = _project_to_feasible(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.3)
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-6 or stall > 40:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project_to_feasible(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: L-BFGS annealing from parent
    z_parent = _h_to_z(h, n0)
    deadline1 = t0 + budget_s * 0.20
    betas1 = [5.0, 20.0, 80.0, 300.0, 1000.0, 4000.0]
    h1, c5_1, z1 = _annealed_lbfgs(z_parent, n0, betas1, 120, deadline1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        z_parent = z1

    # Phase 2: H-space soft descent from current best
    deadline2 = t0 + budget_s * 0.35
    h2, c5_2 = _h_space_soft_descent(best_h, deadline2, 20.0, 3000.0, 8000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More L-BFGS with different schedules
    deadline3 = t0 + budget_s * 0.50
    z_best = _h_to_z(best_h, best_n)
    schedules = [
        [10.0, 60.0, 300.0, 1500.0],
        [3.0, 20.0, 100.0, 500.0, 2000.0],
        [30.0, 150.0, 800.0, 5000.0],
    ]
    for sched in schedules:
        if time.time() > deadline3:
            break
        time_rem = deadline3 - time.time()
        if time_rem < 1.0:
            break
        h3, c5_3, z_new = _annealed_lbfgs(z_best, best_n, sched, 80,
                                           min(deadline3, time.time() + time_rem / max(3, 1)))
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            z_best = z_new

    # Phase 4: Population-based exploration in z-space
    deadline4 = t0 + budget_s * 0.72
    z_best = _h_to_z(best_h, best_n)
    pool = [(z_best.copy(), best_c5)]

    for s in range(1, 10):
        h_shift = np.roll(best_h, s)
        c5_shift = _exact_c5(h_shift)
        if c5_shift < best_c5 + 0.04:
            pool.append((_h_to_z(h_shift, best_n), c5_shift))

    for scale in [0.3, 0.5, 0.7, 1.4, 1.6]:
        z_s = scale * z_best
        h_s = _logit_map(z_s, best_n)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5 + 0.08:
            pool.append((z_s.copy(), c5_s))

    for sigma in [0.05, 0.1, 0.2, 0.4, 0.6]:
        z_p = z_best + rng.normal(0, sigma, best_n)
        h_p = _logit_map(z_p, best_n)
        c5_p = _exact_c5(h_p)
        if c5_p < best_c5 + 0.06:
            pool.append((z_p.copy(), c5_p))

    pool.sort(key=lambda x: x[1])
    pool = pool[:25]

    opt_betas = [(8.0, 40.0, 200.0), (20.0, 100.0, 500.0), (5.0, 30.0, 150.0),
                 (15.0, 80.0, 400.0)]
    trial = 0
    for idx in range(len(pool)):
        if time.time() > deadline4:
            break
        time_per = (deadline4 - time.time()) / max(len(pool) - idx, 1)
        time_per = min(time_per, budget_s * 0.015)
        if time_per < 0.1:
            break
        sched = opt_betas[trial % len(opt_betas)]
        trial += 1
        h_pop, c5_pop, z_opt = _annealed_lbfgs(pool[idx][0], best_n, list(sched), 60,
                                                min(deadline4, time.time() + time_per))
        if c5_pop < best_c5:
            best_c5 = c5_pop
            best_h = h_pop.copy()
            z_best = z_opt
            pool = [(z_best.copy(), best_c5)] + pool[:14]
            pool = pool[:15]

    # Phase 5: Crossover and mutation
    deadline5 = t0 + budget_s * 0.82
    z_best = _h_to_z(best_h, best_n)
    for gen in range(300):
        if time.time() > deadline5:
            break
        if len(pool) >= 2:
            i = int(rng.integers(0, len(pool)))
            j = int(rng.integers(0, len(pool)))
            while j == i and len(pool) > 1:
                j = int(rng.integers(0, len(pool)))
            if i == j:
                continue
            alpha = rng.uniform(0.2, 0.8)
            z_child = alpha * pool[i][0] + (1.0 - alpha) * pool[j][0]
        else:
            continue
        sigma = rng.choice([0.03, 0.08, 0.15, 0.25])
        z_child = z_child + rng.normal(0, sigma, best_n)
        h_child = _logit_map(z_child, best_n)
        c5_child = _exact_c5(h_child)
        if c5_child < best_c5 + 0.02:
            time_per = (deadline5 - time.time()) / max(300 - gen, 1)
            time_per = min(time_per, budget_s * 0.006)
            if time_per < 0.05:
                break
            h_opt, c5_opt, z_opt = _annealed_lbfgs(z_child, best_n, [15.0, 80.0,
                                                                      400.0, 2000.0], 50,
                                                    min(deadline5, time.time() + time_per))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                z_best = z_opt
                pool = [(z_best.copy(), best_c5)] + pool[:12]
                pool = pool[:13]
            elif c5_opt < best_c5 + 0.015:
                pool.append((z_child.copy(), c5_opt))
                if len(pool) > 13:
                    pool.pop()

    # Phase 6: Resample to different n
    deadline6 = t0 + budget_s * 0.92
    n_candidates = []
    for dn in range(-12, 13, 4):
        n_try = best_n + dn
        if 32 <= n_try <= 128 and n_try % 2 == 0 and n_try != best_n:
            n_candidates.append(n_try)
    for n_try in n_candidates:
        if time.time() > deadline6:
            break
        h_r = _resample(best_h, n_try)
        z_r = _h_to_z(h_r, n_try)
        time_per = min((deadline6 - time.time()) / max(len(n_candidates), 1),
                       budget_s * 0.012)
        if time_per < 0.1:
            break
        h_opt, c5_opt, _ = _annealed_lbfgs(z_r, n_try, [10.0, 60.0, 300.0, 1500.0], 60,
                                            min(deadline6, time.time() + time_per))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 7: Final high-beta L-BFGS
    if time.time() < t0 + budget_s * 0.95:
        z_final = _h_to_z(best_h, best_n)
        h_f, c5_f, _ = _annealed_lbfgs(z_final, best_n, [50.0, 250.0, 1000.0, 5000.0,
                                                          15000.0], 150,
                                        t0 + budget_s * 0.96)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Coordinate descent polish
    if time.time() < t0 + budget_s * 0.985:
        h_cd, c5_cd = _coord_descent(best_h, t0 + budget_s * 0.99, max_passes=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final validation
    best_h = _project_to_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
