import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, d:d + m] += (1.0 - h[:m])
            G[r, :m] += -h[d:d + m]
        else:
            g = -d
            G[r, :m] += (1.0 - h[g:g + m])
            G[r, g:g + m] += -h[:m]
    return G * dx

def _golden_line_search(h, p, f0):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.999)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.999)
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 20
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.empty(n_samp)
    for i in range(n_samp):
        vals[i] = _exact_c5(h + alphas[i] * p)
    best_i = int(np.argmin(vals))
    best_f = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_f
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:

        if f1 < f2:

            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = 0.5 * (lo + hi)
    opt_f = _exact_c5(h + opt_a * p)
    if opt_f < best_f:
        return opt_a, opt_f
    return best_a, best_f

def _pnorm_optimize(h, deadline, beta_start=8, beta_end=400, n_steps=60, step_scale=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    step_mult = 1.0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        alpha = step_scale * step_mult
        improved = False
        for _ in range(30):
            h_new = _project(h + alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.65
            if alpha < 1e-12:
                break
        if improved:
            stall = 0
            step_mult = min(step_mult * 1.05, 3.0)
        else:
            stall += 1
            step_mult *= 0.7
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    from scipy.optimize import linprog
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-14:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.12, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _golden_line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.4)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 12 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _targeted_push(h, deadline, max_iters=200):

    n = len(h)

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.RandomState(123)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_pairs = min(800, n * (n - 1) // 2)
        for _ in range(n_pairs):
            if time.time() > deadline:
                break

            if h[j] < 1e-7 or h[j] > 1.0 - 1e-7 or h[k] < 1e-7 or h[k] > 1.0 - 1e-7:
                continue
            for sign in (1, -1):

                    f1 = _exact_c5(h1)

                    h2[j] += sign * m2
                    h2[k] -= sign * m2
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                a_opt = 0.5 * (lo + hi)
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
