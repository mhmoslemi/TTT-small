def _score_grad(h, d):
    """Gradient of Q_d(h) with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project(v, target):
    """Project v onto {x: 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_optimize(h, i, j, max_iter=100):
    """Optimize transfer between h[i] and h[j] to minimize C5"""
    n = len(h)
    h_work = h.copy()
    
    # Feasible range for t
    t_min = max(-h_work[j], -1.0 + h_work[i])  # h[i]+t <= 1, h[j]-t >= 0
    t_max = min(1.0 - h_work[i], h_work[j])
    
    # Golden section search on [t_min, t_max]
    # Minimize C5(h_work + t*(e_i - e_j))
    
    def obj(t):
        h_t = h_work.copy()
        h_t[i] += t
        h_t[j] -= t
        h_t = np.clip(h_t, 0.0, 1.0)
        return _exact_c5(h_t)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_min, t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(max_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    t_opt = (a + b) / 2.0
    h_work[i] += t_opt
    h_work[j] -= t_opt
    h_work = np.clip(h_work, 0.0, 1.0)
    return h_work

        per_dl = time.time() + min(rem, 30.0)
        result = _slsqp_solve(cur_h, active_lags, per_dl,
                              maxiter=200, ftol=1e-13)
        if result is None:
            break
        h_try, c5_try = result
        if c5_try < cur_c5 - 1e-14:
            cur_h = h_try
            cur_c5 = c5_try
            scores = _all_scores(cur_h)
            mx = float(np.max(scores))
            close = scores >= mx - 0.05 * max(1.0, mx / n)
            top_k = max(2, int(np.sum(close)))
            top_k = min(top_k, top_k_max)
        else:
            top_k = max(2, top_k - 2)
            if top_k < 3:
                break

    return cur_h, cur_c5

def _pair_cd(h_init, deadline, max_sweeps=200):
    n = len(h_init)
    target = n / 2.0
    cur_h = h_init.copy()
    cur_c5 = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    stall = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        sweep_improved = False
        g = _subgradient_vec(cur_h, beta=60.0)
        idx_pos = np.where(g > 0.01)[0]
        idx_neg = np.where(g < -0.01)[0]
        if len(idx_pos) == 0 or len(idx_neg) == 0:
            break

        pairs = []
        for i in idx_pos[:20]:
            for j in idx_neg[:20]:
                pairs.append((int(i), int(j)))
        np.random.shuffle(pairs)

        for (i, j) in pairs[:80]:
            if time.time() > deadline:
                break
            t_min = max(-cur_h[j], cur_h[i] - 1.0)
            t_max = min(1.0 - cur_h[i], cur_h[j])
            if t_max - t_min < 1e-12:
                continue
            a, b = t_min, t_max
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            hc = cur_h.copy(); hc[i] += c; hc[j] -= c
            fc = _exact_c5(np.clip(hc, 0, 1))
            hd = cur_h.copy(); hd[i] += d; hd[j] -= d
            fd = _exact_c5(np.clip(hd, 0, 1))
            for _ in range(25):
                if b - a < 1e-13:
                    break
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a)
                    hc = cur_h.copy(); hc[i] += c; hc[j] -= c
                    fc = _exact_c5(np.clip(hc, 0, 1))
                else:
                    a = c; c = d; fc = fd
                    d = a + gr * (b - a)
                    hd = cur_h.copy(); hd[i] += d; hd[j] -= d
                    fd = _exact_c5(np.clip(hd, 0, 1))
            t_opt = (a + b) / 2.0
            new_h = cur_h.copy()
            new_h[i] += t_opt
            new_h[j] -= t_opt
            new_h = np.clip(new_h, 0.0, 1.0)
            if abs(float(np.sum(new_h)) - target) > 1e-10:
                new_h = _project(new_h, target)
            c5_new = _exact_c5(new_h)
            if c5_new < cur_c5 - 1e-14:
                cur_h = new_h
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = new_h.copy()
                sweep_improved = True

        if sweep_improved:
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def _subgrad_phase(h_init, deadline, target):
    n = len(h_init)
    cur_h = h_init.copy()
    cur_c5 = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    alpha0 = 0.02
    beta = 50.0
    it = 0
    stall = 0

    while time.time() < deadline:
        it += 1
        alpha = alpha0 / (1.0 + it * 0.05)
        g = _subgradient_vec(cur_h, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn * alpha
        new_h = _project(cur_h - g, target)
        c5_new = _exact_c5(new_h)
        if c5_new < cur_c5 - 1e-15:
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
            cur_h = new_h
            cur_c5 = c5_new
            stall = 0
        else:
            cur_h = new_h
            stall += 1
            if stall > 80:
                alpha0 *= 0.7
                beta = min(beta * 1.3, 300.0)
                stall = 0
                if alpha0 < 1e-5:
                    break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl_total = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Adaptive SLSQP with broad active set (20%)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _adaptive_slsqp(h, dl1, top_k_max=12, max_rounds=8)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient polish (15%)
    dl2 = t0 + budget_s * 0.35
    h2, c5_2 = _subgrad_phase(best_h, dl2, best_n / 2.0)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP with tight active set (20%)
    dl3 = t0 + budget_s * 0.55
    h3, c5_3 = _adaptive_slsqp(best_h, dl3, top_k_max=6, max_rounds=10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair coordinate descent (25%)
    dl4 = t0 + budget_s * 0.80
    h4, c5_4 = _pair_cd(best_h, dl4, max_sweeps=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final SLSQP polish (10%)
    dl5 = t0 + budget_s * 0.90
    h5, c5_5 = _adaptive_slsqp(best_h, dl5, top_k_max=4, max_rounds=6)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Another pair CD pass (5%)
    dl6 = t0 + budget_s * 0.95
    h6, c5_6 = _pair_cd(best_h, dl6, max_sweeps=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Try nearby resolutions (5%)
    dl7 = t0 + budget_s * 0.99
    for dn in [2, -2, 4, -4, 8, -8]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        c5_r_init = _exact_c5(h_r)
        if c5_r_init >= best_c5 + 0.001:
            continue
        h_r2, c5_r2 = _adaptive_slsqp(h_r, min(dl7, time.time() + 15.0),
                                       top_k_max=6, max_rounds=4)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        # projection degraded slightly, try without
        h_no_proj = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        c5_no_proj = _exact_c5(h_no_proj)
        if c5_no_proj < final_c5:
            best_h = h_no_proj
            final_c5 = c5_no_proj
            best_n = len(h_no_proj)

    return best_h, final_c5, best_n
