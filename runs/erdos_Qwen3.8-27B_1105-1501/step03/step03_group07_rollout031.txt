import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_shift_values(h):
    """Compute all 2n-1 shift values (unscaled, i.e., F_r for r=-(n-1)..n-1)"""
    n = len(h)
    dx = 2.0 / n
    vals = np.zeros(2*n-1)
    one_minus_h = 1.0 - h
    for r in range(-(n-1), n):
        if r >= 0:
            L = n - r
            vals[r + n - 1] = dx * np.dot(h[:L], one_minus_h[r:r+L])
        else:
            k = -r
            L = n - k
            vals[r + n - 1] = dx * np.dot(h[k:], one_minus_h[:L])
    return vals

def _shift_gradient(h, r):
    """Gradient of F_r (unscaled) with respect to h"""
    n = len(h)
    grads = np.zeros(n)
    if r >= 0:
        L = n - r
        # First term: h[i] for i=0..L-1, contribution (1-h[i+r])
        grads[:L] += (1.0 - h[r:r+L])
        # Second term: -h[i]*h[i+r], h[i] part gives -h[i+r] (already included above)
        # h[i+r] part gives -h[i] for i=0..L-1, so indices r..n-1 get -h[0..L-1]
        grads[r:] -= h[:L]
    else:
        k = -r
        L = n - k
        # First term: h[j] for j=k..n-1, contribution (1-h[j-k])
        grads[k:] += (1.0 - h[:L])
        # Second term: -h[j]*h[j-k], 
        # h[j] part already included; h[j-k] part: j-k = 0..L-1, so indices 0..L-1 get -h[k..n-1]
        grads[:L] -= h[k:]
    return grads

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _shift_grad(h, r):
    n = len(h)
    g = np.zeros(n)
    if r >= 0:
        L = n - r
        if L > 0:
            g[:L] += (1.0 - h[r:r+L])
            g[r:] -= h[:L]
    else:
        k = -r
        L = n - k
        if L > 0:
            g[k:] += (1.0 - h[:L])
            g[:L] -= h[k:]
    return g

def _eval_lse(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = dx * corr
    vmax = float(np.max(vals))
    shifted = (vals - vmax) / tau
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    if sum_exp < 1e-300:
        obj = vmax
        w = np.zeros(len(vals))
        w[int(np.argmax(vals))] = 1.0
    else:
        obj = vmax + tau * np.log(sum_exp)
        w = exp_s / sum_exp
    return obj, w, vals

def _grad_lse(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-12:
            continue
        r = (n - 1) - idx
        grad += w[idx] * dx * _shift_grad(h, r)
    return grad

def _slsqp_lse(h0, tau, deadline, target, maxiter=2000):
    n = len(h0)
    h0 = np.clip(h0, 1e-12, 1.0 - 1e-12)

    def objfun(hv):
        obj, w, _ = _eval_lse(hv, tau)
        return obj

    def gradfun(hv):
        _, w, _ = _eval_lse(hv, tau)
        return _grad_lse(hv, w)

    constraints = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target}]
    bounds = [(1e-12, 1.0 - 1e-12)] * n

    try:
        res = minimize(objfun, h0, jac=gradfun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.fun < 10.0:
            hv = np.clip(res.x, 0.0, 1.0)
            hv = _project(hv, target)
            c5 = _exact_c5(hv)
            return hv, c5
    except Exception:
        pass
    return h0, _exact_c5(h0)

def _proj_subgrad_lse(h0, deadline, target, tau=0.02, max_iters=50000):

    c5 = _exact_c5(h)

    beta = 0.5

        if it % 2000 == 0 and stall > 100:
            break
        _, w, _ = _eval_lse(h, tau)
        grad = _grad_lse(h, w)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))

            break
        grad /= gnorm
        v = beta * v - alpha * grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_search(h, direction, target, c5_cur, max_alpha=1.5):
    n = len(h)
    a_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            a_max = min(a_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            a_max = min(a_max, h[j] / (-direction[j]))
    a_max = min(max(a_max, 1e-6), 5.0)
    best_a = 0.0
    best_c = c5_cur
    alphas = np.linspace(0.005, a_max, 15)
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-13:
            best_c = c_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo = max(0.0, best_a * 0.7)
    hi = min(a_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * direction, target))
    f2 = _exact_c5(_project(h + c2 * direction, target))
    for _ in range(15):
        if hi - lo < 1e-9:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * direction, target))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * direction, target))
    opt_a = (lo + hi) / 2.0
    opt_c = _exact_c5(_project(h + opt_a * direction, target))
    if opt_c < best_c:
        return opt_a, opt_c
    return best_a, best_c

                               tau_list=[0.003, 0.0015, 0.0008, 0.0004, 0.0002])
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-start NLP (20%)
    dl4 = t0 + budget_s * 0.80
    if time.time() < dl4:
        h4, c5_4 = _multi_start_nlp(best_h, dl4, best_n / 2.0, rng, n_starts=12)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Resolution sweep (12%)
    dl5 = t0 + budget_s * 0.92
    for dn in [-8, -4, -2, 2, 4, 8, -12, 12, 16, -16, 24, -24]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200 or n_try % 2 != 0:
            continue
        remaining = dl5 - time.time()
        if remaining < 5.0:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + min(25.0, remaining * 0.5))
        # Quick LSE at new resolution
        base = max(_exact_c5(h_r), 0.05)
        tau = max(base * 0.05, 0.005)
        h_r, c5_r = _slsqp_lse(h_r, tau, sub_dl, n_try / 2.0, maxiter=600)
        h_r = _project(h_r, n_try / 2.0)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final fine polish (8%)
    dl6 = t0 + budget_s * 0.99
    h6, c5_6 = _fine_polish(best_h, dl6, best_n / 2.0)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
