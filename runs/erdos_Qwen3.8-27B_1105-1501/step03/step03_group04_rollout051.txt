            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    return grad

np.correlate(h, 1-h, mode='full') gives the cross-correlation of h with (1-h). The output has length 2n-1.

For a sequence h of length n and q = 1-h of length n:
result[k] = sum_i h[i] * q[i + k - (n-1)]

where the sum is over valid i.

So result[n-1] = sum_i h[i] * q[i] = sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]²

result[n-1+d] = sum_i h[i+d] * q[i] = sum_{i=0}^{n-1-d} h[i+d](1-h[i]) for d ≥ 0

result[n-1-d] = sum_i h[i] * q[i+d] = sum_{i=0}^{n-1-d} h[i](1-h[i+d]) for d ≥ 0

So C₅ = max over all k of result[k] * (2/n).

For the binary case (h ∈ {0,1}), result[n-1] = 0 (since h(1-h)=0). The other lags give counts of specific patterns.

Now, for a non-binary h, result[n-1] = sum h[i] - sum h[i]² = n/2 - sum h[i]². To minimize this, we want sum h[i]² to be as large as possible, which means h should be as "binary" as possible (concentrated at 0 and 1). So the optimal solution is likely close to binary!

This is a key insight: the optimal h is close to {0,1}-valued. This means I should try optimizations that push h towards binary values.

For a binary h, the C₅ is:
C₅ = max_d count_10(d) * (2/n)

where count_10(d) = number of i such that h[i]=1 and h[i-d]=0 (for d>0) or h[i]=0 and h[i+d]=1 (for d<0), and count_10(0) = 0.

Wait, for d=0: count = sum h[i](1-h[i]) = 0 for binary h. Good.

For d>0: count_10(d) = number of i in [0, n-1-d] such that h[i+d]=1 and h[i]=0.
For d<0 (let g=-d>0): count_10(-g) = number of i in [0, n-1-g] such that h[i]=1 and h[i+g]=0.

Note that count_10(d) + count_10(-d) = Hamming distance between h and shift_d(h) (for the overlapping part).

So for binary h, C₅ = max_d count_10(d) * (2/n).

This is a combinatorial problem! For n=64, we need a binary sequence of length 64 with exactly 32 ones, such that the maximum "one-before-zero" count at any shift is minimized.

This is related to the concept of "balanced sequences" and "low-discrepancy sequences." The Thue-Morse sequence has the property that its autocorrelation is bounded, which might be optimal here.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:m]
    return grad

def _binary_local_search(h_start, deadline, max_rounds=2000):
    n = len(h_start)
    target = n // 2
    h = h_start.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    h[ones] = 1.0
    h[zeros] = 0.0
    cur_max = float(np.max(_scores(h)))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > 50:
            break
        improved = False
        one_list = list(np.where(h > 0.5)[0])
        zero_list = list(np.where(h <= 0.5)[0])
        for i in range(len(one_list)):
            oi = one_list[i]
            for j in range(len(zero_list)):
                zj = zero_list[j]
                h[oi] = 0.0
                h[zj] = 1.0
                new_max = float(np.max(_scores(h)))
                if new_max < cur_max - 1e-12:
                    cur_max = new_max
                    one_list[i] = zj
                    zero_list[j] = oi
                    improved = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                    break
                else:
                    h[oi] = 1.0
                    h[zj] = 0.0
            if time.time() > deadline:
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_max * (2.0 / n)

def _soft_descent(h, deadline, beta_start=30.0, beta_end=8000.0, max_iters=30000, top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.2
    stall = 0
    n_scores = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        frac = min(it / max(max_iters, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        if top_k is not None and top_k < n_scores:
            top_idx = np.argsort(scores)[::-1][:top_k]
            top_scores = scores[top_idx]
            exps = np.exp(beta * (top_scores - top_scores[0]))
            weights = np.zeros(n_scores)
            sw = float(np.sum(exps))
            if sw > 1e-30:
                weights[top_idx] = exps / sw
        else:
            m = float(np.max(scores))
            exps = np.exp(beta * (scores - m))
            sw = float(np.sum(exps))
            if sw < 1e-30:
                weights = np.zeros(n_scores)
                weights[int(np.argmax(scores))] = 1.0
            else:
                weights = exps / sw
        g = _grad(h, weights)
        g -= np.mean(g)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        direction = -g / gnorm
        alpha = lr
        improved = False
        best_a = 0.0
        best_c5_try = cur_c5
        for trial in range(35):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_a = alpha
            if c5_try < cur_c5 - 1e-14:
                improved = True
                break
            alpha *= 0.55
        if improved or (best_a > 0 and best_c5_try < cur_c5 - 1e-14):
            h = _project(h + best_a * direction, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.6)
        else:
            stall += 1
            lr *= 0.78
            if lr < 1e-6:
                break
    return best_h, best_c5

def _lp_polish(h, deadline, radius_init=0.08, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        m_val = float(np.max(scores))
        n_scores = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_scores, n + 1))
        G = np.zeros((n_scores, n))
        hh = h
        for k in range(n_scores):
            d = k - (n - 1)
            if d >= 0:
                mm = n - d
                if mm > 0:
                    G[k, d:] += (1.0 - hh[:mm])
                    G[k, :mm] -= hh[d:]
            else:
                gg = -d
                mm = n - gg
                if mm > 0:
                    G[k, :mm] += (1.0 - hh[gg:])
                    G[k, gg:] -= hh[:mm]
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores + m_val + 0.001
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo_b = max(-radius, -h[j])
            hi_b = min(radius, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 5000})
            if res.success:
                p = res.x[:n]
                pmax = float(np.max(np.abs(p)))
                if pmax < 1e-12:
                    stall += 1
                    radius *= 0.8
                    if radius < 1e-5:
                        break
                    continue
                alpha_max = 1.0
                for j in range(n):
                    if p[j] > 1e-14:
                        alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                    elif p[j] < -1e-14:
                        alpha_max = min(alpha_max, h[j] / (-p[j]))
                alpha_max = max(alpha_max, 1e-8)
                best_al = 0.0
                best_c5_al = cur_c5
                for a in np.concatenate([np.geomspace(0.02, alpha_max, 15),
                                         np.linspace(0, alpha_max, 8)]):
                    h_t = _project(h + a * p, target)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5_al:
                        best_c5_al = c5_t
                        best_al = a
                if best_c5_al < cur_c5 - 1e-12:
                    h = _project(h + best_al * p, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.15, 0.5)
                else:
                    stall += 1
                    radius *= 0.75
                    if stall % 15 == 0 and radius < 0.02:
                        radius = 0.05
            else:
                stall += 1
                radius *= 0.85
        except Exception:
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def _perturb(h, rng, scale):
    n = len(h)
    ptype = rng.integers(0, 5)
    if ptype == 0:
        h_new = h + rng.normal(0, scale, n)
    elif ptype == 1:
        h_new = h.copy()
        k = rng.integers(2, max(3, n // 4))
        start = rng.integers(0, n - k)
        shift = rng.integers(1, n // 2)
        block = h_new[start:start+k].copy()
        h_new[start:start+k] = h_new[(start+shift):(start+shift+k) % n]
        h_new[(start+shift):(start+shift+k) % n] = block if (start+shift+k) <= n else np.concatenate([block[n-(start+shift+k)+start:], block[:n-(start+shift+k)+start]])
        h_new = h_new + rng.normal(0, scale * 0.5, n)
    elif ptype == 2:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        h_new = h + scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif ptype == 3:
        h_new = h.copy()
        for _ in range(rng.integers(2, n // 4)):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = h_new + rng.normal(0, scale * 0.3, n)
    else:
        h_new = h + scale * rng.standard_t(df=4, size=n)
    return _project(h_new, n / 2.0)
