def _project_box_sum(v, target):
    n = len(v)
    # Binary search on the Lagrange multiplier
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_max(h, beta=50.0):
    """Compute smoothed gradient of max score using softmax weighting."""
    n = len(h)
    scores = _scores(h)
    max_s = scores.max()
    exp_vals = np.exp(beta * (scores - max_s))
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[scores.argmax()] = 1.0
    else:
        w = exp_vals / w_sum
    # Build gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[n-1+d] * dx
        if abs(wd) < 1e-15:

            m = n - d
            if m > 0:
                grad[:m] += wd * (1.0 - h[d:])
                grad[d:] -= wd * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[g:]

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_scores(h):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += dx * (1.0 - h[:m])
                G[r, :m] -= dx * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += dx * (1.0 - h[:m])
                G[r, :m] -= dx * h[g:]
    return G

def _smooth_grad(h, beta):

    scores = _scores(h)

        w[np.argmax(scores)] = 1.0
    else:
        w = exp_vals / w_sum
    G = _grad_scores(h)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))

        pass
    return None

                continue
            for sign in [1.0, -1.0]:
                a_max = 1e9
                if sign > 0:
                    a_max = min(a_max, (1.0 - h[i]) / 1.0, h[j] / 1.0)
                else:
                    a_max = min(a_max, h[i] / 1.0, (1.0 - h[j]) / 1.0)
                if a_max < 1e-10:
                    continue
                for frac in [1.0, 0.5, 0.2]:
                    a = a_max * frac
                    h_try = h.copy()
                    h_try[i] += sign * a
                    h_try[j] -= sign * a
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        cur_c5 = c5_try
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _multi_start_cont(h_candidates, deadline, beta_start=5.0, n_beta=8):
    best_h = None
    best_c5 = 1e9
    for h_c in h_candidates:
        if time.time() > deadline:
            break
        per_time = max(1.0, (deadline - time.time()) / max(1, len(h_candidates)))
        h_r, c5_r = _subgrad_optimize(h_c, min(deadline, time.time() + per_time),
                                       beta_start=beta_start, n_beta=n_beta)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 64
        h0 = (np.arange(n0) % 2).astype(float)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary SA at parent resolution (25% time)
    deadline1 = t0 + budget_s * 0.25
    if budget_s > 60:
        hb, c5b = _sa_binary(best_h, deadline1, rng, n_restarts=3)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()

    # Phase 2: Continuous subgradient optimization (20%)
    deadline2 = t0 + budget_s * 0.45
    h2, c5_2 = _subgrad_optimize(best_h, deadline2, beta_start=3.0, beta_end=100.0, n_beta=10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP trust region (20%)
    deadline3 = t0 + budget_s * 0.65
    h3, c5_3 = _lp_refine(best_h, deadline3, radius_init=0.2, max_iters=5000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Subgradient again from LP result (10%)
    deadline4 = t0 + budget_s * 0.75
    h4, c5_4 = _subgrad_optimize(best_h, deadline4, beta_start=10.0, beta_end=500.0, n_beta=8)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: LP again (10%)
    deadline5 = t0 + budget_s * 0.85
    h5, c5_5 = _lp_refine(best_h, deadline5, radius_init=0.1, max_iters=3000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution exploration (10%)
    deadline6 = t0 + budget_s * 0.95
    n_tries = []
    for dn in [-16, -8, -4, 4, 8, 16, 24, 32]:
        nt = best_n + dn
        if 24 <= nt <= 200:
            n_tries.append(nt)
    n_tries = sorted(set(n_tries))
    for n_try in n_tries:
        if time.time() > deadline6:
            break
        per_time = max(2.0, (deadline6 - time.time()) / max(1, len(n_tries)))
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline6, time.time() + per_time * 0.6)
        h_r, c5_r = _subgrad_optimize(h_r, dl_r, beta_start=5.0, n_beta=6)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if time.time() < deadline6:
                dl_lp = min(deadline6, time.time() + per_time * 0.4)
                h_rp, c5_rp = _lp_refine(best_h, dl_lp, radius_init=0.1, max_iters=500)
                if c5_rp < best_c5:
                    best_c5 = c5_rp
                    best_h = h_rp.copy()

    # Phase 7: Face moves polish (5%)
    if time.time() < t0 + budget_s * 0.98:
        deadline7 = t0 + budget_s * 0.98
        h7, c5_7 = _face_moves(best_h, deadline7, max_iters=3000)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final LP
    if time.time() < t0 + budget_s * 0.99:
        h8, c5_8 = _lp_refine(best_h, t0 + budget_s * 0.995, radius_init=0.05, max_iters=1000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            best_n = len(best_h)

    # Final validation
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
