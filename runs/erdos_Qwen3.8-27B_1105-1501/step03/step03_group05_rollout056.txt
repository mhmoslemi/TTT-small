    G = _build_G(h)  # already includes dx scaling

    # Variables: [x (n), t (1)]

    c[-1] = 1.0

    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))  # t >= 0

def _coord_pair_descent(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline or stall > 50:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for dj in [-3, -2, -1, 1, 2, 3]:
                j = i + dj
                if j < 0 or j >= n:
                    continue
                if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    delta_max = min(h[i] if sign > 0 else 1.0 - h[i],
                                    (1.0 - h[j]) if sign > 0 else h[j])
                    if delta_max < 1e-8:
                        continue
                    for frac in [0.5, 1.0, 0.25, 0.75]:
                        delta = delta_max * frac
                        h[i] -= sign * delta
                        h[j] += sign * delta
                        c5 = _exact_c5(h)
                        if c5 < best_c5 - 1e-13:
                            best_c5 = c5
                            best_h = h.copy()
                            improved = True
                            break
                        h[i] += sign * delta
                        h[j] -= sign * delta
                    if improved:
                        break
            if improved:
                break

        else:

def _resample(dense_h, n_new):
    n_old = len(dense_h)
    if n_new == n_old:
        return dense_h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, dense_h)
    return _project_box_sum(h_new, n_new / 2.0)

    n_candidates = sorted(set([
        best_n - 8, best_n - 4, best_n + 4, best_n + 8,
        best_n + 16, best_n - 16, 48, 56, 72, 80, 96, 112, 128, 64
    ]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 160]
    for n_try in n_candidates:
        if time.time() > d6:
            break
        remaining = d6 - time.time()
        if remaining < 10:
            break
        per_time = min(remaining * 0.4, budget_s * 0.035)
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_optimize(h_r, time.time() + per_time, radius_init=0.07)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if time.time() < d6:
                h_r2, c5_r2 = _coord_pair_descent(best_h, time.time() + 20, max_passes=30)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 7: Diversification - try different starts
    d7 = min(t0 + budget_s * 0.93, safe_deadline)
    if time.time() < d7 and budget_s > 300:
        for trial in range(3):
            if time.time() > d7:
                break
            h_pert = _random_feasible_perturb(best_h, rng, strength=0.15)
            h_pert = _lp_optimize(h_pert, time.time() + budget_s * 0.02, radius_init=0.06)
            c5_p = _exact_c5(h_pert)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()

    # Phase 8: Final polish
    d8 = min(t0 + budget_s * 0.97, safe_deadline)
    if time.time() < d8:
        hf, c5f = _lp_optimize(best_h, d8, radius_init=0.03)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()
        hc, c5c = _coord_pair_descent(best_h, min(d8 + 5, t0 + budget_s * 0.98), max_passes=50)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Final validation
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
