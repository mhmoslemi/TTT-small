def _scores_grad(h, weights):

    n = len(h)
    grad_h = np.zeros(n)
    nscores = len(weights)

        w = weights[r]
        if w < 1e-14:

            if mm <= 0:

            grad_h[:mm] -= w * h[s:]
        else:
            g = -s
            mm = n - g
            if mm <= 0:
                continue
            grad_h[:mm] += w * (1.0 - h[g:])
            grad_h[g:] -= w * h[:mm]

    z = np.log(h_clip) - np.log(1.0 - h_clip)

            x[mask] += diff / cnt
