Now for the LP approach:

Variables: t, p_0,...,p_{n-1}, q_0,...,q_{n-1} where d_i = p_i - q_i
- p_i ≥ 0, q_i ≥ 0
- If h_i ≈ 0: q_i = 0 (can only increase)
- If h_i ≈ 1: p_i = 0 (can only decrease)
- Σ(p_i - q_i) = 0 (preserve sum)
- Σ(p_i + q_i) ≤ R (bound the step size)
- For each active offset k: J_k + grad_k · d ≤ t (tangent plane constraint)

Objective: minimize t + λ * Σ(p_i + q_i)

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _offset_grad(h, d):
    """Gradient of overlap at offset d"""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad

def _lp_direction(h, raws, n_active=20, R=None, lam=0.0, eps=0.01):
    """Solve tangent LP to get descent direction"""
    n = len(h)
    if R is None:
        R = 0.3 * n
    
    # Select active offsets
    max_raw = float(np.max(raws))
    threshold = max_raw - 0.15 * max_raw  # offsets within 15% of max
    active = np.where(raws >= threshold)[0]
    if len(active) > n_active:
        active = np.argsort(-raws)[:n_active]
    
    # Variables: [t, p_0, ..., p_{n-1}, q_0, ..., q_{n-1}]
    # Total: 1 + 2n
    
    # Objective: min t + lam * sum(p + q)
    c = np.zeros(1 + 2 * n)
    c[0] = 1.0
    c[1:] = lam
    
    # A_ub constraints: for each active offset k
    # J_k + grad_k · (p - q) <= t
    # => -t + grad_k · p - grad_k · q <= -J_k
    A_rows = []
    b_rows = []
    
    for k in active:
        d = int(k) - (n - 1)
        g = _offset_grad(h, d)
        row = np.zeros(1 + 2 * n)
        row[0] = -1.0
        row[1:1+n] = g
        row[1+n:1+2*n] = -g
        A_rows.append(row)
        b_rows.append(-float(raws[k]))
    
    # Norm constraint: sum(p + q) <= R
    row = np.zeros(1 + 2 * n)
    row[1:] = 1.0
    A_rows.append(row)
    b_rows.append(R)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(p) - sum(q) = 0
    A_eq = np.zeros((1, 1 + 2 * n))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:1+2*n] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)]  # t is free
    for i in range(n):
        if h[i] < eps:
            bounds.append((0, None))  # p_i free
            bounds.append((0, 0))     # q_i = 0
        elif h[i] > 1 - eps:
            bounds.append((0, 0))     # p_i = 0
            bounds.append((0, None))  # q_i free
        else:
            bounds.append((0, None))
            bounds.append((0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    x = res.x
    p = x[1:1+n]
    q = x[1+n:1+2*n]
    d = p - q
    
    # Normalize
    norm_d = float(np.linalg.norm(d))
    if norm_d < 1e-12:
        return None
    d /= norm_d
    
    return d

def _line_search(h, direction, target, c5_cur, max_alpha=2.0, n_trials=12):
    """Find best step along direction"""
    n = len(h)
    # Find max valid alpha
    alpha_max = max_alpha
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = min(max(alpha_max, 1e-6), 5.0)
    
    best_a = 0.0
    best_c = c5_cur
    alphas = np.geomspace(0.01, max(alpha_max, 0.01), n_trials)
    alphas = alphas[alphas <= alpha_max + 1e-10]
    
    for a in alphas:
        h_t = _project(h + a * direction, target)
        c_t = _exact_c5(h_t)
        if c_t < best_c - 1e-14:
            best_c = c_t
            best_a = a
    
    # Refine with golden section
    if best_a > 0:
        lo = max(0.0, best_a * 0.7)
        hi = min(alpha_max, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * direction, target))
        f2 = _exact_c5(_project(h + c2 * direction, target))
        for _ in range(15):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * direction, target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * direction, target))
        opt_a = (lo + hi) / 2.0
        opt_c = _exact_c5(_project(h + opt_a * direction, target))
        if opt_c < best_c:
            best_a, best_c = opt_a, opt_c
    
    return best_a, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, target, max_passes=30):
    """Coordinate descent pair moves"""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        raws = np.correlate(h, 1.0 - h, mode='full')
        
        # Gradient-guided selection
        shifted = raws - float(np.max(raws))
        w = np.exp(shifted / 0.2)
        w /= float(np.sum(w))
        grad = np.zeros(n)
        for r in range(2*n-1):
            if w[r] < 1e-8:
                continue
            d = int(r) - (n-1)
            grad += w[r] * _offset_grad(h, d)
        grad -= np.mean(grad)
        
        top = np.argsort(-np.abs(grad))[:min(n, 16)]
        
        for ii in range(len(top)):
            if time.time() > deadline:
                return best_h, best_c5
            j = top[ii]
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for jj in range(len(top)):
                if jj == ii:
                    continue
                k = top[jj]
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:

                    a_max_k = (1.0 - h[k])/dk if dk > 0 else h[k]/(-dk)
                    a_max = min(a_max_j, a_max_k)
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.5, 0.7, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c_t = _exact_c5(h_t)
                        if c_t < best_c5 - 1e-13:
                            best_c5 = c_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        # Random pairs
        for _ in range(100):
            if time.time() > deadline:
                return best_h, best_c5
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j == k or h[j] <= 1e-8 or h[j] >= 1.0-1e-8 or h[k] <= 1e-8 or h[k] >= 1.0-1e-8:
                continue
            dj = 1.0 if rng.random() > 0.5 else -1.0
            dk = -dj
            a_max = min((1-h[j])/dj if dj>0 else h[j]/(-dj),
                        (1-h[k])/dk if dk>0 else h[k]/(-dk))
            if a_max <= 0:

            for a in [a_max*0.4, a_max*0.7, a_max]:
                h_t = h.copy()
                h_t[j] += dj*a
                h_t[k] += dk*a
                c_t = _exact_c5(h_t)
                if c_t < best_c5 - 1e-13:
                    best_c5 = c_t
                    best_h = h_t.copy()
                    h = h_t.copy()
                    improved = True
                    break
        if not improved:

    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    v = np.nan_to_num(np.asarray(v, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _offset_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    else:
        l = -d
        m = n - l
        if m > 0:
            grad[:m] += (1.0 - h[l:])
            grad[l:] -= h[:m]
    return grad


def _lp_direction(h, raws, n_active=15, R=None, lam=0.0, eps=0.005):
    n = len(h)
    if R is None:
        R = 0.25 * n
    max_raw = float(np.max(raws))
    if max_raw < 1e-12:
        return None
    threshold = max_raw - 0.20 * max_raw
    active = np.where(raws >= threshold)[0]
    if len(active) > n_active:
        active = np.argsort(-raws)[:n_active]
    if len(active) == 0:
        active = np.array([np.argmax(raws)])

    nv = 1 + 2 * n
    c = np.zeros(nv)
    c[0] = 1.0
    c[1:] = lam

    A_rows = []
    b_rows = []
    for k in active:
        d = int(k) - (n - 1)
        g = _offset_grad(h, d)
        row = np.zeros(nv)
        row[0] = -1.0
        row[1:1 + n] = g
        row[1 + n:1 + 2 * n] = -g
        A_rows.append(row)
        b_rows.append(-float(raws[k]))

    row = np.zeros(nv)
    row[1:] = 1.0
    A_rows.append(row)
    b_rows.append(R)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:1 + 2 * n] = -1.0
    b_eq = np.zeros(1)

    bounds = [(-1e4, 1e4)]
    for i in range(n):
        if h[i] < eps:
            bounds.append((0.0, None))
            bounds.append((0.0, 0.0))
        elif h[i] > 1.0 - eps:
            bounds.append((0.0, 0.0))
            bounds.append((0.0, None))
        else:
            bounds.append((0.0, None))
            bounds.append((0.0, None))

    try:

    if not res.success or res.x is None:
        return None
    p = res.x[1:1 + n]
    q = res.x[1 + n:1 + 2 * n]
    d = p - q
    d -= np.mean(d)
    norm_d = float(np.linalg.norm(d))
    if norm_d < 1e-10:

    return d / norm_d

def _softmax_grad(h, raws, T=0.25):

    w = np.exp(shifted / T)

    if wsum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(raws))] = 1.0

        w /= wsum
    grad = np.zeros(n)

        if w[r] < 1e-7:
            continue
        d = int(r) - (n - 1)
        grad += w[r] * _offset_grad(h, d)
    return grad

        elif r < 0.65:
            shift = int(rng.integers(1, max(2, best_n // 5)))
            h_try = _project(np.roll(best_h, shift), target0)
        elif r < 0.8:
            h_rand = (rng.random(best_n) > 0.5).astype(float)
            h_try = _project(0.5 * best_h + 0.5 * h_rand, target0)
        elif r < 0.9:
            # Structured: periodic
            half = best_n // 2
            u = 0.5 * (best_h[:half] + best_h[half:])
            h_try = _project(np.tile(u, 2) if len(u) * 2 == best_n else np.tile(u, max(1, best_n // len(u))), target0)
            h_try = _project(h_try[:best_n], target0)
        else:
            # Scale perturbation
            h_try = _project(0.5 + (best_h - 0.5) * (0.7 + 0.6 * rng.random()), target0)

        h_opt, c5_opt = _lm_optimize(h_try, sub_dl, target0, max_iters=3000)
        if c5_opt < best_c5 - 1e-12:
            # Polish the new best
            sub_dl2 = min(dl3, time.time() + 5.0)
            h_opt, c5_opt = _pair_polish(h_opt, sub_dl2, target0, max_passes=15)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Resolution sweep (15%)
    dl4 = t0 + budget_s * 0.85
    for dn in [-4, -2, 2, 4, -8, 8, -12, 12, -16, 16, -20, 20, 24, -24]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + 12.0)
        h_r, c5_r = _lm_optimize(h_r, sub_dl, n_try / 2.0, max_iters=2000)
        if c5_r < best_c5 - 1e-10:
            sub_dl2 = min(dl4, time.time() + 5.0)
            h_r, c5_r = _pair_polish(h_r, sub_dl2, n_try / 2.0, max_passes=10)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 5: Final fine polish (10%)
    dl5 = t0 + budget_s * 0.985
    h5, c5_5 = _fine_local(best_h, dl5, best_n / 2.0)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-8:
        best_h = _project(h0, target0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
