3. All pairwise crossing points of parabolas r and s within [tlo, thi]

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    diff = target - float(np.sum(v))
    if abs(diff) < 1e-10:
        return v
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) * 0.5
    x = np.clip(v - shift, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-9:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_abc(h, d):
    n = len(h)
    M = 2 * n - 1
    A = np.empty(M)
    B = np.empty(M)
    C = np.empty(M)
    one_minus_h = 1.0 - h
    for r in range(M):
        l = r - (n - 1)
        if l == 0:
            A[r] = float(np.dot(h, one_minus_h))
            B[r] = float(np.dot(d, one_minus_h - h))
            C[r] = -float(np.dot(d, d))
        elif l > 0:
            m = n - l
            hl = h[l:l+m]
            h0 = h[:m]
            dl = d[l:l+m]
            d0 = d[:m]
            A[r] = float(np.dot(hl, 1.0 - h0))
            B[r] = float(np.dot(dl, 1.0 - h0) - np.dot(hl, d0))
            C[r] = -float(np.dot(dl, d0))
        else:
            g = -l
            m = n - g
            h0 = h[:m]
            hg = h[g:g+m]
            d0 = d[:m]
            dg = d[g:g+m]
            A[r] = float(np.dot(h0, 1.0 - hg))
            B[r] = float(np.dot(d0, 1.0 - hg) - np.dot(h0, dg))
            C[r] = -float(np.dot(d0, dg))
    return A, B, C

def _line_search(h, d):
    n = len(h)
    M = 2 * n - 1
    tlo, thi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if di > 1e-15:
            tlo = max(tlo, -h[i] / di)
            thi = min(thi, (1.0 - h[i]) / di)
        elif di < -1e-15:
            tlo = max(tlo, (1.0 - h[i]) / di)
            thi = min(thi, -h[i] / di)
    tlo = min(tlo, 0.0)
    thi = max(thi, 0.0)
    if thi - tlo < 1e-14:
        return 0.0, None
    A, B, C = _compute_abc(h, d)
    cands = [tlo, thi]
    mask_v = np.abs(C) > 1e-14
    if np.any(mask_v):
        vc = -B[mask_v] / (2.0 * C[mask_v])
        vc = vc[(vc >= tlo) & (vc <= thi)]
        cands.extend(vc.tolist())
    for i in range(M):
        Cai = C[i]
        Bi_ = B[i]
        Ai_ = A[i]
        for j in range(i + 1, M):
            a = Cai - C[j]
            b = Bi_ - B[j]
            c = Ai_ - A[j]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:
                    tr = -c / b
                    if tlo <= tr <= thi:
                        cands.append(tr)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= 0.0:
                    sd = np.sqrt(disc)
                    t1 = (-b + sd) / (2.0 * a)
                    t2 = (-b - sd) / (2.0 * a)
                    if tlo <= t1 <= thi:
                        cands.append(t1)
                    if tlo <= t2 <= thi:
                        cands.append(t2)
    cands = np.array(cands)
    cands = np.clip(cands, tlo, thi)
    cands = np.unique(np.round(cands, 14))
    t2 = cands * cands
    vals = A[None, :] + B[None, :] * cands[:, None] + C[None, :] * t2[:, None]
    E_vals = vals.max(axis=1)
    bi = int(np.argmin(E_vals))
    return float(cands[bi]), float(E_vals[bi])

def _grad_score(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = n - lag
        g[:m] += -h[lag:lag+m]
        g[lag:] += (1.0 - h[:m])
    else:
        gg = -lag
        m = n - gg
        g[:m] += (1.0 - h[gg:gg+m])
        g[gg:] += -h[:m]
    return g

def _weighted_dir(h, weights):
    n = len(h)
    M = 2 * n - 1
    d = np.zeros(n)
    for r in range(M):
        if abs(weights[r]) > 1e-15:
            d += weights[r] * _grad_score(h, r - (n - 1))
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None
    return -d / norm

def _random_dir(rng, n):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = float(np.max(np.abs(d)))
    if norm < 1e-14:
        return None
    return d / norm

def _pair_dirs(h, rng, n, count=4):
    dirs = []
    interior = np.where((h > 0.02) & (h < 0.98))[0]
    if len(interior) < 2:
        return dirs
    for _ in range(count):
        i = interior[int(rng.integers(len(interior)))]
        j = interior[int(rng.integers(len(interior)))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        dirs.append(d)
    return dirs

    rng = np.random.default_rng(seed)

    beta = 30.0

    it = 0
    pair_count = 3

        if best_c5 <= target_c5:

        # Reseed on long stall

            h_try = _project(best_h + rng.normal(0, scale, n), target_sum)

            if c5_try < best_c5:

                best_h = h_try.copy()
                h = h_try.copy()
            else:
                h = best_h.copy()
            stall = 0
            beta = 20.0
            continue

        cur_scores = _scores_arr(h)
        cur_max = float(np.max(cur_scores))
        cur_c5 = cur_max * (2.0 / n)

        dirs = []

        # 1. Smooth max gradient
        max_s = cur_max
        w = np.exp(beta * (cur_scores - max_s))

                    h = h_new
                    cur_c5 = new_c5
                    if new_c5 < best_c5 - 1e-14:
                        best_c5 = new_c5
                        best_h = h.copy()

                pair_count = min(8, pair_count + 1)

    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n
