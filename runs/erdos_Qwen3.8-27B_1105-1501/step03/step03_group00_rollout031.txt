def _line_search(h, p):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 20
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lower = max(-radius, -h[j])
        upper = min(radius, 1.0 - h[j])
        bounds.append((lower, upper))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _smooth_grad(h, tau):
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp((scores - max_s) / tau)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _nesterov_phase(h, deadline, tau_init=0.015, tau_end=0.002, max_iters=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    beta = 0.90
    stall = 0
    alpha = 0.08
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        frac = float(it) / max(1, max_iters)
        tau = tau_init * (tau_end / tau_init) ** frac
        beta = 0.90 - 0.20 * frac
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        y = _project(y, target)
        grad = _smooth_grad(y, tau)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        gnorm = grad / gn
        v = beta * v + gnorm
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v = v / vn
        alpha_try = alpha
        h_new = _project(h - alpha_try * v * gn, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            alpha = min(alpha * 1.05, 0.3)
        else:
            stall += 1
            alpha *= 0.85
            if alpha < 0.001:
                stall = 500
    return best_h, best_c5

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.12, max_iters=6000):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.4)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _ga_phase(h, deadline, n_moves=None, T_init=0.004, T_end=0.0003):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    if n_moves is None:
        n_moves = 20000
    T = T_init
    cool = (T_end / T_init) ** (1.0 / max(1, n_moves))
    rng = np.random.default_rng(12345)
    for mv in range(n_moves):
        if time.time() > deadline:
            break
        T *= cool
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        d_max = min(h[i], 1.0 - h[j])
        if d_max < 1e-10:
            continue
        d = rng.uniform(-1.0, 1.0) * min(0.04, d_max)
        if abs(d) < 1e-12:
            continue
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c5_new = _exact_c5(h)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _structured_starts(h0, rng):
    n = len(h0)
    candidates = []
    candidates.append(h0.copy())
    candidates.append(h0[::-1].copy())
    candidates.append(1.0 - h0)
    candidates.append(_project(1.0 - h0, n / 2.0))
    for shift in (1, 2, 3, 5, 8, n // 4, n // 3):
        candidates.append(_project(np.roll(h0, shift), n / 2.0))
        candidates.append(_project(np.roll(h0, -shift), n / 2.0))
    kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
    hc = np.convolve(h0, kernel, mode='same')
    candidates.append(_project(hc, n / 2.0))
    for freq in (1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        x = (np.arange(n) + 0.5) / n
        for amp in (0.05, 0.1, 0.15):
            base = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
            candidates.append(_project(base, n / 2.0))
    return candidates
