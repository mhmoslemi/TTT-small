    """Solve LP to minimize max over active lags."""
    n = len(h)
    # The LP:
    # minimize t
    # subject to: G[active] @ d - t <= 0 (for each active lag)
    #             sum(d) = 0
    #             -radius <= d_i <= radius (and box constraints)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = G[active_lags]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_lags))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    return G * (2.0 / n)

def _line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 0.8
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 40
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(45):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _solve_lp(h, active_idx, G, radius, n):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    m = len(active_idx)
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active_idx]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] - 1e-10), min(radius, 1.0 - h[j] + 1e-10)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _remez_optimize(h, deadline, max_active=15, radius_init=0.2, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    # Initialize active set: top 10 lags
    scores = _scores(h)
    active = list(np.argsort(scores)[-max_active:])
    active_set = set(active)
    max_active_size = max_active
    min_active_size = 5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        G = _grad_matrix(h)
        # Solve LP with current active set
        direction = _solve_lp(h, active, G, radius, n)
        if direction is None or np.max(np.abs(direction)) < 1e-14:
            # Try with smaller active set
            if len(active) > min_active_size:
                scores_cur = _scores(h)
                active = list(np.argsort(scores_cur)[-min_active_size:])
                active_set = set(active)
                continue
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, direction, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * direction
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.6)
            # Update active set: Remez exchange
            scores_new = _scores(h)
            new_max_idx = int(np.argmax(scores_new))
            if new_max_idx not in active_set:
                active.append(new_max_idx)
                active_set.add(new_max_idx)
                if len(active) > max_active_size:
                    # Remove the least active lag
                    scor = scores_new
                    min_score_in_active = min(scor[a] for a in active)
                    remove_idx = [a for a in active if scor[a] == min_score_in_active][0]
                    active.remove(remove_idx)
                    active_set.discard(remove_idx)
        else:
            stall += 1
            radius *= 0.8
            if stall % 10 == 0:
                # Refresh active set
                scores_cur = _scores(h)
                active = list(np.argsort(scores_cur)[-max_active_size:])
                active_set = set(active)
    return best_h, best_c5

def _subgrad_momentum(h, deadline, n_steps=2000, beta0=10.0, beta1=200.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    prev_h = h.copy()
    moment = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta0 + (beta1 - beta0) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        grad /= gn
        # Nesterov extrapolation
        moment = 0.9 * moment + grad
        direction = -moment
        # Line search
        alpha = 0.1
        improved = False
        for _ in range(50):
            h_new = _project(h + alpha * direction, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                prev_h = h.copy()
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved:
            moment *= 0.5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=15):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    target = n / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _cyclic_coord_descent(h, deadline, n_cycles=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    G_cache = _grad_matrix(h)
    for cyc in range(n_cycles):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            # Find best pair j and delta to pair with i
            G_i = G_cache[:, i]
            # For each j, the direction is e_i - e_j (or reversed)
            # Change in score r: delta * (G[r,i] - G[r,j])
            # We want to minimize max_r (scores[r] + delta * (G[r,i] - G[r,j]))
            # Try a few candidate j values
            best_j = -1
            best_delta = 0.0
            best_c5_try = best_c5
            # Sample some j values
            j_candidates = list(range(n))
            np.random.shuffle(j_candidates)
            j_candidates = j_candidates[:min(20, n)]
            for j in j_candidates:
                if j == i:
                    continue
                if abs(h[i] - h[j]) < 1e-12:
                    continue
                diff = G_i - G_cache[:, j]
                scores_cur = _scores(h)
                # Line search for delta in [-1, 1] scaled
                # Try positive delta (increase h[i], decrease h[j])
                a_max_pos = min(1.0 - h[i], h[j])
                a_max_neg = min(h[i], 1.0 - h[j])
                for sign_val in [1.0, -1.0]:
                    if sign_val > 0:
                        a_max = a_max_pos
                    else:
                        a_max = a_max_neg
                    if a_max < 1e-10:
                        continue
                    d = sign_val * diff
                    # Find alpha that minimizes max(scores + alpha * d)
                    # subject to 0 <= h[i] + alpha*sign_val <= 1 and 0 <= h[j] - alpha*sign_val <= 1
                    # Binary search on maximum
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(20):
                        mid_a = (lo_a + hi_a) / 2.0
                        s_try = scores_cur + mid_a * d
                        max_try = np.max(s_try)
                        if max_try < best_c5_try - 1e-15:
                            lo_a = mid_a
                        else:
                            hi_a = mid_a
                    a_opt = lo_a
                    if a_opt > 1e-10:
                        h_try = h.copy()
                        h_try[i] += sign_val * a_opt
                        h_try[j] -= sign_val * a_opt
                        h_try = np.clip(h_try, 0.0, 1.0)
                        h_try = _project(h_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5_try:
                            best_c5_try = c5_try
                            best_j = j
                            best_delta = sign_val * a_opt
            if best_j >= 0 and best_c5_try < best_c5 - 1e-14:
                h[i] += best_delta
                h[best_j] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                G_cache = _grad_matrix(h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Remez-LP from parent (35% budget)
    dl1 = t0 + budget_s * 0.35
    h, c5 = _remez_optimize(h, dl1, max_active=15, radius_init=0.25, max_iters=5000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Subgradient with momentum (15% budget)
    if time.time() < t0 + budget_s * 0.45:
        dl2 = t0 + budget_s * 0.50
        h2, c52 = _subgrad_momentum(best_h, dl2, n_steps=1500, beta0=5.0, beta1=150.0)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()

    # Phase 3: Remez-LP again with smaller radius (15% budget)
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.65
        h3, c53 = _remez_optimize(best_h, dl3, max_active=12, radius_init=0.1, max_iters=5000)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

    # Phase 4: Resolution annealing (15% budget)
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.78
        for n_try in [best_n - 4, best_n - 8, best_n + 4, best_n + 8, best_n - 16, best_n + 16]:
            if time.time() > dl4:
                break
            if n_try < 24 or n_try > 128:
                continue
            h_r = _resample(best_h, n_try)
            c5_r_init = _exact_c5(h_r)
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            h_r, c5_r = _remez_optimize(h_r, sub_dl, max_active=10, radius_init=0.15, max_iters=2000)
            if c5_r < best_c5 - 1e-11:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 5: Diverse restarts (15% budget)
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.88
        scale = 0.03
        n_restarts = 200
        for attempt in range(n_restarts):
            if time.time() > dl5:
                break
            if attempt % 50 == 0 and attempt > 0:
                scale = max(0.005, scale * 0.8)
            h_pert = best_h.copy()
            pert_type = attempt % 4
            if pert_type == 0:
                h_pert += rng.normal(0, scale, best_n)
            elif pert_type == 1:
                # Circular shift
                shift = rng.integers(1, best_n // 4)
                h_pert = np.roll(best_h, shift)
            elif pert_type == 2:
                # Block swap
                bsz = rng.integers(2, best_n // 4)
                i1 = rng.integers(0, best_n - 2 * bsz)
                i2 = rng.integers(0, best_n - 2 * bsz)
                if abs(i1 - i2) < bsz:
                    i2 = (i2 + bsz + 1) % (best_n - 2 * bsz)
                tmp = h_pert[i1:i1+bsz].copy()
                h_pert[i1:i1+bsz] = h_pert[i2:i2+bsz]
                h_pert[i2:i2+bsz] = tmp
            else:
                # Binary rounding perturbation
                h_bin = (best_h > 0.5).astype(float)
                n_ones = int(np.sum(h_bin))
                if n_ones > best_n // 2:
                    idx = np.where(h_bin == 1.0)[0]
                    to_zero = rng.choice(idx, n_ones - best_n // 2, replace=False)
                    h_bin[to_zero] = 0.0
                elif n_ones < best_n // 2:
                    idx = np.where(h_bin == 0.0)[0]
                    to_one = rng.choice(idx, best_n // 2 - n_ones, replace=False)
                    h_bin[to_one] = 1.0
                h_pert = 0.7 * best_h + 0.3 * h_bin
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_try, c5_try = _remez_optimize(h_pert, sub_dl, max_active=10, radius_init=0.1, max_iters=500)
            if c5_try < best_c5 - 1e-11:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Cyclic coordinate descent (5% budget)
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.93
        h6, c56 = _cyclic_coord_descent(best_h, dl6, n_cycles=50)
        if c56 < best_c5:
            best_c5 = c56
            best_h = h6.copy()

    # Phase 7: Pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl7, max_passes=8)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final ultra-fine Remez polish
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _remez_optimize(best_h, t0 + budget_s * 0.995, max_active=8, radius_init=0.02, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
