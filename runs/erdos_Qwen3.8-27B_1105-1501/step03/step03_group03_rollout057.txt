def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, -5.0, 5.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _feasible_interval(h, d):
    a_min = 0.0
    a_max = 0.0
    for j in range(len(h)):
        di = d[j]
        if di > 1e-14:
            a_min = max(a_min, -h[j] / di)
            aa = (1.0 - h[j]) / di
            a_max = min(a_max, aa)
        elif di < -1e-14:
            a_min = max(a_min, (1.0 - h[j]) / di)
            aa = -h[j] / di
            a_max = min(a_max, aa)
    return a_min, a_max


def _line_search(h, d, n_coarse=20, golden_iters=25, deadline=None):
    a_min, a_max = _feasible_interval(h, d)
    if a_max <= a_min + 1e-10:
        return h.copy(), _exact_c5(h)
    margin = max((a_max - a_min) * 0.005, 1e-8)
    a_lo = a_min + margin
    a_hi = a_max - margin
    if a_hi <= a_lo:
        return h.copy(), _exact_c5(h)
    best_a = 0.0
    best_c5 = _exact_c5(h)
    alphas = np.linspace(a_lo, a_hi, n_coarse)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return h.copy(), best_c5
    spread = (a_hi - a_lo) / n_coarse * 2.5
    lo = max(a_lo, best_a - spread)
    hi = min(a_hi, best_a + spread)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(golden_iters):
        if deadline is not None and time.time() > deadline:
            break
        if hi - lo < 1e-8:
            break
        if f1 <= f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)

        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)

    return h_new, final_c5


def _score_gradient(h, r, n):
    g = np.zeros(n)
    d_lag = r - (n - 1)
    if d_lag >= 0:
        m = n - d_lag
        if m <= 0:
            return g
        for k in range(n):
            if d_lag <= k < d_lag + m:
                g[k] += (1.0 - h[k - d_lag])
            if 0 <= k < m:
                g[k] -= h[k + d_lag]
    else:
        gg = -d_lag
        m = n - gg
        if m <= 0:
            return g
        for k in range(n):
            if 0 <= k < m:
                g[k] += (1.0 - h[k + gg])
            if gg <= k < gg + m:
                g[k] -= h[k - gg]
    return g


def _active_gradient(h, band_frac=0.06, top_count=5):
    n = len(h)
    s = _scores(h)
    s_max = float(np.max(s))
    if s_max < 1e-12:
        return np.zeros(n)
    band = band_frac * s_max
    active = np.where(s >= s_max - band)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(s))])
    if len(active) > top_count:
        active = active[np.argsort(s[active])[-top_count:]]
    grad = np.zeros(n)
    weights = np.maximum(s[active] - (s_max - band), 0.0)
    wsum = np.sum(weights)
    if wsum < 1e-12:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / wsum
    for idx, r in enumerate(active):
        grad += weights[idx] * _score_gradient(h, int(r), n)
    grad -= np.mean(grad)
    return grad


def _random_zs_dir(n, rng):

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h0 = np.full(64, 0.5)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: Gradient + random line search
    dl1 = t0 + budget_s * 0.25
    h, best_c5 = _gradient_phase(h, best_c5, dl1, rng, max_iters=300)
    best_h = h.copy()
    best_n = len(h)

    # Phase 2: Pair exchange focused on active scores
    dl2 = t0 + budget_s * 0.45
    h, best_c5 = _pair_phase(h, best_c5, dl2, rng, max_rounds=500)
    if best_c5 < _exact_c5(best_h) + 1e-10:
        best_h = h.copy()
        best_n = len(h)

    # Phase 3: Coordinate cycling
    dl3 = t0 + budget_s * 0.58
    h, best_c5 = _cycling_phase(h, best_c5, dl3, rng, max_passes=5)
    if best_c5 < _exact_c5(best_h) + 1e-10:
        best_h = h.copy()
        best_n = len(h)

    # Phase 4: Resolution changes
    dl4 = t0 + budget_s * 0.72
    for dn in [2, -2, 4, -4, 1, -1, 3, -3, 6, -6]:
        if time.time() > dl4:
            break
        nt = best_n + dn
        if nt < 20 or nt > 160:
            continue
        hr = _resample(best_h, nt)
        c5r = _exact_c5(hr)
        if c5r < best_c5 - 1e-10:
            best_c5 = c5r
            best_h = hr.copy()
            best_n = nt
            h = hr.copy()
        elif abs(c5r - best_c5) < 5e-5 and nt < best_n:
            best_c5 = c5r
            best_h = hr.copy()
            best_n = nt
            h = hr.copy()

    # Phase 5: Restart with perturbations
    dl5 = t0 + budget_s * 0.85
    for sigma in [0.02, 0.04, 0.08, 0.12, 0.03, 0.06]:
        if time.time() > dl5:
            break
        hp = _coordinate_perturb(best_h, best_n, rng, sigma)
        c5p = _exact_c5(hp)
        h_try = hp.copy()
        dl_sub = min(dl5, time.time() + budget_s * 0.03)
        h_try, c5_try = _gradient_phase(h_try, min(best_c5, c5p), dl_sub, rng, max_iters=80)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
            h = h_try.copy()

    # Phase 6: Final intensive pair + gradient polish
    dl6 = t0 + budget_s * 0.93
    h, best_c5 = _pair_phase(h, best_c5, dl6, rng, max_rounds=400)
    if best_c5 < _exact_c5(best_h) + 1e-10:
        best_h = h.copy()
        best_n = len(h)

    # Phase 7: Final gradient polish
    dl7 = t0 + budget_s * 0.97
    h, best_c5 = _gradient_phase(h, best_c5, dl7, rng, max_iters=200)
    if best_c5 < _exact_c5(best_h) + 1e-10:
        best_h = h.copy()
        best_n = len(h)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
