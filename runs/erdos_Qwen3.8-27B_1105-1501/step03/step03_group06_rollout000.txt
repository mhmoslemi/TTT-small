    constraints.append({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac})
    
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n
    x0 = np.zeros(n + 1)
    x0[0] = max_c + 0.01
    x0[1:] = h
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        return h_new
    except Exception:
        return h.copy()

def _lp_direction(h, active_lags):
    """Find direction minimizing max of active lag gradients."""
    n = len(h)
    K = len(active_lags)
    
    grads = np.zeros((K, n))
    for idx, k_idx in enumerate(active_lags):
        shift = k_idx - (n - 1)
        if shift >= 0:
            m = n - shift
            for i in range(m):
                grads[idx, i] += (1.0 - h[i+shift])
                grads[idx, i+shift] += h[i]
        else:
            gs = -shift
            m = n - gs
            for i in range(m):
                grads[idx, i] += (1.0 - h[i+gs])
                grads[idx, i+gs] += h[i]
    
    # LP: min t s.t. grads@d <= t, sum(d)=0, -1<=d<=1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, target):
    """Golden section line search for h + a*d."""
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if d[j] > 1e-10:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-10:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a, best_f = 0.0, _exact_c5(_project(h, target))
    lo, hi = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, target))
    f2 = _exact_c5(_project(h + c2 * d, target))
    
    for _ in range(25):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, target))
    
    a = (lo + hi) / 2.0
    f = _exact_c5(_project(h + a * d, target))
    if f < best_f:
        best_a, best_f = a, f
    return best_a, best_f
