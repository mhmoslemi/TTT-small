    pos = x >= 0

    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

    zc = z - np.mean(z)
    n = len(zc)
    lo = -np.max(zc) - 60.0
    hi = -np.min(zc) + 60.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(_sigmoid(zc + mid))
        if s > target:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    return _sigmoid(zc + c)

    return z - np.mean(z)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[r, k:] += (1.0 - h[:m])
                G[r, :m] -= h[k:]
        else:
            g = -k
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

    """Objective and gradient for L-BFGS in z-space."""

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

    return z

def _subgrad_polish(z, target, deadline, max_iters=2000, topm=3):
    n = len(z)
    z = z.copy()
    cur_c5 = _exact_c5(_z_to_h(z, target))
    best_z = z.copy()
    best_c5 = cur_c5
    step = 0.5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h = _z_to_h(z, target)
        scores = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(scores)[-topm:]
        weights = np.zeros(len(scores))
        weights[idx] = 1.0 / topm
        G = _compute_G(h)
        grad_h = G.T @ weights
        A = h * (1.0 - h)
        S = float(np.sum(A))
        if S < 1e-15:
            break
        SA = float(np.sum(A * grad_h)) / S
        grad_z = A * (grad_h - SA)
        gn = float(np.linalg.norm(grad_z))
        if gn < 1e-14:
            break
        direction = -grad_z / gn
        alpha = step
        improved = False
        for _ in range(20):
            h_try = _z_to_h(z + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                z = z + alpha * direction
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_z = z.copy()
                stall = 0
                step = min(step * 1.1, 2.0)
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            step *= 0.7
            if step < 1e-6 or stall > 80:
                break
    return best_z, best_c5

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _run_lbgfs_anneal(z, target, deadline, beta_list, maxiter):
    """Run L-BFGS with increasing betas, returning best h and z."""
    best_h = None
    best_c5 = float('inf')
    current_z = z.copy()
    for beta in beta_list:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        per_beta = max(5, int(remaining / max(1, len(beta_list))))
        nz = current_z
        _opt_ctx['target'] = target
        _opt_ctx['beta'] = float(beta)
        bounds = [(-25.0, 25.0)] * len(z)
        try:
            res = minimize(_obj_fn, nz, jac=_grad_fn, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': per_beta, 'ftol': 1e-12,
                                    'gtol': 1e-8, 'maxls': 30})
            if res.x is not None and np.all(np.isfinite(res.x)):
                current_z = res.x.copy()
        except Exception:
            pass
        h_try = _z_to_h(current_z, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    if best_h is None:
        best_h = _z_to_h(current_z, target)
        best_c5 = _exact_c5(best_h)
    return current_z, best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 = n0 // 2 * 2
        h0 = h0[:n0]
    target = n0 / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    best_z = _h_to_z(h)

    # Phase 1: Main L-BFGS annealing from parent (35% budget)
    d1 = t0 + budget_s * 0.35
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]
    z1, h1, c5_1 = _run_lbgfs_anneal(best_z, target, d1, beta_schedule, 50)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_z = z1.copy()

    # Phase 2: Subgradient polish on best (10% budget)
    d2 = t0 + budget_s * 0.45
    z2, c5_2 = _subgrad_polish(best_z, target, d2, max_iters=3000, topm=3)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_z = z2.copy()
        best_h = _z_to_h(z2, target)

    # Phase 3: Restarts with perturbations (30% budget)
    d3 = t0 + budget_s * 0.75
    scales = [0.3, 0.6, 1.0, 1.5, 2.5, 4.0]
    n_restarts = 60
    for attempt in range(n_restarts):
        if time.time() > d3:
            break
        if best_c5 <= 0.38080:
            break
        remaining = d3 - time.time()
        if remaining < 2.0:
            break
        time_per = remaining / max(1, n_restarts - attempt)
        time_per = min(time_per, 12.0)
        sub_deadline = time.time() + time_per

        # Choose perturbation type
        rtype = attempt % 4
        if rtype == 0:
            # Gaussian noise
            z_pert = best_z + rng.normal(0, scales[attempt % len(scales)], best_n)
        elif rtype == 1:
            # Circular shift
            shift = int(rng.integers(1, best_n))
            z_pert = np.roll(best_z, shift)
        elif rtype == 2:
            # Sign flip on random subset
            mask = rng.random(best_n) < 0.3
            z_pert = best_z.copy()
            z_pert[mask] *= -1.0
        else:
            # Interpolation with noise
            z_pert = best_z * rng.uniform(0.5, 1.0) + rng.normal(0, 0.5, best_n)

        # Run short L-BFGS from perturbation
        short_betas = [20, 80, 320, 1280]
        z_try, h_try, c5_try = _run_lbgfs_anneal(z_pert, target, sub_deadline, short_betas, 30)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_z = z_try.copy()

    # Phase 4: Try different n values (15% budget)
    d4 = t0 + budget_s * 0.90
    n_candidates = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, -12, 12, -16, 16]:
        nc = best_n + dn
        if 32 <= nc <= 128 and nc % 2 == 0 and nc != best_n:
            n_candidates.append(nc)
    for nc in n_candidates[:6]:
        if time.time() > d4:
            break
        h_r = _resample_h(best_h, nc)
        z_r = _h_to_z(h_r)
        tgt_r = nc / 2.0
        short_betas = [10, 40, 160, 640]
        z_r2, h_r2, c5_r2 = _run_lbgfs_anneal(z_r, tgt_r, time.time() + 8.0, short_betas, 30)
        if c5_r2 < best_c5 - 1e-12:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = nc
            best_z = z_r2.copy()

    # Phase 5: Final L-BFGS + subgradient polish (10% budget)
    d5 = t0 + budget_s * 0.97
    final_betas = [64, 256, 1024, 4096]
    z_f, h_f, c5_f = _run_lbgfs_anneal(best_z, target, min(d5, time.time() + (d5 - time.time()) * 0.6), final_betas, 80)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_z = z_f.copy()

    # Subgradient final
    z_sp, c5_sp = _subgrad_polish(best_z, target, d5, max_iters=2000, topm=2)
    if c5_sp < best_c5:
        best_c5 = c5_sp
        best_h = _z_to_h(z_sp, target)

    # Phase 6: Final exact coordinate polish
    if time.time() < t0 + budget_s * 0.99:
        h_cd = best_h.copy()
        tgt = best_n / 2.0
        cd_c5 = _exact_c5(h_cd)
        cd_deadline = t0 + budget_s * 0.995
        for _pass in range(50):
            if time.time() > cd_deadline:
                break
            h_cur = _z_to_h(best_z, tgt)
            scores = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            G = _compute_G(h_cur)
            grad_h = G[np.argmax(scores)]
            # Find best pair to swap
            A = h_cur * (1.0 - h_cur)
            S = float(np.sum(A))
            if S < 1e-15:
                break
            SA = float(np.sum(A * grad_h)) / S
            grad_z = A * (grad_h - SA)
            # Move mass from high-z to low-z where gradient suggests
            i_src = int(np.argmin(grad_z))
            i_dst = int(np.argmax(grad_z))
            if i_src == i_dst:
                break
            for delta in [0.005, 0.01, 0.02, 0.04, 0.06]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h_cd[i_dst] + d < 0 or h_cd[i_dst] + d > 1:
                        continue
                    if h_cd[i_src] - d < 0 or h_cd[i_src] - d > 1:
                        continue
                    h_try = h_cd.copy()
                    h_try[i_dst] += d
                    h_try[i_src] -= d
                    h_try = _project(h_try, tgt)
                    c5_try = _exact_c5(h_try)
                    if c5_try < cd_c5 - 1e-13:
                        cd_c5 = c5_try
                        h_cd = h_try
                        break
            if h_cd is not best_h and not np.allclose(h_cd, best_h, atol=1e-10):
                best_c5 = cd_c5
                best_h = h_cd
                best_z = _h_to_z(best_h)

    # Final validation and projection
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection changed things slightly, recompute
        best_c5 = final_c5

    return best_h, best_c5, best_n
