import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_of_score(h, k):
    """Gradient of output[k] w.r.t. h.
    grad[m] = (1 - 2*h[k-m]) if 0 <= k-m < n, else 0
    """
    n = len(h)
    m_start = max(0, k - (n - 1))
    m_end = min(n - 1, k)
    if m_start > m_end:
        return np.zeros(n)
    indices = np.arange(m_start, m_end + 1)
    val_indices = k - indices  # = k-m for each m in indices
    grad = np.zeros(n)
    grad[indices] = 1.0 - 2.0 * h[val_indices]
    return grad

def _grad_matrix(h):
    """Matrix where row k is grad_of_score(h, k). Shape: (2n-1, n)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    one_minus_2h = 1.0 - 2.0 * h
    for k in range(2 * n - 1):
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_start <= m_end:
            val_indices = k - np.arange(m_start, m_end + 1)
            G[k, m_start:m_end+1] = one_minus_2h[val_indices]
    return G

def _project(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = 0.0, 1.0
    # Binary search for tau such that sum(clip(v-tau, 0, 1)) = target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgrad_descent(h, deadline, n_iters=20000, step_init=0.1):
    """Subgradient method with momentum on the exact max objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    beta_mom = 0.9
    step = step_init
    stall = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        mx = float(np.max(scores))
        # Active set: scores within 10% of max
        active = scores >= mx - 0.1 * mx - 0.01
        # Compute subgradient as average of active gradients
        G = _grad_matrix(h)
        idx = np.where(active)[0]
        grad = G[idx].mean(axis=0)
        grad -= np.mean(grad)  # Project out the sum constraint direction
        
        # Momentum
        momentum = beta_mom * momentum - step * grad
        h_try = h + momentum
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            step *= 0.95
            stall += 1
            if stall > 100:
                step = step_init * 0.1
                stall = 0
                momentum *= 0.5
    
    return best_h, best_c5

def _two_swap_search(h, deadline, max_pairs=None):
    """Try 2-swap moves: h[j] += delta, h[k] -= delta for various j,k,delta."""
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    
    # Random sampling of pairs if too many
    if max_pairs > 5000:
        rng = np.random.default_rng(42)
        pairs = [(int(rng.integers(n)), int(rng.integers(n))) for _ in range(5000)]
        pairs = [(a, b) for a, b in pairs if a != b]
    else:
        pairs = [(j, k) for j in range(n) for k in range(j+1, n)]
    
    deltas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]
    
    count = 0
    for j, k in pairs:
        if time.time() > deadline:
            break
        for d in deltas:
            for sign in [1, -1]:
                delta = sign * d
                if h[j] + delta < -1e-9 or h[j] + delta > 1 + 1e-9:
                    continue
                if h[k] - delta < -1e-9 or h[k] - delta > 1 + 1e-9:
                    continue
                h_try = h.copy()
                h_try[j] += delta
                h_try[k] -= delta
                h_try = np.clip(h_try, 0, 1)
                # Re-project to maintain exact sum
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    count += 1
    return best_h, best_c5

def _annealing_search(h, deadline, T_init=0.005, T_end=1e-6, n_iter=50000):
    """Simulated annealing with 2-swap moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_end / T_init) ** (it / n_iter)
        
        # Random 2-swap
        j = int(rng.integers(n))
        k = int(rng.integers(n))
        if j == k:
            continue
        d = rng.uniform(-0.3, 0.3)
        if h[j] + d < -1e-9 or h[j] + d > 1 + 1e-9:
            continue
        if h[k] - d < -1e-9 or h[k] - d > 1 + 1e-9:
            continue
        
        h_try = h.copy()
        h_try[j] += d
        h_try[k] -= d
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        delta = c5_try - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _active_set_lp(h, scores, active_idx, radius):
    """Find best direction using LP on active set."""
    n = len(h)
    G = _grad_matrix(h)
    A = G[active_idx]  # (|A|, n)
    
    # LP: min z s.t. A @ t <= z - scores[active], |t| <= radius, sum(t) = 0
    # Variables: t (n), z (1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = radius * np.ones(len(active_idx))  # A@t - z <= radius (simplified)
    
    # Actually: A@t <= z - scores[active] → A@t - z <= -scores[active]
    # But we want to find direction that decreases the max, so:
    # We want A@t to be as negative as possible (decrease all active scores)
    # min z s.t. A[i]@t <= z for all i (shifted), bounds on t
    
    # Simpler: find t that minimizes max_i (scores[active_i] + G[active_i]@t)
    # = min z s.t. G[active_i]@t <= z - scores[active_i]
    
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -scores[active_idx]
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0  # z doesn't appear in equality
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # z is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline, radius_init=0.05, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

            break

        active_idx = np.where(active)[0]
        
        p = _active_set_lp(h, scores, active_idx, radius)
        if p is None or np.max(np.abs(p)) < 1e-10:
            stall += 1
            radius *= 0.9
            if radius < 1e-5:

        alpha_max = 1.0

            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-8)
        
        best_a = 0.0
        best_c5_try = cur_c5

            stall = 0

output[k] = ∑_{i: 0≤i<n, 0≤k-i<n} h[i] * (1-h[k-i])

Let j = k-i, so i = k-j:

∂/∂h[m]:
- h[m] appears as h[k-j] when k-j = m, i.e., j = k-m. Valid if 0≤k-m<n and 0≤m<n.
  Contribution: (1-h[j]) = (1-h[k-m])

  Contribution: -h[k-m]

Both conditions are the same: 0≤k-m<n.

If k-m ≠ m, the two contributions are from different terms, and the total is (1-h[k-m]) - h[k-m] = 1-2h[k-m].

def _solve_lp_direction(h, scores, active_idx, radius):

    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -scores[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
