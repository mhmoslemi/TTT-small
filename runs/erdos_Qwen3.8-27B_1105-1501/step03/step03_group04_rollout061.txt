    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # For lag k = 0 (index n-1):
    G[n-1, :] = 1.0 - 2.0 * h
    
    # For positive lags k = 1, 2, ..., n-1 (indices n, n+1, ..., 2n-2):
    for k in range(1, n):

        # dc[k]/dh[j] = (1-h[j-k]) if j >= k, else 0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        idx = n - 1 + k
        m = n - k
        G[idx, k:] = 1.0 - h[:m]
        G[idx, :m] += -h[k:n]
    for d in range(1, n):
        idx = n - 1 - d
        m = n - d
        G[idx, :m] = 1.0 - h[d:n]
        G[idx, d:] += -h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:

            mask = x < 1.0
        else:

            hi_a = min(alpha_max, best_a * 1.3)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * p, target))
            f2 = _exact_c5(_project(h + c2 * p, target))
            for _ in range(15):
                if hi_a - lo_a < 1e-12:
                    break
                if f1 < f2:
                    hi_a = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = _exact_c5(_project(h + c1 * p, target))
                else:
                    lo_a = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f2 = _exact_c5(_project(h + c2 * p, target))
            opt_a = (lo_a + hi_a) / 2.0
            opt_c5 = _exact_c5(_project(h + opt_a * p, target))
            if opt_c5 < best_val:
                best_val = opt_c5
                best_a = opt_a
            
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _coord2d_descent(h, deadline, max_passes=30):
    """2D coordinate descent: vary pairs of coordinates."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Random subset of pairs for diversity
        pairs = [(i, j) for i in range(n) for j in range(i+1, n) if 0 < h[i] < 1 and 0 < h[j] < 1]
        if not pairs:
            break
        # Shuffle and sample
        np.random.shuffle(pairs)
        n_check = min(len(pairs), 500)
        for (i, j) in pairs[:n_check]:
            if time.time() > deadline:
                break
            # Try moving mass from i to j and vice versa
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta < 1e-6:
                continue
            deltas = np.linspace(-max_delta*0.8, max_delta*0.8, 9)
            best_delta = 0.0
            best_val = best_c5
            for d in deltas:
                if abs(d) < 1e-10:
                    continue
                h_try = h.copy()
                h_try[i] += d
                h_try[j] -= d
                if h_try[i] < -1e-9 or h_try[i] > 1+1e-9 or h_try[j] < -1e-9 or h_try[j] > 1+1e-9:
                    continue
                h_try = np.clip(h_try, 0, 1)
                c5_try = _exact_c5(h_try)
                if c5_try < best_val:
                    best_val = c5_try
                    best_delta = d
            if best_delta != 0 and best_val < best_c5 - 1e-13:
                h[i] += best_delta
                h[j] -= best_delta
                h = np.clip(h, 0, 1)
                best_c5 = best_val
                best_h = h.copy()
                improved = True
        if not improved:
            break
    
    return best_h, best_c5

def _coord1d_descent(h, deadline, max_passes=20):
    """1D coordinate descent with compensation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            # Try changing h[j] and compensating on the best other coordinate
            for delta in [0.005, 0.01, 0.02, 0.05, 0.1]:
                for sign in [1.0, -1.0]:
                    d = sign * delta
                    if h[j] + d < 0 or h[j] + d > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] += d
                    # Compensate: distribute -d among coordinates with room
                    if d > 0:
                        mask = (h_try < 1.0)
                        mask[j] = False
                    else:
                        mask = (h_try > 0.0)
                        mask[j] = False
                    if not np.any(mask):
                        continue
                    room = np.where(mask, 1.0 - h_try, 0.0) if d > 0 else np.where(mask, h_try, 0.0)
                    total_room = room.sum()
                    if total_room < abs(d):
                        continue
                    h_try[mask] -= d * room[mask] / total_room
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    
    return best_h, best_c5

def _equalize_active(h, deadline):
    """Try to equalize the top few lags using Newton-like iterations."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for outer in range(50):
        if time.time() > deadline:
            break
        scores = _scores(h)
        m = float(np.max(scores))
        # Find lags within 5% of max
        threshold = m - 0.05 * (m - float(np.min(scores)))
        active_idx = np.where(scores >= threshold)[0]
        n_active = len(active_idx)
        
        if n_active < 2:
            break
        
        G = _grad_matrix(h)
        # Build equalization system: f_k(h) = f_0(h) for k in active, k != 0
        # Jacobian: G[active[k]] - G[active[0]]
        a0 = active_idx[0]
        J = G[active_idx[1:]] - G[a0]
        # Project to zero-sum subspace
        J = J - np.outer(J.mean(axis=1), np.ones(n) / n)
        F = scores[active_idx[1:]] - scores[a0]
        
        if float(np.max(np.abs(F))) < 1e-12:
            # Already equalized, try to reduce the common value
            avg_grad = G[active_idx].mean(axis=0)
            avg_grad -= np.mean(avg_grad)
            gnorm = float(np.linalg.norm(avg_grad))
            if gnorm < 1e-14:
                break
            direction = -avg_grad / gnorm
            for alpha in [0.1, 0.05, 0.02]:
                h_try = _project(h + alpha * direction, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    break
            continue
        
        try:
            delta, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)
        except:
            break
        
        delta -= np.mean(delta)  # Ensure zero-sum
        
        improved = False
        for alpha in [0.5, 1.0, 2.0, 0.25, 0.1]:
            h_try = _project(h + alpha * delta, target)
            s_try = _scores(h_try)
            spread = float(np.max(s_try[active_idx]) - np.min(s_try[active_idx]))
            c5_try = _exact_c5(h_try)
            if spread < float(np.max(F)) * 0.5 and c5_try <= best_c5 + 0.001:
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_c5

def _random_perturb(h, rng, scale):
    n = len(h)
    h_new = h + rng.normal(0, scale, n)
    return _project(h_new, n / 2.0)

def _mix_perturb(h, h_ref, rng, mix_frac):
    n = len(h)
    h_new = (1 - mix_frac) * h + mix_frac * h_ref + rng.normal(0, 0.01, n)
    return _project(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0.copy(), target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Soft descent (broad exploration)
    d1 = t0 + budget_s * 0.18
    h, c5 = _soft_descent(h, d1, beta_start=10.0, beta_end=3000.0, max_iters=8000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 2: Active set descent (precise)
    d2 = t0 + budget_s * 0.32
    h, c5 = _active_descent(h, d2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 3: LP polish
    d3 = t0 + budget_s * 0.45
    h, c5 = _lp_descent(h, d3, radius=0.12, max_iters=2000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 4: Equalize active set
    d4 = t0 + budget_s * 0.55
    h, c5 = _equalize_active(h, d4)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 5: Basin hopping with restarts
    d5 = t0 + budget_s * 0.75
    scale = 0.04
    for attempt in range(200):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        # Various perturbation strategies
        strat = attempt % 4
        if strat == 0:
            h_pert = _random_perturb(best_h, rng, scale)
        elif strat == 1:
            # Flip a random subset
            h_pert = best_h.copy()
            k = rng.integers(4, 12)
            idx = rng.choice(best_n, size=k, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = _project(h_pert, best_n / 2.0)
        elif strat == 2:
            # Gaussian with larger scale on random half
            h_pert = best_h.copy()
            mask = rng.random(best_n) < 0.5
            h_pert[mask] += rng.normal(0, scale * 2, mask.sum())
            h_pert = _project(h_pert, best_n / 2.0)
        else:
            # Small structured perturbation (sine)
            h_pert = best_h + 0.02 * np.sin(2 * np.pi * rng.integers(1, 5) * np.arange(best_n) / best_n)
            h_pert = _project(h_pert, best_n / 2.0)
        
        # Quick optimization from perturbed start
        t_dead = min(d5, time.time() + budget_s * 0.012)
        h_try, c5_try = _active_descent(h_pert, t_dead)
        
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.1)
    
    # Phase 6: Try different resolutions
    d6 = t0 + budget_s * 0.85
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        t_dead = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _active_descent(h_r, t_dead)
        t_dead2 = min(d6, time.time() + budget_s * 0.008)
        h_r, c5_r = _lp_descent(h_r, t_dead2, radius=0.08, max_iters=800)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 7: Final intensive polish
    d7 = t0 + budget_s * 0.92
    # Active descent
    h, c5 = _active_descent(best_h, min(d7, time.time() + budget_s * 0.03))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    # Equalize
    h, c5 = _equalize_active(best_h, min(d7, time.time() + budget_s * 0.02))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    # LP
    h, c5 = _lp_descent(best_h, min(d7, time.time() + budget_s * 0.03), radius=0.06, max_iters=2000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 8: 2D coordinate descent
    d8 = t0 + budget_s * 0.975
    h, c5 = _coord2d_descent(best_h, d8, max_passes=15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 9: 1D coordinate + final LP
    d9 = t0 + budget_s * 0.995
    h, c5 = _coord1d_descent(best_h, min(d9, time.time() + budget_s * 0.01), max_passes=10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Final projection and verification
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
