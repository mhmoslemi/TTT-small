def _pair_score_change(h, i, j, alpha, d):
    """Compute the change in score at lag d after moving alpha from i to j.
    d is the lag index in np.correlate(h, 1-h, 'full') which is offset by n-1.
    The actual lag is d - (n-1).
    """
    n = len(h)
    actual_d = d - (n - 1)  # actual lag
    # Valid x range: x in [max(0, -actual_d), min(n-1, n-1-actual_d)]
    x_lo = max(0, -actual_d)
    x_hi = min(n - 1, n - 1 - actual_d)
    
    affected = set()
    if x_lo <= i <= x_hi:
        affected.add(i)
    if x_lo <= j <= x_hi:
        affected.add(j)
    if x_lo <= i - actual_d <= x_hi and i - actual_d != i:
        affected.add(i - actual_d)
    if x_lo <= j - actual_d <= x_hi and j - actual_d != j:
        affected.add(j - actual_d)
    
    change = 0.0
    for x in affected:
        old = h[x] * (1.0 - h[x + actual_d])
        # new values
        hx = h[x] - alpha if x == i else (h[x] + alpha if x == j else h[x])
        hxd = h[x + actual_d] - alpha if (x + actual_d) == i else (h[x + actual_d] + alpha if (x + actual_d) == j else h[x + actual_d])
        new = hx * (1.0 - hxd)
        change += new - old
    return change
