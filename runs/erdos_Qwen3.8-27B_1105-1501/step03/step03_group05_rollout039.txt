def exact_pair_line_search(h, i, j, direction=1):
    """
    Find best alpha for h[i] += direction*alpha, h[j] -= direction*alpha
    """
    n = len(h)
    dx = 2.0/n
    current_scores = np.correlate(h, 1.0-h, mode='full')
    current_c5 = current_scores.max() * dx
    
    # Feasible range for alpha
    if direction == 1:
        alpha_min = max(-h[i], h[j]-1)  # h[i]+a >= 0, h[j]-a >= 0
        alpha_max = min(1-h[i], h[j])    # h[i]+a <= 1, h[j]-a <= 1
    else:
        alpha_min = max(-(1-h[j]), h[i])  # h[j]+a >= 0, h[i]-a >= 0
        alpha_max = min(1-h[j], h[i])    # h[j]+a <= 1, h[i]-a <= 1
    
    if alpha_max <= alpha_min:
        return 0, current_c5
    
    # Coarse scan
    n_scan = 40
    best_a = 0
    best_val = current_c5
    for a in np.linspace(alpha_min, alpha_max, n_scan):
        h_new = h.copy()
        h_new[i] += direction * a
        h_new[j] -= direction * a
        c5_new = np.correlate(h_new, 1.0-h_new, mode='full').max() * dx
        if c5_new < best_val:
            best_val = c5_new
            best_a = a
    
    # Golden section refine around best
    ...
    
    return best_a, best_val

    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _binary_descend(h, deadline, max_passes=1000):
    """Binary swap descent: iteratively find and apply the best swap."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

                new_corr = corr.copy()

                jpd = j + d_range
                imd = i - d_range
                jmd = j - d_range
                
                v1 = (ipd >= 0) & (ipd < n)
                v2 = (jpd >= 0) & (jpd < n)
                v3 = (imd >= 0) & (imd < n)
                v4 = (jmd >= 0) & (jmd < n)

                new_corr[v2] += (1.0 - h[jpd[v2]])
                new_corr[v3] += h[imd[v3]]
                new_corr[v4] += h[jmd[v4]]
                
                new_corr[n-1] -= 2.0
                dj = j - i
                if 1 <= abs(dj) <= n-1:
                    new_corr[n-1 + dj] += 1.0
                    new_corr[n-1 - dj] += 1.0
                
                c5_new = float(np.max(new_corr)) * dx
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_swap = (i, j, new_corr)
