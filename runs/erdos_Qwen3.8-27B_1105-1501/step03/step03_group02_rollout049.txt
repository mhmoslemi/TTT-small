def _scores(h):

def _c5(h):

def _repair(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-10:
        return v
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-9:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _quadratic_coeffs(h, d):

def _smooth_max_grad(h, scores, beta=50.0):
    """Compute smooth-max gradient direction (descent, zero-sum)."""
    n = len(h)
    M = 2 * n - 1
    max_s = float(np.max(scores))
    
    # Weights
    diffs = scores - max_s
    w = np.exp(beta * diffs)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(M)
        w[idx] = 1.0
    else:
        w /= wsum
    
    # Compute gradient: sum of w_r * grad_r
    G = np.zeros(n)
    for r in range(M):
        if w[r] < 1e-10:
            continue
        L = r - (n - 1)
        if L == 0:
            G += w[r] * (1.0 - 2.0 * h)
        elif L > 0:
            m = n - L
            G[:m] += -w[r] * h[L:L+m]
            G[L:] += w[r] * (1.0 - h[:m])

            g = -L
            m = n - g
            G[:m] += w[r] * (1.0 - h[g:g+m])
            G[g:] += -w[r] * h[:m]
    
    # Project to zero-sum
    d = -(G - np.mean(G))
    norm = float(np.max(np.abs(d)))
    if norm < 1e-15:
        return None
    return d / norm

def _smooth_max_grad(h, scores, beta=50.0):
    n = len(h)
    M = 2 * n - 1
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(M)
        w[idx] = 1.0

        w /= wsum

        if w[r] < 1e-12:
            continue
        L = r - (n - 1)

        elif L > 0:

            G[:m] += -w[r] * h[L:L+m]
            G[L:] += w[r] * (1.0 - h[:m])

            g = -L
            m = n - g
            G[:m] += w[r] * (1.0 - h[g:g+m])
            G[g:] += -w[r] * h[:m]

            d2 = _lag_direction(h, cur_scores, top_k=3)
            if d2 is not None:
                t_opt, _ = _exact_1d_min(h, d2)
                if t_opt is not None and abs(t_opt) > 1e-10:
                    h_new = _repair(h + t_opt * d2, target)
                    c5_new = _c5(h_new)
                    if c5_new < cur_c5 - 1e-12:
                        h = h_new
                        cur_c5 = c5_new
                        cur_scores = _scores(h)
                        stall = 0
                        improved = True
        
        if not improved:
            # Try random directions
            if rng is None:
                rng = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
            for _ in range(3):
                if time.time() > deadline:
                    break
                d_rand = rng.normal(0, 1, n)
                d_rand -= np.mean(d_rand)
                norm = float(np.max(np.abs(d_rand)))
                if norm < 1e-15:
                    continue
                d_rand /= norm
                t_opt, _ = _exact_1d_min(h, d_rand)
                if t_opt is not None and abs(t_opt) > 1e-10:
                    h_new = _repair(h + t_opt * d_rand, target)
                    c5_new = _c5(h_new)
                    if c5_new < cur_c5 - 1e-12:
                        h = h_new
                        cur_c5 = c5_new
                        cur_scores = _scores(h)
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            if stall % 5 == 0:
                beta = min(300.0, beta * 1.15)
    
    return h, cur_c5

def _lp_direction(h, scores, top_k=20, max_norm=1.0):
    """LP-based direction: minimize max first-order change over active lags."""
    n = len(h)
    M = 2 * n - 1
    top_idx = np.argsort(scores)[::-1][:top_k]
    
    # Build gradient matrix for top lags
    Gmat = np.zeros((top_k, n))
    for i, r in enumerate(top_idx):
        L = r - (n - 1)
        if L == 0:
            Gmat[i, :] = (1.0 - 2.0 * h)
        elif L > 0:
            m = n - L
            Gmat[i, :m] += -h[L:L+m]
            Gmat[i, L:] += (1.0 - h[:m])
        else:
            g = -L
            m = n - g
            Gmat[i, :m] += (1.0 - h[g:g+m])
            Gmat[i, g:] += -h[:m]
    
    # LP: min nu s.t. Gmat @ d <= nu, sum(d)=0, |d_i| <= 1
    # Variables: d[0..n-1], nu
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * top_k + 2, n + 1))
    b_ub = np.zeros(2 * top_k + 2)
    # Gmat @ d - nu <= 0
    A_ub[:top_k, :n] = Gmat
    A_ub[:top_k, n] = -1.0
    # -Gmat @ d - nu <= 0 ( minimize max absolute)
    A_ub[top_k:2*top_k, :n] = -Gmat
    A_ub[top_k:2*top_k, n] = -1.0
    # sum(d) = 0 handled by A_eq
    # |d_i| <= 1 handled by bounds
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs',
                      options={'time_limit': min(5.0, max(1.0, 2.0))})
        if res.success:
            d = res.x[:n]
            d -= np.mean(d)
            norm = float(np.max(np.abs(d)))
            if norm > 1e-10:
                return d / norm
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target = n / 2.0
    h = _repair(h_parent, target)
    best_h = h.copy()
    best_c5 = _c5(h)
    
    def _check_deadline(frac):
        return time.time() < t0 + budget_s * frac
    
    # Phase 1: Local descent from parent with varied betas
    deadline1 = t0 + budget_s * 0.25
    for beta_try in [40.0, 60.0, 80.0, 120.0]:
        if time.time() > deadline1:
            break
        sub_dl = min(deadline1, time.time() + (deadline1 - t0) / 4.0)
        h_try, c5_try = _local_descent(best_h.copy(), sub_dl, beta=beta_try, rng=rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 2: Population search
    deadline2 = t0 + budget_s * 0.70
    seeds = [best_h.copy()]
    seeds.append(_repair(1.0 - best_h, target))
    seeds.append(_repair(best_h[::-1].copy(), target))
    seeds.append(_repair(1.0 - best_h[::-1].copy(), target))
    
    # Add perturbed seeds
    for scale in [0.03, 0.06, 0.10, 0.15]:
        h_pert = best_h + rng.normal(0, scale, n)
        seeds.append(_repair(h_pert, target))
    
    # Random binary-like seeds
    for _ in range(2):
        h_bin = (rng.random(n) < 0.5).astype(float)
        # Adjust to exact target
        h_bin = _repair(h_bin, target)
        seeds.append(h_bin)
    
    seed_c5s = [_c5(s) for s in seeds]
    order = np.argsort(seed_c5s)
    seeds_sorted = [(seeds[i], seed_c5s[i]) for i in order]
    
    n_seeds = len(seeds_sorted)
    per_seed = max(3.0, min(20.0, (deadline2 - time.time()) / n_seeds))
    
    for si in range(n_seeds):
        if time.time() > deadline2:
            break
        h_s, c5_s = seeds_sorted[si]
        if c5_s > best_c5 + 0.005:
            continue
        sub_dl = min(deadline2, time.time() + per_seed)
        h_try, c5_try = _local_descent(h_s, sub_dl, beta=50.0, rng=rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 3: LP-based refinement
    if time.time() < t0 + budget_s * 0.82:
        deadline3 = t0 + budget_s * 0.85
        for _ in range(5):
            if time.time() > deadline3:
                break
            scores = _scores(best_h)
            d_lp = _lp_direction(best_h, scores, top_k=15)
            if d_lp is None:
                break
            t_opt, _ = _exact_1d_min(best_h, d_lp)
            if t_opt is not None and abs(t_opt) > 1e-10:
                h_new = _repair(best_h + t_opt * d_lp, target)
                c5_new = _c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
                else:
                    break
            else:
                break
    
    # Phase 4: Final intensive local descent
    if time.time() < t0 + budget_s * 0.95:
        deadline4 = t0 + budget_s * 0.97
        # Multiple beta sweeps
        for beta_try in [30.0, 50.0, 70.0, 100.0, 150.0, 200.0]:
            if time.time() > deadline4:
                break
            per_dl = min(deadline4, time.time() + 15.0)
            h_try, c5_try = _local_descent(best_h.copy(), per_dl, beta=beta_try, rng=rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 5: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.98:
        deadline5 = t0 + budget_s * 0.99
        for dn in [4, -4, 8, -8]:
            if time.time() > deadline5:
                break
            n_try = n + dn
            if n_try < 32 or n_try > 128 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            c5_r = _c5(h_r)
            if c5_r < best_c5 + 0.001:
                # Local descent at new resolution
                sub_dl = min(deadline5, time.time() + 10.0)
                h_r2, c5_r2 = _local_descent(h_r, sub_dl, beta=50.0, rng=rng)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2
                    n = n_try
    
    # Final validation
    best_h = _repair(best_h, int(len(best_h)) / 2.0)
    final_c5 = _c5(best_h)
    final_n = len(best_h)
    
    return best_h, final_c5, final_n
