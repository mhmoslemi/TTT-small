def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_obj(h, beta):
    scores = _scores(h)
    m = float(np.max(scores))
    if m < -1e10:
        return 0.0
    shifted = scores - m
    log_sum = np.log(np.sum(np.exp(beta * shifted))) + m
    return log_sum / beta

def _soft_grad(h, beta):

    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    weights = exps / float(np.sum(exps))
    G = _grad_matrix(h)
    return G.T @ weights

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_grad(h, beta):
    scores = _scores(h)
    m = float(np.max(scores))
    exps = np.exp(beta * (scores - m))
    weights = exps / float(np.sum(exps))
    G = _grad_matrix(h)
    return G.T @ weights

def _slsqp_optimize(h, deadline, beta=200.0):

    target = n / 2.0

    def obj(x):
        return _soft_obj_fn(x, beta)

    def obj_grad(x):
        return _soft_grad(x, beta)

    def eq_con(x):
        return np.sum(x) - target

    def eq_con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}]
    bounds = [(0, 1)] * n
    x0 = h.copy()
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-16})
        if res.success:
            h_opt = _project(res.x, target)
            c5_opt = _exact_c5(h_opt)
            return h_opt, c5_opt
    except Exception:

    return h, _exact_c5(h)

def _soft_obj_fn(x, beta):
    scores = _scores(x)
    m = float(np.max(scores))
    shifted = scores - m
    return np.log(np.sum(np.exp(beta * shifted))) / beta + m

def _pair_swap(h, rng, n_swaps=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _ in range(n_swaps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        swap_amt = rng.uniform(0.1, 0.5) * (h[i] - h[j])
        h[i] -= swap_amt
        h[j] += swap_amt
    return _project(h, target)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_init(n, rng, method='random'):
    if method == 'random':
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:n//2]] = 1.0
        h = h + rng.normal(0, 0.1, n)
    elif method == 'sine':
        x = np.arange(n) / n
        freq = rng.integers(1, 5)
        amp = rng.uniform(0.2, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    elif method == 'triangle':
        x = np.arange(n) / n
        freq = rng.integers(1, 4)
        amp = rng.uniform(0.2, 0.4)
        phase = rng.uniform(0, 1.0)
        t = (freq * x + phase) % 1.0
        h = 0.5 + amp * (2.0 * np.abs(2.0 * t - 1.0) - 1.0)
    elif method == 'block':
        nblocks = rng.integers(3, 10)
        block_len = n // nblocks
        h = np.zeros(n)
        for b in range(nblocks):
            start = b * block_len
            end = min(start + block_len + 1, n)
            h[start:end] = rng.uniform(0, 1)
    else:
        h = np.full(n, 0.5)
    return _project(h, n / 2.0)
