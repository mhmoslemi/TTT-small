def _project(v, target):

    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)

        else:

def _raw_scores(h):

def _score_gradient(h, d):
    n = len(h)

        m = n - d
        if m <= 0:
            return g
        idx1 = np.arange(m)
        idx2 = np.arange(d, d + m)
        g[idx1] += -(h[idx2])
        g[idx2] += (1.0 - h[idx1])

        idx2 = np.arange(gg, gg + m)

    d -= np.mean(d)
    norm = float(np.sqrt(np.sum(d * d)))

            a_max = min(a_max, (1.0 - h[i]) / di)
        elif di < -eps:
            a_min = max(a_min, (1.0 - h[i]) / di)
            a_max = min(a_max, -h[i] / di)
    if a_max <= a_min:

    return a_min, a_max
