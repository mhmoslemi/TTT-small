        length = n - k

            grad[:length] -= dx * h[k:]

    # Minimize max_r (scores[r] + grad_diff[r] * t) over t in [t_min, t_max]
    
    # This is a 1D convex piecewise-linear minimization
    # The optimum is at a breakpoint or at a boundary
    
    # Method: bisection on the subgradient
    # The subgradient of f(t) = max_r (scores[r] + grad_diff[r]*t) at t is:
    # { weighted average of grad_diff[r] for active r } (convex hull of active gradients)
    # At the minimum, 0 is in the subgradient
    
    # Simpler: ternary search (since f is convex)
    def f(t):
        vals = scores + grad_diff * t
        return float(np.max(vals))
    
    # Ternary search
    lo, hi = t_min, t_max
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    
    h_new = h.copy()
    h_new[i] = a + t_opt
    h_new[j] = b - t_opt
    h_new = np.clip(h_new, 0.0, 1.0)
    # Re-project to ensure exact sum
    h_new = _project(h_new, target)
    
    return h_new, _exact_c5(h_new)
