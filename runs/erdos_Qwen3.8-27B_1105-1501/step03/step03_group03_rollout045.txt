    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_golden):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h, deadline, n_active, radius_init, max_iters, radius_min):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        s_max = float(np.max(scores))
        sorted_idx = np.argsort(-scores)
        active = sorted_idx[:n_active]
        s_min_active = scores[active].min()
        for extra_i in range(n_active, min(2 * n_active, len(scores))):
            if scores[sorted_idx[extra_i]] >= s_min_active:
                active = np.append(active, sorted_idx[extra_i])
                break
        p = _lp_step(h, scores, G, active, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < radius_min:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.65
            if radius < radius_min:
                break
    return best_h, best_c5


def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-10 or h[k] >= 1.0 - 1e-10:
                    continue
                p = np.zeros(n)
                p[j] = 1.0
                p[k] = -1.0
                a, c5_new = _line_search_exact(h, p, _exact_c5(h), n_coarse=12, n_golden=20)
                if c5_new < best_c5 - 1e-13:
                    h = h + a * p
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                p2 = np.zeros(n)
                p2[j] = -1.0
                p2[k] = 1.0
                a2, c5_new2 = _line_search_exact(h, p2, _exact_c5(h), n_coarse=12, n_golden=20)
                if c5_new2 < best_c5 - 1e-13:
                    h = h + a2 * p2
                    best_c5 = c5_new2
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _random_direction(n, rng, scale=0.1):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = np.sqrt(np.sum(d ** 2))
    if norm < 1e-12:
        return np.zeros(n)
    return d / norm * scale


def _steepest_descent_dir(h, scores, G, n_top=8):
    s_max = float(np.max(scores))
    sorted_idx = np.argsort(-scores)
    active = sorted_idx[:n_top]
    band = 0.1 * s_max
    active = sorted_idx[scores >= s_max - band]
    if len(active) == 0:
        active = sorted_idx[:n_top]
    d = G[active].mean(axis=0)
    d -= np.mean(d)
    norm = np.sqrt(np.sum(d ** 2))
    if norm < 1e-12:
        return None
    return -d / norm


def _smooth_obj(x, beta, G_cache=None):
    scores = _scores(x)
    s_max = float(np.max(scores))
    return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta


def _slsqp_quick(h, target, beta, max_iter=60):
    from scipy.optimize import minimize
    n = len(h)

    def obj(x):
        scores = _scores(x)
        s_max = float(np.max(scores))
        return s_max + np.log(np.sum(np.exp(beta * (scores - s_max)))) / beta

    def jac(x):
        G = _grad_matrix(x)
        scores = _scores(x)
        s_max = float(np.max(scores))
        exp_s = np.exp(beta * (scores - s_max))
        w = exp_s / np.sum(exp_s)
        return G.T @ w

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(len(x))

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.97

    # Initialize from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP quick improvement (15% budget)
    deadline1 = t0 + budget_s * 0.15
    for beta in [10.0, 30.0, 80.0]:
        if time.time() > deadline1:
            break
        h_try = _slsqp_quick(h, target0, beta, max_iter=80)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()

    # Phase 2: Main LP local search (55% budget)
    deadline2 = t0 + budget_s * 0.70
    n = len(best_h)
    target = n / 2.0
    h_cur = best_h.copy()

    # Run LP loop with various parameters
    lp_configs = [
        (8, 0.15, 5000, 1e-6),
        (12, 0.08, 5000, 1e-6),
        (6, 0.20, 5000, 1e-6),
        (10, 0.05, 8000, 1e-7),
        (15, 0.03, 8000, 1e-7),
    ]
    for cfg in lp_configs:
        if time.time() > deadline2:
            break
        na, ri, mi, rm = cfg
        h_lp, c5_lp = _lp_loop(h_cur, deadline2, na, ri, mi, rm)
        if c5_lp < best_c5 - 1e-12:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h_cur = h_lp.copy()

    # Phase 3: Pair polish (15% budget)
    deadline3 = t0 + budget_s * 0.85
    if time.time() < deadline3:
        hp, c5p = _pair_polish(best_h, deadline3, max_passes=40)
        if c5p < best_c5 - 1e-14:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 4: NPC - random perturbation + LP (10% budget)
    deadline4 = t0 + budget_s * 0.95
    if time.time() < deadline4:
        for attempt in range(30):
            if time.time() > deadline4:
                break
            if best_c5 < 0.38070:
                break
            scale = 0.02 + 0.03 * rng.random()
            d = _random_direction(best_n, rng, scale)
            h_jump = _project(best_h + d, target)
            c5_j = _exact_c5(h_jump)
            if c5_j < best_c5 + 0.002:
                h_lp, c5_lp = _lp_loop(h_jump, min(deadline4, time.time() + budget_s * 0.02),
                                       10, 0.10, 2000, 1e-6)
                if c5_lp < best_c5 - 1e-12:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()

    # Phase 5: Resolution sweep
    if time.time() < t0 + budget_s * 0.97:
        deadline5 = t0 + budget_s * 0.975
        for dn in [1, -1, 2, -2, 3, -3, 4, -4]:
            if time.time() > deadline5:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 120:
                continue
            h_r = _resample(best_h, n_try)
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5 - 1e-13:
                h_lp, c5_lp = _lp_loop(h_r, min(deadline5, time.time() + budget_s * 0.015),
                                       10, 0.08, 1500, 1e-6)
                if c5_lp < best_c5 - 1e-13:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_try

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995, 12, 0.03, 2000, 1e-7)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, best_n
