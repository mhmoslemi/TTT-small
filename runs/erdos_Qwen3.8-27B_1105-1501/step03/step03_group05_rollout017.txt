import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Binary search for the threshold
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_gradient_matrix(h):
    """Build the gradient matrix G where G[r, j] = d(score_r)/d(h[j]) * dx"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _cvx_trust_step(h, rho, solver='CLARABEL'):
    """Solve the trust-region minimax problem using CVXPY"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = _scores(h) * dx
    G = _build_gradient_matrix(h)
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n / 2.0,
        F + G @ (x - h) <= t,
        cp.norm(x - h, 2) <= rho,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return x.value
    except Exception:
        pass
    
    # Try other solvers
    for s in ['ECOS', 'SCS', 'OSQP', 'MOSEK']:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return x.value
        except Exception:
            continue
    return None

def _smooth_grad(h, beta):
    """Compute smooth gradient of max score"""
    n = len(h)
    scores = _scores(h)
    max_s = float(np.max(scores))
    exp_vals = np.exp(beta * (scores - max_s))
    w = exp_vals / float(np.sum(exp_vals))
    
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if abs(wd) < 1e-14:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            grad[q] += wd * (1.0 - h[p])
            grad[p] -= wd * h[q]
    return grad

def _line_search(h, d, cur_c5, n_steps=20):
    """Line search along direction d"""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = cur_c5
    alphas = np.linspace(0.0, alpha_max, n_steps)
    for a in alphas:
        if a < 1e-10:
            continue
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, cur_c5
    
    lo = max(0.0, best_a - alpha_max / n_steps)
    hi = min(alpha_max, best_a + alpha_max / n_steps)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(12):
        if hi - lo < 1e-9 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _gradient_descent(h, deadline, beta_start=5.0, beta_end=300.0, n_beta_steps=15):
    """Continuation method with gradient descent"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for beta in np.geomspace(beta_start, beta_end, n_beta_steps):
        if time.time() > deadline:
            break
        step_size = 0.1
        stall = 0
        for step in range(5000):
            if time.time() > deadline:
                break
            g = _smooth_grad(h, beta)
            g -= np.mean(g)
            norm = float(np.max(np.abs(g)))
            if norm < 1e-14:
                break
            d = -g / norm * step_size
            a, new_c5 = _line_search(h, d, cur_c5)
            if new_c5 < cur_c5 - 1e-12:
                h = h + a * d
                h = _project_box_sum(h, target)

            else:

def _cvx_descent(h, deadline, sigma_start=0.05, sigma_min=0.005, sigma_max=0.3):
    """Trust-region optimization using CVXPY"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sigma = sigma_start
    stall = 0
    
    step_factors = [1.0, 0.75, 0.5, 0.3, 0.15]
    
    for it in range(200):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        rho = sigma * np.sqrt(n)
        x_new = _cvx_trust_step(h, rho)
        
        if x_new is not None:
            found_improvement = False
            for factor in step_factors:
                trial = h + factor * (x_new - h)
                trial = _project_box_sum(trial, target)
                c5_trial = _exact_c5(trial)
                if c5_trial < cur_c5 - 1e-11:

                    cur_c5 = c5_trial

                stall = 0
                sigma = min(sigma * 1.3, sigma_max)
            else:
                stall += 1
                sigma = max(sigma * 0.7, sigma_min)
        else:
            stall += 1
            sigma = max(sigma * 0.6, sigma_min)
    
    return best_h, best_c5

def _binary_local_search(h, deadline, max_iters=100000, rng=None):

    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)

    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        if not ones or not zeros:
            break
        
        best_swap = None
        best_val = cur_c5
        batch = min(50, len(ones) * len(zeros))
        for _b in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = _exact_c5(h_bin)
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            if c5_new < best_val - 1e-14:
                best_val = c5_new
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    """Resample h to n_new points using linear interpolation"""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _lp_step(h, scores, radius):
    """LP-based direction finding"""
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += (1.0 - h[p])
            G[r, p] -= h[q]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores * dx  # We want G @ delta <= t, i.e., G @ delta - t <= 0
    
    # Actually let me reformulate: minimize t s.t. scores + G@delta <= t, i.e., G@delta - t <= -scores
    # With bounds on delta: -radius*(1-h) <= delta_j <= radius*h_j for those at bounds
    # But we also need sum(delta) = 0 (to preserve the sum constraint)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})
        if res.success:
            return res.x[:n]

    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _smooth_grad(h, beta):
    n = len(h)
    scores = _scores_raw(h)
    mx = float(np.max(scores))
    ev = np.exp(beta * (scores - mx))
    w = ev / float(np.sum(ev))
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[n - 1 + d]
        if wd < 1e-15:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            grad[q] += wd * (1.0 - h[p])
            grad[p] -= wd * h[q]
    return grad

def _topk_grad(h, k=5):
    n = len(h)
    scores = _scores_raw(h)
    top = np.argsort(-scores)[:k]
    grad = np.zeros(n)
    for r in top:
        d = r - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            grad[q] += (1.0 - h[p])
            grad[p] -= h[q]
    return grad / float(k)

def _line_search(h, d, cur_c5, n_steps=16):
    n = len(h)
    alpha_max = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    alpha_max = max(alpha_max, 1e-8)
    
    best_a = 0.0
    best_c5 = cur_c5
    for a in np.linspace(1e-8, alpha_max, n_steps):
        c5_t = _exact_c5(h + a * d)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, cur_c5
    
    lo = max(0.0, best_a - alpha_max / n_steps)
    hi = min(alpha_max, best_a + alpha_max / n_steps)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = _exact_c5(h + c1 * d), _exact_c5(h + c2 * d)
    for _ in range(10):
        if hi - lo < 1e-10 * max(alpha_max, 1e-8):
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
