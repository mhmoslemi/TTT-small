import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Water-filling algorithm
    for _ in range(100):
        diff = target - np.sum(v)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        # Check if step would violate bounds
        if diff > 0:
            max_step = np.min(1.0 - v[mask])
            step = min(step, max_step)
        else:
            max_step = np.min(v[mask])
            step = -min(-step, max_step)
        v[mask] += step
    # Final correction
    diff = target - np.sum(v)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if np.any(mask):
            v[mask] += diff / np.sum(mask)
    return np.clip(v, 0.0, 1.0)

def _solve_lp(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx  # current scores
    G = _build_gradient(h)  # gradient matrix (includes dx factor)
    
    # LP: min t s.t. F + G@(x-h) <= t, 0<=x<=1, sum(x) = n/2
    # Variables: [x_0, ..., x_{n-1}, t]
    # => G@x - t <= G@h - F
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        for i in range(n):
            if 0 <= i + d < n:
                G[r, i] -= dx * h[i + d]
            if 0 <= i - d < n:
                G[r, i] += dx * (1.0 - h[i - d])
    return G

def _solve_lp(h):
    n = len(h)
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    G = _build_gradient(h)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _lp_optimize(h, deadline, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        
        x_star = _solve_lp(h)
        if x_star is None:
            stall += 5
            continue
        
        if np.max(np.abs(x_star - h)) < 1e-11:
            stall += 1
            continue
        
        # Try the LP solution directly
        new_h = _project_box_sum(x_star, target)
        new_c5 = _exact_c5(new_h)
        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            continue
        
        # Line search
        d = x_star - h
        best_alpha = 0.0
        best_trial = cur_c5
        alphas = [0.5, 0.3, 0.7, 0.2, 0.9, 0.1]
        for a in alphas:
            trial = _project_box_sum(h + a * d, target)
            tc = _exact_c5(trial)
            if tc < best_trial - 1e-12:
                best_trial = tc
                best_alpha = a
        
        if best_alpha > 0:
            h = _project_box_sum(h + best_alpha * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _random_perturb(h, rng, strength=0.15):
    n = len(h)
    target = n / 2.0
    h_p = h.copy()
    n_transfers = max(4, n // 4)
    for _ in range(n_transfers):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-strength, strength)
        # Clamp to maintain [0,1]
        if h_p[i] + delta > 1:
            delta = 1.0 - h_p[i]
        if h_p[i] + delta < 0:
            delta = -h_p[i]
        if h_p[j] - delta > 1:
            delta = h_p[j] - 1.0
        if h_p[j] - delta < 0:
            delta = h_p[j]
        if abs(delta) < 1e-10:
            continue
        h_p[i] += delta
        h_p[j] -= delta
    return _project_box_sum(h_p, target)

def _resample_pc(h, n_new):
    """Piecewise constant resampling"""
    n = len(h)
    if n == n_new:
        return h.copy()
    h_new = np.zeros(n_new)
    for j in range(n_new):
        new_start = j / n_new
        new_end = (j + 1) / n_new
        mass = 0.0
        for i in range(n):
            old_start = i / n
            old_end = (i + 1) / n
            overlap = min(new_end, old_end) - max(new_start, old_start)
            if overlap > 0:
                mass += h[i] * overlap
        h_new[j] = mass / (1.0 / n_new)
    return _project_box_sum(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=20):
    """Polish by trying pairwise adjustments"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try adjusting pairs of interior points
        interior = [j for j in range(n) if 1e-6 < h[j] < 1.0 - 1e-6]
        if len(interior) < 2:
            break
        # Limit pairs to keep it fast
        pairs = []
        for a in range(len(interior)):
            for b in range(a+1, min(a+10, len(interior))):
                pairs.append((interior[a], interior[b]))
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            for di, dj in [(0.01, -0.01), (-0.01, 0.01), (0.1, -0.1), (-0.1, 0.1)]:
                if h[i] + di < 0 or h[i] + di > 1:
                    continue
                if h[j] + dj < 0 or h[j] + dj > 1:
                    continue
                trial = h.copy()
                trial[i] += di
                trial[j] += dj
                tc = _exact_c5(trial)
                if tc < best_c5 - 1e-12:
                    h = trial
                    best_c5 = tc
                    best_h = h.copy()
                    improved = True
    return best_h, best_c5
