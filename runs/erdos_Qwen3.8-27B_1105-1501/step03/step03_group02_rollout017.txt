def _golden_section_1d(f, a, b, tol=1e-10, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

def _pairwise_refine(h_init, deadline, n_pairs=200):

            break

        if i == j:
            continue

        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])

        h_save = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[:m] -= h[d:d+m]
            g[d:] += 1.0 - h[:m]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += 1.0 - h[gg:gg+m]
            g[gg:] -= h[:m]
    return g

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        return float(np.sum(h[d:d+m] * (1.0 - h[:m]))) if m > 0 else 0.0
    else:
        gg = -d
        m = n - gg
        return float(np.sum(h[:m] * (1.0 - h[gg:gg+m]))) if m > 0 else 0.0

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-14) if res > 0 else (x > 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_val_grad(h, beta):
    n = len(h)
    s = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(s))
    sh = s - mx
    e = np.exp(beta * sh)
    ess = float(np.sum(e))
    if ess < 1e-300:
        val = mx / beta
        g = np.zeros(n)
        am = int(np.argmax(s))
        d = am - (n - 1)
        g = _score_grad(h, d) / beta
        return val, g
    w = e / ess
    val = (mx + float(np.log(ess))) / beta
    g = np.zeros(n)
    for idx in range(len(s)):
        if w[idx] < 1e-14:
            continue
        d = idx - (n - 1)
        g += w[idx] * _score_grad(h, d)
    g /= beta
    return val, g

def _slsqp_lse(h0, dl, beta, maxiter=200, ftol=1e-12):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        v, _ = _lse_val_grad(x, beta)
        return v
    def jac(x):
        _, g = _lse_val_grad(x, beta)
        return g

def _golden(f, a, b, tol=1e-11, mi=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a); d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    for _ in range(mi):
        if b - a < tol: break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a); fc = f(c)
        else:
            a, c, fc = c, d, fc
            d = a + gr * (b - a); fd = f(d)
    return (a + b) / 2.0

def _pairwise(h0, dl, mi=3000, seed=7):

            a, b = tlo, thi

        h = np.clip(h, 0.0, 1.0)

def _resample(h, nn):
