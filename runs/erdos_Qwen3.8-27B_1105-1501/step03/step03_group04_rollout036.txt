**Active-set trust-region method:**

At each iteration:
1. Compute scores = _scores(h), find max score M and the set of "active" lags (those within some tolerance of M)
2. Compute gradients for active lags only
3. Trust-region subproblem: find direction p that minimizes the max of (scores[r] + grad_r · p) over active r, subject to ||p||_inf ≤ Δ (trust region), sum(p) = 0
4. This is an LP! (minimize t subject to grad_r · p ≤ t - scores[r] for each active r, ||p||_inf ≤ Δ, sum(p)=0)
5. Take a step in direction p with step size α determined by exact evaluation
6. If the actual reduction is less than expected (say < 0.1 * predicted), shrink the trust region
7. If the actual reduction is good (> 0.5 * predicted), expand the trust region

This is a proper trust-region method for the max problem.

The LP has:
- Variables: p (length n) and t (scalar) → n+1 variables
- Constraints: K active-set constraints (K × (n+1)), n box constraints (2n), 1 equality
- Total: K + 2n + 1 constraints, n+1 variables

For n=64 and K=20 (say 20 active lags), this is a small LP that scipy's HiGHS should solve in milliseconds.

**Step size determination:**
After getting direction p from the LP, find α by:
- Checking that h + α*p stays in [0,1]
- Evaluating C₅(h + α*p) exactly
- Using golden section or bisection to find the best α

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_arr(h):

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, r):
    n = len(h)
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            if d > 0:
                grad[d:] -= h[:m]
            else:
                grad[:] -= h[:]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[g:] += (1.0 - h[:m])
            grad[:m] -= h[g:]
    return grad

        base = _thue_morse_seq(half)
        tm[:half] = base
        tm[half:] = 1 - base
    elif mode == 'random':

        tm[idx[:n//2]] = 1
    elif mode == 'sine_round':

                    tm[ones[i]] = 0
            else:
                for i in range(n // 2 - s):
                    tm[zeros[i]] = 1
    else:
        tm = np.zeros(n, dtype=int)
    return tm

def _trust_region_lp(h, deadline, delta_init=0.08, active_count=15, max_iters=3000):

    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    delta = delta_init
    stall = 0
    n_lags = 2 * n - 1

    for it in range(max_iters):

            break
        if stall > 100:
            break

        scores = _scores_arr(h)
        max_score = float(np.max(scores))

        # Select active set: top active_count lags
        active = np.argsort(scores)[::-1][:active_count]

        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        for k, r in enumerate(active):
            g = _score_grad(h, r)
            A_ub[k, :n] = g
            A_ub[k, n] = -1.0
            b_ub[k] = -scores[r]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        lb = np.maximum(-delta, -h)
        ub = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lb, ub))
        bounds.append((None, None))

        try:
            res = linprog(np.zeros(n + 1), A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

            continue

        if not res.success:
            stall += 1
            delta *= 0.7
            continue

        predicted_t = res.x[n]

            delta *= 0.7

                break
            continue

def _soft_max_descent(h, deadline, beta_start=15.0, beta_end=3000.0, max_iters=8000):

        improved = False
        alpha = lr

def _block_nelder_mead(h, block_size=6, deadline=None, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_blocks = n // block_size

    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved_this_pass = False
        rng_local = np.random.default_rng(_pass * 31 + 7)
        block_order = rng_local.permutation(n_blocks)
        for bi in block_order:
            if deadline is not None and time.time() > deadline:
                break
            start = bi * block_size
            end = start + block_size
            x0 = h[start:end].copy()
            rest = h.copy()

            def _block_obj(xb):
                rest[start:end] = xb
                # Adjust sum: need sum(rest) = target
                diff = target - float(np.sum(rest))
                if abs(diff) > 1e-10:
                    # Distribute equally among non-block entries
                    mask = np.ones(n, dtype=bool)
                    mask[start:end] = False
                    cnt = int(np.sum(mask))
                    if cnt > 0:
                        rest[mask] += diff / cnt
                        rest = np.clip(rest, 0.0, 1.0)
                    # Re-adjust block
                    diff2 = target - float(np.sum(rest))
                    if abs(diff2) > 1e-10:
                        xb_adj = np.clip(xb + diff2 / block_size, 0.0, 1.0)
                        rest[start:end] = xb_adj
                return _exact_c5(rest)

            x0_clipped = np.clip(x0, 0.0, 1.0)
            res = minimize(_block_obj, x0_clipped, method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
            if res.fun < best_c5 - 1e-10:
                h[start:end] = res.x
                # Re-project
                h = _project(h, target)
                c5_now = _exact_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                    improved_this_pass = True

        if not improved_this_pass:
            break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturbed_restart(h_best, rng, scale=0.04):
    h_pert = h_best + rng.normal(0, scale, len(h_best))
    return _project(h_pert, len(h_best) / 2.0)
