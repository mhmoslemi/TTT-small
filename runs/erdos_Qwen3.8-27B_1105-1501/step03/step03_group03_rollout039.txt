```python
                G[s, d:] += (1.0 - h[:m])
```
```python
        else:

                G_rows[i, g:] -= h[:m]
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _build_G(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        s = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[s, :m] -= h[d:]
                G[s, d:] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[s, :m] += (1.0 - h[g:])
                G[s, g:] -= h[:m]
    return G


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if np.any(~np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lp_direction(h, scores, G, active_idx, radius):
    n = len(h)
    m = len(active_idx)
    if m == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    for i, k in enumerate(active_idx):
        A_ub[i, :n] = G[k, :]
        A_ub[i, n] = -1.0
    b_ub = np.array([-scores[k] for k in active_idx])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.fun
    except Exception:
        pass
    return None, None


def _line_search(h, p, c5_cur):
    n = len(p)
    alpha_max = 1e10
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.99)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.99)
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 5.0)

    best_a = 0.0
    best_c5 = c5_cur
    n_coarse = 32
    alphas = np.linspace(0.0, alpha_max, n_coarse)
    for a in alphas:
        if a < 1e-12:
            continue
        c5_t = _exact_c5(h + a * p)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a

    if best_a == 0.0:
        return 0.0, c5_cur

    lo = max(0.0, best_a - alpha_max / n_coarse)
    hi = min(alpha_max, best_a + alpha_max / n_coarse)
    if hi - lo < 1e-12:
        return best_a, best_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _lp_loop(h, deadline, max_iters=100000, radius_init=0.15,
             active_thresh=0.03, active_max=40):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    stall2 = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break

        scores = _scores(h)
        G = _build_G(h)
        s_max = float(np.max(scores))
        thresh = s_max - active_thresh * max(abs(s_max), 0.05)
        active = np.where(scores >= thresh)[0]
        if len(active) > active_max:
            order = np.argsort(scores[active])[::-1][:active_max]
            active = active[order]

        p, t_val = _lp_direction(h, scores, G, active, radius)

        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.9
            if stall % 25 == 0 and radius < 0.02:
                radius = 0.08
            if radius < 1e-7:
                break
            continue

        a, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + a * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            stall2 = 0
            radius = min(radius * 1.04, 0.5)
        else:
            stall += 1
            stall2 += 1
            radius *= 0.85
            if stall2 % 30 == 0 and radius < 0.03:
                radius = 0.12

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 16:
        n_new = 16
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _active_descent(h, deadline, max_iters=5000, step_init=0.08):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break

        scores = _scores(h)
        s_max = float(np.max(scores))
        thresh = s_max - 0.04 * max(abs(s_max), 0.05)
        active = np.where(scores >= thresh)[0]
        G = _build_G(h)
        g = G[active, :].mean(axis=0)
        g -= np.mean(g)
        gnorm = float(np.sqrt(np.sum(g * g)))
        if gnorm < 1e-12:
            break
        g = g / gnorm

        h_try = _project(h - step * g, target)
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.05, 0.3)
        else:
            stall += 1
            step *= 0.85
            if step < 1e-6:
                break

    return best_h, best_c5


def _explore(best_h, best_c5, deadline, rng, per_try_time):
    n = len(best_h)
    h = best_h.copy()
    c5 = best_c5
    attempts = 0
    max_attempts = int((deadline - time.time()) / max(per_try_time, 1.0))
    max_attempts = min(max_attempts, 200)

    for _ in range(max_attempts):
        if time.time() > deadline:
            break
        attempts += 1
        scale = rng.uniform(0.01, 0.10)
        h_jump = h + scale * rng.standard_normal(n)
        h_jump = _project(h_jump, n / 2.0)
        c5_jump = _exact_c5(h_jump)
        # Only try LP if the jump didn't make things too bad
        if c5_jump < c5 + 0.005:
            h_try, c5_try = _lp_loop(h_jump, time.time() + per_try_time,
                                     radius_init=0.15, max_iters=5000)
            if c5_try < c5 - 1e-13:
                c5 = c5_try
                h = h_try.copy()

    return h, c5


def _pair_polish(h, deadline, max_passes=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        for ii in range(len(interior)):
            if time.time() > deadline:
                break
            j = interior[ii]
            for kk in range(ii + 1, min(ii + 8, len(interior))):
                k = interior[kk]
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 0.5 * min(h[j] if dj < 0 else 1.0 - h[j],
                                      h[k] if dk < 0 else 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += dj * a
                        h_t[k] += dk * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            h = h_t.copy()
                            best_h = h.copy()
                            improved = True
                            break
        if not improved:
            break

    return best_h, best_c5
```
