def project_feasible(v, target):
    n = len(v)

        return np.zeros(n)

        return np.ones(n)

    for _ in range(50):

        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)

            grad[j] += (1.0 - h[j - lag])
        if 0 <= j + lag <= n - 1:
            grad[j] -= h[j + lag]
    return grad

        if hj <= 1e-8 or hj >= 1.0 - 1e-8:
            continue
        for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
            new_val = hj + delta
            if new_val < 0 or new_val > 1:
                continue
            # Need to redistribute: move delta to a random set of other coords
            n_others = n - 1
            share = -delta / n_others
            h_t = h.copy()
            h_t[j] = new_val
            for k in range(n):
                if k != j:
                    h_t[k] += share
            if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
                continue
            h_t = np.clip(h_t, 0.0, 1.0)
            h_t = project_feasible(h_t, target)
            c5_t = exact_c5(h_t)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
                h = h_t.copy()
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Start from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project_feasible(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Quick multi-resolution sweep
    deadline1 = t0 + budget_s * 0.30
    n_candidates = []
    for dn in range(-15, 16, 2):
        n_try = n0 + dn
        if n_try < 30 or n_try > 160:
            continue
        n_candidates.append(n_try)
    # Also add some non-standard values
    n_candidates.extend([n0 - 1, n0 + 1])
    n_candidates = sorted(set(n_candidates))

    per_n_budget = (deadline1 - time.time()) / max(len(n_candidates), 1)
    for n_try in n_candidates:
        if time.time() > deadline1 - 5:
            break
        per_budget = max(2.0, per_n_budget)
        sub_dl = min(deadline1, time.time() + per_budget)
        if n_try == n0:
            h_try = h.copy()
        else:
            h_try = resample_to(best_h, n_try)
        h_opt, c5_opt = optimize_fast(h_try, sub_dl, rng, n_iters=400)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 2: Focused optimization on best and neighbors
    deadline2 = t0 + budget_s * 0.75
    focus_ns = [best_n]
    for dn in [-3, -2, -1, 1, 2, 3, -5, 5]:
        nn = best_n + dn
        if 30 <= nn <= 160 and nn not in focus_ns:
            focus_ns.append(nn)
    focus_ns = focus_ns[:6]

    per_focus = (deadline2 - time.time()) / max(len(focus_ns), 1)
    for n_f in focus_ns:
        if time.time() > deadline2 - 5:
            break
        sub_dl = min(deadline2, time.time() + per_focus)
        if n_f == best_n:
            h_f = best_h.copy()
        else:
            h_f = resample_to(best_h, n_f)
        h_opt, c5_opt = optimize_thorough(h_f, sub_dl, rng, n_iters=15000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_f

    # Phase 3: Try a few more n values if we have time
    deadline3 = t0 + budget_s * 0.85
    for dn in [-7, 7, -10, 10, -12, 12]:
        if time.time() > deadline3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 160:
            continue
        h_try = resample_to(best_h, n_try)
        sub_dl = min(deadline3, time.time() + 15.0)
        h_opt, c5_opt = optimize_fast(h_try, sub_dl, rng, n_iters=600)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 4: Pairwise refinement
    deadline4 = t0 + budget_s * 0.92
    if time.time() < deadline4:
        best_h, best_c5 = pairwise_refine(best_h, deadline4, max_passes=30)

    # Phase 5: Coordinate refine + final line search
    deadline5 = t0 + budget_s * 0.97
    if time.time() < deadline5:
        best_h, best_c5 = coordinate_refine(best_h, min(deadline5, time.time() + 60))
        # Final line search polish
        sub_dl = min(deadline5, time.time() + 30)
        best_h, best_c5 = optimize_fast(best_h, sub_dl, rng, n_iters=500)

    # Final project and evaluate
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
