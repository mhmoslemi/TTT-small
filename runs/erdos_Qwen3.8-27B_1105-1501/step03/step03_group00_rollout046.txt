    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p_val = p_start * (p_end / p_start) ** frac
        beta = beta0 * (10.0 ** frac)
        scores = _scores(h)
        max_s = np.max(scores)
        shifted = np.maximum(scores - max_s + 1e-15, 1e-30)
        w = (shifted / max_s) ** p_val
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_scores(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = momentum * v + (1.0 - momentum) * grad / gn
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v = v / vn
        found = False
        alpha = 0.08
        for _ in range(35):
            h_try = h + alpha * v
            h_projected = _project(h_try, target)
            c5_new = _exact_c5(h_projected)
            if c5_new < cur_c5 - 1e-14:
                h = h_projected
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                found = True
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
        if not found:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None, thresh_frac=0.0):
    n = len(h)
    n_lags = len(scores)
    max_s = np.max(scores)
    if thresh_frac > 0:
        threshold = max_s - thresh_frac * max_s
        active = np.where(scores >= threshold)[0]
    elif top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        active = idx
    else:
        active = np.arange(n_lags)
    if len(active) == 0:
        return None, None
    scores_k = scores[active]
    G_k = G[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-12)
        hi = min(radius, 1.0 - h[j] - 1e-12)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.12, max_iters=3000, thresh_init=0.005):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    thresh = thresh_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_scores(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk, thresh_frac=thresh)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            thresh = max(0.001, thresh * 0.7)
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.4)
            thresh = min(thresh * 1.1, 0.02)
        else:
            stall += 1
            radius *= 0.75
            thresh = max(0.001, thresh * 0.7)
            ti += 1
            if stall > 20 and radius < 0.02:
                radius = 0.06
                thresh = 0.005
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_start, deadline, n_iters=5000, T0=None, T_end=None):
    n = len(h_start)
    target = n / 2.0
    if T0 is None:
        T0 = 0.002
    if T_end is None:
        T_end = 1e-5
    h = np.round(h_start)
    n_ones = int(np.sum(h))
    diff = int(round(target)) - n_ones
    if diff > 0:
        zeros_idx = np.where(h < 0.5)[0]
        if len(zeros_idx) >= diff:
            h[zeros_idx[:diff]] = 1.0
    elif diff < 0:
        ones_idx = np.where(h > 0.5)[0]
        if len(ones_idx) >= -diff:
            h[ones_idx[:-diff]] = 0.0
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(1, n_iters - 1))
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(np.random.randint(len(ones_idx)))
        j = int(np.random.randint(len(zeros_idx)))
        ii = ones_idx[i]
        jj = zeros_idx[j]
        h[ii] = 0.0
        h[jj] = 1.0
        c5_new = _exact_c5(h)
        if c5_new < cur_c5:
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            delta = c5_new - cur_c5
            if np.log(np.random.uniform()) < -delta / max(T, 1e-10):
                cur_c5 = c5_new
            else:
                h[ii] = 1.0
                h[jj] = 0.0
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _block_initial(n, n_blocks, rng):
    h = np.zeros(n)
    block_size = n // n_blocks
    remaining = n - block_size * n_blocks
    pos = 0
    for b in range(n_blocks):
        sz = block_size + (1 if b < remaining else 0)
        if rng.random() < 0.5:
            h[pos:pos + sz] = 1.0
        pos += sz
    return _project(h, n / 2.0)

def _periodic_initial(n, period, duty, rng):
    h = np.zeros(n)
    k = max(1, int(period * duty))
    for i in range(n):
        if i % period < k:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _sinusoidal_initial(n, freq, phase, rng):
    x = np.linspace(0, 1, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Coarse-to-fine multi-resolution
    # Try lower resolutions first to find good basins
    dl1 = t0 + budget_s * 0.12
    for n_try in [32, 40, 48]:
        if time.time() > dl1:
            break
        h_c = _resample(best_h, n_try)
        sub_dl = min(dl1, time.time() + budget_s * 0.03)
        h_c, c5_c = _smooth_optimize(h_c, sub_dl, p_start=4, p_end=50, n_steps=80, beta0=6.0, momentum=0.4)
        if c5_c < _exact_c5(_resample(best_h, n_try)) - 0.01:
            h_up = _resample(h_c, best_n)
            c5_up = _exact_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
    
    # Phase 2: Diverse multi-start at target resolution
    dl2 = t0 + budget_s * 0.28
    candidates = []
    # Structured patterns
    for nb in [2, 3, 4, 5, 6]:
        for _ in range(3):
            h_b = _block_initial(best_n, nb, rng)
            candidates.append(h_b)
    for per in [3, 4, 5, 6, 7, 8]:
        for duty in [0.3, 0.4, 0.5]:
            h_p = _periodic_initial(best_n, per, duty, rng)
            candidates.append(h_p)
    for freq in [1.5, 2.0, 2.5, 3.0]:
        for ph in [0, np.pi / 3, np.pi, 2 * np.pi / 3]:
            h_s = _sinusoidal_initial(best_n, freq, ph, rng)
            candidates.append(h_s)
    # Random perturbations of current best
    for scale in [0.1, 0.2, 0.3]:
        h_r = best_h + rng.normal(0, scale, best_n)
        candidates.append(_project(h_r, best_n / 2.0))
    
    # Evaluate and pick best few
    for h_c in candidates:
        if time.time() > dl2:
            break
        c5_c = _exact_c5(h_c)
        if c5_c < best_c5 + 0.02:
            sub_dl = min(dl2, time.time() + budget_s * 0.02)
            h_opt, c5_opt = _smooth_optimize(h_c, sub_dl, p_start=4, p_end=80, n_steps=60, beta0=8.0, momentum=0.35)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 3: Main smooth optimization with increasing p
    dl3 = t0 + budget_s * 0.42
    h, c5 = _smooth_optimize(best_h, dl3, p_start=6, p_end=300, n_steps=300, beta0=10.0, momentum=0.3)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 4: LP optimization
    dl4 = t0 + budget_s * 0.60
    h, c5 = _lp_optimize(best_h, dl4, top_k_list=[None, 12, 6, 4], radius_init=0.15, max_iters=3000, thresh_init=0.008)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 5: Binary SA from current best
    if time.time() < t0 + budget_s * 0.70:
        dl5 = t0 + budget_s * 0.72
        h_bin, c5_bin = _binary_sa(best_h, dl5, n_iters=8000, T0=0.003, T_end=1e-5)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            # Relax binary solution
            sub_dl = min(t0 + budget_s * 0.76, time.time() + budget_s * 0.04)
            h_rel, c5_rel = _smooth_optimize(best_h, sub_dl, p_start=8, p_end=60, n_steps=100, beta0=12.0, momentum=0.25)
            if c5_rel < best_c5:
                best_c5 = c5_rel
                best_h = h_rel.copy()
    
    # Phase 6: Perturbation restarts with LP
    dl6 = t0 + budget_s * 0.82
    scale = 0.03
    for attempt in range(300):
        if time.time() > dl6:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.02)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k_list=[8, 5, None], radius_init=0.08, max_iters=300, thresh_init=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)
    
    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.90:
        dl7 = t0 + budget_s * 0.92
        hp, c5p = _pair_polish(best_h, dl7, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
    
    # Phase 8: Resolution sweep
    if time.time() < t0 + budget_s * 0.95:
        dl8 = t0 + budget_s * 0.96
        for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10, 12]:
            if time.time() > dl8:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 120:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl8, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k_list=[8, None], radius_init=0.08, max_iters=200, thresh_init=0.005)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 9: Final ultra-fine LP
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _lp_optimize(best_h, t0 + budget_s * 0.995, top_k_list=[3, 5, 4, 2], radius_init=0.01, max_iters=3000, thresh_init=0.002)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
