    n = len(h)

        h2[i] += d
        h2[j] -= d
        scores = _all_scores(h2)
        val = float(np.max(scores))

        hi_t = deltas[idx+1]
    else:
        lo_t = lo
        hi_t = hi

    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    h_c1 = h.copy(); h_c1[i] += c1; h_c1[j] -= c1
    h_c2 = h.copy(); h_c2[i] += c2; h_c2[j] -= c2
    f1 = float(np.max(_all_scores(h_c1)))
    f2 = float(np.max(_all_scores(h_c2)))

        if hi_t - lo_t < 1e-12:

            f2 = f1
