def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Repair
    res = target - float(x.sum())
    if abs(res) > 1e-10:
        if res > 0:
            for i in np.argsort(-x):
                add = min(res, 1.0 - x[i])
                x[i] += add; res -= add
                if res < 1e-12: break

            res = -res
            for i in np.argsort(x):
                red = min(res, x[i])
                x[i] -= red; res -= red
                if res < 1e-12: break
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _smooth_obj(x, alpha):
    s = np.correlate(x, 1.0 - x, mode='full')
    ms = float(np.max(s))
    vals = np.clip(alpha * (s - ms), -500.0, 500.0)
    ev = np.exp(vals)
    f = ms + (1.0/alpha) * np.log(float(np.sum(ev)))
    w = ev / float(np.sum(ev))
    G = _grad_matrix(x)
    g = G.T @ w
    g -= np.mean(g)
    return f, g

def _lbfgs(h, alpha, iters=200):

    bounds = [(0.0, 1.0)] * n
