import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    """Compute all correlation scores (raw, before dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    """Build gradient matrix G where G[k, j] = d(correlation_k)/d(h_j)."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m_cnt = n - d
            if m_cnt > 0:
                G[r, d:] += (1.0 - h[:m_cnt])
                G[r, :m_cnt] += -h[d:]
        else:
            g = -d
            m_cnt = n - g
            if m_cnt > 0:
                G[r, :m_cnt] += (1.0 - h[g:])
                G[r, g:] += -h[:m_cnt]
    return G

def _lp_direction_reduced(h, scores, G, radius, top_m=10):
    """LP direction using only top-m active constraints."""
    n = len(h)
    m = len(scores)
    # Select top-m constraints
    idx = np.argsort(scores)[::-1][:top_m]
    G_sub = G[idx, :]
    b_sub = -scores[idx]
    
    # LP: max delta_t
    # s.t. G_sub @ (h + p) - (scores_sub + delta_t) <= 0
    # i.e. G_sub @ p - delta_t <= -scores_sub
    # sum(p) = 0
    # |p_j| <= radius, box constraints
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize delta_t
    
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = b_sub  # = -scores[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur):
    """Exact line search along direction p."""
    n = len(h)
    alpha_max = 1e8
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(min(alpha_max, 2.0), 1e-8)
    
    best_a = 0.0
    best_c5 = c5_cur
    # Coarse grid
    n_grid = 16
    for i in range(1, n_grid + 1):
        a = alpha_max * i / n_grid
        h_t = h + a * p
        if np.any(h_t < -1e-6) or np.any(h_t > 1.0 + 1e-6):
            continue
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a = c2; c2 = c1; f2 = f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_a = c1; c1 = c2; f1 = f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coordinate_descent(h, deadline, max_rounds=100):
    """Pairwise coordinate descent."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for round_ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            # Try moving all pairs with j
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try (dj, dk) = (eps, -eps) and (-eps, eps)
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        am = min((1.0 - h[j]), h[k])
                    else:
                        am = min(h[j], (1.0 - h[k]))
                    if am < 1e-10:
                        continue
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        eps = am * frac
                        h_t = h.copy()
                        h_t[j] += sign * eps
                        h_t[k] -= sign * eps
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def _make_block_h(n, block_frac, seed=0):

    rng = np.random.default_rng(seed)

def _make_symmetric_h(n, seed=0):
    """Create an anti-symmetric initial h: h(i) + h(n-1-i) ≈ 1."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    half = n // 2
    # Random values in first half
    h[:half] = rng.uniform(0.2, 0.8, size=half)
    # Anti-symmetric completion
    h[half:] = 1.0 - h[:half][::-1]
    if n % 2 == 1:
        h[half] = 0.5
    return _project(h, n / 2.0)

def _soft_gradient(h, tau):
    """Compute softmax-weighted gradient of max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = _scores(h)
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        j_lo = max(0, d)
        j_hi = min(n - 1, d + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n - 1, n - 1 - d)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[idx2] += wk * (-h[idx2 + d])
    return grad

def _grad_descent(h, deadline, tau=50.0, lr=0.01, momentum=0.9):
    """Softmax gradient descent with momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad = _soft_gradient(h, tau)
        mom = momentum * mom + (1.0 - momentum) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            lr *= 0.92
            if stall > 40:
                break
    return best_h, best_c5
