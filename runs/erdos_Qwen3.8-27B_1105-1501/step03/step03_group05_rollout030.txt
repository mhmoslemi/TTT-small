import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_delta(h, i, j):
    """Compute linear and quadratic coefficients for pair transfer h[i]+=α, h[j]-=α."""
    n = len(h)
    lin = np.zeros(2 * n - 1)
    quad = np.zeros(2 * n - 1)
    
    for d in range(-(n - 1), n):
        r = n - 1 + d
        val = 0.0
        if 0 <= i - d < n:
            val += (1.0 - h[i - d])
        if 0 <= j - d < n:
            val -= (1.0 - h[j - d])
        if 0 <= i + d < n:
            val -= h[i + d]
        if 0 <= j + d < n:
            val += h[j + d]
        lin[r] = val
        
        q = 0.0
        if d == 0:
            q = -2.0
        if d == j - i:
            q += 1.0
        if d == i - j:
            q += 1.0
        quad[r] = q
    
    return lin, quad

def _pair_line_search(h, i, j, scores, dx):
    """Exact line search for pair transfer. Returns (alpha, objective)."""
    n = len(h)
    cur_best = float(np.max(scores) * dx)
    alpha_min = max(-h[i], h[j] - 1.0)
    alpha_max = min(1.0 - h[i], h[j])
    if alpha_max - alpha_min < 1e-14:
        return 0.0, cur_best
    
    lin, quad = _pair_delta(h, i, j)
    
    # Coarse grid
    n_pt = 60
    alphas = np.linspace(alpha_min, alpha_max, n_pt)
    mat = scores[np.newaxis, :] + alphas[:, np.newaxis] * lin[np.newaxis, :] + \
          (alphas[:, np.newaxis]) ** 2 * quad[np.newaxis, :]
    vals = np.max(mat, axis=1) * dx
    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    best_a = float(alphas[best_idx])
    
    if best_val >= cur_best - 1e-14:
        return 0.0, cur_best
    
    # Fine grid around best
    step = (alpha_max - alpha_min) / (n_pt - 1)
    lo = max(alpha_min, best_a - 3.0 * step)
    hi = min(alpha_max, best_a + 3.0 * step)
    alphas2 = np.linspace(lo, hi, 300)
    mat2 = scores[np.newaxis, :] + alphas2[:, np.newaxis] * lin[np.newaxis, :] + \
           (alphas2[:, np.newaxis]) ** 2 * quad[np.newaxis, :]
    vals2 = np.max(mat2, axis=1) * dx
    best_idx2 = int(np.argmin(vals2))
    if float(vals2[best_idx2]) < best_val:
        best_val = float(vals2[best_idx2])
        best_a = float(alphas2[best_idx2])
    
    return best_a, best_val

def _pair_local_search(h, deadline, max_passes=2000, seed=0):
    """Iterated pair-transfer local search with exact line search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _scores(h)
    
    for path in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(seed + path * 7919)
        order = rng.permutation(n)
        for idx in range(n):
            if time.time() > deadline:
                break
            i = int(order[idx])
            for jj in range(idx + 1, n):
                j = int(order[jj])
                a, val = _pair_line_search(h, i, j, scores, dx)
                if a != 0.0 and val < cur_c5 - 1e-13:
                    h[i] += a
                    h[j] -= a
                    # Ensure box constraint (should be satisfied by line search range)
                    h[i] = min(max(h[i], 0.0), 1.0)
                    h[j] = min(max(h[j], 0.0), 1.0)
                    cur_c5 = val
                    scores = _scores(h)
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_delta_vec(h, i, j):
    n = len(h)
    d = np.arange(-(n - 1), n)
    lin = np.zeros(2 * n - 1)
    m1 = (d >= i - n + 1) & (d <= i)
    if np.any(m1):
        lin[m1] += (1.0 - h[i - d[m1]])
    m2 = (d >= j - n + 1) & (d <= j)
    if np.any(m2):
        lin[m2] -= (1.0 - h[j - d[m2]])
    m3 = (d >= -i) & (d <= n - 1 - i)
    if np.any(m3):
        lin[m3] -= h[i + d[m3]]
    m4 = (d >= -j) & (d <= n - 1 - j)
    if np.any(m4):
        lin[m4] += h[j + d[m4]]
    quad = np.zeros(2 * n - 1)
    idx0 = n - 1
    quad[idx0] = -2.0
    dj = j - i
    if dj != 0:
        quad[n - 1 + dj] += 1.0
        quad[n - 1 - dj] += 1.0
    return lin, quad

def _pair_line_search(h, i, j, scores, dx):
    cur_best = float(np.max(scores) * dx)
    alpha_min = max(-h[i], h[j] - 1.0)
    alpha_max = min(1.0 - h[i], h[j])
    if alpha_max - alpha_min < 1e-14:
        return 0.0, cur_best
    lin, quad = _pair_delta_vec(h, i, j)
    n_pt = 80
    alphas = np.linspace(alpha_min, alpha_max, n_pt)
    a2 = alphas * alphas
    mat = scores[np.newaxis, :] + alphas[:, np.newaxis] * lin[np.newaxis, :] + a2[:, np.newaxis] * quad[np.newaxis, :]
    vals = np.max(mat, axis=1) * dx
    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    best_a = float(alphas[best_idx])
    if best_val >= cur_best - 1e-14:
        return 0.0, cur_best
    step = (alpha_max - alpha_min) / (n_pt - 1)
    lo = max(alpha_min, best_a - 3.0 * step)
    hi = min(alpha_max, best_a + 3.0 * step)
    alphas2 = np.linspace(lo, hi, 400)
    a2_2 = (alphas2 * alphas2)
    mat2 = scores[np.newaxis, :] + alphas2[:, np.newaxis] * lin[np.newaxis, :] + a2_2[:, np.newaxis] * quad[np.newaxis, :]
    vals2 = np.max(mat2, axis=1) * dx
    best_idx2 = int(np.argmin(vals2))
    if float(vals2[best_idx2]) < best_val:
        best_val = float(vals2[best_idx2])
        best_a = float(alphas2[best_idx2])
    return best_a, best_val

def _pair_local_search(h, deadline, max_passes=5000, seed=0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _scores(h)
    for path in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(seed + path * 104729)
        order = rng.permutation(n)
        for idx in range(n):
            if time.time() > deadline:
                break
            i = int(order[idx])
            for jj in range(idx + 1, n):
                j = int(order[jj])
                a, val = _pair_line_search(h, i, j, scores, dx)
                if a != 0.0 and val < cur_c5 - 1e-13:
                    h[i] += a
                    h[j] -= a
                    h[i] = min(max(h[i], 0.0), 1.0)
                    h[j] = min(max(h[j], 0.0), 1.0)
                    cur_c5 = val
                    scores = _scores(h)
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _cyclic_pattern(n, shift=0):
    h = np.zeros(n)
    for i in range(n):
        if ((i + shift) % (n // 2) < n // 4):
            h[i] = 1.0
    count = int(np.sum(h))
    if count != n // 2:
        h = _project_box_sum(h, n / 2.0)
    return h
