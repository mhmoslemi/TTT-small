def _gradient_descent(h, deadline, lr_init=0.1, max_iters=5000):
    """simple gradient descent with projection"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        # Gradient of max: use the active lag(s)
        max_val = float(np.max(scores))
        active = scores >= max_val - 1e-10
        grad = G[active, :].mean(axis=0)
        grad -= np.mean(grad)  # maintain sum constraint
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        # Line search
        alpha = lr
        found = False
        for _ in range(20):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 0.5)
                found = True
                break
            alpha *= 0.5
        if not found:
            stall += 1
            lr *= 0.8
            if lr < 1e-6 or stall > 100:
                break
    return best_h, best_c5

def _soft_descent(h, deadline, beta_start=30.0, beta_end=3000.0, max_iters=20000):
    """Soft-max weighted gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.08
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.4)
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-5 or stall > 60:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    R = 2 * n - 1
    G = np.zeros((R, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:n] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:n] += (1.0 - h[:m])
                G[r, :m] -= h[g:g+m]
    return G

            # Recompute scores and G for this block
            scores = _scores(h)
            G = _grad_matrix(h)
            
            block_end = min(block_start + block_size, npairs)
            for bi in range(block_start, block_end):
                if time.time() > deadline:
                    break
                i, j = pairs[order[bi]]
                t0 = h[i]
                t1 = h[j]
                s_val = t0 + t1
                tlo = max(0.0, s_val - 1.0)
                thi = min(1.0, s_val)
                if tlo >= thi - 1e-14:
                    continue
                
                a = G[:, i] - G[:, j]
                dlo = tlo - t0
                dhi = thi - t0
                
                # Grid search on linear approx
                grid = np.linspace(dlo, dhi, 32)
                best_d = 0.0
                best_val = float(np.max(scores))
                for d in grid:
                    val = float(np.max(scores + a * d))
                    if val < best_val:
                        best_val = val
                        best_d = d
                
                # Golden section refine
                span = (dhi - dlo) / 31
                lo = max(dlo, best_d - span)
                hi = min(dhi, best_d + span)
                if hi - lo < 1e-12:
                    continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = float(np.max(scores + a * c1))
                f2 = float(np.max(scores + a * c2))
                for _ in range(20):
                    if hi - lo < 1e-10:
                        break
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = float(np.max(scores + a * c1))
                    else:
                        lo, c1, f1 = c1, c2, f1
                        c2 = lo + gr * (hi - lo)
                        f2 = float(np.max(scores + a * c2))
                opt_d = (lo + hi) / 2.0
                opt_t = t0 + opt_d
                h[i], h[j] = opt_t, s_val - opt_t
                
                # Exact verification
                c5_new = _exact_c5(h)
                if c5_new > cur_c5 + 1e-12:
                    # Revert (quadratic terms made it worse)
                    h[i], h[j] = t0, t1
                else:
                    cur_c5 = c5_new
                    improved_this_pass += 1
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        
        if improved_this_pass == 0:
            break
    
    return best_h, best_c5

def _soft_descent(h, deadline, beta_start=30.0, beta_end=3000.0, max_iters=20000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.08
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        weights = exps / float(np.sum(exps))
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = _project(h + alpha * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.05, 0.4)
                improved = True
                break
            else:
                alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.75
            if lr < 1e-5 or stall > 60:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
