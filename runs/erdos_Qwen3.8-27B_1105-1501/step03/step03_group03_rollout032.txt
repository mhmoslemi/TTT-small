def _grad_scores(h):
    """Return G matrix where G[r,j] = d(score_r)/d(h_j), scores include dx factor"""

def _lp_loop(h, deadline, radius_init=0.1, max_iters=5000):
    """LP-based local search loop"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = _scores(h)
        G = _grad_scores(h)
        
        # Try active-set LP first (faster)
        p = _lp_direction_active(h, scores, G, radius, k_active=20)
        if p is None:
            p = _lp_direction(h, scores, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        
        alpha, new_c5 = _line_search_c5(h, p, cur_c5)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.75
            continue
        
        h_new = h + alpha * p
        h_new = _project(h_new, target)
        actual_c5 = _exact_c5(h_new)
        
        if actual_c5 < best_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.03
    
    return best_h, best_c5

def _subgradient_descent(h, deadline, step_init=0.05, momentum=0.7, 
                         active_tol=0.01, n_steps=5000):
    """Subgradient descent with momentum for the max of scores"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    step = step_init
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        s_max = float(np.max(scores))
        
        # Active set: scores within active_tol of max
        active = np.where(scores >= s_max - active_tol)[0]
        G = _grad_scores(h)
        grad = G[active, :].mean(axis=0)
        grad -= np.mean(grad)  # ensure sum(grad) = 0
        
        # Momentum update
        v = momentum * v + step * grad
        h_new = _project(h - v, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.01, 0.2)
        else:
            step *= 0.85
            v *= 0.5
            # Don't accept, try smaller step
            v_small = 0.3 * v
            h_try = _project(h - v_small, target)
            c5_try = _exact_c5(h_try)
            if c5_try < _exact_c5(h):
                h = h_try
                v = v_small
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=10):
    """Local pair search: adjust pairs of coordinates"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Only adjust pairs where both are strictly interior
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        idx = np.where(interior)[0]
        
        for i_pos in range(len(idx)):
            if time.time() > deadline:
                break
            j = idx[i_pos]
            for k_pos in range(i_pos + 1, len(idx)):
                k = idx[k_pos]
                # Try moving h[j] up and h[k] down (and vice versa)
                for sign in [1.0, -1.0]:
                    # Maximum step
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Try a few fractions
                    for frac in [0.3, 0.5, 0.7, 0.9, 1.0]:
                        a = a_max * frac
                        h_t = h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    
    return h, best_c5

- Second term (j, j+d): valid if 0 ≤ j < n and 0 ≤ j+d < n. Since 0 ≤ j < n is always true, we need 0 ≤ j+d < n.

So the gradient is simply:
∂c[r]/∂h[j] = (1-h[j-d]) * [0 ≤ j-d < n] + (-h[j+d]) * [0 ≤ j+d < n]

And the score (with dx factor) is:
score_r = dx * c[r]
∂score_r/∂h[j] = dx * [(1-h[j-d]) * [0 ≤ j-d < n] + (-h[j+d]) * [0 ≤ j+d < n]]

Let me verify with a small example. n=3, h = [a, b, c], d=1 (r = n-1+d = 3):
c[3] = sum_{n: 0≤n<3, 0≤n+1<3} h[n+1]*(1-h[n]) = h[1]*(1-h[0]) + h[2]*(1-h[1]) = b(1-a) + c(1-b)

∂c[3]/∂a = -(h[1]) = -b (from second term: j=0, d=1, j+d=1, -h[1] = -b; first term: j-d = -1, not valid)
∂c[3]/∂b = (1-h[0]) + (-h[2]) = (1-a) - c (from first term: j=1, j-d=0, 1-h[0]=1-a; second term: j=1, j+d=2, -h[2]=-c)
∂c[3]/∂c = (1-h[1]) = (1-b) (from first term: j=2, j-d=1, 1-h[1]=1-b; second term: j+d=3, not valid)

Check: c[3] = b - ab + c - bc
∂/∂a = -b ✓
∂/∂b = 1 - a - c ✓
∂/∂c = 1 - b ✓

Wait, max of convex functions is convex. Each s_d(t) is quadratic (hence convex or concave depending on the coefficient of t²). If the coefficient of t² is negative, s_d(t) is concave, and the max of concave functions is not necessarily convex.

Let me check: s_d = dx * sum_i h_i(t) * (1 - h_{i+d}(t)) where h_i(t) = h_i + t*p_i.

s_d(t) = dx * sum_i (h_i + t*p_i) * (1 - h_{i+d} - t*p_{i+d})
       = dx * sum_i [h_i(1-h_{i+d}) + t*p_i(1-h_{i+d}) - t*h_i*p_{i+d} - t²*p_i*p_{i+d}]
       = s_d(0) + t * dx * sum_i [p_i(1-h_{i+d}) - h_i*p_{i+d}] - t² * dx * sum_i p_i*p_{i+d}

The coefficient of t² is -dx * sum_i p_i * p_{i+d}, which can be positive or negative depending on the direction p.

If sum_i p_i * p_{i+d} > 0, then s_d(t) is concave in t (opens downward).
If sum_i p_i * p_{i+d} < 0, then s_d(t) is convex in t (opens upward).

So the max of these quadratics is not necessarily convex. The golden section search might not work perfectly, but it should still find a good local minimum.

def _anti_periodic_project(h):
    """Project h onto anti-periodic subspace: h[i] + h[i+n//2] = 1"""
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    h_ap = np.zeros(n)
    half = n // 2
    for i in range(half):
        val = (h[i] + (1.0 - h[i + half])) / 2.0
        h_ap[i] = val
        h_ap[i + half] = 1.0 - val
    return h_ap


def _mirror_project(h):

    n = len(h)
    h_m = np.zeros(n)
    for i in range(n // 2):

        h_m[i] = val
        h_m[n - 1 - i] = 1.0 - val

    return h_m
