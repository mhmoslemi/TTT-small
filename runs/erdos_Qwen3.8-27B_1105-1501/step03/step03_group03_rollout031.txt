def _exact_c5(h):

def _scores(h):

def _scores_grad(h):
    """Compute scores and the gradient matrix G."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]

def _project(v, target):

                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _pair_refine(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores, G = _scores_grad(h)
        grad_mag = np.abs(G).sum(axis=0)
        active_idx = np.argsort(grad_mag)[::-1][:min(16, n)]
        for j in active_idx:
            if time.time() > deadline:
                break
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for k in active_idx:
                if k == j:
                    continue
                if h[k] <= 1e-10 or h[k] >= 1.0 - 1e-10:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    a_max = 1e9
                    if dj > 0:
                        a_max = min(a_max, (1.0 - h[j]) / abs(dj))
                    else:
                        a_max = min(a_max, h[j] / abs(dj))
                    if dk > 0:
                        a_max = min(a_max, (1.0 - h[k]) / abs(dk))
                    else:
                        a_max = min(a_max, h[k] / abs(dk))
                    if a_max < 1e-12:
                        continue
                    for t in [0.3, 0.6, 1.0]:
                        a = a_max * t
                        h[j] += dj * a
                        h[k] += dk * a
                        c5_t = _exact_c5(h)
                        if c5_t < best_c5:
                            best_c5 = c5_t
                            improved = True
                            break
                        else:
                            h[j] -= dj * a
                            h[k] -= dk * a
        if not improved:
            break
    return h, best_c5


def _subgradient_kick(h, target, rng, strength):
    n = len(h)
    scores, G = _scores_grad(h)
    s_max = float(np.max(scores))
    active = np.where(scores >= s_max - 0.02 * s_max)[0]
    if len(active) == 0:
        active = np.argmax(scores)
    grad = G[active].mean(axis=0)
    grad = grad - np.mean(grad)
    norm = float(np.linalg.norm(grad))
    if norm > 1e-14:
        direction = -grad / norm + rng.standard_normal(n) * 0.3
    else:
        direction = rng.standard_normal(n)
    direction = direction - np.mean(direction)
    h_new = _project(h + strength * direction, target)
    return h_new


def _fourier_init(n, m, a_scale=0.12):
    h = np.full(n, 0.5)
    i = np.arange(n)
    for j in range(1, m + 1):
        a = a_scale * (0.7 + 0.3 * np.cos(2 * np.pi * j / (m + 1)))
        h += a * np.cos(2 * np.pi * j * i / n + 0.5 * j)
    return _project(h, n / 2.0)


def _block_init(n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    for _ in range(20):
        idx = np.random.permutation(n)
        h[idx] = np.random.uniform(0, 1, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Intensive LP from initial point
    deadline1 = t0 + budget_s * 0.28
    h1, c5_1 = _lp_local_search(h, deadline1, radius_init=0.18, max_iters=20000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP + LP refinement
    if time.time() < t0 + budget_s * 0.38:
        deadline2 = t0 + budget_s * 0.38
        for beta in [30.0, 80.0, 150.0]:
            if time.time() > deadline2:
                break
            h_s = _slsqp(best_h, best_n / 2.0, beta, max_iter=60)
            c5_s = _exact_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            h_lp, c5_lp = _lp_local_search(h_s, min(deadline2, time.time() + 25),
                                           radius_init=0.1, max_iters=5000)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 3: Structured exploration with kicks
    if time.time() < t0 + budget_s * 0.62:
        deadline3 = t0 + budget_s * 0.62
        n_attempts = 0
        while time.time() < deadline3:
            n_attempts += 1
            if best_c5 < 0.3806:
                break
            # Vary kick strength
            strength = 0.02 + 0.04 * (0.5 + 0.5 * np.sin(n_attempts * 0.7))
            h_kick = _subgradient_kick(best_h, best_n / 2.0, rng, strength)
            # Also try pure random kicks every few attempts
            if n_attempts % 4 == 0:
                kick = rng.standard_normal(best_n) * 0.04
                kick -= kick.mean()
                h_kick = _project(best_h + kick, best_n / 2.0)
            # Quick LP optimization
            time_left = deadline3 - time.time()
            if time_left < 3:
                break
            alloc = min(20, time_left / 3.0)
            h_try, c5_try = _lp_local_search(h_kick, time.time() + alloc,
                                             radius_init=0.1, max_iters=4000)
            if c5_try < best_c5 - 1e-11:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 4: Pair refinement
    if time.time() < t0 + budget_s * 0.72:
        deadline4 = t0 + budget_s * 0.72
        h_p, c5_p = _pair_refine(best_h, deadline4, max_passes=100)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
        # Follow up with LP
        h_f, c5_f = _lp_local_search(h_p, min(deadline4, time.time() + 15),
                                      radius_init=0.05, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Try different resolutions
    if time.time() < t0 + budget_s * 0.88:
        deadline5 = t0 + budget_s * 0.88
        for dn in [16, -16, 32, 8, -8, 48]:
            if time.time() > deadline5:
                break
            n_try = best_n + dn
            if n_try < 32 or n_try > 192:
                continue
            h_r = _resample(best_h, n_try)
            time_left = deadline5 - time.time()
            if time_left < 5:
                break
            h_r2, c5_r2 = _lp_local_search(h_r, time.time() + min(25, time_left / 2.0),
                                           radius_init=0.12, max_iters=6000)
            if c5_r2 < best_c5 - 1e-11:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_try

    # Phase 6: Alternate between LP and SLSQP with different betas
    if time.time() < t0 + budget_s * 0.95:
        deadline6 = t0 + budget_s * 0.95
        for beta in [40.0, 100.0, 200.0]:
            if time.time() > deadline6:
                break
            h_s = _slsqp(best_h, best_n / 2.0, beta, max_iter=50)
            h_lp, c5_lp = _lp_local_search(h_s, min(deadline6, time.time() + 20),
                                           radius_init=0.06, max_iters=4000)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 7: Final intensive LP polish
    final_deadline = t0 + budget_s * 0.995
    h_f, c5_f = _lp_local_search(best_h, final_deadline,
                                  radius_init=0.04, max_iters=20000, top_k=25)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(best_h)
    # One more pass without top_k
    if time.time() < final_deadline:
        h_f2, c5_f2 = _lp_local_search(best_h, final_deadline,
                                        radius_init=0.02, max_iters=10000)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            best_n = len(best_h)

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
