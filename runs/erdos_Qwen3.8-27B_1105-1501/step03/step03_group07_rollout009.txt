import numpy as np
import time
import cvxpy as cp

# === Core functions ===

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_shift(h, m):
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        L = n - m
        if L <= 0:
            return g
        g[:L] += (1.0 - h[m:m+L])
        g[m:] -= h[:L]
    else:
        k = -m
        L = n - k
        if L <= 0:
            return g
        g[k:] += (1.0 - h[:L])
        g[:L] -= h[k:n]
    return g

def _project(v, target):
    # Water-filling projection onto {v: 0<=v[i]<=1, sum(v)=target}
    n = len(v)
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    # Bisection for water level
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_step(h, active_indices, n, target):
    """Solve one LP step of the active set method"""
    dx = 2.0 / n
    vals = _values(h)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        t_var >= 0

        shift = r - (n - 1)
        G = _grad_shift(h, shift) * dx
        # Linearized: vals[r](h) + G . (h_var - h) <= t_var
        # => G @ h_var - t_var <= G . h - vals[r]
        rhs = float(np.dot(G, h)) - vals[r]
        constraints.append(G @ h_var - t_var <= rhs)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)

        except:

    return h_var.value, t_var.value

            if c_damped < max_val - 1e-14:
                h = h_damped
                stall = 0
    
    return best_h, best_c5
