def _project_box_sum(v, target):
    # Water-filling projection onto {x: 0<=x[i]<=1, sum(x)=target}
    n = len(v)

            mask = x > 0.0

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x
