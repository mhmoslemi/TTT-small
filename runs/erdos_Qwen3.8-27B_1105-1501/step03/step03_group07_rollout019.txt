    # So the correction to the linear formula is: 2t - 2t² = 2t(1-t)
    c[n-1] += 2.0 * t * (1.0 - t)
    return c

def _pair_optimize(h, j, k, target, dx):
    """Optimize pair (j,k) to minimize C5, maintaining h[j]+h[k] = s"""
    n = len(h)
    s = h[j] + h[k]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-14:
        return False, 0.0
    
    slope, intercept = _pair_coeffs(h, j, k)
    
    # Find t that minimizes max(c(t)) subject to t in [t_lo, t_hi]
    # c(t) = intercept + t*slope (for r≠n-1) and intercept[n-1] + t*slope[n-1] + 2t(1-t) (for r=n-1)
    
    # Since the function is the max of linear functions (plus one concave quadratic),
    # it's convex. Use golden section.
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a0, b0 = t_lo, t_hi
    c1 = b0 - gr * (b0 - a0)
    c2 = a0 + gr * (b0 - a0)
    f1 = float(np.max(_eval_c_at_t(slope, intercept, c1, n)))
    f2 = float(np.max(_eval_c_at_t(slope, intercept, c2, n)))
    
    for _ in range(60):
        if b0 - a0 < 1e-12:
            break
        if f1 < f2:
            b0 = c2
            c2 = c1
            f2 = f1
            c1 = b0 - gr * (b0 - a0)
            f1 = float(np.max(_eval_c_at_t(slope, intercept, c1, n)))
        else:
            a0 = c1
            c1 = c2
            f1 = f2
            c2 = a0 + gr * (b0 - a0)
            f2 = float(np.max(_eval_c_at_t(slope, intercept, c2, n)))
    
    t_opt = (a0 + b0) / 2.0
    dt = t_opt - h[j]
    if abs(dt) < 1e-12:
        return False, 0.0
    
    h[j] = t_opt
    h[k] = s - t_opt
    return True, dt
