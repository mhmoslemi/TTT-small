def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} using water-filling"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-12:
        return v
    # Binary search for threshold
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    """G[r, j] = ∂score(d)/∂h[j], d = r-(n-1)"""
    n = len(h)
    G = np.zeros((2*n-1, n))

            if j + d < n:
                c += 1.0 - h[j+d]
            if j - d >= 0:
                c += -h[j-d]

    return G

    cur_c5 = float(np.max(scores))
    
    for p in range(max_passes):
        improved = False
        for j in range(n):
            cj = G[:, j]
            k_idx = np.array([k for k in range(n) if k != j])
            B = cj[None, :] - G[:, k_idx].T  # (2n-1, n-1)
            
            dmin = np.maximum(-h[j], h[k_idx] - 1.0)
            dmax = np.minimum(1.0 - h[j], h[k_idx])
            
            valid = (dmax - dmin) > 1e-12
            if not np.any(valid):
                continue
            
            Bv = B[:, valid]
            dminv = dmin[valid]
            dmaxv = dmax[valid]
            kv = k_idx[valid]
            
            delta, fvals = solve_1d_batch(scores, Bv, dminv, dmaxv)
            
            idx = int(np.argmin(fvals))
            if fvals[idx] < cur_c5 - tol:
                kb = int(kv[idx])
                db = float(delta[idx])
                if abs(db) > 1e-12:
                    h[j] += db
                    h[kb] -= db
                    cur_c5 = float(fvals[idx])
                    improved = True
                    G = compute_grad_matrix(h)
                    scores = np.correlate(h, 1.0 - h, mode='full')
        
        if not improved:
            break
    
    return h, cur_c5 * (2.0/n)

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)
