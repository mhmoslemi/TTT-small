    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _repair_box_sum(v, target):
    """Project v onto [0,1]^n with sum = target using bisection."""
    n = len(v)
    v = np.array(v, dtype=float)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build gradient matrix G where G[r,i] = dS_r/dh_i."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += (1.0 - h[p])
            G[r, p] -= h[q]
    return G

def _active_direction(h, G, scores, beta):
    """Generate direction from active gradient."""
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w = w / w_sum
    g = G.T @ w
    v = -g
    v = v - np.mean(v)
    nv = float(np.linalg.norm(v))
    if nv < 1e-14:
        return None
    return v / nv

def _topk_direction(h, G, scores, k):

    g = G.T @ w
    v = -g
    v = v - np.mean(v)
    nv = float(np.linalg.norm(v))

        return None
    return v / nv

def _resample(h, n_new):
    """Resample h to new resolution."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_box_sum(h_new, n_new / 2.0)

    n = len(h)
    k = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:k]] = 1.0
    return h_bin

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_box_sum(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    if np.any(np.isnan(v)):
        v = np.nan_to_num(v, nan=0.5)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, v):

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += (1.0 - h[p])
            G[r, p] -= h[q]
    return G

    v = -g

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _repair_box_sum(h_new, n_new / 2.0)
