def _c5_val(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def _max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr))


def _enum_best_binary(n, time_limit):
    t0 = time.time()
    half = n // 2
    best = n + 1
    best_seq = None
    for combo in combinations(range(n), half):
        if time.time() - t0 > time_limit:
            break
        seq = np.zeros(n)
        seq[list(combo)] = 1.0
        mc = _max_corr(seq)
        if mc < best:
            best = mc
            best_seq = seq.copy()
            if best == 1:
                break
    return best_seq, best


def _proj_sum(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        mask = x < 1.0 if diff > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _scores(h):

def _grad_top_scores(h, w):
