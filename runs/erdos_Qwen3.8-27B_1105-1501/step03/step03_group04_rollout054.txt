from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad(h, weights):
    """Compute gradient of weighted sum of scores.
    grad[j] = sum_k weights[k] * ((1-h[j+d_k])*[0<=j+d_k<n] - h[j-d_k]*[0<=j-d_k<n])
    where d_k = k - (n-1)
    """
    n = len(h)
    grad = np.zeros(n)
    # For each j, d_k ranges over values where j+d_k or j-d_k is in [0,n-1]

    #   k in [n-1-j, n-1+j-1], weight is (1-h[j+d_k]) = (1-h[j+k-n+1])
    # Second term: -h[j-d_k] for d_k in [j-n+1, j]  (so j-d_k in [0, n-1])
    #   k in [n-1+j-n+1, n-1+j] = [j, n-1+j], weight is -h[j-d_k] = -h[j-k+n-1]
    
    for j in range(n):
        # First term: k from max(0, n-1-j) to min(2n-2, n-1+j-1)

        k_hi1 = min(2*n - 2, n - 2 + j)
        if k_lo1 <= k_hi1:

            # = j + k_lo1 - n + 1 to j + k_hi1 - n + 1

            w_slice = weights[k_lo1:k_hi1+1]
            h_slice = 1.0 - h[idx_lo:idx_hi+1]
            grad[j] += float(np.dot(w_slice, h_slice))
        
        # Second term: k from max(0, j) to min(2n-2, n-1+j)

            h_slice = h[idx_lo2:idx_hi2+1]

    return grad

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_fast(h, weights):
    n = len(h)
    nm1 = n - 1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        # First term: (1-h[j+d]) for d such that 0 <= j+d < n
        # d ranges from -j to n-1-j
        # k = nm1 + d, so k ranges from nm1-j to nm1+n-1-j = 2n-2-j
        k_lo = nm1 - j
        k_hi = nm1 + (n - 1 - j)  # = 2*n - 2 - j
        k_lo = max(k_lo, 0)
        k_hi = min(k_hi, 2*n - 2)
        if k_lo <= k_hi:
            # h index = j + d = j + (k - nm1) = j + k - nm1
            idx_lo = j + k_lo - nm1  # = max(0, 2j-n+1)大致
            idx_hi = j + k_hi - nm1
            grad[j] += float(np.dot(weights[k_lo:k_hi+1], one_minus_h[idx_lo:idx_hi+1]))
        
        # Second term: -h[j-d] for d such that 0 <= j-d < n
        # d ranges from j-n+1 to j
        # k = nm1 + d, so k ranges from nm1+j-n+1 to nm1+j = j to nm1+j
        k_lo2 = nm1 + (j - n + 1)  # = j
        k_hi2 = nm1 + j
        k_lo2 = max(k_lo2, 0)
        k_hi2 = min(k_hi2, 2*n - 2)
        if k_lo2 <= k_hi2:
            # h index = j - d = j - (k - nm1) = j + nm1 - k
            # As k goes from k_lo2 to k_hi2, h index goes from j+nm1-k_lo2 to j+nm1-k_hi2 (reverse)
            h_idx_lo = j + nm1 - k_hi2  # smallest h index
            h_idx_hi = j + nm1 - k_lo2  # largest h index
            # weights[k_lo2..k_hi2] corresponds to h[h_idx_hi..h_idx_lo] (reversed)
            grad[j] -= float(np.dot(weights[k_lo2:k_hi2+1][::-1], h[h_idx_lo:h_idx_hi+1]))
    return grad

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = (x < 1.0) & (x > 0.0)
            if np.sum(mask) == 0:
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0)
            if np.sum(mask) == 0:
                mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_descent(h, deadline, beta_start=30.0, beta_end=5000.0, max_iters=20000, lr_init=0.08):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        frac = min(it / max(iter(max_iters * 0.1, 100)), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        m = float(np.max(scores))
        exps = np.exp(beta * (scores - m))
        s_exp = float(np.sum(exps))
        if s_exp < 1e-300:
            weights = np.zeros(2*n-1)
            weights[np.argmax(scores)] = 1.0
        else:
            weights = exps / s_exp
        grad = _grad_fast(h, weights)
        grad -= np.mean(grad)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = h + alpha * direction
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.5)
                improved = True
                break
            else:
                alpha *= 0.55
        if not improved:
            stall += 1
            lr *= 0.7
            if lr < 1e-6 or stall > 80:
                break
    return best_h, best_c5

def _binary_flip_search(h, deadline, max_iters=50000):
    """Local search on near-binary solutions by flipping pairs."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    # Fix count
    s = float(np.sum(h))
    if s > target:
        ones = np.where(h == 1.0)[0]
        rng = np.random.default_rng(42)
        to_flip = rng.choice(ones, int(s - target), replace=False)
        h[to_flip] = 0.0
    elif s < target:
        zeros = np.where(h == 0.0)[0]
        rng = np.random.default_rng(42)
        to_flip = rng.choice(zeros, int(target - s), replace=False)
        h[to_flip] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        # Randomly pick a 1 and a 0 to swap
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        a = ones[i]
        b = zeros[j]
        if a == b:
            continue
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
            # Update tracking
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    return best_h, best_c5

def _lp_polish(h, deadline, radius_init=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        # Build G matrix for LP (only needed here, not in gradient)
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                if m > 0:
                    G[r, d:] += (1.0 - h[:m])
                    G[r, :m] -= h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, :m] += (1.0 - h[g:])
                    G[r, g:] -= h[:m]
        
        # LP: min T s.t. G@(h-new) + scores <= T, sum(new)=target, bounds
        # i.e., G@new <= T - scores + G@h
        # Variables: new[0..n-1], T
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores + G @ h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        bounds = [(max(h[j]-radius, 0.0), min(h[j]+radius, 1.0)) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
            else:
                stall += 1
                radius *= 0.85
                if radius < 1e-5:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        
        if float(np.max(np.abs(p - h))) < 1e-12:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        
        # Line search along h -> p
        best_a = 0.0
        best_cv = cur_c5
        for a in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            h_t = _project(h + a * (p - h), target)
            cv = _exact_c5(h_t)
            if cv < best_cv - 1e-13:
                best_cv = cv
                best_a = a
        if best_a > 0:
            # Golden section refinement
            lo_a, hi_a = max(0, best_a * 0.5), min(1.0, best_a * 1.5)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = _exact_c5(_project(h + c1 * (p - h), target))
            f2 = _exact_c5(_project(h + c2 * (p - h), target))
            for _ in range(15):
                if hi_a - lo_a < 1e-10:
                    break
                if f1 < f2:
                    hi_a = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = _exact_c5(_project(h + c1 * (p - h), target))
                else:
                    lo_a = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f2 = _exact_c5(_project(h + c2 * (p - h), target))
            opt_a = (lo_a + hi_a) / 2.0
            opt_cv = _exact_c5(_project(h + opt_a * (p - h), target))
            if opt_cv < best_cv:
                best_a, best_cv = opt_a, opt_cv
            h = _project(h + best_a * (p - h), target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse_init(n):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = np.where(tm == 1, 0.0, 1.0).astype(float)
    return _project(h, n / 2.0)

def _m_sequence_init(n):
    """Generate an m-sequence (maximum length sequence) of length n if n = 2^k - 1."""
    # For n not of form 2^k-1, fall back to Thue-Morse
    k = int(np.log2(n + 1))
    if 2**k - 1 != n:
        return _thue_morse_init(n)
    # Generate m-sequence using LFSR
    # Use a primitive polynomial of degree k
    # For small k, use known taps
    taps = {2: [0, 1], 3: [0, 1], 4: [0, 1, 3], 5: [0, 2, 4], 6: [0, 3, 5],
            7: [0, 1, 6], 8: [0, 1, 4, 7], 9: [0, 4, 8], 10: [0, 3, 7, 9]}
    if k not in taps:
        return _thue_morse_init(n)
    t = taps[k]
    state = 1
    seq = []
    for i in range(n):
        seq.append(state & 1)
        # Compute next bit
        fb = 0
        for bit in t:
            fb ^= (state >> bit) & 1
        state = (state >> 1) | (fb << (k - 1))
    h = np.array(seq, dtype=float)
    # Ensure exactly n/2 ones
    return _project(h, n / 2.0)
