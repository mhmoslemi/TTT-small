```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lse_objective(h, tau):
    vals = _corr_values(h)
    m = np.max(vals)
    if tau < 1e-12:
        return m
    return m + tau * np.log(np.sum(np.exp((vals - m) / tau)))

def _lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    shifted = (vals - m) / tau
    w = np.exp(shifted)
    wsum = np.sum(w)
    w = w / wsum
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            grad[d:] += w[r] * (1.0 - h[:length])
            grad[:length] -= w[r] * h[d:]
        else:
            k = -d
            length = n - k
            grad[k:] += w[r] * (1.0 - h[:length])
            grad[:length] -= w[r] * h[k:]
    grad *= dx
    return grad

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-10:
        return v
    # Bisection for water-filling
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _slsqp_lse(h0, tau, deadline, maxiter=100):
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        return _lse_objective(h, tau)
    
    def grad(h):
        return _lse_gradient(h, tau)
    
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    
    h_opt = _project(result.x, target)
    return h_opt, _exact_c5(h_opt)
```

```python
from scipy.optimize import minimize
import time

_tau_box = [0.1]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lse_objective(h, tau):
    vals = _corr_values(h)
    m = float(np.max(vals))
    if tau < 1e-12:
        return m
    shifted = (vals - m) / tau
    return m + tau * float(np.log(np.sum(np.exp(shifted))))

def _lag_grad(h, d):
    """Gradient of the correlation at shift d (signed) w.r.t. h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        length = n - d
        if length > 0:
            g[d:] += (1.0 - h[:length])
            g[:length] -= h[d:]
    else:
        k = -d
        length = n - k
        if length > 0:
            g[k:] += (1.0 - h[:length])
            g[:length] -= h[k:]
    return g * dx

def _lse_gradient(h, tau):
    n = len(h)
    vals = _corr_values(h)
    m = float(np.max(vals))
    shifted = (vals - m) / tau
    w = np.exp(shifted)
    wsum = float(np.sum(w))
    w = w / wsum
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        d = r - (n - 1)
        grad += w[r] * _lag_grad(h, d)
    return grad

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.clip(v, 0.0, 1.0)
    if abs(float(np.sum(v)) - target) < 1e-12:
        return v.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _obj_fn(h):
    return _lse_objective(h, _tau_box[0])

def _grad_fn(h):
    return _lse_gradient(h, _tau_box[0])

def _slsqp_lse(h0, tau, maxiter=200):
    n = len(h0)
    target = n / 2.0
    _tau_box[0] = tau
    constraints = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(_obj_fn, h0, jac=_grad_fn, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = _project(result.x, target)
        return h_opt, _exact_c5(h_opt)
    except Exception:
```
