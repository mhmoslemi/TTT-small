import cvxpy as cp

def _qp_solve_qp(w, n, target):
    """Solve min_h sum_d w_d * s_d(h) s.t. sum h = target, 0 <= h <= 1"""
    h = cp.Variable(n)
    
    # Build the objective
    # f(h) = sum_d w_d * s_d(h)
    # s_d(h) = sum_i h[i]*(1-h[i+d])
    
    obj = 0
    for d_idx, d in enumerate(range(-(n-1), n)):
        w_d = w[d_idx]
        if w_d < 1e-15:
            continue
        if d >= 0:
            m = n - d
            # s_d = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
            obj += w_d * cp.sum(h[:m] * (1 - h[d:]))
        else:
            g = -d
            m = n - g
            # s_d = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
            obj += w_d * cp.sum(h[g:] * (1 - h[:m]))
    
    constraints = [cp.sum(h) == target, h >= 0, h <= 1]
    problem = cp.Problem(cp.Minimize(obj), constraints)
    problem.solve(solver=cp.ECOS)
    
    if h.value is not None:
        return h.value
    return None
