def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        G[r] = _score_grad(h, d)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

                'fun': lambda x, dd=d: x[m] - _score_tau(_antipodal_h(x[:m], n), dd),
                'jac': lambda x, dd=d: _antipodal_grad(x[:m], n, dd)
            })

        def _cb_a(xk):
            if time.time() > deadline:
                raise RuntimeError("timeout")

        try:
            result = minimize(obj_a, x0, method='SLSQP', jac=obj_jac_a,
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 60, 'ftol': 1e-10, 'disp': False},
                              callback=_cb_a)
        except (RuntimeError, Exception):
            continue
        y_out = np.clip(result.x[:m], 0.0, 1.0)
        h_out = _antipodal_h(y_out, n)
        c5_out = _exact_c5(h_out)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_y = y_out.copy()
    return _antipodal_h(best_y, n), best_c5

def _antipodal_grad(y, n, d):
    """Gradient of score at lag d wrt y, for antipodal h=[y, 1-y_rev]"""
    m = n // 2
    h = _antipodal_h(y, n)
    gh = _score_grad(h, d)
    # dh/dy: h[i]=y[i] for i<m, h[m+i]=1-y[m-1-i]
    gy = np.zeros(m)
    # contribution from h[i] = y[i] for i < m
    gy[:m] += gh[:m]
    # contribution from h[m+j] = 1 - y[m-1-j], so dh[m+j]/dy[m-1-j] = -1
    # i.e., dh[m+j]/dy[k] = -1 if k = m-1-j, i.e., j = m-1-k
    gy += -gh[m:][::-1]
    full_grad = np.zeros(2 * m + 1)
    full_grad[:m] = gy
    full_grad[m] = 1.0
    return full_grad

def _subgrad_descent(h, deadline, max_iters=20000):
    """Smoothed subgradient descent with momentum"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    momentum = np.zeros(n)
    mu = 0.7
    step_base = 0.03
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        beta = 60.0
        w = np.exp(beta * (scores - max_s) / max(1e-10, abs(max_s)))
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= wsum
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-10:
                d = r - (n - 1)
                grad += w[r] * _score_grad(h, d)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        momentum = mu * momentum - grad
        momentum -= np.mean(momentum)
        step = step_base / (1.0 + 0.001 * it)
        h_new = h + step * momentum
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < c5 - 1e-14:
            h = h_new
            c5 = new_c5
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
            step_base *= 0.95
    return best_h, best_c5

def _pair_descent(h, deadline, max_sweeps=20):
    """Pair exchange descent: for each pair (j,k), find optimal transfer"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Random subset of pairs for speed
        pairs = []
        for _ in range(min(200, n * n // 4)):
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j != k:
                pairs.append((j, k))
        # Always include some structured pairs
        for j in range(0, n, 4):
            for k in range(j + 1, min(j + 8, n)):
                pairs.append((j, k))

        for (j, k) in pairs:
            if time.time() > deadline:
                break
            # Try transferring mass from j to k (and vice versa)
            if abs(h[j] - h[k]) < 1e-10:
                continue
            # Direction: decrease h[j], increase h[k] (or opposite)
            # Try a few delta values
            best_delta = 0.0
            best_c5_pair = cur_c5
            for delta_sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[j] = -delta_sign * 1.0
                d[k] = delta_sign * 1.0
                # Line search
                alpha_max = 0.15
                if delta_sign > 0:
                    alpha_max = min(alpha_max, h[j], 1.0 - h[k])
                else:
                    alpha_max = min(alpha_max, h[k], 1.0 - h[j])
                if alpha_max < 1e-6:
                    continue
                for a in np.concatenate([np.logspace(-3, np.log10(alpha_max), 8),
                                         np.linspace(0, alpha_max, 5)]):
                    if a < 1e-8 or time.time() > deadline:
                        continue
                    h_t = h + a * d
                    if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5_pair - 1e-14:
                        best_c5_pair = c5_t
                        best_delta = a * delta_sign
            if best_delta != 0.0:
                h[j] += best_delta * (-1.0 if best_delta > 0 else 1.0) * (1.0 if best_delta > 0 else -1.0)
                # Simpler: just store and reconstruct
                # Redo properly:
                h = np.clip(h, 0.0, 1.0)
                # Actually let me redo this more carefully
                # h[j] should change by -delta_sign * alpha, h[k] by delta_sign * alpha
                # best_delta = alpha * delta_sign means h[j] += -best_delta, h[k] += best_delta
                # Wait no, let me just recompute
                pass
            # This is getting complicated, let me simplify
        # Recompute
        h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 64
        h0 = 0.5 + rng.normal(0, 0.1, n0)
        h0 = np.clip(h0, 0, 1)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0.copy(), target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP multi-pass
    dl1 = t0 + budget_s * 0.15
    h1, c5_1 = _slsqp_multi_pass(h, dl1, passes=3)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP loop
    dl2 = t0 + budget_s * 0.30
    h2, c5_2 = _lp_loop(best_h, dl2, radius_init=0.12, max_iters=5000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient descent
    dl3 = t0 + budget_s * 0.42
    h3, c5_3 = _subgrad_descent(best_h, dl3, max_iters=15000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # SLSQP polish after subgrad
    if time.time() < dl3 + budget_s * 0.05:
        h3b, c5_3b = _slsqp_multi_pass(best_h, dl3 + budget_s * 0.05, passes=2)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Antipodal family optimization
    dl4 = t0 + budget_s * 0.60
    for n_try in [n0, n0 - 8, n0 + 8]:
        if n_try < 24 or n_try > 128 or time.time() > dl4:
            continue
        per_dl = min(dl4, time.time() + 20.0)
        h_ant, c5_ant = _antipodal_optimize(n_try, per_dl, rng)
        if c5_ant < best_c5:
            best_c5 = c5_ant
            best_h = h_ant.copy()
            best_n = n_try

    # Phase 5: Multi-restart with SLSQP
    dl5 = t0 + budget_s * 0.82
    scale = 0.05
    attempts = 0
    while time.time() < dl5:
        attempts += 1
        if best_c5 < 0.3806:
            break
        scale = 0.05 * (0.93 ** min(attempts, 25))
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        per_budget = max(3.0, min(20.0, (dl5 - time.time()) * 0.25))
        per_dl = time.time() + per_budget
        result = _slsqp_solve(h_pert, per_dl, top_k=40, maxiter=80, ftol=1e-10)
        if result is not None:
            h_try, c5_try = result
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.01, scale * 0.8)
            else:
                scale = min(0.12, scale * 1.05)
        else:
            h_lp2, c5_lp2 = _lp_loop(h_pert, time.time() + min(8.0, (dl5 - time.time()) * 0.15),
                                      radius_init=0.08, max_iters=400)
            if c5_lp2 < best_c5 - 1e-12:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()

    # Phase 6: SLSQP refinement with tight active set
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.92
        h6, c5_6 = _slsqp_multi_pass(best_h, dl6, passes=4)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.96
        for dn in [-4, 4, -8, 8, -2, 2, -16, 16]:
            if time.time() > dl7:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 160 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            c5_r_init = _exact_c5(h_r)
            dl_r = min(dl7, time.time() + 15.0)
            result = _slsqp_solve(h_r, dl_r, top_k=35, maxiter=70, ftol=1e-10)
            if result is not None:
                h_try, c5_try = result
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 8: Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.995,
                              radius_init=0.04, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
