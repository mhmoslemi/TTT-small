def _sine_init(n, freq, amp, phase, rng):
    x = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return _project(h, n / 2.0)

def _shifted_init(h, n, shift, rng):
    h2 = np.roll(h, shift)
    h2 = h2 + rng.normal(0, 0.03, n)
    return _project(h2, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline_abs = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)
    
    # Get parent
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(64, 0.5)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        c = _exact_c5(h_cand)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_h = h_cand.copy()
            best_n = n_cand
        elif c <= best_c5 + 1e-10 and n_cand < best_n:
            best_c5 = c
            best_h = h_cand.copy()
            best_n = n_cand
    
    # Phase 1: Multi-beta L-BFGS on parent (40%)
    dl1 = t0 + budget_s * 0.40
    h, c5 = _multibeta(h, best_n, dl1, betas=[3.0, 8.0, 20.0, 50.0, 120.0, 300.0, 800.0, 2000.0], maxiter_base=50)
    try_update(h, best_n)
    
    # Phase 2: Soft descent refinement (15%)
    dl2 = t0 + budget_s * 0.55
    h, c5 = _soft_descent(best_h, dl2, beta_start=30.0, beta_end=3000.0, max_iters=8000)
    try_update(h, best_n)
    
    # Phase 3: Multi-beta again on best (10%)
    dl3 = t0 + budget_s * 0.65
    h, c5 = _multibeta(best_h, best_n, dl3, betas=[50.0, 200.0, 600.0, 1500.0, 4000.0], maxiter_base=40)
    try_update(h, best_n)
    
    # Phase 4: Population exploration (20%)
    dl4 = t0 + budget_s * 0.85
    n_opts = [best_n] + [best_n + d for d in [-8, -4, -2, 2, 4, 8] if 24 <= best_n + d <= 128]
    n_opts = sorted(set(n_opts))[:6]
    
    seeds = []
    # Shifted parent
    for sh in [1, 2, 3, -1, -2, -3]:
        seeds.append(_shifted_init(best_h, best_n, sh, rng))
    # Sine inits at various n
    for n_s in n_opts[:3]:
        for freq in [1, 2, 3]:
            seeds.append(_sine_init(n_s, freq, rng.uniform(0.2, 0.4), 0, rng))
    # Walsh
    seeds.append(_random_walsh(best_n, rng))
    # Random binary
    idx = rng.permutation(best_n)
    hb = np.zeros(best_n)
    hb[idx[:best_n // 2]] = 1.0
    hb += rng.normal(0, 0.05, best_n)
    seeds.append(_project(hb, best_n / 2.0))
    
    n_seeds = len(seeds)
    time_per_seed = (dl4 - time.time()) / max(n_seeds, 1)
    
    for seed_h in seeds:
        if time.time() > dl4:
            break
        if best_c5 < 0.3807:
            break
        ns = len(seed_h)
        t_end = min(time.time() + min(time_per_seed, budget_s * 0.02), dl4)
        # Quick multi-beta
        h_s, c5_s = _multibeta(seed_h, ns, t_end, betas=[10.0, 50.0, 200.0, 600.0], maxiter_base=30)
        try_update(h_s, ns)
    
    # Phase 5: Resample best to other n and optimize (10%)
    dl5 = t0 + budget_s * 0.93
    for dn in [-8, -4, -2, 2, 4, 8]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128 or n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        t_end = min(time.time() + budget_s * 0.012, dl5)
        h_r, c5_r = _multibeta(h_r, n_try, t_end, betas=[20.0, 100.0, 400.0, 1500.0], maxiter_base=30)
        try_update(h_r, n_try)
    
    # Phase 6: Final soft descent + coordinate descent (7%)
    dl6 = deadline_abs
    if time.time() < dl6 - 5:
        h_f, c5_f = _soft_descent(best_h, min(dl6 - 3, time.time() + budget_s * 0.04), 
                                   beta_start=200.0, beta_end=5000.0, max_iters=5000)
        try_update(h_f, best_n)
    
    if time.time() < dl6 - 1:
        h_cd, c5_cd = _coord_descent(best_h, dl6, max_passes=100)
        try_update(h_cd, best_n)
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
