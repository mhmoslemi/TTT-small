Let me think about the smooth gradient more carefully.

The objective is f(h) = max_k Q_k(h) where Q_k(h) = sum_j h[j+k']*(1-h[j]) for appropriate ranges.

The log-sum-exp approximation: F_β(h) = (1/β) * log(sum_k exp(β*Q_k(h)))

As β → ∞, F_β → max_k Q_k.

The gradient: ∇F_β = (1/β) * sum_k exp(β*Q_k) * ∇Q_k / sum_k exp(β*Q_k)
= sum_k w_k * ∇Q_k where w_k = exp(β*(Q_k - max_Q)) / sum_j exp(β*(Q_j - max_Q))

This is a weighted average of the gradients of individual scores, weighted by how close each score is to the maximum.

For β large, this concentrates on the active (near-maximal) scores.

The descent direction is -∇F_β, projected to be orthogonal to the constraint (sum h = n/2) and respecting bounds.

def _pair_optimize(h, i, j, deadline):
    """Optimize single-pair transfer."""
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return 0.0, _exact_c5(h)
    
    c5_cur = _exact_c5(h)
    best_t = 0.0
    best_c5 = c5_cur
    
    for t in np.linspace(t_lo, t_hi, 10):
        if time.time() > deadline:
            break
        ht = h.copy()
        ht[i] += t
        ht[j] -= t
        c5_t = _exact_c5(ht)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_t = t
    
    if best_t == 0.0:
        return 0.0, c5_cur
    
    # Golden section refine
    step = (t_hi - t_lo) / 10.0
    lo = max(t_lo, best_t - step)
    hi = min(t_hi, best_t + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def ev(t):
        ht = h.copy()
        ht[i] += t
        ht[j] -= t
        return _exact_c5(ht)
    
    f1, f2 = ev(c1), ev(c2)
    for _ in range(20):
        if hi - lo < 1e-13 or time.time() > deadline:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = ev(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = ev(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_c5 = ev(opt_t)
    if opt_c5 < best_c5:
        return opt_t, opt_c5
    return best_t, best_c5

def _resample(h, n_new):

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] += -h[:m]
    return grad

def _project_sum(h, target):

        cnt = int(np.sum(mask))

def _smooth_dir(h, beta):
    n = len(h)

        return None
    return -grad / norm

def _slsqp_solve(h, deadline, top_k=20, maxiter=200):

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
