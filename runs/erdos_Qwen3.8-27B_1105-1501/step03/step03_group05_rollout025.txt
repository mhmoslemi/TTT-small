import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {x: 0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Use iterative water-filling
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import cvxpy as cp

def _socp_trust_step(h, deadline, sigma=0.05, max_iters=100, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = sigma * np.sqrt(n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Compute current scores and gradient matrix
        raw = np.correlate(h, 1.0 - h, mode='full')
        F = raw * dx  # current scaled scores
        
        # Build gradient matrix G (2n-1 x n)
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            low = max(0, -d)
            high = min(n, n - d)
            for p in range(low, high):
                q = p + d
                G[r, q] += dx * (1.0 - h[p])
                G[r, p] -= dx * h[q]
        
        # Solve SOCP
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            G @ x + (F - G @ h) <= t,
            x >= 0,
            x <= 1,
            cp.sum(x) == target,
            cp.norm(x - h) <= rho
        ]
        problem = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            problem.solve(solver=cp.CLARABEL)
        except:
            try:
                problem.solve(solver=cp.SCS)
            except:
                break
        
        if x.value is None:
            break
        
        x_val = x.value
        # Project to ensure feasibility
        x_val = _project_box_sum(x_val, target)
        
        # Try different step sizes
        best_step = None
        best_c5_trial = cur_c5
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15]:
            trial = h + factor * (x_val - h)
            trial = _project_box_sum(trial, target)
            c5_trial = _exact_c5(trial)
            if c5_trial < best_c5_trial - 1e-12:
                best_c5_trial = c5_trial
                best_step = trial
        
        if best_step is not None:
            h = best_step
            cur_c5 = best_c5_trial
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            rho = min(rho * 1.3, 0.5 * np.sqrt(n))
        else:
            rho *= 0.6
            if rho < 0.001 * np.sqrt(n):
                break
    
    return best_h, best_c5

S(d) = [terms not involving i or j] 
     + x*(1-h(i-d)) [if i-d valid]
     + h(i+d)*(1-x) [if i valid]
     + (c-x)*(1-h(j-d)) [if j-d valid]
     + h(j+d)*(1-(c-x)) [if j valid]

The x terms:
- x*(1-h(i-d)): linear in x
- h(i+d)*(1-x) = h(i+d) - h(i+d)*x: linear in x
- (c-x)*(1-h(j-d)) = c*(1-h(j-d)) - x*(1-h(j-d)): linear in x
- h(j+d)*(1-c+x) = h(j+d)*(1-c) + h(j+d)*x: linear in x

So S(d) is LINEAR in x for fixed h[i]+h[j]=c! 

Wait, that's only if i and j are different positions and don't "interact" (i.e., i+j ≠ 2p for any p that appears in both terms). Let me check the case where both conditions for i hold:

If i-d is valid AND i is valid, then S(d) contains:
x*(1-h(i-d)) + h(i+d)*(1-x) = x - x*h(i-d) + h(i+d) - x*h(i+d)
= h(i+d) + x*(1 - h(i-d) - h(i+d))

This is still linear in x! Because both terms are x * (constant).

What about the case where i = j+d (i.e., the same term involves both h[i] and h[j])? This happens when j = i-d, i.e., d = i-j. In that case:

The term p=i-d=j gives: h(p+d)*(1-h(p)) = h(j+d)*(1-h(j)) = h[i]*(1-h[j]) = x*(1-(c-x)) = x - x*c + x²

Oh! This IS quadratic! When d = i-j (or d = j-i), the term h[i]*(1-h[j]) = x*(1-c+x) = x - cx + x² appears.

So S(d) for d = i-j contains x². The coefficient of x² is +1, so it's convex (opens upward).

Similarly for d = j-i.

So for most lags, S(d) is linear in x. For at most two lags (d=i-j and d=j-i), S(d) is quadratic (convex) in x.

Therefore, max_d S(d) is the max of:
- Several linear functions of x
- At most two convex quadratic functions of x

The max of linear functions is convex (piecewise linear). The max of a convex function and linear functions is also convex. So max_d S(d) is CONVEX in x!

This is a great result! It means the 1D problem is a convex optimization, and golden section search will find the global optimum exactly.

Actually wait, I need to double-check. The quadratic terms x² - cx + x = x² + (1-c)x have the form x² + bx for some b. The second derivative is 2 > 0, so yes, convex.

And the max of convex functions is convex. So the 1D objective is convex, and golden section (or bisection on the derivative) will find the global minimum.

This confirms that the pairwise GS is doing exact 1D convex optimization at each step. The only question is convergence to a global optimum of the full problem, which is not guaranteed (the full problem is nonconvex), but multiple passes with different orderings help.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(v, target):
    n = len(v)
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - x.sum()
    if r > 1e-12:
        m = x < 1.0
        x[m] += r / m.sum()
    elif r < -1e-12:
        m = x > 0.0
        x[m] += r / m.sum()
    return np.clip(x, 0.0, 1.0)

def _pair_opt(h, i, j, c5):

    c2 = a + gr * (b - a)
    h[i], h[j] = c1, c - c1
    f1 = _exact_c5(h)
    h[i], h[j] = c2, c - c2
    f2 = _exact_c5(h)

def _socp(h, dl, sigma=0.06, rng=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = _exact_c5(h)
    bh, bc5 = h.copy(), c5
    rho = sigma * np.sqrt(n)
    stall = 0
    for _ in range(2000):
        if time.time() > dl or rho < 1e-4 * np.sqrt(n):
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        F = raw * dx
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            lo = max(0, -d)
            hi = min(n, n - d)
            for p_ in range(lo, hi):
                q = p_ + d
                G[r, q] += dx * (1.0 - h[p_])
                G[r, p_] -= dx * h[q]
        x = cp.Variable(n)
        t = cp.Variable()
        cons = [G @ x + (F - G @ h) <= t, x >= 0, x <= 1, cp.sum(x) == n/2.0, cp.norm(x - h) <= rho]
        prob = cp.Problem(cp.Minimize(t), cons)
        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or x.value is None:
            rho *= 0.5
            stall += 1
            if stall > 5:
                break
            continue
        xv = _proj(x.value, n / 2.0)
        improved = False
        for fac in [1.0, 0.7, 0.5, 0.3]:
            trial = _proj(h + fac * (xv - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < c5 - 1e-12:
                h = trial; c5 = tc5
                improved = True
                break
        if improved:
            if c5 < bc5:
                bc5 = c5; bh = h.copy()
            rho = min(rho * 1.2, 0.4 * np.sqrt(n))
            stall = 0
        else:
            rho *= 0.6
            stall += 1
            if stall > 10:
                break
    return bh, bc5
