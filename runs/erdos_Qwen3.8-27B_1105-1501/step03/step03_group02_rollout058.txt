```python
def _exact_c5(h):

    dx = 2.0 / n

def _project(v, target):

    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:
            hi = mid
```

For a pairwise transfer (i gets +delta, j gets -delta):
The score at lag d changes by:
delta * (coefficient of h[i] at lag d) - delta * (coefficient of h[j] at lag d)

This is a 1D convex optimization problem! The optimal delta is where the maximum is minimized, which can be found by bisection or by finding the intersection of the two "active" lines.

This is actually a very powerful approach because:
1. It's exactly feasible (sum preserved, box constraints maintained)
2. Each step is a 1D convex optimization (fast)

```python
def _pairwise_score_change(h, i, j, scores):
    """Compute a_d = change in score_d per unit delta when h[i]+=delta, h[j]-=delta"""
    n = len(h)
    a = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        # Coefficient of h[i] in score at lag d
        ci = _coeff(h, i, d)
        # Coefficient of h[j] in score at lag d
        cj = _coeff(h, j, d)
        a[d + n - 1] = ci - cj
    return a

def _coeff(h, idx, d):
    """Coefficient of h[idx] in the score at lag d"""
    n = len(h)
    c = 0.0
    if d == 0:
        return 1.0 - 2.0 * h[idx]
    elif d > 0:
        # h[idx] appears as h[j+d] if idx >= d and idx < n, j = idx-d
        if d <= idx < n:
            c += (1.0 - h[idx - d])
        # h[idx] appears as h[j] if idx < n-d, j = idx
        if idx < n - d:
            c += -h[idx + d]
    else:
        g = -d
        # h[idx] appears as h[j] if idx < n-g, j = idx
        if idx < n - g:
            c += (1.0 - h[idx + g])
        # h[idx] appears as h[j+g] if idx >= g and idx < n, j = idx-g
        if g <= idx < n:
            c += -h[idx - g]
    return c
```

Then for each pair, the optimal delta minimizes max_d (s_d + delta * a_d). This is the smallest epsilon such that s_d + delta * a_d <= s_max + epsilon for all d. By strong duality, the optimal value is the minimum over delta of the maximum, which equals the maximum over "dual variables" (probability distributions on the lags) of the minimum over delta of the expected linear function.

In practice, for a 1D problem, the optimal delta is at a point where two or more of the lines s_d + delta * a_d are simultaneously at their maximum. We can find this by:
1. For each pair of lags (p, q), find the delta where s_p + delta*a_p = s_q + delta*a_q
2. Check if at that delta, it's indeed the maximum
3. Take the best such delta

Or more simply, since it's 1D, we can just do a golden section search over delta.

```python
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        grad[:m] -= h[d:]
        grad[d:] += 1.0 - h[:m]
    else:
        g = -d
        m = n - g
        grad[:m] += 1.0 - h[g:]
        grad[g:] -= h[:m]
    return grad

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_direction(h, scores, beta):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= ws
    d = np.zeros(n)
    for r in range(len(scores)):
        if w[r] > 1e-13:
            d += w[r] * _score_grad(h, r - (n - 1))
    d -= float(np.mean(d))
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-15:
        return None
    return -d / nrm

def _linesearch(h, d, deadline, iters=45):
    amax = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if pos.any():
        amax = min(amax, float(((1.0 - h)[pos] / d[pos]).min()))
    if neg.any():
        amax = min(amax, float((h[neg] / (-d[neg])).min()))
    amax = max(amax, 1e-12)
    c5_0 = _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, amax
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * d, 0.0, 1.0))
    f2 = _exact_c5(np.clip(h + c2 * d, 0.0, 1.0))
    best_a = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    for _ in range(iters):
        if time.time() > deadline or (hi - lo) < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * d, 0.0, 1.0))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * d, 0.0, 1.0))
        if f1 < best_f:
            best_a, best_f = c1, f1
        if f2 < best_f:
            best_a, best_f = c2, f2
    if best_a < 1e-13 or best_f >= c5_0 - 1e-15:
        return h, c5_0
    target = len(h) / 2.0
    h_new = _project(h + best_a * d, target)
    return h_new, _exact_c5(h_new)

def _grad_descent(h, deadline, beta=50, maxit=200):
    best_h, best_c5 = h.copy(), _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    b = beta
    for _ in range(maxit):
        if time.time() > deadline or stall > 50:
            break
        scores = _all_scores(h)
        d = _smooth_direction(h, scores, b)
        if d is None:
            break
        h_new, c5_new = _linesearch(h, d, deadline)
        if c5_new < cur_c5 - 1e-14:
            h, cur_c5 = h_new, c5_new
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 5 == 0:
                b = min(600.0, b * 1.4)
    return best_h, best_c5

def _coeff_at(h, idx, d):
    n = len(h)
    if d == 0:
        return 1.0 - 2.0 * h[idx]
    elif d > 0:
        c = 0.0
        if d <= idx:
            c += 1.0 - h[idx - d]
        if idx < n - d:
            c -= h[idx + d]
        return c
    else:
        g = -d
        c = 0.0
        if idx < n - g:
            c += 1.0 - h[idx + g]
        if g <= idx:
            c -= h[idx - g]
        return c

def _pair_step(h, i, j, scores):
    n = len(h)
    a = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        a[d + n - 1] = _coeff_at(h, i, d) - _coeff_at(h, j, d)
    dmax = min(1.0 - h[i], h[j])
    dmin = -min(h[i], 1.0 - h[j])
    if dmax < 1e-13 and dmin > -1e-13:
        return 0.0, float(np.max(scores))
    f0 = float(np.max(scores))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = dmin, dmax
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + c1 * a))
    f2 = float(np.max(scores + c2 * a))
    if f1 < f2:
        best_a, best_f = c1, f1
    else:
        best_a, best_f = c2, f2
    if best_f >= f0 - 1e-14:
        return 0.0, f0
    for _ in range(30):
        if (hi - lo) < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + c1 * a))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * a))
        if f1 < best_f:
            best_a, best_f = c1, f1
        if f2 < best_f:
            best_a, best_f = c2, f2
    if best_f >= f0 - 1e-14:
        return 0.0, f0
    h[i] += best_a
    h[j] -= best_a
    return best_a, best_f

def _pairwise_search(h, deadline, n_rounds=150, n_try=60, seed=0):
    n = len(h)
    h = h.copy()
    rng = np.random.RandomState(seed)
    best_h, best_c5 = h.copy(), _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    for rnd in range(n_rounds):
        if time.time() > deadline or stall > 30:
            break
        scores = _all_scores(h)
        improved = False
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta, f_new = _pair_step(h, i, j, scores)
            if abs(delta) > 1e-13 and f_new < cur_c5 - 1e-13:
                cur_c5 = f_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
                scores = _all_scores(h)
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anti_periodic_search(h, deadline, seed=0):
    n = len(h)
    half = n // 2
    rng = np.random.RandomState(seed)
    y0 = np.clip(h[:half], 0.0, 1.0)
    y0 = _project(y0, half / 2.0) * 2.0
    y0 = np.clip(y0, 0.0, 1.0)
    y = y0.copy()
    best_y = y.copy()
    best_c5 = _exact_c5(np.concatenate([y, 1.0 - y]))
    pop_size = 30
    pop = np.zeros((pop_size, half))
    pop[0] = y
    for i in range(1, pop_size):
        pop[i] = np.clip(y + 0.08 * rng.randn(half), 0.0, 1.0)
    sc = np.zeros(pop_size)
    for i in range(pop_size):
        sc[i] = _exact_c5(np.concatenate([pop[i], 1.0 - pop[i]]))
    F, CR = 0.7, 0.6
    max_iter = 400
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        idx = _ % pop_size
        a, b, c = [], []
        while len(c) < 3:
            k = int(rng.randint(pop_size))
            if k != idx and k not in c:
                c.append(k)
        a, b, c = c[0], c[1], c[2]
        v = pop[a] + F * (pop[b] - pop[c])
        trial = pop[idx].copy()
        mask = rng.random(half) < CR
        if not mask.any():
            mask[int(rng.randint(half))] = True
        trial[mask] = np.clip(v[mask], 0.0, 1.0)
        trial = np.clip(trial, 0.0, 1.0)
        h_trial = np.concatenate([trial, 1.0 - trial])
        c5_t = _exact_c5(h_trial)
        if c5_t < sc[idx]:
            pop[idx] = trial
            sc[idx] = c5_t
            if c5_t < best_c5:
                best_c5 = c5_t
                best_y = trial.copy()
    return np.concatenate([best_y, 1.0 - best_y]), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h, best_c5, best_n = h.copy(), _exact_c5(h), n0

    # Phase 1: Anti-periodic global search
    if time.time() - t0 < budget_s * 0.15:
        dl1 = t0 + budget_s * 0.15
        h_ap, c5_ap = _anti_periodic_search(h, dl1, seed=seed)
        if c5_ap < best_c5:
            best_c5, best_h = c5_ap, h_ap.copy()
        # Gradient polish from AP result
        if c5_ap < best_c5 + 0.01:
            h_g, c5_g = _grad_descent(h_ap, dl1, beta=50, maxit=80)
            if c5_g < best_c5:
                best_c5, best_h = c5_g, h_g.copy()

    # Phase 2: Beta sweep gradient descent
    if time.time() - t0 < budget_s * 0.35:
        dl2 = t0 + budget_s * 0.35
        for beta in [20, 40, 70, 120, 200, 400]:
            if time.time() > dl2:
                break
            h_g, c5_g = _grad_descent(best_h, min(dl2, time.time() + 20), beta=beta, maxit=120)
            if c5_g < best_c5:
                best_c5, best_h = c5_g, h_g.copy()

    # Phase 3: Multi-resolution search
    if time.time() - t0 < budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, 12, -12, 16, -16, 20]:
            if time.time() > dl3:
                break
            nt = best_n + dn
            if nt < 40 or nt > 128 or nt % 2 != 0:
                continue
            hr = _resample(best_h, nt)
            hg, cg = _grad_descent(hr, min(dl3, time.time() + 18), beta=60, maxit=100)
            if cg < best_c5 - 1e-12:
                best_c5, best_h, best_n = cg, hg.copy(), nt

    # Phase 4: Pairwise search (main new method)
    if time.time() - t0 < budget_s * 0.78:
        dl4 = t0 + budget_s * 0.78
        h_p, c5_p = _pairwise_search(best_h, dl4, n_rounds=200, n_try=80, seed=seed)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 5: Random restarts
    if time.time() - t0 < budget_s * 0.90:
        dl5 = t0 + budget_s * 0.90
        rng = np.random.RandomState(seed + 1)
        for att in range(40):
            if time.time() > dl5:
                break
            if best_c5 < 0.3807:
                break
            sc = 0.025 + 0.03 * rng.random()
            hp = _project(best_h + rng.normal(0, sc, best_n), best_n / 2.0)
            b = [25, 50, 100, 200][att % 4]
            hg, cg = _grad_descent(hp, min(dl5, time.time() + 8), beta=b, maxit=60)
            if cg < best_c5 - 1e-12:
                best_c5, best_h = cg, hg.copy()

    # Phase 6: Final polish - gradient + pairwise
    if time.time() - t0 < budget_s * 0.99:
        dl6 = t0 + budget_s * 0.99
        for b in [10, 30, 80, 200, 500]:
            if time.time() > dl6:
                break
            h_r, c5_r = _grad_descent(best_h, min(dl6, time.time() + 12), beta=b, maxit=120)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r.copy()
        if time.time() < dl6:
            h_p2, c5_p2 = _pairwise_search(best_h, dl6, n_rounds=100, n_try=100, seed=seed+99)
            if c5_p2 < best_c5:
                best_c5, best_h = c5_p2, h_p2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
```
