def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_onto_simplex(v, target):
    """Project v onto {x: 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = 0.0
    hi = 1.0
    # Actually need wider range
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    # Fix rounding
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_rows(h):
    """Return all correlation values (not scaled by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_of_corr(h):
    """
    Return matrix G of shape (2n-1, n) where G[r, j] = d(corr[r])/d(h[j])
    corr[r] corresponds to lag d = r - (n-1)

    m = 2 * n - 1
    G = np.zeros((m, n))

            #   If j+d < n and j >= d: contributes -h[j-d]  (from second factor when i = j-d)
            #   Wait, j-d >= 0 means j >= d, and j-d < n-d means j < n
            #   So for j in d..n-1: -h[j-d]

            #   So sum_{i=g}^{n-1} h[i] * (1-h[i+d]) = sum_{i=g}^{n-1} h[i] * (1-h[i-g])

def _subgradient_direction(h, top_k=5):
    """
    Find the min-norm negative subgradient of max_r corr[r].
    Returns direction d (mean-zero not required, we'll project).
    """
    n = len(h)
    scores = _corr_rows(h)
    # Find top-k active constraints
    idx = np.argsort(scores)[::-1][:top_k]
    G = _grad_of_corr(h)
    active_G = G[idx, :]  # shape (k, n)
    # Find lambda >= 0, sum lambda = 1, minimizing ||G_active^T @ lambda||^2
    # This is: min lambda^T (G G^T) lambda s.t. sum lambda = 1, lambda >= 0
    # KKT: 2 G G^T lambda + 1 - mu @ active_mask = 0
    # For simplicity, since k is small, use a simple approach
    # If k=1, direction is -G[idx[0], :]
    if len(idx) == 1:
        d = -G[idx[0], :].copy()
    else:
        # Solve small QP: min lambda^T A lambda s.t. sum=lambda=1, lambda>=0
        # A = active_G @ active_G.T
        A = active_G @ active_G.T
        k = len(idx)
        # Try: if unconstrained solution (solve (k-1)x(k-1) system) is feasible, use it
        # Unconstrained: A @ lambda = c * 1 for some scalar c, sum lambda = 1
        # Actually: min lambda^T A lambda s.t. 1^T lambda = 1
        # Lagrangian: 2A lambda + mu*1 = 0, 1^T lambda = 1
        # => lambda = -mu * A^{-1} * 1 / 2
        # 1^T A^{-1} 1 * (-mu/2) = 1 => mu = -2 / (1^T A^{-1} 1)
        try:
            A_inv_1 = np.linalg.solve(A + 1e-12 * np.eye(k), np.ones(k))
            s = float(np.sum(A_inv_1))
            if s > 1e-12:
                lam = A_inv_1 / s
                if np.all(lam >= -1e-10):
                    lam = np.maximum(lam, 0)
                    lam /= lam.sum()
                else:
                    # Some negative, project
                    lam = np.clip(lam, 0, None)
                    lam /= lam.sum()
            else:
                lam = np.ones(k) / k
        except:
            lam = np.ones(k) / k
        d = -(active_G.T @ lam)
    # Make mean-zero
    d -= d.mean()
    # Normalize
    norm = np.sqrt(float(np.dot(d, d)))
    if norm < 1e-15:
        return None
    return d / norm

def _line_search(h, d, c5_cur, target):
    """
    Golden section search along h + alpha*d, maintaining box constraints.
    d should be mean-zero so that the sum constraint is preserved.
    Returns (best_alpha, best_c5).
    """
    n = len(h)
    # Find feasible range for alpha
    alpha_min = 0.0
    alpha_max = float('inf')
    for i in range(n):
        if d[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    alpha_max = min(alpha_max, 2.0)  # Cap step size
    if alpha_max < 1e-10:
        return 0.0, c5_cur
    
    # Coarse grid to find approximate min
    n_grid = 30
    alphas = np.linspace(0, alpha_max, n_grid)
    best_a = 0.0
    best_c5 = c5_cur
    for a in alphas:
        h_t = h + a * d
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    if best_a == 0.0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max / n_grid)
    hi = min(alpha_max, best_a + alpha_max / n_grid)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _lp_direction(h, scores, G, radius):
    """LP-based direction finding (same as before)"""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_onto_simplex(h_new, n_new / 2.0)
