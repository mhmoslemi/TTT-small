def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = (x < 1.0) & (x > 1e-12)
            if not np.any(mask):
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0 - 1e-12)
            if not np.any(mask):
                mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d+m] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:g+m] -= h[:m]
    return G

def _feasible_interval(h, d):
    """Find [alpha_min, alpha_max] such that h + alpha*d stays in [0,1]."""
    n = len(h)
    a_min = -1e10
    a_max = 1e10
    d_eps = 1e-14
    for i in range(n):
        di = d[i]
        hi = h[i]
        if di > d_eps:
            a_min = max(a_min, -hi / di)
            a_max = min(a_max, (1.0 - hi) / di)
        elif di < -d_eps:
            a_min = max(a_min, (1.0 - hi) / di)
            a_max = min(a_max, -hi / di)
    a_min += 1e-13
    a_max -= 1e-13
    return a_min, a_max

    n = len(h)
    a_min, a_max = _feasible_interval(h, d)
    if a_max - a_min < 1e-12:
        return h, _exact_c5(h)
    
    # Grid search
    best_alpha = 0.0
    best_c5 = _exact_c5(h)
    alphas = np.linspace(a_min, a_max, n_grid)
    for a in alphas:
        h_try = h + a * d
        # Check feasibility
        if h_try.min() < -1e-9 or h_try.max() > 1.0 + 1e-9:
            continue
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a

def _active_direction(h, band_frac=0.10, n_top=8):
    """Compute descent direction from active scores."""
    n = len(h)
    scores = _scores(h)
    M = float(np.max(scores))
    if M < 1e-10:
        return None
    # Select active scores
    threshold = M - band_frac * M
    active_idx = np.where(scores >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.where(scores == M)[0]
    # Weight by proximity to max
    weights = scores[active_idx] - (M - band_frac * M)
    weights = np.maximum(weights, 1e-6)
    weights = weights / np.sum(weights)
    
    G = _grad_matrix(h)
    G_active = G[active_idx, :]
    direction = -(weights @ G_active)
    direction -= np.mean(direction)
    norm = np.sqrt(np.sum(direction**2))
    if norm < 1e-14:
        return None
    return direction / norm

def _resample(h, n_new):

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):

        else:

    return x


def _grad_matrix(h):

    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d + m] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:g + m] -= h[:m]
    return G


def _feasible_interval(h, d):

def _random_dir(n, rng):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = np.sqrt(np.sum(d * d))
    if norm < 1e-14:
        d = rng.standard_normal(n)
        d -= np.mean(d)
        norm = np.sqrt(np.sum(d * d))

def _active_dir(h, band_frac=0.10):
    n = len(h)
    scores = _scores(h)
    M = float(np.max(scores))
    if M < 1e-10:
        return None
    threshold = M * (1.0 - band_frac)
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.where(scores == M)[0]
    weights = scores[active] - threshold
    weights = np.maximum(weights, 1e-8)
    wsum = float(np.sum(weights))
    if wsum < 1e-12:
        weights = np.ones(len(active))
    else:
        weights /= wsum
    G = _grad_matrix(h)
    direction = -(weights @ G[active, :])
    direction -= np.mean(direction)
    norm = np.sqrt(np.sum(direction * direction))
    if norm < 1e-14:
        return None
    return direction / norm


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_obj(x, beta):
    scores = np.correlate(x, 1.0 - x, mode='full')
    s_max = float(np.max(scores))
    shifted = beta * (scores - s_max)
    shifted = np.clip(shifted, -500, 500)
    return s_max + np.log(np.sum(np.exp(shifted))) / beta


def _slsqp_step(h, target, beta, max_iter=60):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def _sf(x):
        return float(np.sum(x)) - target

    def _sj(x):
        return np.ones(n)

    def _so(x):
        return _smooth_obj(x, beta)

    try:
        res = minimize(_so, h, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(res.x, target)
    except Exception:
        pass
    return h.copy()


def _lp_direction(h, scores, G, radius):
    n = len(h)
    nf = n + 1
    c = np.zeros(nf)
    c[-1] = 1.0
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, nf))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, nf))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            d = res.x[:n]
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None


def _lp_step(h, radius=0.1):

                d /= norm_d
            else:
                d = _random_dir(n, rng)
        else:
            # LP step occasionally
            h_try, c5_try = _lp_step(h, 0.05)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_h = h.copy()
                best_c5 = c5_try
                no_improve = 0
                continue
            else:
                d = _random_dir(n, rng)

        h_new, c5_new = _line_search(h, d, n_grid=14, n_golden=10)

        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            # Simulated annealing: accept worse with probability
            delta = c5_new - best_c5
            if T_sa > T_min and delta < 0.01:
                if np.random.random() < np.exp(-delta / T_sa):
                    h = h_new
            no_improve += 1

        T_sa *= 0.9995
        if T_sa < T_min:
            T_sa = T_min

        # Periodic escape
        if no_improve > 150:
            d_esc = _random_dir(n, rng)
            h_esc = h + 0.03 * d_esc
            h_esc = _project(h_esc, n / 2.0)
            c5_esc = _exact_c5(h_esc)
            if c5_esc < best_c5 + 0.005:
                h = h_esc
            no_improve = 0
            T_sa = max(T_sa, 0.003)

    # Phase 4: Resolution transfer (15% budget)
    deadline4 = t0 + budget_s * 0.85
    for dn in [8, -8, 16, -16, 4, -4, 32, -32, 12, -12]:
        if time.time() > deadline4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        # Quick optimize at new resolution
        for _ in range(8):
            if time.time() > deadline4:
                break
            d_r = _active_dir(h_r, band_frac=0.10)
            if d_r is None:
                d_r = _random_dir(n_try, rng)
            h_r, c5_r2 = _line_search(h_r, d_r, n_grid=12, n_golden=8)
            c5_r = min(c5_r, c5_r2)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            h = h_r.copy()
            n = n_try

    # Phase 5: Final polish (10% budget)
    deadline5 = hard_deadline
    n = best_n
    h = best_h.copy()
    cur_c5 = best_c5

    # Tight active gradient search
    while time.time() < deadline5:
        d = _active_dir(h, band_frac=0.04)
        if d is not None:
            h_try, c5_try = _line_search(h, d, n_grid=10, n_golden=8)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                continue

        # Random
        d = _random_dir(n, rng)
        h_try, c5_try = _line_search(h, d, n_grid=10, n_golden=6)
        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try

        # Pair
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) >= 2:
            i, j = rng.choice(interior, size=2, replace=False)
            d = np.zeros(n)
            d[i] = 0.7
            d[j] = -0.7
            d -= np.mean(d)
            h_try, c5_try = _line_search(h, d, n_grid=8, n_golden=6)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            best_n = n

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
