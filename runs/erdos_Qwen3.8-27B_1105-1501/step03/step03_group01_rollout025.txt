    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

            hi = mid

        else:

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx

    n = len(h0)
    target = n / 2.0
    t_start = time.time()
    call_count = [0]
    
    def obj(x):
        call_count[0] += 1
        if time.time() - t_start > time_limit:
            raise TimeoutError()
        v, g = soft_obj_grad(x, tau)
        return v, g
    
    def cons_eq(x):
        return np.array([np.sum(x) - target])

        return np.array([np.ones(n)])
    
    bounds = [(0.0, 1.0)] * n

            return h_new
    except (TimeoutError, Exception):

    n = len(h0)

    cur_c5 = exact_c5(h)

    for tau in tau_seq:

            break
        tau_stall = 0
        while time.time() < deadline - 5:
            if tau_stall > 80:
                break
            val, grad = soft_obj_grad(h, tau)
            momentum = beta * momentum + (1 - beta) * grad

                lr = min(lr * 1.02, 0.1)

                lr *= 0.85
                if tau_stall > 40:

        best_c5_try = cur_c5
        for i in range(15):
            a = alpha_max * (i + 1) / 15.0
            c5_try = exact_c5(h + a * p)
            if c5_try < best_c5_try - 1e-12:
                best_c5_try = c5_try
                best_a = a
        if best_a == 0.0:
            stall += 1
            radius *= 0.7
            continue
        # Fine line search (golden section)
        lo_a = max(0.0, best_a * 0.7)
        hi_a = min(alpha_max, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = exact_c5(h + c1 * p)
        f2 = exact_c5(h + c2 * p)
        for _ in range(15):
            if hi_a - lo_a < 1e-12:
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1
                f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = exact_c5(h + c1 * p)
            else:
                lo_a = c1
                c1 = c2
                f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = exact_c5(h + c2 * p)
        opt_a = (lo_a + hi_a) / 2.0
        opt_c5 = exact_c5(h + opt_a * p)
        if opt_c5 < best_c5_try:
            best_a, best_c5_try = opt_a, opt_c5
        new_c5 = best_c5_try
        if new_c5 < cur_c5 - 1e-11:
            h = project(h + best_a * p, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def pair_polish(h0, deadline, max_passes=50):
    "Pair-move local search."
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(12345)
    for _pass in range(max_passes):
        if time.time() > deadline - 1:
            break
        improved = False
        # Try adjacent pairs and some random pairs
        pairs = [(j, j+1) for j in range(n-1) if 0 < h[j] < 1 and 0 < h[j+1] < 1]
        # Add some random pairs
        for _ in range(min(200, n * n // 4)):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j and 0 < h[i] < 1 and 0 < h[j] < 1:
                pairs.append((i, j))
        for (i, j) in pairs:
            if time.time() > deadline - 1:
                break
            for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                if di > 0:
                    am = (1.0 - h[i]) / di
                else:
                    am = h[i] / (-di)
                if dj > 0:
                    am = min(am, (1.0 - h[j]) / dj)
                else:
                    am = min(am, h[j] / (-dj))
                if am < 1e-10:
                    continue
                # Try a few fractions
                best_frac = 0
                best_c5_p = best_c5
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    h_t = h.copy()
                    h_t[i] += di * am * frac
                    h_t[j] += dj * am * frac
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5_p - 1e-12:
                        best_c5_p = c5_t
                        best_frac = frac
                if best_frac > 0:
                    h[i] += di * am * best_frac
                    h[j] += dj * am * best_frac
                    best_c5 = best_c5_p
                    best_h = h.copy()
                    improved = True
    return best_h, best_c5
