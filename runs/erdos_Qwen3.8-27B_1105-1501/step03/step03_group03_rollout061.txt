import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            # s = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # ds/dh[j] = (1-h[j-d]) for d<=j<=n-1, minus h[j+d] for j<=n-1-d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            # s = sum_{i=0}^{m-1} h[i]*(1-h[i+g])
            # ds/dh[j] = (1-h[j+g]) for j<=m-1, minus h[j-g] for g<=j<=n-1
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] -= h[:m]
    return G

def _feasible_interval(h, d):
    # d is zero-sum direction
    # find [a_min, a_max] such that h + a*d is in [0,1]^n
    a_min = 0.0
    a_max = 0.0
    for j in range(len(h)):
        if d[j] > 1e-15:
            a_min = max(a_min, -h[j] / d[j])
            a_max = min(a_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-15:
            a_min = max(a_min, (1.0 - h[j]) / d[j])
            a_max = min(a_max, -h[j] / d[j])
    return a_min, a_max

def _line_search_exact(h, d, n_samples=40, golden_iters=20):
    # Line search along h + alpha*d, d is zero-sum
    a_min, a_max = _feasible_interval(h, d)
    if a_max - a_min < 1e-14:
        return h.copy(), _exact_c5(h)
    
    # Sample
    best_a = 0.0
    best_c5 = _exact_c5(h)
    alphas = np.linspace(a_min, a_max, n_samples)
    for a in alphas:
        h_t = h + a * d
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    # Golden section refine around best
    if best_a > a_min and best_a < a_max:
        lo = max(a_min, best_a - (a_max-a_min)*0.15)
        hi = min(a_max, best_a + (a_max-a_min)*0.15)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr*(hi-lo)
        c2 = lo + gr*(hi-lo)
        f1 = _exact_c5(h + c1*d)
        f2 = _exact_c5(h + c2*d)
        for _ in range(golden_iters):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr*(hi-lo)
                f1 = _exact_c5(h + c1*d)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr*(hi-lo)
                f2 = _exact_c5(h + c2*d)
        opt_a = (lo+hi)/2.0
        opt_c5 = _exact_c5(h + opt_a*d)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_a = opt_a
    
    return h + best_a * d, best_c5

def _lp_direction(h, scores, G, t_current):
    # Solve: min t
    # s.t. G @ d + scores <= t
    #      -h <= d <= 1-h
    #      sum(d) = 0
    n = len(h)
    # Variables: d_0, ..., d_{n-1}, t => n+1 variables
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: G @ d + scores <= t => G @ d - t <= -scores
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h_i <= d_i <= 1-h_i, t unbounded below (but will be positive)
    bounds = [(max(0, -h[i]), min(1, 1-h[i])) for i in range(n)]
    # Wait, d_i can be negative (go below 0) or positive (go above 1)
    # But h_i + d_i must be in [0,1], so -h_i <= d_i <= 1-h_i
    bounds = [(-h[i], 1-h[i]) for i in range(n)]
    bounds.append((None, None))  # t is free
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1], res.success
    return None, None, False

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] -= h[:m]
    return G


def _feasible_interval(h, d):
    a_min = 0.0
    a_max = 0.0
    for j in range(len(h)):
        if d[j] > 1e-15:
            a_min = max(a_min, -h[j] / d[j])
            a_max = min(a_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-15:
            a_min = max(a_min, (1.0 - h[j]) / d[j])
            a_max = min(a_max, -h[j] / d[j])
    return a_min, a_max


def _line_coefficients(h, d):
    n = len(h)
    c = _scores(h)
    G = _grad_matrix(h)
    b = G @ d
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            a[k] = -np.dot(d[:m], d[lag:lag + m])
        else:
            g = -lag
            m = n - g
            a[k] = -np.dot(d[g:], d[:m])
    return a, b, c


def _c5_from_coeffs(a, b, c, alpha, n):
    vals = a * alpha * alpha + b * alpha + c
    return float(np.max(vals) * (2.0 / n))


def _line_search_fast(h, d, n_samples=200, golden_iters=30):
    n = len(h)
    a_min, a_max = _feasible_interval(h, d)
    if a_max - a_min < 1e-14:
        return h.copy(), _exact_c5(h)

    a, b, c = _line_coefficients(h, d)
    best_a = 0.0
    best_c5 = _exact_c5(h)

    alphas = np.linspace(a_min, a_max, n_samples)
    for al in alphas:
        ct = _c5_from_coeffs(a, b, c, al, n)
        if ct < best_c5:
            best_c5 = ct
            best_a = al

    # Also check vertices of individual quadratics
    for k in range(len(a)):
        if abs(a[k]) > 1e-14:
            av = -b[k] / (2.0 * a[k])
            if a_min < av < a_max:
                ct = _c5_from_coeffs(a, b, c, av, n)
                if ct < best_c5:
                    best_c5 = ct
                    best_a = av

    # Golden section around best
    if best_a > a_min + 1e-14 and best_a < a_max - 1e-14:
        spread = (a_max - a_min) * 0.12
        lo = max(a_min, best_a - spread)
        hi = min(a_max, best_a + spread)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _c5_from_coeffs(a, b, c, c1, n)
        f2 = _c5_from_coeffs(a, b, c, c2, n)
        for _ in range(golden_iters):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _c5_from_coeffs(a, b, c, c1, n)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _c5_from_coeffs(a, b, c, c2, n)
        opt_a = (lo + hi) / 2.0
        opt_c5 = _c5_from_coeffs(a, b, c, opt_a, n)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_a = opt_a

    h_new = h + best_a * d
    # Verify with exact computation
    actual_c5 = _exact_c5(h_new)
    if actual_c5 < best_c5 - 1e-12:
        best_c5 = actual_c5
        # Re-find best alpha with exact
        best_a2 = best_a
        alphas2 = np.linspace(max(0, best_a - 0.01), min(best_a + 0.01, a_max), 20)
        for al in alphas2:
            h_t = h + al * d
            ct = _exact_c5(h_t)
            if ct < best_c5:
                best_c5 = ct
                best_a2 = al
        h_new = h + best_a2 * d
        best_c5 = _exact_c5(h_new)

    return h_new, best_c5


def _lp_solve(h, scores, G):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        d = res.x[:n].copy()
        d -= np.mean(d)  # ensure zero sum (numerical)
        return d
    return None


def _slp_loop(h, deadline, max_stall=30):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    while time.time() < deadline:
        scores = _scores(h)
        G = _grad_matrix(h)
        d = _lp_solve(h, scores, G)
        if d is None or np.max(np.abs(d)) < 1e-14:
            stall += 1
            if stall > max_stall:
                break
            continue

        h_new, c5_new = _line_search_fast(h, d)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

    return best_h, best_c5


def _pair_exchange_all(h, deadline, max_passes=20):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    a, b, c = _line_coefficients(h, np.zeros(n))  # placeholder

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = _scores(h)

        # Focus on pairs that affect high scores
        top_scores = np.argsort(-scores)[:40]

            if time.time() > deadline:

            if h[p] <= 1e-9 or h[p] >= 1.0 - 1e-9:

            for q in range(p + 1, n):
                if h[q] <= 1e-9 or h[q] >= 1.0 - 1e-9:
                    continue

                gp = G[top_scores[:10], p]

def _random_direction_search(h, deadline, rng, n_trials=500):

            if s_mid > target:

            else:
                hi = mid

def _resample(h, n_new):

    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    # Project
    lo, hi = -1.0, 2.0
    for _ in range(60):

        if s_mid > target:

        else:
            hi = mid

    lo, hi = float(np.min(h)) - 2.0, float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.97

    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(64, 0.5)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0.copy(), target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLP with LP direction (main optimization)
    deadline1 = t0 + budget_s * 0.40
    h_s, c5_s = _slp_loop(h, deadline1, max_stall=50)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 2: Pair exchanges
    deadline2 = t0 + budget_s * 0.55
    h_p, c5_p = _pair_exchange_all(best_h, deadline2, max_passes=30)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 3: More SLP after pair exchanges
    deadline3 = t0 + budget_s * 0.62
    h_s2, c5_s2 = _slp_loop(best_h, deadline3, max_stall=40)
    if c5_s2 < best_c5:
        best_c5 = c5_s2
        best_h = h_s2.copy()

    # Phase 4: Random direction search + restarts
    deadline4 = t0 + budget_s * 0.75
    for attempt in range(30):
        if time.time() > deadline4:
            break
        if best_c5 < 0.3805:
            break
        sigma = rng.uniform(0.005, 0.03)
        h_try, c5_try = _perturb_and_optimize(best_h, rng, sigma,
                                              min(deadline4, time.time() + budget_s * 0.02))
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Random direction search on best
    deadline5 = t0 + budget_s * 0.80
    h_r, c5_r = _random_direction_search(best_h, deadline5, rng, n_trials=300)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()

    # Phase 6: Final SLP + pairs
    deadline6 = t0 + budget_s * 0.86
    h_s3, c5_s3 = _slp_loop(best_h, deadline6, max_stall=30)
    if c5_s3 < best_c5:
        best_c5 = c5_s3
        best_h = h_s3.copy()

    deadline6b = t0 + budget_s * 0.90
    h_p2, c5_p2 = _pair_exchange_all(best_h, deadline6b, max_passes=15)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()

    # Phase 7: Resolution sweep
    deadline7 = t0 + budget_s * 0.95
    for dn in [1, -1, 2, -2, 3, -3, 8, -8]:
        if time.time() > deadline7:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        h_opt, c5_opt = _slp_loop(h_r, min(deadline7, time.time() + budget_s * 0.015),
                                   max_stall=20)
        if c5_opt < best_c5 - 1e-11:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
        elif abs(c5_opt - best_c5) < 1e-12 and n_try < best_n:
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 8: Final polish
    deadline8 = hard_deadline
    h_f, c5_f = _slp_loop(best_h, deadline8, max_stall=20)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation and cleanup
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        # Projection slightly changed things, use projected value
        best_c5 = final_c5

    return best_h, best_c5, best_n
