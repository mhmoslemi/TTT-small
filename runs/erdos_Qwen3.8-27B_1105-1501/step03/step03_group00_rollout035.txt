    # H is list of Hessian matrices or we compute on the fly

    # Quadratic prediction: s_k(α) = score_k + α * (G[k] @ p) + α²/2 * (p^T H_k p)

    # H_k for lag d has -1 at (i, i+d) and (i+d, i)
    # p^T H_k p = -2 * Σ_{i=0}^{n-1-d} p[i]*p[i+d] for d > 0
    #            = 0 for d = 0

        quad[n-1-d] = -2.0 * pdot * dx

    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + c1 * Gp + 0.5 * c1 * c1 * quad_terms))
    f2 = float(np.max(scores + c2 * Gp + 0.5 * c2 * c2 * quad_terms))
    for _ in range(25):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + c1 * Gp + 0.5 * c1 * c1 * quad_terms))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * Gp + 0.5 * c2 * c2 * quad_terms))
    return (lo + hi) / 2.0

def _actual_line_search(h, p, alpha_max, n_evals=35):
    """Golden section line search on actual C5."""
    if alpha_max < 1e-14:
        return 0.0
    lo = 0.0
    hi = alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(n_evals):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    return (lo + hi) / 2.0

def _box_alpha_max(h, p):
    alpha_max = 0.5
    for j in range(len(h)):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    return max(alpha_max, 1e-14)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _exp_gradient_direction(h, scores, G, beta):
    """Compute gradient direction with exponential weights on scores."""
    max_s = np.max(scores)
    w = np.exp(beta * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _quad_optimize(h, deadline, top_k_list=None, radius_init=0.12, max_iters=5000, fine=False):
    if top_k_list is None:
        top_k_list = [None, 15, 8, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        
        # Get LP direction
        tk = top_k_list[ti % n_tk]
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        
        # Also compute gradient direction as backup
        beta = 50.0 if not fine else 100.0
        p_grad = _exp_gradient_direction(h, scores, G, beta)
        gn = float(np.linalg.norm(p_grad))
        if gn > 1e-14:
            p_grad /= gn
            # Scale to be within radius
            p_grad *= min(radius, 0.1)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = p_grad
            if float(np.max(np.abs(p))) < 1e-14:
                stall += 1
                radius *= 0.7
                ti += 1
                if radius < 1e-7:
                    break
                continue
            t_val = None
        
        # Compute quadratic terms for predictive step
        alpha_max = _box_alpha_max(h, p)
        Gp = G @ p
        qt = _quad_terms(n, p, 2.0 / n)
        
        # Predictive alpha
        alpha_pred = _predictive_alpha(h, p, scores, Gp, qt, alpha_max)
        # Cap predictive alpha
        alpha_pred = min(alpha_pred, alpha_max * 0.95)
        
        # Try predicted alpha
        h_try = h + alpha_pred * p
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        # Also try with actual line search (cheaper version)
        if c5_try >= cur_c5 - 1e-13:
            alpha_actual = _actual_line_search(h, p, alpha_max, n_evals=20)
            h_try2 = h + alpha_actual * p
            h_try2 = _project(h_try2, target)
            c5_try2 = _exact_c5(h_try2)
            if c5_try2 < c5_try:
                h_try = h_try2
                c5_try = c5_try2
        
        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            ti += 1
            if stall % 10 == 0:
                radius *= 0.7
            if stall > 20 and radius < 0.02:
                # Try gradient direction
                p2 = _exp_gradient_direction(h, scores, G, 80.0)
                gn2 = float(np.linalg.norm(p2))
                if gn2 > 1e-14:
                    p2 = p2 / gn2 * min(radius * 2, 0.05)
                    am2 = _box_alpha_max(h, p2)
                    a2 = _actual_line_search(h, p2, am2, n_evals=15)
                    h2 = _project(h + a2 * p2, target)
                    c5_2 = _exact_c5(h2)
                    if c5_2 < cur_c5 - 1e-13:
                        h = h2
                        cur_c5 = c5_2
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.1, 0.2)
                    else:
                        radius *= 0.7
    return best_h, best_c5

def _smooth_phase(h, deadline, beta_start=20.0, beta_end=300.0, n_steps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.02
        found = False
        for _ in range(30):
            h_new = _project(h + alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                found = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not found and step > n_steps // 2:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if time.time() > deadline:
                    break
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _structured_starts(n, rng):
    """Generate diverse structured starting points."""
    candidates = []
    
    # Block patterns: 1,1,0,0 repeated with variations
    for block_size in [2, 3, 4, 5, 6, 7, 8]:
        pattern = []
        half = block_size // 2
        for _ in range(n // block_size + 1):
            pattern.extend([1.0] * half + [0.0] * (block_size - half))
        h = np.array(pattern[:n])
        h = _project(h, n / 2.0)
        candidates.append(h)
    
    # Random with block structure
    for _ in range(5):
        n_blocks = rng.integers(4, n // 4 + 1)
        block_starts = np.sort(rng.choice(np.arange(3, n - 3), size=n_blocks, replace=False))
        h = np.zeros(n)
        for bs in block_starts:
            bw = rng.integers(2, min(8, n - bs))
            h[bs:bs + bw] = 1.0
        h = _project(h, n / 2.0)
        candidates.append(h)
    
    # Alternating with random gaps
    for _ in range(3):
        h = np.zeros(n)
        i = 0
        while i < n:
            gap = rng.integers(1, 4)
            h[i] = 1.0
            i += gap
        h = _project(h, n / 2.0)
        candidates.append(h)
    
    return candidates

def _multi_res_search(best_h, best_n, deadline, rng, best_c5):
    """Try different resolutions and structured starts."""
    n = best_n
    best_local_h = best_h.copy()
    best_local_c5 = best_c5
    
    # Try nearby resolutions
    res_list = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, -1, 1, -3, 3]:
        nn = n + dn
        if 24 <= nn <= 120:
            res_list.append(nn)
    # Also try halving
    if n >= 32:
        res_list.append(n // 2)
    
    for nn in res_list:
        if time.time() > deadline:
            break
        h_r = _resample(best_h, nn)
        sub_dl = min(deadline, time.time() + 15.0)
        h_r, c5_r = _quad_optimize(h_r, sub_dl, top_k_list=[10, 5, None], radius_init=0.15, max_iters=800)
        if c5_r < best_local_c5 - 1e-11:
            best_local_c5 = c5_r
            best_local_h = h_r.copy()
    
    # Try structured starts at original resolution
    starts = _structured_starts(n, rng)
    for h_s in starts:
        if time.time() > deadline:
            break
        c5_s = _exact_c5(h_s)
        if c5_s < best_local_c5 + 0.05:
            sub_dl = min(deadline, time.time() + 10.0)
            h_s, c5_s = _quad_optimize(h_s, sub_dl, top_k_list=[10, 5, None], radius_init=0.2, max_iters=500)
            if c5_s < best_local_c5 - 1e-11:
                best_local_c5 = c5_s
                best_local_h = h_s.copy()
    
    # Perturbation restarts from best
    scale = 0.03
    for _ in range(200):
        if time.time() > deadline:
            break
        h_p = best_local_h + rng.normal(0, scale, len(best_local_h))
        h_p = _project(h_p, len(best_local_h) / 2.0)
        sub_dl = min(deadline, time.time() + 8.0)
        h_p, c5_p = _quad_optimize(h_p, sub_dl, top_k_list=[8, 5, None], radius_init=0.08, max_iters=300)
        if c5_p < best_local_c5 - 1e-11:
            best_local_c5 = c5_p
            best_local_h = h_p.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.1)
    
    return best_local_h, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(64, 0.5)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Main quadratic optimization (40% budget)
    dl1 = t0 + budget_s * 0.40
    h1, c51 = _quad_optimize(h, dl1, top_k_list=[None, 12, 6, 4], radius_init=0.15, max_iters=3000)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()
        best_n = len(h1)
    
    # Phase 2: Smooth gradient refinement (10% budget)
    if time.time() < t0 + budget_s * 0.52:
        dl2 = t0 + budget_s * 0.52
        h2, c52 = _smooth_phase(best_h, dl2, beta_start=30.0, beta_end=200.0, n_steps=150)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
            best_n = len(h2)
    
    # Phase 3: Multi-resolution and diverse starts (25% budget)
    if time.time() < t0 + budget_s * 0.60:
        dl3 = t0 + budget_s * 0.78
        h3, c53 = _multi_res_search(best_h, best_n, dl3, rng, best_c5)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()
            best_n = len(h3)
    
    # Phase 4: Fine LP polish (10% budget)
    if time.time() < t0 + budget_s * 0.82:
        dl4 = t0 + budget_s * 0.90
        h4, c54 = _quad_optimize(best_h, dl4, top_k_list=[4, 3, 5, None], radius_init=0.01, max_iters=5000, fine=True)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()
            best_n = len(h4)
    
    # Phase 5: Pair polish (5% budget)
    if time.time() < t0 + budget_s * 0.92:
        dl5 = t0 + budget_s * 0.95
        h5, c55 = _pair_polish(best_h, dl5, max_passes=6)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5.copy()
            best_n = len(h5)
    
    # Final: one more quick quad optimization
    if time.time() < t0 + budget_s * 0.97:
        h6, c56 = _quad_optimize(best_h, t0 + budget_s * 0.995, top_k_list=[3, 4, 5], radius_init=0.005, max_iters=2000, fine=True)
        if c56 < best_c5:
            best_c5 = c56
            best_h = h6.copy()
            best_n = len(h6)
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
