        g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return g / norm

def _gen_fourier_dir(h, rng, k_min=1, k_max=5):
    """Generate a direction from random Fourier modes."""
    n = len(h)
    P = np.zeros(n)
    nr = min(k_max, n // 2 - 1)
    for k in range(k_min, k_min + nr):
        amp = rng.standard_normal()
        phase = rng.uniform(0, 2 * np.pi)
        x = (np.arange(n) + 0.5) / n * 2.0
        P += amp * np.cos(2 * np.pi * k * (x / 2.0) + phase)
    P -= np.mean(P)
    norm = np.linalg.norm(P)
    if norm < 1e-14:
        return None
    return P / norm

def _gen_pair_dir(h, rng):
    """Generate a direction from a random pair exchange."""
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    P = np.zeros(n)
    P[i] = 1.0
    P[j] = -1.0
    return P

def _gen_block_dir(h, rng, block_size_range=(3, 12)):
    """Generate a direction from block transfers."""
    n = len(h)
    bs = rng.integers(block_size_range[0], block_size_range[1] + 1)
    if bs >= n:
        return None
    i = rng.integers(0, n - bs + 1)
    j = rng.integers(0, n - bs + 1)
    if abs(i - j) < bs:
        return None
    P = np.zeros(n)
    P[i:i+bs] = 1.0 / bs
    P[j:j+bs] = -1.0 / bs
    return P

def _gen_random_dir(h, rng, smoothness=0.3):
    """Generate a smooth random mean-zero direction."""
    n = len(h)
    P = rng.standard_normal(n)
    # Smooth it
    if smoothness > 0:
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / smoothness) ** 2)
        kernel /= np.sum(kernel)
        P = np.convolve(P, kernel, mode='same')
    P -= np.mean(P)
    norm = np.linalg.norm(P)
    if norm < 1e-14:
        return None
    return P / norm

def _optimize_at_n(h, deadline, rng, max_iters=2000):
    """Optimize h at fixed resolution n using line search."""
    n = len(h)
    target = n / 2.0
    h = h.astype(float).copy()
    h = _project(h, target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    dir_strategies = [
        lambda: _gen_subgrad_dir(h, rng, tau_top=3, noise_scale=0.05),
        lambda: _gen_subgrad_dir(h, rng, tau_top=5, noise_scale=0.1),
        lambda: _gen_subgrad_dir(h, rng, tau_top=10, noise_scale=0.0),
        lambda: _gen_fourier_dir(h, rng),
        lambda: _gen_random_dir(h, rng, smoothness=0.5),
        lambda: _gen_block_dir(h, rng),
        lambda: _gen_pair_dir(h, rng),
    ]
    
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 80:
            break
        # Cycle through strategies
        strat = dir_strategies[it % len(dir_strategies)]
        P = strat()
        if P is None:
            stall += 1
            continue
        
        t_opt, new_c5_est = _line_search(h, P, cur_c5)
        if t_opt != 0.0:

            if actual_c5 < cur_c5 - 1e-12:

                cur_c5 = actual_c5
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
