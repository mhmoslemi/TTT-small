def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = max(1, int(np.sum(mask)))

        x[mask[:cnt]] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    """Gradient of score at lag d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] -= h[d:d+m]
            grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:] -= h[:m]
    return grad

def _quadratic_coeffs(h, d, scores_current):

    n = len(h)
    M = 2 * n - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    
    for r in range(M):

        if lag == 0:
            A[r] = float(np.sum(h * (1.0 - h)))
            B[r] = float(np.sum(d * (1.0 - 2.0 * h)))

            B[r] = float(np.sum(d[:m] * (1.0 - h[g:g+m]) - h[:m] * d[g:g+m]))
            C[r] = -float(np.sum(d[:m] * d[g:g+m]))
    
    return A, B, C

def _feaspd_interval(h, d):

def _solve_1d(h, d, deadline):
    """Solve the exact 1D minimax problem along direction d."""
    n = len(h)
    M = 2 * n - 1
    
    A, B, C = _quadratic_coeffs(h, d, _exact_scores(h))
    t_lo, t_hi = _feaspd_interval(h, d)
    
    if t_hi - t_lo < 1e-14:
        return None, float('inf')
    
    # Collect candidate t values
    candidates = set()
    candidates.add(t_lo)
    candidates.add(t_hi)
    candidates.add(0.0)
    
    # Stationary points of each quadratic
    for r in range(M):
        if abs(C[r]) > 1e-15:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                candidates.add(max(t_lo, min(t_hi, t0)))
    
    # Pairwise crossing points
    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            
            if abs(a) < 1e-14:
                # Linear: b*t + c = 0
                if abs(b) > 1e-14:
                    t0 = -c / b
                    if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                        candidates.add(max(t_lo, min(t_hi, t0)))
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-12:
                    disc = max(0.0, disc)
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b - sqrt_disc) / (2.0 * a)
                    t2 = (-b + sqrt_disc) / (2.0 * a)
                    for t0 in [t1, t2]:
                        if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                            candidates.add(max(t_lo, min(t_hi, t0)))
    
    if time.time() > deadline:
        return None, float('inf')
    
    # Evaluate max of quadratics at each candidate
    best_t = 0.0
    best_val = float(np.max(_exact_scores(h)))
    
    # Use current scores as initial best
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        val = float(np.max(A + B * t + C * t * t))
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t == 0.0 and best_val >= float(np.max(_exact_scores(h))):
        return None, float('inf')
    
    # Verify with exact evaluation
    h_trial = h + best_t * d
    h_trial = np.clip(h_trial, 0.0, 1.0)
    # Check feasibility
    n = len(h_trial)
    target = n / 2.0
    s = float(np.sum(h_trial))
    if abs(s - target) > 1e-6:
        h_trial = _project(h_trial, target)
    
    exact_val = float(np.max(_exact_scores(h_trial)))
    
    if exact_val < float(np.max(_exact_scores(h))) - 1e-12:
        return best_t, exact_val
    
    return None, float('inf')

def _smooth_max_direction(h, scores, top_k=20, beta=5.0):
    """Generate a direction using soft-max weighting of top scores."""
    n = len(h)
    max_s = float(np.max(scores))
    # Soft max weights
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= wsum
    
    # Compute weighted gradient
    G = np.zeros(n)
    for r in range(len(scores)):
        lag = r - (n - 1)
        G += w[r] * _score_grad(h, lag)
    
    # Make zero-sum
    G -= np.mean(G)
    norm = float(np.max(np.abs(G)))
    if norm < 1e-15:
        return None
    return G / norm

def _lp_direction(h, scores, n_active=20):
    """LP to find direction minimizing max first-order score increase."""
    n = len(h)
    M = 2 * n - 1
    
    # Get top active scores
    top_idx = np.argsort(scores)[::-1][:n_active]
    
    # Build constraint matrix
    # Variables: d[0..n-1], nu
    # Minimize nu
    # Subject to: grad_r · d + nu <= 0 for r in active
    #              sum(d) = 0
    #              -1 <= d_i <= 1
    
    A_ub = np.zeros((len(top_idx), n + 1))
    for k, r in enumerate(top_idx):
        lag = r - (n - 1)
        A_ub[k, :n] = _score_grad(h, lag)
        A_ub[k, n] = 1.0
    b_ub = np.zeros(len(top_idx))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            d -= np.mean(d)  # ensure zero sum
            norm = float(np.max(np.abs(d)))
            if norm < 1e-12:
                return None
            return d / norm
    except Exception:
        pass
    return None

    scores = np.correlate(h, 1.0 - h, mode='full')
    
    A = scores.copy()  # A_r = current score

    C = np.zeros(M)

        if lag == 0:
            B[r] = float(np.sum(d * (1.0 - 2.0 * h)))
            C[r] = -float(np.sum(d * d))
        elif lag > 0:

            m = n - lag
            B[r] = float(np.sum(d[lag:lag + m] * (1.0 - h[:m]) - h[lag:lag + m] * d[:m]))
            C[r] = -float(np.sum(d[lag:lag + m] * d[:m]))
        else:
            g = -lag
            m = n - g
            B[r] = float(np.sum(d[:m] * (1.0 - h[g:g + m]) - h[:m] * d[g:g + m]))
            C[r] = -float(np.sum(d[:m] * d[g:g + m]))
    return A, B, C

def _calibrate_offset(h, n):
    """Determine if np.correlate convention matches our lag formula or is reversed."""
    d_test = np.zeros(n)
    d_test[0] = 1.0
    d_test[-1] = -1.0
    d_test -= np.mean(d_test)
    eps = 0.005
    h_test = h + eps * d_test
    h_test = np.clip(h_test, 0.0, 1.0)
    scores_direct = _exact_scores(h_test)
    A, B, C = _quadratic_coeffs(h, d_test)
    scores_pred = A + B * eps + C * eps * eps
    diff_fwd = float(np.max(np.abs(scores_direct - scores_pred)))
    scores_pred_rev = A[::-1] + B[::-1] * eps + C[::-1] * eps * eps
    diff_rev = float(np.max(np.abs(scores_direct - scores_pred_rev)))
    if diff_fwd <= diff_rev:
        return 'forward'
    else:
        return 'reversed'

def _solve_1d(h, d, deadline, orient='forward'):
    n = len(h)
    M = 2 * n - 1
    A, B, C = _quadratic_coeffs(h, d)
    if orient == 'reversed':
        A = A[::-1].copy()
        B = B[::-1].copy()
        C = C[::-1].copy()

    # Feasible interval
    t_lo = -np.inf
    t_hi = np.inf
    pos = d > 1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    neg = d < -1e-15
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.max(-h[neg] / d[neg])))

    if t_hi - t_lo < 1e-14:
        return None, float('inf')
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)

    cur_raw = float(np.max(A))

    candidates = [t_lo, t_hi, 0.0]

    # Stationary points
    for r in range(M):
        if abs(C[r]) > 1e-14:
            t0 = -B[r] / (2.0 * C[r])
            if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                candidates.append(max(t_lo, min(t_hi, t0)))

    # Pairwise crossings
    for i in range(M):
        for j in range(i + 1, M):
            a = C[i] - C[j]
            b = B[i] - B[j]
            c = A[i] - A[j]
            if abs(a) < 1e-13:
                if abs(b) > 1e-13:
                    t0 = -c / b
                    if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                        candidates.append(max(t_lo, min(t_hi, t0)))
            else:
                disc = b * b - 4.0 * a * c
                if disc >= -1e-11:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    for t0 in [(-b - sq) / (2.0 * a), (-b + sq) / (2.0 * a)]:
                        if t_lo - 1e-10 <= t0 <= t_hi + 1e-10:
                            candidates.append(max(t_lo, min(t_hi, t0)))

    if time.time() > deadline:
        return None, float('inf')

    # Evaluate at candidates (vectorized)
    cands = np.array(candidates)
    # Deduplicate
    if len(cands) > 1:
        cands = np.sort(cands)
        keep = [0]
        for i in range(1, len(cands)):
            if cands[i] - cands[keep[-1]] > 1e-10:
                keep.append(i)
        cands = cands[keep]

    best_t = 0.0
    best_val = cur_raw
    for t in cands:
        val = float(np.max(A + B * t + C * t * t))
        if val < best_val:
            best_val = val
            best_t = t

    h_trial = h + best_t * d
    h_trial = np.clip(h_trial, 0.0, 1.0)
    target = n / 2.0
    if abs(float(np.sum(h_trial)) - target) > 1e-6:
        h_trial = _project(h_trial, target)
    exact_val = float(np.max(_exact_scores(h_trial)))
    if exact_val < cur_raw - 1e-12:
        return best_t, exact_val
    return None, float('inf')

def _smooth_max_direction(h, scores, top_k=20, beta=6.0):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    G = np.zeros(n)
    for r in range(len(scores)):
        if w[r] > 1e-6:
            lag = r - (n - 1)
            G += w[r] * _score_grad(h, lag)
    G -= np.mean(G)
    norm = float(np.max(np.abs(G)))
    if norm < 1e-14:
        return None
    return G / norm

def _lp_direction(h, scores, n_active=20):
    n = len(h)
    M = 2 * n - 1
    top_idx = np.argsort(scores)[::-1][:n_active]
    A_ub = np.zeros((len(top_idx), n + 1))
    for k, r in enumerate(top_idx):
        lag = int(r) - (n - 1)
        A_ub[k, :n] = _score_grad(h, lag)
        A_ub[k, n] = -1.0
    b_ub = np.zeros(len(top_idx))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n].copy()
            d -= np.mean(d)
            norm = float(np.max(np.abs(d)))
            if norm < 1e-12:
                return None
            return d / norm
    except Exception:
        pass
    return None

def _local_descent(h, deadline, rng, max_iters=300, orient='forward'):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break

        scores = _exact_scores(best_h)
        cur_raw = float(np.max(scores))
        improved = False

        dirs = []
        # Smooth max gradient (both signs)
        d_sm = _smooth_max_direction(best_h, scores, top_k=15, beta=8.0)
        if d_sm is not None:
            dirs.append(d_sm)
            dirs.append(-d_sm)
        # LP direction
        d_lp = _lp_direction(best_h, scores, n_active=15)
        if d_lp is not None:
            dirs.append(d_lp)
        # Pair directions
        for _ in range(8):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                dirs.append(d)
        # Random directions
        for _ in range(4):
            d_r = rng.normal(0, 1, n)
            d_r -= np.mean(d_r)
            norm = float(np.max(np.abs(d_r)))
            if norm > 1e-10:
                dirs.append(d_r / norm)

        for d in dirs:
            if time.time() > deadline:
                break
            t_star, val = _solve_1d(best_h, d, deadline, orient)
            if t_star is not None:
                h_new = best_h + t_star * d
                h_new = np.clip(h_new, 0.0, 1.0)
                if abs(float(np.sum(h_new)) - target) > 1e-6:
                    h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(64, 0.5)

    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    orient = _calibrate_offset(h, n0)

    # Phase 1: Initial local descent
    deadline1 = t0 + budget_s * 0.25
    h1, c51 = _local_descent(h, deadline1, rng, max_iters=200, orient=orient)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Multi-restart
    deadline2 = t0 + budget_s * 0.75
    scale = 0.07
    attempts = 0
    while time.time() < deadline2:
        if best_c5 <= 0.38080:
            break
        attempts += 1
        scale = 0.07 * (0.93 ** min(attempts, 30))
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        per_dl = time.time() + max(3.0, min(18.0, (deadline2 - time.time()) * 0.3))
        h_try, c5_try = _local_descent(h_pert, per_dl, rng, max_iters=150, orient=orient)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 3: Try nearby resolutions
    if time.time() < t0 + budget_s * 0.85:
        dl3 = t0 + budget_s * 0.93
        for dn in [4, -4, 8, -8, 2, -2]:
            if time.time() > dl3:
                break
            n_try = best_n + dn
            if n_try < 24 or n_try > 128 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            orient_r = _calibrate_offset(h_r, n_try)
            h_r, c5_r = _local_descent(h_r, min(dl3, time.time() + 12.0), rng,
                                       max_iters=80, orient=orient_r)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
                orient = orient_r

    # Phase 4: Final polish
    if time.time() < t0 + budget_s * 0.96:
        h_f, c5_f = _local_descent(best_h, t0 + budget_s * 0.99, rng,
                                   max_iters=150, orient=orient)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    if final_c5 > best_c5 + 1e-6:
        pre_h = np.clip(best_h, 0.0, 1.0)
        if abs(float(np.sum(pre_h)) - best_n / 2.0) < 1e-4:
            final_c5 = _exact_c5(pre_h)
            best_h = pre_h

    return best_h, final_c5, best_n
