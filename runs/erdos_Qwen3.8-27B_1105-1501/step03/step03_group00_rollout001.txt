    beta = 0.9  # momentum coefficient
    tau = tau_init
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

        n_steps = 500

        if time.time() > deadline:
            break
        
        scores = _scores(h)
        mx = np.max(scores)
        # Softmax weights
        w = np.exp((scores - mx) / tau)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        # Gradient
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        
        # Nesterov update
        m = beta * m + grad
        # Adaptive step size
        step_size = 0.02 * (1.0 / (1.0 + step * 0.01))

def _active_set_lp(h, deadline, top_k=10, radius=0.1, max_iters=2000):

    n = len(h)
    target = n / 2.0

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = _scores(h)
        # Get top-k active lags
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        
        G = _grad_matrix(h)
        G_k = G[idx]
        
        # LP: minimize t subject to G_k @ p <= t - scores_k, sum(p) = 0, |p| <= radius
        n_vars = n + 1  # p[0..n-1] and t
        c = np.zeros(n_vars)
        c[-1] = 1.0  # minimize t
        
        A_ub = np.zeros((top_k, n_vars))
        A_ub[:, :n] = G_k
        A_ub[:, -1] = -1.0
        b_ub = -scores_k
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo_b = max(-radius, -h[j])
            hi_b = min(radius, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-13:
                    alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=0.4)
                    if new_c5 < cur_c5 - 1e-12:
                        h = _project(h + alpha * p, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                        radius *= 0.8
                else:
                    stall += 1
                    top_k = min(top_k + 2, 2*n-1)
            else:
                stall += 1
                radius *= 0.8
        except Exception:
            stall += 1
            radius *= 0.8
        
        if radius < 1e-6:
            break
        if stall > 5 and it % 10 == 0:
            radius = max(radius, 0.05)

    return best_h, best_c5

    cur_c5 = best_c5

        if time.time() > deadline:

        frac = it / max(1, n_iters)
        cur_t = t_init * (t_final / t_init) ** frac

        # Choose two disjoint blocks
        pos1 = rng.integers(0, n - block_len + 1)

        b2 = slice(pos2, pos2 + block_len)
        
        # Random delta

        h_try = h.copy()

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, n_harmonics=3, rng=None):
    """Generate a sine-based initial construction."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = (np.arange(n) + 0.5) / n * 2.0
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.25)
        freq = rng.integers(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return _project(h, n / 2.0)

def _block_init(n, n_blocks=4, rng=None):
    """Generate a block-based initial construction."""
    if rng is None:
        rng = np.random.default_rng(0)
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        val = rng.uniform(0.0, 1.0)
        h[b * block_size:(b + 1) * block_size] = val
    if n_blocks > 1:
        h[-(n - n_blocks * block_size):] = 0.5
    return _project(h, n / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    """Polish by trying all pairs of coordinates."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                # Try transferring from j to k and vice versa
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Ternary search for optimal transfer
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5
