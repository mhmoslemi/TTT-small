def _project(v, target):

    if target >= n:
        return np.ones(n)

    if best_i == 0:
        return 0.0, best_c5
    if best_i == n_samp - 1:
        return best_a, best_c5
    
    # Golden section refinement
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

def _pnorm_anneal(h, deadline, p_start=12, p_end=500, n_steps=200):

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

            break
        if stall > 40:
            break
        
        frac = step / max(1, n_steps - 1)
        # Exponential ramp from p_start to p_end
        p = p_start * (p_end / p_start) ** frac
        
        # Actually use a smoother ramp
        p = p_start + (p_end - p_start) * frac ** 2
        
        scores = _scores(h)
        max_s = float(np.max(scores))
        eps = 1e-12
        
        # Compute p-norm weights
        s_shifted = scores + eps
        # To avoid overflow, use log-space
        log_s = np.log(s_shifted)
        log_max = np.max(log_s)
        w = np.exp(p * (log_s - log_max))
        sw = float(np.sum(w))
        if sw < 1e-300:

            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        # Gradient
        G = _grad_matrix(h)
        grad = G.T @ w

        grad -= np.mean(grad)
        
        # Check step size

def _equalize_active(h, deadline, threshold=0.005, max_iters=500):

    n = len(h)

    best_h = h.copy()

            break

        max_s = float(np.max(scores))

            break

        G_act = G[active]

        d = -np.mean(G_act, axis=0)
        d -= np.mean(d)  # preserve sum

def _lp_direction(h, scores, G, radius, top_k=None):
    """Find LP trust region direction"""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _simulate_anneal_pair(h, deadline, T0=0.002, T_end=1e-6, n_iters=5000):
    """Simulated annealing on pair exchanges"""
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        T = T0 * (T_end / T0) ** (it / max(1, n_iters - 1))
        
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Pick amount
        # Choose from a distribution that's scaled
        max_move = min(h[i], 1.0 - h[j]) if rng.random() < 0.5 else min(h[j], 1.0 - h[i])
        if max_move < 1e-8:
            continue
        
        eps = rng.uniform(0.001, max_move)
        
        if h[i] >= h[j]:
            h[i] -= eps
            h[j] += eps
        else:
            h[i] += eps
            h[j] -= eps
        
        # Check bounds
        if h[i] < -1e-12 or h[i] > 1 + 1e-12 or h[j] < -1e-12 or h[j] > 1 + 1e-12:
            # Revert
            if h[i] >= h[j] - 2*eps:
                h[i] += eps
                h[j] -= eps
            else:
                h[i] -= eps
                h[j] += eps
            continue
        
        c5_new = _exact_c5(h)
        delta = c5_new - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Revert
            if h[i] >= h[j] - 2*eps:
                h[i] += eps
                h[j] -= eps
            else:
                h[i] -= eps
                h[j] += eps
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):
    """Targeted pair move polish on interior variables"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Only consider interior variables
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) < 2:
        return best_h, best_c5
    
    idx_list = list(interior)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for ii in range(len(idx_list)):
            if time.time() > deadline:
                break
            j = idx_list[ii]
            for kk in range(ii + 1, len(idx_list)):
                k = idx_list[kk]
                # Try moving mass from j to k and vice versa
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    
                    # Ternary search
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def _create_structured_start(n, pattern_type, rng):
    """Create structured starting vectors"""
    if pattern_type == 0:
        # Uniform
        h = np.full(n, 0.5)
    elif pattern_type == 1:
        # Two blocks
        h = np.zeros(n)
        h[:n//2] = 1.0
    elif pattern_type == 2:
        # Alternating blocks of size 2
        h = np.zeros(n)
        for i in range(0, n, 4):
            h[i:i+2] = 1.0
    elif pattern_type == 3:
        # Sinusoidal
        x = np.linspace(0, 2*np.pi, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(x)
    elif pattern_type == 4:
        # Random binary
        h = (rng.random(n) < 0.5).astype(float)
    elif pattern_type == 5:
        # Block of size n/4 in center
        h = np.zeros(n)
        start = n//4
        h[start:3*n//4] = 1.0
    else:
        h = np.full(n, 0.5)
    
    return _project(h, n / 2.0)

        resolutions = [32, 40, 48, 56, 64, 72, 80]
    
    n = len(h)

    for n_try in resolutions:
