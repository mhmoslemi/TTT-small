from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.where(np.isfinite(v), v, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        k = n - 1 + d
        if d >= 0:
            m = n - d
            G[k, :m] = (1.0 - h[d:])
            G[k, d:n] = G[k, d:n] - h[:m]
        else:
            g = -d
            m = n - g
            G[k, g:n] = (1.0 - h[:m])
            G[k, :m] = G[k, :m] - h[g:]
    return G * dx

def lp_direction(h, scores, G, radius, active_tolerance=0.001):
    """Find LP descent direction using active lags."""
    n = len(h)
    n_lags = len(scores)
    max_score = np.max(scores)
    # Use lags within tolerance of max
    active = np.where(scores >= max_score - active_tolerance)[0]
    if len(active) < 3:
        active = np.argsort(scores)[-5:]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -np.zeros(len(active))  # G[active] @ d - t <= 0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, d, c5_cur):
    """Find best step size along direction d."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Grid search
    n_grid = 20
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.empty(n_grid)
    for i, a in enumerate(alphas):
        vals[i] = exact_c5(h + a * d)
    
    best_i = np.argmin(vals)
    best_alpha = alphas[best_i]
    best_val = vals[best_i]
    
    # Refine with golden section
    if 0 < best_i < n_grid - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = exact_c5(h + c1 * d)
        f2 = exact_c5(h + c2 * d)
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = exact_c5(h + c1 * d)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = exact_c5(h + c2 * d)
        opt_a = (lo + hi) / 2.0
        opt_v = exact_c5(h + opt_a * d)
        if opt_v < best_val:
            best_alpha = opt_a
            best_val = opt_v
    
    return best_alpha, best_val

def lp_optimize(h, deadline, radius_init=0.1, max_iters=5000, tol=0.001):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = all_scores(h)
        G = grad_matrix(h)
        d = lp_direction(h, scores, G, radius, active_tolerance=tol)
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        
        alpha, new_c5 = line_search(h, d, cur_c5)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.85
            if stall > 20 and radius < 0.01:
                radius = 0.05
    
    return best_h, best_c5

Actually, let me reconsider the problem structure. The key observation is:

score_k = sum_i h[i] * (1-h[i+k']) * dx

For the "zero lag" (k = n-1 in the full correlation):
score_{n-1} = sum_i h[i] * (1-h[i]) * dx = (sum_i h[i] - sum_i h[i]²) * dx = (n/2 - sum h²) * dx

To minimize this, we want to maximize sum h². Since sum h = n/2 and h ∈ [0,1], the maximum of sum h² is achieved when as many h[i] are at extreme values (0 or 1) as possible.

If h is binary (all 0s and 1s with n/2 of each), sum h² = n/2 and score_{n-1} = 0.
If h is all 1/2, sum h² = n/4 and score_{n-1} = (n/2 - n/4) * 2/n = 1/4 = 0.25.

So the zero lag score ranges from 0 (binary) to 0.25 (uniform). The non-zero lags are what matter for the C₅ value of ~0.38.

For a binary h with n/2 ones, the non-zero lag scores are:
score_d = (n/2 - R(d)) * dx = 1 - R(d) * 2/n

where R(d) = sum_i h[i]*h[i+d] is the autocorrelation.

To minimize max_d score_d, we want to maximize min_d R(d).

For n=64, dx = 2/64 = 1/32. We need max_d (1 - R(d)/32) < 0.38, i.e., R(d) > 32*(1-0.38) = 32*0.62 = 19.84 for all d.

The average R(d) is (n/2)²/(2n-1) = 32²/127 ≈ 8.09. So we need min R(d) > 19.84, but the average is only 8.09. This is impossible for a binary sequence!

So binary sequences can't achieve C₅ < 0.38. We need non-binary values.

For non-binary h, the zero lag score is (n/2 - sum h²) * dx. If we want this < 0.38:
(n/2 - sum h²) * 2/n < 0.38
1 - 2*sum(h²)/n < 0.38
sum(h²) > n*0.62/2 = 0.31n

For n=64: sum(h²) > 19.84

def _exact_c5(h):

def _project(v, target):

    v = np.where(np.isfinite(v), v, 0.5)

    lo = float(v.min()) - 1.0

        if s > target:

        else:

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

            G[k, :m] = (1.0 - h[d:])

            m = n - g
            G[k, g:n] = (1.0 - h[:m])
            G[k, :m] = G[k, :m] - h[g:]
    return G * dx
