import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    n = len(h)
    if d == 0:
        score = float(np.dot(h, 1.0 - h))
        grad = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0, np.zeros(n)
        score = float(np.dot(h[d:d+m], 1.0 - h[:m]))
        grad = np.zeros(n)
        grad[:m] += -h[d:d+m]
        grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0, np.zeros(n)
        score = float(np.dot(h[:m], 1.0 - h[g:g+m]))
        grad = np.zeros(n)
        grad[:m] += (1.0 - h[g:g+m])
        grad[g:] += -h[:m]
    return score, grad

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lp_direction(h, scores, radius, top_k=5):
    n = len(h)
    # Get top-k score indices
    top_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in top_idx]
    
    # Build LP: minimize z s.t. grad_i · d - z <= -score_i, sum(d)=0, |d|<=radius
    # Variables: [d_0, ..., d_{n-1}, z]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    for idx in range(top_k):
        lag = active_lags[idx]
        _, grad = _score_grad(h, lag)
        A_ub[idx, :n] = grad
        A_ub[idx, n] = -1.0
        b_ub[idx] = -scores[top_idx[idx]]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo > hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _golden_search(h, direction, deadline, max_iter=35):
    n = len(h)
    # Max step

def _smooth_grad_direction(h, beta=60.0):

    scores = _all_scores(h)
    max_s = float(np.max(scores))
    a = beta * (scores - max_s)

    w = np.exp(a)
    w_sum = float(np.sum(w))

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, d):
    n = len(h)
    if d == 0:
        score = float(np.dot(h, 1.0 - h))
        grad = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0, np.zeros(n)
        score = float(np.dot(h[d:d + m], 1.0 - h[:m]))
        grad = np.zeros(n)
        grad[:m] += -h[d:d + m]
        grad[d:] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0, np.zeros(n)
        score = float(np.dot(h[:m], 1.0 - h[g:g + m]))
        grad = np.zeros(n)
        grad[:m] += (1.0 - h[g:g + m])
        grad[g:] += -h[:m]
    return score, grad

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lp_direction(h, scores, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    active_lags = [int(k - (n - 1)) for k in top_idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    for idx in range(top_k):
        lag = active_lags[idx]
        _, grad = _score_grad(h, lag)
        A_ub[idx, :n] = grad
        A_ub[idx, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo > hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _golden_search(h, direction, deadline):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 1e-10)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * direction, 0.0, 1.0))
    f2 = _exact_c5(np.clip(h + c2 * direction, 0.0, 1.0))
    for _ in range(35):
        if time.time() > deadline:
            break
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * direction, 0.0, 1.0))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * direction, 0.0, 1.0))
    opt = (lo + hi) / 2.0
    return np.clip(h + opt * direction, 0.0, 1.0)

def _smooth_grad_direction(h, beta):
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    a = beta * (scores - max_s)
    a = a - np.max(a)
    w = np.exp(a)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return None
    w /= w_sum
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        _, g = _score_grad(h, d)
        grad += w[r] * g
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-15:
        return None
    return -grad / norm

def _antipodal_init(n, rng):

def _resample(h, n_new):
