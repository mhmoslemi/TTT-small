Let me focus on making the CVXPY trust-region work well. The key insight is:

At point h, the scores are S_d = sum_p h[p+d]*(1-h[p]) * dx for each lag d.
The gradient of S_d with respect to h[j] is:
- For each p where j = p (i.e., p = j): contribution is -h[j+d] * dx (if 0 ≤ j+d < n)
- For each p where j = p+d (i.e., p = j-d): contribution is (1-h[j-d]) * dx (if 0 ≤ j-d < n)

So G[r, j] = sum over valid p such that either p=j or p=j-d of the appropriate terms.

Let me verify: S_d = sum_{p=max(0,-d)}^{min(n, n-d)-1} h[p+d] * (1-h[p]) * dx

dS_d/dh[j] = dx * [sum over p where p+d=j and valid: (1-h[p]) + sum over p where p=j and valid: -h[j+d]]
= dx * [(1-h[j-d]) if 0≤j-d<n and valid for lag d, else 0] + [-h[j+d] if 0≤j+d<n and valid for lag d, else 0]

- First term: p = j-d, valid if max(0,-d) ≤ j-d ≤ min(n, n-d)-1

    F = scores_raw * (2.0 / n)
    G = _build_G(h)
    
    # L-infinity trust region: x in [h-sigma, h+sigma] ∩ [0,1]
    x = cp.Variable(n)
    t = cp.Variable()

        x <= np.minimum(1.0, h + sigma),

        else:
            prob.solve(verbose=False)

    max_s = float(np.max(scores))

def _lp_trust_step(h, target, sigma):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = _scores_raw(h) * dx  # F
    G = _build_G(h)
    
    # LP: minimize t
    # Variables: [x_0, ..., x_{n-1}, t]  (n+1 variables)
    # Constraints:
    #   G @ x - t <= (F - G @ h)   [L constraints]
    #   x_i >= max(0, h_i - sigma)  [n lower bounds]
    #   x_i <= min(1, h_i + sigma)  [n upper bounds]
    #   sum(x) = target             [1 equality]
    # Objective: minimize t
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ [x, t] <= b_ub
    # G @ x - t <= F - G @ h
    # [G, -1] @ [x; t] <= F - G @ h
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = scores - G @ h
    
    # Equality: sum(x) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    lo = np.maximum(0.0, h - sigma)
    hi = np.minimum(1.0, h + sigma)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))  # t is unbounded (but will be positive)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

Actually wait - I just realized something important. The LP trust-region approach I described has a potential issue: the linearization F + G@(x-h) <= t is a FIRST-ORDER approximation. If the true scores S(x) are concave in x (which they might be in some directions), then the linearization is an upper bound, and minimizing the linear approximation gives a valid upper bound on the true objective. In that case, the LP solution is guaranteed to not increase the true objective (or at least, the linearized objective).

But if the scores are convex in some directions, the linearization is a lower bound, and the LP might underestimate the true scores. This is why the line search safety net is important.

For the specific problem at hand: S_d(x) = sum_p x[p+d]*(1-x[p]) * dx. This is a concave function of x (it's a sum of products of affine functions, which is concave on [0,1]^n since each x[p] ∈ [0,1]). Wait, no. x[p+d]*(1-x[p]) is bilinear, not necessarily concave or convex.

This means the linearization can be in either direction. The trust region and line search are essential for correctness.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        for p in range(low, high):
            q = p + d
            G[r, q] += dx * (1.0 - h[p])
            G[r, p] -= dx * h[q]
    return G

def _lp_trust_loop(h, deadline, sigma_init=0.05, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sigma = sigma_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50 or sigma < 0.002:
            break
        
        dx = 2.0 / n
        L = 2 * n - 1
        F = _scores_raw(h) * dx
        G = _build_G(h)
        
        # LP: minimize t s.t. G@x - t <= F - G@h, bounds, sum(x)=target
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = F - G @ h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        lo = np.maximum(0.0, h - sigma)
        hi = np.minimum(1.0, h + sigma)
        bounds = list(zip(lo, hi)) + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 1000})
        except Exception:

            sigma *= 0.7
            continue

            stall += 1
