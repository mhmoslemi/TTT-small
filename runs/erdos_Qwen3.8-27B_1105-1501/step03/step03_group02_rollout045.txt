def _score_tau(h, d):
    n = len(h)
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    elif d > 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:d+m] * (1.0 - h[:m])))
    else:
        g = -d
        m = n - g
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * (1.0 - h[g:g+m])))

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
