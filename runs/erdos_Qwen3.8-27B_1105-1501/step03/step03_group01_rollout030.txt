def _lp_dir(h, G, scores, K, radius):
    n = len(h)
    nd = 2 * n - 1
    if K <= 0:
        K = nd
    K = min(K, nd)
    order = np.argsort(-scores)[:K]
    G_act = G[order]
    s_act = scores[order]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = 0.0
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            P = res.x[:n]
            if float(np.max(np.abs(P))) > 1e-14:
                return P
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, K_init=20, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    K = K_init
    stall = 0
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_rows(h)
        P = _lp_dir(h, G, scores, K, radius)
        if P is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.08
                stall = 0
            continue
        P = P - P.mean()
        t_max = 1.0
        for j in range(n):
            if P[j] > 1e-14:
                t_max = min(t_max, (1.0 - h[j]) / P[j])
            elif P[j] < -1e-14:
                t_max = min(t_max, h[j] / (-P[j]))
        t_max = max(t_max, 1e-12)
        S0, L, C = _line_coeffs(h, P)
        alpha, f_obj = _line_search_q(t_max, S0, L, C, 2.0 / n)
        if f_obj < cur_c5 - 1e-11:
            h_new = h + alpha * P
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                stall = 0
                radius = min(radius * 1.12, 0.35)
                K = min(K + 1, 2 * n - 1)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0:
                K = max(5, K // 2)
    return best_h, best_c5

def _softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:])
                grad[d:] += wk * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wk * (1.0 - h[:m])
                grad[:m] += wk * (-h[g:])
    return grad

def _grad_optimize(h, deadline, tau=60.0, lr=0.008, beta=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        g = _softmax_grad(h, tau)
        g = g - g.mean()
        mom = beta * mom + (1.0 - beta) * g
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            lr = min(lr * 1.03, 0.05)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.92
            if stall % 25 == 0:
                h = best_h + np.random.normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                mom = np.zeros(n)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / dj
                    else:
                        am = h[j] / (-dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / dk)
                    else:
                        am = min(am, h[k] / (-dk))
                    if am < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def _make_start(n, seed, h_ref=None):
    rng = np.random.default_rng(seed)
    if h_ref is not None:
        h = _resample(h_ref, n)
        h += rng.normal(0, 0.03, n)
    else:
        x = (np.arange(n) + 0.5) / n
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
        h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Intensive LP on initial n
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _lp_optimize(h, d1, K_init=20, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient-based to escape local optima
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _grad_optimize(best_h, d2, tau=50.0, lr=0.006, beta=0.8)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Follow up with LP
    d2b = t0 + budget_s * 0.42
    h2b, c5_2b = _lp_optimize(best_h, d2b, K_init=15, radius_init=0.1)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: Multi-resolution search
    d3 = t0 + budget_s * 0.70
    n_candidates = list(range(max(30, best_n - 30), min(200, best_n + 31), 2))
    if best_n % 2 != 0:
        n_candidates.append(best_n)
    n_candidates = sorted(set(n_candidates))
    # Prioritize n values close to best_n
    n_candidates.sort(key=lambda x: abs(x - best_n))
    
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try == best_n:
            continue
        if n_try < 24 or n_try > 200:
            continue
        h_r = _make_start(n_try, seed, best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, K_init=12, radius_init=0.12)
        if time.time() < d3 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_r, c5_r = _lp_optimize(h_r, sub_dl2, K_init=20, radius_init=0.06)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Perturbation + re-optimization at best_n
    d4 = t0 + budget_s * 0.85
    scale = 0.03
    for attempt in range(100):
        if time.time() > d4:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, K_init=15, radius_init=0.1)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.2)

    # Phase 5: Final LP polish
    if time.time() < t0 + budget_s * 0.93:
        d5 = t0 + budget_s * 0.93
        h_f, c5_f = _lp_optimize(best_h, d5, K_init=20, radius_init=0.05)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Pair refinement
    if time.time() < t0 + budget_s * 0.97:
        d6 = t0 + budget_s * 0.97
        h_p, c5_p = _pair_refine(best_h, d6, max_passes=80)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final LP micro-polish
    if time.time() < t0 + budget_s * 0.995:
        h_f2, c5_f2 = _lp_optimize(best_h, t0 + budget_s * 0.998, K_init=10, radius_init=0.02)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
