def c5_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_box_sum(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:

    if target >= n:

        else:
            hi = mid

    x = np.clip(v - lam, 0.0, 1.0)

def line_objective(S_r, B_r, Q_r, t_vals, dx):

def grad_soft_weights(h, tau, dx):
    """Compute softmax weights on the lag scores."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    return w, corr * dx

def grad_direction(h, G, w, target):
    """Compute gradient direction from weighted sum of lag gradients."""
    grad = G.T @ w
    # Make zero-sum
    grad = grad - float(np.mean(grad))
    # Negate for descent
    D = -grad
    return D

def build_grad_matrix(h):
    """Build the gradient matrix G where G[r, i] = d(corr_r)/d(h_i)."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m_len = n - d
            if m_len > 0:
                G[r, d:] += (1.0 - h[:m_len])
                G[r, :m_len] += -h[d:]
        else:
            g = -d
            m_len = n - g
            if m_len > 0:
                G[r, :m_len] += (1.0 - h[g:])
                G[r, g:] += -h[:m_len]
    return G

    start2 = available[int(rng.integers(0, len(available)))]
    
    D[start1:start1+bs] = 1.0
    D[start2:start2+bs] = -1.0
    return D

def c5_exact(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def line_eval(S_r, B_r, Q_r, t, dx):
    """Evaluate C5 at parameter t along a line."""
    vals = S_r + t * B_r - t * t * Q_r
    return float(np.max(vals) * dx)

def line_eval_array(S_r, B_r, Q_r, t_arr, dx):

def soft_weights(scores, tau):
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    s = float(np.sum(exp_s))
    if s < 1e-300:
        return np.ones(len(scores)) / len(scores)
    return exp_s / s

def build_G(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    one_minus = 1.0 - h
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            mlen = n - d
            G[r, d:] += one_minus[:mlen]
            G[r, :mlen] -= h[d:]
        else:
            g = -d
            mlen = n - g
            G[r, :mlen] += one_minus[g:]
            G[r, g:] -= h[:mlen]
    return G
