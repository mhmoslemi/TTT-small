            lag2 = -lag
            m_valid = n - lag2
            if m_valid <= 0:
                continue
            b[r] = np.sum(d[:m_valid] * (1 - h[lag2:]) - h[:m_valid] * d[lag2:])
            a[r] = -np.sum(d[:m_valid] * d[lag2:])
    
    # Now find min_{t in [0, alpha_max]} max_r (a_r*t² + b_r*t + c0_r)
    # Candidate points:
    candidates = [0.0, alpha_max]
    
    # Vertices of individual quadratics
    for r in range(m):
        if a[r] > 1e-15:
            t_v = -b[r] / (2 * a[r])
            if 0 < t_v < alpha_max:
                candidates.append(t_v)
        elif abs(a[r]) <= 1e-15:
            # Linear: no interior minimum unless b=0
            pass
    
    # Pairwise intersections (only between "important" quadratics to save time)
    # For n=64, m=127, C(127,2)=8001 pairs, each giving at most 2 points
    # That's ~16000 points, each evaluation is O(m)=127, total ~2M ops. Fine.
    # But let's limit to top quadratics to be safe
    # Actually, let's do all pairs for n <= 64
    
    for r in range(m):
        for s in range(r+1, m):
            # Solve a_r*t² + b_r*t + c_r = a_s*t² + b_s*t + c_s
            # (a_r - a_s)*t² + (b_r - b_s)*t + (c_r - c_s) = 0
            da = a[r] - a[s]
            db = b[r] - b[s]
            dc = c0[r] - c0[s]
            if abs(da) < 1e-15:
                # Linear: db*t = -dc
                if abs(db) > 1e-15:
                    t_i = -dc / db
                    if 0 < t_i < alpha_max:
                        candidates.append(t_i)
            else:
                disc = db*db - 4*da*dc
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2*da)
                    t2 = (-db - sqrt_disc) / (2*da)
                    for t_i in [t1, t2]:
                        if 0 < t_i < alpha_max:
                            candidates.append(t_i)
    
    # Evaluate at all candidates
    candidates = np.array(candidates)
    candidates = np.clip(candidates, 0, alpha_max)
    candidates = np.unique(candidates)  # remove duplicates
    
    # Evaluate max_r (a_r*t² + b_r*t + c_r) at each candidate
    # Shape: (len(candidates), m)
    vals = a[np.newaxis, :] * candidates[:, np.newaxis]**2 + b[np.newaxis, :] * candidates[:, np.newaxis] + c0[np.newaxis, :]
    max_vals = vals.max(axis=1)
    
    best_idx = np.argmin(max_vals)
    best_t = candidates[best_idx]
    best_raw = max_vals[best_idx]
    best_c5 = best_raw * dx
    
    return best_t, best_c5

    vals_at_t = a * t_approx**2 + b * t_approx + c0
    top_k = np.argsort(vals_at_t)[-10:]  # top 10

    candidates = [0.0, alpha_max, t_approx]
    for i in range(len(top_k)):
        for j in range(i+1, len(top_k)):

            t_v = -b[r] / (2*a[r])
            if 0 <= t_v <= alpha_max:
