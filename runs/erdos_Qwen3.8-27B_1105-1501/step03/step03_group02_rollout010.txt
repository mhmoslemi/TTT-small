Let me also add an LP-based direction finder as a backup:

The LP approach: given current h and current scores s, find a direction d such that:
- sum(d) = 0
- ||d||_∞ ≤ 1 (or some bound)
- For all k: s_k + gradient_k · d ≤ s_max (i.e., we don't increase any score above current max)
- Maximize the decrease in the max: minimize max_k (s_k + gradient_k · d)

This is an LP! (Linear in d since we're linearizing the scores.)

Variables: d_1, ..., d_n, z
Minimize: z
Subject to:
- For all k: ∑_i grad_k[i] * d_i + s_k ≤ z
- sum(d) = 0
- -1 ≤ d_i ≤ 1

This gives a direction that (in the linearized sense) minimizes the new max. Then we do a line search in this direction.

def _lp_direction(h, scores, deadline):
    """Find LP direction to reduce max score (linearized)"""
    n = len(h)
    num_lags = 2 * n - 1
    
    # Build gradient matrix: G[k, i] = grad of score k w.r.t. h_i
    G = np.zeros((num_lags, n))
    for k in range(num_lags):
        d = k - (n - 1)
        G[k] = _score_grad(h, d)
    
    # LP: min z s.t. G @ d + scores <= z, sum(d) = 0, -1 <= d <= 1
    # Variables: d_1,...,d_n, z (total n+1 vars)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize z
    
    # Inequality: G @ d - z <= -scores
    # i.e., G @ d + 0*... - 1*z <= -scores
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            grad[:m] += -h[d:m+d]
            grad[d:n] += (1.0 - h[:m])
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += (1.0 - h[g:g+m])
            grad[g:g+m] += -h[:m]
    return grad

        return np.zeros(n)

        return np.ones(n)

            actual_c5 = _exact_c5(h)
            if actual_c5 < best_c5 - 1e-15:
                best_c5 = actual_c5
                best_h = h.copy()
                stall = 0
                max_step = min(0.5, max_step * 1.05)
            else:
                h = best_h.copy()
                stall += 1
                max_step *= 0.9
        else:
            stall += 1
            max_step *= 0.8
            if max_step < 1e-6:
                break
    return best_h, best_c5

def _combined_optimize(h, deadline, use_grad=True, use_lp=True):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if use_grad:
        t_mid = time.time() + (deadline - time.time()) * 0.6
        h_g, c5_g = _grad_descent(best_h, t_mid)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
    if use_lp and time.time() < deadline:
        h_l, c5_l = _lp_refine(best_h, deadline)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    
    best_h = _proj_feasible(h_init, target0)
    best_c5 = _exact_c5(best_h)
    best_n = n0
    
    def _elapsed():
        return time.time() - t0
    
    def _remaining():
        return max(0.0, budget_s - _elapsed())
    
    # Phase 1: Gradient descent (30% budget)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _grad_descent(best_h, dl1, beta_init=100.0, max_iters=15000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: LP refinement (15% budget)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = _lp_refine(best_h, dl2, max_iters=8000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
    
    # Phase 3: Multi-restart (35% budget)
    dl3 = t0 + budget_s * 0.80
    attempt = 0
    scale = 0.04
    while time.time() < dl3:
        attempt += 1
        if best_c5 < 0.38070:
            break
        per_budget = min(20.0, _remaining() * 0.15)
        if per_budget < 2.0:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _proj_feasible(h_pert, best_n / 2.0)
        per_dl = time.time() + per_budget
        h_r, c5_r = _combined_optimize(h_pert, per_dl, use_grad=True, use_lp=True)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r
            scale = max(0.008, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.15)
            if scale > 0.1 and attempt > 20:
                break
    
    # Phase 4: Try different resolutions (15% budget)
    dl4 = t0 + budget_s * 0.95
    for n_try in [best_n + 2, best_n - 2, best_n + 4, best_n + 8, best_n + 16, best_n - 4]:
        if time.time() > dl4:
            break
        if n_try < 32 or n_try > 160 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        per_dl = min(time.time() + 15.0, dl4)
        h_opt, c5_opt = _combined_optimize(h_r, per_dl, use_grad=True, use_lp=True)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
    
    # Phase 5: Final polish (5% budget)
    if time.time() < t0 + budget_s * 0.99:
        dl5 = t0 + budget_s * 0.995
        h_f, c5_f = _lp_refine(best_h, dl5, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_feasible(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, best_n
