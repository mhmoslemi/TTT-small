from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _weighted_grad(h, p):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2*n-1):
        k = idx - (n-1)
        pk = p[idx]
        if abs(pk) < 1e-15:
            continue
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += pk * (1.0 - h[:m])
                grad[:m] -= pk * h[k:]
        else:
            g = -k
            m = n - g
            if m > 0:
                grad[:m] += pk * (1.0 - h[g:])
                grad[g:] -= pk * h[:m]
    return grad

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _softmax_weights(scores, beta):
    s = scores - scores.max()
    e = np.exp(beta * s)
    return e / e.sum()

def _topk_weights(scores, k):
    top_idx = np.argsort(scores)[-k:]
    w = np.zeros(len(scores))
    w[top_idx] = 1.0 / k
    return w

    """Maximize weighted sum of scores via projected gradient ascent."""
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_val = float(np.dot(p, _scores(h)))
    for it in range(max_iters):
        grad = _weighted_grad(h, p)  # gradient of weighted sum
        grad_mean = np.mean(grad)
        gnorm = np.linalg.norm(grad - grad_mean)
        if gnorm < 1e-12:

        best_f = best_val
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:

            val = float(np.dot(p, _scores(h_try)))

                best_alpha = alpha
        if best_alpha is None:

        h = _project(h + best_alpha * direction, target)
        val = float(np.dot(p, _scores(h)))
        if val > best_val:
            best_val = val
            best_h = h.copy()
    return best_h

            break
        frac = it / max_iters
        beta = beta_min * (beta_max / beta_min) ** frac
        scores = _scores(h)
        top_val = scores.max()

        s = scores - top_val

                break

            if stall > 30:
                break

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _wgrad(h, p):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2*n-1):
        k = idx - (n-1)
        pk = p[idx]
        if abs(pk) < 1e-14:
            continue
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += pk * (1.0 - h[:m])
                grad[:m] -= pk * h[k:]

            g = -k
            m = n - g
            if m > 0:
                grad[:m] += pk * (1.0 - h[g:])
                grad[g:] -= pk * h[:m]
