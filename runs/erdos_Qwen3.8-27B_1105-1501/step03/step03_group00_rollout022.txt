    opt_c5 = _exact_c5(h + opt_a * p)

        return opt_a, opt_c5
    return best_a, best_c5

def _pnorm_descent(h, deadline, p_start=4, p_end=500, n_iter=500, momentum=None):
    """Adaptive p-norm gradient descent with Nesterov momentum."""
    if momentum is None:
        momentum = 0.9
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)  # momentum velocity
    
    n_actually_done = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Adaptive p
        frac = min(1.0, it / max(1, n_iter - 1))
        p = p_start * (p_end / p_start) ** frac
        
        scores = _scores(h)
        G = _grad_matrix(h)
        grad = _pnorm_grad(scores, p, G)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Nesterov step
        v = momentum * v - grad / gn
        # Apply step with adaptive learning rate
        lr = 0.3 / np.sqrt(n)
        
        # Try several step sizes
        best_local_c5 = cur_c5
        best_local_h = h.copy()
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            h_try = _project(h + scale * v, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_local_c5 - 1e-14:
                best_local_c5 = c5_try
                best_local_h = h_try
        
        if best_local_c5 < cur_c5 - 1e-13:
            h = best_local_h
            cur_c5 = best_local_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Reset momentum if no progress
            v *= 0.5
    
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _equalize_active(h, deadline, k_active=5):
    """Try to equalize the top-k active scores."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

        top_idx = np.argsort(scores)[-k_active:]
        top_scores = scores[top_idx]
        gap = top_scores.max() - top_scores.min()
        
        if gap < 1e-6 * top_scores.max():
            break
        
        # Weights: inverse of score (push down higher scores more)
        w = 1.0 / (top_scores + 1e-10)
        w /= w.sum()

        G_top = G[top_idx]
        grad = G_top.T @ w
        grad -= np.mean(grad)
        
        gn = np.linalg.norm(grad)

            break

        alpha, new_c5 = _line_search(h, direction, best_c5, max_alpha=0.1)

            if cur_c5 < best_c5:

                best_h = h.copy()

def _binary_sa(h_start, deadline, n_iter=20000):
    """Binary simulated annealing."""
    n = len(h_start)
    if n % 2 != 0:
        return h_start.copy(), _exact_c5(h_start)
    
    target = n // 2
    # Threshold to binary
    thresh = np.sort(h_start)[target]
    h_bin = (h_start >= thresh).astype(float)
    # Ensure exactly target ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Remove excess ones (those closest to threshold)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_start[ones])
        for idx in order[:n_ones - target]:
            h_bin[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_start[zeros])
        for idx in order[:target - n_ones]:
            h_bin[zeros[idx]] = 1
    
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    T = 0.01
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= 0.9997
        
        # Swap a 1 and a 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_try = h_bin.copy()
        h_try[i] = 0
        h_try[j] = 1
        c5_try = _exact_c5(h_try)
        
        delta = c5_try - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h_bin = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=8):

    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if len(interior) > 30:

            for k in range(j + 1, n):

                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)

                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 25
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = np.argmin(vals)
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _pnorm_descent(h, deadline, p_start=4, p_end=400, n_iter=600, mom=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, n_iter - 1))
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        G = _grad_matrix(h)
        sc = scores + 1e-15
        if p < 50:
            w = sc ** (p - 1)
            w = w / (w.sum() + 1e-300)
        else:
            max_s = np.max(sc)
            w = sc * np.exp(p * (sc - max_s))
            sw = np.sum(w)
            if sw < 1e-300:
                w = np.zeros(len(sc))
                w[np.argmax(sc)] = 1.0
            else:
                w = w / sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = mom * v - grad / gn
        lr = 0.25 / np.sqrt(n)
        best_local = cur_c5
        best_local_h = h.copy()
        for scale in [lr, lr * 0.5, lr * 0.2, lr * 2.0]:
            h_try = _project(h + scale * v, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_local - 1e-14:
                best_local = c5_try
                best_local_h = h_try
        if best_local < cur_c5 - 1e-13:
            h = best_local_h
            cur_c5 = best_local
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            v *= 0.5
    return best_h, best_c5

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None
