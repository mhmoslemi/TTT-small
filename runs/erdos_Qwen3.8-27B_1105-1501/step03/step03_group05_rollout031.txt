    cur_c5 = float(np.max(S)) * dx

    a_min = max(-h[i], h[j] - 1.0)
    a_max = min(1.0 - h[i], h[j])
    if a_max - a_min < 1e-14:

    d_range = np.arange(-(n-1), n)  # length 2n-1
    # h_ext[x] = h[x] if 0<=x<n else 0

import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _compute_G(h, i, j, n):
    """Compute G vector for pair transfer i gets, j loses."""
    d_range = np.arange(-(n-1), n)
    # Helper: h_ext(idx) = h[idx] if 0<=idx<n else 0
    # Vectorized:
    jd = j + d_range
    id_ = i + d_range
    id_neg = i - d_range
    jd_neg = j - d_range
    
    t1 = np.where((jd >= 0) & (jd < n), h[jd.clip(0, n-1)], 0.0)
    t2 = np.where((id_ >= 0) & (id_ < n), h[id_.clip(0, n-1)], 0.0)
    t3 = np.where((id_neg >= 0) & (id_neg < n), h[id_neg.clip(0, n-1)], 0.0)
    t4 = np.where((jd_neg >= 0) & (jd_neg < n), h[jd_neg.clip(0, n-1)], 0.0)
    
    return t1 - t2 - t3 + t4

def _pair_line_search(S, G, a_min, a_max, dx):
    """Minimize f(a) = max(S + a*G) * dx over [a_min, a_max]."""
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_min
    hi = a_max
    if hi - lo < 1e-14:
        return a_min, float(np.max(S)) * dx
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(S + c * G))

            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(S + c * G))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(S + d * G))
    
    a_opt = (lo + hi) / 2.0
    f_opt = float(np.max(S + a_opt * G))

import numpy as np
import time
from scipy.optimize import linprog

def _project_box_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_exact(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _compute_G(h, i, j, n):
    d_range = np.arange(-(n - 1), n)
    jd = j + d_range
    idv = i + d_range
    idn = i - d_range
    jdn = j - d_range
    t1 = np.where((jd >= 0) & (jd < n), h[np.clip(jd, 0, n - 1)], 0.0)
    t1 = np.where((jd >= 0) & (jd < n), t1, 0.0)
    t2 = np.where((idv >= 0) & (idv < n), h[np.clip(idv, 0, n - 1)], 0.0)
    t2 = np.where((idv >= 0) & (idv < n), t2, 0.0)
    t3 = np.where((idn >= 0) & (idn < n), h[np.clip(idn, 0, n - 1)], 0.0)
    t3 = np.where((idn >= 0) & (idn < n), t3, 0.0)
    t4 = np.where((jdn >= 0) & (jdn < n), h[np.clip(jdn, 0, n - 1)], 0.0)
    t4 = np.where((jdn >= 0) & (jdn < n), t4, 0.0)
    return t1 - t2 - t3 + t4

def _pair_line_search(S, G, a_min, a_max):
    if a_max - a_min < 1e-14:
        return a_min, float(np.max(S))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_min
    hi = a_max
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(S + c * G))
    fd = float(np.max(S + d * G))
    for _ in range(60):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(S + c * G))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(S + d * G))
    a_opt = (lo + hi) / 2.0
    f_opt = float(np.max(S + a_opt * G))
    f_lo = float(np.max(S + a_min * G))
    f_hi = float(np.max(S + a_max * G))
    if f_lo < f_opt:
        a_opt, f_opt = a_min, f_lo
    if f_hi < f_opt:
        a_opt, f_opt = a_max, f_hi
    return a_opt, f_opt

def _pair_descent(h, deadline, rng=None, max_sweeps=300):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    S = _scores(h)
    cur_max = float(np.max(S))
    cur_c5 = cur_max * dx
    best_h = h.copy()
    best_c5 = cur_c5
    pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
    if rng is not None:
        for s in range(max_sweeps):
            rng.shuffle(pairs)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if rng is not None:
            rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            a_min_v = max(-h[i], h[j] - 1.0)
            a_max_v = min(1.0 - h[i], h[j])
            if a_max_v - a_min_v < 1e-13:
                continue
            G = _compute_G(h, i, j, n)
            a_opt, f_opt = _pair_line_search(S, G, a_min_v, a_max_v)
            if f_opt < cur_max - 1e-12:
                h[i] += a_opt
                h[j] -= a_opt
                h[i] = min(max(h[i], 0.0), 1.0)
                h[j] = min(max(h[j], 0.0), 1.0)
                S = S + a_opt * G
                new_max = float(np.max(S))
                if new_max < cur_max:
                    cur_max = new_max
                    cur_c5 = cur_max * dx
                    improved = True
                    if cur_c5 < best_c5:

        if not improved:

    best_c5_final = _c5_exact(best_h)

    return best_h, best_c5

def _lp_refine(h, deadline, max_iters=500):

            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, g:] = (1.0 - h[:m])
                    G[r, :m] = -h[g:]
        c_vec = np.zeros(n + 1)

        A_ub[:, n] = -1.0

        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.zeros(1)

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 200})
            if res.success:
                direction = np.clip(res.x[:n], -0.5, 0.5)
                norm = float(np.max(np.abs(direction)))
                if norm > 1e-10:
                    direction /= norm

                    step = 0.1

                        stall += 1
                else:
                    stall += 1

                stall += 1

            stall += 1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _multi_start_descent(h, n, deadline, rng, n_starts=3, drift=0.15):

    for s in range(n_starts):

            break
        if s == 0:
            h_try = h.copy()
        else:
            h_try = h + rng.normal(0, drift, size=n)
            h_try = _project_box_sum(h_try, n / 2.0)

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5
