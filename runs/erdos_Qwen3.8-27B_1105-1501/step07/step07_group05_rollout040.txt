def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr * dx
    vmax = float(np.max(val))
    shifts = beta * (val - vmax)
    w = np.exp(shifts)
    wsum = float(np.sum(w))
    obj = vmax + math.log(wsum) / beta
    g = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < wsum * 1e-10:
            continue
        sh = s - (n - 1)
        if sh >= 0:
            m = n - sh
            if m > 0:
                g[:m] += ws * (1.0 - h[sh:sh + m])
                g[sh:] -= ws * h[:m]
        else:
            m = n + sh
            if m > 0:
                s2 = -sh
                g[s2:] += ws * (1.0 - h[:m])
                g[:m] -= ws * h[s2:s2 + m]
    g *= dx / wsum
    return obj, g

def _adam_lse_phase(h, deadline, beta_start, beta_end, lr, n_iters_per_beta, beta_mult):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    beta = beta_start
    best_h = h.copy()
    best_c = _exact_c5(h)
    stale = 0

    while time.time() < deadline:
        obj, g = _grad_lse(h, beta)
        t += 1
        m_v = 0.9 * m_v + 0.1 * g
        v_v = 0.999 * v_v + 0.001 * g * g
        m_hat = m_v / (1.0 - 0.9 ** t)
        v_hat = v_v / (1.0 - 0.999 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = _project(h + step, target)
        c_new = _exact_c5(h)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if t % n_iters_per_beta == 0:
            if stale > n_iters_per_beta:
                break
            beta = min(beta * beta_mult, beta_end)
            if beta >= beta_end:
                stale = 0
                if stale > n_iters_per_beta * 2:
                    break
    return best_h, best_c, h

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    best_val = float('inf')
    best_a = 0.0
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    cands.extend(np.linspace(a_lo, a_hi, 48).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    return best_a, best_val

    # Phase 5: Pair optimization (12%)
    d5 = t0 + budget_s * 0.89
    h5 = _project(best_h.copy(), len(best_h) / 2.0)
    h5, c5 = _pair_opt(h5, d5, max_it=40000)
    if c5 < best_c:
        best_c = c5
        best_h = h5.copy()

    # Phase 5b: Direction opt on pair result
    d5b = t0 + budget_s * 0.93
    if time.time() < d5b:
        h5b, c5b = _dir_opt(best_h, d5b, r0=0.06, stall_lim=25, nd=3)
        if c5b < best_c:
            best_c = c5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution exploration (3%)
    d6 = t0 + budget_s * 0.96
    if time.time() < d6:
        nc = len(best_h)
        for dn in [-4, 4, -8, 8]:
            if time.time() > d6:
                break
            n_try = nc + dn
            if n_try < 80 or n_try > 220:
                continue
            hr = _resample(best_h, n_try)
            sd = min(d6, time.time() + budget_s * 0.008)
            hr, cr = _dir_opt(hr, sd, r0=0.10, stall_lim=20, nd=3)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()

    # Phase 7: Final polish (4%)
    d7 = deadline
    if time.time() < d7 - 10:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _subgrad_line_opt(hf, min(d7, time.time() + budget_s * 0.02), r0=0.03, stall_lim=20)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if time.time() < d7 - 5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, min(d7, time.time() + budget_s * 0.01), max_it=15000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if time.time() < d7 - 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _dir_opt(hf, d7, r0=0.02, stall_lim=15, nd=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
