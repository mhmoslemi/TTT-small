import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:g+m]
    return G

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, p, dx):
    """Exact line search along direction p from h."""
    n = len(h)
    S0 = _raw_scores(h)
    # T_k = G[k,:] · p = change in score k per unit step
    # P_k = correlate(p,p, mode='full')[k] = quadratic coefficient
    # score_k(t) = S0[k] + t*T[k] - t^2*P[k]
    T = np.zeros(2 * n - 1)
    P = np.correlate(p, p, mode='full')
    # Compute T via gradient
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(p[:m] * (1.0 - h[lag:lag+m]) - h[:m] * p[lag:lag+m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(p[g:] * (1.0 - h[:m]) - h[g:] * p[:m]))
    
    # Find t_hi from box constraints
    t_hi = 1.0
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search + golden section
    n_grid = 50
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * P)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    if hi_b - lo_b > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * P)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * P)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * P)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * P)) * dx
    
    # Check endpoints too
    candidates = [0.0, t_hi]
    opt_t = (lo_b + hi_b) / 2.0
    candidates.append(opt_t)
    best_t = min(candidates, key=lambda t: float(np.max(S0 + t * T - t * t * P)) * dx)
    
    new_h = h + best_t * p
    new_h = _project_sum(new_h, n / 2.0)
    new_c5 = _exact_c5(new_h)
    return best_t, new_c5

def _epigraph_lp(h, scores, G, margin=0.5, max_rows=360):
    """Full epigraph LP: find direction p minimizing max score."""
    n = len(h)
    # Active rows: those with score near max
    max_score = float(np.max(scores))
    active = np.where(scores > max_score - margin)[0]
    if len(active) > max_rows:
        # Keep top max_rows
        active = active[np.argsort(scores[active])[::-1][:max_rows]]
    
    m = len(active)
    if m == 0:
        return None
    
    # Variables: p_0..p_{n-1}, eta (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: G[active,j]*p_j - eta <= -scores[active]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[j] <= p_j <= 1-h[j]
    bounds = [(max(-h[j], -1.0), min(1.0 - h[j], 1.0)) for j in range(n)]
    bounds.append((None, None))  # eta free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _focused_lp(h, scores, G, top_k, margin=0.3):
    """Focused LP with top_k constraints."""
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-h[j], -1.0), min(1.0 - h[j], 1.0)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _coord_pair_descent(h, deadline, n_iter=20000, seed=42):
    """Coordinate pair descent: p = e_i - e_j."""
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_c5 = cur_c5
    best_cur = cur.copy()
    rng = np.random.default_rng(seed)
    stall = 0
    dx = 2.0 / n
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        # Direction: move mass from j to i
        if cur[j] < 1e-10 or cur[i] > 1 - 1e-10:
            stall += 1
            continue
        p = np.zeros(n)
        p[i] = 1.0
        p[j] = -1.0
        t, new_c5 = _line_search(cur, p, dx)
        if new_c5 < cur_c5 - 1e-13:
            cur = cur + t * p
            cur = _project_sum(cur, n / 2.0)
            cur_c5 = _exact_c5(cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_cur = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_cur, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=123):
    """Simulated annealing on binary sequence."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=20):
    """Exhaustive pair swap that improves."""
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _lp_loop(h, deadline, use_full=True, top_k=6, margin=0.5, stall_limit=50):
    """Main LP optimization loop."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline:
        scores = _raw_scores(h)
        if use_full:
            p = _epigraph_lp(h, scores, _grad_matrix(h), margin=margin)
        else:
            p = _focused_lp(h, scores, _grad_matrix(h), top_k)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            # Try a smaller focus
            if use_full:
                p = _focused_lp(h, scores, _grad_matrix(h), top_k=min(top_k, 4))
            if p is None or float(np.max(np.abs(p))) < 1e-12:
                stall += 1
                if stall > stall_limit:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, p, dx)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project_sum(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
    
    return best_h, best_c5

def _grad_matrix(h):

                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:g + m]

    h1, c5_1 = _lp_loop(h1, d1, margin=1.5, stall_limit=40, use_full=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Main continuous optimization (42%)
    d2 = t0 + budget_s * 0.50
    # Full epigraph LP - first pass with generous margin
    h2 = _project_sum(best_h.copy(), n0 / 2.0)
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h2, c5_2 = _lp_loop(h2, sub_dl, margin=1.0, stall_limit=80, use_full=True)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Coordinate pair descent
    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.08)
        h2, c5_2 = _coord_pair_descent(best_h, sub_dl, n_iter=50000, seed=seed + 1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Tighter LP
    if time.time() < d2:
        h2 = _project_sum(best_h.copy(), n0 / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.08)
        h2, c5_2 = _lp_loop(h2, sub_dl, margin=0.6, stall_limit=60, use_full=True)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Another coordinate pass
    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.06)
        h2, c5_2 = _coord_pair_descent(best_h, sub_dl, n_iter=40000, seed=seed + 2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Final LP in phase 2
    if time.time() < d2:
        h2 = _project_sum(best_h.copy(), n0 / 2.0)
        h2, c5_2 = _lp_loop(h2, d2, margin=0.5, stall_limit=50, use_full=True)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Binary optimization (12%)
    d3 = t0 + budget_s * 0.62
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=seed + 3)
    sub_dl2 = min(d3, time.time() + budget_s * 0.02)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=20)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < d3:
        h_pol = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _lp_loop(h_pol, sub_dl3, margin=0.5, stall_limit=40, use_full=True)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (18%)
    d4 = t0 + budget_s * 0.80
    n_cur = len(best_h)
    tried_ns = set()
    n_offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24]
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 80 or n_try > 220 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, margin=0.8, stall_limit=30, use_full=True)
        if time.time() < d4 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d4, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=12000, T_init=0.002, seed=n_try)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if time.time() < d4 - 2:
            h_pol = _project_sum(h_r.copy(), n_try / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.006)
            h_pol, c5_p = _lp_loop(h_pol, sub_dl3, margin=0.5, stall_limit=20, use_full=False)
            if c5_p < c5_r:
                h_r, c5_r = h_pol, c5_p
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 5: Basin hopping (12%)
    d5 = t0 + budget_s * 0.92
    scale = 0.05
    n_restarts = 0
    while time.time() < d5 and n_restarts < 120:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, margin=0.8, stall_limit=20, use_full=False)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.06)

    # Phase 6: Final polish (8%)
    d6 = deadline
    h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _lp_loop(h6, min(d6, time.time() + budget_s * 0.03), margin=0.4, stall_limit=40, use_full=True)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    if deadline - time.time() > 5:
        h6, c5_6 = _coord_pair_descent(best_h, min(d6, time.time() + budget_s * 0.02), n_iter=15000, seed=seed + 99)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    if deadline - time.time() > 4:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, min(d6, time.time() + budget_s * 0.015), max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
