```python
    # Minimize -beta
    c = np.zeros(2 * n + 1)
    c[-1] = -1.0
    
    # Equality constraints: sum(sp) - sum(sn) = 0, sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, 2 * n + 1))
    A_eq[0, :n] = 1.0

    A_eq[1, :n] = 1.0
```

```python
import numpy as np
import time
from scipy.optimize import linprog

# Core evaluation
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

# Feasibility projection
def _project(v, target):
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    # Bisection to find theta such that sum(clip(v-theta, 0, 1)) = target
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

# Gradient matrix
def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    G = np.zeros((L, n))
    for m in range(L):
        ell = m - (n - 1)
        for i in range(n):
            j1 = i - ell
            j2 = i + ell
            val = 0.0
            if 0 <= j1 < n:
                val += o[j1]
            if 0 <= j2 < n:
                val -= h[j2]
            G[m, i] = val
    return G

# Line search along arbitrary direction
def _line_search_dir(h, d, R0, t_cap=1.5):
    n = len(h)
    o = 1.0 - h
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Feasible t range
    t_lo = -t_cap
    t_hi = t_cap
    for i in range(n):
        if d[i] > 1e-14:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0))
    
    # Grid search
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_i = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_i - 2)]
    hi = ts[min(n_grid - 1, best_i + 2)]
    if hi - lo < 1e-12:
        best_t = float(ts[best_i])
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = np.max(R0 + c1*A - c1*c1*B)
        f2 = np.max(R0 + c2*A - c2*c2*B)
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = np.max(R0 + c1*A - c1*c1*B)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = np.max(R0 + c2*A - c2*c2*B)
        best_t = (lo + hi) / 2.0
    
    best_val = np.max(R0 + best_t*A - best_t*best_t*B)
    
    # Also check endpoints and original grid best
    candidates = [0.0, t_lo, t_hi, float(ts[best_i]), best_t]
    best_v = np.inf
    for ct in candidates:
        if t_lo - 1e-10 <= ct <= t_hi + 1e-10:
            cv = np.max(R0 + ct*A - ct*ct*B)
            if cv < best_v:
                best_v = cv
                best_t = ct
    
    return best_t, float(best_v)

# LP direction
def _lp_direction(h, R, peak_count):
    n = len(h)
    L = 2 * n - 1
    top_idx = np.argsort(R)[-peak_count:]
    
    G = _grad_matrix(h)
    
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[-1] = -1.0
    
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    A_ub = np.zeros((peak_count, nv))
    for k, r in enumerate(top_idx):
        A_ub[k, :n] = G[r, :]
        A_ub[k, n:2*n] = -G[r, :]
        A_ub[k, 2*n] = 1.0
    b_ub = np.zeros(peak_count)
    
    bounds = [(0, 1)] * (2 * n) + [(-1e6, 1e6)]
    
    try:
        res = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs')
    except:
        return None
    
    if not res.success or res.x is None:
        return None
    
    u = res.x[:n] - res.x[n:2*n]
    beta = float(res.x[2*n])
    
    if beta <= 1e-8:
        return None
    
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-8:
        return None
    u = u / l1
    
    return u

# Main LP-based local descent
def _lp_descent(h, deadline, peak_counts=None, t_cap=1.0, max_no_improve=50):
    if peak_counts is None:
        peak_counts = [4, 6, 8, 12, 20]
    
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    R0 = np.correlate(h, 1.0 - h, mode='full')
    
    while time.time() < deadline:
        improved = False
        
        # Try LP directions with different peak counts
        for pc in peak_counts:
            if time.time() > deadline:
                break
            if pc > 2 * n - 1:
                continue
            u = _lp_direction(h, R0, pc)
            if u is None:
                continue
            t_opt, val_opt = _line_search_dir(h, u, R0, t_cap)
            if abs(t_opt) < 1e-12:
                continue
            h_new = h + t_opt * u
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                R0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
                improved = True
                break  # restart with new R0
        
        if not improved:
            # Fallback: pair moves on worst peak
            worst = int(np.argmax(R0))
            ell = worst - (n - 1)
            # Find coordinates that affect this peak
            # R[worst] involves h[i] and h[i+ell] (or similar)
            # Simple fallback: random pair moves
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                t_opt, val_opt = _line_search_dir(h, d, R0, 0.3)
                if abs(t_opt) < 1e-12:
                    continue
                h_new = h + t_opt * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    R0 = np.correlate(h, 1.0 - h, mode='full')
                    if cur_c5 < best_c5 - 1e-14:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        no_improve = 0
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
    
    return best_h, best_c5

# Resampling
def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

# Binary conversion
def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

# Binary local search (swap-based)
def _binary_local(h_bin, deadline, max_no_improve=200):

    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    no_improve = 0
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    while time.time() < deadline and no_improve < max_no_improve:
        # Try all pair swaps (or a subset for larger n)
        if len(ones) * len(zeros) <= 50000:
            candidates = [(oi, oz) for oi in ones for oz in zeros]
            rng.shuffle(candidates)
        else:
            candidates = [(ones[int(rng.integers(0, len(ones)))], 
                         zeros[int(rng.integers(0, len(zeros)))]] for _ in range(200)]
        
        improved = False
        for (oi, oz) in candidates:
            if time.time() > deadline:
                break
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                ones.remove(oi)
                zeros.remove(oz)
                ones.append(oz)
                zeros.append(oi)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    no_improve = 0
                improved = True
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _project(v, target):
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    G = np.zeros((L, n))
    for m in range(L):
        ell = m - (n - 1)
        i_lo = max(0, ell)
        i_hi = min(n, n + ell)
        if i_lo < i_hi:
            G[m, i_lo:i_hi] += o[i_lo - ell:i_hi - ell][::-1]
        i_lo2 = max(0, -ell)
        i_hi2 = min(n, n - ell)
        if i_lo2 < i_hi2:
            G[m, i_lo2:i_hi2] -= h[i_lo2 + ell:i_hi2 + ell]
    return G


def _line_search_dir(h, d, R0, t_cap=1.5):
    n = len(h)
    o = 1.0 - h
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')

    t_lo = -t_cap
    t_hi = t_cap
    for i in range(n):
        di = d[i]
        if di > 1e-14:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0))

    n_grid = 90
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_i = int(np.argmin(vals))
    lo = ts[max(0, best_i - 2)]
    hi = ts[min(n_grid - 1, best_i + 2)]
    best_t = float(ts[best_i])
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        best_t = (lo + hi) / 2.0

    candidates = [0.0, float(t_lo), float(t_hi), float(ts[best_i]), best_t]
    best_v = np.inf
    for ct in candidates:
        if t_lo - 1e-10 <= ct <= t_hi + 1e-10:
            cv = float(np.max(R0 + ct * A - ct * ct * B))
            if cv < best_v:
                best_v = cv
                best_t = ct
    return best_t, best_v


def _solve_lp(G, top_idx, n):
    pc = len(top_idx)
    if pc == 0:
        return None
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[-1] = -1.0
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2 * n] = 1.0
    b_eq = np.array([0.0, 1.0])
    A_ub = np.zeros((pc, nv))
    for k, r in enumerate(top_idx):
        A_ub[k, :n] = G[r, :]
        A_ub[k, n:2 * n] = -G[r, :]
        A_ub[k, 2 * n] = 1.0
    b_ub = np.zeros(pc)
    bounds = [(0, 1)] * (2 * n) + [(-1e6, 1e6)]
    try:
        res = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    u = res.x[:n] - res.x[n:2 * n]
    beta = float(res.x[2 * n])
    if beta <= 1e-8:
        return None
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-8:
        return None
    return u / l1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _binary_local(h_bin, deadline, max_no_improve=300, seed=777):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    no_improve = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])

    while time.time() < deadline and no_improve < max_no_improve:
        if len(ones) * len(zeros) <= 60000:
            candidates = [(oi, oz) for oi in ones for oz in zeros]
            rng.shuffle(candidates)
        else:
            candidates = [(ones[int(rng.integers(0, len(ones)))],
                          zeros[int(rng.integers(0, len(zeros)))]] for _ in range(300))
        improved = False
        for (oi, oz) in candidates:
            if time.time() > deadline:
                break
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                ones.remove(oi)
                zeros.remove(oz)
                ones.append(oz)
                zeros.append(oi)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    no_improve = 0
                improved = True
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
        if not improved:
            no_improve += 1
```
