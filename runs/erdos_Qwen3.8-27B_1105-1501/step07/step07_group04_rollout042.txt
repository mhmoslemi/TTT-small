Let me analyze the problem and the previous solution carefully.

The Erdős minimum overlap problem asks us to find a step function h: [0,2] → [0,1] that minimizes the maximum overlap integral C₅ = max_k ∫ h(x)(1-h(x+k)) dx.

The previous solution achieved C₅ = 0.380889, and the current record is 0.38085. The target is 0.3808.

Key observations from the previous code:
1. It uses subgradient methods, binary local search, block swaps, LP polish, pair swaps, and basin hopping
2. The initial_h_values has n=180 samples
3. The main bottleneck seems to be local optimization around a local optimum

Let me think about what could be improved:

1. **LP-based trust region approach**: The strategy suggests using a proper LP trust region subproblem. This is fundamentally different from the subgradient approach. Instead of just following the subgradient of the max correlation, we build a small LP that simultaneously handles multiple active correlation constraints.

2. **Better active set management**: The LP approach can handle multiple active constraints simultaneously rather than averaging them.

3. **Smarter binary search**: Instead of random swaps, we can use the LP direction to guide which binary swaps to try.

4. **Multi-resolution approach**: Try different n values more systematically.

Let me design a solution that:
- Uses an LP trust region method as the core optimizer
- Builds the exact gradient matrix for active correlation constraints
- Uses proper line search with the quadratic expansion
- Combines with smart binary search guided by the LP direction
- Tries multiple resolutions

Key implementation details:
- n=180 gives 359 correlation lags
- The LP has variables: d (n variables) + tau (1 variable) = n+1 variables
- Constraints: sum(d) = 0, tau >= F_r(h) + grad_r * d for active r, |d_i| bounded
- Objective: minimize tau + eps * sum(|d_i|)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, k, n):
    """Gradient of correlation at lag k (k can be negative, zero, or positive)."""
    g = np.zeros(n)
    dx = 2.0 / n
    if k > 0:
        # F(h) = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
        # dF/dh[j] for j >= k: (1 - h[j-k])
        # dF/dh[j] for j < n-k: -h[j+k]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:n]
    elif k < 0:
        u = -k
        # F(h) = sum_{i=0}^{n-u-1} h[i] * (1-h[i+u])
        # dF/dh[j] for j < n-u: (1 - h[j+u])
        # dF/dh[j] for j >= u: -h[j-u]
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    else:
        # F(h) = sum h[i] * (1-h[i])
        g[:] = 1.0 - 2.0 * h
    return g * dx

def _build_active_grads(h, n, active_lags):
    """Build gradient rows for active lags."""
    rows = []
    for lag_idx in active_lags:
        k = lag_idx - (n - 1)
        rows.append(_corr_grad(h, k, n))
    return np.array(rows)

def _line_search_exact(h, d, dx, alpha_max, n_grid=40):
    """Exact line search using quadratic expansion."""
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    def f_alpha(a):
        return float(np.max(C0 + a * L1 - a * a * Q2)) * dx
    
    if alpha_max < 1e-14:
        return 0.0, float(np.max(C0)) * dx
    
    grid = np.linspace(0.0, alpha_max, n_grid)
    vals = np.array([f_alpha(a) for a in grid])
    best_idx = int(np.argmin(vals))
    
    # Golden section refinement
    lo = grid[max(0, best_idx - 2)]
    hi = grid[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f_alpha(c1), f_alpha(c2)
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f_alpha(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f_alpha(c2)
    
    opt_alpha = (lo + hi) / 2.0
    opt_val = f_alpha(opt_alpha)
    
    # Check endpoints
    candidates = [(opt_alpha, opt_val), (0.0, f_alpha(0.0)), (alpha_max, f_alpha(alpha_max))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_direction(h, active_grads, active_vals, cur_c5, radius):
    """Solve LP trust region to find descent direction.
    Variables: d[0..n-1], tau
    Minimize: tau + eps * sum(|d_i|)
    Subject to:
        sum(d) = 0
        active_vals[i] + active_grads[i] @ d <= tau  for each active constraint
        -min(radius, h_i) <= d_i <= min(radius, 1-h_i) for each i
        -cur_c5 - 0.5 <= tau <= cur_c5 + 1e-6
    """
    n = len(h)
    m = len(active_grads)
    eps = 1e-5
    
    # Variables: d[0..n-1], u[0..n-1] (for |d_i|), tau
    # Total: 2n + 1 variables
    nvar = 2 * n + 1
    d_offset = 0
    u_offset = n
    tau_idx = 2 * n
    
    # Objective: tau + eps * sum(u_i)
    c_obj = np.zeros(nvar)
    c_obj[tau_idx] = 1.0
    c_obj[u_offset:2*n] = eps
    
    # Inequality constraints: A_ub @ x <= b_ub
    # tau - sum(active_grads[i] * d) >= active_vals[i]
    # => -active_grads[i] @ d + tau >= active_vals[i]
    # => active_grads[i] @ d - tau <= -active_vals[i]
    # In standard form A_ub @ x <= b_ub:
    # Row for each active constraint: active_grads[i] for d, 0 for u, -1 for tau <= -active_vals[i]
    
    A_ub = np.zeros((m, nvar))
    b_ub = np.zeros(m)
    for i in range(m):
        A_ub[i, d_offset:n] = active_grads[i]
        A_ub[i, tau_idx] = -1.0
        b_ub[i] = -active_vals[i]
    
    # u_i >= d_i => -d_i + u_i >= 0 => d_i - u_i <= 0
    # u_i >= -d_i => -u_i - d_i >= 0 => d_i + u_i <= 0... wait
    # Actually: u_i >= |d_i| means u_i >= d_i AND u_i >= -d_i
    # u_i - d_i >= 0 => -u_i + d_i <= 0
    # u_i + d_i >= 0 => -u_i - d_i <= 0 ... that's not right either
    # Let me think again. We want u_i >= |d_i|.
    # This is: u_i - d_i >= 0 and u_i + d_i >= 0
    # In <= form: -(u_i - d_i) <= 0 => d_i - u_i <= 0
    # and -(u_i + d_i) <= 0 => -d_i - u_i <= 0
    
    A_ineq = np.zeros((2*n, nvar))
    b_ineq = np.zeros(2*n)
    for i in range(n):
        # d_i - u_i <= 0
        A_ineq[i, d_offset + i] = 1.0
        A_ineq[i, u_offset + i] = -1.0
        # -d_i - u_i <= 0
        A_ineq[n+i, d_offset + i] = -1.0
        A_ineq[n+i, u_offset + i] = -1.0
    
    # Combine all A_ub
    A_full = np.vstack([A_ub, A_ineq])
    b_full = np.concatenate([b_ub, b_ineq])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, d_offset:n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_d = -min(radius, h[i]) + 1e-15
        hi_d = min(radius, 1.0 - h[i]) + 1e-15
        bounds.append((lo_d, hi_d))
    for i in range(n):
        bounds.append((0.0, 2.0 * radius))
    # tau bounds
    bounds.append((cur_c5 - 0.5, cur_c5 + 1e-6))
    
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[d_offset:n].copy()
            # Remove mean to help with numerical issues
            d = d - np.mean(d)
            return d
        else:
            return None
    except Exception:
        return None

def _lp_optimize(h, deadline, radius_init=0.08, max_active=80, stall_limit=50):
    """LP-based trust region optimization."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and radius > 1e-5:
        if stall > stall_limit:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_scaled = corr * dx
        cur_c5 = float(np.max(corr_scaled))
        
        # Select active constraints
        max_val = float(corr_scaled.max())
        # Include constraints within a tolerance of the max
        tol = max(0.005, 0.05 * max_val)
        active = np.where(corr_scaled >= max_val - tol)[0]
        if len(active) > max_active:
            # Sort by value and take top
            val_order = np.argsort(-corr_scaled)
            active = val_order[:max_active]
        
        # Build gradient matrix for active constraints
        active_grads = _build_active_grads(h, n, active)
        active_vals = corr_scaled[active]
        
        # Solve LP
        d = _lp_direction(h, active_grads, active_vals, cur_c5, radius)
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            radius *= 0.7
            continue
        
        # Line search
        # Compute feasible alpha_max
        alpha_max = radius
        pos = d > 1e-15
        neg = d < -1e-15
        if np.any(pos):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
        
        if alpha_max < 1e-12:
            stall += 1
            radius *= 0.5
            continue
        
        alpha, new_c5 = _line_search_exact(h, d, dx, alpha_max)
        
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * d
            h = _project(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_search_guided(h_bin, deadline, max_iters=100000, seed=42):
    """Binary local search guided by correlation peaks."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    temp = 0.003
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        if not ones or not zeros:
            break
        
        # Find the worst correlation peak
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        
        # Try swaps that might reduce the worst peak
        # Focus on positions involved in the worst correlation
        if k >= 0:
            relevant_ones = [i for i in ones if 0 <= i + k < n]
            relevant_zeros = [i for i in zeros if k <= i < n]
        else:
            u = -k
            relevant_ones = [i for i in ones if 0 <= i < n - u]
            relevant_zeros = [i for i in zeros if u <= i < n]
        
        # Also include random positions
        if not relevant_ones:
            relevant_ones = ones
        if not relevant_zeros:
            relevant_zeros = zeros
        
        n_try = min(50, len(relevant_ones) * len(relevant_zeros))
        best_delta = 0.0
        best_move = None
        
        for _ in range(n_try):
            if relevant_ones and relevant_zeros:
                oi = relevant_ones[rng.integers(0, len(relevant_ones))]
                oz = relevant_zeros[rng.integers(0, len(relevant_zeros))]
            else:
                oi = ones[rng.integers(0, len(ones))]
                oz = zeros[rng.integers(0, len(zeros))]
            
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5 = _exact_c5(trial)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz)
        
        if best_move and best_delta < -1e-14:
            oi, oz = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 = _exact_c5(cur)
            # Update lists
            if oi in ones:
                ones.remove(oi)
            ones.append(oz)
            if oz in zeros:
                zeros.remove(oz)
            zeros.append(oi)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            # Try random swap with simulated annealing
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5 = _exact_c5(trial)
            delta = new_c5 - cur_c5
            if delta < -1e-14 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
                cur = trial
                cur_c5 = new_c5
                ones[i] = oz
                zeros[j] = oi
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                else:
                    stall += 1
                temp = max(0.0003, temp * 0.9998)
            else:
                stall += 1
    
    return best_h, best_c5

def _resample_2d(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _independent_set_construction(n, rng):
    """Construct a binary sequence using a greedy approach that models the overlap problem."""
    h = np.zeros(n)
    ones_needed = n // 2
    
    # Start with a simple pattern and refine
    # Use a constructive approach: try to place ones to minimize max overlap
    best = None
    best_score = float('inf')
    
    for trial in range(3):
        h_trial = np.zeros(n)
        # Initialize with random placement
        positions = rng.permutation(n)[:ones_needed]
        h_trial[positions] = 1.0
        score = _exact_c5(h_trial)
        
        # Greedy local improvement
        improved = True
        iter_count = 0
        while improved and iter_count < 200:
            improved = False
            iter_count += 1
            corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            max_k = int(np.argmax(corr))
            dx = 2.0 / n
            cur_score = float(np.max(corr)) * dx
            
            # Try to fix the worst correlation
            k = max_k - (n - 1)
            if k > 0:
                ones_in_range = [i for i in positions if i + k < n]
                zeros_in_range = [i for i in range(n) if h_trial[i] == 0 and 0 <= i - k and i - k >= 0]
                # Find a one to move
                # Try moving each one that contributes to the worst peak
                for oi in ones_in_range[:10]:
                    for oz in zeros_in_range[:10]:
                        if h_trial[oz] == 0:
                            h_trial[oi] = 0.0
                            h_trial[oz] = 1.0
                            new_score = _exact_c5(h_trial)
                            if new_score < cur_score - 1e-14:
                                cur_score = new_score
                                if oz not in positions:
                                    positions = list(positions)
                                    positions.remove(oi)
                                    positions.append(oz)
                                improved = True
                            else:
                                h_trial[oi] = 1.0
                                h_trial[oz] = 0.0
            else:
                # Center correlation - just try random swaps
                ones_list = list(positions)
                zeros_list = [i for i in range(n) if h_trial[i] == 0]
                for _ in range(50):
                    oi = ones_list[rng.integers(0, len(ones_list))]
                    oz = zeros_list[rng.integers(0, len(zeros_list))]
                    h_trial[oi] = 0.0
                    h_trial[oz] = 1.0
                    new_score = _exact_c5(h_trial)
                    if new_score < cur_score - 1e-14:
                        cur_score = new_score
                        positions = list(positions)
                        positions.remove(oi)
                        positions.append(oz)
                        improved = True
                    else:
                        h_trial[oi] = 1.0
                        h_trial[oz] = 0.0
        
        if cur_score < best_score:
            best_score = cur_score
            best = h_trial.copy()
    
    return best

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, alpha_max, n_grid=35):
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(C0 + a * L1 - a * a * Q2)) * dx

    if alpha_max < 1e-14:
        return 0.0, f(0.0)

    grid = np.linspace(0.0, alpha_max, n_grid)
    vals = np.array([f(a) for a in grid])
    best_idx = int(np.argmin(vals))
    lo = grid[max(0, best_idx - 2)]
    hi = grid[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(28):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_a = (lo + hi) / 2.0

    cands = [(opt_a, opt_v), (0.0, f(0.0)), (alpha_max, f(alpha_max))]

def _subgradient(h, corr, n):

        if k >= 0:

                g[:n-k] -= h[k:]
            else:
                g[:] = 1.0 - 2.0 * h
        else:
            ka = -k
            g[:n-ka] += (1.0 - h[ka:])
            g[ka:] -= h[:n-ka]
    g /= len(active)
    return g

            f2 = fv(c2)
    m = 0.5 * (lo + hi)
    best = min([(m, fv(m)), (0.0, fv(0.0)), (amax, fv(amax))], key=lambda x: x[1])
    return best


def _subgrad(h, corr, n):
    mx = float(corr.max())
    mask = corr >= mx - 1e-10
    cnt = int(mask.sum())
    g = np.zeros(n)
    for m in np.where(mask)[0]:
        k = int(m - (n - 1))
        if k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        elif k < 0:
            u = -k
            g[:n - u] += (1.0 - h[u:])
            g[u:] -= h[:n - u]
        else:
            g += 1.0 - 2.0 * h
    return g / cnt


def _sg_optimize(h, deadline, radius0=0.10):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c = h.copy(), cur
    rad, stall = radius0, 0
    while time.time() < deadline and rad > 1e-5:
        if stall > 100:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = -_subgrad(h, corr, n)
        d -= d.mean()
        a, nc = _line_search(h, d, dx, rad)
        if nc < cur - 1e-13:
            h = _project(h + a * d, n / 2.0)
            cur = _exact_c5(h)
            if cur < best_c:
                best_c, best_h = cur, h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.35)
        else:
            stall += 1
            rad *= 0.72
    return best_h, best_c


def _to_binary(h):
    n = len(h)
    hb = np.zeros(n)
    hb[np.argsort(h)[-n // 2:]] = 1.0
    return hb


def _bin_sa(h, deadline, seed=42, maxit=200000):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c = h.copy(), cur
    R = np.random.default_rng(seed)
    on = list(np.where(h > 0.5)[0])
    ze = list(np.where(h <= 0.5)[0])
    stall, temp = 0, 0.004
    for _ in range(maxit):
        if time.time() > deadline or stall > 4000:
            break
        if not on or not ze:
            break
        i = int(R.integers(len(on)))
        j = int(R.integers(len(ze)))
        h[on[i]], h[ze[j]] = 0.0, 1.0
        nc = _exact_c5(h)
        if nc < cur - 1e-14 or R.random() < np.exp(-(nc - cur) / max(temp, 1e-15)):
            cur = nc
            on[i], ze[j] = ze[j], on[i]
            if cur < best_c:
                best_c, best_h, stall = cur, h.copy(), 0
            else:
                stall += 1
            temp = max(3e-4, temp * 0.9998)
        else:
            h[on[i]], h[ze[j]] = 1.0, 0.0
            stall += 1
    return best_h, best_c
