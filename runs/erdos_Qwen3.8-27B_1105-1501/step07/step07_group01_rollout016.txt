def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-15)
        else:
            mask = (x > 1e-15)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(0, n):

        G[idx, k:] = 1.0 - h[:n-k]

        G[idx, :n-b] = 1.0 - h[b:n]

        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _build_grad(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for k in range(n):
        idx = n - 1 + k
        G[idx, k:] = 1.0 - h[:n-k]
        G[idx, :n-k] += -h[k:]
    for b in range(1, n):
        idx = n - 1 - b
        G[idx, :n-b] = 1.0 - h[b:]
        G[idx, b:] += -h[:n-b]
    return G

def _lp_one_shot(h, radius, top_k, seed=0):
    """Single LP step. Returns (new_h, improved)."""
    n = len(h)
    target = n / 2.0
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = R.max()
    
    sorted_idx = np.argsort(R)[::-1][:top_k]
    G = _build_grad(h)
    G_act = G[sorted_idx, :]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([G_act, -np.ones(len(sorted_idx))])
    b_ub = -R[sorted_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo >= hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                  bounds=bounds, method='highs')
    if not res.success:
        return h, False
    
    p = res.x[:n]
    if np.max(np.abs(p)) < 1e-14:
        return h, False
    
    h_new = _project(h + p, target)
    return h_new, True
