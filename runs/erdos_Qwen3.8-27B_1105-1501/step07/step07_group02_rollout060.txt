    """Compute G where G[k,j] = d(corr[k])/d(h[j])."""
    n = len(h)
    L = 2*n - 1
    G = np.zeros((L, n))
    one_minus_h = 1.0 - h
    for k in range(L):
        lag = k - (n - 1)
        # G[k,j] = (1-h[j-lag]) * 1[0<=j-lag<n] - h[j+lag] * 1[0<=j+lag<n]
        # Part 1: (1-h[j-lag]) for valid j
        # j-lag valid means 0 <= j-lag < n, i.e., lag <= j <= lag+n-1
        j_min = max(0, lag)

        G[k, j_min:j_max+1] += one_minus_h[j_min-lag:j_max+1-lag]
        # Part 2: -h[j+lag] for valid j

        j_min2 = max(0, -lag)
        j_max2 = min(n-1, n-1-lag)
        G[k, j_min2:j_max2+1] -= h[j_min2+lag:j_max2+1+lag]

def _soft_max_gradient(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    c5 = float(np.max(vals))
    L = len(vals)
    exp_vals = np.exp(beta * (vals - c5))
    sum_exp = float(np.sum(exp_vals))
    w = exp_vals / sum_exp
    one_h = 1.0 - h
    grad = np.zeros(n)
    for k in range(L):
        if w[k] < 1e-16:
            continue
        lag = k - (n - 1)
        uw = w[k] * dx
        j_lo = max(0, lag)
        j_hi = min(n - 1, lag + n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += uw * one_h[j_lo - lag:j_hi + 1 - lag]
        j_lo2 = max(0, -lag)
        j_hi2 = min(n - 1, n - 1 - lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= uw * h[j_lo2 + lag:j_hi2 + 1 + lag]
    return c5, grad

def _gradient_descent(h, deadline, beta=50.0, max_iter=800, step0=0.02):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step0
    momentum = np.zeros(n)
    mu = 0.85
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = _soft_max_gradient(h, beta)
        momentum = mu * momentum - step * grad
        h_new = _project(h + momentum, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            step = min(step * 1.04, 0.06)
        else:
            step *= 0.65
            if step < 1e-7:
                break
    return best_h, best_c5

def _nlp_solve(h, deadline, K=35, outer_iters=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    def _make_nlp(top_lags, cur_t):
        def obj(x):
            return x[-1]
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        cons = []
        for lag in top_lags:
            def make_cf(lv):
                def cf(x):
                    hx = x[:-1]
                    total = 0.0
                    for i in range(n):
                        if 0 <= i - lv < n:
                            total += hx[i] * (1.0 - hx[i - lv])
                    return x[-1] - total * dx
                def cj(x):
                    hx = x[:-1]
                    jv = np.zeros(n + 1)
                    jv[-1] = 1.0
                    for i in range(n):
                        if 0 <= i - lv < n:
                            jv[i] -= dx * (1.0 - hx[i - lv])
                        if 0 <= i + lv < n:
                            jv[i + lv] += dx * hx[i]
                    return jv
                return (cf, cj)
            cf, cj = make_cf(int(lag))
            cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})
        hq = (np.ones(n), 0.0)
        eqs = [{'type': 'eq', 'fun': lambda x: np.sum(x[:-1]) - target,
                'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        bounds = [(0.0, 1.0)] * n + [(0.0, cur_t + 0.01)]
        return obj, obj_grad, cons, eqs, bounds

    cur_t = cur_c5
    for outer in range(outer_iters):
        if time.time() > deadline:
            break
        vals = _corr_vals(h)
        order = np.argsort(vals)
        top_k_indices = order[-K:]
        top_lags = top_k_indices - (n - 1)
        x0 = np.concatenate([h, [cur_t]])
        obj, obj_grad, cons, eqs, bounds = _make_nlp(top_lags, cur_t)
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                           bounds=bounds, constraints=cons + eqs,
                           options={'maxiter': 80, 'ftol': 1e-9})
        except Exception:
            break
        if res.success or res.status <= 2:
            h_trial = _project(res.x[:-1], target)
            c5_trial = _exact_c5(h_trial)
            if c5_trial < best_c5 - 1e-9:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                cur_t = c5_trial
            elif res.x[-1] < cur_t - 1e-9:
                h = _project(res.x[:-1], target)
                cur_t = float(res.x[-1])
                c5_now = _exact_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
        else:
            break
    return best_h, best_c5

def _line_search_pair(h, sink, source, S0, radius=0.15):
    n = len(h)
    one_mh = 1.0 - h
    d = np.zeros(n)
    d[sink] = 1.0
    d[source] = -1.0
    T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = max(-h[sink], h[source] - 1.0, -radius)
    t_hi = min(1.0 - h[sink], h[source], radius)
    if t_hi - t_lo < 1e-13:
        return None, None
    n_pts = 30
    ts = np.linspace(t_lo, t_hi, n_pts)
    obj_vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    bi = int(np.argmin(obj_vals))
    lo = ts[max(0, bi - 2)]
    hi = ts[min(n_pts - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    t_opt = (lo + hi) / 2.0
    return t_opt, f1

def _pair_optimize(h, deadline, radius=0.12, n_pairs=40, beta=25.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        vals = S0 * (2.0 / n)
        M = float(np.max(vals))
        w = np.exp(beta * (vals - M))
        s = float(np.sum(w))
        if s < 1e-30:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= s
        G = np.zeros((len(vals), n))
        for k in range(len(vals)):
            lag = k - (n - 1)
            j_lo = max(0, lag)
            j_hi = min(n - 1, lag + n - 1)
            if j_lo <= j_hi:
                G[k, j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi + 1 - lag])
            j_lo2 = max(0, -lag)
            j_hi2 = min(n - 1, n - 1 - lag)
            if j_lo2 <= j_hi2:
                G[k, j_lo2:j_hi2 + 1] -= h[j_lo2 + lag:j_hi2 + 1 + lag]
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        sources = sort_idx[-min(n_pairs, n // 2):]
        sinks = sort_idx[:min(n_pairs, n // 2)]
        pairs = [(int(sinks[i]), int(sources[i])) for i in range(min(n_pairs, len(sinks), len(sources)))]
        for _ in range(min(10, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_opt, val = _line_search_pair(h, sink, source, S0, radius)
            if t_opt is None:
                continue
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _make_seeds(h0, n, rng):
    seeds = [h0.copy()]
    p1 = h0.copy()
    p1 += 0.05 * np.sin(np.pi * np.arange(n) / n)
    seeds.append(_project(p1, n / 2.0))
    p2 = h0.copy()
    p2 += 0.04 * rng.normal(0.0, 1.0, n)
    seeds.append(_project(p2, n / 2.0))
    p3 = h0.copy()
    p3 += 0.03 * np.cos(2.0 * np.pi * 2.0 * np.arange(n) / n)
    seeds.append(_project(p3, n / 2.0))
    return seeds

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Multi-start gradient descent at n0 (30% time)
    d1 = t_start + budget_s * 0.30
    seeds = _make_seeds(h_init, n0, rng)
    for si, s in enumerate(seeds):
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + (d1 - t_start) / max(1, len(seeds) - si))
        h_g, c5_g = _gradient_descent(s, sub_dl, beta=60.0, max_iter=300, step0=0.015)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        if time.time() < d1:
            h_g2, c5_g2 = _gradient_descent(s, sub_dl, beta=120.0, max_iter=200, step0=0.01)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()

    # Also try different n values
    n_candidates = [n0, n0 - 4, n0 + 4, n0 - 8, n0 + 8, n0 - 12, n0 + 12]
    tried = set()
    for nc in n_candidates:
        if time.time() > d1:
            break
        if nc < 60 or nc > 240 or nc % 2 == 0 and False:
            pass
        if nc < 60 or nc > 240 or nc in tried:
            continue
        tried.add(nc)
        h_r = _resample(best_h, nc)
        sub_dl = min(d1, time.time() + 15.0)
        h_gr, c5_gr = _gradient_descent(h_r, sub_dl, beta=50.0, max_iter=200, step0=0.012)
        if c5_gr < best_c5:
            best_c5 = c5_gr
            best_h = h_gr.copy()

    # Phase 2: NLP solve (25% time)
    d2 = t_start + budget_s * 0.55
    n_curr = len(best_h)
    if time.time() < d2:
        for K_val in [30, 50, 80]:
            if time.time() > d2:
                break
            sub_dl = min(d2, time.time() + budget_s * 0.06)
            h_n, c5_n = _nlp_solve(best_h, sub_dl, K=K_val, outer_iters=30)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()

    # Phase 3: Pair optimization (25% time)
    d3 = t_start + budget_s * 0.80
    n_curr = len(best_h)
    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.10)
        h_p, c5_p = _pair_optimize(best_h, sub_dl, radius=0.12, n_pairs=50, beta=25.0, stale_limit=100, seed=111)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
        # Try nearby n
        for dn in [-4, 4, -8, 8, -12, 12]:
            if time.time() > d3:
                break
            n_try = n_curr + dn
            if n_try < 60 or n_try > 240:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.02)
            h_pr, c5_pr = _pair_optimize(h_r, sub_dl2, radius=0.10, n_pairs=35, beta=22.0, stall_limit=50, seed=200+dn)
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr.copy()

    # Phase 4: NLP + gradient refinement (15% time)
    d4 = deadline - 5
    if time.time() < d4:
        n_curr = len(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_g4, c5_g4 = _gradient_descent(best_h, sub_dl, beta=100.0, max_iter=400, step0=0.008)
        if c5_g4 < best_c5:
            best_c5 = c5_g4
            best_h = h_g4.copy()
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + budget_s * 0.04)
            h_n4, c5_n4 = _nlp_solve(best_h, sub_dl2, K=40, outer_iters=30)
            if c5_n4 < best_c5:
                best_c5 = c5_n4
                best_h = h_n4.copy()

    # Phase 5: Final pair polish (5% time)
    if time.time() < deadline - 2:
        n_curr = len(best_h)
        sub_dl = min(deadline - 2, time.time() + budget_s * 0.03)
        h_f, c5_f = _pair_optimize(best_h, sub_dl, radius=0.06, n_pairs=30, beta=35.0, stall_limit=40, seed=999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection and verification
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5
    return best_h, float(best_c5), int(n_final)
