def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_c5(h, d, dx, alpha_max):
    """Line search along direction d from h, finding alpha that minimizes C5.
    Uses the expansion C(alpha) = C0 + alpha*T - alpha^2*Q."""
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    if alpha_max < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search
    n_grid = 40
    grid = np.linspace(0.0, alpha_max, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        a = grid[i]
        vals[i] = float(np.max(C0 + a * T - a * a * Q)) * dx
    
    best_i = int(np.argmin(vals))
    lo_g = grid[max(0, best_i - 2)]
    hi_g = grid[min(n_grid - 1, best_i + 2)]
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_g - lo_g < 1e-10:
        best_a = grid[best_i]
    else:
        c1 = hi_g - gr * (hi_g - lo_g)
        c2 = lo_g + gr * (hi_g - lo_g)
        f1 = float(np.max(C0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(C0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if hi_g - lo_g < 1e-13:
                break
            if f1 < f2:
                hi_g, c2, f2 = c2, c1, f1
                c1 = hi_g - gr * (hi_g - lo_g)
                f1 = float(np.max(C0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_g, c1, f1 = c1, c2, f2
                c2 = lo_g + gr * (hi_g - lo_g)
                f2 = float(np.max(C0 + c2 * T - c2 * c2 * Q)) * dx
        best_a = (lo_g + hi_g) / 2.0
    
    # Check endpoints
    f_best = float(np.max(C0 + best_a * T - best_a * best_a * Q)) * dx
    f_0 = float(np.max(C0)) * dx
    f_end = float(np.max(C0 + alpha_max * T - alpha_max * alpha_max * Q)) * dx
    
    if f_0 <= f_best and f_0 <= f_end:
        return 0.0, f_0
    elif f_end <= f_best and f_end <= f_0:
        return alpha_max, f_end
    else:
        return best_a, f_best

def _lp_direction(h, corr, radius, n_active=60):
    """Compute LP direction to reduce max of active correlation constraints."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    
    # Find active constraints (top n_active by correlation value)
    max_val = float(corr.max())
    tol = max(1e-10, 0.01 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    if len(active) < n_active:
        # Pad with next highest
        sorted_idx = np.argsort(-corr)
        active = np.union1d(active, sorted_idx[:n_active])
    
    active = active[:n_active]
    
    # Build gradient matrix for active constraints
    # For corr[n-1+j] = sum_{i=0}^{n-j-1} h[i+j]*(1-h[i]):
    #   d/dh[i+j] = (1-h[i]), d/dh[i] = -h[i+j]
    # For corr[n-1-j] = sum_{i=0}^{n-j-1} h[i]*(1-h[i+j]):
    #   d/dh[i] = (1-h[i+j]), d/dh[i+j] = -h[i]
    
    G = np.zeros((len(active), n))
    center = n - 1
    for idx, r in enumerate(active):
        k = int(r) - center
        if k == 0:
            # corr[center] = sum h[i]*(1-h[i]), grad = 1-2h[i]
            G[idx, :] = (1.0 - 2.0 * h) * dx
        elif k > 0:
            # corr[n-1+k] = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
            # grad[i+k] += (1-h[i]), grad[i] += -h[i+k]
            G[idx, k:n] += (1.0 - h[:n-k]) * dx
            G[idx, :n-k] += -h[k:n] * dx
        else:
            j = -k
            # corr[n-1-j] = sum_{i=0}^{n-j-1} h[i]*(1-h[i+j])
            # grad[i] += (1-h[i+j]), grad[i+j] += -h[i]
            G[idx, :n-j] += (1.0 - h[j:n]) * dx
            G[idx, j:n] += -h[:n-j] * dx
    
    # LP: minimize t + eps*sum(|d_i|)
    # s.t. G @ d - t <= 0 (for each active constraint, linearized)
    #      sum(d) = 0
    #      -min(radius, h_i) <= d_i <= min(radius, 1-h_i)
    #      d_i - u_i^+ <= 0, -d_i - u_i^- <= 0 (for penalty)
    
    n_vars_total = n + 1  # d + t (we'll handle |d| with variable bounds)
    # Actually, let's simplify: minimize t subject to G@d <= t, sum(d)=0, box on d
    # Use linprog with c = [0,...,0, 1] for t at the end
    
    n_constr = len(active) + 1  # active + mass
    A_ub = np.hstack([G, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    d_lo = -np.minimum(radius, h) + 1e-12
    d_hi = np.minimum(radius, 1.0 - h) - 1e-12
    bounds = list(zip(d_lo, d_hi))
    bounds.append((-1.0, max_val + 0.1))  # t can be up to current max + small
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            # Remove mean to ensure exact zero sum
            d = d - np.mean(d)
            if np.max(np.abs(d)) < 1e-12:

    return None

def _subgradient_direction(h, corr):
    """Compute negative subgradient of max(corr)."""
    n = len(h)
    max_val = float(corr.max())
    active = np.where(corr >= max_val - 1e-10)[0]
    
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for r in active:
        k = int(r) - center
        if k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[k:] += (1.0 - h[:n-k]) * dx
            g[:n-k] -= h[k:] * dx
        else:
            j = -k
            g[:n-j] += (1.0 - h[j:]) * dx
            g[j:] -= h[:n-j] * dx
    g /= len(active)
    return -g

def _binary_sa(h_bin, deadline, max_iters=200000, temp_init=0.003, seed=42):
    """Simulated annealing on binary h with exact n/2 ones."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = float(cur_corr.max())
    best_h = cur.copy()
    best_max = cur_max
    best_c5 = cur_max * dx
    
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode='full')
        new_max = float(new_corr.max())
        delta = new_max - cur_max
        
        if delta < -1e-14 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_corr = new_corr
            cur_max = new_max
            ones[i] = oz
            zeros[j] = oi
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = cur.copy()
                best_c5 = cur_max * dx
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            best_stall += 1
            T = max(0.0003, T * 0.9995)
    
    return best_h, best_c5

            cur[oi] = 1.0
            cur[oz] = 0.0
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j, new_corr = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_corr = new_corr
            cur_max += best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = cur.copy()
                best_c5 = cur_max * dx
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _lp_polish(h, deadline, radius_init=0.06, max_iter=2000):
    """LP-based continuous optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Try LP direction
        d = _lp_direction(h, corr, radius, n_active=min(80, 2*n-1))
        if d is None:
            # Fallback to subgradient
            d = _subgradient_direction(h, corr)
            d = d - np.mean(d)
        
        # Compute alpha_max
        pos = d > 1e-12
        neg = d < -1e-12
        alpha_max = radius
        if np.any(pos):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
        
        if alpha_max < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search_c5(h, d, dx, alpha_max)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.2, 5.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _block_swap(h_bin, deadline, max_iters=10000, seed=77):
    """Try swapping blocks of equal sum."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(6, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        block1 = cur[i:i+length].copy()
        block2 = cur[j:j+length].copy()
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue
        cur[i:i+length] = block2
        cur[j:j+length] = block1
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            cur[i:i+length] = block1
            cur[j:j+length] = block2
            stall += 1
    
    return best_h, best_c5
