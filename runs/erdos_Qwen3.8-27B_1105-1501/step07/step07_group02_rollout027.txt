import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smooth_weights(S0, beta=20.0):
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    shifted = np.clip(beta * (S0 - M), -50.0, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _grad_weighted(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        if w[s] < 1e-10:
            continue
        ell = s - (n - 1)
        if ell >= 0:
            m = n - ell
            grad[:m] += w[s] * dx * (1.0 - h[ell:])
            grad[ell:] -= w[s] * dx * h[:m]
        else:
            p = -ell
            m = n - p
            grad[:m] += w[s] * dx * (1.0 - h[p:])
            grad[p:] -= w[s] * dx * h[:m]
    return grad

def _golden_search(S0, T, C, t_lo, t_hi, max_iter=45):
    """Minimize max_k(S0[k] + t*T[k] - t^2*C[k]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-15:
        return t_lo, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(max_iter):
        if t_hi - t_lo < 1e-13:
            break
        if f1 < f2:
            t_hi = c2
            c2 = c1
            f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            t_lo = c1
            c1 = c2
            f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _dense_descent(h, deadline, beta0=10.0, beta_rate=0.05, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    it = 0
    beta = beta0
    
    while time.time() < deadline and stall < stall_limit:
        it += 1
        beta = min(beta0 + beta_rate * it, 100.0)
        
        w = _smooth_weights(S0, beta)
        g = _grad_weighted(h, w)
        
        # Sum-zero projection
        d = g - np.mean(g)
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-12:
            stall += 5
            continue
        d = d / norm_d
        
        # Feasible step
        pos = d > 1e-12
        neg = d < -1e-12
        t_max = 1.0
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_max = min(t_max, float(np.min(-h[neg] / d[neg])))
        
        if t_max < 1e-10:
            stall += 1
            continue
        
        one_mh = 1.0 - h
        T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        alpha, val = _golden_search(S0, T, C, 0.0, t_max)
        
        if val < cur_c5 * (1.0 - 1e-10):
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                S0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _grad_weighted(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        if w[s] < 1e-10:
            continue
        ell = s - (n - 1)
        if ell >= 0:
            # f = sum_{i=0}^{n-1-ell} h[i] * (1-h[i+ell])
            # dc/dh[k] = (1-h[k+ell]) * I(k+ell < n) - h[k-ell] * I(k >= ell)
            m = n - ell
            grad[:m] += w[s] * dx * (1.0 - h[ell:])
            grad[ell:] -= w[s] * dx * h[:m]
        else:
            p = -ell
            m = n - p
            # f = sum_{i=0}^{n-1-p} h[i+p] * (1-h[i])
            # dc/dh[k] = (1-h[k-p]) * I(k >= p) - h[k+p] * I(k+p < n)
            grad[p:] += w[s] * dx * (1.0 - h[:m])
            grad[:m] -= w[s] * dx * h[p:]
    return grad
