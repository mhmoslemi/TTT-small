import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _smooth_weights(R, beta=20.0):
    M = float(np.max(R))
    if M < 1e-15:
        return np.ones(len(R)) / len(R)
    shifted = R - M  # all <= 0
    w = np.exp(beta * shifted)  # since shifted <= 0, this is exp(negative) <= 1
    s = float(np.sum(w))
    if s < 1e-30:
        # Fallback: uniform over top-k
        top_k = max(1, len(R) // 10)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _line_search_exact(R0, A, B, t_lo, t_hi):
    """Minimize max_r(R0[r] + t*A[r] - t^2*B[r]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R0))
    
    # Grid search first
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine around best
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    
    if hi - lo > 1e-10:
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))

        opt_val = float(np.max(R0 + opt_t * A - opt_t * opt_t * B))

        opt_t = float(ts[best_idx])
        opt_val = float(vals[best_idx])

    val_hi = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))

    if best_val == val_hi:

def _dense_descent(h, deadline, n_pairs=40, beta_init=10.0, stall_limit=80, seed=42):

    best_h = h.copy()

            break

            beta *= 1.5

            stall += 1
            beta *= 1.3
            continue

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 350:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _block_descent(h, deadline, beta=15.0, n_blocks=8, seed=7):
    """Try block-based coordinated moves."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    stall = 0

    while time.time() < deadline and stall < 50:
        w = _smooth_weights(R0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w

        # Create blocks
        block_size = max(2, n // n_blocks)
        n_b = n // block_size

        # For each pair of blocks, try shifting mass from one to the other
        best_move = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h

        for _ in range(min(30, 2 * n_b)):
            bi = int(rng.integers(0, n_b))
            bj = int(rng.integers(0, n_b))
            if bi == bj:
                continue
            # Block i will receive mass, block j will lose mass
            i_start = bi * block_size
            i_end = min(n, (bi + 1) * block_size)
            j_start = bj * block_size
            j_end = min(n, (bj + 1) * block_size)

            # Direction: add to block i, subtract from block j
            d = np.zeros(n)
            d[i_start:i_end] = 1.0 / block_size
            d[j_start:j_end] = -1.0 / block_size

            # Feasible range
            pos_mask = d > 0
            neg_mask = d < 0
            t_hi = 1.0
            if np.any(pos_mask):
                t_hi = min(t_hi, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
            t_lo = -1.0
            if np.any(neg_mask):
                t_lo = max(t_lo, float(np.min(h[neg_mask] / (-d[neg_mask]))))
            t_hi = min(t_hi, 0.5)
            t_lo = max(t_lo, -0.5)
            if t_hi - t_lo < 1e-12:
                continue

            A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            B = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search_exact(R0, A, B, t_lo, t_hi)
            h_try = h + t_opt * d
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_move = h_try

        if best_move is not None:
            h = best_move
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            R0 = np.correlate(h, 1.0 - h, mode='full')
            one_mh = 1.0 - h
            stall = 0
        else:
            stall += 1
            beta *= 1.3

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense gradient descent (15% time)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _dense_descent(h_init, d1, beta_init=8.0, stall_limit=50, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair optimization (10%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _pair_optimize(best_h, d2, radius=0.18, n_pairs=55, beta=18.0, stall_limit=100, seed=2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Block descent (8%)
    d3 = t0 + budget_s * 0.33
    h3, c5_3 = _block_descent(best_h, d3, beta=12.0, n_blocks=6, seed=3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-n exploration (17%)
    d4 = t0 + budget_s * 0.50
    candidate_ns = []
    for dn in [-20, -16, -12, -8, -4, 4, 8, 12, 16, 20, 24, 28]:
        n_try = n0 + dn
        if 70 <= n_try <= 280:
            candidate_ns.append(n_try)

    for n_try in candidate_ns:
        if time.time() > d4:
            break
        remaining = d4 - time.time()
        if remaining < 3.0:
            break
        sub_dl = time.time() + max(2.0, remaining * 0.07)
        sub_dl = min(sub_dl, d4)
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _dense_descent(h_r, sub_dl, beta_init=12.0, stall_limit=35, seed=100 + n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Binary diversification (20%)
    d5 = t0 + budget_s * 0.70
    n_curr = len(best_h)

    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=45000, T_init=0.004, seed=777)
        sub_dl2 = min(d5, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _pair_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            n_curr = len(best_h)

        # Random binary starts
        rng_rt = np.random.default_rng(5000)
        for attempt in range(4):
            if time.time() > d5:
                break
            h_rand = _random_binary(n_curr, rng_rt)
            sub_dl3 = min(d5, time.time() + budget_s * 0.015)
            h_rand, c5_rand = _binary_sa(h_rand, sub_dl3, max_iter=20000, T_init=0.006, seed=6000 + attempt)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand.copy()

    # Phase 6: Polish with continuous optimization (15%)
    d6 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    if time.time() < d6:
        h_pol = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.05)
        h_pol, c5_pol = _pair_optimize(h_pol, sub_dl, radius=0.12, n_pairs=55, beta=22.0, stall_limit=80, seed=555)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

        if time.time() < d6:
            sub_dl2 = min(d6, time.time() + budget_s * 0.04)
            h_pol2, c5_pol2 = _dense_descent(best_h, sub_dl2, beta_init=18.0, stall_limit=45, seed=666)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

        if time.time() < d6:
            sub_dl3 = min(d6, time.time() + budget_s * 0.03)
            h_pol3, c5_pol3 = _block_descent(best_h, sub_dl3, beta=20.0, n_blocks=8, seed=777)
            if c5_pol3 < best_c5:
                best_c5 = c5_pol3
                best_h = h_pol3.copy()

    # Phase 7: Final intensive refinement (12%)
    d7 = deadline - 3
    n_curr = len(best_h)
    h7 = _project(best_h.copy(), n_curr / 2.0)

    sub_dl = min(d7, time.time() + budget_s * 0.04)
    h7, c5_7 = _dense_descent(h7, sub_dl, beta_init=25.0, stall_limit=35, seed=888)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    if time.time() < d7:
        sub_dl2 = min(d7, time.time() + budget_s * 0.04)
        h7b, c5_7b = _pair_optimize(best_h, sub_dl2, radius=0.08, n_pairs=50, beta=30.0, stall_limit=70, seed=999)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Try nearby n for final squeeze
    n_curr = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12, 16, -16]:
        if time.time() > d7:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 300:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl3 = min(d7, time.time() + budget_s * 0.01)
        h_r, c5_r = _pair_optimize(h_r, sub_dl3, radius=0.10, n_pairs=40, beta=28.0, stall_limit=40, seed=444 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
