            best_val = val_opt
            best_alpha = opt

def _lp_step(h, rho, K, target):
    """Solve LP for one optimization step. Returns (lp_solution, success)."""
    n = len(h)
    raw = _raw_profile(h)
    max_raw = float(np.max(raw))
    
    # Select top K lags
    top_idx = np.argsort(-raw)[:K]
    
    # Compute gradients for these lags
    G = _grad_for_lags(h, top_idx)  # (n, K)
    
    # Compute c_k = sum_i G[i,k] * h[i] - raw[top_idx[k]]
    c = G.T @ h - raw[top_idx]  # shape (K,)
    
    # LP: minimize t
    # s.t. G^T x - t <= c (K constraints)
    #      sum x = target
    #      lb_i <= x_i <= ub_i
    #      x_i - t is in variable array: [x_0..x_{n-1}, t]
    
    # Variable order: x_0, ..., x_{n-1}, t
    # Objective: minimize [0, ..., 0, 1] @ vars
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # For each lag k: sum_i G[i,k] * x_i - t <= c_k
    # => G[:,k] @ x - 1 * t <= c_k
    A_ub = np.zeros((K, n + 1))
    for k in range(K):
        A_ub[k, :n] = G[:, k]
        A_ub[k, n] = -1.0
    b_ub = c.copy()
    
    # Equality: sum x = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: max(0, h_i - rho) <= x_i <= min(1, h_i + rho), and -inf <= t <= inf
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    
    if res.success:
        x_lp = res.x[:n]
        # Verify feasibility
        if abs(float(np.sum(x_lp)) - target) > 1e-6:
            # Project to fix sum constraint
            x_lp = _project(x_lp, target)
        if np.min(x_lp) < -1e-9 or np.max(x_lp) > 1.0 + 1e-9:
            x_lp = np.clip(x_lp, 0.0, 1.0)
            x_lp = _project(x_lp, target)
        return x_lp, True
    return h, False

def _gradient_direction(h, raw, K):
    """Compute weighted gradient direction from top K lags."""
    n = len(h)
    top_idx = np.argsort(-raw)[:K]
    # Use softmax weights
    shifted = raw[top_idx] - raw[top_idx[0]]
    weights = np.exp(20.0 * shifted)
    weights /= weights.sum()
    
    G = _grad_for_lags(h, top_idx)
    g = G @ weights
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm > 1e-12:
        g /= norm
    return g

def _pair_swap_cont(h, deadline, n_pairs, radius, seed):
    """Continuous pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Try swap: decrease h[i], increase h[j] (and vice versa)
        for sign in [1.0, -1.0]:
            d = np.zeros(n)
            d[i] = sign
            d[j] = -sign
            raw = _raw_profile(h)
            alpha, c_try = _line_search(h, d, raw, dx, radius)
            if alpha < 1e-10:
                continue
            h_new = h + alpha * d

            if actual < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                break
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):

    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        raw = _raw_profile(h)
        for sign in [1.0, -1.0]:
            d = np.zeros(n)
            d[i] = sign
            d[j] = -sign
            alpha, c_try = _line_search(h, d, raw, dx, radius)
            if alpha < 1e-10:
                continue
            h_new = _project(h + alpha * d, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                break
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter, T_init):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9994
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 / 2.0
    h = _project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick gradient warmup (4%)
    d1 = t0 + budget_s * 0.04
    while time.time() < d1:
        raw = _raw_profile(h)
        g = _gradient_dir(h, raw, K=20, weight_beta=20.0)
        if np.linalg.norm(g) < 1e-12:
            break
        alpha, c_try = _line_search(h, g, raw, 2.0/n0, max_alpha=0.15)
        if alpha < 1e-10:
            break
        h_new = _project(h + alpha * g, target)
        actual = _exact_c5(h_new)
        if actual < _exact_c5(h) - 1e-13:
            h = h_new
            if actual < best_c5:
                best_c5 = actual
                best_h = h.copy()
        else:
            break

    # Phase 2: LP-based main optimization (62%)
    d2 = t0 + budget_s * 0.66
    rho = 0.35
    K_schedule = [15, 20, 25, 30, 40, 50, 60, 80]
    K_phase = 0
    stall = 0
    iter_count = 0

    while time.time() < d2:
        iter_count += 1
        K = K_schedule[min(K_phase, len(K_schedule) - 1)]

        # LP step
        x_lp, success = _lp_solve(h, rho, K, target)
        if success and not np.allclose(x_lp, h, atol=1e-9):
            c_lp = _exact_c5(x_lp)
            if c_lp < best_c5:
                best_c5 = c_lp
                best_h = x_lp.copy()
                h = x_lp.copy()
                stall = 0
                rho = min(rho * 1.08, 0.5)
                continue

            # Line search along LP direction
            dd = x_lp - h
            dd_norm = np.linalg.norm(dd)
            if dd_norm > 1e-12:
                dd_scaled = dd  # keep as-is, trust region already bounded
                raw = _raw_profile(h)
                alpha, c_try = _line_search(h, dd_scaled, raw, 2.0/n0, max_alpha=1.0)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * dd_scaled, target)
                    actual = _exact_c5(h_new)
                    if actual < _exact_c5(h) - 1e-14:
                        h = h_new
                        if actual < best_c5:
                            best_c5 = actual
                            best_h = h.copy()
                        stall = 0
                        rho = min(rho * 1.03, 0.45)
                        continue

        # Gradient direction fallback
        raw = _raw_profile(h)
        g = _gradient_dir(h, raw, K=min(K, 30), weight_beta=25.0)
        if np.linalg.norm(g) > 1e-12:
            alpha, c_try = _line_search(h, g, raw, 2.0/n0, max_alpha=rho * 0.5)
            if alpha > 1e-10:
                h_new = _project(h + alpha * g, target)
                actual = _exact_c5(h_new)
                if actual < _exact_c5(h) - 1e-14:
                    h = h_new
                    if actual < best_c5:
                        best_c5 = actual
                        best_h = h.copy()
                    stall = 0
                    continue

        # Stalling handling
        stall += 1
        if stall % 8 == 0:
            rho *= 0.65
            if rho < 0.04:
                rho = 0.30
                K_phase = (K_phase + 1) % len(K_schedule)
        if stall > 40:
            # Perturb from best (not current) to escape basin
            h = _smooth_perturb(best_h, 0.08, rng)
            h = _project(h, target)
            stall = 0
            rho = 0.30

    # Phase 3: Pair swap + binary refinement (20%)
    d3 = t0 + budget_s * 0.86
    # Continuous pair swap
    h_pc = _project(best_h.copy(), target)
    h_pc, c5_pc = _pair_swap(h_pc, min(d3, time.time() + budget_s * 0.08), n_pairs=6000, radius=0.25, seed=77)
    if c5_pc < best_c5:
        best_c5 = c5_pc
        best_h = h_pc.copy()

    # Binary SA
    if time.time() < d3 - 5:
        h_bin = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.06)
        h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()

    # Polish binary with gradient
    if time.time() < d3 - 3:
        h_pol = _project(best_h.copy(), target)
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        while time.time() < sub_dl:
            raw = _raw_profile(h_pol)
            g = _gradient_dir(h_pol, raw, K=20, weight_beta=25.0)
            if np.linalg.norm(g) < 1e-12:
                break
            alpha, c_try = _line_search(h_pol, g, raw, 2.0/n0, max_alpha=0.1)
            if alpha < 1e-10:
                break
            h_new = _project(h_pol + alpha * g, target)
            actual = _exact_c5(h_new)
            if actual < _exact_c5(h_pol) - 1e-14:
                h_pol = h_new
                if actual < best_c5:
                    best_c5 = actual
                    best_h = h_pol.copy()
            else:
                break

    # Phase 4: Final LP polish (12%)
    d4 = deadline
    h = _project(best_h.copy(), target)
    rho = 0.12
    stall2 = 0
    while time.time() < d4 - 2:
        for K in [20, 30, 40, 50]:
            if time.time() > d4 - 2:
                break
            x_lp, success = _lp_solve(h, rho, K, target)
            if success and not np.allclose(x_lp, h, atol=1e-9):
                c_lp = _exact_c5(x_lp)
                if c_lp < best_c5:
                    best_c5 = c_lp
                    best_h = x_lp.copy()
                    h = x_lp.copy()
                    stall2 = 0
                    break
                dd = x_lp - h
                raw = _raw_profile(h)
                alpha, c_try = _line_search(h, dd, raw, 2.0/n0, max_alpha=1.0)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * dd, target)
                    actual = _exact_c5(h_new)
                    if actual < _exact_c5(h) - 1e-14:
                        h = h_new
                        if actual < best_c5:
                            best_c5 = actual
                            best_h = h.copy()
                        stall2 = 0
                        break
        else:
            stall2 += 1
            if stall2 % 4 == 0:
                rho *= 0.7
                if rho < 0.015:
                    break
            if stall2 > 20:
                break

    # Final: small pair swaps
    if deadline - time.time() > 5:
        h_fs = _project(best_h.copy(), target)
        h_fs, c5_fs = _pair_swap(h_fs, deadline, n_pairs=2000, radius=0.12, seed=1234)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    best_h = _project(best_h, target)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
