import numpy as np
import time

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * (2.0 / n))

def _lse_val(h, tau):
    dx = 2.0 / len(h)
    s = _scores(h) * dx
    m = float(s.max())
    u = (s - m) / tau
    mx = u.max()
    z = np.exp(u - mx)
    zsum = z.sum()
    if zsum < 1e-300:
        return m
    return float(m + tau * (mx + np.log(zsum)))

def _lse_probs(h, tau):
    dx = 2.0 / len(h)
    s = _scores(h) * dx
    m = float(s.max())
    u = (s - m) / tau
    mx = u.max()
    z = np.exp(u - mx)
    zsum = z.sum()
    if zsum < 1e-300:
        p = np.zeros(len(s))
        mask = s == m
        p[mask] = 1.0 / mask.sum()
        return p
    return z / zsum

def _lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    p = _lse_probs(h, tau)
    grad = np.zeros(n)
    mid = n - 1
    grad += dx * p[mid] * (1.0 - 2.0 * h)
    for d in range(1, n):
        pd = p[mid + d] * dx
        if pd > 1e-16:
            m = n - d
            grad[:m] += pd * (1.0 - h[d:])
            grad[d:] += -pd * h[:m]
        pn = p[mid - d] * dx
        if pn > 1e-16:
            m = n - d
            grad[d:] += pn * (1.0 - h[:m])
            grad[:m] += -pn * h[d:]
    return grad

def _exact_line_search(h1, h2, n_grid=30):

    target = n / 2.0
    best_f = _exact_c5(h1)
    best_t = 0.0

        ht = _project_sum(ht, target)
        ft = _exact_c5(ht)
        if ft < best_f:
            best_f = ft
            best_t = t
    lo = max(0.0, best_t - 2.0 / n_grid)
    hi = min(1.0, best_t + 2.0 / n_grid)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project_sum((1-c1)*h1 + c1*h2, target))
    f2 = _exact_c5(_project_sum((1-c2)*h1 + c2*h2, target))
    for _ in range(30):
        if hi - lo < 1e-8:

        if f1 < f2:

            c2 = c1
            f2 = f1

        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)

def _pair_polish(h, deadline, max_rounds=20):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

    # Phase 2: Hybrid refinement (20%)
    d2 = t0 + budget_s * 0.65
    h2, c5_2 = _refine_hybrid(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Dimension exploration (15%)
    d3 = t0 + budget_s * 0.80
    n_cur = len(best_h)
    n_tries = [n_cur - 8, n_cur - 4, n_cur + 4, n_cur + 8, n_cur - 12, n_cur + 12]
    for dn in n_tries:
        if time.time() > d3:
            break
        n_try = n_cur + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _anneal_lse(h_r, sub_dl, tau_init=0.10, tau_min=0.01,
                                n_iter_per_tau=100, seed=n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 4: Basin hopping (12%)
    d4 = t0 + budget_s * 0.92
    h4, c5_4 = _basin_hop(best_h, d4, rng, scale=0.04, max_hops=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final polish (8%)
    h5, c5_5 = _refine_hybrid(best_h, deadline, rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final: ensure feasibility and exact evaluation
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
