def _lag_gradient(h, t):
    """Compute gradient of correlation at output index t with respect to h."""
    n = len(h)
    g = np.zeros(n)
    lag = t - (n - 1)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += 1.0 - h[lag:lag + L]
            g[lag:lag + L] -= h[:L]
    else:
        s = -lag
        L = n - s
        if L > 0:
            g[s:s + L] += 1.0 - h[:L]
            g[:L] -= h[s:s + L]
    return g

def _agg_gradient(h, beta, top_k=10):
    """Compute weighted aggregate gradient focusing on top lags."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    
    # Use top-k weighted approach
    top_idx = np.argsort(-S)[:top_k]
    weights = np.zeros(len(S))
    for rank, idx in enumerate(top_idx):
        weights[idx] = math.exp(-beta * (sc[idx] - mx))
    
    g = np.zeros(n)
    for t in range(len(S)):
        if weights[t] > 1e-10:
            g += weights[t] * _lag_gradient(h, t)
    g -= np.mean(g)
    return -g  # descent direction

def _top_lag_dirs(h, top_k=8):
    """Generate directions from individual top-lag gradients."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-S)[:top_k]
    dirs = []
    for t in top_idx:
        g = _lag_gradient(h, int(t))
        g -= np.mean(g)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    return dirs

def _lp_direction(h, top_k=10, r=0.1):
    """Solve LP to find direction that reduces top K lags simultaneously."""
    try:
        from scipy.optimize import linprog
    except:
        return None
    
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-S)[:top_k]
    
    # Build constraint matrix: for each active lag, g_l^T s <= -tau
    # Variables: s (n vars) + tau (1 var) = n+1 vars
    # Constraints:
    #   A_eq: [1...1, 0] @ [s, tau] = 0  (sum s = 0)
    #   A_ub: [g_l, 1] @ [s, tau] <= 0 for each l (g_l^T s + tau <= 0)
    #   Bounds: -r*h_i <= s_i <= r*(1-h_i), tau >= 0
    
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    for i, t in enumerate(top_idx):
        g = _lag_gradient(h, int(t))
        A_ub[i, :n] = g
        A_ub[i, n] = 1.0  # +tau
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = -r * h[i]
        hi = r * (1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # tau >= 0
    
    # Objective: minimize -tau (i.e., maximize tau)
    c = np.zeros(n + 1)
    c[n] = -1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.func < -1e-10:
            s = res.x[:n]
            s -= np.mean(s)  # ensure zero mean (numerical fix)
            if np.linalg.norm(s) > 1e-10:
                return s
    except:
        pass
    return None

        idx = rng.integers(0, n, 3)
        if len(set(idx)) < 3:
            stall += 1
            continue
        i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
        # Try all 6 permutations of (from, to1, to2) with equal split
        improved = False
        for perm in [(i,j,k), (i,k,j), (j,i,k), (j,k,i), (k,i,j), (k,j,i)]:
            src, dst1, dst2 = perm
            if h[src] < 0.01:
                continue
            # Move epsilon from src to dst1 and dst2
            for eps in [0.02, 0.05, 0.1]:

                        cur = c5a

                            best_h = h.copy()
                        stall = 0
                        improved = True
                        break

                break

            stall += 1

def _lag_gradient(h, t):

        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _focused_lag_descent(h, deadline, n_tries=500):
    """Focus on the single worst lag and try to reduce it with random directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(5555)
    for _ in range(n_tries):
        if time.time() > deadline - 0.3:
            break
        S = np.correlate(h, 1.0 - h, mode='full')
        worst_t = int(np.argmax(S))
        g = _lag_gradient(h, worst_t)
        g -= np.mean(g)
        nrm = np.linalg.norm(g)
        if nrm < 1e-12:
            continue
        d = g / nrm
        for rad in [0.05, 0.1, 0.2, 0.03]:
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < cur - 1e-14:
                h_new = h + a * d
                h_new = _project(h_new, target)
                c5a = _exact_c5(h_new)
                if c5a < cur - 1e-14:
                    h = h_new
                    cur = c5a
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Local opt with top-lag + LP directions (18%)
    d1 = t0 + budget_s * 0.18
    h1, c1 = _local_opt(h_init, d1, r0=0.20, stall_lim=55, use_lp=True)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Focused worst-lag descent (7%)
    d2 = t0 + budget_s * 0.25
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _focused_lag_descent(h2, d2, n_tries=800)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair optimization (8%)
    d3 = t0 + budget_s * 0.33
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=35000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: More local opt (12%)
    d4 = t0 + budget_s * 0.45
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    h4, c4 = _local_opt(h4, d4, r0=0.15, stall_lim=50, use_lp=True)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (22%)
    d5 = t0 + budget_s * 0.67
    nc = len(best_h)
    offsets = list(range(-12, 13, 2)) + [-14, 14, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5 - 2:
            break
        n_try = nc + dn
        if n_try < 100 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.018)
        hr, cr = _local_opt(hr, sd, r0=0.15, stall_lim=30, use_lp=True)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
        elif cr < best_c + 0.003 and time.time() < d5 - 3:
            sd2 = min(d5, time.time() + budget_s * 0.007)
            hr, cr2 = _pair_opt(hr, sd2, max_it=4000)
            if cr2 < best_c:
                best_c = cr2
                best_h = hr.copy()
                nc = n_try

    # Phase 6: Intensive alternation local+pair (14%)
    d6 = t0 + budget_s * 0.81
    for r_init in [0.14, 0.07, 0.035]:
        if time.time() > d6 - 3:
            break
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=45, use_lp=True)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _project(best_h.copy(), len(best_h) / 2.0)
            sd = min(d6, time.time() + budget_s * 0.025)
            h6, c6 = _pair_opt(h6, sd, max_it=8000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Basin hopping (9%)
    d7 = deadline
    scale = 0.035
    for _ in range(100):
        if time.time() > d7 - 1:
            break
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.15 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.005)
        ht, ct = _local_opt(hp, sd, r0=0.06, stall_lim=12, use_lp=False)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.09, scale * 1.07)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.025, stall_lim=20, use_lp=True)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    assert final_c <= best_c + 1e-10
    return best_h, final_c, len(best_h)
