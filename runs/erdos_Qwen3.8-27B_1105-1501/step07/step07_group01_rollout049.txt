```python
    # grad[m] = (1/alpha) * sum_s w[n-1+s] * ((1-h[m+s]) - h[m-s]) for valid indices
    # = (1/alpha) * (sum_s w[n-1+s] - sum_s w[n-1+s]*h[m+s] - sum_s w[n-1+s]*h[m-s]) for valid
    # The three terms are:
    # T1[m] = sum of w[k] for k in [n-1-m, 2n-2-m] (window sum)
    # T2[m] = sum of w[k]*h[k-(n-1)+m] for k in [n-1-m, 2n-2-m]
    # T3[m] = sum of w[k]*h[k-(n-1)-m] for k in [n-1-m, 2n-2-m]
    
    # T1: sliding window sum of w
    w_ext = np.concatenate([[0], w, [0]])  # length 2n+1
    # T1[m] = sum(w[n-1-m : 2n-1-m+1]) = cumulative sum difference
    cs = np.cumsum(w)
    # T1[m] = cs[2n-2-m] - cs[n-2-m] (for n-1-m >= 0)
    # This is getting index-heavy. Let me just loop over m.
    
    grad = np.zeros(n)
    for m in range(n):
        total = 0.0
        for s in range(-(n-1), n):
            k = n - 1 + s
            val = 0.0
            if 0 <= m + s < n:
                val += (1.0 - h[m+s])
            if 0 <= m - s < n:
                val += (-h[m-s])
            total += w[k] * val
        grad[m] = total / alpha
    return grad
```
