`np.correlate(a, v, mode='full')` computes:
`c[k] = sum_n a[n+k] * conj(v[n])` for the full cross-correlation.

Actually, `np.correlate(a, v, mode='full')[k]` = sum over valid n of `a[n] * v[n - (k - (len(v)-1))]`.

Let me just use the standard: for h of length n, `np.correlate(h, 1-h, mode='full')` gives a vector of length 2n-1 where element at index m corresponds to lag `m - (n-1)`.

For the gradient: d/dh_i of R_m where R_m = sum_valid h[j] * (1-h[j+lag]):
- If i appears in the h[j] factor: contribution is (1-h[j+lag])
- If i appears in the (1-h[j+lag]) factor: contribution is -h[j-lag]

Actually, let me just compute it numerically as the strategy suggests:
G[:, i] = correlate(e_i, 1-h, 'full') - correlate(h, e_i, 'full')

where e_i is the unit vector.

Let me think about the LP more carefully. We want to find a direction u (with sum u_i = 0, 0 <= h_i + t*u_i <= 1 for some t > 0) that reduces the active peaks.

For active peaks at indices A, we want:
G[a, :] @ u < 0 for all a in A (first-order decrease)

The LP:
maximize beta
subject to:
  G[a,:] @ u + beta <= 0 for a in A (i.e., u reduces peak a by at least beta)
  sum(u) = 0
  -1 <= u_i <= 1 (normalization)

Actually, to make it a proper LP, use split variables:
u_i = p_i - n_i where p_i >= 0, n_i >= 0
sum(p_i) + sum(n_i) <= 1 (normalization, or = 1 for unit direction)
sum(p_i) - sum(n_i) = 0 (balance)

Then G[a,:] @ (p - n) + beta <= 0

minimize -beta
subject to:
  sum_i G[a,i]*p_i - sum_i G[a,i]*n_i + beta <= 0 for a in A
  sum(p) - sum(n) = 0
  sum(p) + sum(n) = 1 (or <= 1)
  0 <= p_i <= 1, 0 <= n_i <= 1
  beta free (or bounded)

This is a standard LP that scipy.optimize.linprog can solve.

    #   sum(p) + sum(n) <= 1
    #   0 <= p_i, n_i <= 1

    c = np.zeros(n_vars)

        b_ub[k] = 0.0

    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)

The main differentiator is the LP direction. Let me think about whether the LP formulation is correct and will actually produce useful directions.

The issue: if all active peaks have the same value M and we want to reduce them all, the LP is finding a direction that reduces the weighted sum. But the objective is the MAX, not the sum. So reducing the sum doesn't necessarily reduce the max.

A better approach: focus on the top 1-3 peaks only, and find a direction that reduces them. If the top peak is at index m*, then we want G[m*, :] @ u < 0. The LP with just 1 constraint is trivial: u = -G[m*, :] (normalized), and the line search finds the best t.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    one_mh = 1.0 - h
    G = np.empty((L, n))
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        G[:, i] = np.correlate(e, one_mh, mode='full') - np.correlate(h, e, mode='full')
    return G

def _line_search_quad(R0, A, B, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R0))
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(R0 + opt_t * A - opt_t * opt_t * B))
    e1 = float(np.max(R0 + t_lo * A - t_lo * t_lo * B))
    e2 = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))
    best_val = min(opt_val, float(vals[best_idx]), e1, e2)
    if best_val == opt_val:
        return opt_t, best_val
    if best_val == float(vals[best_idx]):
        return float(ts[best_idx]), best_val
    if best_val == e1:
        return t_lo, best_val
    return t_hi, best_val

def _try_direction(h, d, R0, t_cap=0.5):

    t_lo = -t_cap
    t_hi = t_cap
    for i in range(n):
        di = d[i]
        if di > 1e-12:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-12:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)

    if t_hi - t_lo < 1e-12:
        return None
    one_mh = 1.0 - h
    A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    t_opt, _ = _line_search_quad(R0, A, B, t_lo, t_hi)
    if abs(t_opt) < 1e-12:
        return None

def _lp_direction(G, R0, peak_count=6):
    n = G.shape[1]
    M = float(np.max(R0))
    idx = np.argsort(R0)[-peak_count:]
    idx = np.sort(idx.astype(int))
    n_vars = 2 * n + 1
    c = np.zeros(n_vars)
    c[-1] = -1.0
    n_constraints = len(idx) + 1
    A_ub = np.zeros((n_constraints, n_vars))
    b_ub = np.zeros(n_constraints)
    for k, a in enumerate(idx):
        A_ub[k, :n] = G[a, :]
        A_ub[k, n:2 * n] = -G[a, :]
        A_ub[k, 2 * n] = 1.0
        b_ub[k] = 0.0
    A_ub[len(idx), :n] = 1.0
    A_ub[len(idx), n:2 * n] = 1.0
    b_ub[len(idx)] = 1.0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    b_eq = np.zeros(1)
    bounds = [(0.0, 1.0)] * (2 * n) + [(-5.0, 5.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n] - res.x[n:2 * n]
            norm = float(np.sum(np.abs(u)))
            if norm > 1e-10:
                u = u / norm
                u = u - np.mean(u)
                return u
    except Exception:
        pass
    return None

def _steepest_single(R0, G):

    u = -G[m_star, :]
    u = u - np.mean(u)
    norm = float(np.sum(np.abs(u)))
    if norm < 1e-12:
        return None
    return u / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h6, c5_6 = _local_descent(h6, deadline - 1.0, peak_count=12, t_cap=0.15, n_pairs=50, seed=1234)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final repair
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
