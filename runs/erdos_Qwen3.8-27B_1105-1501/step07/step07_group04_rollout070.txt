        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=vb, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    d = res.x[:n].copy()
    d[np.abs(d) < 1e-13] = 0.0
    if float(np.sum(np.abs(d))) < 1e-10:
        return None
    return d

def _corr_scaled_val(h, r_idx, dx):
    n = len(h)
    c = _corr_vec(h)
    return float(c[r_idx] * dx)

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _line_search_quad(h, d, dx, radius_max=1.0):
    n = len(h)
    C0 = _corr_vec(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    alpha_max = radius_max
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    if alpha_max < 1e-13:
        return 0.0, float(np.max(C0) * dx)
    n_grid = 50
    grid = np.linspace(0.0, alpha_max, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        t = grid[i]
        vals[i] = float(np.max(C0 + t * T - t * t * Q)) * dx
    best_i = int(np.argmin(vals))
    lo = grid[max(0, best_i - 1)]
    hi = grid[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(C0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(C0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(C0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(C0 + c2 * T - c2 * c2 * Q)) * dx
    alpha_opt = (lo + hi) / 2.0
    val_opt = float(np.max(C0 + alpha_opt * T - alpha_opt * alpha_opt * Q)) * dx
    val_0 = float(np.max(C0) * dx)
    val_end = float(np.max(C0 + alpha_max * T - alpha_max * alpha_max * Q)) * dx
    candidates = [(0.0, val_0), (alpha_opt, val_opt), (alpha_max, val_end)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _subgrad_fallback(h, corr, dx):
    n = len(h)
    center = n - 1
    cur_max = float(corr.max()) * dx
    tol = max(0.01, 0.15 * cur_max)
    active = np.where(corr * dx >= cur_max - tol)[0]
    g = np.zeros(n)
    for r in active:
        k = int(r) - center
        g += _grad_row(h, k, n) * dx
    g /= len(active)
    d = -g
    d -= np.mean(d)
    return d

def _local_lp_optimize(h_start, deadline, radius_init=0.08, max_iter=2000):
    n = len(h_start)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_start.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    lp_fail_count = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if radius < 1e-6:
            break
        corr = _corr_vec(h)
        corr_s = corr * dx
        active, cur_max_s = _select_active(corr_s, n, max_count=50)
        d = None
        if len(active) >= 2:
            d = _lp_direction(h, active, cur, radius, dx)
        if d is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                d = _subgrad_fallback(h, corr, dx)
            else:
                radius *= 0.7
                stall += 1
                continue
        lp_fail_count = 0
        alpha, new_c5 = _line_search_quad(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur - 1e-13:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.30)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_eval(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr

def _binary_targeted(h_bin, deadline, max_iters=100000, seed=42, max_trials=100):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_eval(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        max_idx = int(np.argmax(cur_corr))
        best_delta = 0.0
        best_move = None
        n_try = min(max_trials, max(50, len(ones) + len(zeros)))
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            cur[oi] = 0.0
            cur[oz] = 1.0
            trial_c5, _ = _binary_eval(cur)
            delta = trial_c5 - cur_c5
            cur[oi] = 1.0
            cur[oz] = 0.0
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iters=80000, temp_init=0.004, seed=123):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, _ = _binary_eval(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    T = temp_init
    stall = 0
    best_stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5, _ = _binary_eval(cur)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            ones[i] = oz
            zeros[j] = oi
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            best_stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    n_harms = 5
    for _ in range(n_harms):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.5 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=10000, radius=0.15, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, new_c5 = _line_search_quad(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur - 1e-13:
                h = h_new
                cur = actual
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _block_swap(h_bin, deadline, max_iters=10000, seed=77):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        length = int(rng.integers(2, min(8, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        b1 = cur[i:i+length].copy()
        b2 = cur[j:j+length].copy()
        if np.sum(b1) != np.sum(b2):
            stall += 1
            continue
        cur[i:i+length] = b2
        cur[j:j+length] = b1
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            cur[i:i+length] = b1
            cur[j:j+length] = b2
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None and len(h0) > 20:
        n0 = len(h0)
        h_init = _project(h0.copy(), n0 / 2.0)
    else:
        n0 = 180
        h_init = _project(0.5 * np.ones(n0), n0 / 2.0)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP-based continuous optimization (25% of budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _local_lp_optimize(best_h, d1, radius_init=0.07, max_iter=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Second LP pass with different params
    if time.time() < d1:
        h1b, c5_1b = _local_lp_optimize(best_h, min(d1, time.time() + 10), radius_init=0.04, max_iter=1000)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Binary refinement (35% of budget)
    d2 = t0 + budget_s * 0.60
    n_best = len(best_h)
    h_bin = _to_binary(best_h, n_best)
    c5_bin = _exact_c5(h_bin)

    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_b, c5_b = _binary_targeted(h_bin, sub_dl, max_iters=100000, seed=101, max_trials=80)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.08)
        h_b, c5_b = _binary_sa(h_bin, sub_dl, max_iters=80000, temp_init=0.003, seed=202)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_b, c5_b = _block_swap(h_bin, sub_dl, max_iters=8000, seed=303)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_targeted(h_bin, sub_dl, max_iters=80000, seed=404, max_trials=120)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # LP polish on best
    if time.time() < d2:
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _local_lp_optimize(h_pol, sub_dl, radius_init=0.05, max_iter=1000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Basin hopping (15% of budget)
    d3 = t0 + budget_s * 0.75
    scale = 0.03
    for _ in range(80):
        if time.time() > d3:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d3, time.time() + budget_s * 0.006)
        h_t, c5_t = _local_lp_optimize(h_pert, sub_dl, radius_init=0.05, max_iter=500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 4: Multi-n exploration (15% of budget)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    tried = set([n_curr])
    for dn in range(-24, 25, 3):
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _local_lp_optimize(h_r, sub_dl, radius_init=0.06, max_iter=800)
        if time.time() < d4 - 5:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_targeted(h_b, sub_dl2, max_iters=40000, seed=1000 + n_try, max_trials=60)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final intensive refinement (10% of budget)
    d5 = deadline
    n_best = len(best_h)

    if time.time() < d5 - 10:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h4, c5_4 = _local_lp_optimize(h4, sub_dl, radius_init=0.03, max_iter=1500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    if time.time() < d5 - 10:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=8000, radius=0.15, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    if time.time() < d5 - 10:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_targeted(h_b, sub_dl, max_iters=100000, seed=777, max_trials=150)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d5 - 8:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iters=60000, temp_init=0.0015, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final nearby-n check
    n_fin = len(best_h)
    for dn in [-2, 2, -4, 4, -3, 3]:
        if time.time() > d5 - 2:
            break
        n_try = n_fin + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 1, time.time() + 3)
        h_r, c5_r = _local_lp_optimize(h_r, sub_dl, radius_init=0.03, max_iter=300)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
