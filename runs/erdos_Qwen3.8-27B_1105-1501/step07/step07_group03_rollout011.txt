import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    k = np.arange(n)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        idx1 = k - d
        m1 = (idx1 >= 0) & (idx1 < n)
        if np.any(m1):
            G[r, m1] += 1.0 - h[idx1[m1]]
        idx2 = k + d
        m2 = (idx2 >= 0) & (idx2 < n)
        if np.any(m2):
            G[r, m2] += -h[idx2[m2]]
    return G

def _lp_solve(h, solvers=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    if solvers is None:
        solvers = ['ECOS', 'SCS', 'CLARABEL', 'GLPK']
    G = _build_G(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    rhs = G @ h - raw
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        G @ h_var - t_var <= rhs
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    h_new = None
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                h_new = np.clip(np.asarray(h_var.value).flatten(), 0.0, 1.0)
                break
        except Exception:
            continue
    if h_new is None:
        return h, _exact_c5(h)
    h_new = _project(h_new, target)
    return h_new, _exact_c5(h_new)

def _line_search_c5(h, direction, raw, dx, max_alpha=1.0):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(direction, u, mode='full') - np.correlate(h, direction, mode='full')
    Q = np.correlate(direction, direction, mode='full')
    best_a = 0.0
    best_val = float(np.max(raw)) * dx
    alphas = np.linspace(0.0, max_alpha, 21)
    for a in alphas:
        val = float(np.max(raw + a * T - a * a * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_a = a
    if best_a < 1e-9:
        return 0.0, best_val
    a0 = best_a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, a0 - 0.05)
    hi = min(max_alpha, a0 + 0.05)
    if hi - lo > 1e-10:
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_a = opt
    return best_a, best_val

def _lp_optimize(h, deadline, max_iter=500, min_gain=1e-10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        h_lp, c5_lp = _lp_solve(h)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
        direction = h_lp - h
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-10:
            stall += 1
            continue
        direction = direction / norm_d
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search_c5(h, direction, raw, 2.0 / n, max_alpha=norm_d / np.max(np.abs(direction)) if np.max(np.abs(direction)) > 0 else 1.0)
        if alpha > 1e-9:
            h_try = _project(h + alpha * direction, target)
            c5_actual = _exact_c5(h_try)
            if c5_actual < cur_c5 - min_gain:
                h = h_try
                cur_c5 = c5_actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                continue
        if c5_lp < cur_c5 - min_gain:
            h = h_lp
            cur_c5 = c5_lp
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _grad_optimize(h, deadline, beta=20.0, radius=0.20, stall_limit=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _ in range(2000):
        if time.time() > deadline or stall > stall_limit:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        u = 1.0 - h
        g = np.zeros(n)
        n2 = len(raw)
        for r in range(n2):
            wv = w[r]
            if wv < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += wv * u[:m]
                    g[:m] -= wv * h[d:]
            else:
                gp = -d
                m = n - gp
                if m > 0:
                    g[:m] += wv * u[gp:]
                    g[gp:] -= wv * h[:m]
        g -= np.mean(g)
        d = -g / (np.linalg.norm(g) + 1e-12)
        alpha, c5_try = _line_search_c5(h, d, raw, dx, max_alpha=radius)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = _project(h + alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9994
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-15:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.25, seed=111):

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()

        d = np.zeros(n)
        d[i] = 1.0
