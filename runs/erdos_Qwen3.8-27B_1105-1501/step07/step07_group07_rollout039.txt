    m = len(scores)
    smax = float(np.max(scores))

                cur[a] = 0.0
                cur_max += best_delta
                cur_scores = _scores_arr(cur)
                zeros = list(zeros)
                ones = list(ones)
                if cur_max < best_max - 1e-14:
                    best_max = cur_max
                    best = cur.copy()
                improved = True
                break
        if not improved:
            break
    return best, float(best_max * 2.0 / n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=5000, seed=111):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        # Simple line search
        best_alpha = 0.0
        best_c5_try = cur_c5
        for alpha_frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
            trial = _project_sum(h + alpha_frac * d, target)
            c5_try = _exact_c5(trial)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_alpha = alpha_frac
        if best_alpha > 0:
            h = _project_sum(h + best_alpha * d, target)
            cur_c5 = best_c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    n0_adj = (n0 // 2) * 2
    if n0_adj != n0 or n0_adj == 0:
        n0_adj = 180
        h0 = _resample(h0, n0_adj)
    h_init = _project_sum(h0.copy(), n0_adj / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE-based continuous optimization (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _lse_optimize(h_init, d1, tau_start=1.5, tau_min=0.05, tau_rate=0.90, n_inner=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary SA (20%)
    d2 = t0 + budget_s * 0.30
    h_bin = _to_binary(best_h)
    h_bin, c5_bin = _binary_sa(h_bin, d2, T_init=0.006, seed=1001)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 3: LSE polish on binary result (5%)
    d3 = t0 + budget_s * 0.35
    h_pol = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_pol, c5_pol = _lse_optimize(h_pol, d3, tau_start=0.5, tau_min=0.02, tau_rate=0.88, n_inner=25)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (25%)
    d4 = t0 + budget_s * 0.60
    n_best = len(best_h)
    tried_ns = set([n_best])
    n_offsets = list(range(-30, 31, 2))
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        h_r_bin = _to_binary(h_r)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r_bin, c5_r = _binary_sa(h_r_bin, sub_dl, T_init=0.004, seed=n_try * 7 + 1)
        # Quick LSE polish
        if time.time() < d4 - 2:
            h_r_cont = _project_sum(h_r_bin.copy(), n_try / 2.0)
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r_cont, c5_r2 = _lse_optimize(h_r_cont, sub_dl2, tau_start=0.8, tau_min=0.03, tau_rate=0.90, n_inner=20)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r_bin = _to_binary(h_r_cont)
                h_r_bin, c5_r3 = _binary_sa(h_r_bin, min(d4, time.time() + 1.0), T_init=0.003, seed=n_try * 13)
                if c5_r3 < c5_r:
                    c5_r = c5_r3
                    h_r_bin = h_r_bin
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r_bin.copy()
            n_best = n_try

    # Phase 5: Intensive binary SA with multiple restarts (20%)
    d5 = t0 + budget_s * 0.80
    h_b = _to_binary(best_h)
    for attempt in range(30):
        if time.time() > d5 - 5:
            break
        h_start = _perturb(best_h, rng, 0.04 + 0.02 * (attempt % 5))
        h_b = _to_binary(h_start)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_b, c5_b = _binary_sa(h_b, sub_dl, T_init=0.005, seed=seed + attempt * 100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Follow-up LSE polish
            if time.time() < d5 - 3:
                h_pc = _project_sum(best_h.copy(), len(best_h) / 2.0)
                sub_dl2 = min(d5, time.time() + budget_s * 0.008)
                h_pc, c5_pc = _lse_optimize(h_pc, sub_dl2, tau_start=0.5, tau_min=0.02, tau_rate=0.88, n_inner=20)
                if c5_pc < best_c5:
                    best_c5 = c5_pc
                    best_h = h_pc.copy()

    # Phase 6: Pair swap refinement (5%)
    d6 = t0 + budget_s * 0.85
    h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _pair_swap_loop(h6, d6, n_pairs=6000, seed=555)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Deterministic local search on binary (5%)
    d7 = t0 + budget_s * 0.90
    h_b7 = _to_binary(best_h)
    h_b7, c5_7 = _binary_det_local(h_b7, d7, max_rounds=50)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h_b7.copy()

    # Phase 8: Final SA polish (5%)
    d8 = deadline
    h_b8 = _to_binary(best_h)
    h_b8, c5_8 = _binary_sa(h_b8, d8, T_init=0.003, seed=9999)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h_b8.copy()

    # Final LSE micro-polish if time remains
    if deadline - time.time() > 3:
        h_final = _project_sum(best_h.copy(), len(best_h) / 2.0)
        h_final, c5_f = _lse_optimize(h_final, deadline, tau_start=0.3, tau_min=0.01, tau_rate=0.85, n_inner=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()
        else:
            h_final = _to_binary(best_h)
            h_final, c5_f2 = _binary_sa(h_final, deadline, T_init=0.002, seed=4242)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_final.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
