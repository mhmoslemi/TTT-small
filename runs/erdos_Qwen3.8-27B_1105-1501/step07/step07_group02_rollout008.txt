import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lag_grad(h, k):
    """Gradient of corr at lag k wrt h. Returns array of length n."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = k
        if n - m > 0:
            g[:n-m] += (1.0 - h[m:])
        if m < n:
            g[m:] -= h[:n-m]
    else:
        m = -k
        if n - m > 0:
            g[:n-m] -= h[m:]
        if m < n:
            g[m:] += (1.0 - h[:n-m])
    return g

def _max_subgrad(h, beta=50.0):
    """Weighted subgradient of max_k corr_k(h) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(corr))
    w = np.exp(beta * (corr - M))
    ws = float(np.sum(w))
    if ws > 1e-30:
        w /= ws
    grad = np.zeros(n)
    for r in range(len(corr)):
        if w[r] < 1e-12:
            continue
        k = n - 1 - r
        grad += w[r] * _lag_grad(h, k)
    return grad * dx

def _proj_grad_opt(h, deadline, beta=50.0, n_max_iter=3000, seed=42):

    alpha = 0.02
    stall = 0

                G[row, :m] -= h[p:]

        best_c5_try = cur_c5

        one_mh = 1.0 - h
