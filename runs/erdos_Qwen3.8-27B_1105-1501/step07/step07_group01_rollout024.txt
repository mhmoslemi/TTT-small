def _project(v, target):
    # Water-filling projection: find theta such that sum(clip(v-theta, 0, 1)) = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    s = np.arange(n)  # lags 0, 1, ..., n-1
    d = np.zeros(2 * n - 1)
    
    # Positive lag t = n-1+s
    # Term 1: (1-h[fig + s]) for i + s < n, i.e., s < n - i
    mask1_i = s < (n - i)
    mask1_j = s < (n - j)
    term1_i = np.zeros(n)
    term1_j = np.zeros(n)
    term1_i[mask1_i] = 1.0 - h[i + s[mask1_i]]
    term1_j[mask1_j] = 1.0 - h[j + s[mask1_j]]
    
    # Term 2: h[i-s] for i-s >= 0, i.e., s <= i
    mask2_i = s <= i
    mask2_j = s <= j
    term2_i = np.zeros(n)
    term2_j = np.zeros(n)
    term2_i[mask2_i] = h[i - s[mask2_i]]
    term2_j[mask2_j] = h[j - s[mask2_j]]

    termA_i = np.zeros(n-1)
    termB_i = np.zeros(n-1)
    termA_j = np.zeros(n-1)
    termB_j = np.zeros(n-1)
    termA_i[maskA_i] = 1.0 - h[i - s1[maskA_i]]

def _compute_pressure(h, corr, beta=5.0):
    """Compute smoothed gradient at each position."""
    n = len(h)
    # w[t] = exp(beta * (corr[t] - max_corr)) / sum
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    w /= np.sum(w)
    # P[i] = sum_t w[t] * S[i,t]
    # S[i, n-1+s] = (1-h[i+s])*[i<=n-s-1] - h[i-s]*[i>=s]
    P = np.zeros(n)
    for i in range(n):
        P[i] = 0.0
        for s in range(n):
            # Contribution from positive lag n-1+s
            w_pos = w[n-1+s]
            if i <= n-s-1:
                P[i] += w_pos * (1.0 - h[i+s])
            if i >= s:
                P[i] -= w_pos * h[i-s]

                if i >= s:
                    P[i] += w_neg * (1.0 - h[i-s])
                if i <= n-s-1:
