from scipy.optimize import minimize, linprog
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 0.01
_g_G = None  # Gradient matrix

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build gradient matrix G where G[r, k] = d(raw[r])/d(h[k])."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    # Zero lag
    G[mid, :] = 1.0 - 2.0 * h
    # Positive lags d = 1, ..., n-1
    for d in range(1, n):
        m = n - d
        # row = mid + d: sum_{i=0}^{m-1} h[i]*(1-h[i+d])
        # d/dh[i] = 1-h[i+d] for i < m
        # d/dh[i+d] = -h[i] for i < m
        G[mid + d, :m] += (1.0 - h[d:d+m])
        G[mid + d, d:d+m] += -h[:m]
        # row = mid - d: sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        # d/dh[i+d] = 1-h[i] for i < m
        # d/dh[i] = -h[i+d] for i < m
        G[mid - d, d:d+m] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:d+m]
    return G

def _verify_G(h, G):

    n = len(h)
    err = 0.0
    delta = 1e-6
    for k in range(min(3, n)):
        h_plus = h.copy()
        h_plus[k] = min(1.0, h[k] + delta)
        raw0 = _raw_scores(h)
        raw1 = _raw_scores(h_plus)

        # Account for boundary: if h[k]+delta > 1, the derivative is different
        if h[k] + delta > 1.0:
            continue
        err = max(err, float(np.max(np.abs(fd - G[:, k]))))

def _smooth_obj(x):
    """Log-sum-exp objective."""
    h = np.clip(x, 0.0, 1.0)
    s = _raw_scores(h) * _g_dx
    tau = _g_tau
    mval = float(np.max(s))
    u = (s - mval) / tau
    # Handle underflow
    z = np.exp(np.minimum(u, 500))
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        # Fallback: uniform over max rows
        mask = s >= mval - 1e-10
        return mval
    return mval + tau * np.log(zsum)

def _smooth_jac(x):
    """Gradient of log-sum-exp objective."""
    h = np.clip(x, 0.0, 1.0)
    s = _raw_scores(h) * _g_dx
    tau = _g_tau
    mval = float(np.max(s))

    else:

    G = _build_G(h)
    return G.T @ probs * _g_dx

def _slsqp_smooth(h, tau, maxiter=50):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _line_search_pair(h, i, j, n_iter=30):
    """Line search along direction e_i - e_j."""
    n = len(h)
    dx = 2.0 / n
    hi = h[i]
    hj = h[j]
    # t ranges so that h[i]+t in [0,1] and h[j]-t in [0,1]
    t_lo = max(-hi, -(1.0 - hj))  # h[i]+t >= 0, h[j]-t <= 1 => t >= hj-1
    t_hi = min(1.0 - hi, hj)       # h[i]+t <= 1, h[j]-t >= 0 => t <= hj
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    
    def eval_at(t):
        h_new = h.copy()
        h_new[i] = h[i] + t
        h_new[j] = h[j] - t
        h_new = np.clip(h_new, 0.0, 1.0)
        return _exact_c5(h_new)
    
    fc = eval_at(c)
    fd = eval_at(d_pt)
    
    for _ in range(n_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_at(c)
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = eval_at(d_pt)
    
    t_opt = (a + b) / 2.0
    f_opt = eval_at(t_opt)
    # Also check endpoints
    f_lo = eval_at(t_lo)
    f_hi = eval_at(t_hi)
    f_0 = eval_at(0.0)
    
    best_t, best_f = 0.0, f_0
    if f_lo < best_f:
        best_t, best_f = t_lo, f_lo
    if f_hi < best_f:
        best_t, best_f = t_hi, f_hi
    if f_opt < best_f:
        best_t, best_f = t_opt, f_opt
    
    return best_t, best_f

    G = _build_G(h)
    return G.T @ probs * _g_dx

def _eq_fun(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones(len(x))

def _slsqp_step(h, tau, maxiter=30):
    global _g_n, _g_target, _g_dx, _g_tau
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    _g_tau = tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lse_obj, h, method='SLSQP', jac=_lse_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h.copy()

def _projected_grad(h, tau, n_iter=60):
    n = len(h)
    cur = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(cur)
    lr = 0.5
    for it in range(n_iter):
        s = _raw_scores(cur) * dx
        mval = float(np.max(s))
        u = (s - mval) / tau
        z = np.exp(np.minimum(u, 500.0))
        zsum = float(np.sum(z))
        if zsum < 1e-300:
            mask = s >= mval - 1e-10
            probs = np.zeros(len(s))
            probs[mask] = 1.0 / max(1, int(np.sum(mask)))
        else:
            probs = z / zsum
        G = _build_G(cur)
        grad = G.T @ probs * dx
        step = lr * grad
        new_h = _project_sum(cur - step, n / 2.0)
        new_c5 = _exact_c5(new_h)
        if new_c5 < cur_c5:
            cur = new_h
            cur_c5 = new_c5
            lr = min(lr * 1.1, 5.0)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return cur, cur_c5

def _anneal_smooth(h, deadline, n_stages=5, tau_start=None):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    if tau_start is None:
        s = _raw_scores(cur) * (2.0 / n)
        tau_start = max(0.02, 0.25 * float(np.std(s)))
    tau = tau_start
    tau_floor = max(0.002, tau_start * 0.03)
    for stage in range(n_stages):
        if time.time() > deadline - 3:
            break
        maxiter = max(10, int(30 * (tau / max(tau_floor, 1e-9))))
        new_h = _slsqp_step(cur, tau, maxiter=min(maxiter, 50))
        new_c5 = _exact_c5(new_h)
        if new_c5 < cur_c5:
            cur = new_h
            cur_c5 = new_c5
        if tau < tau_floor * 2:
            new_h2, c5_2 = _projected_grad(cur, max(tau, tau_floor), n_iter=40)
            if c5_2 < cur_c5:
                cur = new_h2
                cur_c5 = c5_2
        tau *= 0.5
    return cur, cur_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_swap_binary(h, deadline, max_rounds=40):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline - 1:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline - 1:
                break
            best_oz = -1
            best_c = cur_c5
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < best_c:
                    best_c = c_try
                    best_oz = oz
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
            if best_oz >= 0:
                cur[oi] = 0.0
                cur[best_oz] = 1.0
                cur_c5 = best_c
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                improved = True
                ones = list(np.where(cur > 0.5)[0])
                zeros = list(np.where(cur <= 0.5)[0])
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _line_search_pair(h, i, j, n_iter=20):
    n = len(h)
    hi_val = h[i]
    hj_val = h[j]
    t_lo = max(-hi_val, hj_val - 1.0)
    t_hi = min(1.0 - hi_val, hj_val)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    def ev(t):
        h_new = h.copy()
        h_new[i] = h[i] + t
        h_new[j] = h[j] - t
        return _exact_c5(h_new)
    fc = ev(c_pt)
    fd = ev(d_pt)
    for _ in range(n_iter):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = ev(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = ev(d_pt)
    t_opt = (a + b) / 2.0
    f_opt = ev(t_opt)
    f0 = ev(0.0)
    f_lo = ev(t_lo)
    f_hi = ev(t_hi)
    best_t, best_f = 0.0, f0
    if f_lo < best_f:
        best_t, best_f = t_lo, f_lo
    if f_hi < best_f:
        best_t, best_f = t_hi, f_hi
    if f_opt < best_f:
        best_t, best_f = t_opt, f_opt
    return best_t, best_f

def _cont_pair_loop(h, deadline, n_trials=3000, seed=42):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline - 2:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        t, f_new = _line_search_pair(cur, i, j, n_iter=15)
        if f_new < cur_c5 - 1e-13:
            cur[i] = cur[i] + t
            cur[j] = cur[j] - t
            cur = _project_sum(cur, n / 2.0)
            cur_c5 = _exact_c5(cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 15.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += 0.1 * rng.normal(0, 1, n)
    pert *= amplitude
    return _project_sum(h + pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        n0 = (n0 // 2) * 2
        h0 = _resample(np.array(initial_h_values, dtype=float) if 'initial_h_values' in dir() else h0, n0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary local search (8%)
    d1 = t0 + budget_s * 0.08
    h_bin = _to_binary(best_h)
    h_bin, c5_1 = _pair_swap_binary(h_bin, d1, max_rounds=25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h_bin.copy()

    # Phase 2: LSE-SLSQP annealing (17%)
    d2 = t0 + budget_s * 0.25
    h_smooth = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_smooth, c5_2 = _anneal_smooth(h_smooth, d2, n_stages=5)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h_smooth.copy()

    # Phase 3: Binary refinement (8%)
    d3 = t0 + budget_s * 0.33
    h_bin3 = _to_binary(best_h)
    h_bin3, c5_3 = _pair_swap_binary(h_bin3, d3, max_rounds=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h_bin3.copy()

    # Phase 4: Multi-n exploration (22%)
    d4 = t0 + budget_s * 0.55
    n_cur = len(best_h)
    n_offsets = list(range(-16, 17, 4))
    tried_ns = set()
    for dn in n_offsets:
        if time.time() > d4 - 15:
            break
        n_try = n_cur + dn
        if n_try < 80 or n_try > 240 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        h_br = _to_binary(h_r)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_br, c5_br = _pair_swap_binary(h_br, sub_dl, max_rounds=12)
        if c5_br < best_c5 + 0.004 and time.time() < d4 - 10:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_sr, c5_sr = _anneal_smooth(_project_sum(h_br, n_try / 2.0), sub_dl2, n_stages=3)
            if c5_sr < c5_br:
                c5_br = c5_sr
                h_br = h_sr
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            n_cur = n_try

    # Phase 5: Intensive continuous (20%)
    d5 = t0 + budget_s * 0.75
    n_best = len(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.07)
    h_cont = _project_sum(best_h.copy(), n_best / 2.0)
    h_cont, c5_cont = _cont_pair_loop(h_cont, sub_dl, n_trials=4000, seed=777)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()

    if time.time() < d5 - 5:
        h_pol = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h_pol, c5_pol = _anneal_smooth(h_pol, sub_dl, n_stages=3)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() < d5 - 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_swap_binary(h_b, d5, max_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Basin hopping (13%)
    d6 = t0 + budget_s * 0.88
    scale = 0.05
    for _ in range(300):
        if time.time() > d6 - 2:
            break
        h_p = _perturb(best_h, rng, scale)
        sub_dl = min(d6, time.time() + budget_s * 0.004)
        h_pb = _to_binary(h_p)
        h_pb, c5_pb = _pair_swap_binary(h_pb, sub_dl, max_rounds=6)
        if c5_pb < best_c5:
            best_c5 = c5_pb
            best_h = h_pb.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.07)

    # Phase 7: Final polish (12%)
    d7 = deadline
    n_best = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > d7 - 8:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_br = _to_binary(h_r)
        sub_dl = min(d7, time.time() + budget_s * 0.007)
        h_br, c5_br = _pair_swap_binary(h_br, sub_dl, max_rounds=15)
        if c5_br < best_c5 and time.time() < d7 - 5:
            sub_dl2 = min(d7, time.time() + budget_s * 0.005)
            h_sr, c5_sr = _anneal_smooth(_project_sum(h_br, n_try / 2.0), sub_dl2, n_stages=2)
            if c5_sr < c5_br:
                c5_br = c5_sr
                h_br = h_sr
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()

    # Final SLSQP
    if deadline - time.time() > 8:
        h_fin = _project_sum(best_h.copy(), len(best_h) / 2.0)
        h_fin, c5_fin = _anneal_smooth(h_fin, deadline, n_stages=2)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final binary sweep
    if deadline - time.time() > 4:
        h_fb = _to_binary(best_h)
        h_fb, c5_fb = _pair_swap_binary(h_fb, deadline, max_rounds=8)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
