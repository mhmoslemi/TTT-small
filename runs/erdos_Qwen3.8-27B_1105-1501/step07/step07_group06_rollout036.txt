import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_anti(h):
    n = len(h)
    if n % 2 != 0:
        n2 = n + 1
        x_old = (np.arange(len(h)) + 0.5) / len(h)
        x_new = (np.arange(n2) + 0.5) / n2
        h = np.interp(x_new, x_old, h)
        h = _project(h, n2 / 2.0)
        n = n2
    m = n // 2
    x = 0.5 * h[:m] + 0.5 * (1.0 - h[n-1:m-1:-1])
    return np.clip(x, 0.0, 1.0)

def _from_anti(x):
    m = len(x)
    n = 2 * m
    h = np.zeros(n)
    h[:m] = x
    h[m:] = 1.0 - x[m-1::-1]
    return h

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anti_smooth_grad(x, beta):
    m = len(x)
    n = 2 * m
    h = _from_anti(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    raw = corr
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        w = np.zeros(len(raw))
        w[idx] = 1.0
    else:
        w /= wsum
    S = mx + np.log(max(wsum, 1e-300)) / beta
    G = np.zeros(n)
    for r in range(len(raw)):
        if w[r] < 1e-10:
            continue
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mt = n - tau
            if mt > 0:
                G[tau:] += wr * (1.0 - h[:mt])
                G[:mt] -= wr * h[tau:tau + mt]
        else:
            d = -tau
            mt = n - d
            if mt > 0:
                G[:mt] += wr * (1.0 - h[d:])
                G[d:] -= wr * h[:mt]
    grad_x = G[:m] - G[n-1:m-1:-1]
    return S, grad_x

def _anti_lbfgsb(x0, deadline, beta_max=250, max_outer=30):
    m = len(x0)
    bounds = [(0.0, 1.0)] * m
    x = x0.copy()
    best_x = x0.copy()
    h_b = _from_anti(x0)
    best_c5 = _exact_c5(h_b)
    beta = 10.0
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        def fun(xv):
            S, _ = _anti_smooth_grad(xv, beta)
            return S
        def jac(xv):
            _, g = _anti_smooth_grad(xv, beta)
            return g
        try:
            res = minimize(fun, x, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 150, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
            x_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            x_new = x.copy()
        h_new = _from_anti(x_new)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_x = x_new.copy()
            x = x_new
        elif c5_new < best_c5 + 1e-10:
            x = x_new
        if not res.success and 'Maximum' in res.message:
            beta = min(beta * 1.6, beta_max)
        elif beta >= beta_max * 0.85 and res.success:
            break
        else:
            beta = min(beta * 1.4, beta_max)
    return best_x, best_c5

def _full_smooth_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    raw = corr
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        w = np.zeros(len(raw))
        w[idx] = 1.0
    else:
        w /= wsum
    S = mx + np.log(max(wsum, 1e-300)) / beta
    G = np.zeros(n)
    for r in range(len(raw)):
        if w[r] < 1e-10:
            continue
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mt = n - tau
            if mt > 0:
                G[tau:] += wr * (1.0 - h[:mt])
                G[:mt] -= wr * h[tau:tau + mt]
        else:
            d = -tau
            mt = n - d
            if mt > 0:
                G[:mt] += wr * (1.0 - h[d:])
                G[d:] -= wr * h[:mt]
    G -= np.mean(G)
    return S, G

def _line_search_full(h, d, radius=0.25):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = _exact_c5(h + c1 * d), _exact_c5(h + c2 * d)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_t = (lo + hi) / 2.0
    opt_v = _exact_c5(h + opt_t * d)
    for t in [a_lo, a_hi, 0.0]:
        ft = _exact_c5(h + t * d)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _full_opt(h0, deadline, max_iters=150, beta_start=12.0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_start
    stall = 0
    rng = np.random.default_rng(31415)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 25:
            break
        S, G = _full_smooth_grad(h, beta)
        gn = np.linalg.norm(G)
        if gn < 1e-15:
            beta = min(beta * 1.4, 200.0)
            continue
        d = G / gn
        alpha, c_try = _line_search_full(h, d, radius=0.2)
        if alpha != 0.0 and c_try < best_c5 - 1e-14:
            h_new = _project(h + alpha * d, target)
            c_new = _exact_c5(h_new)
            if c_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Try random direction
            dr = rng.normal(0, 1, n)
            dr -= np.mean(dr)
            dr /= np.linalg.norm(dr)
            alpha, c_try = _line_search_full(h, dr, radius=0.15)
            if alpha != 0.0 and c_try < best_c5 - 1e-14:
                h_new = _project(h + alpha * dr, target)
                c_new = _exact_c5(h_new)
                if c_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall % 5 == 0:
            beta = min(beta * 1.2, 200.0)
    return best_h, best_c5

def _binary_opt(h_bin, deadline, sa_iters=30000, max_stall=400):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(9999)
    T = 0.004
    stall = 0
    for _ in range(sa_iters):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[rng.integers(len(ones))]
        oz = zeros[rng.integers(len(zeros))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    # Greedy improvement
    h_g = best_h.copy()
    for _ in range(5):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h_g > 0.5)[0]
        zeros = np.where(h_g <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_g[oi] = 0.0
                h_g[oz] = 1.0
                c = _exact_c5(h_g)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = h_g.copy()
                    improved = True
                    break
                else:
                    h_g[oz] = 0.0
                    h_g[oi] = 1.0
            if improved:
                h_g = best_h.copy()
                break
    return best_h, best_c5

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.15 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _anti_lbfgsb_multi(x0, deadline, betas=(10, 25, 50, 100, 200, 400)):
    """Run L-BFGS-B with increasing beta values."""
    m = len(x0)
    bounds = [(0.0, 1.0)] * m
    x = x0.copy()
    best_x = x0.copy()
    best_c5 = _exact_c5(_from_anti(x0))
    for beta in betas:
        if time.time() > deadline:
            break
        def fun(xv):
            S, _ = _anti_smooth_grad(xv, beta)
            return S
        def jac(xv):
            _, g = _anti_smooth_grad(xv, beta)
            return g
        try:
            res = minimize(fun, x, jac=jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 60})
            x = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
        c5_cur = _exact_c5(_from_anti(x))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_x = x.copy()
    return best_x, best_c5
