h(x) = sum_{k=-inf}^{inf} c_k * exp(2*pi*i*k*x/2) = sum_{k=-inf}^{inf} c_k * exp(pi*i*k*x)

For real h: h(x) = c_0 + sum_{k=1}^{inf} (a_k * cos(pi*k*x) + b_k * sin(pi*k*x))

The constraint integral_0^2 h(x) dx = 1 means c_0 = 0.5 (since integral of cos/sin over [0,2] is 0).

The constraint 0 <= h <= 1 constrains the amplitudes.

import numpy as np
import time
import math

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _score_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, t):
    n = len(h)
    l = t - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:l+m])
            g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s+m]
    return g

def _line_search(h, d, dx, radius, raw=None):
    n = len(h)
    if raw is None:
        S0 = np.correlate(h, 1.0 - h, mode='full')
    else:
        S0 = raw
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        cap = float(np.min((1.0 - h[pos]) / d[pos]))
        a_hi = min(a_hi, cap)
    if np.any(neg):
        cap = float(np.min(h[neg] / (-d[neg])))
        a_lo = max(a_lo, cap)
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    gr_r = (math.sqrt(5.0) - 1.0) / 2.0
    lo_r, hi_r = a_lo, a_hi
    c1 = hi_r - gr_r * (hi_r - lo_r)
    c2 = lo_r + gr_r * (hi_r - lo_r)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    best_a = c1 if f1 < f2 else c2
    best_val = min(f1, f2)
    for _ in range(45):
        if hi_r - lo_r < 1e-13:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr_r * (hi_r - lo_r)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr_r * (hi_r - lo_r)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fm = float(np.max(S0 + mid * T - mid*mid * Q)) * dx
        if fm < best_val:
            best_val = fm
            best_a = mid
    return best_a, best_val

def _softmax_dir(h, beta, raw=None):
    n = len(h)
    if raw is None:
        raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    v = np.clip(beta * (raw - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(raw)
        w[np.argmax(raw)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[n-1+l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l+m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s+m]
    g -= np.mean(g)
    return g

def _topk_dir(h, K, raw=None):
    n = len(h)
    if raw is None:
        raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:K]
    g = np.zeros(n)
    for t in top_idx:
        g += _lag_grad(h, int(t))
    g /= K
    g -= np.mean(g)
    return g

def _norm_scale(d, radius):
    nrm = np.linalg.norm(d)
    if nrm < 1e-12:
        return None
    d = d / nrm * min(radius, 0.25)
    d -= np.mean(d)
    return d

def _freq_dir(rng, n, radius):
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 14.0)
        p = rng.uniform(0, 2*np.pi)
        d += rng.uniform(0.4, 1.2) * np.sin(2*np.pi*f*x/n + p)
    d -= np.mean(d)
    return _norm_scale(d, radius)

def _block_smooth_dir(g, n, bs, radius):
    nb = n // bs
    if nb < 1:
        return None
    d = np.zeros(n)
    for b in range(nb):
        d[b*bs:(b+1)*bs] = np.mean(g[b*bs:(b+1)*bs])
    if nb*bs < n:
        d[nb*bs:] = np.mean(g[nb*bs:])
    d -= np.mean(d)
    return _norm_scale(d, radius)

def _local_opt(h, deadline, r0=0.15, stall_lim=50, use_topk=True, topk_list=None):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur, raw_cur = _score_raw(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    if topk_list is None:
        topk_list = [3, 7, 15, 30]
    betas = [12.0, 25.0, 50.0, 100.0]
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = []
        for beta in betas:
            g = _softmax_dir(h, beta, raw_cur)
            d = _norm_scale(-g, rad)
            if d is not None:
                dirs.append(d)
        if use_topk:
            for K in topk_list:
                g = _topk_dir(h, K, raw_cur)
                d = _norm_scale(-g, rad)
                if d is not None:
                    dirs.append(d)
        d = _freq_dir(rng, n, rad)
        if d is not None:
            dirs.append(d)
        g25 = _softmax_dir(h, 25.0, raw_cur)
        for bs in [6, 12, 24]:
            bd = _block_smooth_dir(-g25, n, bs, rad)
            if bd is not None:
                dirs.append(bd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            a, c_try = _line_search(h, d, dx, rad * 2.0, raw_cur)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur, raw_cur = _score_raw(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.35)
        else:
            stall += 1
            rad *= 0.75
            if rad < 0.006:
                pert = rng.normal(0, 0.08, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp, _ = _score_raw(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur, raw_cur = _score_raw(hp)
                    stall = 0
                    rad = 0.12
                elif stall % 25 == 0:
                    rad = 0.07
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _score(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _score(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n//2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _gen_perturbation(rng, n, scale, mode=0):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    if mode == 0:
        for _ in range(4):
            f = rng.uniform(0.5, 16.0)
            p = rng.uniform(0, 2*np.pi)
            pert += rng.uniform(0.4, 1.3) * np.sin(2*np.pi*f*x/n + p)
        pert += 0.4 * scale * rng.normal(0, 1, n)
    elif mode == 1:
        n_flips = max(3, int(n * scale * 2))
        flips = rng.integers(0, n, min(n_flips, n//4))
        for fi in flips:
            pert[fi] = rng.choice([-1.0, 1.0]) * rng.uniform(0.5, 1.5)
    else:
        bs = int(rng.integers(4, 16))
        nb = n // bs
        if nb > 1:
            perm = rng.permutation(nb)
            for b in range(nb):
                pert[b*bs:(b+1)*bs] = rng.uniform(-0.3, 0.3) * (perm[b] - (nb-1)/2.0) / max(nb, 1)
    mxp = np.max(np.abs(pert))
    if mxp > 1e-10:
        pert *= scale / mxp
    return pert
