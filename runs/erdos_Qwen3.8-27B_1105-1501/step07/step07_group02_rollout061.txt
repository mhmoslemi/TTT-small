import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_S(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _active_gradient(h, S, epsilon, temperature, top_m=0):
    n = len(h)
    L = 2 * n - 1
    C = float(np.max(S))
    if top_m > 0 and top_m < L:
        thresh = np.sort(S)[-top_m]
        active = S >= thresh
    else:
        active = S >= (C - epsilon)
    if not np.any(active):
        active = S >= (C - max(epsilon * 10, 0.001))
    if not np.any(active):
        return None
    idx_active = np.nonzero(active)[0]
    S_act = S[idx_active]
    w = np.exp((S_act - C) / max(temperature, 1e-10))
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.ones(len(S_act)) / len(S_act)
    else:
        w /= wsum
    shifts = idx_active - (n - 1)
    g = np.zeros(n)
    for shift, wt in zip(shifts, w):
        if wt < 1e-10:
            continue
        if shift >= 0:
            g[shift:] += wt * h[:n - shift]
            g[:n - shift] -= wt * (1.0 - h[shift:])
        else:
            p = -shift
            g[:n - p] += wt * h[p:]
            g[p:] -= wt * (1.0 - h[:n - p])
    g -= np.mean(g)
    return g

def _line_search(h, g, S, t_max, n_evals=16):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_max = float(np.max(S))
    if t_max <= 1e-14:
        return 0.0, cur_max
    ts = np.linspace(0.0, t_max, n_evals)
    best_t = 0.0
    best_val = cur_max
    for t in ts[1:]:
        h_try = h + t * g
        if np.any(h_try < -1e-10) or np.any(h_try > 1 + 1e-10):
            continue
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _project(h_try, target)
        S_try = np.correlate(h_try, 1.0 - h_try, mode='full') * dx
        val = float(np.max(S_try))
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    # Refine with golden section around best_t
    if best_t > 0:
        lo = max(0.0, best_t - t_max / n_evals)
        hi = min(t_max, best_t + t_max / n_evals)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(_compute_S(_project(np.clip(h + c1 * g, 0, 1), target))))
        f2 = float(np.max(_compute_S(_project(np.clip(h + c2 * g, 0, 1), target))))
        for _ in range(20):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(_compute_S(_project(np.clip(h + c1 * g, 0, 1), target))))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(_compute_S(_project(np.clip(h + c2 * g, 0, 1), target))))
        t_opt = (lo + hi) / 2.0
        h_opt = _project(np.clip(h + t_opt * g, 0, 1), target)
        v_opt = float(np.max(_compute_S(h_opt)))
        if v_opt < best_val:
            best_t = t_opt
            best_val = v_opt
    return best_t, best_val

    stall = 0

            temp *= 0.6
            step_factor *= 0.6

        g_norm = float(np.max(np.abs(g)))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_solve(h_init, deadline, top_k=25, max_iter=100):
    """SLSQP epigraph solve."""
    from scipy.optimize import minimize
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h0 = _project(h_init.copy(), target)
    S0 = _compute_S(h0)
    t0 = float(np.max(S0)) + 0.005
    x0 = np.append(h0, t0)
    
    # Top k lags
    top_lags = np.argsort(S0)[-top_k:]
    shifts = top_lags - (n - 1)
    
    def obj(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Equality: sum(h) - n/2 = 0
    def eq_con(z):
        return float(np.sum(z[:n]) - target)
    
    def eq_grad(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Inequality constraints: S_r(h) - t <= 0
    # Each returns -S_r(h) + t >= 0 (for SLSQP: constraint >= 0)
    # Actually SLSQP uses: fun(x) >= 0 for inequalities
    # So we need: t - S_r(h) >= 0
    
    constraints = [{'type': 'eq', 'fun': eq_con, 'jac': eq_grad}]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    for lag in top_lags:
        shift = int(lag) - (n - 1)
        def make_incon(shift_val):
            def incon(z):
                h = z[:n]
                # Compute S at this lag
                if shift_val >= 0:
                    m = n - shift_val
                    c = float(np.sum(h[:m] * (1.0 - h[shift_val:])))
                else:
                    p = -shift_val
                    m = n - p
                    c = float(np.sum(h[p:] * (1.0 - h[:m])))
                return z[n] - c * dx
            def incon_jac(z):
                h = z[:n]
                g = np.zeros(n + 1)
                if shift_val >= 0:
                    m = n - shift_val
                    # d(c)/d(h[i]) for i in [0, m): (1-h[i+shift])
                    # d(c)/d(h[j]) for j in [shift, n): -h[j-shift]
                    g[:m] += dx * (1.0 - h[shift_val:])
                    g[shift_val:] += dx * (-h[:m])
                else:
                    p = -shift_val
                    m = n - p
                    g[p:] += dx * (1.0 - h[:m])
                    g[:m] += dx * (-h[p:])
                g[n] = 1.0
                return g
            return {'type': 'ineq', 'fun': incon, 'jac': incon_jac}
        constraints.append(make_incon(shift))
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       constraints=constraints, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-8})
        if res.success or res.x is not None:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < float(np.max(S0)):
                return h_new, c5_new
    except Exception:
        pass
    return h_init, float(np.max(S0))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = 0.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-11:

        else:

        if cnt > 0:

            x[mask] += res / cnt

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

class _EpiIneq:
    def __init__(self, n, dx, shift):
        self.n = n
        self.dx = dx
        self.shift = shift
    def fun(self, z):
        h = z[:self.n]
        s = self.shift
        if s >= 0:
            m = self.n - s
            c = float(np.dot(h[:m], 1.0 - h[s:]))
        else:
            p = -s
            m = self.n - p
            c = float(np.dot(h[p:], 1.0 - h[:m]))
        return z[self.n] - c * self.dx
    def jac(self, z):
        h = z[:self.n]
        n = self.n
        g = np.zeros(n + 1)
        s = self.shift
        if s >= 0:
            m = n - s
            g[:m] += self.dx * (1.0 - h[s:])
            g[s:] += self.dx * (-h[:m])
        else:
            p = -s
            m = n - p
            g[p:] += self.dx * (1.0 - h[:m])
            g[:m] += self.dx * (-h[p:])
        g[n] = 1.0
        return g

def _epi_eq_fun(z, target):
    return float(np.sum(z[:len(z) - 1]) - target)

def _epi_eq_jac(z):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def _epi_obj(z):
    return z[-1]

def _epi_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _slsqp_solve(h_init, deadline, top_k=22, max_iter=120):

    except ImportError:
        return h_init, _exact_c5(h_init)

    t0 = float(np.max(S0)) + 0.003

    ineqs = []
    for lag in top_lags:
        shift = int(lag) - (n - 1)
        obj_c = _EpiIneq(n, dx, shift)
        ineqs.append({'type': 'ineq', 'fun': obj_c.fun, 'jac': obj_c.jac})
    constraints = [{'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac}]
    constraints[0]['fun'] = lambda z: _epi_eq_fun(z, target)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(_epi_obj, x0, method='SLSQP', jac=_epi_obj_jac,

        if res.x is not None:

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.010)
            h_r2, c5_r2 = _pair_refine(h_r, sub_dl2, n_pairs=25, radius=0.10, seed=300 + dn)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

    # Phase 4: SLSQP polish + dense (15% time)
    d4 = t0 + budget_s * 0.85
    if time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.08)
        h4, c5_4 = _slsqp_solve(best_h, sub_dl, top_k=20, max_iter=100)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + budget_s * 0.06)
            h4b, c5_4b = _dense_descend(best_h, sub_dl2, epsilon=0.002, temperature=0.004, max_iter=4000, seed=555)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Final refinement (10% time)
    if time.time() < deadline:
        h5, c5_5 = _dense_descend(best_h, deadline, epsilon=0.001, temperature=0.003, max_iter=5000, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        for dn in [-2, 2, -4, 4]:
            if time.time() > deadline:
                break
            n_try = len(best_h) + dn
            if n_try < 80 or n_try > 240 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _dense_descend(h_r, deadline, epsilon=0.001, temperature=0.002, max_iter=2000, seed=999 + dn)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
