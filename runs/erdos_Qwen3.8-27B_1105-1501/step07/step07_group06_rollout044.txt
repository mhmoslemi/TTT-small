def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_anti(h):
    """Convert h to anti-palindromic form, returning x (first half)."""
    n = len(h)
    m = n // 2
    # x[i] = 0.5 * h[i] + 0.5 * (1 - h[n-1-i])
    x = 0.5 * h[:m] + 0.5 * (1.0 - h[n-1:m-1:-1])
    return np.clip(x, 0.0, 1.0)

def _from_anti(x):
    """Convert anti-palindromic x back to full h."""
    m = len(x)
    n = 2 * m
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _smooth_obj_and_grad(x, beta):
    """Compute smooth-max objective and gradient in anti-palindromic space."""
    m = len(x)
    n = 2 * m
    h = _from_anti(x)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    R = len(raw)  # 2n-1
    
    # Smooth max
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    exp_z = np.exp(z)
    wsum = float(np.sum(exp_z))
    w = exp_z / wsum
    S = M + np.log(wsum) / beta
    
    # Gradient of S with respect to h
    # dS/dh[i] = sum_r w[r] * (d raw[r]/d h[i])
    G = np.zeros(n)
    for r in range(R):
        if w[r] < 1e-10:

        if k >= 0:

            G[k:] += w[r] * u[:mk]
            G[:mk] -= w[r] * h[k:k+mk]
        else:

            mk = n - d
            G[:mk] += w[r] * u[d:]
            G[d:] -= w[r] * h[:mk]
    
    # dS/dx[i] = G[i] + G[n-1-i]
    grad = G[:m] + G[n-1:m-1:-1]

The L-BFGS-B approach with scipy will call the objective+gradient function maybe 50-200 times per run. Each call involves:
- Computing h from x: O(n)
- Computing raw = correlate(h, 1-h): O(n^2) ≈ 32K
- Computing weights: O(R) ≈ 359
- Computing gradient: O(R*n) ≈ 65K
Total: ~100K operations per call, ~200 calls = 20M operations ≈ 0.1-0.5 seconds.

This is very fast, so I can afford many L-BFGS-B runs with different betas and restarts.

import numpy as np
import time
import scipy.optimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _anti_map(x):
    """x of length m -> h of length 2m with anti-palindromic structure."""
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _anti_project(h):
    """Extract anti-palindromic x from h."""
    n = len(h)
    m = n // 2
    x = 0.5 * h[:m] + 0.5 * (1.0 - h[n-1:m-1:-1].copy())
    return np.clip(x, 0.0, 1.0)

def _smooth_grad_anti(x, beta):
    """Smooth-max objective and gradient for anti-palindromic parameterization."""
    m = len(x)
    n = 2 * m
    h = _anti_map(x)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    R = len(raw)
    
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 50.0)
    exp_z = np.exp(z - np.max(z))
    wsum = float(np.sum(exp_z))
    w = exp_z / wsum
    S = M + np.log(wsum) / beta
    
    # Gradient w.r.t. h
    G = np.zeros(n)
    # Only compute where w is significant
    for r in range(R):
        wr = w[r]
        if wr < 1e-10:
            continue
        k = r - (n - 1)
        if k >= 0:
            mk = n - k
            G[k:] += wr * u[:mk]
            G[:mk] -= wr * h[k:k+mk]
        else:
            d = -k
            mk = n - d
            G[:mk] += wr * u[d:]
            G[d:] -= wr * h[:mk]
    
    # dS/dx[i] = G[i] + G[n-1-i]
    grad = G[:m].copy()
    grad += G[n-1:m-1:-1]
    
    return S, grad

def _lbfgs_anti(x0, beta, max_time, max_iter=None):
    """Run L-BFGS-B on anti-palindromic parameterization."""
    m = len(x0)
    bounds = [(0.0, 1.0)] * m
    
    def fun_and_grad(x):
        S, g = _smooth_grad_anti(x, beta)
        return S, g
    
    options = {'maxiter': max_iter or 300, 'ftol': 1e-14, 'gtol': 1e-8, 'maxcor': 20}
    t_start = time.time()
    
    # Use callback to check time
    state = {'stop': False}
    def callback(xk):
        if time.time() - t_start > max_time:
            state['stop'] = True
        return state['stop']
    
    result = scipy.optimize.minimize(
        fun_and_grad, x0, jac=True, method='L-BFGS-B',
        bounds=bounds, options=options, callback=callback
    )
    
    x_opt = result.x
    h = _anti_map(x_opt)
    # Fix any numerical issues
    x_opt = np.clip(x_opt, 0.0, 1.0)
    return x_opt, h

def _line_search_exact(h, d, radius):
    """Exact line search for the C5 objective along direction d."""
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])) - 1e-15)
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])) + 1e-15)
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])) - 1e-15)
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg])) + 1e-15)
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Golden section + grid
    ng = 40
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    
    # Check vertex of each active quadratic
    top_idx = np.argsort(S0)[-10:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo - 1e-12 <= av <= a_hi + 1e-12:
                av = max(a_lo, min(a_hi, av))
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    
    return opt_t, opt_v

def _smooth_opt(h_init, deadline, beta_start=5.0, beta_end=60.0, radius=0.15):
    """Smooth gradient descent with exact line search."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    total_iters = 0
    
    while time.time() < deadline and stall < 60:
        total_iters += 1
        frac = min(1.0, float(total_iters) / 100.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        M = float(np.max(raw))
        z = beta * (raw - M)
        z = np.clip(z, -50.0, 50.0)
        exp_z = np.exp(z - np.max(z))
        w = exp_z / float(np.sum(exp_z))
        
        # Gradient
        G = np.zeros(n)
        for r in range(len(raw)):
            wr = w[r]
            if wr < 1e-10:
                continue
            k = r - (n - 1)
            if k >= 0:
                mk = n - k
                G[k:] += wr * u[:mk]
                G[:mk] -= wr * h[k:k+mk]
            else:
                d = -k
                mk = n - d
                G[:mk] += wr * u[d:]
                G[d:] -= wr * h[:mk]
        G -= np.mean(G)
        
        # Generate directions
        dirs = []
        gn = np.linalg.norm(G)
        if gn > 1e-12:
            dirs.append(G / gn)
        
        # Single-shift gradient
        r_max = int(np.argmax(raw))
        Gs = np.zeros(n)
        k = r_max - (n - 1)
        if k >= 0:
            mk = n - k
            Gs[k:] += u[:mk]
            Gs[:mk] -= h[k:k+mk]
        else:
            d = -k
            mk = n - d
            Gs[:mk] += u[d:]
            Gs[d:] -= h[:mk]
        Gs -= np.mean(Gs)
        Gsn = np.linalg.norm(Gs)
        if Gsn > 1e-12:
            dirs.append(Gs / Gsn)
        
        # Block directions
        for blk in [4, 8]:
            if n % blk == 0:
                gb = G.reshape(n // blk, blk).mean(axis=1)
                d = np.tile(gb, (1, blk)).reshape(n)
                d -= np.mean(d)
                dn = np.linalg.norm(d)
                if dn > 1e-12:
                    dirs.append(d / dn)
        
        # Random directions
        for _ in range(2):
            d = rng.normal(0, 1, n)
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                dirs.append(d / dn)
        
        # Try each direction
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search_exact(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        
        if best_d is not None and abs(best_alpha) > 1e-12:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
            else:
                stall += 1
                radius = max(0.03, radius * 0.8)
        else:
            stall += 1
            radius = max(0.03, radius * 0.85)
    
    return best_h, best_c5

def _greedy_improve(h_bin, deadline):

    n = len(h_bin)

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)

Hmm, wait. I realize there might be an issue with the L-BFGS-B approach. The smooth-max objective is not exactly the C5 objective, so L-BFGS-B might converge to a point that's good for the smooth approximation but not for the exact C5. I should always evaluate the exact C5 and only accept improvements based on that.

Also, I realize the `_smooth_grad_anti` function has a potential issue: when I compute the gradient, I need to be more careful about the indexing.

Let me reconsider the gradient computation.

For h of length n, raw[r] = sum_j h[j+k] * (1-h[j]) where k = r - (n-1).

d(raw[r])/d(h[i]):
- If k >= 0: raw[r] = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
  - h[i] appears as h[j+k] when j = i-k (valid if 0 <= i-k <= n-1-k, i.e., k <= i <= n-1)
    Contribution: (1-h[i-k])
  - h[i] appears as h[j] (in 1-h[j]) when j = i (valid if 0 <= i <= n-1-k)
    Contribution: -h[i+k]
  
  So d(raw[r])/d(h[i]) = (1-h[i-k]) * [k <= i <= n-1] - h[i+k] * [0 <= i <= n-1-k]

- If k < 0, let d = -k: raw[r] = sum_{j=d}^{n-1} h[j-d] * (1-h[j])
  = sum_{m=0}^{n-1-d} h[m] * (1-h[m+d])
  - h[i] appears as h[m] when m = i (valid if 0 <= i <= n-1-d)
    Contribution: (1-h[i+d])
  - h[i] appears as h[m+d] (in 1-h[m+d]) when m+d = i, i.e., m = i-d (valid if 0 <= i-d <= n-1-d, i.e., d <= i <= n-1)
    Contribution: -h[i-d]
  
  So d(raw[r])/d(h[i]) = (1-h[i+d]) * [0 <= i <= n-1-d] - h[i-d] * [d <= i <= n-1]

Let me verify with the example: h = [1,0,0,1], n=4.
For r=3 (k=0): raw[3] = sum_{j=0}^{3} h[j]*(1-h[j]) = 0
d(raw[3])/d(h[0]) = (1-h[0]) * [0<=0<=3] - h[0] * [0<=0<=3] = 0 - 1 = -1
d(raw[3])/d(h[1]) = (1-h[1]) * [0<=1<=3] - h[1] * [0<=1<=3] = 1 - 0 = 1
d(raw[3])/d(h[2]) = (1-h[2]) * [0<=2<=3]
