def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute LSE smoothed objective and its gradient."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    m = scores.max()
    u = (scores - m) / tau
    exp_u = np.exp(u)
    L = exp_u.sum()
    p = exp_u / L
    obj = m + tau * np.log(L)
    
    # Build gradient: sum_r p_r * grad_r(h)
    # grad_r for lag d:
    # d >= 0: grad[0:n-d] += (1-h[d:n]), grad[d:n] += -h[0:n-d]
    # d < 0 (g=-d): grad[g:n] += (1-h[0:n-g]), grad[0:n-g] += -h[g:n]

                        best_h = h.copy()
                    break
                alpha *= 0.5
            else:
                break
    return best_h, best_c5

                break

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _projected_grad_lse(h, deadline, tau_init=5.0, tau_end=0.03, n_taus=35,
                        iters_per_tau=25, lr_init=0.005, momentum_beta=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    taus = np.geomspace(tau_init, tau_end, n_taus)
    momentum = np.zeros(n)
    total_stall = 0

    for ti, tau in enumerate(taus):
        if time.time() > deadline:
            break
        lr = lr_init * (1.0 + 0.5 * ti / n_taus)
        for it in range(iters_per_tau):
            if time.time() > deadline:
                break
            obj, grad = _lse_obj_and_grad(h, tau)
            grad_proj = grad - grad.mean()
            gnorm = np.linalg.norm(grad_proj)
            if gnorm < 1e-10:
                break
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad_proj
            found = False
            for _ in range(40):
                h_try = _project_sum(h - lr * momentum, target)
                obj_try, _ = _lse_obj_and_grad(h_try, tau)

def _slsqp_lse(h, tau, maxiter=80):
    n = len(h)
    target = n / 2.0

    def obj_func(x):
        o, _ = _lse_obj_and_grad(x, tau)
        return o

    def grad_func(x):
        _, g = _lse_obj_and_grad(x, tau)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            return _project_sum(res.x, target)
    except Exception:
        pass
    return h.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _line_search_direction(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h).copy()
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

                                    n_pairs=12000, radius=0.25, seed=222)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary refinement
    if time.time() < d4:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_sa(h_b, min(d4, time.time() + budget_s * 0.04),
                               max_iter=35000, T_init=0.002, seed=777)
        h_b, c5_b = _pair_sweep(h_b, min(d4, time.time() + budget_s * 0.02), max_rounds=25)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # SLSQP final
    if time.time() < d4:
        h_sl = _project_sum(best_h.copy(), best_h.size / 2.0)
        h_sl, c5_sl = _slsqp_anneal(h_sl, min(d4, time.time() + budget_s * 0.03),
                                     tau_init=0.5, tau_end=0.03, n_taus=6, maxiter=50)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.90
    scale = 0.04
    n_restarts = 0
    while time.time() < d5 and n_restarts < 150:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _projected_grad_lse(h_pert, sub_dl, tau_init=1.5, tau_end=0.03,
                                         n_taus=15, iters_per_tau=10, lr_init=0.004)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 6: Final polish (10%)
    d6 = deadline
    n_best = len(best_h)
    for dn in [-4, 4, -8, 8, -2, 2, -12, 12]:
        if time.time() > d6 - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_r, c5_r = _projected_grad_lse(h_r, sub_dl, tau_init=1.0, tau_end=0.02,
                                         n_taus=15, iters_per_tau=10)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final LSE
    h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _projected_grad_lse(h6, deadline - 0.1, tau_init=0.5, tau_end=0.005,
                                    n_taus=20, iters_per_tau=15, lr_init=0.002)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Final pair sweep
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, deadline - 0.1, max_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
