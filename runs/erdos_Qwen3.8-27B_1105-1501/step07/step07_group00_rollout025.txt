def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        else:
            mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):

    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Golden section search on phi(t) = max(S0 + t*T - t^2*C) * dx
    # First find a good bracket via grid
    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(n_grid-1, best_idx + 3)]

def _active_lp(h, scores_raw, radius, n_active=12, gap=0.004):
    n = len(h)
    mx = float(np.max(scores_raw))
    # Active set: lags where score is within gap of max
    active = np.where(scores_raw >= mx - gap * mx)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores_raw))])
    if len(active) > n_active:
        order = np.argsort(scores_raw[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _gradient_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores_raw[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores_raw, beta=20.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores_raw))
    shifted = np.clip(beta * (scores_raw - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores_raw))
        w[int(np.argmax(scores_raw))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores_raw)) - (n - 1)
    G = _gradient_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Clip to stay within box
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _milp_binary(n, deadline, K_init=None):
    """Solve MILP to find binary sequence with min max overlap."""
    if n % 2 != 0:
        return None
    if n < 20 or n > 120:
        return None
    remaining = deadline - time.time()
    if remaining < 20:
        return None
    
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    # Variables
    x = cp.Variable(n, boolean=True)
    # Auxiliary variables for |x_{j+d} - x_j|
    # For each d from 1 to n-1, and j from 0 to n-d-1
    aux_vars = {}
    for d in range(1, n):
        for j in range(n - d):
            aux_vars[(j, d)] = cp.Variable(boolean=True)
    
    # Objective variable K
    K = cp.Variable(integer=True)
    
    constraints = [x >= 0, x <= 1]
    constraints.append(cp.sum(x) == n // 2)
    
    # For each distance d, compute E_d and B_d
    for d in range(1, n):
        E_d = 0
        B_d = 0
        for j in range(n - d):
            y = aux_vars[(j, d)]
            E_d += y
            B_d += x[j + d] - x[j]
            # Constraints: y >= x_{j+d} - x_j
            constraints.append(y >= x[j + d] - x[j])
            # y >= x_j - x_{j+d}
            constraints.append(y >= x[j] - x[j + d])
        
        # R_+(d) = (E_d + B_d) / 2 <= K
        constraints.append(E_d + B_d <= 2 * K)
        # R_-(d) = (E_d - B_d) / 2 <= K
        constraints.append(E_d - B_d <= 2 * K)
    
    if K_init is not None:
        constraints.append(K <= K_init)
    
    # Minimize K
    obj = cp.Minimize(K)
    prob = cp.Problem(obj, constraints)
    
    # Try to solve
    solve_time = min(remaining * 0.8, 300)
    try:
        prob.solve(solver=cp.CBC, time_limit=solve_time)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP, time_limit=solve_time)
        except Exception:
            return None
    
    if x.value is None:
        return None
    
    xv = np.array(x.value).flatten()
    # Round to binary
    xv = np.round(xv)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(xv))
    if n_ones != n // 2:
        # Fix by adjusting
        if n_ones > n // 2:
            # Remove some ones
            ones_idx = np.where(xv == 1)[0]
            to_remove = n_ones - n // 2
            xv[ones_idx[:to_remove]] = 0
        else:
            zeros_idx = np.where(xv == 0)[0]
            to_add = n // 2 - n_ones
            xv[zeros_idx[:to_add]] = 1
    
    xv = xv.astype(float)
    return xv

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):

    idx = np.argsort(h)[-n // 2:]

    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 4
    freqs = rng.uniform(0.5, 10.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amplitudes = rng.uniform(0.4, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
