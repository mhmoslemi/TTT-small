The number of changed entries: for pair (i,j), the changed lags are those d where the sum for lag d involves position i or j. As I computed earlier, this is at most about 2n lags (but in practice, for a random pair, it's about 2*(n/2) = n lags on each side, so about 2n total... hmm that's a lot).

Wait, let me reconsider. For a specific lag d, the sum is over m = 0 to n-d-1. The term at m involves positions m and m+d. So position i is involved in the sum for lag d if:
- i = m for some m in [0, n-d-1], i.e., d ≤ n-1-i
- i = m+d for some m in [0, n-d-1], i.e., m = i-d ∈ [0, n-d-1], which requires d ≤ i and i-d ≤ n-d-1 (the latter is i ≤ n-1, always true)

So for position i, the affected positive lags d are: d = 0, 1, ..., max(n-1-i, i).
The number is max(n-1-i, i) + 1.

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target} using water filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_optimize(h, i, j, R0):
    """Find delta that minimizes max of correlate(h+delta*e_i - delta*e_j, 1-that)
    where e_i, e_j are standard basis vectors.
    R0 is the precomputed correlate(h, 1-h, full) for the current h."""
    n = len(h)
    # Feasible range for delta
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = d_lo
    b = d_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = h.copy()
    hd = h.copy()
    hc[i] += c; hc[j] -= c
    hd[i] += d; hd[j] -= d
    fc = float(np.max(np.correlate(hc, 1.0 - hc, mode='full')))
    fd = float(np.max(np.correlate(hd, 1.0 - hd, mode='full')))
    
    for _ in range(50):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            hc[i] = h[i] + c; hc[j] = h[j] - c
            fc = float(np.max(np.correlate(hc, 1.0 - hc, mode='full')))
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            hd[i] = h[i] + d; hd[j] = h[j] - d
            fd = float(np.max(np.correlate(hd, 1.0 - hd, mode='full')))
    
    delta = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = h[i] + delta
    h_new[j] = h[j] - delta
    h_new = np.clip(h_new, 0.0, 1.0)
    f_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))
    return delta, f_new
