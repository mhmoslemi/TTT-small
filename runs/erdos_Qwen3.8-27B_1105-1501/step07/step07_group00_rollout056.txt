    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _quadratic_line_search(h, d, dx, t_max):
    # Exact quadratic model along direction d
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find feasible t range
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = t_max
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Sample the quadratic
    n_samp = 80
    t_grid = np.linspace(0.0, t_hi, n_samp)
    f_grid = np.empty(n_samp)
    for i in range(n_samp):
        t = t_grid[i]
        f_grid[i] = float(np.max(S0 + t * T - t * t * C)) * dx
    
    best_i = int(np.argmin(f_grid))
    # Golden section on local neighborhood
    lo = t_grid[max(0, best_i - 3)]
    hi = t_grid[min(n_samp-1, best_i + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-12:
        return t_grid[best_i], f_grid[best_i]
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    
    t_opt = (lo + hi) / 2.0
    f_opt = float(np.max(S0 + t_opt*T - t_opt*t_opt*C)) * dx
    
    # Check endpoints too
    cands = [(t_opt, f_opt), (0.0, float(np.max(S0))*dx), (t_hi, float(np.max(S0+t_hi*T-t_hi*t_hi*C))*dx), (t_grid[best_i], f_grid[best_i])]
    return min(cands, key=lambda x: x[1])

def _lp_direction(h, scores, radius, n_active, gap):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_descent_direction(h, scores, beta, radius):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    
    # Clip to feasible direction
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g[pos & (h < 1.0 - 1e-12)] = np.clip(g[pos & (h < 1.0 - 1e-12)], -h[pos & (h < 1.0 - 1e-12)]/radius, (1.0-h[pos & (h < 1.0 - 1e-12)])/radius)
    g[neg & (h > 1e-12)] = np.clip(g[neg & (h > 1e-12)], -h[neg & (h > 1e-12)]/radius, (1.0-h[neg & (h > 1e-12)])/radius)
    g -= float(np.mean(g))
    return g

def _smooth_perturb(h, amplitude, rng):
    n = len(h)

                if cur < best_c - 1e-16:
                    best_c = cur
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for k in range(1, 7):
        f = k + rng.exponential(1.5)
        ph = rng.uniform(0, 2 * np.pi)
        a = amp / (k + 0.5) * rng.uniform(0.3, 1.2)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p += amp * 0.15 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def _swap_perturb(h, rng, n_swaps=5):
    n = len(h)
    h2 = h.copy()
    idx = rng.permutation(n)
    n_ones = int(np.sum(h2 > 0.5))
    for s in range(n_swaps):
        i = idx[s % n]
        j = idx[(s * 7 + 3) % n]
        if i != j:
            h2[i], h2[j] = h2[j], h2[i]
    return _project(h2, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Initial local optimization (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _local_opt(h_init, d1, radius_init=0.12, max_iters=4000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair sweep (8%)
    d2 = t0 + budget_s * 0.26
    h2, c5_2 = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), d2, max_rounds=200, rng_seed=777)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (22%)
    d3 = t0 + budget_s * 0.48
    n_curr = len(best_h)
    tried = set([n_curr])
    offsets = list(range(-24, 25, 2))
    offsets.sort(key=lambda x: abs(x))
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.10, max_iters=800)
        if c5_r < best_c5 + 0.002 and time.time() < d3 - 10:
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, max_rounds=60, rng_seed=1234)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Basin hopping with diverse perturbations (25%)
    d4 = t0 + budget_s * 0.73
    scale = 0.035
    n_restart = 0
    while time.time() < d4 and n_restart < 400:
        n_restart += 1
        if n_restart % 5 == 0:
            h_pert = _swap_perturb(best_h, rng, n_swaps=3)
        else:
            h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.005)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.07, max_iters=600)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.18, scale * 1.12)

    # Phase 5: Intensive final polish (20%)
    d5 = deadline - 3.0
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, d5, radius_init=0.05, max_iters=8000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final pair sweep if time remains
    rem = deadline - time.time()
    if rem > 8:
        h_fs = _project(best_h.copy(), len(best_h) / 2.0)
        h_fs, c5_fs = _pair_sweep(h_fs, deadline - 1, max_rounds=150, rng_seed=9999)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Tiny final touch
    if deadline - time.time() > 4:
        n_ff = len(best_h)
        h_ff = _project(best_h.copy(), n_ff / 2.0)
        h_ff, c5_ff = _local_opt(h_ff, deadline - 0.5, radius_init=0.03, max_iters=1000)
        if c5_ff < best_c5:
            best_c5 = c5_ff
            best_h = h_ff.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
