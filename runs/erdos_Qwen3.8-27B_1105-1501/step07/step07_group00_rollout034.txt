def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    res = target - float(np.sum(x))

```python
def _binary_sa(b, deadline, max_iter=500000):
    n = len(b)
    dx = 2.0 / n
    b = b.copy()
    counts = np.correlate(b, 1-b, 'full')
    cur_max = int(counts.max())
    best_b = b.copy()
    best_max = cur_max
    rng = np.random.default_rng(123)
    T = 2.0  # in units of count (not C5)
    ones = np.where(b == 1)[0].tolist()
    zeros = np.where(b == 0)[0].tolist()
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 10000:
            break
        oi = int(rng.integers(len(ones)))
        zi = int(rng.integers(len(zeros)))
        r = ones[oi]  # position to turn off
        s = zeros[zi]  # position to turn on
        
        # Compute new counts
        b[r] = 0
        b[s] = 1
        new_counts = np.correlate(b, 1-b, 'full')
        new_max = int(new_counts.max())
        
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            ones[oi] = s
            zeros[zi] = r
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
                stall = 0
            T *= 0.9999
        else:
            b[r] = 1
            b[s] = 0
            stall += 1
            T = max(T * 0.9999, 0.5)
    
    return best_b, best_max * dx
```
