    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = float(F.max())

    exps = np.exp(diff)

    p = exps / S
    c = n - 1

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = float(F.max())
    diff = (F - m) / tau
    diff = np.clip(diff, -500.0, 0.0)
    exps = np.exp(diff)
    S = float(np.sum(exps))
    value = m + tau * np.log(S)
    p = exps / S
    c = n - 1
    grad = np.zeros(n)
    grad += p[c] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = p[c + k]
        wm = p[c - k]
        wsum = wp + wm
        if n - k > 0:
            grad[:n-k] += wm - wsum * h[k:n]
        grad[k:] += wp - wsum * h[:n-k]
    grad *= dx
    return value, grad

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = float(F.max())
    diff = (F - m) / tau
    diff = np.clip(diff, -500.0, 0.0)
    exps = np.exp(diff)
    S = float(np.sum(exps))
    value = m + tau * np.log(S)
    p = exps / S
    c = n - 1
    grad = np.zeros(n)
    grad += p[c] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = p[c + k]
        wm = p[c - k]
        wsum = wp + wm
        if n - k > 0:
            grad[:n-k] += wm - wsum * h[k:n]
        grad[k:] += wp - wsum * h[:n-k]
    grad *= dx
    return value, grad

def _sum_constraint(x, target):
    return float(np.sum(x) - target)

def _sum_jacobian(x, target):
    return np.ones_like(x)

def slsqp_optimize(h_init, tau, target, maxiter=150, tol=1e-9):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: _sum_constraint(x, target),
                    'jac': lambda x: _sum_jacobian(x, target)}]
    try:
        res = minimize(
            lambda x: smooth_value_grad(x, tau)[0],
            h_init.copy(),
            jac=lambda x: smooth_value_grad(x, tau)[1],
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'iprint': 0, 'ftol': tol}
        )
        return res.x
    except Exception:
        return h_init.copy()

def to_binary(h, n):

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_wave = 5
    perturb = np.zeros(n)
    for _ in range(n_wave):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return project(h + perturb, n / 2.0)

def line_search_dir(h, d, target, radius=0.1):

def grad_descent_loop(h, deadline, radius_init=0.08, max_iter=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        active = np.where(corr >= max_val - 1e-10)[0]
        g = np.zeros(n)
        for m in active:
            k = int(m - (n - 1))
            if k >= 0:
                idx1 = np.arange(k, n)
                g[idx1] += (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                g[idx2] -= h[idx2 + k]
            else:
                ka = -k
                idx1 = np.arange(0, n - ka)
                g[idx1] += (1.0 - h[idx1 + ka])
                idx2 = np.arange(ka, n)
                g[idx2] -= h[idx2 - ka]
        g /= len(active)
        d = -g
        d = d - np.mean(d)
        alpha, new_c5 = line_search_dir(h, d, target, radius)
        if new_c5 < cur - 1e-13:

            radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
    return best_h, best_c5

    n_best = len(best_h)
    h_bin = to_binary(best_h, n_best)
    c5_bin = exact_c5(h_bin)
    if c5_bin < best_c5 - 1e-12:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Targeted 2-exchange
    sub_dl = min(d3, time.time() + budget_s * 0.10)
    h_b, c5_b = binary_2exchange(h_bin, sub_dl, seed=101)
    if c5_b < best_c5 - 1e-12:
        best_c5 = c5_b
        best_h = h_b.copy()

    # More 2-exchange with different seed
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_b, c5_b = binary_2exchange(h_bin, sub_dl, seed=202)
    if c5_b < best_c5 - 1e-12:
        best_c5 = c5_b
        best_h = h_b.copy()

    # SA
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_b, c5_b = binary_sa(h_bin, sub_dl, seed=303, temp_init=0.002)
    if c5_b < best_c5 - 1e-12:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Block swaps
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_b, c5_b = block_swap(h_bin, sub_dl, seed=404)
    if c5_b < best_c5 - 1e-12:
        best_c5 = c5_b
        best_h = h_b.copy()

    # More 2-exchange from best binary
    if time.time() < d3 - 10:
        h_b2 = to_binary(best_h, len(best_h))
        sub_dl = min(d3, time.time() + budget_s * 0.06)
        h_b, c5_b = binary_2exchange(h_b2, sub_dl, seed=505)
        if c5_b < best_c5 - 1e-12:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: Multi-n exploration (15%)
    d4 = t0 + budget_s * 0.85
    tried_ns = set([len(best_h)])
    for dn in range(-30, 31, 3):
        if time.time() > d4:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = resample(best_h, n_try)
        # Quick SLSQP
        if time.time() < d4 - 5:
            target_r = n_try / 2.0
            h_r = slsqp_optimize(h_r, 0.002, target_r, maxiter=80, tol=1e-7)
            h_r = project(h_r, target_r)
        c5_r = exact_c5(h_r)
        # Try binary
        if time.time() < d4 - 3:
            h_br = to_binary(h_r, n_try)
            sub_dl = min(d4, time.time() + budget_s * 0.01)
            h_br, c5_br = binary_2exchange(h_br, sub_dl, seed=1000 + n_try, max_rounds=30000)
            if c5_br < c5_r:
                c5_r = c5_br
                h_r = h_br
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final intensive polish (20%)
    d5 = deadline
    n_fin = len(best_h)
    target_fin = n_fin / 2.0

    # SLSQP with very small tau
    if time.time() < d5 - 20:
        sub_dl = min(d5, time.time() + budget_s * 0.06)
        h_s = slsqp_optimize(best_h, 0.0002, target_fin, maxiter=150, tol=1e-10)
        h_s = project(h_s, target_fin)
        c5_s = exact_c5(h_s)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Gradient descent final
    if time.time() < d5 - 15:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_gd, c5_gd = grad_descent_loop(best_h, sub_dl, radius_init=0.03, max_iter=1500)
        if c5_gd < best_c5 - 1e-12:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    # Binary final
    if time.time() < d5 - 10:
        h_bf = to_binary(best_h, n_fin)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h_bf, c5_bf = binary_2exchange(h_bf, sub_dl, seed=777, max_rounds=80000)
        if c5_bf < best_c5 - 1e-12:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Pair swap on continuous
    if time.time() < d5 - 10:
        n_ps = len(best_h)
        target_ps = n_ps / 2.0
        h_ps = best_h.copy()
        cur_ps = best_c5
        stall_ps = 0
        radius_ps = 0.15
        for _ in range(5000):
            if time.time() > d5 - 5:
                break
            if stall_ps > 150:
                break
            i = int(rng.integers(0, n_ps))
            j = int(rng.integers(0, n_ps))
            if i == j:
                stall_ps += 1
                continue
            d = np.zeros(n_ps)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = line_search_dir(h_ps, d, target_ps, radius_ps)
            if c5_try < cur_ps - 1e-13:
                h_ps = h_ps + alpha * d
                h_ps = project(h_ps, target_ps)
                cur_ps = exact_c5(h_ps)
                if cur_ps < best_c5:
                    best_c5 = cur_ps
                    best_h = h_ps.copy()
                stall_ps = 0
            else:
                stall_ps += 1

    # Basin hopping
    if time.time() < d5 - 8:
        scale = 0.015
        for _ in range(30):
            if time.time() > d5 - 3:
                break
            h_p = smooth_perturb(best_h, scale, rng)
            h_p = slsqp_optimize(h_p, 0.001, len(h_p) / 2.0, maxiter=60, tol=1e-8)
            h_p = project(h_p, len(h_p) / 2.0)
            c5_p = exact_c5(h_p)
            if c5_p < best_c5 - 1e-12:
                best_c5 = c5_p
                best_h = h_p.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.05, scale * 1.1)

    # Final projection and verification
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
