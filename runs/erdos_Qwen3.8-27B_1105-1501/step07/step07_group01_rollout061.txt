    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            mask = (x < 1.0)
        else:
            mask = (x > 0.0)

            x[mask] += res / cnt

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.2, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(25):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-13:
            if best_v == f0:
                continue
            elif best_v == opt_val:
                h = h + opt_t * d
            else:
                h = h + t_hi * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _subgrad_step(h, beta_top=5.0):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    S_max = S.max()
    active = S >= S_max - beta_top / n
    n_active = int(np.sum(active))
    if n_active == 0:
        return np.zeros(n)
    W = np.zeros(len(S))
    W[active] = 1.0 / n_active
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = W[k]
        if w < 1e-16:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            grad[s:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[s:])
        else:
            sp = -s
            m = n - sp
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[sp:])
            grad[sp:] += w * (-h[:m])
    return grad

def _subgrad_optimize(h, deadline, lr_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        if stall > 300:
            if lr > 0.01:
                lr *= 0.5
                stall = 0
            else:
                break
        grad = _subgrad_step(h)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12:
            break
        t = lr
        improved = False
        for _ in range(8):
            h_t = _project(h - t * grad, target)
            f_t = _exact_c5(h_t)
            if f_t < best_c5 - 1e-15:
                h = h_t
                best_c5 = f_t
                best_h = h.copy()
                improved = True
                break
            t *= 0.4
        if improved:
            stall = 0
        else:
            stall += 1
            lr = max(0.005, lr * 0.8)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE gradient descent (20% of budget)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lse_optimize(h_init, d1, beta_init=30.0, beta_max=5000.0, lr_init=0.25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (30%)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.12), max_sweeps=8000, seed=111)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    h2b, c5_2b = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.10), max_sweeps=6000, seed=222)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: LP polish (10%)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _lp_polish(best_h, d3, radius_init=0.07, stall_limit=50, top_k=14)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Subgradient method (8%)
    d4 = t0 + budget_s * 0.68
    h4, c5_4 = _subgrad_optimize(best_h, d4, lr_init=0.12)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.83
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=800, seed=3000 + n_restarts)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.06)

    # Phase 6: Multi-n exploration (7%)
    d6 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    tried_ns = set()
    offset_list = list(range(-16, 17, 3))
    for dn in offset_list:
        if time.time() > d6:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns or n_try == n_curr:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=400, seed=5000 + dn)
        if time.time() < d6 - 3:
            sub_dl2 = min(d6, time.time() + budget_s * 0.003)
            h_r, c5_r2 = _lse_optimize(h_r, sub_dl2, beta_init=50.0, beta_max=3000.0, lr_init=0.15)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = len(best_h)

    # Phase 7: Final intensive optimization (10%)
    d7 = deadline
    n_final = len(best_h)
    sub_dl = min(d7, time.time() + budget_s * 0.03)
    h7, c5_7 = _coord_descent(best_h, sub_dl, max_sweeps=5000, seed=9999)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7 - 3:
        sub_dl = min(d7, time.time() + budget_s * 0.025)
        h7, c5_7 = _lp_polish(best_h, sub_dl, radius_init=0.04, stall_limit=40, top_k=18)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    if time.time() < d7 - 3:
        sub_dl = min(d7, time.time() + budget_s * 0.02)
        h7, c5_7 = _pair_swap(best_h, sub_dl, n_pairs=3000, radius=0.15, seed=7777)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    if time.time() < d7 - 2:
        sub_dl = min(d7, time.time() + budget_s * 0.015)
        h7, c5_7 = _coord_descent(best_h, sub_dl, max_sweeps=3000, seed=12345)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final projection and exact evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
