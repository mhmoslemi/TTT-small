def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _autocorr(p):
    return np.correlate(p, p, mode='full')

def _project_sum(v, target):
    n = len(v)
    if n == 0:
        return np.zeros(0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _safe_alpha(h, p, max_alpha):
    pos = p > 1e-15
    neg = p < -1e-15
    hi = max_alpha
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        hi = min(hi, float(np.min(h[neg] / (-p[neg]))))
    return max(0.0, hi)

def _line_search_exact(h, R, T, P, dx, alpha_max):
    S0 = R
    n_grid = 60
    grid_t = np.linspace(0.0, alpha_max, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * P)) * dx
    best_idx = int(np.argmin(grid_f))
    best_val = grid_f[best_idx]
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-12:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * P)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * P)) * dx
        for _ in range(50):

                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)

            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)

        opt_t = (lo_b + hi_b) / 2.0

def _full_epigraph_direction(h, R, G, radius, margin, max_active):
    n = len(h)
    max_score = float(np.max(R))
    active = np.where(R >= max_score - margin)[0]
    if len(active) > max_active:
        active = np.argsort(R)[::-1][:max_active]
    m = len(active)
    if m == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -R[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15, margin_init=0.3, shrink=0.75, grow=1.2,

    margin = margin_init
    stall = 0

    while time.time() < deadline:
        R = _scores(h)
        G = _grad_matrix(h)
        p = _full_epigraph_direction(h, R, G, radius, margin, max_active)

            stall += 1
            radius *= shrink
            margin *= 0.9
            if radius < min_radius:
                break
            if stall > stall_limit:
                break
            continue
        T = G @ p
        P = _autocorr(p)
        alpha_max = _safe_alpha(h, p, 1.0)

            stall += 1
            radius *= shrink
            continue
        best_t, best_f = _line_search_exact(h, R, T, P, dx, alpha_max)

                stall += 1
                radius *= shrink
        else:
            stall += 1
            radius *= shrink
            margin = max(0.02, margin * 0.8)

def _pair_descent(h, deadline, n_attempts=20000, seed=42):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_attempts):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        di = h[i]
        dj = h[j]
        if di <= 0 or dj >= 1:
            stall += 1
            continue
        delta = -min(di, 1.0 - dj) * rng.uniform(0.3, 1.0)
        h[i] = di + delta
        h[j] = dj - delta
        h[i] = float(np.clip(h[i], 0.0, 1.0))
        h[j] = float(np.clip(h[j], 0.0, 1.0))
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-13:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            h[i] = di
            h[j] = dj
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    half = n // 2
    idx = np.argsort(h)[-half:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _slsqp_optimize(h, deadline, tau_start=0.5, tau_end=0.01, n_tau=6):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_start
    ts = np.linspace(np.log(tau_start), np.log(tau_end), n_tau)
    for li in range(n_tau):
        if time.time() > deadline:
            break
        tau = float(np.exp(ts[li]))
        def obj(v):
            R = _scores(v)
            scaled = R / tau
            mx = float(np.max(scaled))
            if mx > 500:
                return 1e6
            s = float(np.sum(np.exp(scaled - mx)))
            return tau * (mx + np.log(s)) * dx
        def grad(v):
            R = _scores(v)
            G = _grad_matrix(v)
            scaled = R / tau
            mx = float(np.max(scaled))
            exps = np.exp(scaled - mx)
            w = exps / float(np.sum(exps))
            return (G.T @ w) * dx
        cons = [{'type': 'eq', 'fun': lambda v: float(np.sum(v)) - target},
                {'type': 'ineq', 'fun': lambda v: float(np.min(v))},
                {'type': 'ineq', 'fun': lambda v: float(np.min(1.0 - v))}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success or res.fun < best_c5:
                h_try = _project_sum(res.x, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        except Exception:
            pass
    return best_h, best_c5

                             stall_limit=35, max_active=200)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.007)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: SLSQP smooth optimization (15%)
    d4 = t0 + budget_s * 0.62
    if time.time() < d4:
        h_s = _project_sum(best_h.copy(), target)
        sub_dl = min(d4, time.time() + budget_s * 0.12)
        h_s, c5_s = _slsqp_optimize(h_s, sub_dl, tau_start=0.3, tau_end=0.02, n_tau=5)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        # LP refinement after SLSQP
        if time.time() < d4:
            h_l = _project_sum(best_h.copy(), target)
            sub_dl2 = min(d4, time.time() + budget_s * 0.03)
            h_l, c5_l = _lp_loop(h_l, sub_dl2, radius_init=0.08, margin_init=0.2,
                                 stall_limit=30, max_active=180)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()

    # Phase 5: Intensive LP + pair descent (20%)
    d5 = t0 + budget_s * 0.82
    n_cur = len(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.08)
    h5 = _project_sum(best_h.copy(), target)
    h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.12, margin_init=0.4,
                        shrink=0.72, grow=1.2, max_radius=0.35,
                        stall_limit=80, max_active=220)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < d5:
        sub_dl2 = min(d5, time.time() + budget_s * 0.06)
        h5, c5_5 = _pair_descent(best_h, sub_dl2, n_attempts=15000, seed=333)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Binary refinement
    if time.time() < d5:
        h_b = _to_binary(best_h)
        sub_dl3 = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl3, max_iter=30000, T_init=0.002, seed=555)
        if time.time() < d5:
            sub_dl4 = min(d5, time.time() + budget_s * 0.015)
            h_b, c5_b = _pair_sweep_binary(h_b, sub_dl4, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Final LP on best
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), target)
        sub_dl5 = min(d5, time.time() + budget_s * 0.02)
        h5, c5_5 = _lp_loop(h5, sub_dl5, radius_init=0.05, margin_init=0.15,
                            stall_limit=40, max_active=180)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping (12%)
    d6 = t0 + budget_s * 0.94
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 and n_restarts < 300:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d6, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.10, margin_init=0.3,
                             shrink=0.75, grow=1.2, max_radius=0.25,
                             stall_limit=20, max_active=150)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.05)

    # Phase 7: Final polish (6%)
    d7 = deadline
    # Try nearby n values
    n_best = len(best_h)
    for dn in [-4, 4, -2, 2, -6, 6]:
        if time.time() > d7 - 3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.06, margin_init=0.2,
                             stall_limit=25, max_active=180)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final LP
    if time.time() < d7 - 2:
        h7 = _project_sum(best_h.copy(), target)
        sub_dl = min(d7, time.time() + budget_s * 0.03)
        h7, c5_7 = _lp_loop(h7, sub_dl, radius_init=0.04, margin_init=0.1,
                            shrink=0.8, grow=1.1, max_radius=0.15,
                            stall_limit=60, max_active=200)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    # Final binary check
    if deadline - time.time() > 4:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep_binary(h_b, deadline, max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
