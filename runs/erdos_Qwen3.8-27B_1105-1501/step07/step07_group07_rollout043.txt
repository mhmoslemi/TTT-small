def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    S0 = S0.copy()

        return float(np.max(S0 + t * T - t * t * C)) * dx

    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([_f(t) for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Golden section
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]

    else:

        c2 = lo_b + gr * (hi_b - lo_b)

    try:

    scores is the full (2n-1) score array before the swap.
    Returns the updated scores array.

    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _binary_max(h):
    return float(np.max(_scores(h)))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, T_init=0.8, seed=42, use_2swap=True):
    n = len(h_init)
    dx = 2.0 / n
    cur = h_init.copy()
    cur_s = _scores(cur)
    cur_max = float(np.max(cur_s))
    best = cur.copy()
    best_max = cur_max
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    while time.time() < deadline:
        if stall > 8000:
            break
        do_2swap = use_2swap and stall > 2000 and rng.random() < 0.3
        if do_2swap and len(ones) >= 2 and len(zeros) >= 2:
            o1 = int(rng.integers(0, len(ones)))
            o2 = int(rng.integers(0, len(ones)))
            if o2 == o1:
                o2 = (o1 + 1) % len(ones)
            z1 = int(rng.integers(0, len(zeros)))
            z2 = int(rng.integers(0, len(zeros)))
            if z2 == z1:
                z2 = (z1 + 1) % len(zeros)
            a, b = ones[o1], ones[o2]
            c, d_ = zeros[z1], zeros[z2]
            cur[a] = 0.0; cur[b] = 0.0; cur[c] = 1.0; cur[d_] = 1.0
            new_s = _scores(cur)
            new_max = float(np.max(new_s))
            delta = new_max - cur_max
            if delta < -1e-12 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_s = new_s
                cur_max = new_max
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                if cur_max < best_max - 1e-12:
                    best_max = cur_max
                    best = cur.copy()
                    stall = 0
                T = max(0.05, T * 0.9997)
            else:
                cur[a] = 1.0; cur[b] = 1.0; cur[c] = 0.0; cur[d_] = 0.0
                stall += 1
        else:
            oi = int(rng.integers(0, len(ones)))
            oz = int(rng.integers(0, len(zeros)))
            i, j = ones[oi], zeros[oz]
            cur[i] = 0.0
            cur[j] = 1.0
            new_s = _scores(cur)
            new_max = float(np.max(new_s))
            delta = new_max - cur_max
            if delta < -1e-12 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_s = new_s
                cur_max = new_max
                ones[oi] = j
                zeros[oz] = i
                if cur_max < best_max - 1e-12:
                    best_max = cur_max
                    best = cur.copy()
                    stall = 0
                T = max(0.05, T * 0.9999)
            else:
                cur[i] = 1.0
                cur[j] = 0.0
                stall += 1
    return best, best_max * dx

def _greedy_sweep(h_init, deadline, max_rounds=30):
    n = len(h_init)
    dx = 2.0 / n
    cur = h_init.copy()
    cur_max = float(np.max(_scores(cur)))
    best = cur.copy()
    best_max = cur_max
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0].tolist()
        zeros = np.where(cur <= 0.5)[0].tolist()
        for i in ones:
            if time.time() > deadline:
                return best, best_max * dx
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                nm = float(np.max(_scores(cur)))
                if nm < cur_max - 1e-12:
                    cur_max = nm
                    improved = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best = cur.copy()
                    break
                else:
                    cur[i] = 1.0
                    cur[j] = 0.0
        if not improved:
            break
    return best, best_max * dx

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(1, n):
        r = n - 1 + lag
        m = n - lag
        T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag+m]) - h[:m] * d[lag:lag+m]))
        C[r] = float(np.sum(d[:m] * d[lag:lag+m]))
        T[n-1-lag] = float(np.sum(d[lag:] * (1.0 - h[:m]) - h[lag:] * d[:m]))
        C[n-1-lag] = float(np.sum(d[lag:] * d[:m]))
    
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    def _f(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    n_grid = 25
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([_f(t) for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    if hi_b - lo_b < 1e-12:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1, f2 = _f(c1), _f(c2)
        for _ in range(25):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c1, hi_b - gr*(hi_b - lo_b), f1
                f1 = _f(c1)
            else:
                lo_b, c1, f1 = c2, lo_b + gr*(hi_b - lo_b), f2
                f2 = _f(c2)
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = _f(opt_t)
    
    cands = [(0.0, float(np.max(S0)*dx)), (t_hi, _f(t_hi)), (opt_t, opt_val)]
    return min(cands, key=lambda x: x[1])

def _lp_direction(h, scores, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, n] = -1.0
    b_ub = np.zeros(top_k)
    for idx, r in enumerate(top_idx):
        lag = r - (n - 1)
        if lag >= 0:
            m = n - lag
            A_ub[idx, :m] += (1.0 - h[lag:lag+m])
            A_ub[idx, lag:lag+m] += (-h[:m])
        else:
            g = -lag
            m = n - g
            A_ub[idx, g:] += (1.0 - h[:m])
            A_ub[idx, :m] += (-h[g:])
        b_ub[idx] = -scores[r]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius=0.08, top_k=4, stall_limit=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius, top_k)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.2)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _perturb_binary(h_bin, rng, n_swaps):
    n = len(h_bin)
    cur = h_bin.copy()
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(n_swaps):
        if not ones or not zeros:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        cur[ones[oi]] = 0.0
        cur[zeros[oz]] = 1.0
        ones[oi] = zeros[oz]
        zeros[oz] = cur[ones[oi]] if False else ones[oz] if oz < len(ones) else zeros[oz]
        # Simpler: just redo
    # Redo properly
    cur = h_bin.copy()
    for _ in range(n_swaps):
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        cur[ones[i]] = 0.0
        cur[zeros[j]] = 1.0
    return cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
    n0 = (len(h0) // 2) * 2
    if n0 != len(h0):
        h0 = _resample(h0, n0)
    best_h = _project_sum(h0.copy(), len(h0) / 2.0)
    best_c5 = _exact_c5(best_h)
    
    # Phase 1: Binary SA + greedy on parent-projected, multiple n (55%)
    d1 = t0 + budget_s * 0.55
    n_center = len(best_h)
    n_list = []
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16]:
        nt = n_center + dn
        if 100 <= nt <= 260 and nt % 2 == 0:
            n_list.append(nt)
    n_list = sorted(set(n_list))
    
    for n_try in n_list:
        if time.time() > d1:
            break
        h_r = _resample(best_h, n_try)
        h_bin = _to_binary(h_r)
        # SA
        sub_dl = min(d1, time.time() + budget_s * 0.025)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, T_init=1.0, seed=n_try * 7 + 1)
        # Greedy
        if time.time() < d1 - 2:
            sub_dl2 = min(d1, time.time() + budget_s * 0.008)
            h_bin, c5_bin = _greedy_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    # Phase 2: More binary SA with different starts (20%)
    d2 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    for restart in range(15):
        if time.time() > d2:
            break
        n_swaps = n_cur // 4 + restart * 3
        h_pert = _perturb_binary(best_h, rng, n_swaps)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_b, c5_b = _binary_sa(h_pert, sub_dl, T_init=0.6, seed=5000 + restart * 13)
        if time.time() < d2 - 1:
            sub_dl2 = min(d2, time.time() + budget_s * 0.004)
            h_b, c5_b = _greedy_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Phase 3: Explore wider n range with random structured starts (15%)
    d3 = t0 + budget_s * 0.90
    for n_try in range(120, 261, 2):
        if time.time() > d3:
            break
        if abs(n_try - n_cur) < 4:
            continue
        # Random start
        h_rand = np.zeros(n_try)
        idx = rng.choice(n_try, size=n_try // 2, replace=False)
        h_rand[idx] = 1.0
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_b, c5_b = _binary_sa(h_rand, sub_dl, T_init=1.5, seed=n_try * 3 + 100)
        if time.time() < d3 - 1:
            sub_dl2 = min(d3, time.time() + budget_s * 0.004)
            h_b, c5_b = _greedy_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Phase 4: Non-binary LP polish (10%)
    d4 = deadline
    n_best = len(best_h)
    h_pol = _project_sum(best_h.copy(), n_best / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h_pol, c5_pol = _lp_loop(h_pol, sub_dl, radius=0.05, top_k=4, stall_limit=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Final binary polish if still binary-like
    if time.time() < deadline - 5:
        h_b = _to_binary(best_h)
        sub_dl = min(deadline, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_sa(h_b, sub_dl, T_init=0.3, seed=9999)
        sub_dl2 = min(deadline, time.time() + budget_s * 0.01)
        h_b, c5_b = _greedy_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
