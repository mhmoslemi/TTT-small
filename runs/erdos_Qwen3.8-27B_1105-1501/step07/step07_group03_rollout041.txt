```python
def _binary_local_search(h_bin, deadline, max_iters=100000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0])
    zeros = set(np.where(cur <= 0.5)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find best improving pair-swap
        best_delta = 1e-15
        best_move = None
        # Sample a subset of moves if too many
        ones_list = list(ones)
        zeros_list = list(zeros)
        # Try all moves if feasible, else sample
        if len(ones_list) * len(zeros_list) < 50000:
            candidates = [(oi, oz) for oi in ones_list for oz in zeros_list]
        else:
            rng = np.random.default_rng(it)
            candidates = [(ones_list[rng.integers(len(ones_list))], 
                          zeros_list[rng.integers(len(zeros_list))]) for _ in range(50000)]
        
        for oi, oz in candidates:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            delta = c_try - cur_c5
            if delta < -best_delta:
                best_delta = -delta
                best_move = (oi, oz)
            cur[oi] = 1.0
            cur[oz] = 0.0
        
        if best_move is None:
            break
        oi, oz = best_move
        cur[oi] = 0.0
        cur[oz] = 1.0
        cur_c5 -= best_delta
        ones.remove(oi)
        ones.add(oz)
        zeros.remove(oz)
        zeros.add(oi)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v onto {x: 0≤x≤1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-8
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_optimize(h, deadline, beta=20.0, radius=0.2, stall_limit=100):
    # Simplified gradient-based optimization
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta_cur = beta
    
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta_cur * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        
        # Compute gradient
        g = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += w[r] * (1.0 - h[:m])
                    g[:m] -= w[r] * h[d:]
            else:
                gp = -d
                m = n - gp
                if m > 0:
                    g[:m] += w[r] * (1.0 - h[gp:])
                    g[gp:] -= w[r] * h[:m]
        g -= np.mean(g)
        
        if np.linalg.norm(g) < 1e-10:
            break
        
        # Line search
        d = -g / (np.linalg.norm(g) + 1e-30)
        u = 1.0 - h
        T_arr = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
        Q_arr = np.correlate(d, d, mode='full')
        
        a_hi = radius
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-8
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)

    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _structured_init(n, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        h = rng.uniform(0, 1, n)
    elif mode == 1:
        x = np.arange(n, dtype=float) / n
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        x = np.arange(n, dtype=float) / n
        freqs = rng.uniform(1, 8, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        h = 0.5
        for f, p in zip(freqs, phases):
            h += 0.25 * np.sin(2 * np.pi * f * x + p)
    h = np.clip(h, 0, 1)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Multi-n exploration with binary focus (55% of budget)
    d1 = t0 + budget_s * 0.55
    n_candidates = sorted(set([80, 90, 100, 110, 120, 130, 140, 150, 160, 170, 180, 190, 200, 210]))
    n_candidates = [n for n in n_candidates if 60 <= n <= 240 and n % 2 == 0]
    # Sort: n0 first, then by distance
    n_candidates.sort(key=lambda n: (abs(n - n0), n))

    total_alloc = d1 - t0
    per_n = total_alloc / len(n_candidates)

    for ni, n in enumerate(n_candidates):
        if time.time() > d1 - 10:
            break
        remaining = d1 - time.time()
        alloc = min(per_n * 1.5, remaining * 0.35)
        if alloc < 3:
            break

        # Multiple initial conditions
        inits = []
        if n == n0:
            inits.append(_to_binary(h_init))
        else:
            inits.append(_to_binary(_resample(h_init, n)))
        inits.append(_to_binary(_structured_init(n, rng, mode=int(rng.integers(3)))))
        inits.append(_to_binary(_structured_init(n, rng, mode=int(rng.integers(3)))))

        best_n_c5 = np.inf
        best_n_h = None
        for ii, h_b in enumerate(inits):
            if time.time() > d1 - 5:
                break
            sub_alloc = alloc / len(inits)
            # Binary local search
            h_lb, c5_lb = _binary_local_search(h_b, time.time() + sub_alloc * 0.6,
                                               max_iter=30000, samples=2000, seed=seed + ni * 10 + ii)
            # Full sweep for small n
            if n <= 140 and time.time() < d1 - 5:
                h_lb, c5_lb = _binary_full_sweep(h_lb, min(time.time() + sub_alloc * 0.2, d1 - 5))
            # SA if time allows
            if time.time() < d1 - 5:
                h_lb, c5_lb = _binary_sa(h_lb, min(time.time() + sub_alloc * 0.2, d1 - 5),
                                         max_iter=15000, T_init=0.003, seed=seed + 200 + ni * 10 + ii)
            if c5_lb < best_n_c5:
                best_n_c5 = c5_lb
                best_n_h = h_lb

        if best_n_h is None:
            continue

        # Continuous refinement if promising
        if best_n_c5 < best_c5 + 0.015 and time.time() < d1 - 5:
            h_c = _project(best_n_h.copy(), n / 2.0)
            sub_dl = min(time.time() + alloc * 0.2, d1 - 5)
            h_c, c5_c = _gradient_optimize(h_c, sub_dl, beta=25, radius=0.15, stall_limit=40)
            if c5_c < best_n_c5:
                best_n_c5 = c5_c
                best_n_h = h_c

        if best_n_c5 < best_c5:
            best_c5 = best_n_c5
            best_h = best_n_h.copy()

    # Phase 2: Intensive optimization (30%)
    d2 = t0 + budget_s * 0.85
    n_best = len(best_h)

    # Gradient on continuous
    if time.time() < d2 - 10:
        sub_dl = min(d2 - 10, time.time() + budget_s * 0.10)
        h2 = _project(best_h.copy(), n_best / 2.0)
        h2, c5_2 = _gradient_optimize(h2, sub_dl, beta=30, radius=0.12, stall_limit=150)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Binary intensive
    if time.time() < d2 - 10:
        h_b = _to_binary(best_h)
        sub_dl = min(d2 - 10, time.time() + budget_s * 0.10)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iter=100000, samples=3000, seed=seed + 700)
        if n_best <= 140 and time.time() < d2 - 5:
            h_b, c5_b = _binary_full_sweep(h_b, min(time.time() + budget_s * 0.04, d2 - 5))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # SA from binary
    if time.time() < d2 - 5:
        h_b = _to_binary(best_h)
        sub_dl = min(d2 - 5, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=50000, T_init=0.002, seed=seed + 800)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Pair swap continuous
    if time.time() < d2 - 5:
        h_ps = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d2 - 5, time.time() + budget_s * 0.06)
        h_ps, c5_ps = _pair_swap_inlinear(h_ps, sub_dl, n_pairs=6000, radius=0.2, seed=seed + 900)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Final gradient
    if time.time() < d2 - 3:
        h2 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d2 - 3, time.time() + budget_s * 0.04)
        h2, c5_2 = _gradient_optimize(h2, sub_dl, beta=40, radius=0.08, stall_limit=60)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Basin hopping (12%)
    d3 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d3 - 15 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d3 - 10, time.time() + budget_s * 0.008)
        h_t, c5_t = _gradient_optimize(h_pert, sub_dl, beta=25, radius=0.12, stall_limit=35)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.06)

    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _gradient_optimize(h_f, deadline, beta=50, radius=0.05, stall_limit=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if deadline - time.time() > 4:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _pair_swap_inlinear(h_f, deadline, n_pairs=2000, radius=0.1, seed=seed + 999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
