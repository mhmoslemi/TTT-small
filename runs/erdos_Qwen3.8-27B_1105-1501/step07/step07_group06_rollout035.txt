```python
import numpy as np
import time
from scipy.optimize import minimize

# --- Core evaluation ---
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

# --- Projection ---
def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

# --- Anti-palindromic helpers ---
def _anti_map(x):
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _anti_unmap(h):
    m = len(h) // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])

# --- Smooth-max objective and gradient in anti-palindromic space ---
def _smooth_anti_obj(x, beta):
    h = _anti_map(x)
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    lse = M + np.log(np.sum(np.exp(z))) / beta
    return lse * (2.0 / n)

def _smooth_anti_obj_grad(x, beta):
    h = _anti_map(x)
    n = len(h)
    m = n // 2
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(w.sum())
    w = w / wsum
    S = (M + np.log(wsum) / beta) * (2.0 / n)
    
    G = np.zeros(n)
    for r in range(len(raw)):
        wr = w[r]
        if wr < 1e-10: continue
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            G[tau:] += wr * (1.0 - h[:mm])

        else:
            d = -tau

            G[:mm] += wr * (1.0 - h[d:])
            G[d:] -= wr * h[:mm]
    
    grad_x = (G[:m] - G[m:][::-1]) * (2.0 / n)
    return S, grad_x

# --- L-BFGS-B wrapper with beta annealing ---
def _anti_lbfgs(h_init, deadline, betas=None, maxiter_each=200):
    n = len(h_init)
    m = n // 2
    if m % 2 != 0:  # need even m for binary, but for continuous it's fine
        pass  # continuous works for any m
    x0 = _anti_unmap(h_init)
    x0 = np.clip(x0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * m
    best_x = x0.copy()
    best_c5 = _exact_c5(_anti_map(x0))
    
    if betas is None:
        betas = [5, 10, 20, 40, 80]
    
    for beta in betas:
        if time.time() > deadline - 5: break
        # Use the smooth-max function for optimization
        res = minimize(
            _smooth_anti_obj, x0, args=(beta,),
            jac=lambda x: _smooth_anti_obj_grad(x, beta)[1],
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter_each, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        x0 = res.x
        h_try = _anti_map(x0)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_x = x0.copy()
    
    return _anti_map(best_x), best_c5

# --- Full-dimensional smooth gradient with line search ---
def _smooth_grad_full(h, beta):
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(w.sum())
    w = w / wsum
    
    G = np.zeros(n)
    for r in range(len(raw)):
        wr = w[r]
        if wr < 1e-10: continue
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:] += wr * (1.0 - h[:mm])
                G[:mm] -= wr * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * (1.0 - h[d:])
                G[d:] -= wr * h[:mm]
    G -= np.mean(G)  # ensure sum-preserving
    return G

def _line_search_exact(h, d, radius):
    """Exact line search along direction d (sum(d)=0)."""
    n = len(h)
    dx = 2.0 / n
    # Find feasible interval
    a_lo, a_hi = -radius, radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-12:
        return 0.0, _exact_c5(h)
    
    # Golden section + parabolic
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Coarse grid
    ng = 40
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.array([f(t) for t in gt])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(30):
        if hi - lo < 1e-12: break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    
    # Check quadratic vertex for top shifts
    top = np.argsort(S0)[-5:]
    for r in top:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo < av < a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    
    # Check endpoints and 0
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    
    return opt_t, opt_v

def _pair_sweep(h_bin, deadline):
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _anti_map(x):
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _anti_unmap(h):
    m = len(h) // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])

def _smooth_anti_obj(x, beta):
    h = _anti_map(x)
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    lse = M + np.log(np.sum(np.exp(z))) / beta
    return lse * (2.0 / n)

def _smooth_anti_grad(x, beta):
    h = _anti_map(x)
    n = len(h)
    m = n // 2
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(w.sum())
    w = w / wsum
    G = np.zeros(n)
    for r in range(len(raw)):
        wr = w[r]
        if wr < 1e-10:
            continue
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            G[tau:] += wr * (1.0 - h[:mm])
            G[:mm] -= wr * h[tau:tau + mm]
        else:
            d = -tau
            mm = n - d
            G[:mm] += wr * (1.0 - h[d:])
            G[d:] -= wr * h[:mm]
    grad_x = (G[:m] - G[m:][::-1]) * (2.0 / n)
    return grad_x

def _anti_lbfgs(h_init, deadline, betas=None, maxiter_each=300):
    n = len(h_init)
    m = n // 2
    x0 = _anti_unmap(h_init)
    x0 = np.clip(x0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * m
    best_x = x0.copy()
    best_c5 = _exact_c5(_anti_map(x0))
    if betas is None:

def _smooth_grad_full(h, beta):
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = np.max(raw)
    z = beta * (raw - M)

def _line_search_exact(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    a_lo, a_hi = -radius, radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-12:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    ng = 40
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top = np.argsort(S0)[-6:]
    for r in top:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo < av < a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

        h_pol, c5_pol = _full_smooth_opt(h_pol, sub_dl, beta=60.0, max_iter=150, radius_init=0.06)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Try a few nearby n values for final push
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5 - 8:
            break
        nc = len(best_h) + dn
        if 60 <= nc <= 300 and nc % 2 == 0:
            h_r = _resample(best_h, nc)
            sub_dl = min(d5 - 4, time.time() + budget_s * 0.015)
            h_r, c5_r = _anti_lbfgs(h_r, sub_dl, betas=[30, 80, 150], maxiter_each=120)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Absolute final: exact pair sweep on best
    if deadline - time.time() > 10:
        n_f = len(best_h)
        h_bin = np.zeros(n_f)
        h_bin[np.argsort(best_h)[-n_f // 2:]] = 1.0
        h_bin, c5_bin = _pair_sweep(h_bin, deadline - 3)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
```
