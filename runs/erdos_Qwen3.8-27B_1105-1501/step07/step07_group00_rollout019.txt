def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    scores = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(scores))
    z = (scores - m) / beta
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw < 1e-300:
        return m
    return float(m + beta * np.log(sw))

def _smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    scores = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(scores))
    z = (scores - m) / beta
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
        sw = 1.0
    w /= sw
    grad = np.zeros(n)
    n_scores = len(scores)
    for ell in range(n_scores):
        wi = w[ell]
        if wi * dx < 1e-16:
            continue
        d = ell - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                wd = wi * dx
                grad[:L] -= wd * x[d:d + L]
                grad[d:d + L] += wd * (1.0 - x[:L])
        else:
            g = -d
            L = n - g
            if L > 0:
                wd = wi * dx
                grad[:L] += wd * (1.0 - x[g:g + L])
                grad[g:g + L] -= wd * x[:L]
    return grad

def _grad_rows(h, lags):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        d = int(d)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    G *= dx
    return G

def _line_search(h, d, dx, t_hi):

def _active_lp(h, scores, radius, n_active, gap):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = int(rng.integers(3, 7))
    freqs = rng.uniform(0.5, 10.0, size=n_freqs)
    phases = rng.uniform(0, 2.0 * np.pi, size=n_freqs)
    amps = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude / max(1.0, float(np.max(np.abs(perturb))))
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _rotate_interpolate(h, frac, n_new=None):
    n = len(h)
    if n_new is None:
        n_new = n
    shift = frac * n
    n_shift = int(round(shift))
    frac_shift = shift - n_shift
    h_rot = np.zeros(n)
    for i in range(n):

    if n_new != n:

    return _project(h_rot, n_new / 2.0)

                if c5_sm2 < c5_sm:
                    c5_sm = c5_sm2
                    h_sm = h_sm.copy()
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 6: Final polish (10% remaining)
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _slsqp_run(h_f, [0.015, 0.006, 0.002, 0.0008], maxiter_each=40, deadline=deadline)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
    h_f2, c5_f2 = _lp_polish(h_f2, deadline, radius_init=0.04, max_iters=1000)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
