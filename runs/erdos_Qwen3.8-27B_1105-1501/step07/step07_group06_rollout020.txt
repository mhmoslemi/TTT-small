def _swap_delta(h, a, b, corr, n):
    """Compute new max corr after swapping positions a(1->0) and b(0->1)."""
    h[a] = 0.0
    h[b] = 1.0
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    h[a] = 1.0
    h[b] = 0.0
    return float(np.max(new_corr))

def _local_search_bin(h, deadline, max_iters=500, rng=None):
    n = len(h)
    cur = h.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode='full')))
    best_h = cur.copy()
    best_max = cur_max
    if rng is None:
        rng = np.random.default_rng(0)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        cur_max = float(np.max(corr))
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        beta = 12.0
        grad = _compute_grad(cur, corr, beta)
        best_swap_val = cur_max
        best_a, best_b = -1, -1
        n_try = min(60, len(ones) * len(zeros))
        for _t in range(n_try):
            a = int(ones[rng.integers(0, len(ones))])
            b = int(zeros[rng.integers(0, len(zeros))])
            if a == b:
                continue
            new_max = _swap_delta(cur, a, b, corr, n)
            if new_max < best_swap_val - 1e-12:
                best_swap_val = new_max
                best_a, best_b = a, b
        if best_a >= 0:
            cur[best_a] = 0.0
            cur[best_b] = 1.0
            if best_swap_val < best_max - 1e-14:
                best_max = best_swap_val
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                a = int(ones[rng.integers(0, len(ones))])
                b = int(zeros[rng.integers(0, len(zeros))])
                cur[a], cur[b] = cur[b], cur[a]
    return best_h, best_max * (2.0 / n)

def _cont_local(h, deadline, max_iters=200, beta_start=8.0, beta_end=50.0, radius=0.15):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        frac = float(it) / max(max_iters, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        grad = _compute_grad(cur, corr, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        d = -grad / gn
        # Line search
        best_a = 0.0
        best_v = cur_c5
        for a in np.linspace(-radius, radius, 30):
            hc = cur + a * d
            hc = _project(hc, target)
            vc = _exact_c5(hc)
            if vc < best_v:
                best_v = vc
                best_a = a
        if best_a != 0.0:
            new_h = _project(cur + best_a * d, target)
            new_c5 = _exact_c5(new_h)
            if new_c5 < cur_c5 - 1e-13:
                cur = new_h
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5
