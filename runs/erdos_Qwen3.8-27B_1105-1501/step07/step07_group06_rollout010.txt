import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v


def _lp_direction(h, raw, top_m, max_time=2.0):
    n = len(h)
    top_indices = np.argsort(raw)[-top_m:]
    A_rows = []
    raw_vals = []
    for r in top_indices:
        tau = r - (n - 1)
        g = np.zeros(n)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[:m] -= h[tau:tau + m]
                g[tau:tau + m] += (1.0 - h[:m])
        else:
            dd = -tau
            m = n - dd
            if m > 0:
                g[:m] += (1.0 - h[dd:dd + m])
                g[dd:dd + m] -= h[:m]
        A_rows.append(g)
        raw_vals.append(raw[r])
    A = np.array(A_rows)
    rv = np.array(raw_vals)
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    parts_a = []
    parts_b = []
    parts_a.append(np.hstack([A, -np.ones((len(A), 1))]))
    parts_b.append(-rv)
    parts_a.append(np.hstack([np.eye(n), np.zeros((n, 1))]))
    parts_b.append(1.0 - h)
    parts_a.append(np.hstack([-np.eye(n), np.zeros((n, 1))]))
    parts_b.append(h)
    A_ub = np.vstack(parts_a)
    b_ub = np.concatenate(parts_b)
    A_eq = np.hstack([np.ones(n), np.array([0.0])])
    b_eq = np.array([0.0])
    t_start = time.time()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if time.time() - t_start > max_time:
        return None
    if res.success and res.status == 0:
        d = res.x[:n].copy()
        d -= np.mean(d)
        dn = np.max(np.abs(d))
        if dn < 1e-12:
            return None
        return d
    return None


def _gradient_direction(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        w = np.zeros(len(raw))
        w[idx] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for r in range(len(raw)):
        if w[r] < 1e-10:
            continue
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[:m] -= wr * h[tau:tau + m]
                g[tau:tau + m] += wr * (1.0 - h[:m])
        else:
            dd = -tau
            m = n - dd
            if m > 0:
                g[:m] += wr * (1.0 - h[dd:dd + m])
                g[dd:dd + m] -= wr * h[:m]
    g -= float(np.mean(g))
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        g /= gn
    return g

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_continuous(h4, sub_dl, n_iter=4000, radius=0.12, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    scale = 0.03
    n_restarts = 0
    while time.time() < deadline - 15 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.007)
        h_t, c5_t = _refine_lp(h_p, sub_dl, top_m_init=10, top_m_max=30,
                               radius_init=0.08, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.10)

    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _refine_lp(h_f, deadline, top_m_init=15, top_m_max=40,
                               radius_init=0.05, max_stall=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final binary check
    if deadline - time.time() > 5:
        h_bf = _to_binary(best_h)
        sub_dl = min(deadline, time.time() + budget_s * 0.01)
        h_bf, c5_bf = _binary_swap_opt(h_bf, sub_dl, max_passes=50)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
