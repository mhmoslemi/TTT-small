import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[r, :m] += (1.0 - h[lag:lag + m])
                G[r, lag:] += -h[:m]
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] -= h[g:]
    return G

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _full_lp(h, G, radius):
    n = len(h)
    m = 2 * n - 1
    scores = _raw_scores(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_full(h, p, G, radius):
    """Exact quadratic line search along direction p."""
    n = len(h)
    S0 = _raw_scores(h)
    T = G @ p
    P = np.correlate(p, p, mode='full')
    
    # Compute safe t_hi
    t_hi = radius
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0))
    
    # Grid search
    n_grid = 50
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * P))
    best_idx = int(np.argmin(grid_f))
    
    # Golden section
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-12:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * P))
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * P))
        for _ in range(60):
            if hi_b - lo_b < 1e-14:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * P))
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * P))
        opt_t = (lo_b + hi_b) / 2.0
    else:
        opt_t = grid_t[best_idx]
    
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * P))
    # Also check endpoints
    val_0 = float(np.max(S0))
    val_end = float(np.max(S0 + t_hi * T - t_hi * t_hi * P))
    
    if val_0 < opt_val:
        return 0.0, val_0
    if val_end < opt_val:
        return t_hi, val_end
    return opt_t, opt_val

def _lp_optimize(h, deadline, radius_init=0.10, max_iters=None):
    """Main LP optimization loop with full epigraph."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    max_stall = 200
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        G = _grad_matrix(h)
        p = _full_lp(h, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 5
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        alpha, pred_max = _line_search_full(h, p, G, radius)
        if alpha < 1e-14:
            stall += 1
            radius *= 0.85
            continue
        
        # Apply step and check
        h_new = h + alpha * p
        h_new = _project_sum(h_new, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_iters=200000):
    """Local search for binary solutions using pair swaps."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    stall = 0
    max_stall = 2000
    ones = list(np.where(cur == 1)[0])
    zeros = list(np.where(cur == 0)[0])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        # Random pair swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
    
    return best_h, best_c5

def _systematic_pair_swap(h_bin, deadline, max_rounds=50):
    """Systematic pair swap: try all (one, zero) pairs."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur == 1)[0])
        zeros = list(np.where(cur == 0)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c = _exact_c5(cur)
                if c < cur_c5 - 1e-15:
                    cur_c5 = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    
    return best_h, best_c5
