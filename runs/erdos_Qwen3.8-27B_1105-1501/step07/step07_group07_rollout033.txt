    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj(x):
    """LSE smooth approximation of max score."""
    x = np.clip(x, 0.0, 1.0)
    raw = np.correlate(x, 1.0-x, mode='full')
    s = raw * _g_dx
    mval = np.max(s)
    u = (s - mval) / _g_tau
    u = np.minimum(u, 0.0)
    u = np.maximum(u, -50.0)
    z = np.exp(u)
    zsum = np.sum(z)
    return float(mval + _g_tau * np.log(zsum))

def _lse_grad(x):
    """Gradient of LSE objective."""
    x = np.clip(x, 0.0, 1.0)
    raw = np.correlate(x, 1.0-x, mode='full')
    s = raw * _g_dx
    mval = np.max(s)
    u = (s - mval) / _g_tau
    u = np.minimum(u, 0.0)
    u = np.maximum(u, -50.0)
    z = np.exp(u)
    p = z / np.sum(z)
    G = _grad_matrix(x)
    return _g_dx * (G.T @ p)

    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones(len(x))

def _lse_anneal(h, deadline, tau_start=0.08, tau_end=0.005, n_steps=5):
    """Run LSE temperature annealing with SLSQP."""
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    h = _project_sum(h.copy(), n/2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    taus = np.geomspace(tau_start, tau_end, n_steps)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac}
    
    for tau in taus:
        if time.time() > deadline:
            break
        _g_tau = tau
        try:
            res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            if res.x is not None:
                h_new = _project_sum(np.clip(res.x, 0.0, 1.0), n/2.0)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                # Always update h for next iteration (even if not best)
                if c5_new < _exact_c5(h) + 0.01:  # accept if not much worse
                    h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
