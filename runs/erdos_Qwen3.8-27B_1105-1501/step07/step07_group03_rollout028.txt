    n = len(h)
    # Variables: [delta_0, ..., delta_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: G @ delta - t <= -raw
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -raw.copy()
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = list(zip(-h, 1 - h)) + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        t_val = res.x[n]
        return delta, t_val
    return None, 0.0

        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad(h):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += u[d:]
                G[r, d:] -= h[:m]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                G[r, gp:] += u[:m]
                G[r, :m] -= h[gp:]
    return G

def _solve_lp(h, raw):
    n = len(h)
    G = _build_grad(h)
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0
    A_ub = np.empty((2 * n - 1, nv))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -raw.copy()
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lb = -h
    ub = 1.0 - h
    bounds = list(zip(lb, ub)) + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, 0.0

def _lp_optimize(h, deadline, max_stall=40, seed=42):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline - 1:
        if stall > max_stall:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(raw))
        delta, t_val = _solve_lp(h, raw)
        if delta is None:
            stall += 1
            continue
        
        M - t_val  # predicted improvement
        if M - t_val < 1e-12:
            pert = rng.normal(0, 0.08, n)
            h = _project(h + pert, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall += 1
            continue
        
        # Line search
        best_a = 0.0
        best_v = cur_c5
        for a in (1.0, 0.6, 0.4, 0.25):
            ht = h + a * delta
            vt = _exact_c5(ht)
            if vt < best_v - 1e-14:
                best_v = vt
                best_a = a
        
        if best_a < 0.05:
            stall += 1
            if stall % 8 == 0:
                pert = rng.normal(0, 0.03, n)
                h = _project(h + pert, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            continue
        
        h = h + best_a * delta
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        stall = 0
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_sa(hb, deadline, max_iter=50000, T_init=0.004):
    n = len(hb)
    cur = hb.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9992
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(hb, deadline):
    n = len(hb)
    cur = hb.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    improved = True
    while improved and time.time() < deadline - 1:
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline - 1:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)
