    n = len(h)
    u = 1.0 - h
    # T[r] = derivative of raw[r] along d at h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    # Q[r] = second derivative / 2
    Q = np.correlate(d, d, mode='full')
    # raw[r](a) = raw[r] + a*T[r] - a^2*Q[r]
    return T, Q

    if hi - lo > 1e-11:

        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

def _grad_lag(h, d_lag):
    """Gradient of raw(d_lag) w.r.t. h at current h."""
    n = len(h)
    coeff = np.zeros(n)
    if d_lag >= 0:
        m = n - d_lag
        coeff[:m] += (1.0 - h[d_lag:d_lag + m])
        coeff[d_lag:] -= h[:m]
    else:
        gd = -d_lag
        m = n - gd
        coeff[gd:] += (1.0 - h[:m])
        coeff[:m] -= h[gd:gd + m]
    return coeff

def _lp_solve(h0, n_select=10, delta=0.12, solver=None):
    """Solve LP to find a point that reduces the max of top lags."""
    n = len(h0)
    target = n / 2.0
    raw0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Select top lags
    top_idx = np.argsort(-raw0)[:n_select]
    
    # Build constraint matrix
    # For each selected lag, the linearized value is:
    # raw0[r] + grad_r @ (x - h0) <= t
    # => grad_r @ x <= t - raw0[r] + grad_r @ h0
    grads = np.zeros((n_select, n))
    rhs = np.zeros(n_select)
    for k, r in enumerate(top_idx):
        d_lag = int(r) - (n - 1)
        g = _grad_lag(h0, d_lag)
        grads[k] = g
        rhs[k] = raw0[r] - g @ h0  # so constraint is grads[k] @ x + rhs[k] <= t
    
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.multiply(grads, x) + rhs <= t,  # shape (n_select,) <= scalar
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h0 - delta,
        x <= h0 + delta
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if solver is None:
        solver = cp.CLARABEL
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iter=20000, eps=1e-9)
        except:
            try:
                prob.solve(solver=cp.ECOS)
            except:
                return None
    if x.value is None:
        return None
    return np.clip(x.value, 0.0, 1.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_sweep(h, deadline, max_rounds=5):
    """Greedy pair swap."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _main_loop(h, deadline, rng, lp_interval=3, n_select=10, delta_init=0.15):
    """Main optimization loop combining LP directions and gradient directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    delta = delta_init
    beta = 15.0
    step_count = 0
    rng2 = np.random.default_rng(12345)
    
    while time.time() < deadline:
        step_count += 1
        raw = np.correlate(h, 1.0 - h, mode='full')
        
        # Build direction set
        directions = []
        
        # 1. Gradient direction (softmax weighted)
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -70, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        # Gradient of max approximated by weighted gradient
        g = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-12:
                continue
            d_lag = r - (n - 1)
            g += w[r] * _grad_lag(h, d_lag)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            directions.append(-g / gn)
        
        # 2. LP direction (every lp_interval steps)
        if step_count % lp_interval == 1:
            x_lp = _lp_solve(h, n_select=n_select, delta=delta)
            if x_lp is not None:
                d_lp = x_lp - h
                d_lp -= np.mean(d_lp)
                dln = np.linalg.norm(d_lp)
                if dln > 1e-10:
                    directions.append(d_lp / dln)
        
        # 3. Random smooth perturbation direction (every 5 steps)
        if step_count % 5 == 3:
            d_rand = _smooth_perturb(h, 0.05, rng2) - h
            d_rand -= np.mean(d_rand)
            drn = np.linalg.norm(d_rand)
            if drn > 1e-10:
                directions.append(d_rand / drn)
        
        # Try each direction
        accepted = False
        for d in directions:
            alpha, quad_c5 = _line_search_quad(h, d, raw, dx, a_hi=0.3, n_top=14)
            if alpha < 1e-10:
                continue
            h_new = _project(h + alpha * d, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                accepted = True
                break
        
        if not accepted:
            stall += 1
            if stall % 5 == 0:
                beta *= 0.9
                delta = max(0.03, delta * 0.92)
            if stall > 80:
                # Reset with perturbation
                h = _smooth_perturb(h, 0.08, rng2)
                cur_c5 = _exact_c5(h)
                stall = 0
                delta = 0.10
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(9999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

- For shift d ≥ 0: ∂raw/∂h[p] = (1-h[p+d]) if p ≤ n-1-d, plus -h[p-d] if d ≤ p ≤ n-1

     + Σ_{d=1}^{n-1-p} w[n-1-d] * (-h[p+d])  [for negative shifts, p is first factor]
     + Σ_{d=1}^{n-1-p} ...

Actually, let me use a smarter approach. I can compute the gradient using the same correlation machinery:

For the full correlation raw = np.correlate(h, 1-h, mode='full'):
raw[r] = Σ_{valid i} h[i] * (1-h[i + r - n + 1])

The gradient w.r.t. h[p]:
∂raw/∂h[p] = Σ_{valid i} [δ(i,p)*(1-h[i+d]) + h[i]*(-δ(i+d,p))]
where d = r - (n-1)

= (1-h[p+d]) * [p is valid first index for this r] + (-h[p-d]) * [p-d is valid first index for this r]

For d ≥ 0 (r ≥ n-1): first index range is [0, n-1-d]
  - p is valid first index: 0 ≤ p ≤ n-1-d → contributes (1-h[p+d])
  - p-d is valid first index: 0 ≤ p-d ≤ n-1-d → d ≤ p ≤ n-1 → contributes -h[p-d]

For d < 0 (r < n-1): let gd = -d, valid first index range is [0, n-1-gd], and the formula is raw = Σ h[i+gd]*(1-h[i])
  - p is the "first" factor (h[i+gd]): p = i+gd, i = p-gd, valid: 0 ≤ p-gd ≤ n-1-gd → gd ≤ p ≤ n-1 → contributes (1-h[p-gd])
  - p is the "second" factor (h[i]): i = p, valid: 0 ≤ p ≤ n-1-gd → contributes -h[p+gd]

So:
For r (which corresponds to shift d = r - n + 1):
If d ≥ 0:
  ∂raw[r]/∂h[p] = (1-h[p+d]) * [0 ≤ p ≤ n-1-d] + (-h[p-d]) * [d ≤ p ≤ n-1]
If d < 0 (gd = -d > 0):

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, d_lag):
    n = len(h)
    coeff = np.zeros(n)
    if d_lag >= 0:
        m = n - d_lag
        coeff[:m] += (1.0 - h[d_lag:d_lag + m])
        coeff[d_lag:] -= h[:m]
    else:
        gd = -d_lag
        m = n - gd
        coeff[gd:] += (1.0 - h[:m])
        coeff[:m] -= h[gd:gd + m]
    return coeff

def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-13:
            continue
        d_lag = r - (n - 1)
        g += w[r] * _grad_lag(h, d_lag)
    return g

def _line_search_quad(h, d, raw, dx, a_hi=0.3, n_top=14):

    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, min(len(top_idx), 8)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A2 = Q[ri] - Q[rj]
            B2 = T[ri] - T[rj]
            C2 = raw[ri] - raw[rj]
            if abs(A2) > 1e-14:
                disc = B2 * B2 + 4.0 * A2 * C2
                if disc >= 0:
                    sd = np.sqrt(disc)
                    for aa in [(B2 + sd) / (2.0 * A2), (B2 - sd) / (2.0 * A2)]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)

                aa = -C2 / B2
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    grid = np.linspace(0.0, a_hi, 28)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    step = a_hi / 28
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx

        if val_opt < best_val:

def _lp_direction(h0, n_select=10, delta=0.12):
    n = len(h0)
    target = n / 2.0
    raw0 = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(-raw0)[:n_select]
    grads = np.zeros((n_select, n))
    rhs = np.zeros(n_select)
    for k, r in enumerate(top_idx):
        d_lag = int(r) - (n - 1)
        g = _grad_lag(h0, d_lag)
        grads[k] = g
        rhs[k] = raw0[r] - float(g @ h0)
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.matmul(grads, x) + rhs <= t,
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h0 - delta,
        x <= h0 + delta
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except:
            continue
    if not solved or x.value is None:
        return None
    return np.clip(x.value, 0.0, 1.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)

        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
