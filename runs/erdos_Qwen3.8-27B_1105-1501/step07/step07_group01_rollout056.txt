        k_hi = min(2*n-2, 2*n-2-i)
        for k in range(k_lo, k_hi+1):
            idx = i + k - n + 1
            if 0 <= idx < n:
                T[k] -= (1.0 - h[idx])
        
        # Compute C: at most 3 non-zero entries
        C = np.zeros(2*n-1)
        pos_jj = n - 1  # corr(e_j, e_j)
        C[pos_jj] -= 1.0
        pos_ii = n - 1  # corr(e_i, e_i)
        C[pos_ii] -= 1.0
        # corr(e_j, e_i) at position i-j+n-1
        pos_ji = i - j + n - 1
        if 0 <= pos_ji < 2*n-1:
            C[pos_ji] += 1.0
        # corr(e_i, e_j) at position j-i+n-1
        pos_ij = j - i + n - 1
        if 0 <= pos_ij < 2*n-1:
            C[pos_ij] += 1.0
        
        # Note: C = -corr(d,d) where d = e_j - e_i
        # corr(d,d) = corr(e_j,e_j) - corr(e_j,e_i) - corr(e_i,e_j) + corr(e_i,e_i)
        # So -corr(d,d) = -corr(e_j,e_j) + corr(e_j,e_i) + corr(e_i,e_j) - corr(e_i,e_i)
        # corr(e_j,e_j) = 1 at n-1
        # corr(e_i,e_i) = 1 at n-1
        # corr(e_j,e_i) = 1 at i-j+n-1 (if valid)
        # corr(e_i,e_j) = 1 at j-i+n-1 (if valid)
        # So C = -1 at n-1 - 1 at n-1 + 1 at (i-j+n-1) + 1 at (j-i+n-1)
        # = -2 at n-1 + 1 at (i-j+n-1) + 1 at (j-i+n-1)
        # (if i≠j, the off-diagonal terms are at different positions from n-1)
        
        # Wait, if i-j+n-1 = n-1, that means i=j, which we excluded.
        # So the -2 at n-1 and the +1s at other positions are distinct.
        
        # Find t_hi: the maximum t such that 0 ≤ h + t*(e_j - e_i) ≤ 1
        # h_j + t ≤ 1 → t ≤ 1 - h_j
        # h_i - t ≥ 0 → t ≤ h_i
        t_hi = min(1.0 - h[j], h[i])
        if t_hi < 1e-12:
            stall += 1
            continue
        
        # Golden section search for t in [0, t_hi]
        # Minimize max(corr + t*T + t^2*C)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t = 0.0
        hi_t = t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(corr + c1 * T + c1*c1 * C))
        f2 = float(np.max(corr + c2 * T + c2*c2 * C))
        for _ in range(40):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(corr + c1 * T + c1*c1 * C))
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(corr + c2 * T + c2*c2 * C))
        
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(corr + opt_t * T + opt_t*opt_t * C))
        f0 = float(np.max(corr))
        # Also check t_hi
        f_end = float(np.max(corr + t_hi * T + t_hi*t_hi * C))
        
        # Pick best t
        best_t = 0.0
        best_f = f0
        if opt_val < best_f:
            best_f = opt_val
            best_t = opt_t
        if f_end < best_f:
            best_f = f_end
            best_t = t_hi
        
        if best_f < f0 - 1e-12:
            # Apply the move
            h[j] += best_t
            h[i] -= best_t
            corr = corr + best_t * T + best_t*best_t * C
            cur_c5 = float(np.max(corr)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(corr - m)
    total = float(np.sum(exps))
    if total < 1e-300:
        return np.zeros(n)
    w = exps / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = n - s
            if ms > 0:
                grad[:ms] += wk * (1.0 - h[s:s + ms])
                grad[s:] -= wk * h[:ms]
        else:
            p = -s
            mp = n - p
            if mp > 0:
                grad[p:] += wk * (1.0 - h[:mp])
                grad[:mp] -= wk * h[p:]
    return grad * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_modes = 4
    perturb = np.zeros(n)
    for _ in range(n_modes):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2.0 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Smooth-max fine tune
    sub_dl = min(deadline, time.time() + remaining * 0.2)
    h6, c5_6 = _smooth_anneal(best_h, sub_dl, beta_start=200.0, beta_end=20000.0, seed=9500)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Final pair moves
    sub_dl = min(deadline, time.time() + remaining * 0.3)
    h6, c5_6 = _pair_move_opt(best_h, sub_dl, n_iters=500000, seed=9900, strategy="structured")
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
