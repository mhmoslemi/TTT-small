def _adam_optimize(h_init, deadline, beta_start=5.0, beta_end=50.0, lr=0.01, n_epochs=100):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    while time.time() < deadline:
        t += 1
        # Anneal beta
        progress = min(1.0, t / n_epochs)
        beta = beta_start + (beta_end - beta_start) * progress
        
        obj, grad = _lse_grad(h, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_try = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_try = _project(h_try, target)
        
        # Backtracking line search
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        h = h_try
    
    return best_h, best_c5

def _adam_optimize(h0, deadline, beta_start=10.0, beta_end=100.0, lr=0.005, decay=0.999):
    n = len(h0)
    target = n / 2.0
    h = _project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    cur_lr = lr
    current_beta = beta_start
    stall = 0
    
    while time.time() < deadline:
        t += 1
        # Slowly increase beta
        if t % 50 == 0:
            current_beta = min(current_beta * 1.1, beta_end)
            cur_lr *= decay
        
        obj, grad = _lse_obj_grad(h, current_beta)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        step = -cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        h_try = _project(h + step, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        h = h_try
    
    return best_h, best_c5

def _slsqp_solve(h0, deadline, max_iters=200):
    """Use SLSQP to minimize max_k corr[k]*dx via log-sum-exp."""
    n = len(h0)
    target = n / 2.0
    h0 = _project(h0.copy(), target)
    dx = 2.0 / n
    L = 2 * n - 1
    
    def objective(x):
        h = x[:n]
        beta = 50.0
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr = corr_raw * dx
        m = float(np.max(corr))
        shifted = np.clip(50.0 * (corr - m), -700, 700)
        exp_s = np.exp(shifted)
        s = float(np.sum(exp_s))
        if s < 1e-300:
            return float(np.max(corr))
        return m + float(np.log(s)) / 50.0
    
    def grad(x):
        h = x[:n]
        beta = 50.0
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr = corr_raw * dx
        m = float(np.max(corr))
        shifted = np.clip(beta * (corr - m), -700, 700)
        exp_s = np.exp(shifted)
        s = float(np.sum(exp_s))
        w = exp_s / s
        g = np.zeros(n)
        for idx in range(L):
            wi = w[idx]
            if wi < 1e-14:
                continue
            ell = idx - (n - 1)
            if ell >= 0:
                m2 = n - ell
                g[:m2] += wi * (1.0 - h[ell:ell + m2])
                g[ell:] -= wi * h[:m2]
            else:
                p = -ell
                m2 = n - p
                g[p:] += wi * (1.0 - h[:m2])
                g[:m2] -= wi * h[p:p + m2]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
             'jac': lambda x: np.concatenate([np.ones(n), np.zeros(1)])}]
    bounds = [(0, 1)] * n
    
    x0 = h0.copy()
    result = minimize(objective, x0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iters, 'ftol': 1e-8})
    
    if result.success or result.fun < _exact_c5(h0):
        h_sol = _project(result.x[:n].copy(), target)
        c5_sol = _exact_c5(h_sol)
        return h_sol, c5_sol
    return h0, _exact_c5(h0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h0, deadline, T_init=0.004, seed=42):
    n = len(h0)
    cur = h0.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    while time.time() < deadline:
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):
    """Pair-based local optimization (from previous approach, kept as backup)."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    one_mh = 1.0 - h
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        # Compute gradient of smoothed max
        corr = S0 * (2.0 / n)
        m = float(np.max(corr))
        shifted = np.clip(beta * (corr - m), -500, 500)
        exp_s = np.exp(shifted)
        s = float(np.sum(exp_s))
        w = exp_s / s
        
        # Compute coordinate gradient
        coord_grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            ell = idx - (n - 1)
            if ell >= 0:
                m2 = n - ell
                coord_grad[:m2] += wi * (1.0 - h[ell:ell + m2])
                coord_grad[ell:] -= wi * h[:m2]
            else:
                p = -ell
                m2 = n - p
                coord_grad[p:] += wi * (1.0 - h[:m2])
                coord_grad[:m2] -= wi * h[p:p + m2]
        
        # Select pairs: source (high grad) -> sink (low grad)
        sort_idx = np.argsort(coord_grad)
        n_sel = min(n_pairs, n // 4)
        sinks = sort_idx[:n_sel]
        sources = sort_idx[-n_sel:]
        
        best_step = None
        for ii in range(len(sinks)):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            
            # Exact line search: C5(t) = max_k (S0[k] + t*T[k] - t^2*C[k]) * dx
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            dx_val = 2.0 / n
            
            # Minimize max_k S0[k]*dx + t*T[k]*dx - t^2*C[k]*dx
            # = dx * max_k (S0[k] + t*T[k] - t^2*C[k])
            # Since dx > 0, minimize max_k (S0[k] + t*T[k] - t^2*C[k])
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = t_lo, t_hi
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
            for _ in range(40):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
            
            t_opt = (lo + hi) / 2.0
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_step = h_try
        
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if radius < 0.03:
                radius = 0.10
            if stall > stall_limit:
                break
    
    return best_h, best_c5

def _dense_line_search(h, deadline, n_dirs=50, max_step=0.3):
    """Dense direction line search: move along -grad of smoothed objective."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for iteration in range(2000):
        if time.time() > deadline:
            break
        beta = 30.0 + 20.0 * (iteration % 10)
        obj, grad = _lse_obj_grad(h, beta)
        
        # Generate several candidate directions
        for d_idx in range(min(n_dirs, 10)):
            if time.time() > deadline:
                break
            if d_idx == 0:
                # Pure negative gradient
                direction = -grad
            elif d_idx == 1:
                # Negative gradient with random perturbation
                direction = -grad + 0.1 * rng.standard_normal(n)
            elif d_idx < 5:
                # Sparse direction: only top gradient coordinates
                top_k = max(5, n // 10)
                idx = np.argsort(np.abs(grad))[-top_k:]
                direction = np.zeros(n)
                direction[idx] = -grad[idx]
            else:
                # Random feasible direction
                direction = rng.standard_normal(n)
                direction -= np.mean(direction)
            
            # Normalize and clip step
            direction = np.clip(direction, -max_step, max_step)
            direction -= np.mean(direction)  # keep zero-sum approximately
            
            # Project onto feasible set
            h_try = _project(h + direction, target)
            
            # Try multiple step sizes
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                h_try = _project(h + alpha * direction, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    break
                elif c5_try < _exact_c5(h):
                    # Accept if better than current
                    h = h_try
                    break
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_s = np.exp(shifted)
    s = float(np.sum(exp_s))
    if s < 1e-300:
        top_k = np.argsort(corr)[-3:]
        w = np.zeros(len(corr))
        w[top_k] = 1.0 / 3.0
        return m, None, w
    w = exp_s / s
    obj = m + float(np.log(s)) / beta
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        ell = idx - (n - 1)
        if ell >= 0:
            m2 = n - ell
            grad[:m2] += wi * (1.0 - h[ell:ell + m2])
            grad[ell:] -= wi * h[:m2]
        else:
            p = -ell
            m2 = n - p
            grad[p:] += wi * (1.0 - h[:m2])
            grad[:m2] -= wi * h[p:p + m2]
    return obj, grad, w

def _adam_run(h0, deadline, beta_start=15.0, beta_end=300.0, lr=0.012, decay=0.9997, seed=0):
    n = len(h0)
    target = n / 2.0
    h = _project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    cur_lr = lr
    cur_beta = beta_start
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        t += 1
        if t % 30 == 0:
            cur_beta = min(cur_beta * 1.08, beta_end)
            cur_lr *= decay
        obj, grad, _ = _lse_grad(h, cur_beta)
        if grad is None:
            break
        m1 = b1 * m1 + (1.0 - b1) * grad
        m2 = b2 * m2 + (1.0 - b2) * (grad * grad)
        m1_hat = m1 / (1.0 - b1 ** t)
        m2_hat = m2 / (1.0 - b2 ** t)
        step = -cur_lr * m1_hat / (np.sqrt(m2_hat) + eps)
        h_try = _project(h + step, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        h = h_try
    return best_h, best_c5

def _pair_optimize(h0, deadline, radius=0.15, n_pairs=50, stall_limit=80, seed=42):

        corr = S0 * dx_val
        beta_ls = 40.0

        exp_s = np.exp(shifted)
        s_tot = float(np.sum(exp_s))
        if s_tot < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0

        coord_grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            ell = idx - (n - 1)
            if ell >= 0:
                m2 = n - ell
                coord_grad[:m2] += wi * (1.0 - h[ell:ell + m2])
                coord_grad[ell:] -= wi * h[:m2]
            else:
                p = -ell
                m2 = n - p
                coord_grad[p:] += wi * (1.0 - h[:m2])
                coord_grad[:m2] -= wi * h[p:p + m2]
        sort_idx = np.argsort(coord_grad)
        n_sel = min(n_pairs, n // 4)
        sinks = sort_idx[:n_sel]
        sources = sort_idx[-n_sel:]
        best_step_h = None
        best_step_c5 = best_c5
        for ii in range(len(sinks)):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -cur_radius)
            t_hi = min(1.0 - h[sink], h[source], cur_radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_gs, hi_gs = t_lo, t_hi
            c1 = hi_gs - gr * (hi_gs - lo_gs)
            c2 = lo_gs + gr * (hi_gs - lo_gs)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
            for _ in range(35):

                    break
                if f1 < f2:
                    hi_gs, c2, f2 = c2, c1, f1
                    c1 = hi_gs - gr * (hi_gs - lo_gs)
                    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                else:
                    lo_gs, c1, f1 = c1, c2, f2
                    c2 = lo_gs + gr * (hi_gs - lo_gs)
                    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
            t_opt = (lo_gs + hi_gs) / 2.0
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_step_c5 - 1e-14:
                best_step_c5 = c5_try
                best_step_h = h_try
        if best_step_h is not None:
            h = _project(best_step_h, target)
            cur_c5 = _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
