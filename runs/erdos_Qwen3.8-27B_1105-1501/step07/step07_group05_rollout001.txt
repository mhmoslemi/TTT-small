    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Bisection projection onto [0,1]^n with sum = target
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

def _line_search_max(h, d, dx, radius):

    # = max_k [S0_k + a*T_k - a^2*Q_k] * dx
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')

    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    a_hi = radius
    pos = d > 1e-14

    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)

    # - endpoints

    # - intersection of two quadratics
    cands = [a_lo, a_hi, 0.0]

def _grad_active(h, K=5, beta=30.0):
    # Compute gradient of the soft-max of the overlap terms
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    
    # Soft weights
    v = np.clip(beta * (sc - mx), -50.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    
    # Also add uniform weight on top-K
    top_k_idx = np.argsort(-sc)[:K]
    w_top = np.zeros_like(w)
    w_top[top_k_idx] = 1.0 / K
    w = 0.5 * w + 0.5 * w_top
    
    # Compute gradient
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _block_grad(h, K, beta, bs):
    # Block-averaged gradient for smoother directions
    g = _grad_active(h, K=K, beta=beta)
    n = len(g)
    nb = n // bs
    if nb < 1:
        return g
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

def _grad_active(h, K, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -50.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    top_k_idx = np.argsort(-sc)[:K]
    w_top = np.zeros_like(w)
    w_top[top_k_idx] = 1.0 / K
    w = 0.5 * w + 0.5 * w_top
    
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return g.copy()
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

def _local_opt(h, deadline, r0=0.15, stall_lim=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    k_cycle = [1, 3, 5, 8, 12]
    k_idx = 0
    beta_cycle = [20.0, 40.0, 80.0, 15.0]
    beta_idx = 0
    bs_cycle = [6, 12, 18, 24]
    bs_idx = 0
    
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        
        K = k_cycle[k_idx % len(k_cycle)]
        beta = beta_cycle[beta_idx % len(beta_cycle)]
        
        # Generate directions
        dirs = []
        g1 = _grad_active(h, K, beta)
        nrm = np.linalg.norm(g1)
        if nrm > 1e-10:
            dirs.append(g1 / nrm)
        
        bs = bs_cycle[bs_idx % len(bs_cycle)]
        g2 = _block_dir(g1, bs)
        nrm2 = np.linalg.norm(g2)
        if nrm2 > 1e-10:
            dirs.append(g2 / nrm2)
        
        # Add a higher-level block direction
        g3 = _block_dir(g1, bs * 2)
        nrm3 = np.linalg.norm(g3)
        if nrm3 > 1e-10:
            dirs.append(g3 / nrm3)
        
        # Random direction
        d_rand = rng.normal(0, 1, n)
        d_rand -= np.mean(d_rand)
        nrm_r = np.linalg.norm(d_rand)
        if nrm_r > 1e-10:
            dirs.append(d_rand / nrm_r)
        
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
            # Cycle through parameters
            if stall == 0:
                k_idx += 1
        else:
            stall += 1
            rad *= 0.7
            if rad < 0.005:
                # Perturbation
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                    k_idx += 1
                    bs_idx += 1
                elif stall % 20 == 0:
                    rad = 0.06
                    k_idx += 1
                    beta_idx += 1
                    bs_idx += 1
    
    return best_h, best_c

def _bin_sa(hb, deadline, mx=30000, Ti=0.003):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(31415)
    T = Ti
    stall = 0
    
    for it in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        
        # Choose move type
        r = rng.random()
        if r < 0.5:
            # Single pair swap
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            hn = cur.copy()
            hn[oi] = 0.0
            hn[oz] = 1.0
        elif r < 0.8:
            # Double pair swap
            if len(ones) < 2 or len(zeros) < 2:
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                oi, oz = ones[i], zeros[j]
                hn = cur.copy()
                hn[oi] = 0.0
                hn[oz] = 1.0
            else:
                i1 = int(rng.integers(0, len(ones)))
                i2 = int(rng.integers(0, len(ones)))
                j1 = int(rng.integers(0, len(zeros)))
                j2 = int(rng.integers(0, len(zeros)))
                if i1 == i2 or j1 == j2:
                    i = int(rng.integers(0, len(ones)))
                    j = int(rng.integers(0, len(zeros)))
                    oi, oz = ones[i], zeros[j]
                    hn = cur.copy()
                    hn[oi] = 0.0
                    hn[oz] = 1.0
                else:
                    hn = cur.copy()
                    hn[ones[i1]] = 0.0
                    hn[ones[i2]] = 0.0
                    hn[zeros[j1]] = 1.0
                    hn[zeros[j2]] = 1.0
        else:
            # Block move: move a contiguous block of ones to new positions
            blen = int(rng.integers(2, 5))
            if len(ones) < blen:
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                hn = cur.copy()
