def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 0.000001
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_c5(h, d, dx, a_max=1.0):
    """Line search: minimize max(raw + a*T - a^2*Q) * dx over feasible a."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible range
    pos = d > 1e-12
    neg = d < -1e-12
    a_hi = a_max
    a_lo = -a_max
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Evaluate at key points
    cands = [a_lo, a_hi, 0.0, (a_lo + a_hi) / 2.0]
    # Add vertices from top lags
    top_idx = np.argsort(-S0)[:min(8, len(S0))]
    for ki in top_idx:
        if Q[ki] > 1e-14:
            av = T[ki] / (2.0 * Q[ki])
            if a_lo <= av <= a_hi:
                cands.append(av)
    # Grid
    ng = max(20, int((a_hi - a_lo) * 100))
    cands.extend(np.linspace(a_lo, a_hi, min(ng, 64)).tolist())
    
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.1 * (a_hi - a_lo))
    hi_r = min(a_hi, best_a + 0.1 * (a_hi - a_lo))
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(20):
            if hi_r - lo_r < 1e-11:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

    # where k = t-(n-1)

    # When h[i] changes by dv, and i=a:
    #   New term: (h[i]+dv) * (1-h[i+k])
    #   Change: dv * (1-h[i+k])

def _swap_delta_fast(h, i, j, n):
    delta = np.zeros(2*n-1)
    # For position p going by dv:
    # Part A: t in [n-1-i, n-1+(n-1-i)] = [n-1-i, 2n-2-i], contribution = dv*(1-h[i+k]) where k=t-(n-1)
    #   So at index t: contribution = dv*(1-h[i+t-n+1])
    #   Valid range of t: n-1-i <= t <= 2n-2-i (and i+k = i+t-n+1 in [0,n))
    # Part B: t in [i, n-1+i], contribution = -dv*h[i-k] = -dv*h[i-(t-(n-1))] = -dv*h[i-t+n-1]
    
    def add_swap_contribution(p, dv):
        # Part A: t from max(n-1-p, 0) to min(2n-2-p, 2n-2)
        t_lo = max(n-1-p, 0)
        t_hi = min(2*n-2-p, 2*n-2)
        if t_lo <= t_hi:
            k_vals = np.arange(t_lo, t_hi+1) - (n-1)  # these are the k values
            idx_a = t_lo:t_hi+1
            # 1-h[p+k_vals]
            idx_into_h = p + k_vals
            # idx_into_h should be in [0, n)
            delta[idx_a] += dv * (1.0 - h[idx_into_h])
        
        # Part B: t from max(p, 0) to min(n-1+p, 2n-2)
        t_lo2 = max(p, 0)
        t_hi2 = min(n-1+p, 2*n-2)
        if t_lo2 <= t_hi2:
            k_vals2 = np.arange(t_lo2, t_hi2+1) - (n-1)
            idx_into_h2 = p - k_vals2  # = p - (t - (n-1)) = p + n - 1 - t
            delta[t_lo2:t_hi2+1] += -dv * h[idx_into_h2]
    
    add_swap_contribution(i, -1.0)
    add_swap_contribution(j, +1.0)
    return delta

    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    cands = [a_lo, a_hi, 0.0, (a_lo + a_hi) / 2.0]
    top_idx = np.argsort(-S0)[:min(8, len(S0))]
    for ki in top_idx:
        if Q[ki] > 1e-14:
            av = T[ki] / (2.0 * Q[ki])
            if a_lo <= av <= a_hi:
                cands.append(av)
    cands.extend(np.linspace(a_lo, a_hi, 48).tolist())
    
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a

    hi_r = min(a_hi, best_a + 0.15 * (a_hi - a_lo))
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1, f2 = f(c1), f(c2)
        for _ in range(22):
            if hi_r - lo_r < 1e-11:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)

                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = f(c2)
        mid = (lo_r + hi_r) / 2.0
        fm = f(mid)
        if fm < best_val:
            best_val = fm
            best_a = mid
    
    return best_a, best_val

- As the "second" factor (1-h[a+k])=(1-h[p]): contributes -dv * h[p-k] where k = t-(n-1), valid when 0≤p-k<n

def _compute_swap_change(h, i, j, n, old_corr):
    """Compute new_corr after swapping h[i]=1->0 and h[j]=0->1."""
    # Change at position i (dv=-1):
    # As first: -(1-h[i+k]) for k in [-i, min(n-1-i, n-1)]
    # As second: +h[i-k] for k in [max(i-(n-1), -(n-1)), min(i, n-1)]
    
    new_corr = old_corr.copy()
    
    # Position i, dv=-1
    # First factor: k from -i to n-1-i
    k_min = max(-i, -(n-1))
    k_max = min(n-1-i, n-1)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max+1)
        ts = ks + n - 1
        new_corr[ts] -= (1.0 - h[i + ks])
    
    # Second factor: k from i-(n-1) to i
    k_min2 = max(i - (n-1), -(n-1))
    k_max2 = min(i, n-1)
    if k_min2 <= k_max2:

        new_corr[ts2] += h[i - ks2]
