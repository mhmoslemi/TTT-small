Let me define the set of j values that are "affected" by the swap for lag m:

- j = b-m (if valid)

These might overlap (e.g., a = b-m means a+m = b, etc.)

The cleanest approach is to just compute the old and new values of the affected terms directly:

For each affected j, compute old_term = h_old[j]*(1-h_old[j+m]) and new_term = h_new[j]*(1-h_new[j+m]), and sum the differences.

In practice, there are at most 4 affected j values per lag (could be fewer due to overlaps). So for each swap, the total work is O(1) per affected lag, but we need to check ALL lags (there are 2n-1 of them).

- a is a valid j for lag m iff 0≤a≤n-1 and 0≤a+m≤n-1, i.e., -a ≤ m ≤ n-1-a.

So the affected lags are those in the union of:
[-a, n-1-a] ∪ [a-(n-1), a] ∪ [-b, n-1-b] ∪ [b-(n-1), b]

Each of these is an interval of length n. The union could cover all 2n-1 lags in the worst case. So in the worst case, we still need to check all lags.

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _initial_binary(n, rng, mode='random'):
    if mode == 'thue_morse':
        # Thue-Morse sequence of length n
        t = np.zeros(n)
        for i in range(1, n):
            t[i] = t[i >> 1] ^ (i & 1)
        # Adjust to have exactly n/2 ones
        # Thue-Morse of length 2^k has exactly 2^(k-1) ones
        # For arbitrary n, we might need to adjust
        if t.sum() != n // 2:
            # Swap some 1s and 0s to get exactly n/2
            ones = np.where(t == 1)[0]
            zeros = np.where(t == 0)[0]
            if t.sum() > n // 2:
                for idx in ones[:int(t.sum() - n//2)]:
                    t[idx] = 0
            else:
                for idx in zeros[:int(n//2 - t.sum())]:
                    t[idx] = 1
        return t
    else:
        # Random binary with n/2 ones
        pos = rng.permutation(n)[:n//2]
        h = np.zeros(n)
        h[pos] = 1.0
        return h

def _binary_sa(h, deadline, n_iter_max=5000000, Ti=0.003, moves_fraction_block=0.3):

    zeros = np.where(h <= 0.5)[0].tolist()

    n_eval = 0

    while time.time() < deadline and n_eval < n_iter_max:
        # Choose move type

            # Block swap: swap blocks of size 1-4
            bs = int(rng.integers(1, 5))
            # Pick a random 1-block and 0-block of size bs
            # Find consecutive runs

            i = int(rng.integers(0, n - bs))
            j = int(rng.integers(0, n - bs))
            # Check if block i has more 1s and block j has more 0s (or vice versa)
            # Actually for simplicity, just do pair swaps for now
            # Let's do a different block move: pick two random positions and swap
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a == b or h[a] == h[b]:
                continue
        else:
            # Pair swap
            if not ones or not zeros:
                break
            a = ones[int(rng.integers(0, len(ones)))]
            b = zeros[int(rng.integers(0, len(zeros)))]
        
        if h[a] == h[b]:
            continue
        
        # Perform swap
        h[a], h[b] = h[b], h[a]
        new_score = _score(h)
        n_eval += 1
        
        delta = new_score - cur
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur = new_score
            # Update ones and zeros lists
            if h[a] == 1:
                ones.append(b)
                ones.remove(a)
            else:
                zeros.append(b)
                zeros.remove(a)
            if cur < best - 1e-12:
                best = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:

            h[a], h[b] = h[b], h[a]
            stall += 1
        
        # Cooling
        T *= 0.99999
        if stall > 500:
            T = max(T * 2, 0.001)
            stall = 0
    
    return best_h, best
