def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {0 ≤ x ≤ 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Build G matrix where G[r,j] = dR[r]/dh[j]."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):

                                        best_h = h.copy()
                                    stall = 0
        else:
            stall += 1
            beta = max(beta_start, beta * 0.9)

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=60, seed=42):
    """Pair-based optimization (from previous code, kept as fallback)."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(R))
        w = np.exp(beta * (R - M))
        ws = float(np.sum(w))
        if ws < 1e-30:
            top_k = max(1, len(R) // 10)
            idx = np.argsort(R)[-top_k:]
            w = np.zeros(len(R))
            w[idx] = 1.0 / top_k
        else:
            w /= ws
        
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        n_half = min(n_pairs, n // 2)
        sources = sort_idx[-n_half:]
        sinks = sort_idx[:n_half]
        pairs = list(zip(sinks.astype(int), sources.astype(int)))
        
        # Add some random pairs
        for _ in range(min(10, max(1, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            B = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(R, A, B, t_lo, t_hi)
            if abs(t_opt) < 1e-14:
                continue
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            R = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=20000, T_init=0.005, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[i], cur[j] = cur[j], cur[i]  # revert
            stall += 1
    return best_h, best_c5

def _binary_sweep(h_bin, deadline):
    """Greedy pair swap sweep for binary vectors."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _multi_beta_dense(h, deadline, betas=[3.0, 8.0, 20.0, 40.0], seed=42):
    """Run dense optimization with multiple beta values."""
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n = len(h)
    for bi, beta in enumerate(betas):
        t_sub = deadline - (deadline - time.time()) * (len(betas) - bi - 1) / len(betas)
        t_sub = max(time.time() + 5, min(t_sub, deadline - 2))
        h_cur, c5_cur = _dense_optimize(h, t_sub, beta_start=beta, beta_end=beta * 3, n_iter_max=200, seed=seed + bi * 100)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            h = h_cur.copy()
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        lag = r - (n - 1)
        j1s = max(0, -lag)
        j1e = min(n - 1, n - 1 - lag)
        if j1s <= j1e:
            G[r, j1s:j1e + 1] += (1.0 - h[j1s + lag:j1e + lag])
        j2s = max(0, lag)
        j2e = min(n - 1, n - 1 + lag)
        if j2s <= j2e:
            G[r, j2s:j2e + 1] -= h[j2s - lag:j2e - lag]
    return G

                # Random dense direction
                dr = rng.standard_normal(n)
                dr -= dr.mean()
                dr /= (np.max(np.abs(dr)) + 1e-14)
                pr = dr > 0
                nr = dr < 0
                thi_r = float(np.min((1.0 - h[pr]) / dr[pr])) if np.any(pr) else 1.0
                tlo_r = -float(np.min(h[nr] / (-dr[nr]))) if np.any(nr) else -0.3
                thi_r = min(thi_r, 0.4)
                tlo_r = max(tlo_r, -0.4)
                if thi_r - tlo_r > 1e-10:
                    Ar = np.correlate(dr, 1.0 - h, mode='full') - np.correlate(h, dr, mode='full')
                    Br = np.correlate(dr, dr, mode='full')
                    tr, vr = _line_search(R, Ar, Br, tlo_r, thi_r)
                    if vr < M - 1e-13 and abs(tr) > 1e-10:
                        hn = _project(h + tr * dr, target)
                        cn = _exact_c5(hn)
                        if cn < cur_c5 - 1e-14:
                            h = hn
                            cur_c5 = cn
                            R = np.correlate(h, 1.0 - h, mode='full')
                            stall = 0
                            if cn < best_c5 - 1e-12:
                                best_c5 = cn
                                best_h = h.copy()

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=60, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(R))
        w = np.exp(beta * (R - M))
        ws = float(np.sum(w))
        if ws < 1e-30:
            top_k = max(1, len(R) // 10)
            idx = np.argsort(R)[-top_k:]
            w = np.zeros(len(R))
            w[idx] = 1.0 / top_k
        else:
            w /= ws
        G = _grad_matrix(h)
        cg = G.T @ w
        si = np.argsort(cg)
        nh = min(n_pairs, n // 2)
        sources = si[-nh:]
        sinks = si[:nh]
        pairs = list(zip(sinks.astype(int), sources.astype(int)))
        for _ in range(min(12, max(1, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            B = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search(R, A, B, t_lo, t_hi)
            if abs(t_opt) < 1e-14:
                continue
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            R = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=20000, T_init=0.005, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _binary_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense optimization with multiple betas (30% time)
    d1 = t0 + budget_s * 0.30
    remaining_at_d1 = d1 - t0
    for beta in [5.0, 12.0, 30.0]:
        sub_dl = min(d1, time.time() + remaining_at_d1 * 0.38)
        hr, cr = _dense_optimize(best_h, sub_dl, beta_start=beta, beta_max=beta * 4,
                                  n_iter_max=300, seed=seed + int(beta))
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()

    # Phase 2: Pair optimization (15% time)
    d2 = t0 + budget_s * 0.45
    hp, cp = _pair_optimize(best_h, d2, radius=0.18, n_pairs=55, beta=25.0,
                            stall_limit=80, seed=seed + 100)
    if cp < best_c5:
        best_c5 = cp
        best_h = hp.copy()

    # Phase 3: Multi-n exploration (20% time)
    d3 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    n_tried = set()
    for dn in [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -28, 28, 32, -32]:
        if time.time() > d3:
            break
        nt = n0 + dn
        if nt < 80 or nt > 300 or nt % 2 != 0 or nt in n_tried:
            continue
        n_tried.add(nt)
        h_r = _resample(best_h, nt)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c_r = _dense_optimize(h_r, sub_dl, beta_start=10.0, beta_max=40.0,
                                    n_iter_max=80, seed=seed + 200 + abs(dn))
        sub_dl2 = min(d3, time.time() + budget_s * 0.008)
        h_r, c_r = _pair_optimize(h_r, sub_dl2, radius=0.1, n_pairs=30, beta=25.0,
                                   stall_limit=25, seed=seed + 400 + abs(dn))
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 4: Binary diversification (15% time)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    if time.time() < d4:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=25000, T_init=0.004, seed=seed + 600)
        sub_dl2 = min(d4, time.time() + budget_s * 0.025)
        h_bin, c5_bin = _binary_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d4:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _dense_optimize(h_pol, sub_dl3, beta_start=15.0, beta_max=60.0,
                                             n_iter_max=200, seed=seed + 700)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Final refinement (10% time)
    d5 = deadline - 3
    n_curr = len(best_h)
    if time.time() < d5:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + (d5 - time.time()) * 0.6)
        h5, c5_5 = _dense_optimize(h5, sub_dl, beta_start=6.0, beta_max=70.0,
                                    n_iter_max=400, seed=seed + 800)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < d5:
            h5b = _project(best_h.copy(), n_curr / 2.0)
            h5b, c5_5b = _pair_optimize(h5b, min(d5, time.time() + budget_s * 0.02),
                                         radius=0.08, n_pairs=60, beta=35.0,
                                         stall_limit=40, seed=seed + 900)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()
        # Try a couple nearby n values
        for dn in [-4, 4]:
            if time.time() > d5:
                break
            nt = n_curr + dn
            if nt < 80 or nt > 300 or nt % 2 != 0:
                continue
            h_r = _resample(best_h, nt)
            h_r, c_r = _dense_optimize(h_r, min(d5, time.time() + budget_s * 0.01),
                                        beta_start=15.0, beta_max=50.0, n_iter_max=60, seed=seed + 1000 + abs(dn))
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
