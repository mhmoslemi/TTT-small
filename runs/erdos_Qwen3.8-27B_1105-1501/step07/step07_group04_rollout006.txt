import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _get_grads(h, active_lags):
    """Compute gradients of correlation at active lags.
    Returns array of shape (len(active_lags), n).
    """
    n = len(h)
    center = n - 1
    dx = 2.0 / n
    grads = np.zeros((len(active_lags), n))
    for idx, r in enumerate(active_lags):
        k = r - center
        g = np.zeros(n)
        if k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        elif k < 0:
            u = -k
            g[:n-u] += (1.0 - h[u:])
            g[u:] -= h[:n-u]
        else:
            g[:] = 1.0 - 2.0 * h
        grads[idx] = g * dx
    return grads

def _lp_direction(h, corr, radius, max_active=80):
    """Solve LP to find direction minimizing max gradient over active lags."""
    n = len(h)
    m = len(corr)  # 2n-1
    
    # Select active lags: those with corr close to max
    max_corr = float(corr.max())
    tol = max(1e-10, 0.01 * max_corr)
    active = np.where(corr >= max_corr - tol)[0]
    
    if len(active) > max_active:
        # Keep only top max_active
        order = np.argsort(-corr[active])[:max_active]
        active = active[order]
    
    a_lags = active  # array of lag indices
    
    # Build gradient matrix
    G = _get_grads(h, a_lags)  # shape (len(a_lags), n), already scaled by dx
    
    # LP: minimize t
    # s.t. G @ d <= t (i.e., G @ d - t <= 0)
    #      sum(d) = 0
    #      -radius <= d[i] <= radius
    #
    # Variables: d[0..n-1], t (total n+1 variables)
    
    # Objective: c = [0,...,0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag r: G[r,:] @ d - t <= 0
    A_ub = np.zeros((len(a_lags), n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(len(a_lags))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-radius, radius)] * n
    bounds.append((max_corr * 0.5, max_corr * 1.5))  # t bounds (loose)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        d = res.x[:n]
        # Ensure sum is exactly 0
        d = d - np.mean(d)
        return d
    return None

def _line_search(h, d, dx, max_alpha=1.0):
    """Exact line search along direction d.
    Uses the quadratic expansion (which is exact for bilinear correlations).
    """
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find max feasible alpha
    alpha_max = max_alpha
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    if alpha_max < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # The objective as function of alpha: max(C0 + alpha*L - alpha^2*Q) * dx
    # This is max of concave quadratics, so it's concave in alpha.
    # Use golden section search on [0, alpha_max]
    
    def f(alpha):
        return float(np.max(C0 + alpha * L - alpha * alpha * Q)) * dx
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    opt_alpha = (lo + hi) / 2.0
    opt_val = f(opt_alpha)
    
    # Also check endpoints
    candidates = [(0.0, float(np.max(C0) * dx)),
                  (opt_alpha, opt_val),
                  (alpha_max, f(alpha_max))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_optimize(h, deadline, radius_init=0.08, max_iter=5000, max_active=80):
    """Main LP-based optimization loop."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _lp_direction(h, corr, radius, max_active=max_active)
        
        if d is None or np.max(np.abs(d)) < 1e-14:
            # Fallback: subgradient
            max_val = float(corr.max())
            active = np.where(corr >= max_val - 1e-10)[0]
            g = np.zeros(n)
            for r in active:
                k = r - (n - 1)
                if k > 0:
                    g[k:] += (1.0 - h[:n-k])
                    g[:n-k] -= h[k:]
                elif k < 0:
                    u = -k
                    g[:n-u] += (1.0 - h[u:])
                    g[u:] -= h[:n-u]
                else:
                    g[:] = 1.0 - 2.0 * h
            g /= len(active)
            d = -g - np.mean(-g)
        
        alpha, new_c5 = _line_search(h, d, dx, max_alpha=radius + 0.5)
        
        if new_c5 < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-5:
                break
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')

def _binary_targeted(h_bin, deadline, max_iters=100000, seed=42):

    n = len(h_bin)
    dx = 2.0 / n

    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        max_idx = int(np.argmax(cur_corr))
        best_delta = 0.0
        best_move = None
        
        # Try to find moves that reduce the max correlation
        n_try = min(300, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            
            # Compute delta efficiently
            # Moving 1 from oi to oz changes correlation at each lag
            # For lag k: the change is related to h[oi±k] and h[oz±k]
            # Simple approach: just compute the new c5
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_corr = np.correlate(trial, 1.0 - trial, mode='full')
            new_c5 = float(np.max(new_corr) * dx)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j, new_corr)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j, new_corr = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = new_corr
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)
