def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _compute_deltas(h, n, M):
    D_plus = np.zeros((n, M))
    center = n - 1
    for p in range(n):
        for k in range(1, n):
            r_pos = center + k
            if p >= k and p <= n - 1:
                D_plus[p, r_pos] += (1.0 - h[p - k])
            if p <= n - k - 1:
                D_plus[p, r_pos] -= h[p + k]
            r_neg = center - k

                D_plus[p, r_neg] += (1.0 - h[p + k])
            if p >= k:
                D_plus[p, r_neg] -= h[p - k]

    best_h = h.copy()
    best_c5 = cur_c5

            break
        if stall >= 2:
            break
        D_plus = _compute_deltas(h, n, M)
        D_minus = -D_plus
        ones = np.where(h > 0.5)[0]

def _continuous_refine(h_start, deadline, max_solves=15):
    n = len(h_start)
    M = 2 * n - 1
    dx = 2.0 / n
    s = n / 2.0
    h = _project(h_start.copy(), s)

    best_h = h.copy()

    x0 = np.concatenate([h, [cur_c5 + 0.001]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': lambda x: _nlp_ineq(x, n, dx, s),
         'jac': lambda x: _nlp_ineq_jac(x, n, dx, s)},
        {'type': 'eq', 'fun': lambda x: _nlp_eq(x, n, dx, s),
         'jac': lambda x: _nlp_eq_jac(x, n, dx, s)}
    ]
    for _ in range(max_solves):
        if time.time() > deadline:
            break
        try:
            res = minimize(_nlp_objective, x0, method='SLSQP',
                           jac=_nlp_objective_grad, args=(n, dx, s),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12})
            if res.x is not None:

                c5_try = _exact_c5(h_try)

            break

def _perturb_and_refine(h, deadline, rng, n_perturbations=10):

    best_c5 = _exact_c5(h)
    for _ in range(n_perturbations):
        if time.time() > deadline:
            break
        amplitude = rng.uniform(0.01, 0.05)
        noise = amplitude * rng.normal(0, 1, n)
        h_pert = _project(h + noise, n / 2.0)

        if c5_r < best_c5 - 1e-12:

            h[best_pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n0 = len(h_parent)
    h_init = _project(h_parent.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary full-swap search (35% of budget)
    d1 = t0 + budget_s * 0.35
    h_bin = _to_binary(h_init, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    h_bin, c5_bin = _binary_full_swap(h_bin, d1, max_passes=500)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Stochastic escape + 2-swaps (20%)
    d2 = t0 + budget_s * 0.55
    h_b2, c5_b2 = _binary_stochastic(best_h, min(d2, time.time() + budget_s * 0.08), seed=seed + 1, temp=0.004)
    if c5_b2 < best_c5:
        best_c5 = c5_b2
        best_h = h_b2.copy()
    h_b3, c5_b3 = _binary_2swap(best_h, min(d2, time.time() + budget_s * 0.06), seed=seed + 2)
    if c5_b3 < best_c5:
        best_c5 = c5_b3
        best_h = h_b3.copy()
    # Continue full-swap from new state
    if time.time() < d2 - 5:
        h_b4, c5_b4 = _binary_full_swap(best_h, d2, max_passes=200)
        if c5_b4 < best_c5:
            best_c5 = c5_b4
            best_h = h_b4.copy()

    # Phase 3: Continuous refinement (25%)
    d3 = t0 + budget_s * 0.80
    h_cont = _project(best_h.copy(), len(best_h) / 2.0)
    h_c1, c5_c1 = _continuous_refine(h_cont, min(d3, time.time() + budget_s * 0.12), max_solves=10)
    if c5_c1 < best_c5:
        best_c5 = c5_c1
        best_h = h_c1.copy()
    # Perturb and refine
    if time.time() < d3 - 20:
        n_best = len(best_h)
        h_c2, c5_c2 = _perturb_and_refine(best_h, d3, rng, n_perturbations=8)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2.copy()

    # Phase 4: Multi-n exploration (12%)
    d4 = t0 + budget_s * 0.92
    n_curr = len(best_h)
    for dn in [0, 6, -6, 12, -12, 18, -18, 24]:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _continuous_refine(h_r, sub_dl, max_solves=5)
        if time.time() < d4 - 3:
            h_br = _to_binary(h_r, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_br, c5_br = _binary_full_swap(h_br, sub_dl2, max_passes=100)
            if c5_br < c5_r:
                c5_r = c5_br
                h_r = h_br
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final intensive polish (8%)
    d5 = deadline - 3.0
    n_fin = len(best_h)
    # Full-swap on binary at best n
    h_fin_bin = _to_binary(best_h, n_fin)
    if _exact_c5(h_fin_bin) < best_c5:
        best_h = h_fin_bin
        best_c5 = _exact_c5(h_fin_bin)
    sub_dl = min(d5, time.time() + budget_s * 0.03)
    h_fin_bin, c5_fin = _binary_full_swap(h_fin_bin, sub_dl, max_passes=200)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin_bin.copy()
    # Continuous on final
    if time.time() < d5 - 5:
        h_fin = _project(best_h.copy(), n_fin / 2.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_fin, c5_fin2 = _continuous_refine(h_fin, sub_dl2, max_solves=8)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin.copy()
    # Stochastic polish
    if time.time() < d5 - 3:
        h_sp, c5_sp = _binary_stochastic(best_h, d5 - 1, seed=seed + 99, temp=0.002)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()

    # Final projection and evaluation
    best_h = _project(best_h.copy(), len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
