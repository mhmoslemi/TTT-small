import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(v.sum())
    if abs(s - target) < 1e-14:
        return v
    # Water filling
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(x.sum())
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(mask.sum())
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_c(h, scores, T):
    """Compute gradient of log-sum-exp of scores with temperature T."""
    n = len(h)
    # Weights
    shifted = T * (scores - np.max(scores))
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient: sum_d w_d * dc_d/dh
    # dc_d/dh[i] for lag d = (i - (n-1)):
    # For d >= 0: dc_d/dh[j+d] += (1-h[j]), dc_d/dh[j] -= h[j+d]
    # For d < 0 (let g = -d): dc_d/dh[j] += (1-h[j+g]), dc_d/dh[j+g] -= h[j]
    g = np.zeros(n)
    lags = np.arange(len(scores)) - (n - 1)
    for idx in range(len(scores)):
        wi = w[idx]
        if wi < 1e-15:
            continue
        d = lags[idx]
        if d >= 0:
            m = n - d
            g[d:] += wi * (1.0 - h[:m])
            g[:m] -= wi * h[d:]
        else:
            gg = -d
            m = n - gg
            g[:m] += wi * (1.0 - h[gg:])
            g[gg:] -= wi * h[:m]
    return g

def _lse_optimize(h, deadline, T_start=1.0, T_end=500.0, max_iters=5000, lr_start=0.01):
    """Log-sum-exp smooth optimization with adaptive T and learning rate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_start
    lr = lr_start
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        mx = np.max(scores)
        
        # Adaptive T: increase as we focus on the max
        T = min(T_end, T * 1.02)
        
        g = _grad_c(h, scores, T)
        # Project gradient: subtract mean for sum constraint
        g -= g.mean()
        
        # Gradient descent with projection
        h_new = _project(h - lr * g, target)
        
        # Check improvement
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            lr *= 0.8
            if stall > 100:
                # Reset T and try again
                T = max(T_start, T * 0.5)
                lr = lr_start * 0.5
                stall = 0
            if lr < 1e-6:
                break

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 50)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(49, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    return min(cands, key=lambda x: x[1])

def _active_lp(h, scores, radius, n_active=16, gap=0.006):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = np.zeros((len(active), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += 1.0 - h[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += 1.0 - h[g:]
                G[idx, g:] += -h[:m]
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_dir(h, scores, beta=25.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += 1.0 - h[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += 1.0 - h[g:]
                G[idx, g:] += -h[:m]
    gvec = G.T @ w
    gvec -= float(np.mean(gvec))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    gvec[pos] = np.minimum(gvec[pos], (1.0 - h[pos]) / radius)
    gvec[neg] = np.maximum(gvec[neg], -h[neg] / radius)
    gvec[h >= 1.0 - 1e-12] = 0.0
    gvec[h <= 1e-12] = 0.0
    gvec -= float(np.mean(gvec))
    return gvec

def _lse_optimize(h, deadline, T_start=5.0, T_end=200.0, lr_start=0.005):

            break
        scores_raw = np.correlate(h, 1.0 - h, mode='full')
        shifted = T * (scores_raw - np.max(scores_raw))
        w = np.exp(shifted)
        w /= w.sum()
        g = np.zeros(n)
        lags = np.arange(n_lags) - (n - 1)
        for idx in range(n_lags):
            wi = w[idx]
            if wi < 1e-14:
                continue
            d = lags[idx]
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += wi * (1.0 - h[:m])
                    g[:m] += -wi * h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    g[:m] += wi * (1.0 - h[gg:])
                    g[gg:] += -wi * h[:m]
        g -= g.mean()
        h_new = _project(h - lr * g, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = new_c5

                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0

            T = min(T * 1.01, T_end)
        else:
            stall += 1
            lr *= 0.85
            if stall % 50 == 0:
                T = max(T_start, T * 0.6)

def _binary_sa_exact(b, deadline, T_init=0.004, max_iter=200000):
    n = len(b)
    half = n // 2
    b = b.copy()

    stall = 0
    n_lags = 2 * n - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        oi = int(rng.integers(half))
        zi = int(rng.integers(half))
        r = ones_idx[oi]
        s = zeros_idx[zi]
        if r == s:
            continue
        delta = np.zeros(n_lags)
        for d_val in range(-(n-1), n):
            idx = d_val + (n - 1)
            total = 0.0
            j1 = r
            jd1 = r + d_val
            if 0 <= jd1 < n and jd1 != r:
                total -= (1.0 - b[jd1]) if jd1 != s else 0.0
            j2 = r - d_val
            if 0 <= j2 < n and j2 != r:
                total += 1.0 if (j2 == s) else b[j2]
            j3 = s
            jd3 = s + d_val
            if 0 <= jd3 < n and jd3 != s:
                total += (1.0 - b[jd3]) if jd3 != r else 1.0
            j4 = s - d_val
            if 0 <= j4 < n and j4 != s and j4 != j2:
                if j4 == j1:
                    pass
                elif j4 == r:
                    total -= 1.0
                else:
                    total -= b[j4]
            elif j4 == j2:
                pass
            delta[idx] = total
        new_counts = counts + delta
        new_max = float(np.max(new_counts))
        delta_c5 = new_max - cur_max
        if delta_c5 <= 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-15)):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 10.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.4 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _random_balanced(n, rng, seed_type=0):
    half = n // 2
    b = np.zeros(n)
    if seed_type == 0:
        idx = rng.choice(n, size=half, replace=False)
    elif seed_type == 1:
        step = 1
        for s in range(2, n):
            if np.gcd(s, n) == 1:
                step = s
                break
        idx = (np.arange(half) * step) % n
    elif seed_type == 2:
        idx = np.sort(rng.choice(n, size=half, replace=False))
    else:
        pos = np.linspace(0, n - 1, half, endpoint=False)
        idx = np.unique(pos.astype(int))
        if len(idx) < half:
            remaining = np.setdiff1d(np.arange(n), idx)
            extra = rng.choice(remaining, size=half - len(idx), replace=False)
            idx = np.sort(np.concatenate([idx, extra]))
    b[idx] = 1.0
    return b

    h2, c5_2 = _lse_optimize(_project(best_h.copy(), len(best_h) / 2.0), d2,
                              T_start=5.0, T_end=300.0, lr_start=0.004)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution binary SA (30%)
    d3 = t0 + budget_s * 0.55
    n_base = len(best_h)
    tried_ns = set()
    # Try a range of n values around n_base
    for dn in sorted(range(-20, 21, 4), key=lambda x: abs(x)):
        if time.time() > d3 - 5:
            break
        n_try = n_base + dn
        if n_try < 100 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        # Resample and optimize
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.08, max_iters=600)
        if c5_r >= best_c5 - 0.003:
            continue
        # Binary SA at this resolution
        if time.time() < d3 - 3:
            h_bin = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.02)
            b_best, max_cnt = _binary_sa_exact(h_bin, sub_dl2, T_init=0.004, max_iter=80000)
            bin_c5 = max_cnt * (2.0 / n_try)
            if bin_c5 < c5_r:
                h_r = b_best.astype(float)
                c5_r = bin_c5
            # Continuous polish from binary
            if time.time() < d3 - 2:
                h_pol = _project(h_r.copy(), n_try / 2.0)
                sub_dl3 = min(d3, time.time() + budget_s * 0.008)
                h_pol, c5_pol = _local_opt(h_pol, sub_dl3, radius_init=0.05, max_iters=400)
                if c5_pol < c5_r:
                    c5_r = c5_pol
                    h_r = h_pol
            if time.time() < d3 - 1:
                sub_dl4 = min(d3, time.time() + budget_s * 0.005)
                h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl4, max_rounds=30)
                if c5_r2 < c5_r:
                    c5_r = c5_r2
                    h_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: More diverse binary starts (15%)
    d4 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    for st in range(6):
        if time.time() > d4 - 3:
            break
        for stype in range(3):
            if time.time() > d4 - 2:
                break
            b_rand = _random_balanced(n_curr, rng, seed_type=stype + st)
            b_try, max_try = _binary_sa_exact(b_rand,
                                               min(d4, time.time() + budget_s * 0.01),
                                               T_init=0.006, max_iter=50000)
            bin_c5 = max_try * (2.0 / n_curr)
            if bin_c5 < best_c5 + 0.005:
                h_from_bin = _project(b_try.astype(float), n_curr / 2.0)
                sub_dl = min(d4, time.time() + budget_s * 0.006)
                h_pol, c5_pol = _local_opt(h_from_bin, sub_dl, radius_init=0.05, max_iters=300)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.85
    scale = 0.015
    n_restarts = 0
    while time.time() < d5 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.05, max_iters=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.06)

    # Phase 6: LSE refinement (5%)
    d6 = t0 + budget_s * 0.90
    h_lse = _lse_optimize(_project(best_h.copy(), len(best_h) / 2.0), d6,
                           T_start=20.0, T_end=500.0, lr_start=0.003)
    if h_lse[1] < best_c5:
        best_c5 = h_lse[1]
        best_h = h_lse[0].copy()

    # Phase 7: Final polish (10%)
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    h_f, c5_f = _local_opt(h_f, deadline - 1, radius_init=0.04, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if deadline - time.time() > 3:
        h_f2, c5_f2 = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), deadline, max_rounds=40)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
