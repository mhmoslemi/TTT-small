def _pair_optimize(h, i, j):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h[i], h[j], None  # no freedom
    
    # Compute a_r and b_r
    h_save_i, h_save_j = h[i], h[j]
    h[i], h[j] = lo, s - lo
    corr_lo = np.correlate(h, 1.0 - h, mode='full')
    h[i], h[j] = hi, s - hi
    corr_hi = np.correlate(h, 1.0 - h, mode='full')
    h[i], h[j] = h_save_i, h_save_j
    
    a_r = corr_hi - corr_lo  # (2n-1,)
    b_r = corr_lo  # (2n-1,)
    
    # f(t) = max_r(a_r * t + b_r) for t in [lo, hi]
    # Find minimum using ternary/golden section
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(a_r * c1 + b_r)
    f2 = np.max(a_r * c2 + b_r)
    
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(a_r * c1 + b_r)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(a_r * c2 + b_r)
    
    t_opt = (lo + hi) / 2.0
    f_opt = np.max(a_r * t_opt + b_r)
    return t_opt, s - t_opt, f_opt

def _full_sweep_best(h, corr):
    """Find the best single swap (1→0 at i, 0→1 at j) that reduces C5."""
    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    best_c5 = float(np.max(corr))
    best_pair = None

                best_pair = (i, j)

            h[i] = 1.0
            h[j] = 0.0
    
    return best_pair, best_c5, best_new_corr

def _swap_delta_vec(h, i, j):
    """Vectorized computation of delta for swap h[i]:1->0, h[j]:0->1."""

    # i-d must be in [0, n-1], i.e., d in [i-(n-1), i]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _c5_from_corr(corr, n):
    return float(np.max(corr)) * 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):

def _swap_delta(h, i, j):
    n = len(h)
    d = np.arange(-(n - 1), n)
    m = 2 * n - 1
    delta = np.zeros(m)
    mask_A = (d >= i - (n - 1)) & (d <= i)
    if np.any(mask_A):
        idx_A = i - d[mask_A]
        delta[mask_A] += -(1.0 - h[idx_A])
    mask_B = (d >= -i) & (d <= n - 1 - i)
    if np.any(mask_B):
        idx_B = i + d[mask_B]
        delta[mask_B] += h[idx_B]
    mask_C = (d >= j - (n - 1)) & (d <= j)
    if np.any(mask_C):
        idx_C = j - d[mask_C]
        delta[mask_C] += (1.0 - h[idx_C])
    mask_D = (d >= -j) & (d <= n - 1 - j)
    if np.any(mask_D):
        idx_D = j + d[mask_D]
        delta[mask_D] += -h[idx_D]
    r_ij = (i - j) + (n - 1)
    if 0 <= r_ij < m:
        delta[r_ij] += 1.0
    return delta

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _four_pt_swap_delta(h, i1, i2, j1, j2):

    # Phase 7: Final polish (8%)
    d7 = deadline
    n_curr = len(best_h)
    sub_dl = min(d7, time.time() + budget_s * 0.04)
    h_final = _project(best_h.copy(), n_curr / 2.0)
    h_final, c5_f = _pair_cd_fast(h_final, sub_dl, n_sweeps=3000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_final.copy()

    if deadline - time.time() > 5:
        h_final = _project(best_h.copy(), len(best_h) / 2.0)
        h_final, c5_f = _pair_cd_fast(h_final, deadline, n_sweeps=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)[0]
    return best_h, final_c5, len(best_h)
