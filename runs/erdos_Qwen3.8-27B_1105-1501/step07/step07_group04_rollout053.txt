import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C2 = np.correlate(d, d, mode='full')
    
    # Feasible alpha range
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Golden section search
    f0 = float(np.max(C0) * dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = 0.0
    hi_b = t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(C0 + c1 * L1 - c1*c1 * C2) * dx)
    f2 = float(np.max(C0 + c2 * L1 - c2*c2 * C2) * dx)
    
    for _ in range(60):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(C0 + c1 * L1 - c1*c1 * C2) * dx)
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(C0 + c2 * L1 - c2*c2 * C2) * dx)
    
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(C0 + opt_t * L1 - opt_t*opt_t * C2) * dx)
    
    # Check endpoints too
    candidates = [(opt_t, opt_val), (0.0, f0), (t_hi, float(np.max(C0 + t_hi*L1 - t_hi*t_hi*C2)*dx))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_direction(h, radius):
    """Solve LP to find minimax descent direction."""
    n = len(h)
    dx = 2.0 / n
    m_total = 2 * n - 1
    
    # Compute current correlation values
    C0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradients for all lags
    # grad[m, j] = (1-h[j-m]) * [0<=j-m<n] - h[j+m] * [0<=j+m<n]
    # where m is the lag, and the full index is n-1+m
    # For full index k, lag m = k - (n-1)
    
    # Build gradient matrix: shape (m_total, n)
    # For lag m = k - (n-1):
    # grad[j] = (1-h[j-m]) * [0<=j-m<n] - h[j+m] * [0<=j+m<n]
    
    # Let's build it efficiently
    grad = np.zeros((m_total, n))
    for k in range(m_total):
        m = k - (n - 1)
        if m == 0:
            grad[k, :] = 1.0 - 2.0 * h
        elif m > 0:
            # j >= m: (1 - h[j-m])
            grad[k, m:] = 1.0 - h[:n-m]
            # j <= n-1-m: -h[j+m]
            grad[k, :n-m] -= h[m:]
        else:
            u = -m
            # j <= n-1-u: (1 - h[j+u])
            grad[k, :n-u] = 1.0 - h[u:]
            # j >= u: -h[j-u]
            grad[k, u:] -= h[:n-u]
    
    # We want to minimize t subject to:
    # C0[k] + grad[k] @ d <= t for all k
    # sum(d) = 0
    # -min(radius, h[i]) <= d[i] <= min(radius, 1-h[i])
    
    # Variables: d[0..n-1], t
    # Objective: min t => c = [0,...,0, 1]
    m = n + 1
    c_obj = np.zeros(m)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # C0[k] + grad[k] @ d - t <= 0
    # => grad[k] @ d - t <= -C0[k]
    A_ub = np.zeros((m_total, m))
    for k in range(m_total):
        A_ub[k, :n] = grad[k]
        A_ub[k, n] = -1.0
    b_ub = -C0
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(radius, h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((C0.max() - 2.0, C0.max() + 2.0))  # t bounds
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    if np.max(np.abs(d)) < 1e-14:
        return None
    
    # Center the direction (should already be centered by constraint, but just in case)
    d = d - np.mean(d)
    return d

def _lp_optimize(h_start, deadline, radius_init=0.08, max_iter=2000):
    n = len(h_start)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h_start.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0

        if time.time() > deadline:
            break
        if stall > 100:
            break

            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue

def _resample(h, n_new):

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.2, 5.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
