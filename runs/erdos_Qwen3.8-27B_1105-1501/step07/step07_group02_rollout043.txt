```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)
```

```python
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(n)
    for r in range(L):
        k = r - (n - 1)  # lag
        if w[r] < 1e-15:
```

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_gradient(h, w):
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(n)
    for r in range(L):
        if w[r] < 1e-15:
            continue

            j2 = idx2 + k
            g[idx2] -= w[r] * h[j2]
    return g

    max_shift = float(np.max(shifted))
    if max_shift > 50.0 / beta:

        s = float(np.sum(w))
        if s < 1e-300:

        return t_lo, float(np.max(R0))

    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 4)]
    hi = ts[min(n_grid - 1, best_idx + 4)]
    if hi - lo > 1e-12:
        ts2 = np.linspace(lo, hi, 80)
        vals2 = np.max(R0[None, :] + ts2[:, None] * A[None, :] - (ts2[:, None]) ** 2 * B[None, :], axis=1)
        best_idx2 = int(np.argmin(vals2))
        t_opt = float(ts2[best_idx2])
        val_opt = float(vals2[best_idx2])

        t_opt = float(ts[best_idx])
        val_opt = float(vals[best_idx])
    v_lo = float(np.max(R0 + t_lo * A - t_lo * t_lo * B))
    v_hi = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))
    if v_lo < val_opt:
        t_opt, val_opt = t_lo, v_lo
    if v_hi < val_opt:
        t_opt, val_opt = t_hi, v_hi
    return t_opt, val_opt

def _line_search_pair(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo > 1e-12:
        ts2 = np.linspace(lo, hi, 40)
        vals2 = np.max(S0[None, :] + ts2[:, None] * T[None, :] - (ts2[:, None]) ** 2 * C[None, :], axis=1)
        best_idx2 = int(np.argmin(vals2))
        t_opt = float(ts2[best_idx2])
        val_opt = float(vals2[best_idx2])
    else:
        t_opt = float(ts[best_idx])
        val_opt = float(vals[best_idx])
    v_lo = float(np.max(S0 + t_lo * T - t_lo * t_lo * C))
    v_hi = float(np.max(S0 + t_hi * T - t_hi * t_hi * C))
    if v_lo < val_opt:
        t_opt, val_opt = t_lo, v_lo
    if v_hi < val_opt:
        t_opt, val_opt = t_hi, v_hi
    return t_opt, val_opt

def _dense_descent(h, deadline, beta_init=4.0, cap=0.3, seed=42, stall_limit=80):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline - 1:
        if stall > stall_limit:
            break
        R = np.correlate(h, 1.0 - h, mode='full')
        W = _soft_weights(R, beta)
        g = _compute_gradient(h, W)
        d = g - np.mean(g)
        d = -d
        d_max = float(np.max(np.abs(d)))
        if d_max < 1e-12:
            break

        pos = d > 1e-15
        neg = d < -1e-15
        t_lo = 0.0
        t_hi = np.inf
        if np.any(pos):
            t_lo = min(t_lo, float(np.max(-h[pos] / d[pos])))
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_lo = min(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))

        t_lo = max(t_lo, -cap)
        t_hi = min(t_hi, cap)
        if t_hi - t_lo < 1e-10:
            stall += 1
            continue

        one_mh = 1.0 - h
        A_vec = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        B_vec = np.correlate(d, d, mode='full')

        t_opt, _ = _line_search_dense(R, A_vec, B_vec, t_lo, t_hi)
        h_try = h + t_opt * d
        s_corr = target - float(np.sum(h_try))
        if abs(s_corr) > 1e-14:
            interior = (h_try > 1e-10) & (h_try < 1.0 - 1e-10)
            if np.any(interior):
                h_try[interior] += s_corr / float(np.sum(interior))
            else:
                h_try += s_corr / n
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = _exact_c5(h_try)

        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            beta = min(beta * 1.12, 200.0)
        else:
            stall += 1
            beta = max(beta * 0.85, 2.0)
            if stall % 8 == 0:
                pert = rng.standard_normal(n)
                pert -= np.mean(pert)
                pnorm = float(np.sum(np.abs(pert)))
                if pnorm > 1e-10:
                    h_try2 = h + 0.03 * pert / pnorm
                    h_try2 = _project(h_try2, target)
                    c5_2 = _exact_c5(h_try2)
                    if c5_2 < cur_c5:
                        h = h_try2
                        cur_c5 = c5_2

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline - 1:
        if stall > stall_limit or radius < 1e-7:
            break
        M = float(np.max(S0))
        shifted = S0 - M
        w = np.exp(beta * shifted)
        ws = float(np.sum(w))
        if ws < 1e-300:
            w = np.zeros_like(S0)
            w[np.argmax(S0)] = 1.0
        else:
            w /= ws

        G = np.zeros((2 * n - 1, n))
        for k in range(-(n - 1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                G[row, k:] += (1.0 - h[:m])
                G[row, :m] -= h[k:]
            else:
                p = -k
                m = n - p
                G[row, p:] += (1.0 - h[:m])
                G[row, :m] -= h[p:]

        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        sources = sort_idx[-min(n_pairs, n // 2):]
        sinks = sort_idx[:min(n_pairs, n // 2)]
        pairs = []
        for ii in range(min(n_pairs, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))

        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search_pair(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=15000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _pair_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline - 1:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def _hybrid_optimize(h, deadline, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mode = 0
    stall_total = 0
    beta = 4.0
    rng = np.random.default_rng(seed)

    while time.time() < deadline - 1:
        if stall_total > 60:
            break
        if mode == 0:
            sub_dl = min(deadline, time.time() + 8.0)
            h_new, c5_new = _dense_descent(h, sub_dl, beta_init=beta, cap=0.25, seed=seed + int(stall_total))
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                stall_total = 0
                beta = min(beta * 1.2, 100.0)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall_total += 1
                beta = max(beta * 0.8, 2.0)
            mode = 1
        else:
            sub_dl = min(deadline, time.time() + 6.0)
            h_new, c5_new = _pair_optimize(h, sub_dl, radius=0.12, n_pairs=35, beta=20.0, stall_limit=30, seed=seed + 100 + int(stall_total))
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                stall_total = 0
                beta = min(beta * 1.1, 80.0)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall_total += 1
            mode = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Hybrid dense+pair optimization at parent n (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _hybrid_optimize(h_init, d1, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (25% time)
    d2 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    candidate_ns = []
    for dn in [-20, -16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, 24, 28]:
        n_try = n0 + dn
        if n_try >= 80 and n_try <= 280 and n_try % 2 == 0:
            candidate_ns.append(n_try)
    rng.shuffle(candidate_ns)

    for n_try in candidate_ns:
        if time.time() > d2:
            break
        h_r = _resample(best_h, n_try)
        c5_r_init = _exact_c5(h_r)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_r, c5_r = _hybrid_optimize(h_r, sub_dl, seed=200 + n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.004, seed=777)
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _pair_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary result
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _hybrid_optimize(h_pol, sub_dl3, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Try a different binary start and diversify (10% time)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    if time.time() < d4:
        # Create a periodic-like binary pattern
        h_periodic = np.zeros(n_curr)
        period = n_curr // 3
        for blk in range(0, n_curr - period + 1, 2 * period):
            h_periodic[blk:blk + period] = 1.0
        h_periodic = _project(h_periodic, n_curr / 2.0)
        c5_per = _exact_c5(h_periodic)
        if c5_per < best_c5 + 0.02:
            sub_dl = min(d4, time.time() + budget_s * 0.04)
            h_per, c5_per = _hybrid_optimize(h_periodic, sub_dl, seed=555)
            if c5_per < best_c5:
                best_c5 = c5_per
                best_h = h_per.copy()

        # Random start
        h_rand = np.zeros(n_curr)
        idx = rng.choice(n_curr, size=n_curr // 2, replace=False)
        h_rand[idx] = 1.0
        c5_rand = _exact_c5(h_rand)
        if c5_rand < best_c5 + 0.03:
            sub_dl = min(d4, time.time() + budget_s * 0.03)
            h_rd, c5_rd = _hybrid_optimize(h_rand, sub_dl, seed=666)
            if c5_rd < best_c5:
                best_c5 = c5_rd
                best_h = h_rd.copy()

    # Phase 5: Intensive refinement on best (10% time)
    d5 = t0 + budget_s * 0.92
    n_curr = len(best_h)
    if time.time() < d5:
        sub_dl = min(d5, deadline - 5)
        h5, c5_5 = _hybrid_optimize(best_h, sub_dl, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final polish with tight params (8% time)
    d6 = deadline - 3
    n_curr = len(best_h)
    if time.time() < d6:
        h6 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d6, deadline - 2)
        h6, c5_6 = _pair_optimize(h6, sub_dl, radius=0.05, n_pairs=50, beta=30.0, stall_limit=40, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
