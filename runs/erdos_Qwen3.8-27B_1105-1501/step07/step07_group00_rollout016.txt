import numpy as np
from scipy.optimize import linprog

def _c5(h):
    """Compute C5 exactly."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    """Compute all shift scores (unnormalized)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} using water-filling."""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    # Bisection on threshold
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1
        else: mask = x > 0
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _grad_all(h):
    """Compute gradient matrix for all lags. Returns G with shape (2n-1, n)."""
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)  # signed lag
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, d:] += u[:m]
                G[k, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, :m] += u[g:]
                G[k, g:] += -h[:m]
    return G

def _full_shift_lp(h, scores, radius):
    """Full-shift LP: minimize max_k (S_k + G_k^T d) subject to constraints."""
    n = len(h)
    # Build gradient matrix
    G = _grad_all(h)
    # LP: min tau
    # s.t. G_k^T d - tau <= -S_k for all k
    #      sum(d) = 0
    #      -radius*(1-h[i]) <= d[i] <= radius*(1-h[i]) for h[i]>0... 
    #      Actually: |d[i]| <= radius AND h[i]+d[i] in [0,1]
    m = 2*n - 1
    # Variables: d[0..n-1], tau
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize tau
    A_ub = np.zeros((m, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[i]), min(radius, 1-h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

def _line_search(h, d, dx, radius):
    """Polynomial line search: maximize Q(t) = max_k (S_k(t)) where S_k(t) is quadratic in t."""
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    # Find valid t range
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos): t_hi = min(t_hi, np.min((1-h[pos])/d[pos]))
    if np.any(neg): t_hi = min(t_hi, np.min(h[neg]/(-d[neg])))
    if t_hi < 1e-14: return 0.0, float(np.max(S0))*dx
    # Golden section search
    a, b = 0.0, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    for _ in range(50):
        if b - a < 1e-12: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    t_opt = (a+b)/2
    f_opt = float(np.max(S0 + t_opt*T - t_opt*t_opt*C))*dx
    # Also check endpoints
    f0 = float(np.max(S0))*dx
    fhi = float(np.max(S0 + t_hi*T - t_hi*t_hi*C))*dx
    best = min([(0, f0), (t_opt, f_opt), (t_hi, fhi)], key=lambda x: x[1])
    return best[0], best[1]

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time
import math

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_all(h):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, d:] += u[:m]
                G[k, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, :m] += u[g:]
                G[k, g:] += -h[:m]
    return G

def _full_shift_lp(h, scores, radius):
    n = len(h)
    G = _grad_all(h)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    a, b = 0.0, t_hi
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(S0 + t_opt * T - t_opt * t_opt * C)) * dx
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
    best = min([(0.0, f0), (t_opt, f_opt), (t_hi, fhi)], key=lambda x: x[1])
    return best[0], best[1]

def _local_opt_full(h, deadline, radius=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best_c = cur
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = _scores(h)
        d = _full_shift_lp(h, scores, radius)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            # Fallback: steepest descent on max score
            idx = int(np.argmax(scores))
            lag = idx - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                m = n - lag
                g[lag:] += (1.0 - h[:m])
                g[:m] += -h[lag:]
            else:
                g2 = -lag
                m = n - g2
                g[:m] += (1.0 - h[g2:])
                g[g2:] += -h[:m]
            g -= float(np.mean(g))
            d = g
            if float(np.max(np.abs(d))) < 1e-12:
                break
        alpha, new_c = _line_search(h, d, radius)
        if new_c < cur - 1e-14:
            h = _project(h + alpha * d, target)
            cur = _c5(h)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
    return best_h, best_c

def _binary_quantize(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _binary_cdescent(b, deadline, max_rounds=200):
    n = len(b)
    b = b.copy()
    cur = _c5(b)
    best_b = b.copy()
    best_c = cur
    rng = np.random.default_rng(999)
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try all swaps if small enough, else sample
        total_pairs = len(ones) * len(zeros)
        max_pairs = 15000
        if total_pairs <= max_pairs:
            indices = np.arange(total_pairs)
        else:
            indices = rng.integers(0, total_pairs, size=max_pairs)
        best_delta = -1e-18
        best_move = None
        for idx in indices:
            oi = ones[idx // len(zeros)]
            zi = zeros[idx % len(zeros)]
            b[oi] = 0.0
            b[zi] = 1.0
            c_try = _c5(b)
            b[oi] = 1.0
            b[zi] = 0.0
            if c_try < cur - 1e-14:
                delta = cur - c_try
                if delta > best_delta:
                    best_delta = delta
                    best_move = (oi, zi)
        if best_move is not None:
            oi, zi = best_move
            b[oi] = 0.0
            b[zi] = 1.0
            cur = _c5(b)

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse_binary(n):

    while len(t) < n:

    h = np.zeros(n)

    elif n_ones < n // 2:
        zeros = np.where(h <= 0.5)[0]
        h[zeros[:n // 2 - n_ones]] = 1
    return h

def _structured_binary(n, period):

        h[ones[:n_ones - n // 2]] = 0
    elif n_ones < n // 2:
        zeros = np.where(h <= 0.5)[0]
        h[zeros[:n // 2 - n_ones]] = 1
    return h

def _random_binary(n, rng):

    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
