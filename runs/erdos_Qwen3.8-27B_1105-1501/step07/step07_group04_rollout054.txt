def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Phase 0: Initialize
    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(h_init)
    best_h = h_init.copy()
    
    # Try complement
    h_comp = 1.0 - h_init
    h_comp = _project(h_comp, n0 / 2.0)
    c5_comp = _exact_c5(h_comp)
    if c5_comp < best_c5:
        best_c5 = c5_comp
        best_h = h_comp.copy()
    
    # Phase 1: LP optimization on initial n (35% of budget)
    d1 = t0 + budget_s * 0.35
    
    sub_dl = min(d1, time.time() + budget_s * 0.20)
    h1, c5_1 = _lp_optimize(best_h, sub_dl, radius_init=0.08, max_iter=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    if time.time() < d1 - 2:
        sub_dl = min(d1, time.time() + budget_s * 0.12)
        h1, c5_1 = _subgrad_optimize(best_h, sub_dl, radius_init=0.06, max_iter=4000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Multi-n exploration (30% of budget)
    d2 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    
    n_candidates = []
    for dn in range(-40, 41, 2):
        n_try = n0 + dn
        if 60 <= n_try <= 320:
            n_candidates.append(n_try)
    
    rng.shuffle(n_candidates)
    
    for n_try in n_candidates:
        if time.time() > d2 - 10:
            break
        if n_try == n_curr:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.06, max_iter=2000)
        
        if c5_r < best_c5 + 0.002 and time.time() < d2 - 15:
            sub_dl2 = min(d2, time.time() + budget_s * 0.008)
            h_b = _to_binary(h_r, n_try)
            h_b, c5_b = _binary_swap_search(h_b, sub_dl2, max_iters=30000, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 3: Binary + basin hopping (25% of budget)
    d3 = t0 + budget_s * 0.90
    n_best = len(best_h)
    
    h_bin = _to_binary(best_h, n_best)
    c5_bin_init = _exact_c5(h_bin)
    
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin.copy()
    
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_b, c5_b = _binary_swap_search(h_bin, sub_dl, max_iters=80000, seed=202)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    
    # Basin hopping
    scale = 0.03
    for _ in range(80):
        if time.time() > d3 - 3:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d3, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.04, max_iter=500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.88)
        else:
            scale = min(0.08, scale * 1.1)
    
    # Phase 4: Final polish (10% of budget)
    d4 = deadline - 3
    
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4, c5_4 = _lp_optimize(h4, sub_dl, radius_init=0.03, max_iter=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _pair_swap_lp(h4, sub_dl, n_pairs=5000, radius=0.12, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    if time.time() < d4 - 5:
        h_b2 = _to_binary(best_h, len(best_h))
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_b2, c5_b2 = _binary_swap_search(h_b2, sub_dl, max_iters=50000, seed=999)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
    
    # Try nearby n values one more time
    n_fin = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > d4 - 2:
            break
        n_try = n_fin + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 1, time.time() + 4)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.02, max_iter=1000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
