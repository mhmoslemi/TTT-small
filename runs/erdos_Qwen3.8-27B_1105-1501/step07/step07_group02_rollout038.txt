import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _feasible_range(h, u):
    """Compute [t_lo, t_hi] such that h + t*u stays in [0,1]."""
    n = len(h)
    t_lo = -1e10
    t_hi = 1e10
    for i in range(n):
        ui = u[i]
        if ui > 1e-14:
            t_lo = max(t_lo, -h[i] / ui)
            t_hi = min(t_hi, (1.0 - h[i]) / ui)
        elif ui < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / ui)
            t_hi = min(t_hi, -h[i] / ui)
    return t_lo, t_hi

def _line_search(h, u, t_cap=1.0):
    """Exact 1D line search along direction u from point h.
    Returns (best_t, best_c5) or (0, inf) if no improvement possible."""
    n = len(h)
    t_lo, t_hi = _feasible_range(h, u)
    t_lo = max(t_lo, -t_cap)
    t_hi = min(t_hi, t_cap)
    if t_hi - t_lo < 1e-12:
        return 0.0, _exact_c5(h)
    
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(u, o, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')
    
    def f(t):
        return float(np.max(R0 + t * A - t * t * B))
    
    # Grid search
    n_grid = 100
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([f(float(t)) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(float(c1))
    f2 = f(float(c2))
    for _ in range(40):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(float(c1))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(float(c2))
    
    opt_t = float((lo + hi) / 2.0)
    best_t = float(ts[best_idx])
    best_val = float(vals[best_idx])
    for candidate in [opt_t, t_lo, t_hi, 0.0]:
        if candidate < t_lo - 1e-12 or candidate > t_hi + 1e-12:
            continue
        v = f(candidate)
        if v < best_val:
            best_val = v
            best_t = candidate

def _project(v, target):

    v = np.clip(v.copy(), 0.0, 1.0)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:

    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt

def _feasible_range(h, u):

            t_hi = min(t_hi, (1.0 - h[i]) / ui)

            t_lo = max(t_lo, (1.0 - h[i]) / ui)
            t_hi = min(t_hi, -h[i] / ui)
    return t_lo, t_hi

def _line_search(h, u, t_cap=1.0):

    t_lo = max(t_lo, -t_cap)

        return 0.0, _exact_c5(h)
    o = 1.0 - h

    A = np.correlate(u, o, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')

        return float(np.max(R0 + t * A - t * t * B))

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _feasible_range(h, u):
    n = len(h)
    t_lo = -1e10
    t_hi = 1e10
    for i in range(n):
        ui = u[i]
        if ui > 1e-14:
            t_lo = max(t_lo, -h[i] / ui)
            t_hi = min(t_hi, (1.0 - h[i]) / ui)
        elif ui < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / ui)
            t_hi = min(t_hi, -h[i] / ui)
    return t_lo, t_hi

def _line_search(h, u, t_cap=1.0):
    n = len(h)
    t_lo, t_hi = _feasible_range(h, u)
    t_lo = max(t_lo, -t_cap)
    t_hi = min(t_hi, t_cap)
    if t_hi - t_lo < 1e-12:
        return 0.0, _exact_c5(h)
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(u, o, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')

    def f(t):
        return float(np.max(R0 + t * A - t * t * B))

    n_grid = 100
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        vals[i] = f(float(ts[i]))
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(float(c1))
    f2 = f(float(c2))
    for _ in range(45):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(float(c1))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(float(c2))
    opt_t = float((lo + hi) / 2.0)
    best_t = float(ts[best_idx])
    best_val = float(vals[best_idx])
    for candidate in [opt_t, t_lo, t_hi, 0.0]:
        if candidate < t_lo - 1e-9 or candidate > t_hi + 1e-9:
            continue
        cv = f(candidate)
        if cv < best_val:
            best_val = cv
            best_t = candidate
    dx = 2.0 / n
    return best_t, best_val * dx

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.empty((L, n))
    for m in range(L):
        lag = m - (n - 1)
        if lag == 0:
            G[m, :] = 1.0 - 2.0 * h
        elif lag > 0:
            G[m, :] = 1.0
            G[m, :n - lag] -= h[lag:]
            G[m, lag:] -= h[:n - lag]
        else:
            p = -lag
            G[m, :] = 1.0
            G[m, p:] -= h[:n - p]
            G[m, :n - p] -= h[p:]
    return G

def _lp_direction(h, peak_count=8):
    n = len(h)
    o = 1.0 - h
    R = np.correlate(h, o, mode='full')
    peak_idx = np.argsort(R)[-peak_count:]
    G_a = np.zeros((peak_count, n))
    for idx, m in enumerate(peak_idx):
        lag = m - (n - 1)
        if lag == 0:
            G_a[idx, :] = 1.0 - 2.0 * h
        elif lag > 0:
            G_a[idx, :] = 1.0
            G_a[idx, :n - lag] -= h[lag:]
            G_a[idx, lag:] -= h[:n - lag]
        else:
            p = -lag
            G_a[idx, :] = 1.0
            G_a[idx, p:] -= h[:n - p]
            G_a[idx, :n - p] -= h[p:]
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    A_eq = np.zeros((2, nv))
    b_eq = np.zeros(2)
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq[1] = 1.0
    A_ub = np.zeros((peak_count, nv))
    for r in range(peak_count):
        A_ub[r, :n] = G_a[r, :]
        A_ub[r, n:2*n] = -G_a[r, :]
        A_ub[r, -1] = 1.0
    b_ub = np.zeros(peak_count)
    bounds = [(0.0, 1.0)] * (2 * n) + [(-100.0, 100.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n].copy() - res.x[n:2*n].copy()
            s = float(np.sum(u))
            if abs(s) > 1e-8:
                u = u - s / n
            nrm = float(np.linalg.norm(u))
            if nrm > 1e-10:
                return u, float(-res.fun)
    except Exception:
        pass
    return None, 0.0

def _lp_descent(h, deadline, seed=42, peak_counts=None):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    if peak_counts is None:
        peak_counts = [8, 6, 12, 4, 16, 20, 10, 14]
    pc_cycle = list(peak_counts)
    pc_idx = 0
    stall = 0
    t_cap = 1.0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 50 == 0:
            t_cap = max(0.3, 1.0 - it * 0.003)
        if stall > 80:
            if stall % 40 == 0 and np.any(h > 1e-6) and np.any(h < 1.0 - 1e-6):
                h = _project(h + 0.02 * rng.standard_normal(n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
            else:
                break
        found = False
        for _ in range(len(pc_cycle)):
            if time.time() > deadline:
                break
            pc = pc_cycle[pc_idx % len(pc_cycle)]
            pc_idx += 1
            u, beta = _lp_direction(h, pc)
            if u is None or beta <= 1e-8:
                continue
            t_opt, c5_pred = _line_search(h, u, t_cap)
            if abs(t_opt) < 1e-12:
                continue
            h_new = h + t_opt * u
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                found = True
                break
        if not found:
            stall += 1
    return best_h, best_c5

def _pair_pass(h, n_pairs=30, radius=0.15, seed=0):
    n = len(h)
    target = n / 2.0
    o = 1.0 - h
    S0 = np.correlate(h, o, mode='full')
    M = float(np.max(S0))
    w = np.exp(25.0 * (S0 - M))
    ws = float(np.sum(w))
    if ws > 1e-30:
        w /= ws
    else:
        w = np.ones(len(S0)) / len(S0)
    G = _grad_matrix(h)
    cg = G.T @ w
    idx = np.argsort(cg)
    sinks = idx[:min(n_pairs, n // 4)]
    sources = idx[-min(n_pairs, n // 4):]
    rng = np.random.default_rng(seed)
    pairs = [(int(sinks[i]), int(sources[i])) for i in range(len(sinks))]
    for _ in range(5):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a != b:
            pairs.append((a, b))
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for (si, so) in pairs:
        if si == so:
            continue
        t_lo = max(-h[si], h[so] - 1.0, -radius)
        t_hi = min(1.0 - h[si], h[so], radius)
        if t_hi - t_lo < 1e-12:
            continue
        d = np.zeros(n)
        d[si] = 1.0
        d[so] = -1.0
        T = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        n_g = 25
        ts_g = np.linspace(t_lo, t_hi, n_g)
        vals_g = np.array([float(np.max(S0 + t * T - t * t * C)) for t in ts_g])
        bi = int(np.argmin(vals_g))
        best_t = float(ts_g[bi])
        best_v = float(vals_g[bi])
        for ct in [t_lo, t_hi, 0.0]:
            cv = float(np.max(S0 + ct * T - ct * ct * C))
            if cv < best_v:
                best_v = cv
                best_t = ct
        h_try = h.copy()
        h_try[si] += best_t
        h_try[so] -= best_t
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def _pair_descent(h, deadline, seed=42, radius=0.15, stall_limit=120):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rad = radius
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > stall_limit or rad < 1e-6:
            break
        h_try, c5_try = _pair_pass(h, n_pairs=40, radius=rad, seed=seed + it)
        if c5_try < cur_c5 - 1e-13:
            h = _project(h_try, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.02, 0.25)
        else:
            stall += 1
            rad *= 0.82
            if stall % 40 == 0 and rad < 0.02:
                rad = 0.10
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=50000, T_init=0.004, seed=777):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-13:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-13:
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5
