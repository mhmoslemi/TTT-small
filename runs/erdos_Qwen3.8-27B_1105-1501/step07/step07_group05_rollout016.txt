def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, radius=1.0):
    """Exact line search along direction d from h. Returns (best_alpha, best_c5)."""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find valid alpha range
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Candidate alphas: grid + critical points
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    
    # Critical points of each quadratic
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Uniform grid
    ng = min(48, max(24, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    
    # Evaluate
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement around best
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    # Verify with exact computation
    h_new = h + best_a * d
    h_new = np.clip(h_new, 0.0, 1.0)
    exact_val = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx)
    if exact_val < best_val - 1e-12:
        best_val = exact_val
    
    return best_a, best_val

def _lp_direction(h, top_k=20, r_max=1.0):
    """Find direction via LP that maximizes minimum first-order decrease of top constraints."""
    n = len(h)
    S0 = _raw_corr(h)
    dx = 2.0 / n
    
    # Select top K active lags
    top_idx = np.argsort(-S0)[:top_k]
    
    # Build constraint matrix A where A @ s gives T_s for active lags
    # T_s[t] for lag xi = t - (n-1)
    A = np.zeros((top_k, n))
    for i, t in enumerate(top_idx):
        xi = t - (n - 1)
        s_abs = abs(xi)
        if s_abs == 0:
            A[i, :] = 1.0 - 2.0 * h
        else:
            # c[j] = (1-h[j+s_abs]) * [j <= n-1-s_abs] - h[j-s_abs] * [j >= s_abs]
            j = np.arange(n)
            mask1 = j <= (n - 1 - s_abs)
            mask2 = j >= s_abs
            A[i, :] = np.where(mask1, 1.0 - h[np.clip(j + s_abs, 0, n-1)], 0.0) - \
                       np.where(mask2, h[np.clip(j - s_abs, 0, n-1)], 0.0)
    
    # LP: minimize z s.t. A @ s <= z, sum(s) = 0, -h <= s <= 1-h
    # Variables: s[0..n-1], z
    # Using scipy linprog
    try:
        from scipy.optimize import linprog
        # c = [0, ..., 0, 1] (minimize z)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A @ s - z <= 0  =>  A, -1] @ [s, z] <= 0
        A_ub = np.hstack([A, -np.ones((top_k, 1))])
        b_ub = np.zeros(top_k)
        
        # Equality: sum(s) = 0 => [1, ..., 1, 0] @ [s, z] = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds: -h[i] <= s[i] <= 1-h[i], -inf <= z <= inf
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
        bounds.append((None, None))
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            s_opt = res.x[:n]
            s_opt -= np.mean(s_opt)  # ensure zero mean
            return s_opt
    except:
        pass
    
    return None

def _project(v, target):

def _build_coeff(h, xi):
    n = len(h)
    s_abs = abs(xi)
    if s_abs == 0:
        return 1.0 - 2.0 * h
    j = np.arange(n)

        m1 = (j >= xi) & (j < n)
        m2 = (j >= 0) & (j <= n - 1 - xi)
        c[m1] = 1.0 - h[j[m1] - xi]

    else:

        m2 = (j >= s) & (j < n)

def _lp_direction(h, top_k=20):

    try:

        c_obj = np.zeros(n + 1)

        b_ub = np.zeros(top_k)
        A_eq = np.zeros((1, n + 1))

            s_opt -= np.mean(s_opt)

def _softmax_grad(h, beta=30.0):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -100.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue

            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])

                g[s:] += wl * (1.0 - h[:m])

    return -g

def _sinusoidal_init(n, f, phase, amp):
    x = np.arange(n, dtype=float) / n
    v = 0.5 + amp * np.sin(2 * np.pi * f * x + phase)
    return _project(np.clip(v, 0, 1), n / 2.0)

def _binary_mirror(h):
    n = len(h)
    hb = _to_bin(h, n)
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization from parent (15%)
    d1 = t0 + budget_s * 0.15
    h1, c1 = _local_opt_lp(h_init, d1, r0=0.20, stall_lim=40, nd=6)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair swap refinement (8%)
    d2 = t0 + budget_s * 0.23
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=25000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary + SA exploration (10%)
    d3 = t0 + budget_s * 0.33
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d3, mx=12000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _local_opt_lp(hp, sd, r0=0.10, stall_lim=30, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Alternative starting points (15%)
    d4 = t0 + budget_s * 0.48
    nc = len(best_h)
    alts = []
    alts.append(_to_bin(best_h, nc))
    alts.append(_sinusoidal_init(nc, 2.0, 0.0, 0.3))
    alts.append(_sinusoidal_init(nc, 3.0, np.pi / 4, 0.35))
    alts.append(_sinusoidal_init(nc, 5.0, 1.0, 0.25))
    alts.append(_sinusoidal_init(nc, 1.0, 0.5, 0.4))
    xc = np.arange(nc, dtype=float)
    v_blk = np.zeros(nc)
    for bi in range(nc // 12):
        blk = bi * 12
        if bi % 2 == 0:
            v_blk[blk:blk+12] = 1.0
    alts.append(_project(v_blk, nc / 2.0))
    for alt in alts:
        if time.time() > d4:
            break
        alt = _project(alt.copy(), nc / 2.0)
        ca = _exact_c5(alt)
        if ca < best_c + 0.02:
            sd = min(d4, time.time() + budget_s * 0.015)
            ha, ca2 = _local_opt_lp(alt, sd, r0=0.15, stall_lim=30, nd=5)
            if ca2 < best_c:
                best_c = ca2
                best_h = ha.copy()

    # Phase 5: Multi-resolution (17%)
    d5 = t0 + budget_s * 0.65
    nc = len(best_h)
    offsets = [-10, -6, -4, 4, 6, 10, -14, 14, -8, 8, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.014)
        hr, cr = _local_opt_lp(hr, sd, r0=0.12, stall_lim=30, nd=5)
        if cr < best_c + 0.005 and time.time() < d5 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.003 and time.time() < d5 - 2:
            sd3 = min(d5, time.time() + budget_s * 0.008)
            hr, cr = _local_opt_lp(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive LP polishing (20%)
    d6 = t0 + budget_s * 0.85
    nc = len(best_h)
    for r_init in [0.15, 0.08, 0.04, 0.02]:
        if time.time() > d6 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.05)
        h5, c5 = _local_opt_lp(h5, sd, r0=r_init, stall_lim=50, nd=6)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d6 - 5:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.025)
            h5, c5 = _pair_opt(h5, sd, max_it=12000)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()

    # Phase 7: Basin hopping (12%)
    d7 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d7 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_opt_lp(hp, sd, r0=0.06, stall_lim=15, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.08)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt_lp(hf, deadline, r0=0.03, stall_lim=25, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
