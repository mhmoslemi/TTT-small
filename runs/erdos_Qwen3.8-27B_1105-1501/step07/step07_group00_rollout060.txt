```python
def _swap_delta(h, i, j, n):
    """Compute the change in the correlation vector when swapping h[i]:0->1 and h[j]:1->0.
    Returns a vector of length 2n-1 (same as np.correlate output)."""
    d = np.arange(n-1, 0, -1)  # positive distances: n-1, n-2, ..., 1
    # For positive lag d:
    # delta(d) = (1-h[i-d])_clipped - h[i+d]_clipped - (1-h[j-d])_clipped + h[j+d]_clipped + (1 if j-i==d else 0)
    
    # Vectorized:
    # (1-h[i-d]): valid when i>=d, i.e., d<=i
    term1 = np.zeros(n-1)
    valid1 = d <= i
    term1[valid1] = 1 - h[i - d[valid1]]
    
    # h[i+d]: valid when i+d < n, i.e., d < n-i
    term2 = np.zeros(n-1)
    valid2 = (i + d) < n
    term2[valid2] = h[i + d[valid2]]
    
    # (1-h[j-d]): valid when j>=d
    term3 = np.zeros(n-1)
    valid3 = d <= j
    term3[valid3] = 1 - h[j - d[valid3]]
    
    # h[j+d]: valid when j+d < n
    term4 = np.zeros(n-1)
    valid4 = (j + d) < n
    term4[valid4] = h[j + d[valid4]]
    
    # interaction
    interact = (d == (j - i)).astype(float)
    
    delta_pos = term1 - term2 - term3 + term4 + interact
    delta_neg = delta_pos  # by symmetry (same formula with |d|)
    
    # Full correlation vector: indices 0 to 2n-2
    # Lag -(n-1) is at index 0, lag 0 is at index n-1, lag (n-1) is at index 2n-2
    # So positive lag d is at index n-1+d, negative lag -d is at index n-1-d
    full_delta = np.zeros(2*n-1)
    full_delta[n:] = delta_pos  # indices n to 2n-2 correspond to lags 1 to n-1
    full_delta[:n-1] = delta_neg[::-1]  # indices 0 to n-2 correspond to lags -(n-1) to -1
    
    return full_delta
```
