def _smooth_max_objective(h, tau):
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    # Shift to avoid overflow
    max_corr = np.max(corr)
    exp_vals = np.exp(tau * (corr - max_corr))
    return (1.0/tau) * (np.log(np.sum(exp_vals)) + tau * max_corr)

def _smooth_max_gradient(h, tau):
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(tau * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)

        # ∂c_k/∂h_i = (1 - h_{i-k}) * [0 ≤ i-k ≤ n-1] - h_{i+k} * [0 ≤ i+k ≤ n-1]

def _project_exact(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} exactly."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        bt, bv = _line_search_quad(S0, T, C, dx, 0.0, t_hi)
        f0 = float(np.max(S0)) * dx
        if bv < f0 - 1e-12 and bt > 1e-12:
            h = h + bt * p
            h = _project_exact(h, target)
            cur_c5 = _c5_exact(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 5.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amplitudes = rng.uniform(0.4, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_exact(h + perturb, n / 2.0)


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project_exact(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _c5_exact(h_init)

    # Phase 1: Smooth-max gradient descent (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _smooth_max_descent(h_init, d1, tau_init=2.0, tau_min=30.0, seed=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient with Nesterov momentum (15% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _subgradient_nesterov(best_h, d2, k_active=12, seed=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP polish (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _lp_polish(best_h, d3, radius=0.08, top_k=15, stall_limit=50)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Gradient-guided pair search (20% time)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _grad_guided_pair_search(best_h, d4, max_rounds=3000, top_k=12, seed=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Coordinate descent (15% time)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _coord_descent_exact(best_h, d5, max_sweeps=3000, seed=400)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary SA + polish (5% time)
    d6 = t0 + budget_s * 0.85
    h_bin = _to_binary(best_h)
    h_bin, c5_bin = _binary_sa(h_bin, d6, max_iter=20000, T_init=0.003, seed=500)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    h_pol = _project_exact(h_bin.copy(), n0 / 2.0)
    h_pol, c5_pol = _lp_polish(h_pol, min(d6, time.time() + 5), radius=0.05, top_k=10, stall_limit=20)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 7: Multi-resolution exploration (7% time)
    d7 = t0 + budget_s * 0.92
    n_curr = len(best_h)
    tried_ns = {n_curr}
    for dn in [-10, -6, -4, 4, 6, 10, -8, 8, 12, -12]:
        if time.time() > d7:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 250 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = _coord_descent_exact(h_r, sub_dl, max_sweeps=400, seed=700 + dn)
        if time.time() < d7 - 2:
            h_r, c5_r2 = _lp_polish(h_r, min(d7 - 1, time.time() + budget_s * 0.004),
                                     radius=0.05, top_k=10, stall_limit=15)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Final intensive polish (8% time)
    d8 = deadline - 2
    n_final = len(best_h)
    # Coordinate descent round 1
    sub_dl = min(d8, time.time() + budget_s * 0.025)
    h8, c5_8 = _coord_descent_exact(best_h, sub_dl, max_sweeps=2000, seed=800)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    # LP polish
    sub_dl = min(d8, time.time() + budget_s * 0.02)
    h8, c5_8 = _lp_polish(best_h, sub_dl, radius=0.04, top_k=20, stall_limit=40)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    # Basin hopping
    scale = 0.015
    n_restarts = 0
    while time.time() < d8 - 1 and n_restarts < 40:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d8 - 1, time.time() + budget_s * 0.005)
        h_t, c5_t = _coord_descent_exact(h_pert, sub_dl, max_sweeps=300, seed=2000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.1)
    # Final coordinate descent
    if time.time() < d8:
        h8, c5_8 = _coord_descent_exact(best_h, d8, max_sweeps=2000, seed=9999)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final projection and evaluation
    best_h = _project_exact(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    return best_h, final_c5, len(best_h)
