import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def _anti_to_full(x):
    m = len(x)
    h = np.empty(2*m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h):
    n = len(h)
    m = n // 2
    # x[i] = (h[i] + (1 - h[n-1-i])) / 2 = (h[i] + 1 - h[n-1-i]) / 2
    x = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return x

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_c5(x, beta):
    h = _anti_to_full(x)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Log-sum-exp smoothing
    s = logsumexp(beta * corr) / beta
    return float(s * 2.0 / n)

def _smooth_c5_grad(x, beta):
    h = _anti_to_full(x)
    n = len(h)
    m = n // 2
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = beta * (corr - corr.max())
    w = np.exp(z)
    wsum = w.sum()
    w = w / wsum
    # Full gradient G
    G = np.zeros(n)
    for r in range(len(corr)):
        if w[r] < 1e-10:
            continue
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            G[tau:] += w[r] * (1.0 - h[:mm])
            G[:mm] -= w[r] * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            G[:mm] += w[r] * (1.0 - h[d:])
            G[d:] -= w[r] * h[:mm]
    # Anti-palindromic gradient
    grad = G[:m] - G[m:][::-1]
    return grad * 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_full(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    eps = 1e-14
    pos = d > eps
    neg = d < -eps
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _c5_exact(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Golden section + boundary checks
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    # Check vertex of quadratic for each active shift
    top_idx = np.argsort(S0)[-10:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    return opt_t, opt_v

def _optimize_anti(x, deadline, beta=20.0, max_iter=200):
    """Optimize anti-palindromic representation using L-BFGS-B"""
    m = len(x)
    x = np.clip(x, 0.0, 1.0).copy()
    
    def obj_and_grad(xk):
        val = _smooth_c5(xk, beta)
        g = _smooth_c5_grad(xk, beta)
        return val, g
    
    bounds = [(0.0, 1.0)] * m
    x0 = x.copy()
    
    try:
        res = minimize(obj_and_grad, x0, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-8})
        x_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x_opt = x0
    
    h_opt = _anti_to_full(x_opt)
    c5_opt = _c5_exact(h_opt)
    
    # Compare with initial
    h_init = _anti_to_full(x0)
    c5_init = _c5_exact(h_init)
    
    if c5_opt <= c5_init:
        return x_opt, c5_opt
    return x0, c5_init

def _grad_full(h, raw, beta):
    n = len(h)
    z = beta * (raw - raw.max())
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = w.sum()
    if wsum < 1e-30:
        # Fall back to single max
        r = int(np.argmax(raw))
        tau = r - (n - 1)
        g = np.zeros(n)
        if tau >= 0:
            m = n - tau
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau+m]
        else:
            d = -tau
            m = n - d
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
        g -= np.mean(g)
        return g
    w /= wsum
    active = w > 1e-8
    g = np.zeros(n)
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            g[tau:] += wr * (1.0 - h[:m])
            g[:m] -= wr * h[tau:tau+m]
        else:
            d = -tau
            m = n - d
            g[:m] += wr * (1.0 - h[d:])
            g[d:] -= wr * h[:m]
    g -= np.mean(g)
    return g

def _opt_full(h, deadline, beta_lo=5.0, beta_hi=50.0, radius=0.15, max_stall=60):
    """Full-space smooth optimization with exact sum constraint"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline - 1.0:
        if stall > max_stall:
            break
        total_iters += 1
        frac = min(1.0, total_iters / 50.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        g = _grad_full(h, raw, beta)
        
        # Direction: negative gradient
        d = -g
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            stall += 1
            continue
        d = d / dn
        
        alpha, val = _line_search_full(h, d, radius)
        
        improved = False
        if val < cur_c5 - 1e-14 and abs(alpha) > 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _c5_exact(h_new)
            if nc < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                improved = True
        
        if not improved:
            # Try random perturbations
            for attempt in range(3):
                rd = rng.standard_normal(n)
                rd -= np.mean(rd)
                rn = np.linalg.norm(rd)
                if rn < 1e-12:
                    continue
                rd = rd / rn
                alpha2, val2 = _line_search_full(h, rd, radius * 0.5)
                if val2 < cur_c5 - 1e-14 and abs(alpha2) > 1e-14:
                    h_new = h + alpha2 * rd
                    h_new = _project(h_new, target)
                    nc = _c5_exact(h_new)
                    if nc < cur_c5 - 1e-15:
                        h = h_new
                        cur_c5 = nc
                        if nc < best_c5:
                            best_c5 = nc
                            best_h = h.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            radius = max(0.03, radius * 0.85)
    
    return best_h, best_c5

def _binary_exchange(h, deadline, max_iter=20000, T_init=0.003):
    """Binary swap optimization with simulated annealing"""
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = _c5_exact(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    
    # Precompute shift contributions for fast evaluation
    for iteration in range(max_iter):
        if time.time() > deadline - 1.0:
            break
        if stall > 300:
            break
        
        # Find a 1 and a 0 to swap
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        
        # Quick estimate: compute C5 after swap
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _c5_exact(cur)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best_h, best_c5

def _pair_sweep(h, deadline):
    """Try all pairs of swaps that improve"""
    n = len(h)
    cur = h.copy()
    cur_c5 = _c5_exact(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for oi in ones:
        if time.time() > deadline - 1.0:
            break
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _c5_exact(cur)
            if c_try < cur_c5 - 1e-15:
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
    
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0

                if nc < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = nc
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                    stall = 0
                    improved = True
        
        if not improved:
            # Try a few random directions
            for _ in range(4):
                rd = rng.standard_normal(n)
                rd -= np.mean(rd)
                rn = np.linalg.norm(rd)
                if rn < 1e-12:
                    continue
                rd /= rn
                alpha2, val2 = _line_search_full(h, rd, radius * 0.4)
                if val2 < cur_c5 - 1e-14 and abs(alpha2) > 1e-14:
                    h_new = h + alpha2 * rd
                    h_new = _project(h_new, target)
                    nc = _c5_exact(h_new)
                    if nc < cur_c5 - 1e-15:
                        h = h_new
                        cur_c5 = nc
                        if nc < best_c5:
                            best_c5 = nc
                            best_h = h.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            radius = max(0.03, radius * 0.88)
    
    return best_h, best_c5

def _pair_continuous(h, deadline, n_iter=3000, radius=0.18, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline - 1.0:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search_full(h, d, radius)
        if c_try < cur_c5 - 1e-13 and abs(alpha) > 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _c5_exact(h_new)
            if nc < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _c5_exact(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline - 1.0:
            break
        if stall > 250:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _c5_exact(cur)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline):
    n = len(h)
    cur = h.copy()
    cur_c5 = _c5_exact(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for oi in ones:
        if time.time() > deadline - 1.0:
            break
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _c5_exact(cur)
            if c_try < cur_c5 - 1e-15:
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.15 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)
    
    h0 = _project(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
    best_h = h0.copy()
    best_c5 = _c5_exact(h0)
    n0 = len(h0)
    
    # Phase 1: Anti-palindromic L-BFGS optimization (25%)
    d1 = t0 + budget_s * 0.25
    if n0 % 2 == 0:
        x0 = _full_to_anti(h0)
        x_opt, c5_opt = _optimize_anti(x0, d1, beta_seq=[8.0, 20.0, 40.0, 80.0, 150.0], max_iter=200)
        h_opt = _anti_to_full(x_opt)
        c5_check = _c5_exact(h_opt)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_opt.copy()
    
    # Phase 2: Full-space smooth optimization (15%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _opt_full(best_h.copy(), d2, beta_lo=4.0, beta_hi=35.0, radius=0.18, max_stall=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Anti-palindromic refinement with high beta (15%)
    d3 = t0 + budget_s * 0.55
    if len(best_h) % 2 == 0:
        x3 = _full_to_anti(best_h)
        x_opt, c5_opt = _optimize_anti(x3, d3, beta_seq=[50.0, 100.0, 200.0], max_iter=300)
        h_opt = _anti_to_full(x_opt)
        c5_check = _c5_exact(h_opt)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_opt.copy()
    
    # Phase 4: Continuous pair moves (10%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _pair_continuous(best_h.copy(), d4, n_iter=5000, radius=0.15, seed=777)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Binary optimization (15%)
    d5 = t0 + budget_s * 0.80
    h_b = _to_binary(best_h)
    h_b, c5_b = _binary_sa(h_b, d5, max_iter=40000, T_init=0.005)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d5 - 2:
        h_ps, c5_ps = _pair_sweep(best_h, d5)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
    
    # Phase 6: Multi-resolution (10%)
    d6 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    for dn in [-16, -12, -8, -4, 4, 8, 12, 16, -20, 20]:
        if time.time() > d6 - 3:
            break
        n_try = n_curr + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6 - 2, time.time() + budget_s * 0.01)
        h_r, c5_r = _opt_full(h_r, sub_dl, beta_lo=8.0, beta_hi=30.0, radius=0.12, max_stall=25)
        if n_try % 2 == 0 and time.time() < d6 - 1:
            x_r = _full_to_anti(h_r)
            x_o, c5_o = _optimize_anti(x_r, d6, beta_seq=[30.0, 80.0], max_iter=80)
            h_o = _anti_to_full(x_o)
            c5_o = _c5_exact(h_o)
            if c5_o < c5_r:
                c5_r = c5_o
                h_r = h_o
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 7: Basin hopping + final polish (10%)
    d7 = deadline
    scale = 0.012
    n_restarts = 0
    while time.time() < d7 - 12 and n_restarts < 50:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d7 - 8, time.time() + budget_s * 0.007)
        if len(h_p) % 2 == 0:
            x_p = _full_to_anti(h_p)
            x_t, c5_t = _optimize_anti(x_p, sub_dl, beta_seq=[40.0, 100.0], max_iter=60)
            h_t = _anti_to_full(x_t)
            c5_t = _c5_exact(h_t)
        else:
            h_t, c5_t = _opt_full(h_p, sub_dl, beta_lo=15.0, beta_hi=60.0, radius=0.08, max_stall=20)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.04, scale * 1.12)
    
    # Final polish
    if deadline - time.time() > 4:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        if len(h_f) % 2 == 0:
            x_f = _full_to_anti(h_f)
            x_t, _ = _optimize_anti(x_f, deadline, beta_seq=[80.0, 200.0], max_iter=200)
            h_f = _anti_to_full(x_t)
        c5_f = _c5_exact(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
