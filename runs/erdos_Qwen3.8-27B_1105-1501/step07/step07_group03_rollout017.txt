def _solve_lp(h, active_lags, target, radius):
    """Solve the trust-region LP. Returns solution x or None."""
    n = len(h)
    K = len(active_lags)
    if K == 0:
        return None
    
    # Build LP: minimize t
    # s.t. c_r @ x - t <= -const_r for each active lag r
    #      sum(x) = target
    #      h[i]-radius <= x[i] <= h[i]+radius
    #      0 <= x[i] <= 1
    
    # Variables: [x_0, ..., x_{n-1}, t], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, r in enumerate(active_lags):
        c_r, const_r = _lag_coeffs(h, r)
        A_ub[i, :n] = c_r
        A_ub[i, n] = -1.0
        b_ub[i] = -const_r

        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))

def _exact_c5(h):
    n = len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _lag_coeffs(h, r, n):
    d = r - (n - 1)
    c = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return c, 0.0
        idx1 = np.arange(m)
        idx2 = idx1 + d
        const = float(np.sum(h[idx1] * (1.0 - h[idx2])))
        c[idx1] += (1.0 - h[idx2])
        c[idx2] -= h[idx1]
    else:

        if m <= 0:

        const = float(np.sum(h[idx1] * (1.0 - h[idx2])))

        c[idx2] -= h[idx1]

    if K == 0:

    c_obj = np.zeros(n + 1)

        cr, const_r = _lag_coeffs(h, r, n)
        A_ub[i, :n] = cr
        A_ub[i, n] = -1.0
        b_ub[i] = -const_r
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

    n = len(h)

    w = vp / (float(np.sum(vp)) + 1e-30)
    g = np.zeros(n)

    nm = np.linalg.norm(g)

                Tvec = np.correlate(d, 1.0 - cur, mode='full') - np.correlate(cur, d, mode='full')
                raw_new = np.correlate(cur, 1.0 - cur, mode='full') + Tvec
                c_try = float(np.max(raw_new)) * dx
                if c_try < cur_c5 - 1e-15:
                    cur = cur.copy()
                    cur[oi] = 0.0
                    cur[oz] = 1.0
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: LP-based optimization (40%)
    d1 = t0 + budget_s * 0.40
    h1, c5_1 = _lp_optimize(h_init, d1, K=25, radius_init=0.12, stall_limit=200, beta=15.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary refinement (10%)
    d2 = t0 + budget_s * 0.50
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_b = _binary_flip_optimize(h_bin, sub_dl, max_iter=25000)
    sub_dl = min(d2, time.time() + budget_s * 0.02)
    h_bin, c5_b = _pair_sweep_bin(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_pol, c5_p = _lp_optimize(h_pol, sub_dl, K=20, radius_init=0.10, stall_limit=40, beta=12.0)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()
    
    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in range(-12, 13, 2):
        if time.time() > d3 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, K=20, radius_init=0.10, stall_limit=50, beta=14.0)
        if c5_r < best_h is not None and c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.005)
            h_b, c5_b2 = _binary_flip_optimize(h_b, sub_dl2, max_iter=8000)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 4: Intensive LP refinement (20%)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    for trial in range(6):
        if time.time() > d4 - 2:
            break
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _lp_optimize(h4, sub_dl, K=30, radius_init=0.08, stall_limit=80, beta=25.0)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, K=20, radius_init=0.08, stall_limit=30, beta=18.0)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.06, scale * 1.10)
    
    # Final polish
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline, K=35, radius_init=0.05, stall_limit=50, beta=30.0)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
