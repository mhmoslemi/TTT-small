def _line_search(S, A, B, t_lo, t_hi, n_grid=50):
    """Minimize max_k(S_k + t*A_k - t^2*B_k) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return 0.0
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, n_grid)
    # f(t) = max_k(S_k + t*A_k - t^2*B_k)
    # Shape: (n_grid, L)
    vals = S[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :]
    obj = np.max(vals, axis=1)
    best_idx = int(np.argmin(obj))
    
    # Golden section in neighborhood of best
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx])
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * A - c1*c1 * B))
    f2 = float(np.max(S + c2 * A - c2*c2 * B))
    
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * A - c1*c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * A - c2*c2 * B))
    
    t_opt = (lo + hi) / 2.0
    return t_opt

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_grads(h, row_indices):
    n = len(h)
    K = len(row_indices)
    G = np.zeros((K + 1, n))
    for i, r in enumerate(row_indices):
        ell = r - (n - 1)
        if ell >= 0:
            m = n - ell
            G[i, ell:ell + m] += (1.0 - h[:m])
            G[i, :m] -= h[ell:]
        else:
            p = -ell
            m = n - p
            G[i, :m] += (1.0 - h[p:])
            G[i, p:] -= h[:m]
    G[K, :] = 1.0
    return G

def _compute_AB(h, d):
    A = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    return A, B

def _feasible_interval(h, d):
    t_lo = -1e10
    t_hi = 1e10
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    return t_lo, t_hi

def _line_search(S, A, B, t_lo, t_hi, n_grid=60):
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S))
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = S[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :]
    obj = np.max(vals, axis=1)
    best_idx = int(np.argmin(obj))
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    if hi - lo < 1e-13:
        return float(ts[best_idx]), float(obj[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * A - c1 * c1 * B))
    f2 = float(np.max(S + c2 * A - c2 * c2 * B))
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * A - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * A - c2 * c2 * B))
    t_opt = (lo + hi) / 2.0
    f_opt = float(np.max(S + t_opt * A - t_opt * t_opt * B))
    # Also check endpoints and grid best
    candidates = [(t_opt, f_opt)]
    if t_lo > -1e9:
        candidates.append((t_lo, float(np.max(S + t_lo * A - t_lo * t_lo * B))))
    if t_hi < 1e9:
        candidates.append((t_hi, float(np.max(S + t_hi * A - t_hi * t_hi * B))))
    candidates.append((float(ts[best_idx]), float(obj[best_idx])))
    candidates.append((0.0, float(np.max(S))))
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _make_dense_dir(h, S, row_indices, target, use_weight=False):
    G = _row_grads(h, row_indices)
    K = len(row_indices)
    active_vals = np.array([S[r] for r in row_indices])
    b = np.zeros(K + 1)
    b[:K] = target - active_vals
    b[K] = 0.0
    if use_weight:
        W_inv = 1.0 / (h * (1.0 - h) + 1e-5)
        GW = G * W_inv[None, :]
        M = GW @ G.T
    else:
        M = G @ G.T
    M += 1e-10 * np.eye(K + 1)
    try:
        lam = np.linalg.solve(M, b)
    except Exception:
        try:
            lam = np.linalg.lstsq(M, b, rcond=None)[0]
        except Exception:
            return None
    if use_weight:
        d = W_inv * (G.T @ lam)
    else:
        d = G.T @ lam
    if not np.all(np.isfinite(d)):
        return None
    d -= np.mean(d)
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-12:
        return None
    d /= max_abs
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h.copy(), n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
