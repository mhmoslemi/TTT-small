import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : 0<=x<=1, sum(x)=target}
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, tau):
    # Gradient of corr(tau) w.r.t. h
    n = len(h)
    g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:tau+m] += (1.0 - h[:m])
            g[:m] -= h[tau:tau+m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:d+m] -= h[:m]
    return g

def _smooth_grad(h, beta):
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    mx = raw.max()
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = w.sum()
    if wsum < 1e-30:
        return _grad_lag(h, int(np.argmax(raw)) - (n-1))
    w /= wsum
    active = w > 1e-8
    g = np.zeros(n)
    for r in np.where(active)[0]:
        tau = r - (n - 1)
        g += w[r] * _grad_lag(h, tau)
    g -= g.mean()  # project to zero-sum
    return g

def _waterfill_dir(h, q):
    # Find d = x - h where x = argmax q'x s.t. 0<=x<=1, sum(x)=n/2
    n = len(h)
    target = n // 2
    idx = np.argsort(q)[::-1]  # descending
    x = np.zeros(n)
    x[idx[:target]] = 1.0
    d = x - h
    # d already sums to 0 (both x and h sum to n/2)
    return d

def _line_search_exact(h, d):
    # Exact line search: minimize f(a) = dx * max(S0 + a*T - a^2*Q)
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible interval for a
    a_lo, a_hi = -1e15, 1e15
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, np.min(-h[pos] / d[pos]))
        a_hi = min(a_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        a_lo = max(a_lo, np.min((1.0 - h[neg]) / d[neg]))
        a_hi = min(a_hi, np.min(h[neg] / (-d[neg])))
    
    if a_hi <= a_lo + 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Grid search
    ng = 50
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.array([f(t) for t in gt])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi-2)]
    hi = gt[min(ng-1, bi+2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    opt_a = (lo + hi) / 2.0
    opt_v = f(opt_a)
    
    # Check critical points
    top_idx = np.argsort(S0)[-10:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_a, opt_v = av, fv
    
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_a, opt_v = t, ft
    
    return opt_a, opt_v

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, tau):
    n = len(h)
    g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:tau + m] += 1.0 - h[:m]
            g[:m] -= h[tau:tau + m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += 1.0 - h[d:]
            g[d:d + m] -= h[:m]
    return g

def _waterfill_dir(h, q):
    n = len(h)
    target = n // 2
    idx = np.argsort(-q)
    x = np.zeros(n)
    x[idx[:target]] = 1.0
    d = x - h
    dn = np.linalg.norm(d)
    if dn > 1e-14:
        return d / dn
    return d

def _line_search(h, d, S0=None):
    n = len(h)
    dx = 2.0 / n
    if S0 is None:
        u = 1.0 - h
        S0 = np.correlate(h, u, mode='full')
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo, a_hi = -1e15, 1e15
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, float(np.max(S0) * dx)
    a_lo = max(a_lo, -5.0)
    a_hi = min(a_hi, 5.0)
    cands = [a_lo, a_hi, 0.0]
    nz = Q > 1e-15
    if np.any(nz):
        ac = T[nz] / (2.0 * Q[nz])
        for v in ac:
            if a_lo - 1e-12 <= v <= a_hi + 1e-12:
                cands.append(float(np.clip(v, a_lo, a_hi)))
    grid = np.linspace(a_lo, a_hi, 40)
    cands.extend(grid.tolist())
    best_a, best_v = 0.0, float('inf')
    for a in cands:
        fv = float(np.max(S0 + a * T - a * a * Q)) * dx
        if fv < best_v:
            best_v, best_a = fv, a
    return best_a, best_v

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_dirs(h, raw, beta, rng, n):
    dirs = []
    u = 1.0 - h
    mx = float(raw.max())
    v_beta = np.clip(beta * (raw - mx), -50.0, 0.0)
    w = np.exp(v_beta)
    wsum = float(w.sum())
    g = None
    if wsum > 1e-30:
        w = w / wsum
        g = np.zeros(n)
        active = w > 1e-8
        if np.any(active):
            for r in np.where(active)[0]:
                g += w[r] * _grad_lag(h, r - (n - 1))
        g -= g.mean()
    if g is not None and np.linalg.norm(g) > 1e-12:
        d = -g
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
        wd = _waterfill_dir(h, -g)
        if np.linalg.norm(wd) > 1e-12:
            dirs.append(wd)
    top_k = np.argsort(raw)[-3:]
    for k in range(1, 4):
        gk = np.zeros(n)
        for r in top_k[:k]:
            gk += _grad_lag(h, r - (n - 1))
        gk -= gk.mean()
        gkn = np.linalg.norm(gk)
        if gkn > 1e-12:
            dirs.append(-gk / gkn)
    if rng is not None:
        d_r = rng.normal(0, 1, n)
        d_r -= d_r.mean()
        drn = np.linalg.norm(d_r)
        if drn > 1e-12:
            dirs.append(d_r / drn)
        d_m = np.zeros(n)
        i1 = int(rng.integers(0, n))
        i2 = int(rng.integers(0, n))
        if i1 != i2:
            d_m[i1] = 1.0
            d_m[i2] = -1.0
            d_m -= d_m.mean()
            dm_n = np.linalg.norm(d_m)
            if dm_n > 1e-12:
                dirs.append(d_m / dm_n)
    return dirs

def _refine_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    iters = 0
    rng = np.random.default_rng(54321)
    while time.time() < deadline and stall < max_stall:
        iters += 1
        frac = min(1.0, iters / 80.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        dirs = _make_dirs(h, raw, beta, rng, n)
        S0 = raw.copy()
        best_alpha, best_val, best_d = 0.0, cur_c5, None
        for d in dirs:
            if time.time() > deadline:
                break
            a, val = _line_search(h, d, S0)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, a, d
        if best_d is not None and abs(best_alpha) > 1e-13:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5, best_h = nc, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5
