def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt

def _lag_grad(h, k):
    """Gradient of the k-th correlation output with respect to h."""
    n = len(h)
    g = np.zeros(n)
    # k is index in 'full' correlation output (0 to 2n-2)
    # lag = k - (n-1)
    lag = k - (n - 1)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:lag+m])
            g[lag:lag+m] -= h[:m]
    else:
        s = -lag
        m = n - s
        if m > 0:
            g[s:s+m] += (1.0 - h[:m])
            g[:m] -= h[s:s+m]
    return g

def _active_subgrad(h, top_k=5, weight_decays=None):
    """Compute subgradient from top-K active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:top_k]
    if weight_decays is not None:
        weights = np.array(weight_decays[:top_k])
    else:
        # Exponential decay
        weights = np.exp(-0.3 * np.arange(top_k))
        weights /= np.sum(weights)
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        g += weights[i] * _lag_grad(h, int(idx))
    g -= np.mean(g)
    return g, top_idx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _many_block_dir(h, block_sizes):
    """Generate directions by averaging gradient over blocks."""
    n = len(h)
    _, top_idx = _active_subgrad(h, top_k=5)
    g = np.zeros(n)
    for idx in top_idx:
        g += _lag_grad(h, int(idx))

    g -= np.mean(g)

    for bs in block_sizes:
        if n % bs != 0:

def _lag_grad(h, k):
    n = len(h)

    else:

        m = n - s
        if m > 0:
            g[s:s + m] += (1.0 - h[:m])
            g[:m] -= h[s:s + m]
    return g

def _subgrad_dir(h, K, decay=0.3):

    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(9999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _score_only(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.997
        else:
            stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _construct_binary(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = _construct_binary(180, seed)

    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _score_only(h_init)

    # Phase 1: Local opt on parent (14%)
    d1 = t0 + budget_s * 0.14
    h1, c1 = _local_opt(h_init, d1, K_init=3, K_max=16, stall_limit=35)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Binary SA + fractional (16%)
    d2 = t0 + budget_s * 0.30
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _binary_sa(hbin, d2, max_iter=18000, T_init=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d2 - 1:
        hd = _project(hbin.copy(), nb / 2.0)
        sd = min(d2, time.time() + budget_s * 0.03)
        hd, cd = _local_opt(hd, sd, K_init=4, K_max=12, stall_limit=25)
        if cd < best_c:
            best_c = cd
            best_h = hd.copy()

    # Phase 3: Multi-resolution (26%)
    d3 = t0 + budget_s * 0.56
    nc = len(best_h)
    offsets = [0, -4, 4, -8, 8, -6, 6, -12, 12, -10, 10, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d3, time.time() + budget_s * 0.014)
        hr, cr = _local_opt(hr, sd, K_init=4, K_max=12, stall_limit=22)
        if cr < best_c + 0.004 and time.time() < d3 - 3:
            hb = _to_bin(hr)
            sd2 = min(d3, time.time() + budget_s * 0.007)
            hb, cb2 = _binary_sa(hb, sd2, max_iter=5000, T_init=0.0015)
            if cb2 < cr:
                cr = cb2
                hr = hb
            if time.time() < d3 - 3:
                hr_f = _project(hr.copy(), n_try / 2.0)
                sd3 = min(d3, time.time() + budget_s * 0.008)
                hr_f, cr_f = _local_opt(hr_f, sd3, K_init=3, K_max=10, stall_limit=18)
                if cr_f < cr:
                    cr = cr_f
                    hr = hr_f
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 4: Intensive optimization (30%)
    d4 = t0 + budget_s * 0.86
    nc = len(best_h)
    for trial in range(8):
        if time.time() > d4 - 2:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        if trial % 3 == 0:
            sd = min(d4, time.time() + budget_s * 0.035)
            h5, c5 = _local_opt(h5, sd, K_init=3, K_max=18, stall_limit=45)
        elif trial % 3 == 1:
            hb5 = _to_bin(best_h)
            sd_b = min(d4, time.time() + budget_s * 0.02)
            hb5, cb5 = _binary_sa(hb5, sd_b, max_iter=10000, T_init=0.002)
            h5f = _project(hb5.copy(), nc / 2.0)
            sd = min(d4, time.time() + budget_s * 0.02)
            h5, c5 = _local_opt(h5f, sd, K_init=4, K_max=14, stall_limit=30)
        else:
            pert = rng.normal(0, 0.08, nc)
            pert -= np.mean(pert)
            hp = _project(h5 + pert, nc / 2.0)
            sd = min(d4, time.time() + budget_s * 0.025)
            h5, c5 = _local_opt(hp, sd, K_init=3, K_max=16, stall_limit=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Phase 5: Basin hopping + final polish (14%)
    d5 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d5 - 2 and n_re < 35:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * math.sin(2 * math.pi * f * x / nc + p)
        pert += 0.25 * scale * rng.normal(0, 1, nc)
        mxp = float(np.max(np.abs(pert)))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d5 - 2, time.time() + budget_s * 0.005)
        ht, ct = _local_opt(hp, sd, K_init=3, K_max=10, stall_limit=15)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.7)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 3:
        nf = len(best_h)
        hf = _project(best_h.copy(), nf / 2.0)
        hf, cf = _local_opt(hf, deadline, K_init=3, K_max=15, stall_limit=35)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _score_only(best_h)
    return best_h, final_c, len(best_h)
