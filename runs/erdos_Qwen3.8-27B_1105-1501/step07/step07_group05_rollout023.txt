- The constraint ||G^T α|| ≤ r is a quadratic constraint: α^T (G G^T) α ≤ r².

def _project(v, target):

    n = len(v)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:lag+m])
            g[lag:] -= h[:m]
    else:
        s = -lag
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s+m]
    g -= np.mean(g)
    return g

def _line_search(h, d, dx, radius):
    """Line search along direction d from h."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S) * dx)
    
    # Model: f(a) = max(S + a*T - a*a*Q) * dx
    # Find minimizer
    
    # Candidate points:
    cands = [a_lo, a_hi]
    width = a_hi - a_lo
    
    # Stationary points of each lag's parabola
    for k in range(len(S)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Grid
    ng = min(48, max(24, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    
    # Find best
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-12:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12: break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

def _lp_direction(h, n_top=10):
    """Find direction using active-set LP minimax."""
    n = len(h)
    nc = 2*n - 1
    raw = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top lags
    idx = np.argsort(-raw)[:n_top]
    grads = []
    raws = []
    for i in idx:
        lag = int(i) - (n - 1)
        g = _lag_grad(h, lag)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            grads.append(g)
            raws.append(raw[i])
    
    K = len(grads)
    if K == 0:
        return np.zeros(n)
    if K == 1:
        return -grads[0] / np.linalg.norm(grads[0])
    
    # Build M = G G^T (K x K), where G has grads as rows
    G = np.vstack(grads)  # K x n
    M = G @ G.T  # K x K
    c_top = np.array(raws)
    
    # LP: min t subject to c_top + M@alpha <= t
    # x = [alpha (K), t]
    # minimize: [0, ..., 0, 1]^T @ x
    # constraints: for each k: M[k,:] @ alpha - t <= -c_top[k]
    A = np.hstack([M, -np.ones((K, 1))])  # K x (K+1)
    b = -c_top
    c_obj = np.zeros(K + 1)
    c_obj[-1] = 1.0
    bounds = [(-1.0, 1.0)] * K + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            alpha = res.x[:K]
            d = G.T @ alpha
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-10:
                return d / nrm
    except:
        pass
    
    # Fallback: negative weighted gradient
    w = np.exp(-(c_top - c_top.max()) / 0.02)
    w /= w.sum()
    d = sum(wi * g for wi, g in zip(w, grads))
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-10:
        return -d / nrm  # negative gradient direction
    return np.zeros(n)

def _binary_best_flip(cur):
    """Find the single flip that most reduces max correlation."""
    n = len(cur)
    cur_raw = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = cur_raw.max()
    
    best_new_max = cur_max
    best_j = -1
    
    for j in range(n):
        # Compute change in all correlations from flipping j
        # Using the formula above
        delta = 1.0 - 2.0 * cur[j]  # +1 for 0→1, -1 for 1→0
        change = np.zeros(2*n - 1)
        
        for k in range(2*n - 1):
            l = k - (n - 1)
            if l >= 0:
                # First term: (1-h[j-l]) if j >= l
                if j >= l:
                    change[k] += delta * (1.0 - cur[j - l])
                # Second term: -h[j+l] if j+l < n
                if j + l < n:
                    change[k] -= delta * cur[j + l]
            else:
                s = -l
                # First term: (1-h[j+s]) if j+s < n
                if j + s < n:
                    change[k] += delta * (1.0 - cur[j + s])
                # Second term: -h[j-s] if j >= s
                if j >= s:
                    change[k] -= delta * cur[j - s]
        
        new_raw = cur_raw + change
        new_max = new_raw.max()
        if new_max < best_new_max - 1e-10:
            best_new_max = new_max
            best_j = j
    
    return best_j, best_new_max
