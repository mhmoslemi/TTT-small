    stall = 0
    for _ in range(n_tries):
        if time.time() > deadline or stall > 500:
            break
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur <= 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(len(ones), 2, replace=False)
        j1, j2 = rng.choice(len(zeros), 2, replace=False)
        i1, i2 = ones[i1], ones[i2]
        j1, j2 = zeros[j1], zeros[j2]
        if i1 == i2 or j1 == j2:
            continue
        h_cur[i1] = 0.0
        h_cur[i2] = 0.0
        h_cur[j1] = 1.0
        h_cur[j2] = 1.0
        c_try = _exact_c5(h_cur)
        if c_try < cur_c5 - 1e-12:
            cur_c5 = c_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
            stall = 0
        else:
            h_cur[i1] = 1.0
            h_cur[i2] = 1.0
            h_cur[j1] = 0.0
            h_cur[j2] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_continuous(h, deadline, n_iter=4000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for k in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    idx = rng.choice(n, min(n // 4, 20), replace=False)
    pert[idx] += amp * 0.5 * rng.normal(0, 1, len(idx))
    return _project(h + pert, n / 2.0)

def _binary_perturb(h_bin, n_flips, rng):
    h_new = h_bin.copy()
    n = len(h_bin)
    ones = np.where(h_new > 0.5)[0]
    zeros = np.where(h_new <= 0.5)[0]
    for _ in range(n_flips):
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        h_new[ones[oi]] = 0.0
        h_new[zeros[oz]] = 1.0
        ones = np.where(h_new > 0.5)[0]
        zeros = np.where(h_new <= 0.5)[0]
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous optimization (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _opt_smooth(h_init, d1, beta_lo=8.0, beta_hi=50.0, radius_init=0.15, max_stall=50)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary local search - the key phase (35%)
    d2 = t0 + budget_s * 0.43
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    c5_b = _exact_c5(h_b)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    h_b2, c5_b2 = _binary_local_search(h_b, d2, max_sweeps=80)
    if c5_b2 < best_c5:
        best_c5 = c5_b2
        best_h = h_b2.copy()
    if time.time() < d2 - 5:
        h_b3, c5_b3 = _binary_2swap(h_b2, min(d2, time.time() + budget_s * 0.05), n_tries=8000)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
        h_b4, c5_b4 = _binary_local_search(h_b3, min(d2, time.time() + budget_s * 0.04), max_sweeps=30)
        if c5_b4 < best_c5:
            best_c5 = c5_b4
            best_h = h_b4.copy()

    # Phase 3: Continuous refinement from best (15%)
    d3 = t0 + budget_s * 0.58
    n_c = len(best_h)
    h_c = _project(best_h.copy(), n_c / 2.0)
    h_c, c5_c = _opt_smooth(h_c, d3, beta_lo=15.0, beta_hi=80.0, radius_init=0.12, max_stall=60)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()
    if time.time() < d3 - 3:
        h_c2, c5_c2 = _pair_continuous(best_h, min(d3, time.time() + budget_s * 0.04), n_iter=5000, radius=0.15, seed=555)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2.copy()

    # Phase 4: Multi-n exploration (18%)
    d4 = t0 + budget_s * 0.76
    n_curr = len(best_h)
    n_candidates = []
    for dn in range(-40, 41, 8):
        nt = n_curr + dn
        if 80 <= nt <= 280 and nt % 2 == 0:
            n_candidates.append(nt)
    if not n_candidates:
        n_candidates = [n_curr]
    rng.shuffle(n_candidates)
    for nt in n_candidates:
        if time.time() > d4:
            break
        h_r = _resample(best_h, nt)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _opt_smooth(h_r, sub_dl, beta_lo=10.0, beta_hi=50.0, radius_init=0.12, max_stall=30)
        if time.time() < d4 - 3:
            h_rb = _to_binary(h_r)
            sub_dl2 = min(d4, time.time() + budget_s * 0.012)
            h_rb, c5_rb = _binary_local_search(h_rb, sub_dl2, max_sweeps=15)
            if c5_rb < c5_r:
                c5_r = c5_rb
                h_r = h_rb
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = nt

    # Phase 5: Basin hopping (17%)
    d5 = deadline
    scale_cont = 0.04
    scale_bin = 3
    n_restarts = 0
    while time.time() < d5 - 10 and n_restarts < 120:
        n_restarts += 1
        h_bin = _to_binary(best_h)
        if n_restarts % 3 == 0 and scale_bin > 1:
            h_p = _binary_perturb(h_bin, scale_bin, rng)
            c5_p = _exact_c5(h_p)
            sub_dl = min(d5 - 5, time.time() + budget_s * 0.006)
            h_t, c5_t = _binary_local_search(h_p, sub_dl, max_sweeps=8)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale_bin = max(1, scale_bin - 1)
            else:
                scale_bin = min(8, int(scale_bin * 1.3) + 1)
        else:
            h_p = _smooth_perturb(best_h, scale_cont, rng)
            sub_dl = min(d5 - 5, time.time() + budget_s * 0.007)
            h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=12.0, beta_hi=60.0, radius_init=0.10, max_stall=25)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale_cont = max(0.004, scale_cont * 0.85)
            else:
                scale_cont = min(0.10, scale_cont * 1.06)

    # Phase 6: Final intensive polish (remaining time)
    if deadline - time.time() > 15:
        n_f = len(best_h)
        h_f = _project(best_h.copy(), n_f / 2.0)
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h_f, c5_f = _opt_smooth(h_f, sub_dl, beta_lo=20.0, beta_hi=100.0, radius_init=0.06, max_stall=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if deadline - time.time() > 5:
            h_f2 = _project(best_h.copy(), n_f / 2.0)
            sub_dl2 = deadline
            h_f2, c5_f2 = _pair_continuous(h_f2, sub_dl2, n_iter=3000, radius=0.12, seed=777)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
