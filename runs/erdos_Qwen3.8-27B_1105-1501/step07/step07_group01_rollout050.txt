```python
def _smooth_max_grad(h, alpha):
    """Compute gradient of smooth-max of correlation values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_shifted = corr - corr.max()  # numerical stability
    w = np.exp(alpha * corr_shifted)
    w = w / w.sum()
    
    # Gradient: for each j, grad[j] = sum_k w[k] * (1 - 2*h[k-j]) * [k-j in [0,n-1]]
    # Equivalent: grad[j] = sum_s w[j+s] * (1-2*h[s]) for s in [0,n-1] where j+s in [0,2n-2]
    
    grad = np.zeros(n)
    for j in range(n):
        # k ranges from j to j+n-1 (so that k-j ranges from 0 to n-1)
        k_lo = max(0, j)
        k_hi = min(2*n-2, j+n-1)
        s = np.arange(k_lo - j, k_hi - j + 1)  # s = k - j
        grad[j] = float(np.sum(w[k_lo:k_hi+1] * (1.0 - 2.0 * h[s])))
```

```python
def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

            x[mask] += res / cnt
```

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-15) if res > 0 else (x > 1e-15)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, alpha):
    n = len(h)
    corr = _get_corr(h)
    cmax = float(corr.max())
    w = np.exp(alpha * (corr - cmax))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    grad = np.correlate(1.0 - 2.0 * h, w, mode='valid')
    return grad, cmax

def _adam_smooth(h, deadline, alpha_start=20.0, alpha_end=500.0,
                 lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, stall_limit=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    S = _get_corr(h)
    dx = 2.0 / n
    cur_c5 = float(S.max()) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    while time.time() < deadline:
        t += 1
        frac = min(1.0, t / 3000.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        grad, _ = _smooth_grad(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        step_norm = np.sqrt(np.sum(step * step))
        if step_norm > 1e-12:
            step = step / step_norm * lr
        best_lr = 0.0
        best_val = cur_c5
        try_lr = lr
        for _ in range(20):
            h_try = _project(h - try_lr * step, target)
            S_try = _get_corr(h_try)
            val = float(S_try.max()) * dx
            if val < best_val:
                best_val = val
                best_lr = try_lr
            try_lr *= 0.5
            if try_lr < 1e-12:
                break
        if best_lr > 0:
            h = _project(h - best_lr * step, target)
            S = _get_corr(h)
            cur_c5 = float(S.max()) * dx
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            break
    return best_h, best_c5

def _fw_step(h, alpha):
    n = len(h)
    target = n / 2.0
    grad, _ = _smooth_grad(h, alpha)
    order = np.argsort(grad)
    s = np.zeros(n)
    s[order[:int(target)]] = 1.0
    return s

def _fw_loop(h, deadline, alpha=100.0, stall_limit=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    S = _get_corr(h)
    dx = 2.0 / n
    cur_c5 = float(S.max()) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    t = 0
    while time.time() < deadline:
        t += 1
        s = _fw_step(h, alpha)
        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h + gamma * s
        h_try = _project(h_new, target)
        S_try = _get_corr(h_try)
        val = float(S_try.max()) * dx
        if val < cur_c5 - 1e-14:
            h = h_try
            S = S_try
            cur_c5 = val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n / 2.0
    if target == int(target):
        idx = np.argsort(h)[int(n - target):]
        hb = np.zeros(n)
        hb[idx] = 1.0
    else:
        idx = np.argsort(h)
        k = int(target)
        hb = np.zeros(n)
        hb[idx[k:]] = 1.0
        hb[idx[k - 1]] = target - k
    return hb

def _binary_local(hb, deadline, top_k=8, max_rounds=2000):
    n = len(hb)
    dx = 2.0 / n
    h = hb.copy()
    S = _get_corr(h)
    cur_c5 = float(S.max()) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    ones = set(np.where(h >= 0.999)[0].tolist())
    zeros = set(np.where(h <= 0.001)[0].tolist())
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        improved = False
        M = float(S.max())
        thresh = M - max(1.0, 0.01 * M)
        crit_idx = np.where(S >= thresh)[0]
        for i in list(ones):
            if time.time() > deadline:
                break
            gi = np.zeros(2 * n - 1)
            klo = max(0, i)
            khi = min(2 * n - 2, i + n - 1)
            gi[klo:khi + 1] = 1.0 - 2.0 * h[range(klo - i, khi - i + 1)]
            best_j = -1
            best_new_c5 = cur_c5
            for j in zeros:
                gj = np.zeros(2 * n - 1)
                klo2 = max(0, j)
                khi2 = min(2 * n - 2, j + n - 1)
                gj[klo2:khi2 + 1] = 1.0 - 2.0 * h[range(klo2 - j, khi2 - j + 1)]
                dS = gj - gi
                new_S = S + dS
                new_max = float(new_S.max())
                if new_max < M - 1e-12:
                    nc = new_max * dx
                    if nc < best_new_c5:
                        best_new_c5 = nc
                        best_j = j
            if best_j >= 0:
                h[i] = 0.0
                h[best_j] = 1.0
                S = _get_corr(h)
                cur_c5 = float(S.max()) * dx
                ones.discard(i)
                zeros.discard(best_j)
                ones.add(best_j)
                zeros.add(i)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.sin(2.0 * np.pi * f * x / n + ph)
    p += amp * 0.2 * rng.standard_normal(n)
    return _project(h + p, n / 2.0)
```
