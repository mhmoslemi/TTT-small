```python
def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt

def _compute_grads(h):
    """Compute gradient of each correlation value w.r.t. each h[j].
    Returns a (2n-1, n) matrix where G[m, j] = d(correlate[m])/d(h[j]).
    This is O(n^2) which is fine for n < 300.
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2*n-1, n))
    # For each shift s = m - (n-1):
    # corr[m] = sum_i h[i] * (1 - h[i+s]) for valid i
    # d(corrm)/d(h[j]) = (1-h[j+s]) if j+s in [0,n) + (-h[j-s]) if j-s in [0,n)
    #                    (the first term if j is the "h" part, second if j is the "(1-h)" part)
    for m in range(2*n-1):
        s = m - (n-1)
        if s >= 0:
            # corr[m] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])

                if j + s < n:
                    g += (1.0 - h[j+s])
                if j - s >= 0:
                    g += -h[j-s]

        else:
            ss = -s

                if j >= ss:
                    g += (1.0 - h[j-ss])
                if j + ss < n:
                    g += -h[j+ss]
                G[m, j] = g
    return corr, G

def _smooth_max_grad(corr, beta):
    """Compute softmax-weighted gradient direction.
    Returns (weights, weighted_grad) where weighted_grad = sum_k w_k * G[k,:]
    """
    n_corr = len(corr)
    n = (n_corr + 1) // 2
    shifted = corr - corr.max()
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    return weights

def _pair_line_search(corr, p, q, h, max_iter=60):

    Returns (best_t, best_c5_raw) where c5_raw = max of new corr values.
    """

    # Linear coefficient: a_s for each lag

    # a_s = -(h[p+s] - h[q+s] + h[p] - h[q]) for valid indices
    hp = h[p]
    hq = h[q]
    const = -(hp - hq)

    # where read_p[s] = h[p+s] if 0<=p+s<n else 0

    idx_p = np.where((s_vals >= -p) & (s_vals < n-p))[0]
    read_p[idx_p] = h[p + s_vals[idx_p]]
    read_q = np.zeros(2*n-1)
    idx_q = np.where((s_vals >= -q) & (s_vals < n-q))[0]
    read_q[idx_q] = h[q + s_vals[idx_q]]
    
    a = -(read_p - read_q) + const
    b = np.zeros(2*n-1)
    # b[0] corresponds to s=0, index n-1
    b[n-1] = -2.0
    # s = q - p → index (q-p) + (n-1)

    if -(n-1) <= s_qp <= n-1:

    s_pq = p - q

        b[s_pq + n - 1] = 1.0

        return 0.0, float(np.max(corr))
    
    # Golden section search on [0, t_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)

        else:

            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)

    best_t = (lo + hi) / 2.0

        grad[j] = g
    
    # Find pairs (p, q) that maximize grad[q] - grad[p]
    # (we want to increase h[p] and decrease h[q], direction is e_p - e_q)
    # The directional derivative is grad[p] - grad[q]... wait
    # Actually for direction d = e_p - e_q, the directional derivative of the objective
    # (which we want to minimize) is sum_j grad[j] * d[j] = grad[p] - grad[q]
    # We want this to be negative, so we want grad[p] < grad[q]
    # i.e., we want to move mass from high-grad to low-grad positions
    
    # Find the best O(n_pairs) pairs
    # Sort by grad value
    order = np.argsort(grad)  # ascending
    # Best pairs: p from low grad, q from high grad
    # But we need to be more careful - the pair (p,q) with p in low half, q in high half
    mid = n // 2
    # For efficiency, take top L from low and top L from high
    L = min(30, n//4)
    low_candidates = order[:L]
    high_candidates = order[-L:]
    
    # Score each pair by grad[p] - grad[q] (want most negative)
    scores = []
    for p in low_candidates:
        for q in high_candidates:
            if p != q:
                scores.append((grad[p] - grad[q], p, q))
    scores.sort(key=lambda x: x[0])
    
    pairs = [(p, q) for _, p, q in scores[:n_pairs]]
    return pairs


def _optimize_h(h, deadline, max_iters=20000, beta_init=5.0, seed=42):
    """Main optimization loop using gradient-guided pair search + exact line search.
    """
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    cur_c5 = cur_max * dx
    best_h = h.copy()
    best_c5 = cur_c5
    best_corr_max = cur_max
    rng = np.random.default_rng(seed)
    beta = beta_init
    stall = 0
    improvement_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        
        # Select pairs: mix of gradient-guided and random
        n_guided = max(5, 20 - stall // 50)
        pairs = _grad_guided_pairs(corr, h, beta, n_pairs=n_guided)
        
        # Add some random pairs for diversity
        n_random = max(5, 30 - n_guided)
        for _ in range(n_random):
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            if p != q:
                pairs.append((p, q))
        
        found_improvement = False
        # Shuffle pairs
        rng.shuffle(pairs)
        
        for p, q in pairs:
            if time.time() > deadline:
                break
            best_t, best_val = _pair_line_search(corr, p, q, h, max_iter=40)
            if best_val < cur_max - 1e-13 and best_t > 1e-12:
                # Apply the move
                h[p] += best_t
                h[q] -= best_t
                # Clip to [0,1]
                h[p] = np.clip(h[p], 0, 1)
                h[q] = np.clip(h[q], 0, 1)
                # Recompute correlation
                corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = float(np.max(corr))
                if new_max < cur_max:
                    cur_max = new_max
                    cur_c5 = cur_max * dx
                    improvement_count += 1
                    found_improvement = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        best_corr_max = cur_max
                    stall = 0
                else:
                    # The line search prediction was off (due to the quadratic
                    # approximation being local); revert
                    h[p] -= best_t
                    h[q] += best_t
                    h[p] = np.clip(h[p], 0, 1)
                    h[q] = np.clip(h[q], 0, 1)
        
        if not found_improvement:
            stall += 1
            # Reduce beta to broaden the search (less peaked)
            beta *= 0.995
            # Occasionally do a random perturbation
            if stall % 50 == 0:
                idx = rng.integers(0, n, size=2)
                if idx[0] != idx[1]:
                    t = rng.uniform(0.1, 0.5)
                    h[idx[0]] = np.clip(h[idx[0]] + t, 0, 1)
                    h[idx[1]] = np.clip(h[idx[1]] - t, 0, 1)
                    h = _project(h, n/2.0)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    cur_max = float(np.max(corr))
                    cur_c5 = cur_max * dx
    
    return best_h, best_c5


def _binary_initialize(n, seed):
    """Create a random binary sequence with n/2 ones."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _fast_coord_descent(h, deadline, max_sweeps=2000, seed=123):
    """Fast coordinate descent using precomputed gradients."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            # Compute gradient at position j
            # d(corrm)/d(h[j]) for all m
            g = np.zeros(2*n-1)
            for m in range(2*n-1):
                s = m - (n-1)
                gm = 0.0
                if s >= 0:
                    if j + s < n:
                        gm += (1.0 - h[j+s])
                    if j - s >= 0:
                        gm += -h[j-s]
                else:
                    ss = -s
                    if j >= ss:
                        gm += (1.0 - h[j-ss])
                    if j + ss < n:
                        gm += -h[j+ss]
                g[m] = gm
            
            # f_j(t) = max(corr + (t - old + old) * ... ) 
            # Actually: new_h[j] = old + delta, where delta in [-(old), 1-old]
            # new_corr[m] = corr[m] + delta * g[m] (linear in delta since we're
            # only changing one coordinate and the correlation is bilinear)
            # Wait, it's not linear! f_s = sum_i h_i(1-h_{i+s})
            # If we change h[j] to h[j]+delta:
            # The terms involving h[j] are: h[j](1-h[j+s]) and h[j-s](1-h[j])
            # New: (h[j]+delta)(1-h[j+s]) + h[j-s](1-h[j]-delta)
            # = h[j](1-h[j+s]) + delta(1-h[j+s]) + h[j-s](1-h[j]) - delta*h[j-s]
            # = old_terms + delta*((1-h[j+s]) - h[j-s])
            # So it IS linear in delta! The g[m] I computed is the slope.
            
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            
            # Minimize max(corr + delta * g) over delta in [lo_d, hi_d]
            # This is a 1D piecewise-linear minimization
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_t, hi_t = lo_d, hi_d
            c1 = hi_t - gr * (hi_t - lo_t)
            c2 = lo_t + gr * (hi_t - lo_t)
            f1 = float(np.max(corr + c1 * g))
            f2 = float(np.max(corr + c2 * g))
            for _ in range(40):
                if hi_t - lo_t < 1e-11:
                    break
                if f1 < f2:
                    hi_t = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_t - gr * (hi_t - lo_t)
                    f1 = float(np.max(corr + c1 * g))
                else:
                    lo_t = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_t + gr * (hi_t - lo_t)
                    f2 = float(np.max(corr + c2 * g))
            opt_d = (lo_t + hi_t) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                corr += delta * g
                improved = True
        
        if improved:
            h = _project(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5


def _lp_refine(h, deadline, radius_init=0.1, stall_limit=80, top_k=15):
    """LP-based refinement focusing on active constraints."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr.copy()
        idx = np.argsort(scores)[-top_k:]
        
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj])
            hi_b = min(radius, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        # Line search on h + t*p
        S0 = corr
        T = np.zeros(2*n-1)
        C = np.zeros(2*n-1)
        # T[m] = d(corr[m])/dt at t=0 in direction p
        # C[m] = d²(corr[m])/dt² in direction p
        for m in range(2*n-1):
            s = m - (n-1)
            Tm = 0.0
            Cm = 0.0
            if s >= 0:
                for i in range(n - s):
                    Tm += p[i] * (1.0 - h[i+s]) + h[i] * (-p[i+s])
                    Cm += p[i] * (-p[i+s])
            else:
                ss = -s
                for i in range(ss, n):
                    Tm += p[i] * (1.0 - h[i-ss]) + h[i] * (-p[i-ss])
                    Cm += p[i] * (-p[i-ss])
            T[m] = Tm
            C[m] = Cm
        
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, fe)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        
        if best_v < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.70
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    
    return best_h, best_c5


def _perturb(h, scale, rng):
    """Add smooth perturbation and project back."""
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freq = 3
    perturb = np.zeros(n)
    for _ in range(n_freq):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= scale
    perturb += scale * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
```

```python
def _pair_line_search_fast(corr, p, q, h, max_iter=40):
    n = len(h)
    s_vals = np.arange(-(n-1), n)
    
    # read_p[s] = h[p+s] if valid
    idx_p = s_vals + p  # indices into h
    valid_p = (idx_p >= 0) & (idx_p < n)
    read_p = np.zeros(2*n-1)
    read_p[valid_p] = h[idx_p[valid_p]]
    
    idx_q = s_vals + q
    valid_q = (idx_q >= 0) & (idx_q < n)
    read_q = np.zeros(2*n-1)
    read_q[valid_q] = h[idx_q[valid_q]]
    
    a = -(read_p - read_q) + (h[q] - h[p])
    b = np.zeros(2*n-1)
    b[n-1] = -2.0
    s_qp = q - p
    if -(n-1) <= s_qp <= n-1:
        b[s_qp + n - 1] += 1.0
    s_pq = p - q
    if -(n-1) <= s_pq <= n-1 and s_pq != 0:
        b[s_pq + n - 1] += 1.0
    
    t_max = min(1.0 - h[p], h[q])  # max t such that h[p]+t ≤ 1 and h[q]-t ≥ 0
    if t_max < 1e-15:
        return 0.0, float(np.max(corr))
    
    # Golden section
    gr = 0.6180339887498949
    lo, hi = 0.0, t_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(corr + c1 * a + c1*c1 * b))
    f2 = float(np.max(corr + c2 * a + c2*c2 * b))
    for _ in range(max_iter):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(corr + c1 * a + c1*c1 * b))
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(corr + c2 * a + c2*c2 * b))
    
    best_t = (lo + hi) / 2.0
    best_val = float(np.max(corr + best_t * a + best_t*best_t * b))
    f0 = float(np.max(corr))
    ft = float(np.max(corr + t_max * a + t_max*t_max * b))
    if f0 <= best_val and f0 <= ft:
        return 0.0, f0
    elif ft <= best_val:
        return t_max, ft
    return best_t, best_val
```
