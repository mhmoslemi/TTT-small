def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

            hi = mid

    # if d >= 0: raw = sum_{i=0}^{n-1-d} h[i] * (1-h0[i+d])

def _line_search(h, d, dx, radius, n_top=16):

    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')

            break

            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1*c1 * Q)) * dx
        else:

def _subgradient_step(h, alpha, n_selected, radius):
    """Subgradient step for the max of correlations."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    
    # Softmax weighting over top lags
    top_idx = np.argsort(-raw)[:n_selected]
    shifted = raw[top_idx] - mx
    weights = np.exp(50.0 * shifted)
    weights /= weights.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i, r in enumerate(top_idx):
        w = weights[i]
        if w < 1e-10:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            g[:m] += w * (1.0 - h[d:])
            g[d:] -= w * h[:m]
        else:
            gp = -d
            m = n - gp
            g[:m] += w * (1.0 - h[gp:])
            g[gp:] -= w * h[:m]
    
    # Remove mean to maintain sum constraint
    g -= np.mean(g)
    
    # Line search
    d = -g / (np.linalg.norm(g) + 1e-15)
    alpha_ls, val = _line_search(h, d, dx, radius)
    if alpha_ls < 1e-12:
        return h, _exact_c5(h)
    
    h_new = h + alpha_ls * d
    h_new = _project(h_new, target)
    c5_new = _exact_c5(h_new)
    return h_new, c5_new

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius, n_top=16):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)

    # Build candidate alphas from critical points
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, min(i + 6, len(top_idx))):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)
                    a2 = (B - sd) / (2.0 * A)
                    for aa in [a1, a2]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)

    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha

    # Golden section refinement around best
    step = a_hi / 30.0
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(45):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt

    return best_alpha, best_val

def _lp_solve(h0, n_selected, box_radius, deadline):
    """Solve LP with box trust region to minimize max linearized overlap."""
    n = len(h0)
    target = n / 2.0
    raw0 = np.correlate(h0, 1.0 - h0, mode='full')
    sorted_idx = np.argsort(-raw0)
    selected = sorted_idx[:n_selected]

    x = cp.Variable(n)
    t = cp.Variable()

    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == target,
        x >= h0 - box_radius,
        x <= h0 + box_radius
    ]

    for r in selected:
        d = int(r) - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                coeff = np.array(1.0 - h0[d:d + m])
                constraints.append(cp.sum(cp.multiply(cp.Parameter(len(m), value=coeff), x[:m])) <= t)
        else:
            gp = -d
            m = n - gp
            if m > 0:
                coeff = np.array(1.0 - h0[:m])
                constraints.append(cp.sum(cp.multiply(cp.Parameter(len(m), value=coeff), x[gp:gp + m])) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    if x.value is None:
        return None
    return np.array(x.value).flatten()

def _lp_round(h, deadline, n_sel_list=None, radii=None, max_lp_calls=50):
    if n_sel_list is None:
        n_sel_list = [8, 12, 16, 20, 25]
    if radii is None:
        radii = [0.5, 0.35, 0.25, 0.15, 0.1, 0.07]

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lp_count = 0

    for ns in n_sel_list:
        for radius in radii:
            if time.time() > deadline or lp_count >= max_lp_calls:
                break
            x_lp = _lp_solve(h, ns, radius, deadline)
            lp_count += 1
            if x_lp is None:
                continue
            d = x_lp - h
            nm = np.linalg.norm(d)
            if nm < 1e-12:
                continue
            d = d / nm
            alpha, _ = _line_search(h, d, dx, radius)
            if alpha >= 1e-12:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()

    return best_h, best_c5

def _gradient_optimize(h, deadline, beta=20.0, radius=0.2, stall_limit=80, n_sel=12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta_cur = beta

    while time.time() < deadline:
        if stall > stall_limit:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        top_idx = np.argsort(-raw)[:n_sel]
        shifted = raw[top_idx] - mx
        vp = np.exp(np.clip(beta_cur * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)

        g = np.zeros(n)
        for i, r in enumerate(top_idx):
            wi = w[i]
            if wi < 1e-12:
                continue
            d = int(r) - (n - 1)
            if d >= 0:
                m = n - d
                g[:m] += wi * (1.0 - h[d:])
                g[d:] -= wi * h[:m]
            else:
                gp = -d
                m = n - gp
                g[:m] += wi * (1.0 - h[gp:])
                g[gp:] -= wi * h[:m]

        g -= np.mean(g)
        nm = np.linalg.norm(g)
        if nm < 1e-14:
            break
        d = -g / nm
        alpha, _ = _line_search(h, d, dx, radius)
        if alpha >= 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.02, 0.35)
                beta_cur = min(beta_cur * 1.04, 50.0)
            else:
                stall += 1
                radius *= 0.85
                beta_cur = max(beta_cur * 0.9, 3.0)
        else:
            stall += 1
            radius *= 0.8
            beta_cur = max(beta_cur * 0.9, 3.0)

    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=25000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(54321)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

                best_c5 = c5_bs
                best_h = h_bs.copy()

    # Phase 6: Basin hopping with LP (10% remaining)
    d6 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d6 - 8 and n_restarts < 60:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.007)
        h_t, c5_t = _lp_round(h_pert, sub_dl,
                              n_sel_list=[10, 15],
                              radii=[0.2, 0.12, 0.08])
        if time.time() < d6 - 3:
            sub_dl2 = min(d6 - 3, time.time() + budget_s * 0.004)
            h_t, c5_t = _gradient_optimize(h_t, sub_dl2, beta=30.0, radius=0.05, stall_limit=20, n_sel=15)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.06)

    # Final polish
    if deadline - time.time() > 4:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _gradient_optimize(h_f, deadline, beta=40.0, radius=0.04, stall_limit=30, n_sel=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if deadline - time.time() > 3:
        h_bin = _to_binary(best_h)
        h_b, c5_b = _binary_sa(h_bin, min(deadline, time.time() + budget_s * 0.02), max_iter=10000, T_init=0.001)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
