            x[mask] += res / cnt

def _smoothed_obj_and_grad(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = len(corr)
    values = mu * dx * corr
    m = float(np.max(values))
    exp_vals = np.exp(values - m)
    Z = float(np.sum(exp_vals))
    weights = exp_vals / Z
    fval = (m + np.log(Z)) / mu
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(L):
        w = weights[k]
        if w < 1e-14:

        ell = k - (n - 1)
        if ell >= 0:
            mvalid = n - ell
            if mvalid > 0:
                grad[:mvalid] += w * dx * (1.0 - h[ell:ell+mvalid])
                grad[ell:ell+mvalid] += w * dx * (-h[:mvalid])
        else:
            p = -ell
            mvalid = n - p
            if mvalid > 0:
                grad[p:p+mvalid] += w * dx * (1.0 - h[:mvalid])
                grad[:mvalid] += w * dx * (-h[p:p+mvalid])
    
    return fval, grad

def _nlp_solve(h_init, mu, deadline, maxiter=200):
    n = len(h_init)
    target = n / 2.0
    h0 = _project(h_init.copy(), target)
    
    def obj(x):
        f, g = _smoothed_obj_and_grad(x, mu)
        return f
    
    def jac(x):
        f, g = _smoothed_obj_and_grad(x, mu)
        return g

    if s < 1e-30:
        top_k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    best_val = min(opt_val, float(vals[best_idx]))
    if best_val == opt_val:
        return opt_t, best_val
    return float(ts[best_idx]), best_val

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        sources = sort_idx[-min(n_pairs, n // 2):]
        sinks = sort_idx[:min(n_pairs, n // 2)]
        pairs = []
        for ii in range(min(n_pairs, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, max(1, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-15:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.02, 0.30)
        else:
            stall += 1
            radius *= 0.80
            if stall % 50 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _run_nlp_phase(h_init, n_init, deadline, rng):
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Homotopy at main n
    mu_sched = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 250.0]
    sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.45)
    h_nlp, c5_nlp = _nlp_homotopy(h_init, sub_dl, mu_schedule=mu_sched, maxiter=100)
    if c5_nlp < best_c5:
        best_c5 = c5_nlp
        best_h = h_nlp.copy()
    # Multiple restarts at high mu
    if time.time() < deadline:
        for restart in range(6):
            if time.time() > deadline:
                break
            pert = rng.normal(0, 0.03, size=n_init)
            h_r = _project(best_h + pert, n_init / 2.0)
            h_r = _nlp_solve(h_r, 100.0, maxiter=80)
            h_r = _project(h_r, n_init / 2.0)
            c5_r = _exact_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    return best_h, best_c5

def _run_multi_n(h_best, n_best, deadline, rng):
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    n_init = len(initial_h_values) if 'initial_h_values' in dir() else len(h_best)
    offsets = [-8, 8, -12, 12, -4, 4, -16, 16, -20, 20, -24, 24, -28, 28]
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_init + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 30.0)
        h_r, c5_r = _nlp_homotopy(h_r, sub_dl,
                                   mu_schedule=[10.0, 30.0, 80.0], maxiter=60)
        # Quick pair polish
        sub_dl2 = min(deadline, time.time() + 15.0)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.12, n_pairs=35,
                                    beta=20.0, stall_limit=60, seed=200+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: NLP homotopy at parent n (45% time)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _run_nlp_phase(h_init, n0, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (25% time)
    d2 = t0 + budget_s * 0.70
    n_curr = len(initial_h_values)
    h2, c5_2 = _run_multi_n(best_h, n_curr, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair optimization polish (20% time)
    d3 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h3 = _project(best_h.copy(), n_curr / 2.0)
    h3, c5_3 = _pair_optimize(h3, d3, radius=0.14, n_pairs=55,
                               beta=25.0, stall_limit=150, seed=555)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Final NLP + polish (7% time)
    d4 = deadline - 2.0
    n_curr = len(best_h)
    if time.time() < d4 - 10:
        h4 = _nlp_solve(best_h, 200.0, maxiter=60)
        h4 = _project(h4, n_curr / 2.0)
        c5_4 = _exact_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h5 = _project(best_h.copy(), len(best_h) / 2.0)
        h5, c5_5 = _pair_optimize(h5, d4, radius=0.08, n_pairs=40,
                                   beta=30.0, stall_limit=80, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
