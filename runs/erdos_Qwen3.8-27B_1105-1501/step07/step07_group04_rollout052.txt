import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_gradient_rows(h, active_indices):
    """Build gradient rows for correlation constraints at given lag indices."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    rows = []
    for m in active_indices:
        k = m - center
        g = np.zeros(n)
        if k >= 0:
            # F_k = sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                # d/dh[j] for j in [0, n-k-1]: (1-h[j+k])
                g[:n-k] += (1.0 - h[k:])
                # d/dh[j] for j in [k, n-1]: -h[j-k]
                g[k:] -= h[:n-k]
        else:
            u = -k
            # F_{-u} = sum_{j=0}^{n-u-1} h[j+u]*(1-h[j])
            # d/dh[j] for j in [u, n-1]: (1-h[j-u])
            g[u:] += (1.0 - h[:n-u])
            # d/dh[j] for j in [0, n-u-1]: -h[j+u]
            g[:n-u] -= h[u:]
        rows.append(g * dx)
    return np.array(rows)

def _lp_direction(h, corr_scaled, active_indices, radius, eps=1e-5):
    """Solve LP to find direction that reduces max of active constraints."""
    n = len(h)
    m = len(active_indices)
    if m == 0:
        return None
    
    # Build gradient matrix
    G = _build_gradient_rows(h, active_indices)  # shape (m, n)
    
    # Current values of active constraints
    F_active = corr_scaled[active_indices]  # shape (m,)
    
    # Variables: d (n vars), t (1 var)
    # Minimize: t + eps * sum(|d_i|) -> use t + eps * sum(u_i) with -u_i <= d_i <= u_i
    # Actually, let's keep it simple: just minimize t with bounded d
    # Variables: d_0,...,d_{n-1}, t
    # Total vars: n + 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Constraints: G @ d - t <= -F_active (i.e., G@d <= t - F_active)
    # In standard form A_ub @ x <= b_ub:
    # [G, -1] @ [d; t] <= -F_active
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -F_active
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -radius <= d_i <= radius, t unbounded below (or bounded)
    bounds = [(-radius, radius)] * n + [(-10.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    # Check that it actually reduces the max
    new_F = F_active + G @ d
    if new_F.max() >= F_active.max() - 1e-12:
        return None
    
    return d

def _line_search_exact(h, d, dx, t_hi=None):
    """Exact line search using quadratic expansion."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    # Find feasible t_hi
    if t_hi is None:
        t_hi = 1.0
        pos = d > 1e-15
        neg = d < -1e-15
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * L1 - t*t * Q2)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement
    lo_b = grid_t[max(0, best_idx - 1)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    if hi_b - lo_b < 1e-12:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * L1 - c1*c1 * Q2)) * dx
        f2 = float(np.max(S0 + c2 * L1 - c2*c2 * Q2)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * L1 - c1*c1 * Q2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * L1 - c2*c2 * Q2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    
    opt_val = float(np.max(S0 + opt_t * L1 - opt_t*opt_t * Q2)) * dx
    return opt_t, opt_val

def _lp_optimize(h, deadline, radius_init=0.08, max_iter=5000):
    """Main LP-based optimization loop."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    n_active = 30  # number of active constraints to use
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_max = float(corr.max())
        
        # Select active constraints: those close to the max
        threshold = cur_max - max(0.005, cur_max * 0.02)
        active = np.where(corr >= threshold)[0]
        
        # Also include top n_active for diversity
        top_idx = np.argsort(-corr)[:n_active]
        active = np.unique(np.concatenate([active, top_idx]))
        
        # LP direction
        d = _lp_direction(h, corr, active, radius)
        
        if d is None:
            # Fallback: negative subgradient
            G = _build_gradient_rows(h, active)
            g = G.mean(axis=0)
            d = -g
            d = d - np.mean(d)
            d = np.clip(d, -radius, radius)
            d = d - np.mean(d)
        
        if float(np.max(np.abs(d))) < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        
        alpha, new_c5 = _line_search_exact(h, d, dx)
        
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.72
    
    return best_h, best_c5

def _binary_local_search(h_bin, deadline, max_iters=100000, temp_init=0.004, seed=42):
    """Simulated annealing on binary vectors."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 4000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.2, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _build_grad(h, active_indices):
    """Build gradient rows for correlation constraints. Returns (m, n) array."""
    n = len(h)

    for idx, mi in enumerate(active_indices):
        k = int(mi) - center
        if k >= 0:
            if k == 0:
                G[idx, :] = (1.0 - 2.0 * h) * dx
            else:
                G[idx, :n - k] += (1.0 - h[k:]) * dx
                G[idx, k:] -= h[:n - k] * dx
        else:
            u = -k
            G[idx, u:] += (1.0 - h[:n - u]) * dx
            G[idx, :n - u] -= h[u:] * dx

    A_ub[:, n] = -1.0
    b_ub = -F_act
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    try:

        return None

        return None

        best_c5 = c5_comp
        best_h = h_comp.copy()
    
    # Phase 1: Core LP optimization (25% of budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_optimize(best_h, d1, radius_init=0.08, max_iter=4000, n_active_base=35)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary refinement (20%)
    d2 = t0 + budget_s * 0.45
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=40000, seed=101)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    sub_dl = min(d2, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iters=80000, temp_init=0.004, seed=202)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=30000, seed=303)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # LP polish on best
    sub_dl = min(d2, time.time() + budget_s * 0.03)
    h_pol = _project(best_h.copy(), n0 / 2.0)
    h_pol, c5_pol = _lp_optimize(h_pol, sub_dl, radius_init=0.05, max_iter=2000, n_active_base=40)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Phase 3: Multi-n exploration (30%)
    d3 = t0 + budget_s * 0.75
    tried = set()
    # Try nearby n values
    for dn in sorted(range(-24, 25, 2), key=abs):
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r0 = _exact_c5(h_r)
        # Quick LP pass
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.07, max_iter=1200, n_active_base=25)
        # Binary pass if time
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iters=30000, temp_init=0.003, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Try greedy constructions for a few n values
    for n_try in [n0 - 12, n0 + 12, n0 - 24, n0 + 6]:
        if time.time() > d3:
            break
        if n_try < 80 or n_try > 280:
            continue
        h_g = _greedy_construct(n_try, max_time=3.0)
        c5_g = _exact_c5(h_g)
        if c5_g < best_c5 + 0.003:
            sub_dl = min(d3, time.time() + budget_s * 0.01)
            h_g, c5_g = _lp_optimize(h_g, sub_dl, radius_init=0.06, max_iter=800, n_active_base=25)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
    
    # Phase 4: Basin hopping + final refinement (20%)
    d4 = deadline
    n_best = len(best_h)
    scale = 0.012
    for _ in range(60):
        if time.time() > d4 - 5:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.04, max_iter=800, n_active_base=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.12)
    
    # Pair swap
    if time.time() < d4 - 20:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_swap_optimize(h4, sub_dl, n_pairs=20000, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Final binary SA
    if time.time() < d4 - 15:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iters=60000, temp_init=0.002, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Final LP polish
    if time.time() < d4 - 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, d4 - 1, radius_init=0.02, max_iter=800, n_active_base=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
