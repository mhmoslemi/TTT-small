    # Positive lags: C[n-1+tau]
    # Term 1: +delta*(1-h[p+tau]) for tau = 1..min(n-1-p, n-1)
    # Term 2: -delta*h[p-tau] for tau = 1..min(p, n-1)
    # Negative lags: C[n-1-d]
    # Term 1: +delta*(1-h[p-d]) for d = 1..min(p, n-1)
    # Term 2: -delta*h[p+d] for d = 1..min(n-1-p, n-1)
    
    if p + delta == h[p]:  # no change
        return
    old = h[p]
    new = old + delta
    
    # Positive lags
    max_t1 = min(n - 1 - p, n - 1)  # for p+tau < n, tau < n-p
    if max_t1 >= 1:
        taus = np.arange(1, max_t1 + 1)
        C[n - 1 + taus] += delta * (1.0 - h[p + taus])
    max_t2 = min(p, n - 1)
    if max_t2 >= 1:
        taus = np.arange(1, max_t2 + 1)

    # Negative lags
    max_d1 = min(p, n - 1)  # for p-d >= 0, d <= p

    if max_d2 >= 1:

                elif lag < 0:
                    d = -lag
                    src = [i for i in range(d, n) if h[i] > 0.5 and h[i-d] <= 0.5]
                    tgt = [i for i in range(0, n-d) if h[i] <= 0.5 and h[i+d] > 0.5]
                else:
                    src = []; tgt = []
                hot_sources.extend(src[:10])
                hot_targets.extend(tgt[:10])
            hot_sources = list(set(hot_sources))
            hot_targets = list(set(hot_targets))
        ones_set = np.where(h > 0.5)[0]
        zeros_set = np.where(h <= 0.5)[0]
        if len(ones_set) == 0 or len(zeros_set) == 0:
            break
        use_targeted = len(hot_sources) > 0 and len(hot_targets) > 0 and rng.random() < target_prob
        if use_targeted:
            oi = hot_sources[int(rng.integers(0, len(hot_sources)))]
            oz = hot_targets[int(rng.integers(0, len(hot_targets)))]
            if h[oi] <= 0.5 or h[oz] > 0.5:
                oi = ones_set[int(rng.integers(0, len(ones_set)))]
                oz = zeros_set[int(rng.integers(0, len(zeros_set)))]
            if oi == oz:
                oi = ones_set[int(rng.integers(0, len(ones_set)))]
        else:
            oi = ones_set[int(rng.integers(0, len(ones_set)))]
            oz = zeros_set[int(rng.integers(0, len(zeros_set)))]
        if oi == oz:
            continue
        _apply_swap(C, h, oi, oz, n)
        new_raw_max = float(np.max(C))
        new_c5 = new_raw_max * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            cur_raw_max = new_raw_max
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
                best_stall_reset += 1
            T *= 0.9997
        else:
            _undo_swap(C, h, oi, oz, n)
            stall += 1
            T *= 0.9999
    return best_h, best_c5

def _grad_smooth(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    w /= wsum
    active = w > 1e-8
    if not np.any(active):
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    g = np.zeros(n)
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[tau:tau + m]
        else:
            gg = -tau
            m = n - gg
            if m > 0:
                g[:m] += wr * (1.0 - h[gg:])
                g[gg:] -= wr * h[:m]
    g -= float(np.mean(g))
    return g

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        gg = -tau
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    ng = 80
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _make_dirs(h, g, raw, rng, n_max=8):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    r_max = int(np.argmax(raw))
    gs = _grad_single(h, r_max)
    gsn = np.linalg.norm(gs)
    if gsn > 1e-12:
        d = gs / gsn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for blk in [4, 8, 16]:
        if n % blk != 0:
            continue
        nblk = n // blk
        gb = g.reshape(nblk, blk).mean(axis=1)
        d = np.tile(gb, (1, blk)).reshape(n).copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for freq in [1, 2, 3]:
        x = np.arange(n, dtype=float)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    idx_s = np.argsort(g)
    for s in idx_s[:3]:
        for k in idx_s[-3:]:
            if s == k:
                continue
            d = np.zeros(n)
            d[s] = 1.0
            d[k] = -1.0
            d -= np.mean(d)
            dirs.append(d)
            break
    return dirs[:n_max]

def _opt_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.15, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(777 + n)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = float(total_iters) / float(total_iters + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 2.5)
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        g = _grad_smooth(h, raw, beta)
        dirs = _make_dirs(h, g, raw, rng, 6)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.04, radius * 0.78)
        else:
            stall += 1
            radius = max(0.04, radius * 0.80)
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=6000, radius=0.18, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa_multistart(h_bin, deadline, n_starts=3):
    n = len(h_bin)
    rng = np.random.default_rng(42)
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    per_dl = (deadline - time.time()) / n_starts
    for s in range(n_starts):
        sub_dl = t0_global + (s + 1) * per_dl  # placeholder, handled below
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + per_dl)
        if s == 0:
            h_cur = h_bin.copy()
        else:
            pert = np.zeros(n)
            k = int(n * rng.uniform(0.02, 0.06))
            idx = rng.choice(n, size=k * 2, replace=False)
            ones_in = idx[h_cur[idx] > 0.5]
            zeros_in = idx[h_cur[idx] <= 0.5]
            if len(ones_in) > 0 and len(zeros_in) > 0:
                h_cur = h_cur.copy()
                m = min(len(ones_in), len(zeros_in))
                h_cur[ones_in[:m]] = 0.0
                h_cur[zeros_in[:m]] = 1.0
            else:
                h_cur = h_bin.copy()
        h_res, c5_res = _binary_sa_fast(h_cur, sub_dl, max_iter=200000, T_init=0.003)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5

t0_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0 = time.time()
    t0_global = t0
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous optimization (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _opt_smooth(h_init, d1, beta_lo=4.0, beta_hi=35.0, radius_init=0.12, max_stall=60)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Intensive fast binary SA (50%)
    d2 = t0 + budget_s * 0.58
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    c5_b_start = _exact_c5(h_b)
    if c5_b_start < best_c5:
        best_c5 = c5_b_start
        best_h = h_b.copy()
    # Main binary SA with lots of iterations
    sub_dl = min(d2, time.time() + budget_s * 0.35)
    h_b, c5_b = _binary_sa_fast(h_b, sub_dl, max_iter=1000000, T_init=0.004, T_floor=1e-6, target_prob=0.6, refresh_int=150)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    # Second SA round from best
    if time.time() < d2 - 5:
        sub_dl2 = min(d2, time.time() + budget_s * 0.10)
        h_b2, c5_b2 = _binary_sa_fast(best_h.copy(), sub_dl2, max_iter=400000, T_init=0.002, T_floor=1e-6, target_prob=0.7, refresh_int=100)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 3: Continuous polish of best binary (18%)
    d3 = t0 + budget_s * 0.76
    n_c = len(best_h)
    h_pol = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_pol, c5_pol = _opt_smooth(h_pol, sub_dl, beta_lo=8.0, beta_hi=50.0, radius_init=0.10, max_stall=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    if time.time() < d3 - 3:
        h_pc = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d3, time.time() + budget_s * 0.05)
        h_pc, c5_pc = _pair_continuous(h_pc, sub_dl2, n_iter=8000, radius=0.15, seed=555)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
    if time.time() < d3 - 2:
        h_pol2 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.04)
        h_pol2, c5_pol2 = _opt_smooth(h_pol2, sub_dl3, beta_lo=20.0, beta_hi=80.0, radius_init=0.06, max_stall=40)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    # Phase 4: Multi-n exploration + basin hopping (16%)
    d4 = t0 + budget_s * 0.92
    n_curr = len(best_h)
    for dn in range(-16, 17, 4):
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 300 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _opt_smooth(h_r, sub_dl, beta_lo=10.0, beta_hi=40.0, radius_init=0.10, max_stall=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Basin hopping
    if time.time() < d4 - 5:
        scale = 0.025
        n_restarts = 0
        while time.time() < d4 - 3 and n_restarts < 60:
            n_restarts += 1
            h_p = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4 - 3, time.time() + budget_s * 0.006)
            h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=12.0, beta_hi=60.0, radius_init=0.08, max_stall=25)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.004, scale * 0.75)
            else:
                scale = min(0.07, scale * 1.12)

    # Phase 5: Final polish (8%)
    d5 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h_f, c5_f = _opt_smooth(h_f, sub_dl, beta_lo=25.0, beta_hi=100.0, radius_init=0.05, max_stall=35)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d5 - 2:
        h_bf = _to_binary(best_h)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_bf, c5_bf = _binary_sa_fast(h_bf, sub_dl2, max_iter=200000, T_init=0.001, T_floor=1e-6, target_prob=0.8, refresh_int=80)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            # Quick continuous from new binary
            h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
            sub_dl3 = min(d5, time.time() + budget_s * 0.015)
            h_f2, c5_f2 = _opt_smooth(h_f2, sub_dl3, beta_lo=30.0, beta_hi=100.0, radius_init=0.04, max_stall=30)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
