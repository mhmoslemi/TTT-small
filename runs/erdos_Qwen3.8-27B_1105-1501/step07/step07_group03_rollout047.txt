def _lp_optimize(h, k_active, radius, deadline):
    """LP-based minimax optimizer."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline:
        if stall > 50:
            break
        
        corr = _correlation_profile(h)
        top_idx = np.argsort(-corr)[:k_active]
        
        # Compute gradients for active lags
        grads = np.zeros((k_active, n))
        for i, r in enumerate(top_idx):
            d = int(r - (n - 1))
            g = np.zeros(n)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += (1.0 - h[:m])
                    g[:m] -= h[d:]
            else:
                p = -d
                m = n - p
                if m > 0:
                    g[:m] += (1.0 - h[p:])
                    g[p:] -= h[:m]
            grads[i] = g
        
        # Current overlap values for active lags
        curr_vals = corr[top_idx].copy()
        
        # Set up LP
        x = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [
            cp.sum(x) == target,
            x >= 0,
            x <= 1,
            x >= h - radius,
            x <= h + radius
        ]
        
        for i in range(k_active):
            # Linearization: f_i(x) ≈ curr_vals[i] + grads[i] · (x - h)
            # Want: curr_vals[i] + grads[i] · (x - h) <= t
            # i.e.: grads[i] · x <= t - curr_vals[i] + grads[i] · h
            rhs = t - curr_vals[i] + float(np.dot(grads[i], h))
            constraints.append(float(np.dot(grads[i], x)) <= rhs)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.CLVXR if 'CLVXR' in dir(cp) else cp.SCS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                stall += 1
                continue
        
        if x.value is None:
            stall += 1

    # Try nearby n values
    offset_list = list(range(-20, 21, 3)) + list(range(-10, 11, 2))
    for dn in offset_list:
        if time.time() > d3 - 10:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 250 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, k_active=18, radius=0.12, stall_limit=35, seed=n_try)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    # LP with more active constraints
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, k_active=35, radius=0.10, stall_limit=80, seed=100)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Pair swaps
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _pair_swap_cont(h4, sub_dl, n_pairs=10000, radius=0.25, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Binary round
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.0015)
        if time.time() < d4 - 1:
            sub_dl2 = min(d4, time.time() + budget_s * 0.02)
            h_b, c5_b = _pair_sweep_fast(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # More LP
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _lp_optimize(h4, sub_dl, k_active=40, radius=0.06, stall_limit=50, seed=200)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: Basin hopping (5%)
    d5 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 2, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, k_active=20, radius=0.10, stall_limit=25, seed=500 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.10, scale * 1.10)
    
    # Final polish
    if deadline - time.time() > 3:
        nf = len(best_h)
        h6 = _project(best_h.copy(), nf / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline - 1, k_active=50, radius=0.04, stall_limit=40, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    if deadline - time.time() > 2:
        nf = len(best_h)
        h6 = _project(best_h.copy(), nf / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=2000, radius=0.10, seed=777)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
