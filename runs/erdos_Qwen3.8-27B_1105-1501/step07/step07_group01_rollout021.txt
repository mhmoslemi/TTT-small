    """Compute gradient of sum_a weights[a] * R_a(h) with respect to h.
    
    R_a = correlate(h, 1-h, 'full')[a]
    
    For a = n-1+s (s >= 0): R = sum_{i=0}^{n-s-1} h[i]*(1-h[i+s])
    For a = n-1-s (s > 0, i.e., a < n-1): R = sum_{i=0}^{n-s-1} h[i+s]*(1-h[i])
    """
    n = len(h)
    g = np.zeros(n)
    q = 1.0 - h
    
    for a in range(2 * n - 1):
        w = weights[a]
        if abs(w) < 1e-14:
            continue
        s = a - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            # R = sum_{i=0}^{m-1} h[i] * q[i+s]
            # dR/dh[j]: 
            #   j as h[i]: i=j, need j < m => j < n-s
            #     contribution: q[j+s]
            #   j as q[i+s]: i+s=j => i=j-s, need 0<=j-s<m => s<=j<s+m
            #     contribution: -h[j-s] (derivative of h[j] wrt h[j] in -h[i]*h[i+s] is -h[i])
            # Wait, R = sum h[i]*(1-h[i+s]) = sum h[i] - sum h[i]*h[i+s]
            # dR/dh[j] = sum of contributions:
            #   From h[i] term: if i=j and j < m: contributes 1
            #   From -h[i]*h[i+s] term: 
            #     if i=j and j < m: contributes -h[j+s]
            #     if i+s=j, i=j-s: contributes -h[j-s]

            #       = [j<m]*(1-h[j+s]) - [s<=j<s+m]*h[j-s]
            
            # Vectorized:
            # j in [0, m-1]: contribution += (1 - h[j+s])
            # j in [s, s+m-1] = [s, n-1]: contribution += -h[j-s]
            g[:m] += w * q[s:s+m]  # h[j+s] for j=0..m-1 => h[s:s+m]

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

                m = n - b
                if m > 0:
                    G[ai, b:b + m] += (1.0 - h[:m])
                    G[ai, :m] += (-h[b:b + m])
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -R[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj])
            hi_b = min(radius, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
        except Exception:
            stall += 1
                radius *= 0.7
                continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            continue
        t_max = _t_max_for_direction(h, p)
        t_max = min(t_max, radius)
        if t_max < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        T1 = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C1 = np.correlate(p, p, mode='full')
        S0 = R
        t_opt = _line_search_max_quadratic(S0, T1, C1, t_max)
        f0 = float(np.max(S0)) * dx
        f_end = float(np.max(S0 + t_max * T1 - t_max * t_max * C1)) * dx
        f_opt = float(np.max(S0 + t_opt * T1 - t_opt * t_opt * C1)) * dx
        best_t = min([(0.0, f0), (t_opt, f_opt), (t_max, f_end)],
                     _key_func)[0]
        if best_t > 1e-12:
            h_new = h + best_t * p
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.25)
            else:
                stall += 1
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _key_func(pair):
    return pair[1]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_from_h(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick coordinate descent warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _coord_descent(best_h, d1, max_sweeps=800, seed=111)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed subgradient (12%)
    d2 = t0 + budget_s * 0.20
    h2, c5_2 = _smoothed_subgradient(best_h, d2, n_iter=500, seed=222)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Follow with coord descent
    sub_dl = min(d2, time.time() + budget_s * 0.02)
    h2b, c5_2b = _coord_descent(best_h, sub_dl, max_sweeps=500, seed=333)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: Intensive coordinate descent (30%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _coord_descent(best_h, d3, max_sweeps=8000, seed=444)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP + subgradient refinement (15%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _lp_active(best_h, min(d4, time.time() + budget_s * 0.04),
                          radius_init=0.06, stall_limit=40, top_k=15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    h4b, c5_4b = _smoothed_subgradient(best_h, min(d4, time.time() + budget_s * 0.04),
                                       n_iter=200, tau_init=0.3, alpha_init=0.03, seed=555)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()
    h4c, c5_4c = _coord_descent(best_h, min(d4, time.time() + budget_s * 0.03),
                                max_sweeps=2000, seed=666)
    if c5_4c < best_c5:
        best_c5 = c5_4c
        best_h = h4c.copy()

    # Phase 5: Pair descent (10%)
    d5 = t0 + budget_s * 0.75
    h5, c5_5 = _pair_descent(best_h, d5, n_pairs=50000, seed=777)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    h5b, c5_5b = _coord_descent(best_h, min(d5, time.time() + budget_s * 0.02),
                                max_sweeps=2000, seed=888)
    if c5_5b < best_c5:
        best_c5 = c5_5b
        best_h = h5b.copy()

    # Phase 6: Multi-resolution exploration (12%)
    d6 = t0 + budget_s * 0.87
    tried_ns = set()
    offsets = list(range(-20, 21, 4))
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 250 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=400, seed=1000 + dn)
        if time.time() < d6 - 3:
            sub_dl2 = min(d6, time.time() + budget_s * 0.003)
            h_r, c5_r2 = _pair_descent(h_r, sub_dl2, n_pairs=3000, seed=2000 + dn)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Basin hopping + final polish (13%)
    d7 = deadline
    scale = 0.015
    n_restarts = 0
    while time.time() < d7 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d7 - 5, time.time() + budget_s * 0.004)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=300, seed=3000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.1)

    # Final intensive polish
    sub_dl = min(d7, time.time() + budget_s * 0.02)
    h_f, c5_f = _coord_descent(best_h, sub_dl, max_sweeps=5000, seed=9999)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d7 - 2:
        h_f2, c5_f2 = _pair_descent(best_h, min(d7, time.time() + budget_s * 0.01),
                                    n_pairs=10000, seed=8888)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
