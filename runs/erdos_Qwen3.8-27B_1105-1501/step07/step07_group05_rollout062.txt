import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _active_gradients(h, active_idx):
    """Compute gradients for active output indices of the full correlation."""
    n = len(h)
    grads = []
    for t in active_idx:
        l = t - (n - 1)
        g = np.zeros(n)
        if l >= 0:
            m = n - l
            g[:m] += (1.0 - h[l:l+m])
            g[l:] -= h[:m]
        else:
            s = -l
            m = n - s
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s+m]
        norm = np.linalg.norm(g)
        if norm > 1e-10:
            g /= norm
        grads.append(g)
    return np.array(grads)

def _lp_direction(h, active_idx, radius):
    """Solve LP to find direction that decreases all active constraints."""
    import cvxpy as cp
    n = len(h)
    G = _active_gradients(h, active_idx)
    K = G.shape[0]
    if K == 0:
        return None
    lb = np.minimum(h, radius)
    ub = np.minimum(1.0 - h, radius)
    s_var = cp.Variable(n)
    tau_var = cp.Variable()
    constraints = [
        cp.sum(s_var) == 0,
        tau_var >= 0,
        s_var >= -lb,
        s_var <= ub
    ]
    for i in range(K):
        constraints.append(G[i] @ s_var <= -tau_var)
    prob = cp.Problem(cp.Maximize(tau_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8, reltol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-8)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    tau_val = float(tau_var.value)
    if tau_val < 1e-10:
        return None
    s_val = np.array(s_var.value).flatten()
    # Verify sum is close to 0 and bounds are respected
    s_val[np.where(s_val > ub)[0]] = ub[np.where(s_val > ub)[0]]
    s_val[np.where(s_val < -lb)[0]] = -lb[np.where(s_val < -lb)[0]]
    s_val -= np.mean(s_val)
    return s_val

def _aggregate_direction(h, active_idx, radius):
    """Fallback: weighted aggregate gradient direction."""
    n = len(h)
    G = _active_gradients(h, active_idx)
    G_agg = np.sum(G, axis=0)
    norm = np.linalg.norm(G_agg)
    if norm < 1e-10:
        return None
    d = -G_agg / norm
    # Clip to feasible range
    lb = np.minimum(h, radius)
    ub = np.minimum(1.0 - h, radius)
    d = np.clip(d, -lb, ub)
    d -= np.mean(d)
    norm2 = np.linalg.norm(d)
    if norm2 < 1e-10:
        return None
    return d / norm2

def _line_search(h, d, dx, radius):
    """Exact line search for h + alpha*d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Determine feasible alpha range
    pos = d > 1e-14
    neg = d < -1e-14
    a_hi = radius
    a_lo = -radius
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Sample the model f(alpha) = max(S0 + alpha*T - alpha^2*Q) * dx
    best_val = float('inf')
    best_a = 0.0
    
    # Collect candidate alphas
    cands = [a_lo, a_hi, (a_lo + a_hi) / 2.0]
    
    # Optimal for each output index
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Grid
    width = a_hi - a_lo
    cands.extend(np.linspace(a_lo, a_hi, 64).tolist())
    
    # Top M outputs: pairwise intersections
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    
    # Evaluate all candidates
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

def _local_opt_lp(h, deadline, r0=0.10, K_list=(3, 9, 24, 70), stall_lim=50):

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rad = r0
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        
        S = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-S)
        
        best_new = cur
        best_a = 0.0
        best_d = None
        
        for K in K_list:
            K_actual = min(K, len(S))
            active = top_idx[:K_actual]
            
            # Try LP direction
            d = _lp_direction(h, active, rad)
            if d is not None and np.linalg.norm(d) > 1e-10:
                a, c_try = _line_search(h, d, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d
            
            # Try aggregate gradient
            d = _aggregate_direction(h, active, rad)
            if d is not None:
                a, c_try = _line_search(h, d, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d
        
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.25)
        else:
            stall += 1
            rad *= 0.8
            if rad < 0.005:
                # Perturbation escape
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp_val = _exact_c5(hp)
                if cp_val < cur - 1e-12:
                    h = hp
                    cur = cp_val
                    stall = 0
                    rad = 0.10
                elif stall % 20 == 0:
                    rad = 0.05
    
    return best_h, best_c

def _pair_opt(h, deadline, max_it=30000):
    """Pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _fourier_dir(h, rng, radius):

    norm2 = np.linalg.norm(d)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _active_gradients(h, active_idx):
    n = len(h)
    grads = []
    for t in active_idx:
        l = t - (n - 1)
        g = np.zeros(n)
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += (1.0 - h[l:l+m])
                g[l:] -= h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += (1.0 - h[:m])
                g[:m] -= h[s:s+m]
        norm = np.linalg.norm(g)
        if norm > 1e-10:
            g /= norm
        grads.append(g)
    return np.array(grads)

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-14
    neg = d < -1e-14
    a_hi = radius
    a_lo = -radius
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, (a_lo + a_hi) / 2.0]
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    width = a_hi - a_lo
    cands.extend(np.linspace(a_lo, a_hi, 48).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _local_opt_lp(h, deadline, r0=0.10, K_list=(3, 9, 24, 70), stall_lim=50, use_lp=True):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rad = r0
    stall = 0
    rng = np.random.default_rng(777)
    it = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        it += 1
        S = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-S)
        best_new = cur
        best_a = 0.0
        best_d = None
        for K in K_list:
            K_actual = min(K, len(S))
            active = top_idx[:K_actual]
            d_try = None
            if use_lp:
                d_try = _lp_direction(h, active, rad)
            if d_try is not None and np.linalg.norm(d_try) > 1e-10:
                a, c_try = _line_search(h, d_try, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d_try
                continue
            d_try = _agg_direction(h, active, rad)
            if d_try is not None:
                a, c_try = _line_search(h, d_try, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d_try
                continue
            d_try = _weighted_agg(h, active, S, rad, beta=12.0)
            if d_try is not None:
                a, c_try = _line_search(h, d_try, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d_try
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.25)
        else:
            stall += 1
            rad *= 0.8
            if rad < 0.005:
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp_val = _exact_c5(hp)
                if cp_val < cur - 1e-12:
                    h = hp
                    cur = cp_val
                    stall = 0
                    rad = 0.10
                elif stall % 15 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_opt(h, deadline, max_it=25000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=12000, Ti=0.002):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _fourier_dir(h, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-10:
        return None
    return d / norm

def _random_subset_active(S, top_frac=2, rng=None):
    n = len(S)
    top_k = min(n // 4, 40 * top_frac)
    top_idx = np.argsort(-S)[:top_k]
    if rng is not None:
        sub_k = max(3, top_k // 2)
        perm = rng.permutation(top_k)
        return top_idx[perm[:sub_k]]
    return top_idx

def _lp_opt_extended(h, deadline, r0=0.10, stall_lim=50):
    """Extended LP optimization with varied active sets and radii."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rad = r0
    stall = 0
    rng = np.random.default_rng(54321)
    K_options = [3, 5, 9, 15, 24, 40, 70]
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        S = np.correlate(h, 1.0 - h, mode='full')
        top_idx_np = np.argsort(-S)
        best_new = cur
        best_a = 0.0
        best_d = None
        n_K = len(K_options)
        K_choose = int(rng.integers(0, n_K))
        K_try_list = [K_options[(K_choose + di) % n_K] for di in range(min(3, n_K))]
        for K in K_try_list:
            K_actual = min(K, len(S))
            if rng.random() < 0.3 and K_actual > 6:
                active = _random_subset_active(S, top_frac=2, rng=rng)
                K_use = len(active)
                if K_use > K_actual:
                    active = active[:K_actual]
            else:
                active = top_idx_np[:K_actual]
            d_try = _lp_direction(h, active, rad)
            if d_try is not None and np.linalg.norm(d_try) > 1e-10:
                a, c_try = _line_search(h, d_try, dx, min(rad, 1.0))
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d_try
            else:
                d_try = _agg_direction(h, active, rad)
                if d_try is not None:
                    a, c_try = _line_search(h, d_try, dx, min(rad, 1.0))
                    if c_try < best_new - 1e-14:
                        best_new = c_try
                        best_a = a
                        best_d = d_try
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.02, 0.20)
        else:
            stall += 1
            rad *= 0.85
            if rad < 0.004:
                pert = rng.normal(0, 0.025, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp_val = _exact_c5(hp)
                if cp_val < cur - 1e-12:
                    h = hp
                    cur = cp_val
                    stall = 0
                    rad = 0.08
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c1 = _local_opt_lp(h_init, d1, r0=0.12, K_list=(3, 9, 24, 70), stall_lim=60, use_lp=True)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Extended LP with varied active sets (15%)
    d2 = t0 + budget_s * 0.35
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _lp_opt_extended(h2, d2, r0=0.10, stall_lim=50)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair optimization (10%)
    d3 = t0 + budget_s * 0.45
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=20000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Binary SA + fractional polish (15%)
    d4 = t0 + budget_s * 0.60
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _bin_sa(hbin, d4, mx=10000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.035)
        hp, cp = _local_opt_lp(hp, sd, r0=0.08, K_list=(3, 9, 24), stall_lim=35, use_lp=True)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.75
    nc = len(best_h)
    offsets = [-8, -4, 4, 8, -12, 12, -6, 6, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.012)
        hr, cr = _local_opt_lp(hr, sd, r0=0.10, K_list=(3, 9, 24), stall_lim=30, use_lp=True)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
            tried.clear()
            tried.add(n_try)

    # Phase 6: Intensive LP (12%)
    d6 = t0 + budget_s * 0.87
    nc = len(best_h)
    for r_init in [0.06, 0.03, 0.015]:
        if time.time() > d6 - 3:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _local_opt_lp(h6, sd, r0=r_init, K_list=(3, 5, 9, 15, 30, 70), stall_lim=50, use_lp=True)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 2:
            h6b = _project(best_h.copy(), nc / 2.0)
            sd_b = min(d6, time.time() + budget_s * 0.015)
            h6b, c6b = _pair_opt(h6b, sd_b, max_it=8000)
            if c6b < best_c:
                best_c = c6b
                best_h = h6b.copy()

    # Phase 7: Basin hopping (8%)
    d7 = deadline
    scale = 0.025
    n_re = 0
    while time.time() < d7 and n_re < 50:
        n_re += 1
        nc = len(best_h)
        pert = _fourier_dir(best_h, rng)
        if pert is None:
            pert = rng.normal(0, 1, nc)
            pert -= np.mean(pert)
            nrm = np.linalg.norm(pert)
            if nrm > 1e-10:
                pert /= nrm
        hp = _project(best_h + scale * pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.005)
        ht, ct = _local_opt_lp(hp, sd, r0=0.04, K_list=(3, 9, 24), stall_lim=15, use_lp=True)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt_lp(hf, deadline, r0=0.01, K_list=(3, 9), stall_lim=20, use_lp=True)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    if deadline - time.time() > 1.5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=4000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
