```python
def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    # Project v onto {x : 0 <= x <= 1, sum(x) = target}
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def pair_line_search(h, i, j, dx):
    """Exact line search for direction d = e_i - e_j"""
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    B = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible alpha range
    alpha_hi = min(1.0 - h[i], h[j])  # h[i]+a <= 1, h[j]-a >= 0
    alpha_lo = max(h[i] - 1.0, -h[j])  # h[i]+a >= 0, h[j]-a <= 1
    # Actually: h[i] + a >= 0 => a >= -h[i]; h[i] + a <= 1 => a <= 1-h[i]
    #           h[j] - a >= 0 => a <= h[j]; h[j] - a <= 1 => a >= h[j]-1
    alpha_lo = max(-h[i], h[j] - 1.0)
    alpha_hi = min(1.0 - h[i], h[j])
    
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, float(np.max(B) * dx)
    
    # Evaluate on grid + golden section
    def f(alpha):
        vals = B + alpha * T - alpha * alpha * Q
        return float(np.max(vals)) * dx
    
    # Grid
    n_grid = 40
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    fvals = np.array([f(a) for a in grid])
    best_idx = int(np.argmin(fvals))
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = grid[max(0, best_idx - 2)]

        optimal = grid[best_idx]
        opt_val = fvals[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = f(c1)
        f2 = f(c2)
        for _ in range(50):
```

```python
def compute_active_gradient(h, corr, n_active=10, tau=0.0):
    """Compute weighted gradient from active lags"""
    n = len(h)
    max_val = float(corr.max())
    
    if tau > 0:
        # Soft weights
        weights = np.exp((corr - max_val) / tau)
        weights /= weights.sum()
    else:
        # Hard: top n_active
        top_idx = np.argsort(corr)[-n_active:]
        weights = np.zeros(2*n-1)
        weights[top_idx] = 1.0 / n_active
    
    g = np.zeros(n)
    center = n - 1
    for r in range(2*n-1):
        if weights[r] < 1e-10:
            continue
        k = r - center  # lag
        w = weights[r]
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            # corr at positive lag k: sum_{a=0}^{n-k-1} h[a+k]*(1-h[a])
            # d/dh[i] for i >= k: (1-h[i-k])
            # d/dh[i] for i <= n-k-1: -h[i+k]
            idx_inc = np.arange(0, n-k)
            g[idx_inc] += w * (-h[idx_inc + k])
            idx_dec = np.arange(k, n)
            g[idx_dec] += w * (1.0 - h[idx_dec - k])
        else:
            ka = -k
            # corr at negative lag: sum_{a=0}^{n-ka-1} h[a]*(1-h[a+ka])
            # d/dh[i] for i < n-ka: (1-h[i+ka])
            # d/dh[i] for i >= ka: -h[i-ka]
            idx_inc = np.arange(0, n-ka)
            g[idx_inc] += w * (1.0 - h[idx_inc + ka])
            idx_dec = np.arange(ka, n)
            g[idx_dec] += w * (-h[idx_dec - ka])
    
    return g
```

Now for pair selection:
- Sort indices by gradient: top ascend (want to increase) and top descend (want to decrease)
- Generate pairs: (inc_candidate, dec_candidate)
- Try each pair, line search, accept if improves

Let me think about the number of pairs to try. If I have top 30 inc and top 30 dec, that's 900 pairs. Each pair takes ~5000 operations, so ~4.5M operations per round. At ~100M ops/sec in numpy, that's about 0.05 seconds per round. So I can do many rounds.

But I should be smarter: after each accepted move, recompute the correlation and gradients. This costs O(n^2) for the correlation. So the full cycle is:
1. Compute corr: O(n^2) = ~32K ops
2. Compute gradient: O(n * n_active) = ~1800 ops
3. Generate and test pairs: ~4.5M ops (but only until first improvement or all tested)
4. Update h, recompute corr: O(n^2)

So each "round" (one accepted move + recomputation) costs about 5M operations. At 100M ops/sec, that's 0.05 sec. In 100 seconds, I can do ~2000 rounds.

But actually, I don't need to recompute corr after every single move. I can do multiple moves and then recompute. But the pairs depend on the gradient, which depends on corr. So I should recompute periodically.

Let me structure the algorithm:
- Outer loop: recompute corr and gradients
  - Inner loop: test pairs until stall or max_pairs
    - For each pair, line search
    - If improvement, accept (but don't recompute corr yet)
    - After K accepted moves, recompute corr

Actually, this is getting complex. Let me simplify: test pairs in order of predicted improvement, accept the first one that actually improves, then recompute. This is the "best-improvement" strategy.

Or even simpler: test all generated pairs, find the best one, accept it, recompute. This is clean but slow.

Let me go with a compromise: test pairs in order, accept the first improvement found, recompute. If no improvement found in all pairs, stall.

2. My approach uses individual pair directions, which gives more flexibility in the search space.
3. The pair directions are 1-parameter families that preserve feasibility exactly.
4. The exact line search along each pair finds the optimal step, not just a small step.

The key advantage: by trying many different pairs, I can navigate the non-convex landscape more effectively. The subgradient method gets stuck when the gradient points in a direction that doesn't lead to improvement (due to the max over lags).

For the binary case, the pair-pivot becomes: swap a 1 at position i with a 0 at position j. The line search alpha is either 0 or 1 (or the maximum feasible). So the "line search" simplifies to just checking if the swap improves C5.

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_line_search(h, i, j, B, dx):
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    omh = 1.0 - h
    T = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(B) * dx)
    f0 = float(np.max(B) * dx)
    n_grid = 32
    grid = np.linspace(a_lo, a_hi, n_grid)
    fvals = np.empty(n_grid)
    for idx in range(n_grid):
        a = grid[idx]
        fvals[idx] = float(np.max(B + a * T - a * a * Q)) * dx
    best_idx = int(np.argmin(fvals))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = grid[max(0, best_idx - 2)]
    hi_b = grid[min(n_grid - 1, best_idx + 1)]
    if hi_b - lo_b < 1e-13:
        opt_a = grid[best_idx]
        opt_v = fvals[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(B + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-15:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(B + c2 * T - c2 * c2 * Q)) * dx
        opt_a = (lo_b + hi_b) / 2.0
        opt_v = float(np.max(B + opt_a * T - opt_a * opt_a * Q)) * dx
    f_lo = float(np.max(B + a_lo * T - a_lo * a_lo * Q)) * dx
    f_hi = float(np.max(B + a_hi * T - a_hi * a_hi * Q)) * dx
    best = min([(opt_a, opt_v), (a_lo, f_lo), (a_hi, f_hi), (0.0, f0)], key=lambda x: x[1])
    return best[0], best[1]

def _compute_grad(h, corr, n_active=15, tau_frac=0.0):
    n = len(h)
    center = n - 1
    max_val = float(corr.max())
    if tau_frac > 0:
        tau = max(1e-6, tau_frac * max(1.0, max_val))
        w = np.exp((corr - max_val) / tau)
        w /= w.sum()
    else:
        top_idx = np.argsort(corr)[-n_active:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / n_active
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        k = r - center
        wt = w[r]
        if k == 0:
            g += wt * (1.0 - 2.0 * h)
        elif k > 0:
            idx1 = np.arange(0, n - k)
            g[idx1] += wt * (-h[idx1 + k])
            idx2 = np.arange(k, n)
            g[idx2] += wt * (1.0 - h[idx2 - k])
        else:
            ka = -k
            idx1 = np.arange(0, n - ka)
            g[idx1] += wt * (1.0 - h[idx1 + ka])
            idx2 = np.arange(ka, n)
            g[idx2] += wt * (-h[idx2 - ka])
    return g

def _pair_pivot_search(h, deadline, max_stall=100, n_top=25, alpha_random=0.05):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    rng = np.random.default_rng(12345)
    tau_sched = [0.0, 0.0, 0.0, 0.005, 0.01, 0.02, 0.04]
    tau_idx = 0
    while time.time() < deadline:
        if stall > max_stall:
            break
        B = np.correlate(h, 1.0 - h, mode='full')
        g = _compute_grad(h, B, n_active=n_top, tau_frac=tau_sched[min(tau_idx, len(tau_sched)-1)])
        tg = g - np.mean(g)
        inc_pool = np.argsort(tg)[:n_top]
        dec_pool = np.argsort(tg)[-n_top:]
        if inc_pool[0] == dec_pool[-1]:
            stall += 1
            if stall % 7 == 0:
                tau_idx = min(tau_idx + 1, len(tau_sched) - 1)
            continue
        accepted = False
        tried = 0
        max_tries = n_top * n_top + 20
        for _ in range(max_tries):
            if time.time() > deadline:
                break
            if rng.random() < alpha_random:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
            else:
                i = int(inc_pool[int(rng.integers(0, n_top))])
                j = int(dec_pool[int(rng.integers(0, n_top))])
            if i == j:
                continue
            tried += 1
            alpha, new_c5 = _pair_line_search(h, i, j, B, dx)
            if new_c5 < cur - 1e-13 and abs(alpha) > 1e-12:
                h = h + alpha * 0
                h[i] += alpha
                h[j] -= alpha
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                tau_idx = 0
                accepted = True
                break
        if not accepted:
            stall += 1
            if stall % 10 == 0:
                alpha_random = min(0.3, alpha_random * 1.3)
            elif alpha_random > 0.05:
                alpha_random *= 0.95
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _greedy_construct(n, seed=0):
    h = np.zeros(n)
    ones_needed = n // 2
    rng = np.random.default_rng(seed)
    order = rng.permutation(n)
    for step in range(ones_needed):
        best_pos = -1
        best_c5 = float('inf')
        for pos in order:
            if h[pos] == 0:
                h[pos] = 1.0
                c5 = _exact_c5(h)
                h[pos] = 0.0
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
        if best_pos >= 0:

    return h

        h_r, c5_r = _pair_pivot_search(h_r, sub_dl, max_stall=40, n_top=20, alpha_random=0.1)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Also try greedy constructions
    for n_try in [n0, n0 - 8, n0 + 8, n0 - 16, n0 + 16]:
        if time.time() > d4:
            break
        if n_try < 100 or n_try > 240:
            continue
        h_g = _greedy_construct(n_try, seed=500 + n_try)
        c5_g = _exact_c5(h_g)
        if c5_g < best_c5 + 0.003:
            sub_dl = min(d4, time.time() + budget_s * 0.015)
            h_g, c5_g = _pair_pivot_search(h_g, sub_dl, max_stall=50, n_top=25, alpha_random=0.1)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 5: Final intensive refinement (10%)
    d5 = deadline - 2
    n_best = len(best_h)

    # Pair-pivot on best
    sub_dl = min(d5, time.time() + budget_s * 0.03)
    h5, c5_5 = _pair_pivot_search(best_h, sub_dl, max_stall=60, n_top=35, alpha_random=0.05)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Basin hopping
    if time.time() < d5 - 10:
        scale = 0.03
        for _ in range(80):
            if time.time() > d5 - 3:
                break
            h_p = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d5, time.time() + budget_s * 0.003)
            h_t, c5_t = _pair_pivot_search(h_p, sub_dl, max_stall=30, n_top=20, alpha_random=0.15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.005, scale * 0.85)
            else:
                scale = min(0.08, scale * 1.1)

    # Final binary pass
    if time.time() < d5 - 8:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_targeted(h_b, sub_dl, max_iters=60000, seed=999)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final nearby n check
    n_fin = len(best_h)
    for dn in [-2, 2, -4, 4]:
        if time.time() > deadline - 3:
            break
        n_try = n_fin + dn
        if n_try < 100:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline - 1, time.time() + 4)
        h_r, c5_r = _pair_pivot_search(h_r, sub_dl, max_stall=30, n_top=20, alpha_random=0.1)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
