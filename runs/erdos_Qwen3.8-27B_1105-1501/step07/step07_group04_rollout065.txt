import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_adj_matrix(h):
    """Build the adjacency/gradient structure for all correlation lags."""
    n = len(h)
    return n

def _grad_row(h, r):
    """Compute gradient of correlation at index r with respect to h."""
    n = len(h)
    center = n - 1
    k = r - center
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # h[j] in first factor: j from k to n-1, contribution (1-h[j-k])
        g[k:] += (1.0 - h[:n-k])
        # h[j] in second factor: j from 0 to n-k-1, contribution -h[j+k]
        g[:n-k] -= h[k:]
    else:
        u = -k
        # h[j] in first factor: j from 0 to n-u-1, contribution (1-h[j+u])
        g[:n-u] += (1.0 - h[u:])
        # h[j] in second factor: j from u to n-1, contribution -h[j-u]
        g[u:] -= h[:n-u]
    return g

def _lp_direction(h, corr, radius, n_active=80):
    """Find LP direction that reduces the maximum correlation."""
    n = len(h)
    dx = 2.0 / n
    m = len(corr)  # 2n-1
    
    # Select active constraints (top correlations)
    sorted_idx = np.argsort(-corr)
    active = sorted_idx[:min(n_active, m)]
    
    # Build constraint matrix
    # For each active constraint r: grad_r . d - t <= -corr_r
    # i.e., grad_r . d - t + corr_r <= 0
    # In standard form: A_ub @ x <= b_ub
    # where x = [d (n), t (1)]
    
    A_rows = []
    b_rows = []
    
    for r in active:
        g = _grad_row(h, int(r))
        row = np.zeros(n + 1)
        row[:n] = g * dx  # scaled gradient
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corr[r] * dx)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Objective: minimize t + eps * sum(d^2 approximated by L1)
    # Actually just minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    # Add small regularization
    c_obj[:n] = 1e-6  # encourage small moves
    
    # Bounds for d and t
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    # t is unbounded above (or bounded by current value)
    cur_c5 = float(np.max(corr) * dx)
    bounds.append((-2.0, cur_c5 + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0],
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[:n].copy()
            # Remove mean to ensure sum = 0 (numerical)
            d = d - np.mean(d)
            if np.max(np.abs(d)) > 1e-12:
                return d
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    """Exact line search along direction d."""
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find max feasible alpha
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search + golden section
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(C0 + t * T - t * t * C)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(C0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(C0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(C0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(C0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    
    opt_val = float(np.max(C0 + opt_t * T - opt_t * opt_t * C)) * dx
    cur_val = float(np.max(C0) * dx)
    end_val = float(np.max(C0 + t_hi * T - t_hi * t_hi * C)) * dx
    
    best_c = min([(opt_t, opt_val), (0.0, cur_val), (t_hi, end_val)], key=lambda x: x[1])
    return best_c[0], best_c[1]

def _subgradient(h, corr):
    """Compute averaged subgradient from active constraints."""
    n = len(h)
    dx = 2.0 / n
    max_val = float(corr.max())
    active = np.where(corr >= max_val - 1e-10)[0]
    g = np.zeros(n)
    for r in active:
        g += _grad_row(h, int(r))
    g /= len(active)
    g *= dx
    return g

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=10000, radius=0.20, seed=111):

            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:

            continue
        d = np.zeros(n)

            stall = 0
