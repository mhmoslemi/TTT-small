    # B[p, m] = [0 <= p - d[m] < n] * h[p - d[m]]
    pd2 = p - d_mat
    valid_pd2 = (pd2 >= 0) & (pd2 < n)
    B_ones = np.zeros((len(ones), L))
    for i in range(len(ones)):
        valid = valid_pd2[i]
        B_ones[i, valid] = h[ones[i] - d[valid]]
    
    # Similarly for zeros
    p = zeros[:, None]
    pd = p + d_mat
    valid_pd = (pd >= 0) & (pd < n)
    A_zeros = np.zeros((len(zeros), L))
    for i in range(len(zeros)):
        valid = valid_pd[i]
        A_zeros[i, valid] = (1.0 - h[zeros[i] + d[valid]])
    
    pd2 = p - d_mat
    valid_pd2 = (pd2 >= 0) & (pd2 < n)
    B_zeros = np.zeros((len(zeros), L))
    for i in range(len(zeros)):
        valid = valid_pd2[i]
        B_zeros[i, valid] = h[zeros[i] - d[valid]]
    
    # For swap (i, j): Δc[m] = -A_ones[i,m] + B_ones[i,m] + A_zeros[j,m] - B_zeros[j,m]
    # new_corr[m] = corr[m] + Δc[m]
    # new_max = max(new_corr)
    
    # For each i, find best j:
    best_max = float(np.max(corr))
    best_i = -1
    best_j = -1
    
    for i in range(len(ones)):
        if time.time() > deadline:
            break
        base = corr - A_ones[i] + B_ones[i]  # shape (L,)
        # For each j: new_corr[m] = base[m] + A_zeros[j,m] - B_zeros[j,m]
        # new_max[j] = max_m(base[m] + A_zeros[j,m] - B_zeros[j,m])
        vals = base[None, :] + A_zeros - B_zeros  # shape (n_zeros, L)
        new_maxs = vals.max(axis=1)  # shape (n_zeros,)
        j_best = int(np.argmin(new_maxs))
        if new_maxs[j_best] < best_max - 1e-12:
            best_max = new_maxs[j_best]
            best_i = i
            best_j = j_best
    
    if best_i < 0:
        return h, corr, False
    
    h_new = h.copy()
    h_new[ones[best_i]] = 0.0
    h_new[zeros[best_j]] = 1.0
    corr_new = _get_corr(h_new)
    return h_new, corr_new, True
