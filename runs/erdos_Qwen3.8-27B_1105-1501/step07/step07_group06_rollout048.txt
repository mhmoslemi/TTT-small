import numpy as np
import time
from scipy.optimize import minimize as scipy_min

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _x_to_h(x, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _antipal_seed(h_full, n):
    m = n // 2
    x = 0.5 * h_full[:m] + 0.5 * (1.0 - h_full[n-1-m:n-1:-1])
    return np.clip(x, 0.0, 1.0)

def _smooth_obj_grad_anti(x, n, beta, k_active=30):
    m = n // 2
    h = _x_to_h(x, n)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -500.0, 0.0)
    ez = np.exp(z)
    total = float(np.sum(ez))
    LSE = float(np.log(total))
    S_raw = M + LSE / beta
    w = ez / total
    # Select top active
    order = np.argsort(w)[::-1]
    active = order[:k_active]
    G = np.zeros(n)
    for r in active:
        wr = w[r]
        if wr < 1e-10:
            break
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:] += wr * (1.0 - h[:mm])
                G[:mm] -= wr * h[tau:tau + mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * (1.0 - h[d:])
                G[d:] -= wr * h[:mm]
    dx_scale = 2.0 / n
    grad_x = (G[:m] - G[n-1:n-1-m:-1]) * dx_scale
    return S_raw * dx_scale, grad_x

    raw = np.correlate(h, u, mode='full')

def _line_search(h, d, radius, n):
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_lo, a_hi
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    fc, fd = f(c), f(dd)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if fc < fd:
            b, dd, fd = dd, c, fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = c, dd, fd
            dd = a + gr * (b - a)
            fd = f(dd)
    opt = (a + b) / 2.0
    opt_v = f(opt)
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt, opt_v = t, ft
    top_idx = np.argsort(S0)[-8:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt, opt_v = av, fv
    return opt, opt_v

def _make_dirs(h, G, n, rng, n_dirs=6):
    m = n // 2
    dirs = []
    d = -G.copy()
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12:
        dirs.append(d / dn)
    # Top raw direction
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    r_max = int(np.argmax(raw))
    tau = r_max - (n - 1)
    if tau >= 0:
        mm = n - tau
        g = np.zeros(n)
        g[tau:] += (1.0 - h[:mm])
        g[:mm] -= h[tau:tau + mm]
    else:
        dd2 = -tau
        mm = n - dd2
        g = np.zeros(n)
        g[:mm] += (1.0 - h[dd2:])
        g[dd2:] -= h[:mm]
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        dirs.append(-g / gn)
    # Fourier modes
    x = np.arange(n, dtype=float)
    for freq in [1, 2, 3]:
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    # Block direction
    blk = 4
    if n % blk == 0:
        nblk = n // blk
        gb = G.reshape(nblk, blk).mean(axis=1)
        d = np.tile(-gb, (1, blk)).reshape(n)
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    return dirs[:n_dirs]

def _resample_to_n(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_init(n, n_modes, rng, base=0.5):
    x = np.arange(n, dtype=float)
    h = np.full(n, base)
    for i in range(n_modes):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4) / (i + 1)
        h += amp * np.sin(2.0 * np.pi * freq * x / n + phase)

        h_p = _perturb(best_h, scale, n_best, rng)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.012)
        # Try anti-palindromic
        m = n_best // 2
        x_seed = _antipal_seed(h_p, n_best)
        sub_dl_anti = min(sub_dl, time.time() + 3.0)
        x_opt, c5_a = _optimize_anti_bfgs(x_seed, n_best, 30.0, sub_dl_anti, maxiter=100)
        h_a = _x_to_h(x_opt, n_best)
        # Also try full
        sub_dl_full = min(sub_dl, time.time() + 3.0)
        h_f, c5_f = _optimize_full_smooth(h_p, n_best, sub_dl_full, beta_lo=20.0, beta_hi=80.0, max_stall=25)
        c5_t = min(c5_a, c5_f)
        h_t = h_a if c5_a <= c5_f else h_f
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 5: Final n-sweep and polish (10%)
    d5 = deadline
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > d5 - 10:
            break
        nt = n_final + dn
        if 80 <= nt <= 260 and nt % 2 == 0:
            h_r = _resample_to_n(best_h, nt)
            sub_dl = min(d5 - 5, time.time() + budget_s * 0.02)
            m = nt // 2
            x_seed = _antipal_seed(h_r, nt)
            x_opt, c5_a = _optimize_anti_bfgs(x_seed, nt, 30.0, sub_dl, maxiter=150)
            h_a = _x_to_h(x_opt, nt)
            c5_a = _exact_c5(h_a)
            if time.time() < d5 - 5:
                h_f, c5_f = _optimize_full_smooth(h_r, nt, min(d5 - 3, time.time() + 8.0), beta_lo=20.0, beta_hi=80.0, max_stall=30)
            else:
                c5_f = c5_a
                h_f = h_a
            c5_t = min(c5_a, c5_f)
            h_t = h_a if c5_a <= c5_f else h_f
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                n_final = nt

    # Final polish
    if deadline - time.time() > 5:
        n_f = len(best_h)
        h_f = _project(best_h.copy(), n_f / 2.0)
        h_f, c5_f = _optimize_full_smooth(h_f, n_f, deadline, beta_lo=30.0, beta_hi=120.0, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # One more anti-pal pass
        if deadline - time.time() > 3:
            m = n_f // 2
            x_seed = _antipal_seed(best_h, n_f)
            x_opt, c5_a = _optimize_anti_bfgs(x_seed, n_f, 50.0, deadline, maxiter=200)
            h_a = _x_to_h(x_opt, n_f)
            c5_a = _exact_c5(h_a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
