def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search_max_quadratic(R0, A, B, t_lo, t_hi):
    """Minimize max_r (R0[r] + t*A[r] - t^2*B[r]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R0))
    
    # Grid initial search
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1*c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2*c2 * B))

            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1*c1 * B))
        else:

    # Check endpoints and grid best too

        if v < best_val:

    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.max(-h[neg] / d[neg])))
    
    return max(t_lo, -max_t), min(t_hi, max_t)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _dense_optimize(h, deadline, beta0=8.0, beta_max=50.0, max_stall=80, max_t=0.3, seed=42):
    """Dense gradient-based optimization."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = _exact_corr(h)
    rng = np.random.default_rng(seed)
    
    beta = beta0
    stall = 0
    t_scale = max_t
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        
        # Compute smooth gradient
        grad, w = _smooth_grad(h, R, beta)
        
        # Project to zero-sum (tangent direction)
        d = grad - np.mean(grad)
        norm = np.sqrt(np.sum(d**2))
        if norm < 1e-12:
            break
        d = d / norm * t_scale
        
        # Try both directions (positive and negative)
        best_step = None
        best_c5_try = cur_c5
        
        for sign in [1.0, -1.0]:
            ds = sign * d
            t_lo, t_hi = _feasible_range(h, ds, max_t)
            if t_hi - t_lo < 1e-10:
                continue
            
            # Compute A and B for the line
            A = np.correlate(ds, 1.0 - h, mode='full') - np.correlate(h, ds, mode='full')
            B = np.correlate(ds, ds, mode='full')
            
            t_opt, val_opt = _line_search_max_quadratic(R, A, B, t_lo, t_hi)
            
            h_try = h + t_opt * ds
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            
            if c5_try < best_c5_try - 1e-15:
                best_c5_try = c5_try
                best_step = h_try
        
        if best_step is not None:
            h = best_step
            cur_c5 = best_c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            R = _exact_corr(h)
        else:
            stall += 1
            beta = min(beta * 1.3, beta_max)
            t_scale *= 0.85
            if t_scale < 0.01:
                # Try random perturbations
                d_rand = rng.standard_normal(n)
                d_rand -= np.mean(d_rand)
                norm_r = np.sqrt(np.sum(d_rand**2))
                if norm_r > 1e-12:
                    d_rand = d_rand / norm_r * t_scale
                    t_lo, t_hi = _feasible_range(h, d_rand, max_t)
                    if t_hi - t_lo > 1e-10:
                        A = np.correlate(d_rand, 1.0-h, mode='full') - np.correlate(h, d_rand, mode='full')
                        B = np.correlate(d_rand, d_rand, mode='full')
                        t_opt, _ = _line_search_max_quadratic(R, A, B, t_lo, t_hi)
                        h_try = np.clip(h + t_opt * d_rand, 0.0, 1.0)
                        h_try = _project(h_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try
                            cur_c5 = c5_try
                            R = _exact_corr(h)
                            stall = 0
                t_scale = 0.2
    
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=80, seed=42):
    """Pairwise optimization (fallback / local refinement)."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = _exact_corr(h)
    stall = 0
    rng = np.random.default_rng(seed)
    radius = float(radius)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        
        # Weighted gradient for pair selection
        M = float(np.max(R))
        deltas = M - R
        w = np.exp(-beta * deltas)
        s = float(np.sum(w))
        if s < 1e-300:
            top_k = max(1, (2*n-1) // 10)
            w = np.zeros(2*n-1)
            w[np.argsort(R)[-top_k:]] = 1.0 / top_k
        else:
            w /= s
        
        grad = np.zeros(n)
        for idx in range(2*n-1):
            wi = w[idx]
            if wi < 1e-15:
                continue
            lag = idx - (n-1)
            m = n - abs(lag)
            if lag >= 0:
                grad[lag:lag+m] += wi * (1.0 - h[:m])
                grad[:m] -= wi * h[lag:lag+m]
            else:
                p = -lag
                grad[:m] += wi * (1.0 - h[p:p+m])
                grad[p:p+m] -= wi * h[:m]
        
        # Select pairs
        sort_idx = np.argsort(grad)
        n_half = min(n_pairs, n // 2)
        sources = sort_idx[-n_half:]
        sinks = sort_idx[:n_half]
        pairs = []
        for ii in range(n_half):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        # Add random pairs
        for _ in range(min(15, n_pairs // 3)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            B = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search_max_quadratic(R, A, B, t_lo, t_hi)
            
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            R = _exact_corr(h)
            stall = 0
            radius = min(radius * 1.01, 0.3)
        else:
            stall += 1
            radius *= 0.78
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.12
                beta = min(beta * 1.5, 80.0)
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, T_init=0.005, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_greedy(h_bin, deadline):
    """Greedy pairwise swap for binary."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones[:]:
            if time.time() > deadline:
                break
            for oz in zeros[:]:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5
