import numpy as np
import time
import scipy.optimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, r, n):
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    a_lo, a_hi = -radius, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    grid = np.linspace(a_lo, a_hi, 50)
    gf = np.empty(len(grid))
    for i in range(len(grid)):
        gf[i] = f(grid[i])
    bi = int(np.argmin(gf))
    lo = grid[max(0, bi - 3)]
    hi = grid[min(len(grid) - 1, bi + 3)]

    try:
        res = scipy.optimize.minimize_scalar(f, bounds=(lo, hi), method='bounded',
                                            options={'xatol': 1e-15, 'maxiter': 300})
        best_a = float(res.x)
        best_v = float(res.fun)
    except Exception:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1, f2 = f(c1), f(c2)
        for _ in range(80):
            if hi - lo < 1e-15:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = f(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = f(c2)
        best_a = (lo + hi) / 2.0
        best_v = f(best_a)

    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < best_v:
            best_a, best_v = t, ft
    top_idx = np.argsort(S0)[-10:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < best_v:
                    best_a, best_v = av, fv
    return best_a, best_v

                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Main refined descent (28%)
    d1 = t0 + budget_s * 0.28
    h1, c5_1 = _opt_refined(h_init, d1, beta_lo=4.0, beta_hi=50.0, radius_init=0.16, max_stall=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary optimization (20%)
    d2 = t0 + budget_s * 0.48
    h_b = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=70000, T_init=0.005)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d2 - 5:
        sub_dl2 = min(d2, time.time() + budget_s * 0.06)
        h_b, c5_b = _greedy_binary(h_b, sub_dl2, max_passes=30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d2:
        h_pol = _project(h_b.copy(), len(h_b) / 2.0)
        sub_dl3 = min(d2, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _opt_refined(h_pol, sub_dl3, beta_lo=10.0, beta_hi=60.0, radius_init=0.10, max_stall=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: N-sweep (20%)
    d3 = t0 + budget_s * 0.68
    n_curr = len(best_h)
    for dn in [-8, -4, 4, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.018)
        h_r, c5_r = _opt_refined(h_r, sub_dl, beta_lo=8.0, beta_hi=40.0, radius_init=0.12, max_stall=50)
        if time.time() < d3 - 3:
            h_br = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_br, c5_br = _binary_sa(h_br, sub_dl2, max_iter=15000, T_init=0.003)
            if c5_br < c5_r:
                c5_r = c5_br
                h_r = h_br
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive continuous optimization (20%)
    d4 = t0 + budget_s * 0.88
    n_c = len(best_h)
    h4 = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.07)
    h4, c5_4 = _opt_refined(h4, sub_dl, beta_lo=15.0, beta_hi=80.0, radius_init=0.10, max_stall=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _pair_continuous(h4, sub_dl2, n_iter=6000, radius=0.15, seed=999)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _opt_refined(h4, sub_dl3, beta_lo=30.0, beta_hi=120.0, radius_init=0.06, max_stall=50)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping + final polish (12%)
    d5 = deadline
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 - 15 and n_restarts < 100:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 8, time.time() + budget_s * 0.007)
        h_t, c5_t = _opt_refined(h_p, sub_dl, beta_lo=10.0, beta_hi=60.0, radius_init=0.07, max_stall=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.12)

    # Final: try nearby n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5 - 10:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.012)
        h_r, c5_r = _opt_refined(h_r, sub_dl, beta_lo=15.0, beta_hi=60.0, radius_init=0.07, max_stall=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_refined(h_f, deadline, beta_lo=20.0, beta_hi=100.0, radius_init=0.04, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
