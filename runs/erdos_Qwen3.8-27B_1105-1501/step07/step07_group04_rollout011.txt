def _binary_swap_delta(h, i, j):
    """Compute the change in correlation vector when h[j]: 1->0, h[i]: 0->1.
    h is binary. Returns delta array of length 2n-1."""
    n = len(h)
    delta = np.zeros(2*n-1)
    # For each shift s (can be negative), compute the contribution change
    # shift = idx - (n-1), so idx ranges from 0 to 2n-2, shift from -(n-1) to n-1
    
    # j going from 1 to 0:
    # As h-position: contributes at shifts where j-shift is in [0, n-1] and j is in [shift, n-1+shift]... 
    # Let me use the formulation: corr[idx] = sum over valid I of h[I] * (1-h[I-shift])
    # where shift = idx - (n-1)
    
    # Actually let me just think in terms of shift directly.
    # For shift s: corr at index (n-1+s) = sum_{I: 0<=I<n, 0<=I-s<n} h[I] * (1-h[I-s])
    
    # j: 1->0
    # As h[I] with I=j: valid if 0 <= j-s < n, i.e., -j < s <= n-1-j, i.e., s in [-(j), n-1-j] wait
    # 0 <= I-s < n means 0 <= j-s < n, so s <= j and j-s < n i.e. s > j-n

    # Contribution: h[j]*(1-h[j-s]) = 1*(1-h[j-s]) = 1-h[j-s] (old), 0 (new)
    # Delta: -(1-h[j-s]) for s in [j-n+1, j]
    
    # As (1-h[I-s]) with I-s=j, i.e., I=j+s: valid if 0 <= j+s < n, so s in [-j, n-1-j]
    # Contribution: h[j+s]*(1-h[j]) = h[j+s]*0 = 0 (old)

    # i: 0->1

    # Old: 0, New: 1*(1-h[i-s]) = 1-h[i-s]
    # Delta: +(1-h[i-s]) for s in [i-n+1, i]

    # Old: h[i+s]*(1-0) = h[i+s], New: h[i+s]*(1-1) = 0
    # Delta: -h[i+s] for s in [-i, n-1-i]

    # j removal as h: s in [max(j-n+1, -(n-1)), min(j, n-1)]
    # j addition as (1-h): s in [max(-j, -(n-1)), min(n-1-j, n-1)]
    # i addition as h: s in [max(i-n+1, -(n-1)), min(i, n-1)]
    # i removal as (1-h): s in [max(-i, -(n-1)), min(n-1-i, n-1)]
    
    # Let me just loop over all shifts and compute:
    # This would be O(n) which is fine.
    
    for s in range(-(n-1), n):  # all possible shifts
        d = 0.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _swap_delta(h, i, j):
    """O(n) delta computation for binary swap h[j]:1->0, h[i]:0->1"""
    n = len(h)
    delta = np.zeros(2*n-1)
    for s in range(-(n-1), n):
        d = 0.0
        # j: 1->0 as h position (I=j, need 0<=j-s<n)
        idx = j - s
        if 0 <= idx < n:
            d -= (1.0 - h[idx])
        # j: 1->0 as (1-h) position (I=j+s, need 0<=j+s<n)
        idx = j + s
        if 0 <= idx < n:
            d += h[idx]
        # i: 0->1 as h position (I=i, need 0<=i-s<n)
        idx = i - s
        if 0 <= idx < n:
            d += (1.0 - h[idx])
        # i: 0->1 as (1-h) position (I=i+s, need 0<=i+s<n)
        idx = i + s
        if 0 <= idx < n:
            d -= h[idx]
        delta[n-1+s] = d
    return delta
