def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

    # Initialize correlations (all zero since h is all zero)
    R = np.zeros(2*n - 1)
    # Actually R_l = sum_j h[j](1-h[j+l]) = 0 when h=0
    
    for step in range(target):
        if step % 100 == 0:
            pass
        best_i = -1
        best_max = float('inf')
        for i in range(n):
            if h[i] >= 1.0:
                continue
            # Compute new max if we set h[i] = 1
            # ΔR_l = (1-h[i+l]) - h[i-l] for valid l
            new_R = R.copy()
            for l in range(-(n-1), n):
                tl = n - 1 + l
                # Check if i is a valid index for this lag
                # First factor: i+l must be in [0, n-1]
                if 0 <= i+l < n:
                    new_R[tl] += (1 - h[i+l])
                # Second factor: i-l must be in [0, n-1]
                if 0 <= i-l < n:
                    new_R[tl] -= h[i-l]
            mx = float(np.max(new_R))
            if mx < best_max:
                best_max = mx
                best_i = i
        h[best_i] = 1.0
        # Update R
        for l in range(-(n-1), n):
            tl = n - 1 + l
            if 0 <= best_i+l < n:
                R[tl] += (1 - h[best_i+l])
            if 0 <= best_i-l < n:
                R[tl] -= h[best_i-l]
    
    return h

                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.1
                elif stall % 20 == 0:
                    rad = 0.06
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        best_h = _project(h0.copy(), n0 / 2.0)
    except NameError:
        n0 = 180
        best_h = _greedy_binary(n0, seed)

    best_c = _exact_c5(best_h)
    n = len(best_h)

    # Phase 1: FW optimization (22%)
    d1 = t0 + budget_s * 0.22
    h1, c1 = _fw_opt(_project(best_h.copy(), n / 2.0), d1,
                     betas=(20.0, 40.0, 80.0, 160.0, 100.0))
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Guided pair optimization (13%)
    d2 = t0 + budget_s * 0.35
    h2, c2 = _pair_opt_guided(_project(best_h.copy(), n / 2.0), d2, max_it=40000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA with double flips (13%)
    d3 = t0 + budget_s * 0.48
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa_double(hbin, d3, mx=12000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 2:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _fw_opt(hp, sd, betas=(30.0, 60.0, 120.0))
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-direction local opt (15%)
    d4 = t0 + budget_s * 0.63
    h4, c4 = _multi_dir_opt(_project(best_h.copy(), n / 2.0), d4,
                            r0=0.12, stall_lim=50)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (17%)
    d5 = t0 + budget_s * 0.80
    nc = len(best_h)
    offsets = [-6, -4, -2, 2, 4, 6, -10, 10, -14, 14]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 250 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.02)
        hr, cr = _fw_opt(hr, sd, betas=(30.0, 60.0, 120.0, 50.0))
        if cr < best_c + 0.003 and time.time() < d5 - 3:
            sd2 = min(d5, time.time() + budget_s * 0.01)
            hr2, cr2 = _pair_opt_guided(_project(hr.copy(), n_try / 2.0), sd2, max_it=8000)
            if cr2 < cr:
                cr = cr2
                hr = hr2
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive final optimization (15%)
    d6 = deadline
    nc = len(best_h)
    for round_i in range(4):
        if time.time() > d6 - 3:
            break
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _pair_opt_guided(_project(best_h.copy(), nc / 2.0), sd, max_it=15000)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 5:
            sd2 = min(d6, time.time() + budget_s * 0.03)
            h6, c6 = _fw_opt(_project(best_h.copy(), nc / 2.0), sd2,
                             betas=(40.0, 80.0, 160.0, 25.0))
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()
        if time.time() < d6 - 3:
            sd3 = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _multi_dir_opt(_project(best_h.copy(), nc / 2.0), sd3,
                                    r0=0.05, stall_lim=30)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt_guided(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
