        return np.zeros(n, dtype=float)

def _swap_delta_fast(h, a, b):
    """Compute correlation delta for binary swap: h[a]: 1->0, h[b]: 0->1. O(n)."""
    n = len(h)
    L = 2 * n - 1
    delta = np.zeros(L, dtype=np.float64)
    # For lag k, delta[center+k] = h[a+k] + h[a-k] - h[b+k] - h[b-k]
    # where h[j] = 0 for j out of [0, n-1]
    center = n - 1
    # Vectorized:
    ks = np.arange(-(n-1), n, dtype=np.intp)  # lags
    # a+k
    idx1 = a + ks
    m1 = (idx1 >= 0) & (idx1 < n)
    delta[m1] += h[idx1[m1]]
    # a-k
    idx2 = a - ks
    m2 = (idx2 >= 0) & (idx2 < n)
    delta[m2] += h[idx2[m2]]
    # b+k
    idx3 = b + ks
    m3 = (idx3 >= 0) & (idx3 < n)
    delta[m3] -= h[idx3[m3]]
    # b-k
    idx4 = b - ks
    m4 = (idx4 >= 0) & (idx4 < n)
    delta[m4] -= h[idx4[m4]]
    return delta

def _find_contributing(h, k):
    """For lag k, find positions i where h[i]=1 and h[i+k]=0 (positive lag)."""
    n = len(h)
    if k > 0:
        mask = (h[:n-k] > 0.5) & (h[k:] <= 0.5)
        return np.where(mask)[0].tolist()
    elif k < 0:
        ka = -k
        mask = (h[ka:] > 0.5) & (h[:n-ka] <= 0.5)
        return (np.where(mask)[0] + ka).tolist()
    else:
        return []

def _continuous_optimize(h, deadline, radius_init=0.10, max_iter=3000):

        best_c5 = float('inf')
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = 1.0
                c5 = _exact_c5(h)
                h[pos] = 0.0
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h

def _periodic_pattern(n, period, frac):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        pos_in_period = i % period
        if pos_in_period < int(period * frac):
            h[i] = 1.0
    return _project(h, n / 2.0)

def _random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    # Initialize from parent
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float).flatten()
            n0 = len(h0)
            h_init = _project(h0.copy(), n0 / 2.0)
        except:
            n0 = 180
            x = (np.arange(n0) + 0.5) / n0
            base = 0.5 + 0.18 * np.sin(2 * np.pi * x) - 0.08 * np.cos(4 * np.pi * x)
            h_init = _project(base, n0 / 2.0)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Continuous optimization (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _continuous_optimize(h_init, d1, radius_init=0.06, max_iter=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary optimization (52%)
    d2 = t0 + budget_s * 0.60
    h_bin = _to_binary(best_h)
    n0 = len(h_bin)

    # First: systematic attack from the initial binary
    sub_dl = min(d2, time.time() + budget_s * 0.18)
    h_b, c5_b = _binary_attack(h_bin, sub_dl, seed=1000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Second round with different seed
    sub_dl = min(d2, time.time() + budget_s * 0.12)
    h_b, c5_b = _binary_attack(h_b.copy(), sub_dl, seed=2000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Try a different starting binary (complement shift)
    h_alt = _to_binary(np.roll(best_h, n0 // 3))
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_b, c5_b = _binary_attack(h_alt, sub_dl, seed=3000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Third attack round on best
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_cur = _to_binary(best_h)
    h_b, c5_b = _binary_attack(h_cur, sub_dl, seed=4000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Continuous polish on best binary
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_pol = _project(best_h.copy(), len(best_h) / 2.0)
    h_pol, c5_pol = _continuous_optimize(h_pol, sub_dl, radius_init=0.04, max_iter=1500)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.85
    tried_ns = set([len(best_h)])
    n_base = len(best_h)

    # Try nearby n values
    for dn in [-6, -4, -3, -2, 2, 3, 4, 6, 8, 12, -8, -12]:
        if time.time() > d3:
            break
        n_try = n_base + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample_h(best_h, n_try)
        # Continuous polish
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _continuous_optimize(h_r, sub_dl, radius_init=0.06, max_iter=800)
        # Binary optimization
        if time.time() < d3 - 10:
            h_rb = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.02)
            h_rb, c5_rb = _binary_attack(h_rb, sub_dl2, seed=5000 + n_try)
            if c5_rb < c5_r:
                c5_r = c5_rb
                h_r = h_rb
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_base = n_try

    # Try greedy construction for a couple of n values
    for n_try in [n_base, n_base - 10, n_base + 10]:
        if time.time() > d3 - 15:
            break
        if n_try < 80 or n_try > 260:
            continue
        h_g = _greedy_construction(n_try, rng)
        c5_g = _exact_c5(h_g)
        if c5_g < best_c5 + 0.01:
            sub_dl = min(d3 - 5, time.time() + budget_s * 0.02)
            h_g, c5_g = _binary_attack(h_g, sub_dl, seed=6000 + n_try)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Try periodic patterns
    for period in [7, 9, 11, 13, 17]:
        if time.time() > d3 - 10:
            break
        n_try = len(best_h)
        h_p = _periodic_pattern(n_try, period, 0.5)
        c5_p = _exact_c5(h_p)
        if c5_p < best_c5 + 0.01:
            sub_dl = min(d3 - 5, time.time() + budget_s * 0.015)
            h_pb = _to_binary(h_p)
            h_pb, c5_pb = _binary_attack(h_pb, sub_dl, seed=7000 + period)
            if c5_pb < best_c5:
                best_c5 = c5_pb
                best_h = h_pb.copy()

    # Phase 4: Final intensive refinement (15%)
    d4 = deadline
    n_best = len(best_h)

    # More binary attack rounds on the best
    for rnd in range(4):
        if time.time() > d4 - 10:
            break
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.03)
        h_cur = _to_binary(best_h)
        h_b, c5_b = _binary_attack(h_cur, sub_dl, seed=8000 + rnd * 100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Continuous polish final
    if time.time() < d4 - 10:
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.04)
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        h_pol, c5_pol = _continuous_optimize(h_pol, sub_dl, radius_init=0.03, max_iter=2000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Try nearby n one more time
    n_fin = len(best_h)
    for dn in [-2, 2]:
        if time.time() > d4 - 8:
            break
        n_try = n_fin + dn
        if n_try < 60:
            continue
        h_r = _resample_h(best_h, n_try)
        sub_dl = min(d4 - 2, time.time() + 4.0)
        h_r, c5_r = _continuous_optimize(h_r, sub_dl, radius_init=0.03, max_iter=500)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        else:
            h_rb = _to_binary(h_r)
            sub_dl2 = min(d4 - 2, time.time() + 3.0)
            h_rb, c5_rb = _binary_attack(h_rb, sub_dl2, seed=9000 + n_try)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Re-evaluate without projection issues
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
