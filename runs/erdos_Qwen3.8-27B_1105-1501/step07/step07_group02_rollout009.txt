import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_mh = 1.0 - h
    for l in range(-(n-1), n):
        row = (n-1) + l
        if l >= 0:
            G[row, :n-l] += one_mh[l:]
            G[row, l:] -= h[:n-l]
        else:
            p = -l
            G[row, p:] += one_mh[:n-p]
            G[row, :n-p] -= h[p:]
    return G

def _softmax_weights(R, beta):
    M = float(np.max(R))
    w = np.exp(beta * (R - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(R) // 10)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _line_search_dense(R, A, B, t_lo, t_hi):
    """Find t in [t_lo, t_hi] minimizing max_k(R[k] + t*A[k] - t^2*B[k])"""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R))
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R + c1 * A - c1*c1 * B))
    f2 = float(np.max(R + c2 * A - c2*c2 * B))
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R + c1 * A - c1*c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R + c2 * A - c2*c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(R + opt_t * A - opt_t*opt_t * B))
    best_val = min(opt_val, float(vals[best_idx]))

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    omh = 1.0 - h
    for l in range(-(n-1), n):
        row = (n - 1) + l
        if l >= 0:
            G[row, :n-l] += omh[l:]
            G[row, l:] -= h[:n-l]
        else:
            p = -l
            G[row, p:] += omh[:n-p]
            G[row, :n-p] -= h[p:]
    return G

def _line_search(R, A, B, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R))
    n_grid = 60
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R + c1 * A - c1*c1 * B))
    f2 = float(np.max(R + c2 * A - c2*c2 * B))
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1

        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R + c2 * A - c2*c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(R + opt_t * A - opt_t*opt_t * B))

    return float(ts[best_idx]), float(vals[best_idx])
