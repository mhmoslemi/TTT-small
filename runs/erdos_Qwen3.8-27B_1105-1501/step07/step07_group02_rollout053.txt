def _project(v, target):
    """Project v onto {x : 0≤x≤1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(S)
    if beta * (M - M) > 500:  # numerical stability
        return M
    expS = np.exp(beta * (S - M))
    logsum = M + np.log(np.sum(expS)) / beta
    return logsum

import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(S)
    shifted = beta * (S - M)
    max_shift = np.max(shifted)
    if max_shift > 500:
        expS = np.exp(shifted - max_shift)
        logsum = M + max_shift / beta + np.log(np.sum(expS)) / beta
        w = expS / np.sum(expS)
    else:
        expS = np.exp(shifted)
        logsum = M + np.log(np.sum(expS)) / beta
        w = expS / np.sum(expS)
    L = len(S)
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(L):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        lag = k - n_half
        if lag >= 0:
            m = n - lag
            if m <= 0:

            grad[:m] += wk * (1.0 - h[lag:])
            grad[lag:] += -wk * h[:m]
        else:
            p = -lag
            m = n - p
            if m <= 0:
                continue
            grad[p:] += wk * (1.0 - h[:m])
            grad[:m] += -wk * h[p:]
    return logsum, grad

        return v

        h = lx[:n]
        _, g = _lse_val_grad(h, beta)

            f, x0, jac=jac, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        if hasattr(result, 'x') and result.x is not None:
            h_sol = _project(result.x[:n].copy(), target)
            c5_sol = _exact_c5(h_sol)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol
    except Exception:
        pass
    if time.time() > deadline:
        return best_h, best_c5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    n_half = n - 1
    G = np.zeros((L, n))
    for k in range(L):
        lag = k - n_half
        if lag >= 0:
            m = n - lag
            if m <= 0:
                continue
            G[k, :m] += (1.0 - h[lag:])
            G[k, lag:] += -h[:m]
        else:
            p = -lag
            m = n - p
            if m <= 0:
                continue
            G[k, p:] += (1.0 - h[:m])
            G[k, :m] += -h[p:]
    return G

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    best_val = min(opt_val, float(vals[best_idx]))
    if best_val == opt_val:
        return opt_t, best_val
    return float(ts[best_idx]), best_val

def _smooth_weights(S, beta=30.0):
    M = float(np.max(S))
    if M < 1e-15:
        return np.ones(len(S)) / len(S)
    w = np.exp(beta * (S - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(S) // 10)
        idx = np.argsort(S)[-top_k:]
        w = np.zeros(len(S))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=25.0, stall_limit=100, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        w = _smooth_weights(S0, beta)
        G = _make_grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        n_p = min(n_pairs, n // 2)
        sources = sort_idx[-n_p:]
        sinks = sort_idx[:n_p]
        pairs = []
        for ii in range(n_p):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, max(3, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-15:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.75
            if stall % 50 == 0 and radius < 0.02:
                radius = 0.12
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=25000, T_init=0.0025):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            stall += 1
    return best_h, best_c5

def _binary_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.where(np.arange(n0) < n0 // 2, 1.0, 0.0)
        h0 = _project(h0, n0 / 2.0)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE-SLSQP at parent n (25% time)
    d1 = t0 + budget_s * 0.25
    h_curr = h_init.copy()
    for beta in [30, 80, 150, 300]:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.05)
        h_curr, c5 = _lse_slsqp(h_curr, beta, sub_dl, maxiter=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()

    # Phase 2: Multi-n exploration with LSE (30% time)
    d2 = t0 + budget_s * 0.55
    n_base = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n_base + dn
        if n_try < 64 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.025)
        for beta in [60, 200]:
            if time.time() > sub_dl:
                break
            h_r, c5_r = _lse_slsqp(h_r, beta, sub_dl, maxiter=50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Pair optimization on best (20% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    h3 = _project(best_h.copy(), n_curr / 2.0)
    h3, c5_3 = _pair_optimize(h3, d3, radius=0.15, n_pairs=50, beta=25.0, stall_limit=120, seed=111)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary diversification (15% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    if time.time() < d4:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d4, time.time() + budget_s * 0.07)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=30000, T_init=0.003)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        sub_dl2 = min(d4, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on best binary
        if time.time() < d4:
            n_cur = len(best_h)
            h_pol = _project(best_h.copy(), n_cur / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.03)
            for bv in [100, 250]:
                if time.time() > sub_dl3:
                    break
                h_pol, c5_pol = _lse_slsqp(h_pol, bv, sub_dl3, maxiter=40)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Final refinement (10% time)
    d5 = deadline - 2
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d5, deadline)
    h5, c5_5 = _pair_optimize(h5, sub_dl, radius=0.08, n_pairs=50, beta=35.0, stall_limit=80, seed=777)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Last LSE polish
    if time.time() < deadline - 3:
        n_cur = len(best_h)
        h_l = _project(best_h.copy(), n_cur / 2.0)
        h_l, c5_l = _lse_slsqp(h_l, 200, deadline - 3, maxiter=40)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
