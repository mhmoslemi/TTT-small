import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import time

# Global state for SLSQP
_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 1.0
_g_h = None  # not needed, SLSQP passes x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # Zero lag
    G[n-1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        # Positive lag: row = n-1+d
        G[n-1+d, :m] += (1.0 - h[d:d+m])
        G[n-1+d, d:] += -h[:m]
        # Negative lag: row = n-1-d
        G[n-1-d, d:] += (1.0 - h[:m])
        G[n-1-d, :m] += -h[d:]
    return G

def _lse_objective(x):
    n = _g_n
    tau = _g_tau
    raw = np.correlate(x, 1.0 - x, mode='full')
    m_max = float(np.max(raw))
    exp_arg = (raw - m_max) / tau
    w = np.exp(exp_arg)
    sw = float(np.sum(w))
    lse = m_max + tau * np.log(sw)
    return float(lse)

def _lse_gradient(x):
    n = _g_n
    tau = _g_tau
    raw = np.correlate(x, 1.0 - x, mode='full')
    m_max = float(np.max(raw))
    exp_arg = (raw - m_max) / tau
    w = np.exp(exp_arg)
    sw = float(np.sum(w))
    p = w / sw
    G = _grad_matrix(x)
    return G.T @ p

def _eq_constraint(x):
    return sum(x) - _g_target

def _eq_jacobian(x):
    return np.ones((1, len(x)))

def _slsqp_solve(x0, tau, maxiter=100):
    global _g_n, _g_target, _g_tau
    _g_n = len(x0)
    _g_target = _g_n / 2.0
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    constraints = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian}]
    res = minimize(_lse_objective, x0, jac=_lse_gradient, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success or res.x is not None:
        return _project_sum(res.x, _g_target)
    return _project_sum(x0, _g_target)

def _lse_anneal(h, deadline, tau_start=10.0, tau_end=0.15, n_tau=7):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    taus = np.geomspace(tau_start, tau_end, n_tau)
    for tau in taus:
        if time.time() > deadline:

        x_new = _slsqp_solve(h, tau, maxiter=80)
        c5_new = _exact_c5(x_new)
        if c5_new < best_c5:
            best_h = x_new.copy()
            best_c5 = c5_new
        # Always continue from the new point (even if slightly worse) for exploration
        h = x_new
    return best_h, best_c5

            best_c5 = c5_new

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_tau = 1.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[n - 1 + d, :m] += (1.0 - h[d:d + m])
        G[n - 1 + d, d:] += -h[:m]
        G[n - 1 - d, d:] += (1.0 - h[:m])
        G[n - 1 - d, :m] += -h[d:]
    return G

def _lse_objective(x):
    tau = _g_tau
    raw = np.correlate(x, 1.0 - x, mode='full')
    m_max = float(np.max(raw))
    w = np.exp((raw - m_max) / tau)
    return float(m_max + tau * np.log(float(np.sum(w))))

def _lse_gradient(x):
    tau = _g_tau
    raw = np.correlate(x, 1.0 - x, mode='full')
    m_max = float(np.max(raw))
    w = np.exp((raw - m_max) / tau)
    sw = float(np.sum(w))
    p = w / sw
    G = _grad_matrix(x)
    return G.T @ p

def _eq_fun(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones((1, len(x)))

def _slsqp_solve(x0, tau, maxiter=100):
    global _g_n, _g_target, _g_tau
    _g_n = len(x0)
    _g_target = _g_n / 2.0
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lse_objective, x0, jac=_lse_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        x_new = res.x
    except Exception:
        x_new = x0
    return _project_sum(x_new, _g_target)

def _lse_anneal(h, deadline, tau_start=12.0, tau_end=0.12, n_tau=8):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    taus = np.geomspace(tau_start, tau_end, n_tau)
    for tau in taus:
        if time.time() > deadline - 2:
            break
        x_new = _slsqp_solve(h, tau, maxiter=100)
        c5_new = _exact_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
        h = x_new
    return best_h, best_c5

def _line_search_exact(h, d, max_radius=0.5):
    n = len(h)
    dx = 2.0 / n
    n_grid = 32
    t_hi = max_radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-12:
        return 0.0, _exact_c5(h)
    grid_t = np.linspace(0.0, t_hi, n_grid)
    best_t = 0.0
    best_val = _exact_c5(h)
    for t in grid_t:
        cand = h + t * d
        cand = _project_sum(cand, n / 2.0)
        v = _exact_c5(cand)
        if v < best_val:
            best_val = v
            best_t = t
    lo = grid_t[max(0, int(np.argmin([_exact_c5(_project_sum(h + t * d, n/2.0)) for t in grid_t])) - 2)]
    hi = grid_t[min(n_grid - 1, int(np.argmin([_exact_c5(_project_sum(h + t * d, n/2.0)) for t in grid_t])) + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project_sum(h + c1 * d, n / 2.0))
    f2 = _exact_c5(_project_sum(h + c2 * d, n / 2.0))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project_sum(h + c1 * d, n / 2.0))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project_sum(h + c2 * d, n / 2.0))
    opt_t = (lo + hi) / 2.0
    opt_val = _exact_c5(_project_sum(h + opt_t * d, n / 2.0))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _pair_polish(h, deadline, n_trials=5000, radius=0.3, seed=42):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search_exact(h, d, max_radius=radius)
        if c5_try < cur_c5 - 1e-14:
            h = _project_sum(h + alpha * d, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_init, deadline, max_iter=30000, T_init=0.003, seed=77):
    n = len(h_init)
    h_bin = np.zeros(n)
    h_bin[np.argsort(h_init)[-n // 2:]] = 1.0
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(rng.integers(0, len(ones)))]
        j = zeros[int(rng.integers(0, len(zeros)))]
        cur[i] = 0.0
        cur[j] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    pert = np.zeros(n)
    x = np.arange(n, dtype=float)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 1, n) * 0.3
    pert *= amplitude
    return _project_sum(h + pert, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
