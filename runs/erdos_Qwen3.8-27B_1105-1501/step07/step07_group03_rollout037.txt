import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x: sum(x) = target, 0 <= x <= 1}
    n = len(v)
    v = np.clip(v, 0, 1)
    # Use bisection on threshold
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    # Fix rounding
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        x[mask] += res / mask.sum()
        x = np.clip(x, 0, 1)
    return x

def _build_A(u, n):
    # Build the correlation coefficient matrix
    # A[r, j] = coefficient of h[j] in correlation term r
    A = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        k = r - (n-1)
        if k >= 0:
            A[r, k:] = u[:n-k]
        else:
            p = -k
            A[r, :n-p] = u[p:]
    return A

def _lp_step(h, solvers=None):
    if solvers is None:
        solvers = ['ECOS', 'SLS', 'SCS', 'CLARABEL']
    n = len(h)
    u = 1.0 - h
    target = n / 2.0
    A = _build_A(u, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    obj = cp.Minimize(t_var)
    cons = [A @ h_var - t_var <= 0,
            cp.sum(h_var) == target,
            h_var >= 0, h_var <= 1]
    prob = cp.Problem(obj, cons)
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                return np.array(h_var.value, dtype=float)
        except:
            continue
    return None

def _lp_iterate(h, max_iters=15, deadline=None, solvers=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        h_lp = _lp_step(h_cur, solvers)
        if h_lp is None:
            break
        h_lp = _project(h_lp, n/2.0)
        c5_lp = _exact_c5(h_lp)
        
        if c5_lp < cur_c5 - 1e-10:
            # Accept: move towards LP solution
            h_cur = h_lp
            cur_c5 = c5_lp
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
        else:
            # LP didn't help, try line search
            d = h_lp - h_cur
            if np.linalg.norm(d) < 1e-10:
                break
            # Binary search for best alpha in [0, 1]
            best_alpha = 0.0
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                h_try = _project(h_cur + alpha * d, n/2.0)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-12:
                    best_alpha = alpha
                    cur_c5 = c5_try
                    h_cur = h_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h_cur.copy()
            if best_alpha == 0.0:
                break
    return best_h, best_c5

For binary solutions with n/2 ones, the problem is: place n/2 ones in n positions to minimize the maximum number of (1,0) pairs at any lag.

This is related to "Golay sequences" or "optimal difference sets". A difference set is a subset D of a group G such that every non-identity element of G is expressed as a difference of elements of D exactly λ times.

If we find a (v, k, λ) difference set with v=n, k=n/2, then every non-zero lag would have exactly λ pairs, giving C₅ = λ * 2/n.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-9:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-7
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _build_A(u):

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-8
        cnt = int(mask.sum())
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(u):
    n = len(u)
    A = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        if k >= 0:
            A[r, k:] = u[:n - k]
        else:
            p = -k
            A[r, :n - p] = u[p:]
    return A

def _solve_lp(h, solver_order=None):
    if solver_order is None:
        solver_order = ['ECOS', 'SCS', 'SLS', 'GLPK', 'CBC']
    n = len(h)
    u = 1.0 - h
    A = _build_A(u)
    target = n / 2.0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [A @ h_var - t_var <= 0,
            cp.sum(h_var) == target,
            h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for s in solver_order:

            if prob.status in ['optimal', 'optimal_inaccurate']:

    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 5.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + amp * p, n / 2.0)

    scale = 0.04
    for _ in range(60):
        if time.time() > d5 - 8:
            break
        h_p = _perturb(best_h, scale, rng)
        sub_dl = min(d5 - 4, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_refine(h_p, sub_dl, max_iters=10)
        if time.time() < d5 - 3:
            sub_dl2 = min(d5 - 3, time.time() + budget_s * 0.005)
            h_t, c5_t2 = _gradient_refine(_project(h_t, len(h_t) / 2.0), sub_dl2,
                                          max_iters=40, beta_init=22.0, radius_init=0.12)
            if c5_t2 < c5_t:
                c5_t = c5_t2
                h_t = _project(h_t, len(h_t) / 2.0)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.09)

    # Final polish
    if deadline - time.time() > 5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, c5_f = _lp_refine(hf, min(deadline - 2, time.time() + budget_s * 0.02), max_iters=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, c5_f = _gradient_refine(hf, min(deadline - 1, time.time() + budget_s * 0.015),
                                    max_iters=100, beta_init=40.0, radius_init=0.06)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
