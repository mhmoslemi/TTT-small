    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_full(h):
    n = len(h)
    n_lags = 2 * n - 1
    u = 1.0 - h
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        r = k - (n - 1)
        if r >= 0:
            m = n - r
            if m > 0:
                G[k, r:r+m] += u[:m]
                G[k, :m] += -h[r:r+m]
        else:
            g = -r
            m = n - g
            if m > 0:
                G[k, :m] += u[g:g+m]
                G[k, g:g+m] += -h[:m]
    return G
