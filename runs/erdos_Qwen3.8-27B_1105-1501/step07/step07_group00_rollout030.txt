def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -2.0, 2.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """
    Swap ones position r (was 1, becomes 0) with zeros position s (was 0, becomes 1).
    Returns updated counts.
    r is index in ones array, s is index in zeros array (or actual positions?)

def _bin_counts(b):

def _bin_swap_counts(counts, n, r_pos, s_pos, ones_arr, zeros_arr):
    new_counts = counts.copy()
    mask_z = zeros_arr != s_pos
    z_arr = zeros_arr[mask_z]
    new_counts[s_pos - z_arr + n - 1] += 1
    new_counts[r_pos - z_arr + n - 1] -= 1
    mask_r = ones_arr != r_pos
    r_arr = ones_arr[mask_r]
    new_counts[r_arr - r_pos + n - 1] += 1
    new_counts[r_arr - s_pos + n - 1] -= 1
    new_counts[s_pos - r_pos + n - 1] += 1
    new_counts[r_pos - s_pos + n - 1] -= 1
    return new_counts

def _thue_masgen(n):
    half = n // 2
    tmasgen = np.zeros(n, dtype=np.int8)
    for i in range(n):
        tmasgen[i] = i & (i >> 1) if False else 0
    # Use proper Thue-Morse: parity of popcount
    tmasgen = np.zeros(n, dtype=np.int8)
    for i in range(n):
        tmasgen[i] = bin(i).count('1') % 2
    # Ensure exactly half ones
    ones = np.where(tmasgen == 1)[0]
    zeros = np.where(tmasgen == 0)[0]
    if len(ones) > half:
        excess = len(ones) - half
        remove = rng.integers(0, len(ones), size=excess) if 'rng' in dir() else np.arange(excess)
        tmasgen[ones[remove]] = 0
    elif len(ones) < half:
        deficit = half - len(ones)
        add = np.arange(deficit)
        tmasgen[zeros[add]] = 1
    return tmasgen

    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous local opt from parent (5% time)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _continuous_local(h_init, d1, radius_init=0.10, max_iters=1000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary search (45% time)
    d2 = t0 + budget_s * 0.50
    n_b = n0
    half_b = n_b // 2
    
    # Generate diverse seeds
    seeds = []
    # Seed 1: threshold parent
    seeds.append(_to_binary(best_h).astype(np.int8))
    # Seed 2: Thue-Morse
    seeds.append(_thue_masgen_f(n_b, rng))
    # Seed 3: Kronecker
    seeds.append(_km_balanced(n_b, rng))
    # Seed 4: AP with step coprime
    for step in [2, 3, 5, 7]:
        if np.gcd(step, n_b) == 1 or step < n_b:
            seeds.append(_ap_balanced(n_b, step, rng))
    # Seed 5-8: Random
    for _ in range(4):
        seeds.append(_rnd_balanced(n_b, rng))
    
    best_bin = None
    best_bin_c5 = best_c5
    
    # Evaluate and sort seeds by initial quality
    seed_scores = []
    for s in seeds:
        if len(s) == n_b and int(s.sum()) == half_b:
            c = _bin_counts(s)
            c5_s = int(c.max()) * 2.0 / n_b
            seed_scores.append((c5_s, s))
    
    # Binary SA on best seeds
    n_best_seeds = min(6, len(seed_scores))
    for si in range(n_best_seeds):
        if time.time() > d2:
            break
        time_left = d2 - time.time()
        if time_left < 5:
            break
        # Allocate more time to better seeds
        frac = 0.5 / (si + 1)
        sub_dl = min(d2, time.time() + max(10, time_left * frac))
        s_in = seed_scores[si][1]
        b_res, c5_res = _binary_sa_multi(s_in, n_b, sub_dl, n_replicas=4, T_base=0.015)
        if c5_res < best_bin_c5:
            best_bin_c5 = c5_res
            best_bin = b_res.copy()
    
    # Try a few more n values for binary search
    for dn in [-4, 4, -8, 8]:
        if time.time() > d2 - 10:
            break
        n_try = n_b + dn
        if n_try < 60 or n_try > 250:
            continue
        if n_try % 2 != 0:
            continue
        sub_dl = min(d2, time.time() + max(5, (d2 - time.time()) * 0.15))
        b_seed = _rnd_balanced(n_try, rng)
        # Also try Thue-Morse at this size
        b_tm = _thue_masgen_f(n_try, rng)
        c5_rnd = int(_bin_counts(b_seed).max()) * 2.0 / n_try
        c5_tm = int(_bin_counts(b_tm).max()) * 2.0 / n_try
        b_best_seed = b_seed if c5_rnd < c5_tm else b_tm
        b_res, c5_res = _binary_sa_multi(b_best_seed, n_try, sub_dl, n_replicas=4, T_base=0.015)
        if c5_res < best_bin_c5:
            best_bin_c5 = c5_res
            best_bin = b_res.copy()
            best_h = _project(b_res.astype(float), n_try / 2.0)
            best_c5 = c5_res
    
    # Use best binary result
    if best_bin is not None and best_bin_c5 < best_c5:
        n_use = len(best_bin)
        best_h = _project(best_bin.astype(float), n_use / 2.0)
        best_c5 = _exact_c5(best_h)

    # Phase 3: Continuous polish from best (15% time)
    d3 = t0 + budget_s * 0.65
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c5_3 = _continuous_local(h3, d3, radius_init=0.08, max_iters=2000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair sweep (10% time)
    d4 = t0 + budget_s * 0.75
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    h4, c5_4 = _pair_sweep(h4, d4, max_rounds=150)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (15% time)
    d5 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in list(range(-16, 17, 4)):
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = _continuous_local(h_r, sub_dl, radius_init=0.08, max_iters=500)
        if time.time() < d5 - 2 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d5, time.time() + budget_s * 0.005)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, max_rounds=30)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        # Also try binary search at this n
        if time.time() < d5 - 3:
            h_bin_r = _to_binary(h_r)
            sub_dl3 = min(d5, time.time() + budget_s * 0.008)
            b_res_r, c5_bin_r = _binary_sa_multi(h_bin_r.astype(np.int8), n_try, sub_dl3, n_replicas=3, T_base=0.012)
            if c5_bin_r < c5_r:
                h_r = _project(b_res_r.astype(float), n_try / 2.0)
                c5_r = _exact_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Basin hopping + final polish (10% time)
    d6 = deadline - 2
    scale = 0.015
    n_restarts = 0
    while time.time() < d6 * 0.7 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 * 0.7, time.time() + budget_s * 0.003)
        h_t, c5_t = _continuous_local(h_pert, sub_dl, radius_init=0.05, max_iters=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)

    # Final polish
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    sub_dl_f = deadline - 1
    h_f, c5_f = _continuous_local(h_f, sub_dl_f, radius_init=0.04, max_iters=1500)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    if deadline - time.time() > 3:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_f2, c5_f2 = _pair_sweep(h_f2, deadline, max_rounds=40)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
