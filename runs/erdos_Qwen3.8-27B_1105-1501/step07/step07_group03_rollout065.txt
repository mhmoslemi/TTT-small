```python
import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, d, n):
    """Gradient of correlation lag d w.r.t. h. Vectorized."""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n-d])
            g[:n-d] -= h[d:]
    else:
        gg = -d
        if gg < n:
            g[:n-gg] += (1.0 - h[gg:])
            g[gg:] -= h[:n-gg]
    return g

def _line_search(h, d, dx, a_max):
    """Exact line search along direction d. Returns (best_alpha, best_c5)."""
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    raw = np.correlate(h, u, mode='full')
    
    n = len(h)
    a_hi = a_max
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    
    if a_hi < 1e-13:
        return 0.0, float(np.max(raw)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(80):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    opt = (lo + hi) / 2.0
    val = float(np.max(raw + opt * T - opt * opt * Q)) * dx
    return opt, val

def _lp_optimize(h, raw, radius, n_lags):
    """LP trust region. Returns proposed x or None."""
    n = len(h)
    target = n / 2.0
    
    top_idx = np.argsort(-raw)[:n_lags]
    
    # Compute gradients for top lags
    grads = np.zeros((n_lags, n))
    for m, idx in enumerate(top_idx):
        d = idx - (n - 1)
        grads[m] = _lag_grad(h, d, n)
    
    x_lo = np.maximum(0.0, h - radius)
    x_hi = np.minimum(1.0, h + radius)
    
    if np.any(x_lo > x_hi + 1e-10):
        return None
    if float(np.sum(x_lo)) > target + 1e-8 or float(np.sum(x_hi)) < target - 1e-8:
        return None
    
    x_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = []
    for m in range(n_lags):
        rhs = float(raw[top_idx[m]]) - float(grads[m] @ h)
        constraints.append(grads[m] @ x_var - t_var <= rhs + 1e-10)
    constraints.append(cp.sum(x_var) == target)
    constraints.append(x_var >= x_lo)
    constraints.append(x_var <= x_hi)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.CBC)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None
    
    if x_var.value is None:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    x_lp = np.clip(x_var.value, 0.0, 1.0)
    x_lp = _project(x_lp, target)
    return x_lp

def _optimize_hybrid(h, deadline, n_lags=12, radius_init=0.30, stall_limit=100):
    """Hybrid LP + gradient + pair swap optimizer."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    rng = np.random.default_rng(54321)
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        
        raw = _raw_corr(h)
        
        # Try LP
        accepted = False
        x_lp = _lp_optimize(h, raw, radius, n_lags)
        if x_lp is not None:
            d = x_lp - h
            d_norm = np.linalg.norm(d)
            if d_norm > 1e-10:
                d = d / d_norm
                a_max = min(radius * 1.5, 0.5)
                alpha, new_c5 = _line_search(h, d, dx, a_max)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        accepted = True
        
        if not accepted:
            # Gradient direction (softmax weighted)
            mx = float(np.max(raw))
            shifted = raw - mx
            beta = 25.0
            w = np.exp(np.clip(beta * shifted, -80, 0))
            w = w / (float(np.sum(w)) + 1e-30)
            
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-12:
                    continue
                d = k - (n - 1)
                g += w[k] * _lag_grad(h, d, n)
            g -= np.mean(g)
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-12:
                d_dir = -g / g_norm
                alpha, _ = _line_search(h, d_dir, dx, radius)
                if
