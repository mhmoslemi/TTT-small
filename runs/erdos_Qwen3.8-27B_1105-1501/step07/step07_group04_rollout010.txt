                # 1->0: d=-1, delta = -(val) where val = (1-h[p+shift])-h[p-shift]
                # Reducing means delta > 0, i.e., val < 0, i.e., h[p+shift]+h[p-shift] > 1
                # Same condition as above for binary
                cond = True
                if p + shift < n:
                    if h[p + shift] < 0.5:
                        cond = False
                else:
                    cond = False
                if p - shift >= 0:
                    if h[p - shift] < 0.5:
                        cond = False
                else:
                    cond = False
                if cond:
                    candidates.append((p, -1))
    elif shift < 0:
        u = -shift
        # delta[center-u] = d * ((1-h[p-u]) - h[p+u])
        # 0->1 reduces: (1-h[p-u]) - h[p+u] < 0 => h[p-u]=1 and h[p+u]=1
        # 1->0 reduces: same condition
        for p in range(n):
            if p - u >= n or p + u < 0:
                pass
            h_pu = h[p - u] if 0 <= p - u < n else None
            h_pup = h[p + u] if 0 <= p + u < n else None
            if h[p] < 0.5:
                cond = True
                if 0 <= p - u < n:
                    if h[p - u] < 0.5:
                        cond = False
                else:
                    cond = False
                if 0 <= p + u < n:
                    if h[p + u] < 0.5:
                        cond = False
                else:
                    cond = False
                if cond:
                    candidates.append((p, 1))
            else:
                cond = True
                if 0 <= p - u < n:
                    if h[p - u] < 0.5:
                        cond = False
                else:
                    cond = False
                if 0 <= p + u < n:
                    if h[p + u] < 0.5:
                        cond = False
                else:
                    cond = False
                if cond:
                    candidates.append((p, -1))
    return candidates

def _binary_greedy_exact(h, deadline, max_iters=500000, seed=42):
    """Fast binary local search with O(n) exact incremental updates."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(seed)
    stall = 0
    total_iters = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        total_iters += 1
        if stall > 500:
            break
        
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        
        # Get candidates for the max lag
        cands = _candidates_for_lag(h, shift, n)
        
        if not cands:
            # Try random flips as fallback
            if not ones or not zeros:
                break
            if rng.random() < 0.5:
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                cands = [(ones[i], -1), (zeros[j], 1)]
            else:
                stall += 1
                continue
        
        # Try up to 30 candidates
        best_move = None
        best_new_c5 = cur_c5
        n_try = min(30, len(cands))
        if n_try < len(cands):
            idxs = rng.choice(len(cands), size=n_try, replace=False)
            cands_try = [cands[i] for i in idxs]
        else:
            cands_try = cands
        
        for p, d in cands_try:
            if d == 1 and h[p] > 0.5:
                continue
            if d == -1 and h[p] <= 0.5:
                continue
            delta = _flip_delta_vec(h, p, d, n)
            new_corr = corr + delta
            new_max = float(np.max(new_corr)) * (2.0 / n)
            if new_max < best_new_c5:
                best_new_c5 = new_max
                best_move = (p, d, delta, new_corr)
        
        if best_move is not None and best_new_c5 < cur_c5 - 1e-14:
            p, d, delta, new_corr = best_move
            h[p] += d
            corr = new_corr
            cur_c5 = best_new_c5
            # Update ones/zeros
            if d == 1:
                ones.append(p)
                # Remove p from zeros
                for i, z in enumerate(zeros):
                    if z == p:
                        zeros[i] = zeros[-1]
                        zeros.pop()
                        break
            else:
                zeros.append(p)
                for i, o in enumerate(ones):
                    if o == p:
                        ones[i] = ones[-1]
                        ones.pop()
                        break
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_sa_exact(h, deadline, max_iters=300000, temp=0.003, seed=42):
    """Simulated annealing with O(n) exact updates."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(seed)
    T = temp
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        p1, p2 = ones[i], zeros[j]
        
        # Flip p1: 1->0, p2: 0->1
        d1 = -1
        d2 = 1
        delta1 = _flip_delta_vec(h, p1, d1, n)
        # Need to compute delta for p2 considering h AFTER p1 flip
        # For exactness, compute sequentially
        h[p1] += d1
        delta2 = _flip_delta_vec(h, p2, d2, n)
        h[p1] -= d1  # undo for now
        
        total_delta = delta1 + delta2
        new_corr = corr + total_delta
        new_max = float(np.max(new_corr)) * (2.0 / n)
        delta_c5 = new_max - cur_c5
        
        if delta_c5 < -1e-14 or rng.random() < np.exp(-delta_c5 / max(T, 1e-15)):
            h[p1] += d1
            h[p2] += d2
            corr = new_corr
            cur_c5 = new_max
            ones[i] = p2
            zeros[j] = p1
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
                best_stall += 1
            T = max(0.0003, T * 0.9998)
        else:
            stall += 1
            best_stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    h = h.copy()
    if n != len(h):
        h = _resample(h, n)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(len(h))
    h_bin[idx] = 1.0
    return h_bin

def _continuous_polish(h, deadline, radius=0.05, max_iter=500):
    """Simple projected subgradient with line search for continuous refinement."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    r = radius
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        active = np.where(corr >= max_val - 1e-10)[0]
        
        g = np.zeros(n)
        for m in active:
            k = int(m - (n - 1))
            if k >= 0:
                for i in range(n - k):
                    g[i + k] += (1.0 - h[i])
                    g[i] -= h[i + k]
            else:
                ka = -k
                for i in range(n - ka):
                    g[i] += (1.0 - h[i + ka])
                    g[i + ka] -= h[i]
        g /= len(active)
        
        d = -g
        d -= np.mean(d)
        d = np.clip(d, -r, r)
        d -= np.mean(d)
        
        if np.max(np.abs(d)) < 1e-14:
            break
        
        # Line search
        S0 = corr.copy()
        T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C_arr = np.correlate(d, d, mode='full')
        
        t_hi = 1.0
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        t_hi = max(t_hi, 1e-14)
        
        grid_t = np.linspace(0.0, t_hi, 20)
        best_t = 0.0
        best_f = cur
        for t in grid_t:
            f = float(np.max(S0 + t * T_arr - t * t * C_arr)) * dx
            if f < best_f:
                best_f = f
                best_t = t
        
        if best_t > 0 and best_f < cur - 1e-13:
            h = h + best_t * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            r = min(r * 1.05, 0.2)
        else:
            r *= 0.7
            if r < 1e-6:
                break
    
    return best_h, best_c5

def _multi_lag_greedy(h, deadline, max_iters=200000, k_lags=3, seed=42):
    """Greedy search considering top-k lags simultaneously."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(seed)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if not ones or not zeros:
            break
        
        # Get top-k lags
        top_k = np.argsort(corr)[-k_lags:][::-1]
        
        # Collect all candidates from top-k lags
        cands = set()
        for m in top_k:
            shift = int(m) - (n - 1)
            lc = _candidates_for_lag(h, shift, n)
            for p, d in lc:
                cands.add((p, d))
        
        if not cands:
            # Fallback: random pair
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            cands = {(ones[i], -1), (zeros[j], 1)}
        
        best_move = None
        best_new_c5 = cur_c5
        cands_list = list(cands)
        n_try = min(40, len(cands_list))
        if n_try < len(cands_list):
            idxs = rng.choice(len(cands_list), size=n_try, replace=False)
            cands_list = [cands_list[i] for i in idxs]
        
        for p, d in cands_list:
            if d == 1 and h[p] > 0.5:
                continue
            if d == -1 and h[p] <= 0.5:
                continue
            delta = _flip_delta_vec(h, p, d, n)
            new_corr = corr + delta
            new_max = float(np.max(new_corr)) * (2.0 / n)
            if new_max < best_new_c5:
                best_new_c5 = new_max
                best_move = (p, d, delta, new_corr)
        
        if best_move is not None and best_new_c5 < cur_c5 - 1e-14:
            p, d, delta, new_corr = best_move
            h[p] += d
            corr = new_corr
            cur_c5 = best_new_c5
            if d == 1:
                ones.append(p)
                for i, z in enumerate(zeros):
                    if z == p:
                        zeros[i] = zeros[-1]
                        zeros.pop()
                        break
            else:
                zeros.append(p)
                for i, o in enumerate(ones):
                    if o == p:
                        ones[i] = ones[-1]
                        ones.pop()
                        break
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _random_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _pair_swap_binary(h, deadline, max_iters=100000, seed=42):
    """Try swapping pairs of positions (one 1 and one 0)."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(seed)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        p1, p2 = ones[i], zeros[j]
        
        delta1 = _flip_delta_vec(h, p1, -1, n)
        h[p1] = 0.0
        delta2 = _flip_delta_vec(h, p2, 1, n)
        h[p1] = 1.0
        
        new_corr = corr + delta1 + delta2
        new_max = float(np.max(new_corr)) * (2.0 / n)
        
        if new_max < cur_c5 - 1e-14:
            h[p1] = 0.0
            h[p2] = 1.0
            corr = new_corr
            cur_c5 = new_max
            ones[i] = p2
            zeros[j] = p1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    
    # Get initial values
    try:
        h0_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h0_raw = np.full(180, 0.5)
    
    n0 = len(h0_raw)
    h_init = _project(h0_raw.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Convert to binary for main search
    h_bin = _to_binary(h_init, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Phase 1: Multi-lag greedy on n0 (30% of budget)
    d1 = t0 + budget_s * 0.25
    h_r, c5_r = _multi_lag_greedy(h_bin, d1, max_iters=300000, k_lags=3, seed=seed)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    
    # Phase 2: Pair swap (15%)
    d2 = t0 + budget_s * 0.40
    h_r, c5_r = _pair_swap_binary(best_h, d2, max_iters=200000, seed=seed + 1)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    
    # Phase 3: SA on best (15%)
    d3 = t0 + budget_s * 0.55
    h_r, c5_r = _binary_sa_exact(best_h, d3, max_iters=200000, temp=0.003, seed=seed + 2)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    
    # Phase 4: Multi-n exploration (25%)
    d4 = t0 + budget_s * 0.80
    n_list = []
    for dn in range(-24, 25, 4):
        nt = n0 + dn
        if 100 <= nt <= 260 and nt % 2 == 0:
            n_list.append(nt)
    # Remove n0 (already done)
    n_list = [x for x in n_list if x != n0]
    # Sort by distance from n0
    n_list.sort(key=lambda x: abs(x - n0))
    
    time_per_n = max(3, (d4 - time.time()) / max(1, len(n_list)))
    for nt in n_list:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + time_per_n)
        h_r = _resample(best_h, nt)
        h_r = _to_binary(h_r, nt)
        # Quick greedy
        h_r, c5_r = _multi_lag_greedy(h_r, sub_dl, max_iters=50000, k_lags=2, seed=seed + 100 + nt)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Final refinement (20%)
    d5 = deadline
    n_best = len(best_h)
    
    # More greedy iterations
    sub_dl = min(d5, time.time() + budget_s * 0.06)
    h_r, c5_r = _multi_lag_greedy(best_h, sub_dl, max_iters=200000, k_lags=4, seed=seed + 500)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    
    # SA with lower temperature
    if time.time() < d5 - 20:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_r, c5_r = _binary_sa_exact(best_h, sub_dl, max_iters=100000, temp=0.001, seed=seed + 600)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Continuous polish
    if time.time() < d5 - 15:
        n_best = len(best_h)
        h_cont = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_r, c5_r = _continuous_polish(h_cont, sub_dl, radius=0.03, max_iter=200)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Final: try nearby n values
    if time.time() < d5 - 10:
        n_fin = len(best_h)
        for dn in [-2, 2, -4, 4, -6, 6]:
            if time.time() > d5 - 5:
                break
            nt = n_fin + dn
            if nt < 80:
                continue
            h_r = _resample(best_h, nt)
            h_r = _to_binary(h_r, nt)
            sub_dl = min(d5 - 3, time.time() + 5)
            h_r, c5_r = _multi_lag_greedy(h_r, sub_dl, max_iters=30000, k_lags=3, seed=seed + 900 + nt)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Final polish
    if time.time() < deadline - 2:
        h_final = _project(best_h.copy(), len(best_h) / 2.0)
        final_c5 = _exact_c5(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
        else:
            pass  # keep original
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
