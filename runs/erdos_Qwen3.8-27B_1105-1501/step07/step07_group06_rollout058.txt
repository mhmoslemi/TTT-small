import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _anti_to_full(x, m):
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h, m):
    x1 = h[:m].copy()
    x2 = (1.0 - h[m:])[::-1].copy()
    x = 0.5 * (x1 + x2)
    return np.clip(x, 0.0, 1.0)

def _anti_obj(x, m, beta):
    h = _anti_to_full(x, m)
    raw = np.correlate(h, 1.0 - h, mode='full')
    n = 2 * m
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 50.0)
    lse = float(np.log(np.sum(np.exp(z))))
    return (M + lse / beta) * (2.0 / n)

def _anti_grad(x, m, beta):
    h = _anti_to_full(x, m)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    n = 2 * m
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return np.zeros(m)
    w /= wsum
    
    G = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for r in active:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:tau+mm] += wr * u[:mm]
                G[:mm] -= wr * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * u[d:d+mm]
                G[d:d+mm] -= wr * h[:mm]
    
    grad = G[:m] - G[2*m-1:m-1:-1]
    grad *= (2.0 / n)
    return grad

def _opt_anti(x, m, deadline, beta_start=6.0, beta_mult=2.0, n_stages=4):
    """Multi-stage L-BFGS-B optimization in anti-palindromic subspace."""
    bounds = [(0.0, 1.0)] * m
    best_x = x.copy()
    best_c5 = _exact_c5(_anti_to_full(x, m))
    beta = beta_start
    
    for stage in range(n_stages):
        if time.time() > deadline:
            break
        res = minimize(
            _anti_obj, x, args=(m, beta),
            jac=_anti_grad,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
        )
        if hasattr(res, 'x') and res.x is not None:
            x_new = np.clip(res.x, 0.0, 1.0)
            c5_new = _exact_c5(_anti_to_full(x_new, m))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_x = x_new.copy()
            x = x_new
        beta *= beta_mult
        if beta > 200:
            break
    
    return best_x, best_c5

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample_full(h, n_new):
    """Resample h to new length n_new, preserving sum constraint."""
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0, right=0)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    """Simulated annealing for binary h."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(1234)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300 or len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        oi = ones[oi]
        oz = zeros[oz]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T = max(0.0001, T * 0.9997)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best_h, best_c5

def _pair_sweep(h_bin, deadline, max_rounds=10):
    """Iterated pair swap local search."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    improved = True
    
    for rnd in range(max_rounds):
        if time.time() > deadline or not improved:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            best_oz = -1
            best_c = cur_c5
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < best_c:
                    best_c = c_try
                    best_oz = oz
            if best_oz >= 0:
                cur[oi] = 0.0
                cur[best_oz] = 1.0
                cur_c5 = best_c
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0]
                zeros = np.where(cur <= 0.5)[0]
            else:
                cur[oi] = 1.0
    
    return best_h, best_c5

def _full_grad_descent(h, deadline, beta_start=10.0, n_iter=500, radius=0.15):
    """Full-space gradient descent with smooth-max and exact line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_start
    rng = np.random.default_rng(555)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        M = float(np.max(raw))
        z = beta * (raw - M)
        z = np.clip(z, -50, 50)
        w = np.exp(z)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            r_max = int(np.argmax(raw))
            tau = r_max - (n - 1)
            g = np.zeros(n)
            if tau >= 0:
                mm = n - tau
                g[tau:] += u[:mm]
                g[:mm] -= h[tau:tau+mm]
            else:
                d = -tau
                mm = n - d
                g[:mm] += u[d:d+mm]
                g[d:d+mm] -= h[:mm]
        else:
            w /= wsum
            g = np.zeros(n)
            for r in np.where(w > 1e-12)[0]:
                wr = w[r]
                tau = r - (n - 1)
                if tau >= 0:
                    mm = n - tau
                    if mm > 0:
                        g[tau:tau+mm] += wr * u[:mm]
                        g[:mm] -= wr * h[tau:tau+mm]
                else:
                    d = -tau
                    mm = n - d
                    if mm > 0:
                        g[:mm] += wr * u[d:d+mm]
                        g[d:d+mm] -= wr * h[:mm]
        
        # Normalize direction
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn
        
        # Exact line search
        # Find valid range for alpha
        pos = d > 1e-14
        neg = d < -1e-14
        a_lo = -radius
        a_hi = radius
        if np.any(pos):
            a_lo = max(a_lo, float(np.min((-h + 0.01) / d)[pos]) if False else float(np.min(-h[pos] / d[pos])))
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
            a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
        
        # Handle numerical issues
        if a_lo >= a_hi:
            a_lo, a_hi = -radius * 0.1, radius * 0.1
        a_lo = max(a_lo, -radius)
        a_hi = min(a_hi, radius)
        
        # Golden section on C5(h + alpha*d)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = a_hi - gr * (a_hi - a_lo)
        c2 = a_lo + gr * (a_hi - a_lo)
        f1 = _exact_c5(_project(h + c1 * d, target))
        f2 = _exact_c5(_project(h + c2 * d, target))
        
        for _ in range(30):
            if a_hi - a_lo < 1e-10:
                break
            if f1 < f2:
                a_hi, c2, f2 = c2, c1, f1
                c1 = a_hi - gr * (a_hi - a_lo)
                f1 = _exact_c5(_project(h + c1 * d, target))
            else:
                a_lo, c1, f1 = c1, c2, f2
                c2 = a_lo + gr * (a_hi - a_lo)
                f2 = _exact_c5(_project(h + c2 * d, target))
        
        alpha = (a_lo + a_hi) / 2.0
        h_new = _project(h + alpha * d, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # No improvement, try smaller radius
            radius *= 0.8
            if radius < 0.01:
                break
        
        # Increase beta periodically for sharper gradient
        if it % 20 == 19:
            beta *= 1.5
    
    return best_h, best_c5

where raw[r] is a bilinear form in h. If I fix all but two coordinates of h, the objective becomes a function of those two coordinates that is piecewise quadratic (since the max of linear functions is convex, and each raw[r] is linear in each h[i] when others are fixed).

import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _anti_to_full(x, m):
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h, m):
    x1 = h[:m].copy()
    x2 = (1.0 - h[m:])[::-1].copy()
    x = 0.5 * (x1 + x2)
    return np.clip(x, 0.0, 1.0)

def _anti_obj(x, m, beta):
    h = _anti_to_full(x, m)
    raw = np.correlate(h, 1.0 - h, mode='full')
    n = 2 * m
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 50.0)
    lse = float(np.log(np.sum(np.exp(z))))
    return (M + lse / beta) * (2.0 / n)

def _anti_grad(x, m, beta):
    h = _anti_to_full(x, m)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    n = 2 * m
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return np.zeros(m)
    w /= wsum
    G = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for r in active:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:tau+mm] += wr * u[:mm]
                G[:mm] -= wr * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * u[d:d+mm]
                G[d:d+mm] -= wr * h[:mm]
    grad = G[:m] - G[2*m-1:m-1:-1]
    grad *= (2.0 / n)
    return grad

def _opt_anti(x, m, deadline, beta_start=6.0, n_stages=4):
    bounds = [(0.0, 1.0)] * m
    best_x = x.copy()
    best_c5 = _exact_c5(_anti_to_full(x, m))
    beta = float(beta_start)
    for stage in range(n_stages):
        if time.time() > deadline:

        res = minimize(_anti_obj, x, args=(m, beta), jac=_anti_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 40})
        if hasattr(res, 'x') and res.x is not None:
            x_new = np.clip(res.x, 0.0, 1.0)
            c5_new = _exact_c5(_anti_to_full(x_new, m))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_x = x_new.copy()
            x = x_new
        beta *= 2.0
        if beta > 300:
            break
    return best_x, best_c5

def _project(v, target):
    n = len(v)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample_full(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0, right=0)
    return _project(h_new, n_new / 2.0)

            if cur_c5 < best_c5:

def _pair_sweep(h_bin, deadline, max_rounds=50, seed=123):

    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for oi in ones:
            if time.time() > deadline:
                break
            best_oz = -1
            best_c = cur_c5
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0

                    best_oz = oz
