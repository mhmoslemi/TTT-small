def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):

    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Find minimum of max(S0 + a*T - a^2*Q) over a in [a_lo, a_hi]
    # Use golden section search with refinement
    best_val = float('inf')
    best_a = 0.0
    
    # Check endpoints and some interior points
    cands = [a_lo, a_hi]
    width = a_hi - a_lo

                cands.append(av)

        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement around best
    lo_r = max(a_lo, best_a - 0.1 * width)
    hi_r = min(a_hi, best_a + 0.1 * width)
    if hi_r - lo_r > 1e-12:

        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new

def _struct_grad(h, k, dx):
    """
    Compute a structured gradient focusing on balancing the top-k shifts.
    Uses the subgradient of the max function.
    """
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-c)[:k]
    top_val = c[top_idx[0]]
    
    # Subgradient: uniform average of subgradients of active terms
    # For each active shift l, the subgradient of c[l] w.r.t. h is:
    #   for l >= 0: (1-h) on [0, n-l) and -h on [l, n)
    #   for l < 0: (1-h) on [-l, n) and -h on [0, n+l-1]
    
    g = np.zeros(n)
    for l_idx in top_idx:
        l = int(l_idx) - (n - 1)
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += (1.0 - h[l:l+m])
                g[l:] -= h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += (1.0 - h[:m])
                g[:m] -= h[s:s+m]
    g /= float(k)
    g -= np.mean(g)
    return g

    return best_h, best_c

def _reweight_opt(h, deadline, eta=4.0, r0=0.10):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    stall = 0
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.exp(eta * (c - np.max(c)))
    rad = r0
    while time.time() < deadline and stall < 70:
        if deadline - time.time() < 0.4:
            break
        wn = w / float(np.sum(w))
        g = _grad_weighted(h, wn)
        nrm = np.linalg.norm(g)
        if nrm < 1e-12:
            break
        d = -g / nrm
        a, c_try = _line_search(h, d, dx, rad)
        if c_try < cur - 1e-15:
            h = h + a * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-15:
                best_c = cur
                best_h = h.copy()
            stall = 0
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            w = np.exp(eta * (c - np.max(c)))
        else:
            stall += 1
            rad *= 0.88
            if rad < 0.003:
                break
    return best_h, best_c

def _pair_opt(h, deadline, max_it=30000, seed=31337):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = _project(h + a * d, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _bin_sa(hb, deadline, mx=30000, Ti=0.003, seed=42, guided=False):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(seed)
    T = Ti
    stall = 0
    dx = 2.0 / n
    c_cur = np.correlate(cur, 1.0 - cur, mode='full')
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if not ones or not zeros:
            break
        if guided and len(ones) > 0 and len(zeros) > 0:
            r = rng.random()
            if r < 0.6:
                top_c = np.argsort(-c_cur)[:10]
                oi = ones[int(rng.integers(0, len(ones)))]
                oz = zeros[int(rng.integers(0, len(zeros)))]
            else:
                oi = ones[int(rng.integers(0, len(ones)))]
                oz = zeros[int(rng.integers(0, len(zeros)))]
        else:
            oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            if guided:
                c_cur = np.correlate(cur, 1.0 - cur, mode='full')
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9988
        else:
            stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _greedy_binary(hb, schedule=3, max_passes=50):
    n = len(hb)
    dx = 2.0 / n
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for p in range(max_passes):
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        for _ in range(schedule * n_ones):
            if not ones or not zeros:
                break
            oi = ones[int(np.random.randint(0, len(ones)))]
            oz = zeros[int(np.random.randint(0, len(zeros)))]
            hn = cur.copy()
            hn[oi] = 0.0
            hn[oz] = 1.0
            nc = _exact_c5(hn)
            if nc < cc - 1e-15:
                cur = hn
                cc = nc
                ones.remove(oi)
                zeros.remove(oz)
                ones.append(oz)
                zeros.append(oi)
                if cc < bc:
                    bc = cc
                    bh = cur.copy()
                improved = True
                break
        if not improved:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Top-k gradient optimization (14%)
    d1 = t0 + budget_s * 0.14
    h1, c1 = _local_topk(h_init, d1, ks=(3, 5, 8), r0=0.15, stall_lim=50)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Reweighting (10%)
    d2 = t0 + budget_s * 0.24
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    for eta in [3.0, 5.0, 8.0]:
        if time.time() > d2 - 3:
            break
        h2, c2 = _reweight_opt(h2, min(d2, time.time() + budget_s * 0.025), eta=eta, r0=0.10)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Top-k with wider k range (8%)
    d3 = t0 + budget_s * 0.32
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _local_topk(h3, d3, ks=(3, 4, 5, 7, 10, 15, 25), r0=0.10, stall_lim=60)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Pair optimization (10%)
    d4 = t0 + budget_s * 0.42
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    h4, c4 = _pair_opt(h4, d4, max_it=25000)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: Binary SA - main phase (20%)
    d5 = t0 + budget_s * 0.62
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, min(d5, time.time() + budget_s * 0.08), mx=40000, Ti=0.004, seed=42)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d5 - 10:
        hbin2 = _to_bin(_project(best_h + rng.normal(0, 0.15, nb), nb / 2.0), nb)
        hbin2, cb2 = _bin_sa(hbin2, min(d5, time.time() + budget_s * 0.06), mx=30000, Ti=0.003, seed=123, guided=True)
        if cb2 < best_c:
            best_c = cb2
            best_h = hbin2.copy()
    if time.time() < d5 - 5:
        hbin3 = _to_bin(_project(best_h + rng.normal(0, 0.08, nb), nb / 2.0), nb)
        hbin3, cb3 = _bin_sa(hbin3, d5, mx=30000, Ti=0.002, seed=999)
        if cb3 < best_c:
            best_c = cb3
            best_h = hbin3.copy()

    # Phase 6: Multi-resolution (15%)
    d6 = t0 + budget_s * 0.77
    nc = len(best_h)
    tried_sizes = {nc}
    for dn in [0, -3, 3, -6, 6, -9, 9, -12, 12, -15, 15]:
        if time.time() > d6 - 4:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 260 or n_try in tried_sizes:
            continue
        tried_sizes.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d6, time.time() + budget_s * 0.016)
        hr, cr = _local_topk(hr, sd, ks=(3, 5, 8), r0=0.12, stall_lim=35)
        if cr < best_c + 0.004 and time.time() < d6 - 6:
            hb = _to_bin(hr, n_try)
            sd2 = min(d6, time.time() + budget_s * 0.008)
            hb, cb2 = _bin_sa(hb, sd2, mx=10000, Ti=0.002, seed=n_try)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.002 and time.time() < d6 - 4:
            sd3 = min(d6, time.time() + budget_s * 0.008)
            hr, cr = _local_topk(_project(hr.copy(), n_try / 2.0), sd3, ks=(3, 4, 5), r0=0.05, stall_lim=20)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 7: Intensive polishing (12%)
    d7 = t0 + budget_s * 0.89
    nc = len(best_h)
    for r_init in [0.08, 0.04, 0.02]:
        if time.time() > d7 - 4:
            break
        h7 = _project(best_h.copy(), nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.035)
        h7, c7 = _local_topk(h7, sd, ks=(3, 4, 5, 8, 12), r0=r_init, stall_lim=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
        if time.time() < d7 - 4:
            h7 = _project(best_h.copy(), nc / 2.0)
            sd = min(d7, time.time() + budget_s * 0.015)
            h7, c7 = _pair_opt(h7, sd, max_it=12000, seed=555)
            if c7 < best_c:
                best_c = c7
                best_h = h7.copy()

    # Phase 8: Basin hopping (10%)
    d8 = deadline
    scale = 0.018
    while time.time() < d8:
        nc = len(best_h)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * np.arange(nc) / nc + p)
        pert += 0.2 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d8, time.time() + budget_s * 0.005)
        ht, ct = _local_topk(hp, sd, ks=(3, 5, 8), r0=0.04, stall_lim=18)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.05, scale * 1.12)

    # Final polish
    if deadline - time.time() > 1.5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=4000, seed=4242)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
