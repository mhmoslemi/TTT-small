import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_grad(h, tau):
    n = len(h)
    scores = _scores(h)
    m = float(np.max(scores))
    u = (scores - m) / tau
    exp_u = np.exp(u)
    sum_exp = float(np.sum(exp_u))
    p = exp_u / sum_exp
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = p[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                grad[d:d+m_] += w * (1.0 - h[:m_])
                grad[:m_] += -w * h[d:d+m_]
        else:
            g = -d
            m_ = n - g
            if m_ > 0:
                grad[:m_] += w * (1.0 - h[g:g+m_])
                grad[g:] += -w * h[:m_]
    return grad

def _lse_pg(h, tau, n_iter, step_init, deadline):
    """Projected gradient descent on LSE objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    step = step_init
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    prev_c5 = best_c5
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        grad = _lse_grad(h, tau)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break
        # Gradient step with projection
        alpha = step
        for _ in range(20):
            h_try = _project_sum(h - alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        if alpha > 1e-10:
            h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                stall = 0
                step = min(alpha * 1.5, 0.5)
            else:
                stall += 1
                step *= 0.7
        else:
            stall += 1
            step *= 0.5
    return best_h, best_c5

def _slsqp_lse(h, tau, max_iter=30, deadline=None):
    """SLSQP on LSE objective."""
    n = len(h)
    target = n / 2.0
    
    def obj_and_grad(x):

        scores = _scores(x)
        m = float(np.max(scores))
        u = (scores - m) / tau
        exp_u = np.exp(u)
        sum_exp = float(np.sum(exp_u))
        f = m + tau * np.log(sum_exp)
        p = exp_u / sum_exp
        grad = np.zeros(n)

                m_ = n - g
                if m_ > 0:
                    grad[:m_] += w * (1.0 - x[g:g+m_])
                    grad[g:] += -w * x[:m_]
        return f, grad
    
    constraints = [{'type': 'eq',
                    'fun': lambda x: float(np.sum(x)) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:

                       bounds=bounds, constraints=constraints,

        if res.x is not None:
            h_new = _project_sum(res.x, target)

    except Exception:

    return h

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_sweep(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

    S0 = _scores(h)  # shape (2n-1,)
    
    # score[r](alpha) = S0[r] + alpha * T[r] - alpha^2 * C[r]
    # where T[r] = sum of directional derivative contributions
    # and C[r] = quadratic coefficient
    
    T = np.zeros(2*n-1)
    C = np.zeros(2*n-1)
    for r in range(2*n-1):
        d_val = r - (n-1)
        if d_val >= 0:
            m_ = n - d_val
            if m_ > 0:
                # score = sum h[i+d_val]*(1-h[i])
                # derivative w.r.t. alpha: sum (d[i+d_val]*(1-h[i]) - h[i+d_val]*d[i])
                # quadratic: -sum d[i+d_val]*d[i]
                T[r] = float(np.sum(d[d_val:d_val+m_] * (1.0 - h[:m_]) - h[d_val:d_val+m_] * d[:m_]))
                C[r] = float(np.sum(d[d_val:d_val+m_] * d[:m_]))
        else:
            g = -d_val
            m_ = n - g
            if m_ > 0:
                # score = sum h[i]*(1-h[i+g])
                T[r] = float(np.sum(d[:m_] * (1.0 - h[g:g+m_]) - h[:m_] * d[g:g+m_]))
                C[r] = float(np.sum(d[:m_] * d[g:g+m_]))
    
    # Find feasible alpha range
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section on f(alpha) = max_r (S0[r] + alpha*T[r] - alpha^2*C[r]) * dx
    lo = 0.0
    hi = t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr*(hi-lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr*(hi-lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    
    opt = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt*T - opt*opt*C)) * dx
    # Also check endpoints
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx
    best_a = 0.0
    best_v = f0
    if opt_val < best_v:
        best_a = opt
        best_v = opt_val
    if fhi < best_v:
        best_a = t_hi
        best_v = fhi
    
    return best_a, best_v

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_grad(h, tau):
    n = len(h)
    scores = _scores(h)
    m = float(np.max(scores))
    u = (scores - m) / tau
    exp_u = np.exp(u)
    sum_exp = float(np.sum(exp_u))
    p = exp_u / sum_exp
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = p[r]
        if w < 1e-13:
            continue
        d = r - (n - 1)
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                grad[d:d + m_] += w * (1.0 - h[:m_])
                grad[:m_] += -w * h[d:d + m_]
        else:
            g = -d
            m_ = n - g
            if m_ > 0:
                grad[:m_] += w * (1.0 - h[g:g + m_])
                grad[g:] += -w * h[:m_]
    return grad

def _pg_loop(h, tau, deadline, max_iters, step_init):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        g = _lse_grad(h, tau)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        alpha = step
        h_try = None
        c5_try = None
        for _bt in range(25):
            h_try = _project_sum(h - alpha * g, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        if h_try is not None and c5_try is not None and c5_try < cur_c5:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(alpha * 1.3, 0.3)
        else:
            stall += 1
            step *= 0.6

    return best_h, best_c5

def _slsqp_lse(h, tau, max_iter=25):
    n = len(h)
    target = n / 2.0
    def obj_grad(x):
        scores = _scores(x)
        m = float(np.max(scores))
        u = (scores - m) / tau
        exp_u = np.exp(u)
        sum_exp = float(np.sum(exp_u))
        f = m + tau * np.log(sum_exp)
        p = exp_u / sum_exp
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            w = p[r]
            if w < 1e-13:
                continue
            d = r - (n - 1)
            if d >= 0:
                m_ = n - d
                if m_ > 0:
                    grad[d:d + m_] += w * (1.0 - x[:m_])
                    grad[:m_] += -w * x[d:d + m_]
            else:
                g2 = -d
                m_ = n - g2
                if m_ > 0:
                    grad[:m_] += w * (1.0 - x[g2:g2 + m_])
                    grad[g2:] += -w * x[:m_]
        return f, grad
    cons = [{'type': 'eq',
             'fun': lambda x: float(np.sum(x)) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        from scipy.optimize import minimize
        res = minimize(obj_grad, h, method='SLSQP', bounds=bounds,
                       constraints=cons, maxiter=max_iter,
                       options={'ftol': 1e-12})
        if res.x is not None:
            return _project_sum(res.x, target)
    except Exception:
        pass
    return h

def _exact_1d(h, d, dx, max_alpha):

    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    lo, hi = 0.0, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):

        else:

    opt = (lo + hi) / 2.0

    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx

def _pair_direction_loop(h, deadline, n_pairs=10000, seed=99):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()

    best_h = h.copy()

        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        if h[i] <= h[j] + 0.01:
            stall += 1
            continue
        d = np.zeros(n)
        d[j] = 1.0
        d[i] = -1.0

        if c5_try < cur_c5 - 1e-14:

def _to_binary(h):

    idx = np.argsort(h)[-n // 2:]

def _pair_sweep(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.3, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)
