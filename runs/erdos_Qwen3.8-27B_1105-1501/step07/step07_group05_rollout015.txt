```python
def _weighted_grad(h, w):
    """Compute gradient of sum_l w_l * R_l(h) w.r.t. h.
    h: array of length n
    w: array of length 2n-1 (weights for lags -(n-1) to n-1)
    Returns: array of length n
    """
    n = len(h)
    # Pad w to length 2n-1 (it already is)
    # (w * h) at position m = sum_j w[m-j] h[j]
    # This is cross-correlation of h and w (or convolution with reversed w)
    # np.convolve(h, w, 'full') gives length 3n-1, we need positions n-1 to 2n-2
    # Actually, let me use np.correlate
    # (w * h)[m] = sum_j w[m-j] h[j] for m=0..n-1
    # This is the same as np.correlate(w, h, 'full')[n-1:n-1+n]
    # Wait, np.correlate(a, v)[k] = sum_n a[n+k] * v[n] for valid n
    # So np.correlate(w, h, 'full') has length len(w)+len(h)-1 = 3n-1
    # At index k: sum_n w[n+k] * h[n]
    # We want sum_j w[m-j] h[j] = sum_j w[m-j] h[j]
    # Let k = m: sum_j w[m-j] h[j]
    # This is like the cross-correlation with a sign flip on one argument
    # np.correlate(w, h, 'full')[k] = sum_n w[n+k]*h[n]
    # For k = m: sum_n w[n+m]*h[n] -- not quite what we want
    
    # Let me think differently.
    # We want c[m] = sum_j w[m-j] * h[j] for m = 0, ..., n-1
    # This is the convolution (w * h)[m] where w is indexed from -(n-1) to n-1
    # To use numpy, I need to shift w so it's 0-indexed: w_shifted[j] = w[j - (n-1)]
    # Then c[m] = sum_j w_shifted[m-j + (n-1)] * h[j] for valid indices
    # = (w_shifted * h)[m + (n-1)] (standard convolution)
    
    # w_shifted has length 2n-1, index 0 to 2n-2, representing lag -(n-1) to n-1
    # h has length n
    # w_shifted * h has length 3n-2
    # The m-th element of c is (w_shifted * h)[m + (n-1)]
    
    w_shifted = w  # already in the right order: w[0] = w_{-(n-1)}, ..., w[n-1] = w_0, ..., w[2n-2] = w_{n-1}
    conv = np.convolve(w_shifted, h, 'full')  # length 3n-2
    ch = conv[n-1:n-1+n]  # length n, ch[m] = sum_j w[m-j] h[j]
    
    # Similarly for (W^T * h)[m] = sum_j w[j-m] h[j]
    # = sum_j w[-(m-j)] h[j] = sum_j w_rev[m-j] h[j] where w_rev[k] = w[-k]
    # w_rev_shifted[j] = w_rev[j-(n-1)] = w[-(j-(n-1))] = w[n-1-j]
    w_rev_shifted = w_shifted[::-1]  # reverse
    conv_t = np.convolve(w_rev_shifted, h, 'full')
    cht = conv_t[n-1:n-1+n]
    
    g = 1.0 - ch - cht
    return g
```
