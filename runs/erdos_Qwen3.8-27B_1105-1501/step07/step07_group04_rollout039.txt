def basin_hopping_binary(h, deadline, rng, n_jumps=30):

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def to_binary(h, n):
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def make_random_start(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def make_periodic_start(n, period, rng):
    h = np.zeros(n)
    ones_per_period = period // 2
    offset = int(rng.integers(0, period))
    for i in range(n):
        pos_in_period = (i + offset) % period
        h[i] = 1.0 if pos_in_period < ones_per_period else 0.0
    # Fix count
    cnt = int(np.sum(h))
    if cnt != n // 2:
        h = to_binary(project(h, n/2.0), n)
    return h

def make_greedy_start(n, deadline, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones_needed = n // 2
    for step in range(ones_needed):
        if time.time() > deadline:
            break
        best_pos = -1
        best_c5 = float('inf')
        # Try positions in shuffled order for diversity, but evaluate all
        order = list(range(n))
        rng.shuffle(order)
        for pos in order:
            h[pos] = 1.0
            c5 = exact_c5(h)
            h[pos] = 0.0
            if c5 < best_c5:
                best_c5 = c5
                best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h

def make_shifted_start(h_base, n, shift):
    """Cyclically shift a base pattern to length n."""
    if len(h_base) == n:
        return np.roll(h_base, shift % n)
    # Resample then shift
    h_r = resample(h_base, n)
    h_b = to_binary(h_r, n)
    return np.roll(h_b, shift % n)

def basin_hopping(h_bin, deadline, seed=42, n_rounds=15):

    rng = np.random.default_rng(seed)

    best_c5 = exact_c5(h_bin)
    for r in range(n_rounds):
        if time.time() > deadline - 3:
            break
        h_pert = best_h.copy()
        # Random block swap perturbation
        blen = int(rng.integers(2, min(8, n // 4)))
        i = int(rng.integers(0, n - blen))
        j = int(rng.integers(0, n - blen))
        if abs(i - j) >= blen:
            h_pert[i:i+blen], h_pert[j:j+blen] = h_pert[j:j+blen].copy(), h_pert[i:i+blen].copy()
        # Ensure balanced
        if int(np.sum(h_pert)) != n // 2:

        # Quick local search
