    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c = n - 1  # center index
    m = corr.max()
    diff = (corr - m) / tau
    # Clip for numerical stability
    diff = np.clip(diff, -50, 0)
    exps = np.exp(diff)
    S = exps.sum()
    p = exps / S  # softmax weights
    F = (m + tau * np.log(S)) * dx
    
    grad = np.zeros(n)
    # Center term (k=0)
    grad += p[c] * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., n-1
    for k in range(1, n):
        pk = p[c + k]
        if pk < 1e-15:
            continue
        # j is upper index: j from k to n-1, contribution (1-h[j-k])
        grad[k:] += pk * (1.0 - h[:-k] if k < n else np.zeros(0))
        # Wait, h[:-k] has length n-k, and grad[k:] has length n-k. Good.
        # j is lower index: j from 0 to n-1-k, contribution -h[j+k]
        grad[:n-k] += pk * (-h[k:])

    for k in range(1, n):

        if pk < 1e-15:

        # j is lower index: j from 0 to n-1-k, contribution (1-h[j+k])

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:

            x[mask] += res / cnt

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-8
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_val_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = n - 1
    m = float(corr.max())
    diff = (corr - m) / tau
    diff = np.clip(diff, -50.0, 0.0)
    exps = np.exp(diff)
    S = float(exps.sum())
    p = exps / S
    val = (m + tau * np.log(S)) * dx
    
    # Active set optimization
    active = np.where(p > 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        pk = p[idx]
        k = int(idx - c)
        if k == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += pk * (1.0 - h[:n-k])
            grad[:n-k] += pk * (-h[k:])
        else:
            d = -k
            grad[:n-d] += pk * (1.0 - h[d:])
            grad[d:] += pk * (-h[:n-d])
    grad *= dx
    return val, grad

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
