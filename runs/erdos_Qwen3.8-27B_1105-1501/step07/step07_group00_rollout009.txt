def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    G[n-1, :] = 1.0 - 2.0 * h  # lag 0
    for L in range(1, n):
        m_L = n - L
        idx = n - 1 + L
        G[idx, L:] += (1.0 - h[:m_L])
        G[idx, :m_L] += -h[L:]
    for g in range(1, n):
        m_g = n - g
        idx = n - 1 - g
        G[idx, :m_g] += (1.0 - h[g:])
        G[idx, g:] += -h[:m_g]
    return G

def _line_search(h, d, dx, radius):
    """Exact line search along h + t*d."""
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find t_hi from bounds
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Evaluate at grid points
    n_grid = 50
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Refine with golden section
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    # Check boundaries
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
    if f0 < opt_val:
        opt_t, opt_val = 0.0, f0
    if fhi < opt_val:
        opt_t, opt_val = t_hi, fhi
    
    return opt_t, opt_val

def _lp_direction(h, scores, radius):
    """Full-lag LP to find descent direction."""
    n = len(h)
    G = _build_G(h)
    m = len(scores)
    
    # LP: minimize tau
    # s.t. G[k,d] - tau <= -scores[k] for all k
    #      sum(d) = 0
    #      -radius <= d_i <= radius
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize tau
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            d = res.x[:n]
            d -= float(np.mean(d))  # Ensure zero mean (numerical)
            return d
    except Exception:
        pass
    return None

def _gradient_fallback(h, scores, radius):
    """Weighted gradient descent direction as fallback."""
    n = len(h)
    mx = float(np.max(scores))
    beta = 30.0
    shifted = beta * (scores - mx)
    w = np.exp(np.clip(shifted, -50, 0))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _build_G(h)
    g = G.T @ w
    g -= float(np.mean(g))
    # Clip to respect bounds
    g = np.clip(g, -radius, radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g
