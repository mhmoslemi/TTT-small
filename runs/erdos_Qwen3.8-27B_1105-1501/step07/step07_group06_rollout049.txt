    gr = (np.sqrt(5)-1)/2
    lo, hi = a_lo, a_hi
    c1 = hi - gr*(hi-lo); c2 = lo + gr*(hi-lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi-lo < 1e-13: break
        if f1 < f2: hi, c2, f2 = c2, c1, f1; c1 = hi-gr*(hi-lo); f1 = f(c1)
        else: lo, c1, f1 = c1, c2, f2; c2 = lo+gr*(hi-lo); f2 = f(c2)
    return 0.5*(lo+hi), f(0.5*(lo+hi))

def _full_opt(h0, deadline, max_stall=60):
    n = len(h0); target = n/2.0
    h = _project(h0, target).copy()
    cur = _exact_c5(h); best_h = h.copy(); best_c = cur
    radius = 0.15; stall = 0; rng = np.random.default_rng(123)
    while time.time() < deadline and stall < max_stall:
        beta = 30.0
        g = _full_grad(h, beta)
        d = -g; d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14: stall += 1; continue
        d /= dn
        a, val = _line_search(h, d, radius)
        if a != 0 and val < cur - 1e-14:
            h_new = _project(h + a*d, target)
            nc = _exact_c5(h_new)
            if nc < cur - 1e-14:
                h = h_new; cur = nc; stall = 0
                if cur < best_c: best_c = cur; best_h = h.copy()
                radius = min(radius*1.05, 0.4)
            else: stall += 1; radius = max(0.03, radius*0.7)
        else:
            stall += 1; radius = max(0.03, radius*0.75)
    return best_h, best_c

def _binary_sa(hb, deadline, max_iter=40000, T0=0.003):
    n = len(hb); cur = hb.copy(); cc = _exact_c5(cur)
    best_h = cur.copy(); best_c = cc
    rng = np.random.default_rng(555)
    T = T0; stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300: break
        if not ones or not zeros: break
        oi = ones[int(rng.integers(len(ones)))]
        oz = zeros[int(rng.integers(len(zeros)))]
        cur[oi] = 0; cur[oz] = 1
        nc = _exact_c5(cur)
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d/max(T, 1e-12)):
            cc = nc; T *= 0.999
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cc < best_c: best_c = cc; best_h = cur.copy(); stall = 0
        else:
            cur[oi] = 1; cur[oz] = 0; stall += 1
    return best_h, best_c

def _pair_sweep(hb, deadline):
    n = len(hb); cur = hb.copy(); cc = _exact_c5(cur)
    best_h = cur.copy(); best_c = cc
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for oi in ones:
        if time.time() > deadline: break
        for oz in zeros:
            cur[oi] = 0; cur[oz] = 1
            ct = _exact_c5(cur)
            if ct < cc - 1e-14:
                cc = ct
                if ct < best_c: best_c = ct; best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                break
            else: cur[oi] = 1; cur[oz] = 0
    return best_h, best_c

def _perturb_anti(x, amp, rng):
    m = len(x)
    t = np.arange(m, dtype=float)
    p = np.zeros(m)
    for _ in range(3):
        p += np.sin(2*np.pi*rng.uniform(0.5,5)*t/m + rng.uniform(0, 2*np.pi))
    p *= amp/3
    p += amp*0.3*rng.normal(0, 1, m)
    return np.clip(x + p, 0, 1)
