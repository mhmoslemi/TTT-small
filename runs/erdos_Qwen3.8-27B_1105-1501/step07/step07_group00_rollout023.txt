def _project(v, target):
    # Water-filling projection
    n = len(v)
    v = np.array(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]

def _active_lp(h, scores, radius, n_active, gap):

    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            d = res.x[:n].copy()
            d -= float(np.mean(d))  # Ensure sum=0
            return d
    except Exception:
        pass
    return None

def _weighted_gradient(h, scores, beta, radius):

    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Clip to respect box constraints
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    gn = float(np.linalg.norm(g))
    if gn > 1e-15:
        g /= gn
    return g

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Intensive LP local optimization (30%)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _local_opt(h_init, d1, radius_init=0.15, max_iters=10000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair sweep + coordinate descent (15%)
    d2 = t0 + budget_s * 0.42
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _pair_sweep(h2, min(d2, time.time() + budget_s * 0.06))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    if time.time() < d2:
        h3 = _project(best_h.copy(), len(best_h) / 2.0)
        h3, c5_3 = _coord_descent(h3, d2, max_rounds=300)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.67
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    # Try a range of even n values
    n_candidates = sorted(set([n_curr + dn for dn in range(-40, 41, 4) if 80 <= n_curr + dn <= 260 and (n_curr + dn) % 2 == 0]))
    rng.shuffle(n_candidates)
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.12, max_iters=2000)
        if c5_r < best_c5 + 0.003:
            sub_dl2 = min(d3, time.time() + budget_s * 0.007)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, max_rounds=60)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Binary optimization (20%)
    d4 = t0 + budget_s * 0.87
    n_b = len(best_h)
    h_bin = _to_binary(best_h)
    c5_bin_init = _exact_c5(h_bin)
    # Also try from a slightly perturbed fractional start
    h_frac = _smooth_perturb(best_h, 0.03, rng)
    h_bin2 = _to_binary(h_frac)
    c5_bin2_init = _exact_c5(h_bin2)
    if c5_bin2_init < c5_bin_init:
        h_bin = h_bin2
        c5_bin_init = c5_bin2_init
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=100000, T_init=0.005)
    if time.time() < d4:
        h_bin_s, c5_bin_s = _pair_sweep(h_bin, min(d4, time.time() + budget_s * 0.04), max_rounds=100)
        if c5_bin_s < c5_bin:
            h_bin, c5_bin = h_bin_s, c5_bin_s
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on best binary
    if time.time() < d4 - 2:
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl3, radius_init=0.08, max_iters=1500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Basin hopping (13%)
    d5 = deadline
    scale = 0.015
    n_restarts = 0
    while time.time() < d5 - 30 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 30, time.time() + budget_s * 0.005)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.08, max_iters=600)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.80)
        else:
            scale = min(0.12, scale * 1.06)

    # Phase 6: Final polish (remaining)
    if deadline - time.time() > 5:
        n_f = len(best_h)
        h_f = _project(best_h.copy(), n_f / 2.0)
        h_f, c5_f = _local_opt(h_f, deadline, radius_init=0.05, max_iters=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
