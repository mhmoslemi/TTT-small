The problem is to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx, subject to ∫₀² h(x) dx = 1.

The current record is C₅ ≤ 0.38085, and the previous program achieved 0.380889. I need to improve on this.

Let me think about what makes this problem hard and what approaches could work:

1. The objective is max_k of (2n-1) quadratic functions
2. At the optimum, many lags are "active" (near the maximum)
3. Moving one or two coordinates helps some lags but hurts others
4. A dense direction that moves many coordinates simultaneously can reduce multiple active lags together

Let me design the algorithm:

2. **Dense active-row optimization**: 
   - Identify the top-K most active correlation rows
   - Build their gradient matrix

   - Perform exact line search along that direction
   - Accept if improved

3. **Multi-resolution exploration**: Try different n values
4. **Binary refinement**: Convert to binary and use swaps
5. **Final polish**: Fine-grained optimization

Let me think about the line search more carefully. Given direction d (zero-sum, normalized), the objective along the line is:

max_r [S_r + t*U_r - t²*B_r]

where S_r = R_r(h), U_r = R_r(d, 1-h) - R_r(h, d), B_r = R_r(d, d).

This is a max of concave quadratics, so it's not concave in general. But we can do:
- Grid search over feasible t
- For each t, compute the max exactly
- Golden section refinement

The feasible interval for t is determined by box constraints: h_i + t*d_i ∈ [0,1] for all i.

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on theta: sum(clip(v - theta, 0, 1)) = target
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_gradient(h, lag):
    """Gradient of R_r with respect to h, where lag = r - (n-1)."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])   # (1 - h_j) for j in [0, m)
            g[:m] -= h[lag:]            # -h_{j+lag} for j in [0, m)
    else:
        p = -lag
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def _line_search_quad(S, U, B, t_lo, t_hi, n_grid=40):
    """Minimize max(S + t*U - t^2*B) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S))
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Vectorized: shape (n_grid, L)
    vals = np.max(S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    
    # Golden section refinement around best
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), best_val
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1*c1 * B))
    f2 = float(np.max(S + c2 * U - c2*c2 * B))
    
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1*c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2*c2 * B))
    
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S + opt_t * U - opt_t*opt_t * B))
    
    # Also check endpoints
    v_lo = float(np.max(S + t_lo * U - t_lo*t_lo * B))
    v_hi = float(np.max(S + t_hi * U - t_hi*t_hi * B))
    
    best_val = min(best_val, opt_val, v_lo, v_hi)
    if best_val == opt_val:
        return opt_t, best_val
    if best_val == v_lo:
        return t_lo, best_val
    return t_hi, best_val

def _feasible_interval(h, d, cap=1.0):
    """Find t range where h + t*d stays in [0,1]."""
    t_lo = -cap
    t_hi = cap
    mask_pos = d > 1e-14
    mask_neg = d < -1e-14
    if np.any(mask_pos):
        t_lo = max(t_lo, float(np.max(-h[mask_pos] / d[mask_pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[mask_pos]) / d[mask_pos])))
    if np.any(mask_neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[mask_neg]) / d[mask_neg])))
        t_hi = min(t_hi, float(np.min(-h[mask_neg] / d[mask_neg])))
    return t_lo, t_hi

def _dense_direction(h, S, K, target_mode, rng, weight_type=0):
    """Solve weighted least-norm direction to reduce top-K active rows toward target."""
    n = len(h)
    L = 2 * n - 1
    # Select top-K rows
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argsort(S)[-K:]
    
    # Compute target
    u = S[idx]
    if target_mode == 0:  # mean
        t_target = np.mean(u) - 0.02 * np.ptp(u)
    elif target_mode == 1:  # min
        t_target = np.min(u)
    elif target_mode == 2:  # median
        t_target = np.median(u)
    elif target_mode == 3:  # slightly below mean
        t_target = np.mean(u) - 0.05 * np.ptp(u)
    else:
        t_target = np.mean(u)
    t_target = min(t_target, float(np.max(S)))
    
    # Build gradient matrix A (K rows)
    A = np.zeros((K, n))
    for i, r in enumerate(idx):
        lag = r - (n - 1)
        A[i] = _row_gradient(h, lag)
    
    # Add sum constraint row
    A_full = np.vstack([A, np.ones(n)])
    b_full = np.zeros(K + 1)
    for i in range(K):
        b_full[i] = t_target - S[idx[i]]
    b_full[K] = 0.0  # sum constraint
    
    # Weighted least-norm: min d^T W d s.t. A_full @ d = b_full
    # Solution: d = W^{-1} A^T (A W^{-1} A^T)^{-1} b
    
    if weight_type == 0:
        W_inv = np.ones(n)  # identity metric
    elif weight_type == 1:
        W_inv = 1.0 / (h * (1.0 - h) + 1e-6)  # boundary-aware
    else:
        W_inv = np.ones(n)
    
    A_scaled = A_full * W_inv[None, :]
    M = A_scaled @ A_full.T  # (K+1) x (K+1)
    ridge = 1e-8 * np.trace(M) / (K + 1)
    M += ridge * np.eye(K + 1)
    
    try:
        lam = np.linalg.solve(M, b_full)
    except np.linalg.LinAlgError:
        return None
    
    d = W_inv * (A_full.T @ lam)
    
    # Verify zero-sum (should be approximately zero due to constraint)
    d -= np.mean(d)
    
    # Check finite
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-15:
        return None
    
    return d

def _soft_direction(h, S, beta=15.0):
    """Smoothed direction using all rows weighted by softmax."""
    n = len(h)
    L = 2 * n - 1
    M = float(np.max(S))
    w = np.exp(beta * (S - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, L // 10)
        idx = np.argsort(S)[-top_k:]
        w = np.zeros(L)
        w[idx] = 1.0 / top_k
    else:
        w /= s
    
    # Compute weighted gradient
    d = np.zeros(n)
    for r in range(L):
        if w[r] > 1e-10:
            lag = r - (n - 1)
            d += w[r] * _row_gradient(h, lag)
    
    d -= np.mean(d)
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-15:
        return None
    return d

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        p = -lag
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def _line_search_quad(S, U, B, t_lo, t_hi, n_grid=40):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S))
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S + opt_t * U - opt_t * opt_t * B))
    v_lo = float(np.max(S + t_lo * U - t_lo * t_lo * B))
    v_hi = float(np.max(S + t_hi * U - t_hi * t_hi * B))
    best_val = min(best_val, opt_val, v_lo, v_hi)
    if best_val == opt_val:
        return opt_t, best_val
    if best_val == v_lo:
        return t_lo, best_val
    return t_hi, best_val

def _feasible_interval(h, d, cap=1.0):
    t_lo = -cap
    t_hi = cap
    mp = d > 1e-14
    mn = d < -1e-14
    if np.any(mp):
        t_lo = max(t_lo, float(np.max(-h[mp] / d[mp])))
        t_hi = min(t_hi, float(np.min((1.0 - h[mp]) / d[mp])))
    if np.any(mn):
        t_lo = max(t_lo, float(np.max((1.0 - h[mn]) / d[mn])))
        t_hi = min(t_hi, float(np.min(-h[mn] / d[mn])))
    return t_lo, t_hi

def _dense_direction(h, S, K, target_mode, weight_type):
    n = len(h)
    L = 2 * n - 1
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argsort(S)[-K:]
    u = S[idx]
    spread = max(float(np.ptp(u)), 1e-10)
    if target_mode == 0:
        t_target = float(np.mean(u)) - 0.01 * spread
    elif target_mode == 1:
        t_target = float(np.min(u))
    elif target_mode == 2:
        t_target = float(np.median(u))
    else:
        t_target = float(np.mean(u)) - 0.04 * spread
    t_target = min(t_target, float(np.max(S)))
    A = np.zeros((K, n))
    for i in range(K):
        lag = int(idx[i]) - (n - 1)
        A[i] = _row_gradient(h, lag)
    A_full = np.vstack([A, np.ones(n)])
    b_full = np.zeros(K + 1)
    for i in range(K):
        b_full[i] = t_target - S[idx[i]]
    b_full[K] = 0.0
    if weight_type == 1:
        W_inv = 1.0 / (h * (1.0 - h) + 1e-4)
    else:
        W_inv = np.ones(n)
    A_scaled = A_full * W_inv[None, :]
    M = A_scaled @ A_full.T
    ridge = 1e-8 * float(np.trace(M)) / (K + 1)
    M += ridge * np.eye(K + 1)
    try:
        lam = np.linalg.solve(M, b_full)
    except np.linalg.LinAlgError:
        return None
    d = W_inv * (A_full.T @ lam)
    d -= np.mean(d)
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-14:
        return None
    return d

def _soft_direction(h, S, beta=15.0):
    n = len(h)
    L = 2 * n - 1
    mx = float(np.max(S))
    w = np.exp(beta * (S - mx))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, L // 10)
        idx = np.argsort(S)[-top_k:]
        w = np.zeros(L)
        w[idx] = 1.0 / top_k
    else:
        w /= s
    d = np.zeros(n)
    for r in range(L):
        if w[r] > 1e-10:
            lag = r - (n - 1)
            d += w[r] * _row_gradient(h, lag)
    d -= np.mean(d)
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-14:
        return None
    return d

def _pair_direction(h, S, n_pairs, beta, rng):
    n = len(h)
    mx = float(np.max(S))
    w = np.exp(beta * (S - mx))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, (2 * n - 1) // 10)
        idx = np.argsort(S)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / top_k
    else:
        w /= s
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        lag = r - (n - 1)
        G[r] = _row_gradient(h, lag)
    grad = G.T @ w
    sort_idx = np.argsort(grad)
    sinks = sort_idx[:n_pairs]
    sources = sort_idx[-n_pairs:]
    best_h_try = None
    best_c5_try = float('inf')
    one_mh = 1.0 - h
    for ii in range(n_pairs):
        si = int(sinks[ii])
        so = int(sources[ii])
        if si == so:
            continue
        t_lo_p = max(-h[si], h[so] - 1.0, -0.25)
        t_hi_p = min(1.0 - h[si], h[so], 0.25)
        if t_hi_p - t_lo_p < 1e-13:
            continue
        d = np.zeros(n)
        d[si] = 1.0
        d[so] = -1.0
        T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        S0 = np.correlate(h, 1.0 - h, mode='full')
        t_opt, _ = _line_search_quad(S0, T, C, t_lo_p, t_hi_p, n_grid=30)
        h_try = h.copy()
        h_try[si] += t_opt
        h_try[so] -= t_opt
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5_try:
            best_c5_try = c5_try
            best_h_try = h_try.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
