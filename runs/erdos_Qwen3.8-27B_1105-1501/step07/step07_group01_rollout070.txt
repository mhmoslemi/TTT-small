def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)
