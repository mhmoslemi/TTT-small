def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _correlation_scores(h):
    """Compute the n distinct score values (positive lags 0 to n-1)."""
    n = len(h)
    scores = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(n):
        m = n - k
        scores[k] = np.dot(h[k:], one_minus_h[:m])
    return scores

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_grad(h, beta):
    """Compute gradient of smooth max approximation.
    
    F(h) = (1/beta) * log(sum_k exp(beta * score_k))
    Returns (F_value, gradient) scaled so that F * (2/n) gives C5 estimate.
    """
    n = len(h)
    dx = 2.0 / n
    scores = _correlation_scores(h)
    # For numerical stability
    s_max = scores.max()
    exp_scores = np.exp(beta * (scores - s_max))
    log_sum = np.log(np.sum(exp_scores)) 
    F_val = (s_max + log_sum / beta) * dx
    
    # Weights
    w = exp_scores / np.sum(exp_scores)
    
    # Gradient computation using convolution
    one_minus_h = 1.0 - h
    # First part: conv(w, 1-h)[:n]
    part1 = np.convolve(w, one_minus_h, mode='full')[:n]
    # Second part: conv(w[::-1], h)[n-1:2*n-1]
    part2 = np.convolve(w[::-1], h, mode='full')[n-1:2*n-1]
    
    grad = (part1 - part2) * dx  # Scale by dx to match C5
    
    return F_val, grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _coord_descent(h, deadline, max_sweeps=3000, seed=123):

    stall = 0

        if time.time() > deadline:
            break
        if stall > 100:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            for _ in range(40):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1 * g))
                f2 = float(np.max(S0 + m2 * g))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx

def _smooth_perturb(h, amplitude, rng):

def _exact_c5(h):

def _scores(h):

    n = len(h)

    """Compute (smooth_max_value, gradient) in count units (not scaled by dx)."""
    n = len(h)
    sc = _scores(h)

    # Clip for stability
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-300:
        w = np.full(n, 1.0 / n)
    else:
        w = exp_s / total
    F_val = s_max + np.log(total) / beta
    one_minus_h = 1.0 - h
    part1 = np.convolve(w, one_minus_h, mode='full')[:n]

    h2, c5_2 = _lp_active(best_h, d2, radius_init=0.10, stall_limit=80, top_k=20)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Coordinate descent (15%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _coord_descent(best_h, d3, max_sweeps=5000, seed=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Second smooth max with higher beta (10%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _smooth_descent(best_h, d4, beta_start=10.0, beta_end=200.0, lr=0.003, momentum=0.90, seed=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary search + LP polish (15%)
    d5 = t0 + budget_s * 0.80
    h_bin = _to_binary(best_h)
    h5, c5_5 = _binary_search(h_bin, d5, max_iter=80000, seed=400)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < deadline - 40:
        h_bp = _project(h5.copy(), n0 / 2.0)
        sub_dl = min(deadline - 20, time.time() + budget_s * 0.03)
        h5p, c5_5p = _lp_active(h_bp, sub_dl, radius_init=0.06, stall_limit=50, top_k=20)
        if c5_5p < best_c5:
            best_c5 = c5_5p
            best_h = h5p.copy()

    # Phase 6: Basin hopping (15%)
    d6 = deadline - 10
    scale = 0.03
    n_iter = 0
    while time.time() < d6 and n_iter < 80:
        n_iter += 1
        rem = d6 - time.time()
        if rem < 5:
            break
        h_p = _perturb(best_h, scale, rng)
        sub_dl = min(d6, time.time() + min(rem * 0.35, budget_s * 0.012))
        h_t, c5_t = _smooth_descent(h_p, sub_dl, beta_start=10.0, beta_end=80.0, lr=0.005, momentum=0.90, max_iter=600, seed=500 + n_iter)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.88)
        else:
            scale = min(0.08, scale * 1.06)

    # Phase 7: Multi-n exploration
    if time.time() < deadline - 20:
        for dn in range(-12, 13, 3):
            if time.time() > deadline - 15:
                break
            n_try = n0 + dn
            if n_try < 60 or n_try > 280:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(deadline - 15, time.time() + 3.0)
            h_r, c5_r = _smooth_descent(h_r, sub_dl, beta_start=5.0, beta_end=60.0, lr=0.005, momentum=0.9, max_iter=300, seed=700 + dn)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final polish
    if time.time() < deadline - 3:
        h_f, c5_f = _coord_descent(best_h, deadline - 1, max_sweeps=3000, seed=999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
