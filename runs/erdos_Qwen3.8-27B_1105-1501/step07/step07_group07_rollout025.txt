import numpy as np
from scipy.optimize import minimize
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, r):
    n = len(h)
    grad = np.zeros(n)
    lag = r - (n - 1)
    if lag >= 0:
        k = lag
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            idx1 = np.arange(k, n)
            grad[idx1] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] += -h[idx2 + k]
    else:
        g = -lag
        if g == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            idx1 = np.arange(0, n - g)
            grad[idx1] += (1.0 - h[idx1 + g])
            idx2 = np.arange(g, n)
            grad[idx2] += -h[idx2 - g]
    return grad

def _lse_grad(h, tau):
    n = len(h)
    s = _scores(h)
    m = np.max(s)
    if tau < 1e-8:
        max_idx = int(np.argmax(s))
        return _score_grad(h, max_idx)
    ps = np.exp((s - m) / tau)
    total = np.sum(ps)
    weights = ps / total
    # Only compute gradient for significant weights
    significant = weights > 1e-6
    grad = np.zeros(n)
    for r in np.where(significant)[0]:
        grad += weights[r] * _score_grad(h, int(r))
    return grad

def _lse_optimize(h, tau, max_iter=200, lr_init=0.01):
    """Projected gradient descent on LSE objective."""
    n = len(h)
    target = n / 2.0
    h_curr = _project_sum(h.copy(), target)
    c5 = _exact_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5
    lr = lr_init
    for it in range(max_iter):
        g = _lse_grad(h_curr, tau)
        # Project gradient onto the constraint sum(h)=target (subtract mean)
        g_proj = g - np.mean(g)
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(50):
            h_try = _project_sum(h_curr - alpha * g_proj, target)
            c5_try = _exact_c5(h_try)
            if c5_try < c5 - 1e-14:
                h_curr = h_try
                c5 = c5_try
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
        lr = min(lr * 1.1, 0.1)
    return best_h, best_c5

def _lse_anneal(h, deadline, tau_init=1.0, tau_final=0.01, n_stages=20):
    """Anneal tau from tau_init to tau_final."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(best_h)
    taus = np.geomspace(tau_init, tau_final, n_stages)
    for tau in taus:
        if time.time() > deadline:
            break
        h_new, c5_new = _lse_optimize(best_h, tau, max_iter=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

import numpy as np
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_grad(h, r):
    n = len(h)
    grad = np.zeros(n)
    lag = r - (n - 1)
    if lag >= 0:
        k = lag
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            idx1 = np.arange(k, n)
            grad[idx1] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] += -h[idx2 + k]
    else:
        g = -lag
        if g == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            idx1 = np.arange(0, n - g)
            grad[idx1] += (1.0 - h[idx1 + g])
            idx2 = np.arange(g, n)
            grad[idx2] += -h[idx2 - g]
    return grad

def _lse_grad(h, tau):
    n = len(h)
    s = _scores(h)
    m = float(np.max(s))
    if tau < 1e-8:
        return _score_grad(h, int(np.argmax(s)))
    ps = np.exp((s - m) / tau)
    total = float(np.sum(ps))
    weights = ps / total
    grad = np.zeros(n)
    sig = np.where(weights > 1e-5)[0]
    for r in sig:
        grad += weights[r] * _score_grad(h, int(r))
    return grad

    target = n / 2.0
    h_curr = _project_sum(h.copy(), target)
    c5 = _exact_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5
    mom = np.zeros(n)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _lse_grad(h_curr, tau)
        g_proj = g - float(np.mean(g))
        mom = momentum * mom + (1.0 - momentum) * g_proj
        norm_g = float(np.linalg.norm(g_proj))
        if norm_g < 1e-10:
            break
        alpha = lr / (1.0 + norm_g)
        accepted = False
        for _ in range(60):
            h_try = _project_sum(h_curr - alpha * mom, target)
            c5_try = _exact_c5(h_try)
            if c5_try < c5 - 1e-14:
                h_curr = h_try
                c5 = c5_try
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        if not accepted:
            lr *= 0.7
            if lr < 1e-6:
                break
        else:
            lr = min(lr * 1.05, 0.15)
    return best_h, best_c5

def _lse_anneal(h, deadline, tau_init=2.0, tau_final=0.003, n_stages=30):
    best_h = _project_sum(h.copy(), len(h) / 2.0)
    best_c5 = _exact_c5(best_h)
    taus = np.geomspace(tau_init, tau_final, n_stages)
    for si in range(len(taus)):
        if time.time() > deadline:
            break
        tau = taus[si]
        remaining = max(1, int(0.9 * (deadline - time.time()) / max(0.1, deadline - time.time() + (len(taus) - si) * 0.2)))
        stage_dl = min(deadline, time.time() + max(1.0, (deadline - time.time()) / (len(taus) - si)))
        h_new, c5_new = _lse_pg(best_h, tau, stage_dl, max_iter=min(300, remaining * 10))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amp):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert *= amp
    pert += amp * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _lp_polish(h, deadline, radius_init=0.1, top_k_init=4, max_top_k=10, stall_limit=80):
    from scipy.optimize import linprog
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        top_idx = np.argsort(scores)[::-1][:top_k]
        top_scores = scores[top_idx]
        G_rows = np.array([_score_grad(h, int(r)) for r in top_idx])
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G_rows
        A_ub[:, n] = -1.0
        b_ub = -top_scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
            else:
                d = None
        except Exception:
            d = None
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search_lp(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.03, 0.2)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 40 == 0 and radius < 0.015:
                radius = 0.06
                top_k = top_k_init
    return best_h, best_c5

def _line_search_lp(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    best_t = grid_t[0]
    best_val = float(np.max(S0) * dx)
    for t in grid_t:
        val = float(np.max(S0 + t * T - t * t * C)) * dx
        if val < best_val:
            best_val = val
            best_t = t
    lo_b = grid_t[max(0, int(np.argmin([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])) - 2)]
    hi_b = grid_t[min(n_grid - 1, int(np.argmin([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])) + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-12:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        best_t = (lo_b + hi_b) / 2.0
        best_val = float(np.max(S0 + best_t * T - best_t * best_t * C)) * dx
    return best_t, best_val

def _coord_descent(h, deadline, max_rounds=30):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old_val = cur[j]
            best_local = old_val
            best_local_c5 = cur_c5
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if frac == old_val:
                    continue
                cur[j] = frac
                tmp = _project_sum(cur, target)
                c_try = _exact_c5(tmp)
                if c_try < best_local_c5 - 1e-14:
                    best_local_c5 = c_try
                    best_local = frac
            if best_local != old_val:
                cur = _project_sum(cur.copy(), target)
                cur[closest_idx(cur, old_val)] = best_local if False else cur[closest_idx(cur, old_val)]
                cur = _project_sum(cur, target)
                cur_c5 = _exact_c5(cur)
                if cur_c5 < cur_c5 + 1e-14:
                    pass
                if best_local_c5 < cur_c5:
                    cur = _project_sum(cur, target)
                    cur[j] = min(1.0, max(0.0, best_local))
                    cur = _project_sum(cur, target)
                    cur_c5 = best_local_c5
                improved = True
        if not improved:
            break
    return cur, cur_c5

def closest_idx(arr, val):
    return int(np.argmin(np.abs(arr - val)))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    n0 = (n0 // 2) * 2
    if n0 != len(h0):
        h0 = _resample(h0, n0)
        n0 = len(h0)

    best_h = _project_sum(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: LSE annealing (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lse_anneal(best_h, d1, tau_init=3.0, tau_final=0.002, n_stages=25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish on best (15%)
    d2 = t0 + budget_s * 0.40
    h2 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _lp_polish(h2, d2, radius_init=0.1, top_k_init=4, max_top_k=10, stall_limit=80)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.65
    n_best = len(best_h)
    n_candidates = sorted(set(n_best + dn for dn in range(-36, 37, 4)
                              if 96 <= n_best + dn <= 280) |
                          {n_best + dn for dn in [-12, -8, -4, 4, 8, 12]
                           if 96 <= n_best + dn <= 280})
    n_candidates = [n for n in n_candidates if n % 2 == 0]
    tried = set()
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try in tried or n_try == n_best:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        stage_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_pg(h_r, 0.1, stage_dl, max_iter=150, momentum=0.6)
        if time.time() < d3 - 3:
            stage_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_polish(h_r, stage_dl2, radius_init=0.06, top_k_init=3, max_top_k=8, stall_limit=40)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Binary exploration (15%)
    d4 = t0 + budget_s * 0.80
    h_bin = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=999)
    sub_dl2 = min(d4, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _binary_local(h_bin, sub_dl2, max_rounds=40)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Try different binary at different n
    if time.time() < d4 - 5:
        for n_try in [n_best - 8, n_best + 8, n_best - 16, n_best + 16]:
            if n_try < 96 or n_try > 280 or n_try % 2 != 0:
                continue
            if time.time() > d4:
                break
            h_r = _resample(best_h, n_try)
            h_b = _to_binary(h_r)
            sub_dl3 = min(d4, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(h_b, sub_dl3, max_iter=15000, T_init=0.002, seed=n_try)
            sub_dl4 = min(d4, time.time() + budget_s * 0.01)
            h_b, c5_b = _binary_local(h_b, sub_dl4, max_rounds=20)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 5: Basin hopping with LSE (10%)
    d5 = t0 + budget_s * 0.90
    scale = 0.04
    for _ in range(80):
        if time.time() > d5:
            break
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _lse_pg(h_pert, 0.05, sub_dl, max_iter=80, momentum=0.5)
        if time.time() < d5 - 2:
            sub_dl2 = min(d5, time.time() + budget_s * 0.003)
            h_t, c5_t = _lp_polish(h_t, sub_dl2, radius_init=0.04, top_k_init=3, max_top_k=6, stall_limit=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 6: Final intensive polish (10%)
    d6 = deadline
    n_cur = len(best_h)
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6 = _project_sum(best_h.copy(), n_cur / 2.0)
    h6, c5_6 = _lp_polish(h6, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=7, stall_limit=60)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6 - 5:
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        h6 = _project_sum(best_h.copy(), n_cur / 2.0)
        h6, c5_6 = _lse_pg(h6, 0.005, sub_dl, max_iter=100, momentum=0.3)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if time.time() < d6 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d6, time.time() + budget_s * 0.01)
        h_b, c5_b = _binary_local(h_b, sub_dl, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d6 - 2:
        h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_polish(h6, d6, radius_init=0.02, top_k_init=3, max_top_k=5, stall_limit=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
