    d = -g / ng

import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    h_var = cp.Variable(n)
    T = cp.Variable()
    constraints = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1, T >= 0]
    for k in range(2 * n - 1):

        i_hi = min(n - 1, k)
        if i_lo > i_hi:
