import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, l):
    """Compute gradient of raw_corr at output index corresponding to lag l."""
    n = len(h)
    G = np.zeros(n)
    if l >= 0:
        L = n - l
        if L > 0:
            G[:L] += (1.0 - h[l:l+L])
            G[l:l+L] -= h[:L]
    else:
        s = -l
        L = n - s
        if L > 0:
            G[s:s+L] += (1.0 - h[:L])
            G[:L] -= h[s:s+L]
    return G

def _line_search_exact(h, s, dx, raw0):
    """Find best alpha in [0,1] such that h + alpha*s is feasible and minimizes C5."""
    n = len(h)
    # Compute T and Q
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    
    # Feasible range of alpha
    a_lo, a_hi = 0.0, 1.0
    pos = s > 1e-14
    neg = s < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / s[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-s[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(raw0) * dx)
    
    def f(alpha):
        return float(np.max(raw0 + alpha * T - alpha * alpha * Q)) * dx
    
    # Grid search
    n_grid = 40
    alphas = np.linspace(a_lo, a_hi, n_grid)
    vals = np.array([f(a) for a in alphas])
    idx = int(np.argmin(vals))
    
    # Golden section refinement
    if idx > 0 and idx < n_grid - 1:
        lo_r = alphas[max(0, idx-1)]
        hi_r = alphas[min(n_grid-1, idx+1)]
    else:
        lo_r, hi_r = a_lo, a_hi
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(40):
        if hi_r - lo_r < 1e-10:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr * (hi_r - lo_r)
            f1 = f(c1)
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = f(c2)
    best_a = (lo_r + hi_r) / 2.0
    best_v = f(best_a)
    
    # Also check grid best
    if vals[idx] < best_v:
        best_v = vals[idx]
        best_a = alphas[idx]
    
    # Clamp
    best_a = max(a_lo, min(a_hi, best_a))
    best_v = f(best_a)
    
    return best_a, best_v

def _lp_direction(h, raw, K, radius):
    """Solve LP to find direction that decreases top K lags."""
    n = len(h)
    n_corr = 2 * n - 1
    
    # Select top K lags
    top_idx = np.argsort(-raw)[:K]
    
    # Compute gradients for these lags
    grads = []
    for t in top_idx:
        l = int(t) - (n - 1)
        g = _lag_gradient(h, l)
        norm = np.linalg.norm(g)
        if norm > 1e-12:
            grads.append(g / norm)
        else:
            grads.append(g)
    
    if not grads:
        return None
    
    A_g = np.column_stack(grads)  # K x n
    
    # LP: max tau
    # s.t. A_g @ s <= -tau  =>  A_g @ s + tau <= 0
    #      sum(s) = 0
    #      -radius * min(h, 1) <= s_i <= radius * min(1-h, 1)  (approx)
    #      Actually: -h_i * radius_eff <= s_i <= (1-h_i) * radius_eff
    #      Let's use: -radius * h_i <= s_i <= radius * (1-h_i) for small radius
    #      But we want sum=0, so let's just use -radius <= s_i <= radius and sum=0
    
    # Variables: [s_0, ..., s_{n-1}, tau] (n+1 vars)
    # Objective: minimize -tau => c = [0, ..., 0, -1]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality: A_g @ s + tau <= 0 for each row
    # In linprog: A_ub @ x <= b_ub
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A_g
    A_ub[:, n] = 1.0  # tau coefficient
    b_ub = np.zeros(K)
    
    # Bounds
    lo = -radius * np.maximum(h, 1e-8)  # s_i >= -radius * h_i
    hi = radius * np.maximum(1.0 - h, 1e-8)  # s_i <= radius * (1-h_i)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))  # tau >= 0
    
    # Equality: sum(s) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x[-1] > 1e-10:
            s = res.x[:n]
            s -= np.mean(s)  # Ensure zero sum
            return s
    except:
        pass
    return None

def _smooth_grad(h, raw, beta):
    """Soft-max weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    mx = float(np.max(raw))
    v = np.clip(beta * (raw - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(raw)] = 1.0
    else:
        w /= ws
    
    g = np.zeros(n)
    for t in range(2*n-1):
        wt = w[t]
        if wt < 1e-15:
            continue
        l = t - (n-1)
        g += wt * _lag_gradient(h, l)
    
    g -= np.mean(g)
    return -g  # Negative gradient (descent direction)

def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        d[b*bs:(b+1)*bs] = np.mean(g[b*bs:(b+1)*bs])
    if nb * bs < n:
        d[nb*bs:] = np.mean(g[nb*bs:])
    d -= np.mean(d)
    return d

def _pair_opt(h, deadline, max_it=30000):
    """Pair swap optimization."""
    n = len(h)
    target = n / 2.0

    cur = float(np.max(raw) * dx)
    best_h = h.copy()

    stall = 0

            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:

        d = np.zeros(n)

                cur = c5a

                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

def _bin_sa(hb, deadline, mx=15000, Ti=0.003):
    """Binary simulated annealing."""
    n = len(hb)
    dx = 2.0 / n
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    raw = _raw_corr(cur)
    cc = float(np.max(raw) * dx)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        raw_n = _raw_corr(hn)
        nc = float(np.max(raw_n) * dx)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            raw = raw_n
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9985
        else:
            stall += 1
    
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):

    hb[idx] = 1.0
    return hb

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lag_gradient(h, l):
    n = len(h)
    G = np.zeros(n)
    if l >= 0:
        L = n - l
        if L > 0:
            G[:L] += (1.0 - h[l:l + L])
            G[l:l + L] -= h[:L]
    else:
        s = -l
        L = n - s
        if L > 0:
            G[s:s + L] += (1.0 - h[:L])
            G[:L] -= h[s:s + L]
    return G


def _line_search_exact(h, s, dx, raw0):
    n = len(h)
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    a_lo, a_hi = 0.0, 1.0
    pos = s > 1e-14
    neg = s < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / s[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-s[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(raw0) * dx)

    def f_at(alpha):
        return float(np.max(raw0 + alpha * T - alpha * alpha * Q)) * dx

    n_grid = 30
    alphas = np.linspace(a_lo, a_hi, n_grid)
    vals = np.array([f_at(a) for a in alphas])
    idx = int(np.argmin(vals))

    if 0 < idx < n_grid - 1:
        lo_r = alphas[idx - 1]
        hi_r = alphas[idx + 1]
    else:
        lo_r, hi_r = a_lo, a_hi

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = f_at(c1)
    f2 = f_at(c2)
    for _ in range(35):
        if hi_r - lo_r < 1e-10:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr * (hi_r - lo_r)
            f1 = f_at(c1)
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = f_at(c2)
    best_a = (lo_r + hi_r) / 2.0
    best_v = f_at(best_a)
    if vals[idx] < best_v:
        best_v = float(vals[idx])
        best_a = float(alphas[idx])
    best_a = max(a_lo, min(a_hi, best_a))
    best_v = f_at(best_a)
    return best_a, best_v


def _lp_direction(h, raw, K, radius):
    n = len(h)
    n_corr = 2 * n - 1
    if K > n_corr:
        K = n_corr
    top_idx = np.argsort(-raw)[:K]
    grads = []
    for t in top_idx:
        l = int(t) - (n - 1)
        g = _lag_gradient(h, l)
        if np.linalg.norm(g) > 1e-12:
            grads.append(g)
    if not grads:
        return None
    A_g = np.column_stack(grads)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((len(grads), n + 1))
    A_ub[:, :n] = A_g
    A_ub[:, n] = 1.0
    b_ub = np.zeros(len(grads))
    lo_b = -radius * np.maximum(h, 1e-8)
    hi_b = radius * np.maximum(1.0 - h, 1e-8)
    bounds = list(zip(lo_b, hi_b))
    bounds.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x[-1] > 1e-8:
            s = res.x[:n].copy()
            s -= np.mean(s)
            if np.linalg.norm(s) > 1e-10:
                return s
    except Exception:
        pass
    return None


def _smooth_grad(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = np.clip(beta * (raw - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros(len(raw))
        w[np.argmax(raw)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for t in range(len(raw)):
        wt = w[t]
        if wt < 1e-15:
            continue
        l = t - (n - 1)
        g += wt * _lag_gradient(h, l)
    g -= np.mean(g)
    return -g


def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d


def _gen_directions(h, raw, rng, r0):
    n = len(h)
    dirs = []
    for K in [5, 15, 40]:
        s = _lp_direction(h, raw, K, r0)
        if s is not None:
            dirs.append(s)
    for beta in [25.0, 60.0]:
        g = _smooth_grad(h, raw, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm * r0)
    g20 = _smooth_grad(h, raw, 25.0)
    for bs in [10, 25]:
        d = _block_dir(g20, bs)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm * r0)
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-10:
        dirs.append(d / nrm * r0)
    return dirs


def _local_opt(h, deadline, r0=0.15, stall_lim=60, nd=6):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    raw = _raw_corr(h)
    cur = float(np.max(raw) * dx)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(12345)
    rad = r0
    stall = 0
    iter_count = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.4:
            break
        iter_count += 1
        raw = _raw_corr(h)
        dirs = _gen_directions(h, raw, rng, rad)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            pos = d > 1e-14
            neg = d < -1e-14
            scale = 1.0
            if np.any(pos):
                shr = float(np.min((1.0 - h[pos]) / d[pos]))
                if shr < scale:
                    scale = shr
            if np.any(neg):
                shr = float(np.min(h[neg] / (-d[neg])))
                if shr < scale:
                    scale = shr
            if scale < 1e-10:
                continue
            d = d * min(scale, 1.0)
            a, c_try = _line_search_exact(h, d, dx, raw)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.06, 0.35)
        else:
            stall += 1
            rad *= 0.70
            if rad < 0.005:
                pert = rng.normal(0, 0.05, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 25 == 0:
                    rad = 0.07
    return best_h, best_c


def _pair_opt(h, deadline, max_it=30000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    raw = _raw_corr(h)
    cur = float(np.max(raw) * dx)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search_exact(h, d, dx, raw)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                raw = _raw_corr(h)
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c


def _bin_sa(hb, deadline, mx=15000, Ti=0.003):
    n = len(hb)
    dx = 2.0 / n
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    raw = _raw_corr(cur)
    cc = float(np.max(raw) * dx)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        raw_n = _raw_corr(hn)
        nc = float(np.max(raw_n) * dx)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            raw = raw_n
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9985
        else:
            stall += 1
    return bh, bc


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)


def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb


def _construct_fallback(n=180):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    return _project(h, half)


def _fourier_start(n, rng, n_terms=8):
    h = np.zeros(n)
    x = np.arange(n, dtype=float)
    for k in range(1, n_terms + 1):
        amp = rng.uniform(0.1, 0.8) / k
        freq = rng.randint(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / n + phase)
    h += 0.5
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = _construct_fallback(180)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local opt on parent (22%)
    d1 = t0 + budget_s * 0.22
    h1, c1 = _local_opt(h_init, d1, r0=0.15, stall_lim=55, nd=6)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair optimization (8%)
    d2 = t0 + budget_s * 0.30
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=20000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA + follow-up (15%)
    d3 = t0 + budget_s * 0.45
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d3, mx=10000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.025)
        hp, cp = _local_opt(hp, sd, r0=0.10, stall_lim=35, nd=5)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.65
    nc = len(best_h)
    offsets = [-10, -5, 5, 10, -15, 15, -20, 20, -8, 8, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.014)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, nd=5)
        if cr < best_c + 0.004 and time.time() < d4 - 4:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=4000, Ti=0.002)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.002 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, nd=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Alternative starts (10%)
    d5 = t0 + budget_s * 0.75
    for alt_seed in [101, 202, 303]:
        if time.time() > d5:
            break
        rng_alt = np.random.default_rng(alt_seed)
        na = nc
        ha = _fourier_start(na, rng_alt, n_terms=10)
        sd = min(d5, time.time() + budget_s * 0.02)
        ha, ca = _local_opt(ha, sd, r0=0.15, stall_lim=30, nd=5)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 6: Intensive LP polish (12%)
    d6 = t0 + budget_s * 0.87
    nc = len(best_h)
    for r_init in [0.08, 0.04, 0.02]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.03)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=50, nd=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.015)
            h6, c6 = _pair_opt(h6, sd, max_it=8000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Basin hopping (final stretch)
    d7 = deadline
    scale = 0.025
    n_re = 0
    while time.time() < d7 and n_re < 40:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.2 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=18, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.06, scale * 1.12)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.012, stall_lim=35, nd=5)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=4000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
