def _milp_binary(n, time_limit):
    m = n // 2
    x = cp.Variable(n, boolean=True)
    
    # Create auxiliary variables for each pair
    # s_{j,d} for j=0..n-d-1, d=1..n-1
    # Flatten into a single variable for efficiency
    # Total number: n*(n-1)/2
    total_pairs = n * (n - 1) // 2
    
    # Actually, let's build it more carefully
    # For each d, we have (n-d) pairs
    # s[d] will be a vector of length (n-d)
    
    M = cp.Variable(nonneg=True)
    
    constraints = [sum(x) == m]
    
    for d in range(1, n):
        n_pairs = n - d
        s = cp.Variable(n_pairs, boolean=True)
        
        # s[j] = |x[j+d] - x[j]| for j=0..n-d-1
        for j in range(n_pairs):
            constraints.append(s[j] >= x[j+d] - x[j])
            constraints.append(s[j] >= x[j] - x[j+d])
        
        D_d = sum(s)  # This is sum of s[j]
        B_d = sum(x[j+d] for j in range(n_pairs)) - sum(x[j] for j in range(n_pairs))
        
        constraints.append(D_d + B_d <= 2 * M)
        constraints.append(D_d - B_d <= 2 * M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solver_opts = {'seconds': time_limit}
    if cp.CBC is not None:
        prob.solve(solver=cp.CBC, **solver_opts)
    elif cp.GLPK is not None:
        prob.solve(solver=cp.GLPK, **solver_opts)
    else:
        return None
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        x_val = np.round(x.value)
        if x_val is not None and len(x_val) == n and sum(x_val) == m:
            if np.all((x_val == 0) | (x_val == 1)):
                return x_val.astype(float)

            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.3)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amplitudes = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 120
        h0 = _to_binary(np.sort(np.random.default_rng(0). random(n0))[::-1])
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: MILP for small n (25% time)
    d1 = t0 + budget_s * 0.25
    milp_ns = [40, 60, 80]
    for mn in milp_ns:
        if time.time() > d1 - 5:
            break
        alloc = min(25, max(10, int((d1 - time.time()) / max(1, len(milp_ns)))))
        b_milp = _milp_solve(mn, time_limit=alloc)
        if b_milp is not None:
            c5_m = _exact_c5(b_milp)
            if c5_m < best_c5:
                best_c5 = c5_m
                best_h = b_milp.copy()
            # Try fractional relaxation of MILP solution
            if time.time() < d1 - 3:
                h_rel = _resample(b_milp, n0)
                sub_dl = min(d1, time.time() + 10)
                h_r, c5_r = _local_opt(h_rel, sub_dl, radius_init=0.08, max_iters=500)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()

    # Phase 2: Binary optimization at n0 (20% time)
    d2 = t0 + budget_s * 0.45
    b0 = _to_binary(best_h if len(best_h) == n0 else h_init)
    sub_dl = min(d2, time.time() + budget_s * 0.12)
    b_sa, c5_sa = _binary_sa(b0, sub_dl, T_init=0.004, max_iter=150000)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = b_sa.copy()
    if time.time() < d2 - 5:
        sub_dl2 = min(d2, time.time() + budget_s * 0.08)
        b_t, c5_t = _binary_targeted(b_sa, sub_dl2, max_iter=50000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = b_t.copy()

    # Phase 3: Fractional optimization (20% time)
    d3 = t0 + budget_s * 0.65
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    n_f = len(h_f)
    sub_dl = min(d3, time.time() + budget_s * 0.10)
    h_f, c5_f = _local_opt(h_f, sub_dl, radius_init=0.12, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d3 - 3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.06)
        h_p, c5_p = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), sub_dl2, max_rounds=80)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: Multi-resolution (15% time)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    tried = set([n_curr])
    for dn in range(-16, 17, 4):
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 250 or n_try in tried:
            continue
        if n_try % 2 != 0:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.10, max_iters=400)
        if c5_r < best_c5 + 0.003 and time.time() < d4 - 2:
            sub_dl2 = min(d4, time.time() + budget_s * 0.005)
            h_r2, c5_r2 = _pair_sweep(h_r, sub_dl2, max_rounds=30)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Basin hopping (10% time)
    d5 = t0 + budget_s * 0.90
    scale = 0.03
    for _ in range(200):
        if time.time() > d5:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.003)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.05, max_iters=200)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: Binary + fractional on best (5% time)
    d6 = deadline - 2
    if time.time() < d6:
        b_best = _to_binary(best_h)
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        b_pol, c5_pol = _binary_targeted(b_best, sub_dl, max_iter=20000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = b_pol.copy()
    if time.time() < d6:
        h_fin = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline, time.time() + budget_s * 0.02)
        h_fin, c5_fin = _local_opt(h_fin, sub_dl, radius_init=0.04, max_iters=500)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
