    exp_scaled = np.exp(scaled - scaled_max)
    Z = np.sum(exp_scaled)
    obj = (scaled_max + np.log(Z)) / tau  # = logsumexp(scaled) / tau = logsumexp(tau*dx*corr) / tau
    
    # Weights

    # Gradient: grad[p] = sum_k w[k] * drect[k]/dh[p]
    # drect[k]/dh[p] = (1-h[p-m]) * I[0<=p-m<n] - h[p+m] * I[0<=p+m<n] where m = k-(n-1)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

            x[mask] += res / cnt

    exp_s = np.exp(scaled - smax)
