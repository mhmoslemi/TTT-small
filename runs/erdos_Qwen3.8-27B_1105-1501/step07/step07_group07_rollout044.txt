    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Build gradient matrix G where G[r, j] = d(raw[r])/d(h[j])."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    m = n - 1
    G[m, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        mm = n - d
        # row = m + d (positive lag)
        r = m + d
        G[r, d:] += (1.0 - h[:mm])
        G[r, :mm] += -h[d:]
        # row = m - d (negative lag)
        r2 = m - d
        G[r2, :mm] += (1.0 - h[d:])
        G[r2, d:] += -h[:mm]
    return G

def _verify_grad(h, G):
    """Verify gradient matrix against finite differences."""
    n = len(h)
    max_err = 0.0
    test_cols = list(range(0, n, max(1, n // 8)))
    for j in test_cols:
        h_plus = h.copy()
        h_minus = h.copy()
        delta = 1e-6
        if h[j] + delta <= 1.0:
            h_plus[j] += delta
        else:
            h_plus[j] -= delta
            delta = -delta
        if h[j] + delta >= 0.0:
            h_minus[j] = h[j] - (2 * delta if delta > 0 else -2 * delta)
        fd = (_raw_scores(h_plus) - _raw_scores(h_minus)) / (2 * abs(delta))
        err = float(np.max(np.abs(fd - G[:, j])))
        max_err = max(max_err, err)
    return max_err
