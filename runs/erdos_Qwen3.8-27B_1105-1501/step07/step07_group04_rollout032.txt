```python
            oi = ones.index(best_i)
            oj = zeros.index(best_j)
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones[oi] = best_j
            zeros[oj] = best_i
            # Recompute correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(corr))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            # Try random swaps to escape
            if stall % 100 == 0:
                if ones and zeros:
                    i_idx = int(rng.integers(0, len(ones)))
                    j_idx = int(rng.integers(0, len(zeros)))
                    i = ones[i_idx]
                    j = zeros[j_idx]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones[i_idx] = j
                    zeros[j_idx] = i
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    cur_max = float(np.max(corr))
    
    return best_h, best_max * dx
```

For continuous optimization, the key idea from the strategy is the "pair-pivot" with exact quadratic line search. Let me implement that:

For a direction d = e_i - e_j (swap between positions i and j, continuous version), the correlation is:
corr(alpha) = corr(0) + alpha * T - alpha² * Q

where:
- T[r] = corr_swap_linear[r] (the linear term)
- Q[r] = corr_swap_quadratic[r] (the quadratic term)

For d = e_i - e_j:
- T[r] = (1-h[i]) * (contribution of i to corr[r]) - h[j] * (contribution of j to corr[r])... 

Actually, let me just use the general formula. For any direction d with sum(d) = 0:
corr_h + alpha*d, 1 - h - alpha*d = corr(h,1-h) + alpha*(corr(d,1-h) - corr(h,d)) - alpha²*corr(d,d)

So:
B = corr(h, 1-h)
T = corr(d, 1-h) - corr(h, d)
Q = corr(d, d)

And corr(alpha) = B + alpha*T - alpha²*Q.

The objective is f(alpha) = max_r (B + alpha*T - alpha²*Q)[r] * dx.

This is the max of quadratic functions, which is convex in alpha (each quadratic -alpha²*Q[r] + alpha*T[r] + B[r] is concave in alpha when Q[r] > 0... wait, -Q[r]*alpha² is concave, so the max of concave functions is not necessarily convex or concave).

Hmm, actually each term B[r] + alpha*T[r] - alpha²*Q[r] is a concave quadratic in alpha (since -Q[r] ≤ 0). The max of concave functions is not necessarily concave. So f(alpha) is not necessarily concave, and the line search might have multiple local minima.

But in practice, for small d, the function should be well-behaved near alpha=0. The golden section or exhaustive search over a small interval should work.

Let me implement the continuous pair search:

```python
def _continuous_pair_search(h_init, deadline, max_pairs=50000, seed=42):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    h = _project(h, target)
    cur_c5 = _exact_c5(h)
    B = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(B))
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    stall = 0
    for it in range(max_pairs):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        # Choose a pair (i, j)
        # Strategy: pick i and j based on gradient
        corr = B  # current correlation
        max_val = float(np.max(corr))
        g = np.zeros(n)
        # Subgradient at the max correlation
        active = np.where(corr >= max_val - 1e-8)[0]
        for m in active[:10]:  # limit to top 10
            k = int(m - (n - 1))
            if k >= 0:
                for i in range(n - k):
                    g[i] += (1 - h[i+k]) / len(active)
                    g[i+k] += (-h[i]) / len(active)
            else:
                ka = -k
                for i in range(n - ka):
                    g[i] += ... 
        
        # This is getting complicated. Let me just use random pairs with exact line search.
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        
        # Feasible alpha range
        lo = max(-h[i], h[j] - 1.0)  # h[i] + alpha >= 0, h[j] - alpha >= 0
        hi = min(1.0 - h[i], h[j])    # h[i] + alpha <= 1, h[j] - alpha <= 1
        # Wait: h_new[i] = h[i] + alpha, h_new[j] = h[j] - alpha
        # Constraints: 0 <= h[i]+alpha <= 1 → -h[i] <= alpha <= 1-h[i]
        #              0 <= h[j]-alpha <= 1 → h[j]-1 <= alpha <= h[j]
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        
        if hi - lo < 1e-12:
            stall += 1
            continue
        
        # Find best alpha in [lo, hi]
        # Use grid + golden section
        n_grid = 20
        alphas = np.linspace(lo, hi, n_grid)
        vals = np.array([float(np.max(B + a * T - a * a * Q)) for a in alphas])
        best_a = alphas[int(np.argmin(vals))]
        
        # Refine with golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo = max(lo, best_a - (hi - lo) / n_grid)
        a_hi = min(hi, best_a + (hi - lo) / n_grid)
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        f1 = float(np.max(B + a1 * T - a1 * a1 * Q))
        f2 = float(np.max(B + a2 * T - a2 * a2 * Q))
        for _ in range(30):
            if a_hi - a_lo < 1e-10:
                break
            if f1 < f2:
                a_hi = a2
                a2 = a1
                f2 = f1
                a1 = a_hi - gr * (a_hi - a_lo)
                f1 = float(np.max(B + a1 * T - a1 * a1 * Q))
            else:
                a_lo = a1
                a1 = a2
                f1 = f2
                a2 = a_lo + gr * (a_hi - a_lo)
                f2 = float(np.max(B + a2 * T - a2 * a2 * Q))
        opt_a = (a_lo + a_hi) / 2.0
        opt_val = float(np.max(B + opt_a * T - opt_a * a_lo * ... ))
        
        # Also check endpoints
        candidates = [opt_a, lo, hi, 0.0]
        best_cand_a = min(candidates, key=lambda a: float(np.max(B + a*T - a*a*Q)))
        
        new_max = float(np.max(B + best_cand_a * T - best_cand_a * best_cand_a * Q))
        
        if new_max < cur_max - 1e-12:
            h[i] += best_cand_a
            h[j] -= best_cand_a
            h = _project(h, target)  # Ensure feasibility (should be a no-op if exact)
            cur_max = float(np.max(B + best_cand_a * T - best_cand_a * best_cand_a * Q))
            # Recompute B for next iteration
            B = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(B))
            if cur_max < best_c5 / dx:
                best_c5 = cur_max * dx
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5
```
