def _targeted_flip(corr, h, n):
    """Find the best single pair flip by targeting the max lag."""
    max_lag = int(np.argmax(corr))  # index in [0, 2n-2]
    l = max_lag - (n-1)  # actual lag value
    
    # Find all pairs at this lag: (i, i+l) where h[i+l]=1, h[i]=0
    # For l > 0: pairs are (i, i+l) for i = 0, ..., n-1-l
    # For l < 0: pairs are (i+l, i) for i = -l, ..., n-1, i.e., (j, j-|l|) for j = |l|, ..., n-1
    
    if l > 0:
        candidates = []
        for i in range(n - l):
            if h[i+l] > 0.5 and h[i] < 0.5:
                candidates.append(i)  # flip pair: h[i+l]→0, h[i]→1
    elif l < 0:
        abs_l = -l
        candidates = []
        for i in range(n - abs_l):
            if h[i] > 0.5 and h[i+abs_l] < 0.5:
                candidates.append(i)  # flip pair: h[i]→0, h[i+abs_l]→1
    else:
        return None, 0.0
    
    if not candidates:
        return None, 0.0
    
    # For each candidate, compute the new C5
    best = None
    best_c5 = _c5_from_corr(corr, n)
    for i in candidates:
        if l > 0:
            p = i + l  # position going 1→0
            q = i      # position going 0→1
        else:
            p = i      # position going 1→0
            q = i + abs_l  # position going 0→1
        
        # Compute new corr
        new_corr = corr.copy()
        # Apply delta for p: 1→0
        new_corr += _flp_delta_1to0(h, p, n)
        # Apply delta for q: 0→1 = negative of 1→0 delta
        new_corr -= _flp_delta_1to0(h, q, n)
        
        c5_new = _c5_from_corr(new_corr, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = (p, q)
    
    return best, best_c5

import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

            break
        # Targeted approach: focus on top lags
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]
        best_gain = -1e-14
        best_p, best_q = -1, -1
        best_new_c5 = cur_c5
        
        for ti in range(top_k):
            li = int(top_idx[ti])
            l_val = li - (n - 1)
            # Find pairs at this lag
            if l_val > 0:
                cands = []
                for i in range(n - l_val):
                    if hb[i + l_val] > 0.5 and hb[i] < 0.5:
                        cands.append(i)
            elif l_val < 0:
                al = -l_val
                cands = []
                for i in range(n - al):
                    if hb[i] > 0.5 and hb[i + al] < 0.5:
                        cands.append(i)
            else:
                continue
            
            # Sample up to 20 candidates from this lag
            if len(cands) > 20:
                cands = list(rng.choice(cands, size=20, replace=False))
            
            for i in cands:
                if l_val > 0:
                    p = i + l_val
                    q = i
                else:
                    p = i
                    q = i + (-l_val)
                
                # Compute new corr
                new_corr = corr.copy()
                new_corr += _flip_delta(hb, p, n)
                new_corr -= _flip_delta(hb, q, n)
                new_c5 = float(np.max(new_corr)) * (2.0 / n)
                
                if new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_p, best_q = p, q
                    best_gain = cur_c5 - new_c5
        
        if best_p >= 0 and best_new_c5 < cur_c5 - 1e-14:
            hb[best_p] = 0.0
            hb[best_q] = 1.0
            ones.remove(best_p)
            ones.append(best_q)
            zeros.remove(best_q)
            zeros.append(best_p)
            corr = np.correlate(hb, 1.0 - hb, mode='full')
            cur_c5 = float(np.max(corr)) * (2.0 / n)
            stall = 0
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_hb = hb.copy()
        else:
            # Try random pair
            if len(ones) > 0 and len(zeros) > 0:
                p = ones[int(rng.integers(0, len(ones)))]
                q = zeros[int(rng.integers(0, len(zeros)))]
                new_corr = corr.copy()
                new_corr += _flip_delta(hb, p, n)
                new_corr -= _flip_delta(hb, q, n)
                new_c5 = float(np.max(new_corr)) * (2.0 / n)
                if new_c5 < cur_c5 - 1e-14:
                    hb[p] = 0.0
                    hb[q] = 1.0
                    ones.remove(p)
                    ones.append(q)
                    zeros.remove(q)
                    zeros.append(p)
                    corr = np.correlate(hb, 1.0 - hb, mode='full')
                    cur_c5 = float(np.max(corr)) * (2.0 / n)
                    stall = 0
                    if cur_c5 < best_c5 - 1e-14:
                        best_c5 = cur_c5
                        best_hb = hb.copy()
                else:
                    stall += 1
            else:
                stall += 50
    
    return best_hb, best_c5

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    best_val = float('inf')
    best_a = 0.0
    width = a_hi - a_lo
    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo_r, hi_r = a_lo, a_hi
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(30):
        if hi_r - lo_r < 1e-10:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr * (hi_r - lo_r)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    mid = (lo_r + hi_r) / 2.0
    fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
    if fmid < best_val:
        best_val = fmid
        best_a = mid
    # Also check 0
    f0 = float(np.max(S0)) * dx
    if f0 < best_val:
        best_val = f0
        best_a = 0.0
    return best_a, best_val

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _smooth_opt(h, deadline, r0=0.20, stall_lim=60, nd=4):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(777)
    rad = r0
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        # Generate directions
        dirs = []
        for beta in [30.0, 60.0]:
            g = _smooth_grad(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)
        # Random direction
        rd = rng.normal(0, 1, n)
        rd -= np.mean(rd)
        nrm = np.linalg.norm(rd)
        if nrm > 1e-10:
            dirs.append(rd / nrm)
        
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.40)
        else:
            stall += 1
            rad *= 0.7
            if rad < 0.01:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 30 == 0:
                    rad = 0.06
    
    return best_h, best_c

def _pairstep(h, deadline, max_it=80000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(555)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 0.5)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c = _exact_c5(h0)
    
    # Phase 1: Binary systematic optimization (30%)
    d1 = t0 + budget_s * 0.30
    hb = _to_binary(h0, n0)
    hb, cb = _bin_opt_systematic(hb, d1, n0, max_iters=800000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Smooth optimization from binary (20%)
    d2 = t0 + budget_s * 0.50
    hs = _project(hb.copy(), n0 / 2.0)
    hs, cs = _smooth_opt(hs, d2, r0=0.25, stall_lim=80, nd=4)
    if cs < best_c:
        best_c = cs
        best_h = hs.copy()
    
    # Phase 3: Pair optimization (10%)
    d3 = t0 + budget_s * 0.60
    hp = _project(best_h.copy(), len(best_h) / 2.0)
    hp, cp = _pairstep(hp, d3, max_it=50000)
    if cp < best_c:
        best_c = cp
        best_h = hp.copy()
    
    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.80
    nc = len(best_h)
    offsets = [-6, 6, -12, 12, -18, 18, -4, 4, -24, 24, -30, 30]
    for dn in offsets:
        if time.time() > d4 - 2:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0:
            continue
        hr = _resample(best_h, n_try)
        # Quick binary opt
        sd_bin = min(d4, time.time() + budget_s * 0.012)
        hbr = _to_binary(hr, n_try)
        hbr, cbr = _bin_opt_systematic(hbr, sd_bin, n_try, max_iters=200000)
        if cbr < best_c:
            best_c = cbr
            best_h = hbr.copy()
            nc = n_try
        # Quick smooth opt
        if time.time() < d4 - 3:
            sd_sm = min(d4, time.time() + budget_s * 0.012)
            hsr = _project(hbr.copy(), n_try / 2.0)
            hsr, csr = _smooth_opt(hsr, sd_sm, r0=0.15, stall_lim=40, nd=3)
            if csr < best_c:
                best_c = csr
                best_h = hsr.copy()
                nc = n_try
    
    # Phase 5: Intensive local (8%)
    d5 = t0 + budget_s * 0.88
    nc = len(best_h)
    for _ in range(2):
        if time.time() > d5 - 2:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.025)
        h5, c5 = _smooth_opt(h5, sd, r0=0.10, stall_lim=50, nd=4)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # Phase 6: Basin hopping + polish (6%)
    d6 = deadline
    scale = 0.02
    for _ in range(100):
        if time.time() > d6 - 2:
            break
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hpert = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _smooth_opt(hpert, sd, r0=0.08, stall_lim=25, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.7)
        else:
            scale = min(0.06, scale * 1.15)
    
    # Final: pair polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pairstep(hf, deadline, max_it=20000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    # Final binary check
    if deadline - time.time() > 2:
        nc = len(best_h)
        hbf = _to_binary(best_h, nc)
        sd = min(deadline, time.time() + budget_s * 0.008)
        hbf, cbf = _bin_opt_systematic(hbf, sd, nc, max_iters=100000)
        if cbf < best_c:
            best_c = cbf
            best_h = hbf.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
