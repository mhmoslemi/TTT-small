Let me refine the 1D subproblem solution:

Given c_k (current values) and w_k (coefficients), find δ ∈ [δ_lo, δ_hi] that minimizes f(δ) = max_k (c_k + δ*w_k).

The function f(δ) is convex piecewise linear. Its minimum is at:
- A point where the derivative changes sign (a "knee"), or
- A boundary (δ_lo or δ_hi)

To find the knees, we need to find all δ values where two lines c_k + δ*w_k cross. But there are O(n²) such crossings. However, we only need the ones that are on the upper envelope.

- The subgradient at a point δ is the set of slopes of the active lines at that point.
- At a "knee" (where the argmax changes), the subgradient is the interval [min active slope, max active slope].
- The minimum is at a point where 0 is in the subgradient, i.e., at a knee where some active lines have negative slope and some have positive slope, or at a flat point.

- At δ_lo, compute the set of active lags (those achieving f(δ_lo)) and their slopes w_k.
  - If all slopes are ≥ 0, the minimum is at δ_lo.
  - If all slopes are < 0, we need to go right.
  - Otherwise (some positive, some negative), the minimum is somewhere in between.

def _coord_descent(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    
    # Initialize correlation values
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(c)) * dx
    best_h = h.copy()
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Randomize order
        order = np.random.permutation(n)  # or fixed order
        for i in order:
            # Compute weights w[k] for coordinate i
            w = np.zeros(2*n - 1)
            # For d >= 0: index = n-1+d
            for d in range(0, n):
                idx = n - 1 + d
                wval = 0.0
                if d <= i:
                    wval += (1.0 - h[i-d])
                if i + d < n:
                    wval += (-h[i+d])
                w[idx] = wval
            # For d < 0: d = -g, index = n-1-d
            for g in range(1, n):
                idx = n - 1 - g
                wval = 0.0
                if i + g < n:
                    wval += (1.0 - h[i+g])
                if g <= i:
                    wval += (-h[i-g])
                w[idx] = wval
            
            # Solve 1D: minimize max(c + delta*w) subject to -h[i] <= delta <= 1-h[i]
            dlo = -h[i]
            dhi = 1.0 - h[i]
            if dhi - dlo < 1e-14:
                continue
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a = dlo
            b = dhi
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = np.max(c + c_pt * w)
            fd = np.max(c + d_pt * w)
            for _ in range(50):
                if b - a < 1e-13:
                    break
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = np.max(c + c_pt * w)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = np.max(c + d_pt * w)
            opt = (a + b) / 2.0
            opt_val = np.max(c + opt * w)
            
            # Also check boundaries
            val_lo = np.max(c + dlo * w)  # should be current max if dlo=0
            val_hi = np.max(c + dhi * w)
            if val_lo < opt_val:
                opt = dlo
                opt_val = val_lo
            if val_hi < opt_val:
                opt = dhi
                opt_val = val_hi
            
            if opt_val < np.max(c) - 1e-13:
                c += opt * w
                h[i] += opt
                h[i] = np.clip(h[i], 0.0, 1.0)
                improved = True
        
        if not improved:
            break
    
    # Final projection to ensure exact constraint
    h = _project(h, target)
    final_c5 = _exact_c5(h)
    if final_c5 < best_c5:
        return h, final_c5
    return best_h, best_c5

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):

        if s > target:

        else:
            hi = mid

    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

        cnt = int(np.sum(mask))

def _coord_weights(h, i, n):
    """Compute the gradient of the correlation vector with respect to h[i]."""
    w = np.zeros(2 * n - 1)
    # For d >= 0: signed lag d, index = n-1+d
    # w(d) = [d <= i] * (1-h[i-d]) + [i+d <= n-1] * (-h[i+d])
    for d in range(0, n):
        idx = n - 1 + d
        wv = 0.0
        if d <= i:
            wv += (1.0 - h[i - d])
        if i + d < n:
            wv += (-h[i + d])
        w[idx] = wv
    # For d < 0: signed lag -g, index = n-1-g
    # w(-g) = [i+g <= n-1] * (1-h[i+g]) + [g <= i] * (-h[i-g])
    for g in range(1, n):
        idx = n - 1 - g
        wv = 0.0
        if i + g < n:
            wv += (1.0 - h[i + g])
        if g <= i:
            wv += (-h[i - g])
        w[idx] = wv
    return w
