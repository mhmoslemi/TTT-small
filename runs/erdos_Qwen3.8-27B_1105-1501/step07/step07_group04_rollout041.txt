def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_T(i, j, h):
    n = len(h)

    k_hi = min(i, n - 1)

        ks = np.arange(k_lo, k_hi + 1)
        T[ks + n - 1] += (1.0 - h[i - ks])
    k_lo = max(j - (n - 1), -(n - 1))
    k_hi = min(j, n - 1)

def _line_search_fast(B, T, Q, alpha_lo, alpha_hi):
    n_grid = 12
    if alpha_hi - alpha_lo < 1e-12:
        return 0.0, float(np.max(B))
    ag = np.linspace(alpha_lo, alpha_hi, n_grid)
    fm = B[None, :] + ag[:, None] * T[None, :] - (ag[:, None] ** 2) * Q[None, :]
    mf = fm.max(axis=1)
    bi = int(np.argmin(mf))
    lo_b = ag[max(0, bi - 1)]
    hi_b = ag[min(n_grid - 1, bi + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(20):
        if hi_b - lo_b < 1e-10:
            break
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(B + c1 * T - c1 * c1 * Q))
        f2 = float(np.max(B + c2 * T - c2 * c2 * Q))
        if f1 < f2:
            hi_b = c2
        else:
            lo_b = c1
    opt_a = (lo_b + hi_b) / 2.0
    opt_v = float(np.max(B + opt_a * T - opt_a * opt_a * Q))
    cur_v = float(np.max(B))
    if opt_v > cur_v:
        return 0.0, cur_v
    return opt_a, opt_v
