def _project_sum(v, target):
    """Project v to {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):

        else: hi = mid

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 0.0

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[n - 1 + d, d:] += (1.0 - h[:m])
        G[n - 1 + d, :m] -= h[d:]
        G[n - 1 - d, :m] += (1.0 - h[d:])
        G[n - 1 - d, d:] -= h[:m]
    return G

def _lse_fun(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = float(np.max(corr))
    u = (corr - mval) / tau
    u_max = float(np.max(u))
    z = np.exp(u - u_max)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        max_mask = np.abs(corr - mval) < tau * 0.01
        probs = np.zeros(2 * n - 1)
        probs[max_mask] = 1.0 / max(int(np.sum(max_mask)), 1)
    else:
        probs = z / zsum
    G = _build_G(h)
    grad = G.T @ (probs * dx)
    obj = mval + tau * (np.log(zsum) + u_max)
    return obj, grad

def _lse_eq(x):
    return np.sum(x) - _g_target

def _lse_eq_jac(x):
    return np.ones((1, _g_n))

def _lsq_single(h0, tau, maxiter=150, ftol=1e-10):
    global _g_n, _g_target, _g_dx, _g_tau
    _g_n = len(h0)
    _g_target = _g_n / 2.0
    _g_dx = 2.0 / _g_n
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    try:
        res = minimize(_lse_fun, h0, jac=True, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _lse_eq, 'jac': _lse_eq_jac}],
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.success and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return None

def _exact_line_search(h1, h2, n_eval=30):

    best_t = 0.0

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE annealing from multiple seeds
    d1 = t0 + budget_s * 0.35
    seeds = _make_seeds(h_init, rng, n_seeds=6)
    for si, h_seed in enumerate(seeds):
        if time.time() > d1 - 5:
            break
        rem = d1 - time.time()
        n_stages = min(8, max(2, int(rem / 2.0)))
        tau_start = 0.10
        tau_end = 0.001
        ratio = (tau_end / tau_start) ** (1.0 / max(n_stages - 1, 1))
        taus = [tau_start * (ratio ** i) for i in range(n_stages)]
        sub_dl = min(d1, time.time() + max(1.0, rem / (len(seeds) - si)))
        h_res = _lse_anneal(h_seed, sub_dl, taus=taus, maxiter_per_tau=100)
        c5_res = _exact_c5(h_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 2: Refine best with lower-temperature LSE
    d2 = t0 + budget_s * 0.50
    h_cur = best_h.copy()
    tau_lo = [0.005, 0.003, 0.0015, 0.0008, 0.0004]
    for tau in tau_lo:
        if time.time() > d2 - 2:
            break
        h_new = _lsq_single(h_cur, tau, maxiter=150, ftol=1e-12)
        if h_new is not None:
            h_proj = _project_sum(h_new, len(h_cur) / 2.0)
            h_blend, c5_blend = _exact_line_search(h_cur, h_proj, n_eval=25)
            if c5_blend < _exact_c5(h_cur):
                h_cur = _project_sum(h_blend, len(h_cur) / 2.0)
        else:
            h_proj = _project_sum(h_new if h_new is not None else h_cur, len(h_cur) / 2.0)
        c5_cur = _exact_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 3: Coordinate polish and binary SA
    d3 = t0 + budget_s *  0.70
    h_pol = best_h.copy()
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_pol, c5_pol = _coord_polish(h_pol, sub_dl, n_grid=6)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    if time.time() < d3:
        h_bin = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.06)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=seed + 1)
        sub_dl2 = min(d3, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=12)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 4: Dimension exploration
    d4 = t0 + budget_s * 0.88
    n_best = len(best_h)
    n_candidates = sorted(set(n_best + dn for dn in [-16, -12, -8, -4, 4, 8, 12, 16, -24, 24]
                              if 60 <= n_best + dn <= 260 and (n_best + dn) % 2 == 0))
    for n_try in n_candidates:
        if time.time() > d4 - 8:
            break
        h_res = _resample(best_h, n_try)
        c5_res = _exact_c5(h_res)
        if c5_res > best_c5 + 0.003:
            continue
        taus_short = [0.05, 0.02, 0.008, 0.003]
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r = _lse_anneal(h_res, sub_dl, taus=taus_short, maxiter_per_tau=60)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.001 and time.time() < d4 - 5:
            taus_fin = [0.003, 0.001, 0.0005]
            sub_dl2 = min(d4, time.time() + budget_s * 0.01)
            h_r2 = _lse_anneal(h_r, sub_dl2, taus=taus_fin, maxiter_per_tau=80)
            c5_r2 = _exact_c5(h_r2)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            h_b = _to_binary(h_r)
            sub_dl3 = min(d4, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_sa(h_b, sub_dl3, max_iter=15000, T_init=0.002, seed=n_try)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Basin hopping with LSE
    d5 = deadline - budget_s * 0.03
    scale = 0.02
    for iter_idx in range(200):
        if time.time() > d5:
            break
        h_pert = _perturb(best_h, rng, scale)
        taus_mini = [0.03, 0.01, 0.003]
        sub_dl = min(d5, time.time() + max(1.0, budget_s * 0.005))
        h_t = _lse_anneal(h_pert, sub_dl, taus=taus_mini, maxiter_per_tau=50)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5 - 1e-6:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.08)

    # Phase 6: Final polish
    h_fin = _project_sum(best_h.copy(), len(best_h) / 2.0)
    c5_fin = _exact_c5(h_fin)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    else:
        best_h = h_fin.copy()

    # Final low-temp LSE
    if deadline - time.time() > 3:
        taus_final = [0.002, 0.0008, 0.0003]
        h_f2 = _lse_anneal(best_h, deadline - 1, taus=taus_final, maxiter_per_tau=100)
        c5_f2 = _exact_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final projection and verification
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
