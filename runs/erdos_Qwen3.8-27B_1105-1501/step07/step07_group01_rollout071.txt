        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, fe)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-13:
            h = h + best_t * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.1 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient with annealing (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _anneal_smooth(best_h, d1, max_iter_total=100000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (20%)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _coord_descent(best_h, d2, max_sweeps=3000, seed=111)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Second round with different seed
    if time.time() < d2:
        h2b, c5_2b = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.08), max_sweeps=2000, seed=222)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: LP active set (10%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _lp_active(best_h, d3, radius_init=0.08, stall_limit=60, top_k=15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Smooth gradient again from LP result (10%)
    d4 = t0 + budget_s * 0.75
    if time.time() < d4:
        h4, c5_4 = _anneal_smooth(best_h, d4, max_iter_total=40000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Pair swaps (10%)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _pair_swap(best_h, d5, n_pairs=6000, radius=0.30, seed=333)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-n exploration (7%)
    d6 = t0 + budget_s * 0.92
    for dn in [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d6:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.007)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=400, seed=500 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Basin hopping (8%)
    d7 = deadline
    scale = 0.03
    n_restarts = 0
    while time.time() < d7 - 1 and n_restarts < 80:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d7 - 1, time.time() + budget_s * 0.003)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=150, seed=1000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Final
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
