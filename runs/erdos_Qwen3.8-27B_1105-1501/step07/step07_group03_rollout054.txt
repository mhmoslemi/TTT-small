import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_lag_gradient(h, k):
    """Compute gradient of O(k; x) at x=h."""
    n = len(h)
    g = np.zeros(n)
    # ∂O(k)/∂x[i] = (1-x[i-k]) * [valid(i-k)] - x[i+k] * [valid(i+k)]
    # valid(i-k) means 0 <= i-k < n AND 0 <= i < n
    # => max(0, k) <= i <= min(n-1, n-1+k)
    lo1 = max(0, k)
    hi1 = min(n - 1, n - 1 + k)
    if hi1 >= lo1:
        idx = np.arange(lo1, hi1 + 1)
        g[idx] += (1.0 - h[idx - k])
    # valid(i+k) means 0 <= i+k < n AND 0 <= i < n
    # => max(0, -k) <= i <= min(n-1, n-1-k)
    lo2 = max(0, -k)
    hi2 = min(n - 1, n - 1 - k)
    if hi2 >= lo2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 + k]
    return g

def _line_search(h, d, raw, dx, radius, n_top=16):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    candidates = [0.0, a_hi]
    
    # Add quadratic minimizers for top lags
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    
    # Grid
    n_grid = 30
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha_c = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha_c * T + alpha_c * alpha_c * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha_c
    
    # Golden section refinement
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T + c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T + c2 * c2 * Q)) * dx
        for _ in range(40):
            if hi - lo < 1e-13:
                break
            if f1 < f2:

def _lp_optimize(h, deadline, radius=0.2, n_active=20, max_iter=200):

            if raw[r] > mx - 0.005 * mx:
                active_idx.append(r)
        if len(active_idx) > n_act:
            active_idx = active_idx[:n_act]
        if len(active_idx) < 4:
            active_idx = top_lags[:n_act].tolist()
        x_lp = _lp_solve(h, active_idx, radius, target)
        if x_lp is None:
            stall += 2
            radius *= 0.8
            continue
        d = x_lp - h
        nm = np.linalg.norm(d)
        if nm < 1e-10:
            stall += 1
            radius *= 0.8
            continue
        alpha, new_c5 = _line_search(h, d, raw, dx, radius, n_top=16)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.85
            continue
        h_new = _project(h + alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
            n_act = min(n_act + 1, 45)
        else:
            stall += 1
            radius *= 0.85
            n_act = max(5, n_act - 2)
    return best_h, best_c5

def _grad_optimize(h, deadline, beta=20.0, radius=0.2, stall_limit=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta_cur = beta
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > stall_limit:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta_cur * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        g = np.zeros(n)
        u = 1.0 - h
        for r in range(2 * n - 1):
            wi = w[r]
            if wi < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m2 = n - d
                if m2 > 0:
                    g[d:] += wi * u[:m2]
                    g[:m2] -= wi * h[d:]
            else:
                gp = -d
                m2 = n - gp
                if m2 > 0:
                    g[:m2] += wi * u[gp:]
                    g[gp:] -= wi * h[:m2]
        g -= np.mean(g)
        directions = [None]
        nm = np.linalg.norm(-g)
        if nm > 1e-12:
            directions[0] = -g / nm
        else:
            directions = []
        for _ in range(2):
            d_rand = rng.normal(0, 1, n)
            d_rand -= np.mean(d_rand)
            nm = np.linalg.norm(d_rand)
            if nm > 1e-12:
                directions.append(d_rand / nm)
        accepted = False
        for d in directions:
            if d is None:
                continue
            alpha, new_c5 = _line_search(h, d, raw, dx, radius, n_top=16)
            if alpha < 1e-10:
                continue
            h_new = _project(h + alpha * d, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                accepted = True
                break
        if not accepted:
            stall += 1
            radius *= 0.8
            beta_cur = max(beta_cur * 0.93, 5.0)
            if stall % 25 == 0 and radius < 0.05:
                radius = 0.15
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_swap(h, deadline, n_pairs=3000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search(h, d, raw, dx, radius, n_top=8)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = _project(h + alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9993
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP-based optimization (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_optimize(h_init, d1, radius=0.25, n_active=20, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient polish + binary (15%)
    d2 = t0 + budget_s * 0.40
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _grad_optimize(h2, min(d2, time.time() + budget_s * 0.06), beta=20.0, radius=0.15, stall_limit=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    if time.time() < d2 - 1:
        h_bin = _to_binary(best_h)
        h_bin, c5_b = _binary_sa(h_bin, min(d2, time.time() + budget_s * 0.04), max_iter=20000, T_init=0.002)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()

    # Phase 3: Multi-n exploration (15%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = list(range(-12, 13, 2))
    for dn in offsets:
        if time.time() > d3 - 3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 220 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius=0.12, n_active=15, stall_limit=50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization with LP + pair swap (25%)
    d4 = t0 + budget_s * 0.80
    for attempt in range(4):
        if time.time() > d4 - 5:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.035)
        rad = max(0.08, 0.18 - attempt * 0.02)
        n_act = 20 + attempt * 5
        h4, c5_4 = _lp_optimize(_project(best_h.copy(), n_curr / 2.0), sub_dl,
                                radius=rad, n_active=n_act, stall_limit=100)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < d4 - 3:
            h4, c5_4 = _pair_swap(best_h, min(d4, time.time() + budget_s * 0.012),
                                  n_pairs=2000, radius=0.12, seed=400 + attempt)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
        if time.time() < d4 - 2:
            h4, c5_4 = _grad_optimize(_project(best_h.copy(), n_curr / 2.0),
                                      min(d4, time.time() + budget_s * 0.01),
                                      beta=25.0, radius=0.08, stall_limit=40)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = deadline
    scale = 0.035
    n_restarts = 0
    while time.time() < d5 - 3 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 2, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius=0.10, n_active=15, stall_limit=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.07)

    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, min(deadline, time.time() + budget_s * 0.02),
                                radius=0.04, n_active=35, stall_limit=50)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap(h6, deadline, n_pairs=1000, radius=0.06, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
