    n = len(h)

    g = np.zeros(n)

        g[:m] += (1.0 - h[l:l+m])
        g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        g[s:] += (1.0 - h[:m])

def _topk_grad(h, k=3):

    n = len(h)

def _line_search(h, d, dx, radius):

    S0 = np.correlate(h, 1.0 - h, mode='full')

    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(48, max(24, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]

        worst_lag = int(np.argmax(corr))
        l = worst_lag - (n - 1)
        # Gradient at worst lag tells us which pairs to flip

        neg_g = -g

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

    starts = []
    # 1. Block pattern
    h = np.zeros(n)
    block = n // 4
    h[:block] = 1.0
    h[n - block:] = 1.0
    h = _project(h, n / 2.0)
    starts.append(h)

    h = np.zeros(n)
    period = max(4, n // 10)
    for i in range(n):
        if (i // period) % 2 == 0:
            h[i] = 1.0
    h = _project(h, n / 2.0)
    starts.append(h)
    
    # 3. Sine-based
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    h = _project(h, n / 2.0)
    starts.append(h)
    
    # 4. Random binary
    h = (rng.random(n) < 0.5).astype(float)
    # Fix sum
    ones = np.sum(h)
    if ones > n / 2:
        idx = np.where(h == 1.0)[0]
        extra = int(ones - n / 2)
        to_flip = rng.choice(idx, extra, replace=False)
        h[to_flip] = 0.0
    elif ones < n / 2:
        idx = np.where(h == 0.0)[0]
        missing = int(n / 2 - ones)
        to_flip = rng.choice(idx, missing, replace=False)
        h[to_flip] = 1.0
    starts.append(h)
    
    # 5. Two-frequency

    h = _project(h, n / 2.0)
    starts.append(h)
    
    return starts

import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_at_lag(h, lag_idx):
    n = len(h)
    l = lag_idx - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        g[:m] += (1.0 - h[l:l + m])
        g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:s + m]
    return g

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(S))
    v = np.clip(beta * (S - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _topk_grad(h, k=4):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(-S)[:k]
    g = np.zeros(n)
    for ki in top_k:
        g += _grad_at_lag(h, int(ki))
    g /= k
    g -= np.mean(g)
    return -g

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14

    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(-corr)[:5]
        best_pair_c = cur
        best_d = None
        best_a = 0.0
        for lag_idx in top_lags[:3]:
            g = _grad_at_lag(h, int(lag_idx))
            neg_g = -g
            pos_idx = np.argsort(-neg_g)[:8]
            neg_idx = np.argsort(neg_g)[:8]
            for ii in range(min(5, len(pos_idx))):
                for jj in range(min(5, len(neg_idx))):
                    i = int(pos_idx[ii])
                    j = int(neg_idx[jj])
                    if i == j:
                        continue
                    d = np.zeros(n)
                    d[i] = 1.0
                    d[j] = -1.0
                    a, c5t = _line_search(h, d, dx, 1.0)
                    if c5t < best_pair_c - 1e-14:
                        best_pair_c = c5t
                        best_a = a
                        best_d = d
        if best_d is not None and best_pair_c < cur - 1e-14:
            h_new = h + best_a * best_d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall % 50 == 0:
                # Random restart within neighborhood
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur:
                    h = hp
                    cur = cp
                    stall = 0
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _gen_starts(n, rng, ref_h=None):
    starts = []
    # Sine-based starts with different frequencies
    x = np.linspace(0, 2, n, endpoint=False)
    for freqs in [(1, 0), (2, 0), (3, 1), (2, 5), (1, 4), (3, 7)]:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * freqs[0] * x)
        if freqs[1] > 0:
            h += 0.15 * np.sin(2 * np.pi * freqs[1] * x + 0.7)
        h = _project(h, n / 2.0)
        starts.append(h)
    # Block-based
    for block_frac in [0.2, 0.25, 0.3, 0.4]:
        h = np.zeros(n)
        block = int(n * block_frac)
        h[:block] = 1.0
        h[n - block:] = 1.0
        h = _project(h, n / 2.0)
        starts.append(h)
    # Random binary
    for _ in range(3):
        h = (rng.random(n) < 0.5).astype(float)
        ones = int(np.sum(h))
        if ones > n // 2:
            idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(idx, ones - n // 2, replace=False)
            h[to_flip] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(idx, n // 2 - ones, replace=False)
            h[to_flip] = 1.0
        starts.append(h)
    # Blend with reference
    if ref_h is not None and len(ref_h) == n:
        for alpha in [0.3, 0.5, 0.7]:
            sb = alpha * ref_h + (1 - alpha) * rng.uniform(0, 1, n)
            sb = _project(sb, n / 2.0)
            starts.append(sb)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Optimize initial (18%)
    d1 = t0 + budget_s * 0.18
    h1, c1 = _local_opt(h_init, d1, r0=0.18, stall_lim=50)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Targeted worst-lag opt (12%)
    d2 = t0 + budget_s * 0.30
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _targeted_opt(h2, d2, max_it=40000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair opt (8%)
    d3 = t0 + budget_s * 0.38
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=30000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Try different n (22%)
    d4 = t0 + budget_s * 0.60
    nc = len(best_h)
    offsets = sorted(range(-24, 25, 4))
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 50 or n_try > 300:
            continue
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.010)
        hr, cr = _local_opt(hr, sd, r0=0.15, stall_lim=35)
        if cr < best_c + 0.003 and time.time() < d4 - 3:
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hr2, cr2 = _targeted_opt(_project(hr.copy(), n_try / 2.0), sd2, max_it=8000)
            if cr2 < cr:
                cr = cr2
                hr = hr2
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Multi-start (18%)
    d5 = t0 + budget_s * 0.78
    nc = len(best_h)
    starts = _gen_starts(nc, rng, best_h)
    for s in starts:
        if time.time() > d5 - 2:
            break
        sd = min(d5, time.time() + budget_s * 0.012)
        hs, cs = _local_opt(s, sd, r0=0.20, stall_lim=35)
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()
        if cs < best_c + 0.002 and time.time() < d5 - 3:
            sd2 = min(d5, time.time() + budget_s * 0.005)
            hs2, cs2 = _targeted_opt(_project(hs.copy(), nc / 2.0), sd2, max_it=8000)
            if cs2 < best_c:
                best_c = cs2
                best_h = hs2.copy()

    # Phase 6: Intensive final opt (12%)
    d6 = deadline
    for r_init in [0.12, 0.06, 0.03]:
        if time.time() > d6 - 4:
            break
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        sd = min(d6, time.time() + budget_s * 0.03)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=45)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _project(best_h.copy(), len(best_h) / 2.0)
            sd = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _pair_opt(h6, sd, max_it=12000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _targeted_opt(hf, deadline, max_it=15000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=25)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
