def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    S_max = float(np.max(S))
    w = np.exp(beta * (S - S_max))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= w_sum
    omh = 1.0 - h
    k_arr = np.arange(2 * n - 1, dtype=float)
    grad = np.zeros(n)
    for j in range(n):
        p = j + k_arr - (n - 1)
        mask = (p >= 0) & (p < n)
        p_int = p.astype(int)
        part1 = np.zeros(2 * n - 1)
        part1[mask] = omh[p_int[mask]]
        p2 = k_arr + (n - 1) - j
        mask2 = (p2 >= 0) & (p2 < n)
        p2_int = p2.astype(int)
        part2 = np.zeros(2 * n - 1)
        part2[mask2] = h[p2_int[mask2]]
        grad[j] = np.dot(w, part1 - part2)
    return grad, S

    target = n / 2.0

    best_h = h.copy()

    beta = beta_init
    alpha = alpha_init
    vel = np.zeros(n)
    momentum = 0.7
    stall = 0

        if time.time() > deadline:
            break
        if stall > 200:
            break
        grad, S = _smooth_grad(h, beta)
        step = -alpha * (grad + momentum * vel)
        vel = step
        h_new = _project(h + step, target)

def _coord_descent(h, deadline, max_sweeps=5000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall_sweeps = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall_sweeps > 100:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            for _ in range(45):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1 * g))
                f2 = float(np.max(S0 + m2 * g))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall_sweeps = 0
        else:
            stall_sweeps += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _multi_smooth(h, deadline, rng, n_starts=5):

    best_c5 = _exact_c5(h)
    for s in range(n_starts):
        if time.time() > deadline:

        h_s = _perturb(h, 0.03 + 0.02 * s, rng)
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (n_starts - s) * 0.9)
        h_s, c5_s = _smooth_descent(h_s, sub_dl, beta_init=2.0 + s, beta_max=20.0)

            best_h = h2b.copy()

    # Phase 3: LP active set refinement (20%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_active(best_h, d3, radius_init=0.08, stall_limit=120, top_k=14)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Coordinate descent + pair swap (15%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _coord_descent(best_h, min(d4, time.time() + budget_s * 0.06), max_sweeps=4000, seed=777)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 2:
        h4b, c5_4b = _pair_swap(best_h, d4, n_pairs=5000, radius=0.25, seed=333)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-n exploration (15%)
    d5 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    tried_ns = set()
    for dn in range(-20, 21, 2):
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.006)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=300, seed=500 + dn)
        if time.time() < d5 - 3:
            sub_dl2 = min(d5, time.time() + budget_s * 0.004)
            h_r, c5_r2 = _lp_active(h_r, sub_dl2, radius_init=0.05, stall_limit=30, top_k=10)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Final intensive polish (15%)
    d6 = deadline - 2
    n_final = len(best_h)
    # Coordinate descent
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6, c5_6 = _coord_descent(best_h, sub_dl, max_sweeps=5000, seed=999)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # LP
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6, c5_6 = _lp_active(best_h, sub_dl, radius_init=0.04, stall_limit=80, top_k=20)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Pair swap
    if time.time() < d6 - 3:
        h6, c5_6 = _pair_swap(best_h, min(d6, time.time() + budget_s * 0.02), n_pairs=3000, radius=0.15, seed=555)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    # Basin hopping with smooth descent
    scale = 0.015
    n_restarts = 0
    while time.time() < d6 and n_restarts < 80:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d6, time.time() + budget_s * 0.003)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=300, seed=3000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.1)

    # Final projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
