The valid range of t:

To find min of f(t) = max_d (R[d] + t*delta[d]) over this interval:
- f is convex piecewise linear
- Use golden section search (or bisection on the subgradient)

Actually, let me optimize: instead of golden section, I can use the fact that f is piecewise linear convex. The minimum of a piecewise linear convex function on an interval occurs at:
1. A boundary, or
2. A point where the slope changes sign (a "kink")

The slope of f at a non-kink point t is delta[d*] where d* is the unique argmax.

So I can:
1. Find d* at the left boundary → slope_left = delta[d*]
2. Find d* at the right boundary → slope_right = delta[d*]
3. If slope_left ≥ 0: minimum is at left boundary
4. If slope_right ≤ 0: minimum is at right boundary
5. Otherwise: the minimum is at a kink between the two boundaries

For step 5, I can use bisection: find the point where the argmax changes from a "low slope" lag to a "high slope" lag. At that point, f(t) is the intersection of two lines.

Actually, a simpler approach: since f is convex, I can just use ternary search (which works for unimodal functions, and convex functions are unimodal). 40 iterations of ternary search gives precision (2/3)^40 ≈ 10^-8, which is plenty.

import numpy as np
from scipy.optimize import minimize, linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _paper_raw(h):
    """Compute the raw correlation array (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-15)
        else:
            mask = (x > 1e-15)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_pair(h, i, j, R):
    """
    Find optimal t for moving mass from j to i.
    h[i] += t, h[j] -= t, t in [t_lo, t_hi].
    Returns (t_opt, f_at_t_opt) where f(t) = max_d (R[d] + t*delta[d]).
    R is the current raw correlation array.
    """
    n = len(h)
    # Compute delta for all lags
    # delta[d] = (h[j+d] - h[i+d]) + (h[j-d] - h[i-d]) with boundary handling
    # In the 'full' correlation indexing:
    # R[k] corresponds to lag d = k - (n-1)
    # R[d] = sum_m h[m] * (1-h[m+d]) for valid m
    
    # For lag d, the change when h[i]+=t, h[j]-=t is:
    # t * [(1-h[i+d]) - h[i-d] - (1-h[j+d]) + h[j-d]]
    # where h[idx] = 0 if idx out of [0, n-1]
    
    # Let's compute this using the actual indexing
    # h[i+d] is valid if 0 <= i+d < n
    # h[i-d] is valid if 0 <= i-d < n
    # similarly for j
    
    hi_d = 1.0 - h[i]  # t <= 1-h[i]
    lo_d = -h[i]       # t >= -h[i]
    hi_d = min(hi_d, h[j])       # t <= h[j]
    lo_d = max(lo_d, h[j]-1.0)   # t >= h[j]-1
    
    if hi_d - lo_d < 1e-15:
        return 0.0, float(np.max(R))
    
    # Compute delta array
    # delta[k] where k indexes into R (0 to 2n-2), corresponding to lag d = k-(n-1)
    delta = np.zeros(2*n-1)
    
    # Contribution from i: +t*(1-h[i+d]) - t*h[i-d]
    # i+d valid: d in [-i, n-1-i]
    # i-d valid: d in [i-n+1, i]
    d_min_i = -i
    d_max_i = n - 1 - i
    d_min_im = i - n + 1
    d_max_im = i
    # Combined: d in [min(d_min_i, d_min_im), max(d_max_i, d_max_im)]
    # = [min(-i, i-n+1), max(n-1-i, i)]
    
    # For each valid d, the contribution to delta from position i is:
    # (1-h[i+d]) if i+d valid, else 0
    # minus h[i-d] if i-d valid, else 0
    
    # Let me just compute it directly for efficiency
    # For position i:
    # Term 1: if i+d in [0,n), add (1-h[i+d]) to delta[d]
    # Term 2: if i-d in [0,n), subtract h[i-d] from delta[d]
    
    # i+d in [0,n) → d in [-i, n-1-i]
    for d in range(-i, n - i):
        if d >= -(n-1) and d <= n-1:
            delta[n-1+d] += (1.0 - h[i+d])
    # i-d in [0,n) → d in [i-n+1, i]
    for d in range(i-n+1, i+1):
        if d >= -(n-1) and d <= n-1:
            delta[n-1+d] -= h[i-d]
    
    # For position j (with opposite sign since h[j] decreases):
    for d in range(-j, n - j):
        if d >= -(n-1) and d <= n-1:
            delta[n-1+d] -= (1.0 - h[j+d])
    for d in range(j-n+1, j+1):
        if d >= -(n-1) and d <= n-1:
            delta[n-1+d] += h[j-d]
    
    # Now f(t) = max_k (R[k] + t * delta[k]) for k = 0..2n-2
    # Find min of this over [lo_d, hi_d]
    
    # Evaluate at boundaries
    f_lo = float(np.max(R + lo_d * delta))
    f_hi = float(np.max(R + hi_d * delta))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_d, hi_d
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(R + c * delta))
    fd = float(np.max(R + d * delta))
    
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(R + c * delta))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(R + d * delta))
    
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(R + t_opt * delta))
    
    # Also check boundaries
    if f_lo < f_opt:
        t_opt = lo_d
        f_opt = f_lo
    if f_hi < f_opt:
        t_opt = hi_d
        f_opt = f_hi
    
    return t_opt, f_opt

```python
def _compute_delta(h, i, j):
    n = len(h)
    # delta[k] for k in 0..2n-2
    # corresponds to lag d = k - (n-1)
    # delta[d] = (1-h[i+d])*1[valid] - h[i-d]*1[valid] - (1-h[j+d])*1[valid] + h[j-d]*1[valid]
    
    delta = np.zeros(2*n-1)
    
    # For position i, term (1-h[i+d]):
    # i+d valid: 0 <= i+d <= n-1, so -i <= d <= n-1-i
    # In k-index: k = n-1+d, so n-1-i <= k <= 2n-2-i
    # value at k: 1 - h[i + (k-n+1)] = 1 - h[k - (n-1) + i]
    k_start = n - 1 - i
    k_end = 2*n - 2 - i + 1  # exclusive
    idx = np.arange(k_start, k_end)
    delta[idx] += 1.0 - h[idx - (n - 1) + i]
    
    # For position i, term -h[i-d]:
    # i-d valid: 0 <= i-d <= n-1, so i-n+1 <= d <= i
    # In k-index: n-1+i-n+1 <= k <= n-1+i, i.e., i <= k <= n-1+i
    k_start = i
    k_end = n - 1 + i + 1
    idx = np.arange(k_start, k_end)
    delta[idx] -= h[i - (idx - (n-1))]
    
    # For position j, term -(1-h[j+d]):
    k_start = n - 1 - j
    k_end = 2*n - 2 - j + 1
    idx = np.arange(k_start, k_end)
    delta[idx] -= 1.0 - h[idx - (n - 1) + j]
    
    # For position j, term +h[j-d]:
    k_start = j
    k_end = n - 1 + j + 1
    idx = np.arange(k_start, k_end)
    delta[idx] += h[j - (idx - (n-1))]
    
    return delta
```
