        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_quad(h, d, raw, dx, radius, n_top=24):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)
                    a2 = (B - sd) / (2.0 * A)
                    for aa in [a1, a2]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    n_grid = 50
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    cset = set()
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = round(max(0.0, min(alpha, a_hi)), 10)
        if alpha in cset:
            continue
        cset.add(alpha)
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    step = a_hi / n_grid
    lo_g = max(0.0, best_alpha - step)
    hi_g = min(a_hi, best_alpha + step)
    if hi_g - lo_g > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_g - gr * (hi_g - lo_g)
        c2 = lo_g + gr * (hi_g - lo_g)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(70):
            if hi_g - lo_g < 1e-13:
                break
            if f1 < f2:
                hi_g, c2, f2 = c2, c1, f1
                c1 = hi_g - gr * (hi_g - lo_g)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_g, c1, f1 = c1, c2, f2
                c2 = lo_g + gr * (hi_g - lo_g)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo_g + hi_g) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    return best_alpha, best_val

def _lp_direction(h, n_active, trust_radius):
    n = len(h)
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:n_active]
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h - trust_radius,
        x <= h + trust_radius,
    ]
    for idx in top_idx:
        d_lag = int(idx) - (n - 1)
        m = n - abs(d_lag)
        if m <= 0:
            continue
        if d_lag >= 0:
            a_plus = (1.0 - h[:m]).copy()
            a_minus = h[d_lag:d_lag + m].copy()
            const = float(np.sum(h[:m] * h[d_lag:d_lag + m]))
            constraints.append(a_plus @ x[d_lag:d_lag + m] - a_minus @ x[:m] + const <= t)
        else:
            s = -d_lag
            a_plus = (1.0 - h[s:s + m]).copy()
            a_minus = h[:m].copy()
            const = float(np.sum(h[:m] * h[s:s + m]))
            constraints.append(a_plus @ x[:m] - a_minus @ x[s:s + m] + const <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=20000, eps=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=20000)
        except Exception:
            return None
    if x.value is None:
        return None
    return np.array(x.value, dtype=float).flatten()

def _lp_optimize(h, deadline, n_active=30, initial_radius=0.35, stall_limit=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = initial_radius
    stall = 0
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        x_lp = _lp_direction(h, n_active, radius)
        if x_lp is None:
            break
        x_lp = _project(x_lp, target)
        d = x_lp - h
        dmax = float(np.max(np.abs(d)))
        if dmax < 1e-10:
            stall += 1
            radius *= 0.9
            continue
        r_eff = min(radius, dmax * 2.0)
        alpha, c5_try = _line_search_quad(h, d, raw, dx, r_eff, n_top=24)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.88
            continue
        h_new = _project(h + alpha * d, target)
        c5_actual = _exact_c5(h_new)
        if c5_actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_actual
            if cur_c5 < best_c5:
                best_c5 = c5_actual
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.50)
        else:
            stall += 1
            radius *= 0.87
            if radius < 0.03:
                break
    return best_h, best_c5

def _grad_direction(h, beta=20.0):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        wi = w[r]
        if wi < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wi * u[:m]
                g[:m] -= wi * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wi * u[gp:]
                g[gp:] -= wi * h[:m]
    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm > 1e-14:
        g /= nm
    return g

def _grad_optimize(h, deadline, beta=20.0, radius=0.25, stall_limit=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta_cur = beta
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        d = _grad_direction(h, beta_cur)
        if np.linalg.norm(d) < 1e-14:
            break
        alpha, c5_try = _line_search_quad(h, d, raw, dx, radius, n_top=20)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.85
            beta_cur = max(beta_cur * 0.9, 5.0)
            continue
        h_new = _project(h + alpha * d, target)
        c5_actual = _exact_c5(h_new)
        if c5_actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_actual
            if cur_c5 < best_c5:
                best_c5 = c5_actual
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.40)
            beta_cur = min(beta_cur * 1.04, 60.0)
        else:
            stall += 1
            radius *= 0.82
            beta_cur = max(beta_cur * 0.92, 5.0)
            if stall % 40 == 0 and radius < 0.06:
                radius = 0.20
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        h_new = cur.copy()
        h_new[ones[i]] = 0.0
        h_new[zeros[j]] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9993
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.3, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search_quad(h, d, raw, dx, radius, n_top=12)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = _project(h + alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_optimize(h_init, d1, n_active=35, initial_radius=0.35, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient refinement (10%)
    d2 = t0 + budget_s * 0.35
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _grad_optimize(h2, d2, beta=22.0, radius=0.20, stall_limit=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary refinement (15%)
    d3 = t0 + budget_s * 0.50
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=60000, T_init=0.004)
    sub_dl = min(d3, time.time() + budget_s * 0.04)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d3 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_pol, c5_p = _lp_optimize(h_pol, sub_dl, n_active=25, initial_radius=0.15, stall_limit=40)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (25%)
    d4 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in list(range(-24, 25, 3)):
        if time.time() > d4 - 5:
            break
        n_try = n_curr + dn
        if n_try < 50 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.014)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_active=25, initial_radius=0.30, stall_limit=50)
        if time.time() < d4 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d4, time.time() + budget_s * 0.007)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.003 and time.time() < d4 - 2:
            sub_dl3 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, n_active=20, initial_radius=0.12, stall_limit=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Intensive optimization (20%)
    d5 = deadline
    n_curr = len(best_h)
    # LP intensive
    sub_dl = min(d5, time.time() + budget_s * 0.08)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _lp_optimize(h5, sub_dl, n_active=40, initial_radius=0.20, stall_limit=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Pair swap continuous
    if time.time() < d5 - 4:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.07)
        h5, c5_5 = _pair_swap_cont(h5, sub_dl, n_pairs=10000, radius=0.25, seed=333)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Binary round
    if time.time() < d5 - 4:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=50000, T_init=0.0015)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Final LP polish
    if time.time() < d5 - 2:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h5, c5_5 = _lp_optimize(h5, sub_dl, n_active=45, initial_radius=0.10, stall_limit=80)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping (remaining time)
    scale = 0.04
    n_restarts = 0
    while time.time() < deadline - 8 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 6, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, n_active=25, initial_radius=0.15, stall_limit=35)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.10)

    # Final polish
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline - 1, n_active=50, initial_radius=0.06, stall_limit=50)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline - 0.5, n_pairs=2000, radius=0.10, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
