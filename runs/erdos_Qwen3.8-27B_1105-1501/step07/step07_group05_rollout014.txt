def _weighted_grad(h, w):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    return g

    sc = S * dx
    mx = float(np.max(sc))

    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = _weighted_grad(h, w)
    g -= np.mean(g)
    return -g

    h1, c1 = _local_opt(h_init, d1, r0=0.15, stall_lim=55, mode='focus')
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    # Phase 2: Mixed mode local opt (15%)
    d2 = t0 + budget_s * 0.35
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _local_opt(h2, d2, r0=0.12, stall_lim=50, mode='mixed')
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    # Phase 3: Pair optimization (15%)
    d3 = t0 + budget_s * 0.50
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=40000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()
    # Phase 4: Binary SA (10%)
    d4 = t0 + budget_s * 0.60
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _bin_sa(hbin, d4, mx=12000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.02)
        hp, cp = _local_opt(hp, sd, r0=0.08, stall_lim=30, mode='focus')
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.75
    nc = len(best_h)
    offsets = [-6, 6, -3, 3, -9, 9, -12, 12, -15, 15]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 320 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.012)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=25, mode='mixed')
        if cr < best_c + 0.002 and time.time() < d5 - 2:
            sb = min(d5, time.time() + budget_s * 0.005)
            hr, cr = _pair_opt(hr, sb, max_it=6000)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
    # Phase 6: Basin hopping (12%)
    d6 = t0 + budget_s * 0.87
    h6, c6 = _basin_hop(best_h, d6, rng, max_restarts=35)
    if c6 < best_c:
        best_c = c6
        best_h = h6.copy()
    # Phase 7: Final intensive polish (8%)
    d7 = deadline
    nc = len(best_h)
    for r_init in [0.08, 0.04, 0.02]:
        if time.time() > d7 - 1:
            break
        h7 = _project(best_h.copy(), nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.025)
        h7, c7 = _local_opt(h7, sd, r0=r_init, stall_lim=40, mode='focus')
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=8000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
