def _exact_c5(h):

def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

            hi = mid

    res = target - float(np.sum(x))

        else:

def _grad_at_lag(h, r, n):
    """Compute gradient of c[r] with respect to h."""
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += 1.0 - h[:m]
            g[:m] -= h[d:]
    else:
        gp = -d
        m = n - gp
        if m > 0:
            g[:m] += 1.0 - h[gp:]
            g[gp:] -= h[:m]
    return g

def _active_grad(h, n, top_k=5):

    raw = np.correlate(h, 1.0 - h, mode='full')

    g = np.zeros(n)
    for r in idx:
        g += _grad_at_lag(h, int(r), n)
    g /= len(idx)
    g -= np.mean(g)
    return g, raw, idx

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 0.9999999) if res > 0 else (x > 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _grad_lag(h, r, n):
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += 1.0 - h[:m]
            g[:m] -= h[d:]
    else:
        gp = -d
        m = n - gp
        if m > 0:
            g[:m] += 1.0 - h[gp:]
            g[gp:] -= h[:m]
    return g

def _ls_quad(h, d, raw, dx, radius, n_top=14):
    n = len(h)
    u = 1.0 - h

    Q = np.correlate(d, d, mode='full')
    a_hi = radius

    a_hi = max(a_hi, 1e-12)
