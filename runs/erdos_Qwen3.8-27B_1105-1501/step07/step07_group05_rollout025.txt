def _soft_grad(h, beta):
    """Compute smooth gradient of the max objective."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    sc = c * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[n-1+l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l+m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s+m]
    g -= g.mean()
    return -g  # negative gradient (descent direction)

def _line_search(h, d, dx, radius):
    """Line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section
    best_val = float('inf')
    best_a = 0.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo_r, hi_r = a_lo, a_hi
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx

def _smooth_opt(h, deadline, r0=0.15, stall_lim=50):

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _score(h)
    best_h = h.copy()
    best_c = cur

    rad = r0
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        
        # Generate directions
        dirs = []
        for beta in [15.0, 40.0, 100.0]:
            g = _soft_grad(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)

        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(3):
            f = rng.uniform(1.0, 10.0)
            p = rng.uniform(0, 2*np.pi)
            d += np.sin(2*np.pi*f*x/n + p)
        d -= d.mean()
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
        
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - d.mean()
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _score(h)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
        else:
            stall += 1
            rad *= 0.7
            if rad < 0.01:
                # Perturb and restart

                pert -= pert.mean()

                if cp < cur:

                    cur = cp

    return best_h, best_c

def _resample(h, n_new):
    n = len(h)

    xo = (np.arange(n) + 0.5) / n
