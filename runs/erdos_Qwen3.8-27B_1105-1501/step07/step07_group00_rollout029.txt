    if int(t.sum()) > n_half:
        # Too many ones, remove some
        one_indices = np.where(t == 1)[0]
        # Remove the ones with least perturbation
        order = np.argsort(perturbed[one_indices])
        n_remove = n_ones - n_half
        remove_set = set(one_indices[order[:n_remove]].tolist())
        t[remove_set] = 0
    else:
        zero_indices = np.where(t == 0)[0]
        n_add = n_half - n_ones
        order = np.argsort(-perturbed[zero_indices])
        add_set = zero_indices[order[:n_add]].tolist()
        t[add_set] = 1
    return t

def _golden_balanced(n, rng):
    """Generate balanced binary using golden ratio placement."""
    n_half = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    # Use the three-gap theorem for even distribution
    positions = np.array([int(np.floor(i * phi) % n) for i in range(n_half)])
    # Ensure uniqueness
    if len(np.unique(positions)) < n_half:
        # Fallback to van der Corput
        positions = np.array([int(np.floor(i * n / n_half)) for i in range(n_half)])
    # If still not unique, use a different approach
    seen = set()
    final_positions = []
    for p in positions:
        if p not in seen:
            seen.add(p)
            final_positions.append(p)
    while len(final_positions) < n_half:
        p = int(rng.integers(0, n))
        if p not in seen:
            seen.add(p)
            final_positions.append(p)
    h = np.zeros(n)
    h[final_positions[:n_half]] = 1.0
    return h

def _random_balanced(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 1.0
    return h

def _greedy_construct(n, rng):
    """Greedy construction: add each 1 to minimize predicted max."""
    n_half = n // 2
    h = np.zeros(n)
    remaining = n - 1  # remaining positions
    for step in range(n_half):
        best_pos = -1
        best_score = float('inf')
        # Try positions sorted by some heuristic
        candidates = rng.permutation(n)[:(remaining if remaining < 40 else 40)]
        for pos in candidates:
            if h[pos] == 0:
                h[pos] = 1
                # Quick eval
                scores = np.correlate(h, 1 - h, mode='full')
                s = float(np.max(scores))
                h[pos] = 0
                if s < best_score:
                    best_score = s
                    best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1
        else:
            # Fallback
            zero_positions = np.where(h == 0)[0]
            h[zero_positions[0]] = 1
    return h

def _update_scores_swap(sums, h, i, j, n):
    """
    Update the unweighted score vector after swapping h[i]=0->1, h[j]=1->0.
    sums = np.correlate(h, 1-h, mode='full') before swap.
    Returns the updated sums array (in-place modification).
    """
    # The score at index k (0 to 2n-2) corresponds to lag d = k - (n-1)
    # For d >= 0: score = sum_{k=0}^{n-1-d} h[k+d]*(1-h[k])
    # For d < 0 (let g = -d): score = sum_{k=0}^{n-1-g} h[k]*(1-h[k+g])
    
    # We need to compute the change at each affected lag.
    # For simplicity, just recompute the affected lags.
    # Position i (0->1) affects lags d where:
    #   - d >= 0 and i+d < n: the term h[i+d]*(1-h[i]) changes
    #   - d >= 0 and i-d >= 0 and i-d <= n-1-d: the term h[i]*(1-h[i-d]) changes
    #   - d < 0 (g=-d): h[i]*(1-h[i+g]) changes if i+g < n; h[i-g]*(1-h[i]) changes if i-g >= 0
    # Similarly for j (1->0).
    
    # Actually, the simplest and most reliable approach for correctness:
    # Just recompute the full score vector. For n up to ~200, this is fast.
    return np.correlate(h, 1.0 - h, mode='full')

def _binary_search_focused(h, deadline, max_iters=200000):
    """
    Focused binary minimax local search.
    Targets the worst lag specifically.
    """
    n = len(h)
    h = h.copy().astype(float)
    assert abs(h.sum() - n/2) < 1e-9
    
    cur_scores = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(cur_scores))
    best_h = h.copy()
    best_max = cur_max
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    rng = np.random.default_rng(777)
    stall = 0
    T = 0.5
    n_swaps_per_iter = 50
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            # Try a random restart within the same n
            # Just keep going with higher temperature
            T = min(T * 1.5, 5.0)
            stall = 0
            if T > 10:
                break
        
        # Find the top few worst lags
        max_val = float(np.max(cur_scores))
        # Get indices where score is close to max
        threshold = max_val - 0.5
        active = np.where(cur_scores >= threshold)[0]
        
        improved = False
        for _ in range(n_swaps_per_iter):
            if time.time() > deadline:
                break
            
            # Pick a random swap
            i = zeros[rng.integers(0, len(zeros))]
            j = ones[rng.integers(0, len(ones))]
            
            # Quick check: does this swap help at the worst lag?
            # The worst lag is at index active[0] (or the argmax)
            worst_idx = int(np.argmax(cur_scores))
            d_worst = worst_idx - (n - 1)  # signed lag
            
            # Compute the change at the worst lag
            # For the full score vector, position i (0->1) and j (1->0):
            # We'll just do a quick partial evaluation
            
            # Actually, for speed, let's just do the swap and evaluate
            h[i] = 1.0
            h[j] = 0.0
            new_scores = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_scores))
            
            # Restore for now
            h[i] = 0.0
            h[j] = 1.0
            
            if new_max < cur_max:
                # Accept
                h[i] = 1.0
                h[j] = 0.0
                cur_scores = new_scores
                cur_max = new_max
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                stall = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                improved = True
                break
            elif new_max - cur_max < T and rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-10)):
                # Annealing accept
                h[i] = 1.0
                h[j] = 0.0
                cur_scores = new_scores
                cur_max = new_max
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                stall = 0
        if not improved:
            stall += 1
            T = max(T * 0.998, 0.05)
    
    return best_h, best_max * (2.0/n)
