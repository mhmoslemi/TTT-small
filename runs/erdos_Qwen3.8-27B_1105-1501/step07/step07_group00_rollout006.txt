            x[mask] += res / cnt

    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        gv = -d
        m = n - gv
        if m > 0:
            g[:m] += (1.0 - h[gv:])
            g[gv:] += -h[:m]
    return g * dx
