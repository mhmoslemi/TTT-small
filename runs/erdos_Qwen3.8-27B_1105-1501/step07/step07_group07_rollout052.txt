```python
def _grad_matrix(h):
    n = len(h)
    mid = n - 1
    G = np.zeros((2*n-1, n))
    # d = 0
    G[mid] = 1.0 - 2.0*h
    for d in range(1, n):
        r_pos = mid + d  # positive lag
        r_neg = mid - d  # negative lag
        m = n - d
        # Positive lag d: raw = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        # d/dh[j]: if j in [d, n-1]: += (1-h[j-d]); if j in [0, m-1]: -= h[j+d]
        G[r_pos, d:] += (1.0 - h[:m])  # h[j-d] for j in [d, d+m-1] = h[0:m]
        G[r_pos, :m] -= h[d:]  # h[j+d] for j in [0, m-1] = h[d:d+m]
        
        # Negative lag -d: raw = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
        # d/dh[j]: if j in [0, m-1]: += (1-h[j+d]); if j in [d, n-1]: -= h[j-d]
        G[r_neg, :m] += (1.0 - h[d:])
        G[r_neg, d:] -= h[:m]
    return G
```

            best_t = t
    h_out = _project_sum((1.0 - best_t) * h1 + best_t * h2, target)
    return h_out, best_c5

def _pair_polish(h, deadline, n_trials=5000, seed=99):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        h2 = h.copy()
        h2[i] += 1.0
        h2[j] -= 1.0
        h2 = _project_sum(h2, target)
        _, c5_try = _exact_line_search(h, h2, deadline, n_grid=8)
        if c5_try < cur_c5 - 1e-13:
            h = _project_sum((h + h2) * 0.5, target)
            h, c5_try2 = _exact_line_search(h, h2, deadline, n_grid=8)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(ones[rng.integers(0, len(ones))])
        oz = int(zeros[rng.integers(0, len(zeros))])
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new % 2 != 0:
        n_new += 1
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    target = n / 2.0
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert += amplitude * rng.normal(0, 1, n)
    pert *= amplitude
    return _project_sum(h + pert, target)

def _pair_sweep_binary(h_bin, deadline, max_rounds=20):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _transforms_of(h):
    n = len(h)
    target = n / 2.0
    results = [h.copy()]
    results.append(_project_sum(h[::-1].copy(), target))
    results.append(_project_sum(1.0 - h, target))
    results.append(_project_sum(1.0 - h[::-1], target))
    results.append(_project_sum(0.5 * (h + h[::-1]), target))
    return results

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is None or len(h0) == 0:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        n0 += 1
        h0 = _resample(h0, n0)
    n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Generate transforms
    transforms = _transforms_of(h_init)
    for t_h in transforms:
        c5_t = _exact_c5(t_h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = t_h.copy()

    # Phase 1: Main LSE annealing on best transform (40% of budget)
    d1 = t0 + budget_s * 0.40
    h1, c5_1 = _anneal_loop(best_h.copy(), d1, tau_init=0.08, tau_min=0.002, decay=0.62, maxiter=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Also try annealing on other transforms with remaining time
    d1b = t0 + budget_s * 0.50
    for t_h in transforms[1:]:
        if time.time() >= d1b:
            break
        sub_dl = min(d1b, time.time() + budget_s * 0.04)
        h_t, c5_t = _anneal_loop(t_h.copy(), sub_dl, tau_init=0.06, tau_min=0.003, decay=0.65, maxiter=80)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 2: Binary refinement (15% of budget)
    d2 = t0 + budget_s * 0.65
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003, seed=421)
    sub_dl2 = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _pair_sweep_binary(h_bin, sub_dl2, max_rounds=20)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # LP-style polish on binary result (convert to continuous)
    if time.time() < d2:
        sub_dl3 = min(d2, time.time() + budget_s * 0.03)
        h_cont = _project_sum(h_bin.copy(), len(h_bin) / 2.0)
        h_cont, c5_cont = _anneal_loop(h_cont, sub_dl3, tau_init=0.03, tau_min=0.002, decay=0.6, maxiter=80)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 3: Multi-n exploration (20% of budget)
    d3 = t0 + budget_s * 0.85
    n_best = len(best_h)
    n_offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in sorted(n_offsets, key=lambda x: abs(x)):
        if time.time() >= d3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _anneal_loop(h_r, sub_dl, tau_init=0.06, tau_min=0.003, decay=0.62, maxiter=70)
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.012)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try)
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b = _pair_sweep_binary(h_b, sub_dl3, max_rounds=10)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Basin hopping (10% of budget)
    d4 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d4 - 10 and n_restarts < 150:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.006)
        h_t, c5_t = _anneal_loop(h_pert, sub_dl, tau_init=0.05, tau_min=0.004, decay=0.6, maxiter=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 5: Final polish
    if time.time() < deadline:
        n_cur = len(best_h)
        sub_dl = min(deadline, time.time() + budget_s * 0.02)
        h_fin = _project_sum(best_h.copy(), n_cur / 2.0)
        h_fin, c5_fin = _anneal_loop(h_fin, sub_dl, tau_init=0.02, tau_min=0.001, decay=0.6, maxiter=100)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final pair polish
    if deadline - time.time() > 10:
        n_cur = len(best_h)
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h_pp, c5_pp = _pair_polish(best_h.copy(), sub_dl, n_trials=3000, seed=777)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp.copy()

    # Final binary check
    if deadline - time.time() > 5:
        h_b = _to_binary(best_h)
        c5_b = _exact_c5(h_b)
        h_b2, c5_b2 = _pair_sweep_binary(h_b, deadline, max_rounds=10)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Ensure feasibility
    n_final = len(best_h)
    best_h = _project_sum(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, n_final
