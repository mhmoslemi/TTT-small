            x[mask] += res / cnt

def _obj_lse(x):
    n = len(x)
    dx = 2.0 / n
    xe = np.clip(x, 0.0, 1.0)
    sc = np.correlate(xe, 1.0 - xe, mode='full') * dx
    m = float(np.max(sc))
    z = (sc - m) / _BETA
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw < 1e-300:
        return m
    return float(m + _BETA * np.log(sw))

    n = len(x)
    dx = 2.0 / n
    xe = np.clip(x, 0.0, 1.0)
    sc = np.correlate(xe, 1.0 - xe, mode='full') * dx
    m = float(np.max(sc))

    w = np.exp(z)
    sw = float(np.sum(w))
    if sw < 1e-300:
        k = int(np.argmax(sc))
        w = np.zeros(len(sc))
        w[k] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    for k in range(len(sc)):
        wk = w[k]
        if wk < 1e-10:
            continue
        d = k - (n - 1)
        if d >= 0:
            ml = n - d
            if ml > 0:
                grad[d:] += wk * (1.0 - xe[:ml])
                grad[:ml] += -wk * xe[d:]
        else:
            g = -d
            ml = n - g
            if ml > 0:
                grad[:ml] += wk * (1.0 - xe[g:])
                grad[g:] += -wk * xe[:ml]
    return grad * dx

def _eq_fun(x):
    return float(np.sum(x) - len(x) / 2.0)

def _eq_jac(x):
    return np.ones(len(x))

def _slsqp_run(h0, beta, deadline, maxiter=200):
    global _BETA
    _BETA = beta
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.fun < _obj_lse(h0):
            h_new = _project(res.x, n / 2.0)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h0.copy(), _exact_c5(h0)

    c5_cur = _exact_c5(h_cur)
    best_h = h_cur.copy()

        else:

        # Active set
        active = np.where(scores >= mx - 0.005)[0]
        if len(active) > n_active:
            # Keep top n_active
            order = np.argsort(scores[active])[::-1][:n_active]
            active = active[order]
        # Compute subgradient (average of active score gradients)
        n_act = len(active)
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            if d >= 0:
                ml = n - d
                if ml > 0:

            else:

                ml = n - g
                if ml > 0:
                    grad[:ml] += (1.0 - h[g:])
                    grad[g:] += -h[:ml]

        # Projected gradient step
        grad -= np.mean(grad)  # Ensure sum = 0
        step = lr * grad

                best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
            stall = 0
        else:
            stall += 1
            lr *= 0.7
            if lr < 1e-6:
