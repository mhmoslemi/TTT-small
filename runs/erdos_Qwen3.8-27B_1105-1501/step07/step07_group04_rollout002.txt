def project_to_box_sum(v, target):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        else:

    n = len(h)
    dx = 2.0 / n

    corr = corr_raw * dx  # scaled correlation values
