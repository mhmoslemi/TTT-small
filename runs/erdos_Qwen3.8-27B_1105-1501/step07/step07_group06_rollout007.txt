import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_weighted(h, beta, k_top=20):
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = raw.max()
    v = beta * (raw - M)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = w.sum()
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    w /= wsum
    g = np.zeros(n)
    for r in range(len(raw)):
        if w[r] < 1e-8:
            continue
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += wr * u[:m]
                g[:m] -= wr * h[tau:tau+m]
        else:
            d = -tau
            m = n - d
            if m > 0:
                g[:m] += wr * u[d:]
                g[d:] -= wr * h[:m]
    g -= g.mean()
    return g, raw

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau+m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    g -= g.mean()
    return g

def _to_binary(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[-m:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    while time.time() < deadline:
        if stall > max_stall or not ones or not zeros:
            break
        oi = ones[rng.integers(0, len(ones))]
        oz = zeros[rng.integers(0, len(zeros))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            T *= cool
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            T *= cool
    return best_h, best_c5

def _continuous_line_search(h, d, radius=0.2):
    """Line search along direction d while maintaining constraints."""
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Grid search
    ng = 50
    gt = np.linspace(max(a_lo, -10.0), min(a_hi, 10.0), ng)
    gf = np.array([f(t) for t in gt])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    
    # Check quadratic vertices
    for r in np.argsort(S0)[-6:]:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    
    for t in [a_lo, a_hi, 0.0]:
        if abs(t) > 1e-15:
            ft = f(t)
            if ft < opt_v:
                opt_t, opt_v = t, ft
    
    return opt_t, opt_v

def _continuous_optimize(h, deadline, beta_start=8.0, beta_end=50.0, max_iter=60):
    """Smooth gradient descent for continuous h."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(555)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 20:
            break
        
        beta = beta_start + (beta_end - beta_start) * (it / max_iter)
        g, raw = _grad_weighted(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        # Create several directions
        dirs = [g / gn]
        # Block-averaged
        for blk in [4, 8]:
            if n % blk == 0:
                nb = n // blk
                gb = g.reshape(nb, blk).mean(axis=1)
                d = np.tile(gb, (nb, 1)).reshape(n).copy()
                d -= d.mean()
                dn = np.linalg.norm(d)
                if dn > 1e-12:
                    dirs.append(d / dn)
        # Random smooth perturbation
        d_rand = np.sin(2.0 * np.pi * rng.uniform(1, 4) * np.arange(n) / n)
        d_rand -= d_rand.mean()
        dn = np.linalg.norm(d_rand)
        if dn > 1e-12:
            dirs.append(d_rand / dn)
        
        # Remove mean from all dirs (preserve sum)
        dirs = [d - d.mean() for d in dirs]
        
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _continuous_line_search(h, d, radius=0.15)
            if val < best_val - 1e-13:
                best_val, best_alpha, best_d = val, alpha, d
        
        if best_d is not None and abs(best_alpha) > 1e-12:
            h_new = _project(h + best_alpha * best_d, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _compression_eval(b, alphas=None):
    """Evaluate compressed versions of binary sequence."""
    if alphas is None:
        alphas = [0.0, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    n = len(b)
    best_c5 = float('inf')
    best_h = b.copy()
    for a in alphas:
        h_c = a * 0.5 + (1.0 - a) * b
        c5 = _exact_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
    return best_h, best_c5

        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.07)
        h_pol, c5_pol = _continuous_optimize(h_pol, sub_dl, beta_lo=12.0, beta_hi=60.0, radius=0.10, max_stall=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Basin hopping + final (10%)
    d5 = deadline
    scale = 0.025
    rng_final = np.random.default_rng(seed + 100)
    while time.time() < d5 - 5:
        n = len(best_h)
        x = np.arange(n, dtype=float)
        pert = np.zeros(n)
        for _ in range(3):
            freq = rng_final.uniform(0.5, 5.0)
            phase = rng_final.uniform(0, 2 * np.pi)
            pert += np.sin(2.0 * np.pi * freq * x / n + phase)
        pert *= scale / 3.0
        pert += 0.01 * scale * rng_final.normal(0, 1, n)
        h_p = _project(best_h + pert, n / 2.0)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.005)
        h_t, c5_t = _continuous_optimize(h_p, sub_dl, beta_lo=10.0, beta_hi=40.0, radius=0.08, max_stall=20)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)

    # Final: try nearby sizes
    n_final = len(best_h)
    for dn in [-6, 6, -12, 12]:
        if time.time() > d5 - 10:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 220 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.012)
        h_r, c5_r = _continuous_optimize(h_r, sub_dl, beta_lo=15.0, beta_hi=50.0, radius=0.08, max_stall=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _continuous_optimize(h_f, deadline, beta_lo=20.0, beta_hi=80.0, radius=0.05, max_stall=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    return best_h, final_c5, len(best_h)
