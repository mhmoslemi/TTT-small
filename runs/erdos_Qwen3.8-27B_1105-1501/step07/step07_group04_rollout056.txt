def _lp_minimax_direction(h, radius):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    F = corr_raw * dx  # scaled correlation values
    
    m = 2 * n - 1  # number of lags
    
    # Build constraint matrix
    # Variables: d_0, ..., d_{n-1}, t
    # A_ub: m rows, n+1 cols
    # b_ub: m values
    
    A_rows = []
    b_vals = []
    
    for r in range(m):
        k = r - (n - 1)
        g = _lag_gradient(h, k)  # already scaled by dx
        # Constraint: g · d - t ≤ -F[r]
        row = np.append(g, -1.0)
        A_rows.append(row)
        b_vals.append(-F[r])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Mass constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(radius, h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((-10.0, 10.0))  # t is free (with large bounds)
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        return d, t

        return None, None
