                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _gen_dirs(h, rng, nd=5):
    n = len(h)
    dirs = []
    for beta in [15.0, 30.0, 50.0]:
        g = _smooth_grad(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
        if len(dirs) >= nd:
            break
    x = np.arange(n, dtype=float)
    for _ in range(nd - len(dirs)):
        d = np.zeros(n)
        for _ in range(3):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

def _local_opt(h, deadline, r0=0.18, stall_lim=60, nd=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(12345)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = _gen_dirs(h, rng, nd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.35)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.006:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 40 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(55555)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_to_cont(hb, t=0.05):
    n = len(hb)
    hc = np.where(hb > 0.5, 1.0 - t, t)
    return _project(hc, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Binary optimization (12% of budget)
    d1 = t0 + budget_s * 0.12
    # Start from thresholded continuous
    hb0 = _to_bin(h_init, n0)
    hb0, cb0 = _binary_best_swap(hb0, d1)
    if cb0 * 2.0 / n0 < best_c:
        best_c = cb0 * 2.0 / n0
        best_h = _project(hb0.astype(float), n0 / 2.0)

    # Multi-start binary at current n
    hb_ms, cb_ms = _binary_multi_start(n0, d1, n_starts=15)
    if hb_ms is not None and cb_ms * 2.0 / n0 < best_c:
        best_c = cb_ms * 2.0 / n0
        best_h = _project(hb_ms.astype(float), n0 / 2.0)

    # Try different n for binary
    for nb_try in [n0 + 10, n0 - 10, n0 + 20, n0 - 20, n0 + 30]:
        if time.time() > d1 or nb_try < 80 or nb_try > 280:
            continue
        # Resample current best to nb_try then binarize
        hr = _resample(best_h, nb_try)
        hb_r = _to_bin(hr, nb_try)
        hb_r, cb_r = _binary_best_swap(hb_r, min(d1, time.time() + budget_s * 0.01))
        if cb_r * 2.0 / nb_try < best_c:
            best_c = cb_r * 2.0 / nb_try
            best_h = _project(hb_r.astype(float), nb_try / 2.0)

    # Phase 2: Continuous opt from best binary and original (18%)
    d2 = t0 + budget_s * 0.30
    nc = len(best_h)
    # From best binary continuation
    hc = _bin_to_cont(_to_bin(best_h, nc), t=0.08)
    hc, cc = _local_opt(hc, min(d2, time.time() + budget_s * 0.05), r0=0.15, stall_lim=40, nd=5)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()

    # From original continuous
    h2 = _project(h_init.copy(), n0 / 2.0)
    h2, c2 = _local_opt(h2, min(d2, time.time() + budget_s * 0.05), r0=0.15, stall_lim=40, nd=5)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # From average of binary-continued and original
    avg_h = _project(0.5 * best_h + 0.5 * h_init[:min(len(best_h), n0)] + 0.5 * h_init[len(best_h):len(h_init)] if len(best_h) != n0 else 0.5 * best_h + 0.5 * h_init, len(best_h) / 2.0)
    if time.time() < d2:
        avg_h, avg_c = _local_opt(avg_h, min(d2, time.time() + budget_s * 0.03), r0=0.10, stall_lim=30, nd=4)
        if avg_c < best_c:
            best_c = avg_c
            best_h = avg_h.copy()

    # Phase 3: Pair optimization (12%)
    d3 = t0 + budget_s * 0.42
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=40000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Multi-resolution continuous (18%)
    d4 = t0 + budget_s * 0.60
    nc = len(best_h)
    offsets = [-6, 6, -12, 12, -18, 18, -24, 24, n0 - nc + 10, n0 - nc - 10]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.02)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=35, nd=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
        elif cr < best_c + 0.003:
            # Also try binary at this resolution then back to continuous
            hb_r = _to_bin(hr, n_try)
            hb_r, cb_r = _binary_best_swap(hb_r, min(d4, time.time() + budget_s * 0.006))
            if cb_r * 2.0 / n_try < cr:
                hc_r = _bin_to_cont(hb_r, t=0.06)
                hc_r, cc_r = _local_opt(hc_r, min(d4, time.time() + budget_s * 0.01), r0=0.08, stall_lim=25, nd=4)
                if cc_r < best_c:
                    best_c = cc_r
                    best_h = hc_r.copy()
                    nc = n_try

    # Phase 5: Intensive local + pair (20%)
    d5 = t0 + budget_s * 0.80
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025, 0.012]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.04)
        h5, c5 = _local_opt(h5, sd, r0=r_init, stall_lim=50, nd=5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    if time.time() < d5:
        h5 = _project(best_h.copy(), nc / 2.0)
        h5, c5 = _pair_opt(h5, min(d5, time.time() + budget_s * 0.03), max_it=20000)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Phase 6: Basin hopping (10%)
    d6 = deadline
    scale = 0.02
    for _ in range(80):
        if time.time() > d6:
            break
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.2 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.006)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.015, stall_lim=40, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=8000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
