def _project(v, target):
    # Project onto simplex-like constraint: sum = target, 0 <= v <= 1
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_at_lag(h, lag):
    """Compute correlation at a specific lag (offset from center)."""
    n = len(h)
    # np.correlate(h, 1-h, mode='full') has indices 0..2n-2
    # lag = index - (n-1), so lag ranges from -(n-1) to (n-1)
    # For positive lag d >= 0: sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
    # For negative lag d < 0: let d' = -d, sum_{i=0}^{n-1-d'} h[i+d'] * (1-h[i])
    u = 1.0 - h
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return 0.0
        return float(np.sum(h[:m] * u[lag:]))
    else:
        d = -lag
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.sum(h[d:] * u[:m]))

def _gradient_at_lag(h, lag):
    """Gradient of correlation at lag with respect to h."""
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return g
        # C = sum_{i=0}^{m-1} h[i] * u[i+lag]
        # dC/dh[i] = u[i+lag] for i in 0..m-1
        # dC/dh[i+lag] = -h[i] for i in 0..m-1, i.e., h[i+lag] gets -h[i-lag]
        g[:m] += u[lag:]
        g[lag:] -= h[:m]
    else:
        d = -lag
        m = n - d
        if m <= 0:
            return g
        # C = sum_{i=0}^{m-1} h[i+d] * u[i]
        # dC/dh[i+d] = u[i] for i in 0..m-1
        # dC/dh[i] = -h[i+d] for i in 0..m-1
        g[d:] += u[:m]
        g[:m] -= h[d:]
    return g

def _lp_direction(h, raw, top_k=30, delta=0.1):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    # Select active lags: top_k highest correlation values
    idx = np.argsort(-raw)[:top_k]
    # Build LP
    # Variables: [t, d_0, ..., d_{n-1}]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = []
    b_ub = []
    
    for lag_idx in idx:
        lag = int(lag_idx) - (n - 1)
        g = _gradient_at_lag(h, lag)
        # g · d <= t - C0(lag)
        # -g · d + t >= C0(lag)  =>  t - g · d >= C0(lag)
        # In standard form: -t + g · d <= -C0(lag)
        row = np.zeros(n + 1)
        row[0] = -1.0  # coefficient of t
        row[1:] = g
        A_ub.append(row)
        b_ub.append(-raw[lag_idx])
    
    # Box constraints on d: -h_i <= d_i <= 1 - h_i
    # Also |d_i| <= delta (trust region)
    lo = np.maximum(-h, -delta)
    hi = np.minimum(1.0 - h, delta)
    
    # For linprog: bounds on variables
    bounds = [(None, None)]  # t is free
    for i in range(n):
        bounds.append((lo[i], hi[i]))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = [0.0]
    
    if not A_ub:
        return None
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[1:]
    # Normalize to have max abs = delta
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-12:
        return None
    d = d / max_abs * delta
    return d

import time
from scipy.optimize import linprog

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

    return x

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _neg(d, h, t, Q):
    """Quadratic model value at step size d for many lags at once."""
    return t + d - Q * d * d

def _line_search_exact(h, d, raw, dx, a_max):
    """Exact line search: find best alpha in [0, a_max] minimizing max(raw + alpha*T - alpha^2*Q)*dx"""
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = 0.5 * np.correlate(d, d, mode='full')
    # Objective: max_k (raw[k] + alpha*T[k] - alpha^2*Q[k]) * dx
    # Wait, let me recompute.
    # C(k) = sum_i h[i] * (1-h[i+k])
    # C(k; h + alpha*d) = sum_i (h[i]+alpha*d[i]) * (1-h[i+k]-alpha*d[i+k])
    # = sum_i h[i]*(1-h[i+k]) - alpha*sum_i h[i]*d[i+k] + alpha*sum_i d[i]*(1-h[i+k]) - alpha^2*sum_i d[i]*d[i+k]
    # = raw[k] + alpha*(T[k]) - alpha^2*Q[k]
    # where T[k] = sum_i d[i]*(1-h[i+k]) - h[i]*d[i+k]
    # and Q[k] = sum_i d[i]*d[i+k]
    # So T = correlate(d, 1-h) - correlate(h, d)
    # and Q = correlate(d, d)
    # (Note: Q is always >= 0 for real d, since it's like a "self-correlation")
    
    # Actually wait: Q[k] = sum_i d[i]*d[i+k] which is the correlation of d with itself at lag k.
    # For k=0: Q[0] = sum d[i]^2 >= 0
    # For k != 0: Q[k] can be negative!
    # So the quadratic in alpha for each k is: raw[k] + T[k]*alpha - Q[k]*alpha^2
    # This is a concave quadratic if Q[k] > 0, convex if Q[k] < 0.
    # The max over k of concave/convex functions... this is tricky.
    
    # For line search, I'll use a combination of:
    # 1. Exact solutions for each active lag (where derivative = 0)
    # 2. Golden section search on [0, a_max]
    
    # First, find the feasible range: 0 <= h[i] + alpha*d[i] <= 1
    if a_max <= 0:
        return 0.0, float(np.max(raw)) * dx
    
    pos = d > 1e-12
    neg = d < -1e-12
    a_hi = a_max
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 0.0)
    if a_hi < 1e-14:
        return 0.0, float(np.max(raw)) * dx
    
    # Collect candidate alpha values
    candidates = set()
    candidates.add(0.0)
    candidates.add(a_hi)
    
    # Top k lags by current value
    top_k_idx = np.argsort(-raw)[:20]
    for idx in top_k_idx:
        qi = Q[idx]
        ti = T[idx]
        ri = raw[idx]
        if abs(qi) > 1e-14:
            # -2*qi*alpha + ti = 0 => alpha = ti / (2*qi)
            a_star = ti / (2.0 * qi)
            if 0 < a_star < a_hi:
                candidates.add(a_star)
        # Also where this lag equals the current max
        # raw[idx] + ti*alpha - qi*alpha^2 = max_raw
        # -qi*alpha^2 + ti*alpha + (raw[idx] - max_raw) = 0
        max_raw = float(np.max(raw))
        a = -qi
        b = ti
        c_val = ri - max_raw
        if abs(a) > 1e-14:
            disc = b*b - 4*a*c_val
            if disc >= 0:
                for a_cand in [(b + np.sqrt(disc))/(2*a), (b - np.sqrt(disc))/(2*a)]:
                    if 0 < a_cand < a_hi:
                        candidates.add(a_cand)
    
    # Grid
    for i in range(1, 40):
        candidates.add(a_hi * i / 40.0)
    
    # Evaluate
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    
    # Golden section refinement
    step = a_hi / 40.0
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(60):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    
    return best_alpha, best_val

def _lp_step(h, raw, dx, n_active=None, delta=0.15):
    """LP-based step: find direction that reduces max of top active lags."""
    n = len(h)
    if n_active is None:
        # Use lags above 90% of max
        thresh = 0.90 * float(np.max(raw))
        active_idx = np.where(raw > thresh)[0]
        if len(active_idx) < 5:
            active_idx = np.argsort(-raw)[:20]
        elif len(active_idx) > 80:
            active_idx = np.argsort(-raw)[:80]
    else:
        active_idx = np.argsort(-raw)[:n_active]
    
    # Build LP
    # Variables: [t, d_0, ..., d_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    A_rows = []
    b_vals = []
    
    for li in active_idx:
        lag = int(li) - (n - 1)
        # Gradient of C(lag) at h
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[:m] = (1.0 - h[lag:lag+m])
                g[lag:lag+m] -= h[:m]
        else:
            d_val = -lag
            m = n - d_val
            if m > 0:
                g[d_val:d_val+m] = (1.0 - h[:m])
                g[:m] -= h[d_val:d_val+m]
        # Constraint: t - g·d >= raw[li]  =>  -t + g·d <= -raw[li]
        row = np.zeros(n + 1)
        row[0] = -1.0
        row[1:] = g
        A_rows.append(row)
        b_vals.append(-raw[li])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Box: -min(h, delta) <= d_i <= min(1-h, delta)
    lo = np.maximum(-h, -delta)
    hi = np.minimum(1.0 - h, delta)
    
    bounds = [(None, None)] + list(zip(lo.tolist(), hi.tolist()))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[1:].copy()
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-12:
        return None
    # Don't rescale - the LP already respects the trust region
    return d

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_corr(h):

def _line_search_quad(h, d, raw, dx, a_max, n_top=25):

        return 0.0, float(np.max(raw)) * dx
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]

def _lp_direction(h, raw, n_active=50, delta=0.15):
    n = len(h)
    thresh = 0.92 * float(np.max(raw))
    active_mask = raw > thresh
    if np.sum(active_mask) < 10:
        active_idx = np.argsort(-raw)[:n_active]
    elif np.sum(active_mask) > 120:
        active_idx = np.argsort(-raw)[:n_active]
    else:
        active_idx = np.where(active_mask)[0]
    active_idx = active_idx[:n_active]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_rows = []
    b_vals = []
    for li in active_idx:
        lag = int(li) - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[:m] = (1.0 - h[lag:lag + m])
                g[lag:lag + m] -= h[:m]
        else:
            dv = -lag
            m = n - dv
            if m > 0:
                g[dv:dv + m] = (1.0 - h[:m])
                g[:m] -= h[dv:dv + m]
        row = np.zeros(n + 1)
        row[0] = -1.0
        row[1:] = g
        A_rows.append(row)
        b_vals.append(-raw[li])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    lo = np.maximum(-h, -delta)
    hi = np.minimum(1.0 - h, delta)
    bounds = [(None, None)] + list(zip(lo.tolist(), hi.tolist()))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    d = res.x[1:].copy()
    if np.max(np.abs(d)) < 1e-12:
        return None
    return d

def _gradient_smooth(h, raw, beta=20.0):
    n = len(h)
    mx = float(np.max(raw))
    shifted = raw - mx
    w = np.exp(np.clip(beta * shifted, -80, 0))
    w = w / (float(np.sum(w)) + 1e-30)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wi = w[r]
        if wi < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += wi * (1.0 - h[d:d + m])
                g[d:d + m] -= wi * h[:m]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[gp:gp + m] += wi * (1.0 - h[:m])
                g[:m] -= wi * h[gp:gp + m]
    g -= np.mean(g)
    return g

    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(9999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-15:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.3, 7.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.25 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP + gradient optimization at initial n (35%)
    d1 = min(t0 + budget_s * 0.35, deadline - 20)
    h1, c5_1 = _main_optimize(h_init, d1, n_active=50, delta=0.18, stall_limit=120, beta=22.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (25%)
    d2 = min(t0 + budget_s * 0.60, deadline - 20)
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in list(range(-24, 25, 2)):
        if time.time() > d2 - 5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2 - 3, time.time() + budget_s * 0.009)
        h_r, c5_r = _main_optimize(h_r, sub_dl, n_active=40, delta=0.14, stall_limit=35, beta=18.0)
        if c5_r < best_c5 - 0.00005:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Intensive multi-method (25%)
    d3 = min(t0 + budget_s * 0.85, deadline - 10)
    n_curr = len(best_h)
    # LP intensive
    h3 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h3, c5_3 = _main_optimize(h3, sub_dl, n_active=70, delta=0.10, stall_limit=100, beta=30.0)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Binary round
    if time.time() < d3 - 5:
        h_b = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.06)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.002)
        sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Polish binary with LP
        if time.time() < d3 - 3:
            h_pol = _project(h_b.copy(), len(h_b) // 2.0)
            n_pol = len(h_pol)
            if n_pol % 2 != 0:
                h_pol = _project(h_b.copy(), n_pol / 2.0)
            sub_dl3 = min(d3 - 1, time.time() + budget_s * 0.03)
            h_pol, c5_p = _main_optimize(h_pol, sub_dl3, n_active=50, delta=0.08, stall_limit=50, beta=25.0)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()
    # More LP with tighter delta
    if time.time() < d3 - 3:
        h3 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d3 - 1, time.time() + budget_s * 0.05)
        h3, c5_3 = _main_optimize(h3, sub_dl, n_active=80, delta=0.05, stall_limit=80, beta=35.0)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Basin hopping (15%)
    d4 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d4 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.010)
        h_t, c5_t = _main_optimize(h_pert, sub_dl, n_active=50, delta=0.12, stall_limit=35, beta=25.0)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.12)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _main_optimize(h6, deadline, n_active=100, delta=0.03, stall_limit=80, beta=40.0)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
