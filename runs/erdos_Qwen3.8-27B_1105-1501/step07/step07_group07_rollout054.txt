import numpy as np
from scipy.optimize import minimize
import time

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    # lag 0
    G[mid] = 1.0 - 2.0 * h
    for k in range(1, n):

        r = mid + k
        m = n - k

        G[r, :m] -= h[k:]
        # negative lag -k: row = mid - k
        r = mid - k
        G[r, k:] += 1.0 - h[:m]
        G[r, :m] -= h[k:]
    return G

import numpy as np

import time

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        r_pos = mid + k
        G[r_pos, k:] += 1.0 - h[:m]
        G[r_pos, :m] -= h[k:]
        r_neg = mid - k
        G[r_neg, k:] += 1.0 - h[:m]
        G[r_neg, :m] -= h[k:]
    return G

                          bounds=bounds, constraints=constraints,

            pass

Let me also add a "coordinate descent" step: for each coordinate j, try to find the best value of h[j] (with all others fixed) that minimizes the max score. This is a 1D optimization problem.

For coordinate j, the scores are:
score_r(j) = sum over valid i of the terms involving h[j]

For a fixed j, each score_r is linear in h[j] (since the score is bilinear in h):
score_r = a_r * h[j] + b_r

where a_r and b_r depend on j and the other h values.

So the max of {a_r * h[j] + b_r} over r is the upper envelope of lines in h[j]. We want to find h[j] in [lo_j, hi_j] (respecting the sum constraint) that minimizes this upper envelope.

This is a 1D convex optimization problem (min of upper envelope of lines = convex function). We can solve it by:
1. Finding the breakpoints of the upper envelope
2. Checking the minimum at each breakpoint and at the boundaries

But computing the upper envelope for 2n-1 lines is O(n^2) in the worst case. For n=180, that's about 359 lines, and the envelope has at most 359 breakpoints. Computing it takes O(n^2 log n) or O(n^2) with a divide-and-conquer approach.

Actually, for a simpler approach: just scan h[j] over a grid of, say, 100 points in [0, 1] (or [lo_j, hi_j]), evaluate the max score at each, and pick the best. Then refine with a few bisections.

But this has to respect the sum constraint. If we change h[j] from its current value to a new value v, we need to adjust another coordinate to maintain the sum. This makes it no longer a simple 1D problem.

Alternative: use the "pair direction" approach. Pick two coordinates (i, j) and vary h[i] and h[j] while keeping h[i] + h[j] constant. This maintains the sum constraint.

For a pair (i, j) with h[i] + h[j] = s (constant), let h[i] = t and h[j] = s - t for t in [max(0, s-1), min(1, s)].

Each score_r becomes a quadratic in t (since it's bilinear in h, and h[i]=t, h[j]=s-t):
score_r(t) = a_r * t + b_r * (s-t) + c_r * t * (s-t) + d_r
           = (a_r - b_r) * t - c_r * t^2 + c_r * s * t + (b_r * s + d_r)
           = -c_r * t^2 + (a_r - b_r + c_r * s) * t + (b_r * s + d_r)

So each score is a concave quadratic in t. The max of concave quadratics is not necessarily concave, but it is continuous. We can find its minimum by scanning.

For a binary h with n/2 ones:

import numpy as np
from scipy.optimize import minimize
import time

_STATE = {}

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        r = mid + k
        G[r, k:] += 1.0 - h[:m]
        G[r, :m] -= h[k:]
        r2 = mid - k
        G[r2, k:] += 1.0 - h[:m]
        G[r2, :m] -= h[k:]
    return G

def _lse_obj(x):
    tau = _STATE['tau']
    dx = _STATE['dx']
    scores = _raw_scores(x) * dx
    m = float(np.max(scores))
    u = (scores - m) / tau
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        mask = scores >= m - 1e-15
        cnt = max(int(np.sum(mask)), 1)
        return m + tau * np.log(float(cnt))
    return m + tau * np.log(zsum)

def _lse_grad(x):
    tau = _STATE['tau']
    dx = _STATE['dx']
    scores = _raw_scores(x) * dx
    m = float(np.max(scores))
    u = (scores - m) / tau
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        mask = scores >= m - 1e-15
        cnt = max(int(np.sum(mask)), 1)
        w = np.where(mask, 1.0 / cnt, 0.0)
    else:
        w = z / zsum
    G = _grad_matrix(x)
    return dx * (G.T @ w)

def _eq_con(x):
    return float(np.sum(x)) - _STATE['target']

def _eq_jac(x):
    return np.ones(len(x))

def _lse_anneal(h, tau_schedule, deadline, maxiter=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    _STATE['dx'] = dx
    _STATE['target'] = target
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    x = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
    
    for tau in tau_schedule:
        if time.time() > deadline:
            break
        _STATE['tau'] = tau
        try:
            res = minimize(_lse_obj, x, method='SLSQP', jac=_lse_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                x_new = _project_sum(res.x, target)
                c5_new = _exact_c5(x_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = x_new.copy()
                x = x_new
        except Exception:
            pass
    
    return best_h, best_c5
