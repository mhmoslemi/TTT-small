        w = vp / (float(np.sum(vp)) + 1e-30)
        g = _weighted_grad(h, w)
        blk = np.zeros(n)
        for b in range(nb):
            sl = slice(b * bl, min((b + 1) * bl, n))
            blk[sl] = np.mean(g[sl])
        db = -blk.copy()
        db -= np.mean(db)
        nm = np.linalg.norm(db)
        if nm > 1e-12:
            directions.append(db / nm)
    idx_pos = np.argsort(-h)[
:4]
    idx_neg = np.argsort(h)[:4]
    for ip in idx_pos[:3]:
        for ino in idx_neg[:3]:
            if ip == ino:
                continue
            dp = np.zeros(n)
            dp[ip] = -1.0
            dp[ino] = 1.0
            dp -= np.mean(dp)
            nm = np.linalg.norm(dp)
            if nm > 1e-12:
                directions.append(dp / nm)
    for _ in range(5):
        dr = rng.standard_normal(n)
        dr -= np.mean(dr)
        nm = np.linalg.norm(dr)
        if nm > 1e-12:
            directions.append(dr / nm)
    return directions[:14]

def _optimize(h, deadline, beta_list=None, radius=0.25, stall_limit=150):
    if beta_list is None:
        beta_list = [8.0, 20.0, 50.0]
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5, raw = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(777)
    rad = radius
    while time.time() < deadline:
        if stall > stall_limit:
            break
        directions = _make_dirs(h, raw, n, rng, rad, beta_list)
        accepted = False
        for d in directions:
            alpha, new_c5 = _line_search(h, d, raw, dx, rad, n_top=20)
            if alpha < 1e-10:
                continue
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual, raw_new = _exact_c5(h_new)
            if actual < cur_c5 - 1e-13:
                h = h_new
                raw = raw_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                rad = min(rad * 1.03, 0.45)
                accepted = True
                break
        if not accepted:
            stall += 1
            rad *= 0.7
            if stall % 40 == 0 and rad < 0.06:
                rad = 0.22
        if time.time() > deadline:
            break
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5, _ = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5, _ = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
            T *= 0.9992
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5, _ = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try, _ = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _perturb(h, scale, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= scale
    p += scale * 0.1 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def _pair_swap_opt(h, deadline, n_pairs=5000, radius=0.3, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5, raw = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, raw, dx, radius, n_top=10)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual, raw_new = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            raw = raw_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _coordinate_descent(h, deadline, n_sweeps=20):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5, raw = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n - 1):
            if time.time() > deadline:
                break
            j = i + 1
            s = h[i] + h[j]
            if s <= 0 or s >= 2:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, _ = _line_search(h, d, raw, dx, 0.5, n_top=12)
            if alpha > 1e-10:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual, raw_new = _exact_c5(h_new)
                if actual < cur_c5 - 1e-13:
                    h = h_new
                    raw = raw_new
                    cur_c5 = actual
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)[0]

    # Phase 1: Initial gradient optimization (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _optimize(h_init, d1, beta_list=[10.0, 25.0, 60.0], radius=0.25, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary refinement (12%)
    d2 = t0 + budget_s * 0.30
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=45000, T_init=0.003)
    sub_dl = min(d2, time.time() + budget_s * 0.035)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 3:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_pol, c5_p = _optimize(h_pol, sub_dl, beta_list=[12.0, 30.0], radius=0.15, stall_limit=50)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (30%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in range(-36, 37, 2):
        if time.time() > d3 - 8:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.014)
        h_r, c5_r = _optimize(h_r, sub_dl, beta_list=[10.0, 25.0, 50.0], radius=0.22, stall_limit=45)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 4:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.007)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.002 and time.time() < d3 - 2:
            sub_dl3 = min(d3, time.time() + budget_s * 0.007)
            h_r, c5_r = _optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, beta_list=[20.0, 50.0], radius=0.10, stall_limit=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _optimize(h4, sub_dl, beta_list=[8.0, 20.0, 50.0, 100.0], radius=0.18, stall_limit=180)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.07)
        h4, c5_4 = _pair_swap_opt(h4, sub_dl, n_pairs=10000, radius=0.25, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _coordinate_descent(h4, sub_dl, n_sweeps=15)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=30000, T_init=0.0012)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _optimize(h4, sub_dl, beta_list=[30.0, 80.0], radius=0.08, stall_limit=80)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 120:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d5 - 4, time.time() + budget_s * 0.005)
        h_t, c5_t = _optimize(h_pert, sub_dl, beta_list=[15.0, 40.0], radius=0.12, stall_limit=35)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.80)
        else:
            scale = min(0.09, scale * 1.1)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _optimize(h6, deadline - 2, beta_list=[20.0, 50.0, 100.0], radius=0.05, stall_limit=60)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_opt(h6, deadline - 1, n_pairs=2000, radius=0.1, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)[0]
    return best_h, final_c5, len(best_h)
