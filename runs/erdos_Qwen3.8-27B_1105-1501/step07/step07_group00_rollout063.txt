    cur_raw = np.correlate(b, 1.0 - b, mode='full')
    cur_max = float(np.max(cur_raw))
    best_b = b.copy()
    best_max = cur_max
    dx = 2.0 / n
    ones = np.where(b > 0.5)[0].tolist()
    zeros = np.where(b <= 0.5)[0].tolist()
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    batch_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 8000:
            break
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_raw = None
        best_new_max = 0.0
        for _ in range(batch_size):
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            b[i] = 0.0
            b[j] = 1.0
            new_raw = np.correlate(b, 1.0 - b, mode='full')
            new_max = float(np.max(new_raw))
            b[i] = 1.0
            b[j] = 0.0
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j
                best_new_raw = new_raw
                best_new_max = new_max
        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            cur_raw = best_new_raw
            cur_max = best_new_max
            ii = ones.index(best_i)
            jj = zeros.index(best_j)
            ones[ii] = best_j
            zeros[jj] = best_i
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
                stall = 0
                batch_no_improve = 0
                T = max(T * 0.97, 0.002)
            else:
                stall += 1
                batch_no_improve += 1
        else:
            batch_no_improve += 1
            if batch_no_improve > 30:
                i = ones[int(rng.integers(len(ones)))]
                j = zeros[int(rng.integers(len(zeros)))]
                b[i] = 0.0
                b[j] = 1.0
                new_raw = np.correlate(b, 1.0 - b, mode='full')
                new_max = float(np.max(new_raw))
                delta = new_max - cur_max
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    cur_raw = new_raw
                    cur_max = new_max
                    ii = ones.index(i)
                    jj = zeros.index(j)
                    ones[ii] = j
                    zeros[jj] = i
                    if cur_max < best_max:
                        best_max = cur_max
                        best_b = b.copy()
                        stall = 0
                    batch_no_improve = 0
                    T = min(T * 1.005, 0.15)
                else:
                    b[i] = 1.0
                    b[j] = 0.0
                    stall += 1
            else:
                stall += 1
    return best_b, best_max * dx

def _focused_binary_optimize(b, deadline, seed=777):
    """Focus on reducing the top lags specifically."""
    n = len(b)
    b = b.copy().astype(float)
    cur_raw = np.correlate(b, 1.0 - b, mode='full')
    cur_max = float(np.max(cur_raw))
    best_b = b.copy()
    best_max = cur_max
    dx = 2.0 / n
    ones = np.where(b > 0.5)[0].tolist()
    zeros = np.where(b <= 0.5)[0].tolist()
    rng = np.random.default_rng(seed)
    n_top = 8
    for it in range(200000):
        if time.time() > deadline:
            break
        top_idx = np.argsort(cur_raw)[-n_top:]
        top_vals = cur_raw[top_idx]
        cur_worst = float(np.max(top_vals))
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_raw = None
        best_new_max = 0.0
        attempts = min(400, 200000 - it)
        for _ in range(attempts):
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            b[i] = 0.0
            b[j] = 1.0
            new_raw = np.correlate(b, 1.0 - b, mode='full')
            new_top = new_raw[top_idx]
            new_worst = float(np.max(new_top))
            if new_worst < cur_worst - 1e-15:
                new_max_all = float(np.max(new_raw))
                delta = new_max_all - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
                    best_new_raw = new_raw
                    best_new_max = new_max_all
            b[i] = 1.0
            b[j] = 0.0
        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            cur_raw = best_new_raw
            cur_max = best_new_max
            ii = ones.index(best_i)
            jj = zeros.index(best_j)
            ones[ii] = best_j
            zeros[jj] = best_i
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        else:
            i = ones[int(rng.integers(len(ones)))]
            j = zeros[int(rng.integers(len(zeros)))]
            b[i] = 0.0
            b[j] = 1.0
            new_raw = np.correlate(b, 1.0 - b, mode='full')
            new_max = float(np.max(new_raw))
            if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / 0.02):
                cur_raw = new_raw
                cur_max = new_max
                ii = ones.index(i)
                jj = zeros.index(j)
                ones[ii] = j
                zeros[jj] = i
                if cur_max < best_max:
                    best_max = cur_max
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
    return best_b, best_max * dx

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _active_lp(h, scores, radius, n_active=10, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_dir(h, scores, beta=15.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -40.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g -= float(np.mean(g))
    return g

def _local_polish(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_act = 10
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores_raw(h) * dx
        d = _active_lp(h, scores, radius, n_active=n_act, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_dir(h, scores, beta=15.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.4
                if radius < 1e-7:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
            gap = max(gap * 0.92, 0.001)
        else:
            stall += 1
            radius *= 0.75
            gap *= 1.3
            n_act = min(n_act + 2, 20)
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=150):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    rng = np.random.default_rng(555)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(max(1, len(ones) // 4)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi = zeros[i]
            oj = ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.2)
            if c_try < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amp
    perturb += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Binary optimization from parent (35% time)
    d1 = t0 + budget_s * 0.35
    b0 = _to_binary(h_init)
    b_best, c5_bin = _binary_optimize(b0, d1, T_init=0.04, batch_size=200, seed=42)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = b_best.copy()

    # Phase 2: Focused binary optimization (15% time)
    d2 = t0 + budget_s * 0.50
    b_foc, c5_foc = _focused_binary_optimize(b_best, d2, seed=777)
    if c5_foc < best_c5:
        best_c5 = c5_foc
        best_h = b_foc.copy()

    # Phase 3: Multi-n exploration (20% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    n_tried = {n_curr}
    offset_list = [-8, -4, 4, 8, 12, 16, 20, 24, -12]
    for dn in offset_list:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 120 or n_try > 260 or n_try % 2 != 0 or n_try in n_tried:
            continue
        n_tried.add(n_try)
        b_res = _resample_binary(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        b_t, c5_t = _binary_optimize(b_res, sub_dl, T_init=0.03, batch_size=120, seed=n_try)
        if c5_t < best_c5 and time.time() < d3 - 2:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            b_t2, c5_t2 = _focused_binary_optimize(b_t, sub_dl2, seed=n_try + 1)
            if c5_t2 < c5_t:
                c5_t = c5_t2
                b_t = b_t2
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = b_t.copy()
            n_curr = n_try

    # Phase 4: Fractional polish (20% time)
    d4 = t0 + budget_s * 0.90
    n_f = len(best_h)
    h_frac = _project(best_h.copy(), n_f / 2.0)
    h_pol, c5_pol = _local_polish(h_frac, min(d4, time.time() + budget_s * 0.10), radius_init=0.08, max_iters=2000)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    if time.time() < d4:
        h_ps, c5_ps = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), d4, max_rounds=80)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 5: Basin hopping + final polish (10% time)
    n_restarts = 0
    while time.time() < deadline - 10 and n_restarts < 100:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, 0.015, rng)
        sub_dl = min(deadline - 10, time.time() + budget_s * 0.003)
        h_t, c5_t = _local_polish(h_pert, sub_dl, radius_init=0.05, max_iters=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Final: try binary on the best fractional, then polish
    if deadline - time.time() > 10:
        b_fin = _to_binary(best_h)
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.03)
        b_fin, c5_fin = _binary_optimize(b_fin, sub_dl, T_init=0.015, batch_size=300, seed=999)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = b_fin.copy()

    # Final fractional polish
    if deadline - time.time() > 3:
        h_final = _project(best_h.copy(), len(best_h) / 2.0)
        h_final, c5_final = _local_polish(h_final, deadline - 1, radius_init=0.04, max_iters=1000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
