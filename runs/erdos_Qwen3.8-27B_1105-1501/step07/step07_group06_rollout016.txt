def _compute_effects(h):
    # effect[i][k] = change in c_k when h[i] goes from 0 to 1
    # c_k = sum_n h[n+k] * (1-h[n])
    # dc_k/dh[i] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    n = len(h)
    effects = np.zeros((n, 2*n-1))
    for i in range(n):
        # (1-h[i-k]) for k = i-n+1 to i-1 (where 0 <= i-k < n means k in [i-(n-1), i])
        # In the full correlation output, index = n-1+k, so k = idx - (n-1)
        # (1-h[i-k]) contributes to output index n-1+k where k = i-k_val... 
        # Let me use a different parameterization.
        # For shift tau (where tau = idx - (n-1)):
        # dc_{tau}/dh[i] = (1-h[i-tau])*[0<=i-tau<n] - h[i+tau]*[0<=i+tau<n]
        tau = np.arange(-(n-1), n)  # all possible shifts
        # (1-h[i-tau]) where 0 <= i-tau < n
        idx1 = i - tau
        valid1 = (idx1 >= 0) & (idx1 < n)
        # h[i+tau] where 0 <= i+tau < n
        idx2 = i + tau
        valid2 = (idx2 >= 0) & (idx2 < n)
        eff = np.zeros(2*n-1)
        eff[valid1] += 1.0 - h[idx1[valid1]]
        eff[valid2] -= h[idx2[valid2]]
        effects[i] = eff
    return effects
