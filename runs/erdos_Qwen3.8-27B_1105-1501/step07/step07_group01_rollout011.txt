        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _smoothed_obj_and_grad(h, beta):
    """Returns (smoothed_c5_value, gradient) for the log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    # softmax weights
    e = np.exp(beta * (corr - cmax))
    w = e / e.sum()
    val = (cmax + np.log(e.sum()) / beta) * dx
    
    # Gradient: for each i, grad[i] = sum_d w[n-1+d] * [(1-h[i-d])*I(0<=i-d<n) - h[i+d]*I(0<=i+d<n)] * dx
    # This is O(n^2) but fast for small n
    gh = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[n-1+d]
        if wd < 1e-14: continue
        # (1-h[i-d]) contribution: valid for i in [max(0,d), min(n, n+d))
        lo1, hi1 = max(0, d), min(n, n + d)
        if lo1 < hi1:

        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2) + d  # these are the i+d values
            gh[lo2:hi2] -= wd * h[idx2]

Since we need sum(h) = n/2, we can write h_i = 1/2 + g_i where sum(g) = 0. Then:
h_i(1-h_{i+d}) = (1/2+g_i)(1/2-g_{i+d}) = 1/4 - g_i*g_{i+d} + (1/2)(g_i - g_{i+d})

The sum over i for a given lag d:
corr_d = (n-|d|)/4 - sum_i g_i*g_{i+d} + (1/2)(sum_i g_i - sum_i g_{i+d})

The last term: sum_i g_i = 0 (by constraint), and sum_i g_{i+d} depends on which i are valid.

For d > 0: sum_{i=0}^{n-d-1} g_{i+d} = sum_{j=d}^{n-1} g_j = -sum_{j=0}^{d-1} g_j (since total sum is 0)

So corr_d = (n-d)/4 - sum_{i=0}^{n-d-1} g_i*g_{i+d} - (1/2)sum_{j=0}^{d-1} g_j

Similarly for d < 0 (by symmetry of the problem, we'd get a similar expression).

The key term to control is sum g_i*g_{i+d}, which is like an autocorrelation of g. We want this to be as large as possible (to minimize corr_d) for all d, while respecting |g_i| ≤ 1/2.

This is reminiscent of the "Golay sequence" or "Complementary set" problems.

def _smoothed_val_grad(h, beta):
    """Log-sum-exp smoothed C5 and its gradient. Returns (value, gradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    # Stable log-sum-exp
    s = beta * (corr - cmax)
    s_max = float(s.max())
    e = np.exp(s - s_max)
    log_sum_exp = s_max + float(np.log(e.sum()))
    val = (cmax + log_sum_exp / beta) * dx
    w = e / e.sum()
    # Gradient

    for d in range(-(n - 1), n):

        lo2, hi2 = max(0, -d), min(n, n - d)

def _adam_optimize(h0, deadline, n_iter=50000, lr=0.05, beta_start=1.0, beta_end=15.0,

    target = n * 0.5
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_start
