def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lag_gradient(h, d, dx):
    """Exact gradient of score at lag d with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:d+m] += (1.0 - h[:m])  # contribution from h[j+d]
            g[:m] += -h[d:d+m]          # contribution from (1-h[j])
    else:
        g_ = -d
        m = n - g_
        if m > 0:
            g[:m] += (1.0 - h[g_:g_+m])  # contribution from h[j]
            g[g_:g_+m] += -h[:m]          # contribution from (1-h[j+g_])
    return g * dx

def _lsq_objective_grad(h, beta):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(h)
    scores = _scores(h)
    m = np.max(scores)
    z = (scores - m) / beta
    w = np.exp(z)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    value = m + beta * np.log(max(sw, 1e-300))
    
    # Gradient: weighted sum of lag gradients
    dx = 2.0 / n
    n_lags = len(scores)
    grad = np.zeros(n)
    for k in range(n_lags):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        grad += w[k] * _lag_gradient(h, d, dx)
    return value, grad
