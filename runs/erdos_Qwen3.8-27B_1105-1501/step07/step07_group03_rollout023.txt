def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    idx = np.argsort(h)[-n//2:]

    h_bin[idx] = 1.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    """Fast 1-swap local search for binary h."""
    n = len(h)
    h = h.copy()
    if rng is None:

    def _optimize(h):

            h[best_j] = 1.0

def _subgradient_polish(h, deadline, steps=200, lr=0.01):
    """Simple subgradient method for continuous h."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(456)
    
    for t in range(steps):
        if time.time() > deadline:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        c5 = mx * dx
        
        # Find active lags (within epsilon of max)
        active = raw >= mx - 1e-6 * mx
        
        # Subgradient: for each active lag, compute gradient
        # d(raw[r])/d(h[p]) for the term at lag d = r-(n-1)
        # = (1-h[p+d]) if p is first index + (-h[p-d]) if p is second index (for d≠0)
        g = np.zeros(n)
        for r in np.where(active)[0]:
            d = r - (n - 1)
            for p in range(n):
                # p as first index
                if d != 0 and 0 <= p + d < n:
                    g[p] += (1.0 - h[p+d])
                # p as second index
                if d != 0 and 0 <= p - d < n:
                    g[p] += (-h[p-d])
                # d=0: h[p]*(1-h[p]) derivative = 1-2h[p]
                if d == 0:
                    g[p] += (1.0 - 2.0*h[p])
        
        # Scale learning rate
        lr_t = lr / (1.0 + 0.01 * t)
        
        # Update
        h_new = h - lr_t * g
        h_new = _project(h_new, n / 2.0)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

```python
def _compute_on_effects(h):
    n = len(h)
    m = 2 * n - 1
    effects = np.zeros((n, m))
    # Vectorize using numpy
    # For each p, the effect at lag d (raw index n-1+d):
    # As first index: (1-h[p+d]) for d in [-p, n-1-p], d != 0
    # As second index: -h[p-d] for d in [p-n+1, p], d != 0
    
    # This is hard to fully vectorize due to the variable range, but we can use a trick:
    # Create a full (n, m) grid and use masks
    
    p = np.arange(n)[:, None]  # shape (n, 1)
    d = np.arange(-(n-1), n)  # shape (2n-1,)
    # d_idx[r] = r - (n-1) gives the lag for raw index r
    # For raw index r, lag d = r - (n+1) ... wait let me re-derive
    
    # raw[r] corresponds to lag d = r - (n-1)
    # So for raw index r, the lag is d = r - (n-1)
    # r ranges from 0 to 2n-2, d ranges from -(n-1) to n-1
    
    r = np.arange(m)[None, :]  # shape (1, m)
    d_grid = (r - (n-1))  # shape (1, m), the lag for each raw index
    
    # First index condition: 0 <= p+d < n, d != 0
    # p is shape (n,1), d_grid is shape (1,m)
    pd = p + d_grid  # shape (n, m), p+d for each (p, r)
    mask1 = (pd >= 0) & (pd < n) & (d_grid != 0)
    # effects[p, r] += (1-h[p+d]) where mask1 is True
    # h[p+d] where pd is in [0,n): use pd as index
    # But pd can be out of range, so we need to handle that
    pd_clipped = np.clip(pd, 0, n-1)
    first_contrib = (1.0 - h[pd_clipped]) * mask1
    effects += first_contrib
    
    # Second index condition: 0 <= p-d < n, d != 0
    pmd = p - d_grid  # shape (n, m), p-d for each (p, r)
    mask2 = (pmd >= 0) & (pmd < n) & (d_grid != 0)
    pmd_clipped = np.clip(pmd, 0, n-1)
    second_contrib = (-h[pmd_clipped]) * mask2
    effects += second_contrib
    
    return effects
```

This is O(n*m) = O(n^2) but fully vectorized in numpy. For n=180, m=359, that's 64K operations per call, which should be very fast (~0.001s).
