def _exact_linesearch(h, d, radius=0.2):
    """Exact line search in direction d (sum(d)=0). Returns (alpha, c5_value)."""
    n = len(h)
    dx = 2.0 / n
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_lo
    hi = a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(60):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_a = (lo + hi) / 2.0
    opt_v = f(opt_a)
    # Check endpoints and 0
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_v = ft
            opt_a = t
    return opt_a, opt_v

def _fullspace_refine(h, deadline, beta=20.0, radius=0.15, max_iters=500):
    """Gradient-based refinement in full space with sum constraint."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        mx = float(np.max(raw))
        z = beta * (raw - mx)
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        wsum = float(np.sum(w))
        w /= wsum
        
        G = np.zeros(n)
        for r in range(len(raw)):
            wr = w[r]
            if wr < 1e-10:
                continue
            tau = r - (n - 1)
            if tau >= 0:
                m2 = n - tau
                G[tau:] += wr * (1.0 - h[:m2])
                G[:m2] += -wr * h[tau:tau + m2]
            else:
                d = -tau
                m2 = n - d
                G[:m2] += wr * (1.0 - h[d:d + m2])
                G[d:] += -wr * h[:m2]
        G -= np.mean(G)
        
        # Try negative gradient direction and a few perturbations
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        
        dirs = [G / (np.linalg.norm(G) + 1e-15)]
        # Add a random perturbation
        rp = rng.normal(0, 1, n)
        rp -= np.mean(rp)
        dirs.append(-rp / (np.linalg.norm(rp) + 1e-15))
        # Try the steepest descent (negative gradient)
        dirs.insert(0, -G / (np.linalg.norm(G) + 1e-15))
        
        for d in dirs:
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-12:
                continue
            d /= dn
            alpha, val = _exact_linesearch(h, d, radius)
            if val < best_val - 1e-13:
                best_val = val
                best_alpha = alpha
                best_d = d
        
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

            opt_v = ft
            opt_a = t
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_v = fv
                    opt_a = av
    return opt_a, opt_v

def _fullspace_refine(h, deadline, beta=30.0, radius=0.12, max_iters=300, seed=99):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        mx = float(np.max(raw))
        z = beta * (raw - mx)
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        wsum = float(np.sum(w))
        w /= wsum
        G = np.zeros(n)
        active = w > 1e-10
        for r in np.where(active)[0]:
            wr = w[r]
            tau = r - (n - 1)
            if tau >= 0:
                m2 = n - tau
                G[tau:] += wr * (1.0 - h[:m2])
                G[:m2] -= wr * h[tau:tau + m2]
            else:
                d = -tau
                m2 = n - d
                G[:m2] += wr * (1.0 - h[d:d + m2])
                G[d:] -= wr * h[:m2]
        G -= np.mean(G)
        gnorm = np.linalg.norm(G)
        if gnorm < 1e-14:
            break
        dirs = [-G / gnorm]
        rp = rng.normal(0, 1, n)
        rp -= np.mean(rp)
        rpn = np.linalg.norm(rp)
        if rpn > 1e-12:
            dirs.append(-rp / rpn)
        idx_s = np.argsort(G)
        s = idx_s[:3]
        k = idx_s[-3:]
        for si in s[:2]:
            for ki in k[:2]:
                if si != ki:
                    dd = np.zeros(n)
                    dd[si] = 1.0
                    dd[ki] = -1.0
                    dd -= np.mean(dd)
                    dirs.append(dd)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-12:
                continue
            d /= dn
            alpha, val = _exact_linesearch(h, d, radius)
            if val < best_val - 1e-13:
                best_val = val
                best_alpha = alpha
                best_d = d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_greedy(h_bin, deadline, max_sweeps=50):
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h <= 0.5)[0].tolist()
        if not ones or not zeros:
            break
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h[oi] = 0.0
                h[oz] = 1.0
                c_try = _exact_c5(h)
                if c_try < cur_c5 - 1e-13:
                    cur_c5 = c_try
                    improved = True
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h.copy()
                        zeros = [z for z in zeros if z != oz]
                        ones.append(oz)
                        ones.remove(oi)
                    break
                else:
                    h[oi] = 1.0
                    h[oz] = 0.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.25 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _coarse_search(h_start, n, deadline, beta_list=(20.0, 60.0, 150.0)):
    h = _project(h_start, n / 2.0)
    x = _antipal_from_h(h)
    for beta in beta_list:
        if time.time() > deadline:
            break
        x = _bfgs_anti(x, beta, max_iter=150)
    h = _h_from_antipal(x)
    h = _project(h, n / 2.0)
    if time.time() < deadline - 3:
        h, _ = _fullspace_refine(h, deadline, beta=50.0, radius=0.08, max_iters=60)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Anti-palindromic BFGS with increasing beta (25%)
    d1 = t0 + budget_s * 0.25
    x_init = _antipal_from_h(h_init)
    for beta in [6.0, 15.0, 40.0, 100.0, 250.0, 500.0]:
        if time.time() > d1 - 5:
            break
        x_init = _bfgs_anti(x_init, beta, max_iter=250)
        h_try = _h_from_antipal(x_init)
        h_try = _project(h_try, n0 / 2.0)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: Full-space gradient refinement (15%)
    d2 = t0 + budget_s * 0.40
    h2 = best_h.copy()
    h2, c5_2 = _fullspace_refine(h2, d2, beta=30.0, radius=0.12, max_iters=250, seed=55)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    n_list = sorted(set([n_curr + dn for dn in range(-24, 25, 4) if n_curr + dn >= 60 and n_curr + dn <= 220 and (n_curr + dn) % 2 == 0]))
    for n_try in n_list:
        if time.time() > d3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3 - 2, time.time() + budget_s * 0.018)
        h_r = _coarse_search(h_r, n_try, sub_dl, beta_list=(30.0, 80.0, 200.0))
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Binary local search on best (12%)
    d4 = t0 + budget_s * 0.72
    h_b = _to_binary(best_h)
    h_b, c5_b = _binary_greedy(h_b, d4, max_sweeps=80)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d4 - 3:
        h_pol = _project(h_b.copy(), len(h_b) / 2.0)
        h_pol, c5_pol = _fullspace_refine(h_pol, d4, beta=60.0, radius=0.06, max_iters=150, seed=77)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: High-beta anti-palindromic refinement + full space (15%)
    d5 = t0 + budget_s * 0.87
    x5 = _antipal_from_h(best_h)
    for beta in [100.0, 300.0, 800.0]:
        if time.time() > d5 - 5:
            break
        x5 = _bfgs_anti(x5, beta, max_iter=200)
        h5 = _h_from_antipal(x5)
        h5 = _project(h5, len(h5) / 2.0)
        c5_5 = _exact_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    if time.time() < d5 - 3:
        h5b = best_h.copy()
        h5b, c5_5b = _fullspace_refine(h5b, d5, beta=80.0, radius=0.05, max_iters=200, seed=33)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Basin hopping (12%)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 8 and n_restarts < 50:
        n_restarts += 1
        h_p = _perturb(best_h, scale, rng)
        sub_dl = min(d6 - 4, time.time() + budget_s * 0.012)
        h_t, c5_t = _fullspace_refine(h_p, sub_dl, beta=50.0, radius=0.08, max_iters=80, seed=100+n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.07, scale * 1.06)

    # Final: try nearby n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d6 - 12:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 220 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.012)
        h_r = _coarse_search(h_r, n_try, sub_dl, beta_list=(60.0, 200.0))
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 5:
        h_f = best_h.copy()
        h_f, c5_f = _fullspace_refine(h_f, deadline, beta=100.0, radius=0.04, max_iters=150, seed=42)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
