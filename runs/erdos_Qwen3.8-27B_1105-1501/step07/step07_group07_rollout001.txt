        f2 = float(np.max(R + c2 * T - c2 * c2 * Q) * dx)

            if hi_b - lo_b < 1e-14:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(R + c1 * T - c1 * c1 * Q) * dx)
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(R + c2 * T - c2 * c2 * Q) * dx)
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(R + opt_t * T - opt_t * opt_t * Q) * dx)
    
    # Check endpoints
    candidates = [(opt_t, opt_val),
                  (0.0, float(np.max(R) * dx)),
                  (t_hi, float(np.max(R + t_hi * T - t_hi * t_hi * Q) * dx))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _full_lp_loop(h, deadline, radius_init=0.15, max_radius=0.3, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        R = _raw_scores(h)
        G = _grad_matrix(h)
        p = _full_epi_lp(h, R, G, radius)
        if p is None:
            stall += 2
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, R, G)
        
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, max_radius)
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=20):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _coord_pair_loop(h, deadline, n_tries=5000, seed=111):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        # Pick two indices, one with high h and one with low h
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 0.05:
            stall += 1
            continue
        # Direction: move mass from higher to lower
        if h[i] > h[j]:
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        
        R = _raw_scores(h)
        G = _grad_matrix(h)
        alpha, new_c5 = _line_search(h, d, R, G)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * d
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
    
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 -= 1
        h0 = h0[:n0]
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Try symmetries
    candidates = [h_init]
    h_flip = h_init[::-1].copy()
    candidates.append(_project_sum(h_flip, n0 / 2.0))
    h_comp = 1.0 - h_init
    candidates.append(_project_sum(h_comp, n0 / 2.0))
    h_flip_comp = (1.0 - h_init[::-1]).copy()
    candidates.append(_project_sum(h_flip_comp, n0 / 2.0))
    h_avg = 0.5 * (h_init + h_init[::-1])
    candidates.append(_project_sum(h_avg, n0 / 2.0))
    
    for cand in candidates:
        c = _exact_c5(cand)
        if c < best_c5:
            best_c5 = c
            best_h = cand.copy()
    
    # Phase 1: Full epigraph LP (20% of budget)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _full_lp_loop(best_h, d1, radius_init=0.15, max_radius=0.30, stall_limit=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary SA + pair sweep (10%)
    d2 = t0 + budget_s * 0.30
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=31337)
    sub_dl2 = min(d2, time.time() + budget_s * 0.03)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=20)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish
    if time.time() < d2:
        h_pol = _project_sum(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl3 = min(d2, time.time() + budget_s * 0.02)
        h_pol, c5_pol = _full_lp_loop(h_pol, sub_dl3, radius_init=0.08, max_radius=0.15, stall_limit=30)
        if c5_pol < best_c5:
            best_c5 = c5_pol
