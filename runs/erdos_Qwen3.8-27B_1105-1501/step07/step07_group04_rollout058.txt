        g[:n-k] -= h[k:]
    elif k < 0:
        u = -k
        # F_{-u} = sum_{i=u}^{n-1} h[i-u]*(1-h[i])
        # dF/dh[j] = (1-h[j+u]) if j+u < n, minus h[j-u] if j >= u
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    else:
        g = 1.0 - 2.0 * h
    return g

def _build_lp(h, active_lags, radius):
    # Build and solve the LP for the minimax direction
    n = len(h)
    m = len(active_lags)
    # Variables: d[0..n-1], t
    # Minimize: t (plus small regularization on d)
    # Constraints:
    #   For each active lag r: grad_r · d - t <= -F_r(h)  [i.e., F_r + grad_r·d - t <= 0]
    #   sum(d) = 0
    #   Bounds on d: -min(radius, h[i]) <= d[i] <= min(radius, 1-h[i])
    #   t is free (but we can bound it)
    
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    
    # Build constraint matrix
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, lag in enumerate(active_lags):
        p = center + lag  # index in raw_corr
        g = _corr_grad_at_lag(h, lag)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -raw_corr[p]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t + eps * ||d||_1 (approximated by t for now)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    # Small regularization to prefer smaller moves
    c_obj[:n] = 1e-6
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(radius, h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[n]

def _line_search_quadratic(h, d, dx, t_max):
    # Line search using quadratic expansion
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    # Feasible alpha range
    alpha_max = t_max
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
    
    if alpha_max < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search + golden section
    n_grid = 40
    grid = np.linspace(0, alpha_max, n_grid)
    vals = np.array([np.max(C0 + a * L1 - a * a * Q2) * dx for a in grid])
    best_idx = np.argmin(vals)
    
    # Refine with golden section
    if best_idx == 0:
        return 0.0, float(vals[0])
    if best_idx == n_grid - 1:
        return alpha_max, float(vals[-1])
    
    lo = grid[best_idx - 1]
    hi = grid[best_idx + 1]
    gr = (np.sqrt(5) - 1) / 2
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(C0 + c1 * L1 - c1 * c1 * Q2) * dx
    f2 = np.max(C0 + c2 * L1 - c2 * c2 * Q2) * dx
    
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(C0 + c1 * L1 - c1 * c1 * Q2) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(C0 + c2 * L1 - c2 * c2 * Q2) * dx
    
    alpha = (lo + hi) / 2
    val = np.max(C0 + alpha * L1 - alpha * alpha * Q2) * dx
    return alpha, float(val)

def _lp_minimax_loop(h, deadline, radius_init=0.1, max_iter=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
            
        raw_corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = np.max(raw_corr)
        
        # Select active lags: those within 2.0 of the max (in raw units)
        # Plus always include the argmax
        threshold = cur_max - 3.0
        active_mask = raw_corr >= threshold
        active_idx = np.where(active_mask)[0]
        
        # Limit to top 60 for LP speed
        if len(active_idx) > 60:
            top_vals = raw_corr[active_idx]
            top_order = np.argsort(-top_vals)[:60]
            active_idx = active_idx[top_order]
        
        active_lags = active_idx - (n - 1)
        
        # Solve LP
        d, t_pred = _build_lp(h, active_lags, radius)
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        # Line search
        alpha, new_c5 = _line_search_quadratic(h, d, dx, 1.0)
        
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
    
    return best_h, best_c5
