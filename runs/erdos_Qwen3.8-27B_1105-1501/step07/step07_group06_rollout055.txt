        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    idx_s = np.argsort(g)
    for s in idx_s[:2]:
        for k in idx_s[-2:]:
            if s != k:
                d = np.zeros(n)
                d[s] = 1.0
                d[k] = -1.0
                d -= d.mean()
                dn = np.linalg.norm(d)
                if dn > 1e-12:
                    dirs.append(d / dn)
                break
    return dirs[:n_max]

def _optimize_full(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.18, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(12345)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = min(1.0, float(total_iters) / 100.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        raw = _raw_corr(h)
        g = _smooth_grad_full(h, beta)
        dirs = _make_dirs(h, g, raw, rng, 8)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search_quad(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.04, radius * 0.72)
        else:
            stall += 1
            radius = max(0.04, radius * 0.75)
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500 or len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9993
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for oi in ones:
        if time.time() > deadline:
            break
        improved = False
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                improved = True
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
        if not improved:
            continue
    return best_h, best_c5

def _thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = 1.0 - t[i >> 1]
    return t

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=5000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search_quad(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Anti-palindromic L-BFGS-B optimization (25%)
    d1 = t0 + budget_s * 0.25
    m0 = n0 // 2
    x0 = _anti_from_h(h_init)
    for beta in [10.0, 25.0, 50.0]:
        if time.time() > d1 - 3:
            break
        x_opt = _lbfgs_anti(x0, beta, maxiter=80)
        h_anti = _anti_map(x_opt)
        c5_anti = _exact_c5(h_anti)
        if c5_anti < best_c5:
            best_c5 = c5_anti
            best_h = h_anti.copy()
            x0 = _anti_from_h(h_anti)

    # Also try Thue-Morse starting point
    if time.time() < d1 - 5:
        h_tm = _thue_morse(n0)
        c5_tm = _exact_c5(h_tm)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
        x_tm = _anti_from_h(h_tm)
        x_opt = _lbfgs_anti(x_tm, 25.0, maxiter=60)
        h_anti2 = _anti_map(x_opt)
        c5_anti2 = _exact_c5(h_anti2)
        if c5_anti2 < best_c5:
            best_c5 = c5_anti2
            best_h = h_anti2.copy()

    # Phase 2: Full-dimensional smooth optimization (15%)
    d2 = t0 + budget_s * 0.40
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    h_f, c5_f = _optimize_full(h_f, d2, beta_lo=8.0, beta_hi=50.0, radius_init=0.15, max_stall=60)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Phase 3: Multi-resolution exploration (20%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    for dn in [-20, -10, 10, 20, -30, 30, -40, 40]:
        if time.time() > d3 - 3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260:
            continue
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 80 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        # Anti-palindromic opt on resampled
        x_r = _anti_from_h(h_r)
        x_opt = _lbfgs_anti(x_r, 20.0, maxiter=50)
        h_r2 = _anti_map(x_opt)
        c5_r = _exact_c5(h_r2)
        h_r_final = h_r2 if c5_r < _exact_c5(h_r) else h_r
        # Full opt
        sub_dl2 = min(d3, time.time() + budget_s * 0.015)
        h_r3, c5_r3 = _optimize_full(h_r_final, sub_dl2, beta_lo=10.0, beta_hi=60.0, radius_init=0.12, max_stall=40)
        if c5_r3 < c5_r:
            c5_r = c5_r3
            h_r_final = h_r3
        # Try binary
        if time.time() < d3 - 2:
            h_rb = _to_binary(h_r_final)
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_rb, c5_rb = _binary_sa(h_rb, sub_dl3, max_iter=10000, T_init=0.002)
            if c5_rb < c5_r:
                c5_r = c5_rb
                h_r_final = h_rb
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r_final.copy()
            n_curr = n_try

    # Phase 4: Binary optimization (15%)
    d4 = t0 + budget_s * 0.75
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=60000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d4:
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4:
        h_pol = _project(h_b.copy(), n_b / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _optimize_full(h_pol, sub_dl3, beta_lo=15.0, beta_hi=80.0, radius_init=0.08, max_stall=30)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Intensive continuous (10%)
    d5 = t0 + budget_s * 0.85
    n_c = len(best_h)
    h4 = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h4, c5_4 = _optimize_full(h4, sub_dl, beta_lo=20.0, beta_hi=100.0, radius_init=0.08, max_stall=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d5:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_continuous(h4, sub_dl2, n_iter=4000, radius=0.15, seed=777)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Anti-palindromic refinement
    if time.time() < d5:
        x_c = _anti_from_h(best_h)
        x_opt = _lbfgs_anti(x_c, 50.0, maxiter=60)
        h_anti_f = _anti_map(x_opt)
        c5_af = _exact_c5(h_anti_f)
        if c5_af < best_c5:
            best_c5 = c5_af
            best_h = h_anti_f.copy()

    # Phase 6: Basin hopping + final (10%)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 15 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 - 10, time.time() + budget_s * 0.007)
        h_t, c5_t = _optimize_full(h_p, sub_dl, beta_lo=12.0, beta_hi=60.0, radius_init=0.08, max_stall=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.07, scale * 1.1)

    # Final nearby n values
    n_final = len(best_h)
    for dn in [-6, 6, -12, 12]:
        if time.time() > d6 - 20:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        x_r = _anti_from_h(h_r)
        x_opt = _lbfgs_anti(x_r, 40.0, maxiter=40)
        h_r2 = _anti_map(x_opt)
        c5_r = _exact_c5(h_r2)
        sub_dl = min(d6 - 10, time.time() + budget_s * 0.012)
        h_r3, c5_r3 = _optimize_full(_project(h_r2, n_try / 2.0), sub_dl, beta_lo=15.0, beta_hi=70.0, radius_init=0.08, max_stall=30)
        if c5_r3 < c5_r:
            c5_r = c5_r3
            h_r2 = h_r3
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r2.copy()

    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _optimize_full(h_f, deadline, beta_lo=30.0, beta_hi=100.0, radius_init=0.05, max_stall=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # One more anti-palindromic pass
        x_f = _anti_from_h(best_h)
        x_opt = _lbfgs_anti(x_f, 80.0, maxiter=40)
        h_af = _anti_map(x_opt)
        c5_af = _exact_c5(h_af)
        if c5_af < best_c5:
            best_c5 = c5_af
            best_h = h_af.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
