import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_profile(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w * u[:m]
                g[:m] -= w * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w * u[gp:]
                g[gp:] -= w * h[:m]
    return g

def _line_search(h, d, raw, dx, radius, n_grid=28):
    n = len(h)
    target = n / 2.0
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    pos = d > 1e-12
    neg = d < -1e-12
    a_hi = radius
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    grid = np.linspace(0.0, a_hi, n_grid)
    for alpha in grid:
        h_try = _project(h + alpha * d, target)
        val = _exact_c5(h_try)
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * d, target))
        f2 = _exact_c5(_project(h + c2 * d, target))
        for _ in range(45):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * d, target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * d, target))
        opt = (lo + hi) / 2.0
        val_opt = _exact_c5(_project(h + opt * d, target))
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    return best_alpha, best_val

def _lp_direction(h, raw, n_active=5, trust=0.12):
    n = len(h)
    top_k = n_active + 2
    idx = np.argsort(-raw)[:top_k]
    mx = raw[idx[0]]
    active = idx[raw[idx] >= mx - 0.08 * mx]
    if len(active) == 0:
        return None
    active = active[:n_active]
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for ai, r in enumerate(active):
        d_shift = r - (n - 1)
        A_ub[ai, 0] = -1.0
        for j in range(n):
            coeff = 0.0
            i1 = j - d_shift
            if 0 <= i1 < n and max(0, -d_shift) <= i1 < min(n, n - d_shift):
                coeff += (1.0 - h[i1])
            if max(0, -d_shift) <= j < min(n, n - d_shift):
                j2 = j + d_shift
                if 0 <= j2 < n:
                    coeff -= h[j2]
            A_ub[ai, j + 1] = coeff
        b_ub[ai] = -raw[r]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    bounds = [(-trust, 0.0)] + [(-trust, trust)] * n
    try:
        from scipy.optimize import linprog
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=bounds, method='highs')
        if result.status != 0:
            return None
        d = result.x[1:]
        if result.x[0] >= -1e-10:
            return None
        return d
    except Exception:
        return None

def _make_dirs(h, g, n, raw, radius, use_lp):
    dirs = []
    d0 = -g.copy()
    d0 -= np.mean(d0)
    nm = np.linalg.norm(d0)
    if nm > 1e-12:
        dirs.append(d0 / nm)
    # Top-2 lag gradient
    top2 = np.argsort(-raw)[:2]
    raw2 = raw[top2]
    w2 = raw2 / raw2.sum() if raw2.sum() > 0 else np.ones(2) / 2
    g2 = np.zeros(n)
    for wi, ri in zip(w2, top2):
        g2 += wi * _smooth_grad(h, np.zeros(2*n-1)) * 0  # placeholder
    # Use weighted gradient of top-2
    w_top = np.zeros(2*n-1)
    for ri in top2:
        w_top[ri] = 1.0
    g2 = _smooth_grad(h, w_top / w_top.sum())
    g2 -= np.mean(g2)
    nm2 = np.linalg.norm(g2)
    if nm2 > 1e-12:
        dirs.append(-g2 / nm2)
    # Block directions
    for bl in [6, 12]:
        nb = n // bl
        if nb < 2:
            continue
        blk = np.zeros(n)
        for b in range(nb):
            sl = slice(b * bl, min((b+1)*bl, n))
            blk[sl] = np.mean(g[sl])
        db = -blk.copy()
        db -= np.mean(db)
        nmb = np.linalg.norm(db)
        if nmb > 1e-12:
            dirs.append(db / nmb)
    if use_lp:
        d_lp = _lp_direction(h, raw, n_active=4, trust=radius * 0.7)
        if d_lp is not None:
            d_lp -= np.mean(d_lp)
            nml = np.linalg.norm(d_lp)
            if nml > 1e-12:
                dirs.append(d_lp / nml)
    return dirs

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

    for dn in [-10, -8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 20, 30, 60, 80]:
        nt = n_curr + dn
        if 80 <= nt <= 360 and nt % 2 == 0 and nt not in tried_ns:
            n_options.append(nt)
    n_options.sort(key=lambda x: abs(x - n_curr))
    for n_try in n_options:
        if time.time() > d3 - 10:
            break
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _main_opt(h_r, sub_dl, beta=18.0, radius=0.18, stall_limit=50, use_lp=True)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.005)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=10000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
        elif c5_r < best_c5 + 0.002 and time.time() < d3 - 3:
            sub_dl3 = min(d3, time.time() + budget_s * 0.006)
            h_r2, c5_r2 = _main_opt(_project(h_r.copy(), n_try/2.0), sub_dl3, beta=15.0, radius=0.10, stall_limit=35, use_lp=True)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                n_curr = n_try
    
    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.83
    n_curr = len(best_h)
    # Gradient refinement with high beta
    sub_dl = min(d4, time.time() + budget_s * 0.07)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _main_opt(h4, sub_dl, beta=28.0, radius=0.14, stall_limit=130, use_lp=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Rigorous local pair search
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _rigorous_local(h4, sub_dl)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary round
    if time.time() < d4 - 4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=30000, T_init=0.0012)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Final gradient pass
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _main_opt(h4, sub_dl, beta=35.0, radius=0.07, stall_limit=80, use_lp=True)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: Basin hopping + final polish (15%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 80:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d5 - 4, time.time() + budget_s * 0.007)
        h_t, c5_t = _main_opt(h_pert, sub_dl, beta=22.0, radius=0.10, stall_limit=35, use_lp=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.07, scale * 1.1)
    
    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _main_opt(h6, min(deadline - 2, time.time() + budget_s * 0.02), beta=40.0, radius=0.04, stall_limit=60, use_lp=True)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
