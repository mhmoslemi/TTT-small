def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    """Generate Thue-Morse sequence of length n as 0/1."""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = cnt % 2
    # Balance: keep n/2 ones
    return _to_binary(tm)

    """Pattern based on modular arithmetic."""
    h = np.zeros(n)
    for i in range(n):

            h[i] = 1.0

        h[to_flip] = 0.0
    elif n_ones < n // 2:
        idx = np.where(h <= 0.5)[0]
        to_flip = idx[:n // 2 - n_ones]
        h[to_flip] = 1.0

                h_blk[idx1[perm]] = 0
            else:
                idx0 = np.where(h_blk <= 0.5)[0]
                perm = rng.permutation(len(idx0))[:n // 2 - n_ones]
                h_blk[idx0[perm]] = 1
        patterns.append(h_blk)
    
    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Quick diverse exploration (15%)
    d1 = t0 + budget_s * 0.15
    n_candidates = [160, 180, 200]
    candidates = []
    
    for n_try in n_candidates:
        if time.time() > d1:
            break
        h_r = _resample(h_init, n_try)
        h_bin = _to_binary(h_r)
        c5_bin = _exact_c5(h_bin)
        candidates.append((c5_bin, h_bin, n_try))
    
    # Also try diverse patterns
    patterns = _initial_patterns(n0, rng)
    for pat in patterns:
        if time.time() > d1:
            break
        h_bin = _to_binary(pat)
        c5_bin = _exact_c5(h_bin)
        candidates.append((c5_bin, h_bin, n0))
    
    # Sort by quality
    candidates.sort(key=lambda x: x[0])
    
    # Phase 2: Intensive binary search on top candidates (45%)
    d2 = t0 + budget_s * 0.60
    top_k = min(4, len(candidates))
    for ci in range(top_k):
        if time.time() > d2:
            break
        _, h_cand, n_cand = candidates[ci]
        sub_dl = min(d2, time.time() + budget_s * 0.06 * (top_k - ci))
        h_b, c5_b = _binary_search_fasta(h_cand, sub_dl, max_stall=3000, T_init=0.004, T_decay=0.9997, seed=100 + ci * 7)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Greedy polish
        if time.time() < d2 - 2:
            sub_dl2 = min(d2, time.time() + budget_s * 0.03)
            h_g, c5_g = _binary_greedy(h_b, sub_dl2)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
    
    # Phase 3: Pair sweep on best (10%)
    d3 = t0 + budget_s * 0.70
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_b, c5_b = _pair_sweep(h_b, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    
    # Phase 4: Continuous polishing (15%)
    d4 = t0 + budget_s * 0.85
    n_c = len(best_h)
    h_pol = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_pol, c5_pol = _opt_smooth(h_pol, sub_dl, beta_lo=8.0, beta_hi=40.0, radius_init=0.12, max_stall=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    if time.time() < d4:
        h_pol = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _opt_smooth(h_pol, sub_dl2, beta_lo=20.0, beta_hi=80.0, radius_init=0.08, max_stall=50)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 4, time.time() + budget_s * 0.01)
        h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=10.0, beta_hi=50.0, radius_init=0.08, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.1)
    
    # Final: try nearby n values
    n_final = len(best_h)
    for dn in [-10, 10, -20, 20]:
        if time.time() > d5 - 12:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_br = _to_binary(h_r)
        sub_dl = min(d5 - 4, time.time() + budget_s * 0.015)
        h_br, c5_r = _binary_search_fasta(h_br, sub_dl, max_stall=1500, T_init=0.003, T_decay=0.9997, seed=999 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_br.copy()
    
    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_smooth(h_f, deadline, beta_lo=25.0, beta_hi=100.0, radius_init=0.05, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
