    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid

    res = target - float(np.sum(x))

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    return np.correlate(h, 1.0 - h, mode='full')

def _binary_flip_delta(h, corr, p, n):
    """Compute exact delta in corr when flipping position p (0->1) in binary h."""
    # h is binary, we flip h[p] from 0 to 1
    # Returns array of deltas for each correlation index
    M = 2 * n - 1
    center = n - 1
    delta = np.zeros(M)
    
    # For positive lag k (index r = center + k), k = 1..n-1:
    # delta_F_r = (1-h[p-k]) * I(k<=p) - h[p+k] * I(p+k<n)
    for k in range(1, n):
        r = center + k
        d = 0.0
        if k <= p:
            d += (1.0 - h[p - k])
        if p + k < n:
            d -= h[p + k]
        delta[r] = d
    
    # For negative lag -u (index r = center - u), u = 1..n-1:
    # delta_F_r = (1-h[p+u]) * I(p+u<n) - h[p-u] * I(u<=p)
    for u in range(1, n):
        r = center - u
        d = 0.0
        if p + u < n:
            d += (1.0 - h[p + u])
        if u <= p:
            d -= h[p - u]
        delta[r] = d
    
    return delta

def _apply_flip(h, corr, p, flip_to):
    """Apply flip at position p and update corr. flip_to = 1 means 0->1, 0 means 1->0."""
    n = len(h)
    if flip_to == 1:
        # 0 -> 1
        delta = _binary_flip_delta(h, corr, p, n)
        corr = corr + delta
        h[p] = 1.0
    else:
        # 1 -> 0
        delta = _binary_flip_delta(h, corr, p, n)
        corr = corr - delta  # reverse the flip
        h[p] = 0.0
    return h, corr

    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    corr = _compute_corr(h)
    cur_max = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_max
    best_corr = corr.copy()
    
    rng = np.random.default_rng(seed)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    T = temp_init
    stall = 0
    best_stall = 0
    c5_base = float(np.max(corr))  # in raw units (before *dx)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 20000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i_ones = int(rng.integers(0, len(ones)))
        i_zeros = int(rng.integers(0, len(zeros)))
        p_one = ones[i_ones]
        p_zero = zeros[i_zeros]
        
        # Compute delta for flipping p_one from 1->0 and p_zero from 0->1
        # Net effect: swap positions
        d1 = _binary_flip_delta(h, corr, p_one, n)  # delta if p_one goes 0->1
        d2 = _binary_flip_delta(h, corr, p_zero, n)  # delta if p_zero goes 0->1
        # We want p_one: 1->0 (so subtract d1) and p_zero: 0->1 (so add d2)
        total_delta = d2 - d1
        
        new_max = float(np.max(corr + total_delta))
        new_c5 = new_max * dx
        delta_c5 = new_c5 - cur_max
        
        if delta_c5 < -1e-14 or rng.random() < np.exp(-delta_c5 / max(T, 1e-15)):
            # Accept
            corr = corr + total_delta
            h[p_one] = 0.0
            h[p_zero] = 1.0
            ones[i_ones] = p_zero
            zeros[i_zeros] = p_one
            cur_max = new_max
            T = max(0.001, T * 0.9998)
            
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                best_corr = corr.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
        else:
            stall += 1
            best_stall += 1
    
    return best_h, best_c5, best_corr

def _binary_flip_delta_vec(h, p, n):
    """Vectorized computation of delta in corr when h[p] flips from 0 to 1."""
    M = 2 * n - 1
    center = n - 1
    delta = np.zeros(M)
    
    # Positive lags k=1..n-1, index r = center+k
    # delta_F_r = (1-h[p-k]) * I(k<=p) - h[p+k] * I(p+k<n)
    k_vals = np.arange(1, n)
    r_pos = center + k_vals
    
    # (1-h[p-k]) for k <= p
    mask_pos = k_vals <= p
    if np.any(mask_pos):
        idx = p - k_vals[mask_pos]
        delta[r_pos[mask_pos]] += (1.0 - h[idx])
    
    # -h[p+k] for p+k < n
    mask_neg = (p + k_vals) < n
    if np.any(mask_neg):
        idx = p + k_vals[mask_neg]
        delta[r_pos[mask_neg]] -= h[idx]
    
    # Negative lags -u, r = center-u
    u_vals = np.arange(1, n)
    r_neg = center - u_vals
    
    # (1-h[p+u]) for p+u < n
    mask_pos2 = (p + u_vals) < n

        idx = p + u_vals[mask_pos2]
        delta[r_neg[mask_pos2]] += (1.0 - h[idx])
    
    # -h[p-u] for u <= p
    mask_neg2 = u_vals <= p
    if np.any(mask_neg2):
        idx = p - u_vals[mask_neg2]
        delta[r_neg[mask_neg2]] -= h[idx]
    
    return delta

def _binary_targeted(h_init, deadline, max_iters=200000, seed=99, top_k=5):
    """Targeted binary search focusing on top lags."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_raw = float(np.max(corr))
    best_h = h.copy()
    best_raw = cur_raw
    rng = np.random.default_rng(seed)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        if not ones or not zeros:
            break
        
        # Find top lags
        top_idx = np.argsort(corr)[-top_k:]
        best_delta = 0
        best_move = None
        
        n_try = min(300, len(ones) * len(zeros))
        for _ in range(n_try):
            io = int(rng.integers(0, len(ones)))
            iz = int(rng.integers(0, len(zeros)))
            po, pz = ones[io], zeros[iz]
            d_o = _flip_delta(h, po, n)
            d_z = _flip_delta(h, pz, n)
            net = d_z - d_o
            new_top = corr[top_idx] + net[top_idx]
            new_max = float(np.max(new_top))
            # Also check global max approximately
            approx_new_max = max(float(np.max(corr + net)), new_max)
            delta = approx_new_max - cur_raw
            if delta < best_delta:
                best_delta = delta
                best_move = (po, pz, io, iz, net)
        
        if best_move is not None and best_delta < -1e-14:
            po, pz, io, iz, net = best_move
            corr = corr + net
            h[po] = 0.0
            h[pz] = 1.0
            ones[io] = pz
            zeros[iz] = po
            cur_raw = float(np.max(corr))
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_raw * dx

def _multi_start_binary(h_base, n, deadline, n_starts=5, seed=100):

def _continuous_refine(h_bin, deadline, max_iter=1000, radius=0.15):
    """Brief continuous refinement of a binary pattern."""
    n = len(h_bin)
    target = n / 2.0
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(12345)
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        active = np.where(corr >= max_val - 1e-10)[0]
        
        # Subgradient
        g = np.zeros(n)
        for m in active:
            k = int(m - (n - 1))
            if k >= 0:
                idx1 = np.arange(k, n)
                g[idx1] += (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                g[idx2] -= h[idx2 + k]
            else:
                ka = -k
                idx1 = np.arange(0, n - ka)
                g[idx1] += (1.0 - h[idx1 + ka])
                idx2 = np.arange(ka, n)
                g[idx2] -= h[idx2 - ka]
        g /= len(active)
        
        d = -g
        d = d - np.mean(d)
        
        # Line search
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        t_hi = radius
        pos = d > 1e-15
        neg = d < -1e-15
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            continue
        
        # Golden section on [0, t_hi]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_b, hi_b = 0.0, t_hi
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(corr + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(corr + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(corr + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(corr + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(corr + opt_t * T - opt_t * opt_t * C)) * dx
        
        if opt_val < cur - 1e-13:
            h = h + opt_t * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
        else:
            radius *= 0.7
            if radius < 1e-6:
                break
    
    return best_h, best_c5
