    target = n / 2.0

import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_slsqp(h_init, deadline, betas=None, maxiter=20):

    target = n / 2.0

            c = (2.0 / n) * corr
            cm = float(c.max())

            return cm + np.log(np.sum(np.exp(sh))) / beta

            corr = np.correlate(x, 1.0 - x, mode='full')
            c = (2.0 / n) * corr
            cm = float(c.max())
            sh = beta * (c - cm)
            es = np.exp(sh)
            w = es / np.sum(es)
            G = _grad_matrix(x)
            return (2.0 / n) * (w @ G)
        
        def eqf(x):
            return float(np.sum(x) - target)
        
        def eqj(x):
            return np.ones(n)
