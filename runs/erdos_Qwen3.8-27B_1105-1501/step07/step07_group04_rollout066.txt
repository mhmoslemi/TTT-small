d has d[i]=1, d[j]=-1. The full correlation of d with itself:

def _golden_min(cost_vals, lo, hi, n_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    f1 = cost_vals(gr * (hi - lo))
    x1 = lo + gr * (hi - lo)
    f2 = cost_vals((1.0 - gr) * (hi - lo))
    x2 = lo + (1.0 - gr) * (hi - lo)

def _gradient_from_lags(h, corr, tau):
    n = len(h)
    center = n - 1
    max_corr = float(corr.max())
    weights = np.exp((corr - max_corr) / tau)
    wsum = float(weights.sum())
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    g = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        k = int(r - center)
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] -= w * h[k:]
        else:
            ka = -k
            g[:n-ka] += w * (1.0 - h[ka:])
            g[ka:] -= w * h[:n-ka]
    return g

def _select_pairs_from_grad(g, h, n, k_inc, k_dec):
    """Select pairs (i,j) where i should increase and j should decrease."""
    order = np.argsort(g)
    dec_indices = list(order[:k_dec])
    inc_indices = list(order[-k_inc:])
    pairs = []
    for di in dec_indices:
        for ii in inc_indices:
            if di != ii:
                pairs.append((int(ii), int(di)))
    return pairs

        neg = d < -1e-15
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.8
            continue
        t_hi = max(0.0, t_hi)
        a_opt, c5_try = _pair_line_search(B, T, Q, 0.0, t_hi, dx)
        if c5_try < cur - 1e-12:
            h_new = h + a_opt * d
            h_new = _project(h_new, target)
            c5_check = _exact_c5(h_new)
            if c5_check < cur - 1e-12:
                h = h_new
                cur = c5_check
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.30)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    if n0 < 10:
        n0 = 180
        x = (np.arange(n0) + 0.5) / n0
        h0 = 0.5 + 0.15 * np.sin(2 * np.pi * x)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Subgradient descent (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _subgradient_descent(h_init, d1, radius_init=0.08)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair-pivot local search (20%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _pair_pivot_search(best_h, d2, max_stall=80, tau_init=0.03, k_pairs=max(10, n0 // 12))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Random pair search for escape (10%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _random_pair_search(best_h, d3, n_trials=5000, seed=111)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary optimization (20%)
    d4 = t0 + budget_s * 0.60
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=80000, seed=200)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sub_dl = min(d4, time.time() + budget_s * 0.10)
    h_bin, c5_bin = _binary_local_search(h_bin, sub_dl, max_iters=80000, seed=300)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 5: Multi-resolution exploration (15%)
    d5 = t0 + budget_s * 0.75
    tried_ns = set([n0])
    for dn in range(-20, 21, 4):
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.008)
        h_r, c5_r = _pair_pivot_search(h_r, sub_dl, max_stall=40, tau_init=0.02, k_pairs=max(8, n_try // 15))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_b = _to_binary(h_r, n_try)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=20000, seed=1000 + n_try)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Basin hopping (10%)
    d6 = t0 + budget_s * 0.85
    n_best = len(best_h)
    scale = 0.04
    for _ in range(100):
        if time.time() > d6 - 3:
            break
        h_pert = _perturb(best_h, scale, rng)
        c5_p = _exact_c5(h_pert)
        if c5_p < best_c5 + 0.01:
            sub_dl = min(d6, time.time() + budget_s * 0.005)
            h_t, c5_t = _pair_pivot_search(h_pert, sub_dl, max_stall=30, tau_init=0.02,
                                           k_pairs=max(8, n_best // 15))
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.1)
        else:
            scale = min(0.08, scale * 1.15)

    # Phase 7: Final intensive pair-pivot (15%)
    d7 = deadline - 5
    n_best = len(best_h)
    h7, c5_7 = _pair_pivot_search(best_h, d7, max_stall=100, tau_init=0.015,
                                   k_pairs=max(12, n_best // 10))
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7:
        h3b, c5_3b = _random_pair_search(best_h, d7, n_trials=3000, seed=999)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Final: subgradient polish
    if time.time() < deadline - 3:
        hp, cp = _subgradient_descent(best_h, deadline - 2, radius_init=0.03, max_iter=2000)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
