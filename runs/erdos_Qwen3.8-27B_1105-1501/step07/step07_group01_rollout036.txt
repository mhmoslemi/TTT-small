```python
def _smooth_grad(h, p_shifted):
    """
    h: array of length n
    p_shifted: array of length 2n-1, where p_shifted[k] = p[k-(n-1)] (probability at shift k-(n-1))
    Returns gradient of smooth max of correlation wrt h.

    one_minus_h = 1.0 - h
    
    for j in range(n):

        s_min = max(-(n-1), -j)  # j+s >= 0
```

```python
def _smooth_gradient(h, p):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*S_k)) w.r.t. h.
    h: length n
    p: length 2n-1, normalized probabilities (p[k] corresponds to S[k])
    Returns: gradient of length n
    """
    n = len(h)
    omh = 1.0 - h
    G = np.zeros(n)
    for j in range(n):
        # First term: sum_m omh[m] * p[j-m+n-1] for valid indices
        m_lo = max(0, j - (n - 1))
        m_hi = n - 1
        if m_hi >= m_lo:
            idx = j - np.arange(m_lo, m_hi + 1) + (n - 1)
            G[j] += np.dot(omh[m_lo:m_hi + 1], p[idx])
        # Second term: sum_m h[m] * p[m-j+n-1] for valid indices
        m_lo2 = 0
        m_hi2 = n - 1
        idx2 = np.arange(m_lo2, m_hi2 + 1) - j + (n - 1)
        G[j] -= np.dot(h, p[idx2])
    return G
```
