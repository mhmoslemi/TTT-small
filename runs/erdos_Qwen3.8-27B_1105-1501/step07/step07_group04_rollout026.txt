```python
def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    m = F.max()
    exps = np.exp((F - m) / tau)
    Z = exps.sum()
    p = exps / Z
    value = m + tau * np.log(Z)
    
    g = np.zeros(n)
    c = n - 1
    # Center term
    g += p[c] * (1.0 - 2.0 * h)
    # Off-center terms
    for d in range(1, n):
        wd = p[c + d]
        wn = p[c - d]
        wp = wd + wn
        g[d:] += wd - wp * h[:n-d]
        g[:n-d] += wn - wp * h[d:]
    g *= dx
    return value, g
```
