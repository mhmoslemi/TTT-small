def _single_lag_grad(h, r):
    """Gradient of the correlation term at index r with respect to h."""
    n = len(h)
    L = r - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        # c[r] = sum_{i=0}^{m-1} h[i+L] * (1-h[i])
        # ∂/∂h[k]:
        #   h[k] appears as h[i+L] when i = k-L (valid if 0 <= k-L <= m-1, i.e., L <= k <= n-1)
        #     contribution: (1 - h[k-L])
        #   h[k] appears as h[i] when i = k (valid if 0 <= k <= m-1, i.e., 0 <= k <= n-1-L)
        #     contribution: -h[k+L]
        g[L:] += (1.0 - h[:m])  # for k = L to n-1: (1-h[k-L])
        g[:m] -= h[L:]  # for k = 0 to n-1-L: -h[k+L]
    else:
        d = -L
        m = n - d
        # c[r] = sum_{i=d}^{n-1} h[i-d] * (1-h[i])
        # ∂/∂h[k]:
        #   h[k] appears as h[i-d] when i = k+d (valid if d <= k+d <= n-1, i.e., 0 <= k <= n-1-d)
        #     contribution: (1 - h[k+d])
        #   h[k] appears as h[i] when i = k (valid if d <= k <= n-1)
        #     contribution: -h[k-d]
        g[:m] += (1.0 - h[d:])  # for k = 0 to n-1-d: (1-h[k+d])
        g[d:] -= h[:m]  # for k = d to n-1: -h[k-d]
    g -= np.mean(g)  # ensure sum = 0 for balanced projection
    return g
```

    d2 = t0 + budget_s * 0.7
    h_cont = _project(h_bin.copy(), len(h_bin) / 2.0)
    h_cont, c5_cont = _single_lag_opt(h_cont, d2, rng)
    if c5_cont < best_c5:

        best_h = h_cont.copy()

    if time.time() < d2:
        h_sm = _project(h_cont.copy(), len(h_cont) / 2.0)
        h_sm, c5_sm = _opt_smooth_fast(h_sm, min(d2, time.time() + 100), rng)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
    
    # Phase 3: Multi-n exploration (25%)
    d3 = deadline
    n_curr = len(best_h)
    for dn in [0, -4, 4, -8, 8, -16, 16]:
        if time.time() > d3 - 30:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_rb = _to_binary(h_r)
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_rb, c5_rb = _binary_search_fast(h_rb, sub_dl, rng)
        if time.time() < d3 - 30:
            h_rc = _project(h_rb.copy(), n_try / 2.0)
            sub_dl2 = min(d3, time.time() + budget_s * 0.03)
            h_rc, c5_rc = _single_lag_opt(h_rc, sub_dl2, rng)
            if c5_rc < c5_rb:
                c5_rb = c5_rc
                h_rb = h_rc
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()
    
    # Final polish
    if deadline - time.time() > 10:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _single_lag_opt(h_f, deadline - 2, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, min(final_c5, best_c5), len(best_h)

```python
def _single_lag_opt(h, deadline, rng, max_stall=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    step = 0.05
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        
        # Find max lag
        raw = np.correlate(h, 1.0 - h, mode='full')
        r_star = int(np.argmax(raw))
        cur_max_raw = raw[r_star]
        
        # Compute gradient of the single lag term
        g = _single_lag_grad(h, r_star)
        
        # Line search
        best_alpha = 0.0
        best_val = cur_max_raw
        for alpha in [step, step*0.5, step*0.25, step*2, step*0.1]:
            h_try = h - alpha * g
            h_try = _project(h_try, target)
            # Quick check: does this reduce the target lag?
            raw_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            val_try = raw_try[r_star]
            if val_try < best_val:
                # Also check full max
                full_max = float(np.max(raw_try)) * dx
                if full_max < best_c5:
                    best_val = val_try
                    best_alpha = alpha
        
        if best_alpha > 0:
            h = h - best_alpha * g
            h = _project(h, target)
            new_c5 = _exact_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                stall = 0
                step = min(step * 1.1, 0.3)
            else:
                stall += 1
                step = max(0.005, step * 0.8)
        else:
            stall += 1
            step = max(0.005, step * 0.85)
            # Try a random perturbation to escape
            if stall % 20 == 0:
                h = h + 0.02 * rng.normal(0, 1, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
    
    return best_h, best_c5

Hmm, this is essentially the same as the smooth-max method with very high beta. The difference is that I'm computing the gradient for a single lag (O(n)) instead of a weighted sum over all lags (O(n²) with the weighted approach).

Actually wait, the smooth-max gradient `_grad_smooth` in the previous code already does this in O(n * n_active) where n_active is the number of active (near-max) lags. With high beta, typically only 1-3 lags are active, so it's effectively O(n). So the single-lag approach isn't fundamentally different.

The real difference would be in the LINE SEARCH. The previous code does a full line search (golden section + quadratic fit) which is expensive. My single-lag approach does a simple downhill check (try a few alpha values and take the best that reduces the full max). This is cheaper per step.

**Idea: Use the structure of the problem more directly.**

The objective is max_k c[k] where c[k] = sum_i h[i] * (1-h[i+k]).

For a fixed pattern of which lags are "active" (near the max), the problem becomes:
minimize max_{k in active} sum_i h[i] * (1-h[i+k])
subject to sum_i h[i] = n/2, 0 ≤ h[i] ≤ 1

This is still non-convex (bilinear), but if I fix the support of h (which positions are "on"), it becomes linear.

**Idea: Alternating optimization**
1. Fix the support (binary pattern), optimize the continuous values
2. Fix the continuous values, optimize the support (which positions to include)

If the support is fixed, the problem is:
minimize max_k sum_{i in S} h[i] * (1-h[i+k]) + sum_{i not in S} 0 * (1-h[i+k])
= maximize... hmm this doesn't simplify nicely.

Actually, if h[i] = 0 for i not in S and h[i] ∈ [0,1] for i in S, then:
c[k] = sum_{i in S, i+k valid} h[i] * (1-h[i+k])

This is bilinear in h. Not easy to optimize directly.

**Alternative idea: Use cvxpy to solve a relaxed version.**

The problem min_h max_k c_k(h) s.t. sum h = n/2, 0 ≤ h ≤ 1 is non-convex. But I can relax it:
- Add a variable t
- Minimize t
- Subject to c_k(h) ≤ t for all k, sum h = n/2, 0 ≤ h ≤ 1

The constraint c_k(h) ≤ t is sum_i h[i]*(1-h[i+k]) ≤ t, which is sum_i h[i] - sum_i h[i]*h[i+k] ≤ t. The term sum_i h[i]*h[i+k] is the "correlation" which is not linear. So this isn't convex.

However, if I introduce auxiliary variables z_i = h[i] * h[i+k] for each k, I get a bilinear constraint. This can be linearized using McCormick envelopes for 0 ≤ h ≤ 1, but that gives a relaxation, not the exact solution.

Actually, the function f(h) = sum_i h[i] * (1-h[i+k]) is NOT DCP in general (it's bilinear). So cvxpy can't directly solve this.

What if I use a "sum of squared" formulation? minimize sum_k c_k(h)² + lambda * max_k c_k(h)? The sum of squares is still non-convex.

I think the most practical approach is still the iterative methods (gradient-based, combinatorial). Let me focus on making those as efficient as possible.

**Final algorithm design:**

1. **Fast binary landscape traversal**: 
   - Start from initial binary
   - Do aggressive greedy + SA with exact evaluation
   - Key: minimize per-iteration overhead
   - Use numpy for the correlation, minimize Python-level branching

2. **Iterative single-lag continuous optimization**:
   - Repeatedly find the max lag, compute its gradient, take a step
   - Use a smart step size (backtracking line search)
   - Periodically check the full max to ensure progress
   - This is O(n) per step + O(n²) per full check

3. **Block/structured moves in binary search**:
   - In addition to single swaps, try "complement flips" (flip a long block)
   - Try "shift" moves (shift the pattern by 1-2 positions and re-balance)

4. **Multi-n with interpolation**:
   - Try n = 176, 180, 184, 188
   - For each, quickly optimize

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_bin, deadline, rng, T_init=0.004, max_iter=200000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_raw = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(cur_raw)) * (2.0 / n)
    best_h = cur.copy()
    best_c5 = cur_c5
    best_raw = cur_raw.copy()
    
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    T = T_init
    stall = 0
    n_ones = len(ones)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            if T < 1e-6:
                break
            T = max(T, 0.002)
            stall = 0
        
        # Random swap
        oi = ones[int(rng.integers(n_ones))]
        oz = zeros[int(rng.integers(n_ones))]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        cur_raw = np.correlate(cur, 1.0 - cur, mode='full')
        new_c5 = float(np.max(cur_raw)) * (2.0 / n)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            # Update ones/zeros
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            n_ones = len(ones)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_raw = cur_raw.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            T *= 0.999
    
    return best_h, best_c5

def _greedy_binary(h_bin, deadline, rng, max_passes=50):
    """Repeatedly find and apply the best improving swap."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        improved = False
        
        # Sample candidate swaps
        n_samples = min(n_ones * n_zeros, 5000)
        for _ in range(n_samples):
            if time.time() > deadline:
                break
            oi = ones[int(rng.integers(n_ones))]
            oz = zeros[int(rng.integers(n_zeros))]
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) * dx
            cur[oi] = 1.0
            cur[oz] = 0.0
            if c_try < cur_c5 - 1e-14:
                cur[oi] = 0.0
                cur[oz] = 1.0
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                improved = True
                ones = np.where(cur > 0.5)[0]
                zeros = np.where(cur <= 0.5)[0]
                n_ones = len(ones)
                n_zeros = len(zeros)
        if not improved:
            break
    
    return best_h, best_c5

def _single_lag_grad(h, r):
    n = len(h)
    L = r - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        g[L:] += (1.0 - h[:m])
        g[:m] -= h[L:]
    else:
        d = -L
        m = n - d
        g[:m] += (1.0 - h[d:])
        g[d:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _smooth_max_grad(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        r = int(np.argmax(raw))
        return _single_lag_grad(h, r)
    w /= wsum
    active = w > 1e-7
    g = np.zeros(n)
    for r in np.where(active)[0]:
        wr = w[r]
        L = r - (n - 1)
        if L >= 0:
            m = n - L
            g[L:] += wr * (1.0 - h[:m])
            g[:m] -= wr * h[L:]
        else:
            d = -L
            m = n - d
            g[:m] += wr * (1.0 - h[d:])
            g[d:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _line_search_c5(h, d, radius, n
    dx = 2.0 / n
    # Compute S0, T, Q for the quadratic model
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Golden section + vertex check
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) * 0.5
    opt_v = f(opt_t)
    
    # Check vertices of individual parabolas
    for r in np.argsort(S0)[-6:]:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    
    return opt_t, opt_v

def _cont_opt(h, deadline, rng, beta_lo=8.0, beta_hi=80.0, max_stall=80,
              radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        it += 1
        
        # Compute gradient
        frac = min(1.0, it / 100.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        raw = np.correlate(h, 1.0 - h, mode='full')
        g = _smooth_max_grad(h, raw, beta)
        
        # Also try the single max-lag gradient
        r_max = int(np.argmax(raw))
        gs = _single_lag_grad(h, r_max)
        
        # Try both directions
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        
        for d in [g, gs]:
            dn = np.linalg.norm(d)
            if dn < 1e-12:
                continue
            d = d / dn
            alpha, val = _line_search_c5(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        
        if best_d is not None and abs(best_alpha) > 1e-10:
            h_new = _project(h + best_alpha * best_d, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                radius = min(radius * 1.08, 0.4)
            else:
                stall += 1
                radius = max(0.03, radius * 0.75)
        else:
            stall += 1
            radius = max(0.03, radius * 0.8)
    
    return best_h, best_c5

def _pair_cont(h, deadline, rng, n_iter=3000, radius=0.15):
    """Pair-wise continuous optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search_c5(h, d, radius)
        if c_try < cur_c5 - 1e-13 and abs(alpha) > 1e-10:
            h_new = _project(h + alpha * d, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for k in range(4):
        freq = rng.uniform(0.5, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)
