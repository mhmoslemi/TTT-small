import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, r):
    """Gradient of g_r at h."""
    n = len(h)
    d = abs(r - (n - 1))
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        m = n - d
        g[:m] += (1.0 - h[d:d+m])
        g[d:d+m] -= h[:m]
    return g

def _line_search_exact(h, d, raw, a_max, n_grid=100):
    """Find alpha in [0, a_max] minimizing max_k g_k(h+alpha*d)*dx."""
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(2 * n - 1)
    ds = np.abs(lags - (n - 1))
    
    c = raw.copy()
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        dd = ds[k]
        if dd == 0:
            b[k] = float(np.sum(d * (1.0 - 2.0 * h)))
            a[k] = float(-np.sum(d * d))
        else:
            m = n - dd
            b[k] = float(np.sum(d[:m] * (1.0 - h[dd:dd+m])) - np.sum(h[:m] * d[dd:dd+m]))
            a[k] = float(-np.sum(d[:m] * d[dd:dd+m]))
    
    grid = np.linspace(0.0, a_max, n_grid)
    best_alpha = 0.0
    best_val = float('inf')
    for alpha in grid:
        vals = a * alpha * alpha + b * alpha + c
        v = float(np.max(vals)) * dx
        if v < best_val:
            best_val = v
            best_alpha = alpha
    
    # Refine
    idx = int(np.argmin([float(np.max(a*gg*gg + b*gg + c))*dx for gg in grid]))
    lo = grid[max(0, idx-1)]
    hi = grid[min(n_grid-1, idx+1)]
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a*m1*m1 + b*m1 + c)) * dx
        f2 = float(np.max(a*m2*m2 + b*m2 + c)) * dx
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    mid = (lo + hi) / 2.0
    f_mid = float(np.max(a*mid*mid + b*mid + c)) * dx
    if f_mid < best_val:
        best_val = f_mid
        best_alpha = mid
    
    return best_alpha, best_val

def _lp_direction(h, n_active=30, solver='ECOS'):
    """Solve LP to get a direction that reduces the max overlap."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    
    # Select active lags
    idx = np.argsort(-raw)[:n_active]
    
    # Compute gradients for active lags
    grads = np.array([_lag_gradient(h, r) for r in idx])
    raw_active = raw[idx]
    
    # Build LP
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(x) == n / 2.0,
        x >= 0,
        x <= 1,
    ]
    # Linearized constraints: raw_active[i] + grads[i] . (x - h) <= t
    # i.e., grads[i].x <= t - raw_active[i] + grads[i].h
    rhs = t - raw_active + grads.dot(h)  # shape (n_active,)
    constraints += [grads @ x <= rhs]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
    except:
        return None, None
    
    if x.value is None:
        return None, None
    
    x_val = np.array(x.value).flatten()
    t_val = float(t.value)
    d = x_val - h
    return d, t_val

def _main_optimize_lp(h, deadline, n_active=30, stall_limit=50):
    """Main optimization loop using LP directions."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_act = n_active
    
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        
        # Get LP direction
        d, t_val = _lp_direction(h, n_act)
        if d is None:
            stall += 1
            continue
        
        # Project direction to maintain sum constraint
        d -= np.mean(d)
        
        # Find feasible step size
        pos = d > 1e-12
        neg = d < -1e-12
        a_max = 1.0
        if np.any(pos):
            a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
        a_max = max(a_max, 0.0)
        
        if a_max < 1e-10:
            stall += 1
            continue
        
        # Line search
        alpha, new_c5_est = _line_search_exact(h, d, raw, a_max, n_grid=80)
        
        if alpha < 1e-8:
            stall += 1
            continue
        
        h_new = h + alpha * d
        h_new = _project(h_new, n / 2.0)
        actual = _exact_c5(h_new)
        
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Slowly increase active set
            n_act = min(n_act + 1, 60)
        else:
            stall += 1
            n_act = max(n_act - 2, 10)
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, r):
    n = len(h)

    else:
        m = n - d
        g[:m] += (1.0 - h[d:d + m])
        g[d:d + m] -= h[:m]
    return g
