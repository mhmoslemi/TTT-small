        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        candidates = [(0.0, f0), (opt_t, opt_val), (t_hi, fe)]
        best_t = min(candidates, key=lambda z: z[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.35)
        else:
            stall += 1
            radius *= 0.70
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.08
    return best_h, best_c5

def _structured_perturb(h, rng, mode=0):
    n = len(h)
    target = n / 2.0
    if mode == 0:
        # Smooth multi-frequency perturbation
        x = np.arange(n, dtype=float)
        pertur = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.5, 8.0)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.3, 1.0) * 0.04
            pertur += a * np.sin(2.0 * np.pi * f * x / n + p)
        pertur += 0.01 * rng.normal(0, 1, n)
        return _project(h + pertur, target)
    elif mode == 1:
        # Block flip (flip a contiguous segment)
        h_new = h.copy()
        L = int(rng.integers(max(2, n // 20), max(3, n // 5)))
        start = int(rng.integers(0, n - L))
        h_new[start:start+L] = 1.0 - h[start:start+L]
        return _project(h_new, target)
    elif mode == 2:
        # Cyclic shift
        shift = int(rng.integers(1, max(2, n // 10)))
        direction = 1 if rng.random() < 0.5 else -1
        h_new = np.roll(h, shift * direction)
        return _project(h_new, target)
    elif mode == 3:
        # Scale jitter (modulate the pattern)
        x = np.arange(n, dtype=float)
        scale = rng.uniform(0.7, 1.3)
        h_new = np.exp(-0.5 * ((x - n/2) / (0.25 * n * scale))**2)
        h_new = 0.5 + 0.5 * (2.0 * h - 1.0) * np.exp(-0.1 * (x - n/2)**2 / n)
        return _project(h_new, target)
    else:
        # Random pair mass transfer
        h_new = h.copy()
        for _ in range(rng.integers(2, 6)):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = rng.uniform(0.02, 0.15)
                h_new[i] += amt
                h_new[j] -= amt
        return _project(h_new, target)

def _basin_hopping(h, deadline, rng, max_trials=200):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    trial = 0
    consecutive_fail = 0
    while time.time() < deadline and trial < max_trials:
        trial += 1
        mode = trial % 5
        h_pert = _structured_perturb(h_cur, rng, mode=mode)
        # Quick coordinate descent from perturbed point
        sub_deadline = min(deadline, time.time() + 12.0)
        h_opt, c5_opt = _coord_descent_fast(h_pert, sub_deadline, max_sweeps=300, seed=trial * 17)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_cur = h_opt.copy()
            cur_c5 = c5_opt
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            if consecutive_fail > 30:
                # Bigger jump
                h_pert = _structured_perturb(h_best, rng, mode=1)
                h_pert = _structured_perturb(h_pert, rng, mode=0)
                h_cur = _project(h_pert, target)
                consecutive_fail = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_bin, deadline, max_iter=80000, T_init=0.004, seed=31337):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        dS = np.zeros(2 * n - 1)
        for dd in range(0, n - oi):
            dS[n - 1 + dd] -= (1.0 - cur[oi + dd])
        for dd in range(-oi, 0):
            dS[n - 1 + dd] += cur[oi + dd]
        for dd in range(0, n - oz):
            dS[n - 1 + dd] += (1.0 - cur[oz + dd])
        for dd in range(-oz, 0):
            dS[n - 1 + dd] -= cur[oz + dd]
        new_S0 = S0 + dS
        new_c5 = float(np.max(new_S0)) * dx
        delta_c5 = new_c5 - cur_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            S0 = new_S0
            cur_c5 = new_c5
            ones[i] = zeros[j]
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep_binary(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0].tolist()
        zeros = np.where(cur <= 0.5)[0].tolist()
        for oi in ones:
            if time.time() > deadline:
                break
            g_oi = np.zeros(2 * n - 1)
            for dd in range(0, n - oi):
                g_oi[n - 1 + dd] = (1.0 - cur[oi + dd])
            for dd in range(-oi, 0):
                g_oi[n - 1 + dd] = -cur[oi + dd]
            for oz in zeros:
                g_oz = np.zeros(2 * n - 1)
                for dd in range(0, n - oz):
                    g_oz[n - 1 + dd] = (1.0 - cur[oz + dd])
                for dd in range(-oz, 0):
                    g_oz[n - 1 + dd] = -cur[oz + dd]
                dS = -g_oi + g_oz
                new_c5 = float(np.max(S0 + dS)) * dx
                if new_c5 < cur_c5 - 1e-14:
                    cur[oi] = 0.0
                    cur[oz] = 1.0
                    S0 += dS
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _gradient_descent_step(h, deadline, max_steps=500, lr_init=0.02, seed=77):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    rng = np.random.default_rng(seed)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        g = _subgradient_direction(h, top_k=6)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-10:
            break
        g = g / g_norm
        h_new = _project(h - lr * g, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            lr = min(lr * 1.1, 0.15)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            lr = lr * 0.7
            if lr < 1e-5:
                break
    return best_h, best_c5

def _multi_start(h_best, deadline, n_starts=15, seed=999):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        mode = s % 5
        h_pert = _structured_perturb(h_best, rng, mode=mode)
        h_pert = _structured_perturb(h_pert, rng, mode=(s + 2) % 5)
        sub_dl = min(deadline - 3, time.time() + 20.0)
        h_opt, c5_opt = _coord_descent_fast(h_pert, sub_dl, max_sweeps=500, seed=1000 + s)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Phase 0: Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick coordinate descent warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _coord_descent_fast(h_init, d1, max_sweeps=2000, seed=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement (7%)
    d2 = t0 + budget_s * 0.15
    h2, c5_2 = _lp_refine(best_h, d2, radius_init=0.10, stall_limit=60, top_k=12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient + coordinate descent (10%)
    d3 = t0 + budget_s * 0.25
    h3, c5_3 = _gradient_descent_step(best_h, min(d3, time.time() + budget_s * 0.04), max_steps=200, lr_init=0.03, seed=55)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    h3b, c5_3b = _coord_descent_fast(best_h, d3, max_sweeps=3000, seed=200, momentum=0.4)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()

    # Phase 4: Basin hopping with structured perturbations (22%)
    d4 = t0 + budget_s * 0.47
    h4, c5_4 = _basin_hopping(best_h, d4, rng, max_trials=120)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution sweep (15%)
    d5 = t0 + budget_s * 0.62
    n_curr = len(best_h)
    tried_ns = {n_curr}
    offset_list = list(range(-30, 31, 3))
    offset_list = [o for o in offset_list if (n_curr + o) >= 120 and (n_curr + o) <= 240]
    for dn in offset_list:
        if time.time() > d5 - 2:
            break
        n_try = n_curr + dn
        if n_try in tried_ns or n_try < 100:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 1, time.time() + budget_s * 0.012)
        h_r, c5_r = _coord_descent_fast(h_r, sub_dl, max_sweeps=600, seed=500 + dn)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Binary optimization (12%)
    d6 = t0 + budget_s * 0.74
    h_bin = _to_binary(best_h)
    n_bin = len(h_bin)
    sub_dl = min(d6, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003, seed=4444)
    sub_dl2 = min(d6, time.time() + budget_s * 0.03)
    h_bin, c5_bin = _pair_sweep_binary(h_bin, sub_dl2)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary result
    if time.time() < d6 - 5:
        h_pol = _project(h_bin.copy(), n_bin / 2.0)
        sub_dl3 = min(d6, time.time() + budget_s * 0.025)
        h_pol, c5_pol = _lp_refine(h_pol, sub_dl3, radius_init=0.06, stall_limit=40, top_k=14)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Multi-start exploration (8%)
    d7 = t0 + budget_s * 0.82
    h7, c5_7 = _multi_start(best_h, d7, n_starts=12, seed=7777)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Intensive final optimization (13%)
    d8 = deadline
    # Coordinate descent
    sub_dl = min(d8, time.time() + budget_s * 0.04)
    h8, c5_8 = _coord_descent_fast(best_h, sub_dl, max_sweeps=5000, seed=9999)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    # LP
    sub_dl = min(d8, time.time() + budget_s * 0.035)
    h8, c5_8 = _lp_refine(best_h, sub_dl, radius_init=0.04, stall_limit=60, top_k=18)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    # More basin hopping with smaller perturbations
    if time.time() < d8 - 5:
        sub_dl = min(d8 - 2, time.time() + budget_s * 0.03)
        h8, c5_8 = _basin_hopping(best_h, sub_dl, rng, max_trials=40)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    # Final coord descent
    if time.time() < d8 - 3:
        sub_dl = min(d8 - 1, time.time() + budget_s * 0.02)
        h8, c5_8 = _coord_descent_fast(best_h, sub_dl, max_sweeps=2000, seed=12345)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final: project and evaluate
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
