Let me think about the gradient of the LSE objective:
- score_k(h) = sum over valid (i,j) of h_i * (1 - h_j) where j-i = k (lag)
- d(score_k)/d(h_m) is the gradient matrix G[k, m]
- LSE = τ * log(sum_k exp(score_k / τ))
- d(LSE)/d(h_m) = (1/sum_k exp(score_k/τ)) * sum_k exp(score_k/τ) * G[k,m] / τ
- = sum_k p_k * G[k,m] where p_k = exp(score_k/τ) / sum_j exp(score_j/τ)

So the gradient is a weighted sum of rows of G, with weights being softmax of scores.

For the gradient matrix G[k, m]:
- score_k = sum_{i: i and i+k in [0,n)} h[i] * (1 - h[i+k]) (for k ≥ 0)
- For k < 0, score_k = sum_{i: i and i+k in [0,n)} h[i] * (1 - h[i+k])

Wait, let me look at this more carefully. From numpy docs:

c[k] = sum_i a[i+k] * conj(v[i])

So np.correlate(h, 1-h, 'full'):
c[k] = sum_i h[i+k] * (1-h)[i] for valid i

The length is 2n-1, with k going from -(n-1) to n-1.

For the middle index (k=0): c[n-1] = sum_i h[i] * (1-h)[i]

= sum_{i=0}^{n-1-k} h[i+k] * (1-h)[i]

= sum_{i=-k}^{n-1} h[i+k] * (1-h)[i]

= sum_{j=0}^{n-1-|k|} h[j] * (1-h)[j-|k|]

So for lag d ≥ 0 (corresponding to index n-1+d in the result):
score = sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])

d(score)/d(h[m]):
- If m is in position i+d for some i: contributes (1-h[i])
- If m is in position i for some i: contributes -h[i+d]

So G[n-1+d, m]:
- For m ≥ d: G[n-1+d, m] += (1-h[m-d]) (from h[m] * (1-h[m-d]) term, m plays role of i+d, i = m-d)
- For m ≤ n-1-d: G[n-1+d, m] += -h[m+d] (from h[m+d] * (1-h[m]) term, m plays role of i, h[i+d] = h[m+d])

- h[m] appears in (1-h[i]) when i = m, contribution: -h[m+d]. This is valid when 0 ≤ m ≤ n-1-d.

So G[n-1+d, m] = (1-h[m-d]) * [d ≤ m ≤ n-1] + (-h[m+d]) * [0 ≤ m ≤ n-1-d]

For d=0 (middle): G[n-1, m] = (1-h[m]) + (-h[m]) = 1 - 2*h[m]. Correct!

For negative lags, by symmetry of the correlation:
score_{-d} = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d]) (this is just the conjugate, and for real sequences it's the same as score_d with roles swapped... wait no)

Actually for real sequences: np.correlate(h, 1-h) at lag -d:

= sum_{i=d}^{n-1} h[i-d] * (1-h[i])
Let j = i-d:
= sum_{j=0}^{n-1-d} h[j] * (1-h[j+d])

So score_{-d} = sum_{j=0}^{n-1-d} h[j] * (1-h[j+d])

d(score_{-d})/d(h[m]):
- h[m] in h[j] when j=m: valid when 0 ≤ m ≤ n-1-d. Contribution: (1-h[m+d])
- h[m] in (1-h[j+d]) when j+d=m, j=m-d: valid when d ≤ m ≤ n-1. Contribution: -h[m-d]

So G[n-1-d, m] = (1-h[m+d]) * [0 ≤ m ≤ n-1-d] + (-h[m-d]) * [d ≤ m ≤ n-1]

grad[m] = sum_k p_k * G[k, m] * dx (where dx factor comes from the objective being in integral units)

Wait, actually the objective is max_k (score_k * dx), and the LSE approximation is:
LSE = τ * log(sum_k exp(score_k * dx / τ))

where τ is in the same units as score_k * dx (i.e., the C5 units).

So gradient of LSE w.r.t. h[m]:
= (1/sum_k exp(score_k*d/τ)) * sum_k exp(score_k*d/τ) * dx * G[k,m]
= sum_k p_k * G[k,m] * dx

where p_k = exp(score_k * dx / τ) / sum_j exp(score_j * dx / τ)

This is efficient to compute: just a weighted sum of rows of G with weights p_k (softmax of scaled scores).

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 1.0
_g_h = None
_g_G = None
_g_scores = None
_g_probs = None

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(0, n):
        m = n - d
        # Row n-1+d (positive lag d)
        if d == 0:
            G[n-1, :] = 1.0 - 2.0*h
        else:
            # G[n-1+d, m_idx] for m_idx >= d: (1-h[m_idx-d])
            G[n-1+d, d:] += (1.0 - h[:m])
            # G[n-1+d, m_idx] for m_idx < m: -h[m_idx+d]
            G[n-1+d, :m] += -h[d:]
        # Row n-1-d (negative lag -d), only for d > 0
        if d > 0:
            # G[n-1-d, m_idx] for m_idx < m: (1-h[m_idx+d])
            G[n-1-d, :m] += (1.0 - h[d:])
            # G[n-1-d, m_idx] for m_idx >= d: -h[m_idx-d]
            G[n-1-d, d:] += -h[:m]
    return G

def _lse_obj(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = np.max(scores)
    u = (scores - mval) / tau
    z = np.exp(u)
    lse = mval + tau * np.log(np.sum(z))
    _g_probs = z / np.sum(z)
    return float(lse)

def _lse_grad(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = np.max(scores)
    u = (scores - mval) / tau
    z = np.exp(u)
    probs = z / np.sum(z)
    G = _build_G(h)
    grad = G.T @ probs * dx
    return grad

def _eq_func(x):
    return np.array([np.sum(x) - _g_target])

def _eq_jac(x):
    return np.ones((1, _g_n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_anneal(h, deadline, max_iters=50):
    global _g_n, _g_target, _g_dx, _g_tau
    n = len(h)
    _g_n = n
    _g_target = float(n / 2)
    _g_dx = 2.0 / n

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Temperature schedule
    max_score = float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))
    tau_values = []
    tau = max(0.03, 0.3 * max_score)
    for _ in range(10):
        tau_values.append(tau)
        tau *= 0.6
        if tau < 0.005:
            break
    
    for tau_i, tau_val in enumerate(tau_values):
        if time.time() > deadline - 5:
            break
        _g_tau = tau_val
        # Try SLSQP
        try:
            bounds = [(0.0, 1.0)] * n
            cons = {'type': 'eq', 'fun': _eq_func, 'jac': _eq_jac}
            res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max(10, max_iters // len(tau_values)),
                                  'ftol': 1e-10, 'disp': False})
            if res.x is not None:

                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                # Use this as starting point regardless (it's in the right basin)
                # unless it's much worse
                if c5_new < best_c5 + 0.01:
                    h = h_new
                else:
                    h = best_h.copy()
        except Exception:

    return best_h, best_c5

def _line_search_exact(h_start, h_end, dx, n_grid=60):
    n = len(h_start)
    # h_start and h_end should both be feasible
    # Search along t * (h_end - h_start) + h_start for t in [0,1]
    t_grid = np.linspace(0.0, 1.0, n_grid)
    best_t = 0.0
    best_val = _exact_c5(h_start)

        h_try = _project_sum(h_try, n/2.0)
        val = _exact_c5(h_try)
        if val < best_val:
            best_val = val
            best_t = t
    # Golden section around best
    idx = int(np.argmin([_exact_c5(h_start + t*(h_end-h_start)) if False else 0 for t in [best_t]]))
    # Simpler: just do a few more points around best_t
    t_lo = max(0.0, best_t - 0.05)
    t_hi = min(1.0, best_t + 0.05)
    for _ in range(30):

        c2 = t_lo + 0.618 * (t_hi - t_lo)
        f1 = _exact_c5(_project_sum(h_start + c1*(h_end-h_start), n/2.0))

    t_best = (t_lo + t_hi) / 2.0
    h_best = _project_sum(h_start + t_best*(h_end-h_start), n/2.0)
    val_best = _exact_c5(h_best)

        best_val = val_best
        best_t = t_best
    return h_best, best_val

def _pair_swap_polish(h, deadline, n_tries=10000, seed=42):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Try moving mass from j to i (if h[j] > 0.5) or from i to j (if h[i] > 0.5)
        if h[j] > h[i]:
            # Move from j to i
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        else:
            d = np.zeros(n)
            d[j] = 1.0
            d[i] = -1.0
        # Line search
        h_try = h + d
        # Check bounds
        if h_try[i] > 1.0 or h_try[i] < 0.0 or h_try[j] > 1.0 or h_try[j] < 0.0:
            stall += 1
            continue
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=99):

    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 1.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(0, n):
        m = n - d
        if d == 0:
            G[n - 1, :] = 1.0 - 2.0 * h
        else:
            G[n - 1 + d, d:] += (1.0 - h[:m])
            G[n - 1 + d, :m] += -h[d:]
            G[n - 1 - d, :m] += (1.0 - h[d:])
            G[n - 1 - d, d:] += -h[:m]
    return G

def _lse_value(x):
    dx = _g_dx
    tau = _g_tau
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    z = np.exp(u)
    s = float(np.sum(z))
    lse = mval + tau * np.log(s)
    return lse

def _lse_gradient(x):
    dx = _g_dx
    tau = _g_tau
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    z = np.exp(u)
    probs = z / float(np.sum(z))
    G = _build_G(h)
    return G.T @ probs * dx

def _eq_fun(x):
    return np.array([np.sum(x) - _g_target])

def _eq_jac(x):
    return np.ones((1, _g_n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_step(h, tau, maxiter=50):

    n = len(h)
    _g_n = n
    _g_target = float(n / 2)
    _g_dx = 2.0 / n

        res = minimize(_lse_value, h.copy(), jac=_lse_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project_sum(res.x, n / 2.0)
    except Exception:

    return h

def _projected_grad_descent(h, tau, n_steps=200, lr_init=0.01):

    n = len(h)

def _line_search_exact(h_start, h_end, n_coarse=50, n_fine=40):
    n = len(h_start)
    target = n / 2.0
    t_grid = np.linspace(0.0, 1.0, n_coarse)
    best_t = 0.0
    best_val = _exact_c5(h_start)

        val = _exact_c5(h_try)
        if val < best_val:
            best_val = val

    idx = int(round(best_t * (n_coarse - 1)))

def _pair_sweep(h, deadline, max_rounds=20):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

            d[j] = 1.0
        h_try = cur + d
        if h_try[i] > 1.0 + 1e-9 or h_try[i] < -1e-9 or h_try[j] > 1.0 + 1e-9 or h_try[j] < -1e-9:
            stall += 1
            continue
        h_try = _project_sum(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 - 1e-13:
            cur = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = c5_try
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb_h(h, rng, amp):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += amp * 0.1 * rng.normal(0, 1, n)
    pert *= amp
    return _project_sum(h + pert, n / 2.0)

def _directional_ls(h, deadline, n_dirs=20, seed=200):
    n = len(h)
    rng = np.random.default_rng(seed)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_bin = _to_binary(h)
    # Direction to binary
    h_ls, c5_ls = _line_search_exact(best_h, h_bin)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
    # Direction to complement
    h_comp = _to_binary(1.0 - h)
    if time.time() < deadline:
        h_ls, c5_ls = _line_search_exact(best_h, h_comp)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    # Random directions
    for _ in range(n_dirs):
        if time.time() > deadline:
            break
        d = rng.normal(0, 1, n)
        d -= np.mean(d)
        d /= np.linalg.norm(d)
        h_end = _project_sum(best_h + 0.5 * d, target)
        h_ls, c5_ls = _line_search_exact(best_h, h_end, n_coarse=30, n_fine=30)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        n_init = 180
        h_init = np.zeros(n_init)
        h_init[:n_init // 2] = 1.0

    n = len(h_init)
    if n % 2 != 0:
        n -= 1
        h_init = h_init[:n]
    h_init = _project_sum(h_init, n / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE annealing (35%)
    dl1 = t0 + budget_s * 0.35
    tau_sched = [0.10, 0.06, 0.035, 0.02, 0.012, 0.007, 0.004]
    h1, c5_1 = _smooth_anneal(best_h, dl1, tau_list=tau_sched, slsqp_iters=35)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Directional line search (8%)
    dl2 = t0 + budget_s * 0.43
    h2, c5_2 = _directional_ls(best_h, dl2, n_dirs=15, seed=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA + pair sweep (12%)
    dl3 = t0 + budget_s * 0.55
    h_b = _to_binary(best_h)
    h_b, c5_b = _binary_sa(h_b, min(dl3, t0 + budget_s * 0.12), max_iter=40000, T_init=0.003, seed=4242)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < dl3:
        h_b2, c5_b2 = _pair_sweep(best_h, dl3, max_rounds=15)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 4: LSE polish with tighter tau (12%)
    dl4 = t0 + budget_s * 0.67
    tau_tight = [0.025, 0.015, 0.008, 0.004, 0.002, 0.001]
    n_cur = len(best_h)
    h4, c5_4 = _smooth_anneal(_project_sum(best_h.copy(), n_cur / 2.0), dl4, tau_list=tau_tight, slsqp_iters=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Continuous pair swap (10%)
    dl5 = t0 + budget_s * 0.77
    h5, c5_5 = _pair_swap_continuous(best_h, dl5, n_tries=20000, seed=666)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Basin hopping via LSE (12%)
    dl6 = t0 + budget_s * 0.89
    scale = 0.04
    n_cur = len(best_h)
    n_attempts = 0
    while time.time() < dl6 and n_attempts < 80:
        n_attempts += 1
        h_pert = _perturb_h(best_h, rng, scale)
        tau_mini = max(0.005, scale * 0.5)
        h_t = _slsqp_step(h_pert, tau_mini, maxiter=20)
        h_t = _project_sum(h_t, n_cur / 2.0)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.08)

    # Phase 7: Multi-n exploration (7%)
    dl7 = t0 + budget_s * 0.96
    n_cur = len(best_h)
    tried = set()
    for dn in [-6, -12, 6, 12, -18, 18, -4, 4]:
        if time.time() > dl7:
            break
        n_try = n_cur + dn
        if n_try < 70 or n_try > 280 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.015)
        h_r, c5_r = _smooth_anneal(h_r, sub_dl, tau_list=[0.03, 0.015, 0.008, 0.004], slsqp_iters=20)
        if time.time() < dl7 - 3:
            h_rb = _to_binary(h_r)
            sub_dl2 = min(dl7, time.time() + budget_s * 0.008)
            h_rb, c5_rb = _binary_sa(h_rb, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try + 1)
            if c5_rb < c5_r:
                c5_r = c5_rb
                h_r = h_rb
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish: LSE + pair
    if deadline - time.time() > 5:
        n_cur = len(best_h)
        h_f = _project_sum(best_h.copy(), n_cur / 2.0)
        h_f, c5_f = _smooth_anneal(h_f, deadline - 2, tau_list=[0.005, 0.003, 0.001], slsqp_iters=25)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final pair sweep if time
    if deadline - time.time() > 3:
        h_ps, c5_ps = _pair_sweep(best_h, deadline, max_rounds=10)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
