def _corr_grad(h, k):
    """Gradient of unscaled correlation at lag k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # corr[k] = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
        # d/dh[j]: (1-h[j-k]) for j>=k, minus h[j+k] for j<=n-k-1
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        # corr at lag -u = sum_{i=0}^{n-u-1} h[i] * (1-h[i+u])
        # d/dh[j]: (1-h[j+u]) for j<=n-u-1, minus h[j-u] for j>=u
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

    dm_idx = np.arange(n, 2*n)

    eps = 1e-5
    c = np.zeros(nvar)
    c[dp_idx] = eps
    c[dm_idx] = eps
    c[t_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k: grad_k · (d+ - d-) - t <= -f_k
    A_rows = []
    b_rows = []
    center = n - 1
    for k in active_lags:
        lag = k - center
        g = _corr_grad(h, lag) * dx
        row = np.zeros(nvar)
        row[dp_idx] = g
        row[dm_idx] = -g
        row[t_idx] = -1.0
        A_rows.append(row)
        b_rows.append(-f[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum d+ - sum d- = 0
    A_eq = np.zeros((1, nvar))

    A_eq[0, dm_idx] = -1.0

        bounds.append((0, lo))  # d-
    bounds.append((F - 0.5, F + 0.01))  # t

            oz = cand_z_list[int(rng.integers(0, len(cand_z_list)))]
            if oi == oz:
                continue
            new_c5, _ = _binary_swap_eval(cur, oi, oz)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_oi = oi
                best_oz = oz
        if best_oi >= 0 and best_delta < -1e-14:
            cur[best_oi] = 0.0
            cur[best_oz] = 1.0
            cur_c5 = _exact_c5(cur)
            corr = np.correlate(cur, 1.0 - cur, mode='full')
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _block_swap_search(h_bin, deadline, max_iters=15000, seed=77):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(6, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        b1 = cur[i:i + length].copy()
        b2 = cur[j:j + length].copy()
        if float(np.sum(b1)) != float(np.sum(b2)):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i + length] = b2
        trial[j:j + length] = b1
        new_c5 = _exact_c5(trial)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)


def _pair_swap_loop(h, deadline, n_pairs=10000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    soft_dl = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Load initial
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(180, 0.5)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization on continuous (25%)
    d1 = t0 + budget_s * 0.25
    h_lp, c5_lp = _lp_optimize(h_init, d1, radius_init=0.12, max_lags=50, stall_limit=100)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Second LP pass with smaller radius
    if time.time() < d1 + (soft_dl - d1) * 0.1:
        d1b = min(soft_dl, time.time() + budget_s * 0.05)
        h_lp2, c5_lp2 = _lp_optimize(best_h, d1b, radius_init=0.05, max_lags=60, stall_limit=80)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Phase 2: Binary optimization (35%)
    d2 = t0 + budget_s * 0.60
    n_best = len(best_h)
    h_bin = _to_binary(best_h, n_best)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Targeted binary search
    sub_dl = min(d2, time.time() + budget_s * 0.12)
    h_b, c5_b = _targeted_binary(h_bin, sub_dl, max_iters=80000, seed=101)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # SA binary search
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iters=150000, seed=202)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Block swap
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_b, c5_b = _block_swap_search(h_b, sub_dl, max_iters=15000, seed=303)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # More targeted
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=50000, seed=404)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # LP polish after binary
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_pol = _project(best_h.copy(), n_best / 2.0)
    h_pol, c5_pol = _lp_optimize(h_pol, sub_dl, radius_init=0.06, max_lags=60, stall_limit=50)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.80
    n_best = len(best_h)
    tried_ns = set()
    for dn in range(-24, 25, 4):
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 250 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.08, max_lags=40, stall_limit=40)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b = _targeted_binary(h_b, sub_dl2, max_iters=30000, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Intensive refinement (15%)
    d4 = soft_dl
    n_best = len(best_h)

    # Pair swap on continuous
    if time.time() < d4 - 20:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=6000, radius=0.20, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Binary targeted + SA on best
    n_best = len(best_h)
    if time.time() < d4 - 25:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=60000, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d4 - 15:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iters=60000, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Basin hopping
    if time.time() < d4 - 10:
        scale = 0.03
        for _ in range(40):
            if time.time() > d4 - 3:
                break
            h_pert = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.005)
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.05, max_lags=40, stall_limit=30)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.1)

    # Final LP polish
    if time.time() < d4 - 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, d4 - 1, radius_init=0.03, max_lags=70, stall_limit=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Try a few nearby n values at the end
    n_fin = len(best_h)
    for dn in [-2, 2, -4, 4]:
        if time.time() > d4 - 3:
            break
        n_try = n_fin + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 1, time.time() + 3)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.03, max_lags=40, stall_limit=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
