    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_direction(h, scores, beta, radius):
    """Compute a descent direction using weighted gradient of active scores."""
    n = len(h)
    mx = float(np.max(scores))
    # Use a sharper weighting
    shifted = np.clip(beta * (scores - mx), -100.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Scale to fit within radius
    g_norm = float(np.max(np.abs(g)))
    if g_norm > radius:
        g *= radius / g_norm
    # Project to feasible box
    pos = g > 0
    neg = g < 0
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]))
    g[neg] = np.maximum(g[neg], -h[neg])
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.1, max_iters=5000):
    """Local optimization using LP directions and line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active, gap)
        if d is None or float(np.max(np.abs(d))) < 1e-14:
            d = _grad_direction(h, scores, beta=30.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-14:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Gradually adjust parameters
            radius = min(radius * 1.05, 0.25)
            gap = max(gap * 0.92, 0.0005)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.7
            gap *= 1.4
            n_active = min(n_active + 3, 35)
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200):
    """Try pairwise swaps (increase one, decrease another)."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    rng = np.random.default_rng(12345)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline or stall > 80:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(5, len(ones) // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.3)
            if c_try < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    """Generate a perturbed version of h."""
    n = len(h)
    x = np.arange(n, dtype=float)
    # Use different perturbation types
    rtype = rng.integers(0, 4)
    if rtype == 0:
        # Smooth sinusoidal
        freqs = rng.uniform(0.3, 6.0, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.5, 1.0, size=3)
        perturb = np.zeros(n)
        for f, p, a in zip(freqs, phases, amps):
            perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    elif rtype == 1:
        # Block perturbation
        n_blocks = rng.integers(2, max(3, n // 20))
        perturb = np.zeros(n)
        for _ in range(n_blocks):
            start = int(rng.integers(0, n))
            length = int(rng.integers(2, max(3, n // 8)))
            amp = rng.uniform(-1, 1) * 0.5
            end = min(start + length, n)
            perturb[start:end] += amp
    elif rtype == 2:
        # Random noise
        perturb = rng.normal(0, 0.3, n)
    else:
        # Targeted: perturb in direction that reduces the max score
        scores = _scores(h)
        mx = float(np.max(scores))
        # Weight by how close each score is to the max
        w = np.exp(20 * (scores - mx))
        w /= w.sum()
        lags = np.arange(len(scores)) - (n - 1)
        G = _grad_rows(h, lags)
        perturb = G.T @ w
        perturb -= perturb.mean()
    
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _cvxpy_qp(h, scores, active_idx, radius):
    """Try to solve a QP subproblem using cvxpy."""
    n = len(h)
    # Linearize around current h
    # Solve: min t s.t. G_r^T d - t <= -R_r for r in active, ||d||_inf <= radius, sum(d)=0, h+d in [0,1]
    lags = active_idx - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active_idx)
    
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(d) == 0]
    constraints.append(t >= 0)
    for i in range(m):
        constraints.append(G[i] @ d - t <= -scores[active_idx[i]])
    constraints.append(d <= radius)
    constraints.append(d >= -radius)
    # Box constraints
    constraints.append(d >= -h)
    constraints.append(d <= 1.0 - h)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if d.value is not None and np.isfinite(d.value).all():
            return np.array(d.value).flatten()
    except Exception:
        pass
    return None
