def _compute_DA(h, n):
    """Compute partial derivatives DA[a, ell] for all positions and lags."""
    # S(ell) for ell = d + n - 1 where d is signed lag
    # For positive d: S = sum_{k=0}^{n-d-1} h[k+d]*(1-h[k])
    # ∂S/∂h[a] = (1-h[a-d])*I(d<=a<=n-1) - h[a+d]*I(0<=a<=n-1-d)
    # For negative d (let g=-d>0): S = sum_{k=0}^{n-g-1} h[k]*(1-h[k+g])
    # ∂S/∂h[a] = (1-h[a+g])*I(a+g<=n-1) - h[a-g]*I(a-g>=0)
    #           = (1-h[a+g])*I(a<=n-1-g) - h[a-g]*I(a>=g)
    
    DA = np.zeros((n, 2*n-1))
    center = n - 1
    for a in range(n):
        for d in range(1, n):
            # Positive lag d: ell = d + center
            ell_pos = d + center
            if d <= a <= n-1:
                DA[a, ell_pos] += (1 - h[a-d])
            if 0 <= a <= n-1-d:
                DA[a, ell_pos] += -h[a+d]
            # Negative lag -d: ell = -d + center
            ell_neg = -d + center
            if a <= n-1-d:
                DA[a, ell_neg] += (1 - h[a+d])
            if a >= d:
                DA[a, ell_neg] += -h[a-d]
    return DA

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -1.0, 2.0)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-13:

            x[interior] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _swap_delta(h, a, b):
    n = len(h)
    center = n - 1
    dS = np.zeros(2 * n - 1)
    for d in range(1, n):
        # Positive lag d, index = d + center
        cp = (1.0 - h[a - d]) if (d <= a) else 0.0
        cp -= h[a + d] if (a + d < n) else 0.0
        cm = (-(1.0 - h[b - d])) if (d <= b) else 0.0
        cm += h[b + d] if (b + d < n) else 0.0
        dS[d + center] = cp - cm
        # Negative lag -d, index = -d + center
        cn = (1.0 - h[a + d]) if (a + d < n) else 0.0
        cn -= h[a - d] if (a - d >= 0) else 0.0
        cm2 = (-(1.0 - h[b + d])) if (b + d < n) else 0.0
        cm2 += h[b - d] if (b - d >= 0) else 0.0
        dS[-d + center] = cn - cm2
        # Cross correction
        if abs(a - b) == d:
            dS[d + center] += 1.0
            dS[-d + center] += 1.0
    return dS

def _grad_rows(h, lags):

    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b, hi_b = t_lo, t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [opt_val, float(np.max(S0)) * dx,
             float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx]
    best_val = min(cands)
    if best_val == cands[0]:
        return opt_t, best_val
    elif best_val == cands[1]:
        return 0.0, best_val
    else:
        return t_hi, best_val

def _active_lp(h, scores, radius, n_active, gap):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_dir(h, scores, beta, radius):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _continuous_local_opt(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    beta = 15.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        d = _active_lp(h, scores, radius, n_active, gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_dir(h, scores, beta, radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.6
                gap *= 1.8
                n_active = min(n_active + 5, 35)
                beta *= 1.3
                if radius < 1e-7:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
            gap = max(gap * 0.85, 0.001)
            n_active = max(n_active - 1, 10)
            beta = max(beta * 0.95, 5.0)
        else:
            stall += 1
            radius *= 0.7
            gap *= 1.4
            n_active = min(n_active + 3, 30)
            beta *= 1.15
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.08
                beta = 15.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 5
    freqs = rng.uniform(0.5, 10.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amplitudes = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for fi in range(n_freqs):
        perturb += amplitudes[fi] * np.sin(2.0 * np.pi * freqs[fi] * x / n + phases[fi])
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_sweep(h, deadline, max_rounds=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_trials = max(1, len(ones) // 4)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.2)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Initialize from parent
    has_parent = 'initial_h_values' in dir() or True
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h_init = _project(h0.copy(), n0 / 2.0)
    except Exception:
        n0 = 120
        h_init = np.zeros(n0)
        idx = np.arange(0, n0, 2)
        h_init[idx] = 1.0

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick local opt on parent (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _continuous_local_opt(h_init, d1, radius_init=0.10, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary local search on parent (15%)
    d2 = t0 + budget_s * 0.30
    n_p = len(best_h)
    h_bin = _to_binary(best_h)
    h_bin, max_count = _binary_local_search(h_bin, d2, max_iters=200000, rng=rng)
    bin_c5 = max_count * (2.0 / n_p)
    if bin_c5 < best_c5:
        best_c5 = bin_c5
        best_h = h_bin.copy()
    # Polish binary result with continuous
    if time.time() < d2 + budget_s * 0.03:
        h_pol = _project(h_bin.copy(), n_p / 2.0)
        sub_d = min(t0 + budget_s * 0.35, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _continuous_local_opt(h_pol, sub_d, radius_init=0.06, max_iters=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Multi-resolution exploration (25%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    tried_ns = set()
    # Try various n values
    for dn in [-40, -30, -20, -10, -5, 0, 5, 10, 20, 30, 40]:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 50 or n_try > 250 or n_try in tried_ns:
            continue
        if n_try % 2 != 0:
            n_try += 1
            if n_try in tried_ns:
                continue
        tried_ns.add(n_try)
        # Resample and optimize
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _continuous_local_opt(h_r, sub_dl, radius_init=0.08, max_iters=600)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - budget_s * 0.02:
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_bin_r = _to_binary(h_r)
            h_bin_r, mc = _binary_local_search(h_bin_r, sub_dl2, max_iters=30000, rng=rng)
            bin_c5_r = mc * (2.0 / n_try)
            if bin_c5_r < c5_r:
                c5_r = bin_c5_r
                h_r = h_bin_r.copy()
            # Quick polish
            if time.time() < d3 - 2:
                h_r2, c5_r2 = _continuous_local_opt(_project(h_r.copy(), n_try / 2.0),
                                                     min(d3, time.time() + budget_s * 0.005),
                                                     radius_init=0.05, max_iters=300)
                if c5_r2 < c5_r:
                    c5_r = c5_r2
                    h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.82
    scale = 0.015
    n_restarts = 0
    while time.time() < d4 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.005)
        h_t, c5_t = _continuous_local_opt(h_pert, sub_dl, radius_init=0.05, max_iters=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.06)
        # Occasionally try binary perturbation
        if n_restarts % 10 == 0 and time.time() < d4:
            h_b = _to_binary(best_h)
            n_flips = rng.integers(2, 6)
            ones_list = list(np.where(h_b > 0.5)[0])
            zeros_list = list(np.where(h_b <= 0.5)[0])
            for _ in range(n_flips):
                if not ones_list or not zeros_list:
                    break
                oi = int(rng.integers(0, len(ones_list)))
                zi = int(rng.integers(0, len(zeros_list)))
                h_b[ones_list[oi]] = 0.0
                h_b[zeros_list[zi]] = 1.0
                ones_list[oi] = zeros_list[zi]
                ones_list, zeros_list = zeros_list, ones_list
                # Rebuild
                ones_list = list(np.where(h_b > 0.5)[0])
                zeros_list = list(np.where(h_b <= 0.5)[0])
            sub_dl_b = min(d4, time.time() + budget_s * 0.008)
            h_b2, mc2 = _binary_local_search(h_b, sub_dl_b, max_iters=20000, rng=rng)
            c5_b2 = mc2 * (2.0 / len(h_b))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 5: Pair sweep polish (8%)
    d5 = t0 + budget_s * 0.92
    h_s = _project(best_h.copy(), len(best_h) / 2.0)
    h_s, c5_s = _pair_sweep(h_s, d5, max_rounds=80)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 6: Final polish (8%)
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    h_f, c5_f = _continuous_local_opt(h_f, deadline, radius_init=0.03, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
