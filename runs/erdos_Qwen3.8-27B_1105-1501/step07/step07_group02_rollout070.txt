def _compute_grad(h, w, dx):
    n = len(h)
    k = np.arange(-(n-1), n)  # lags
    # For each i and each lag k_r, compute contribution
    # grad_i = dx * sum_r w_r * [(1-h_{i+k_r}) * 1[0<=i+k_r<n] - h_{i-k_r} * 1[0<=i-k_r<n]]
    
    # Create index matrix: rows = i (0..n-1), cols = r (0..2n-2)
    # idx1[i, r] = i + k_r, valid if 0 <= i+k_r < n
    # idx2[i, r] = i - k_r, valid if 0 <= i-k_r < n
    
    i_idx = np.arange(n)[:, None]  # (n, 1)
    k_vals = k[None, :]  # (1, 2n-1)
    
    idx1 = i_idx + k_vals  # (n, 2n-1)
    idx2 = i_idx - k_vals  # (n, 2n-1)
    
    valid1 = (idx1 >= 0) & (idx1 < n)  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)  # (n, 2n-1)
    
    # For valid1, first term = 1 - h[idx1]
    h1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)

    # Project to satisfy sum constraint

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_grad(h, w, dx):
    n = len(h)
    k = np.arange(-(n - 1), n)
    i_idx = np.arange(n)[:, None]
    k_vals = k[None, :]
    idx1 = i_idx - k_vals
    idx2 = i_idx + k_vals
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    grad = dx * (w[None, :] * (h1 - h2)).sum(axis=1)
    grad -= np.mean(grad)
    return grad

def _soft_weights(S, beta):
    M = float(np.max(S))
    if beta < 0.01:
        return np.ones(len(S)) / len(S)
    w = np.exp(beta * (S - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, int(0.05 * len(S)))
        idx = np.argsort(S)[-top_k:]
        w = np.zeros(len(S))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _dense_optimize(h, deadline, beta_init=15.0, max_steps=50000, stall_limit=300, seed=42):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    step_size = 0.3
    beta = beta_init
    stall = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    S = corr * dx

    for step in range(max_steps):
        if time.time() > deadline:
            break
        if stall > stall_limit or step_size < 1e-8:
            break
        w = _soft_weights(S, beta)
        grad = _compute_grad(h, w, dx)
        grad_norm = float(np.sqrt(np.sum(grad * grad)))
        if grad_norm < 1e-12:
            break
        direction = -grad / grad_norm
        t = step_size
        improved = False
        for ls in range(25):
            h_try = _project(h + t * direction, target)
            c_try = _exact_c5(h_try)
            if c_try < cur_c5 - 1e-12:
                h = h_try
                cur_c5 = c_try
                corr = np.correlate(h, 1.0 - h, mode='full')
                S = corr * dx
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                step_size = min(t * 1.05, 1.5)
                break

        if not improved:

            step_size *= 0.7

            if stall % 100 == 0:

                h = _project(h + perturb, target)
                cur_c5 = _exact_c5(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                S = corr * dx
                beta = beta_init
                step_size = 0.2
    return best_h, best_c5

def _line_search_pair(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    best_val = min(opt_val, float(vals[best_idx]))
    if best_val == opt_val:
        return opt_t, best_val
    return float(ts[best_idx]), best_val

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):

    target = n / 2.0

    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')

        sinks = sort_idx[:min(n_pairs, n // 2)]

            b = int(rng.integers(0, n))
            if a != b:

            if time.time() > deadline:

            if sink == source:

            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            S0_scaled = S0 * dx
            stall = 0
            radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 50 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.002, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _generate_seed(n, idx, seed):
    rng = np.random.default_rng(seed + idx * 1000)
    x = np.linspace(0, 2, n)
    if idx % 5 == 0:
        h = np.clip(0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi)), 0, 1)
    elif idx % 5 == 1:
        h = np.clip(0.5 + 0.25 * np.cos(2 * np.pi * x * rng.uniform(1, 3)), 0, 1)
    elif idx % 5 == 2:
        centers = rng.uniform(0, 2, size=3)
        widths = rng.uniform(0.1, 0.4, size=3)
        h = np.zeros(n)
        for c, wdt in zip(centers, widths):
            h += np.exp(-0.5 * ((x - c) / wdt) ** 2)
        h = np.clip(h / (np.max(h) + 1e-10) * 0.8, 0, 1)
    elif idx % 5 == 3:
        h = rng.uniform(0, 1, n)
        h = np.clip(h + 0.1 * np.sin(3 * np.pi * x), 0, 1)
    else:
        h = np.clip(0.5 + rng.uniform(-0.3, 0.3, n), 0, 1)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense gradient optimization at parent resolution (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _dense_optimize(h_init, d1, beta_init=15.0, max_steps=30000, stall_limit=200, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Dense + pair at parent (15% time)
    d2 = t0 + budget_s * 0.45
    h2 = _project(best_h.copy(), n0 / 2.0)
    h2, c5_2 = _pair_optimize(h2, d2, radius=0.12, n_pairs=50, beta=20.0, stall_limit=80, seed=7)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multiresolution exploration (25% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    tried_ns = set()
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, 24, -24, 32, -32]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _dense_optimize(h_r, sub_dl, beta_init=18.0, max_steps=10000, stall_limit=100, seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Binary diversification (15% time)
    d4 = t0 + budget_s * 0.85
    if time.time() < d4:
        n_curr = len(best_h)
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d4, time.time() + budget_s * 0.08)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.002, seed=777)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary
        if time.time() < d4:
            n_curr = len(best_h)
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d4, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _dense_optimize(h_pol, sub_dl2, beta_init=25.0, max_steps=8000, stall_limit=80, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Alternative seeds (10% time)
    d5 = t0 + budget_s * 0.95
    if time.time() < d5:
        for s_idx in range(4):
            if time.time() > d5:
                break
            h_seed = _generate_seed(n0, s_idx, seed)
            sub_dl = min(d5, time.time() + budget_s * 0.02)
            h_s, c5_s = _dense_optimize(h_seed, sub_dl, beta_init=12.0, max_steps=8000, stall_limit=120, seed=500 + s_idx)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 6: Final polish (5% time)
    d6 = deadline - 1.0
    if time.time() < d6:
        n_curr = len(best_h)
        h6 = _project(best_h.copy(), n_curr / 2.0)
        h6, c5_6 = _dense_optimize(h6, d6, beta_init=25.0, max_steps=10000, stall_limit=60, seed=888)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        n_curr = len(best_h)
        if time.time() < d6:
            h6b = _project(best_h.copy(), n_curr / 2.0)
            sub_dl = min(d6, deadline - 0.5)
            h6b, c5_6b = _pair_optimize(h6b, sub_dl, radius=0.08, n_pairs=40, beta=30.0, stall_limit=40, seed=999)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
