def _project(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_and_grad(h, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
    m = np.max(scores)
    z = (scores - m) / beta
    w = np.exp(z)
    sw = float(np.sum(w))
    value = m + beta * np.log(max(sw, 1e-300))
    w = w / sw
    
    # Compute gradient
    G = np.zeros(n)
    # d >= 0: output index d + n - 1
    for d in range(n):
        wi = w[d + n - 1]
        if wi < 1e-10:
            continue
        m_d = n - d
        G[d:] += wi * (1.0 - h[:m_d])
        G[:m_d] += wi * (-h[d:])

def _slsqp_solve(h0, beta, maxiter=100):
    """Solve using SLSQP with smooth objective."""
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        h_c = np.clip(h, 0.0, 1.0)
        return _smooth_obj_and_grad(h_c, beta)[0]
    
    def obj_grad(h):
        h_c = np.clip(h, 0.0, 1.0)
        return _smooth_obj_and_grad(h_c, beta)
    
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.success or res.x is not None:
        h_new = _project(res.x, target)
        return h_new
    return None

def _active_lp(h, scores, radius, n_active=15, gap=0.005):
    """Active-set LP to find improving direction."""
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    
    lags = active - (n - 1)
    # Build gradient matrix
    G = np.zeros((len(active), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    
    # LP: min tau s.t. G*d - tau <= -scores[active], sum(d)=0, bounds
    m_con = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_con, n + 1))
    A_ub[:, :n] = G * (2.0 / n)  # scale by dx
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    """Exact polynomial line search along direction d."""
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = t_lo
    hi = t_hi
    
    def f_at(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    # Initial bracket with grid
    grid_t = np.linspace(lo, hi, 40)
    grid_f = np.array([f_at(t) for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo = grid_t[max(0, best_idx - 2)]
    hi = grid_t[min(39, best_idx + 2)]
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f_at(c1)
    f2 = f_at(c2)
    
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f_at(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f_at(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_val = f_at(opt_t)
    
    # Check endpoints
    cands = [(opt_t, opt_val), (t_lo, f_at(t_lo)), (t_hi, f_at(t_hi))]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]

    dx = 2.0 / n

            if m > 0:

                G[idx, :m] += -h[d:]

            m = n - g
            if m > 0:

    if np.any(neg):

    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = t_hi
    
    def _f(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    # Grid initial bracket
    grid = np.linspace(lo, hi, 30)

    lo = grid[max(0, bi - 2)]
