import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _antimap(x, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _from_anti(h, n):
    m = n // 2
    x = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(x, 0.0, 1.0)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_full_weighted(h, w, n):
    G = np.zeros(n)
    u = 1.0 - h
    active = w > 1e-10
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m_ = n - tau
            if m_ > 0:
                G[tau:] += wr * u[:m_]
                G[:m_] -= wr * h[tau:tau + m_]
        else:
            d = -tau
            m_ = n - d
            if m_ > 0:
                G[:m_] += wr * u[d:]
                G[d:] -= wr * h[:m_]
    return G

def _smooth_obj(x, n, beta):
    h = _antimap(x, n)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return (2.0 / n) * M
    return (2.0 / n) * (M + np.log(wsum) / beta)

def _smooth_grad(x, n, beta):
    h = _antimap(x, n)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        wc = np.zeros(len(raw))
        wc[idx] = 1.0
        G = _grad_full_weighted(h, wc, n)
    else:
        w /= wsum
        G = _grad_full_weighted(h, w, n)
    m = n // 2
    return (G[:m] - G[m:][::-1]) * (2.0 / n)

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_lo
    hi = a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    # Also check the vertex of the quadratic
    top_idx = np.argsort(S0)[-5:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _full_local_opt(h, deadline, max_iter=2000):
    """Local optimization in full space using coordinate descent style line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    radius = 0.15
    for _ in range(max_iter):
        if time.time() > deadline or stall > 100:
            break
        # Random pair direction
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and abs(alpha) > 1e-10:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
            else:
                stall += 1
                radius = max(0.03, radius * 0.7)
        else:
            stall += 1
            radius = max(0.03, radius * 0.8)
    return best_h, best_c5

def _lbfgs_anti(x0, n, beta, deadline, max_iter=1000):
    """Run L-BFGS-B in anti-palindromic space."""
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    res = minimize(_smooth_obj, x0, method='L-BFGS-B', args=(n, beta),
                   jac=lambda x: _smooth_grad(x, n, beta),
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h = _antimap(res.x, n)
    # Repair any floating point sum error
    h = _project(h, n / 2.0)
    return h, _exact_c5(h)

def _resample_anti(h, n_new):
    """Resample a full h vector to a new even length, then extract anti-palindromic coords."""
    n_old = len(h)
    m_new = n_new // 2
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = _project(h_new, n_new / 2.0)
    return _from_anti(h_new, n_new)

def _perturb_anti(x, n, amp, rng):
    """Perturb anti-palindromic coordinates."""
    m = n // 2
    x_new = x.copy()
    # Smooth perturbation
    t = np.arange(m, dtype=float)
    for _ in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        x_new += (amp / 4.0) * np.sin(2.0 * np.pi * freq * t / m + phase)
    # Add small noise
    x_new += amp * 0.2 * rng.normal(0, 1, m)
    return np.clip(x_new, 0.0, 1.0)

def _to_binary(h):
    n = len(h)
    threshold = n // 2
    idx = np.argsort(h)[-threshold:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):

        else:

    theta = (lo + hi) / 2.0

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = a_lo
    hi = a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(55):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo + hi) / 2.0
    opt_v = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = float(np.max(S0 + av * T - av * av * Q)) * dx
                if fv < opt_v:

def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_anti(x, n, amp, rng):
    m = n // 2
    x_new = x.copy()
    t = np.arange(m, dtype=float)
    for _ in range(5):
        freq = rng.uniform(0.3, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        x_new += (amp / 5.0) * np.sin(2.0 * np.pi * freq * t / m + phase)
    x_new += amp * 0.15 * rng.normal(0, 1, m)
    return np.clip(x_new, 0.0, 1.0)

def _block_dirs(h, n, block_sizes):
    dirs = []
    for blk in block_sizes:
        if n % blk != 0:
            continue
        nblk = n // blk
        for start in range(0, nblk * blk, blk * 2):
            end = min(start + 2 * blk, n)
            d = np.zeros(n)
            mid = start + blk
            d[start:mid] = 1.0 / blk
            d[mid:end] = -1.0 / blk
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                dirs.append(d / dn)
    return dirs

        else:
            d = -tau
            m_ = n - d
            if m_ > 0:
                G[:m_] += u[d:]
                G[d:] -= h[:m_]
        G -= np.mean(G)
        gn = np.linalg.norm(G)
        if gn > 1e-12:
            dirs.append(G / gn)
    return dirs

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 += 1
        h0 = np.append(h0, 0.5)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: L-BFGS-B anti-palindromic at n0 with beta schedule (28%)
    d1 = t0 + budget_s * 0.28
    x_cur = _from_anti(h_init, n0)
    betas = [4.0, 10.0, 25.0, 60.0, 120.0, 200.0]
    for beta in betas:
        if time.time() > d1:
            break
        h_try, c5_try = _lbfgs_anti(x_cur, n0, beta, max_iter=2000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        x_cur = _from_anti(h_try, n0)

    # Phase 2: Anti-midpoint subspace at n0 (8%)
    d2 = t0 + budget_s * 0.36
    x_mid = _from_antimidge(h_init, n0)
    for beta in [10.0, 40.0, 100.0]:
        if time.time() > d2:
            break
        h_try, c5_try = _lbfgs_mid(x_mid, n0, beta, max_iter=1500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        x_mid = _from_antimidge(h_try, n0)

    # Phase 3: Multi-resolution L-BFGS-B (22%)
    d3 = t0 + budget_s * 0.58
    n_list = []
    for dn in [-24, -16, -8, 8, 16, 24, -32, 32, -40, 40]:
        nt = n0 + dn
        if 64 <= nt <= 260 and nt % 2 == 0:
            n_list.append(nt)
    for nt in n_list:
        if time.time() > d3:
            break
        h_r = _resample(best_h, nt)
        x_r = _from_anti(h_r, nt)
        for beta in [15.0, 50.0, 120.0]:
            if time.time() > d3:
                break
            h_try, c5_try = _lbfgs_anti(x_r, nt, beta, max_iter=1200)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            x_r = _from_anti(h_try, nt)
        if time.time() < d3 - 3:
            x_mid_r = _from_antimidge(h_r, nt)
            h_try, c5_try = _lbfgs_mid(x_mid_r, nt, 50.0, max_iter=800)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 4: Full-space local refinement (14%)
    d4 = t0 + budget_s * 0.72
    n_cur = len(best_h)
    h_ref = _project(best_h.copy(), n_cur / 2.0)
    h_ref, c5_ref = _full_local_opt(h_ref, min(d4, t0 + budget_s * 0.62), max_iter=4000, seed=777)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()
    if time.time() < d4:
        h_ref2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_ref2, c5_ref2 = _full_local_opt(h_ref2, d4, max_iter=3000, seed=888)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 5: Basin hopping in anti-palindromic space (18%)
    d5 = t0 + budget_s * 0.90
    n_hop = len(best_h)
    x_hop = _from_anti(best_h, n_hop)
    scale = 0.12
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 120:
        n_restarts += 1
        x_pert = _perturb_anti(x_hop, n_hop, scale, rng)
        h_try, c5_try = _lbfgs_anti(x_pert, n_hop, 80.0, max_iter=800)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            x_hop = _from_anti(h_try, n_hop)
            scale = max(0.008, scale * 0.82)
        else:
            scale = min(0.35, scale * 1.12)

    # Phase 6: Final multi-n + full-space polish (10%)
    d6 = deadline
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12, -16, 16]:
        if time.time() > d6 - 8:
            break
        nt = n_final + dn
        if nt < 64 or nt > 260 or nt % 2 != 0:
            continue
        h_r = _resample(best_h, nt)
        x_r = _from_anti(h_r, nt)
        h_try, c5_try = _lbfgs_anti(x_r, nt, 100.0, max_iter=1000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            n_final = nt
        if time.time() > d6 - 8:
            break
        hf = _project(h_try.copy(), nt / 2.0)
        hf, c5_f = _full_local_opt(hf, min(d6, time.time() + 4), max_iter=800, seed=555)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()
            n_final = nt

    # Final polish
    if deadline - time.time() > 8:
        n_f = len(best_h)
        x_f = _from_anti(best_h, n_f)
        h_f, c5_f = _lbfgs_anti(x_f, n_f, 150.0, max_iter=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if deadline - time.time() > 4:
            hf = _project(best_h.copy(), n_f / 2.0)
            hf, c5_f = _full_local_opt(hf, deadline, max_iter=1500, seed=999)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
