    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

def _grad_direction(h, scores, beta=15.0, radius=0.1):

    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))

        w /= sw
