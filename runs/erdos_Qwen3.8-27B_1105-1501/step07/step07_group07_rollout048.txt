            break

        for i in range(n):

            for j in range(n):
                if j == i:

    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _lse_objective(x):
    scores = np.correlate(x, 1.0 - x, mode='full') * _g_dx
    m = float(np.max(scores))
    u = (scores - m) / _g_tau
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum > 1e-300:
        return m + _g_tau * np.log(zsum)
    return m

def _lse_gradient(x):
    scores = np.correlate(x, 1.0 - x, mode='full') * _g_dx
    m = float(np.max(scores))
    u = (scores - m) / _g_tau
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum > 1e-300:
        w = z / zsum
    else:
        w = np.zeros(len(scores))
        idx = np.where(scores == m)[0]
        w[idx] = 1.0 / len(idx)
    G = _grad_matrix(x)
    return _g_dx * (G.T @ w)

def _eq_fun(x):
    return float(np.sum(x) - _g_target)

def _eq_jac(x):
    return np.ones(_g_n)
