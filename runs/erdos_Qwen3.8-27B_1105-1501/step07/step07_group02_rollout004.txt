Let me analyze the problem and the strategy carefully.

The key improvements I want to make over the previous program:

1. **LP-based multi-coordinate directions**: Instead of only pair moves, use linear programming to find directions that simultaneously reduce multiple active correlation peaks. This is the main algorithmic innovation.

2. **Exact quadratic line search along arbitrary directions**: For any direction d with sum(d)=0, compute the exact quadratic slice and do golden section minimization.

3. **Multi-resolution exploration**: Try different grid sizes around the parent.

4. **Binary diversification**: Use binary states to explore different structural optima.

Let me think about the implementation details:

- `_exact_c5(h)`: Compute `max(np.correlate(h, 1-h, mode='full')) * 2/n`
- `_project(v, target)`: Joint projection onto [0,1]^n ∩ {sum = target}
- `_grad_matrix(h)`: L×n matrix where G[r,i] is the gradient of R_r w.r.t. h_i
- `_line_search(h, u, t_cap)`: Exact 1D minimization along direction u
- `_lp_direction(h, peak_count)`: Solve LP to find a feasible descent direction
- `_local_descent(h, deadline)`: Main loop combining LP directions and pair moves

For the LP direction:
- Active set: top-k correlation peaks
- Variables: sp_i ≥ 0, sn_i ≥ 0 (split u = sp - sn)
- Constraints:
  - G[r,:] @ (sp - sn) + beta ≤ 0 for each active row r
  - sum(sp) - sum(sn) = 0 (zero-sum)
  - sum(sp + sn) = 1 (normalization)
  - 0 ≤ sp_i, sn_i ≤ 1
- Objective: minimize -beta (maximize beta)

The key insight is that beta > 0 means all active peaks decrease first-order.

For the line search along direction u:
- R0 = corr(h, 1-h)
- A = corr(u, 1-h) - corr(h, u)
- B = corr(u, u)
- f(t) = max_m [R0_m + t*A_m - t^2*B_m]
- Find optimal t in feasible interval

Let me think about computational costs:
- n = 180 (parent size), L = 359
- G matrix: 359×180, computed in O(n*L) ≈ 65k operations - very fast
- LP: ~8-30 rows, 2*n+2 variables, ~30+n constraints - solvable in milliseconds with HiGHS
- Line search: O(L) per evaluation, ~100 evaluations - fast
- Total per iteration: maybe 1-5ms

So we can do many iterations within the time budget.

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    G = np.zeros((L, n))
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        G[:, i] = np.correlate(e, o, mode='full') - np.correlate(h, e, mode='full')
    return G

def _line_search(h, u, R0, t_cap=1.0):
    """Minimize max(R0 + t*A - t^2*B) over feasible t."""
    n = len(h)
    t_min = -t_cap
    t_max = t_cap
    for i in range(n):
        ui = u[i]
        if abs(ui) < 1e-15:
            continue
        if ui > 0:
            t_min = max(t_min, -h[i] / ui)
            t_max = min(t_max, (1.0 - h[i]) / ui)
        else:
            t_min = max(t_min, (1.0 - h[i]) / ui)
            t_max = min(t_max, -h[i] / ui)
    t_min += 1e-12
    t_max -= 1e-12
    if t_min >= t_max:
        return None
    o = 1.0 - h
    A = np.correlate(u, o, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')
    
    def f(t):
        return float(np.max(R0 + t * A - t * t * B))
    
    # Grid search
    n_grid = 100
    ts = np.linspace(t_min, t_max, n_grid)
    vals = np.array([f(t) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    candidates = [t_min, t_max, (lo + hi) / 2.0, ts[best_idx]]
    best_t = min(candidates, key=f)
    return best_t

def _lp_direction(G_active, h, peak_count):
    """Solve LP to find descent direction on active peaks."""
    n = h.shape[0]
    m = G_active.shape[0]
    # Variables: sp (n), sn (n), beta (1)
    #Objective: minimize -beta
    # Constraints:
    #   G_active[r,:] @ (sp - sn) + beta <= 0  for r=0..m-1
    #   sum(sp) - sum(sn) = 0
    #   sum(sp) + sum(sn) = 1
    #   0 <= sp_i <= 1, 0 <= sn_i <= 1
    #   beta free (bounded)
    
    N = 2 * n + 1  # sp, sn, beta
    c = np.zeros(N)
    c[-1] = -1.0  # minimize -beta
    
    # Inequality constraints: G[r,:] @ (sp - sn) + beta <= 0
    # => G[r,:] @ sp - G[r,:] @ sn + beta <= 0
    A_ub = np.zeros((m, N))
    for r in range(m):
        A_ub[r, :n] = G_active[r, :]   # sp coefficients
        A_ub[r, n:2*n] = -G_active[r, :]  # sn coefficients
        A_ub[r, -1] = 1.0  # beta coefficient
    b_ub = np.zeros(m)
    
    # Equality constraints:
    # sum(sp) - sum(sn) = 0
    # sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, N))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(-1e4, 1e4)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res is not None and res.status == 0:
        sp = res.x[:n]
        sn = res.x[n:2*n]
        u = sp - sn
        beta = res.x[-1]
        if beta > 1e-8:
            # Normalize to unit L1
            l1 = float(np.sum(np.abs(u)))
            if l1 > 1e-10:
                u = u / l1
            return u, beta
    return None, -1.0

def _pair_move(h, p, q, R0, max_t=1.0):
    """Exact line search for pair move h[p]-=t, h[q]+=t."""
    u = np.zeros(len(h))
    u[q] = 1.0
    u[p] = -1.0
    t_opt = _line_search(h, u, R0, t_cap=max_t)
    if t_opt is None or abs(t_opt) < 1e-13:
        return None
    h_new = h.copy()
    h_new[q] += t_opt
    h_new[p] -= t_opt
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum
    s = float(np.sum(h_new))
    target = len(h) / 2.0
    if abs(s - target) > 1e-10:
        h_new = _project(h_new, target)
    return h_new

def _local_descent(h, deadline, initial_radius=0.3, seed=42):
    """Main local descent: LP directions + pair moves."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    peak_count = 8
    radius = initial_radius
    
    while time.time() < deadline:
        if stall > 200 or radius < 1e-6:
            break
        
        R0 = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(R0))

        top_k = min(peak_count, 2 * n - 1)
        peak_idx = np.argsort(R0)[-top_k:]

        G = _grad_matrix(h)
        G_active = G[peak_idx, :]

        u_lp, beta = _lp_direction(G_active, h, top_k)
        accepted = False

                c_try = _exact_c5(h_try)

                    h = h_try
                    cur_c5 = c_try

            imp_indices = np.argsort(coord_grad)
            # Pick top few sources (high gradient) and sinks (low gradient)

                    if improve > best_pair_improve:

            if best_pair_h is not None:

    return best_h, best_c5

def _line_search_full(h, u, R0, t_cap):
    """Full line search returning the new h."""
    t_opt = _line_search(h, u, R0, t_cap=t_cap)
    if t_opt is None or abs(t_opt) < 1e-14:
        return None
    h_new = h + t_opt * u
    if np.any(h_new < -1e-10) or np.any(h_new > 1 + 1e-10):
        return None
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

            cur_c5 = new_c5
            ones[i] = oz

            if cur_c5 < best_c5 - 1e-14:

                best_h = cur.copy()
                stall = 0

            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            T = max(T * 0.9995, 0.0001)

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    o = 1.0 - h
    L = 2 * n - 1
    G = np.zeros((L, n))
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        G[:, i] = np.correlate(e, o, mode='full') - np.correlate(h, e, mode='full')
    return G

def _line_search(h, u, R0, t_cap=1.0):
    n = len(h)
    t_min = -t_cap
    t_max = t_cap
    for i in range(n):
        ui = u[i]
        if abs(ui) < 1e-15:
            continue
        if ui > 0:
            t_min = max(t_min, -h[i] / ui)
            t_max = min(t_max, (1.0 - h[i]) / ui)
        else:
            t_min = max(t_min, (1.0 - h[i]) / ui)
            t_max = min(t_max, -h[i] / ui)
    t_min += 1e-12
    t_max -= 1e-12
    if t_min >= t_max:
        return None
    o = 1.0 - h
    A = np.correlate(u, o, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')

    def f(t):
        return float(np.max(R0 + t * A - t * t * B))

    n_grid = 120
    ts = np.linspace(t_min, t_max, n_grid)
    vals = np.empty(n_grid)
    for idx in range(n_grid):
        vals[idx] = f(ts[idx])
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-14:
        return float(ts[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    candidates = [t_min, t_max, (lo + hi) / 2.0, ts[best_idx]]
    best_t = t_min
    best_val = f(t_min)
    for ct in candidates:
        cv = f(ct)
        if cv < best_val:
            best_val = cv
            best_t = ct
    return float(best_t)

def _line_search_full(h, u, R0, t_cap):
    t_opt = _line_search(h, u, R0, t_cap=t_cap)
    if t_opt is None or abs(t_opt) < 1e-14:
        return None
    h_new = h + t_opt * u
    if np.any(h_new < -1e-10) or np.any(h_new > 1.0 + 1e-10):
        return None
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _lp_direction(G_active, h):
    n = h.shape[0]
    m = G_active.shape[0]
    N = 2 * n + 1
    c = np.zeros(N)
    c[-1] = -1.0
    A_ub = np.zeros((m, N))
    for r in range(m):
        A_ub[r, :n] = G_active[r, :]
        A_ub[r, n:2*n] = -G_active[r, :]
        A_ub[r, -1] = 1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((2, N))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(-1e4, 1e4)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res is not None and res.status == 0:
            sp = res.x[:n]
            sn = res.x[n:2*n]
            u = sp - sn
            beta = res.x[-1]
            if beta > 1e-8:
                l1 = float(np.sum(np.abs(u)))
                if l1 > 1e-10:
                    u = u / l1
                return u, beta
    except Exception:
        pass
    return None, -1.0

def _pair_move(h, p, q, R0, max_t):
    n = len(h)
    u = np.zeros(n)
    if h[p] > h[q]:
        u[p] = -1.0
        u[q] = 1.0
    else:
        u[p] = 1.0
        u[q] = -1.0
    t_opt = _line_search(h, u, R0, t_cap=max_t)
    if t_opt is None or abs(t_opt) < 1e-13:
        return None
    h_new = h + t_opt * u
    if np.any(h_new < -1e-10) or np.any(h_new > 1.0 + 1e-10):
        return None
    return np.clip(h_new, 0.0, 1.0)

def _local_descent(h, deadline, initial_radius=0.3, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    peak_count = 8
    radius = initial_radius
    lp_fail_count = 0

    while time.time() < deadline:
        if stall > 250 or radius < 1e-7:
            break
        R0 = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(peak_count, 2 * n - 1)
        peak_idx = np.argsort(R0)[-top_k:]
        G = _grad_matrix(h)
        G_active = G[peak_idx, :]

        accepted = False
        u_lp, beta = _lp_direction(G_active, h)
        if u_lp is not None:
            h_try = _line_search_full(h, u_lp, R0, radius)
            if h_try is not None:
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-13:
                    h = h_try
                    cur_c5 = c_try
                    accepted = True
                    lp_fail_count = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        else:
            lp_fail_count += 1

        if not accepted:
            coord_grad = G_active.mean(axis=0)
            imp_indices = np.argsort(coord_grad)
            npair = min(40, n // 2)
            best_improve = 0.0
            best_h_try = None
            for _ in range(npair):
                p = int(imp_indices[rng.integers(0, n)])
                q = int(imp_indices[rng.integers(0, n)])
                if p == q:
                    continue
                h_try = _pair_move(h, p, q, R0, max_t=radius)
                if h_try is not None:
                    c_try = _exact_c5(h_try)
                    improve = cur_c5 - c_try
                    if improve > best_improve:
                        best_improve = improve
                        best_h_try = h_try
            if best_h_try is not None:
                h = best_h_try
                cur_c5 = _exact_c5(h)
                accepted = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

        if accepted:
            h = _project(h, target)
            stall = 0
            radius = min(radius * 1.015, 0.45)
            if rng.random() < 0.08:
                peak_count = int(rng.integers(4, 30))
        else:
            stall += 1
            radius *= 0.84
            if stall > 80:
                peak_count = int(rng.integers(4, 35))
                if lp_fail_count > 20:
                    lp_fail_count = 0
            if stall % 60 == 0 and radius < 0.04:
                radius = 0.18
                pert = rng.normal(0, 0.02, size=n)
                h = _project(h + pert, target)
                cur_c5 = _exact_c5(h)

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=60000, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    stall = 0
    T = 0.0025
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 350:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < -1e-14:
            cur = h_new
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T = max(T * 0.9996, 0.0001)
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            T = max(T * 0.9996, 0.0001)
        else:
            stall += 1
    return best_h, best_c5
