def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _slope_vector(h, j):
    n = len(h)
    center = n - 1
    k = np.arange(2 * n - 1)
    idx1 = j + k - center
    idx2 = j - k + center
    g = np.zeros(2 * n - 1)
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _minimize_pwl1d(corr, d, lo_t, hi_t):
    """Minimize max(corr + t*d) for t in [lo_t, hi_t] using slope bisection."""
    if hi_t - lo_t < 1e-14:
        return (lo_t + hi_t) / 2.0
    for _ in range(45):
        mid = (lo_t + hi_t) / 2.0
        vals = corr + mid * d
        k_max = int(np.argmax(vals))
        if d[k_max] < -1e-11:
            lo_t = mid
        elif d[k_max] > 1e-11:
            hi_t = mid
        else:
            break
    opt_t = (lo_t + hi_t) / 2.0
    cands = [0.0, opt_t, lo_t, hi_t]

def _ls_exp_smoothed(h, deadline, alpha_init=80.0, alpha_min=20.0, max_iters=5000, seed=77):

            break
        if stall > 200:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Log-sum-exp of corr values (in raw units)
        corr_max = float(np.max(corr))
        exp_vals = np.exp(alpha * (corr - corr_max))
        weights = exp_vals / float(np.sum(exp_vals))

        # = sum_k weights[k] * slope_j[k]
        # Compute gradient for all j at once using correlation
        # grad[j] = sum_k weights[k] * (1-h[j+k-center])*[valid] - h[j-k+center]*[valid]
        # This is equivalent to: grad = correlate(weights, 1-h, 'full') - correlate(h, weights, 'full') ... 
        # Actually let me just compute it directly
        grad = np.zeros(n)
        for k_idx in range(len(weights)):
            if weights[k_idx] < 1e-10:
                continue
            k_shift = k_idx - center

            if k_shift >= 0:
                m = n - k_shift
                grad[:m] += w_k * (1.0 - h[k_shift:k_shift+m])
                grad[k_shift:] -= w_k * h[:m]
            else:
                gg = -k_shift
                m = n - gg
                grad[gg:] += w_k * (1.0 - h[:m])
                grad[:m] -= w_k * h[gg:gg+m]
        
        # Line search: h_new = h - lr * grad, projected
        best_step = 0.0
        best_f = float(np.max(corr))
        for lr_candidate in [lr, lr*0.5, lr*0.25, lr*2.0]:
            h_try = h - lr_candidate * grad
            h_try = _project(h_try, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            f_try = float(np.max(corr_try))

            alpha = min(alpha * 1.02, 200.0)
        else:

            lr *= 0.9
            alpha = max(alpha * 0.98, alpha_min)

            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.70
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.06
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    n_freqs = 4
    for _ in range(n_freqs):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude / n_freqs
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_optimize(h_bin, deadline, max_iters=30000, seed=999):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.002
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = int(ones[i])
        oz = int(zeros[j])
        
        # Compute change in correlation from swapping oi and oz
        # d[h] = -1 at oi, +1 at oz
        d = np.zeros(n)
        d[oi] = -1.0
        d[oz] = 1.0
        T_corr = np.correlate(d, 1.0 - cur, mode='full') - np.correlate(cur, d, mode='full')
        new_corr = corr + T_corr
        new_c5 = float(np.max(new_corr)) * dx
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            corr = new_corr
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T = max(T * 0.9998, 0.0001)
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Quick pair-swap CD (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _pair_swap_cd(best_h, d1, max_iters=80000, seed=111)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: LP refinement (12%)
    d2 = t0 + budget_s * 0.24
    h2, c5_2 = _lp_refine(best_h, d2, radius_init=0.08, stall_limit=50, top_k=12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Intensive pair-swap CD (20%)
    d3 = t0 + budget_s * 0.44
    h3, c5_3 = _pair_swap_cd(best_h, d3, max_iters=200000, seed=222)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: LS-exp smoothed gradient (12%)
    d4 = t0 + budget_s * 0.56
    h4, c5_4 = _ls_exp_smoothed(best_h, d4, alpha_init=60.0, max_iters=4000, seed=333)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Basin hopping (16%)
    d5 = t0 + budget_s * 0.72
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 - 3 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        # Quick pair-swap on perturbed
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.005)
        h_t, c5_t = _pair_swap_cd(h_pert, sub_dl, max_iters=8000, seed=5000 + n_restarts)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        elif n_restarts % 5 == 0:
            scale = min(0.08, scale * 1.15)
    
    # Phase 6: Binary optimization + polish (10%)
    d6 = t0 + budget_s * 0.82
    h_bin = _to_binary(best_h)
    sub_dl = min(d6, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_optimize(h_bin, sub_dl, max_iters=40000, seed=444)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d6 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl2 = min(d6, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _lp_refine(h_pol, sub_dl2, radius_init=0.05, stall_limit=40, top_k=12)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 7: Multi-n exploration (10%)
    d7 = t0 + budget_s * 0.92
    tried_ns = set()
    for dn in [-20, -10, -6, -4, -2, 2, 4, 6, 10, 20]:
        if time.time() > d7:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = _pair_swap_cd(h_r, sub_dl, max_iters=15000, seed=7000 + dn)
        if time.time() < d7 - 2 and c5_r < best_c5 + 0.005:
            sub_dl2 = min(d7, time.time() + budget_s * 0.003)
            h_r, c5_r2 = _lp_refine(h_r, sub_dl2, radius_init=0.04, stall_limit=20, top_k=10)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 8: Final intensive polishing (8%)
    d8 = deadline
    sub_dl = min(d8, time.time() + budget_s * 0.035)
    h8, c5_8 = _pair_swap_cd(best_h, sub_dl, max_iters=100000, seed=9999)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline - 5:
        sub_dl2 = min(deadline - 2, time.time() + budget_s * 0.03)
        h8, c5_8 = _lp_refine(best_h, sub_dl2, radius_init=0.03, stall_limit=40, top_k=15)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    # Final coord descent pass
    if time.time() < deadline - 3:
        sub_dl3 = min(deadline - 1, time.time() + budget_s * 0.015)
        h8, c5_8 = _pair_swap_cd(best_h, sub_dl3, max_iters=50000, seed=12345)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    
    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
