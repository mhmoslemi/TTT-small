```python
def _dense_direction_line_search(h, S0, G, d, t_lo=-1.0, t_hi=1.0):
    """Line search along direction d from point h.
    S0: current overlaps (unscaled, L-vector)
    G: gradient matrix (L x n)
    d: direction vector (n-vector, should have sum ~ 0)
    Returns optimal t and the value of max overlap at that t.
    """
    n = len(h)
    dx = 2.0 / n
    T = G @ d  # linear coefficient for each lag
    C = -np.correlate(d, d, mode='full')  # quadratic coefficient for each lag
    
    # Minimize max_k [S0_k + t*T_k + t²*C_k] * dx
    # Since dx is a constant positive factor, we can minimize the unscaled version
    # and multiply by dx at the end.
    
    # Find t that minimizes max_k [S0_k + t*T_k + t²*C_k]
    # Use golden section search on t
    
    def obj(t):
        return float(np.max(S0 + t * T + t * t * C))
    
    # First, find a good bracket
    # Try a grid
    ts = np.linspace(t_lo, t_hi, 50)
    vals = np.array([obj(t) for t in ts])
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(49, best_idx + 2)]
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_val = obj(opt_t)
    
    # Also check the grid best
    best_val = min(opt_val, float(vals[best_idx]))
    if best_val == float(vals[best_idx]):
        opt_t = float(ts[best_idx])
    
    return opt_t, best_val * dx
```

```python
def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n-1), n):
        row = k + n - 1
        if k == 0:
            G[row, :] = 1.0 - 2.0 * h
        elif k > 0:
            m = n - k
            G[row, :m] += (1.0 - h[k:])
            G[row, k:] -= h[:m]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

        return t_lo, float(np.max(S0))

    ts = np.linspace(t_lo, t_hi, n_grid)

    best_idx = int(np.argmin(vals))

    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
```
