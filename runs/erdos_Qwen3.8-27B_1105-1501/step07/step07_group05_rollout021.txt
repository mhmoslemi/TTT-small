def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    corr = _compute_corr(h)
    # corr has length 2n-1, index m corresponds to lag m-(n-1)
    top_idx = np.argsort(-corr)[:M]

def _compute_phi(h, l):
    """Compute phi_l(x) for all x efficiently using numpy."""
    n = len(h)
    phi = np.zeros(n)
    if l >= 0:
        # (1-h[x+l]) for x in [0, n-1-l]
        phi[:n-l] += (1.0 - h[l:])
        # -h[x-l] for x in [l, n-1]
        phi[l:] -= h[:n-l]
    else:
        s = -l

        phi[s:] += (1.0 - h[:n-s])
        # -h[x-l] = -h[x+s] for x in [0, n-1-s]
        phi[:n-s] -= h[s:]
    return phi

import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_phi(h, l):
    n = len(h)
    phi = np.zeros(n)
    if l >= 0:
        if l < n:
            phi[:n-l] += (1.0 - h[l:])
        if l > 0:
            phi[l:] -= h[:n-l]
    else:
        s = -l
        if s < n:
            phi[s:] += (1.0 - h[:n-s])
            phi[:n-s] -= h[s:]
    return phi

def _top_phi(h, M=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:M]
    min_phi = np.full(n, np.inf)
    max_phi = np.full(n, -np.inf)
    for m in top_idx:
        l = m - (n - 1)
        phi = _compute_phi(h, l)
        min_phi = np.minimum(min_phi, phi)
        max_phi = np.maximum(max_phi, phi)
    return min_phi, max_phi

def _binary_lops(hb, deadline, M=30, stall_limit=300):
    """Top-lag-aware binary local search."""
    n = len(hb)
    h = hb.copy()
    cur = _exact_c5(h)
    best = h.copy()
    best_c = cur
    ones_mask = h > 0.5
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline and stall < stall_limit:
        min_phi, max_phi = _top_phi(h, M)
        
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Greedy: best removal (lowest min_phi among ones) + best addition (lowest max_phi among zeros)
        best_i = ones_idx[np.argmin(min_phi[ones_idx])]
        best_j = zeros_idx[np.argmin(max_phi[zeros_idx])]
        
        # With some probability, use random instead
        if rng.random() < 0.1:
            best_i = ones_idx[rng.integers(len(ones_idx))]
            best_j = zeros_idx[rng.integers(len(zeros_idx))]
        
        # Try swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        ones_mask[best_i] = False
        ones_mask[best_j] = True
        new_c = _exact_c5(h)
        
        if new_c < cur - 1e-12:
            cur = new_c
            if cur < best_c:
                best_c = cur
                best = h.copy()
            stall = 0
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            ones_mask[best_i] = True
            ones_mask[best_j] = False
            stall += 1
    
    return best, best_c

def _cont_lops(h, deadline, M=20, stall_limit=200):
    """Top-lag-aware continuous local search using pair moves."""
    n = len(h)
    target = n / 2.0
    h = _project(h, target)
    cur = _exact_c5(h)
    best = h.copy()
    best_c = cur
    rng = np.random.default_rng(67890)
    stall = 0
    
    while time.time() < deadline and stall < stall_limit:
        min_phi, max_phi = _top_phi(h, M)
        
        # Find best pair: i with h[i]>0 and low min_phi, j with h[j]<1 and low max_phi
        # Score for removal: -min_phi[i] (want high = low min_phi)
        # Score for addition: -max_phi[j] (want high = low max_phi)
        
        pos_mask = h > 0.01
        neg_mask = h < 0.99
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        
        # Greedy selection
        best_i = pos_idx[np.argmin(min_phi[pos_idx])]
        best_j = neg_idx[np.argmin(max_phi[neg_idx])]
        
        if rng.random() < 0.2:
            best_i = pos_idx[rng.integers(len(pos_idx))]
            best_j = neg_idx[rng.integers(len(neg_idx))]
        
        if best_i == best_j:
            stall += 1
            continue
        
        # Line search: move δ from i to j
        # h_new = h - δ*e_i + δ*e_j
        # Constraint: 0 ≤ h[i]-δ ≤ 1 and 0 ≤ h[j]+δ ≤ 1
        # So 0 ≤ δ ≤ min(h[i], 1-h[j])
        d = np.zeros(n)
        d[best_i] = -1.0
        d[best_j] = 1.0
        
        delta_max = min(h[best_i], 1.0 - h[best_j])
        if delta_max < 1e-10:
            stall += 1
            continue
        
        # Binary search for best δ
        # The objective as function of δ: max_l C(l) + δ*(φ_l(j) - φ_l(i))
        # φ_l(j) - φ_l(i) is fixed for this pair
        # Let dv = φ(j) - φ(i) for each lag
        # We want to minimize max_l (C(l) + δ * dv(l))
        
        # Compute dv for top lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(-corr)[:M]
        
        # For each top lag, compute φ_l(j) - φ_l(i)
        # φ_l(x) = (1-h[x+l])*I(valid) - h[x-l]*I(valid)
        # Δφ_l = φ_l(j) - φ_l(i)
        
        deltas = np.zeros(len(top_lags))
        for k, m in enumerate(top_lags):
            l = m - (n - 1)
            phi = _compute_phi(h, l)
            deltas[k] = phi[best_j] - phi[best_i]
        
        # Now minimize max_k (corr[top_lags[k]] + δ * deltas[k])
        # subject to 0 ≤ δ ≤ delta_max
        
        # The optimal δ is where the maximum is minimized
        # This is a 1D convex problem (max of linear functions)
        # Use ternary search / bisection
        
        lo = 0.0
        hi = delta_max
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(corr[top_lags] + m1 * deltas)) * (2.0/n)
            f2 = float(np.max(corr[top_lags] + m2 * deltas)) * (2.0/n)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        delta = (lo + hi) / 2.0
        
        if delta < 1e-10:
            stall += 1
            continue
        
        h_new = h.copy()
        h_new[best_i] -= delta
        h_new[best_j] += delta
        h_new = _project(h_new, target)
        new_c = _exact_c5(h_new)
        
        if new_c < cur - 1e-12:
            h = h_new
            cur = new_c
            if cur < best_c:
                best_c = cur
                best = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb
