import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute G[r, j] = d(corr[r])/d(h[j]) CORRECTLY."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_mh = 1.0 - h
    for k in range(-(n-1), n):
        r = k + n - 1
        if k >= 0:
            m = n - k
            G[r, k:] += one_mh[:m]   # (1-h[j-k]) for j in [k, n-1]
            G[r, :m] -= h[k:]         # -h[j+k] for j in [0, m-1]
        else:
            p = -k
            m = n - p
            G[r, :m] += one_mh[p:]    # (1-h[j+p]) for j in [0, m-1]
            G[r, p:] -= h[:m]         # -h[j-p] for j in [p, n-1]
    return G

def _line_search(S0, T, C, t_lo, t_hi):
    """Minimize max_k(S0[k] + t*T[k] - t^2*C[k]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_t = None
    best_val = float('inf')
    # Try a few points first
    for t_test in [t_lo, t_hi, (t_lo+t_hi)/2.0]:
        v = float(np.max(S0 + t_test * T - t_test * t_test * C))
        if v < best_val:
            best_val = v
            best_t = t_test
    # Golden section
    lo, hi = t_lo, t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_val = opt_val
        best_t = opt_t
    return best_t, best_val

def _smooth_weights(S0, beta=20.0):
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    shifted = beta * (S0 - M)
    # Use stable log-sum-exp
    max_s = np.max(shifted)
    w = np.exp(shifted - max_s)
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        n_s = min(n_pairs, n // 2)
        sources = sort_idx[-n_s:]
        sinks = sort_idx[:n_s]
        pairs = []
        for ii in range(n_s):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        # Add some random pairs
        for _ in range(min(10, n_pairs // 5)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo_bound = max(-h[sink], h[source] - 1.0, -radius)
            t_hi_bound = min(1.0 - h[sink], h[source], radius)
            if t_hi_bound - t_lo_bound < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(S0, T, C, t_lo_bound, t_hi_bound)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.02, 0.28)
        else:
            stall += 1
            radius *= 0.78
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

Now for the NLP approach. Let me think about how to implement it efficiently.

For SLSQP with active constraints:
- Variables: [h_0, ..., h_{n-1}, t] (n+1 variables)
- Objective: min t
- Equality: sum(h) = n/2
- Inequalities: t - corr[r]*dx ≥ 0 for selected r

The Jacobian of the objective: [0, 0, ..., 0, 1]
The Jacobian of the equality: [1, 1, ..., 1, 0]
The Jacobian of inequality r: [-dx*G[r,:], 1]

For SLSQP, I need to provide the constraint functions and their Jacobians.

Let me think about the constraint selection. I'll use an iterative approach:
1. Start with top K constraints (K = 20-40)
2. Solve
3. Check all constraints
4. Add any violated ones
5. Re-solve if violations found
6. Repeat until no violations (or time limit)

For the initial point, I'll use the current best h and t = C₅(h) + small margin.

Let me also consider: should I use `minimize` with `constraints` as a list of dicts, or should I reformulate?

SLSQP can handle a list of constraints. Each constraint dict has 'type', 'fun', and optionally 'jac'.

For n=180 and K=30 constraints, the problem has 181 variables and 31 constraints. SLSQP should handle this in a few seconds.

```python
def _nlp_solve(h_init, n, deadline, max_iter=50, K=30, seed=42):
    """Solve the epigraph NLP using SLSQP with active constraint management."""
    dx = 2.0 / n
    target = n / 2.0
    h_init = _project(h_init.copy(), target)
    cur_c5 = _exact_c5(h_init)
    best_h = h_init.copy()
    best_c5 = cur_c5
    
    # Active constraint indices
    corr = np.correlate(h_init, 1.0 - h_init, mode='full')
    active = np.argsort(corr)[::-1][:K]
    
    for outer_iter in range(5):
        if time.time() > deadline:
            break
        
        # Define objective
        def obj(x):
            return x[-1]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        
        # Define constraints
        G = _grad_matrix(h_init)  # Will be recomputed each time in jac
        active_list = list(active)
        
        def eq_fun(x):
            return np.sum(x[:-1]) - target
        
        def eq_jac(x):
            j = np.ones(n + 1)
            j[-1] = 0.0
            return j
        
        # For SLSQP, each inequality constraint is a scalar function
        # We'll pack them into a single vector constraint
        
        def ineq_fun(x):
            h_x = x[:-1]
            t_x = x[-1]
            cx = np.correlate(h_x, 1.0 - h_x, mode='full')
            # Only compute for active constraints
            return t_x - cx[active_list] * dx
        
        def ineq_jac(x):
            h_x = x[:-1]
            G_x = _grad_matrix(h_x)
            J = np.zeros((len(active_list), n + 1))
            for i, r in enumerate(active_list):
                J[i, :n] = -dx * G_x[r, :]
                J[i, n] = 1.0
            return J
        
        constraints = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        ]
        # SLSQP supports vector-valued inequality constraints
        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
        
        bounds = [(0, 1)] * n + [(0, 1)]
        x0 = np.zeros(n + 1)
        x0[:n] = h_init
        x0[n] = cur_c5 + 0.005
        
        try:
            res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                          constraints=constraints, method='SLSQP',
                          options={'maxiter': max_iter, 'ftol': 1e-8})
            if res.success or res.fun < cur_c5:
                h_new = res.x[:n]
                t_new = res.x[n]
                # Project to feasible
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_init = h_new.copy()
                    cur_c5 = c5_new
                    # Update active set
                    corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                    active = np.argsort(corr)[::-1][:max(K, len(active_list))]
        except Exception:
            pass
        
        # Check for violated inactive constraints
        corr_full = np.correlate(h_init, 1.0 - h_init, mode='full')
        t_val = cur_c5
        violations = np.where(corr_full * dx > t_val + 1e-10)[0]
        if len(violations) > 0:
            # Add violated constraints to active set
            active = np.union1d(active, violations)
            active = active[:max(K*2, len(active))]  # Cap the size
        else:
            break
    
    return best_h, best_c5

The gradient is:
∇f_β(h) = Σ_r p_r * dx * G[r, :]
where p_r = exp(β * corr[r] * dx) / Σ exp(β * corr[r] * dx)

This is exactly the smoothed weighted sum of gradients, which is what the pair optimization uses for direction selection.

Full gradient descent:
- Direction: d = -∇f_β(h) (projected to sum=0)
- Line search along d
- Project to [0,1]^n with sum = n/2

```python
def _gradient_descent(h, deadline, beta=30.0, seed=42):
    """Minimize smoothed max using gradient descent with projection."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.time() < deadline:
        if stall > 50:
            break
        S0 = np.correlate(h, 1.0 - h, mode='full')
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        grad = G.T @ w  # Shape: (n,)
        
        # Project gradient to sum=0 subspace
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Normalize
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-10:
            break
        direction = -grad_centered / grad_norm
        
        # Line search with projection
        best_t = 0.0
        best_val = cur_c5
        for t_test in [0.05, 0.1, 0.2, 0.01, 0.5, 0.001]:
            h_try = _project(h + t_test * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_t = t_test
        
        if best_t == 0.0:
            # Try smaller steps
            for t_test in [0.005, 0.002, 0.001, 0.0005]:
                h_try = _project(h + t_test * direction, target)
                c5_try = _exact_c5
