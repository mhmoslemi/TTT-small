import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Use bisection on the threshold
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_grad(h, lag):
    """Compute gradient of correlation row at given lag index (full mode index)."""
    n = len(h)
    ell = lag - (n - 1)  # convert full index to signed lag
    g = np.zeros(n)
    if ell >= 0:
        # g[p] = (1-h[p-ell]) for ell <= p <= n-1, minus h[p+ell] for 0 <= p <= n-1-ell
        m = n - ell
        if m > 0:
            g[ell:] += (1.0 - h[:m])
            g[:m] -= h[ell:]
    else:
        p = -ell
        m = n - p
        if m > 0:
            g[:m] += (1.0 - h[p:])
            g[p:] -= h[:m]
    return g

def _dense_direction(h, S, kags, target_val, weights):
    """
    Solve weighted least-norm for dense direction.
    kags: list of full-correlation indices
    target_val: target value for selected rows
    weights: per-coordinate weights (positive)
    Returns zero-sum direction d (or None if failed).
    """
    n = len(h)
    K = len(kags)
    A = np.zeros((K + 1, n))
    b = np.zeros(K + 1)
    for i, rk in enumerate(kags):
        A[i] = _row_grad(h, rk)
        b[i] = target_val - S[rk]
    A[K] = 1.0 / np.sqrt(n)  # sum constraint (normalized)
    b[K] = 0.0
    
    # Weighted least-norm: d = W^{-1} A^T (A W^{-1} A^T + ridge*I)^{-1} b
    winv = 1.0 / np.maximum(weights, 1e-10)
    AW = A * winv[None, :]  # K+1 x N
    M = AW @ A.T  # (K+1) x (K+1)
    ridge = 1e-8 * np.trace(M) / (K + 1)
    M += ridge * np.eye(K + 1)
    try:
        lam = np.linalg.solve(M, b)
    except np.linalg.LinAlgError:
        try:
            lam = np.linalg.lstsq(M, b, rcond=None)[0]
        except:
            return None
    d = winv * (AW.T @ lam)
    # Ensure zero sum
    d -= np.mean(d)
    # Normalize
    maxabs = np.max(np.abs(d))
    if maxabs < 1e-12:
        return None
    d /= maxabs
    return d

def _line_search(h, S, d, move_cap=0.5):
    """Exact line search along direction d. Returns (t_opt, predicted_val)."""
    n = len(h)
    # Compute U and B
    U = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo, t_hi = -move_cap, move_cap
    for i in range(n):
        if abs(d[i]) < 1e-15:
            continue
        if d[i] > 0:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S))
    
    t_lo += 1e-10
    t_hi -= 1e-10
    if t_hi < t_lo:
        return 0.0, float(np.max(S))
    
    # Coarse grid
    n_grid = 25
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Predicted values: max(S + t*U - t^2*B)
    vals = np.max(S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine with golden section
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1*c1 * B))
    f2 = float(np.max(S + c2 * U - c2*c2 * B))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1*c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2*c2 * B))
    
    t_opt = (lo + hi) / 2.0
    opt_val = float(np.max(S + t_opt * U - t_opt*t_opt * B))
    
    # Also check grid best and endpoints
    candidates = [(float(ts[best_idx]), float(vals[best_idx])),
                  (t_opt, opt_val),
                  (t_lo, float(np.max(S + t_lo*U - t_lo*t_lo*B))),
                  (t_hi, float(np.max(S + t_hi*U - t_hi*t_hi*B)))]
    best_t = min(candidates, key=lambda x: x[1])
    return best_t[0], best_t[1]

    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(22):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    t_opt = (lo + hi) / 2.0
    opt_val = float(np.max(S + t_opt * U - t_opt * t_opt * B))
    cands = [(float(ts[best_idx]), float(vals[best_idx])), (t_opt, opt_val),
             (t_lo, float(np.max(S + t_lo * U - t_lo * t_lo * B))),
             (t_hi, float(np.max(S + t_hi * U - t_hi * t_hi * B)))]
    best = min(cands, key=lambda x: x[1])
    return best[0]

def _refine_dense(h, deadline, move_cap, k_schedule, beta_soft, seed):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    stall = 0
    attempt = 0
    while time.time() < deadline:
        if stall > 120 or move_cap < 0.01:
            break
        L = 2 * n - 1
        sorted_idx = np.argsort(S)[::-1]
        improved = False
        while not improved and time.time() < deadline:
            k_idx = attempt % len(k_schedule)
            K = k_schedule[k_idx]
            K = min(K, L)
            selected = sorted_idx[:K].tolist()
            S_sel = S[selected]
            s_min = float(np.min(S_sel))
            s_max = float(np.max(S_sel))
            s_mean = float(np.mean(S_sel))
            s_range = s_max - s_min
            target_modes = [s_min, s_mean - 0.3 * s_range, s_min + 0.1 * s_range, s_max - 0.15 * s_range]
            t_val = target_modes[attempt % 4]
            w_mode = attempt % 2
            d = _dense_dir(h, S, selected, t_val, w_mode)
            if d is not None:
                t_opt = _line_search(h, S, d, move_cap)
                if abs(t_opt) > 1e-10:
                    h_try = h + t_opt * d
                    if np.all(h_try >= -1e-9) and np.all(h_try <= 1.0 + 1e-9):
                        h_try = np.clip(h_try, 0.0, 1.0)
                        c5_try = _exact_c5(h_try)
                        if c5_try < cur_c5 - 1e-13:
                            h = _project(h_try, target)
                            cur_c5 = _exact_c5(h)
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            S = np.correlate(h, 1.0 - h, mode='full')
                            stall = 0
                            improved = True
                            attempt += 1
                            continue
            attempt += 1
            if attempt % 15 == 0:
                d_soft = _soft_dir(h, S, beta_soft)
                if d_soft is not None:
                    t_opt = _line_search(h, S, d_soft, move_cap * 0.5)
                    if abs(t_opt) > 1e-10:
                        h_try = h + t_opt * d_soft
                        if np.all(h_try >= -1e-9) and np.all(h_try <= 1.0 + 1e-9):
                            h_try = np.clip(h_try, 0.0, 1.0)
                            c5_try = _exact_c5(h_try)
                            if c5_try < cur_c5 - 1e-13:
                                h = _project(h_try, target)
                                cur_c5 = _exact_c5(h)
                                if cur_c5 < best_c5:
                                    best_c5 = cur_c5
                                    best_h = h.copy()
                                S = np.correlate(h, 1.0 - h, mode='full')
                                stall = 0
                                improved = True
                                attempt += 1
                                continue
            if attempt % 20 == 0:
                d_rand = rng.standard_normal(n)
                d_rand -= np.mean(d_rand)
                rm = float(np.max(np.abs(d_rand)))
                if rm > 1e-12:
                    d_rand /= rm
                    t_opt = _line_search(h, S, d_rand, move_cap * 0.3)
                    if abs(t_opt) > 1e-10:
                        h_try = h + t_opt * d_rand
                        if np.all(h_try >= -1e-9) and np.all(h_try <= 1.0 + 1e-9):
                            h_try = np.clip(h_try, 0.0, 1.0)
                            c5_try = _exact_c5(h_try)
                            if c5_try < cur_c5 - 1e-13:
                                h = _project(h_try, target)
                                cur_c5 = _exact_c5(h)
                                if cur_c5 < best_c5:
                                    best_c5 = cur_c5
                                    best_h = h.copy()
                                S = np.correlate(h, 1.0 - h, mode='full')
                                stall = 0
                                improved = True
                                attempt += 1
                                continue
            if attempt > 80:
                break
        if not improved:
            stall += 1
            move_cap *= 0.85
            if stall % 30 == 0 and move_cap < 0.05:
                move_cap = 0.12
            attempt = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_swap_sweep(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    rounds = 0
    while rounds < max_rounds and time.time() < deadline:
        rounds += 1
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
            if time.time() > deadline:
                break
        if not improved:
            break
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        sw = float(np.sum(w))
        if sw < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0 / top_k
        else:
            w /= sw
        n_eff = min(n, 256)
        G = np.zeros((2 * n - 1, n_eff))
        for k in range(-(n - 1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                G[row, k:n_eff - (n_eff - k) if k < n_eff else 0] += 0
                G[row, k:n] if k < n_eff else None
                G[row, k:min(n, n_eff) - 0] += (1.0 - h[:m]) if k + m <= n_eff else 0
            else:
                p = -k
                m = n - p
                G[row, p:min(n, n_eff)] += (1.0 - h[:m])
                G[row, 0:min(n, n_eff) - p if n_eff > p else 0] -= h[p:] if p < n_eff else 0
        # Simplified: just compute coordinate gradients efficiently
        coord_grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-6:
                coord_grad += w[r] * _row_grad(h, r)
        sort_idx = np.argsort(coord_grad)
        n_sel = min(n_pairs, n // 3)
        sources = sort_idx[-n_sel:]
        sinks = sort_idx[:n_sel]
        pairs = []
        for ii in range(n_sel):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(max(3, n_pairs // 5)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_lo2, t_hi2 = t_lo, t_hi
            ts = np.linspace(t_lo2, t_hi2, 15)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
            bi = int(np.argmin(vals))
            t_opt = float(ts[bi])
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    safety = deadline - 3.0
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        n0 = 180
        h0 = 0.5 + 0.3 * np.sin(np.pi * np.arange(n0) / (n0 / 2.0))
        h0 = np.clip(h0, 0, 1)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense direction refinement at parent n (35% time)
    d1 = t0 + budget_s * 0.35
    k_sched = [5, 8, 3, 12, 6, 10, 4, 15, 7, 9]
    h1, c5_1 = _refine_dense(h_init, d1, move_cap=0.6, k_schedule=k_sched, beta_soft=15.0, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (30% time)
    d2 = t0 + budget_s * 0.65
    if time.time() < d2:
        offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20]
        for dn in offsets:
            if time.time() > d2:
                break
            n_try = n0 + dn
            if n_try < 80 or n_try > 250 or n_try in (n0,):
                if abs(dn) == 0:
                    continue
            if n_try < 60:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(d2, time.time() + budget_s * 0.02)
            h_r, c5_r = _refine_dense(h_r, sub_dl, move_cap=0.5, k_schedule=k_sched, beta_soft=18.0, seed=200 + abs(dn))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 3: Binary exploration (15% time)
    d3 = t0 + budget_s * 0.80
    if time.time() < d3:
        n_curr = len(best_h)
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d3, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _pair_swap_sweep(h_bin, sub_dl, max_rounds=30)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish binary with continuous
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
            sub_dl2 = min(d3, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl2, radius=0.12, n_pairs=50, beta=22.0, stall_limit=60, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Intensive dense refinement on best (15% time)
    d4 = t0 + budget_s * 0.95
    if time.time() < d4:
        n_curr = len(best_h)
        h4 = _project(best_h.copy(), n_curr / 2.0)
        k_sched2 = [4, 6, 8, 5, 10, 3, 12, 7]
        h4, c5_4 = _refine_dense(h4, d4, move_cap=0.4, k_schedule=k_sched2, beta_soft=20.0, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Final polish (last 5%)
    d5 = safety
    if time.time() < d5:
        n_curr = len(best_h)
        h5 = _project(best_h.copy(), n_curr / 2.0)
        h5, c5_5 = _refine_dense(h5, d5, move_cap=0.2, k_schedule=[3, 5, 4, 6], beta_soft=25.0, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
