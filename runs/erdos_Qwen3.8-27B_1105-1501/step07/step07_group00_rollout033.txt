```python
def _binary_swap_update(counts, r, s, zeros, ones, n):
    """Apply swap r(1→0), s(0→1) to counts array. Returns new counts."""
    nc = counts.copy()
    # Phase A: r goes 1→0
    # For each z in zeros: counts[idx(r-z)] -= 1
    idxs = (r - zeros) + (n - 1)
    nc[idxs] -= 1
    # For each o in ones, o≠r: counts[idx(o-r)] += 1
    mask = ones != r
    idxs2 = (ones[mask] - r) + (n - 1)
    nc[idxs2] += 1
    
    # Phase B: s goes 0→1
    # New zeros = (zeros - {s}) ∪ {r}
    # New ones = ones - {r}
    zeros_new = np.concatenate([zeros[zeros != s], [r]])
    ones_new = ones[ones != r]
    
    idxs3 = (s - zeros_new) + (n - 1)
    nc[idxs3] += 1
    idxs4 = (ones_new - s) + (n - 1)
    nc[idxs4] -= 1
```

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binary_counts(b, n):
    return np.correlate(b.astype(np.int32), (1 - b).astype(np.int32), mode='full')

def _apply_swap(counts, r, s, zeros_idx, ones_idx, n):
    nc = counts.copy()
    nc[(r - zeros_idx) + (n - 1)] -= 1
    mask = ones_idx != r
    nc[(ones_idx[mask] - r) + (n - 1)] += 1
    mask_z = zeros_idx != s
    zeros_new = np.concatenate([zeros_idx[mask_z], np.array([r], dtype=np.int32)])
    nc[(s - zeros_new) + (n - 1)] += 1
    nc[(ones_idx[mask] - s) + (n - 1)] -= 1
    return nc

def _avg_pool(pool, k, n):
    if len(pool) == 0:
        return None
    top = sorted(pool, key=lambda x: x[1])[:min(k, len(pool))]
    h = np.zeros(n, dtype=float)
    for b, _ in top:
        h += b.astype(float)
    h /= len(top)
    return h

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] -= h[:m]
    return G

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 50)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(49, best_idx + 2)]

    c2 = lo_b + gr * (hi_b - lo_b)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
