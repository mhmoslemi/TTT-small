def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_mh = 1.0 - h
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        G[:, i] = np.correlate(e, one_mh, mode='full') - np.correlate(h, e, mode='full')
    return G

def _feasible_interval(h, u):
    """Find [t_min, t_max] such that 0 <= h + t*u <= 1 for all i."""
    n = len(h)
    t_min = -np.inf
    t_max = np.inf
    pos = u > 1e-14
    neg = u < -1e-14
    if np.any(pos):
        t_min = max(t_min, float(np.max(-h[pos] / u[pos])))
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / u[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((1.0 - h[neg]) / u[neg])))
        t_max = min(t_max, float(np.min(-h[neg] / u[neg])))
    return t_min, t_max

def _exact_line_search(h, u, t_cap=1.0):
    """Minimize max R(t) where R(t) = R0 + t*A - t^2*B along direction u."""
    n = len(h)
    dx = 2.0 / n
    t_min, t_max = _feasible_interval(h, u)
    t_min = max(t_min, -t_cap)
    t_max = min(t_max, t_cap)
    if t_max - t_min < 1e-14:
        return h.copy(), False
    
    R0 = np.correlate(h, 1.0 - h, mode='full')
    A = np.correlate(u, 1.0 - h, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')
    
    # Grid search
    n_grid = 100
    ts = np.linspace(t_min, t_max, n_grid)
    vals = np.array([float(np.max(R0 + t * A - t * t * B)) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    
    def obj(t):
        return float(np.max(R0 + t * A - t * t * B))
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    t_opt = (lo + hi) / 2.0
    t_opt = float(np.clip(t_opt, t_min, t_max))
    
    h_new = h + t_opt * u
    h_new = _project(h_new, n / 2.0)
    
    # Verify
    c5_new = _exact_c5(h_new)
    c5_old = _exact_c5(h)
    if c5_new < c5_old - 1e-12:
        return h_new, True
    return h.copy(), False

def _solve_lp_direction(G_active, n):
    """Solve LP to find direction u that maximizes minimum first-order reduction of active peaks."""
    k = G_active.shape[0]
    # Variables: sp (n), sn (n), beta (1)
    n_var = 2 * n + 1
    # Objective: minimize -beta → c vector with -1 at position 2n
    c = np.zeros(n_var)
    c[2 * n] = -1.0
    
    # Inequality constraints: G_active[r,:] @ u + beta <= 0 for each r
    # u = sp - sn
    # G[r,:] @ sp - G[r,:] @ sn + beta <= 0
    A_ub = np.zeros((k, n_var))
    for r in range(k):
        A_ub[r, :n] = G_active[r, :]
        A_ub[r, n:2*n] = -G_active[r, :]
        A_ub[r, 2*n] = 1.0
    b_ub = np.zeros(k)
    
    # Equality constraints:
    # sum(sp) - sum(sn) = 0
    # sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, n_var))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])

    bounds = []
    for _ in range(2 * n):

    bounds.append((-1e6, 1e6))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    u = sp - sn
    
    # Check numerical quality
    if np.max(np.abs(u)) < 1e-12:
        return None
    if abs(float(np.sum(u))) > 1e-8:
        u = u - float(np.sum(u)) / n  # Project to zero-sum
    
    return u

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_mh = 1.0 - h
    for i in range(n):
        k1s = max(0, n - 1 - i)
        k1e = min(L - 1, 2 * (n - 1) - i)
        if k1s <= k1e:
            idx = k1s - n + 1 + i
            cnt = k1e - k1s + 1
            G[k1s:k1e+1, i] = one_mh[idx:idx+cnt]
        k2s = max(0, i)
        k2e = min(L - 1, i + n - 1)
        if k2s <= k2e:
            idx_start = i - k2s + n - 1
            idx_end = i - k2e + n - 1
            G[k2s:k2e+1, i] -= h[idx_end:idx_start+1][::-1]
    return G

def _line_search(h, u, t_cap=1.0):
    n = len(h)
    t_min, t_max = -np.inf, np.inf
    pos = u > 1e-14
    neg = u < -1e-14
    if np.any(pos):
        t_min = max(t_min, float(np.max(-h[pos] / u[pos])))
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / u[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((1.0 - h[neg]) / u[neg])))
        t_max = min(t_max, float(np.min(-h[neg] / u[neg])))
    t_min = max(t_min, -t_cap)
    t_max = min(t_max, t_cap)
    if t_max - t_min < 1e-14:
        return h.copy(), _exact_c5(h)
    
    R0 = np.correlate(h, 1.0 - h, mode='full')
    A = np.correlate(u, 1.0 - h, mode='full') - np.correlate(h, u, mode='full')
    B = np.correlate(u, u, mode='full')
    
    n_grid = 120
    ts = np.linspace(t_min, t_max, n_grid)
    # Evaluate max(R0 + t*A - t^2*B) for each t
    best_t = t_min
    best_val = np.inf
    for t in ts:
        v = float(np.max(R0 + t * A - t * t * B))
        if v < best_val:
            best_val = v
            best_t = t
    
    # Golden section around best grid point
    best_idx = int(np.argmin([float(np.max(R0 + t * A - t * t * B)) for t in ts]))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    
    t_opt = (lo + hi) / 2.0
    t_opt = float(np.clip(t_opt, t_min, t_max))
    
    h_new = h + t_opt * u
    h_new = _project(h_new, n / 2.0)
    c5_new = _exact_c5(h_new)
    return h_new, c5_new

def _lp_direction(G_active, n):
    k = G_active.shape[0]
    n_var = 2 * n + 1
    c = np.zeros(n_var)
    c[2 * n] = -1.0
    
    A_ub = np.zeros((k, n_var))
    for r in range(k):
        A_ub[r, :n] = G_active[r, :]
        A_ub[r, n:2*n] = -G_active[r, :]
        A_ub[r, 2*n] = 1.0
    b_ub = np.zeros(k)
    
    A_eq = np.zeros((2, n_var))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(-10.0, 10.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res is None or not res.success:
        return None
    u = res.x[:n] - res.x[n:2*n]
    if np.max(np.abs(u)) < 1e-12:
        return None
    u_sum = float(np.sum(u))
    if abs(u_sum) > 1e-10:
        u = u - u_sum / n
    return u

def _pair_direction(h, G, w, n):
    """Find best pair direction from weighted gradient."""
    coord_grad = G.T @ w
    sort_idx = np.argsort(coord_grad)
    best_pair = None
    best_score = np.inf
    n_check = min(30, n // 3)
    sources = sort_idx[-n_check:]
    sinks = sort_idx[:n_check]
    for si in sinks:
        for so in sources:
            si = int(si)
            so = int(so)
            if si == so:

            d = np.zeros(n)
            d[si] = 1.0
            d[so] = -1.0
            score = G @ d
            max_score = float(np.max(score))
            if max_score < best_score:
                best_score = max_score
                best_pair = d
    return best_pair

        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_mh = 1.0 - h
    for i in range(n):
        k1s = max(0, n - 1 - i)
        k1e = min(L - 1, 2 * (n - 1) - i)
        if k1s <= k1e:
            idx = k1s - n + 1 + i
            cnt = k1e - k1s + 1
            G[k1s:k1e + 1, i] = one_mh[idx:idx + cnt]
        k2s = max(0, i)
        k2e = min(L - 1, i + n - 1)
        if k2s <= k2e:
            idx_start = i - k2s + n - 1
            idx_end = i - k2e + n - 1
            G[k2s:k2e + 1, i] -= h[idx_end:idx_start + 1][::-1]
    return G

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    # Also try notably different sizes
    for n_alt in [n0 - 20, n0 - 40, n0 + 20, n0 + 40]:
        if time.time() > d2:
            break
        if n_alt < 80 or n_alt > 320 or n_alt in tried_ns:
            continue
        tried_ns.add(n_alt)
        h_r = _resample(best_h, n_alt)
        sub_dl = min(d2, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_descent(h_r, sub_dl, seed=500 + n_alt, t_cap_init=0.8, stall_limit=50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d3, time.time() + budget_s * 0.09)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, T_init=0.006, max_iter=80000, seed=777)
        sub_dl2 = min(d3, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_swap_local(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Continuous polish from binary
        if time.time() < d3:
            n_b = len(h_bin)
            h_pol = _project(h_bin.copy(), n_b / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _lp_descent(h_pol, sub_dl3, seed=333, t_cap_init=0.4, stall_limit=50)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    # Phase 4: Final intensive polish (10% time)
    d4 = deadline - 3
    n_curr = len(best_h)
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        h4, c5_4 = _lp_descent(h4, d4, seed=555, t_cap_init=0.5, stall_limit=150)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        # Quick check at nearby n
        for dn in [-4, 4]:
            if time.time() > d4:
                break
            n_try = n_curr + dn
            if n_try < 80:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(d4, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_descent(h_r, sub_dl, seed=700 + dn, t_cap_init=0.4, stall_limit=40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
