def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj(x):

    x = np.clip(x, 0.0, 1.0)

    mval = float(np.max(s))

    max_u = float(np.max(u))

        zsum = float(np.sum(z))
        p = z / zsum

    gvec = _g_dx * (_g_G.T @ p)

    return val

def _pair_sweep(h, deadline, max_rounds=20):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0

                    break
                else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)

    top_idx = np.argsort(scores)[::-1][:top_k]
    top_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.1, top_k_init=4, max_top_k=10, stall_limit=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _raw_scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.2)
            if stall == 0 and top_k > top_k_init:
                top_k = max(top_k_init, top_k - 1)
        else:
            stall += 1
            radius *= 0.75
            if stall % 25 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _pair_swap_loop(h, deadline, n_pairs=5000, radius=0.25, seed=111):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * d, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = (len(h0) // 2) * 2
    if n0 != len(h0):
        h0 = _resample(h0, n0)
    n0 = len(h0)

    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE annealing (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lse_anneal(h_init, d1, n_stages=7, tau_init=0.04)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish after LSE (10%)
    d2 = t0 + budget_s * 0.35
    h2 = _project_sum(best_h.copy(), len(best_h) // 2)
    h2, c5_2 = _lp_loop(h2, d2, radius_init=0.08, top_k_init=3, max_top_k=8, stall_limit=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA + pair sweep (15%)
    d3 = t0 + budget_s * 0.50
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003, seed=42)
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=25)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d3:
        sub_dl3 = min(d3, time.time() + budget_s * 0.03)
        h_pol = _project_sum(h_bin.copy(), len(h_bin) / 2.0)
        h_pol, c5_pol = _lp_loop(h_pol, sub_dl3, radius_init=0.05, top_k_init=3, max_top_k=7, stall_limit=30)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.70
    n_best = len(best_h)
    n_offsets = list(range(-20, 21, 4))
    tried_ns = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 240 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_r, c5_r = _lse_anneal(h_r, sub_dl, n_stages=4, tau_init=0.03)
        if time.time() < d4 - 2:
            sub_dl2 = min(d4, time.time() + budget_s * 0.006)
            h_r, c5_r2 = _lp_loop(h_r, sub_dl2, radius_init=0.05, top_k_init=3, max_top_k=7, stall_limit=25)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r
        if c5_r < best_c5 + 0.001 and time.time() < d4 - 2:
            h_b = _to_binary(h_r)
            sub_dl3 = min(d4, time.time() + budget_s * 0.005)
            h_b, c5_b = _binary_sa(h_b, sub_dl3, max_iter=15000, T_init=0.002, seed=n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 5: Intensive local search (15%)
    d5 = t0 + budget_s * 0.85
    n_cur = len(best_h)
    # Pair swap
    if time.time() < d5:
        h4 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=6000, radius=0.20, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary SA on current best
    if time.time() < d5:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=35000, T_init=0.002, seed=555)
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Tight LP
    if time.time() < d5:
        h4 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h4, c5_4 = _lp_loop(h4, sub_dl, radius_init=0.03, top_k_init=2, max_top_k=5, stall_limit=40)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 6: Basin hopping (7%)
    d6 = t0 + budget_s * 0.92
    scale = 0.04
    n_restarts = 0
    while time.time() < d6 and n_restarts < 150:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d6, time.time() + budget_s * 0.003)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=6, stall_limit=20)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.08)

    # Phase 7: Final polish (remaining)
    d7 = deadline
    n_best = len(best_h)
    for dn in [-4, 4, -8, 8, -2, 2]:
        if time.time() > d7 - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=6, stall_limit=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final LSE
    if time.time() < d7 - 10:
        sub_dl = min(d7, time.time() + budget_s * 0.02)
        h7, c5_7 = _lse_anneal(best_h, sub_dl, n_stages=4, tau_init=0.02)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    # Final LP
    if time.time() < d7 - 3:
        h7 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        h7, c5_7 = _lp_loop(h7, d7, radius_init=0.02, top_k_init=2, max_top_k=5, stall_limit=30)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    # Final pair sweep
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, d7, max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
