    if dn > 1e-12:
        dirs.append(d / dn)

    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    mu = momentum_init  # momentum coefficient
    vel = np.zeros(n)  # velocity for momentum
    rng = np.random.default_rng(12345)
    total_iter = 0
    
    while time.time() < deadline:
        total_iter += 1
        if stall > max_stall:
            break

        frac = min(1.0, float(stall) / float(max_stall))
        beta = beta_lo + (beta_hi - beta_lo) * frac

        raw = np.correlate(h, u, mode='full')

        g = _smooth_grad(h, raw, beta)
        
        # Apply momentum
        vel = mu * vel - g
        vel -= np.mean(vel)  # maintain zero-sum
        
        # Generate directions
        dirs = _make_directions(h, vel, raw, dx, rng, radius, n_dirs)
        
        # Precompute for line search
        S0 = raw
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        
        for d in dirs:
            if time.time() > deadline:
                break
            T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
            Q = np.correlate(d, d, mode='full')
            a_lo, a_hi = _alpha_bounds(h, d, radius)
            alpha, val = _line_search_quad(h, d, S0, T, Q, a_lo, a_hi, dx)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        
        if best_d is not None and abs(best_alpha) > 1e-10:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
                mu = min(mu * 1.01, 0.6)
            else:

                radius = max(0.05, radius * 0.8)
                mu *= 0.95
        else:
            stall += 1
            radius = max(0.05, radius * 0.8)
            mu *= 0.9
    
    return best_h, best_c5

import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, tau):
    n = len(h)
    g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:tau+m] += (1.0 - h[:m])
            g[:m] -= h[tau:tau+m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])
            g[d:d+m] -= h[:m]
    g -= np.mean(g)
    return g

def _top_k_weighted_grad(h, raw, k=5):
    n = len(h)
    top_idx = np.argsort(raw)[-k:]
    vals = raw[top_idx]
    mx = float(np.max(vals))
    v = 10.0 * (vals - mx)
    w = np.exp(np.clip(v, -50.0, 0.0))
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return _lag_grad(h, top_idx[0] - (n-1))
    w /= wsum
    g = np.zeros(n)
    for i, r in enumerate(top_idx):
        tau = r - (n - 1)
        g += w[i] * _lag_grad(h, tau)
    g -= np.mean(g)
    return g

def _smooth_weighted_grad(h, raw, beta, max_active=20):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return _lag_grad(h, int(np.argmax(raw)) - (n-1))
    w /= wsum
    # Keep only top max_active by weight
    if np.sum(w > 0) > max_active:
        threshold = np.sort(w)[-max_active]
        w[w < threshold] = 0.0
        w /= np.sum(w)
    g = np.zeros(n)
    active = w > 1e-8
    for r in np.where(active)[0]:
        tau = int(r) - (n - 1)
        g += w[r] * _lag_grad(h, tau)
    g -= np.mean(g)
    return g

def _alpha_bounds(h, d, radius):
    a_lo, a_hi = -radius, radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    return a_lo, a_hi

def _line_search(S0, T, Q, a_lo, a_hi, dx):
    if a_hi <= a_lo + 1e-14:
        return 0.0, float(np.max(S0) * dx)
    def f(a):
        return float(np.max(S0 + a * T - a*a * Q)) * dx
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi - lo < 1e-12: break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo); f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo); f2 = f(c2)
    opt_a, opt_v = (lo+hi)/2.0, f((lo+hi)/2.0)
    # Analytic minimizers for near-max lags
    cur = S0
    mx = float(np.max(cur))
    for r in np.where(cur > mx - 0.1*(mx+1e-8))[0]:
        if Q[r] > 1e-12:
            a_s = T[r] / (2.0 * Q[r])
            if a_lo <= a_s <= a_hi:
                fv = f(a_s)
                if fv < opt_v: opt_v, opt_a = fv, a_s
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v: opt_v, opt_a = ft, t
    return opt_a, opt_v

def _gen_dirs(h, g, raw, rng, n_dirs=6):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn; d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12: dirs.append(d/dn)
    top3 = np.argsort(raw)[-3:]
    for r in top3:
        gs = _lag_grad(h, int(r) - (n-1))
        sn = np.linalg.norm(gs)
        if sn > 1e-12:
            d = gs/sn; d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn > 1e-12: dirs.append(d/dn)
    # Random sparse
    k = max(2, n//12)
    idx = rng.choice(n, size=k, replace=False)
    d = np.zeros(n)
    d[idx[:k//2]] = 1.0; d[idx[k//2:]] = -1.0
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12: dirs.append(d/dn)
    # Block flip
    blk = max(3, n//6)
    st = int(rng.integers(0, n-blk))
    d = np.zeros(n)
    d[st:st+blk] = 1.0
    others = np.setdiff1d(np.arange(n), range(st, st+blk))
    ni = min(blk, len(others))
    d[others[rng.choice(len(others), ni, replace=False)]] = -1.0
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12: dirs.append(d/dn)
    return dirs[:n_dirs]

def _opt_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, rad_init=0.15,
                max_stall=60, mom_init=0.3, n_dir=6):
    n = len(h); target = n/2.0; dx = 2.0/n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    rad, stall, mu, total = rad_init, 0, mom_init, 0
    vel = np.zeros(n)
    rng = np.random.default_rng(54321)
    while time.time() < deadline and stall <= max_stall:
        total += 1
        frac = min(1.0, float(stall)/float(max_stall))
        beta = beta_lo + (beta_hi - beta_lo) * frac
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        if stall < max_stall // 2:
            g = _smooth_weighted_grad(h, raw, beta, max_active=20)
        else:
            g = _top_k_weighted_grad(h, raw, k=min(5, n))
        vel = mu * vel - g
        vel -= np.mean(vel)
        dirs = _gen_dirs(h, vel, raw, rng, n_dir)
        S0 = raw
        b_alpha, b_val, b_d = 0.0, cur_c5, None
        for d in dirs:
            if time.time() > deadline: break
            T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
            Q = np.correlate(d, d, mode='full')
            a_lo, a_hi = _alpha_bounds(h, d, rad)
            a, v = _line_search(S0, T, Q, a_lo, a_hi, dx)
            if v < b_val - 1e-14:
                b_val, b_alpha, b_d = v, a, d
        if b_d is not None and abs(b_alpha) > 1e-10:
            h_n = _project(h + b_alpha * b_d, target)
            nc = _exact_c5(h_n)
            if nc < cur_c5 - 1e-14:
                h = h_n; cur_c5 = nc
                if nc < best_c5: best_c5, best_h = nc, h.copy()
                stall = 0; rad = min(rad*1.05, 0.4); mu = min(mu*1.01, 0.55)
            else:

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(v, tgt):
    n = len(v)
    if tgt <= 0: return np.zeros(n)
    if tgt >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0, 1)) > tgt: lo = mid
        else: hi = mid
    th = (lo + hi) * 0.5
    x = np.clip(v - th, 0, 1)
    r = tgt - np.sum(x)
    if abs(r) > 1e-12:
        m = x < 1.0 if r > 0 else x > 0.0
        c = m.sum()
        if c > 0:
            x = x.copy(); x[m] += r / c
            np.clip(x, 0, 1, out=x)
    return x

def _lg(h, tau):
    n = len(h); g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:tau+m] += 1.0 - h[:m]
            g[:m] -= h[tau:tau+m]
    else:
        d = -tau; m = n - d
        if m > 0:
            g[:m] += 1.0 - h[d:d+m]
            g[d:d+m] -= h[:m]
    g -= g.mean()
    return g

def _sg(h, raw, beta, k=15):
    """Smooth-max gradient: top-k weighted"""
    n = len(h)
    mx = raw.max()
    v = np.clip(beta * (raw - mx), -50, 0)
    w = np.exp(v); s = w.sum()
    if s < 1e-30: return _lg(h, int(np.argmax(raw)) - n + 1)
    w /= s
    if (w > 0).sum() > k:
        thr = np.sort(w)[-k]; w[w < thr] = 0; w /= w.sum()
    g = np.zeros(n)
    for r in np.where(w > 1e-8)[0]:
        g += w[r] * _lg(h, int(r) - n + 1)
    g -= g.mean()
    return g

    a0, a1 = -rad, rad
    p = d > 1e-12; n = d < -1e-12

    return a0, a1

    def fv(a): return float((S0 + a*T - a*a*Q).max()) * dx

        else: lo,c1,f1 = c1,c2,f2; c2=lo+gr*(hi-lo); f2=fv(c2)

def _dirs(h, g, raw, rng, nd=6):

    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g/gn; d -= d.mean(); dn = np.linalg.norm(d)
        if dn > 1e-12: D.append(d/dn)

        if sn > 1e-12:

    d -= d.mean(); dn = np.linalg.norm(d)
