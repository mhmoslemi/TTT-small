    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr


def _directed_binary_search(h_bin, deadline, max_iters=200000, seed=123):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_c5_fast(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    temp = 0.003
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Directed: sample from active lags
        if it % 3 == 0:
            # Find active lags and bias sampling
            max_val = float(cur_corr.max())
            active = np.where(cur_corr >= max_val - dx * 0.1)[0]
            if len(active) > 0:
                am = int(active[int(rng.integers(0, len(active)))])
                k = am - (n - 1)
                if k > 0 and k < n:
                    # Sample from the overlap region of this lag
                    cand_ones = [oz for oz in zeros if oz >= k and oz < n]
                    cand_zeros = [oi for oi in ones if oi >= 0 and oi < n - k]
                    if cand_ones and cand_zeros:
                        oz = cand_ones[int(rng.integers(0, len(cand_ones)))]
                        oi = cand_zeros[int(rng.integers(0, len(cand_zeros)))]
                    else:
                        oz = zeros[int(rng.integers(0, len(zeros)))]
                        oi = ones[int(rng.integers(0, len(ones)))]
                else:
                    oz = zeros[int(rng.integers(0, len(zeros)))]
                    oi = ones[int(rng.integers(0, len(ones)))]
            else:
                oz = zeros[int(rng.integers(0, len(zeros)))]
                oi = ones[int(rng.integers(0, len(ones)))]
        else:
            oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5, _ = _binary_c5_fast(trial)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or (delta < 0.002 * cur_c5 and rng.random() < np.exp(-delta / max(temp, 1e-15))):
            cur = trial
            cur_c5 = new_c5
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
            oi_idx = ones.index(oi)
            oz_idx = zeros.index(oz)
            ones[oi_idx] = oz
            zeros[oz_idx] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                temp = max(0.0005, temp * 0.998)
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def _block_swap_binary(h_bin, deadline, max_iters=20000, seed=456):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(10, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        b1 = cur[i:i + length].copy()
        b2 = cur[j:j + length].copy()
        if int(np.sum(b1)) != int(np.sum(b2)):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i + length] = b2
        trial[j:j + length] = b1
        new_c5 = _exact_c5(trial)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _pair_swap_continuous(h, deadline, n_pairs=5000, radius=0.20, seed=321):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        # Simple line search
        best_alpha = 0.0
        best_val = cur
        for t in np.linspace(0, radius, 15):
            trial = _project(h + t * d, target)
            v = _exact_c5(trial)
            if v < best_val:
                best_val = v
                best_alpha = t
        if best_val < cur - 1e-13:
            h = _project(h + best_alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _gradient_descent_smooth(h, deadline, tau_init=0.02, max_iter=500, seed=789):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    tau = tau_init
    lr = 0.05
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        val, corr, p = _smooth_core(h, tau, n)
        g = _smooth_grad_from_p(h, p, n)
        # Remove mean to maintain sum constraint approximately
        g = g - np.mean(g)
        # Backtracking line search
        best_alpha = 0.0
        best_c5_try = cur
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a = lr * frac
            trial = _project(h - a * g, n / 2.0)
            v = _exact_c5(trial)
            if v < best_c5_try:
                best_c5_try = v
                best_alpha = a
        if best_alpha > 0:
            h = _project(h - best_alpha * g, n / 2.0)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.7
            tau *= 0.95
            if lr < 1e-6:
                break
    return best_h, best_c5


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + ph)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)

    # Initialize from provided values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous optimization (12% of budget)
    d1 = t0 + budget_s * 0.12
    # Try L-BFGS-B with a few tau values
    tau_grid = [0.03, 0.015, 0.008, 0.004]
    for tau in tau_grid:
        if time.time() > d1:
            break
        h_try, c5_try = _lgbfsb_solve(h_init, tau, n0, deadline=d1, maxiter=200, penalty=800.0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_init = h_try.copy()
        h_init = _project(best_h, n0 / 2.0)

    # Gradient descent polish
    if time.time() < d1:
        h_try, c5_try = _gradient_descent_smooth(h_init, d1, tau_init=0.01, max_iter=300)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: Intensive binary optimization (38%)
    d2 = t0 + budget_s * 0.50
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Multiple rounds of directed binary search with different seeds
    sub_tasks = [
        (d2 * 0.30, 50),
        (d2 * 0.30, 51),
        (d2 * 0.20, 52),
    ]
    for frac, s in sub_tasks:
        sub_dl = min(d2, time.time() + budget_s * frac)
        h_bin, c5_bin = _directed_binary_search(h_bin, sub_dl, max_iters=150000, seed=s)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_bin = _to_binary(best_h, n0) if c5_bin >= best_c5 else h_bin

    # Block swaps
    if time.time() < d2 - 5:
        sub_dl = min(d2, time.time() + budget_s * 0.06)
        h_bin, c5_bin = _block_swap_binary(h_bin, sub_dl, max_iters=15000, seed=456)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Another directed search round
    if time.time() < d2 - 5:
        h_bin = _to_binary(best_h, n0)
        sub_dl = min(d2, time.time() + budget_s * 0.06)
        h_bin, c5_bin = _directed_binary_search(h_bin, sub_dl, max_iters=100000, seed=67)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Continuous polish on best
    if time.time() < d2 - 3:
        h_cont = _project(best_h.copy(), n0 / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_cont, c5_cont = _lgbfsb_solve(h_cont, 0.005, n0, deadline=sub_dl, maxiter=150, penalty=1000.0)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
        h_cont, c5_cont = _gradient_descent_smooth(h_cont, sub_dl, tau_init=0.005, max_iter=200)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.75
    n_best = len(best_h)
    tried_ns = set()
    # Try nearby n values
    for dn in range(-20, 21, 2):
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _lgbfsb_solve(h_r, 0.01, n_try, deadline=sub_dl, maxiter=100, penalty=500.0)
        # Binary refine
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b = _directed_binary_search(h_b, sub_dl2, max_iters=40000, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Also try a few specific n values that might work well
    for n_try in [160, 170, 190, 200, 220]:
        if time.time() > d3:
            break
        if n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _lgbfsb_solve(h_r, 0.01, n_try, deadline=sub_dl, maxiter=100, penalty=500.0)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.005)
            h_b, c5_b = _directed_binary_search(h_b, sub_dl2, max_iters=30000, seed=5000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Final intensive refinement (23%)
    d4 = deadline
    n_fin = len(best_h)

    # L-BFGS-B with very small tau
    if time.time() < d4 - 30:
        h_f = _project(best_h.copy(), n_fin / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_f, c5_f = _lgbfsb_solve(h_f, 0.003, n_fin, deadline=sub_dl, maxiter=200, penalty=2000.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Gradient descent with tiny tau
    if time.time() < d4 - 25:
        h_f = _project(best_h.copy(), n_fin / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_f, c5_f = _gradient_descent_smooth(h_f, sub_dl, tau_init=0.003, max_iter=300)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Extended binary search
    if time.time() < d4 - 30:
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(d4, time.time() + budget_s * 0.07)
        h_b, c5_b = _directed_binary_search(h_b, sub_dl, max_iters=150000, seed=999)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Pair swap continuous
    if time.time() < d4 - 20:
        h_f = _project(best_h.copy(), n_fin / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_f, c5_f = _pair_swap_continuous(h_f, sub_dl, n_pairs=5000, radius=0.15, seed=777)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Basin hopping with perturbations
    if time.time() < d4 - 20:
        scale = 0.015
        for _ in range(80):
            if time.time() > d4 - 5:
                break
            h_p = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.003)
            h_t, c5_t = _lgbfsb_solve(h_p, 0.01, len(h_p), deadline=sub_dl, maxiter=80, penalty=500.0)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.002, scale * 0.85)
            else:
                scale = min(0.05, scale * 1.1)

    # Final polish
    if time.time() < d4 - 5:
        h_final = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4 - 2, time.time() + budget_s * 0.02)
        h_final, c5_final = _gradient_descent_smooth(h_final, sub_dl, tau_init=0.002, max_iter=150)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final binary check
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h, len(best_h))
        c5_b = _exact_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    # Never return worse than initial
    if final_c5 > _exact_c5(_project(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)):
        best_h = _project(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
