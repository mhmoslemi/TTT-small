def _softmax_obj_and_grad(x, tau):
    n = len(x)
    q = 1.0 - x
    S = np.correlate(x, q, mode='full')  # length 2n-1
    # softmax weights
    S_shifted = S - S.max()
    exp_S = np.exp(S_shifted / tau)
    w = exp_S / exp_S.sum()  # length 2n-1
    obj = tau * (S.max() + np.log(exp_S.sum()))
    # gradient of obj w.r.t. x
    # grad[j] = sum_k w[k] * dS[k]/dx[j]
    # dS[k]/dx[j] = (1-x[j-m_k]) * I[m_k <= j <= n-1+m_k] - x[j+m_k] * I[0 <= j+m_k <= n-1]
    # where m_k = k - (n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        m = k - (n+1) + 1  # k - n + 1
        wk = w[k]
        if wk < 1e-15:
            continue
        # First term: (1-x[j-m]) for j in [max(0,m), min(n-1, n-1+m)]
        j_lo1 = max(0, m)
        j_hi1 = min(n-1, n-1+m)
        if j_lo1 <= j_hi1:
            if m >= 0:
                j_lo1 = m  # since m >= 0, max(0,m) = m
            grad[j_lo1:j_hi1+1] += wk * (1.0 - x[j_lo1-m:j_hi1-m+1])
        # Second term: -x[j+m] for j in [max(0,-m), min(n-1, n-1-m)]
        j_lo2 = max(0, -m)
        j_hi2 = min(n-1, n-1-m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wk * x[j_lo2+m:j_hi2+m+1]
    return obj, grad

    w = exp_S / exp_S.sum()
    grad = _compute_grad_from_weights(w, h, n)
    
    # Direction: -grad (we want to decrease)
    d = -grad
    # Make it mass-neutral: subtract mean
    d = d - np.mean(d)
    
    # Line search: find best alpha such that h + alpha*d is feasible
    # and maximizes the decrease in the true objective
    # Feasibility: 0 <= h + alpha*d <= 1
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_hi = 0.1  # initial cap
    if np.any(pos):
        alpha_hi = min(alpha_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        alpha_hi = min(alpha_hi, np.min(h[neg] / (-d[neg])))
    
    # Exact line search: minimize max(S + alpha*(T - alpha*C))
    # where T = correlate(d, 1-h) - correlate(h, d)
    # and C = correlate(d, d)
    T = np.correlate(d, 1-h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Try several alpha values
    best_alpha = 0.0
    best_val = cur_c5
    for a in [0, 0.1, 0.25, 0.5, 0.75, 1.0]:
        a = a * alpha_hi
        if a > alpha_hi:
            continue
        val = float(np.max(S + a * T - a * a * C)) * dx
        if val < best_val:
            best_val = val
            best_alpha = a
    # Golden section refinement
    if best_alpha > 0:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = max(0, best_alpha * 0.5), min(alpha_hi, best_alpha * 1.5)
        for _ in range(30):
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(S + c1 * T - c1 * c1 * C)) * dx
            f2 = float(np.max(S + c2 * T - c2 * c2 * C)) * dx
            if f1 < f2:
                hi = c2
            else:
                lo = c1
        opt_a = (lo + hi) / 2
        opt_val = float(np.max(S + opt_a * T - opt_a * opt_a * C)) * dx
        if opt_val < best_val:
            best_alpha = opt_a
            best_val = opt_val
    
    if best_alpha > 1e-12 and best_val < cur_c5 - 1e-13:
        h_new = h + best_alpha * d
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    return h, cur_c5
