The strategy suggests using an LP-based active-set minimax descent method. Let me think about what would be most effective:

1. The LP approach: For each iteration, form a linearized subproblem that minimizes the maximum over all correlation constraints simultaneously. This is different from subgradient which only uses the average of active gradients.

2. The key advantage: When multiple correlation shifts are simultaneously near-maximal, the LP direction explicitly reduces all of them at once, whereas subgradient average might not.

Let me think about the implementation:

For the LP at each step:
- We have n variables d_i (the direction) + 1 variable t (epigraph)
- Constraints: For each correlation lag k where F_k(h) is near-maximal: F_k(h) + ∇F_k · d ≤ t
- Mass: sum(d_i) = 0
- Bounds: -min(radius, h_i) ≤ d_i ≤ min(radius, 1-h_i)
- Objective: minimize t + small_penalty * sum(|d_i|)

Then we do exact line search along the LP direction.

Let me also think about what makes this problem hard:
- The objective is max of many bilinear forms, so it's non-smooth
- Near the optimum, many shifts are simultaneously active
- The binary constraint (h ∈ {0,1} for binary solutions) makes local search hard

For n=180, the correlation has 359 lags. The LP with all 359 constraints × 180 variables = ~64,620 entries, plus 180 bound constraints, 1 equality, + t = 361 variables total. This should be solvable in ~50-200ms with HiGHS.

1. Start with `initial_h_values`, project to exact feasibility.
2. Phase 1 (LP descent): Run the LP-based minimax descent for a good chunk of time.
3. Phase 2 (Binary refinement): Convert to binary, do targeted binary search.
4. Phase 3 (Multi-n): Try different resolutions.
5. Phase 4 (Final polish): Intensive refinement of the best found.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k, n, dx):
    """Compute gradient of F_k(h) * dx where k = lag offset (k=0 is center)."""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # F = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
        # dF/dh[i] for i in [0, n-k-1]: -(h[i+k])
        # dF/dh[i] for i in [k, n-1]: +(1-h[i-k])
        g[:n-k] -= h[k:n]
        g[k:] += (1.0 - h[:n-k])
    else:
        u = -k
        # F = sum_{i=0}^{n-u-1} h[i] * (1-h[i+u])
        # dF/dh[i] for i in [0, n-u-1]: +(1-h[i+u])
        # dF/dh[i] for i in [u, n-1]: -(h[i-u])
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g * dx

def _lp_direction(h, corr_scaled, active_idx, radius, t_max):
    """Solve LP to find direction that reduces max active correlation."""
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    if m == 0:
        return None
    
    # Build constraint matrix A_ub for: F_r + grad_r · d - t <= 0 for each active r
    # Variables: [d_0, ..., d_{n-1}, t]
    # Total vars: n+1
    tv = n + 1
    t_idx = n  # index of t variable
    
    A_rows = []
    b_rows = []
    for r in active_idx:
        k = r - (n - 1)
        grad = _grad_row(h, k, n, dx)
        row = np.zeros(tv)
        row[:n] = grad
        row[t_idx] = -1.0
        A_rows.append(row)
        # F_r(h) + grad·d - t <= 0 => grad·d - t <= -F_r(h)
        # In scipy linprog form: A_ub @ x <= b_ub
        # grad·d - t <= -F_r(h)
        b_rows.append(-corr_scaled[r])  # since corr_scaled = F*dx already
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, tv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_b = -min(radius, h[i] + 1e-14)
        hi_b = min(radius, 1.0 - h[i] + 1e-14)
        bounds.append((lo_b, hi_b))
    bounds.append((-10.0, t_max))  # t variable
    
    # Objective: minimize t + eps * sum(d_i^2 approx) -- since LP can't do quadratic,
    # use obj = t + 0 (the t minimization is sufficient)
    c_obj = np.zeros(tv)
    c_obj[t_idx] = 1.0
    # Small penalty on |d| to prefer smaller moves
    for i in range(n):
        c_obj[i] = 1e-7  # tiny penalty to prefer d=0 when t is achievable
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        d = res.x[:n]
        # Verify mass conservation
        d = d - np.mean(d)
        # Clip to bounds
        for i in range(n):
            d[i] = max(-min(radius, h[i]), d[i])
            d[i] = min(min(radius, 1.0 - h[i]), d[i])
        d = d - np.mean(d)  # re-center after clipping
        if np.max(np.abs(d)) < 1e-12:
            return None
        return d
    except Exception:
        return None

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    grid_t = np.linspace(0.0, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * L1 - t * t * Q2)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * L1 - c1 * c1 * Q2)) * dx
        f2 = float(np.max(S0 + c2 * L1 - c2 * c2 * Q2)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * L1 - c1 * c1 * Q2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * L1 - c2 * c2 * Q2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * L1 - opt_t * opt_t * Q2)) * dx
    
    candidates = [(opt_t, opt_val), 
                  (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * L1 - t_hi * t_hi * Q2)) * dx)]
    best_c = min(candidates, key=lambda x: x[1])
    return best_c[0], best_c[1]

def _lp_descent(h, deadline, radius_init=0.08, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_scaled = corr * dx
        max_val = float(np.max(corr_scaled))
        
        # Select active constraints: those within a fraction of max
        # Use a tiered approach: top constraints + near-active
        threshold1 = max_val - 1e-10  # exactly active
        threshold2 = max_val - 0.01  # near-active (1% below max)
        
        active_strict = np.where(corr_scaled >= threshold1)[0]
        active_near = np.where(corr_scaled >= threshold2)[0]
        
        # Include strict + top of near-active
        active = set(active_strict.tolist())
        remaining = 96 - len(active)
        if remaining > 0:
            near_only = [i for i in active_near if i not in active]
            near_only.sort(key=lambda i: -corr_scaled[i])
            active.update(near_only[:remaining])
        
        active_idx = sorted(active)
        if len(active_idx) == 0:
            active_idx = [int(np.argmax(corr_scaled))]
        
        # Solve LP
        d = _lp_direction(h, corr_scaled, active_idx, radius, max_val + 1e-6)
        if d is None:
            # Fallback: negative subgradient of active constraints
            g = np.zeros(n)
            for r in active_idx:
                k = r - (n - 1)
                g += _grad_row(h, k, n, dx)
            g /= len(active_idx)
            d = -g
            d = d - np.mean(d)
            d = np.clip(d, -radius, radius)
            d = d - np.mean(d)
            if np.max(np.abs(d)) < 1e-13:
                radius *= 0.5
                stall += 1
                continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.30)
        else:
            stall += 1
            radius *= 0.72
    
    return best_h, best_c5

def _binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _fast_binary_eval(trial, n):
    dx = 2.0 / n
    corr = np.correlate(trial, 1.0 - trial, mode='full')
    return float(np.max(corr) * dx)

def _targeted_binary(h_bin, deadline, max_iters=100000, seed=99):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    stall_limit = 5000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Focus on the current max correlation to guide search
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_lag = int(np.argmax(corr))
        k = max_lag - (n - 1)
        
        # Try moves that specifically reduce the max overlap
        # For positive k: overlap is at positions where both cur[i] and cur[i+k] are 1
        # We want to flip a 1 to 0 in positions that contribute to the max overlap
        # or flip a 0 to 1 at positions that DON'T contribute
        n_try = min(300, len(ones) * len(zeros))
        best_delta = 0.0
        best_move = None
        
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5 = _fast_binary_eval(trial, n)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _sa_binary(h_bin, deadline, max_iters=100000, temp_init=0.004, seed=202):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5 = _fast_binary_eval(trial, n)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
            T = max(0.0003, T * 0.9997)
        else:
            best_stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap(h, deadline, n_pairs=5000, radius=0.20, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _subgradient_descent(h, deadline, radius_init=0.06, max_iter=3000):
    """Original subgradient method as fallback."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        active = np.where(corr >= max_val - 1e-10)[0]
        
        g = np.zeros(n)
        for m in active:
            k = int(m - (n - 1))
            if k >= 0:
                g[k:] += (1.0 - h[:n-k]) if k > 0 else (1.0 - 2.0*h)
                if k > 0:
                    g[:n-k] -= h[k:]
            else:
                ka = -k
                g[:n-ka] += (1.0 - h[ka:])
                g[ka:] -= h[:n-ka]
        g /= len(active)
        
        d = -g
        d = d - np.mean(d)
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.30)
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c5

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k, n, dx):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        u = -k
        g[:n - u] += (1.0 - h[u:])
        g[u:] -= h[:n - u]
    return g * dx

def _lp_solve(h, corr_scaled, active_idx, radius, t_max):

def _line_search(h, d, dx, radius):

    h = h.copy()

    best_h = h.copy()

            break
        if stall > 120:
            break

            break

        corr_scaled = corr * dx

        active_near = np.where(corr_scaled >= thresh)[0]
        active_strict = np.where(corr_scaled >= max_val - 1e-10)[0]

                k = r - (n - 1)
                g += _grad_row(h, k, n, dx)
            g /= len(active_idx)
            d = -g
            d = d - np.mean(d)
            d = np.clip(d, -radius, radius)
            d = d - np.mean(d)

                stall += 1

            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0

            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _block_swap(h_bin, deadline, max_iters=20000, seed=77):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(6, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        block1 = cur[i:i + length].copy()
        block2 = cur[j:j + length].copy()
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i + length] = block2
        trial[j:j + length] = block1
        new_c5 = _fast_binary_eval(trial, n)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP descent (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_descent(h_init, d1, radius_init=0.10, max_iters=5000, active_frac=0.02)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary optimization (30%)
    d2 = t0 + budget_s * 0.65
    h_bin = _to_binary(best_h, n0)
    c5_bin = _binary_c5(h_bin)

    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=100000, seed=101, n_try_max=300)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _sa_binary(h_bin, sub_dl, max_iters=100000, temp_init=0.003, seed=202)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sub_dl = min(d2, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=80000, seed=404, n_try_max=200)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _block_swap(h_bin, sub_dl, max_iters=15000, seed=303)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # LP polish on continuous
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_pol = _project(best_h.copy(), n0 / 2.0)
    h_pol, c5_pol = _lp_descent(h_pol, sub_dl, radius_init=0.05, max_iters=1500, active_frac=0.015)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.85
    n_best = len(best_h)
    tried_ns = {n_best}

    for dn in range(-24, 25, 3):
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.015:
            sub_dl = min(d3, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_descent(h_r, sub_dl, radius_init=0.07, max_iters=800, active_frac=0.02)
            if time.time() < d3 - 3:
                h_b = _to_binary(h_r, n_try)
                sub_dl2 = min(d3, time.time() + budget_s * 0.007)
                h_b, c5_b = _targeted_binary(h_b, sub_dl2, max_iters=25000, seed=1000 + n_try, n_try_max=150)
                if c5_b < c5_r:
                    c5_r = c5_b
                    h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Final refinement (15%)
    d4 = deadline
    n_best = len(best_h)

    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = _project(best_h.copy(), n_best / 2.0)
    h4, c5_4 = _lp_descent(h4, sub_dl, radius_init=0.04, max_iters=1000, active_frac=0.01)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    if time.time() < d4 - 15:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _pair_swap(h4, sub_dl, n_pairs=5000, radius=0.15, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    if time.time() < d4 - 12:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=50000, seed=777, n_try_max=200)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d4 - 8:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _sa_binary(h_b, sub_dl, max_iters=50000, temp_init=0.002, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final LP + subgrad
    if time.time() < d4 - 3:
        h6 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4, time.time() + 2)
        h6, c5_6 = _lp_descent(h6, sub_dl, radius_init=0.02, max_iters=300, active_frac=0.01)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
