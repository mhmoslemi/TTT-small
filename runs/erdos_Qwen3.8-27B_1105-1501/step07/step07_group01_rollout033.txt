- And C[k] = -1 for the cross terms, those are convex parabolas

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_gradient(h, raw, tau):
    n = len(h)
    shifted = raw / tau
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd = w[n - 1 + d]
        if wd > 1e-15:
            g[:n - d] += wd * (1.0 - h[d:])
            g[d:] -= wd * h[:n - d]
        wn = w[n - 1 - d]
        if wn > 1e-15:
            g[d:] += wn * (1.0 - h[:n - d])
            g[:n - d] -= wn * h[d:]
    return g

def _projected_direction(h, g):
    d = -g.copy()
    for _ in range(15):
        at_zero = h <= 1e-10
        at_one = h >= 1.0 - 1e-10
        d[at_zero & (d < 0)] = 0.0
        d[at_one & (d > 0)] = 0.0
        d -= d.mean()
    d -= d.mean()
    return d

def _t_max(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = 1.0e6
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    return max(t_hi, 0.0)

def _line_search(h, d, raw, t_hi):
    n = len(h)
    dx = 2.0 / n
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    if t_hi < 1e-14:
        return 0.0, float(np.max(raw)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(raw + c2 * T - c2 * c2 * C)) * dx
    for _ in range(70):
        if hi - lo < 1e-13 * max(1.0, hi):
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * C)) * dx
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * T - c2 * c2 * C)) * dx
    t_opt = (lo + hi) / 2.0
    f_opt = float(np.max(raw + t_opt * T - t_opt * t_opt * C)) * dx
    f0 = float(np.max(raw)) * dx
    f_end = float(np.max(raw + t_hi * T - t_hi * t_hi * C)) * dx
    best_t = 0.0
    best_f = f0
    if f_opt < best_f:
        best_t = t_opt
        best_f = f_opt
    if f_end < best_f:
        best_t = t_hi
        best_f = f_end
    return best_t, best_f

def _pgd_optimize(h, deadline, tau0=2.0, tau_min=0.1, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = float(np.max(raw)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau0
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > 400:
            break
        g = _smooth_gradient(h, raw, tau)
        d = _projected_direction(h, g)
        dnorm = float(np.max(np.abs(d)))
        if dnorm < 1e-14:
            break
        d = d / dnorm
        t_hi = _t_max(h, d)
        if t_hi < 1e-14:
            stall += 1
            continue
        t_opt, f_opt = _line_search(h, d, raw, t_hi)
        if f_opt < cur_c5 - 1e-14:
            h = h + t_opt * d
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            raw = np.correlate(h, 1.0 - h, 'full')
            cur_c5 = float(np.max(raw)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            tau = max(tau_min, tau * 0.98)
        else:
            stall += 1
            if stall % 80 == 1:
                pert = rng.normal(0, 0.005, n)
                h_p = _project(h + pert, target)
                raw_p = np.correlate(h_p, 1.0 - h_p, 'full')
                c5_p = float(np.max(raw_p)) * (2.0 / n)
                if c5_p < best_c5:
                    h = h_p
                    raw = raw_p
                    cur_c5 = c5_p
                    best_c5 = c5_p
                    best_h = h.copy()
                    stall = 0
    return best_h, best_c5

def _slsqp_obj(x, tau, dx):
    raw = np.correlate(x, 1.0 - x, 'full')
    shifted = raw / tau
    shifted -= shifted.max()
    return float(tau * np.log(np.sum(np.exp(shifted)))) * dx

def _slsqp_jac(x, tau, dx):
    raw = np.correlate(x, 1.0 - x, 'full')
    g = _smooth_gradient(x, raw, tau)
    return g * dx

def _mass_constraint(x):
    return np.sum(x) - len(x) / 2.0

def _slsqp_refine(h, deadline, tau=0.3):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _mass_constraint}]
    remaining = deadline - time.time()
    if remaining < 5:
        return best_h, best_c5
    maxit = max(10, int(remaining / 2.0))
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, args=(tau, dx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-12})
        if res.x is not None:
            x = _project(res.x.copy(), target)
            c5 = _exact_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
    except Exception:
        pass
    return best_h, best_c5

def _pair_search_gradient(h, deadline, n_pairs_per_round=200, max_rounds=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = float(np.max(raw)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    round_num = 0
    while time.time() < deadline and round_num < max_rounds:
        round_num += 1
        if stall > 60:
            break
        tau = 0.5
        g = _smooth_gradient(h, raw, tau)
        # Select pairs based on gradient: high g[i], low g[j]
        sorted_idx = np.argsort(-g)
        top_half = sorted_idx[:n // 2]
        bot_half = sorted_idx[n // 2:]
        improved = False
        for _ in range(n_pairs_per_round):
            if time.time() > deadline:
                break
            i = int(top_half[int(np.random.default_rng(_).integers(0, len(top_half)))])
            j = int(bot_half[int(np.random.default_rng(_ + 1000).integers(0, len(bot_half)))])
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            d[-i] += 0.0  # no-op to keep d as vector
            # Check feasibility direction
            if h[i] >= 1.0 or h[j] <= 0.0:
                d[i] = -1.0
                d[j] = 1.0
                if h[i] <= 0.0 or h[j] >= 1.0:
                    continue
            t_hi = _t_max(h, d)
            if t_hi < 1e-14:
                continue
            T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
            C = np.correlate(d, d, 'full')
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, t_hi
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * C)) * dx
            f2 = float(np.max(raw + c2 * T - c2 * c2 * C)) * dx
            for __ in range(50):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(raw + c1 * T - c1 * c1 * C)) * dx
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * C)) * dx
            t_opt = (lo + hi) / 2.0
            f_opt = float(np.max(raw + t_opt * T - t_opt * t_opt * C)) * dx
            f0 = float(np.max(raw)) * dx
            best_f = min(f0, f_opt)
            best_t = 0.0 if f0 <= f_opt else t_opt
            if best_f < cur_c5 - 1e-14:
                h = h + best_t * d
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, target)
                raw = np.correlate(h, 1.0 - h, 'full')
                cur_c5 = float(np.max(raw)) * dx
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        if time.time() > deadline:
            break
        per_n = max(2.0, (deadline - time.time()) / max(1, len(n_list)))
        h_r = _resample.best_h if False else _resample(h_best, n_try)

def _final_polish(h, deadline):
