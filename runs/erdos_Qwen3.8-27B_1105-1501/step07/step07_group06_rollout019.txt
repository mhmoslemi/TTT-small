import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h): return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)

def _update_corr(h, C, p, delta):
    """Update C in place. h[p] still has OLD value."""
    n = len(h)
    m = p
    if m > n - 1: m = n - 1
    if m > 0:
        C[n:n+m] += delta * (1.0 - h[p-m:p][::-1])
        C[n-1-m:n-1] += -delta * h[p-m:p]
    m2 = n - 1 - p
    if m2 > n - 1: m2 = n - 1
    if m2 > 0:
        C[n:n+m2] += -delta * h[p+1:p+1+m2]
        C[n-1-m2:n-1] += delta * (1.0 - h[p+1:p+1+m2][::-1])
    C[n-1] += delta * (1.0 - 2.0 * h[p]) - delta * delta

def _do_swap(h, C, p, q):
    """Binary swap: h[p] 1->0, h[q] 0->1. Updates h and C in place. Returns new max of C."""
    _update_corr(h, C, p, -1.0)
    h[p] = 0.0
    _update_corr(h, C, q, 1.0)
    h[q] = 1.0
    return float(np.max(C))

def _undo_swap(h, C, p, q):
    """Undo: h[p] 0->1, h[q] 1->0."""
    _update_corr(h, C, p, 1.0)
    h[p] = 1.0
    _update_corr(h, C, q, -1.0)
    h[q] = 0.0
    return float(np.max(C))
