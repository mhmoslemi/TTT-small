def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    target = n / 2.0

    rng = np.random.default_rng(seed)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            # Compute direction for coordinate j
            old = h[j]
            g = np.zeros(2 * n - 1)
            # Positive lags: d = 0 to n-j-1
            if n - j > 0:
                g[n-1 : n-1+(n-j)] = 1.0 - h[j:n]
            # Negative lags: d = -j to -1
            if j > 0:
                g[n-1-j : n-1] = -h[0:j]
            
            # Line search: minimize max(S0 + delta * g)

            lo_d = -old

            # Ternary search

                    f2 = f1
                    c1 = hi_d - gr * (hi_d - lo_d)
                    f1 = float(np.max(S0 + c1 * g))

                    c1 = c2
                    f1 = f2
                    c2 = lo_d + gr * (hi_d - lo_d)
                    f2 = float(np.max(S0 + c2 * g))
            opt_d = (lo_d + hi_d) / 2.0

            if abs(new_val - old) > 1e-10:

The effect on the correlation:

- h[j] - δ ≥ 0 → δ ≤ h[j]
- (and δ can be negative: h[i] + δ ≥ 0 → δ ≥ -h[i], h[j] - δ ≤ 1 → δ ≥ h[j] - 1)

def compute_direction(h, i, j):
    n = len(h)
    d = np.zeros(2*n-1)
    # Positive lags: d[n-1+k] = h[j+k] - h[i+k] for valid indices
    max_k = min(n-i, n-j)
    d[n-1:n-1+max_k] = h[j:j+max_k] - h[i:i+max_k]
    # But we need to handle the case where one index is out of range
    # If i+k >= n but j+k < n: d[n-1+k] = h[j+k] - 0 = h[j+k]
    # If j+k >= n but i+k < n: d[n-1+k] = 0 - h[i+k] = -h[i+k]
    for k in range(max_k, n - i):
        d[n-1+k] = -h[i+k]
    for k in range(max_k, n - j):
        d[n-1+k] = h[j+k]
    # Negative lags: d[n-1-k] = h[j-k] - h[i-k] for valid indices
    max_k2 = min(i, j)
    for k in range(1, max_k2+1):
        d[n-1-k] = h[j-k] - h[i-k]
    for k in range(max_k2+1, i+1):
        d[n-1-k] = -h[i-k]
    for k in range(max_k2+1, j+1):
        d[n-1-k] = h[j-k]
    return d

def _project(v, target):

        return np.zeros(n)

        return np.ones(n)
    lo = float(v.min()) - 1.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt

def _lp_refine(h, deadline, radius=0.1, top_k=12, stall_limit=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rad = radius
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(scores)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-rad, -h[jj])
            hi_b = min(rad, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                rad *= 0.7
                if rad < 1e-7:
                    break
                continue
        except Exception:
            stall += 1
            rad *= 0.7
            if rad < 1e-7:
                break
            continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            rad *= 0.7
            if rad < 1e-7:
                break
            continue
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = rad
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            rad *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, fe)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.30)
        else:
            stall += 1
            rad *= 0.7
            if stall % 30 == 0 and rad < 0.01:
                rad = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 5.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Transfer descent (main workhorse) - 35%
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _transfer_descent(h_init, d1, seed=42, n_pairs=300, max_iters=500000, ls_iters=22)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Deterministic transfer descent - 15%
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _transfer_descent_deterministic(best_h, d2, seed=123, ls_iters=18, batch=800)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Coordinate descent with projection - 10%
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _coord_descent(best_h, d3, seed=456, max_sweeps=3000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP refinement - 10%
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _lp_refine(best_h, d4, radius=0.08, top_k=12, stall_limit=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-restart with perturbations - 10%
    d5 = t0 + budget_s * 0.80
    scale = 0.03
    for restart in range(30):
        if time.time() > d5:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_t, c5_t = _transfer_descent(h_pert, sub_dl, seed=1000 + restart, n_pairs=200, max_iters=30000, ls_iters=18)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Phase 6: Multi-n exploration - 5%
    d6 = t0 + budget_s * 0.85
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > d6:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + 8)
        h_r, c5_r = _transfer_descent(h_r, sub_dl, seed=2000 + dn, n_pairs=200, max_iters=20000, ls_iters=18)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Final intensive transfer descent - 10%
    d7 = deadline - 5
    h7, c5_7 = _transfer_descent(best_h, d7, seed=999, n_pairs=400, max_iters=500000, ls_iters=20)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final LP polish
    if time.time() < deadline - 3:
        h8, c5_8 = _lp_refine(best_h, deadline - 1, radius=0.04, top_k=15, stall_limit=30)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
