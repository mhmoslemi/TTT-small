import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for theta
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, t_cap=1.0):
    """Find best t in feasible range for h + t*d minimizing max correlate."""
    n = len(h)
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Find feasible t range
    t_lo = -t_cap
    t_hi = t_cap
    for i in range(n):
        di = d[i]
        if di > 1e-14:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / di)  # di negative, so (1-h[i])/di is more negative
            t_hi = min(t_hi, -h[i] / di)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(R0))
    
    # Grid search
    n_grid = 100
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([np.max(R0 + t * A - t * t * B) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section refinement
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        best_t = float(ts[best_idx])
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        best_t = (lo + hi) / 2.0
    
    best_val = float(np.max(R0 + best_t * A - best_t * best_t * B))
    return best_t, best_val

def _lp_direction(h, peak_indices, max_iter=200):
    """Find LP direction that reduces peaks in peak_indices."""
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    G = np.zeros((L, n))
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        G[:, i] = np.correlate(e, o, mode='full') - np.correlate(h, e, mode='full')
    
    K = len(peak_indices)
    # Variables: sp (n), sn (n), beta (1) => 2n+1 variables
    # Constraints:
    # For each peak r: sum_i G[r,i]*sp_i - sum_i G[r,i]*sn_i + beta <= 0
    # Equality: sum(sp) - sum(sn) = 0
    # Objective: minimize -beta
    
    G_peaks = G[peak_indices, :]  # K x n
    
    # c vector for linprog: minimize c.x
    # x = [sp; sn; beta]
    c = np.zeros(2 * n + 1)
    c[-1] = -1.0  # minimize -beta
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each peak r: G_peak[r,:] @ sp - G_peak[r,:] @ sn + beta <= 0
    A_ub = np.zeros((K, 2 * n + 1))
    for r in range(K):
        A_ub[r, :n] = G_peaks[r, :]
        A_ub[r, n:2*n] = -G_peaks[r, :]
        A_ub[r, -1] = 1.0
    b_ub = np.zeros(K)
    
    # Equality: sum(sp) - sum(sn) = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds: 0 <= sp_i <= 1, 0 <= sn_i <= 1, -1e6 <= beta <= 1e6
    bounds = [(0, 1)] * (2 * n) + [(-1e6, 1e6)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    sp = result.x[:n]
    sn = result.x[n:2*n]
    u = sp - sn
    beta = result.x[-1]
    
    # Check that direction is non-trivial
    if np.max(np.abs(u)) < 1e-10:
        return None
    
    # Normalize to unit L1
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-12:
        return None
    u = u / l1
    
    return u

def _grad_descent_direction(h, peak_indices):
    """Negative weighted gradient direction from active peaks."""
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    R = np.correlate(h, o, mode='full')
    
    # Weight peaks by how close they are to max
    M = float(np.max(R))
    weights = np.zeros(L)
    for idx in peak_indices:
        w = 1.0 / (1.0 + (M - R[idx]) * 10.0)
        weights[idx] = w
    
    if np.sum(weights) < 1e-15:
        return None
    
    # Compute gradient for all peaks
    G_peaks = np.zeros((len(peak_indices), n))
    for k, idx in enumerate(peak_indices):
        for i in range(n):
            e = np.zeros(n)
            e[i] = 1.0
            G_peaks[k, i] = np.correlate(e, o, mode='full')[idx] - np.correlate(h, e, mode='full')[idx]
    
    # Weighted sum of gradients
    u = (G_peaks.T @ weights)  # n-vector
    # Center it (zero sum)
    u -= np.mean(u)
    
    neg_norm = float(np.sum(np.abs(u)))
    if neg_norm < 1e-12:
        return None
    u = -u / neg_norm
    return u

                c5_try = _exact_c5(h_try)

                    best_step = h_try

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:

                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')

        # idx1 = i - m + n - 1, valid if 0 <= idx1 < n
        # idx2 = i + m - n + 1, valid if 0 <= idx2 < n

        idx1 = i - m + n - 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i + m - n + 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        G[k, :] = (1.0 - h[np.where(valid1, idx1, 0)]) * valid1 - h[np.where(valid2, idx2, 0)] * valid2
    return G

def _lp_direction(h, peak_indices):
    n = len(h)
    K = len(peak_indices)
    G = _grad_at_peaks(h, peak_indices)  # K x n
    
    # Variables: [sp(0..n-1), sn(0..n-1), beta]
    # Minimize: -beta
    # Subject to:
    #   G[k, :] @ sp - G[k, :] @ sn + beta <= 0 for k = 0..K-1
    #   sum(sp) - sum(sn) = 0
    #   0 <= sp_i <= 1, 0 <= sn_i <= 1
    #   beta free (bounded by [-1e6, 1e6])
    
    c = np.zeros(2 * n + 1)
    c[-1] = -1.0
    
    A_ub = np.zeros((K, 2 * n + 1))
    for k in range(K):
        A_ub[k, :n] = G[k, :]
        A_ub[k, n:2*n] = -G[k, :]
        A_ub[k, -1] = 1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(0, 1)] * (2 * n) + [(-1e6, 1e6)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not result.success:
        return None
    
    u = result.x[:n] - result.x[n:2*n]
    if np.max(np.abs(u)) < 1e-10:
        return None
    
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-12:
        return None
    return u / l1

def _line_search(h, d, t_cap=1.0):
    n = len(h)
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Feasible t range
    t_lo = -t_cap
    t_hi = t_cap
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    if np.any(pos_mask):
        t_lo = max(t_lo, float(-np.min(h[pos_mask] / d[pos_mask])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg_mask]) / d[neg_mask])))
        t_hi = min(t_hi, float(-np.min(h[neg_mask] / d[neg_mask])))
    
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0))
    
    # Grid search
    n_grid = 200
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([np.max(R0 + t * A - t * t * B) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    if hi - lo > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        for _ in range(45):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        best_t = (lo + hi) / 2.0
        best_val = float(np.max(R0 + best_t * A - best_t * best_t * B))
    else:
        best_t = float(ts[best_idx])
        best_val = float(vals[best_idx])
    
    return best_t, best_val

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_at_peaks(h, peak_indices):
    n = len(h)
    K = len(peak_indices)
    G = np.zeros((K, n))
    i_arr = np.arange(n)
    for k in range(K):
        m = peak_indices[k]
        idx1 = i_arr - m + n - 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i_arr + m - n + 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
        term2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
        G[k, :] = term1 - term2
    return G

def _line_search(h, d, t_cap=1.0):
    n = len(h)
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    t_lo = -t_cap
    t_hi = t_cap
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.max(-h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0))
    
    # Quick grid
    n_grid = 150
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.empty(n_grid)
    for j in range(n_grid):
        t = ts[j]
        vals[j] = np.max(R0 + t * A - t * t * B)
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    if hi - lo > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        for _ in range(40):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        best_t = (lo + hi) / 2.0
        best_val = float(np.max(R0 + best_t * A - best_t * best_t * B))
    else:
        best_t = float(ts[best_idx])
        best_val = float(vals[best_idx])
    
    return best_t, best_val

def _lp_direction(h, peak_indices):
    n = len(h)
    K = len(peak_indices)
    G = _grad_at_peaks(h, peak_indices)
    
    c_vec = np.zeros(2 * n + 1)
    c_vec[-1] = -1.0
    
    A_ub = np.zeros((K, 2 * n + 1))
    for k in range(K):
        A_ub[k, :n] = G[k, :]
        A_ub[k, n:2*n] = -G[k, :]
        A_ub[k, -1] = 1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(-1e6, 1e6)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return None
    
    if not res.success:
        return None
    
    u = res.x[:n] - res.x[n:2*n]
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-10:
        return None
    return u / l1

def _neg_gradient(h, R, n_peaks=8):
    n = len(h)
    top_idx = np.argsort(R)[-n_peaks:]
    G = _grad_at_peaks(h, top_idx)
    M = float(np.max(R))
    diffs = M - R[top_idx]
    w = 1.0 / (1.0 + diffs * 10.0)
    w /= float(np.sum(w))
    grad = G.T @ w
    d = -grad
    d -= float(np.mean(d))
    norm = float(np.sum(np.abs(d)))
    if norm < 1e-12:
        return None
    return d / norm
