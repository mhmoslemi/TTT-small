import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlate_values(h):
    n = len(h)
    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        bounds = (1.0 - h[pos]) / d[pos]
        a_hi = min(a_hi, float(np.min(bounds)))
    if np.any(neg):
        bounds = h[neg] / (-d[neg])
        a_lo = max(a_lo, float(np.min(bounds)))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    # Parabolic minimizers for each correlation lag
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    # Grid
    ng = min(48, max(24, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    # Cross-over points of top M lags
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _grad_softmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _coord_descent(h, target, dx, deadline, max_rounds=100):
    """Optimize individual coordinates one at a time."""
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(12345)
    stall = 0
    order = list(range(n))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > n * 2:
            break
        rng.shuffle(order)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = 1.0
            a, c_try = _line_search(h, d, dx, 0.5)
            if a != 0.0 and c_try < cur - 1e-14:
                h[i] = np.clip(h[i] + a, 0.0, 1.0)
                cur = c_try
                improved = True
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
        if not improved:
            stall += n
        else:
            stall = 0
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(40, max(20, int(width * 80)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _grad_softmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _local_opt(h, deadline, r0=0.20, stall_lim=80, nd=6):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31415)
    rad = r0
    stall = 0
    momentum = np.zeros(n)
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        dirs = []
        for beta in [12.0, 30.0, 60.0]:
            g = _grad_softmax(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)
        if np.linalg.norm(momentum) > 1e-8:
            dirs.append(momentum / np.linalg.norm(momentum))
        g_base = _grad_softmax(h, 25.0)
        for bs in [8, 16, 32]:
            d = np.zeros(n)
            for b in range(0, n, bs):
                e = min(b + bs, n)
                d[b:e] = np.mean(g_base[b:e])
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-10:
                dirs.append(d / nrm)
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.4, 12.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.4, 1.2) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs[:nd + 3]:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            momentum = 0.6 * momentum + 0.4 * best_d
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.45)
        else:
            stall += 1
            rad *= 0.72
            momentum *= 0.5
            if rad < 0.004:
                pert = rng.normal(0, 0.09, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.15
                elif stall % 20 == 0:
                    rad = 0.07
    return best_h, best_c

def _coord_descent(h, deadline, max_rounds=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(999)
    order = list(range(n))
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        rng.shuffle(order)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            d = np.zeros(n)
            d[i] = 1.0
            a, c_try = _line_search(h, d, dx, 0.5)
            if a != 0.0 and c_try < cur - 1e-14:
                h[i] = np.clip(h[i] + a, 0.0, 1.0)
                cur = c_try
                improved = True
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
        if not improved and rnd > 5:
            break
    return best_h, best_c

def _pair_opt(h, deadline, max_it=25000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(27182)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 350:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 0.8)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=12000, Ti=0.0015):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(555)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Thorough local optimization (14%)
    d1 = t0 + budget_s * 0.14
    h1, c1 = _local_opt(h_init, d1, r0=0.22, stall_lim=70, nd=6)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (7%)
    d2 = t0 + budget_s * 0.21
    nc = len(best_h)
    h_cd = _project(best_h.copy(), nc / 2.0)
    h_cd, c_cd = _coord_descent(h_cd, d2, max_rounds=400)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd.copy()

    # Phase 3: Pair optimization (7%)
    d3 = t0 + budget_s * 0.28
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=18000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Binary SA + recover (9%)
    d4 = t0 + budget_s * 0.37
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d4, mx=10000, Ti=0.0012)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4 - 1:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.02)
        hp, cp = _local_opt(hp, sd, r0=0.10, stall_lim=30, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Multi-resolution exploration (18%)
    d5 = t0 + budget_s * 0.55
    nc = len(best_h)
    offsets = [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d5 - 2:
            break
        n_try = nc + dn
        if n_try < 50 or n_try > 320 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5 - 1, time.time() + budget_s * 0.012)
        hr, cr = _local_opt(hr, sd, r0=0.15, stall_lim=40, nd=5)
        if cr < best_c + 0.003 and time.time() < d5 - 5:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5 - 3, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=4000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d5 - 3:
            sd3 = min(d5 - 2, time.time() + budget_s * 0.007)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive local + pair (18%)
    d6 = t0 + budget_s * 0.73
    nc = len(best_h)
    for r_init in [0.15, 0.08, 0.03]:
        if time.time() > d6 - 3:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.05)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=80, nd=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 4:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _pair_opt(h6, sd, max_it=8000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Iterated local search with diverse kicks (14%)
    d7 = deadline
    scale = 0.025
    n_re = 0
    while time.time() < d7 and n_re < 100:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        n_freq = rng.integers(2, 6)
        for _ in range(n_freq):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 1.0)
            pert += amp * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.12)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _coord_descent(hf, deadline, max_rounds=200)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=30, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=4000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
