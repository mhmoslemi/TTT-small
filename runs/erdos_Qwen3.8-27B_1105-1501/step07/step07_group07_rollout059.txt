        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    mid = n - 1
    G = np.zeros((2 * n - 1, n))
    G[mid, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[mid + d, d:] = 1.0 - h[:m]
        G[mid + d, :m] += -h[d:]
        G[mid - d, :m] = 1.0 - h[d:]
        G[mid - d, d:] += -h[:m]
    return G

def _verify_gradient(h):
    n = len(h)
    G = _build_G(h)
    eps = 1e-7
    max_err = 0.0
    for j in [0, n // 4, n // 2, 3 * n // 4, n - 1]:
        h_p = h.copy()
        h_m = h.copy()
        h_p[j] = min(1.0, h[j] + eps)
        h_m[j] = max(0.0, h[j] - eps)
        actual = (_raw_scores(h_p) - _raw_scores(h_m)) / (h_p[j] - h_m[j])
        max_err = max(max_err, float(np.max(np.abs(actual - G[:, j]))))
    return max_err

def _smooth_obj(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    scores = _raw_scores(x) * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    z = np.exp(u)
    s = float(np.sum(z))
    if s < 1e-300:
        return mval
    return mval + tau * np.log(s)

def _smooth_grad(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    G = _build_G(x)
    scores = _raw_scores(x) * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    z = np.exp(u)
    s = float(np.sum(z))
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(scores)] = 1.0
    else:
        p = z / s
    return (G.T @ p) * (dx / tau)

def _eq_fun(x):
    return float(np.sum(x) - _g_target)

def _eq_jac(x):
    return np.ones(_g_n)

def _run_slsqp(h0, tau, maxiter=100):
    global _g_n, _g_target, _g_dx, _g_tau
    n = len(h0)
    _g_n = n
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    _g_tau = tau
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},)
    try:
        res = minimize(_smooth_obj, h0, jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
        x = np.asarray(res.x, dtype=float)
        x = _project_sum(np.clip(x, 0.0, 1.0), n / 2.0)
        return x
    except Exception:
        return h0.copy()

def _lse_anneal(h0, deadline, n_tau=6, maxiter=None):
    n = len(h0)
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _raw_scores(h) * (2.0 / n)
    score_max = float(np.max(scores))
    if maxiter is None:
        maxiter = max(30, min(150, int(150 * (budget_s / 1100.0))))
    taus = np.geomscore_max / 10.0, score_max / 40.0)
    taus = np.geomspace(tau_start, tau_end, n_tau)
    for tau in taus:
        if time.time() > deadline:
            break
        h_new = _run_slsqp(h, tau, maxiter=maxiter)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 + 0.005:
            h = h_new
            cur_c5 = c5_new
        else:
            h = _project_sum(h_new, n / 2.0)
            cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _gen_thue_morse(n):
    bits = np.zeros(n, dtype=int)
    for i in range(1, n):
        bits[i] = bits[i >> 1] ^ (i & 1)
    h = np.zeros(n)
    h[bits == 1] = 1.0
    return _project_sum(h, n / 2.0)

def _gen_golden(n, seed=0):
    rng = np.random.default_rng(seed)
    rho = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n, dtype=float)
    frac = (x * rho + seed * 0.618) % 1.0
    ranked = np.argsort(frac)
    h = np.zeros(n)
    h[ranked[:n // 2]] = 1.0
    return h

def _gen_random_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = (rng.random(n) < 0.5).astype(float)
    idx = np.argsort(rng.random(n))
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def _gen_block(n, ratio=0.5):
    h = np.zeros(n)
    block = max(2, n // 20)
    for i in range(0, n, block):
        if (i // block) % 2 == 0:
            h[i:i + block] = 1.0
    return _project_sum(h, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=200000, T_init=0.02, seed=42, kick_size=1):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    best_stall = 0
    total_kicks = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 5000 and best_stall > 5000:
            break
        if it % 10000 == 0 and it > 0:
            T = max(0.001, T_init * (0.6 ** (it / 10000.0)))
        ks = kick_size if kick_size > 1 else 1
        if ks == 1:
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = ones[rng.integers(len(ones))]
            oz = zeros[rng.integers(len(zeros))]
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_c5 = _exact_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                    best_stall = 0
                else:
                    best_stall += 1
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
                stall += 1
        else:
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if len(ones) < ks or len(zeros) < ks:
                break
            perm1 = rng.permutation(len(ones))[:ks]
            perm0 = rng.permutation(len(zeros))[:ks]
            o_idx = ones[perm1]
            z_idx = zeros[perm0]
            cur[o_idx] = 0.0
            cur[z_idx] = 1.0
            new_c5 = _exact_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                    best_stall = 0
                else:
                    best_stall += 1
            else:
                cur[o_idx] = 1.0
                cur[z_idx] = 0.0
                stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=50):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _kick_sa(h_bin, deadline, seed=7):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.015
    for it in range(50000):
        if time.time() > deadline:
            break
        T = max(0.0005, 0.015 * (0.9997 ** it))
        ks = int(rng.integers(2, 5))
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) < ks or len(zeros) < ks:
            break
        oi = rng.choice(ones, size=ks, replace=False)
        oz = rng.choice(zeros, size=ks, replace=False)
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)
