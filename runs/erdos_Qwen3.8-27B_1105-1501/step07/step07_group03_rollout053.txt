def _lp_direction(h, raw, n, K, delta):
    """
    Find a direction d that reduces the top K raw values simultaneously.
    d has sum(d) = 0, max|d| <= delta.
    We maximize the minimum reduction among top K lags.
    """
    from scipy.optimize import linprog
    
    target = n / 2.0
    # Get top K lag indices
    top_k = np.argsort(-raw)[:K]
    
    # Compute gradients for each top lag
    # grad_r[j] = d(raw[r])/dh[j]
    grads = []
    u = 1.0 - h
    for r in top_k:
        d_lag = r - (n - 1)
        g = np.zeros(n)
        if d_lag >= 0:
            m = n - d_lag
            if m > 0:
                g[d_lag:] += u[:m]
                g[:m] -= h[d_lag:]
        else:
            gp = -d_lag
            m = n - gp
            if m > 0:
                g[:m] += u[gp:]
                g[gp:] -= h[:m]
        grads.append(g)
    
    # LP: maximize t (minimization of -t)
    # subject to: sum_j grad_r[j] * d[j] <= -t  for each r (we want reduction >= t)
    #             sum_j d[j] = 0
    #             -delta <= d[j] <= delta
    #             (also need h[j] + d[j] in [0,1], but we'll handle that with the trust region)
    
    # Variables: d[0], ..., d[n-1], t
    # c = [0, ..., 0, -1] (minimize -t)
    c = np.zeros(n + 1)
    c[n] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each r: sum_j grad_r[j] * d[j] + t <= 0
    A_ub = np.zeros((K, n + 1))
    for i, g in enumerate(grads):
        A_ub[i, :n] = g
        A_ub[i, n] = 1.0
    b_ub = np.zeros(K)
    
    # Equality: sum d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: for d[j], we need max(0, h[j]-delta) <= h[j]+d[j] <= min(1, h[j]+delta)
    # So: max(-h[j], -delta) <= d[j] <= min(1-h[j], delta)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -delta)
        hi = min(1 - h[j], delta)
        bounds.append((lo, hi))
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        t_val = res.x[n]
        return d, t_val
    return None, 0.0

def _exact_c5(h):
    n = len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

    u = 1.0 - h
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w * u[:m]
                g[:m] -= w * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w * u[gp:]
                g[gp:] -= w * h[:m]
    return g

def _line_search_quad(h, d, raw, dx, radius, n_top=18):
    # Same as before - find best alpha along direction d
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)
                    a2 = (B - sd) / (2.0 * A)
                    for aa in [a1, a2]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    n_grid = 36
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    # Golden section refinement
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    return best_alpha, best_val

def _lp_direction(h, raw, n, K, delta):
    """Find direction via LP that reduces top K lags simultaneously."""
    top_k = np.argsort(-raw)[:K]
    u = 1.0 - h
    grads = []
    for r in top_k:
        d_lag = r - (n - 1)
        g = np.zeros(n)
        if d_lag >= 0:
            m = n - d_lag
            if m > 0:
                g[d_lag:] += u[:m]
                g[:m] -= h[d_lag:]
        else:
            gp = -d_lag
            m = n - gp
            if m > 0:
                g[:m] += u[gp:]
                g[gp:] -= h[:m]
        grads.append(g)
    
    c = np.zeros(n + 1)
    c[n] = -1.0
    A_ub = np.zeros((K, n + 1))
    for i in range(K):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = 1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-h[j], -delta)
        hi = min(1.0 - h[j], delta)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[n] > 1e-8:
            d = res.x[:n]
            d -= np.mean(d)  # ensure zero sum
            return d
    except:
        pass
    return None

def _make_directions(h, g, n, rng, radius):
    directions = []
    d0 = -g.copy()
    d0 -= np.mean(d0)
    nm = np.linalg.norm(d0)
    if nm > 1e-12:
        d0 /= nm
        directions.append(d0)
    for bl in [4, 8, 16]:
        nb = n // bl
        if nb < 2:

        db = -blk.copy()
        db -= np.mean(db)
        nm = np.linalg.norm(db)
        if nm > 1e-12:
            directions.append(db / nm)
    idx_pos = np.argsort(g)[-3:]
    idx_neg = np.argsort(g)[:3]
    for ip in idx_pos:
        for ino in idx_neg:
            if ip == ino:
                continue
            dp = np.zeros(n)
            dp[ip] = -1.0
            dp[ino] = 1.0
            dp -= np.mean(dp)
            nm = np.linalg.norm(dp)
            if nm > 1e-12:
                directions.append(dp / nm)
    return directions[:6]

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):

                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.3, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search_quad(h, d, raw, dx, radius, n_top=8)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _subspace_opt(h, deadline, ns=6, iter_per_sub=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    for _ in range(iter_per_sub):
        if time.time() > deadline:
            break
        idx = rng.integers(0, n, size=ns)
        idx = np.sort(idx)
        h_sub = h[idx].copy()
        raw_full = np.correlate(h, 1.0 - h, mode='full')
        for sub_it in range(20):
            g_sub = np.zeros(ns)
            idx_list = idx.tolist()
            for ii, j in enumerate(idx_list):
                # Compute gradient of C5 w.r.t. h[j]
                u = 1.0 - h
                grad_j = 0.0
                for r in range(2 * n - 1):
                    if raw_full[r] < float(np.max(raw_full)) - 0.01:
                        continue
                    d_lag = r - (n - 1)
                    if d_lag >= 0:
                        m = n - d_lag
                        if j < m:
                            grad_j += u[j + d_lag]
                        if j >= d_lag:
                            grad_j -= h[j - d_lag]
                    else:
                        gp = -d_lag
                        m = n - gp
                        if j < m:
                            grad_j += u[j + gp]
                        if j >= gp:
                            grad_j -= h[j - gp]
                g_sub[ii] = grad_j
            step = 0.05
            h_sub_new = h_sub - step * g_sub
            h_sub_new = np.clip(h_sub_new, 0.0, 1.0)
            h_trial = h.copy()
            h_trial[idx] = h_sub_new
            h_trial = _project(h_trial, target)
            c5_trial = _exact_c5(h_trial)
            if c5_trial < cur_c5 - 1e-14:
                h = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                h_sub = h[idx].copy()
            else:
                step *= 0.6
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick gradient optimization (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _main_optimize(h_init, d1, beta=15.0, radius=0.3, stall_limit=60, use_lp=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP-enhanced optimization (15%)
    d2 = t0 + budget_s * 0.23
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _main_optimize(h2, d2, beta=25.0, radius=0.2, stall_limit=150, use_lp=True, lp_delta=0.04, lp_every=3)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary refinement (12%)
    d3 = t0 + budget_s * 0.35
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.0025)
    sub_dl = min(d3, time.time() + budget_s * 0.04)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d3:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_pol, c5_p = _main_optimize(h_pol, sub_dl, beta=20.0, radius=0.12, stall_limit=40, use_lp=True, lp_delta=0.03)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (30%)
    d4 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    n_candidates = []
    for dn in range(-24, 25, 2):
        n_try = n_curr + dn
        if 100 <= n_try <= 260 and n_try % 2 == 0:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))
    results = []
    per_n_time = max(1.0, (d4 - time.time()) * 0.4 / max(len(n_candidates), 1))
    for n_try in n_candidates:
        if time.time() > d4 - 5:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = time.time() + per_n_time
        h_r, c5_r = _main_optimize(h_r, sub_dl, beta=18.0, radius=0.22, stall_limit=50, use_lp=True, lp_delta=0.04)
        if time.time() < d4 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d4 - 2, time.time() + per_n_time * 0.4)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        results.append((c5_r, h_r, n_try))
    results.sort(key=lambda x: x[0])
    # Refine top 3
    for c5_r, h_r, n_try in results[:3]:
        if time.time() > d4 - 3:
            break
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d4, time.time() + budget_s * 0.015)
            h_ref = _project(h_r.copy(), n_try / 2.0)
            h_ref, c5_ref = _main_optimize(h_ref, sub_dl, beta=30.0, radius=0.10, stall_limit=40, use_lp=True, lp_delta=0.02)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 5: Intensive final optimization (22%)
    d5 = t0 + budget_s * 0.87
    n_curr = len(best_h)
    # Deep gradient + LP
    sub_dl = min(d5, time.time() + budget_s * 0.08)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _main_optimize(h5, sub_dl, beta=35.0, radius=0.12, stall_limit=200, use_lp=True, lp_delta=0.015, lp_every=2)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Pair swap
    if time.time() < d5 - 2:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.06)
        h5, c5_5 = _pair_swap_cont(h5, sub_dl, n_pairs=12000, radius=0.20, seed=444)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Binary round
    if time.time() < d5 - 2:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.0012)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Subspace optimization
    if time.time() < d5 - 1:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h5, c5_5 = _subspace_opt(h5, sub_dl, ns=8, iter_per_sub=80)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Final gradient polish
    if time.time() < d5:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h5, c5_5 = _main_optimize(h5, sub_dl, beta=40.0, radius=0.06, stall_limit=60, use_lp=True, lp_delta=0.01)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping (13%)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 3 and n_restarts < 150:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 - 2, time.time() + budget_s * 0.005)
        h_t, c5_t = _main_optimize(h_pert, sub_dl, beta=25.0, radius=0.10, stall_limit=30, use_lp=True, lp_delta=0.02)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)

    # Final polish
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _main_optimize(h6, deadline, beta=50.0, radius=0.04, stall_limit=40, use_lp=True, lp_delta=0.01, lp_every=2)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=2000, radius=0.08, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
