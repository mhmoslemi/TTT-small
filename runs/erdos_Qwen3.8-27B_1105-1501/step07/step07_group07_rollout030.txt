        return np.ones(n)
    lo = -1.0

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:

        else:

            x[mask] += res / cnt

            h = h_new
            alpha *= 0.999
    return best_h, best_c5

def _slsqp_polish(h, deadline, tau=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    def _obj(v):
        p, m, L = _lse_weights(v, tau)
        return m + tau * np.log(L)

    def _obj_grad(v):
        p, m, L = _lse_weights(v, tau)
        return m + tau * np.log(L), _lse_grad(v, p)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - target, 'jac': lambda v: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        if res.success:
            h_new = _project_sum(np.clip(res.x, 0.0, 1.0), target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    except Exception:
        pass
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_sweep(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _lp_direction(h, scores, G, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m])
                C[r] = np.sum(d[:m] * d[lag:lag + m])
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m])
                C[r] = np.sum(d[g:] * d[:m])
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2, f2 = c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b = c1
                c1, f1 = c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_loop(h, deadline, radius_init=0.10, top_k_init=4, max_top_k=10, stall_limit=80):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * p, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.08
                top_k = top_k_init
    return best_h, best_c5

def _random_binary(n, rng, k=20):
    best_h = None
    best_c5 = float('inf')
    for _ in range(k):
        idx = rng.choice(n, size=n // 2, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 - 1)
        n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE projected gradient (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lse_pg_loop(h_init, d1, tau_start=8.0, tau_end=0.1, n_tau=10, alpha_init=0.06)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP + LP (15%)
    d2 = t0 + budget_s * 0.30
    if time.time() < d2:
        h2, c5_2 = _slsqp_polish(best_h, min(d2, time.time() + budget_s * 0.06), tau=0.05)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    if time.time() < d2:
        h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.08, top_k_init=4, max_top_k=10, stall_limit=60)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Binary optimization (15%)
    d3 = t0 + budget_s * 0.45
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.07)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=31337)
    sub_dl2 = min(d3, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=30)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < d3:
        h_pol = _project_sum(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _lse_pg_loop(h_pol, sub_dl3, tau_start=3.0, tau_end=0.05, n_tau=8, alpha_init=0.04)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (25%)
    d4 = t0 + budget_s * 0.70
    n_best = len(best_h)
    tried_ns = set()
    n_offsets = list(range(-20, 21, 4))
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 220 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_pg_loop(h_r, sub_dl, tau_start=4.0, tau_end=0.1, n_tau=6, alpha_init=0.05)
        if time.time() < d4 - 2:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.06, top_k_init=3, max_top_k=8, stall_limit=30)
        if c5_r < best_c5 + 0.001 and time.time() < d4 - 2:
            h_b = _to_binary(h_r)
            sub_dl3 = min(d4, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl3, max_iter=15000, T_init=0.002, seed=n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Random binary restarts at current best n
    if time.time() < d4:
        n_cur = len(best_h)
        h_rb, c5_rb = _random_binary(n_cur, rng, k=15)
        if c5_rb < best_c5 + 0.005:
            sub_dl = min(d4, time.time() + budget_s * 0.02)
            h_rb, c5_rb = _lse_pg_loop(h_rb, sub_dl, tau_start=3.0, tau_end=0.05, n_tau=6, alpha_init=0.05)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()

    # Phase 5: Intensive refinement (15%)
    d5 = t0 + budget_s * 0.85
    n_cur = len(best_h)
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=6, stall_limit=100)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h5, c5_5 = _lse_pg_loop(h5, sub_dl, tau_start=2.0, tau_end=0.03, n_tau=8, alpha_init=0.03)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    if time.time() < d5:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=35000, T_init=0.0015, seed=777)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h5, c5_5 = _slsqp_polish(h5, sub_dl, tau=0.03)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping + final (10%)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 15 and n_restarts < 150:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d6 - 15, time.time() + budget_s * 0.003)
        h_t, c5_t = _lse_pg_loop(h_pert, sub_dl, tau_start=2.0, tau_end=0.05, n_tau=5, alpha_init=0.04)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.07)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline - 2, time.time() + budget_s * 0.02)
        h6, c5_6 = _lp_loop(h6, sub_dl, radius_init=0.02, top_k_init=3, max_top_k=5, stall_limit=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, deadline - 1, max_rounds=12)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
