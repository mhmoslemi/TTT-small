def _to_binary(h):

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):

def _pair_sweep(h_bin, deadline, max_passes=3):

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _lag_target_opt(h, deadline, n_top=3):

        # Build a direction that reduces the top lags
        g = np.zeros(n)
        for r in top_idx:

            if tau >= 0:
                m = n - tau
                if m > 0:
                    g[tau:] += (1.0 - h[:m])
                    g[:m] -= h[tau:tau + m]
            else:
                gg = -tau
                m = n - gg
                if m > 0:
                    g[:m] += (1.0 - h[gg:])
                    g[gg:] -= h[:m]
        g -= np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        d = g / gnorm
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-14 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)

                h = h_new
                cur_c5 = nc

                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.35)
            else:
                stall += 1
                radius = max(0.02, radius * 0.75)
        else:
            # Try random directions as escape
            d = rng.standard_normal(n)
            d -= np.mean(d)
            d /= np.linalg.norm(d)
            alpha, c_try = _line_search(h, d, radius * 0.5)

                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _binary_double_swap(h_bin, deadline, max_iter=20000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 600 or len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[int(rng.integers(0, len(ones)))]
        i2 = ones[int(rng.integers(0, len(ones)))]
        if i1 == i2:
            continue
        j1 = zeros[int(rng.integers(0, len(zeros)))]
        j2 = zeros[int(rng.integers(0, len(zeros)))]
        if j1 == j2:
            continue
        cur[i1] = 0.0
        cur[i2] = 0.0
        cur[j1] = 1.0
        cur[j2] = 1.0
        nc = _exact_c5(cur)
        if nc < cur_c5 - 1e-14:
            cur_c5 = nc
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            stall = 0
        else:
            cur[i1] = 1.0
            cur[i2] = 1.0
            cur[j1] = 0.0
            cur[j2] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline, max_passes=2):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0].tolist()
        zeros = np.where(cur <= 0.5)[0].tolist()
        for oi in ones:
            if time.time() > deadline:
                break
            best_oz = None
            best_c = cur_c5
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_oz = oz
                cur[oi] = 1.0
                cur[oz] = 0.0
            if best_oz is not None:
                cur[oi] = 0.0
                cur[best_oz] = 1.0
                cur_c5 = best_c
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=4000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-14 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5
