        tm_idx += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-14)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
            T = max(T * 1.01, T_init * 0.1)
    return best_h, best_c5

def _binary_greedy_sweep(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in ones[:]:
            if time.time() > deadline:
                break
            for oz in zeros[:]:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _linsc_direction(h, S, move_cap):
    """LP-based direction: minimize max of linearized rows."""
    n = len(h)
    L = 2 * n - 1
    G = _build_grad_matrix(h)
    # We want to find d (zero-sum) that minimizes max_r (S_r + G_r . d)
    # Subject to: |d_i| <= move_cap for all i
    # This is: min C s.t. G d + S <= C*ones, sum(d)=0, |d|<=cap
    # Use coordinate-wise approach as LP solver may not be available
    # Fallback: use the dense direction with all rows weighted
    # Actually let's just use a simpler approach:
    # Find direction that reduces the top rows using weighted gradient
    S_max = float(np.max(S))
    w = np.exp(30.0 * (S - S_max))
    w /= float(np.sum(w)) + 1e-30
    coord_grad = G.T @ w
    # Normalize
    max_abs = float(np.max(np.abs(coord_grad)))
    if max_abs < 1e-15:
        return None
    d = -coord_grad / max_abs  # descend direction
    d -= np.mean(d)
    max_abs2 = float(np.max(np.abs(d)))
    if max_abs2 < 1e-15:
        return None
    d = d / max_abs2 * move_cap
    return d

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Dense optimization at parent n (25% time)
    d1 = t0 + budget_s * 0.25
    K_list = [4, 6, 8, 5, 10, 7, 12]
    h1, c5_1 = _dense_optimize(h_init, d1, K_list, 50000, rng, 0.18, beta=20.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multigrid exploration (35% time)
    d2 = t0 + budget_s * 0.60
    offsets = [-8, 8, -4, 4, -12, 12, -16, 16, -20, 20]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d2 - 5:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2 - 5, time.time() + budget_s * 0.035)
        K_list_r = [3, 5, 7, 4, 8, 6]
        h_r, c5_r = _dense_optimize(h_r, sub_dl, K_list_r, 20000, rng, 0.15, beta=20.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    if time.time() < d3 - 2:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d3 - 2, time.time() + budget_s * 0.07)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.004, seed=777)
        sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_greedy_sweep(h_bin, sub_dl2, max_rounds=30)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary result
        if time.time() < d3 - 2:
            n_curr = len(best_h)
            h_pol = _project(best_h.copy(), n_curr / 2.0)
            sub_dl3 = min(d3 - 2, time.time() + budget_s * 0.04)
            K_list_p = [5, 7, 9, 6, 11]
            h_pol, c5_pol = _dense_optimize(h_pol, sub_dl3, K_list_p, 15000, rng, 0.10, beta=25.0)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 4: Intensive refinement (20% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    K_list_f = [6, 8, 10, 5, 12, 7, 14]
    h4, c5_4 = _dense_optimize(h4, d4, K_list_f, 30000, rng, 0.12, beta=25.0)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Try nearby n values
    n_curr = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > d4 - 3:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl2 = min(d4 - 3, time.time() + budget_s * 0.015)
        h_r, c5_r = _dense_optimize(h_r, sub_dl2, [4, 6, 8], 8000, rng, 0.10, beta=22.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Final polish (5% time)
    d5 = deadline - 2
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _dense_optimize(h5, d5, [8, 10, 12, 6], 10000, rng, 0.06, beta=30.0)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
