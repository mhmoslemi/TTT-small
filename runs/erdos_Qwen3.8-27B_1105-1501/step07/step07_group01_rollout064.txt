                f1 = float(np.max(raw0 + c1*T - c1*c1*Q)) * dx
            else:
                lo_a = c1; c1 = c2; f1 = f2
                c2 = lo_a + gr*(hi_a - lo_a)
                f2 = float(np.max(raw0 + c2*T - c2*c2*Q)) * dx
        best_a = (lo_a + hi_a) / 2.0
        best_obj = float(np.max(raw0 + best_a*T - best_a*best_a*Q)) * dx
    return best_a, best_obj

def _fw_iteration(h, top_k, weight_mode, n_grid):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    raw0 = np.correlate(h, q, mode='full')
    cur_c5 = float(np.max(raw0)) * dx
    
    g = _fw_gradient(h, raw0, top_k, weight_mode)
    target = n / 2.0
    v = _fw_vertex(g, target)
    p = v - h
    
    if np.max(np.abs(p)) < 1e-10:
        return h, cur_c5
    
    T = np.correlate(p, q, mode='full') - np.correlate(h, p, mode='full')
    Q = np.correlate(p, p, mode='full')
    
    best_a, _ = _line_search_c5(raw0, T, Q, dx, n_grid)
    
    if best_a < 1e-8:
        return h, cur_c5
    
    new_h = h + best_a * p
    new_h = _project(new_h, target)
    new_c5 = _exact_c5(new_h)
    
    return new_h, new_c5

def _coord_descent(h, deadline, max_sweeps=2000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(raw)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            q = 1.0 - h
            # Compute gradient at j
            g = 0.0
            # We need d(max_r raw[r])/d(h[j])
            # Only active (top) constraints matter, but for simplicity use all
            # Actually let's use the subgradient from all constraints weighted by proximity to max
            # For efficiency, just use the top few
            top_idx = np.argsort(raw)[::-1][:10]
            for r in top_idx:
                s = r - (n - 1)
                if s >= 0:
                    if s <= j < n and 0 <= j-s <= n-s-1:
                        g += (1.0 - h[j-s])
                    if 0 <= j <= n-s-1:
                        g -= h[j+s]
                else:
                    b = -s
                    if 0 <= j <= n-b-1:
                        g += (1.0 - h[j+b])
                    if b <= j < n:
                        g -= h[j-b]
            
            # Line minimization: we want to move h[j] in direction -sign(g)
            # h[j] + delta, where delta is in [-old, 1-old]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = lo_d + gr*(hi_d - lo_d)
            c2 = hi_d - gr*(hi_d - lo_d)
            
            # Compute the effect of changing h[j] by delta
            # raw_new[r] = raw[r] + delta * d(raw[r])/d(h[j])
            # For each r, the partial derivative is as computed above
            # But for efficiency, precompute the contribution vector
            contrib = np.zeros(2*n - 1)
            s0 = 0  # shift 0
            # For shift s >= 0:
            for s in range(0, n):
                r_pos = n - 1 + s
                m = n - s
                if 0 <= j <= m - 1:
                    contrib[r_pos] -= h[j + s] if j + s < n else 0
                if s <= j < n:
                    contrib[r_pos] += (1.0 - h[j - s]) if 0 <= j - s < m else 0
            # For shift s < 0 (b = -s > 0):
            for b in range(1, n):
                r_neg = n - 1 - b
                m = n - b
                if 0 <= j <= m - 1:
                    contrib[r_neg] += (1.0 - h[j + b]) if j + b < n else 0
                if b <= j < n:
                    contrib[r_neg] -= h[j - b] if 0 <= j - b < m else 0
            
            f1 = float(np.max(raw + c1 * contrib)) * dx
            f2 = float(np.max(raw + c2 * contrib)) * dx
            for _ in range(40):
                if hi_d - lo_d < 1e-10:
                    break
                if f1 < f2:
                    hi_d = c2; c2 = c1; f2 = f1
                    c1 = lo_d
