        hf, cf = _coord_descent(hf, deadline, max_iter=50000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _grad_local(hf, deadline, r0=0.02, stall_lim=25, nd=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
