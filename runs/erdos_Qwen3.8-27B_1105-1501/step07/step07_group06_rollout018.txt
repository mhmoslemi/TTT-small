def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        gg = -tau
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _grad_mixed(h, raw, beta, k_top, rng):

    if np.any(neg):

    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=60000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d3:
        h_b = _to_binary(best_h)
        sub_dl3 = min(d3, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_block_search(h_b, sub_dl3, block_size=2, max_iter=15000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d3:
        h_b = _to_binary(best_h)
        sub_dl4 = min(d3, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_reversal_search(h_b, sub_dl4, max_iter=8000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d3:
        h_pol = _project(h_b.copy(), n_b / 2.0)
        sub_dl5 = min(d3, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _opt_smooth(h_pol, sub_dl5, beta_lo=10.0, beta_hi=60.0, radius_init=0.10, max_stall=40, k_top=10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    d4 = t0 + budget_s * 0.82
    n_c = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = _project(best_h.copy(), n_c / 2.0)
    h4, c5_4 = _opt_smooth(h4, sub_dl, beta_lo=15.0, beta_hi=80.0, radius_init=0.10, max_stall=80, k_top=12)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_continuous(h4, sub_dl2, n_iter=5000, radius=0.18, seed=444)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _block_continuous(h4, sub_dl3, n_iter=3000, radius=0.12, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl4 = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _opt_smooth(h4, sub_dl4, beta_lo=30.0, beta_hi=120.0, radius_init=0.06, max_stall=50, k_top=15)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 10 and n_restarts < 100:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.006)
        h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=10.0, beta_hi=60.0, radius_init=0.08, max_stall=30, k_top=10)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.07, scale * 1.1)

    n_final = len(best_h)
    for dn in [-6, 6, -4, 4, -8, 8, -10, 10]:
        if time.time() > d5 - 15:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.01)
        h_r, c5_r = _opt_smooth(h_r, sub_dl, beta_lo=15.0, beta_hi=60.0, radius_init=0.08, max_stall=35, k_top=10)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_smooth(h_f, deadline, beta_lo=20.0, beta_hi=100.0, radius_init=0.05, max_stall=40, k_top=12)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
