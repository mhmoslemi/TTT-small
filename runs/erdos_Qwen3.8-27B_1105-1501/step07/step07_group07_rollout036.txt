                h[j] = d; h[l] = S - d
                fd = float(np.max(np.correlate(h, 1.0-h, mode='full')))
        
        t_opt = (a + b) / 2.0
        h[j] = t_opt
        h[l] = S - t_opt
        new_c5 = _exact_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5
