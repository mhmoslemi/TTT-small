def _project(v, target):

        return np.ones(n)

def _build_gradient_row(h, k, n):

def _lp_direction(h, active_indices, n, dx, radius):
    """Solve LP to find best descent direction."""
    m = len(active_indices)
    # Build gradient matrix for active constraints
    # Each row is the gradient (scaled by dx) of F_r for that constraint
    A_rows = []
    b_vals = []
    for idx in active_indices:
        k = idx - (n - 1)
        g = _build_gradient_row(h, k, n)
        A_rows.append(g)
        # F_r(h) + g·d <= t  =>  g·d - t <= -F_r(h)
        # We'll store the value F_r(h) separately
    G = np.array(A_rows)  # shape (m, n)
    
    # LP: minimize t
    # Variables: d[0..n-1], t
    # Constraints:
    #   G @ d - t <= -F_r(h) for each active r  (m constraints)
    #   sum(d) = 0 (equality)
    #   -lo[i] <= d[i] <= hi[i] (bounds)
    
    # Compute F_r(h) for each active constraint
    corr = np.correlate(h, 1.0 - h, mode='full')
    F_vals = corr[active_indices] * dx  # scaled values
    
    # Build A_ub, b_ub
    # G @ d + (-1) * t <= -F_vals
    A_ub = np.zeros((m + 1, n + 1))
    A_ub[:m, :n] = G
    A_ub[:m, n] = -1.0  # coefficient of t
    b_ub = -F_vals  # G@d - t <= -F_r(h)
    
    # Mass constraint: sum(d) = 0 -> as equality
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: [(-lo[i], hi[i]) for d], [(-inf, inf) for t]
    bounds = []
    for i in range(n):
        lo_i = min(radius, h[i])
        hi_i = min(radius, 1.0 - h[i])
        bounds.append((-lo_i, hi_i))
    bounds.append((None, None))
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            t = res.x[n]
            # Check if direction is meaningful
            if np.max(np.abs(d)) > 1e-10:
                return d
            return None
        else:
            return None
    except Exception:
        return None

        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        scaled = corr * dx

        threshold = max_val - max(0.005, 0.05 * max_val)
        active = np.where(scaled >= threshold)[0]
        if len(active) > max_active:
            # Take top max_active
            order = np.argsort(-scaled)
            active = order[:max_active]

                g_sum += _build_gradient_row(h, k, n)
            g_sum /= len(active)
            d = -g_sum
            d = d - np.mean(d)
            d = np.clip(d, -radius, radius)
            d = d - np.mean(d)
            lp_fail += 1
        else:
            lp_fail = 0
        
        # Line search
        alpha, new_c5 = _line_search(h, d, dx, radius)

            h_new = _project(h_new, target)

                cur = cur_new
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.30)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

        return np.ones(n)
    lo = float(v.min()) - 5.0
    hi = float(v.max()) + 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_row(h, k, n):
    """Gradient of F_k (scaled by dx) w.r.t. h, where k is offset from center."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        u = -k
        g[:n - u] -= h[u:]
        g[u:] += (1.0 - h[:n - u])
    return g * dx


def _line_search(h, d, dx, radius, n_grid=50):
    S0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')

    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)

    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * L1 - t * t * Q2)) * dx

    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    if hi_b - lo_b < 1e-11:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * L1 - c1 * c1 * Q2)) * dx
        f2 = float(np.max(S0 + c2 * L1 - c2 * c2 * Q2)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * L1 - c1 * c1 * Q2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * L1 - c2 * c2 * Q2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * L1 - opt_t * opt_t * Q2)) * dx

    cands = [(opt_t, opt_val),
             (0.0, float(np.max(S0) * dx)),
             (t_hi, float(np.max(S0 + t_hi * L1 - t_hi * t_hi * Q2)) * dx)]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]


def _lp_direction(h, active, n, radius, cur_max):
    """LP: minimize t s.t. F_r(h)+G_r·d <= t, sum(d)=0, |d_i|<=cap."""
    m = len(active)
    if m == 0:
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    F_vals = corr[active] * (2.0 / n)

    # Build gradient matrix (m x n)
    G = np.zeros((m, n))
    for i, idx in enumerate(active):
        k = idx - (n - 1)
        G[i] = _grad_row(h, k, n)

    var_n = n + 1
    # A_ub: G@d - t <= -F_vals  =>  [G | -1] @ [d; t] <= -F_vals
    A_ub = np.zeros((m, var_n))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -F_vals

    A_eq = np.zeros((1, var_n))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    bounds = []
    for i in range(n):
        lo_i = min(radius, h[i] + 1e-12)
        hi_i = min(radius, 1.0 - h[i] + 1e-12)
        bounds.append((-lo_i, hi_i))
    bounds.append((-10.0, cur_max + 0.1))

    c_obj = np.zeros(var_n)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-9:
                return d
    except Exception:
        pass
    return None


def _subgrad_fallback(h, active, n, radius):
    g_sum = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        g_sum += _grad_row(h, k, n)
    g_sum /= len(active)
    d = -g_sum
    d = d - np.mean(d)
    d = np.clip(d, -radius, radius)
    d = d - np.mean(d)
    return d


def _lp_optimize(h, deadline, radius_init=0.08, max_active=30, stall_limit=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    lp_fails = 0

    while time.time() < deadline:
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        scaled = corr * (2.0 / n)
        max_val = float(scaled.max())

        thresh = max_val - max(0.003, 0.08 * max_val)
        active = np.where(scaled >= thresh)[0]
        if len(active) > max_active:
            order = np.argsort(-scaled)
            active = order[:max_active]

        d = _lp_direction(h, list(active), n, radius, max_val)
        if d is not None:
            lp_fails = 0
        else:
            d = _subgrad_fallback(h, list(active), n, radius)
            lp_fails += 1
            if lp_fails > 15:
                break

        alpha, new_c5 = _line_search(h, d, 2.0 / n, radius)

        if new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            val = _exact_c5(h_new)
            if val < cur - 1e-14:
                h = h_new
                cur = val
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.35)
            else:
                stall += 1
                radius *= 0.85
        else:
            stall += 1
            radius *= 0.85

    return best_h, best_c5


def _binary_swap_optimize(h, deadline, max_iters=100000, seed=42):
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    if abs(np.sum(h) - target) > 1e-10:
        h = _project(h, target)
        h = np.where(h > 0.5, 1.0, 0.0).astype(float)

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        if not ones or not zeros:
            break

        # Pick a move: random or targeted based on correlation
        if len(ones) > 50 and len(zeros) > 50 and rng.random() < 0.6:
            # Targeted: try to reduce max correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = int(np.argmax(corr))
            k = max_idx - (n - 1)
            # Positions involved in max correlation
            if k >= 0:
                cand_ones = [i for i in ones if i + k < n]
                cand_zeros = [j for j in zeros if j - k >= 0]
            else:
                u = -k
                cand_ones = [i for i in ones if i - u >= 0]
                cand_zeros = [j for j in zeros if j + u < n]
            if cand_ones and cand_zeros:
                oi = cand_ones[int(rng.integers(0, len(cand_ones)))]
                oz = cand_zeros[int(rng.integers(0, len(cand_zeros)))]
            else:
                oi = ones[int(rng.integers(0, len(ones)))]
                oz = zeros[int(rng.integers(0, len(zeros)))]
        else:
            oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5

        # Accept if improvement or small probability of worsening
        if delta < -1e-14 or (abs(delta) < 1e-14 and rng.random() < 0.1):
            cur_c5 = new_c5
            # Update lists
            for li, idx in enumerate(ones):
                if idx == oi:
                    ones[li] = oz
                    break
            for li, idx in enumerate(zeros):
                if idx == oz:
                    zeros[li] = oi
                    break
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            stall += 1

    return best_h, best_c5


def _targeted_binary(h, deadline, max_iters=50000, n_try=50, seed=99):
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        best_delta = 0.0
        best_move = None
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            h[oi] = 0.0
            h[oz] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
            h[oi] = 1.0
            h[oz] = 0.0

        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            h[oi] = 0.0
            h[oz] = 1.0
            cur_c5 = cur_c5 + best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    pert = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.3 * rng.standard_normal(n)
    return _project(h + pert, n / 2.0)


def _structured_init(n):
    """Create structured initial patterns."""
    h = np.zeros(n)
    half = n // 2
    # Pattern 1: block pattern
    block_size = max(1, n // 4)
    for i in range(n):
        cycle = i // (2 * block_size) % 2
        if cycle == 0:
            h[i] = 1.0
    h = _project(h, half)

    # Pattern 2: alternating with period
    h2 = np.zeros(n)
    period = max(2, n // 8)
    fill = period // 2
    for i in range(n):
        if (i % period) < fill:
            h2[i] = 1.0
    h2 = _project(h2, half)

    # Pattern 3: random
    rng = np.random.default_rng(12345)
    h3 = (rng.standard_normal(n) > 0).astype(float)
    h3 = _project(h3, half)

    return [h, h2, h3]


def _pair_swap_optimize(h, deadline, n_pairs=5000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, 2.0 / n, radius)
        if c5_try < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            val = _exact_c5(h_new)
            if val < cur - 1e-14:
                h = h_new
                cur = val
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Load initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        h_init = _project(h0.copy(), n0 / 2.0)
        best_h = h_init.copy()
        best_c5 = _exact_c5(h_init)
    else:
        n0 = 180
        best_h = None
        best_c5 = float('inf')
        # Generate structured inits
        for h_s in _structured_init(n0):
            c5_s = _exact_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 1: LP optimization on initial (15% budget)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lp_optimize(best_h, d1, radius_init=0.08, max_active=30, stall_limit=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary optimization (25%)
    d2 = t0 + budget_s * 0.40
    n_curr = len(best_h)

    # Targeted binary
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_t, c5_t = _targeted_binary(best_h, sub_dl, max_iters=40000, n_try=40, seed=101)
    if c5_t < best_c5:
        best_c5 = c5_t
        best_h = h_t.copy()

    # Random binary SA
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_s, c5_s = _binary_swap_optimize(best_h, sub_dl, max_iters=150000, seed=202)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Another targeted pass
    sub_dl = min(d2, time.time() + budget_s * 0.07)
    h_t, c5_t = _targeted_binary(best_h, sub_dl, max_iters=40000, n_try=60, seed=303)
    if c5_t < best_c5:
        best_c5 = c5_t
        best_h = h_t.copy()

    # LP polish on binary result
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_pol = _project(best_h.copy(), n_curr / 2.0)
    h_pol, c5_pol = _lp_optimize(h_pol, sub_dl, radius_init=0.05, max_active=25, stall_limit=25)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (30%)
    d3 = t0 + budget_s * 0.70
    n_best = len(best_h)
    tried_ns = set()

    offsets = list(range(-30, 31, 3))
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.01:
            continue
        # Short LP
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.07, max_active=20, stall_limit=20)
        # Binary
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b = _targeted_binary(h_b, sub_dl2, max_iters=25000, n_try=30, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Try structured inits at a few sizes
    for n_try in [n_best, n_best + 6, n_best - 6, n_best + 12, n_best - 12, n_best + 18]:
        if time.time() > d3:
            break
        if n_try < 60 or n_try > 260:
            continue
        for h_s in _structured_init(n_try):
            if time.time() > d3:
                break
            c5_s = _exact_c5(h_s)
            if c5_s < best_c5 + 0.01:
                sub_dl = min(d3, time.time() + budget_s * 0.015)
                h_s, c5_s = _lp_optimize(h_s, sub_dl, radius_init=0.06, max_active=20, stall_limit=15)
                if time.time() < d3 - 2:
                    h_b = _to_binary(h_s)
                    sub_dl2 = min(d3, time.time() + budget_s * 0.008)
                    h_b, c5_b = _targeted_binary(h_b, sub_dl2, max_iters=20000, n_try=30, seed=5000 + n_try)
                    if c5_b < c5_s:
                        c5_s = c5_b
                        h_s = h_b
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()

    # Phase 4: Intensive refinement (25%)
    d4 = deadline
    n_fin = len(best_h)

    # LP polish
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4 = _project(best_h.copy(), n_fin / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, radius_init=0.04, max_active=35, stall_limit=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Pair swap
    if time.time() < d4 - 20:
        h4 = _project(best_h.copy(), n_fin / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_optimize(h4, sub_dl, n_pairs=6000, radius=0.20, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Binary targeted on best
    if time.time() < d4 - 25:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=60000, n_try=50, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Basin hopping
    if time.time() < d4 - 15:
        scale = 0.015
        for _ in range(40):
            if time.time() > d4 - 5:
                break
            h_pert = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.005)
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.04, max_active=20, stall_limit=15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.1)

    # Final: try a few nearby n
    for dn in [-2, 2, -4, 4]:
        if time.time() > d4 - 6:
            break
        n_try = n_fin + dn
        if n_try < 50:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 1, time.time() + 4)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.03, max_active=20, stall_limit=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final LP polish
    if time.time() < d4 - 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, d4 - 1, radius_init=0.02, max_active=30, stall_limit=20)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
