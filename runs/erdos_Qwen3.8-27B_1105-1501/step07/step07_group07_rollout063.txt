    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    G[n-1, :] = 1.0 - 2.0*h
    for d in range(1, n):
        row = n - 1 + d
        m = n - d
        G[row, d:] += (1.0 - h[:m])
        G[row, :m] += -h[d:]
    for g in range(1, n):
        row = n - 1 - g
        m = n - g
        G[row, :m] += (1.0 - h[g:])
        G[row, g:] += -h[:m]
    return G

def _lse_line_search(h, d, dx, t_max):
    n = len(h)
    raw0 = _scores(h)
    # Compute T_r and C_r for the direction d
    G = _build_grad(h)
    T = G @ d  # T[r] = G[r,:] @ d
    # C_r = sum of d[i]*d[i+d] over valid range
    C = np.zeros(2*n-1)
    for lag in range(-(n-1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                C[r] = float(np.sum(d[:m] * d[lag:lag+m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                C[r] = float(np.sum(d[g:] * d[:m]))
    
    # Now LSE(t) = tau * log(sum exp((raw0[r] + T[r]*t - C[r]*t^2) / tau))
    # We want to minimize max(raw(t)) * dx, which is approximated by LSE(t) * dx
    # But for exactness, let's just evaluate the actual max
    
    # Feasibility: h + t*d must be in [0,1]
    # t must satisfy: -h[i]/d[i] >= 0 for d[i] < 0, and (1-h[i])/d[i] >= 0 for d[i] > 0
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = t_max
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search + golden section
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    best_t = 0.0
    best_val = _exact_c5(h)
    for t in grid_t[1:]:
        h_t = h + t * d
        c5_t = _exact_c5(h_t)
        if c5_t < best_val:
            best_val = c5_t
            best_t = t
    
    # Golden section around best
    idx = int(np.argmin([_exact_c5(h + t * d) for t in grid_t]))
    lo = grid_t[max(0, idx-2)]
    hi = grid_t[min(n_grid-1, idx+2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1; c1 = c2; f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_t = (lo + hi) / 2.0
    opt_val = _exact_c5(h + opt_t * d)
    
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

def _lse_optimize(h, deadline, n_tau=6):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Tau schedule
    # raw scores are around 30-40 for C5 ~ 0.38, dx ~ 0.011
    raw_s = _scores(h)
    tau0 = 0.5 * (raw_s.max() - raw_s.min())  # initial tau in raw units
    tau_sched = [tau0 * (0.6 ** i) for i in range(n_tau)]
    
    for tau in tau_sched:
        if time.time() > deadline:
            break
        # Multiple iterations at this tau
        for iter in range(50):
            if time.time() > deadline:
                break
            G = _build_grad(h)
            raw = _scores(h)
            mval = raw.max()
            u = (raw - mval) / tau
            # Clip for numerical stability
            u = np.clip(u, -50, 50)
            z = np.exp(u)
            z_sum = z.sum()
            if z_sum < 1e-10:
                probs = np.zeros_like(raw)
                probs[raw == mval] = 1.0 / np.sum(raw == mval)
            else:
                probs = z / z_sum
            
            d = G.T @ probs  # gradient direction (negative gradient for minimization)
            
            # Normalize d
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-15:
                break
            d = d / d_norm
            
            # Line search
            t, new_c5 = _lse_line_search(h, -d, dx, t_max=0.5)
            # Wait, we want to minimize, so direction is -gradient
            # But the line search already minimizes along the direction
            # So d should be the direction we move in
            
            if t > 0 and new_c5 < cur_c5 - 1e-14:

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        row = n - 1 + d
        m = n - d
        G[row, d:] += (1.0 - h[:m])
        G[row, :m] += -h[d:]
    for g in range(1, n):
        row = n - 1 - g
        m = n - g
        G[row, :m] += (1.0 - h[g:])
        G[row, g:] += -h[:m]
    return G

def _smooth_obj(x):
    tau = _g_tau
    s = np.correlate(x, 1.0 - x, mode='full')
    mval = float(np.max(s))
    u = (s - mval) / tau
    u = np.clip(u, -60.0, 60.0)
    z = np.exp(u)
    zs = float(z.sum())
    if zs < 1e-300:
        return 1e10
    return mval + tau * np.log(zs)

def _smooth_jac(x):
    tau = _g_tau
    n = _g_n
    s = np.correlate(x, 1.0 - x, mode='full')
    mval = float(np.max(s))
    u = (s - mval) / tau
    u = np.clip(u, -60.0, 60.0)
    z = np.exp(u)
    zs = float(z.sum())
    if zs < 1e-300:
        return np.zeros(n)
    probs = z / zs
    G = _build_grad(x)
    return G.T @ probs

def _eq_con(x):
    return _g_target - float(np.sum(x))

def _eq_jac(x):
    return np.ones(_g_n)

def _line_search(h, d, t_max):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = t_max
    if np.any(pos):

    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)

    n_grid = 25
    grid_t = np.linspace(0.0, t_hi, n_grid)
    vals = np.empty(n_grid)
    for idx in range(n_grid):
        vals[idx] = _exact_c5(h + grid_t[idx] * d)
    best_idx = int(np.argmin(vals))
    best_t = float(grid_t[best_idx])
    best_val = float(vals[best_idx])

    lo = float(grid_t[max(0, best_idx - 2)])
    hi = float(grid_t[min(n_grid - 1, best_idx + 2)])
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * d)
        f2 = _exact_c5(h + c2 * d)
        for _ in range(30):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * d)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * d)
        opt_t = (lo + hi) / 2.0
        opt_val = _exact_c5(h + opt_t * d)
        if opt_val < best_val:

def _pair_search(h, deadline, max_rounds=200, top_k=6):

    target = n / 2.0

        raw = _scores(h)

        if top_k_actual == 0:
            break
        if top_k_actual > top_k:

            top_mask = np.zeros(2 * n - 1, dtype=bool)
            top_mask[top_idx] = True
            top_k_actual = top_k

        w = G[top_mask, :].sum(axis=0)
        w_max_idx = int(np.argmax(w))
        w_min_idx = int(np.argmin(w))

        if w_max_idx == w_min_idx:
            break

        d = np.zeros(n)
        d[w_w_max_idx] = 1.0 if False else 0
        # Move mass from w_max to w_min: h[j] += a, h[i] -= a where i=argmax(w), j=argmin(w)
        d[w_min_idx] = 1.0
        d[w_max_idx] = -1.0

        t, new_c5 = _line_search(h, d, 1.0)
        if t > 1e-12 and new_c5 < cur_c5 - 1e-14:

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Try top-5 candidates
            w_sorted_desc = np.argsort(w)[::-1]
            w_sorted_asc = np.argsort(w)
            found = False
            for ii in range(min(5, n)):
                for jj in range(min(5, n)):
                    i_cand = int(w_sorted_desc[ii])
                    j_cand = int(w_sorted_asc[jj])
                    if i_cand == j_cand:
                        continue
                    d2 = np.zeros(n)
                    d2[j_cand] = 1.0

                    t2, c2 = _line_search(h, d2, 1.0)

                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if not found:
                break

    return best_h, best_c5

def _lp_step(h, radius, top_k):
    n = len(h)
    target = n / 2.0
    G = _build_grad(h)
    raw = _scores(h)
    top_idx = np.argsort(raw)[::-1][:top_k]
    top_scores = raw[top_idx]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15, top_k_init=4, max_top_k=10, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0

    while time.time() < deadline - 2:
        if stall > stall_limit:
            break
        p = _lp_step(h, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        t, new_c5 = _line_search(h, p, radius)
        if t > 1e-12 and new_c5 < cur_c5 - 1e-14:
            h = _project_sum(h + t * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.1
                top_k = top_k_init
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.3, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Binary refinement
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        h_bin, c5_bin = _binary_pair_sweep(h_bin, d3, max_rounds=80)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.75
    n_best = len(best_h)
    n_candidates = []
    for offset in [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24]:
        n_try = n_best + offset
        if n_try >= 60 and n_try <= 250 and n_try % 2 == 0:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.time() > d4 - 5:
            break
        if n_try == n_best:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_r, c5_r = _slsqp_anneal(h_r, sub_dl, n_tau=4, iters_per_tau=20)
        if c5_r < best_c5 + 0.005 and time.time() < d4 - 3:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.08, top_k_init=3, max_top_k=8, stall_limit=30)
        if c5_r < best_c5 and time.time() < d4 - 2:
            sub_dl3 = min(d4, time.time() + budget_s * 0.005)
            h_r, c5_r = _pair_search(h_r, sub_dl3, max_rounds=50, top_k=6)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 5: Basin hopping with LSE (15%)
    d5 = t0 + budget_s * 0.90
    scale = 0.04
    for it in range(150):
        if time.time() > d5 - 2:
            break
        h_pert = _perturb(best_h, rng, scale)
        n_pert = len(h_pert)
        sub_dl = min(d5, time.time() + budget_s * 0.003)
        h_t, c5_t = _slsqp_anneal(h_pert, sub_dl, n_tau=3, iters_per_tau=15)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 6: Final intensive polish (10%)
    d6 = deadline
    n_final = len(best_h)
    # LP with very tight focus
    if time.time() < d6 - 5:
        h6 = _project_sum(best_h.copy(), n_final / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.03)
        h6, c5_6 = _lp_loop(h6, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=6, stall_limit=80)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # SlSQP polish
    if time.time() < d6 - 5:
        h6 = _project_sum(best_h.copy(), n_final / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.025)
        h6, c5_6 = _slsqp_anneal(h6, sub_dl, n_tau=4, iters_per_tau=25)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final pair search
    if time.time() < d6 - 3:
        h6 = _project_sum(best_h.copy(), n_final / 2.0)
        h6, c5_6 = _pair_search(h6, d6, max_rounds=80, top_k=5)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Binary final
    if time.time() < d6 - 2:
        h_bin = _to_binary(best_h)
        h_bin, c5_bin = _binary_pair_sweep(h_bin, d6, max_rounds=50)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
