def _grad_at_offset(h, offset):
    n = len(h)
    grad = np.zeros(n)
    if offset >= 0:
        m = n - offset
        if m <= 0:
            return grad
        # First term: h[i] for i in [0, m-1]
        grad[:m] += (1.0 - h[offset:offset+m])
        # Second term: (1-h[i+offset]) means -h[i+offset] for i+offset in [0, n-1]
        # i+offset ranges from offset to offset+m-1 = n-1
        # So h[i+offset] appears in the sum for i in [0, m-1]
        # Derivative w.r.t. h[j] where j = i+offset, i.e., j in [offset, n-1]
        # Coefficient is -h[i] = -h[j-offset]
        grad[offset:] -= h[:m]
    else:

        # result = sum_{j=0}^{m-1} h[j+o] * (1-h[j])
        # First term: h[j+o], j in [0, m-1], so h[k] for k in [o, n-1]

```python
def _lp_active_set(h, radius=0.2, max_lags=40):
    """Solve the active-set LP relaxation."""
    n = len(h)
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    
    # Start with the argmax lag
    active = [int(np.argmax(raw))]
    t = float(raw[active[0]])
    
    for iteration in range(max_lags):
        # Build LP
        # Variables: x[0..n-1], t (index n)
        # Minimize t
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality constraints: for each r in active, grad_r @ x - t <= b_r
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        
        for idx, r in enumerate(active):
            offset = r - (n - 1)
            grad = _grad_at_offset(h, offset)
            b_r = float(grad @ h) - float(raw[r])
            A_ub[idx, :n] = grad
            A_ub[idx, n] = -1.0
            b_ub[idx] = b_r
        
        # Equality: sum x = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds
        bounds = [(max(0.0, h[i] - radius), min(1.0, h[i] + radius)) for i in range(n)]
        bounds.append((-np.inf, np.inf))  # t unbounded
        
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        x_new = res.x[:n]
        t_new = res.x[n]
        
        # Check for violated constraints
        violated = False
        for r in range(len(raw)):
            if r in active:
                continue
            offset = r - (n - 1)
            grad = _grad_at_offset(h, offset)
            val = float(grad @ x_new) + (float(raw[r]) - float(grad @ h))
            if val > t_new + 1e-8:
                # Add this lag to active set
                active.append(r)
                violated = True
                if len(active) >= max_lags:
                    break
        
        if not violated:
            break
    
    return x_new, active
```

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_at_offset(h, offset):
    n = len(h)
    grad = np.zeros(n)
    if offset >= 0:
        m = n - offset
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[offset:offset+m])
        grad[offset:] -= h[:m]
    else:
        o = -offset
        m = n - o
        if m <= 0:
            return grad
        grad[o:] += (1.0 - h[:m])
        grad[:m] -= h[o:]
    return grad

    n = len(h)
    target = n / 2.0
    raw = _raw_corr(h)
    
    # Start with top lags
    top_k = min(5, len(raw))
    active = list(np.argsort(-raw)[:top_k])
    
    for _ in range(max_active):
        # Build LP
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for r in active:
            offset = r - (n - 1)
            grad = _grad_at_offset(h, offset)
            b_r = float(grad @ h) - float(raw[r])
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(b_r)
        
        if not A_ub_list:
            break
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        lb = np.maximum(0.0, h - radius)
        ub = np.minimum(1.0, h + radius)
        bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs', options={'disp': False})

            break
        
        x_new = res.x[:n].copy()
        t_new = res.x[n]

                continue
            offset = r - (n - 1)

            val = float(grad @ x_new) + (float(raw[r]) - float(grad @ h))
            if val > t_new + tol:
                active.append(r)
                violated = True
                if len(active) >= max_active:
                    break
        
        if not violated:
            break

    # Actually, let's just do direct evaluation for robustness with n <= 200
    
    n = len(h)

def _exact_c5(h):

def _raw_corr(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_at_offset(h, offset):
    n = len(h)
    grad = np.zeros(n)
    if offset >= 0:
        m = n - offset
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[offset:offset + m])
        grad[offset:] -= h[:m]
    else:
        o = -offset
        m = n - o
        if m <= 0:
            return grad
        grad[o:] += (1.0 - h[:m])
        grad[:m] -= h[o:]
    return grad

def _lp_solve(h, radius, top_k, max_active):
    n = len(h)
    target = n / 2.0
    raw = _raw_corr(h)
    active = list(np.argsort(-raw)[:top_k])
    x_lp = h.copy()
    for _ in range(max_active):
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub_rows = []
        b_ub_vals = []
        for r in active:
            offset = r - (n - 1)
            grad = _grad_at_offset(h, offset)
            b_r = float(grad @ h) - float(raw[r])
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(b_r)
        if not A_ub_rows:
            break
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        lb = np.maximum(0.0, h - radius)
        ub = np.minimum(1.0, h + radius)
        bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'disp': False})
        if not res.success:
            break
        x_lp = res.x[:n].copy()
        t_val = res.x[n]
        violated = False
        for r in range(len(raw)):
            if r in active:
                continue
            offset = r - (n - 1)
            grad = _grad_at_offset(h, offset)
            pred = float(grad @ x_lp) + (float(raw[r]) - float(grad @ h))
            if pred > t_val + 1e-7:
                active.append(r)
                violated = True
                if len(active) >= max_active:
                    break
        if not violated:
            break

        pos = d > 1e-12
        neg = d < -1e-12

def _to_binary(h):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 4.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _gradient_direction(h, raw, beta):
    n = len(h)
    shifted = raw - float(np.max(raw))
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wt = w[r]
        if wt < 1e-12:
            continue
        offset = r - (n - 1)
        if offset >= 0:
            m = n - offset
            if m > 0:
                g[:m] += wt * (1.0 - h[offset:offset + m])
                g[offset:] -= wt * h[:m]
        else:
            o = -offset
            m = n - o
            if m > 0:
                g[o:] += wt * (1.0 - h[:m])
                g[:m] -= wt * h[o:]
    g -= np.mean(g)
    return g

        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.007)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, init_radius=0.08, top_k=8, max_active=15, stall_limit=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline - 2, init_radius=0.04, top_k=15, max_active=35, stall_limit=50)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=2000, radius=0.1, seed=777)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
