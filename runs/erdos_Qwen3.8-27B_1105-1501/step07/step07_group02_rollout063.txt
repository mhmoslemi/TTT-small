def _compute_grad(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx] * dx
        if wk < 1e-14:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            # Term 1: j >= l, j-l < n → j in [l, n-1]
            j1 = np.arange(l, n)
            grad[j1] += wk * (1.0 - h[j1 - l])
            # Term 2: j <= n-1-l, j+l < n → j in [0, n-1-l]
            j2 = np.arange(0, n - l)
            grad[j2] -= wk * h[j2 + l]

            # Term 1: j <= n-1-p → j in [0, n-1-p]
            j1 = np.arange(0, n - p)
            grad[j1] += wk * (1.0 - h[j1 + p])
            # Term 2: j >= p → j in [p, n-1]
            j2 = np.arange(p, n)
            grad[j2] -= wk * h[j2 - p]

def _proj_grad_descent(h, deadline, beta=20.0, max_iter=500, alpha_init=0.5):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        R = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(R)
        w = np.exp(beta * (R - M))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= w_sum
        
        grad = _compute_grad(h, w, dx)
        
        # Projected gradient step
        # Try h_new = h - alpha * grad, then project
        # Use backtracking line search
        
        best_alpha = 0.0
        best_trial_c5 = best_c5
        for trial_alpha in [alpha, alpha*0.5, alpha*0.25, alpha*0.125, alpha*0.0625]:
            h_trial = h - trial_alpha * grad
            h_trial = _project(h_trial, target)
            c5_trial = _exact_c5(h_trial)
            if c5_trial < best_trial_c5:
                best_trial_c5 = c5_trial
                best_alpha = trial_alpha
            if c5_trial < best_c5 - 1e-15:
                break
        
        if best_alpha > 0:
            h_new = h - best_alpha * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                alpha = min(alpha * 1.1, 2.0)
                stall = 0
            else:
                h = h_new  # still move even if not improving (for exploration)
                alpha *= 0.7
                stall += 1
        else:
            alpha *= 0.5
            stall += 1
        
        if stall > 50:
            break
    
    return best_h, best_c5

        s = np.sum(w)
        if s < 1e-30:
            active = np.where(R > M - 1e-10)[0]
            w = np.zeros(len(R))
            w[active] = 1.0 / len(active)
        else:
            w /= s
        
        # Weighted gradient (of R values, without dx)
        g = np.zeros(n)
        for k_idx in range(len(R)):
            wk = w[k_idx]
            if wk < 1e-14:
                continue
            g += wk * _grad_lag(h, k_idx, 1.0)
        
        # Project to zero-mean subspace (preserves mass)
        d = g - np.mean(g)
        d = -d  # negative gradient direction
        
        if np.max(np.abs(d)) < 1e-13:
            break
        
        # Determine feasible t range (box constraints only)
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = 100.0
        t_lo = -100.0
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
        
        if t_hi - t_lo < 1e-14:
            break
        
        # Line search (mass is auto-preserved since sum(d) ≈ 0)
        t_opt, val = _line_search_max(h, d, t_lo, t_hi, dx, target)
        
        h_try = h + t_opt * d
        # Due to floating point, sum(h_try) might not be exactly target
        # But it should be very close since sum(d) ≈ 0
        c5_try = _exact_c5(h_try)
        
        # If box constraint slightly violated, project
        if np.any(h_try < -1e-10) or np.any(h_try > 1.0 + 1e-10):
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
        
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            stall = 0
        else:
            # Try half step
            t_half = 0.5 * t_opt
            h_half = h + t_half * d
            if np.any(h_half < -1e-10) or np.any(h_half > 1.0 + 1e-10):
                h_half = np.clip(h_half, 0.0, 1.0)
                h_half = _project(h_half, target)
            c5_half = _exact_c5(h_half)
            if c5_half < best_c5 - 1e-15:
                best_c5 = c5_half
                best_h = h_half.copy()
                h = h_half
            stall += 1
        
        if stall > 100:
            break
    
    return best_h, best_c5

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_lag(h, k_idx, dx):
    n = len(h)
    l = k_idx - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        j1 = np.arange(l, n)
        g[j1] += dx * (1.0 - h[j1 - l])
        if n - l > 0:
            j2 = np.arange(0, n - l)
            g[j2] -= dx * h[j2 + l]
    else:
        p = -l
        if n - p > 0:
            j1 = np.arange(0, n - p)
            g[j1] += dx * (1.0 - h[j1 + p])
        j2 = np.arange(p, n)
        g[j2] -= dx * h[j2 - p]
    return g

def _grad_weighted(h, R, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k_idx in range(len(R)):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            j1 = np.arange(l, n)
            g[j1] += wk * dx * (1.0 - h[j1 - l])
            if n - l > 0:
                j2 = np.arange(0, n - l)
                g[j2] -= wk * dx * h[j2 + l]
        else:
            p = -l
            if n - p > 0:
                j1 = np.arange(0, n - p)
                g[j1] += wk * dx * (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            g[j2] -= wk * dx * h[j2 - p]
    return g

def _line_search_max(R0, A, B, t_lo, t_hi):
    n_grid = 60
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-13:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    for _ in range(45):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    t_opt = (lo + hi) / 2.0
    opt_val = float(np.max(R0 + t_opt * A - t_opt * t_opt * B))
    gv = float(vals[best_idx])
    lv = float(np.max(R0 + t_lo * A - t_lo * t_lo * B))
    hv = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))
    best_val = min(opt_val, gv, lv, hv)
    if best_val == opt_val:
        return t_opt, best_val
    if best_val == gv:
        return float(ts[best_idx]), best_val
    if best_val == lv:
        return t_lo, best_val
    return t_hi, best_val

def _proj_grad_descent(h, deadline, beta=20.0, max_iter=500, stall_limit=120):

            w /= s

def _nlp_solve(h_init, deadline, K=25, max_slsqp_iter=80):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()

        if time.time() > deadline:
            break
        R = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(R)
        active = np.argsort(R, kind='stable')[-K:]
        active = np.sort(active)
        
        def obj(x):
            return x[n]
        
        def cons_fun(x):
            hx = np.clip(x[:n], 0, 1)
            t = x[n]
            Rf = np.correlate(hx, 1.0 - hx, mode='full')
            return t - dx * Rf[active]
        
        x0 = np.concatenate([h, [best_c5 + 0.03]])
        bounds = [(0, 1)] * n + [(0.0, 2.0)]
        con_list = []
        for k_idx in active:

        con_list.append({'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target})
        try:

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.12, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        s = float(np.sum(w))
        if s < 1e-30:
            active = np.where(S0 >= M - 1e-10)[0]
            w = np.zeros(len(S0))
            w[active] = 1.0 / len(active)
        else:
            w /= s
        G = np.zeros((2*n-1, n))
        for k in range(-(n-1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                G[row, k:] += (1.0 - h[:m])
                G[row, :m] -= h[k:]
            else:
                p = -k
                m = n - p
                G[row, p:] += (1.0 - h[:m])
                G[row, :m] -= h[p:]
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        npairs = min(n_pairs, n // 2)
        sources = sort_idx[-npairs:]
        sinks = sort_idx[:npairs]
        pairs = [(int(sinks[i]), int(sources[i])) for i in range(npairs)]
        for _ in range(min(10, npairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search_max(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 50 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=15000, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        i = ones[oi]
        j = zeros[oz]
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-14:
            ones[oi] = j
            zeros[oz] = i
            cur_c5 = new_c5
            stall = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
            ones[oi] = i
            zeros[oz] = j
            stall += 1
    return best_h, best_c5

        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=20000, seed=555)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish binary result
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _proj_grad_descent(h_pol, sub_dl2, beta=30.0, max_iter=100, stall_limit=50)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 6: Final polish (remaining time)
    n_curr = len(best_h)
    h_final = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = deadline - 1
    h_final, c5_final = _proj_grad_descent(h_final, sub_dl, beta=40.0, max_iter=200, stall_limit=60)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    
    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
