            for _ in range(kick_size):

                j = int(zeros_arr[oj])

        oj = int(rng.integers(len(zeros_arr)))
        i = int(ones_arr[oi])
        j = int(zeros_arr[oj])

        hb[i] = 0.0
        hb[j] = 1.0
        one_minus[i] = 1.0
        one_minus[j] = 0.0
        new_corr = np.correlate(hb, one_minus, mode='full')

    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _local_opt(h, deadline, r0=0.20, stall_lim=80, nd=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(8888)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = []
        for beta in [15.0, 30.0, 60.0]:
            g = _smooth_grad(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)
        x = np.arange(n, dtype=float)
        for _ in range(2):
            d = np.zeros(n)
            for _ in range(3):
                f = rng.uniform(1.0, 10.0)
                p = rng.uniform(0, 2 * np.pi)
                d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-10:
                dirs.append(d / nrm)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs[:nd]:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _cont_line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.35)
        else:
            stall += 1
            rad *= 0.75
            if rad < 0.008:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
    return best_h, best_c

def _pair_opt(h, deadline, max_it=50000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _cont_line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _gen_periodic_binary(n, period, ones_in_period, offset=0):
    base = np.zeros(period)
    idx = np.sort(rng_periodic.uniform(0, period, size=ones_in_period, dtype=int)) if False else np.linspace(0, period-1, ones_in_period, dtype=int)
    base[idx] = 1.0
    reps = n // period
    pat = np.tile(base, reps)
    if len(pat) < n:
        pat = np.concatenate([pat, base[:n - len(pat)]])
    if offset > 0:
        pat = np.roll(pat, offset)
    s = pat.sum()
    if s != n // 2:
        # adjust
        if s > n // 2:
            excess = int(s - n // 2)
            ones_pos = np.where(pat > 0.5)[0]
            pat[ones_pos[:excess]] = 0.0
        else:
            deficit = int(n // 2 - s)
            zeros_pos = np.where(pat <= 0.5)[0]
            pat[zeros_pos[:deficit]] = 1.0
    return pat

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Binary local search from initial (30%)
    d1 = t0 + budget_s * 0.30
    hb = _to_binary(h_init, n0)
    hb, cb = _binary_ls(hb, d1, seed=seed, kick_interval=4000, kick_size=3)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Binary SA from best (10%)
    d2 = t0 + budget_s * 0.40
    hb2 = _to_binary(best_h, n0)
    hb2, cb2 = _binary_sa(hb2, d2, seed=seed + 1, T0=0.002, cool=0.9997)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 3: Multi-start binary exploration (15%)
    d3 = t0 + budget_s * 0.55
    starts = []
    # Periodic patterns
    for period in [5, 6, 7, 10, 12, 15, 18, 20, 30]:
        ones_p = period // 2
        pat = np.zeros(period)
        if ones_p <= period // 2:
            for k in range(ones_p):
                pat[int(k * period / ones_p)] = 1.0
        else:
            for k in range(period - ones_p):
                pat[int(k * period / (period - ones_p))] = 0.0
            pat = 1.0 - pat
            pat[0] = 1.0 - pat[0]  # ensure diversity
        reps = n0 // period
        full = np.tile(pat, reps)
        if len(full) < n0:
            full = np.concatenate([full, pat[:n0 - len(full)]])
        # Adjust to exactly n0/2 ones
        s = int(full.sum())
        if s > n0 // 2:
            op = np.where(full > 0.5)[0]
            full[op[:s - n0 // 2]] = 0.0
        elif s < n0 // 2:
            zp = np.where(full <= 0.5)[0]
            full[zp[:n0 // 2 - s]] = 1.0
        starts.append(full)
    # Random balanced
    for _ in range(3):
        rand_b = np.zeros(n0)
        rand_b[rng.integers(0, n0, size=n0 // 2, replace=False)] = 1.0
        starts.append(rand_b)
    # Perturbed initial
    best_b = _to_binary(best_h, n0)
    for _ in range(3):
        pert = best_b.copy()
        for _ in range(10):
            oi = int(rng.integers(n0 // 2))
            ones_pos = np.where(pert > 0.5)[0]
            zeros_pos = np.where(pert <= 0.5)[0]
            i = ones_pos[oi % len(ones_pos)]
            j = zeros_pos[int(rng.integers(len(zeros_pos)))]
            pert[i], pert[j] = pert[j], pert[i]
        starts.append(pert)

    time_per_start = max(0.5, (d3 - time.time()) / len(starts))
    for si, st in enumerate(starts):
        if time.time() > d3:
            break
        st = _to_binary(st, n0)
        st, cs = _binary_ls(st, min(d3, time.time() + time_per_start), seed=seed + 100 + si, kick_interval=3000, kick_size=2)
        if cs < best_c:
            best_c = cs
            best_h = st.copy()

    # Phase 4: Continuous optimization (20%)
    d4 = t0 + budget_s * 0.75
    hc = _project(best_h.copy(), n0 / 2.0)
    for r_init in [0.20, 0.10, 0.05]:
        if time.time() > d4 - 2:
            break
        sd = min(d4, time.time() + budget_s * 0.05)
        hc, cc = _local_opt(hc, sd, r0=r_init, stall_lim=60, nd=5)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
        if time.time() < d4 - 3:
            hc = _project(best_h.copy(), n0 / 2.0)
            sd = min(d4, time.time() + budget_s * 0.03)
            hc, cc = _pair_opt(hc, sd, max_it=20000)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    # Phase 5: Multi-resolution (10%)
    d5 = t0 + budget_s * 0.85
    nc = len(best_h)
    for dn in [-10, -6, 4, 8, -14, 12, -20, 20]:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0:
            continue
        hr = _resample(best_h, n_try)
        hr_b = _to_binary(hr, n_try)
        sd = min(d5, time.time() + budget_s * 0.015)
        hr_b, cr_b = _binary_ls(hr_b, sd, seed=seed + 500, kick_interval=3000, kick_size=2)
        cr = _exact_c5(_project(hr_b.copy(), n_try / 2.0))
        if cr < best_c + 0.003 and time.time() < d5 - 2:
            sd2 = min(d5, time.time() + budget_s * 0.012)
            hr_c, cr_c = _local_opt(_project(hr_b.copy(), n_try / 2.0), sd2, r0=0.10, stall_lim=40, nd=4)
            if cr_c < cr:
                cr = cr_c
                hr = hr_c
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Final intensive (10%)
    d6 = deadline
    nc = len(best_h)
    # Binary refinement
    hb_final = _to_binary(best_h, nc)
    sd = min(d6, time.time() + budget_s * 0.03)
    hb_final, cbf = _binary_ls(hb_final, sd, seed=seed + 999, kick_interval=2000, kick_size=2)
    if cbf < best_c:
        best_c = cbf
        best_h = hb_final.copy()
        nc = len(best_h)
    # Continuous polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), nc / 2.0)
        hf, cf = _local_opt(hf, min(deadline - 1, time.time() + budget_s * 0.03), r0=0.04, stall_lim=30, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=10000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Basin hopping
    if deadline - time.time() > 5:
        scale = 0.03
        n_re = 0
        while time.time() < deadline - 3 and n_re < 30:
            n_re += 1
            nc = len(best_h)
            x = np.arange(nc, dtype=float)
            pert = np.zeros(nc)
            for _ in range(3):
                f = rng.uniform(0.5, 15.0)
                p = rng.uniform(0, 2 * np.pi)
                pert += rng.uniform(0.5, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
            pert += 0.5 * scale * rng.normal(0, 1, nc)
            mxp = np.max(np.abs(pert))
            if mxp > 1e-10:
                pert *= scale / mxp
            hp = _project(best_h + pert, nc / 2.0)
            sd = min(deadline - 3, time.time() + budget_s * 0.004)
            ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, nd=3)
            if ct < best_c:
                best_c = ct
                best_h = ht.copy()
                scale = max(0.003, scale * 0.7)
            else:
                scale = min(0.08, scale * 1.15)

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
