def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to the box [0,1]^n with sum = target using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:

        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_polish(h, deadline, beta=20.0, max_iter=200, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    radius = 0.2
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(raw.max())
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -80, 0))
        w = vp / (float(vp.sum()) + 1e-30)
        
        # Compute gradient: d/dh[i] of sum_k w[k] * L_k(h, 1-h)
        # L_k(h, 1-h) = sum_i h[i+k]*(1-h[i])
        # dL_k/dh[i] = -h[i+k]'s contribution from the (1-h) part
        #                 + (1-h[i+k])'s contribution from the h part
        # More precisely:
        # d/dh[j] L_k = sum over i where i+k=j of (1-h[i]) + sum over i where i=j of (-h[i+k])
        #              = (1-h[j-k]) [if j-k valid] - h[j+k] [if j+k valid]
        # But weighted by w[k]
        
        g = np.zeros(n)
        for k_offset in range(2 * n - 1):
            w_k = w[k_offset]
            if w_k < 1e-12:
                continue
            d = k_offset - (n - 1)
            if d >= 0:
                # L_d = sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
                # dL_d/dh[j] = (1-h[j-d]) for j=d,...,n-1 (from h[i+d] term)
                #             - h[j+d] for j=0,...,n-d-1 (from (1-h[i]) term)
                for j in range(d, n):
                    g[j] += w_k * (1.0 - h[j - d])
                for j in range(n - d):
                    g[j] -= w_k * h[j + d]
            else:
                m = -d
                # L_{-m} = sum_{i=m}^{n-1} h[i-m]*(1-h[i])
                # dL_{-m}/dh[j] = (1-h[j+m]) for j=0,...,n-m-1 (from h[i-m] term, i=j+m)
                #                - h[j-m] for j=m,...,n-1 (from (1-h[i]) term, i=j)
                for j in range(n - m):
                    g[j] += w_k * (1.0 - h[j + m])
                for j in range(m, n):
                    g[j] -= w_k * h[j - m]
        
        g = -g  # descent direction
        g -= np.mean(g)  # preserve sum constraint
        nm = np.linalg.norm(g)
        if nm < 1e-12:
            stall += 1
            if stall > 30:
                break
            continue
        g /= nm
        
        # Line search
        best_alpha = 0.0
        best_val = cur_c5
        for alpha in np.concatenate([np.linspace(0.001, radius, 20), [radius]]):
            h_try = _project(h + alpha * g, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002, seed=99):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    
    while time.time() < deadline and stall < 200:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
        # Check sum constraint
        if not np.isclose(h_new.sum(), n / 2.0):
            stall += 1
            continue
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 4.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
