def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_binary(h):

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h

    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    # Grid + golden section
    ng = 40
    gt = np.linspace(a_lo, a_hi, ng)

        if hi - lo < 1e-13:

        if f1 < f2:
            hi, c2, f2 = c2, c1, f1

            c2 = lo + gr * (hi - lo)
            f2 = f(c2)

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        gg = -tau
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _grad_smooth(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    w /= wsum
    active = w > 1e-6
    if not np.any(active):
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    g = np.zeros(n)
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[tau:tau + m]
        else:
            gg = -tau
            m = n - gg
            if m > 0:
                g[:m] += wr * (1.0 - h[gg:])
                g[gg:] -= wr * h[:m]
    g -= float(np.mean(g))
    return g

def _make_blocks(h, g, blk):
    n = len(h)
    if n % blk != 0:
        return None
    nblk = n // blk
    gb = g.reshape(nblk, blk).mean(axis=1)
    d = np.tile(gb, (1, blk)).reshape(n).copy()
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12:
        return d / dn
    return None

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for k in range(3):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 3.0
    pert += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse_balanced(n):
    t = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c += v & 1
            v >>= 1
        t[i] = c & 1
    cnt = int(np.sum(t))
    target = n // 2
    if cnt > target:
        ones_idx = np.where(t == 1)[0]
        rng_t = np.random.default_rng(1)
        rng_t.shuffle(ones_idx)
        t[ones_idx[cnt - target:cnt]] = 0
    elif cnt < target:
        zeros_idx = np.where(t == 0)[0]
        rng_t = np.random.default_rng(2)
        rng_t.shuffle(zeros_idx)
        t[zeros_idx[:target - cnt]] = 1
    return t

        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _continuous_opt(h_f, deadline, beta_lo=20.0, beta_hi=100.0, radius_init=0.05, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
