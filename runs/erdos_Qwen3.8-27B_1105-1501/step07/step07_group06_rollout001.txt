import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection to [0,1] with sum = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_vec(h):
    """Compute full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag_single(h, tau):
    """Gradient of corr at lag tau (tau = r - (n-1) in correlate indexing)."""
    n = len(h)
    g = np.zeros(n)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau+m]
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    return g  # Note: NOT centering here; centering done in direction generation

def _grad_active(h, raw, beta, K=20):

    n = len(h)
    mx = float(np.max(raw))
    # Use top-K weights more aggressively

            tau = r - (n-1)

def _line_search_exact(h, d, radius):
    """Exact line search over feasible alpha interval."""
    n = len(h)
    dx = 2.0 / n
    # Compute feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a*T - a*a*Q)) * dx
    
    # Golden section + grid

    gr = (np.sqrt(5.0) - 1.0) / 2.0

            c2 = lo + gr*(hi-lo)
            f2 = f(c2)
    opt_t = (lo+hi)/2.0

    # Check boundaries

def _momentum_opt(h, deadline, beta_init=8.0, beta_max=80.0, momentum_init=0.3, max_stall=60):

    h = h.copy()
    v = np.zeros(n)  # velocity for momentum
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = momentum_init
    radius = 0.15
    stall = 0
    beta = beta_init
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        raw = _corr_vec(h)
        g = _grad_active(h, raw, beta, K=15)
        
        # Nesterov step
        v = momentum * v - g
        d = v.copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            stall += 1
            continue
        d = d / dn
        
        alpha, val = _line_search_exact(h, d, radius)
        
        if val < cur_c5 - 1e-13 and alpha != 0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
                # Decay momentum slightly

                # Increase beta over time
                beta = min(beta * 1.02, beta_max)
            else:
                stall += 1
                radius = max(0.03, radius * 0.8)
        else:
            stall += 1
            radius = max(0.03, radius * 0.85)
            # Reset momentum
            v *= 0.5
    
    return best_h, best_c5

    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500 or len(ones) == 0 or len(zeros) == 0:
            break
        
        move_type = rng.random()
        if move_type < 0.4:
            # Single swap
            oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_c5 = _exact_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                T *= 0.9995
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
                stall += 1
        elif move_type < 0.7:
            # Block reversal: reverse a contiguous block
            L = int(rng.integers(2, min(8, n//4)))
            start = int(rng.integers(0, n - L))
            block = cur[start:start+L].copy()
            cur[start:start+L] = block[::-1]
            # Check mass preservation (reversal preserves it)
            new_c5 = _exact_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                T *= 0.9995
            else:
                cur[start:start+L] = block
                stall += 1
        else:
            # Block swap: swap two blocks of same size
            L = int(rng.integers(2, min(5, n//6)))
            i1 = int(rng.integers(0, n - L))
            i2 = int(rng.integers(0, n - L))
            if abs(i1 - i2) < L:
                continue
            b1 = cur[i1:i1+L].copy()
            b2 = cur[i2:i2+L].copy()
            if b1.sum() != b2.sum():
                continue  # Can't swap blocks with different mass
            cur[i1:i1+L] = b2
            cur[i2:i2+L] = b1
            new_c5 = _exact_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                T *= 0.9995
            else:
                cur[i1:i1+L] = b1
                cur[i2:i2+L] = b2
                stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _perturb(h, amp, rng):
    """Smooth perturbation for basin hopping."""
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2*np.pi)
        pert += np.sin(2*np.pi*freq*x/n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n/2.0)

def _directions_for_h(h, raw, rng, n_dirs=6):
    """Generate diverse search directions."""
    n = len(h)
    dirs = []
    
    # Direction 1: smooth-max gradient (top-15)
    g = _grad_active(h, raw, 10.0, K=15)
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d/dn)
    
    # Direction 2: single max gradient
    r_max = int(np.argmax(raw))
    gs = _grad_lag_single(h, r_max - (n-1))
    gsn = np.linalg.norm(gs)
    if gsn > 1e-12:
        d = gs / gsn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d/dn)
    
    # Direction 3: top-5 active set (equal weights)
    top5 = np.argsort(raw)[-5:]
    g5 = np.zeros(n)
    for r in top5:
        tau = r - (n-1)
        g5 += _grad_lag_single(h, tau)
    g5n = np.linalg.norm(g5)
    if g5n > 1e-12:
        d = g5 / g5n
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d/dn)
    
    # Direction 4: Random smooth
    x = np.arange(n, dtype=float)
    f1 = rng.uniform(1, 4)
    f2 = rng.uniform(4, 10)
    ph1 = rng.uniform(0, 2*np.pi)
    ph2 = rng.uniform(0, 2*np.pi)
    d = np.sin(2*np.pi*f1*x/n + ph1) + 0.5*np.sin(2*np.pi*f2*x/n + ph2)
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12:
        dirs.append(d/dn)
    
    # Direction 5: Block-constant
    blk = int(rng.integers(2, 8))
    if n % blk == 0:
        nb = n // blk
        vals = rng.normal(0, 1, nb)
        d = np.tile(vals, blk)
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d/dn)
    
    # Direction 6: Source-sink pair
    idx_sorted = np.argsort(g1 if 'g1' in dir() else raw)
    s = int(idx_sorted[0])
    k = int(idx_sorted[-1])
    d = np.zeros(n)
    d[s] = 1.0
    d[k] = -1.0
    d -= np.mean(d)
    dirs.append(d)
    
    return dirs[:n_dirs]
