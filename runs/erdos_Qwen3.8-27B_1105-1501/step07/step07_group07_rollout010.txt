import numpy as np
from scipy.optimize import minimize, linprog
import time

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _scores_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return _raw_scores(h) * dx

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

# Global state for LSE optimization
_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 0.01
_g_G = None  # Gradient matrix (2n-1, n)

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    # Zero lag
    G[mid] = 1.0 - 2.0 * h
    # Positive lags
    for d in range(1, n):
        m = n - d
        # Row mid + d: score = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        # G[mid+d, i] for i in [0, m-1]: from h[i+d] term -> (1-h[i])
        # G[mid+d, i+d] for i in [0, m-1]: from (1-h[i]) term -> -h[i+d]
        G[mid + d, :m] += (1.0 - h[:m])
        G[mid + d, d:d+m] += -h[:m]
        # Row mid - d: score = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
        # G[mid-d, i] for i in [d, n-1]: from h[i] term -> (1-h[i-d])
        # G[mid-d, i-d] for i in [d, n-1]: from (1-h[i-d]) term -> -h[i]
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:]
    return G

def _lse_obj(x):
    n = _g_n
    tau = _g_tau
    s = _scores_scaled(x)
    mval = float(np.max(s))
    u = (s - mval) / tau
    # Clip for numerical stability
    u = np.clip(u, -500.0, 500.0)
    z = np.exp(u)
    log_sum = np.log(np.sum(z))
    return mval + tau * log_sum

def _lse_grad(x):
    n = _g_n
    tau = _g_tau
    s = _scores_scaled(x)
    mval = float(np.max(s))
    u = (s - mval) / tau
    u = np.clip(u, -500.0, 500.0)
    z = np.exp(u)
    probs = z / np.sum(z)
    G = _build_G(x)
    # grad = G.T @ probs * dx
    return G.T @ probs * _g_dx

def _eq_con(x):
    return float(np.sum(x)) - _g_target

def _eq_con_jac(x):
    return np.ones(_g_n)

def _optimization_bounds(n):
    return [(0.0, 1.0)] * n

def _lse_anneal(h0, deadline, tau_start=0.04, tau_end=0.003, n_taus=8, maxiter_base=80):
    n = len(h0)
    global _g_n, _g_target, _g_dx, _g_tau, _g_G
    _g_n = n
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    n_taus = max(3, n_taus)
    taus = np.geomspace(tau_start, tau_end, n_taus)
    
    for tau in taus:
        if time.time() > deadline:
            break
        _g_tau = float(tau)
        
        # Reduce maxiter as tau gets smaller (converges faster)
        maxiter = max(20, int(maxiter_base * (tau_start / tau) ** 0.3))
        maxiter = min(maxiter, 200)
        
        bounds = _optimization_bounds(n)
        constraints = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac}]
        
        try:
            res = minimize(_lse_obj, h, jac=_lse_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
            if res.x is not None:
                h_new = _project_sum(np.clip(res.x, 0, 1), n / 2.0)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 + 0.001:  # Allow slight worsening for exploration
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

_g_target = 0.0
_g_dx = 0.0
_g_tau = 0.01

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _scores_scaled(h):
    return _raw_scores(h) * (2.0 / len(h))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0).copy()
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[mid + d, :m] += (1.0 - h[:m])
        G[mid + d, d:d+m] += -h[:m]
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:]
    return G

def _lse_obj(x):
    tau = _g_tau
    s = _scores_scaled(x)
    mval = float(np.max(s))
    u = np.clip((s - mval) / tau, -500.0, 500.0)
    z = np.exp(u)
    return mval + tau * float(np.log(np.sum(z)))

def _lse_grad(x):
    tau = _g_tau
    s = _scores_scaled(x)
    mval = float(np.max(s))
    u = np.clip((s - mval) / tau, -500.0, 500.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum <= 0:
        probs = np.full(2*_g_n-1, 1.0/(2*_g_n-1))
    else:
        probs = z / zsum
    G = _build_G(x)
    return G.T @ (probs * _g_dx)

def _eq_fun(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _get_bounds(n):
    return [(0.0, 1.0)] * n

def _check_gradient(h, tol=1e-5):
    n = len(h)
    global _g_n, _g_target, _g_dx, _g_tau
    _g_n = n
    _g_target = n / 2.0

    grad_analytic = _lse_grad(h)

        h_m = h.copy()
