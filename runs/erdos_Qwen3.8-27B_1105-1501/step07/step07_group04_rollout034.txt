def _flip_delta(h, corr, p, n):
    """Compute the change in each correlation when flipping position p.
    If h[p]=1, delta=-1 (flipping to 0). If h[p]=0, delta=+1.
    Returns array of length 2n-1 with the changes."""
    delta = 1.0 if h[p] == 0 else -1.0
    dcorr = np.zeros(2 * n - 1)
    center = n - 1
    
    # For positive lags k=1..n-1: F_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    for k in range(1, n):
        # p appears as h[i+k] when i = p-k, valid if 0 <= p-k and p-k <= n-1-k
        if k <= p <= n - 1:
            dcorr[center + k] += delta * (1.0 - h[p - k])
        # p appears in (1-h[i]) when i = p, valid if 0 <= p <= n-1-k
        if 0 <= p <= n - 1 - k:
            dcorr[center + k] -= delta * h[p + k]
        
        # For negative lags: F_{-k} = sum_{i=k}^{n-1} h[i-k]*(1-h[i])
        # p appears as h[i-k] when i = p+k, valid if k <= p+k <= n-1
        if p + k <= n - 1:
            dcorr[center - k] += delta * (1.0 - h[p + k])
        # p appears in (1-h[i]) when i = p, valid if k <= p <= n-1
        if k <= p <= n - 1:
            dcorr[center - k] -= delta * h[p - k]
    
    return dcorr

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_deltas(h, n):
    d = np.where(h == 0, 1.0, -1.0)  # delta for each position
    center = n - 1
    deltas = np.zeros((n, 2 * n - 1))
    
    for k in range(1, n):
        # Positive lag k
        row_pos = np.zeros(n)
        row_pos[k:] = d[k:] * (1.0 - h[:n-k])
        row_pos[:n-k] += -d[:n-k] * h[k:]
        deltas[:, center + k] = row_pos
        
        # Negative lag k
        row_neg = np.zeros(n)
        row_neg[:n-k] = d[:n-k] * (1.0 - h[k:])
        row_neg[k:] += -d[k:] * h[:n-k]
        deltas[:, center - k] = row_neg
    
    return deltas
