```python
import numpy as np
from scipy.optimize import minimize
import time

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for m in range(1, n):
        # Positive lag: row mid + m
        G[mid + m, m:] += (1.0 - h[:n - m])
        G[mid + m, :n - m] += -h[m:]
        # Negative lag: row mid - m
        G[mid - m, :n - m] += (1.0 - h[m:])
        G[mid - m, m:] += -h[:n - m]
    return G

# Global state for SLSQP
_g_n = 0
_g_target = 0.0
_g_dx = 0.0
_g_tau = 1.0
_g_probs = None

def _smooth_obj(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    x = np.clip(x, 0.0, 1.0)  # soft clip for numerical safety
    raw = _raw_scores(x)
    s = raw * dx
    mval = float(np.max(s))
    u = (s - mval) / tau
    # Safe exp
    u_max = float(np.max(u))
    if u_max > 500:
        u = u - u_max
        z = np.exp(u)
        zsum = float(np.sum(z))
        lse = mval + tau * (float(np.log(zsum)) + u_max)
    else:
        z = np.exp(u)
        zsum = float(np.sum(z))
        lse = mval + tau * float(np.log(zsum))
    # Store probs for gradient
    _g_probs = z / zsum
    return lse

def _smooth_grad(x):
    n = _g_n
    dx = _g_dx
    x = np.clip(x, 0.0, 1.0)
    G = _build_G(x)
    return dx * (G.T @ _g_probs)

def _eq_constraint(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _run_smooth_anneal(h0, deadline, tau_start_ratio=0.15, n_taus=6, maxiter=80):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    global _g_n, _g_target, _g_dx, _g_tau, _g_probs
    
    _g_n = n
    _g_target = target
    _g_dx = dx
    
    h = _project_sum(h0.copy(), target)
    cur_c5 = _exact_c5(h)
    
    raw0 = _raw_scores(h)
    score_range = float(np.max(raw0) - np.min(raw0)) * dx
    tau_start = max(0.01, tau_start_ratio * max(0.05, score_range))
    
    taus = [tau_start * (0.6 ** i) for i in range(n_taus)]
    tau_floor = 0.001
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac}]
    
    for tau in taus:
        if time.time() > deadline - 5:
            break
        if tau < tau_floor:
            break
        _g_tau = tau
        try:
            res = minimize(_smooth_obj, h, method='SLSQP', jac=_smooth_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-8, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project_sum(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5:
                    h = h_new
                    cur_c5 = c5_new
                else:
                    # Still use res.x as warm start for next tau (it might be useful)
                    if c5_new < cur_c5 + 0.01:
                        h = _project_sum(res.x, target)
                        cur_c5 = c5_new
        except Exception:
            pass
    
    return h, cur_c5

def _line_search(h, d, dx, max_step=1.0):
    """Exact line search along direction d from h."""
    n = len(h)
    # Determine max feasible step
    t_hi = max_step
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi <= 0:
        return 0.0, _exact_c5(h)
    
    # Golden section search on [0, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project_sum(h + c1 * d, n / 2.0))
    f2 = _exact_c5(_project_sum(h + c2 * d, n / 2.0))
    
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project_sum(h + c1 * d, n / 2.0))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project_sum(h + c2 * d, n / 2.0))
    
    t_opt = (lo + hi) / 2.0
    f_opt = _exact_c5(_project_sum(h + t_opt * d, n / 2.0))
    
    # Check endpoints
    f0 = _exact_c5(h)
    f1_end = _exact_c5(_project_sum(h + t_hi * d, n / 2.0))
    
    best_t, best_f = 0.0, f0
    if f_opt < best_f:
        best_t, best_f = t_opt, f_opt
    if f1_end < best_f:
        best_t, best_f = t_hi, f1_end
    
    return best_t, best_f

def _pair_polish(h, deadline, n_trials=5000, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, 2.0/n, max_step=1.0)
        if c5_try < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * d, target)
            cur_c5 = _exact_c5(h)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, n_iter=30000, T0=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T0
    stall = 0
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if cur[i] == cur[j]:
            continue
        # Swap: make i=1, j=0 if i was 0, or vice versa
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            # Update sets
            if cur[i] == 1.0:
                ones.add(i)
                zeros.discard(i)
                zeros.add(j)
                ones.discard(j)
            else:
                zeros.add(i)
                ones.discard(i)
                ones.add(j)
                zeros.discard(j)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += amplitude * 0.1 * rng.standard_normal(n)
    return _project_sum(h + amplitude * pert, n / 2.0)
```

```python
def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for m in range(1, n):
        G[mid + m, m:] += (1.0 - h[:n - m])
        G[mid + m, :n - m] += -h[m:]
        G[mid - m, :n - m] += (1.0 - h[m:])
        G[mid - m, m:] += -h[:n - m]
    return G

    gr = (np.sqrt(5.0) - 1.0) / 2.0

            stall += 1
    
    return best_h, best_c5

def _grad_pair_polish(h, deadline, n_trials=3000, seed=77):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        scores = _raw_scores(h)
        top_k = np.argsort(scores)[-3:]
        G = _build_G(h)
        best_dir = None
        best_gain = 0.0
        for _ in range(20):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            gain = float(np.sum(scores[top_k] * (G[top_k, i] - G[top_k, j])))
            if gain < -best_gain:
                best_gain = -gain
                best_dir = d
        if best_dir is None:
            stall += 1
            continue
        alpha, c5_try = _line_search(h, best_dir, max_step=1.0)
        if c5_try < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * best_dir, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_b = np.zeros(n)
    h_b[idx] = 1.0
    return h_b

def _binary_sa(h_bin, deadline, n_iter=30000, T0=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T0
    stall = 0
    ones_list = np.where(cur > 0.5)[0].tolist()
    zeros_list = np.where(cur <= 0.5)[0].tolist()
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        if not ones_list or not zeros_list:
            break
        i = ones_list[int(rng.integers(0, len(ones_list)))]
        j = zeros_list[int(rng.integers(0, len(zeros_list)))]
        cur[i] = 0.0
        cur[j] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += amplitude * 0.15 * rng.standard_normal(n)
    return _project_sum(h + amplitude * pert, n / 2.0)

def _balance_scores(h, deadline, n_iter=2000, seed=555):
    """Targeted search that tries to equalize the top scores."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        scores = _raw_scores(h)
        s_max = float(np.max(scores))
        # Find the lag with max score
        lag_idx = int(np.argmax(scores))
        m = lag_idx - (n - 1)
        # Try to reduce this lag: find i where h[i+m]*(1-h[i]) is large
        # and swap h[i] with some h[j] to reduce it
        if m >= 0:
            pairs = [(i, i + m) for i in range(n - m) if h[i + m] > 0.3 and h[i] < 0.7]
        else:
            g = -m
            pairs = [(i - g, i) for i in range(g, n) if h[i - g] > 0.3 and h[i] < 0.7]
        
        if not pairs:
            stall += 1
            continue
        
        # Try a random pair from this lag
        pi = rng.integers(0, len(pairs))
        i, j = pairs[pi]
        # Try swapping i with a random position
        k = int(rng.integers(0, n))
        if k == i or k == j:
            k = int(rng.integers(0, n))
            if k == i or k == j:
                stall += 1
                continue
        d = np.zeros(n)
        d[i] = -1.0
        d[k] = 1.0
        alpha, c5_try = _line_search(h, d, max_step=1.0)
        if c5_try < cur_c5 - 1e-13:
            h = _project_sum(h + alpha * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
    
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 -= 1
        h0 = h0[:n0]
    
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Smooth annealing from multiple seeds (35%)
    d1 = t0 + budget_s * 0.35
    
    seeds = [h_init.copy()]
    seeds.append(_project_sum(h_init[::-1].copy(), n0 / 2.0))
    seeds.append(_project_sum(1.0 - h_init, n0 / 2.0))
    seeds.append(_project_sum(1.0 - h_init[::-1], n0 / 2.0))
    # Add small perturbations
    for _ in range(2):
        seeds.append(_perturb(h_init, rng, 0.05))
    
    for si, s in enumerate(seeds):
        if time.time() > d1 - 8:
            break
        sub_dl = min(d1 - 3, time.time() + budget_s * 0.05)
        h_s, c5_s = _run_smooth(s, sub_dl, tau_ratio=0.15, n_taus=5, maxiter=50)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 2: Gradient-directed + random pair polish (10%)
    d2 = t0 + budget_s * 0.45
    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_p, c5_p = _grad_pair_polish(best_h, sub_dl, n_trials=2000, seed=111)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    
    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_p, c5_p = _pair_polish(best_h, sub_dl, n_trials=4000, seed=222)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    
    # Phase 3: Binary optimization (10%)
    d3 = t0 + budget_s * 0.55
    if time.time() < d3:
        h_b = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, n_iter=35000, T0=0.003, seed=333)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Smooth polish on binary result
        if time.time() < d3 - 3:
            sub_dl = min(d3 - 2, time.time() + budget_s * 0.03)
            h_r, c5_r = _run_smooth(h_b, sub_dl, tau_ratio=0.1, n_taus=4, maxiter=40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Phase 4: Dimension exploration (20%)
    d4 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    n_list = []
    for dn in range(-28, 29, 2):
        nt = n_cur + dn
        if 80 <= nt <= 250 and nt % 2 == 0:
            n_list.append(nt)
    rng.shuffle(n_list)
    
    for nt in n_list[:14]:
        if time.time() > d4 - 8:
            break
        h_r = _resample(best_h, nt)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.012)
        h_r, c5_r = _run_smooth(h_r, sub_dl, tau_ratio=0.12, n_taus=4, maxiter=35)
        if c5_r < best_c5 + 0.003 and time.time() < d4 - 3:
            sub_dl2 = min(d4 - 2, time.time() + budget_s * 0.006)
            h_r, c5_r = _pair_polish(h_r, sub_dl2, n_trials=1500, seed=nt)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Intensive refinement (20%)
    d5 = t0 + budget_s * 0.95
    n_cur = len(best_h)
    
    # Smooth with tighter tau
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.06)
        h_r, c5_r = _run_smooth(best_h, sub_dl, tau_ratio=0.08, n_taus=5, maxiter=50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Balance scores
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_r, c5_r = _balance_scores(best_h, sub_dl, n_iter=2000, seed=666)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Pair polish
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.06)
        h_r, c5_r = _pair_polish(best_h, sub_dl, n_trials=6000, seed=444)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Grad pair polish
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_r, c5_r = _grad_pair_polish(best_h, sub_dl, n_trials=2000, seed=555)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Binary SA again
    if time.time() < d5:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, sub_dl, n_iter=20000, T0=0.002, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Basin hopping
    if time.time() < d5:
        scale = 0.04
        for _ in range(80):
            if time.time() > d5 - 5:
                break
            h_pert = _perturb(best_h, rng, scale)
            sub_dl = min(d5 - 3, time.time() + budget_s * 0.005)
            h_t, c5_t = _run_smooth(h_pert, sub_dl, tau_ratio=0.1, n_taus=3, maxiter=30)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.01, scale * 0.85)
            else:
                scale = min(0.12, scale * 1.04)
    
    # Final polish
    if deadline - time.time() > 5:
        h_f = _project_sum(best_h, len(best_h) / 2.0)
        h_f, c5_f = _pair_polish(h_f, deadline, n_trials=2000, seed=888)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Final validation
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
```
