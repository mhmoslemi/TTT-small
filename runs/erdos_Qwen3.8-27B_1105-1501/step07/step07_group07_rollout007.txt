def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    for d in range(-(n - 1), n):
        r = mid + d
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if hi1 > lo1:
            G[r, lo1:hi1] = 1.0 - h[lo1 + d:hi1 + d]
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if hi2 > lo2:
            G[r, lo2:hi2] -= h[lo2 - d:hi2 - d]
    return G

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj(x):
    x = np.clip(x, 0.0, 1.0)
    scores = _raw_scores(x) * _G_dx
    tau = _G_tau
    m = float(np.max(scores))
    u = np.clip((scores - m) / tau, -500.0, 0.0)
    z = np.exp(u)
    return m + tau * float(np.log(float(np.sum(z))))

def _lse_grad(x):
    x = np.clip(x, 0.0, 1.0)
    scores = _raw_scores(x) * _G_dx
    tau = _G_tau
    m = float(np.max(scores))
    u = np.clip((scores - m) / tau, -500.0, 0.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        p = np.full(len(scores), 1.0 / len(scores))
    else:
        p = z / zsum
    G = _build_G(x)
    return G.T @ p * _G_dx

def _eq_con(x):
    return np.array([float(np.sum(x)) - _G_target])

def _eq_con_jac(x):
    return np.ones((1, _G_n))

def _smooth_anneal(h, deadline, n_steps=8, tau_start=None, tau_end=None, maxiter_base=60):
    global _G_n, _G_dx, _G_tau, _G_target
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    scores = _raw_scores(h) * dx
    spread = max(0.01, float(np.max(scores) - float(np.min(scores))))
    if tau_start is None:
        tau_start = max(0.06, spread * 0.6)
    if tau_end is None:
        tau_end = max(0.004, tau_start * 0.04)

    _G_n = n
    _G_dx = dx
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac}]

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        _G_tau = tau
        maxiter = max(15, int(maxiter_base * (1.0 + 0.5 * (1.0 - frac))))
        try:
            res = minimize(_lse_obj, h, method='SLSQP', jac=_lse_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                hc = _project_sum(res.x, target)
                c5c = _exact_c5(hc)
                if c5c < best_c5 - 1e-12:
                    best_c5 = c5c
                    best_h = hc.copy()
                    h = hc.copy()
        except Exception:
            pass

    return best_h, best_c5

def _pair_line_search(h, i, j):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n = len(h)
    dx = 2.0 / n
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    n_grid = 20
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_val = _exact_c5(h)
    for idx in range(n_grid):
        t = grid_t[idx]
        h_try = h.copy()
        h_try[i] += t
        h_try[j] -= t
        val = _exact_c5(h_try)
        if val < best_val:
            best_val = val
            best_t = t
    lo_b = grid_t[max(0, int(np.argmin([_exact_c5(h + grid_t[k]*d) for k in range(n_grid)])) - 2)] if n_grid > 3 else grid_t[0]
    hi_b = grid_t[min(n_grid - 1, int(np.argmax([1.0])) + 2)] if n_grid > 3 else grid_t[-1]
    for k in range(n_grid):
        h_try = h.copy()
        h_try[i] += grid_t[k]
        h_try[j] -= grid_t[k]
        if _exact_c5(h_try) <= best_val + 1e-12:
            lo_b = min(lo_b, grid_t[k])
            hi_b = max(hi_b, grid_t[k])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    h1 = h.copy(); h1[i] += c1; h1[j] -= c1
    h2 = h.copy(); h2[i] += c2; h2[j] -= c2
    f1 = _exact_c5(h1)
    f2 = _exact_c5(h2)
    for _ in range(30):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b = c2; c2 = c1; f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            h1 = h.copy(); h1[i] += c1; h1[j] -= c1
            f1 = _exact_c5(h1)
        else:
            lo_b = c1; c1 = c2; f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            h2 = h.copy(); h2[i] += c2; h2[j] -= c2
            f2 = _exact_c5(h2)
    opt_t = (lo_b + hi_b) / 2.0
    h_opt = h.copy(); h_opt[i] += opt_t; h_opt[j] -= opt_t
    opt_val = _exact_c5(h_opt)
    if opt_val < best_val:
        best_val = opt_val
        best_t = opt_t
    return best_t, best_val

def _pair_polish(h, deadline, max_pairs=80000, seed=42):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    sorted_idx = np.argsort(h)
    high_idx = sorted_idx[n // 2:]
    low_idx = sorted_idx[:n // 2]
    for _ in range(max_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, len(high_idx)))
        j = int(rng.integers(0, len(low_idx)))
        oi = high_idx[i]
        oj = low_idx[j]
        t, new_c5 = _pair_line_search(h, int(oi), int(oj))
        if new_c5 < cur_c5 - 1e-13 and abs(t) > 1e-12:
            h[int(oi)] += t
            h[int(oj)] -= t
            h = np.clip(h, 0.0, 1.0)
            h = _project_sum(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                sorted_idx = np.argsort(h)
                high_idx = sorted_idx[n // 2:]
                low_idx = sorted_idx[:n // 2]
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, scale):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0.0, 2.0 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert += scale * 0.2 * rng.normal(0, 1, n)
    pert *= scale
    return _project_sum(h + pert, n / 2.0)

def _coordinate_polish(h, deadline, max_rounds=50):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            t_lo = max(-h[j], 0.0)
            t_hi = min(1.0 - h[j], 0.0)
            if t_hi - t_lo < 1e-14:
                continue
            grid = np.linspace(t_lo, t_hi, 12)
            best_t = 0.0
            best_val = cur_c5
            for t in grid:
                if abs(t) < 1e-14:
                    continue
                h_try = h.copy()
                h_try[j] += t
                h_try = _project_sum(h_try, n / 2.0)
                val = _exact_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_t = t
            if best_val < cur_c5 - 1e-13 and abs(best_t) > 1e-12:
                h[j] += best_t
                h = _project_sum(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 + 1)
        n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE annealing from initial (30% budget)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _smooth_anneal(h_init, d1, n_steps=10, tau_start=0.08, tau_end=0.003, maxiter_base=50)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair polish (10%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _pair_polish(best_h, d2, max_pairs=40000, seed=77)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More LSE with different tau schedule (15%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _smooth_anneal(best_h, d3, n_steps=12, tau_start=0.04, tau_end=0.002, maxiter_base=40)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.75
    scale = 0.04
    n_restarts = 0
    while time.time() < d4 and n_restarts < 200:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d4, time.time() + budget_s * 0.006)
        h_t, c5_t = _smooth_anneal(h_pert, sub_dl, n_steps=5, tau_start=0.05, tau_end=0.005, maxiter_base=30)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.9)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 5: Pair polish again (10%)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _pair_polish(best_h, d5, max_pairs=60000, seed=1337)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Dimension exploration (8%)
    d6 = t0 + budget_s * 0.93
    n_cur = len(best_h)
    for dn in [-8, -4, 4, 8, -16, 16, -24, 24]:
        if time.time() > d6 - 3:
            break
        n_try = n_cur + dn
        if n_try < 60 or n_try > 300 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.008)
        h_r, c5_r = _smooth_anneal(h_r, sub_dl, n_steps=5, tau_start=0.06, tau_end=0.004, maxiter_base=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Final polish (7%)
    d7 = deadline
    h7, c5_7 = _smooth_anneal(best_h, d7, n_steps=6, tau_start=0.02, tau_end=0.001, maxiter_base=40)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    if deadline - time.time() > 10:
        h7 = _pair_polish(best_h, deadline, max_pairs=20000, seed=999)
        if h7[1] < best_c5:
            best_c5 = h7[1]
            best_h = h7[0].copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
