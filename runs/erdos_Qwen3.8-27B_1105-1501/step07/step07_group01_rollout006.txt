def _line_search_quadratic(h, p, S0, radius):
    """Minimize max_d f_d(t) along h + t*p for t in [0, radius]."""
    n = len(h)
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    C = np.correlate(p, p, mode='full')
    # f_d(t) = S0[d] + t*T[d] - t^2*C[d]
    # We want to minimize max_d (S0[d] + t*T[d] - t^2*C[d])
    # Grid search + refinement
    n_grid = 64
    t_vals = np.linspace(0, radius, n_grid)
    scores = np.max(S0[np.newaxis, :] + t_vals[:, np.newaxis] * T[np.newaxis, :] 
                   - (t_vals[:, np.newaxis])**2 * C[np.newaxis, :], axis=1)
    best_idx = int(np.argmin(scores))
    best_t = float(t_vals[best_idx])
    best_score = float(scores[best_idx])
    # Refine around best
    if best_idx > 0 and best_idx < n_grid - 1:
        lo = float(t_vals[best_idx - 1])
        hi = float(t_vals[best_idx + 1])
    elif best_idx == 0:
        lo, hi = 0.0, float(t_vals[1])
    else:
        lo, hi = float(t_vals[-2]), radius
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
        f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
        if f1 < f2:
            hi = c2
        else:
            lo = c1
    best_t = (lo + hi) / 2.0
    best_score = float(np.max(S0 + best_t * T - best_t*best_t * C))
    return best_t, best_score

                    f2 = float(np.max(S0 + c2 * g))

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5
