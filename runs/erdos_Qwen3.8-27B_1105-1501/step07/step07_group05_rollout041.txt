def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _delta_c5(h, i, j):

    n = len(h)

    # Term 1: -(1-h[i-k]) for 0<=i-k<n

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(48, max(24, int(width * 120)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

    n = len(h)
    dirs = []
    for beta in [25.0, 50.0]:
        g = _smooth_grad(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    g25 = _smooth_grad(h, 25.0)
    for bs in [8, 16, 32]:
        if len(dirs) >= nd:
            break
        d = np.zeros(n)
        nb = n // bs
        if nb >= 1:
            for b in range(nb):
                d[b * bs:(b + 1) * bs] = np.mean(g25[b * bs:(b + 1) * bs])
            if nb * bs < n:
                d[nb * bs:] = np.mean(g25[nb * bs:])
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    if len(dirs) < nd:
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)

            dirs.append(d / nrm)
    return dirs[:nd]

    # Also try 2-swap on the binary
    if time.time() < d2:
        h2s, c2s = _two_swap_greedy(_to_bin(best_h), d2)
        if c2s < best_c:
            best_c = c2s
            best_h = h2s.copy()

    # Phase 3: Continuous refinement (20%)
    d3 = t0 + budget_s * 0.48
    hc = _project(best_h.copy(), n0 / 2.0)
    hc, cc = _local_opt(hc, d3, r0=0.10, stall_lim=50, nd=5)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()
    if time.time() < d3:
        hc2 = _project(best_h.copy(), n0 / 2.0)
        hc2, cc2 = _pair_opt(hc2, d3, max_it=20000)
        if cc2 < best_c:
            best_c = cc2
            best_h = hc2.copy()

    # Phase 4: Multi-resolution exploration (22%)
    d4 = t0 + budget_s * 0.70
    nc_cur = len(best_h)
    offsets = [-10, -5, 5, 10, -20, 20, -15, 15, -25, 25, -40, 40]
    tried = set()
    for dn in offsets:
        if time.time() > d4 - 5:
            break
        n_try = nc_cur + dn
        if n_try < 50 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, nd=4)
        if cr < best_c + 0.003 and time.time() < d4 - 5:
            hb2 = _to_bin(hr)
            sd2 = min(d4, time.time() + budget_s * 0.012)
            hb2, cb2 = _binary_greedy(hb2, sd2, max_restarts=100)
            if cb2 < cr:
                cr = cb2
                hr = hb2
        if cr < best_c + 0.002 and time.time() < d4 - 3:
            sd3 = min(d4, time.time() + budget_s * 0.01)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc_cur = n_try

    # Phase 5: Intensive continuous + pair (15%)
    d5 = t0 + budget_s * 0.85
    nc = len(best_h)
    for r_init in [0.08, 0.04, 0.02]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.04)
        h5, c5 = _local_opt(h5, sd, r0=r_init, stall_lim=50, nd=5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d5 - 5:
            h5b = _project(best_h.copy(), nc / 2.0)
            sd2 = min(d5, time.time() + budget_s * 0.025)
            h5b, c5b = _pair_opt(h5b, sd2, max_it=12000)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b.copy()

    # Phase 6: Basin hopping + binary SA (12%)
    d6 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.4 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=20, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Final: binary greedy on best + continuous polish
    if deadline - time.time() > 5:
        hb_final = _to_bin(best_h)
        d_final = min(deadline, time.time() + budget_s * 0.04)
        hb_final, cbf = _binary_greedy(hb_final, d_final, max_restarts=200)
        if cbf < best_c:
            best_c = cbf
            best_h = hb_final.copy()
        hc_final = _project(best_h.copy(), len(best_h) / 2.0)
        d_polish = min(deadline, time.time() + budget_s * 0.03)
        hc_final, cpf = _local_opt(hc_final, d_polish, r0=0.02, stall_lim=30, nd=4)
        if cpf < best_c:
            best_c = cpf
            best_h = hc_final.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
