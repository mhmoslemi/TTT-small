def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _line_search_exact(h, d, dx, radius):

def _logsumexp_gradient(h, beta, top_k=20):
    """Compute gradient of log-sum-exp of top-k scores, projected to sum=0."""
    n = len(h)
    s = _scores(h)  # weighted scores (include dx)
    # Use unweighted for gradient computation
    s_unw = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    s_max = s_unw.max()
    exp_s = np.exp(beta * (s_unw - s_max))
    w = exp_s / exp_s.sum()
    lags = np.arange(len(s_unw)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= g.mean()
    return g

def _barrier_descent(h, deadline, beta_start=10.0, beta_end=200.0, max_iters=5000):
    """Log-sum-exp based descent with increasing beta."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n_iter = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        n_iter += 1
        # Animate beta
        frac = min(1.0, it / max(100, max_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Compute gradient
        g = _logsumexp_gradient(h, beta)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-12:
            break
        
        # Direction (normalize to reasonable step)
        radius = 0.1 * (1.0 + 0.5 * np.cos(np.pi * frac))
        d = g / g_norm * radius
        
        # Project direction to sum = 0
        d -= d.mean()
        
        # Clip to box constraints
        pos = d > 0
        neg = d < 0
        if np.any(pos):
            max_pos = float(np.min((1.0 - h[pos]) / d[pos]))
            d_scale = min(1.0, max_pos)
        else:
            d_scale = 1.0
        if np.any(neg):
            max_neg = float(np.min(h[neg] / (-d[neg])))
            d_scale = min(d_scale, max_neg)
        d *= d_scale
        d -= d.mean()
        
        # Line search
        alpha, new_c5 = _line_search_exact(h, d, dx, 1.0)
        
        if new_c5 < cur_c5 - 1e-12:
            h_new = h + alpha * d
            h = _project(h_new, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

def _active_lp(h, scores_unw, radius, n_active=8, gap=0.003):
    """Active-set LP for minimizing max of selected scores."""
    n = len(h)
    mx = float(np.max(scores_unw))
    active = np.where(scores_unw >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores_unw))])
    if len(active) > n_active:
        order = np.argsort(scores_unw[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores_unw[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _pair_sweep(h, deadline, max_rounds=150):
    """Pair-based local search: swap a 0-ish and a 1-ish position."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    stall = 0
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        improved = False
        ones = np.where(h > 0.4)[0]
        zeros = np.where(h < 0.6)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = min(len(ones) // 2 + 1, 20)
        for _ in range(n_tries):
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[-m:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _qr_sequence(n):
    """Quadratic residue inspired binary sequence of length n (even)."""
    # Find a prime p ≈ n
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(np.sqrt(x)) + 1):
            if x % i == 0:
                return False
        return True
    
    p = n + 1
    while not is_prime(p) or p % 4 != 3:
        p += 2
    # QR mod p
    qr = set()
    for i in range(1, p):
        qr.add((i * i) % p)
    # Create binary seq of length p, then truncate/extend to n
    b_p = np.zeros(p)
    for q in qr:
        if q < p:
            b_p[q] = 1.0
    # Truncate to n
    b = b_p[:n]
    # Adjust to have exactly n/2 ones
    m = n // 2
    cnt = int(b.sum())
    if cnt > m:
        # Remove excess ones
        one_idx = np.where(b == 1)[0]
        rng = np.random.default_rng(int(np.sum(np.arange(n) * b)))
        for idx in rng.permutation(one_idx)[:cnt - m]:
            b[idx] = 0.0
    elif cnt < m:
        zero_idx = np.where(b == 0)[0]
        rng = np.random.default_rng(int(np.sum(np.arange(n) * (1 - b))))
        for idx in rng.permutation(zero_idx)[:m - cnt]:
            b[idx] = 1.0
    return b

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 5
    perturb = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
