import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _compute_rem_add(h, ones_arr, zeros_arr):
    n = len(h)
    rem = np.zeros((len(ones_arr), 2 * n - 1))
    add = np.zeros((len(zeros_arr), 2 * n - 1))
    one_minus_h = 1.0 - h
    for ii in range(len(ones_arr)):
        i = ones_arr[ii]
        rem[ii, i:i + n] += one_minus_h[::-1]
        rem[ii, n - 1 - i:2 * n - 1 - i] += -h
    for jj in range(len(zeros_arr)):
        j = zeros_arr[jj]
        add[jj, j:j + n] += one_minus_h[::-1]
        add[jj, n - 1 - j:2 * n - 1 - j] += -h
    return rem, add

def _binary_steepest(h_bin, deadline, max_passes=300):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    ones_arr = np.where(h > 0.5)[0]
    zeros_arr = np.where(h <= 0.5)[0]
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * 2.0 / n
    best_h = h.copy()
    rem, add = _compute_rem_add(h, ones_arr, zeros_arr)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        best_imp = 0.0
        best_ii = -1
        best_jj = -1
        for ii in range(len(ones_arr)):
            base = corr + rem[ii]
            M = base[None, :] + add
            i_val = ones_arr[ii]
            for jj in range(len(zeros_arr)):
                j_val = zeros_arr[jj]
                M[jj, n - 1] -= 2.0
                idx1 = n - 1 + i_val - j_val
                if 0 <= idx1 < 2 * n - 1:
                    M[jj, idx1] += 1.0
                idx2 = n - 1 + j_val - i_val
                if 0 <= idx2 < 2 * n - 1:
                    M[jj, idx2] += 1.0
            row_max = M.max(axis=1)
            best_j_local = int(np.argmin(row_max))
            if row_max[best_j_local] < cur_max + best_imp - 1e-12:
                best_imp = row_max[best_j_local] - cur_max
                best_ii = ii
                best_jj = best_j_local
        if best_ii < 0:
            break
        i_val = ones_arr[best_ii]
        j_val = zeros_arr[best_jj]
        h[i_val] = 0.0
        h[j_val] = 1.0
        ones_arr = np.delete(ones_arr, best_ii)
        zeros_arr = np.delete(zeros_arr, best_jj)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur_max = float(np.max(corr))
        new_c5 = cur_max * 2.0 / n
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
        rem, add = _compute_rem_add(h, ones_arr, zeros_arr)
    return best_h, best_c5

    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 800:
            break
        if not ones or not zeros:
            break
        if rng.random() < double_frac and len(ones) >= 2 and len(zeros) >= 2:
            i1 = int(rng.integers(0, len(ones)))
            i2 = int(rng.integers(0, len(ones)))
            if i1 == i2:
                i2 = (i2 + 1) % len(ones)
            j1 = int(rng.integers(0, len(zeros)))
            j2 = int(rng.integers(0, len(zeros)))
            if j1 == j2:
                j2 = (j2 + 1) % len(zeros)
            oi1, oi2 = ones[i1], ones[i2]
            oj1, oj2 = zeros[j1], zeros[j2]
            h_new = cur.copy()
            h_new[oi1] = 0.0
            h_new[oi2] = 0.0
            h_new[oj1] = 1.0
            h_new[oj2] = 1.0
            new_c5 = _exact_c5(h_new)
        else:
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            h_new = cur.copy()
            h_new[oi] = 0.0
            h_new[oz] = 1.0
            new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=200, top_k=12):
    n = len(h)
    h = h.copy().astype(float)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    one_minus_h = 1.0 - h
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        for i in range(n):
            if time.time() > deadline:
                break
            B = np.zeros(2 * n - 1)
            B[i:i + n] += one_minus_h[::-1]
            B[n - 1 - i:2 * n - 1 - i] += -h
            A_prime = corr - B * h[i]
            vals_mid = A_prime + B * 0.5
            top_idx = np.argsort(-vals_mid)[:top_k]
            candidates = np.array([0.0, 1.0])
            for a_i in range(len(top_idx)):
                for a_j in range(a_i + 1, len(top_idx)):
                    d1, d2 = top_idx[a_i], top_idx[a_j]
                    denom = B[d1] - B[d2]
                    if abs(denom) > 1e-12:
                        t_cross = (A_prime[d2] - A_prime[d1]) / denom
                        if -0.01 <= t_cross <= 1.01:
                            candidates = np.append(candidates, t_cross)
            best_t = h[i]
            best_val = float(np.max(A_prime + B * h[i]))
            for t in candidates:
                t = max(0.0, min(1.0, float(t)))
                val = float(np.max(A_prime + B * t))
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            if abs(best_t - h[i]) > 1e-10:
                h[i] = best_t
                one_minus_h[i] = 1.0 - best_t
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                improved = True
        h = _project(h, target)
        one_minus_h = 1.0 - h
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5 - 1e-13:
            best_c5 = new_c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5
