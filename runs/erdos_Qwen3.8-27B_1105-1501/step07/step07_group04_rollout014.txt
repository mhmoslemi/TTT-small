```python
def epigraph_nlp(h_start, deadline, max_iter=500):
    n = len(h_start)
    dx = 2.0 / n
    target = n / 2.0
    M = 2 * n - 1
    
    # Initial point
    h0 = np.clip(h_start, 0, 1)
    h0 = project_to_box_sum(h0, target)
    t0 = exact_c5(h0) + 0.01
    
    x0 = np.concatenate([h0, [t0]])
    
    # Objective
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    # Constraints: g_i(x) = t - dx * F_i(h) >= 0 for i = 0..M-1
    # Plus equality: sum(h) - target = 0
    def constraints(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = t - dx * corr
        return np.concatenate([c, [h.sum() - target]])
    
    def constraint_jac(x):
        h = x[:-1]
        # Jacobian of constraint w.r.t. h and t
        # For each lag r, d(t - dx*F_r)/dh[j] = -dx * dF_r/dh[j]
        # d(t - dx*F_r)/dt = 1
        J = np.zeros((M + 1, n + 1))
        center = n - 1
        for r in range(M):
            k = r - center
            if k == 0:
                J[r, :n] = -dx * (1 - 2*h)
            elif k > 0:
                # F_k = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
                # dF_k/dh[j] = [j>=k]*(1-h[j-k]) - [j+k<n]*h[j+k]
                g = np.zeros(n)
                if n > k:
                    g[k:] = (1 - h[:n-k])
                    g[:n-k] -= h[k:]
                J[r, :n] = -dx * g
            else:
                u = -k
                # F_{-u} = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])
                # dF_{-u}/dh[j] = [j<n-u]*(1-h[j+u]) - [j>=u]*h[j-u]
                g = np.zeros(n)
                if n > u:
                    g[:n-u] = (1 - h[u:])
                    g[u:] -= h[:n-u]
                J[r, :n] = -dx * g
            J[r, n] = 1.0  # dt
        
        # Equality constraint gradient
        J[M, :n] = 1.0
        J[M, n] = 0.0
        return J
    
    bounds = [(0, 1)] * n + [(0, 5)]
    
    # Use SLSQP (simpler, works for this size)
    res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': constraints, 'jac': constraint_jac}],
                   method='SLSQP', max_iter=max_iter,
                   options={'ftol': 1e-14, 'maxiter': max_iter})
    
    if res.success or res.fun < t0:
        h_new = project_to_box_sum(res.x[:-1], target)
        return h_new, exact_c5(h_new)
    return h_start, exact_c5(h_start)
```
