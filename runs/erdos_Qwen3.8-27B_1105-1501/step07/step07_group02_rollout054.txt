Let me also think about the "smooth max" PGD approach:

The objective C₅(h) = max_k f_k(h) where f_k(h) = dx * Σ_i h[i]*(1-h[i+k]).

A smooth approximation is: F_β(h) = (1/β) * log(Σ_k exp(β * f_k(h)))

The gradient of F_β is: ∇F_β = Σ_k w_k * ∇f_k where w_k = exp(β*f_k) / Σ exp(β*f_j)

This is the "soft" version. For large β, it approaches the true max. For moderate β (like 20-50), it's a good approximation that's differentiable everywhere.

1. Compute w = softmax(β * corr_values)
2. Compute gradient: g = (dx * G.T @ w) (the weighted sum of row gradients)
3. Take step: h_new = h - lr * g
4. Project: h_new = _project(h_new, target)
5. Accept if C₅ improves

            if res.x is not None and np.all(np.isfinite(res.x)):
                h_try = _project(res.x[:n].copy(), target)
                c_try = _exact_c5(h_try)
                if c_try < best_val - 1e-15:
                    best_val = c_try
                    best_h = h_try.copy()
        except Exception:
            pass

        # Check if we need more constraints
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        t_val = corr[max(np.argsort(corr)[-5:])] * dx
        non_active = [i for i in range(len(corr)) if i not in set(active_idx.tolist())]
        if non_active:
            non_act_corr = corr[non_active]
            if float(np.max(non_act_corr) * dx) > t_val + 1e-8:
                new_active = np.setdiff1d(np.arange(len(corr)), np.array(non_active))
                all_active = np.sort(np.concatenate([active_idx, non_active]))
                top = np.argsort(corr)[-min(80, len(corr)):]
                _NLP_STATE['active_idx'] = np.sort(top)
            else:
                break
        else:
            break
        top_k_init = min(top_k_init + 20, 2 * n - 1)

    return best_h, best_val

def _pgd_solve(h_init, deadline, lr_init=0.003, beta=25.0, max_iter=500):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = _smooth_weights(corr, beta)
        G = _grad_matrix(h)
        grad = G.T @ w

        h_new = h - lr * grad
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)

        if new_c5 < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.015)
        else:
            stall += 1
            lr *= 0.75

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.12, n_pairs=50, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    one_mh = 1.0 - h

    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        nsrc = min(n_pairs, n // 2)
        sources = sort_idx[-nsrc:]
        sinks = sort_idx[:nsrc]
        pairs = []
        for ii in range(nsrc):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(10, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))

        best_step = None
        best_c5_try = cur_c5

        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            n_grid = 40
            ts = np.linspace(t_lo, t_hi, n_grid)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
            best_i = int(np.argmin(vals))
            lo = ts[max(0, best_i - 1)]
            hi = ts[min(n_grid - 1, best_i + 1)]
            if hi - lo < 1e-12:
                t_opt = float(ts[best_i])
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
                for _ in range(20):
                    if hi - lo < 1e-11:
                        break
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                    else:
                        lo, c1, f1 = c1, c2, f2
                        c2 = lo + gr * (hi - lo)
                        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
                t_opt = (lo + hi) / 2.0

            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try

        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            one_mh = 1.0 - h
            stall = 0
            radius = min(radius * 1.015, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08

    return best_h, best_c5

def _make_sine_start(n, freq, phase, rng):
    x = np.arange(n) / n * 2.0
    h = 0.5 + 0.35 * np.sin(np.pi * freq * x + phase)
    h += rng.random(n) * 0.05
    return _project(h, n / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=15000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if not ones or not zeros:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        h_try = cur.copy()
        h_try[ones[oi]] = 0.0
        h_try[zeros[oz]] = 1.0
        c_try = _exact_c5(h_try)
        if c_try < cur_c5 - 1e-14:
            cur = h_try
            cur_c5 = c_try
            ones[oi] = zeros[oz]
            zeros[oz] = ones[oi]
            ones.sort()
            zeros.sort()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Multi-resolution PGD (20% time)
    d1 = t0 + budget_s * 0.20
    n_list = [60, 80, 100, 120, 150, 180]
    coarse_h = h_init.copy()
    for n_try in n_list:
        if time.time() > d1:
            break
        if n_try == n0:
            sub_h = h_init.copy()
        else:
            sub_h = _resample(coarse_h, n_try)
        sub_dl = min(d1, time.time() + budget_s * 0.025)
        sub_h, c5_sub = _pgd_solve(sub_h, sub_dl, lr_init=0.004, beta=20.0, max_iter=300)
        if c5_sub < _exact_c5(_resample(best_h, n_try)):
            coarse_h = sub_h.copy()
        # Try NLP at this resolution if fast enough
        if n_try <= 120 and time.time() < d1:
            sub_dl2 = min(d1, time.time() + budget_s * 0.015)
            h_nlp, c5_nlp = _nlp_solve(sub_h, sub_dl2, maxiter=50, cg_rounds=2, top_k_init=30)
            if c5_nlp < c5_sub:
                coarse_h = h_nlp.copy()
                c5_sub = c5_nlp
        # Upsample for next iteration
        if n_try < 180:
            coarse_h = _resample(coarse_h, n_try + (n_list[n_list.index(n_try) + 1] - n_try))

    # Phase 2: NLP at n=180 (25% time)
    d2 = t0 + budget_s * 0.45
    starts = [h_init, _resample(coarse_h, n0)]
    # Add diverse starts
    for freq in [3, 5, 7]:
        starts.append(_make_sine_start(n0, freq, rng.random() * 2 * np.pi, rng))
    # Shifted parent
    for shift in [10, -10, 25]:
        starts.append(np.roll(h_init, shift).copy())

    for si, h_start in enumerate(starts):
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_nlp, c5_nlp = _nlp_solve(h_start, sub_dl, maxiter=80, cg_rounds=3, top_k_init=40)
        if c5_nlp < best_c5:
            best_c5 = c5_nlp
            best_h = h_nlp.copy()

    # Phase 3: PGD at n=180 from multiple starts (20% time)
    d3 = t0 + budget_s * 0.65
    pgd_starts = [best_h, h_init, _resample(coarse_h, n0)]
    for freq in [4, 6]:
        pgd_starts.append(_make_sine_start(n0, freq, rng.random() * 3.14, rng))

    for ps in pgd_starts:
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_pgd, c5_pgd = _pgd_solve(ps, sub_dl, lr_init=0.003, beta=25.0, max_iter=400)
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()

    # Phase 4: Pair optimization (25% time)
    d4 = t0 + budget_s * 0.90
    h4 = _project(best_h.copy(), n0 / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.12)
    h4, c5_4 = _pair_optimize(h4, sub_dl, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=1)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Try nearby n values
    n_curr = len(best_h)
    for dn in [-6, 6, -12, 12]:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.12, n_pairs=40, beta=22.0, stall_limit=60, seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Binary + final polish (10% time)
    d5 = deadline - 2
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=20000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Final pair polish
    if time.time() < d5:
        n_curr = len(best_h)
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, deadline)
        h5, c5_5 = _pair_optimize(h5, sub_dl, radius=0.06, n_pairs=40, beta=30.0, stall_limit=50, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
