def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))

            x = np.clip(x, 0.0, 1.0)

    A_box[N:, :N] = -np.eye(N)
    b_box = np.concatenate([np.full(N, radius), np.full(N, radius)])
    A_ineq = np.vstack([A_ineq, A_box])
    b_ineq = np.concatenate([b_ineq, b_box])
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.success:
            d = res.x[:N].copy()
            t_val = float(res.x[N])
            d -= np.mean(d)
            if np.linalg.norm(d) > 1e-10:
                return d, t_val
    except Exception:
        pass
    return None, None

def _line_search_quad(h, d, raw, dx, radius, n_top=14):
    n = len(h)
    Q = np.correlate(d, d, mode='full')
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A_ = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A_) > 1e-14:
                disc = B * B + 4.0 * A_ * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    for aa in [(B + sd) / (2.0 * A_), (B - sd) / (2.0 * A_)]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    n_grid = 30
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(40):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    return best_alpha, best_val

def _subgradient(h, raw, active_lags):
    N = len(h)
    g = np.zeros(N)
    for r in active_lags:
        offset = r - (N - 1)
        n_lo = max(0, -offset)
        n_hi = min(N, N - offset)
        for n in range(n_lo, n_hi):
            g[n + offset] += 1.0 - h[n]
            g[n] -= h[n + offset]
    return g

def _gradient_dir(h, raw, active_lags, N, radius):
    g = _subgradient(h, raw, active_lags)
    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return None
    d = -g / nm
    pos = d > 1e-12
    neg = d < -1e-12
    scale = radius
    if np.any(pos):
        scale = min(scale, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        scale = min(scale, float(np.min(h[neg] / (-d[neg]))))
    if scale < 1e-12:
        return None
    return d * scale

def _lp_optimize(h, deadline, top_k=10, radius=0.3, stall_limit=150):
    N = len(h)
    target = N / 2.0
    dx = 2.0 / N
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rad = radius
    for _ in range(10000):
        if time.time() > deadline or stall > stall_limit:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(raw))
        spread = M - float(np.min(raw))
        threshold = M - 0.08 * spread
        active = np.where(raw >= threshold)[0]
        if len(active) > 2 * top_k:
            active = np.argsort(-raw)[:2 * top_k]
        if len(active) < 3:
            active = np.argsort(-raw)[:top_k]
        accepted = False
        # Try LP with various active sets
        for frac in [1.0, 0.7, 0.5]:
            k_eff = max(3, int(len(active) * frac))
            top_active = np.argsort(-raw)[:k_eff]
            d, t_val = _lp_direction(h, N, top_active, rad)
            if d is not None and t_val < -1e-8:
                alpha, _ = _line_search_quad(h, d, raw, dx, rad)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        rad = min(rad * 1.03, 0.5)
                        accepted = True
                        break
        if not accepted:
            d = _gradient_dir(h, raw, active, N, rad)
            if d is not None:
                alpha, _ = _line_search_quad(h, d, raw, dx, rad)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        accepted = True
        if not accepted:
            stall += 1
            rad *= 0.88
            if stall % 25 == 0 and rad < 0.1:
                rad = 0.25
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-15:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[-k:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, _ = _line_search_quad(h, d, raw, dx, radius, n_top=8)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = _project(h + alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization on initial n (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_optimize(h_init, d1, top_k=10, radius=0.3, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary + pair sweep (10%)
    d2 = t0 + budget_s * 0.30
    if time.time() < d2:
        h_bin = _to_binary(best_h)
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=30000, T_init=0.0025)
        if time.time() < d2 - 2:
            sub_dl = min(d2, time.time() + budget_s * 0.03)
            h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()
        # Polish with LP
        if time.time() < d2 - 2:
            h_pol = _project(h_bin.copy(), len(h_bin) // 2 + len(h_bin) % 2 * 0)
            h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
            sub_dl = min(d2, time.time() + budget_s * 0.03)
            h_pol, c5_p = _lp_optimize(h_pol, sub_dl, top_k=8, radius=0.2, stall_limit=50)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()

    # Phase 3: Multi-n exploration with LP (25%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    tried_ns = {n_curr}
    for dn in list(range(-12, 13, 2)):
        if time.time() > d3 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        budget_r = min(budget_s * 0.015, (d3 - time.time()) * 0.5)
        if budget_r < 3:
            continue
        sub_dl = time.time() + budget_r
        h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k=8, radius=0.25, stall_limit=60)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=10000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive LP optimization (25%)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    # Main LP pass
    sub_dl = min(d4, time.time() + budget_s * 0.10)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, top_k=12, radius=0.20, stall_limit=200)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Pair swaps
    if time.time() < d4 - 5:
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _pair_swap_cont(best_h, sub_dl, n_pairs=6000, radius=0.20, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Another LP pass
    if time.time() < d4 - 3:
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4 = _project(best_h.copy(), n_curr / 2.0)
        h4, c5_4 = _lp_optimize(h4, sub_dl, top_k=15, radius=0.12, stall_limit=100)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (12%)
    d5 = deadline - 5
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, top_k=10, radius=0.15, stall_limit=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline - 1, top_k=16, radius=0.08, stall_limit=60)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline - 1, n_pairs=2000, radius=0.10, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
