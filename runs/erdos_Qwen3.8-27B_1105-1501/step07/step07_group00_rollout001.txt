                    mm = n - g2
                    if mm > 0:
                        g[:mm] += w * dx * (1.0 - h[g2:])
                        g[g2:] += -w * dx * h[:mm]
            g -= float(np.mean(g))
            # Normalize
            gnorm = float(np.sqrt(np.sum(g**2)))
            if gnorm < 1e-15:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
            d = g / gnorm * radius
            d[h >= 1.0 - 1e-12] = np.minimum(d[h >= 1.0 - 1e-12], 0.0)
            d[h <= 1e-12] = np.maximum(d[h <= 1e-12], 0.0)
            d -= np.mean(d)
            if float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
            gap = max(gap * 0.92, 0.0005)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.75
            gap *= 1.3
            n_active = min(n_active + 2, 30)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200):
    """Pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    for r in range(max_rounds):
        if time.time() > deadline or stall > 80:
            break
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(max(1, len(ones) // 4)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            if zi == oj:
                continue
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.25)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 5
    freqs = rng.uniform(0.5, 10.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amps = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
