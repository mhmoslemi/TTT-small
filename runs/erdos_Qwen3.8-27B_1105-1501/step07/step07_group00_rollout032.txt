def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    hx = np.clip(h, 0.0, 1.0)
    return float(np.max(np.correlate(hx, 1.0 - hx, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_raw(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

    best_idx = int(np.argmin(grid_f))

def _active_lp(h, scores, radius, n_active=16, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:

        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=30.0, radius=0.1):

    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0

        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    for j in range(n):
        if h[j] > 1.0 - 1e-12:
            g[j] = min(g[j], 0.0)
        elif h[j] < 1e-12:
            g[j] = max(g[j], 0.0)
        else:
            g[j] = min(g[j], (1.0 - h[j]) / radius)
            g[j] = max(g[j], -h[j] / radius)
    g -= float(np.mean(g))
    return g

When we swap r ∈ O to Z and s ∈ Z to O:

- New Z' = Z \ {s} ∪ {r}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    hx = np.clip(h, 0.0, 1.0)
    return float(np.max(np.correlate(hx, 1.0 - hx, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _active_lp(h, scores, radius, n_active=16, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=30.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    for j in range(n):
        if h[j] > 1.0 - 1e-12:
            g[j] = min(g[j], 0.0)
        elif h[j] < 1e-12:
            g[j] = max(g[j], 0.0)
        else:
            g[j] = min(g[j], (1.0 - h[j]) / radius)
            g[j] = max(g[j], -h[j] / radius)
    g -= float(np.mean(g))
    return g

def _bin_counts(b):

def _bin_swap_update(A, n, r, s, ones_arr, zeros_arr):
    """In-place update of A after swapping r(1->0) and s(0->1)."""
    offset = n - 1
    # Remove (r,z) for z in zeros
    for z in zeros_arr:
        A[r - z + offset] -= 1
    # Remove (o,s) for o in ones
    for o in ones_arr:
        A[o - s + offset] -= 1
    # Fix double removal of (r,s)
    A[r - s + offset] += 1
    # Add (s,z) for z in zeros where z != s
    for z in zeros_arr:
        if z != s:
            A[s - z + offset] += 1
    # Add (s,r)
    A[s - r + offset] += 1
    # Add (o,r) for o in ones where o != r
    for o in ones_arr:
        if o != r:
            A[o - r + offset] += 1
    # Add (s,r) again (o=s case)
    A[s - r + offset] += 1
    # Fix double addition of (s,r)
    A[s - r + offset] -= 1

def _bin_dual_swap_update(A, n, r1, r2, s1, s2, ones_arr, zeros_arr):

    cur_max = int(A.max())
    best_b = b.copy()
    best_max = cur_max
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    it = 0
    while it < max_iter:
        if time.time() > deadline:
            break
        if stall > 5000:
            T = max(T, T_init * 0.5)
            stall = 0
        it += 1
        if it % 200000 == 0:
            T *= cool_rate
        if T < 0.001:
            break
        half = n // 2
        if rng.random() < dual_frac and half >= 2:
            # Dual swap
            oi1 = int(rng.integers(half))
            oi2 = int(rng.integers(half - 1))
            if oi2 >= oi1:
                oi2 += 1
            zi1 = int(rng.integers(half))
            zi2 = int(rng.integers(half - 1))
            if zi2 >= zi1:
                zi2 += 1
            r1, r2 = ones[oi1], ones[oi2]
            s1, s2 = zeros[zi1], zeros[zi2]
            A_old_max = A.max()
            _bin_dual_swap_update(A, n, r1, r2, s1, s2, ones, zeros)
            new_max = int(A.max())
            delta = (new_max - int(A_old_max)) * dx
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                b[r1] = 0; b[r2] = 0; b[s1] = 1; b[s2] = 1
                ones[oi1] = s1; ones[oi2] = s2
                zeros[zi1] = r1; zeros[zi2] = r2
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_b = b.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                A_old_max = A.max()
                _bin_dual_swap_update(A, n, s1, s2, r1, r2, zeros, ones)
                stall += 1
        else:
            # Single swap
            oi = int(rng.integers(half))
            zi = int(rng.integers(half))
            r = ones[oi]
            s = zeros[zi]
            A_old_max = int(A.max())
            _bin_swap_update(A, n, r, s, ones, zeros)
            new_max = int(A.max())
            delta = (new_max - A_old_max) * dx
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                b[r] = 0; b[s] = 1
                ones[oi] = s
                zeros[zi] = r
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_b = b.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                _bin_swap_update(A, n, s, r, zeros, ones)
                stall += 1
    return best_b, best_max * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_balanced_binary(h, n, rng):
    m = n // 2
    idx = np.argsort(h)[-m:]
    b = np.zeros(n, dtype=np.int64)
    b[idx] = 1
    return b

def _random_balanced(n, rng):
    m = n // 2
    b = np.zeros(n, dtype=np.int64)
    perm = rng.permutation(n)
    b[perm[:m]] = 1
    return b

def _structured_binary(n, rng, variant=0):
    m = n // 2
    b = np.zeros(n, dtype=np.int64)
    if variant == 0:
        # Alternating
        b[::2] = 1
        b = b[:n]
        if b.sum() != m:
            b = _random_balanced(n, rng)
    elif variant == 1:
        # Arithmetic progression with step = n/m rounded
        step = max(1, n // m)
        for i in range(m):
            pos = (i * step) % n
            b[pos] = 1
        if b.sum() != m:
            extra = m - b.sum()
            if extra > 0:
                open_pos = np.where(b == 0)[0]
                b[open_pos[:extra]] = 1
            elif extra < 0:
                filled_pos = np.where(b == 1)[0]
                b[filled_pos[:extra]] = 0
    elif variant == 2:
        # Low discrepancy via golden ratio
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(m):
            pos = int((i * phi) % 1.0 * n)
            pos = min(pos, n - 1)
            b[pos] = 1
        if b.sum() != m:
            b = _random_balanced(n, rng)
    elif variant == 3:
        # Random blocks
        b = _random_balanced(n, rng)
    else:
        b = _random_balanced(n, rng)
    return b

def _pair_sweep(h, deadline, max_rounds=100):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        improved = False
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ntrials = max(1, len(ones) // 4)
        for _ in range(ntrials):
            if time.time() > deadline:
                break
            i = int(rng.integers(len(zeros)))
            j = int(rng.integers(len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.25)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 10.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    # Get parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous from parent (12% time)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _continuous_opt(h_init, d1, radius_init=0.10, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary SA multi-start (38% time)
    d2 = t0 + budget_s * 0.50
    n_bin = n0
    if n_bin % 2 != 0:
        n_bin += 1
        h_work = _resample(best_h, n_bin)
    else:
        h_work = best_h.copy()

    best_bin_b = None
    best_bin_c5 = best_c5
    n_starts = 12
    time_per_start = max(2.0, (d2 - time.time()) / max(1, n_starts))
    for s in range(n_starts):
        if time.time() > d2 - 2:
            break
        sub_dl = min(d2, time.time() + time_per_start)
        if s == 0:
            b = _to_balanced_binary(h_work, n_bin, rng)
        elif s == 1:
            b = _structured_binary(n_bin, rng, 0)
        elif s == 2:
            b = _structured_binary(n_bin, rng, 1)
        elif s == 3:
            b = _structured_binary(n_bin, rng, 2)
        elif s == 4:
            b = _structured_binary(n_bin, rng, 3)
        else:
            b = _random_balanced(n_bin, rng)
        T_init = 0.3 + 0.3 * (s % 3) / 2.0
        b_out, c5_bin = _binary_sa(b, sub_dl, T_init=T_init, cool_rate=0.9997,
                                    max_iter=10000000, dual_frac=0.08, seed=seed + s * 101)
        if c5_bin < best_bin_c5:
            best_bin_c5 = c5_bin
            best_bin_b = b_out.copy()
            time_per_start = max(2.0, time_per_start * 1.15)
        else:
            time_per_start = max(1.5, time_per_start * 0.85)

    # Convert best binary to continuous and polish
    if best_bin_b is not None and best_bin_c5 < best_c5 + 0.01:
        h_bin = best_bin_b.astype(float)
        # Try small continuous perturbation from binary
        h_bin_pert = _smooth_perturb(h_bin, 0.05, rng)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_bin_pol, c5_bin_pol = _continuous_opt(h_bin_pert, sub_dl, radius_init=0.05, max_iters=800)
        if c5_bin_pol < best_c5:
            best_c5 = c5_bin_pol
            best_h = h_bin_pol.copy()
        # Also try the raw binary
        if best_bin_c5 < best_c5:
            best_c5 = best_bin_c5
            best_h = h_bin.copy()

    # Phase 3: Continuous polish from best (20% time)
    d3 = t0 + budget_s * 0.70
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c5_3 = _continuous_opt(h3, d3, radius_init=0.08, max_iters=3000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Pair sweep
    if time.time() < d3 + budget_s * 0.03:
        sub_dl = min(t0 + budget_s * 0.73, time.time() + budget_s * 0.03)
        h3b = _project(best_h.copy(), len(best_h) / 2.0)
        h3b, c5_3b = _pair_sweep(h3b, sub_dl, max_rounds=80)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (15% time)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    tried_ns = {n_curr}
    ns_to_try = [n_curr + dn for dn in [-20, -12, -8, -4, 4, 8, 12, 20, 28, 36]]
    ns_to_try = [n for n in ns_to_try if 80 <= n <= 260 and n not in tried_ns]
    ns_to_try.sort(key=lambda n: abs(n - n_curr))
    for n_try in ns_to_try[:6]:
        if time.time() > d4 - 3:
            break
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _continuous_opt(h_r, sub_dl, radius_init=0.08, max_iters=600)
        if c5_r < best_c5 and time.time() < d4 - 2:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, max_rounds=40)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Basin hopping (10% time)
    d5 = t0 + budget_s * 0.95
    scale = 0.03
    for _ in range(500):
        if time.time() > d5:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _continuous_opt(h_pert, sub_dl, radius_init=0.05, max_iters=400)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: Binary restart on best continuous (5% time)
    d6 = deadline - 2
    if time.time() < d6 - 5:
        n_b = len(best_h)
        if n_b % 2 == 0:
            b_start = _to_balanced_binary(best_h, n_b, rng)
            sub_dl = min(d6, time.time() + budget_s * 0.04)
            b_out, c5_fin = _binary_sa(b_start, sub_dl, T_init=0.02, cool_rate=0.9998,
                                        max_iter=2000000, dual_frac=0.05, seed=seed + 999)
            h_fin = b_out.astype(float)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
            # Polish
            if time.time() < d6 - 2:
                h_pol = _smooth_perturb(h_fin, 0.03, rng)
                sub_dl2 = min(d6, time.time() + budget_s * 0.015)
                h_pol, c5_pol = _continuous_opt(h_pol, sub_dl2, radius_init=0.04, max_iters=500)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Final polish
    if deadline - time.time() > 3:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _continuous_opt(h_f, deadline - 0.5, radius_init=0.03, max_iters=1000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
