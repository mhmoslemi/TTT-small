def _line_search(h, d, radius):
    """Exact line search for C5(h + alpha*d) subject to 0<=h<=1."""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-15
    pos = d > eps
    neg = d < -eps
    
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])) - 1e-14)
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])) - 1e-14)
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])) - 1e-14)
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))) - 1e-14)
    
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    # Grid search
    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.array([f(t) for t in gt])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    
    # Check vertex of each active quadratic
    for r in np.argsort(S0)[-10:]:

            if a_lo <= av <= a_hi:

                if fv < opt_v:

    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    
    return opt_t, opt_v

def _anti_opt_lbfgsb(x, deadline, beta_start=10.0, beta_end=200.0, n_restarts=3, seed=42):

    best_c5 = _exact_c5(_x_to_h(x))
    rng = np.random.default_rng(seed)
    
    for restart in range(n_restarts):
        beta = beta_start * (beta_end / beta_start) ** (restart / max(1, n_restarts - 1))
        # For the last restart, use very high beta
        if restart == n_restarts - 1:
            beta = beta_end
        
        x0 = best_x.copy()
        if restart > 0:

            x0 = np.clip(x0 + rng.normal(0, 0.01, m), 0.0, 1.0)

            return _smooth_obj_and_grad(xv, beta)[0]
        
        def obj_grad_func(xv):
            _, g = _smooth_obj_and_grad(xv, beta)
            return g
        
        res = minimize(obj_func, x0, jac=obj_grad_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'gtol': 1e-8,
                       'maxls': 50, 'disp': False})
        
        x_new = np.clip(res.x, 0.0, 1.0)

            best_c5 = c5_new
            best_x = x_new.copy()
        
        if time.time() > deadline:
            break
    
    return best_x, best_c5

                if fv < opt_v:
                    opt_t, opt_v = av, fv

    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft

    return opt_t, opt_v

def _anti_lbfgsb(x, deadline, beta_start=10.0, beta_end=200.0, n_restarts=3, seed=42):
    m = len(x)
    bounds = [(0.0, 1.0)] * m
    best_x = x.copy()
    best_c5 = _exact_c5(_x_to_h(x))
    rng = np.random.default_rng(seed)

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        frac = restart / max(1, n_restarts - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        if restart == 0:
            beta = beta_start
        else:
            beta = beta_start * ((beta_end / beta_start) ** (restart / max(1, n_restarts - 1)))

        x0 = best_x.copy()
        if restart > 0:
            amp = 0.02 / restart
            x0 = np.clip(x0 + rng.normal(0, amp, m), 0.0, 1.0)

        def obj_func(xv):
            return _smooth_obj_and_grad(xv, beta)[0]

        def obj_grad_func(xv):
            return _smooth_obj_and_grad(xv, beta)[1]

        try:
            res = minimize(obj_func, x0, jac=obj_grad_func, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 300, 'gtol': 1e-9, 'maxls': 40})
        except Exception:
            break

        x_new = np.clip(res.x, 0.0, 1.0)
        c5_new = _exact_c5(_x_to_h(x_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_x = x_new.copy()

    return best_x, best_c5

def _full_dir_opt(h, deadline, beta=20.0, radius_init=0.15, max_stall=50, n_dirs=6, seed=555):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline and stall < max_stall:
        raw = _raw_from_h(h)
        M = float(np.max(raw))
        z = beta * (raw - M)
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            idx = int(np.argmax(raw))
            w = np.zeros(len(raw))
            w[idx] = 1.0
        else:
            w /= wsum

        G = np.zeros(n)
        for r in range(len(raw)):
            wr = w[r]
            if wr < 1e-10:
                continue
            tau = r - (n - 1)
            if tau >= 0:
                me = n - tau
                if me > 0:
                    G[tau:] += wr * (1.0 - h[:me])
                    G[:me] -= wr * h[tau:tau + me]
            else:
                d = -tau
                me = n - d
                if me > 0:
                    G[:me] += wr * (1.0 - h[d:])
                    G[d:] -= wr * h[:me]
        G -= np.mean(G)

        dirs = []
        gn = np.linalg.norm(G)
        if gn > 1e-12:
            dirs.append(G / gn)

        top_lags = np.argsort(raw)[-4:]
        for r in top_lags:
            gr = _grad_full_h(h, r)
            gr -= np.mean(gr)
            gnr = np.linalg.norm(gr)
            if gnr > 1e-12:
                dirs.append(gr / gnr)

        blk = 4
        if n % blk == 0 and len(dirs) < n_dirs:
            nblk = n // blk
            gb = G.reshape(nblk, blk).mean(axis=1)
            db = np.tile(gb, (1, blk)).reshape(n)
            db -= np.mean(db)
            dbn = np.linalg.norm(db)
            if dbn > 1e-12:
                dirs.append(db / dbn)

        x_arr = np.arange(n, dtype=float)
        for freq in [1, 2]:
            if len(dirs) >= n_dirs:
                break
            dc = np.sin(2.0 * np.pi * freq * x_arr / n + rng.uniform(0, 2 * np.pi))
            dc -= np.mean(dc)
            dcn = np.linalg.norm(dc)
            if dcn > 1e-12:
                dirs.append(dc / dcn)

        best_alpha = 0.0
        best_val = cur_c5
        best_d = None

        for d in dirs[:n_dirs]:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d

        if best_d is not None and best_alpha != 0.0:
            h_new = _project(h + best_alpha * best_d, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.05, radius * 0.80)
        else:
            stall += 1
            radius = max(0.05, radius * 0.80)

    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_hill_climb(h_bin, deadline, max_iter=40000, seed=777):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(max_iter):
        if time.time() > deadline or len(ones) == 0 or len(zeros) == 0:
            break
        if stall > 400:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1

    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for oi in ones:
        if time.time() > deadline:
            break
        improved = False
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                improved = True
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
        if not improved:
            continue
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Anti-palindromic L-BFGS-B with increasing beta (30%)
    d1 = t0 + budget_s * 0.30
    m0 = n0 // 2
    x0 = _h_to_x(best_h)
    x_opt, c5_a = _anti_lbfgsb(x0, d1, beta_start=8.0, beta_end=400.0, n_restarts=5, seed=seed)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = _x_to_h(x_opt).copy()

    # Phase 2: Full-direction optimization (20%)
    d2 = t0 + budget_s * 0.50
    h_full = _project(best_h.copy(), len(best_h) / 2.0)
    h_full, c5_f = _full_dir_opt(h_full, d2, beta=20.0, radius_init=0.15, max_stall=50, n_dirs=6, seed=seed + 1)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_full.copy()

    # Phase 3: Multi-resolution search (25%)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    try_offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16]
    for dn in try_offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        mr = n_try // 2
        xr = _h_to_x(h_r)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        xr_opt, c5_r = _anti_lbfgsb(xr, sub_dl, beta_start=12.0, beta_end=200.0, n_restarts=2, seed=seed + 100 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = _x_to_h(xr_opt).copy()
            n_curr = n_try
        if time.time() < d3 - 4:
            h_rc = _project(_x_to_h(xr_opt), n_try / 2.0)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_rc, c5_rc = _full_dir_opt(h_rc, sub_dl2, beta=25.0, radius_init=0.10, max_stall=30, n_dirs=4, seed=seed + 200 + dn)
            if c5_rc < best_c5:
                best_c5 = c5_rc
                best_h = h_rc.copy()
                n_curr = n_try

    # Phase 4: Binary optimization (15%)
    d4 = t0 + budget_s * 0.90
    h_b = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h_b, c5_b = _binary_hill_climb(h_b, sub_dl, max_iter=50000, seed=seed + 300)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    if time.time() < d4 - 3:
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_b2, c5_b2 = _pair_sweep(best_h, sub_dl2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Polish binary result with continuous
    if time.time() < d4 - 2:
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _full_dir_opt(h_pol, sub_dl3, beta=30.0, radius_init=0.08, max_stall=25, n_dirs=5, seed=seed + 400)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Basin hopping (remaining)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 50:
        n_restarts += 1
        n = len(best_h)
        pert = rng.normal(0, scale, n)
        h_p = _project(best_h + pert, n / 2.0)
        m = n // 2
        x_p = _h_to_x(h_p)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.006)
        x_t, c5_t = _anti_lbfgsb(x_p, sub_dl, beta_start=20.0, beta_end=300.0, n_restarts=1, seed=seed + 500 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = _x_to_h(x_t).copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _full_dir_opt(h_f, deadline, beta=40.0, radius_init=0.05, max_stall=35, n_dirs=6, seed=seed + 900)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure exact feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    final_c5 = min(final_c5, best_c5)

    return best_h, final_c5, len(best_h)
