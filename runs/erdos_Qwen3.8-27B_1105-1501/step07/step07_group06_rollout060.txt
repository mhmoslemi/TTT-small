    """Compute full gradient G where G[i] = sum_r w[r] * d(raw[r])/dh[i]"""
    n = len(h)

    for r in range(2*n-1):

    h = _anti_to_full(x)
    raw = _raw_corr(h)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _anti_to_full(x):
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h):
    m = len(h) // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])

def _smooth_value(x, beta):
    h = _anti_to_full(x)
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = raw.max()
    z = np.clip(beta * (raw - mx), -50.0, 0.0)
    lse = np.log(np.sum(np.exp(z)))
    return (mx + lse / beta) * (2.0 / (2 * len(x)))

def _smooth_grad(x, beta):
    h = _anti_to_full(x)
    n = len(h)
    m = n // 2
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = raw.max()
    z = np.clip(beta * (raw - mx), -50.0, 0.0)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(raw)
        w[int(raw.argmax())] = 1.0
    else:
        w /= ws
    G = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-12:
            continue
        tau = r - (n - 1)
        if tau >= 0:
            mg = n - tau
            G[tau:] += wr * (1.0 - h[:mg])

def _project_sum(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0

        else:
            hi = mid

    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:

        cnt = mask.sum()
        if cnt > 0:

            x[mask] += res / cnt

def _guard_step(h, g, radius, beta):
    n = len(h)
    d = g - np.mean(g)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return h.copy(), _exact_c5(h)
    d = d / dn
    pos = d > 1e-14
    neg = d < -1e-14
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_lo = max(a_lo, -radius)
    a_hi = min(a_hi, radius)
    if a_hi <= a_lo + 1e-14:
        return h.copy(), _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')

    dx_val = 2.0 / n

    def _f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx_val

    ng = 40
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.array([_f(t) for t in gt])

    h_new = h + opt_t * d
    h_new = _project_sum(h_new, n / 2.0)

    n_per = max(2.0, (d2 - t0) / max(len(n_candidates), 1))

    for nt in n_candidates:
        if time.time() > d2 - 8:
            break
        alloc = min(n_per, d2 - 8 - time.time())
        if alloc < 3:
            break
        h_r = _resample(best_h, nt)
        h_r = _ensure_feasible(h_r)
        x_r = _full_to_anti(h_r)
        x_r = np.clip(x_r, 0.0, 1.0)
        # Anti-pal L-BFGS
        sub_dl_a = min(d2 - 8, time.time() + alloc * 0.4)
        for beta in [20.0, 50.0]:
            if time.time() > sub_dl_a:
                break
            xf, cf = _lbfgs_anti(x_r, sub_dl_a, beta=beta, maxiter=150)
            hf = _anti_to_full(xf)
            hf = _ensure_feasible(hf)
            cf_e = _exact_c5(hf)
            if cf_e < best_c5:
                best_c5 = cf_e
                best_h = hf.copy()
            x_r = _full_to_anti(best_h)
            x_r = np.clip(x_r, 0.0, 1.0)
        # Full-space
        if time.time() < d2 - 6:
            sub_dl_f = min(d2 - 6, time.time() + alloc * 0.25)
            hf, cf = _full_opt(h_r, sub_dl_f, beta=30.0, max_iter=60, radius_init=0.10)
            if cf < best_c5:
                best_c5 = cf
                best_h = hf.copy()
                h_r = hf.copy()
        # Binary
        if time.time() < d2 - 4:
            hb = _to_bin(_ensure_feasible(h_r))
            sub_dl_b = min(d2 - 4, time.time() + alloc * 0.25)
            hb, cb = _bin_sa(hb, sub_dl_b, n_iter=12000, T0=0.003, seed=int(rng.integers(99999)))
            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()

    # Phase 3: Basin hopping in anti-pal space (25% time)
    d3 = t0 + budget_s * 0.75
    x_b = _full_to_anti(best_h)
    x_b = np.clip(x_b, 0.0, 1.0)
    m_b = len(x_b)
    scale = 0.04
    n_restarts = 0
    while time.time() < d3 - 8 and n_restarts < 300:
        n_restarts += 1
        x_p = x_b + scale * rng.normal(0, 1, m_b)
        x_p = np.clip(x_p, 0.0, 1.0)
        sub_dl = min(d3 - 5, time.time() + budget_s * 0.004)
        xf, cf = _lbfgs_anti(x_p, sub_dl, beta=30.0, maxiter=80)
        hf = _anti_to_full(xf)
        hf = _ensure_feasible(hf)
        cf_e = _exact_c5(hf)
        if cf_e < best_c5 - 1e-15:
            best_c5 = cf_e
            best_h = hf.copy()
            x_b = np.clip(_full_to_anti(best_h), 0.0, 1.0)
            scale = max(0.008, scale * 0.92)
        else:
            scale = min(0.12, scale * 1.06)

    # Phase 4: Diverse initializations in anti-pal (15% time)
    d4 = t0 + budget_s * 0.90
    n_div = len(best_h)
    m_div = n_div // 2
    n_div_dl_each = max(2.0, (d4 - t0) / 12.0)
    n_div_done = 0
    while time.time() < d4 - 5 and n_div_done < 12:
        n_div_done += 1
        sub_dl = min(d4 - 5, time.time() + n_div_dl_each)
        r = rng.random()
        if r < 0.33:
            freqs = [rng.uniform(0.5, 4.0) for _ in range(3)]
            amps = [rng.uniform(0.1, 0.35) for _ in range(3)]
            phases = [rng.uniform(0, 2 * np.pi) for _ in range(3)]
            x_init = _make_sinusoidal(m_div, freqs, amps, phases, rng)
        elif r < 0.66:
            x_init = _make_random_anti(m_div, rng)
        else:
            x_init = _make_block_anti(m_div, rng, int(rng.integers(2, 8)))
        x_init = np.clip(x_init, 0.0, 1.0)
        xf, cf = _lbfgs_anti(x_init, sub_dl, beta=25.0, maxiter=120)
        hf = _anti_to_full(xf)
        hf = _ensure_feasible(hf)
        cf_e = _exact_c5(hf)
        if cf_e < best_c5:
            best_c5 = cf_e
            best_h = hf.copy()

    # Phase 5: Final binary + polish (10% time)
    d5 = deadline
    hb = _to_bin(_ensure_feasible(best_h))
    sub_dl1 = min(d5 - 15, time.time() + budget_s * 0.04)
    hb, cb = _bin_sa(hb, sub_dl1, n_iter=40000, T0=0.005, seed=999)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb.copy()

    # Greedy binary
    if time.time() < d5 - 10:
        hb2 = _to_bin(_ensure_feasible(best_h))
        sub_dl2 = min(d5 - 10, time.time() + budget_s * 0.03)
        hb2, cb2 = _bin_greedy(hb2, sub_dl2, max_rounds=30)
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2.copy()

    # Final anti-pal polish
    if time.time() < d5 - 3:
        x_f = np.clip(_full_to_anti(best_h), 0.0, 1.0)
        sub_dl3 = min(d5 - 2, time.time() + budget_s * 0.03)
        for beta in [40.0, 80.0]:
            if time.time() > sub_dl3:
                break
            xf, cf = _lbfgs_anti(x_f, sub_dl3, beta=beta, maxiter=150)
            hf = _anti_to_full(xf)
            hf = _ensure_feasible(hf)
            cf_e = _exact_c5(hf)
            if cf_e < best_c5:
                best_c5 = cf_e
                best_h = hf.copy()
            x_f = np.clip(_full_to_anti(best_h), 0.0, 1.0)

    # Final full-space polish
    if time.time() < d5 - 2:
        sub_dl4 = min(d5 - 1, time.time() + budget_s * 0.02)
        hf, cf = _full_opt(best_h, sub_dl4, beta=50.0, max_iter=50, radius_init=0.06)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    best_h = _ensure_feasible(best_h)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
