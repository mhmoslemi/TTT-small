def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_gradient(h, weights):
    """Compute gradient g = sum_r w_r * dR_r/dh for R = correlate(h, 1-h, 'full')"""
    n = len(h)
    g = np.zeros(n)
    for r in range(2*n - 1):
        w = weights[r]
        if w == 0:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            g[d:] += w * (1.0 - h[:m])
            g[:m] -= w * h[d:]
        else:
            b = -d
            m = n - b
            if m <= 0:
                continue
            g[b:] += w * (1.0 - h[:m])
            g[:m] -= w * h[b:]
    return g

    n = len(g)

    v = np.zeros(n)
    m = int(target)
    rem = target - m
    v[order[:m]] = 1.0
    if rem > 1e-12:

    """Minimize max(R + a*T - a^2*C) over a in [0, alpha_max]"""
    if alpha_max < 1e-14:
        return 0.0
    # Coarse grid
    n_grid = 50
    alphas = np.linspace(0, alpha_max, n_grid)
    best_a, best_v = alphas[0], float(np.max(R))
    for a in alphas:
        v = float(np.max(R + a*T - a*a*C))
        if v < best_v:
