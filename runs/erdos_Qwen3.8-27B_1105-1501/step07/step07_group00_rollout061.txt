import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x > 0.001) & (x < 0.999)
        else:
            mask = (x > 0.001) & (x < 0.999)
        if np.sum(mask) == 0:
            mask = x < 1.0 if res > 0 else x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_matrix(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:d+m] += (1.0 - h[:m])
                G[idx, :m] += -h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g+m])
                G[idx, g:g+m] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0-h, mode='full')
    T = np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg]/(-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Golden section search
    gr = (np.sqrt(5.0)-1.0)/2.0
    lo, hi = 0.0, t_hi
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    # Also check endpoints
    f0 = float(np.max(S0)) * dx
    fedge = float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx
    best_t, best_v = opt_t, opt_val
    if f0 < best_v:
        best_t, best_v = 0.0, f0
    if fedge < best_v:
        best_t, best_v = t_hi, fedge
    return best_t, best_v

def _lp_direction(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n-1)
    G = _grad_matrix(h, lags)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.ones((1, n))
    A_eq = np.hstack([A_eq, np.zeros((1,1))])
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0-h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

def _softmax_gradient(h, beta):
    n = len(h)
    scores_raw = np.correlate(h, 1.0-h, mode='full')
    scaled = beta * scores_raw
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    lags = np.arange(len(scores_raw)) - (n-1)
    G = _grad_matrix(h, lags)
    g = G.T @ w
    g -= np.mean(g)  # project onto sum=0
    return g, np.max(scores_raw)

def _local_opt(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or np.max(np.abs(d)) < 1e-12:
            # Fallback to softmax gradient
            g, _ = _softmax_gradient(h, beta=30.0)
            d = -g  # negative gradient (minimize max)
            # Clip to box
            pos = d > 0
            neg = d < 0
            if np.any(pos):
                d[pos] = np.minimum(d[pos], (1.0-h[pos])/radius)
            if np.any(neg):
                d[neg] = np.maximum(d[neg], -h[neg]/radius)
            d[h >= 1.0-1e-12] = 0.0
            d[h <= 1e-12] = 0.0
            d -= np.mean(d)
            if np.max(np.abs(d)) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.5
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, 2.0/n, radius)

            cur_c5 = _exact_c5(h)

                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)

            gap *= 1.4

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _shift_best(h):
    n = len(h)
    best_c5 = _exact_c5(h)
    best_shift = 0
    for s in range(1, n):
        h_shifted = np.roll(h, s)
        c5 = _exact_c5(h_shifted)
        if c5 < best_c5:
            best_c5 = c5
            best_shift = s
    if best_shift != 0:
        h = np.roll(h, best_shift)
    return h

def _block_perturb(h, rng, n_blocks):
    n = len(h)
    h_p = h.copy()
    targets = rng.permutation(n)[:n_blocks]
    t = 0.15 * rng.uniform(-1, 1, n_blocks)
    for i, ti in zip(targets, t):
        h_p[i] = h_p[i] + ti
    return _project(h_p, n/2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    perturb = np.zeros(n)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2*np.pi*f*np.arange(n)/n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n/2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x > 0.001) & (x < 0.999)
        else:
            mask = (x > 0.001) & (x < 0.999)
        if np.sum(mask) == 0:
            mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scores(h):
    n = len(h)
    return _scores_raw(h) * (2.0 / n)

def _grad_matrix(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:d+m] += (1.0 - h[:m])
                G[idx, :m] += -h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g+m])
                G[idx, g:g+m] += -h[:m]
    return G

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    S0 = _scores_raw(h)
    T = np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg]/(-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    f0 = float(np.max(S0)) * dx
    fedge = float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx
    best_t, best_v = opt_t, opt_val
    if f0 < best_v:
        best_t, best_v = 0.0, f0
    if fedge < best_v:
        best_t, best_v = t_hi, fedge
    return best_t, best_v

def _lp_direction(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_matrix(h, lags)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _softmax_gradient(h, beta):
    n = len(h)
    scores_raw = _scores_raw(h)
    scaled = beta * scores_raw
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / float(np.sum(exp_s))
    lags = np.arange(len(scores_raw)) - (n - 1)
    G = _grad_matrix(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            g = _softmax_gradient(h, beta=25.0)
            d = -g
            pos = d > 1e-15
            neg = d < -1e-15
            if np.any(pos):
                d[pos] = np.minimum(d[pos], (1.0 - h[pos]) / radius)
            if np.any(neg):
                d[neg] = np.maximum(d[neg], -h[neg] / radius)
            d[h >= 1.0 - 1e-12] = 0.0
            d[h <= 1e-12] = 0.0
            d -= float(np.mean(d))
            if float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.5
                n_active = min(n_active + 2, 30)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.4
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _lse_descent(h, deadline, beta_start=8.0, max_iters=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_start
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        g = _softmax_gradient(h, beta)
        d = -g
        d -= float(np.mean(d))
        pos = d > 1e-15
        neg = d < -1e-15
        if np.any(pos):
            d[pos] = np.minimum(d[pos], (1.0 - h[pos]) / 0.2)
        if np.any(neg):
            d[neg] = np.maximum(d[neg], -h[neg] / 0.2)
        d[h >= 1.0 - 1e-12] = 0.0
        d[h <= 1e-12] = 0.0
        d -= float(np.mean(d))
        if float(np.max(np.abs(d))) < 1e-12:
            break
        alpha, new_c5 = _line_search(h, d, 0.15)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            beta *= 1.03
        else:
            stall += 1
            beta *= 0.95
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200, seed_val=1234):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed_val)
    stall = 0
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(4, len(ones) // 4)
        for _ in range(n_tries):
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search(h, d, 0.25)
            if c_try < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _shift_best(h):
    n = len(h)
    best_c5 = _exact_c5(h)
    best_shift = 0
    for s in range(1, n):
        h_shifted = np.roll(h, s)
        c5 = _exact_c5(h_shifted)
        if c5 < best_c5:
            best_c5 = c5
            best_shift = s
    if best_shift != 0:
        h = np.roll(h, best_shift)
    return h

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    perturb = np.zeros(n)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2.0 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _block_perturb(h, rng, n_blocks):
    n = len(h)
    h_p = h.copy()
    targets = rng.permutation(n)[:n_blocks]
    deltas = 0.15 * rng.uniform(-1, 1, n_blocks)
    for idx, delta in zip(targets, deltas):
        h_p[idx] += delta
    return _project(h_p, n / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
