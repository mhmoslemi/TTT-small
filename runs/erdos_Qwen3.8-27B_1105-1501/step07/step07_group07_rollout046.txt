Actually, let me think about what step size is appropriate. The LSE gradient has magnitude roughly O(1) (since the softmax weights sum to 1 and each G entry is O(1)). So the direction d (normalized) has unit norm. A step of alpha in this direction changes h by at most alpha in each coordinate. Since h is in [0,1], alpha should be at most 1. In practice, alpha of 0.01-0.1 is probably good.

Let me use a different approach: instead of normalizing the gradient, use the raw gradient and find the optimal step size via a 1D search. The constraint sum(d)=0 is automatically satisfied if we use d = -(grad - mean(grad)).

Actually, I think the cleanest approach is:

1. Compute direction d = -(grad - mean(grad)) (this satisfies sum(d)=0)
2. Find alpha that minimizes C5(h + alpha * d) subject to 0 <= h + alpha * d <= 1
3. Use golden section or similar

The max alpha is determined by bounds. The function C5(h + alpha*d) is piecewise quadratic (each raw score is bilinear in h, so it's quadratic along the line). The max of quadratics is piecewise quadratic, so it's unimodal between the points where the argmax changes. For a practical approach, golden section should work well.

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_lse(h0, tau, target, maxiter=50):
    import scipy.optimize
    n = len(h0)
    
    def obj(x):
        raw = _raw_scores(x)
        mval = float(np.max(raw))
        u = (raw - mval) / tau
        eu = np.exp(u)
        s = float(np.sum(eu))
        if s < 1e-300:
            return float(mval)
        return float(mval + tau * np.log(s))
    
    def jac(x):
        raw = _raw_scores(x)
        mval = float(np.max(raw))
        u = (raw - mval) / tau
        eu = np.exp(u)
        s = float(np.sum(eu))
        if s < 1e-300:
            p = np.zeros(len(raw))
            mask = (raw == mval)
            p[mask] = 1.0 / float(np.sum(mask))
        else:
            p = eu / s
        g = np.zeros(n)
        g += p[n-1] * (1.0 - 2.0 * x)
        for d in range(1, n):
            m = n - d
            g[d:] += p[n-1+d] * (1.0 - x[:m])
            g[:m] -= p[n-1+d] * x[d:]
            g[:m] += p[n-1-d] * (1.0 - x[d:])
            g[d:] -= p[n-1-d] * x[:m]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    try:
        res = scipy.optimize.minimize(obj, h0, jac=jac, method='SLSQP',
                                       constraints=cons, bounds=bounds,
                                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or hasattr(res, 'x'):
            x = _project_sum(res.x, target)
            return x
    except Exception:
        pass
    return h0

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try
    
    # Phase 6: Final intensive polish (10%)
    n_best = len(best_h)
    # More LSE steps with small tau
    sub_dl = min(deadline, time.time() + budget_s * 0.03)
    h_f = best_h.copy()
    h_f, c5_f = _lse_anneal(h_f, sub_dl, tau_start=0.5,
                             tau_ratio=0.6, steps_per_tau=15, tau_floor=0.02)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Pair swap final
    if deadline - time.time() > 5:
        h_ps = best_h.copy()
        h_ps, c5_ps = _pair_swap(h_ps, deadline, n_iter=8000, seed=888)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
    
    # Basin hopping
    if deadline - time.time() > 10:
        scale = 0.02
        for _ in range(50):
            if time.time() > deadline - 3:
                break
            h_pert = _perturb(best_h, rng, scale)
            sub_dl = min(deadline, time.time() + budget_s * 0.005)
            h_t, c5_t = _lse_anneal(h_pert, sub_dl, tau_start=1.0,
                                     tau_ratio=0.6, steps_per_tau=10, tau_floor=0.05)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.15, scale * 1.1)
    
    # Final projection and evaluation
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
