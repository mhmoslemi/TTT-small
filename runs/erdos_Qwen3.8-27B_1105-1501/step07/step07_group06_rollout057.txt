    if target <= 0:

    if target >= n:
        return np.ones(n)

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_anti_x(h):
    n = len(h)
    m = n // 2
    second_rev = h[n-1:n-m-1:-1]
    x = 0.5 * (h[:m] + 1.0 - second_rev)
    return np.clip(x, 0.0, 1.0)

def _from_anti_x(x):
    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _smooth_grad_anti(x, beta):
    h = _from_anti_x(x)
    n = len(h)
    dx = 2.0 / n
    m = n // 2
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    wsum = float(np.sum(e))
    w = e / wsum
    S = (M + float(np.log(wsum)) / beta) * dx
    G = np.zeros(n)
    active_idx = np.where(w > 1e-8)[0]
    for r in active_idx:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:tau+mm] += wr * u[:mm]
                G[:mm] -= wr * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * u[d:]
                G[d:d+mm] -= wr * h[:mm]
    grad_x = (G[:m] - G[m:][::-1]) * dx
    return S, grad_x

def _g_obj(xv):
    return _smooth_grad_anti(xv, _g_beta)[0]

def _g_grad(xv):
    return _smooth_grad_anti(xv, _g_beta)[1]

def _lbfgs_anti(x, beta, max_iter=200, gtol=1e-8):
    global _g_beta
    _g_beta = beta
    m = len(x)
    bounds = [(0.0, 1.0)] * m
    res = minimize(_g_obj, x.copy(), jac=_g_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': gtol})
    h_opt = _from_anti_x(res.x)
    c5_opt = _exact_c5(h_opt)
    return res.x, c5_opt

def _resample(h, n_new):
    if n_new % 2 != 0:
        n_new += 1
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_general(h, beta):
    n = len(h)
    u = 1.0 - h
    raw = np.correlate(h, u, mode='full')
    M = float(np.max(raw))
    z = beta * (raw - M)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    w = e / float(np.sum(e))
    G = np.zeros(n)
    active_idx = np.where(w > 1e-8)[0]
    for r in active_idx:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:tau+mm] += wr * u[:mm]
                G[:mm] -= wr * h[tau:tau+mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * u[d:]
                G[d:d+mm] -= wr * h[:mm]
    return G

def _ls_eval(S0, T, Q, dx, a):
    val = S0 + a * T - a * a * Q
    return float(np.max(val)) * dx

def _line_search_general(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo, a_hi = -radius, radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = _ls_eval(S0, T, Q, dx, gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gratio = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gratio * (hi - lo)
    c2 = lo + gratio * (hi - lo)
    f1 = _ls_eval(S0, T, Q, dx, c1)
    f2 = _ls_eval(S0, T, Q, dx, c2)
    for _ in range(35):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gratio * (hi - lo)
            f1 = _ls_eval(S0, T, Q, dx, c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gratio * (hi - lo)
            f2 = _ls_eval(S0, T, Q, dx, c2)
    opt_t = (lo + hi) * 0.5
    opt_v = _ls_eval(S0, T, Q, dx, opt_t)
    for t in [a_lo, a_hi, 0.0]:
        ft = _ls_eval(S0, T, Q, dx, t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _general_opt(h, deadline, beta_init=15.0, beta_end=80.0, max_iter=200, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(555)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        frac = float(it) / float(max_iter)
        beta = beta_init + (beta_end - beta_init) * frac
        G = _grad_general(h, beta)
        G -= np.mean(G)
        gnorm = np.linalg.norm(G)
        if gnorm < 1e-12:
            break
        d = G / gnorm
        d -= np.mean(d)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = d
        extra_dirs = []
        for blk in [4, 8]:
            if n % blk == 0:
                nblk = n // blk
                gb = G.reshape(nblk, blk).mean(axis=1)
                db = np.tile(gb, (1, blk)).reshape(n).copy()
                db -= np.mean(db)
                dbn = np.linalg.norm(db)
                if dbn > 1e-12:
                    extra_dirs.append(db / dbn)
        x_arr = np.arange(n, dtype=float)
        for freq in [1, 2, 3]:
            ph = rng.uniform(0, 2 * np.pi)
            ds = np.sin(2.0 * np.pi * freq * x_arr / n + ph)
            ds -= np.mean(ds)
            dsn = np.linalg.norm(ds)
            if dsn > 1e-12:
                extra_dirs.append(ds / dsn)
        all_dirs = [best_d] + extra_dirs[:4]
        for dd in all_dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search_general(h, dd, radius)
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha
                best_d = dd
        if best_d is not None and best_alpha != 0.0 and best_val < cur_c5 - 1e-14:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
            else:
                stall += 1
                radius = max(0.04, radius * 0.75)
        else:
            stall += 1
            radius = max(0.04, radius * 0.78)
    return best_h, best_c5

def _binary_local(h, deadline, max_iter=30000, T_init=0.003):
    n = len(h)
    h_bin = np.zeros(n)
    idx = np.argsort(h)[-n // 2:]
    h_bin[idx] = 1.0
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300 or len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _random_anti_x(n, rng):
    m = n // 2
    return rng.uniform(0.0, 1.0, m)

def _perturb_anti(x, amp, rng):
    n = len(x)
    x_p = x + amp * rng.normal(0, 1, n)
    return np.clip(x_p, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Anti-palindromic L-BFGS-B sweep
    d1 = t0 + budget_s * 0.20
    n_candidates = list(range(max(60, n0 - 30), min(n0 + 30, 260), 4))
    n_candidates = [n for n in n_candidates if n % 2 == 0]
    for n_try in n_candidates:
        if time.time() > d1:
            break
        h_r = _resample(best_h, n_try)
        x_start = _to_anti_x(h_r)
        x_best_local = x_start.copy()
        c5_best_local = _exact_c5(_from_anti_x(x_start))
        for beta in [8, 15, 30, 60, 100, 200]:
            if time.time() > d1:
                break
            x_opt, c5_opt = _lbfgs_anti(x_best_local, beta, max_iter=200, gtol=1e-9)
            if c5_opt < c5_best_local:
                c5_best_local = c5_opt
                x_best_local = x_opt.copy()
        if c5_best_local < best_c5:
            best_c5 = c5_best_local
            best_h = _from_anti_x(x_best_local).copy()

    # Phase 2: General optimization on best solution
    d2 = t0 + budget_s * 0.45
    h_g = _project(best_h.copy(), len(best_h) / 2.0)
    h_g, c5_g = _general_opt(h_g, d2, beta_init=10.0, beta_end=60.0, max_iter=150, radius_init=0.12)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    # Phase 3: Binary optimization
    d3 = t0 + budget_s * 0.65
    h_b, c5_b = _binary_local(best_h, d3, max_iter=40000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 3b: Polish binary result with continuous opt
    if time.time() < d3 - 5:
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        h_pol, c5_pol = _general_opt(h_pol, d3, beta_init=20.0, beta_end=80.0, max_iter=80, radius_init=0.08)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Anti-palindromic restarts on best n
    d4 = t0 + budget_s * 0.80
    n_best = len(best_h)
    m_best = n_best // 2
    x_best = _to_anti_x(best_h)
    c5_x_best = best_c5
    n_restarts = 0
    scale = 0.05
    while time.time() < d4 - 5 and n_restarts < 150:
        n_restarts += 1
        x_p = _perturb_anti(x_best, scale, rng)
        for beta in [30, 80, 200]:
            if time.time() > d4 - 5:
                break
            x_opt, c5_opt = _lbfgs_anti(x_p, beta, max_iter=150, gtol=1e-9)
            if c5_opt < c5_x_best:
                c5_x_best = c5_opt
                x_best = x_opt.copy()
                scale = max(0.02, scale * 0.85)
                break
        else:
            scale = min(0.15, scale * 1.1)
    if c5_x_best < best_c5:
        best_c5 = c5_x_best
        best_h = _from_anti_x(x_best).copy()

    # Phase 5: General polishing on anti-palindromic result
    if time.time() < deadline - 20:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _general_opt(h_f, deadline - 10, beta_init=30.0, beta_end=120.0, max_iter=100, radius_init=0.06)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final: try a few nearby n values
    if deadline - time.time() > 30:
        n_final = len(best_h)
        for dn in [-4, 4, -8, 8]:
            if deadline - time.time() < 10:
                break
            n_try = n_final + dn
            if n_try < 60 or n_try > 280 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            x_r = _to_anti_x(h_r)
            for beta in [40, 100]:
                if deadline - time.time() < 8:
                    break
                x_opt, c5_opt = _lbfgs_anti(x_r, beta, max_iter=150, gtol=1e-9)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = _from_anti_x(x_opt).copy()
                    n_final = n_try

    # Final polish
    if deadline - time.time() > 5:
        h_last = _project(best_h.copy(), len(best_h) / 2.0)
        h_last, c5_last = _general_opt(h_last, deadline - 2, beta_init=50.0, beta_end=150.0, max_iter=50, radius_init=0.04)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
