def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    i = np.arange(n)[:, None]  # (n, 1)
    m = np.arange(L)[None, :]  # (1, L)
    idx1 = i - m.T + n - 1  # (L, n)
    idx2 = i + m.T - n + 1  # (L, n)
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    term2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    return term1 - term2

def _exact_c5(h):
    """Compute C5(h) = max_k corr(h, 1-h)_k * 2/n"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    """Compute gradient matrix G where G[m,i] = dR_m/dh_i."""
    n = len(h)
    L = 2 * n - 1

    B = np.correlate(d, d, mode='full')

    t_hi = t_cap

    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0) * dx)

    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1) * dx
    best_i = int(np.argmin(vals))
    
    # Golden section around best grid point

        f1 = float(np.max(R0 + c1 * A - c1*c1 * B) * dx)
        f2 = float(np.max(R0 + c2 * A - c2*c2 * B) * dx)
        for _ in range(n_gs):
            if hi - lo < 1e-13:

            if f1 < f2:
                hi, c2, f2 = c2, c1, f1

def _lp_direction(h, G, active_rows, method='HiGHS'):
    """Solve LP to find descent direction. Returns direction or None."""
    n = len(h)
    k = len(active_rows)
    # Variables: sp(n), sn(n), beta(1) → 2n+1
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0  # minimize -beta
    
    # Inequality constraints: G[r,:] @ d + beta <= 0 for each active row
    # d = sp - sn
    # G[r,:,n] @ sp - G[r,:n] @ sn + beta <= 0
    A_ub = np.zeros((k, nv))
    b_ub = np.zeros(k)
    for j, r in enumerate(active_rows):
        A_ub[j, :n] = G[r, :]
        A_ub[j, n:2*n] = -G[r, :]
        A_ub[j, 2*n] = 1.0
    # b_ub is all zeros
    
    # Equality constraints: sum(sp) - sum(sn) = 0, sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    # Bounds
    bounds = [(0.0, 1.0)] * (2*n) + [(-1e6, 1e6)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
        if not res.success:
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
            except:
                return None
        if not res.success:
            return None
        d = res.x[:n] - res.x[n:2*n]
        beta = res.x[2*n]
        if beta <= 1e-10 or float(np.max(np.abs(d))) < 1e-10:
            return None
        # Normalize to unit L1
        l1 = float(np.sum(np.abs(d)))
        if l1 < 1e-12:
            return None
        d = d / l1
        return d
    except:
        return None

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v.copy(), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    i = np.arange(n)
    m = np.arange(L)
    idx1 = i[None, :] - m[:, None] + (n - 1)
    idx2 = i[None, :] + m[:, None] - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    term2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return term1 - term2

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iters=50000, seed=777):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.004
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline, max_rounds=20):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    parent = np.array(initial_h_values, dtype=float)
    n0 = len(parent)
    h_init = _project(parent.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP descent on parent resolution (32%)
    d1 = t0 + budget_s * 0.32
    h1, c5_1 = _lp_descent(h_init, d1, max_its=5000, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Resolution exploration (33%)
    d2 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -28, 28]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n_curr + dn
        if n_try < 100 or n_try > 300 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.022)
        h_r, c5_r = _lp_descent(h_r, sub_dl, max_its=2000, seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Binary diversification (18%)
    d3 = t0 + budget_s * 0.83
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d3, time.time() + budget_s * 0.07)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iters=60000, seed=777)
        sub_dl2 = min(d3, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl2, max_rounds=15)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.05)
            h_pol, c5_pol = _lp_descent(h_pol, sub_dl3, max_its=2000, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Intensive final polish (14%)
    d4 = deadline - 5.0
    n_curr = len(best_h)
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.07)
        h4, c5_4 = _lp_descent(h4, sub_dl, max_its=5000, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        # Try a few nearby resolutions
        n_curr = len(best_h)
        for dn in [-4, 4, -8, 8]:
            if time.time() > d4:
                break
            n_try = n_curr + dn
            if n_try < 100 or n_try > 300 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_descent(h_r, sub_dl2, max_its=1500, seed=444 + dn)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final: verify and return
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
