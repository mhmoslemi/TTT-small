import numpy as np
import time
import math

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_corr(h)) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, s, r):
    """Exact line search along feasible edge h + alpha*s, alpha in [0, r].
    s must satisfy: sum(s)=0, -h_i/r <= s_i <= (1-h_i)/r for the edge to be feasible."""
    n = len(h)
    dx = 2.0 / n
    raw0 = _raw_corr(h)
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    
    # Find safe range
    a_lo = 0.0
    a_hi = r
    pos = s > 1e-15
    neg = s < -1e-15
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / s[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-s[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(raw0) * dx)
    
    # Build candidate alphas
    cands = [a_lo, a_hi, (a_lo + a_hi) / 2.0]
    for k in range(len(raw0)):
        if Q[k] > 1e-14 and T[k] < 0:
            av = -T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Add a grid
    ng = min(50, max(20, int((a_hi - a_lo) * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    
    # Evaluate
    best_val = float('inf')
    best_a = a_lo
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(raw0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.05 * (a_hi - a_lo))
    hi_r = min(a_hi, best_a + 0.05 * (a_hi - a_lo))
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(raw0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-11:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(raw0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(raw0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(raw0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

def _active_gradients(h, k_top=10):
    """Compute gradients for top-k active constraints."""
    n = len(h)
    raw = _raw_corr(h)
    mx = float(np.max(raw))
    # Get top indices
    thresh = mx - 0.01  # within 1% of max
    active = np.where(raw > thresh)[0]
    if len(active) > k_top:
        idx = np.argsort(-raw)[[:k_top]]
        active = idx
    grads = []
    for t in active:
        xi = t - (n - 1)
        g = np.zeros(n)
        if xi >= 0:
            L = n - xi
            if L > 0:
                g[:L] += (1.0 - h[xi:xi+L])
                g[xi:xi+L] -= h[:L]
        else:
            s = -xi
            L = n - s
            if L > 0:
                g[s:s+L] += (1.0 - h[:L])
                g[:L] -= h[s:s+L]
        grads.append(g)
    return grads, active, raw, mx

def _lp_direction(h, grads, r=0.1):
    """Solve LP to find direction that decreases all active constraints."""
    n = len(h)
    try:
        import cvxpy as cp
        s = cp.Variable(n)
        tau = cp.Variable()
        constraints = [cp.sum(s) == 0, s >= -r * h, s <= r * (1 - h), tau >= 0]
        for g in grads:
            constraints.append(g @ s + tau <= 0)
        prob = cp.Problem(cp.Maximize(tau), constraints)
        # Try solvers
        for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate') and s.value is not None:
                    sv = np.array(s.value).flatten()
                    sv = np.clip(sv, -r, r)
                    sv -= np.mean(sv)
                    # Enforce bounds
                    sv = np.clip(sv, -r*h + 1e-10, r*(1-h) - 1e-10)
                    sv -= np.mean(sv)
                    return sv
            except Exception:
                continue
    except ImportError:
        pass
    return None

def _aggregate_grad(h, grads):
    """Fallback: aggregate gradient direction."""
    g = np.zeros(len(h))
    for gr in grads:
        g += gr
    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-12:
        g /= nrm
    return g

def _pair_opt(h, deadline, max_it=20000):
    """Pair swap optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        s = 0.5 * d
        a, c5t = _line_search(h, s, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * s
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_sa(hb, deadline, mx=10000, Ti=0.001):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        ones[i], zeros[j] = zeros[j], ones[i]
        cur = cur.copy()
        cur[ones[i]] = 0.0
        cur[zeros[j]] = 1.0
        nc = _exact_c5(cur)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cc = nc
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.999
        else:
            cur[ones[i]], cur[zeros[j]] = cur[zeros[j]], cur[ones[i]]
            ones[i], zeros[j] = zeros[j], ones[i]
            stall += 1
    return bh, bc

def _gen_binary_stars(rng, n):
    """Generate diverse binary starting points."""
    stars = []
    # Random binary
    h = (rng.random(n) < 0.5).astype(float)
    h = _project(h, n // 2)
    stars.append(h)
    
    # Periodic patterns
    for period in [7, 11, 13, 17, 23]:
        h = np.zeros(n)
        fill = n // 2
        # Place ones in a pattern
        positions = set()
        while len(positions) < fill:
            start = int(rng.integers(0, n))
            offset = int(rng.integers(0, period))
            for k in range(offset, n, period):
                positions.add(k)
                if len(positions) >= fill:
                    break
        for p in list(positions)[:fill]:
            h[p] = 1.0
        stars.append(h.copy())
    
    # Low-discrepancy (van der Corput-like)
    for base in [2, 3, 5]:
        h = np.zeros(n)
        fill = n // 2
        vals = np.zeros(n)
        for i in range(n):
            v = 0.0
            num = i + 1
            denom = 1.0
            while num > 0:
                denom *= base
                num, rem = divmod(num, base)
                v += rem / denom
            vals[i] = v
        top_idx = np.argsort(-vals)[:fill]
        h[top_idx] = 1.0
        stars.append(h)
    
    return stars

def _lp_dir(h, grads, r):

        s = cp.Variable(n)

        cons = [cp.sum(s) == 0, tau >= 0]

                cons.append(s[i] >= -r * h[i])

            cons.append(g @ s + tau <= 0)
        prob = cp.Problem(cp.Maximize(tau), cons)
        for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:

            except Exception:

    except Exception:
        pass
    return None

def _gen_stars(rng, n):
    stars = []
    # Random binary
    h = (rng.random(n) < 0.5).astype(float)
    stars.append(_proj(h, n//2))
    # Periodic
    for p in [5, 7, 11, 13, 17, 23, 31]:
        h = np.zeros(n)
        pos = set()
        while len(pos) < n//2:
            off = int(rng.integers(0, p))
            for k in range(off, n, p):
                pos.add(k)
                if len(pos) >= n//2:
                    break
        for pp in list(pos)[:n//2]:
            h[pp] = 1.0
        stars.append(h.copy())
    # Low-discrepancy
    for base in [2, 3, 4]:
        vals = np.zeros(n)
        for i in range(n):
            v = 0.0
            num = i + 1
            denom = 1.0
            while num > 0:
                denom *= base
                num, rem = divmod(num, base)
                v += rem / denom
            vals[i] = v
        h = np.zeros(n)
        h[np.argsort(-vals)[:n//2]] = 1.0
        stars.append(h)
    # Smoothed random
    for _ in range(3):
        h = rng.random(n)
        h = _proj(h, n//2)
        stars.append(h)
    return stars

    tried = set()
    for dn in [-8, -4, -6, 4, 6, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > dD:
            break
        n_try = ne + dn
        if n_try < 80 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        cr = _c5(hr)
        if cr < best_c + 0.006:
            sd = min(dD, time.time() + budget_s * 0.02)
            hr, cr = _local_lp(hr, sd, r0=0.12, stall_lim=30)
            if cr < best_c + 0.003 and time.time() < dD - 2:
                hb = _to_bin(hr)
                hb, cb2 = _bin_sa(hb, min(dD, time.time()+budget_s*0.006), mx=4000, Ti=0.0008)
                if cb2 < cr:
                    cr = cb2
                    hr = hb
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                ne = n_try

    # Phase E: Diverse multi-start
    dE = t0 + budget_s * 0.85
    n_main = len(best_h)
    stars = _gen_stars(rng, n_main)
    for star in stars:
        if time.time() > dE:
            break
        ns = len(star)
        sc = _c5(star)
        if sc < best_c + 0.008:
            sd = min(dE, time.time() + budget_s * 0.012)
            hs, cs = _local_lp(_proj(star, ns/2.0), sd, r0=0.10, stall_lim=25)
            if cs < best_c:
                best_c = cs
                best_h = hs.copy()

    # Phase F: Intensive final optimization
    dF = deadline
    nf = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > dF - 3:
            break
        hf = _proj(best_h.copy(), nf/2.0)
        sd = min(dF, time.time() + budget_s * 0.04)
        hf, cf = _local_lp(hf, sd, r0=r_init, stall_lim=45)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
            nf = len(best_h)
        if time.time() < dF - 3:
            hf = _proj(best_h.copy(), len(best_h)/2.0)
            sd = min(dF, time.time() + budget_s * 0.02)
            hf, cf = _pair_opt(hf, sd, max_it=8000)
            if cf < best_c:
                best_c = cf
                best_h = hf.copy()

    # Final polish
    if deadline - time.time() > 2:
        hf = _proj(best_h.copy(), len(best_h)/2.0)
        hf, cf = _local_lp(hf, deadline, r0=0.02, stall_lim=25)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 1.5:
        hf = _proj(best_h.copy(), len(best_h)/2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _proj(best_h, len(best_h)/2.0)
    final_c = _c5(best_h)
    return best_h, final_c, len(best_h)
