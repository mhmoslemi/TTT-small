    n = len(v)

def _greedy_binary(h, deadline, max_stall=200):

    n = len(h)
    dx = 2.0 / n
    h = _to_binary(h.copy())
    S = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(S))

    zeros = list(np.where(h <= 0.5)[0])
    stall = 0
    beta = 30.0
    
    while time.time() < deadline and stall < max_stall:
        max_S = float(np.max(S))
        v = beta * (S - max_S)
        v = np.clip(v, -80, 0)
        w = np.exp(v)
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.zeros_like(S)
            w[np.argmax(S)] = 1.0
        else:
            w /= ws
        
        # Find the weight index corresponding to each shift
        # S has indices 0..2n-2, shift k = idx - (n-1)
        
        # Compute A[i] = sum_k w[k_idx] * (h[i+k] - h[i-k]) for h[i]=1
        # Compute B[j] = sum_k w[k_idx] * (h[j+k] + h[j-k]) for h[j]=0
        # (out of range terms are 0)
        
        # For efficiency, only use significant weights
        w_thresh = w.max() * 0.005
        active_idx = np.where(w > w_thresh)[0]
        
        A = np.zeros(n)
        B = np.zeros(n)
        
        for ki in active_idx:
            k = ki - (n - 1)
            wi = w[ki]
            if k >= 0:
                m = n - k
                A[:m] += wi * h[k:k+m]    # h[i+k] for i in 0..m-1
                A[k:] -= wi * h[:m]        # h[i-k] for i in k..n-1
                B[:m] += wi * h[k:k+m]
                B[k:] += wi * h[:m]
            else:
                s = -k
                m = n - s
                A[s:] += wi * h[:m]        # h[i+k] = h[i-s] for i in s..n-1
                A[:m] -= wi * h[s:s+m]     # h[i-k] = h[i+s] for i in 0..m-1
                B[s:] += wi * h[:m]
                B[:m] += wi * h[s:s+m]
        
        # Find best i (h[i]=1) with minimum A[i]
        # Find best j (h[j]=0) with maximum B[j]
        best_i = -1
        best_A = float('inf')
        for i in ones:
            if A[i] < best_A:
                best_A = A[i]
                best_i = i
        
        best_j = -1
        best_B = -float('inf')
        for j in zeros:
            if B[j] > best_B:
                best_B = B[j]
                best_j = j
        
        if best_i < 0 or best_j < 0:
            break
        
        # Perform swap and check
        h[best_i] = 0.0
        h[best_j] = 1.0
        new_S = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_S))
        
        if new_max < cur_max - 1e-12:
            cur_max = new_max
            S = new_S
            ones.remove(best_i)
            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
            stall = 0
        else:
            # Revert
            h[best_i] = 1.0
            h[best_j] = 0.0
            stall += 1
            beta *= 1.2
            if beta > 500:
                # Try random swaps
                if stall % 10 == 0:
                    ii = ones[np.random.randint(len(ones))]
                    jj = zeros[np.random.randint(len(zeros))]
                    h[ii] = 0.0
                    h[jj] = 1.0
                    new_S = np.correlate(h, 1.0 - h, mode='full')

                        cur_max = new_max
                        S = new_S
                        ones.remove(ii)
                        zeros.remove(jj)
                        ones.append(jj)
                        zeros.append(ii)
                        stall //= 2
                        beta = 30.0
                    else:
                        h[ii] = 1.0
                        h[jj] = 0.0
    
    return h, cur_max * dx

def _line_search_c5(h, d, dx, radius):

def _subgradient(h, beta):
    n = len(h)
    S = _corr_vec(h)
    sc = S * (2.0 / n)
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for ki in range(len(S)):
        wl = w[ki]
        if wl < 1e-14:
            continue
        l = ki - (n - 1)
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l+m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s+m]
    g -= np.mean(g)
    return -g

        if time.time() > d4 - 3:
            break
        h4 = _project(best_h.copy(), nc / 2.0)
        sd = min(d4, time.time() + budget_s * 0.05)
        h4, c4 = _continuous_opt(h4, sd, r0=r_init, max_iter=200)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
    
    # Phase 5: Intensive pair optimization (10%)
    d5 = deadline
    hf = _project(best_h.copy(), len(best_h) / 2.0)
    hf, cf = _pair_opt(hf, min(d5, time.time() + budget_s * 0.06), max_it=15000)
    if cf < best_c:
        best_c = cf
        best_h = hf.copy()
    
    # Final: quick greedy on best
    if deadline - time.time() > 3:
        hg, cg = _greedy_binary(best_h, deadline, max_stall=100, beta_start=40.0)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()
    
    # Final continuous polish
    if deadline - time.time() > 2:
        hf2 = _project(best_h.copy(), len(best_h) / 2.0)
        hf2, cf2 = _continuous_opt(hf2, deadline, r0=0.03, max_iter=80)
        if cf2 < best_c:
            best_c = cf2
            best_h = hf2.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
