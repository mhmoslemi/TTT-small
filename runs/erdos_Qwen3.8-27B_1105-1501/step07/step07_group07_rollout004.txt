def _lp_direction(h, scores, G, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    top_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

        hi = min(radius, 1.0 - h[j])

    bounds.append((None, None))

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = _raw_scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b = c2
            c2 = c1
            f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b = c1
            c1 = c2
            f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    candidates = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

    dx = 2.0 / n

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _raw_scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project_sum(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.04, 0.25)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
                top_k = top_k_init
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 800:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=6000, radius=0.2, seed=111):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project_sum(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
        h0 = _project_sum(h0, n0 / 2.0)

    n0 = len(h0)
    n0 = (n0 // 2) * 2
    if n0 != len(h0):
        h0 = _resample(h0, n0)
        n0 = len(h0)

    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LSE annealed SLSQP (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _lse_optimize(h_init, d1, tau_start=0.04, tau_ratio=0.55, n_tau=8, maxiter=60)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP focused (10%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _lp_loop(_project_sum(best_h.copy(), len(best_h) / 2.0), d2,
                         radius_init=0.10, top_k_init=4, max_top_k=10, stall_limit=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA + pair sweep (15%)
    d3 = t0 + budget_s * 0.40
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003, seed=31337)
    sub_dl2 = min(d3, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=35)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    sub_dl3 = min(d3, time.time() + budget_s * 0.03)
    h_pol = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_pol, c5_pol = _lp_loop(h_pol, sub_dl3, radius_init=0.05, top_k_init=3, max_top_k=8, stall_limit=40)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.60
    n_best = len(best_h)
    n_offsets = list(range(-30, 31, 6))
    tried_ns = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _lse_optimize(h_r, sub_dl, tau_start=0.03, tau_ratio=0.5, n_tau=4, maxiter=50)
        if c5_r > c5_r:
            pass
        if time.time() < d4 - 2 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d4, time.time() + budget_s * 0.010)
            h_r, c5_r = _lp_loop(_project_sum(h_r.copy(), n_try / 2.0), sub_dl2,
                                 radius_init=0.08, top_k_init=4, max_top_k=9, stall_limit=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 5: Intensive optimization (20%)
    d5 = t0 + budget_s * 0.80
    n_cur = len(best_h)
    # LSE refinement
    sub_dl = min(d5, time.time() + budget_s * 0.06)
    h5 = _project_sum(best_h.copy(), n_cur / 2.0)
    h5, c5_5 = _lse_optimize(h5, sub_dl, tau_start=0.02, tau_ratio=0.5, n_tau=5, maxiter=70)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # LP tight
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=7, stall_limit=80)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Pair swap
    if time.time() < d5:
        n_cur = len(best_h)
        h5 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h5, c5_5 = _pair_swap_loop(h5, sub_dl, n_pairs=8000, radius=0.18, seed=222)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    # Binary refinement
    if time.time() < d5:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.002, seed=777)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Final LP on best
    if time.time() < d5:
        h5 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.02, top_k_init=3, max_top_k=6, stall_limit=60)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping (10%)
    d6 = t0 + budget_s * 0.92
    scale = 0.03
    n_restarts = 0
    while time.time() < d6 and n_restarts < 300:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d6, time.time() + budget_s * 0.005)
        h_t, c5_t = _lse_optimize(h_pert, sub_dl, tau_start=0.02, tau_ratio=0.5, n_tau=3, maxiter=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.05)

    # Phase 7: Final polish (8%)
    d7 = deadline
    n_final = len(best_h)
    for dn in [0, -6, 6, -12, 12, -4, 4]:
        if time.time() > d7 - 5:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.04, top_k_init=3, max_top_k=7, stall_limit=30)
        if time.time() < d7 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d7, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.001, seed=444)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Ultimate LP
    h_final = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_final, c5_final = _lp_loop(h_final, d7, radius_init=0.02, top_k_init=3, max_top_k=6, stall_limit=50)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    # Binary pair sweep
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, d7, max_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h.copy(), len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
