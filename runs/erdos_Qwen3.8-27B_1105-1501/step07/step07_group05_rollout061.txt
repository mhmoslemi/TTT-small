def _lp_direction(h, active_indices, r, n_active=None):
    n = len(h)
    if n_active is not None:
        active = active_indices[:n_active]
    else:
        active = active_indices
    
    # Compute gradients for active lags
    # grad[t][j] = d(raw[t])/d(h[j])
    grads = []
    for t in active:
        lag = t - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            # d(Out[t])/dh[j] = (1-h[j-lag]) * [j >= lag] - h[j+lag] * [j+lag <= n-1]
            m = n - lag
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
        else:
            s_abs = -lag
            m = n - s_abs
            g[s_abs:] += (1.0 - h[:m])
            g[:m] -= h[s_abs:]
        grads.append(g)
    
    A = np.array(grads)  # shape (K, n)
    
    s_var = cp.Variable(n)
    tau = cp.Variable()
    
    lo = np.minimum(r, h)
    hi = np.minimum(r, 1.0 - h)
    
    constraints = [
        cp.sum(s_var) == 0,
        s_var >= -lo,
        s_var <= hi,
        A @ s_var <= tau,
    ]
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    
    try:
        prob.solve(solver=cp.CPULEXECUTION or 'CLARABEL' or whatever)
    except:
        return None
    
    if prob.status != 'optimal':
        return None
    
    s_opt = s_var.value
    tau_opt = tau.value
    
    if tau_opt > -1e-8:
        return None  # No improvement possible
    
    return s_opt

def _project(v, target):

        s_var >= -lo,
        s_var <= hi,
        A @ s_var <= tau,
        tau >= -mx * r,  # prevent trivially large negative tau
    ]
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CVXOPT, cp.SCIP, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            break
        except:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    s_opt = np.asarray(s_var.value, dtype=float)
    tau_opt = float(tau.value)
    
    if tau_opt > -1e-10:
        return None
    
    # Ensure feasibility
    if abs(float(np.sum(s_opt))) > 1e-6:
        s_opt -= np.mean(s_opt)
    s_opt = np.clip(s_opt, -lo - 1e-9, hi + 1e-9)
    
    return s_opt

def _smooth_grad(h, beta):
    """Softmax-weighted gradient for C5 objective."""
    n = len(h)
    dx = 2.0 / n
    S = _raw_corr(h)
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s_ = -l
            m = n - s_
            if m > 0:
                g[s_:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s_:s_ + m]
    g -= np.mean(g)
    return -g

def _pair_swap(h, deadline, max_it=50000):
    """Binary pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(4242)
    stall = 0
    
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if h[i] * (1 - h[j]) < 1e-4 and h[j] * (1 - h[i]) < 1e-4:
            stall += 1
            continue
        
        # Direction: move mass from i to j or j to i
        if h[i] > h[j]:
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            max_a = min(h[i], 1.0 - h[j])
        else:
            d = np.zeros(n)
            d[j] = -1.0
            d[i] = 1.0
            max_a = min(h[j], 1.0 - h[i])
        
        a, c5t = _line_search(h, d, dx, max_a)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c

def _to_bin(h, n):

def _resample(h, n_new):

    xn = (np.arange(n_new) + 0.5) / n_new

    active_idx = np.where(raw >= thresh)[0]
    if len(active_idx) > n_top:
        active_idx = active_idx[np.argsort(-raw[active_idx])[:n_top]]
    if len(active_idx) < 1:
        return None
    K = len(active_idx)
    A = np.zeros((K, n))
    for ii, t in enumerate(active_idx):
        lag = t - (n - 1)
        if lag >= 0:
            m = n - lag
            A[ii, lag:] += (1.0 - h[:m])
            A[ii, :m] -= h[lag:]
        else:
            s_abs = -lag
            m = n - s_abs
            A[ii, s_abs:] += (1.0 - h[:m])
            A[ii, :m] -= h[s_abs:]
    s_var = cp.Variable(n)
    tau = cp.Variable()
    lo = np.minimum(r, h)
    hi = np.minimum(r, 1.0 - h)
    constraints = [
        cp.sum(s_var) == 0,
        s_var >= -lo,
        s_var <= hi,
        A @ s_var <= tau,
        tau >= -mx * r * 0.5,
    ]
    prob = cp.Problem(cp.Minimize(tau), constraints)
    solved = False
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None
    s_opt = np.asarray(s_var.value, dtype=float)
    tau_opt = float(tau.value)
    if tau_opt > -1e-10:
        return None
    s_opt -= np.mean(s_opt)
    s_opt = np.clip(s_opt, -(lo + 1e-9), hi + 1e-9)
    nrm = np.linalg.norm(s_opt)
    if nrm < 1e-10:
        return None
    return s_opt

def _local_opt_lp(h, deadline, r0=0.15, stall_lim=50, use_lp=True):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    it = 0
    lp_configs = [(0.0, 5), (0.0, 10), (0.05, 15), (0.1, 25), (0.2, 40), (0.0, 30)]
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.15:
            break
        best_new = cur
        best_a = 0.0
        best_d = None
        it += 1
        if use_lp:
            for ci, (af, nt) in enumerate(lp_configs):
                if time.time() > deadline:
                    break
                d = _lp_direction(h, n, rad, nt, af)
                if d is None:
                    continue
                d -= np.mean(d)
                nrm = np.linalg.norm(d)
                if nrm < 1e-10:
                    continue
                d = d / nrm * rad
                a, c_try = _line_search(h, d, dx, rad)
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d
        g = _smooth_grad(h, 25.0)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            d = -g / nrm * rad
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.30)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.004:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp_ = _exact_c5(hp)
                if cp_ < cur - 1e-12:
                    h = hp
                    cur = cp_
                    stall = 0
                    rad = 0.10
                elif stall % 40 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_swap(h, deadline, max_it=50000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(4242)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if h[i] * (1.0 - h[j]) < 1e-4 and h[j] * (1.0 - h[i]) < 1e-4:
            stall += 1
            continue
        if h[i] > h[j]:
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            max_a = min(h[i], 1.0 - h[j])
        else:
            d = np.zeros(n)
            d[j] = -1.0
            d[i] = 1.0
            max_a = min(h[j], 1.0 - h[i])
        if max_a < 1e-6:
            stall += 1
            continue
        a, c5t = _line_search(h, d, dx, max_a)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _bin_local(hb, deadline, mx=20000, Ti=0.0015):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-guided local opt (30%)
    d1 = t0 + budget_s * 0.30
    h1, c1 = _local_opt_lp(h_init, d1, r0=0.15, stall_lim=60, use_lp=True)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair swap (12%)
    d2 = t0 + budget_s * 0.42
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_swap(h2, d2, max_it=35000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary exploration (18%)
    d3 = t0 + budget_s * 0.60
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hb, cb = _bin_local(hbin, d3, mx=12000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < d3:
        hp = _project(hb.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _local_opt_lp(hp, sd, r0=0.08, stall_lim=35, use_lp=True)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-n exploration (22%)
    d4 = t0 + budget_s * 0.82
    nc = len(best_h)
    offsets = [-8, 8, -4, 4, -12, 12, -6, 6, -16, 16, -10, 10, -20, 20, -14, 14, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 250 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.010)
        hr, cr = _local_opt_lp(hr, sd, r0=0.12, stall_lim=30, use_lp=True)
        if cr < best_c + 0.003 and time.time() < d4 - 4:
            hb2 = _to_bin(hr)
            sd2 = min(d4, time.time() + budget_s * 0.005)
            hb2, cb2 = _bin_local(hb2, sd2, mx=4000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb2.copy()
        if cr < best_c + 0.001 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.007)
            hr, cr = _local_opt_lp(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, use_lp=True)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Intensive LP optimization (18%)
    d5 = deadline
    nc = len(best_h)
    for r_init in [0.12, 0.06, 0.03, 0.02, 0.01]:
        if time.time() > d5 - 2:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.035)
        h5, c5v = _local_opt_lp(h5, sd, r0=r_init, stall_lim=50, use_lp=True)
        if c5v < best_c:
            best_c = c5v
            best_h = h5.copy()
        if time.time() < d5 - 4:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.015)
            h5, c5v = _pair_swap(h5, sd, max_it=8000)
            if c5v < best_c:
                best_c = c5v
                best_h = h5.copy()

    # Final polish
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt_lp(hf, deadline, r0=0.015, stall_lim=25, use_lp=True)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 1.5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_swap(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
