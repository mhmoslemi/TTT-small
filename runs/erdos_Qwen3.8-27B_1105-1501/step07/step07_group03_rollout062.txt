```python
import numpy as np
from scipy.optimize import linprog
import time

def _compute_grad_matrix(h, active_lags):
    """Compute dc[r]/dh[k] for each active lag r and each coordinate k."""
    n = len(h)
    grads = []
    for r in active_lags:
        g = np.zeros(n)
        shift = r - (n - 1)
        # Term 1: h[k] is first factor, (1-h)[k+shift] is second
        # Valid: 0 <= k < n and 0 <= k + shift < n
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)  # k + shift < n => k < n - shift
        if hi1 > lo1:
            # g[lo1:hi1] += 1 - h[lo1+shift : hi1+shift]
            g[lo1:hi1] += 1.0 - h[lo1 + shift : hi1 + shift]
        # Term 2: h[k-shift] is first factor, h[k] is second factor (contribution -h[k-shift])
        # Valid: 0 <= k < n and 0 <= k - shift < n
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)  # k - shift < n => k < n + shift
        if hi2 > lo2:
            # g[lo2:hi2] -= h[lo2 - shift : hi2 - shift]
            g[lo2:hi2] -= h[lo2 - shift : hi2 - shift]
        grads.append(g)
    return np.array(grads)  # shape (K, n)
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad_for_lag(h, r):
    """Compute d(c[r])/d(h[k]) for all k, where c = correlate(h, 1-h, 'full')."""
    n = len(h)
    g = np.zeros(n)
    shift = r - (n - 1)
    # Term 1: h[k] is first factor, (1-h)[k+shift] is second factor
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    if hi1 > lo1:
        g[lo1:hi1] += 1.0 - h[lo1 + shift:hi1 + shift]
    # Term 2: h[k-shift] is first factor, h[k] is second factor
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return g

def _lp_step(h, K=30, delta=0.1):
    """Solve LP to find a descent direction considering top K lags."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    # Select top K lags
    top_k = np.argsort(-raw)[:K]
    # Compute gradients for these lags
    G = np.array([_compute_grad_for_lag(h, int(r)) for r in top_k])  # (K, n)
    c0 = raw[top_k]  # current values at top lags
    
    # LP: minimize t
    # vars: [d_0, ..., d_{n-1}, t]
    # obj: [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G @ d - t <= -c0
    # i.e., for each r: sum_k G[r,k]*d[k] - t <= -c0[r]
    A_ub = np.hstack([G, -np.ones((K, 1))])
    b_ub = -c0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -min(h, delta) <= d[k] <= min(1-h, delta), t >= 0
    lo_bound = np.maximum(-h, -delta)
    hi_bound = np.minimum(1.0 - h, delta)
    bounds = list(zip(lo_bound, hi_bound))
    bounds.append((0.0, None))
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
    
    if not result.success:
        return None, None
    d = result.x[:n]
    t = result.x[-1]
    return d, t

def _line_search(h, d, raw, dx, max_alpha=0.5):
    """Exact line search along h + alpha * d, ensuring 0 <= h <= 1 and sum = target."""
    n = len(h)
    target = n / 2.0
    # Find max alpha such that h + alpha*d is in [0,1]^n
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = max_alpha
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    alpha_max = max(alpha_max, 1e-10)
    
    # Binary search for best alpha
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    # Sample a grid
    grid = np.linspace(0, alpha_max, 40)
    for a in grid:
        h_try = h + a * d
        h_try = np.clip(h_try, 0, 1)
        h_try = _project(h_try, target)
        val = _exact_c5(h_try)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Refine with golden section
    lo = max(0, best_alpha - alpha_max / 40)
    hi = min(alpha_max, best_alpha + alpha_max / 40)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def _eval(a):
        h_try = h + a * d
        h_try = np.clip(h_try, 0, 1)
        h_try = _project(h_try, target)
        return _exact_c5(h_try)
    
    f1, f2 = _eval(c1), _eval(c2)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval(c2)
    opt = (lo + hi) / 2
    val_opt = _eval(opt)
    if val_opt < best_val:
        best_val = val_opt
        best_alpha = opt
    
    return best_alpha, best_val
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_at_lag(h, r):
    n = len(h)
    g = np.zeros(n)
    shift = r - (n - 1)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)
    if hi1 > lo1:
        g[lo1:hi1] += 1.0 - h[lo1 + shift:hi1 + shift]
    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 - shift:hi2 - shift]
    return g

def _lp_step(h, K, delta):

    top_k = np.argsort(-raw)[:K]
    G = np.array([_grad_at_lag(h, int(r)) for r in top_k])
    c0 = raw[top_k].copy()
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((K, 1))])
    b_ub = -c0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_b = np.maximum(-h, -delta)
    hi_b = np.minimum(1.0 - h, delta)
    bounds = list(zip(lo_b.tolist(), hi_b.tolist()))
    bounds.append((0.0, None))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x[:n]
```

We want to minimize `t` such that:

- `c[r](h + d) ≈ c0[r] + g_r · d` where `g_r = grad_at_lag(h, r)`

- Optionally: `|d[k]| <= delta` (trust region)

The LP is:
minimize t
subject to: `g_r · d - t <= -c0[r]` for each active r
           `sum(d) = 0`
           box constraints on d
           t >= 0 (or t >= some lower bound)

In scipy's linprog format:

- `A_ub`: K rows, each `[g_r, -1]`
- `b_ub = -c0`
- `A_eq = [[1, 1, ..., 1, 0]]`

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_at_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    shift = m - (n - 1)
    lo1 = max(0, shift)
    hi1 = min(n, shift + n)
    if hi1 > lo1:
        g[lo1:hi1] += 1.0 - h[lo1 - shift:hi1 - shift]
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
    return g

def _lp_step(h, K, delta, rng=None):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(-raw)[:K]
    if rng is not None and K > 10:
        # Add some diversity: replace some top-k with random from top 2K
        n_rand = min(K // 3, 2 * K - K)
        pool = np.argsort(-raw)[:int(2 * K)]
        fixed = top_k[:int(K - n_rand)]
        extra_pool = list(set(pool.tolist()) - set(fixed.tolist()))
        if extra_pool:
            extra = rng.choice(extra_pool, size=n_rand, replace=False)
            top_k = np.concatenate([fixed, extra])
    G = np.array([_grad_at_lag(h, int(r)) for r in top_k])
    c0 = raw[top_k].copy()
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((len(top_k), 1))])
    b_ub = -c0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_b = np.maximum(-h, -delta)
    hi_b = np.minimum(1.0 - h, delta)
    bounds = list(zip(lo_b.tolist(), hi_b.tolist()))
    bounds.append((0.0, None))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x[:n]

def _try_step(h, d):
    n = len(h)
    target = n / 2.0
    best_val = _exact_c5(h)
    best_h = h.copy()
    for a in [0.5, 0.75, 1.0]:
        h_try = h + a * d
        h_try = _project(h_try, target)
        val = _exact_c5(h_try)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_try.copy()
    return best_h, best_val

def _gradient_step(h, n, beta, max_alpha):
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    w = np.exp(np.clip(beta * shifted, -80, 0))
    w = w / (float(np.sum(w)) + 1e-30)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_at_lag(h, k)
    g -= np.mean(g)
    amax = max_alpha
    pos = g < -1e-12
    neg = g > 1e-12
    if np.any(pos):
        amax = min(amax, float(np.min((1.0 - h[pos]) / (-g[pos]))))
    if np.any(neg):
        amax = min(amax, float(np.min(h[neg] / g[neg])))
    amax = max(amax, 1e-7)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_val = cur_c5
    for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
        a = frac * amax
        h_try = _project(h + a * g, target)
        val = _exact_c5(h_try)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_try.copy()
    return best_h, best_val

def _lp_optimize(h, deadline, K_list, delta_list, rng=None):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    it = 0
    if rng is None:
        rng = np.random.default_rng(12345)
    while time.time() < deadline and stall < 100:
        it += 1
        K = K_list[it % len(K_list)]
        delta = delta_list[(it // len(K_list)) % len(delta_list)]
        d = _lp_step(h, K, delta, rng)
        if d is not None:
            new_h, new_c5 = _try_step(h, d)
            if new_c5 < cur_c5 - 1e-14:
                h = new_h
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                    continue
        # Fallback: gradient step
        h_new, c5_new = _gradient_step(h, n, 20.0, 0.05)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_iter=20000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(42)
    stall = 0
    T = 0.002
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            T *= 0.9994
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amplitude
    p += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_optimize(h_init, d1, [25, 35, 50], [0.06, 0.10, 0.14], rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient polish (10%)
    d2 = t0 + budget_s * 0.30
    n_curr = len(best_h)
    target = n_curr / 2.0
    stall = 0
    while time.time() < d2 and stall < 50:
        h_new, c5_new = _gradient_step(best_h, n_curr, 25.0, 0.04)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

    # Phase 3: Binary refinement (15%)
    d3 = t0 + budget_s * 0.45
    h_bin = _to_binary(best_h)
    h_bin, c5_b = _binary_local_search(h_bin, d3, max_iter=20000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d3:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        h_pol, c5_p = _lp_optimize(h_pol, d3, [30, 45], [0.04, 0.08], rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 4: Multi-resolution exploration (20%)
    d4 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    tried = {n_curr}
    offsets = [-4, 4, -2, 2, -6, 6, -8, 8, -12, 12, -16, 16]
    for dn in offsets:
        if time.time() > d4 - 3:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.014)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, [20, 35, 50], [0.05, 0.10], rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Intensive LP with varied params (20%)
    d5 = t0 + budget_s * 0.85
    h_curr = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _lp_optimize(h_curr, d5, [30, 50, 70, 100], [0.03, 0.05, 0.08, 0.12], rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Second binary round (7%)
    d6 = t0 + budget_s * 0.92
    if best_c5 > 0.380:
        h_bin2 = _to_binary(best_h)
        h_bin2, c5_b2 = _binary_local_search(h_bin2, d6, max_iter=15000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2.copy()

    # Phase 7: Basin hopping (5%)
    scale = 0.015
    n_restarts = 0
    while time.time() < deadline - 5 and n_restarts < 40:
        n_restarts += 1
        h_p = _perturb(best_h, scale, rng)
        sub_dl = min(deadline - 3, time.time() + budget_s * 0.006)
        h_t, c5_t = _lp_optimize(h_p, sub_dl, [30, 45], [0.04, 0.06], rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.12)

    # Final polish
    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _lp_optimize(h_f, deadline, [40, 60], [0.02, 0.04], rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
