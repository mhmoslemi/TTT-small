import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):

        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        return np.zeros(n)

    T = T_init

    while time.time() < deadline and len(ones) > 0 and len(zeros) > 0 if 'ones' in dir() else True:
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if stall > 600:
            break
        oi = ones[int(rng.integers(len(ones)))]
        oz = zeros[int(rng.integers(len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        nc = _exact_c5(cur)
        delta = nc - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = nc
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _binary_hill_climb(h_bin, deadline, max_rounds=200):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(7777)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        improved = False
        for i in ones:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c5
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                nc = _exact_c5(cur)
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                cur[i] = 1.0
                cur[j] = 0.0
            if best_j >= 0:
                cur[i] = 0.0
                cur[best_j] = 1.0
                cur_c5 = best_nc
                ones.remove(i)
                ones.append(best_j)
                zeros.remove(best_j)
                zeros.append(i)
                improved = True
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                break
        if not improved:
            break
    return best_h, best_c5

def _line_search(h, d, radius):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min((1.0 - h[neg]) / d[neg])))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    dx = 2.0 / n

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(55):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-15:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _continuous_opt(h, deadline, beta_start=5.0, beta_max=80.0, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    beta = beta_start
    it = 0
    while time.time() < deadline and stall < max_stall:
        it += 1
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        mx = float(np.max(raw))
        v = beta * (raw - mx)
        v = np.clip(v, -50.0, 0.0)
        w = np.exp(v)
        wsum = float(np.sum(w))
        active = (w / wsum) > 1e-8
        g = np.zeros(n)
        for r in np.where(active)[0]:
            wr = w[r] / wsum
            tau = r - (n - 1)
            if tau >= 0:
                m = n - tau
                if m > 0:
                    g[tau:] += wr * (1.0 - h[:m])
                    g[:m] -= wr * h[tau:tau + m]
            else:
                gg = -tau
                m = n - gg
                if m > 0:
                    g[:m] += wr * (1.0 - h[gg:])
                    g[gg:] -= wr * h[:m]
        g -= float(np.mean(g))
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = g / gn
        best_alpha = 0.0
        best_val = cur_c5
        best_dir = None
        for dd in [d, -d]:
            alpha, val = _line_search(h, dd, 0.25)
            if val < best_val - 1e-15:
                best_val, best_alpha, best_dir = val, alpha, dd
            if time.time() > deadline:
                break
        if best_dir is not None:
            h_new = _project(h + best_alpha * best_dir, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                beta = min(beta_max, beta * 1.03)
            else:
                stall += 1
                beta = min(beta_max, beta * 1.15)
        else:
            stall += 1
            beta = min(beta_max, beta * 1.2)
    return best_h, best_c5

def _pair_continuous(h, deadline, n_iter=5000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = _project(h + alpha * d, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _basin_hop(h, deadline, scale=0.04, max_hops=60):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    for _ in range(max_hops):
        if time.time() > deadline - 5:
            break
        x = np.arange(n, dtype=float)
        pert = np.zeros(n)
        for freq in [rng.uniform(0.5, 6.0), rng.uniform(0.5, 6.0), rng.uniform(0.5, 6.0)]:
            phase = rng.uniform(0, 2 * np.pi)
            pert += np.sin(2.0 * np.pi * freq * x / n + phase)
        pert *= scale / 3.0
        pert += scale * 0.2 * rng.normal(0, 1, n)
        h_p = _project(h + pert, target)
        sub_dl = min(deadline - 3, time.time() + 10.0)
        h_t, c5_t = _continuous_opt(h_p, sub_dl, beta_start=8.0, beta_max=50.0, max_stall=30)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.07, scale * 1.12)
    return best_h, best_c5

def _run_binary_search(start_h, deadline):
    h_bin = _to_binary(start_h)
    best_h = h_bin.copy()
    best_c5 = _exact_c5(h_bin)
    # Hill climb
    sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    hc_h, hc_c5 = _binary_hill_climb(h_bin, sub_dl, max_rounds=300)
    if hc_c5 < best_c5:
        best_c5 = hc_c5
        best_h = hc_h.copy()
    # SA
    if time.time() < deadline - 5:
        sub_dl2 = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        sa_h, sa_c5 = _binary_sa(best_h, sub_dl2, T_init=0.004, max_iter=200000)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_h = sa_h.copy()
    # Final hill climb
    if time.time() < deadline:
        hc2_h, hc2_c5 = _binary_hill_climb(best_h, deadline, max_rounds=200)
        if hc2_c5 < best_c5:
            best_c5 = hc2_c5
            best_h = hc2_h.copy()
    return best_h, best_c5
