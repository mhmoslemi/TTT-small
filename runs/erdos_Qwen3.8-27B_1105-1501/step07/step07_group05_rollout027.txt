def _grad_topk(h, k, dx, beta):
    """Gradient focusing on top-k active lags."""
    n = len(h)
    S = _c5_raw(h) * dx
    mx = S.max()
    v = np.clip(beta * (S - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= g.mean()
    return -g  # negative gradient (descent direction)

def _grad_top_lags(h, k, dx):

        if stall > 250:
            n_flip = int(rng.integers(3, 7))
            idx = rng.choice(n, size=min(n_flip, n), replace=False)
            cur[idx] = 1.0 - cur[idx]
            ones = int(np.sum(cur > 0.5))
            half = n // 2
            if ones > half:
                excess = ones - half
                one_idx = np.where(cur > 0.5)[0]
                fb = rng.choice(one_idx, size=min(excess, len(one_idx)), replace=False)
                cur[fb] = 0.0
            elif ones < half:
                deficit = half - ones
                zero_idx = np.where(cur <= 0.5)[0]
                fv = rng.choice(zero_idx, size=min(deficit, len(zero_idx)), replace=False)
                cur[fv] = 1.0
            cc = _c5_score(cur)
            stall = 0
            kick_count += 1
            if kick_count > n_kicks:
                break
            T = 0.0025
            continue
        mt = int(rng.integers(0, 5))
        hn = cur.copy()
        if mt == 0:
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[int(rng.integers(len(ones)))])
            j = int(zeros[int(rng.integers(len(zeros)))])
            hn[i], hn[j] = 0.0, 1.0
        elif mt == 1:
            blen = int(rng.integers(2, 5))
            if n < 2 * blen:
                continue
            i = int(rng.integers(0, n - blen + 1))
            j = int(rng.integers(0, n - blen + 1))
            if abs(i - j) < blen:
                continue
            tmp = hn[i:i + blen].copy()
            hn[i:i + blen] = hn[j:j + blen]
            hn[j:j + blen] = tmp
        elif mt == 2:
            slen = int(rng.integers(3, min(9, n // 4)))
            if slen < 3:
                continue
            i = int(rng.integers(0, n - slen + 1))
            seg = hn[i:i + slen].copy()
            hn[i:i + slen] = seg[::-1]
        elif mt == 3:
            i = int(rng.integers(0, n - 1))
            if cur[i] != cur[i + 1]:
                hn[i], hn[i + 1] = hn[i + 1], hn[i]
            else:
                continue
        else:
            blen = int(rng.integers(4, min(12, n // 3)))
            if n < 2 * blen:
                continue
            i = int(rng.integers(0, n - blen + 1))
            j = int(rng.integers(0, n - blen + 1))
            if abs(i - j) < blen:
                continue
            tmp = hn[i:i + blen].copy()
            hn[i:i + blen] = hn[j:j + blen]
            hn[j:j + blen] = tmp
        nc = _c5_score(hn)
        dl = nc - cc
        if dl < 0 or (T > 1e-6 and rng.random() < math.exp(-dl / T)):
            cur = hn
            cc = nc
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
                if cc < bc - 1e-4:
                    T = 0.0025
                T = max(T * 0.9996, 0.0003)
        else:
            stall += 1
    return bh, bc

def _local_opt(h, deadline, r0=0.12, stall_lim=60, seed=7777):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _c5_score(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(seed)
    rad = r0
    stall = 0
    momentum = np.zeros(n)
    while time.time() < deadline:
        if deadline - time.time() < 0.5:
            break
        if stall > stall_lim:
            if rad < 0.008:
                pert = rng.normal(0, 0.06, n)
                pert -= pert.mean()
                hp = _project(h + pert, target)
                cp = _c5_score(hp)
                if cp < cur - 1e-13:
                    h, cur = hp, cp
                    stall = 0
                    rad = 0.10
                else:
                    break
            else:
                rad *= 0.7
                stall = 0
        dirs = []
        for beta in [40.0, 100.0]:
            g = _grad_soft(h, beta)
            if np.linalg.norm(g) > 1e-10:
                dirs.append(g)
        g = _grad_top(h, 3)
        if np.linalg.norm(g) > 1e-10:
            dirs.append(g)
        g = _grad_top(h, 7)
        if np.linalg.norm(g) > 1e-10:
            dirs.append(g)
        for bs in [8, 16]:
            g = _grad_block(h, bs, 60.0)
            if np.linalg.norm(g) > 1e-10:
                dirs.append(g)
        d = rng.normal(0, 1, n)
        d -= d.mean()
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
        mom = 0.3
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - d.mean()
            d = mom * momentum + (1 - mom) * d
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            momentum = 0.8 * momentum + 0.2 * best_d
            h = _project(h + best_a * best_d, target)
            cur = _c5_score(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.3)
        else:
            stall += 1
            rad *= 0.76
    return best_h, best_c

def _pair_opt(h, deadline, max_it=15000, seed=31337):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _c5_score(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = _project(h + a * d, target)
            c5a = _c5_score(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _ils_binary(hb, deadline, n_iters=40, seed=55555):
    n = len(hb)
    best_h = hb.copy()
    best_c = _c5_score(hb)
    rng = np.random.default_rng(seed)
    scale = 0.15
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        pert_n = int(max(3, n * scale * rng.uniform(0.5, 1.5)))
        idx = rng.choice(n, size=min(pert_n, n), replace=False)
        hp = best_h.copy()
        hp[idx] = 1.0 - hp[idx]
        ones = int(np.sum(hp > 0.5))
        half = n // 2
        if ones > half:
            excess = ones - half
            oi = np.where(hp > 0.5)[0]
            fb = rng.choice(oi, size=min(excess, len(oi)), replace=False)
            hp[fb] = 0.0
        elif ones < half:
            deficit = half - ones
            zi = np.where(hp <= 0.5)[0]
            fv = rng.choice(zi, size=min(deficit, len(zi)), replace=False)
            hp[fv] = 1.0
        sd = min(deadline, time.time() + max(0.5, (deadline - time.time()) / 4.0))
        hb, cb = _binary_sa(hp, sd, n_kicks=15, seed=int(rng.integers(0, 10**6)))
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            scale = max(0.03, scale * 0.85)
        else:
            scale = min(0.3, scale * 1.15)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.ones(n0) * 0.5
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _c5_score(h_init)

    # Phase 1: Binary SA (20%)
    d1 = t0 + budget_s * 0.20
    hb = _to_binary(h_init, n0)
    hb, cb = _binary_sa(hb, d1, n_kicks=60, seed=1111)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: ILS on binary (15%)
    d2 = t0 + budget_s * 0.35
    nc = len(best_h)
    hb = _to_binary(best_h, nc)
    hb, cb = _ils_binary(hb, d2, n_iters=50, seed=22222)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: Continuous local opt (12%)
    d3 = t0 + budget_s * 0.47
    hc = _project(best_h.copy(), len(best_h) / 2.0)
    hc, cc = _local_opt(hc, d3, r0=0.12, stall_lim=50, seed=3333)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()

    # Phase 4: Pair opt (8%)
    d4 = t0 + budget_s * 0.55
    hp = _project(best_h.copy(), len(best_h) / 2.0)
    hp, cp = _pair_opt(hp, d4, max_it=12000, seed=4444)
    if cp < best_c:
        best_c = cp
        best_h = hp.copy()

    # Phase 5: Multi-resolution (23%)
    d5 = t0 + budget_s * 0.78
    nc = len(best_h)
    offsets = [-10, -5, 5, 10, -15, 15, -20, 20, -8, 8, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        # Quick binary SA
        sb = min(d5, time.time() + max(1.0, budget_s * 0.012))
        hbr = _to_binary(hr, n_try)
        hbr, cbr = _binary_sa(hbr, sb, n_kicks=20, seed=n_try)
        if cbr < best_c + 0.002:
            sc = min(d5, time.time() + max(1.0, budget_s * 0.012))
            hcr = _project(hbr.copy(), n_try / 2.0)
            hcr, ccr = _local_opt(hcr, sc, r0=0.10, stall_lim=35, seed=n_try + 1)
            if ccr < best_c:
                best_c = ccr
                best_h = hcr.copy()
                nc = n_try

    # Phase 6: Intensive refinement (12%)
    d6 = t0 + budget_s * 0.90
    nc = len(best_h)
    for r_init, s in [(0.06, 5555), (0.03, 6666), (0.015, 7777)]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + max(1.0, budget_s * 0.03))
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=45, seed=s)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 2:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd2 = min(d6, time.time() + max(0.5, budget_s * 0.015))
            h6, c6 = _pair_opt(h6, sd2, max_it=6000, seed=s + 1)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Final binary SA + ILS (8%)
    d7 = t0 + budget_s * 0.97
    nc = len(best_h)
    hb_f = _to_binary(best_h, nc)
    hb_f, cb_f = _binary_sa(hb_f, min(d7, d7 - (d7 - time.time()) * 0.5), n_kicks=30, seed=88888)
    if cb_f < best_c:
        best_c = cb_f
        best_h = hb_f.copy()
    if time.time() < d7 - 2:
        nc = len(best_h)
        hb_f = _to_binary(best_h, nc)
        hb_f, cb_f = _ils_binary(hb_f, d7, n_iters=20, seed=99999)
        if cb_f < best_c:
            best_c = cb_f
            best_h = hb_f.copy()

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=25, seed=100000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _c5_score(best_h)
    if final_c > best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
