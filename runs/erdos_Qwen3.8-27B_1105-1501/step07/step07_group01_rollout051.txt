    """Compute smooth objective value and gradient using softmax weighting."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = corr - max_c
    # For numerical stability with large beta
    exps = np.exp(beta * shifted)
    total = float(np.sum(exps))
    weights = exps / total
    smooth_val = max_c + np.log(total) / beta
    grad = np.zeros(n)
    # Only process lags with significant weight
    thresh = 1e-10
    active = weights > thresh
    for k in np.where(active)[0]:
        w = weights[k]
        d = int(k) - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += w * (1.0 - h[d:])
            grad[d:] += w * (-h[:m])
        else:
            dd = -d
            m = n - dd
            grad[dd:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[dd:])
    return smooth_val, grad

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad_smooth(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    exps = np.exp(beta * (corr - max_c))
    total = float(np.sum(exps))
    weights = exps / total
    grad = np.zeros(n)
    thresh = 1e-12
    for k in range(2*n-1):
        w = weights[k]
        if w < thresh:
            continue
        d = k - (n-1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += w * (1.0 - h[d:])
            grad[d:] += w * (-h[:m])
        else:
            dd = -d
            m = n - dd
            grad[dd:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[dd:])
    smooth_val = max_c + np.log(total) / beta
    return smooth_val, grad

def _smooth_grad_descend(h, deadline, beta_start=20.0, beta_end=3000.0,
                          max_iter=10000, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    step = 0.01
    stall = 0
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        
        frac = min(float(it) / max_iter, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        _, grad = _corr_grad_smooth(h, beta)
        # Center for mass constraint
        grad -= np.mean(grad)
        # Box constraints
        grad[h < 1e-8] = np.maximum(grad[h < 1e-8], 0.0)
        grad[h > 1.0 - 1e-8] = np.minimum(grad[h > 1.0 - 1e-8], 0.0)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            stall += 20
            continue
        
        # Momentum update
        momentum = mu * momentum + (1-mu) * grad
        direction = momentum
        
        # Adaptive step with backtracking
        found = False
        s = step
        for _ in range(20):
            h_try = _project(h - s * direction, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-11:

                    best_h = h.copy()
                    stall = 0

                found = True
                break

            step = max(step * 0.75, 1e-6)
            stall += 1

                h = _project(h + 0.005 * rng.normal(0, 1, n), target)
                cur_c5 = _exact_c5(h)
                step = 0.01
                momentum = np.zeros(n)
    
    return best_h, best_c5

                    gk += (1.0 - h[j + d])
                if 0 <= j - d < n:
                    gk += -h[j - d]
                g[k] = gk
            
            lo_t = -old
            hi_t = 1.0 - old
            if hi_t - lo_t < 1e-12:
                continue
            
            binary_mask = is_center.astype(float)
            for _ in range(35):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = float(np.max(S0 + m1 * g - m1 * m1 * binary_mask))
                f2 = float(np.max(S0 + m2 * g - m2 * m2 * binary_mask))
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            
            opt_t = (lo_t + hi_t) / 2.0
            new_val = old + opt_t
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                S0 += delta * g
                S0[n-1] -= delta * delta
                h[j] = new_val
                improved = True
        
        if not improved:
            stall += 1
            continue
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _to_binary(h):
    idx = np.argsort(h)[-len(h)//2:]
    h_bin = np.zeros(len(h))
    h_bin[idx] = 1.0
    return h_bin

def _binary_pair_sweep(h_bin, deadline, max_rounds=50):
    """Iterative pair swap for binary vectors."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    stall = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        progress = False
        # Random subset of pairs to try
        n_pairs = min(len(ones) * len(zeros), 2000)
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            oi = ones[rng.integers(0, len(ones))]
            oz = zeros[rng.integers(0, len(zeros))]
            # Effect on S0: remove contribution of h[oi]=1, add contribution of h[oz]=1
            # dS[k] = -(grad of corr[k] wrt h[oi]) + (grad of corr[k] wrt h[oz])
            # But since it's binary, the change is exact:
            # When we flip h[oi] from 1 to 0 and h[oz] from 0 to 1:
            # For each lag d: corr_d changes by:
            #   -contribution of position oi (was 1, now 0)
            #   +contribution of position oz (was 0, now 1)
            # Contribution of position p to corr at lag d:
            #   If h[p]=1: contributes (1-h[p+d]) to the "first" term and (1-h[p-d])... 
            # Actually let me just compute the delta directly.
            # corr[k] = sum_i h[i]*(1-h[i+d_k])
            # After swap: h[oi]=0, h[oz]=1
            # Delta at corr[k] with lag d:
            #   -1*(1-h[oi+d]) if 0<=oi+d<n  (removing oi's "first" contribution)
            #   -h[oi-d]*(1-1) = 0 ... no
            # Let me be more careful.
            # corr_d = sum_i h[i]*(1-h[i+d]) for valid i,i+d
            # Before: h[oi]=1, h[oz]=0
            # After: h[oi]=0, h[oz]=1
            # Change = -[oi's contribution] + [oz's contribution]
            # oi's contribution to corr_d:
            #   As "first" factor: 1*(1-h[oi+d]) if 0<=oi+d<n
            #   As "second" factor: h[oi-d]*(1-1) = 0 ... no
            #   Wait: corr_d = sum_i h[i](1-h[i+d])
            #   i=oi term: h[oi]*(1-h[oi+d]) = 1*(1-h[oi+d]) if valid
            #   i such that i+d=oi, i.e., i=oi-d: h[oi-d]*(1-h[oi]) = h[oi-d]*0 = 0
            #   So oi's total contribution = (1-h[oi+d]) if 0<=oi+d<n, else 0
            # oz's contribution to corr_d (after flip, h[oz]=1):
            #   As "first" factor: 1*(1-h[oz+d]) if 0<=oz+d<n
            #   As "second" factor: h[oz-d]*(1-h[oz-d+d]) = ... no
            #   i such that i+d=oz: h[oz-d]*(1-1) = 0
            #   So oz's contribution = (1-h[oz+d]) if 0<=oz+d<n, else 0
            # BUT WAIT. When h[oi] changes from 1 to 0, it also affects terms where oi appears as the "second" factor:
            #   corr_d includes term h[i]*(1-h[i+d]) for all valid i
            #   When i+d = oi (i.e., i = oi-d): term is h[oi-d]*(1-h[oi])
            #   Before: h[oi]=1, so this term is h[oi-d]*0 = 0
            #   After: h[oi]=0, so this term is h[oi-d]*1 = h[oi-d]
            #   So the change from this term is +h[oi-d]
            # Similarly when h[oz] changes from 0 to 1:
            #   When i+d = oz (i.e., i = oz-d): term is h[oz-d]*(1-h[oz])
            #   Before: h[oz]=0, term = h[oz-d]*1 = h[oz-d]
            #   After: h[oz]=1, term = h[oz-d]*0 = 0
            #   Change: -h[oz-d]
            
            # Total change in corr_d:
            #   -(1-h[oi+d]) if 0<=oi+d<n  [oi as first factor changes from 1 to 0]
            #   +h[oi-d] if 0<=oi-d<n  [oi as second factor: (1-h[oi]) goes from 0 to 1]
            #   +(1-h[oz+d]) if 0<=oz+d<n  [oz as first factor changes from 0 to 1]
            #   -h[oz-d] if 0<=oz-d<n  [oz as second factor: (1-h[oz]) goes from 1 to 0]
            
            # This is O(n) per pair to compute for all d. Let me compute it.
            dS = np.zeros(2*n-1)
            for d_val in range(-(n-1), n):
                k = n - 1 + d_val
                delta_k = 0.0
                if 0 <= oi + d_val < n:
                    delta_k -= (1.0 - cur[oi + d_val])
                if 0 <= oi - d_val < n:
                    delta_k += cur[oi - d_val]
                if 0 <= oz + d_val < n:
                    delta_k += (1.0 - cur[oz + d_val])
                if 0 <= oz - d_val < n:
                    delta_k -= cur[oz - d_val]
                dS[k] = delta_k
            
            new_c5_val = float(np.max(S0 + dS)) * dx
            if new_c5_val < cur_c5 - 1e-14:
                cur[oi] = 0.0
                cur[oz] = 1.0
                S0 += dS
                cur_c5 = new_c5_val
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                progress = True
                stall = 0
                # Need to update ones and zeros
                break  # restart with new configuration
        if not progress:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
