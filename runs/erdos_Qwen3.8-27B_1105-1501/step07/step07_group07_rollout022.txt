import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    R = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                R[r] = float(np.sum(h[:m] * (1.0 - h[d:d+m])))
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:d+m] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                R[r] = float(np.sum(h[g:] * (1.0 - h[:m])))
                G[r, g:g+m] += (1.0 - h[:m])
                G[r, :m] += -h[g:g+m]
    # Center row (d=0): R = sum h_i(1-h_i), G = 1-2h_i
    # Already computed above since d=0 is handled
    return R, G

def _line_search_exact(h, p, R, T, P):
    """Line search along h + t*p using exact quadratic form.
    R: current raw scores
    T: G @ p (linear term)
    P: correlate(p, p, 'full') (quadratic term)
    score_r(t) = R[r] + t*T[r] - t^2*P[r]
    """
    n = len(h)
    # Find safe t_max
    t_max = 1.0
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d))) if False else float(np.min(h[neg] / (-neg * 0 + p[neg] * -1))))
    # Simplify
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-p[neg]))))
    t_max = max(t_max, 0.0)
    t_max = min(t_max, 2.0)  # cap
    
    if t_max < 1e-12:
        return 0.0, float(np.max(R))
    
    # Grid search
    n_grid = 60
    grid_t = np.linspace(0, t_max, n_grid)
    current_val = float(np.max(R))
    grid_f = np.empty(n_grid)
    for i in range(n_grid):
        t = grid_t[i]
        scores_t = R + t * T - t * t * P
        grid_f[i] = float(np.max(scores_t))
    
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(R + c1 * T - c1*c1 * P))
        f2 = float(np.max(R + c2 * T - c2*c2 * P))
        for _ in range(50):
            if hi_b - lo_b < 1e-14:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(R + c1 * T - c1*c1 * P))
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(R + c2 * T - c2*c2 * P))
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(R + opt_t * T - opt_t*opt_t * P))
    
    # Check endpoints
    candidates = [
        (0.0, float(np.max(R))),
        (t_max, float(np.max(R + t_max * T - t_max*t_max * P))),
        (opt_t, opt_val),
    ]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _full_epi_lp(h, R, G, radius, active_rows=None, max_rows=None):
    """Solve full epigraph LP for descent direction.
    minimize eta
    subject to: G[r,:]@p - eta <= -R[r] for r in active_rows
                sum(p) = 0
                -radius <= p[i] <= radius (and box from h)
    """
    n = len(h)
    if active_rows is None:
        active_rows = np.arange(len(R))
    if max_rows is not None and len(active_rows) > max_rows:
        # Keep most critical rows (highest R values)
        order = np.argsort(R[active_rows])[::-1]
        active_rows = active_rows[order[:max_rows]]
    
    m = len(active_rows)
    if m == 0:
        return None
    
    # Variables: p (n) + eta (1) = n+1
    # Objective: minimize eta → c[n] = 1, rest 0
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: G[ar,:]@p - eta <= -R[ar]
    # → G[ar,:]@[p, eta] <= -R[ar]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G[active_rows]
    A_ub[:, n] = -1.0
    b_ub = -R[active_rows]
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(-radius, -(h[j] + 1e-12))
        hi = min(radius, 1.0 - h[j] + 1e-12)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((-1e10, 1e10))  # eta free (wide bounds)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _active_set_lp(h, R, G, radius, max_iter=50, initial_top=20, expand=10):
    """Active set LP: start with top rows, iteratively add violated rows."""
    n = len(h)
    # Initial active set: top rows by R value
    top_idx = np.argsort(R)[::-1][:initial_top]
    active = list(top_idx)
    
    p = None
    for it in range(max_iter):
        p = _full_epi_lp(h, R, G, radius, active_rows=np.array(active), max_rows=None)
        if p is None or np.max(np.abs(p)) < 1e-14:
            break
        
        # Check which inactive rows are violated
        # A row r is violated if R[r] + G[r,:]@p > eta (the LP's eta)
        # We don't have eta directly, so check: R[r] + G[r,:]@p > max over active
        # Actually, the LP ensures G[active,:]@p - eta <= -R[active]
        # So eta >= R[active] + G[active,:]@p for all active
        # A row r is violated if R[r] + G[r,:]@p > current_eta
        # We can compute: for active rows, the constraint is tight or not
        # Simple approach: compute predicted max score with direction p
        # predicted R[r] + G[r,:]@p for all r, find max, if it's from inactive, add it
        
        pred = R + G @ p  # linear prediction for all rows
        max_pred = float(np.max(pred))
        # Find rows where pred > max_pred * 0.999 or pred close to max
        # Actually, find rows NOT in active where pred is high
        in_active = np.array([1] * len(R))
        for a in active:
            in_active[a] = 0
        
        # Rows that are violated (not in active but high prediction)
        violated = [r for r in range(len(R)) if in_active[r] == 1 and pred[r] > max_pred - 1e-10 and pred[r] > R[r] + 1e-10]
        
        if not violated:
            break
        
        # Add violated rows (limit expansion)
        violated.sort(key=lambda r: -pred[r])
        for v in violated[:expand]:
            if v not in active:
                active.append(v)
        
        if len(active) > 400:  # Cap
            # Remove least critical active rows
            active.sort(key=lambda r: -R[r])
            active = active[:400]
    
    return p

def _lp_loop(h, deadline, radius_init=0.05, radius_max=0.20, top_init=15, expand=10, stall_limit=80, use_active_set=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_R = _raw_scores(h)
    cur_c5 = float(np.max(cur_R)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        
        R, G = _build_G(h)
        
        if use_active_set:
            p = _active_set_lp(h, R, G, radius, max_iter=30, initial_top=top_init, expand=expand)
        else:
            # Use all rows
            p = _full_epi_lp(h, R, G, radius, active_rows=np.arange(len(R)), max_rows=None)
        
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        # Line search
        T = G @ p
        P = np.correlate(p, p, mode='full')
        t_opt, val_opt = _line_search_exact(h, p, R, T, P)
        
        if t_opt > 1e-14:
            h_new = h + t_opt * p
            h_new = _project_sum(h_new, target)
            new_R = _raw_scores(h_new)
            new_c5 = float(np.max(new_R)) * (2.0 / n)
            
            if new_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, radius_max)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

def _pair_swap_loop(h, deadline, n_trials=10000, radius=0.25, seed=42):

        # Quick line search for this direction
        R_cur = _raw_scores(h)
        T = np.zeros(len(R_cur))

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            # Update lists
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _re
esample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _pair_sweep_binary(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = cur > 0.5
        zeros = ~ones
        for oi in np.where(ones)[0]:
            if time.time() > deadline:
                break
            for oz in np.where(zeros)[0]:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

            if hi_b - lo_b < 1e-14:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(R + c1 * T - c1 * c1 * P))
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(R + c2 * T - c2 * c2 * P))
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(R + opt_t * T - opt_t * opt_t * P))

    candidates = [
        (0.0, float(np.max(R))),
        (t_max, float(np.max(R + t_max * T - t_max * t_max * P))),
        (opt_t, opt_val),
    ]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_direction_full(h, R, G, radius):
    n = len(h)
    n_rows = len(R)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -R
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] - 1e-12)
        hi = min(radius, 1.0 - h[j] + 1e-12)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((-1e10, 1e10))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_direction_active(h, R, G, radius, init_top=20, max_rows=300):
    n = len(h)
    n_rows = len(R)
    order = np.argsort(R)[::-1]
    active = list(order[:init_top])

    p = None
    for it in range(40):
        p = _lp_direction_full(h, R, G[active], radius) if len(active) < n_rows else None
        if p is None:
            p = _lp_direction_full(h, R, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-14:
            break
        pred = R + G @ p
        max_pred = float(np.max(pred))
        in_active = np.zeros(n_rows, dtype=bool)
        in_active[active] = True
        violated = [(pred[r], r) for r in range(n_rows)
                    if not in_active[r] and pred[r] > max_pred - 1e-8]
        if not violated:
            break
        violated.sort(key=lambda x: -x[0])
        for _, r in violated[:8]:
            if r not in active and len(active) < max_rows:
                active.append(r)
        if len(active) > max_rows:
            active = list(order[:max_rows])
    return p

def _lp_loop(h, deadline, radius_init=0.05, radius_max=0.20, stall_limit=80,
             use_active=True, top_init=15):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

    while time.time() < deadline:
        if stall > stall_limit:
            break
        R, G = _build_G(h)
        if use_active:
            p = _lp_direction_active(h, R, G, radius, init_top=top_init)
        else:
            p = _lp_direction_full(h, R, G, radius)

        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue

        T = G @ p
        P = np.correlate(p, p, mode='full')
        t_opt, val_opt = _line_search_exact(h, p, R, T, P)

        if t_opt > 1e-14:
            h_new = h + t_opt * p
            h_new = _project_sum(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, radius_max)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            if T > T_init * 0.3:
                T = T_init * 0.5
            else:
                break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep_binary(h_bin, deadline, max_rounds=40):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones_list = np.where(cur > 0.5)[0]
        zeros_list = np.where(cur <= 0.5)[0]
        for oi in ones_list:
            if time.time() > deadline:
                break
            for oz in zeros_list:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _pair_swap_continuous(h, deadline, n_trials=8000, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        improved_here = False
        for alpha in [0.5, 0.25, 0.75, 0.15, 0.1, 0.35]:
            v_i = h[i] + alpha
            v_j = h[j] - alpha
            if v_i < -1e-12 or v_i > 1 + 1e-12 or v_j < -1e-12 or v_j > 1 + 1e-12:
                continue
            h[i] = v_i
            h[j] = v_j
            c_try = _exact_c5(h)
            if c_try < cur_c5 - 1e-13:
                cur_c5 = c_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved_here = True
                break
            else:
                h[i] -= alpha
                h[j] += alpha
        if not improved_here and cur_c5 >= best_c5 - 1e-14:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + ph)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 + 1)
        n0 = len(h0)

    target = n0 / 2.0
    h_init = _project_sum(h0.copy(), target)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Symmetry candidates
    h_rev = _project_sum(h_init[::-1].copy(), target)
    c5_r = _exact_c5(h_rev)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_rev.copy()

    h_comp = _project_sum(1.0 - h_init, target)
    c5_c = _exact_c5(h_comp)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_comp.copy()

    h_sym = _project_sum(0.5 * (h_init + h_init[::-1]), target)
    c5_s = _exact_c5(h_sym)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_sym.copy()

    h_revb = _project_sum(best_h[::-1].copy(), target)
    c5_rb = _exact_c5(h_revb)
    if c5_rb < best_c5:
        best_c5 = c5_rb
        best_h = h_revb.copy()

    # Phase 1: Main LP optimization (38%)
    d1 = t0 + budget_s * 0.38
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.06, radius_max=0.25,
                        stall_limit=120, use_active=True, top_init=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary optimization (14%)
    d2 = t0 + budget_s * 0.52
    h_bin = _to_binary(best_h)
    d2a = min(d2, time.time() + budget_s * 0.07)
    h_bin, c5_bin = _binary_sa(h_bin, d2a, max_iter=60000, T_init=0.003, seed=31337)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    d2b = min(d2, time.time() + budget_s * 0.04)
    h_bin2, c5_bin2 = _pair_sweep_binary(_to_binary(best_h), d2b, max_rounds=35)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()
    # LP polish on best
    if time.time() < d2:
        h_pol = _project_sum(_to_binary(best_h).astype(float), len(best_h) / 2.0)
        d2c = min(d2, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _lp_loop(h_pol, d2c, radius_init=0.04, radius_max=0.10,
                                  stall_limit=50, use_active=True, top_init=10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.72
    n_best = len(best_h)
    n_offsets = [-6, -4, -2, 2, 4, 6, -8, 8, -10, 10, -12, 12, -4, 4]
    tried_ns = set()
    for dn in n_offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 250 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.014)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.05, radius_max=0.15,
                             stall_limit=45, use_active=True, top_init=12)
        if time.time() < d3 - 2:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.007)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try + 1)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Intensive refinement (14%)
    d4 = t0 + budget_s * 0.86
    h4 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    d4a = min(d4, time.time() + budget_s * 0.07)
    h4, c5_4 = _lp_loop(h4, d4a, radius_init=0.03, radius_max=0.10,
                        stall_limit=100, use_active=True, top_init=10)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    if time.time() < d4:
        h4 = _project_sum(best_h.copy(), len(best_h) / 2.0)
        d4b = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_continuous(h4, d4b, n_trials=8000, seed=222)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Binary finish
    if time.time() < d4:
        h_b = _to_binary(best_h)
        d4c = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, d4c, max_iter=30000, T_init=0.002, seed=777)
        d4d = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep_binary(h_b, d4d, max_rounds=25)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: Basin hopping (8%)
    d5 = t0 + budget_s * 0.94
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.04, radius_max=0.12,
                             stall_limit=25, use_active=True, top_init=10)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.1)

    # Phase 6: Final polish (6%)
    h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _lp_loop(h6, deadline, radius_init=0.015, radius_max=0.05,
                        stall_limit=60, use_active=True, top_init=10)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    if deadline - time.time() > 4:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep_binary(h_b, deadline, max_rounds=12)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
