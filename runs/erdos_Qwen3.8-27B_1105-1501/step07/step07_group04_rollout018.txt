def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    if t_hi < 1e-14:
