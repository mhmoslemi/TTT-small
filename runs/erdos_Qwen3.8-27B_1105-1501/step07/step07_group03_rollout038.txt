import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad(h, weights):
    """Compute the weighted sum of gradients of raw correlations."""
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w * u[:m]
                g[:m] -= w * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w * u[gp:]
                g[gp:] -= w * h[:m]
    return g

def _grad_for_lag(h, lag_idx):
    """Compute gradient of raw[lag_idx] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    d = lag_idx - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += u[:m]
            g[:m] -= h[d:]
    else:
        gp = -d
        m = n - gp
        if m > 0:
            g[:m] += u[gp:]
            g[gp:] -= h[:m]
    return g

def _lp_proposal(h, raw, top_lags, radius, target):
    """Use LP to find a better h by linearizing top correlations."""
    n = len(h)
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h - radius,
        h_var <= h + radius
    ]
    
    for lag in top_lags:
        raw_val = raw[lag]
        grad = _grad_for_lag(h, lag)
        # Linearized: raw[lag] + grad · (h_var - h) <= t
        constraints.append(raw_val + grad @ (h_var - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CVXOPT]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                h_new = np.clip(h_var.value, 0.0, 1.0)
                h_new = _project(h_new, target)
                return h_new
        except:
            continue

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, lag_idx):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    d = lag_idx - (n - 1)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += u[:m]
            g[:m] -= h[d:]
    else:
        gp = -d
        m = n - gp
        if m > 0:
            g[:m] += u[gp:]
            g[gp:] -= h[:m]
    return g

def _lp_proposal(h, raw, top_lags, radius, target):
    n = len(h)
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target
    ]
    lb = np.maximum(h - radius, 0.0)
    ub = np.minimum(h + radius, 1.0)
    constraints.append(h_var >= lb)
    constraints.append(h_var <= ub)
    for lag in top_lags:
        raw_val = raw[lag]
        grad = _grad_for_lag(h, lag)
        constraints.append(raw_val + grad @ (h_var - h) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CPLEX, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                h_new = np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
                h_new = _project(h_new, target)
                return h_new
        except Exception:
            continue
    return None

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):

                h_bin_r, c5_br = _binary_local_search(h_bin_r, min(d4 - 2, t0 + budget_s * 0.01), max_iter=10000)
                if c5_br < c5_r:
                    c5_r = c5_br
                    h_r = h_bin_r
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_curr = n_try
    
    # Phase 5: Intensive final optimization (15%)
    d5 = deadline
    n_final = len(best_h)
    # LP pass
    if time.time() < d5 - 5:
        hf = _project(best_h.copy(), n_final / 2.0)
        hf, c5_f = _lp_optimize(hf, min(d5 - 5, t0 + budget_s * 0.06), n_top_lags=50, initial_radius=0.15)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()
    # Gradient pass
    if time.time() < d5 - 3:
        hf = _project(best_h.copy(), n_final / 2.0)
        hf, c5_f = _gradient_descent(hf, min(d5 - 3, t0 + budget_s * 0.04), beta=30.0, radius=0.08, stall_limit=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()
    # Binary final
    if time.time() < d5 - 2:
        hb = _to_binary(best_h)
        hb, c5_bf = _binary_local_search(hb, min(d5 - 2, t0 + budget_s * 0.03), max_iter=20000)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = hb.copy()
        hb2 = _to_binary(best_h)
        if time.time() < d5 - 1:
            hb2, c5_bf2 = _pair_sweep(hb2, d5)
            if c5_bf2 < best_c5:
                best_c5 = c5_bf2
                best_h = hb2.copy()
    
    # Phase 6: Basin hopping (remaining time)
    scale = 0.04
    n_restarts = 0
    while time.time() < deadline - 4 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 3, t0 + budget_s * (0.35 + 0.005 * n_restarts))
        sub_dl = min(deadline - 3, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, n_top_lags=30, initial_radius=0.1, max_iter=40)
        if time.time() < deadline - 3:
            h_t2, c5_t2 = _gradient_descent(h_t, min(deadline - 3, time.time() + budget_s * 0.003), beta=25.0, radius=0.06, stall_limit=20)
            if c5_t2 < c5_t:
                c5_t = c5_t2
                h_t = h_t2
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.1)
    
    # Final project and verify
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
