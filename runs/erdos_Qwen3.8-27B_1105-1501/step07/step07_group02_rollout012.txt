    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    omh = 1.0 - h
    for k in range(-(n - 1), n):
        r = k + n - 1
        if k >= 0:
            m = n - k
            G[r, k:] += omh[:m]
            G[r, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[r, :m] += omh[p:]
            G[r, p:] -= h[:m]
    return G

def _soft_weights(R, beta):
    M = float(np.max(R))
    adj = R - M
    max_adj = float(np.max(adj))
    if max_adj > 700.0 / beta:
        w = np.exp(beta * (adj - max_adj))
    else:
        w = np.exp(beta * adj)
    s = float(np.sum(w))
    if s < 1e-300:
        top_k = max(1, len(R) // 10)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _line_search(R, A, B, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R))
    n_grid = 64
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    bq = int(np.argmax(R[None, :] + ts[best_idx, None] * A[None, :] - ts[best_idx]**2 * B[None, :]))
    b_val = B[bq]
    a_val = A[bq]
    t_q = a_val / (2.0 * b_val) if abs(b_val) > 1e-15 else 0.0
    cands = [t_lo, t_hi, ts[best_idx], t_q]
    lo2 = ts[max(0, best_idx - 1)]
    hi2 = ts[min(n_grid - 1, best_idx + 1)]
    cands.append((lo2 + hi2) / 2.0)
    f_all = []
    for tc in cands:
        tc = float(np.clip(tc, t_lo, t_hi))
        f_all.append(float(np.max(R + tc * A - tc * tc * B)))
    best_i = int(np.argmin(f_all))
    return cands[best_i], f_all[best_i]

def _dense_opt(h, deadline, beta_start=3.0, beta_max=80.0, beta_growth=1.04,
               stall_limit=25, tang_scale=0.3, max_step=0.25, seed=42):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    beta = beta_start
    stall = 0
    radius = max_step
    R = np.correlate(cur, 1.0 - cur, mode='full')
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        w = _soft_weights(R, beta)
        G = _grad_matrix(cur)
        dense_grad = G.T @ w
        d = -(dense_grad - np.mean(dense_grad))
        lnrm = float(np.linalg.norm(d))
        if lnrm < 1e-14:
            d = rng.standard_normal(n)
            d -= np.mean(d)
            lnrm = float(np.linalg.norm(d))
            if lnrm < 1e-14:
                break
        d = d / lnrm * tang_scale
        idx_pos = np.where(d > 1e-12)[0]
        idx_neg = np.where(d < -1e-12)[0]
        t_hi = radius
        t_lo = -radius
        if len(idx_pos) > 0:
            t_hi = min(t_hi, float(np.min((1.0 - cur[idx_pos]) / d[idx_pos])))
            t_lo = max(t_lo, float(np.max(-cur[idx_pos] / d[idx_pos])))
        if len(idx_neg) > 0:
            t_hi = min(t_hi, float(np.min(cur[idx_neg] / (-d[idx_neg]))))
            t_lo = max(t_lo, float(np.max((1.0 - cur[idx_neg]) / d[idx_neg])))
        if t_hi - t_lo < 1e-13:
            radius *= 0.7
            stall += 1
            continue
        omc = 1.0 - cur
        A = np.correlate(d, omc, mode='full') - np.correlate(cur, d, mode='full')
        B = np.correlate(d, d, mode='full')
        t_opt, val_pred = _line_search(R, A, B, t_lo, t_hi)
        cur_try = cur + t_opt * d
        if np.min(cur_try) < -1e-10 or np.max(cur_try) > 1.0 + 1e-10:
            cur_try = np.clip(cur_try, 0.0, 1.0)
            cur_try = _project(cur_try, n / 2.0)
        c5_try = _exact_c5(cur_try)
        if c5_try < best_c5 - 1e-14:
            cur = cur_try
            best_c5 = c5_try
            best_h = cur.copy()
            R = np.correlate(cur, 1.0 - cur, mode='full')
            stall = 0
            beta = min(beta * beta_growth, beta_max)
            radius = min(radius * 1.05, max_step)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.12
                d_rand = rng.standard_normal(n)
                d_rand -= np.mean(d_rand)
                lr = float(np.linalg.norm(d_rand))
                if lr > 1e-14:
                    d_rand = d_rand / lr * 0.05
                    ip = np.where(d_rand > 1e-12)[0]
                    inn = np.where(d_rand < -1e-12)[0]
                    th2 = 0.05
                    tl2 = -0.05
                    if len(ip) > 0:
                        th2 = min(th2, float(np.min((1.0 - cur[ip]) / d_rand[ip])))
                        tl2 = max(tl2, float(np.max(-cur[ip] / d_rand[ip])))
                    if len(inn) > 0:
                        th2 = min(th2, float(np.min(cur[inn] / (-d_rand[inn]))))
                        tl2 = max(tl2, float(np.max((1.0 - cur[inn]) / d_rand[inn])))
                    if th2 - tl2 > 1e-13:
                        omc2 = 1.0 - cur
                        A2 = np.correlate(d_rand, omc2, mode='full') - np.correlate(cur, d_rand, mode='full')
                        B2 = np.correlate(d_rand, d_rand, mode='full')
                        t2, _ = _line_search(R, A2, B2, tl2, th2)
                        cur_try2 = cur + t2 * d_rand
                        cur_try2 = np.clip(cur_try2, 0.0, 1.0)
                        c5_2 = _exact_c5(cur_try2)
                        if c5_2 < best_c5 - 1e-14:
                            cur = cur_try2
                            best_c5 = c5_2
                            best_h = cur.copy()
                            R = np.correlate(cur, 1.0 - cur, mode='full')
                            stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_opt(h_bin, deadline, max_iter=50000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def _pair_opt(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=1):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    one_mh = 1.0 - h
    stall = 0
    rng = np.random.default_rng(seed)
    cur_radius = radius
    while time.time() < deadline:
        if stall > stall_limit or cur_radius < 1e-6:
            break
        S0 = np.correlate(h, one_mh, mode='full')
        M0 = float(np.max(S0))
        adj = S0 - M0
        w = np.exp(beta * adj)
        ws = float(np.sum(w))
        if ws < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx_top = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx_top] = 1.0 / top_k
        else:
            w /= ws
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        ns = min(n_pairs, n // 2)
        sources = sort_idx[-ns:]
        sinks = sort_idx[:ns]
        pairs = []
        for ii in range(min(ns, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(10, max(1, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -cur_radius)
            t_hi = min(1.0 - h[sink], h[source], cur_radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            one_mh = 1.0 - h
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            cur_radius = min(cur_radius * 1.03, 0.28)
        else:
            stall += 1
            cur_radius *= 0.75
            if stall % 40 == 0 and cur_radius < 0.02:
                cur_radius = 0.10
    return best_h, best_c5

        best_h = h6.copy()
    n_cur = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d6:
            break
        n_try = n_cur + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _dense_opt(h_r, sub_dl, beta_start=6.0, beta_max=80.0,
                               beta_growth=1.04, tang_scale=0.2, max_step=0.15, seed=444 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try
    d7 = deadline - 3
    n_cur = len(best_h)
    h7 = _project(best_h.copy(), n_cur / 2.0)
    h7, c5_7 = _dense_opt(h7, d7, beta_start=10.0, beta_max=150.0, beta_growth=1.03,
                          tang_scale=0.12, max_step=0.10, stall_limit=40, seed=888)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h_final = _project(best_h.copy(), len(best_h) / 2.0)
    final_c5 = _exact_c5(h_final)
    return h_final, final_c5, len(h_final)
