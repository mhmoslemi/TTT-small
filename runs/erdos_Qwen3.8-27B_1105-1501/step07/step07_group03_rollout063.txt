import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, raw, n, radius, n_lags=15):

    # Select top lags
    top_idx = np.argsort(-raw)[:n_lags]
    
    # Compute gradients for selected lags
    grads = []
    for r in top_idx:

            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:])
                g[d:] -= h[:m]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[gp:] += (1.0 - h[:m])
                g[:m] -= h[gp:]
        grads.append(g)

        d_var >= -radius,
        d_var <= radius,
        h + d_var >= 0,
        h + d_var <= 1

        try:
            prob.solve(solver=cp.CBC)
        except:
            return None, None
    
    if d_var.value is None:
        return None, None
    
    d_vec = np.array(d_var.value).flatten()
    lp_t = t_var.value
    return d_vec, lp_t

        if val_opt < best_val:

    """Gradient of corr[k] w.r.t. h."""
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    g = np.zeros(n)
    # First part: h[m] * (1-h[k-m]), derivative w.r.t. h[m] is (1-h[k-m])
    # m ranges from lo to hi
    m_range = np.arange(lo, hi + 1)
    k_minus_m = k - m_range
    g[m_range] += (1.0 - h[k_minus_m])
    # Second part: -h[m] * h[k-m], derivative w.r.t. h[k-m] is -h[m]
    # So for index p = k-m where m is in [lo, hi]: g[k-m] -= h[m]
    g[k_minus_m] -= h[m_range]
    return g

        d = _solve_lp(h, raw, n, radius, list(active))

        raw_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        # Find the max lag
        k_max = int(np.argmax(raw_new))
        if k_max not in active:

                break
            # Try adding another lag
            for k in np.argsort(-raw_new)[n_lags:n_lags+5]:
                if k not in active:
                    active.add(k)
                    break
            else:
                break
    
    return d

def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))

                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 4.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _all_lags_lp_optimize(h_init, deadline, n_lags=30, radius=0.10, stall_limit=150):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-raw)[:n_lags]
        grads = _compute_grads(h, n, top_idx)
        d_vec, lp_t = _solve_lp(h, raw, n, radius_cur, top_idx, grads, solver_name='GLOP')
        accepted = False
        if d_vec is not None and np.linalg.norm(d_vec) > 1e-10:
            alpha, _ = _line_search(h, d_vec, raw, dx, radius_cur)
            if alpha > 1e-10:
                h_new = _project(h + alpha * d_vec, target)
                actual = _exact_c5(h_new)
                if actual < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = actual
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius_cur = min(radius_cur * 1.01, 0.25)
                    accepted = True
        if not accepted:
            for k_try in [int(np.argmax(raw)), int(np.argsort(-raw)[1]), int(np.argsort(-raw)[2])]:
                g = _compute_grads(h, n, [k_try])[0]
                g -= np.mean(g)
                gn = np.linalg.norm(g)
                if gn > 1e-12:
                    g /= gn
                    alpha, _ = _line_search(h, g, raw, dx, radius_cur)
                    if alpha > 1e-10:
                        h_new = _project(h + alpha * g, target)
                        actual = _exact_c5(h_new)
                        if actual < cur_c5 - 1e-14:
                            h = h_new
                            cur_c5 = actual
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            stall = 0
                            accepted = True
                            break
        if not accepted:
            stall += 1
            radius_cur *= 0.85
            if radius_cur < 0.015:
                radius_cur = radius
                stall = 0
                if stall > stall_limit:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization from initial point (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _optimize_lp(h_init, d1, n_lags=15, radius=0.15, stall_limit=80, use_qp=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: QP-based refinement (10% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _optimize_lp(best_h.copy(), d2, n_lags=20, radius=0.08, stall_limit=60, use_qp=True)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: All-lags LP intensive (25% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _all_lags_lp_optimize(best_h.copy(), d3, n_lags=30, radius=0.08, stall_limit=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Try different n values (15% time)
    d4 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in [-12, -10, -8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, -16, -14]:
        if time.time() > d4 - 5:
            break
        n_try = n_curr + dn
        if n_try < 70 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.008)
        h_r, c5_r = _optimize_lp(h_r, sub_dl, n_lags=12, radius=0.10, stall_limit=30)
        if c5_r < best_c5 + 0.001 and time.time() < d4 - 3:
            sub_dl2 = min(d4, time.time() + budget_s * 0.005)
            h_r, c5_r = _all_lags_lp_optimize(h_r.copy(), sub_dl2, n_lags=20, radius=0.06, stall_limit=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Binary search + local improvement (10% time)
    d5 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    h_bin = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.0015)
    if time.time() < d5 - 2:
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_bin, c5_b = _pair_sweep_binary(h_bin, sub_dl2)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d5 - 1:
        h_pol = _project(h_bin.copy(), n_curr / 2.0)
        sub_dl3 = min(d5, time.time() + budget_s * 0.015)
        h_pol, c5_p = _optimize_lp(h_pol, sub_dl3, n_lags=15, radius=0.08, stall_limit=40)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 6: Basin hopping (12% time)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 - 3, time.time() + budget_s * 0.005)
        h_t, c5_t = _optimize_lp(h_pert, sub_dl, n_lags=15, radius=0.10, stall_limit=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.06, scale * 1.10)

    # Phase 7: Final intensive LP polish
    if deadline - time.time() > 10:
        h7, c5_7 = _all_lags_lp_optimize(best_h.copy(), deadline, n_lags=25, radius=0.05, stall_limit=100)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Pair swap micro-optimization
    if deadline - time.time() > 5:
        h8 = _project(best_h.copy(), len(best_h) / 2.0)
        h8, c5_8 = _pair_swap_cont(h8, deadline, n_pairs=3000, radius=0.10, seed=777)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
