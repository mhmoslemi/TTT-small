def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 0.9999)
        else:
            mask = (x > 0.0001)
        cnt = int(np.sum(mask))

def _pnorm_scale(corr, p):
    M = np.max(corr)
    if M < 1e-30:
        return 0.0, 0.0, corr
    scaled = np.maximum(corr, 0.0) / M
    return M, S if (S := float(np.sum(scaled ** p))) > 1e-30 else 1e-30, scaled

def _pnorm_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    if M < 1e-30:
        return 0.0
    scaled = np.maximum(corr, 0.0) / M
    S = float(np.sum(scaled ** p))
    if S < 1e-30:
        return 0.0
    return M * S ** (1.0 / p) * dx

def _pnorm_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    if M < 1e-30:
        return np.zeros(n)
    scaled = np.maximum(corr, 0.0) / M
    Sp = float(np.sum(scaled ** p))
    if Sp < 1e-30:
        return np.zeros(n)
    base = Sp ** (1.0 / p - 1.0)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = dx * base * scaled[k] ** (p - 1)
        if w < 1e-20:

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        kmj = k - j
        grad[j] += w * (1.0 - 2.0 * h[kmj])
    return grad

def _mass_eq_jac(h, target):

        if stall > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr[idx]
        A_eq_mat = np.ones((1, n + 1))
        b_eq = np.array([0.0])
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj])
            hi_b = min(radius, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq_mat, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        p_dir = res.x[:n]
        if float(np.max(np.abs(p_dir))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p_dir, 1.0 - h, mode='full') - np.correlate(h, p_dir, mode='full')
        C = np.correlate(p_dir, p_dir, mode='full')
        pos = p_dir > 1e-15
        neg = p_dir < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p_dir[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p_dir[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        vals = [(0.0, f0), (opt_t, opt_val), (t_hi, fe)]
        best_t = min(vals, key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-13:
            h = h + best_t * p_dir
            h = _project(h, target)
            cur_c5 = _c5_exact(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=3000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = lo_d, hi_d
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(S0 + m1 * g))
            f2 = float(np.max(S0 + m2 * g))
            for _ in range(40):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = hi - (hi - lo) / 3.0
                    f1 = float(np.max(S0 + m1 * g))
                else:
                    lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = lo + (hi - lo) / 3.0
                    f2 = float(np.max(S0 + m2 * g))
            opt_d = (lo + hi) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap(h, deadline, n_iters=5000, seed=777):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        t_hi = 1.0
        if h[i] >= 1.0 - 1e-14:
            t_hi = 0.0
        if h[j] <= 1e-14:
            t_hi = 0.0
        if t_hi < 1e-14:
            stall += 1
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-13:
            if best_v == f0:
                stall += 1
                continue
            if best_v == opt_val:
                t_best = opt_t
            else:
                t_best = t_hi
            h = h + t_best * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _multi_n_pipeline(h_current, n_new, deadline, seed):
    h_r = _resample(h_current, n_new)
    target = n_new / 2.0
    h_r = _project(h_r, target)
    for p in [8, 16]:
        if time.time() > deadline - 5:
            break
        h_r = _slsqp_optimize(h_r, p, maxiter=80)
    h_r, c5_r = _coord_descent(h_r, min(deadline - 5, time.time() + 40), max_sweeps=1000, seed=seed)
    if time.time() < deadline - 10:
        h_r, c5_r2 = _lp_active_set(h_r, min(deadline - 5, time.time() + 25),
                                     radius_init=0.06, stall_limit=30, top_k=10)
        if c5_r2 < c5_r:
            h_r = h_r
            c5_r = c5_r2
    c5_r = _c5_exact(h_r)
    return h_r, c5_r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _c5_exact(best_h)

    # Phase 1: SLSQP with p-norm on primary n (30% budget)
    d1 = t0 + budget_s * 0.30
    h_work = h_init.copy()
    for p in [4, 8, 16, 32]:
        if time.time() > d1 - 10:
            break
        h_work = _slsqp_optimize(h_work, p, maxiter=200)
        c5_check = _c5_exact(h_work)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_work.copy()
        else:
            h_work = best_h.copy()

    # Phase 2: Subgradient refinement (10%)
    d2 = t0 + budget_s * 0.40
    h_sg, c5_sg = _subgradient_opt(best_h, min(d2, time.time() + 80), n_steps=3000, tau_init=0.3)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 3: LP active set (15%)
    d3 = t0 + budget_s * 0.55
    h_lp, c5_lp = _lp_active_set(best_h, d3, radius_init=0.08, stall_limit=100, top_k=14)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Coordinate descent (15%)
    d4 = t0 + budget_s * 0.70
    h_cd, c5_cd = _coord_descent(best_h, d4, max_sweeps=4000, seed=42)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
    h_cd2, c5_cd2 = _coord_descent(best_h, min(d4, time.time() + 60), max_sweeps=2000, seed=999)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 5: Multi-n exploration (15%)
    d5 = t0 + budget_s * 0.85
    n_base = len(best_h)
    n_tried = set([n_base])
    offsets = list(range(-20, 25, 4))
    rng.shuffle(offsets)
    for dn in offsets:
        if time.time() > d5 - 15:
            break
        n_try = n_base + dn
        if n_try < 70 or n_try > 260 or n_try in n_tried:
            continue
        n_tried.add(n_try)
        h_try, c5_try = _multi_n_pipeline(best_h, n_try, min(d5 - 5, time.time() + 60), seed=1000 + dn)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 6: Binary + pair swap (5%)
    d6 = t0 + budget_s * 0.93
    h_bin = _to_binary(best_h)
    h_bin = _project(h_bin, len(h_bin) // 2.0)
    c5_bin = _c5_exact(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < d6 - 5:
        h_ps, c5_ps = _pair_swap(best_h, d6, n_iters=4000, seed=555)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 7: Final intensive optimization (7%)
    d7 = deadline
    h_final = best_h.copy()
    for p in [16, 32]:
        if time.time() > d7 - 10:
            break
        h_final = _slsqp_optimize(h_final, p, maxiter=100)
        c5_f = _c5_exact(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()
    if time.time() < d7 - 5:
        h_f2, c5_f2 = _lp_active_set(best_h, min(d7 - 2, time.time() + 30),
                                       radius_init=0.04, stall_limit=50, top_k=20)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
    # Basin hopping
    scale = 0.015
    n_restarts = 0
    while time.time() < d7 - 3 and n_restarts < 40:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d7 - 3, time.time() + 4)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=300, seed=3000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.1)

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    if final_c5 > best_c5:
        best_h = _project(best_h, len(best_h) / 2.0)
    return best_h, final_c5, len(best_h)
