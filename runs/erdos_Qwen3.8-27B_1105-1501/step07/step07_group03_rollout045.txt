    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _build_lp_matrices(h0, lags, n, target, radius):
    """Build LP matrices for the trust-region minimax problem.
    
    Variables: [x_0, ..., x_{n-1}, t]
    Minimize: t
    Subject to:
      For each lag r in lags: linearized_corr_r(x) <= t
      sum(x) = target
      0 <= x_i <= 1
      h0_i - radius <= x_i <= h0_i + radius
    """
    # Linearized correlation for lag r:
    # d = r - (n-1)
    # If d >= 0: L_r(x) = sum_{i=0}^{n-d-1} [(1-h0[i+d])*x[i] - h0[i]*x[i+d]]
    # If d < 0: let g = -d, L_r(x) = sum_{i=0}^{n-g-1} [(1-h0[i])*x[i+g] - h0[i+g]*x[i]]
    #           = sum_{j=g}^{n-1} [(1-h0[j-g])*x[j] - h0[j]*x[j-g]]
    
    # Build coefficient vectors for x in the constraint L_r(x) - t <= 0
    # A_ub * vars <= b_ub
    
    m = len(lags)
    # A_ub shape: (m + 2*n, n+1)
    # We'll build it row by row
    
    A_rows = []
    b_rows = []
    
    for r in lags:
        d = r - (n - 1)
        coeff = np.zeros(n)
        if d >= 0:
            m_size = n - d
            for i in range(m_size):
                coeff[i] += (1.0 - h0[i + d])
                coeff[i + d] -= h0[i]
        else:
            g = -d
            m_size = n - g
            for i in range(m_size):
                coeff[i + g] += (1.0 - h0[i])
                coeff[i] -= h0[i + g]
        
        # L_r(x) <= t  =>  coeff @ x - t <= 0
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # Upper bound on t: t <= t_max (to make problem bounded)
    # Actually we minimize t, so we don't need upper bound on t
    # But we need t >= 0 for numerical stability? No, t can be negative in principle
    # but since all correlations are non-negative, t will be non-negative
    
    # Equality constraint: sum(x) = target
    eq_row = np.zeros(n + 1)
    eq_row[:n] = 1.0
    A_eq = eq_row.reshape(1, -1)
    b_eq = np.array([target])
    
    # Bounds: h0_i - radius <= x_i <= h0_i + radius, also 0 <= x_i <= 1
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    # t is unbounded below (we minimize it)
    bounds.append((None, None))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    return c, A_ub, b_ub, A_eq, b_eq, bounds

def _lp_direction(h0, top_k=15, radius=0.15):

    n = len(h0)
    target = n / 2.0
    raw = np.correlate(h0, 1.0 - h0, mode='full')
    mx = float(np.max(raw))
    
    # Select top lags
    indices = np.argsort(-raw)[:top_k]
    
    # Build LP
    # Variables: [x_0, ..., x_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    for r in indices:
        d = r - (n - 1)
        coeff = np.zeros(n)
        if d >= 0:
            m_size = n - d
            coeff[:m_size] += (1.0 - h0[d:d + m_size])
            coeff[d:d + m_size] -= h0[:m_size]
        else:
            g = -d
            m_size = n - g
            coeff[g:g + m_size] += (1.0 - h0[:m_size])
            coeff[:m_size] -= h0[g:g + m_size]
        
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    ub = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lb, ub))
    bounds.append((None, None))

                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    x_lp = res.x[:n]
    t_lp = res.x[n]
    
    # Line search from h0 to x_lp
    d = x_lp - h0
    # Evaluate C5 at several points along the line
    # For alpha in [0, 1], h = h0 + alpha * d is feasible
    
    best_alpha = 0.0

            best_alpha = alpha

    if 0 < best_alpha < 1:

        hi = min(1.0, best_alpha + 0.05)
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        
        def obj(alpha):

            return _exact_c5(h_try)
        
        f1 = obj(c1)

            if f1 < f2:

                c1 = hi - gr * (hi - lo)
                f1 = obj(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)

        opt = (lo + hi) / 2.0
        val_opt = obj(opt)
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    
    if best_alpha < 1e-6:
        return None, None
    
    h_new = _project(h0 + best_alpha * d, target)
    return h_new, best_val


def _gradient_direction(h0, beta=20.0, radius=0.2):
    """Fallback gradient-based direction."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    raw = np.correlate(h0, 1.0 - h0, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    
    # Compute weighted gradient
    g = np.zeros(n)
    u = 1.0 - h0
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            g[:m] -= w[r] * u[d:]
            g[d:] += w[r] * h0[:m]
        else:
            gp = -d
            m = n - gp
            g[gp:] -= w[r] * h0[:m]
            g[:m] += w[r] * u[gp:]
    g -= np.mean(g)
    
    if np.linalg.norm(g) < 1e-14:
        return None, None
    
    d = -g / np.linalg.norm(g)
    
    # Line search
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h0[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h0[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    
    for alpha in np.linspace(0, a_hi, 15):
        h_try = _project(h0 + alpha * d, target)
        c5 = _exact_c5(h_try)
        if c5 < best_val:
            best_val = c5
            best_alpha = alpha
    
    if best_alpha < 1e-8:
        return None, None
    
    h_new = _project(h0 + best_alpha * d, target)
    return h_new, best_val

def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

    res = target - float(np.sum(x))

        h_t, c5_t = _optimize_block(h_pert, sub_dl, top_k=12, beta=22.0,
                                    radius=0.10, stall_limit=20,
                                    use_lp=True, use_grad=True, use_pairs=False)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.80)
        else:
            scale = min(0.06, scale * 1.10)

    # Final polish
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _optimize_block(h6, deadline - 1, top_k=30, beta=35.0,
                                   radius=0.04, stall_limit=40,
                                   use_lp=True, use_grad=True, use_pairs=True)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
