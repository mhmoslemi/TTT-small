**Key improvement idea: Use the "smooth max" (log-sum-exp) as the objective for gradient-based optimization**

f_μ(h) = (1/μ) * log(sum_k exp(μ * R_k(h)))

For large μ, this approximates max_k R_k(h). The gradient is:
∂f_μ/∂h_i = (1/μ) * sum_k exp(μ * R_k) * ∂R_k/∂h_i / sum_k exp(μ * R_k)

where p_k = exp(μ * R_k) / sum_j exp(μ * R_j) are the softmax weights.

This gives a smooth, differentiable objective whose gradient is a convex combination of the score gradients, weighted by how close each score is to the maximum.

        return 0.0, float(np.max(S0) * dx)
    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t*T - t*t*C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 4)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 4)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-12:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr*(hi_b - lo_b)
        c2 = lo_b + gr*(hi_b - lo_b)
        f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-14:
                break
            if f1 < f2:
                hi_b = c2; c2 = c1; f2 = f1
                c1 = hi_b - gr*(hi_b - lo_b)
                f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
            else:
                lo_b = c1; c1 = c2; f1 = f2
                c2 = lo_b + gr*(hi_b - lo_b)
                f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    end_val = float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx
    best = min([(opt_t, opt_val), (t_lo, float(np.max(S0)*dx)), (t_hi, end_val)], key=lambda x: x[1])
    return best[0], best[1]

def _lp_loop(h, deadline, radius_init=0.10, top_k_init=4, max_top_k=12, stall_limit=100):
    n = len(h)
    dx = 2.0/n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project_sum(h, n/2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 50 == 0 and radius < 0.01:
                radius = 0.05
                top_k = top_k_init
    return best_h, best_c5

def _smooth_max_loop(h, deadline, mu_start=20.0, mu_end=300.0, max_iters=50000):
    n = len(h)
    dx = 2.0/n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        mu_cur = mu_start + (mu_end - mu_start) * min(1.0, it / (max_iters * 0.7))
        obj, grad, p = _smooth_scores_grad(h, n, mu_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        h_trial = h - alpha * grad
        h_trial = _project_sum(h_trial, n/2.0)
        new_c5 = _exact_c5(h_trial)
        if new_c5 < cur_c5 - 1e-15:
            h = h_trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha *= 0.8
            stall += 1
    return best_h, best_c5

def _pair_swap_loop(h, deadline, n_pairs=10000, radius=0.3, seed=111):
    n = len(h)
    dx = 2.0/n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project_sum(h, n/2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_pair_sweep(h_bin, deadline, max_rounds=30):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2*np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2*np.pi*f*x/n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n/2.0)

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 800:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0//2] = 1.0
    n0 = len(h0)
    n0 = (n0 // 2) * 2
    if n0 != len(h0):
        h0 = _resample(h0, n0)
        n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0/2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Phase 1: LP optimization (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.12, top_k_init=5, max_top_k=14, stall_limit=80)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: Smooth max gradient (25%)
    d2 = t0 + budget_s * 0.43
    h2 = _project_sum(best_h.copy(), len(best_h)/2.0)
    h2, c5_2 = _smooth_max_loop(h2, d2, mu_start=15.0, mu_end=400.0, max_iters=80000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Phase 3: LP + smooth max alternation (17%)
    d3 = t0 + budget_s * 0.60
    for _ in range(6):
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s*0.02)
        h3 = _project_sum(best_h.copy(), len(best_h)/2.0)
        h3, c5_3 = _lp_loop(h3, sub_dl, radius_init=0.04, top_k_init=4, max_top_k=10, stall_limit=40)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        sub_dl2 = min(d3, time.time() + budget_s*0.02)
        h3 = _project_sum(best_h.copy(), len(best_h)/2.0)
        h3, c5_3 = _smooth_max_loop(h3, sub_dl2, mu_start=50.0, mu_end=500.0, max_iters=20000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    # Phase 4: Pair swaps (12%)
    d4 = t0 + budget_s * 0.72
    h4 = _project_sum(best_h.copy(), len(best_h)/2.0)
    sub_dl = min(d4, time.time() + budget_s*0.06)
    h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=12000, radius=0.25, seed=222)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    sub_dl2 = min(d4, time.time() + budget_s*0.05)
    h4 = _project_sum(best_h.copy(), len(best_h)/2.0)
    h4, c5_4 = _smooth_max_loop(h4, sub_dl2, mu_start=80.0, mu_end=600.0, max_iters=30000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Phase 5: Binary search + basin hopping (13%)
    d5 = t0 + budget_s * 0.85
    h_b = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s*0.04)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=30000, T_init=0.002, seed=777)
    sub_dl2 = min(d5, time.time() + budget_s*0.025)
    h_b, c5_b = _binary_pair_sweep(h_b, sub_dl2, max_rounds=25)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    # Basin hopping with smooth max
    scale = 0.04
    n_restarts = 0
    while time.time() < d5 and n_restarts < 60:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl3 = min(d5, time.time() + budget_s*0.005)
        h_t, c5_t = _smooth_max_loop(h_pert, sub_dl3, mu_start=30.0, mu_end=200.0, max_iters=8000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)
    # Phase 6: Try other n values (8%)
    d6 = t0 + budget_s * 0.93
    n_cur = len(best_h)
    for dn in [-6, 6, -10, 10, -14, 14, -20, 20]:
        if time.time() > d6:
            break
        n_try = n_cur + dn
        if n_try < 80 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s*0.008)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.08, top_k_init=4, max_top_k=10, stall_limit=50)
        sub_dl2 = min(d6, time.time() + budget_s*0.006)
        h_r, c5_r = _smooth_max_loop(h_r, sub_dl2, mu_start=30.0, mu_end=400.0, max_iters=15000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try
    # Phase 7: Final polish (7%)
    d7 = deadline - 2
    n_best = len(best_h)
    sub_dl = min(d7, time.time() + budget_s*0.03)
    h_last = _project_sum(best_h.copy(), n_best/2.0)
    h_last, c5_last = _smooth_max_loop(h_last, sub_dl, mu_start=100.0, mu_end=800.0, max_iters=20000)
    if c5_last < best_c5:
        best_c5 = c5_last
        best_h = h_last.copy()
    sub_dl2 = min(d7, time.time() + budget_s*0.025)
    h_last = _project_sum(best_h.copy(), len(best_h)/2.0)
    h_last, c5_last = _lp_loop(h_last, sub_dl2, radius_init=0.015, top_k_init=3, max_top_k=8, stall_limit=60)
    if c5_last < best_c5:
        best_c5 = c5_last
        best_h = h_last.copy()
    # Final one-shot pair sweep on binary version
    if deadline - time.time() > 5:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_pair_sweep(h_b, deadline - 2, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Ensure valid
    best_h = _project_sum(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
