def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):

def _line_search(h, direction, raw, dx, radius):
    """Line search: minimize max_k (raw[k] + alpha * T[k] - alpha^2 * Q[k]) over k
    where T[k] = d_dot(1-h) at lag k, Q[k] = d_dot(d) at lag k
    Actually more precisely:
    new_raw[k] at h + alpha*d:
    = correlate(h + alpha*d, 1 - h - alpha*d)
    = correlate(h, 1-h) + alpha*correlate(d, 1-h) - alpha*correlate(h, d) - alpha^2*correlate(d, d)
    = raw + alpha*T - alpha^2*Q
    where T = correlate(d, 1-h) - correlate(h, d) = correlate(d, 1-2h) ... 
    Actually:
    correlate(h+ad, 1-h-ad) = correlate(h,1-h) + a*correlate(d,1-h) + a*correlate(h,-d) + a^2*correlate(d,-d)
    = raw + a*(correlate(d,1-h) - correlate(h,d)) - a^2*correlate(d,d)
    = raw + a*T - a^2*Q
    where T = correlate(d,1-h) - correlate(h,d) and Q = correlate(d,d)
    """
    n = len(h)
    T = np.correlate(direction, 1.0 - h, mode='full') - np.correlate(h, direction, mode='full')
    Q = np.correlate(direction, direction, mode='full')
    
    # Find valid alpha range: 0 <= h + alpha*d <= 1
    a_hi = radius
    pos = direction > 1e-14
    neg = direction < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-direction[neg]))))
    a_hi = max(a_hi, 1e-14)
    
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    # Golden section search on [0, a_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(80):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    opt = (lo + hi) / 2.0
    val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
    if val_opt < best_val:
        best_val = val_opt
        best_alpha = opt
    
    return best_alpha, best_val

def _gradient_direction(h, raw, n_active=20):
    """Compute a gradient-like direction using softmax over active lags."""
    n = len(h)
    # Select top lags
    top_k = max(n_active, 5)
    top_idx = np.argsort(-raw)[:top_k]
    
    # Compute gradient of max (approximated by softmax-weighted sum)
    shifted = raw[top_idx] - raw[top_idx[0]]
    beta = 30.0
    w = np.exp(np.clip(beta * shifted, -50, 0))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        if lag >= 0:
            m = n - lag
            for i in range(m):
                # d/dh[i] of h[i]*(1-h[i+lag]): 1-h[i+lag]
                # d/dh[i+lag]: -h[i]
                grad[i] += w[k] * (1.0 - h[i + lag])
                grad[i + lag] += w[k] * (-h[i])
        else:
            lag_pos = -lag
            m = n - lag_pos
            for i in range(m):
                # h[i+lag_pos]*(1-h[i])
                grad[i] += w[k] * (-h[i + lag_pos])
                grad[i + lag_pos] += w[k] * (1.0 - h[i])
    
    # Project out mean (to maintain sum constraint)
    grad -= np.mean(grad)
    return -grad  # descent direction

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_local_search(h_bin, deadline, max_iter=30000, T_init=0.002):
    """Simulated annealing on binary configurations."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones[j] = oz  # update zeros list
            # Actually need to be more careful
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:

                alpha_g, _ = _line_search(h, gd, dx, delta * 0.4, raw)
                if alpha_g > 1e-12:
                    h_try = _project(h + alpha_g * gd, target)
                    c_try = _exact_c5(h_try)
                    if c_try < cur_c5 - 1e-13:
                        h = h_try
                        cur_c5 = c_try
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        delta = min(delta * 1.02, 0.45)
                        accepted = True
        if not accepted:
            stall += 1
            delta *= 0.88
            if delta < 0.04:
                delta = 0.25
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Primary LP optimization on original n (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_optimize_round(h_init, d1, n_active=18, delta_init=0.3, max_iters=800)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary refinement (15%)
    d2 = t0 + budget_s * 0.50
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.07)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.004, seed=77)
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    # Polish with LP
    if time.time() < d2 - 2:
        hp = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        hp, c5_p = _lp_optimize_round(hp, sub_dl, n_active=15, delta_init=0.15, max_iters=200)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = hp.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = list(range(-14, 15, 2))
    for dn in offsets:
        if time.time() > d3 - 5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 220 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        x_old = (np.arange(n_curr) + 0.5) / n_curr
        x_new = (np.arange(n_try) + 0.5) / n_try
        h_r = np.interp(x_new, x_old, best_h)
        h_r = _project(h_r, n_try / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_optimize_round(h_r, sub_dl, n_active=14, delta_init=0.25, max_iters=200)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (20%)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4, c5_4 = _lp_optimize_round(h4, sub_dl, n_active=20, delta_init=0.2, max_iters=400)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=30000, T_init=0.0015, seed=555)
        sub_dl2 = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _lp_optimize_round(h4, sub_dl, n_active=20, delta_init=0.1, max_iters=300)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.035
    for restart_i in range(80):
        if time.time() > d5 - 8:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        n_p = len(h_pert)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.01)
        h_t, c5_t = _lp_optimize_round(h_pert, sub_dl, n_active=12, delta_init=0.2, max_iters=80)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.09, scale * 1.08)

    # Final polish
    if deadline - time.time() > 5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, c5_f = _lp_optimize_round(hf, deadline - 1, n_active=20, delta_init=0.08, max_iters=200)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
