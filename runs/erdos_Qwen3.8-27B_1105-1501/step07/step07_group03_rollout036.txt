    L = len(top_idx)
    if L < 2:
        return h
    raw = np.correlate(h, 1.0 - h, mode='full')
    # Build linear system: for i = 1..L-1, (g_0 - g_i) · δ = raw[i] - raw[0]
    # Plus: sum(δ) = 0
    # Total: L equations in n unknowns, take min-norm solution
    A_rows = []
    b_vals = []
    g0 = _lag_grad(h, top_idx[0], n)
    for i in range(1, L):

        A_rows.append(g0 - gi)
        b_vals.append(float(raw[top_idx[i]] - raw[top_idx[0]]))

    return delta

                    h[i] += best_a
                    h[j] -= best_a
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    total_moves += 1
                    # Recompute raw and grads
                    raw, top_idx, grads = _compute_top_grads(h, n_active)
                    cur_max = raw[top_idx[0]]
                    # Early exit if threshold varied
                    # (continue inner loops)
            # End j loop
        # End i loop
        if not improved:
            break
