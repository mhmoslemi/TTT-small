import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradients(h, active_lags):
    """Compute gradient of raw[r] w.r.t. h for each active lag r."""
    n = len(h)
    grads = []
    for r in active_lags:
        d = r - (n - 1)
        g = abs(d)
        grad = np.zeros(n)
        # (1-h[k+g]) * [k+g < n]
        valid1 = np.arange(n) + g < n
        grad[valid1] += (1.0 - h[valid1 + g])
        # -h[k-g] * [k-g >= 0]
        valid2 = np.arange(n) - g >= 0
        grad[valid2] -= h[valid2 - g]
        grads.append(grad)
    return np.array(grads)  # shape: (n_active, n)

def _lp_direction(h, n_active, radius, target):

    # This is different from raw[r](h).

    c = np.zeros(n + 1)
    c[-1] = 1.0

    b_ub = rhs

    # A_trust: identity for x, 0 for t

def _line_search_exact(h, d, raw, dx, a_max):
    """Exact line search for h + alpha * d, with d being a zero-sum direction."""
    # raw(h + a*d) = raw(h) + a * T - a^2 * Q
    # where T[r] = grad_raw[r] @ d (linear term)
    # and Q[r] = d-related quadratic term
    # Actually for bilinear: raw[r](h + a*d) = raw[r](h) + a * L[r] + a^2 * Q2[r]
    # where L[r] = sum of (d[i]*(1-h[j]) + h[i]*(-d[j])) for pairs (i,j) in lag r
    # and Q2[r] = -sum of d[i]*d[j] for pairs (i,j) in lag r
    
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # raw[r](a) = raw[r] + a*T[r] - a^2*Q[r]
    # We want to minimize max_r raw[r](a) for a in [0, a_max]
    
    # Also need to ensure 0 <= h[k] + a*d[k] <= 1
    # For d[k] > 0: a <= (1-h[k])/d[k]
    # For d[k] < 0: a <= h[k]/(-d[k])
    
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
    a_max = max(a_max, 1e-12)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, a_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(80):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    best_a = (lo + hi) / 2.0
    best_val = float(np.max(raw + best_a * T - best_a * best_a * Q)) * dx
    return best_a, best_val

def _lp_optimize(h, deadline, n_active_list=None, radius_init=0.15, stall_limit=60):
    """Main LP-based optimization loop."""
    if n_active_list is None:
        n_active_list = [20, 30, 40, 50, 60]
    
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    rng = np.random.default_rng(42)
    
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        
        # Try different numbers of active lags
        found = False
        for n_act in n_active_list:
            n_act = min(n_act, 2 * n - 1)
            x_lp, t_lp = _lp_direction(h, n_act, radius, target)
            if x_lp is None:
                continue
            
            d = x_lp - h
            d -= np.mean(d)  # Ensure zero sum for projection
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-12:
                continue
            
            # Line search from h in direction d
            # Scale d so max step is 1
            d_scaled = d / d_norm
            a_max = radius / d_norm  # trust region limits
            
            alpha, c5_try = _line_search_exact(h, d_scaled, raw, dx, a_max)
            
            if alpha < 1e-10:
                continue
            
            h_new = h + alpha * d_scaled
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.35)
                found = True
                break
        
        if not found:
            stall += 1
            radius *= 0.8
            if stall % 20 == 0 and radius < 0.05:
                # Perturb to escape
                h = _project(h + 0.05 * rng.normal(0, 1, n), target)
                cur_c5 = _exact_c5(h)
                radius = 0.15
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, raw, dx, a_max):
    """Exact 1D line search for bilinear objective along direction d."""
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
    a_max = max(a_max, 1e-12)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, a_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(60):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
    
    best_a = (lo + hi) / 2.0
    best_val = float(np.max(raw + best_a * T - best_a * best_a * Q)) * dx
    return best_a, best_val

def _lp_solve(h, n_active, radius, target):
    """Solve trust-region LP for minimax of active lags."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:n_active]
    
    # Build gradient matrix G (n_active x n)
    G = np.zeros((n_active, n))
    for i, r in enumerate(top_idx):
        d = abs(r - (n - 1))
        k = np.arange(n)
        valid1 = k + d < n
        valid2 = k - d >= 0
        G[i, valid1] += (1.0 - h[valid1 + d])
        if d > 0:
            G[i, valid2] -= h[valid2 - d]
    
    rhs = raw[top_idx] - G @ h
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((n_active, 1))])
    b_ub = rhs.copy()
    
    # Trust region
    I = np.eye(n)
    A_ub = np.vstack([A_ub, I, -I])
    b_ub = np.concatenate([b_ub, h + radius, -(h - radius)])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([target]),
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x[:n]

def _lp_optimize(h, deadline, radius_init=0.12, max_stall=50, n_active_range=(15, 50)):
    """LP-based minimax optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline and stall < max_stall:
        raw = np.correlate(h, 1.0 - h, mode='full')
        found = False
        
        # Try different active set sizes
        for n_act in range(n_active_range[0], n_active_range[1] + 1, 10):
            n_act = min(n_act, 2 * n - 1)
            x_lp = _lp_solve(h, n_act, radius, target)
            if x_lp is None:
                continue
            
            d = x_lp - h
            # Make zero-sum
            d -= np.mean(d)
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-12:
                continue
            
            d_unit = d / d_norm
            a_max = radius / d_norm
            
            alpha, c5_try = _line_search(h, d_unit, raw, dx, a_max)
            if alpha < 1e-10:
                continue
            
            h_new = h + alpha * d_unit
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.02, 0.30)
                found = True
                break
        
        if not found:
            stall += 1
            radius *= 0.85
            if radius < 0.03:
                # Perturb
                amp = 0.03 + 0.02 * rng.random()
                h_p = h + amp * rng.normal(0, 1, n)
                h_p = _project(h_p, target)
                h = h_p
                cur_c5 = _exact_c5(h)
                radius = 0.12
    
    return best_h, best_c5

def _grad_optimize(h, deadline, beta=20.0, radius=0.2, max_stall=40):
    """Softmax-weighted gradient optimization (from previous, kept as backup)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.time() < deadline and stall < max_stall:
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -60, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        
        # Weighted gradient
        g = np.zeros(n)
        u = 1.0 - h
        for r in range(2 * n - 1):
            wi = w[r]
            if wi < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += wi * u[:m]
                    g[:m] -= wi * h[d:]
            else:
                gp = -d
                m = n - gp
                if m > 0:
                    g[:m] += wi * u[gp:]
                    g[gp:] -= wi * h[:m]
        
        g -= np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            stall += 1
            continue
        
        d_unit = -g / gnorm
        a_max = radius / gnorm
        
        alpha, c5_try = _line_search(h, d_unit, raw, dx, a_max)
        if alpha < 1e-10:
            stall += 1
            continue
        
        h_new = h + alpha * d_unit
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.85
            if radius < 0.05:
                h = _project(h + 0.02 * rng.normal(0, 1, n), target)
                cur_c5 = _exact_c5(h)
                radius = 0.15
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    """Simple simulated annealing on binary solutions."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    
    return best_h, best_c5

def _pair_local(h, deadline, n_pairs=5000, seed=42):
    """Random pair swap with line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search(h, d, raw, dx, 0.4)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5
