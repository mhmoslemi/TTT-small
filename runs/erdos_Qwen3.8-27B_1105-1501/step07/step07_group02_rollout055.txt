        best_c5_try = cur_c5
        one_mh = 1.0 - h
        
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            # Quadratic approximation along the direction
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            # g(t) = max_k (S0[k] + t*T[k] - t^2*C[k])
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            # Minimize over t
            # g(t) = max_k (S0[k] + t*T[k] - t^2*C[k])
            # For a fixed active k: optimal t = T[k] / (2*C[k]) if C[k] > 0
            # Use golden section search
            t_lo_s = t_lo
            t_hi_s = t_hi
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1_t = t_hi_s - gr * (t_hi_s - t_lo_s)
            c2_t = t_lo_s + gr * (t_hi_s - t_lo_s)
            
            def g_of_t(t):
                return float(np.max(S0 + t * T - t * t * C) * dx)
            
            f1 = g_of_t(c1_t)
            f2 = g_of_t(c2_t)
            for _ in range(25):
                if t_hi_s - t_lo_s < 1e-10:
                    break
                if f1 < f2:
                    t_hi_s, c2_t, f2 = c2_t, c1_t, f1
                    c1_t = t_hi_s - gr * (t_hi_s - t_lo_s)
                    f1 = g_of_t(c1_t)
                else:
                    t_lo_s, c1_t, f1 = c1_t, c2_t, f2
                    c2_t = t_lo_s + gr * (t_hi_s - t_lo_s)
                    f2 = g_of_t(c2_t)
            t_opt = (t_lo_s + t_hi_s) / 2.0
            h_try = h.copy()
            h_try[sink] = np.clip(h_try[sink] + t_opt, 0.0, 1.0)
            h_try[source] = np.clip(h_try[source] - t_opt, 0.0, 1.0)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = best_step
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
