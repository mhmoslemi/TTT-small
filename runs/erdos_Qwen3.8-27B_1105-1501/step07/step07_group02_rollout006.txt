import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Projection onto {0<=x<=1, sum(x)=target}
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    if s > target:
        # Scale down
        ratio = target / s if s > 0 else 1.0
        v = v * ratio
    else:
        # Push up
        deficit = target - s
        capacity = n - s  # total room to grow
        if capacity > 0:
            v = v + deficit / capacity
        else:
            v = np.ones(n) * (target / n)
    v = np.clip(v, 0.0, 1.0)
    # Fix residual
    res = target - v.sum()
    if abs(res) > 1e-14:
        if res > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if mask.sum() > 0:
            v[mask] += res / mask.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _row_gradient(h, r):
    """Compute gradient of correlation row r w.r.t. h."""
    n = len(h)
    ell = r - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        m = n - ell
        g[ell:] += (1.0 - h[:m])
        g[:m] -= h[ell:]
    else:
        p = -ell
        m = n - p
        g[p:] += (1.0 - h[:m])
        g[:m] -= h[p:]
    return g

def _dense_direction(h, active_indices, target_val, weight_mode=0, ridge=1e-6):
    """
    Solve weighted least-norm:
    min d^T W d s.t. G d = b, sum(d) = 0
    
    where G has rows = row gradients of active rows, b_i = target_val - S[i]
    weight_mode: 0=identity, 1=variance-aware
    """
    n = len(h)
    K = len(active_indices)
    
    # Build gradient matrix G (K x n)
    G = np.zeros((K, n))
    for i, r in enumerate(active_indices):
        G[i] = _row_gradient(h, r)
    
    # Target values
    S = np.correlate(h, 1.0 - h, mode='full')
    b = target_val - S[active_indices]
    
    # Weight vector
    if weight_mode == 1:
        w = 1.0 / (h * (1.0 - h) + 1e-3)
    else:
        w = np.ones(n)
    
    # Build (K+1) x (K+1) system
    # A = G * diag(1/w)  (K x n)
    A = G * (1.0 / w)[None, :]
    
    # System: [A A^T  A 1] [alpha]   [b]
    #         [1^T A  n   ] [beta ] = [0]
    M11 = A @ A.T + ridge * np.eye(K)
    M12 = A @ np.ones(n)
    M22 = float(np.sum(1.0 / w))
    
    M = np.zeros((K + 1, K + 1))
    M[:K, :K] = M11
    M[:K, K] = M12
    M[K, :K] = M12
    M[K, K] = M22
    
    rhs = np.zeros(K + 1)
    rhs[:K] = b
    
    try:
        sol = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        return None
    
    alpha = sol[:K]
    beta = sol[K]
    
    # d = (1/w) * (G^T alpha + beta)
    d = (1.0 / w) * (G.T @ alpha + beta)
    
    # Zero-sum correction (should be approximately zero already)
    d -= d.mean()
    
    # Check if meaningful
    if np.max(np.abs(d)) < 1e-12:
        return None
    if not np.all(np.isfinite(d)):
        return None
    
    return d

def _feasible_interval(h, d, move_cap):
    """Find feasible t interval for h + t*d staying in [0,1]."""
    n = len(h)
    t_lo = -move_cap
    t_hi = move_cap
    
    nz = np.abs(d) > 1e-14
    if nz.any():
        dn = d[nz]
        hn = h[nz]
        pos = dn > 0
        neg = dn < 0
        if pos.any():
            t_lo = max(t_lo, (0.0 - hn[pos]) / dn[pos]).max() if pos.any() else t_lo
            t_hi = min(t_hi, (1.0 - hn[pos]) / dn[pos]).min() if pos.any() else t_hi
        if neg.any():
            t_lo = max(t_lo, (1.0 - hn[neg]) / dn[neg]).max() if neg.any() else t_lo
            t_hi = min(t_hi, (0.0 - hn[neg]) / dn[neg]).max() if neg.any() else t_hi
    
    if t_lo >= t_hi:
        return None
    return t_lo, t_hi

def _line_search(h, S, d, move_cap, deadline):
    """
    Line search: minimize max_r (S_r + t*U_r - t^2*B_r) over feasible t.
    Returns (best_t, best_value) or (0, S.max()).
    """
    one_minus_h = 1.0 - h
    U = np.correlate(d, one_minus_h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    interval = _feasible_interval(h, d, move_cap)
    if interval is None:
        return 0.0, float(np.max(S))
    
    t_lo, t_hi = interval
    cur_val = float(np.max(S))
    
    # Grid search
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Add 0 if in range
    if t_lo < 0 < t_hi:
        ts = np.unique(np.concatenate([ts, [0.0]]))
    
    vals = np.zeros(len(ts))
    for i, t in enumerate(ts):
        vals[i] = float(np.max(S + t * U - t * t * B))
    
    best_idx = int(np.argmin(vals))
    best_val = float(vals[best_idx])
    best_t = float(ts[best_idx])
    
    if best_val >= cur_val:
        return 0.0, cur_val
    
    # Golden section refinement
    if best_idx > 0 and best_idx < len(ts) - 1:
        lo = float(ts[max(0, best_idx - 1)])
        hi = float(ts[min(len(ts)-1, best_idx + 1)])
    else:
        lo, hi = t_lo, t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    
    for _ in range(25):
        if time.time() > deadline:
            break
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    
    mid = (lo + hi) / 2.0
    f_mid = float(np.max(S + mid * U - mid * mid * B))
    if f_mid < best_val:
        best_val = f_mid
        best_t = mid
    
    return best_t, best_val

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    s = float(v.sum())
    if s < 1e-15:
        v[:] = target / n
        return v
    if s > target:
        v *= target / s
    else:
        deficit = target - s
        cap = n - s
        if cap > 1e-15:
            v += deficit / cap
    v = np.clip(v, 0.0, 1.0)
    res = target - float(v.sum())
    if abs(res) > 1e-13:
        if res > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            v[mask] += res / cnt
    return np.clip(v, 0.0, 1.0)

def _row_grad(h, ell):
    n = len(h)
    g = np.zeros(n)
    if ell >= 0:
        m = n - ell
        if m > 0:
            g[ell:] += (1.0 - h[:m])
            g[:m] -= h[ell:]
    else:
        p = -ell
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def _dense_dir(h, S, active_idx, target_val, wmode, ridge):
    n = len(h)
    K = len(active_idx)
    G = np.zeros((K, n))
    for i, r in enumerate(active_idx):
        ell = r - (n - 1)
        G[i] = _row_grad(h, ell)
    b = target_val - S[active_idx]
    if wmode == 1:
        w = 1.0 / (h * (1.0 - h) + 5e-3)
    else:
        w = np.ones(n)
    inv_w = 1.0 / w
    A = G * inv_w[None, :]
    M11 = A @ A.T + ridge * np.eye(K)
    M12 = A @ np.ones(n)
    M22 = float(np.sum(inv_w))
    M = np.zeros((K + 1, K + 1))
    M[:K, :K] = M11
    M[:K, K] = M12
    M[K, :K] = M12
    M[K, K] = M22
    rhs = np.zeros(K + 1)
    rhs[:K] = b
    try:
        sol = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(M, rhs, rcond=None)[0]
        except Exception:
            return None
    d = inv_w * (G.T @ sol[:K] + sol[K])
    d -= d.mean()
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-13:
        return None
    return d

def _fs_interval(h, d, cap):
    t_lo = -cap
    t_hi = cap
    nz = np.abs(d) > 1e-14
    if not nz.any():
        return None
    dn = d[nz]
    hn = h[nz]
    pos = dn > 0
    neg = ~pos
    if pos.any():
        t_lo = max(t_lo, float(np.min(-hn[pos] / dn[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - hn[pos]) / dn[pos])))
    if neg.any():
        t_lo = max(t_lo, float(np.min((1.0 - hn[neg]) / dn[neg])))
        t_hi = min(t_hi, float(np.min(-hn[neg] / dn[neg])))
    if t_lo >= t_hi - 1e-14:
        return None
    return t_lo, t_hi

def _line_search(h, S, d, cap, deadline):
    omh = 1.0 - h
    U = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    iv = _fs_interval(h, d, cap)
    if iv is None:
        return 0.0, float(np.max(S))
    t_lo, t_hi = iv
    cur = float(np.max(S))
    n_g = 36
    ts = np.linspace(t_lo, t_hi, n_g)
    if t_lo < 0.0 < t_hi:
        ts = np.unique(np.append(ts, 0.0))
    vals = np.array([float(np.max(S + t * U - t * t * B)) for t in ts])
    bi = int(np.argmin(vals))
    bv = float(vals[bi])
    bt = float(ts[bi])
    if bv >= cur:
        return 0.0, cur
    if bi > 0 and bi < len(ts) - 1:
        lo = float(ts[bi - 1])
        hi = float(ts[bi + 1])
    else:
        lo, hi = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(22):
        if time.time() > deadline or hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    mid = (lo + hi) / 2.0
    fm = float(np.max(S + mid * U - mid * mid * B))
    if fm < bv:
        bv, bt = fm, mid
    return bt, bv

def _evaluate_move(h, t, d, S, U, B):
    h_new = h + t * d
    if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
        return None
    h_new = np.clip(h_new, 0.0, 1.0)
    S_new = S + t * U - t * t * B
    val = float(np.max(S_new))
    return h_new, val

def _pair_grad(h, S, w):

    L = 2 * n - 1
    G = np.zeros((L, n))
    for ell in range(-(n - 1), n):
