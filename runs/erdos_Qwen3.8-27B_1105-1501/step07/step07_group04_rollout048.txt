            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q2 = np.correlate(d, d, mode='full')
        alpha, new_c5 = _line_search(corr_raw, L1, Q2, dx, radius, t_hi)
        if new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur - 1e-14:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.25)
            else:
                stall += 1
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-7:
                break
    return best_h, best_c5

def _binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_search(h_bin, deadline, max_iters=100000, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    best_stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = min(50, len(ones) * len(zeros))
        best_delta = 0.0
        best_move = None
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_c5, _ = _binary_c5(cur)
            delta = new_c5 - cur_c5
            cur[oi] = 1.0
            cur[oz] = 0.0
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
        else:
            stall += 1
            best_stall += 1
    return best_h, best_c5

def _sa_search(h_bin, deadline, max_iters=100000, temp_init=0.004, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, _ = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    best_stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 10000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5, _ = _binary_c5(cur)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
            T = max(0.0003, T * 0.9998)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            best_stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_refine(h, deadline, radius_init=0.05):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(2000):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _subgradient_dir(h, corr, dx, n_top=15)
        if np.max(np.abs(d)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q2 = np.correlate(d, d, mode='full')
        alpha, new_c5 = _line_search(corr, L1, Q2, dx, radius, t_hi)
        if new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur - 1e-14:
                h = h_new
                cur = actual
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.20)
            else:
                stall += 1
        else:
            stall += 1
            radius *= 0.72
            if radius < 1e-7:
                break
    return best_h, best_c5

def _pair_refine(h, deadline, n_trials=5000, radius=0.15, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, (1.0 - h[i]) / 1.0)
        if np.any(neg):
            t_hi = min(t_hi, h[j] / 1.0)
        if t_hi < 1e-14:
            stall += 1
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q2 = np.correlate(d, d, mode='full')
        alpha, new_c5 = _line_search(corr, L1, Q2, dx, radius, t_hi)
        if new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur - 1e-14:
                h = h_new
                cur = actual
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP-based minimax descent (40% time)
    d1 = t0 + budget_s * 0.40
    sub_dl = min(d1, time.time() + budget_s * 0.15)
    h1, c5_1 = _lp_optimize(h_init, sub_dl, radius_init=0.08, max_iter=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Continue LP with smaller radius
    if time.time() < d1:
        sub_dl = min(d1, time.time() + budget_s * 0.15)
        h1, c5_1 = _lp_optimize(best_h, sub_dl, radius_init=0.04, max_iter=2000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Another LP pass with different params
    if time.time() < d1:
        sub_dl = min(d1, time.time() + budget_s * 0.10)
        h1, c5_1 = _lp_optimize(best_h, sub_dl, radius_init=0.06, max_iter=1500)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Subgradient refinement (10% time)
    d2 = t0 + budget_s * 0.50
    if time.time() < d2:
        sub_dl = min(d2, time.time() + budget_s * 0.08)
        h2, c5_2 = _grad_refine(best_h, sub_dl, radius_init=0.04)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Binary search (25% time)
    d3 = t0 + budget_s * 0.75
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.10)
        h_b, c5_b = _binary_search(best_h, sub_dl, max_iters=80000, seed=201)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.07)
        h_b, c5_b = _sa_search(best_h, sub_dl, max_iters=80000, temp_init=0.003, seed=302)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_search(best_h, sub_dl, max_iters=50000, seed=403)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # LP on binary result
    if time.time() < d3:
        h_pol = _project(best_h.copy(), n0 / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _lp_optimize(h_pol, sub_dl, radius_init=0.05, max_iter=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Multi-n exploration (15% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, -6, 6]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.06, max_iter=600)
        if time.time() < d4 - 5:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_search(h_b, sub_dl2, max_iters=30000, seed=1000 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final intensive refinement (5% time)
    d5 = deadline
    n_best = len(best_h)

    # LP polish
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h5 = _project(best_h.copy(), n_best / 2.0)
        h5, c5_5 = _lp_optimize(h5, sub_dl, radius_init=0.03, max_iter=500)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Pair refinement
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h5 = _project(best_h.copy(), n_best / 2.0)
        h5, c5_5 = _pair_refine(h5, sub_dl, n_trials=3000, radius=0.12, seed=999)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Basin hopping
    if time.time() < d5 - 5:
        scale = 0.015
        for _ in range(30):
            if time.time() > d5 - 3:
                break
            h_pert = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d5, time.time() + budget_s * 0.003)
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.03, max_iter=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.05, scale * 1.1)

    # Final binary + LP
    if time.time() < d5 - 3:
        h_b = _to_binary(best_h, len(best_h))
        sub_dl = min(d5 - 2, time.time() + budget_s * 0.01)
        h_b, c5_b = _binary_search(h_b, sub_dl, max_iters=30000, seed=7777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5 - 1, time.time() + budget_s * 0.005)
        h_pol, c5_pol = _lp_optimize(h_pol, sub_dl, radius_init=0.02, max_iter=300)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
