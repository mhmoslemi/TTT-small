def _grad_rows(h, lags):

    u = 1.0 - h

                G[idx, :m] += -h[d:d + m]

            m = n - g

def _active_lp(h, scores, radius, n_active=14, gap=0.005):

    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    try:

            stall = 0
            radius = min(radius * 1.05, 0.3)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.3
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=150):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(max(1, len(ones) // 4)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.25)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _rotate(h, shift):
    n = len(h)
    return np.roll(h, shift)

def _freq_perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.4, 1.0, size=3)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amplitude
    p += amplitude * 0.3 * rng.standard_normal(n)
    return _project(h + p, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: SLSQP smooth optimization from parent (18%)
    d1 = t0 + budget_s * 0.18
    betas_full = [0.12, 0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]
    h1, c5_1 = _smooth_opt(h_init, betas_full, d1, maxiter=80)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-start SLSQP from diverse perturbations (28%)
    d2 = t0 + budget_s * 0.46
    n_curr = len(best_h)
    pool = [best_h.copy()]
    # Generate diverse starts
    shifts = [7, 13, 23, 37, 53, 71, 97, 127, 150]
    for s in shifts[:6]:
        pool.append(_rotate(best_h, s % n_curr))
    pool.append(best_h[::-1].copy())
    for _ in range(4):
        pool.append(_freq_perturb(best_h, rng, rng.uniform(0.03, 0.08)))
    bn = _to_binary(best_h)
    bn_smooth = _project(bn + 0.05 * rng.standard_normal(n_curr), n_curr / 2.0)
    pool.append(bn_smooth)

    betas_short = [0.06, 0.02, 0.006, 0.002]
    for p_idx in range(len(pool)):
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        hp, cp = _smooth_opt(pool[p_idx], betas_short, sub_dl, maxiter=50)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
            pool = [best_h.copy()]
            for s in [11, 29, 47]:
                pool.append(_rotate(best_h, s % n_curr))
            pool.append(_freq_perturb(best_h, rng, 0.04))

    # Phase 3: Active LP polish (22%)
    d3 = t0 + budget_s * 0.68
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c5_3 = _local_opt(h3, d3, radius_init=0.10, max_iters=2000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b = _project(best_h.copy(), len(best_h) / 2.0)
        h3b, c5_3b = _pair_sweep(h3b, d3, max_rounds=80)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Smooth basin hopping (20%)
    d4 = t0 + budget_s * 0.88
    scale = 0.025
    n_restarts = 0
    while time.time() < d4 and n_restarts < 300:
        n_restarts += 1
        h_pert = _freq_perturb(best_h, rng, scale)
        sub_dl = min(d4, time.time() + budget_s * 0.004)
        betas_bh = [0.04, 0.012, 0.004]
        h_t, c5_t = _smooth_opt(h_pert, betas_bh, sub_dl, maxiter=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 5: Final polish (8%)
    d5 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, min(d5, time.time() + budget_s * 0.03), radius_init=0.04, max_iters=800)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d5 - 2:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_f2, c5_f2 = _pair_sweep(h_f2, min(d5, time.time() + budget_s * 0.02), max_rounds=40)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
    if time.time() < d5 - 3:
        h_f3 = _project(best_h.copy(), len(best_h) / 2.0)
        betas_final = [0.004, 0.001, 0.0003]
        h_f3, c5_f3 = _smooth_opt(h_f3, betas_final, d5, maxiter=50)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
