def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):

    delta[idx_pos[mask1]] += h[a - Ls[mask1]]
    # Term 2: I(b+L<n) * (1-h[b+L])
    mask2 = b + Ls < n
    delta[idx_pos[mask2]] += (1.0 - h[b + Ls[mask2]])
    # Term 3: -I(a+L<n) * (1-h[a+L])
    mask3 = a + Ls < n
    delta[idx_pos[mask3]] -= (1.0 - h[a + Ls[mask3]])
    # Term 4: -I(b-L>=0) * h[b-L]
    mask4 = Ls <= b
    delta[idx_pos[mask4]] -= h[b - Ls[mask4]]
    
    # For L > 0: raw[n-1-L]
    # Change = I(a+L<n)*h[a+L] + I(b>=L)*(1-h[b-L]) - I(a>=L)*(1-h[a-L]) - I(b+L<n)*h[b+L]
    idx_neg = n - 1 - Ls  # indices in raw
    # Term 1: I(a+L<n) * h[a+L]
    mask1 = a + Ls < n
    delta[idx_neg[mask1]] += h[a + Ls[mask1]]
    # Term 2: I(b>=L) * (1-h[b-L])
    mask2 = Ls <= b
    delta[idx_neg[mask2]] += (1.0 - h[b - Ls[mask2]])
    # Term 3: -I(a>=L) * (1-h[a-L])
    mask3 = Ls <= a
    delta[idx_neg[mask3]] -= (1.0 - h[a - Ls[mask3]])
    # Term 4: -I(b+L<n) * h[b+L]
    mask4 = b + Ls < n
    delta[idx_neg[mask4]] -= h[b + Ls[mask4]]
    
    return delta

def _smooth_max_grad(h, raw, beta):
    """Compute gradient of smooth max of raw correlation."""
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        # Just use the argmax
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    w /= wsum
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-8:
            continue
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += w[r] * (1.0 - h[:m])
                g[:m] -= w[r] * h[tau:tau + m]
        else:
            gg = -tau
            m = n - gg
            if m > 0:
                g[:m] += w[r] * (1.0 - h[gg:])
                g[gg:] -= w[r] * h[:m]
    g -= float(np.mean(g))
    return g

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        gg = -tau
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx
    
    ng = 50
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _make_dirs(h, g, raw, rng, n_max=8):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    r_max = int(np.argmax(raw))
    gs = _grad_single(h, r_max)
    gsn = np.linalg.norm(gs)
    if gsn > 1e-12:
        d = gs / gsn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for blk in [4, 8, 16]:
        if n % blk != 0:
            continue
        nblk = n // blk
        gb = g.reshape(nblk, blk).mean(axis=1)
        d = np.tile(gb, (1, blk)).reshape(n).copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for freq in [1, 2, 3]:
        x = np.arange(n, dtype=float)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    idx_s = np.argsort(g)
    for s in idx_s[:3]:
        for k in idx_s[-3:]:
            if s == k:
                continue
            d = np.zeros(n)
            d[s] = 1.0
            d[k] = -1.0
            d -= np.mean(d)
            dirs.append(d)
            break
    return dirs[:n_max]

def _opt_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.18, max_stall=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = float(total_iters) / float(total_iters + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 2.0)
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        g = _smooth_max_grad(h, raw, beta)
        dirs = _make_dirs(h, g, raw, rng, 8)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.06, 0.45)
            else:
                stall += 1
                radius = max(0.04, radius * 0.75)
        else:
            stall += 1
            radius = max(0.04, radius * 0.78)
    return best_h, best_c5

def _binary_opt_informed(h_bin, deadline, max_iter=100000, T_init=0.005):
    """Informed binary optimization using correlation structure to guide swaps."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    raw = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(raw) * dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    best_raw = raw.copy()
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500 or len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose swap strategy
        if rng.random() < 0.7:
            # Informed: target the max lag
            r_max = int(np.argmax(raw))
            tau = r_max - (n - 1)
            if tau >= 0:
                # Positive lag: raw counts h[i]=1, h[i+tau]=0 for i in [0, n-1-tau]
                # Find positions i where h[i]=1 and h[i+tau]=0
                m = n - tau
                pairs = [(i, i + tau) for i in range(m) if cur[i] > 0.5 and cur[i + tau] <= 0.5]
                if pairs:
                    # Try to break one of these pairs
                    oi, oz = pairs[int(rng.integers(0, len(pairs)))]
                    # Find a zero position that's not oz to swap with oi
                    candidates = [z for z in zeros if z != oz]
                    if not candidates:
                        candidates = zeros
                    oz_new = candidates[int(rng.integers(0, len(candidates)))]
                    oi_try, oz_try = oi, oz_new
                else:
                    oi = ones[int(rng.integers(0, len(ones)))]
                    oz = zeros[int(rng.integers(0, len(zeros)))]
                    oi_try, oz_try = oi, oz
            else:
                d = -tau
                # Negative lag: raw counts h[i]=1, h[i-d]=0 for i in [d, n-1]
                pairs = [(i, i - d) for i in range(d, n) if cur[i] > 0.5 and cur[i - d] <= 0.5]
                if pairs:
                    oi, oz = pairs[int(rng.integers(0, len(pairs)))]
                    candidates = [z for z in zeros if z != oz]
                    if not candidates:
                        candidates = zeros
                    oz_new = candidates[int(rng.integers(0, len(candidates)))]
                    oi_try, oz_try = oi, oz_new
                else:
                    oi = ones[int(rng.integers(0, len(ones)))]
                    oz = zeros[int(rng.integers(0, len(zeros)))]
                    oi_try, oz_try = oi, oz
        else:
            # Random swap
            oi_try = ones[int(rng.integers(0, len(ones)))]
            oz_try = zeros[int(rng.integers(0, len(zeros)))]
        
        if oi_try == oz_try:
            stall += 1
            continue
        
        # Evaluate swap
        cur[oi_try] = 0.0
        cur[oz_try] = 1.0
        raw_new = np.correlate(cur, 1.0 - cur, mode='full')
        new_c5 = float(np.max(raw_new) * dx)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            raw = raw_new
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_raw = raw.copy()
                stall = 0
            T = max(0.0005, T * 0.9998)
        else:
            cur[oi_try] = 1.0
            cur[oz_try] = 0.0
            stall += 1
    
    return best_h, best_c5

def _pair_sweep_guided(h_bin, deadline):
    """Pair sweep that targets the max correlation lag."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    raw = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(raw) * dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for oi in ones:
        if time.time() > deadline:
            break
        # Find the top 3 lags and prefer zeros that reduce them
        top_lags = np.argsort(raw)[-3:]
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            raw_try = np.correlate(cur, 1.0 - cur, mode='full')
            c_try = float(np.max(raw_try) * dx)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                raw = raw_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
    return best_h, best_c5

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _continuous_pair_opt(h, deadline, n_iter=3000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _make_periodic(n, period, duty):
    """Create a periodic binary pattern with given period and duty cycle."""
    h = np.zeros(n)
    pattern = np.zeros(period)
    pattern[:int(period * duty)] = 1.0
    for i in range(n):
        h[i] = pattern[i % period]
    return _project(h, n / 2.0)

def _make_blocks(n, n_blocks):
    """Create a block pattern."""
    h = np.zeros(n)
    block_size = n // (2 * n_blocks)
    for b in range(n_blocks):
        start = 2 * b * block_size
        end = start + block_size
        if end <= n:
            h[start:end] = 1.0
    return _project(h, n / 2.0)
