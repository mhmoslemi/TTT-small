def _find_opt_delta(S0, g, lo, hi):
    # Find delta in [lo, hi] minimizing max(S0 + delta * g)
    # f(delta) = max(S0 + delta * g) is convex
    
    # Quick check: if g is all zero, any delta works
    if np.allclose(g, 0):
        return 0.0
    
    # Evaluate at boundaries
    f_lo = float(np.max(S0 + lo * g))
    f_hi = float(np.max(S0 + hi * g))
    
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    
    # Binary search using subgradient
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = S0 + mid * g
        mx = float(np.max(vals))
        # Subgradient: set of g[i] where vals[i] = mx
        active = vals >= mx - 1e-12
        g_active = g[active]
        if len(g_active) == 0:
            break
        gmin = float(np.min(g_active))
        gmax = float(np.max(g_active))
        if gmin <= 0 <= gmax:
            # 0 is in subgradient, mid is optimal
            return mid
        elif gmin > 0:
            # All active slopes positive, need to decrease delta
            hi = mid
        else:
            # All active slopes negative, need to increase delta
            lo = mid
    
    # Check boundaries and mid
    best_d = (lo + hi) / 2.0
    best_f = float(np.max(S0 + best_d * g))
    f_lo2 = float(np.max(S0 + lo * g))
    f_hi2 = float(np.max(S0 + hi * g))
    if f_lo2 < best_f:
        best_d = lo
    elif f_hi2 < best_f:
        best_d = hi
    return best_d

def _active_gradient(h, R, tau):
    n = len(h)
    maxR = float(np.max(R))
    w = np.exp((R - maxR) / tau)
    sw = float(np.sum(w))
    if sw < 1e-15:
        return np.zeros(n)
    w /= sw
    g = np.zeros(n)
    for r in range(len(R)):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d == 0:
            g += w[r] * (1.0 - 2.0 * h)
        else:
            b = abs(d)
            g[b:] += w[r] * (1.0 - h)[:n - b]
            g[:n - b] -= w[r] * h[b:]
    return g
