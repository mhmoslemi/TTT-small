def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

    S0 = np.correlate(h, u, mode='full')

def _lp_direction(h, scores, radius):
    """Full-lag LP direction finding."""
    n = len(h)
    G = _build_G(h)
    n_lags = 2 * n - 1
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        if lo > hi:
            lo, hi = 0.0, 0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})

            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-14:
                # Normalize to have max abs = radius
                scale = float(np.max(np.abs(d)))
                d = d * (radius / scale)
                return d
    except Exception:
        pass
    return None

def _grad_direction(h, scores, beta, radius):
    """Weighted gradient direction (log-sum-exp subgradient)."""
    n = len(h)
    G = _build_G(h)
    n_lags = 2 * n - 1
    
    shifted = beta * (scores - np.max(scores))
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    g = G.T @ w  # shape (n,)
    # Center to satisfy sum constraint
    g -= np.mean(g)
    # Clip to box
    g = np.clip(g, -radius, radius)
    # Respect bounds
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= np.mean(g)
    if np.max(np.abs(g)) < 1e-14:
        return None
    return g

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0

    n = len(h_bin)

    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    T = 0.002

        if time.time() > deadline:
            break
        if stall > 300:
            break
        
        # Random pair flip
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i1 = int(rng.integers(len(ones)))
        i0 = int(rng.integers(len(zeros)))
        oi, oz = ones[i1], zeros[i0]
        
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    
    return best_h, best_c5

                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, t_max):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_feas = t_max
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_feas = min(t_feas, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_feas = min(t_feas, float(np.min(h[neg] / (-d[neg]))))
    if t_feas < 1e-14:
        return 0.0, float(np.max(S0)) * dx

    def f_at(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx

    grid_t = np.linspace(0, t_feas, 30)
    grid_f = np.array([f_at(t) for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(29, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = f_at(c1)
    f2 = f_at(c2)
    for _ in range(25):
        if hi_b - lo_b < 1e-12 * max(t_feas, 1e-10):
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = f_at(c1)
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = f_at(c2)
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = f_at(opt_t)
    cands = [(opt_t, opt_val), (0.0, f_at(0.0)), (t_feas, f_at(t_feas))]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _lp_direction(h, scores, radius):
    n = len(h)
    G = _build_G(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        if lo > hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})
        if res.success:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-14:
                scale = float(np.max(np.abs(d)))
                return d * (radius / scale)
    except Exception:
        pass
    return None

def _grad_direction(h, scores, beta, radius):
    n = len(h)
    G = _build_G(h)
    n_lags = 2 * n - 1
    shifted = beta * (scores - np.max(scores))
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    g = G.T @ w
    g -= np.mean(g)
    g = np.clip(g, -radius, radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= np.mean(g)
    if np.max(np.abs(g)) < 1e-14:
        return None
    return g

def _local_opt(h, deadline, radius_init=0.1, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    beta = 15.0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        d = _lp_direction(h, scores, radius)
        if d is None:
            d = _grad_direction(h, scores, beta, radius)
        if d is None:
            stall += 5
            radius *= 0.7
            beta *= 1.5
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
            beta = max(beta * 0.97, 5.0)
        else:
            stall += 1
            radius *= 0.75
            beta *= 1.3
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=60000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    stall = 0
    T = 0.0025
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = int(rng.integers(len(ones)))
        i0 = int(rng.integers(len(zeros)))
        oi, oz = ones[i1], zeros[i0]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200, n_trials_per_round=8):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        improved = False
        for _ in range(n_trials_per_round):
            if time.time() > deadline:
                break
            idx = np.random.default_rng(r * 1000 + _).integers(0, n, size=2)
            i, j = int(idx[0]), int(idx[1])
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.2)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _mechanical_seed(n, p, q, phase=0.0):
    x = (np.arange(n) * p / q + phase / q) % 1.0
    h = (x < 0.5).astype(float)
    return _project(h, n / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    n_freqs = 5
    for _ in range(n_freqs):
        f = rng.uniform(0.3, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _penalty_opt(h, scores, beta, radius, n_steps=50):
    """Gradient descent on log-sum-exp penalty with projection."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    G = _build_G(h_cur)
    lr = radius / np.sqrt(n)
    for step in range(n_steps):
        s = _scores(h_cur)
        shifted = beta * (s - np.max(s))
        shifted = np.clip(shifted, -50.0, 0.0)
        w = np.exp(shifted)
        sw = float(np.sum(w))
        if sw > 1e-30:
            w /= sw
        g = G.T @ w
        g -= np.mean(g)
        h_cur = h_cur - lr * g
        h_cur = _project(h_cur, target)
        new_c5 = _exact_c5(h_cur)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_cur.copy()
        lr *= 0.95
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Full-lag LP local optimization (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _local_opt(h_init, d1, radius_init=0.12, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution exploration (18%)
    d2 = t0 + budget_s * 0.53
    tried_ns = set([len(best_h)])
    n_base = len(best_h)
    for dn in [-40, -30, -20, -10, 10, 20, 30, 40, -16, 16, -8, 8]:
        if time.time() > d2:
            break
        n_try = n_base + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.10, max_iters=1500)
        if c5_r < best_c5 + 0.003 and time.time() < d2 - 3:
            sub_dl2 = min(d2, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _pair_sweep(h_r, sub_dl2, max_rounds=40, n_trials_per_round=6)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_base = n_try

    # Phase 3: Binary optimization (15%)
    d3 = t0 + budget_s * 0.68
    n_b = len(best_h)
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000)
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_bin2, c5_bin2 = _pair_sweep(h_bin, sub_dl2, max_rounds=60, n_trials_per_round=10)
        if c5_bin2 < c5_bin:
            h_bin, c5_bin = h_bin2, c5_bin2
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d3 - 2:
        h_pol = _project(h_bin.copy(), n_b / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.025)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl3, radius_init=0.06, max_iters=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Structured seeds + basin hopping (22%)
    d4 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    # Try mechanical seeds
    for p_val in [7, 11, 13, 17, 19, 23, 29, 31]:
        if time.time() > d4:
            break
        for q_val in [2*p_val, 2*p_val+1, 2*p_val-1]:
            if time.time() > d4:
                break
            if q_val <= 0 or q_val > 200:
                continue
            h_m = _mechanical_seed(n_cur, p_val, q_val)
            h_m = _smooth_perturb(h_m, 0.05, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.008)
            h_t, c5_t = _local_opt(h_m, sub_dl, radius_init=0.08, max_iters=6
