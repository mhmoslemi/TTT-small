def _lp_direction(h, active_lags, max_step=1.0):
    n = len(h)
    K = len(active_lags)
    if K == 0:
        return None
    
    # Build gradient matrix A (K x n)
    A = np.zeros((K, n))
    norms = np.zeros(K)
    for i, l in enumerate(active_lags):
        g = _lag_grad(h, l)
        A[i] = g
        norms[i] = np.linalg.norm(g)
        if norms[i] < 1e-15:

    if n_top > 4:
        half = n_top // 2
        s2 = _lp_direction(h, active_lags[:half], max_step=max_step)
        if s2 is not None and s2 is not s:
            dirs.append(('lp_half', s2))
    g_agg = np.zeros(n)
    for l in active_lags[:min(5, n_top)]:
        g_agg += _lag_grad(h, l)
    nrm = np.linalg.norm(g_agg)
    if nrm > 1e-10:
        dirs.append(('agg', g_agg / nrm * max_step))
    worst_lag = int(top_idx[0] - (n - 1))
    gw = _lag_grad(h, worst_lag)
    nrmw = np.linalg.norm(gw)
    if nrmw > 1e-10:
        dirs.append(('worst', -gw / nrmw * max_step))
    return dirs

def _local_opt_lp(h, deadline, top_k=12, stall_limit=60):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c = _exact_c5(h)
    best_h = h.copy()
    best_c = cur_c
    stall = 0
    scale = 0.3
    rng = np.random.default_rng(777)
    while time.time() < deadline - 0.3:
        if stall > stall_limit:
            break
        dirs = _gen_dirs_lp(h, top_k=top_k, max_step=scale)
        best_new = cur_c
        best_a = 0.0
        best_d = None
        for name, d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm * scale
            a, c_try = _quad_line_search(h, d, dx)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur_c - 1e-14:
            h_new = h + best_a * best_d
            h_new = _project(h_new, target)
            cur_c = _exact_c5(h_new)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h_new.copy()
            h = h_new
            stall = 0
            scale = min(scale * 1.03, 0.6)
        else:
            stall += 1
            scale *= 0.8
            if scale < 0.005:
                break
            if stall % 20 == 0:
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur_c - 1e-12:
                    h = hp
                    cur_c = cp
                    scale = 0.15
                    stall = 0
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c = _exact_c5(h)
    best_h = h.copy()
    best_c = cur_c
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        if h[i] > h[j]:
            d[i] = -1.0
            d[j] = 1.0
        else:
            d[i] = 1.0
            d[j] = -1.0
        a, c_try = _quad_line_search(h, d, dx)
        if c_try < cur_c - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c_new = _exact_c5(h_new)
            if c_new < cur_c - 1e-14:
                h = h_new
                cur_c = c_new
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, max_it=20000, T_init=0.003):
    n = len(hb)
    ones_list = list(np.where(hb > 0.5)[0])
    zeros_list = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200 or (not ones_list or not zeros_list):
            break
        i = int(rng.integers(0, len(ones_list)))
        j = int(rng.integers(0, len(zeros_list)))
        oi = ones_list[i]
        oz = zeros_list[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones_list = list(np.where(cur > 0.5)[0])
            zeros_list = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _smooth_grad_dir(h, beta=30.0):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(S))
    v = np.clip(beta * (S - mx), -60.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _gen_dirs_mixed(h, rng, scale=0.3):
    n = len(h)
    dirs = []
    corr = _corr_raw(h)
    n_top = min(12, len(corr))
    top_idx = np.argsort(-corr)[:n_top]
    active_lags = [int(t - (n - 1)) for t in top_idx]
    s = _lp_direction(h, active_lags, max_step=scale)
    if s is not None:
        dirs.append(s)
    g_s = _smooth_grad_dir(h, 30.0)
    nrm = np.linalg.norm(g_s)
    if nrm > 1e-10:
        dirs.append(g_s / nrm * scale)
    g_s2 = _smooth_grad_dir(h, 60.0)
    nrm2 = np.linalg.norm(g_s2)
    if nrm2 > 1e-10:
        dirs.append(g_s2 / nrm2 * scale)
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(1.0, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
    d -= np.mean(d)
    nrm3 = np.linalg.norm(d)
    if nrm3 > 1e-10:
        dirs.append(d / nrm3 * scale)
    return dirs

def _local_opt_mixed(h, deadline, stall_limit=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c = _exact_c5(h)
    best_h = h.copy()
    best_c = cur_c
    rng = np.random.default_rng(555)
    scale = 0.25
    stall = 0
    while time.time() < deadline - 0.3:
        if stall > stall_limit:
            break
        dirs = _gen_dirs_mixed(h, rng, scale)
        best_new = cur_c
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm * scale
            a, c_try = _quad_line_search(h, d, dx)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur_c - 1e-14:
            h_new = h + best_a * best_d
            h_new = _project(h_new, target)
            cur_c = _exact_c5(h_new)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h_new.copy()
            h = h_new
            stall = 0
            scale = min(scale * 1.04, 0.5)
        else:
            stall += 1
            scale *= 0.78
            if scale < 0.005:
                break
            if stall % 25 == 0:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur_c - 1e-12:
                    h = hp
                    cur_c = cp
                    scale = 0.12
                    stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization on parent (18%)
    d1 = t0 + budget_s * 0.18
    h1, c1 = _local_opt_lp(h_init, d1, top_k=12, stall_limit=50)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Mixed directions local opt (10%)
    d2 = t0 + budget_s * 0.28
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _local_opt_mixed(h2, d2, stall_limit=45)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair optimization (10%)
    d3 = t0 + budget_s * 0.38
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=35000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP refinement (18%)
    d4 = t0 + budget_s * 0.56
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, min(t0 + budget_s * 0.50, d4), max_it=15000, T_init=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4 - 2:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.025)
        hp, cp = _local_opt_lp(hp, sd, top_k=10, stall_limit=35)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Multi-resolution exploration (20%)
    d5 = t0 + budget_s * 0.76
    nc = len(best_h)
    offsets = [-10, -6, -4, -2, 2, 4, 6, 10, 14, -14, 16, -16]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sr = _exact_c5(hr)
        if sr > best_c + 0.01:
            continue
        sd = min(d5, time.time() + budget_s * 0.018)
        hr, cr = _local_opt_lp(hr, sd, top_k=10, stall_limit=30)
        if cr < best_c + 0.003 and time.time() < d5 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5, time.time() + budget_s * 0.007)
            hb, cb2 = _bin_sa(hb, sd2, max_it=5000, T_init=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d5 - 2:
            sd3 = min(d5, time.time() + budget_s * 0.01)
            hr, cr = _local_opt_mixed(_project(hr.copy(), n_try / 2.0), sd3, stall_limit=20)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive LP polishing (12%)
    d6 = t0 + budget_s * 0.88
    nc = len(best_h)
    for r_init in [0.15, 0.08, 0.04]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _local_opt_lp(h6, sd, top_k=15, stall_limit=50)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _pair_opt(h6, sd, max_it=12000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Basin hopping (8%)
    d7 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d7 - 2 and n_re < 40:
        n_re += 1
        nc = len(best_h)
        pert = np.zeros(nc)
        x = np.arange(nc, dtype=float)
        for _ in range(3):
            f = rng.uniform(0.5, 12.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.5, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7 - 1, time.time() + budget_s * 0.006)
        ht, ct = _local_opt_lp(hp, sd, top_k=8, stall_limit=15)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt_lp(hf, deadline, top_k=10, stall_limit=40)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
