def _binary_swap_search(h, max_sweeps=100):
    """Systematic binary swap search for minimizing C5."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    c = _corr_profile(best_h)
    
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        
        for i in zeros:  # position to make 1
            for j in ones:  # position to make 0
                # Compute change in correlation
                d_range = np.arange(-(n-1), n)
                # prof for position p: (1-h[p+d]) - h[p-d] (with bounds)
                # For position i (0->1):
                pi = np.zeros(2*n-1)
                # i as first factor: partner at i+d, valid if 0<=i+d<n
                valid1 = (i + d_range >= 0) & (i + d_range < n)
                pi[valid1] = (1 - best_h[i + d_range[valid1]])
                # i as second factor: partner at i-d, valid if 0<=i-d<n
                valid2 = (i - d_range >= 0) & (i - d_range < n)
                pi[valid2] -= best_h[i - d_range[valid2]]
                
                # For position j (1->0):
                pj = np.zeros(2*n-1)
                valid1 = (j + d_range >= 0) & (j + d_range < n)
                pj[valid1] = (1 - best_h[j + d_range[valid1]])
                valid2 = (j - d_range >= 0) & (j - d_range < n)
                pj[valid2] -= best_h[j - d_range[valid2]]
                
                delta = pi - pj
                new_c = c + delta
                new_max = float(np.max(new_c))
                dx = 2.0 / n
                new_c5 = new_max * dx
                
                if new_c5 < best_c5 - 1e-15:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    best_c5 = new_c5
                    c = new_c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c5

def _grad_c(r, h, n):
    """Gradient of c_r with respect to h, where d = r - (n-1)."""
    d = r - (n - 1)
    g = np.zeros(n)
    # h[p] as first factor: partner at p+d
    # valid if 0 <= p+d < n, i.e., -d <= p < n-d

    # h[p] as second factor (in 1-h): partner at p-d

    hi2 = min(n, n + d)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return g
