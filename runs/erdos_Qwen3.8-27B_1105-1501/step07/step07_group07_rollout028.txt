import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    # Project v onto {x: sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj_and_grad(h, tau):
    n = len(h)
    scores = _scores(h)
    m = scores.max()
    u = (scores - m) / tau
    eu = np.exp(u)
    se = eu.sum()
    p = eu / se
    obj = m + tau * np.log(se)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if p[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m_lag = n - d
            if m_lag > 0:
                grad[:m_lag] += p[r] * (1.0 - h[d:d+m_lag])
                grad[d:] += -p[r] * h[:m_lag]
        else:
            g = -d
            m_lag = n - g
            if m_lag > 0:
                grad[g:] += p[r] * (1.0 - h[:m_lag])
                grad[:m_lag] += -p[r] * h[g:]
    return obj, grad

def _lse_slsqp(h, tau, max_iter=200):
    n = len(h)
    target = n / 2.0
    
    def fun(x):
        o, _ = _lse_obj_and_grad(x, tau)
        return o
    def jac(x):
        _, g = _lse_obj_and_grad(x, tau)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success:
            h_new = _project_sum(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h, _exact_c5(h)

def _lse_anneal(h, deadline, tau_init=0.5, tau_min=0.05, n_steps=30):
    n = len(h)
    h = _project_sum(h.copy(), n / 2.0)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        tau = tau_init * (tau_min / tau_init) ** (step / max(n_steps - 1, 1))
        h, c5 = _lse_slsqp(h, tau, max_iter=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _perturb(h, rng, amplitude):

    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _lse_obj_grad(h, tau):
    n = len(h)
    scores = _scores(h)
    m = scores.max()
    u = (scores - m) / tau
    eu = np.exp(u)
    se = eu.sum()
    p = eu / se
    obj = m + tau * np.log(se)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if p[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            ml = n - d
            if ml > 0:
                grad[:ml] += p[r] * (1.0 - h[d:d+ml])
                grad[d:] += -p[r] * h[:ml]
        else:
            g = -d
            ml = n - g
            if ml > 0:
                grad[g:] += p[r] * (1.0 - h[:ml])
                grad[:ml] += -p[r] * h[g:]
    return obj, grad

def _lse_slsqp(h, tau, max_iter=100):
    n = len(h)
    target = n / 2.0
    def fun(x):
        o, _ = _lse_obj_grad(x, tau)
        return o
    def jac(x):
        _, g = _lse_obj_grad(x, tau)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success:
            h_new = _project_sum(res.x, target)
            return h_new, _exact_c5(h_new)
    except:
        pass
    return h, _exact_c5(h)

def _lse_anneal(h, deadline, tau_init=0.5, tau_min=0.03, n_steps=20):
    n = len(h)
    h = _project_sum(h.copy(), n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for s in range(n_steps):
        if time.time() > deadline:
            break
        tau = tau_init * (tau_min / tau_init) ** (s / max(n_steps - 1, 1))
        h, c5 = _lse_slsqp(h, tau, max_iter=80)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    # Final exact check
    best_h = _project_sum(best_h, n / 2.0)
    return best_h, _exact_c5(best_h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amp):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)

    pert *= amp
    pert += amp * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)
