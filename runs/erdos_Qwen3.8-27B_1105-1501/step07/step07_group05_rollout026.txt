def _upper_envelope_min(a, b, t_lo=0.0, t_hi=1.0):
    """Find min_{t in [t_lo, t_hi]} max_k (a[k] + b[k]*t)."""
    n = len(a)
    # Sort by slope
    order = np.argsort(b)
    a_s = a[order]
    b_s = b[order]
    
    # Build upper envelope
    # Each element: (a, b, x_intersect_with_prev)
    # Hull is a list of (a, b) and the x-coordinate where this line becomes the topmost
    hull = [(a_s[0], b_s[0])]
    x_starts = [float('-inf')]
    
    for i in range(1, n):
        ai, bi = a_s[i], b_s[i]
        # Find intersection with the last line in hull
        # a_last + b_last * x = ai + bi * x
        # x = (a_last - ai) / (bi - b_last)
        while len(hull) > 0:
            a_last, b_last = hull[-1]
            if abs(bi - b_last) < 1e-15:
                # Parallel lines: keep the one with higher intercept
                if ai > a_last:
                    hull.pop()
                    x_starts.pop()
                    if len(hull) == 0:
                        hull.append((ai, bi))
                        x_starts.append(float('-inf'))
                        break
                    continue
                else:
                    break
            x_int = (a_last - ai) / (bi - b_last)
            if x_int >= x_starts[-1] - 1e-15:
                hull.pop()
                x_starts.pop()
                if len(hull) == 0:
                    hull.append((ai, bi))
                    x_starts.append(float('-inf'))
                    break
            else:
                break
        else:
            # Didn't break from while
            if len(hull) == 0:
                hull.append((ai, bi))
                x_starts.append(float('-inf'))
            else:
                a_last, b_last = hull[-1]
                if abs(bi - b_last) < 1e-15:
                    if ai > a_last:
                        hull[-1] = (ai, bi)
                else:
                    x_int = (a_last - ai) / (bi - b_last)
                    hull.append((ai, bi))
                    x_starts.append(x_int)
        # Hmm, this is getting complicated. Let me use a simpler approach.
    
    # Actually, let me just use the brute-force approach for n~360 lines.
    # O(n²) is fine for this size.
    candidates = [t_lo, t_hi]
    for i in range(n):
        for j in range(i+1, n):
            if abs(b[i] - b[j]) > 1e-14:
                t = (a[j] - a[i]) / (b[i] - b[j])
                if t_lo <= t <= t_hi:
                    candidates.append(t)
    
    best_t = t_lo
    best_val = float('inf')

        val = float(np.max(a + b * t))
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    return best_t, best_val
