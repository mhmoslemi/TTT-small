def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _project_sum(v, target):
    # Euclidean projection onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = 0.0, 1.0  # theta is in [0,1] for this problem since 0 <= target/n <= 1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return x
