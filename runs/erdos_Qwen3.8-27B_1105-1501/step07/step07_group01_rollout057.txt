    # Compute softmax weights
    # Shift to avoid overflow
    c_max = corr.max()
    exp_corr = np.exp(beta * (corr - c_max))
    w = exp_corr / exp_corr.sum()
    
    # grad = sum_m w[m] * dcorr[m]/dh
    # Use the formula derived above
    grad = np.zeros(n)
    
    # For m >= n-1 (s = m-(n-1) >= 0):
    # dcorr/dh[a] = (1-h[a-s])*I(a>=s) + (-h[a+s])*I(a<=n-1-s)

        if ws < 1e-15:

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

            grad[s:] -= ws * h[:n-s]

def _grad_desc(h, deadline, beta_start=3.0, beta_end=200.0, n_steps=5000, lr_init=0.5):
    """Projected gradient descent with annealing beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Anneal beta
        frac = min(step / (n_steps * 0.8), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad = _smooth_grad(h, beta)
        
        # Line search: try lr, halve if worsened
        for attempt in range(10):
            h_new = _project(h - lr * grad, target)

                lr = min(lr * 1.1, 5.0)

                if lr < 1e-6:
                    break
        else:
            # No improvement
            lr = max(lr, lr_init * 0.1)
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    exp_c = np.exp(beta * (corr - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for s in range(n):
        ws = w[n - 1 + s]
        if ws < 1e-11:
            continue
        if n - s > 0:
            grad[s:] += ws * (1.0 - h[:n - s])
            grad[:n - s] -= ws * h[s:]
    for s in range(1, n):
        ws = w[n - 1 - s]
        if ws < 1e-11:
            continue
        if n - s > 0:
            grad[:n - s] += ws * (1.0 - h[s:])
            grad[s:] -= ws * h[:n - s]
    return grad * dx

        frac = min(step / (max_steps * 0.7), 1.0)

def _coord_descent(h, deadline, max_sweeps=2000, seed=123):

    target = n / 2.0

            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            for _ in range(40):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1 * g))
                f2 = float(np.max(S0 + m2 * g))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.3, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.004, seed=777):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        g = np.zeros(2 * n - 1)
        for dd in range(0, n - oi):
            g[n - 1 + dd] = (1.0 - cur[oi + dd])
        for dd in range(-oi, 0):
            g[n - 1 + dd] = -cur[oi + dd]
        g_new = np.zeros(2 * n - 1)
        for dd in range(0, n - oz):
            g_new[n - 1 + dd] = (1.0 - cur[oz + dd])
        for dd in range(-oz, 0):
            g_new[n - 1 + dd] = -cur[oz + dd]
        dS = -g + g_new
        new_c5 = float(np.max(S0 + dS)) * dx
        delta = new_c5 - cur_c5

            cur[oi] = 0.0
            cur[oz] = 1.0
            S0 += dS
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

    sub_dl2 = min(d6, time.time() + budget_s * 0.03)
    h_pol, c5_pol = _lp_refine(h_pol, sub_dl2, radius=0.05, top_k=10, stall_limit=30)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 7: Final intensive polish (12%)
    d7 = deadline - 3
    n_final = len(best_h)
    # Coordinate descent
    sub_dl = min(d7, time.time() + budget_s * 0.03)
    h7, c5_7 = _coord_descent(best_h, sub_dl, max_sweeps=3000, seed=seed + 100)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    # LP
    sub_dl2 = min(d7, time.time() + budget_s * 0.03)
    h7, c5_7 = _lp_refine(best_h, sub_dl2, radius=0.03, top_k=20, stall_limit=40)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    # Smooth gradient final push
    sub_dl3 = min(d7, time.time() + budget_s * 0.02)
    h7, c5_7 = _grad_desc(best_h, sub_dl3, beta_start=50.0, beta_end=600.0, max_steps=5000, lr_init=0.3)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    # Basin hopping
    scale = 0.015
    n_restarts = 0
    while time.time() < d7 - 1 and n_restarts < 50:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl_b = min(d7 - 1, time.time() + budget_s * 0.005)
        h_t, c5_t = _coord_descent(h_pert, sub_dl_b, max_sweeps=300, seed=3000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.1)
    # Final gradient-guided pair swap
    if time.time() < deadline - 1:
        h_fs, c5_fs = _grad_pair_swap(best_h, deadline - 1, beta=100.0, n_iter=3000, rng=np.random.default_rng(seed + 999))
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
