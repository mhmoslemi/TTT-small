            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap(h, deadline, n_pairs=6000, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    q = 1.0 - h
    S0 = np.correlate(h, q, mode='full')
    cur = float(np.max(S0)) * dx
    best_h = h.copy()
    best = cur
    rng = np.random.default_rng(seed)
    stall = 0

    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0

        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = 1.0
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            continue

        T = np.correlate(d, q, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')

        n_grid = 30
        alphas = np.linspace(0, t_hi, n_grid)
        vals = np.max(S0 + alphas[:, None] * T - (alphas**2)[:, None] * C, axis=1) * dx
        bi = int(np.argmin(vals))
        if bi == 0:
            lo_a, hi_a = 0.0, alphas[1]
        elif bi == n_grid - 1:
            lo_a, hi_a = alphas[-2], alphas[-1]
        else:
            lo_a, hi_a = alphas[bi-1], alphas[bi+1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        for _ in range(25):
            if hi_a - lo_a < 1e-13:
                break
            if f1 < f2:
                hi_a, c2, f2 = c2, c1, f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
            else:
                lo_a, c1, f1 = c1, c2, f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        opt_a = (lo_a + hi_a) / 2.0
        if opt_a < 1e-14:
            stall += 1
            continue
        opt_val = float(np.max(S0 + opt_a*T - opt_a*opt_a*C)) * dx
        if opt_val < cur - 1e-13:
            h = h + opt_a * d
            h = _project(h, target)
            q = 1.0 - h
            S0 = np.correlate(h, q, mode='full')
            cur = float(np.max(S0)) * dx
            if cur < best:
                best = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.3, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Heavy FW optimization (45%)
    d1 = t0 + budget_s * 0.45
    G_init = _compute_G(h_init)
    h1, c5_1 = _fw_iterate(h_init, d1, max_steps=100000, top_k=20, stall_limit=100, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent polish (15%)
    d2 = t0 + budget_s * 0.60
    h2, c5_2 = _coord_descent(best_h, d2, max_sweeps=5000, seed=123)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More FW with different parameters (15%)
    d3 = t0 + budget_s * 0.75
    h3, c5_3 = _fw_iterate(best_h, d3, max_steps=50000, top_k=12, stall_limit=60, seed=seed + 100)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair swaps (10%)
    d4 = t0 + budget_s * 0.85
    h4, c5_4 = _pair_swap(best_h, d4, n_pairs=8000, seed=555)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution + basin hopping (15%)
    d5 = deadline
    n_curr = len(best_h)
    tried = set()
    for dn in range(-10, 11, 2):
        if time.time() > d5 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.008)
        h_r, c5_r = _fw_iterate(h_r, sub_dl, max_steps=5000, top_k=15, stall_limit=40, seed=700 + dn)
        sub_dl2 = min(d5 - 5, time.time() + budget_s * 0.004)
        h_r, c5_r = _coord_descent(h_r, sub_dl2, max_sweeps=500, seed=800 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Basin hopping on best
    scale = 0.03
    n_restarts = 0
    while time.time() < deadline - 3 and n_restarts < 40:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 3, time.time() + budget_s * 0.005)
        h_t, c5_t = _fw_iterate(h_pert, sub_dl, max_steps=3000, top_k=15, stall_limit=30, seed=2000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)

    # Final coordinate descent polish
    if time.time() < deadline - 2:
        sub_dl = min(deadline - 1, time.time() + budget_s * 0.02)
        h_f, c5_f = _coord_descent(best_h, sub_dl, max_sweeps=2000, seed=999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
