def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

    return x

def _mech_word(n, alpha):
    """Balanced/mechanical word: b[i] = 1 if {i*alpha} < 0.5"""
    idx = np.arange(n)
    vals = np.mod(idx * alpha, 1.0)
    b = (vals < 0.5).astype(float)
    # Ensure exactly n/2 ones
    if b.sum() != n // 2:
        # Adjust
        diff = n // 2 - int(b.sum())
        if diff > 0:
            # Need more ones: flip smallest vals that are >= 0.5
            above = np.where(vals >= 0.5)[0]
            above_vals = vals[above]
            flip_idx = above[np.argsort(above_vals)[:diff]]
            b[flip_idx] = 1.0
        elif diff < 0:
            below = np.where(vals < 0.5)[0]
            below_vals = vals[below]

def _grad_rows(h, lags):

    G = np.zeros((len(lags), n))

                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g

def _active_lp(h, scores, radius, n_active=10, gap=0.005):

    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))

    A_ub[:, n] = -1.0

def _gradient_dir(h, scores, beta=15.0, radius=0.1):

    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)

    return g

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _mech_word(n, alpha):
    idx = np.arange(n)
    vals = np.mod(idx * alpha, 1.0)
    b = (vals < 0.5).astype(float)
    diff = n // 2 - int(b.sum())
    if diff > 0:
        above = np.where(vals >= 0.5)[0]
        above_sorted = above[np.argsort(vals[above])[:diff]]
        b[above_sorted] = 1.0
    elif diff < 0:
        below = np.where(vals < 0.5)[0]
        below_sorted = below[np.argsort(-vals[below])[:abs(diff)]]
        b[below_sorted] = 0.0
    return b

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):

    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(1.0, 7.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p += 0.3 * amp * rng.normal(0, 1, n)
    p *= amp
    return _project(h + p, n / 2.0)
