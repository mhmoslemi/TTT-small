For the subgradient method, I need to compute ∂c[m]/∂h[j] for m in the active set. Let me think about how to do this efficiently.

Given the active set A (a subset of {0, ..., 2n-2}), I need:
s_j = sum_{m in A} ∂c[m]/∂h[j]

For a given m (lag k = m-(n-1)):
- If k >= 0: ∂c[m]/∂h[j] = (1-h[j-k])*I(k≤j) - h[j+k]*I(j+k<n)
- If k < 0 (s=-k): ∂c[m]/∂h[j] = (1-h[j+s])*I(j+s<n) - h[j-s]*I(j≥s)

For a fixed k ≥ 0, this as a function of j is:
- For j < k: -h[j+k] (if j+k < n)
- For k ≤ j ≤ n-1-k: (1-h[j-k]) - h[j+k]
- For j > n-1-k: (1-h[j-k])
- For j ≥ n (none, j max is n-1)

So it's a "sliding window" convolution! I can compute it for all j simultaneously using numpy.

```python
def _subgradient_step(h, c, active_lags, alpha):
    """Compute subgradient step for active lags."""
    n = len(h)
    s = np.zeros(n)
    for m in active_lags:
        k = m - (n - 1)
        if k >= 0:
            if k > 0:  # k=0 is special: d = 1-2h
                d = np.zeros(n)
                d[k:] += 1 - h[:n-k]
                d[:n-k] -= h[k:]
            else:
                d = 1.0 - 2.0 * h
        else:
            ss = -k
            d = np.zeros(n)
            d[:n-ss] += 1 - h[ss:]
            d[ss:] -= h[:n-ss]
        s += d
    s /= len(active_lags)
    h_new = h - alpha * s
    return h_new
```

```python
            if time.time() > deadline:
                break
```

```python
            g = np.zeros(2*n-1)
            g[n-1] = 1.0 - 2.0*old
            # Positive lags
            for k in range(1, n):
                idx = n - 1 + k
                val = 0.0
                if k <= j:
                    val += (1.0 - h[j-k])
                if j + k < n:
                    val -= h[j+k]
                g[idx] = val
                # Negative lag
                idx2 = n - 1 - k
                val2 = 0.0
                if j + k < n:
                    val2 += (1.0 - h[j+k])
                if k <= j:
                    val2 -= h[j-k]
                g[idx2] = val2
            
            # Ternary search for optimal delta
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            # Use ternary search (40 iterations for precision ~1e-8)
            for _ in range(35):

                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1 * g))
                f2 = float(np.max(S0 + m2 * g))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
```

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
