def _grad_lag(h, t):
    """Gradient of correlation at output index t with respect to h."""
    n = len(h)
    g = np.zeros(n)
    l = t - (n - 1)  # signed lag
    if l >= 0:
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:l+m])
            g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s+m]
    return g

    w = np.exp(v)
    ws = float(np.sum(w))

        w = np.zeros(len(S))
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    
    if top_k is not None:
        # Only use top_k lags
        idx = np.argsort(-w)[:top_k]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w2 /= w2.sum()
        w = w2

        wl = w[n - 1 + l]

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _project(h_new, n_new / 2.0)

def _concentrated_grad(h, top_k=5):
    """Gradient concentrated on the top_k worst lags only."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-S)[:top_k]
    g = np.zeros(n)
    for t in top_idx:
        g += _grad_lag(h, int(t))
    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-10:
        return g / nrm
    return None

        # Fill remaining
        remaining = ones_needed - count
        candidates = np.where(h == 0)[0]
        fill = rng.choice(candidates, size=remaining, replace=False)
        h[fill] = 1.0
    results.append(('sparse_periodic', h))
    
    # 4. Fourier-based smooth

    results.append(('smooth_fourier', h))
    
    # 5. Block structure
    h = np.zeros(n)
    nb = 4
    per_block = n // nb
    for b in range(nb):
        start = b * per_block
        end = start + per_block
        half = end // 2 - start
        h[start:start+half] = 1.0
    results.append(('blocks', h))
    
    # 6. Interleaved
    h = np.zeros(n)
    h[0::2] = 1.0
    if n % 2 == 0:
        h[-1] = 0.0  # ensure exactly n/2 ones
    # Adjust
    ones_count = int(np.sum(h > 0.5))
    needed = n // 2
    if ones_count > needed:
        # Remove excess
        excess = ones_count - needed
        one_idx = np.where(h > 0.5)[0]
        to_remove = rng.choice(one_idx, size=excess, replace=False)
        h[to_remove] = 0.0
    elif ones_count < needed:

        zero_idx = np.where(h <= 0.5)[0]
        to_add = rng.choice(zero_idx, size=deficit, replace=False)
        h[to_add] = 1.0
    results.append(('interleaved', h))
    
    return results

            a, c_try = _line_search_quad(h, d, dx, -rad, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.40)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.008:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 50 == 0:
                    rad = 0.06
    return best_h, best_c

def _intensive_pair(h, deadline, max_it=300000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    while time.time() < deadline:
        if stall > 600:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        max_delta = abs(h[i] - h[j])
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search_quad(h, d, dx, -max_delta, max_delta)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _project(h_new, n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=25000, Ti=0.003):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _multi_start_constructions(n, rng):
    results = []
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    results.append(h)
    h = np.zeros(n)
    for i in range(n):
        if i % 4 < 2:
            h[i] = 1.0
    results.append(_project(h, n / 2.0))
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * 3.0 * x / n) + 0.15 * np.sin(2 * np.pi * 7.0 * x / n + 0.7)
    results.append(_project(h, n / 2.0))
    h = np.zeros(n)
    h[0::3] = 1.0
    h = _project(h, n / 2.0)
    results.append(h)
    h = np.zeros(n)
    h[0::2] = 1.0
    ones_count = int(np.sum(h > 0.5))
    needed = n // 2
    if ones_count > needed:
        excess = ones_count - needed
        one_idx = np.where(h > 0.5)[0]
        to_remove = rng.choice(one_idx, size=excess, replace=False)
        h[to_remove] = 0.0
    elif ones_count < needed:
        deficit = needed - ones_count
        zero_idx = np.where(h <= 0.5)[0]
        to_add = rng.choice(zero_idx, size=deficit, replace=False)
        h[to_add] = 1.0
    results.append(h)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * 1.5 * x / n + 0.3) + 0.2 * np.cos(2 * np.pi * 4.0 * x / n)
    results.append(_project(h, n / 2.0))
    return results

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Quick local opt on parent
    d1 = t0 + budget_s * 0.13
    h1, c1 = _local_opt(h_init, d1, r0=0.18, stall_lim=60, nd=6)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Intensive pair optimization
    d2 = t0 + budget_s * 0.24
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _intensive_pair(h2, d2, max_it=300000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA and follow-up
    d3 = t0 + budget_s * 0.34
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d3, mx=25000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 1:
        hp = _project(best_h.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _local_opt(hp, sd, r0=0.10, stall_lim=40, nd=5)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-start from diverse constructions
    d4 = t0 + budget_s * 0.55
    nc = len(best_h)
    constructions = _multi_start_constructions(nc, rng)
    for hc in constructions:
        if time.time() > d4:
            break
        hc = _project(hc.copy(), nc / 2.0)
        cc = _exact_c5(hc)
        sd = min(d4, time.time() + budget_s * 0.025)
        hc, cc = _local_opt(hc, sd, r0=0.15, stall_lim=40, nd=5)
        if cc < best_c + 0.005:
            if time.time() < d4 - 2:
                hcb = _to_bin(hc, nc)
                sd2 = min(d4 - 1, time.time() + budget_s * 0.01)
                hcb, ccb = _bin_sa(hcb, sd2, mx=8000, Ti=0.002)
                if ccb < cc:
                    cc = ccb
                    hc = hcb
                if time.time() < d4 - 1:
                    sd3 = min(d4, time.time() + budget_s * 0.015)
                    hc, cc = _local_opt(_project(hc.copy(), nc / 2.0), sd3, r0=0.08, stall_lim=30, nd=5)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    # Phase 5: Multi-resolution
    d5 = t0 + budget_s * 0.70
    nc = len(best_h)
    tried_res = set()
    for dn in [-6, -4, -2, 2, 4, 6, -8, 8, -10, 10, -12, 12]:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try in tried_res:
            continue
        tried_res.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=35, nd=5)
        if cr < best_c + 0.004 and time.time() < d5 - 2:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5 - 1, time.time() + budget_s * 0.008)
            hb, cb2 = _bin_sa(hb, sd2, mx=6000, Ti=0.0015)
            if cb2 < cr:
                cr = cb2
                hr = hb
            if time.time() < d5 - 1:
                sd3 = min(d5, time.time() + budget_s * 0.01)
                hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.06, stall_lim=25, nd=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive optimization rounds
    d6 = t0 + budget_s * 0.88
    nc = len(best_h)
    for r_init in [0.15, 0.08, 0.04, 0.02, 0.015]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=80, nd=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6 - 2, time.time() + budget_s * 0.025)
            h6, c6 = _intensive_pair(h6, sd, max_it=150000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Basin hopping
    d7 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d7 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.3) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=20, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.03, stall_lim=40, nd=5)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 1.5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _intensive_pair(hf, deadline, max_it=10000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    c_init = _exact_c5(h_init)
    if final_c > c_init:
        best_h = h_init
        final_c = c_init
    return best_h, final_c, len(best_h)
