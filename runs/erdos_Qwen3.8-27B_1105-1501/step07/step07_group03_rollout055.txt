def _solve_lp(h, active_indices, trust, n):
    """Solve LP to find direction e that minimizes max linearized active lag."""
    n = len(h)
    K = len(active_indices)
    # Variables: e[0..n-1], t → n+1 variables
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(e) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Inequality: for each active lag r_k:
    #   sum_j coeff[k][j] * e[j] - t <= -raw[r_k]
    # Box: -upper[j] <= e[j] <= upper[j]
    
    raw = _raw_correlate(h)
    
    A_ub = []
    b_ub = []
    
    # Active lag constraints
    for idx in active_indices:
        d_lag = idx - (n - 1)
        coeff = _lag_coefficients(h, d_lag, n)
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-raw[idx])
    
    # Box constraints: e[j] <= upper[j] → e[j] <= upper[j]
    #                 -e[j] <= upper[j] → -e[j] <= upper[j]
    for j in range(n):
        upper = min(trust, 1.0 - h[j])
        lower = max(-trust, -h[j])
        
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(upper)
        
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_ub.append(row)
        b_ub.append(-lower)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
