def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    # Handle fractional part (shouldn't be needed if n is even)

    if rem > 1e-10:
        v[sorted_idx[int(target)]] = rem
    return v

    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    raw_max = float(np.max(raw))
    # Find active lags: those within band_frac of max
    band = raw_max * band_frac
    active = np.where(raw > raw_max - band)[0]
    if len(active) == 0:

    # Also include top_k largest

    active = np.unique(np.concatenate([active, top]))
    # Compute weighted subgradient

        else:
            b = -d
            m = n - b
            if m > 0:
                g[:m] += w * (1.0 - h[b:])
                g[b:] -= w * h[:m]

            if val < best_val:

                best_t = t

        if np.max(np.abs(g)) < 1e-10:

        # FW vertex

        direction = v - h
        if np.max(np.abs(direction)) < 1e-12:
            # Try negative gradient step instead
            direction = -g / (np.max(np.abs(g)) + 1e-15)
            # Find t_hi for box constraints
            pos = direction > 1e-12
            neg = direction < -1e-12
            t_hi = 1.0
            if np.any(pos):
                t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
            if np.any(neg):
                t_hi = min(t_hi, float(np.min(h[neg] / (-direction[neg]))))
            t_hi = max(t_hi, 1e-12)
        else:
            t_hi = gap  # FW step: h + gap*(v-h)
        
        # Compute bilinear terms
        T = np.correlate(direction, 1.0 - h, mode='full') - np.correlate(h, direction, mode='full')
        C = np.correlate(direction, direction, mode='full')
        
        # Line search

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Slowly decrease gap for convergence
            gap *= 0.999
        else:
            gap *= 0.9
            if gap < 0.01:
                break
    
    return best_h, best_c5
