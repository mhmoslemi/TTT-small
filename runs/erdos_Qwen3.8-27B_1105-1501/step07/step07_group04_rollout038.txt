def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    lo -= 1.0
    hi += 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_fn_and_grad(x, tau, n, dx):
    # Smoothed max of dx * corr
    corr = np.correlate(x, 1.0 - x, mode='full')
    F = corr * dx  # length 2n-1
    m = float(np.max(F))
    diff = (F - m) / tau
    # Clip for numerical stability
    diff = np.clip(diff, -50.0, 0.0)
    exps = np.exp(diff)
    S = float(np.sum(exps))
    if S < 1e-300:
        S = 1e-300
    value = m + tau * np.log(S)
    
    # Compute weights
    w = exps / S  # length 2n-1
    c = n - 1  # center index
    
    # Compute gradient
    grad = np.zeros(n)
    # d=0 term
    grad += w[c] * (1.0 - 2.0 * x)
    
    for d in range(1, n):
        wp = w[c + d] if c + d < 2*n - 1 else 0.0
        wm = w[c - d] if c - d >= 0 else 0.0
        if wp < 1e-15 and wm < 1e-15:
            continue
        # Positions p = 0, ..., n-1-d: p+d is valid
        # grad[p] += wp*(1-x[p+d]) - wm*x[p+d]
        if wp > 1e-15:
            grad[:n-d] += wp * (1.0 - x[d:])
        if wm > 1e-15:
            grad[:n-d] -= wm * x[d:]
        # Positions p = d, ..., n-1: p-d is valid
        # grad[p] += -wp*x[p-d] + wm*(1-x[p-d])
        if wp > 1e-15:
            grad[d:] -= wp * x[:n-d]
        if wm > 1e-15:
            grad[d:] += wm * (1.0 - x[:n-d])
    
    grad *= dx
    return value, grad

    cons = [
        {'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (target,)}
    ]

    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0

            break
        length = int(rng.integers(2, min(8, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:

            continue
        block1 = cur[i:i+length].copy()
        block2 = cur[j:j+length].copy()
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue

        cur[j:j+length] = block1
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-14:

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            cur[i:i+length] = block1
            cur[j:j+length] = block2
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 5.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_fn_grad(x, tau, n, dx):
    xc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(xc, 1.0 - xc, mode='full')
    F = corr * dx
    m = float(np.max(F))
    diff = (F - m) / tau
    diff = np.clip(diff, -60.0, 0.0)
    exps = np.exp(diff)
    S = float(np.sum(exps))
    if S < 1e-300:
        S = 1e-300
    value = m + tau * np.log(S)
    w = exps / S
    c = n - 1
    grad = np.zeros(n)
    grad += w[c] * (1.0 - 2.0 * xc)
    for d in range(1, n):
        wp = w[c + d] if c + d < 2 * n - 1 else 0.0
        wm = w[c - d] if c - d >= 0 else 0.0
        if wp < 1e-12 and wm < 1e-12:
            continue
        seg = n - d
        if wp > 1e-12:
            grad[:seg] += wp * (1.0 - xc[d:])
            grad[d:] -= wp * xc[:seg]
        if wm > 1e-12:
            grad[:seg] -= wm * xc[d:]
            grad[d:] += wm * (1.0 - xc[:seg])
    grad *= dx
    return value, grad

def _con_fun(x, tau, n, dx, target):
    return float(np.sum(x)) - target

def _con_jac(x, tau, n, dx, target):
    return np.ones(n)

def _slsqp(h0, tau, n, maxiter, ftol, deadline):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (tau, n, dx, target)}]
    try:
        res = minimize(_smooth_fn_grad, h0, args=(tau, n, dx),
                       method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        x_opt = _project(res.x, target)
        c5 = _exact_c5(x_opt)
        return x_opt, c5
    except Exception:
        x_opt = _project(h0, target)
        return x_opt, _exact_c5(x_opt)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _bin_c5(hb):
    n = len(hb)
    dx = 2.0 / n
    corr = np.correlate(hb, 1.0 - hb, mode='full')
    return float(np.max(corr) * dx)

    best_h, best_c5 = cur.copy(), cur_c5

    zeros = [int(i) for i in np.where(cur <= 0.5)[0]]
    stall = 0
    for _ in range(maxit):

            break
        if not ones or not zeros:
            break
        max_idx = int(np.argmax(cur_corr))
        d_off = max_idx - (n - 1)
        ntry = min(80, len(ones) * len(zeros))
        best_delta, best_mv = 0.0, None
        for _ in range(ntry):
            if abs(d_off) > 0 and rng.random() < 0.5:
                if d_off > 0:
                    vo = [o for o in ones if o <= n - 1 - d_off]

            stall = 0

def _resample(h, nn):
    n = len(h)
    if nn == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(nn) + 0.5) / nn
    return _project(np.interp(xn, xo, h), nn / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for k in range(3):
        f = rng.uniform(0.3, 6.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        p += a * np.sin(2 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.15 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)
