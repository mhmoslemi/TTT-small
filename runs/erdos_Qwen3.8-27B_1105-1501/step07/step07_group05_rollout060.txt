def _lp_dir(h, top_idx, r=0.5):
    """Solve LP for minimax descent direction on active constraints."""
    from scipy.optimize import linprog
    n = len(h)
    K = len(top_idx)
    if K == 0: return None
    
    # Compute gradients for active rows
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            A[i, :m] += (1.0 - h[lag:lag+m])
            A[i, lag:] -= h[:m]
        else:
            s = -lag
            m = n - s
            A[i, s:] += (1.0 - h[:m])
            A[i, :m] -= h[s:s+m]
    
    # Normalize rows to avoid scaling issues
    norms = np.linalg.norm(A, axis=1)
    norms[norms < 1e-10] = 1.0
    A /= norms[:, None]
    
    # LP: max tau
    # Variables: x = [s_0, ..., s_{n-1}, tau]
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -tau = maximize tau
    
    # Inequalities: A_ub @ x <= b_ub
    # (1) A_norm @ s + tau <= 0  (K constraints)
    # (2) -h_i/r <= s_i <= (1-h_i)/r  ->  -s_i <= h_i/r and s_i <= (1-h_i)/r (2n constraints)
    # (3) -tau <= 0 (1 constraint)
    
    A_ub = np.zeros((K + 2*n + 1, n + 1))
    b_ub = np.zeros(K + 2*n + 1)
    
    A_ub[:K, :n] = A
    A_ub[:K, n] = 1.0
    
    # -s_i <= h_i/r
    A_ub[K:K+n, :n] = -np.eye(n)
    b_ub[K:K+n] = h / r
    
    # s_i <= (1-h_i)/r
    A_ub[K+n:K+2*n, :n] = np.eye(n)
    b_ub[K+n:K+2*n] = (1.0 - h) / r
    
    # -tau <= 0
    A_ub[K+2*n, n] = -1.0
    
    # Equality: sum(s) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.success and res.x is not None:
            s = res.x[:n].copy()
            tau = res.x[n]
            if tau > 1e-4:
                s -= np.mean(s)  # numerical cleanup
                nrm = np.linalg.norm(s)
                if nrm > 1e-10:
                    return s / nrm  # normalize for line search
    except:
        pass
    return None

    cands = [a_lo, a_hi, (a_lo + a_hi) / 2.0]
    top_idx = np.argsort(-S0)[:min(10, len(S0))]
    for ii in top_idx:
        if Q[ii] > 1e-14:
            av = T[ii] / (2.0 * Q[ii])
            if a_lo <= av <= a_hi:
                cands.append(av)
    width = a_hi - a_lo
    ng = min(60, max(30, int(width * 200)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.04 * width)
    hi_r = min(a_hi, best_a + 0.04 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        if hi_r - lo_r > 1e-13:
            mid = (lo_r + hi_r) / 2.0
            fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
            if fmid < best_val:
                best_val = fmid
                best_a = mid
    return best_a, best_val

def _sharp_grad(h, K=10):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:K]
    g = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            g[:m] += (1.0 - h[lag:lag + m])
            g[lag:] -= h[:m]
        else:
            s = -lag
            m = n - s
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s + m]
    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-12:
        g /= nrm
    return g

def _lp_dir(h, top_idx):
    from scipy.optimize import linprog
    n = len(h)
    K = len(top_idx)
    if K < 2:
        return None
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            A[i, :m] += (1.0 - h[lag:lag + m])
            A[i, lag:] -= h[:m]
        else:
            s = -lag
            m = n - s
            A[i, s:] += (1.0 - h[:m])
            A[i, :m] -= h[s:s + m]
    norms = np.linalg.norm(A, axis=1)
    norms[norms < 1e-10] = 1.0
    A /= norms[:, None]
    c_obj = np.zeros(n + 1)
    c_obj[n] = -1.0
    n_con = K + 2 * n + 1
    A_ub = np.zeros((n_con, n + 1))
    b_ub = np.zeros(n_con)
    A_ub[:K, :n] = A
    A_ub[:K, n] = 1.0
    row = K
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    A_ub[row, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            s = res.x[:n].copy()
            tau = float(res.x[n])
            if tau > 1e-3:
                s -= np.mean(s)
                nrm = np.linalg.norm(s)
                if nrm > 1e-8:
                    return s
    except Exception:
        pass
    return None

def _block_grad(h, K, bs):
    g = _sharp_grad(h, K)
    n = len(g)
    nb = n // bs
    if nb < 1:
        return g
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-12:
        d /= nrm
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _proj(h_new, n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _pair_opt(h, deadline, max_it=20000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-4:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-13:
            h_new = _project_safe(h, d, a, target)
            if h_new is not None:
                c5a = _exact_c5(h_new)
                if c5a < cur - 1e-13:
                    h = h_new
                    cur = c5a
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _project_safe(h, d, a, target):
    h_new = h + a * d
    if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
        return None
    h_new = np.clip(h_new, 0.0, 1.0)
    s = float(np.sum(h_new))
    if abs(s - target) > 1e-6:
        return None
    h_new = _proj(h_new, target)
    return h_new

def _local_opt(h, deadline, r0=0.2, K_lp=10, K_grad=12, stall_lim=60):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rad = r0
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.2:
            break
        it += 1
        raw = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-raw)[:max(K_lp, K_grad)]
        directions = []
        if it % 2 == 0:
            d_lp = _lp_dir(h, top_idx[:K_lp])
            if d_lp is not None:
                directions.append(d_lp)
        g = _sharp_grad(h, K_grad)
        if np.linalg.norm(g) > 1e-12:
            directions.append(g)
        if it % 3 == 0:
            db = _block_grad(h, K_grad, 12)
            if np.linalg.norm(db) > 1e-12:
                directions.append(db)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in directions:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-13:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-13:
            h_new = _project_safe(h, best_d, best_a, target)
            if h_new is None:
                h_new = _proj(h + best_a * best_d, target)
            c_new = _exact_c5(h_new)
            if c_new < cur - 1e-13:
                h = h_new
                cur = c_new
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
                rad = min(rad * 1.03, 0.4)
            else:
                stall += 1
                rad *= 0.8
        else:
            stall += 1
            rad *= 0.75
            if rad < 0.01:
                rng = np.random.default_rng(it * 31 + 7)
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _proj(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 40 == 0:
                    rad = 0.08
    return best_h, best_c

def _bin_sa(hb, deadline, mx=15000, Ti=0.0015):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _proj(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization (25%)
    d1 = t0 + budget_s * 0.25
    h1, c1 = _local_opt(h_init, d1, r0=0.25, K_lp=8, K_grad=12, stall_lim=80)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair swap (10%)
    d2 = t0 + budget_s * 0.35
    h2 = _proj(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=30000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary exploration (15%)
    d3 = t0 + budget_s * 0.50
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _bin_sa(hbin, d3, mx=12000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 2:
        hp = _proj(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _local_opt(hp, sd, r0=0.12, K_lp=6, K_grad=10, stall_lim=40)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Mirror start (10%)
    d4 = t0 + budget_s * 0.60
    if time.time() < d4:
        hmir = best_h[::-1].copy()
        hmir = _proj(hmir, len(hmir) / 2.0)
        cm = _exact_c5(hmir)
        if cm < best_c + 0.01:
            sd = min(d4, time.time() + budget_s * 0.04)
            hm, cmm = _local_opt(hmir, sd, r0=0.15, K_lp=8, K_grad=12, stall_lim=50)
            if cmm < best_c:
                best_c = cmm
                best_h = hm.copy()

    # Phase 5: Multi-resolution (20%)
    d5 = t0 + budget_s * 0.80
    nc = len(best_h)
    offsets = [-16, -8, -4, 4, 8, 16, -12, 12, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.18, K_lp=6, K_grad=10, stall_lim=35)
        if cr < best_c + 0.003 and time.time() < d5 - 2:
            hb = _to_bin(hr)
            sd2 = min(d5, time.time() + budget_s * 0.007)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
        elif cr < best_c + 0.002 and time.time() < d5 - 2:
            sd3 = min(d5, time.time() + budget_s * 0.01)
            hr, cr = _local_opt(_proj(hr.copy(), n_try / 2.0), sd3, r0=0.06, K_lp=4, K_grad=8, stall_lim=25)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()
                nc = n_try

    # Phase 6: Intensive LP optimization (15%)
    d6 = t0 + budget_s * 0.95
    nc = len(best_h)
    for r_init in [0.15, 0.08, 0.04]:
        if time.time() > d6 - 2:
            break
        h6 = _proj(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.04)
        h6, c6 = _local_opt(h6, sd, r0=r_init, K_lp=10, K_grad=15, stall_lim=70)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 3:
            h6 = _proj(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _pair_opt(h6, sd, max_it=10000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Final basin hopping (5%)
    if deadline - time.time() > 5:
        scale = 0.02
        n_re = 0
        while time.time() < deadline and n_re < 40:
            n_re += 1
            nc = len(best_h)
            x = np.arange(nc, dtype=float)
            pert = np.zeros(nc)
            for _ in range(4):
                f = rng.uniform(0.5, 10.0)
                p = rng.uniform(0, 2 * np.pi)
                pert += rng.uniform(0.4, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
            pert += 0.2 * scale * rng.normal(0, 1, nc)
            mxp = np.max(np.abs(pert))
            if mxp > 1e-10:
                pert *= scale / mxp
            hp = _proj(best_h + pert, nc / 2.0)
            sd = min(deadline, time.time() + budget_s * 0.004)
            ht, ct = _local_opt(hp, sd, r0=0.05, K_lp=5, K_grad=8, stall_lim=20)
            if ct < best_c:
                best_c = ct
                best_h = ht.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 2:
        hf = _proj(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, K_lp=5, K_grad=10, stall_lim=30)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 1.5:
        hf = _proj(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _proj(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
