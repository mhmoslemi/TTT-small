def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x[mask] += res / cnt

    return x


def _smooth_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    h_cl = np.clip(h, 0.0, 1.0)
    scores = np.correlate(h_cl, 1.0 - h_cl, mode='full') * dx
    m = float(np.max(scores))
    z = (scores - m) / beta
    z = np.clip(z, -50.0, 50.0)
    exp_z = np.exp(z)
    sum_w = float(np.sum(exp_z))
    if sum_w < 1e-300:
        sum_w = 1e-300
    w = exp_z / sum_w
    value = m + beta * np.log(sum_w)
    grad = np.zeros(n)
    for d in range(n):
        wt = w[d + n - 1]
        if wt < 1e-14:

def _active_lp(h, scores, radius, n_active=15, gap=0.005):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = np.zeros((len(active), n))
    for idx in range(len(active)):
        d = lags[idx]
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = int(-d)
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    m_con = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_con, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _weighted_grad(h, scores, beta=20.0):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    g = np.zeros(n)
    for ell in range(len(scores)):
        wt = w[ell]
        if wt < 1e-14:
            continue
        d = ell - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wt * (1.0 - h[:m])
                g[:m] += -wt * h[d:]
        else:
            gg = int(-d)
            m = n - gg
            if m > 0:
                g[:m] += wt * (1.0 - h[gg:])
                g[gg:] += -wt * h[:m]
    g -= float(np.mean(g))
    return g


def _clip_direction(g, h, radius):
    n = len(h)
    g = g.copy()
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g


def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    best_t = opt_t
    best_v = opt_val
    v0 = float(np.max(S0)) * dx
    if v0 < best_v:
        best_t, best_v = t_lo, v0
    v1 = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
    if v1 < best_v:

    return best_t, best_v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for i in range(4):
        perturb += amps[i] * np.sin(2.0 * np.pi * freqs[i] * x / n + phases[i])
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)


def _sigmoid_transform(h, gamma, rng):
    n = len(h)
    h_cl = np.clip(h, 0.05, 0.95)
    logits = np.log(h_cl / (1.0 - h_cl))
    shift = rng.normal(0, 0.3, n)
    logits += shift
    h_new = 1.0 / (1.0 + np.exp(-gamma * logits))
    return _project(h_new, n / 2.0)


def _rotation_interp(h, frac, n_new=None):
    n = len(h)
    if n_new is None:
        n_new = n
    shift = int(round(frac * n))
    x = np.arange(n_new)
    src_idx = np.mod(x + shift, n)
    frac_part = np.zeros(n_new)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        si = src_idx[i]
        hi_ = h[si % n]
        h_new[i] = hi_
    return _project(h_new, n_new / 2.0)

        if time.time() > d3:
            break
        if c5p > best_c5 + 0.02:
            continue
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_opt, c5_opt = _smooth_sequence(hp, betas_flat, sub_dl, maxiter=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            # Short LP polish on new best
            if time.time() < d3 - 2:
                sub_dl2 = min(d3, time.time() + budget_s * 0.015)
                h_lp, c5_lp = _local_opt(_project(h_opt.copy(), n_b / 2.0), sub_dl2, radius_init=0.06, max_iters=500)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
        # Add result to pool for future combinations
        if c5_opt < best_c5 + 0.01:
            pool.append(h_opt.copy())
            if len(pool) > 8:
                pool = pool[:8]

    # Phase 4: Multi-resolution (15%)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    tried_ns = set()
    offsets = list(range(-20, 21, 4))
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        if n_try % 2 != 0:
            n_try += 1
            if n_try in tried_ns:
                continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.01:
            sub_dl = min(d4, time.time() + budget_s * 0.015)
            betas_r = [max_sc * s for s in [0.08, 0.03, 0.01, 0.003]]
            h_r, c5_r = _smooth_sequence(h_r, betas_r, sub_dl, maxiter=40)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
        elif c5_r < best_c5 + 0.005 and n_try < n_curr:
            sub_dl = min(d4, time.time() + budget_s * 0.01)
            h_r2, c5_r2 = _local_opt(h_r, sub_dl, radius_init=0.08, max_iters=400)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                n_curr = n_try

    # Phase 5: Final polish (15%)
    d5 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, d5, radius_init=0.06, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    if deadline - time.time() > 3:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_f2, c5_f2 = _pair_sweep(h_f2, deadline, max_rounds=50)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final SLSQP micro-polish
    if deadline - time.time() > 2:
        n_f3 = len(best_h)
        max_sc_final = float(np.max(_scores(best_h)))
        betas_final = [max_sc_final * s for s in [0.002, 0.0008, 0.0003]]
        h_f3, c5_f3 = _smooth_sequence(_project(best_h.copy(), n_f3 / 2.0), betas_final, deadline, maxiter=40)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()

    # Final projection and validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    elif final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
