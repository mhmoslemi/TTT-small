    lo_val = h[j]
    t_lo = max(-hi_val, lo_val - 1.0)

    best_idx = int(np.argmin(fs))

    hi = ts[min(n_grid - 1, best_idx + 2)]

            c1 = hi - gr * (hi - lo)
            f1 = _pair_c5_at(h, i, j, c1)

            c2 = lo + gr * (hi - lo)
            f2 = _pair_c5_at(h, i, j, c2)

    tau = _etau
    raw = _raw_scores(x) * dx
    mval = np.max(raw)
    shifted = (raw - mval) / tau
    z = np.exp(shifted)
    p = z / np.sum(z)
    return _lse_grad_direct(x, p, dx)

def _eq_fun(x):
    return float(np.sum(x) - _gn / 2.0)

def _eq_jac(x):
    return np.ones((1, _gn))

def _slsqp_step(h, tau, maxiter=250):
    global _gn, _gdx, _etau
    n = len(h)
    _gn = n
    _gdx = 2.0 / n
    _etau = tau
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},)
    try:
        res = minimize(_obj_func, h.copy(), jac=_jac_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project_sum(np.asarray(res.x, dtype=float), n / 2.0)
    except Exception:
        pass
    return h.copy()

def _anneal(h, tau_list, deadline, maxiter=250):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    for tau in tau_list:
        if time.time() > deadline:
            break
        h_new = _slsqp_step(h_cur, tau, maxiter=maxiter)
        c5_new = _exact_c5(h_new)
        if c5_new <= best_c5 + 0.002:
            h_cur = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5

def _pair_c5_at(h, i, j, t):
    h_new = h.copy()
    h_new[i] += t
    h_new[j] -= t
    return _exact_c5(h_new)

def _pair_line_search(h, i, j):
    hi_val = h[i]
    lo_val = h[j]
    t_lo = max(-hi_val, lo_val - 1.0)
    t_hi = min(1.0 - hi_val, lo_val)
    if t_hi - t_lo < 1e-12:
        return 0.0, _exact_c5(h)
    n_grid = 16
    ts = np.linspace(t_lo, t_hi, n_grid)
    fs = np.array([_pair_c5_at(h, i, j, float(t)) for t in ts])
    best_idx = int(np.argmin(fs))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-12:
        best = min([(float(ts[best_idx]), float(fs[best_idx])),
                    (float(t_lo), float(_pair_c5_at(h, i, j, t_lo))),
                    (float(t_hi), float(_pair_c5_at(h, i, j, t_hi)))], key=lambda x: x[1])
        return best
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _pair_c5_at(h, i, j, float(c1))
    f2 = _pair_c5_at(h, i, j, float(c2))
    for _ in range(35):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _pair_c5_at(h, i, j, float(c1))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _pair_c5_at(h, i, j, float(c2))
    opt_t = (lo + hi) / 2.0
    opt_f = _pair_c5_at(h, i, j, float(opt_t))
    f_lo = _pair_c5_at(h, i, j, float(t_lo))
    f_hi = _pair_c5_at(h, i, j, float(t_hi))
    candidates = [(float(t_lo), f_lo), (float(t_hi), f_hi),
                  (float(opt_t), opt_f), (float(ts[best_idx]), float(fs[best_idx]))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _random_pair_polish(h, deadline, n_pairs=5000, seed=111):
    n = len(h)
    h_cur = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        t, c5_try = _pair_line_search(h_cur, i, j)
        if t != 0.0:
            h_new = h_cur.copy()
            h_new[i] += t
            h_new[j] -= t
            h_new = _project_sum(h_new, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h_cur = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h, deadline, max_iter=30000, T_init=0.005, seed=123):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[ i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=20):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0].tolist()
        zeros = np.where(cur <= 0.5)[0].tolist()
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _proj_grad_descent(h, tau, lr, n_steps=100, deadline=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    for _ in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        raw = _raw_scores(h_cur) * dx
        mval = np.max(raw)
        shifted = (raw - mval) / tau
        z = np.exp(shifted)
        p = z / np.sum(z)
        grad = _lse_grad_direct(h_cur, p, dx)
        h_new = _project_sum(h_cur - lr * grad, target)
        if _exact_c5(h_new) <= _exact_c5(h_cur):
            h_cur = h_new
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        n0 -= 1
        h0 = h0[:n0]
    h_start = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)

    tau_list_full = [0.08, 0.05, 0.03, 0.018, 0.01, 0.005, 0.0025, 0.0012]
    tau_list_short = [0.04, 0.02, 0.01, 0.005, 0.002]

    # Phase 1: SLSQP annealing on primary seed (30%)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _anneal(h_start, tau_list_full, d1, maxiter=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 1b: Continuation from best with tighter tau (10%)
    d1b = t0 + budget_s * 0.40
    tau_tight = [0.015, 0.008, 0.004, 0.002, 0.001]
    h1b, c5_1b = _anneal(best_h, tau_tight, d1b, maxiter=350)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()

    # Phase 2: Diverse starting points (20%)
    d2 = t0 + budget_s * 0.60
    transforms = [best_h.copy()]
    transforms.append(_project_sum(best_h[::-1].copy(), n0 // 2))
    transforms.append(_project_sum(1.0 - best_h, n0 // 2))
    transforms.append(_project_sum(0.5 * (best_h + best_h[::-1]), n0 // 2))
    for _ in range(4):
        transforms.append(_perturb(best_h, rng, 0.06))
    transforms.append(_to_binary(best_h))

    for si, h_seed in enumerate(transforms):
        if time.time() > d2:
            break
        n_s = len(h_seed)
        sub_dl = min(d2, time.time() + budget_s * 0.035)
        tlist = tau_list_short if time.time() > d2 - budget_s * 0.10 else tau_list_full[:5]
        h_s, c5_s = _anneal(h_seed, tlist, sub_dl, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Exact polishing (20%)
    d3 = t0 + budget_s * 0.80
    n_cur = len(best_h)
    # Random pair polish
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h3, c5_3 = _random_pair_polish(best_h, sub_dl, n_pairs=8000, seed=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Binary SA + pair sweep
    if time.time() < d3:
        h_b = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=50000, T_init=0.004, seed=500)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_b2, c5_b2 = _pair_sweep(best_h, sub_dl2, max_rounds=25)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # More pair polish on best
    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h3b, c5_3b = _random_pair_polish(best_h, sub_dl, n_pairs=6000, seed=300)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-n exploration (10%)
    d4 = t0 + budget_s * 0.90
    n_best = len(best_h)
    n_offsets = [-20, -10, -6, -4, -2, 2, 4, 6, 10, 20]
    tried = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _anneal(h_r, tau_list_short[:3], sub_dl, maxiter=150)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
        elif c5_r < best_c5 + 0.001:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _random_pair_polish(h_r, sub_dl2, n_pairs=3000, seed=n_try)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                n_best = n_try

    # Phase 5: Final polish (10%)
    d5 = deadline
    n_fin = len(best_h)
    # SLSQP final tight
    sub_dl = min(d5, time.time() + budget_s * 0.03)
    h_fin, c5_fin = _anneal(_project_sum(best_h.copy(), n_fin // 2),
                            [0.003, 0.0015, 0.0008], sub_dl, maxiter=300)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()

    # Projected gradient with very small tau
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_pg = _proj_grad_descent(best_h, 0.002, lr=0.02, n_steps=200, deadline=sub_dl)
        c5_pg = _exact_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Final pair polish
    if time.time() < d5 - 2:
        h_fp, c5_fp = _random_pair_polish(best_h, d5, n_pairs=5000, seed=999)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final validation and return
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
