    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 80)
    grid_f = np.empty(80)
    for idx in range(80):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(79, best_idx + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx),
             (grid_t[best_idx], grid_f[best_idx])]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _active_lp(h, scores, radius, n_active, gap):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta, radius):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init, max_iters, use_momentum):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 10
    momentum = np.zeros(n)
    mom_alpha = 0.3
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        d = _active_lp(h, scores, radius, n_active, gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_direction(h, scores, 25.0, radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.65
                gap *= 1.8
                n_active = min(n_active + 4, 35)
                if radius < 1e-7:
                    break
                continue
        if use_momentum:
            momentum = mom_alpha * momentum + (1.0 - mom_alpha) * d
            mn = float(np.max(np.abs(momentum)))
            if mn > 1e-15:
                d_mix = momentum / mn * float(np.max(np.abs(d)))
            d = d_mix
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 5e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.35)
            gap = max(gap * 0.85, 0.0008)
            n_active = max(n_active - 1, 6)
            mom_alpha = min(mom_alpha * 1.02, 0.6)
        else:
            stall += 1
            radius *= 0.7
            gap *= 1.4
            n_active = min(n_active + 3, 30)
            mom_alpha = max(mom_alpha * 0.9, 0.1)
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.06
            if stall % 50 == 0:
                gap = 0.003
                n_active = 10
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    rng = np.random.default_rng(12345)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(3, len(ones) // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.3)
            if c_try < cur_c5 - 5e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter, T_init):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    nf = rng.integers(3, 7)
    freqs = rng.uniform(0.5, 10.0, size=nf)
    phases = rng.uniform(0, 2 * np.pi, size=nf)
    amplitudes = rng.uniform(0.3, 1.0, size=nf)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.4 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _random_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def _structured_initial(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    step = max(2, n // 100)
    positions = list(range(0, n, step))
    rng.shuffle(positions)
    for i in range(n // 2):
        pos = positions[i % len(positions)] + (i // len(positions)) * len(positions)
        if pos < n:
            h[pos] = 1.0
    remaining = n // 2 - int(np.sum(h))
    if remaining > 0:
        zero_idx = np.where(h == 0)[0]
        picks = rng.choice(zero_idx, size=min(remaining, len(zero_idx)), replace=False)
        h[picks] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 180
        h0 = _random_balanced(n0, rng)

    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Initial local optimization on parent (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _local_opt(h_init, d1, 0.12, 3000, True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Complement and basic variants (5%)
    d2 = t0 + budget_s * 0.17
    h_comp = _project(1.0 - best_h, len(best_h) / 2.0)
    c5_comp = _exact_c5(h_comp)
    if c5_comp < best_c5:
        best_c5 = c5_comp
        best_h = h_comp.copy()

    # Phase 3: Multi-resolution search (35%)
    d3 = t0 + budget_s * 0.52
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    # Try a range of n values, prioritizing larger (better dx) and nearby
    n_candidates = []
    for dn in range(0, 50, 2):
        n_candidates.append(n_curr + dn)
    for dn in range(0, 50, 2):
        n_candidates.append(n_curr - dn - 2)
    n_candidates = [x for x in n_candidates if 80 <= x <= 260 and x % 2 == 0]
    seen = set()
    n_candidates = [x for x in n_candidates if x not in seen and not seen.add(x)]

    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.01)
        h_r, c5_r = _local_opt(h_r, sub_dl, 0.10, 600, True)
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.005)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, 40)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Binary optimization (18%)
    d4 = t0 + budget_s * 0.70
    n_b = len(best_h)
    h_bin = _to_binary(best_h)
    # Try a few different initial temperatures and iterations
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, 50000, 0.004)
    if time.time() < d4:
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_bin2, c5_bin2 = _pair_sweep(h_bin, sub_dl2, 100)
        if c5_bin2 < c5_bin:
            h_bin, c5_bin = h_bin2, c5_bin2
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary result
    if time.time() < d4 - 2:
        h_pol = _project(h_bin.copy(), n_b / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl3, 0.06, 800, True)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Basin hopping with diverse perturbations (15%)
    d5 = t0 + budget_s * 0.85
    scale = 0.03
    n_restarts = 0
    max_restarts = 300
    while time.time() < d5 and n_restarts < max_restarts:
        n_restarts += 1
        # Vary perturbation type
        rtype = rng.random()
        if rtype < 0.5:
            h_pert = _smooth_perturb(best_h, scale, rng)
        elif rtype < 0.75:
            h_pert = _project(best_h + scale * 0.5 * rng.normal(0, 1, len(best_h)), len(best_h) / 2.0)
        else:
            h_pert = _project(1.0 - best_h + scale * rng.normal(0, 1, len(best_h)), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.003)
        h_t, c5_t = _local_opt(h_pert, sub_dl, 0.05, 300, False)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.06)

    # Phase 6: Final polishing (13%)
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    # Multi-pass polishing with decreasing radius
    d6 = deadline - 5
    for p in range(4):
        if time.time() > d6:
            break
        r_init = [0.06, 0.03, 0.015, 0.008][p]
        sub_dl = min(d6, time.time() + budget_s * 0.025)
        h_f, c5_f = _local_opt(h_f, sub_dl, r_init, 2000, (p == 0))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Pair sweep between passes
        if p < 3 and time.time() < d6:
            sub_dl2 = min(d6, time.time() + budget_s * 0.01)
            h_f, c5_f2 = _pair_sweep(_project(best_h.copy(), n_f / 2.0), sub_dl2, 60)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f.copy()

    # Final: try a few more n values around best
    if deadline - time.time() > 10:
        n_best = len(best_h)
        for dn in [2, -2, 4, -4, 6, -6]:
            if time.time() > deadline - 2:
                break
            n_try = n_best + dn
            if n_try < 60 or n_try > 300 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r, c5_r = _local_opt(h_r, deadline - 1, 0.04, 500, False)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
