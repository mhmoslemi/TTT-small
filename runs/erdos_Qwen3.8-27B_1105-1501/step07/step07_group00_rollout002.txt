def _project(v, target):
    # Water-filling projection to [0,1] with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:
            hi = mid
    theta = (lo + hi) / 2.0

def _grad_rows(h, lags):

            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    # Check endpoints and grid best
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx),
             (grid_t[best_idx], grid_f[best_idx])]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _gradient_direction(h, scores, beta=30.0, radius=0.15):
    """Soft-max weighted gradient direction."""
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Clip to feasible region
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt_full(h, deadline, use_full_lp=True, radius_init=0.15, max_iters=3000):
    """Local optimization with full or active LP."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 20
    use_full = use_full_lp
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        if use_full and n <= 200:
            d = _full_lp_direction(h, scores, radius)
        else:
            d = _active_lp_direction(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-13:
            d = _gradient_direction(h, scores, beta=30.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-13:
                stall += 1
                radius *= 0.6
                gap *= 1.5
                n_active = min(n_active + 5, 40)
                if radius < 1e-7:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
            gap = max(gap * 0.85, 0.001)
            n_active = max(n_active - 1, 10)
            # Occasionally switch to full LP
            if stall == 0 and len(best_h) <= 180 and not use_full:
                use_full = True
        else:
            stall += 1
            radius *= 0.7
            gap *= 1.3
            n_active = min(n_active + 3, 35)
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.1
                if use_full:
                    use_full = False
    return best_h, best_c5

def _fourier_perturb(h, amplitude, rng):
    """Perturb h using random Fourier modes."""
    n = len(h)
    x = np.arange(n, dtype=float)
    n_modes = 5
    freqs = rng.integers(1, max(2, n // 4), size=n_modes)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amplitudes = rng.uniform(0.5, 1.0, size=n_modes)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude / (n_modes * 0.5)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    """Convert to balanced binary."""
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_block_sa(h_bin, deadline, max_iter=50000, T_init=0.002, block_size=2):
    """Binary SA with block moves."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0
    n_ones = n // 2
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        # Random block move: swap a block of 1s with a block of 0s
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Try single swap first (simpler and often effective)
        if cur[i] != cur[j]:
            h_new = cur.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            new_c5 = _exact_c5(h_new)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                    stall = 0
                T = max(T * 0.9997, 1e-5)
            else:
                stall += 1
        else:
            # Find a valid swap partner
            if cur[i] == 1:
                Z = np.where(cur == 0)[0]
            else:
                Z = np.where(cur == 1)[0]
            if len(Z) == 0:
                break
            j2 = int(Z[rng.integers(0, len(Z))])
            h_new = cur.copy()
            h_new[i], h_new[j2] = h_new[j2], h_new[i]
            new_c5 = _exact_c5(h_new)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
