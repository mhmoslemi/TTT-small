def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):

    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14

        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-15:
        return 0.0, float(np.max(S0) * dx)
    
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    
    # Stationary points of each quadratic
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)

                        cands.append(ar)
            else:

                if disc >= 0:

                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)

def _soft_grad(h, beta):
    """Compute soft-max weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _block_avg(v, bs):
    """Block average smoothing."""
    n = len(v)
    nb = n // bs
    if nb < 1:
        return v.copy()
    out = np.zeros(n)
    for b in range(nb):
        out[b*bs:(b+1)*bs] = np.mean(v[b*bs:(b+1)*bs])
    if nb * bs < n:
        out[nb*bs:] = np.mean(v[nb*bs:])
    out -= np.mean(out)
    return out

def _gen_dirs_v2(h, beta_list, block_sizes, rng):
    """Generate multiple directions with different smoothing."""
    n = len(h)
    dirs = []
    for beta in beta_list:

        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
        for bs in block_sizes:
            db = _block_avg(g, bs)
            nrm = np.linalg.norm(db)
            if nrm > 1e-10:
                dirs.append(db / nrm)
    # Add a random smooth direction
    x = np.arange(n, dtype=float)

    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-10:
        dirs.append(d / nrm)
    return dirs

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

        w /= ws

        wl = w[n - 1 + l]

            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]

    return -g

def _block_avg(v, bs):
    n = len(v)
    nb = n // bs
    if nb < 1:
        return v - np.mean(v)
    out = np.zeros(n)
    for b in range(nb):
        out[b*bs:(b+1)*bs] = np.mean(v[b*bs:(b+1)*bs])
    if nb * bs < n:
        out[nb*bs:] = np.mean(v[nb*bs:])
    out -= np.mean(out)
    return out

def _gauss_smooth(v, sigma):
    n = len(v)
    r = int(3 * sigma)
    if r < 1:
        return v.copy()
    k = np.exp(-0.5 * (np.arange(-r, r+1) / sigma) ** 2)
    k /= k.sum()
    out = np.convolve(v, k, mode='same')
    out -= np.mean(out)
    return out

                cur = hn
                cc = nc
                S0_cache = np.correlate(cur, 1.0 - cur, mode='full')
                ones.remove(o1)
                ones.remove(o2)
                ones.extend([z1, z2])
                zeros.remove(z1)
                zeros.remove(z2)
                zeros.extend([o1, o2])
                if cc < bc - 1e-14:
                    bc = cc
                    bh = cur.copy()
                    stall = 0
                T *= 0.9995
            else:
                stall += 1
        else:
            oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]
            hn = cur.copy()
            hn[oi] = 0.0
            hn[oz] = 1.0
            nc = _exact_c5(hn)
            dl = nc - cc
            if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
                cur = hn
                cc = nc
                S0_cache = np.correlate(cur, 1.0 - cur, mode='full')
                ones.remove(oi)
                ones.append(oz)
                zeros.remove(oz)
                zeros.append(oi)
                if cc < bc - 1e-14:
                    bc = cc
                    bh = cur.copy()
                    stall = 0
                T *= 0.9995
            else:
                stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _structured_perturb(h, rng, scale):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 15.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n * 2.0 + p)
    pert += 0.5 * scale * rng.normal(0, 1, n)
    mxp = np.max(np.abs(pert))
    if mxp > 1e-10:
        pert *= scale / mxp
    return pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: High-beta focused gradient descent (28%)
    d1 = t0 + budget_s * 0.28
    h1, c1 = _local_opt(h_init, d1, r0=0.12, stall_lim=50,
                        betas=[500.0, 200.0, 80.0], smooths=[4, 8, 16, 32],
                        momentum=0.2, max_dirs=12)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Medium-beta with momentum (17%)
    d2 = t0 + budget_s * 0.45
    h2, c2 = _local_opt(_project(best_h.copy(), len(best_h)/2.0), d2, r0=0.10,
                        stall_lim=45, betas=[150.0, 50.0, 20.0], smooths=[6, 12, 24],
                        momentum=0.3, max_dirs=10)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Gradient-guided pair optimization (13%)
    d3 = t0 + budget_s * 0.58
    h3, c3 = _pair_opt(_project(best_h.copy(), len(best_h)/2.0), d3, max_it=60000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Another gradient pass from improved point (12%)
    d4 = t0 + budget_s * 0.70
    h4, c4 = _local_opt(_project(best_h.copy(), len(best_h)/2.0), d4, r0=0.08,
                        stall_lim=40, betas=[300.0, 100.0, 40.0], smooths=[4, 8, 16],
                        momentum=0.15, max_dirs=8)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: Binary SA with multi-point moves (15%)
    d5 = t0 + budget_s * 0.85
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _bin_sa(hbin, d5, mx=30000, Ti=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d5:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d5, time.time() + budget_s * 0.025)
        hp, cp = _local_opt(hp, sd, r0=0.08, stall_lim=30,
                           betas=[200.0, 80.0, 30.0], smooths=[6, 12],
                           momentum=0.2, max_dirs=8)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 6: Multi-resolution (7%)
    d6 = t0 + budget_s * 0.92
    nc = len(best_h)
    tried = set()
    for dn in [-8, -4, 4, 8, -12, 12, -16, 16]:
        if time.time() > d6:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d6, time.time() + budget_s * 0.012)
        hr, cr = _local_opt(hr, sd, r0=0.10, stall_lim=25,
                           betas=[200.0, 80.0, 30.0], smooths=[6, 12],
                           momentum=0.2, max_dirs=8)
        if cr < best_c and time.time() < d6 - 2:
            hb2 = _to_bin(hr)
            sd2 = min(d6, time.time() + budget_s * 0.006)
            hb2, cb2 = _bin_sa(hb2, sd2, mx=8000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb2
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 7: Basin hopping + final polish (8%)
    d7 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d7 - 5 and n_re < 40:
        n_re += 1
        nc = len(best_h)
        pert = _structured_perturb(best_h, rng, scale)
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7 - 3, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15,
                           betas=[100.0, 40.0], smooths=[6, 12],
                           momentum=0.0, max_dirs=6)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline - 2, r0=0.03, stall_lim=25,
                           betas=[500.0, 200.0, 80.0], smooths=[4, 8],
                           momentum=0.0, max_dirs=8)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=8000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
