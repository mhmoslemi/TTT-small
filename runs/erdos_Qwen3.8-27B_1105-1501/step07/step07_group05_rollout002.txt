def _exact_c5(h):

def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _topk_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    idx = np.argsort(-sc)[:K]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / K
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _block_smooth_grad(h, bs, beta):
    g = _softmax_grad(h, beta)
    n = len(g)
    nb = n // bs

    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

                    rad = 0.12
                elif stall % 40 == 0:
                    rad = 0.1
    return best_h, best_c

def _pair_opt(h, deadline, max_it=50000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(77777)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=20000, Ti=0.002):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _focussed_opt(h, deadline, n_iter_max=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(54321)
    K_sched = [1, 2, 3, 5, 3, 5, 8, 10, 15, 20]
    stall = 0
    it = 0
    while time.time() < deadline and it < n_iter_max:
        it += 1
        if stall > 50:
            break
        K = K_sched[it % len(K_sched)]
        dirs = []
        g = _topk_grad(h, min(K, 2 * n - 1))
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
        g2 = _softmax_grad(h, 30.0 + 20.0 * (it % 5))
        nrm2 = np.linalg.norm(g2)
        if nrm2 > 1e-10:
            dirs.append(g2 / nrm2)
        if len(dirs) < 3:
            g3 = _block_smooth_grad(h, 12, 40.0)
            nrm3 = np.linalg.norm(g3)
            if nrm3 > 1e-10:
                dirs.append(g3 / nrm3)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            a, c_try = _line_search(h, d, dx, 0.15)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = _project(h + best_a * best_d, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            pert = rng.normal(0, 0.02, n)
            pert -= np.mean(pert)
            hp = _project(h + pert, target)
            cp = _exact_c5(hp)
            if cp < cur - 1e-12:
                h = hp
                cur = cp
                stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Quick local opt (8%)
    d1 = t0 + budget_s * 0.08
    h1, c1 = _local_opt(h_init, d1, r0=0.15, stall_lim=40, nd=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Focused top-K optimization (15%)
    d2 = t0 + budget_s * 0.23
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _focussed_opt(h2, d2, n_iter_max=200)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair swap (12%)
    d3 = t0 + budget_s * 0.35
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=40000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Binary SA + local refinement (13%)
    d4 = t0 + budget_s * 0.48
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d4, mx=10000, Ti=0.0015)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.02)
        hp, cp = _local_opt(hp, sd, r0=0.08, stall_lim=30, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Multi-resolution (18%)
    d5 = t0 + budget_s * 0.66
    nc = len(best_h)
    offsets = [-6, -4, -2, 2, 4, 6, -10, 10, -14, 14, -18, 18, 8, -8]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, nd=4)
        if cr < best_c + 0.003 and time.time() < d5 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=4000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d5 - 2:
            sd3 = min(d5, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive local + pair (20%)
    d6 = t0 + budget_s * 0.86
    nc = len(best_h)
    for r_init in [0.12, 0.08, 0.04, 0.02]:
        if time.time() > d6 - 3:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.05)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=50, nd=5)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 5:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.025)
            h6, c6 = _pair_opt(h6, sd, max_it=12000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()
    if time.time() < d6 - 3:
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.03)
        h6, c6 = _focussed_opt(h6, sd, n_iter_max=100)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Basin hopping (10%)
    d7 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d7 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        if deadline - time.time() < 1.0:
            break
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.3, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.3) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.25 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.006)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=18, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.09, scale * 1.08)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=25, nd=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=4000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
