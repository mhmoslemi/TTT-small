```python
import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x > 1e-9) & (x < 0.9999999)
        else:
            mask = (x > 1e-9) & (x < 0.9999999)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(h):
    """Build the linearization matrix A.
    A has shape (2n-1, n) where A[k, i] is the coefficient of h[i] in raw[k].
    raw[k] = sum_i h[i] * u[i + k - (n-1)] where u = 1-h
    So A[k, i] = u[i + k - (n-1)] if 0 <= i + k - (n-1) < n, else 0.
    """
    n = len(h)
    u = 1.0 - h
    # For lag d = k - (n-1), valid i are: max(0, -d) <= i <= min(n-1, n-1-d)
    A = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        k = d + (n-1)
        # valid i: max(0, -d) <= i <= min(n-1, n-1-d)
        i_start = max(0, -d)
        i_end = min(n-1, n-1-d)
        if i_start <= i_end:
            # A[k, i] = u[i+d] for i in [i_start, i_end]
            A[k, i_start:i_end+1] = u[i_start+d:i_end+d+1]
    return A

def _solve_lp(h_current, solver='CLARABEL', time_limit=5):
    """Solve the LP linearization of the Erdős overlap problem."""
    n = len(h_current)
    target = n / 2.0
    u = 1.0 - h_current
    A = _build_A(h_current)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        A @ h_var <= t,
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        t >= 0
    ]
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False, time_limit=time_limit*1000)
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return None
```

```python
import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x > 1e-9) & (x < 0.9999999)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(h_old):
    n = len(h_old)
    u = 1.0 - h_old
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        d = m - (n-1)
        i_start = max(0, d)
        i_end = min(n-1, n-1+d)
        if i_start <= i_end:
            A[m, i_start:i_end+1] = u[i_start-d : i_end-d+1]
    return A

def _lp_step(h_old, mu=0.1, solver_list=None):
    n = len(h_old)
    target = n / 2.0
    A = _build_A(h_old)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        A @ h_var <= t,
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        t >= 0
    ]
    
    obj = cp.Minimize(t + (mu / 2.0) * cp.sum_squares(h_var - h_old))
    prob = cp.Problem(obj, constraints)
    
    if solver_list is None:
        solver_list = ['CLARABEL', 'ECOS', 'SCS', 'GLOP', 'PDLP']
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            status = prob.status
            if status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                h_new = np.array(h_var.value).flatten().copy()
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = _project(h_new, target)
                return h_new, status
        except Exception:
            continue
    return None, None

def _line_search_lp(h_old, h_lp, dx):
    """Line search on segment [h_old, h_lp]."""
    n = len(h_old)
    target = n / 2.0
    raw_old = np.correlate(h_old, 1.0 - h_old, mode='full')
    best_alpha = 0.0
    best_val = float(np.max(raw_old)) * dx
    
    # Try several alphas
    alphas = [0.0, 0.25, 0.5, 0.75, 1.0]
    for a in alphas:
        h_try = h_old + a * (h_lp - h_old)
        h_try = _project(h_try, target)
        val = _exact_c5(h_try)
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = a
    
    # Golden section refinement
    if best_alpha > 0:
        lo = max(0, best_alpha - 0.25)
        hi = min(1, best_alpha + 0.25)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h_old + c1 * (h_lp - h_old), target))
        f2 = _exact_c5(_project(h_old + c2 * (h_lp - h_old), target))
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h_old + c1 * (h_lp - h_old), target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h_old + c2 * (h_lp - h_old), target))
        opt = (lo + hi) / 2.0
        val_opt = _exact_c5(_project(h_old + opt * (h_lp - h_old), target))
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    
    h_new = _project(h_old + best_alpha * (h_lp - h_old), target)
    return h_new, best_val

def _smooth_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w * u[:m]
                g[:m] -= w * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w * u[gp:]
                g[gp:] -= w * h[:m]
    return g

def _gradient_optimize(h, deadline, beta=20.0, radius=0.2, stall_limit=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(42)
    
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        g = _smooth_grad(h, w)
        g -= np.mean(g)
        
        # Direction: negative gradient
        d = -g.copy()
        nm = np.linalg.norm(d)
        if nm < 1e-14:
            break
        d = d / nm * radius
        
        # Line search
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        
        # Find max step
        a_hi = 1.0
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
        a_hi = max(a_hi, 1e-12)
        
        # Quadratic line search
        candidates = [0.0, a_hi]
        n_top = 10
        top_idx = np.argsort(-raw)[:n_top]
        for idx in top_idx:
            if Q[idx] > 1e-14:
                av = T[idx] / (2.0 * Q[idx])
                if 1e-10 < av < a_hi - 1e-10:
                    candidates.append(av)
        grid = np.linspace(0.0, a_hi, 20)
        candidates.extend(grid.tolist())
        
        best_alpha = 0.0
        best_val = mx * dx
        for alpha in candidates:
            alpha = max(0.0, min(alpha, a_hi))
            val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            break
            
        h_new = _project(h + best_alpha * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.8
            if radius < 0.01:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
```
