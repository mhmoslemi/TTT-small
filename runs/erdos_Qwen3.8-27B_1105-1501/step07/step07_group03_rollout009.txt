        if time.time() > deadline:
            break
        ones_list = list(ones)
        zeros_list = list(zeros)
        best_delta = 1e-10
        best_pair = None
        for oi in ones_list:
            for oz in zeros_list:

                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0

                if c_try < cur_c5 - best_delta:
                    best_delta = cur_c5 - c_try
                    best_pair = (oi, oz, c_try)

        ones.discard(oi)
        zeros.discard(oz)
        ones.add(oz)
        zeros.add(oi)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur.copy()
    
    return best_h, best_c5
