import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, R0, t_cap=1.0):
    """Exact line search along direction d. Returns (best_t, best_value)."""
    n = len(h)
    # Compute A and B
    o = 1.0 - h
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -t_cap
    t_hi = t_cap
    for i in range(n):
        di = d[i]
        if di > 1e-12:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-12:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    
    if t_hi - t_lo < 1e-12:
        return 0.0, float(np.max(R0))
    
    # Grid search
    n_grid = 100
    ts = np.linspace(t_lo, t_hi, n_grid)
    # f(t) = max_m (R0[m] + t*A[m] - t^2*B[m])
    # Vectorized: for each t, compute the max
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1*c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2*c2 * B))
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1*c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2*c2 * B))
    
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(R0 + opt_t * A - opt_t*opt_t * B))
    
    # Also check endpoints
    v_lo = float(np.max(R0 + t_lo * A - t_lo*t_lo * B))
    v_hi = float(np.max(R0 + t_hi * A - t_hi*t_hi * B))
    
    best_val = min(opt_val, float(vals[best_idx]), v_lo, v_hi)
    if best_val == opt_val:
        return opt_t, best_val
    if best_val == v_lo:
        return t_lo, v_lo
    if best_val == v_hi:
        return t_hi, v_hi
    return float(ts[best_idx]), best_val

def _lp_direction(h, peak_indices):
    """Find a direction u via LP that decreases all active peaks."""
    n = len(h)
    L = 2 * n - 1
    k = len(peak_indices)
    
    # Build gradient matrix for active peaks only
    G = np.zeros((k, n))
    for idx, m in enumerate(peak_indices):
        if m >= n - 1:
            lag = m - (n - 1)
            if lag >= 0:
                mm = n - lag
                G[idx, lag:] += (1.0 - h[:mm])
                G[idx, :mm] -= h[lag:]
            else:
                p = -lag
                mm = n - p
                G[idx, p:] += (1.0 - h[:mm])
                G[idx, :mm] -= h[p:]
        # Handle negative lags
        else:
            lag = m - (n - 1)  # negative
            p = -lag
            mm = n - p
            G[idx, p:] += (1.0 - h[:mm])
            G[idx, :mm] -= h[p:]
    
    # LP: minimize -beta
    # vars: [sp(n); sn(n); beta]
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[-1] = -1.0  # minimize -beta
    
    # Inequality constraints: G @ u + beta <= 0 for each peak
    # G @ (sp - sn) + beta <= 0
    # [G, -G, 1] @ [sp; sn; beta] <= 0
    A_ub = np.zeros((k, nv))
    A_ub[:, :n] = G
    A_ub[:, n:2*n] = -G
    A_ub[:, 2*n] = 1.0
    b_ub = np.zeros(k)
    
    # Equality constraints:
    # sum(sp) - sum(sn) = 0
    # sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    # Bounds
    bounds = [(0, 1)] * (2 * n) + [(-10.0, 10.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    u = res.x[:n] - res.x[n:2*n]
    beta = res.x[2*n]
    
    if beta <= 1e-10:
        return None  # No meaningful improvement
    
    # Normalize to unit L1
    l1 = float(np.sum(np.abs(u)))
    if l1 < 1e-10:
        return None
    # u is already unit L1 by construction (sum(sp)+sum(sn)=1 and sp,sn>=0)
    # Actually sum(us) + sum(us) = 1 where us = sp + sn, so ||u||_1 <= 1
    # Let's just return u as is
    
    return u

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    h1 = 1.0 - h
    for i in range(n):
        e_i = np.zeros(n)
        e_i[i] = 1.0
        G[:, i] = np.correlate(e_i, h1, mode='full') - np.correlate(h, e_i, mode='full')
    return G

def _pair_line_search(h, i, j, R0, t_cap=0.5):
    """Line search for moving mass from j to i."""
    d = np.zeros(len(h))
    d[i] = 1.0
    d[j] = -1.0
    return _line_search(h, d, R0, t_cap)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, T_init=0.005, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    while time.time() < deadline:
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5

                continue
            top_idx = np.argsort(R0)[-pc:]
            u = _lp_direction(h, top_idx)
            if u is not None:
                t_opt, _ = _line_search(h, u, R0, t_cap)
                if abs(t_opt) > 1e-12:
                    h_try = h + t_opt * u
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < cur_c5 - 1e-13:
                        h = h_try
                        cur_c5 = c5_try
                        G_cache = None
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        found = True
                        break

        if found:
            continue

        # Fallback: gradient-guided pair moves
        if G_cache is None or not np.array_equal(h, G_cache_h):
            G_cache = _grad_matrix(h)
            G_cache_h = h.copy()
        M = float(np.max(R0))
        beta_w = 25.0
        w = np.exp(beta_w * (R0 - M))
        sw = float(np.sum(w))
        if sw > 1e-30:
            w /= sw
        coord_grad = G_cache.T @ w
        sort_idx = np.argsort(coord_grad)
        n_pairs = min(40, n // 2)
        sources = sort_idx[-n_pairs:]
        sinks = sort_idx[:n_pairs]

        best_c5_try = cur_c5
        best_step = None
        for ii in range(n_pairs):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_opt, _ = _pair_line_search(h, sink, source, R0, t_cap=0.3)
            if abs(t_opt) < 1e-12:
                continue
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-13:
                best_c5_try = c5_try
                best_step = h_try

        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            G_cache = None
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            # Random perturbations to escape
            esc = False
            for _ in range(15):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
                t_opt, _ = _pair_line_search(h, a, b, R0, t_cap=0.25)
                if abs(t_opt) > 1e-12:
                    h_try = h.copy()
                    h_try[a] += t_opt
                    h_try[b] -= t_opt
                    c5_try = _exact_c5(h_try)
                    if c5_try < cur_c5 - 1e-13:
                        h = _project(h_try, target)
                        cur_c5 = c5_try
                        G_cache = None
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        esc = True
                        break
            if not esc:
                stall += 1
                if stall % 50 == 0:
                    # Larger random step
                    a = int(rng.integers(0, n))
                    b = int(rng.integers(0, n))
                    if a != b:
                        h[a] += 0.05
                        h[b] -= 0.05
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        G_cache = None

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, T_init=0.005, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    while time.time() < deadline:
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or (T > 1e-7 and rng.random() < np.exp(-delta / max(T, 1e-12))):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9994
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                if oi == oz:
                    continue
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Primary LP optimization (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _local_optimize_lp(h_init, d1, t_cap=0.8,
                                   peak_counts=[4, 6, 10, 14, 20, 25], seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Resolution exploration (25% time)
    d2 = t0 + budget_s * 0.55
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, 20, -20, 24, -24]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.014)
        h_r, c5_r = _local_optimize_lp(h_r, sub_dl, t_cap=0.6,
                                        peak_counts=[4, 6, 10, 15], seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.72
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.06)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, T_init=0.004, seed=777)
        sub_dl2 = min(d3, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary result
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _local_optimize_lp(h_pol, sub_dl3, t_cap=0.5,
                                                peak_counts=[4, 6, 10, 15], seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Second LP pass with different peaks (13% time)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _local_optimize_lp(h4, d4, t_cap=0.5,
                                   peak_counts=[3, 5, 8, 12, 18, 30, 40], seed=555)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final polish (7% time)
    d5 = deadline - 1.5
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _local_optimize_lp(h5, d5, t_cap=0.25,
                                   peak_counts=[4, 6, 10], seed=888)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
