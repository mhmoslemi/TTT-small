    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

    res = target - float(np.sum(x))

        if res > 0:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _corr_grad(h, n):
    """Compute the (L, n) Jacobian matrix where row r is dc_r/dh for lag k = r-(n-1)."""
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n-1), n):
        r = k + (n - 1)
        if k >= 0:
            m = n - k
            # dc/dh[j] = (1-h[j-k]) for j in [k, n-1] minus h[j+k] for j in [0, m-1]
            if m > 0:
                G[r, k:] += (1.0 - h[:m])
                G[r, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            # dc/dh[j] = (1-h[j+p]) for j in [0, m-1] minus h[j-p] for j in [p, n-1]
            if m > 0:
                G[r, :m] += (1.0 - h[p:])
                G[r, p:] -= h[:m]
    return G

def _nlp_objective(x, n):
    """Objective: minimize t (last element of x)."""
    return x[-1]

def _nlp_objective_grad(x, n):
    """Gradient of objective."""
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _nlp_ineq_fun(x, n):
    """Inequality constraints: t - dx * c_r(h) >= 0 for all r."""
    h = x[:n]
    t = x[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * corr

def _nlp_ineq_jac(x, n):
    """Jacobian of inequality constraints. Shape: (2n-1, n+1)."""
    h = x[:n]
    dx = 2.0 / n
    G = _corr_grad(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def _nlp_eq_fun(x, n):
    """Equality constraint: sum(h) - n/2 = 0."""
    return x[:n].sum() - n / 2.0

def _nlp_eq_jac(x, n):
    """Jacobian of equality constraint."""
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

    cons = [
        {'type': 'ineq', 'fun': lambda x: _nlp_ineq_fun(x, n), 
         'jac': lambda x: _nlp_ineq_jac(x, n)},
        {'type': 'eq', 'fun': lambda x: _nlp_eq_fun(x, n),
         'jac': lambda x: _nlp_eq_jac(x, n)}
    ]
    
    try:
        result = minimize(
            lambda x: _nlp_objective(x, n),
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': tol}
        )
    except Exception:
        return h0, _exact_c5(h0)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]

        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))

def _smooth_weights(S0, beta=20.0):
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    w = np.exp(beta * (S0 - M))
    s = float(np.sum(w))
    if s < 1e-30:

        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

            m = n - p
            G[row, p:] += (1.0 - h[:m])

    d2 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    seeds_phase2 = [1, 7, 13, 23, 42]
    for s in seeds_phase2:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        if s == 1:
            h_start = best_h.copy()
        elif s == 7:
            h_start = _smooth_seed(n_curr, s, shift=0.1, amp=0.25)
        elif s == 13:
            h_start = np.flip(best_h).copy()
            h_start = _project(h_start, n_curr / 2.0)
        elif s == 23:
            rng_s = np.random.default_rng(s)
            h_start = best_h + rng_s.normal(0, 0.03, n_curr)
            h_start = _project(h_start, n_curr / 2.0)
        else:
            h_start = _smooth_seed(n_curr, s, shift=0.3, amp=0.2)
        
        h_new, c5_new = _nlp_solve_single(h_start, n_curr, sub_dl, maxiter=200, active_tol=0.005)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Full-constraint NLP for best candidate (15% time)
    d3 = t0 + budget_s * 0.70
    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.08)
        h_full, c5_full = _nlp_solve_full(best_h, n_curr, sub_dl, maxiter=250)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_full.copy()
    
    # Phase 4: Pair-based refinement (20% time)
    d4 = t0 + budget_s * 0.90
    if time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.10)
        h_ref, c5_ref = _pair_optimize(best_h, sub_dl, radius=0.12, n_pairs=60,
                                       beta=25.0, stall_limit=100, seed=555)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        # Try nearby n values
        for dn in [-4, 4, -8, 8]:
            if time.time() > d4:
                break
            n_try = n_curr + dn
            if n_try < 64 or n_try > 260 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.02)
            h_r2, c5_r2 = _nlp_solve_single(h_r, n_try, sub_dl2, maxiter=100, active_tol=0.005)
            h_r3, c5_r3 = _pair_optimize(h_r2, min(d4, time.time() + budget_s * 0.015),
                                         radius=0.08, n_pairs=40, beta=25.0, stall_limit=40,
                                         seed=400 + dn)
            if c5_r3 < best_c5:
                best_c5 = c5_r3
                best_h = h_r3.copy()
    
    # Phase 5: Binary diversification (8% time)
    d5 = t0 + budget_s * 0.98
    if time.time() < d5:
        n_curr = len(best_h)
        sub_dl_bin = min(d5, time.time() + budget_s * 0.03)
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        h_bin2, c5_bin2 = _binary_sa(h_bin, sub_dl_bin, max_iter=15000, T_init=0.002, seed=777)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        # Polish binary result
        if time.time() < d5:
            sub_dl_pol = min(d5, time.time() + budget_s * 0.02)
            h_pol = _project(best_h.copy(), n_curr / 2.0)
            h_pol2, c5_pol = _pair_optimize(h_pol, sub_dl_pol, radius=0.06, n_pairs=40,
                                            beta=30.0, stall_limit=50, seed=888)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol2.copy()
    
    # Final: project and verify
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        # Fallback to parent
        final_c5 = _exact_c5(h_init)
        best_h = h_init.copy()
    
    return best_h, final_c5, n_final
