import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for theta
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_rows(h, peak_indices):
    """Compute gradient rows for specific lags."""
    n = len(h)
    G = np.zeros((len(peak_indices), n))
    for idx, m in enumerate(peak_indices):
        ell = m - (n - 1)
        g = np.zeros(n)
        # First term: (1-h[j-ell]) for valid j
        lo1 = max(0, ell)
        hi1 = min(n, n + ell)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1 - ell:hi1 - ell])
        # Second term: -h[j+ell] for valid j
        lo2 = max(0, -ell)
        hi2 = min(n, n - ell)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + ell:hi2 + ell]
        G[idx] = g
    return G

def _line_search(h, d, t_cap=1.0):
    """Exact quadratic line search along direction d (zero-sum)."""
    n = len(h)
    o = 1.0 - h
    R0 = np.correlate(h, o, mode='full')
    A = np.correlate(d, o, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Find feasible t range
    t_lo, t_hi = -t_cap, t_cap
    for i in range(n):
        if d[i] > 1e-14:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    
    if t_lo >= t_hi - 1e-14:
        return None, None
    
    # Grid search
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Golden section around best
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-10:
        best_t = float(ts[best_idx])
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
        # Check all candidates
        candidates = [lo, hi, (lo+hi)/2.0, float(ts[best_idx]), t_lo, t_hi]
        best_t = None
        best_val = float('inf')
        for tc in candidates:
            if t_lo - 1e-14 <= tc <= t_hi + 1e-14:
                tc = max(t_lo, min(t_hi, tc))
                vc = float(np.max(R0 + tc * A - tc * tc * B))
                if vc < best_val:
                    best_val = vc
                    best_t = tc
    
    if best_t is None:
        return None, None
    return best_t, best_val

def _lp_direction(h, R, peak_count=10):
    """Solve LP to find multi-coordinate descent direction."""
    n = len(h)
    L = 2 * n - 1
    M = float(np.max(R))
    
    # Select active peaks
    # Use top peaks that are within some tolerance of max
    thresh = M - 0.02  # peaks within 2% of max
    active = np.where(R >= thresh)[0]
    if len(active) < peak_count:
        active = np.argsort(R)[-peak_count:]
    else:
        # Take the top peak_count from active
        active = np.sort(active)
        # Prioritize the most active ones
        scores = R[active]
        order = np.argsort(scores)[::-1][:peak_count]
        active = active[order]
    
    peak_indices = active.tolist()
    G = _grad_rows(h, [int(m) for m in peak_indices])
    
    # LP: maximize beta s.t. G @ u <= -beta, ||u||_1 <= 1, sum(u) = 0
    # Variables: sp(n), sn(n), beta(1). Total: 2n+1
    nvar = 2 * n + 1
    c = np.zeros(nvar)
    c[-1] = -1.0  # minimize -beta
    
    # Inequality constraints: G_j @ (sp - sn) + beta <= 0
    # i.e., G_j @ sp - G_j @ sn + beta <= 0
    A_ub = np.zeros((len(peak_indices) + 1, nvar))
    b_ub = np.zeros(len(peak_indices) + 1)
    
    for j in range(len(peak_indices)):
        A_ub[j, :n] = G[j]
        A_ub[j, n:2*n] = -G[j]
        A_ub[j, -1] = 1.0
    
    # sum(sp + sn) <= 1
    A_ub[-1, :2*n] = 1.0
    b_ub[-1] = 1.0
    
    # Equality: sum(sp - sn) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    bounds = [(0, 1)] * (2 * n) + [(-10.0, 10.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n] - res.x[n:2*n]
            # Check it's meaningful
            if np.max(np.abs(u)) < 1e-10:
                return None
            # Ensure zero sum (should be from LP constraint)
            u -= np.mean(u)
            return u
        else:
            return None
    except Exception:

    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = sorted(np.where(cur > 0.5)[0].tolist())
    zeros = sorted(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.remove(oi)
                    ones.append(oz)
                    ones.sort()
                    zeros.remove(oz)
                    zeros.append(oi)
                    zeros.sort()
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP-based descent (40% time)
    d1 = t0 + budget_s * 0.40
    h1, c5_1 = _lp_descent(h_init, d1, seed=1, peak_counts=(10, 14, 8, 18, 12, 20))
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Resolution sweep (15% time)
    d2 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 260 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_descent(h_r, sub_dl, seed=200 + dn, peak_counts=(8, 12, 6, 14))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=777)
        sub_dl2 = min(d3, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _lp_descent(h_pol, sub_dl3, seed=333, peak_counts=(10, 14, 8))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Intensive refinement (20% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4, c5_4 = _lp_descent(h4, sub_dl, seed=555, peak_counts=(12, 16, 10, 20, 8, 24))
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Try nearby resolutions
    n_curr = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl2 = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_descent(h_r, sub_dl2, seed=444 + dn, peak_counts=(8, 12, 10, 14))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final polish (5% time)
    d5 = deadline - 1.0
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    h5, c5_5 = _lp_descent(h5, d5, seed=888, peak_counts=(14, 18, 10, 22, 16))
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
