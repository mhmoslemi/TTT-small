            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h, mode=0):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        r_pos = mid + d
        r_neg = mid - d
        if mode == 0:
            G[r_pos, d:] += 1.0 - h[:m]
            G[r_pos, :m] += -h[d:]
            G[r_neg, :m] += 1.0 - h[d:]
            G[r_neg, d:] += -h[:m]
        else:
            G[r_pos, :m] += 1.0 - h[d:]
            G[r_pos, d:] += -h[:m]
            G[r_neg, d:] += 1.0 - h[:m]
            G[r_neg, :m] += -h[d:]
    return G

def _verify_orientation(h):
    n = len(h)
    test_cols = [0, n // 4, n // 2, (3 * n) // 4]
    delta = 1e-5
    G0 = _build_G(h, 0)
    G1 = _build_G(h, 1)
    s0 = _scores(h)
    err0 = 0.0
    err1 = 0.0
    for col in test_cols:
        h_p = h.copy()
        h_p[col] = min(1.0, max(0.0, h[col] + delta))
        s_p = _scores(h_p)
        dcol = h_p[col] - h[col]
        if abs(dcol) < 1e-12:
            continue
        fd = (s_p - s0) / dcol
        err0 += float(np.sum((fd - G0[:, col]) ** 2))
        err1 += float(np.sum((fd - G1[:, col]) ** 2))
    return 0 if err0 <= err1 else 1

def _smooth_obj(x):
    n = _g_n
    tau = _g_tau
    s = _scores(x) * _g_dx
    mval = float(np.max(s))
    u = (s - mval) / tau
    max_u = float(np.max(u))
    if max_u > 500:
        mask = (s >= mval - tau * 0.01)
        cnt = max(1, int(np.sum(mask)))
        return mval + tau * np.log(float(cnt))
    z = np.exp(u - max_u)
    lse = mval + tau * (max_u + np.log(float(np.sum(z))))
    return float(lse)

def _smooth_jac(x):
    n = _g_n
    tau = _g_tau
    dx = _g_dx
    s = _scores(x) * dx
    mval = float(np.max(s))
    u = (s - mval) / tau
    max_u = float(np.max(u))
    if max_u > 500:
        mask = (s >= mval - tau * 0.01)
        cnt = max(1, int(np.sum(mask)))
        probs = mask.astype(float) / cnt
    else:
        z = np.exp(u - max_u)
        probs = z / float(np.sum(z))
    G = _build_G(x, _g_grad_mode)
    grad = G.T @ probs * dx
    return grad

def _eq_fun(x):
    return [float(np.sum(x)) - _g_target]

def _eq_jac(x):
    return np.ones((1, len(x)))

def _smooth_anneal(h, deadline, n_tau=5, max_iter=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    _g_n = n
    _g_target = target
    _g_dx = dx
    _g_grad_mode = _verify_orientation(h)
    cur = _project_sum(h, target)
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    s = _scores(cur) * dx
    spread = float(np.max(s) - np.min(s))
    tau_init = max(0.008, spread * 0.35)
    tau_floor = 0.002
    taus = []
    t = tau_init
    for i in range(n_tau):
        taus.append(t)
        t *= 0.55
        if t < tau_floor:
            t = tau_floor
    seen = set()
    taus = [x for x in taus if not (round(x, 7) in seen or seen.add(round(x, 7)))]
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for ti, tau_val in enumerate(taus):
        if time.time() > deadline:
            break
        _g_tau = tau_val
        mi = max(15, int(max_iter * (1.0 - 0.3 * ti / max(1, len(taus)))))
        try:
            res = minimize(
                _smooth_obj, cur, method='SLSQP', jac=_smooth_jac,
                bounds=bounds, constraints=constraints,
                options={'maxiter': mi, 'ftol': 1e-12, 'disp': False}
            )
            if hasattr(res, 'x') and res.x is not None:
                x_new = _project_sum(np.array(res.x, dtype=float), target)
                c5_new = _exact_c5(x_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = x_new.copy()
                if c5_new < cur_c5 + 0.003:
                    cur = x_new
                    cur_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def _pair_polish(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-13:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best, best_c5

def _coord_polish(h, deadline, n_iter=6000, seed=99):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        improved = False
        for step in [0.03, 0.08, 0.15, 0.25, -0.03, -0.08, -0.15, -0.25]:
            trial = cur + step * (np.zeros(n) + 0)
            trial = cur.copy()
            trial[i] += step
            trial[j] -= step
            if trial.min() < -1e-9 or trial.max() > 1.0 + 1e-9:
                continue
            trial = _project_sum(trial, target)
            c_try = _exact_c5(trial)
            if c_try < cur_c5 - 1e-13:
                cur = trial
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best = cur.copy()
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
    return best, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    return _project_sum(h + pert, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _lp_direction_focused(h, scores, G, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    top_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_line_search(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    candidates = [(opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx),
                  (t_lo, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_loop(h, deadline, radius_init=0.1, top_k_init=4, max_top_k=10, stall_limit=60):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _build_G(h, _verify_orientation(h))
        p = _lp_direction_focused(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _lp_line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.08
                top_k = top_k_init
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        n0 -= 1
        h0 = h0[:n0]
    n0 = n0
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Generate diverse seeds
    seeds = [h_init.copy()]
    h_rev = h_init[::-1].copy()
    seeds.append(_project_sum(h_rev, n0 / 2.0))
    h_comp = 1.0 - h_init
    seeds.append(_project_sum(h_comp, n0 / 2.0))
    h_crev = 1.0 - h_rev
    seeds.append(_project_sum(h_crev, n0 / 2.0))
    h_avg = 0.5 * (h_init + h_rev)
    seeds.append(_project_sum(h_avg, n0 / 2.0))
    # Low-freq perturbation seeds
    for amp in [0.08, 0.15]:
        seeds.append(_perturb(h_init, rng, amp))

    # Sort seeds by quality
    seed_c5s = [_exact_c5(s) for s in seeds]
    order = np.argsort(seed_c5s)

    # Phase 1: Smooth annealing on top seeds (30%)
    d1 = t0 + budget_s * 0.30
    for si in order[:4]:
        if time.time() > d1:
            break
        h_seed = seeds[si].copy()
        sub_dl = min(d1, time.time() + budget_s * 0.05)
        h_sm, c5_sm = _smooth_anneal(h_seed, sub_dl, n_tau=5, max_iter=70)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()

    # Phase 1b: LP polish on smooth result
    if time.time() < d1:
        h_lp = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d1, time.time() + budget_s * 0.03)
        h_lp, c5_lp = _lp_loop(h_lp, sub_dl, radius_init=0.06, top_k_init=3, max_top_k=8, stall_limit=40)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 2: Binary optimization (15%)
    d2 = t0 + budget_s * 0.45
    if time.time() < d2:
        h_bin = _to_binary(best_h)
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _pair_polish(h_bin, sub_dl, max_rounds=40)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    if time.time() < d2:
        h_c = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_c, c5_c = _coord_polish(h_c, sub_dl, n_iter=5000, seed=77)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
    if time.time() < d2:
        h_lp = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_lp, c5_lp = _lp_loop(h_lp, sub_dl, radius_init=0.05, top_k_init=3, max_top_k=7, stall_limit=50)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 3: Basin hopping (20%)
    d3 = t0 + budget_s * 0.65
    scale = 0.04
    n_attempts = 0
    while time.time() < d3 and n_attempts < 150:
        n_attempts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_sm, c5_sm = _smooth_anneal(h_pert, sub_dl, n_tau=3, max_iter=40)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
            scale = max(0.008, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.06)

    # Phase 4: Multi-n exploration (12%)
    d4 = t0 + budget_s * 0.77
    n_cur = len(best_h)
    n_offsets = [-8, -4, 4, 8, -16, 16, -12, 12]
    tried = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _smooth_anneal(h_r, sub_dl, n_tau=4, max_iter=50)
        if time.time() < d4 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d4, time.time() + budget_s * 0.006)
            h_b, c5_b = _pair_polish(h_b, sub_dl2, max_rounds=20)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if time.time() < d4 - 3 and c5_r < best_c5 + 0.003:
            h_r2 = _project_sum(h_r, n_try / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _lp_loop(h_r2, sub_dl3, radius_init=0.05, top_k_init=3, max_top_k=7, stall_limit=30)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final intensive polish (16%)
    d5 = deadline
    n_final = len(best_h)
    # Smooth with low tau
    if time.time() < d5 - 8:
        h_f = _project_sum(best_h.copy(), n_final / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_f, c5_f = _smooth_anneal(h_f, sub_dl, n_tau=3, max_iter=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # LP with tight focus
    if time.time() < d5 - 5:
        h_f = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_f, c5_f = _lp_loop(h_f, sub_dl, radius_init=0.03, top_k_init=2, max_top_k=5, stall_limit=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # Binary polish
    if time.time() < d5 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _pair_polish(h_b, sub_dl, max_rounds=50)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Coord polish
    if time.time() < d5 - 2:
        h_f = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_f, c5_f = _coord_polish(h_f, sub_dl, n_iter=4000, seed=333)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # Final smooth
    if time.time() < d5 - 1:
        h_f = _project_sum(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _smooth_anneal(h_f, d5, n_tau=2, max_iter=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
