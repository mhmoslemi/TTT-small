def _thue_morse(n, offset=0):
    """Generate a balanced binary sequence based on Thue-Morse."""
    # Thue-Morse: t(k) = popcount(k) mod 2
    km = np.arange(n)
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(int(km[i])).count('1') % 2
    # Rotate by offset
    tm = np.roll(tm, offset)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(tm))
    if n_ones == n // 2:
        return tm
    elif n_ones > n // 2:

def _ga_step(pop_scores, pop, n, deadline, rng, elite_frac=0.2, children_per_gen=20):
    """One generation of GA."""
    n_one = n // 2
    pop = [p.copy() for p in pop]
    pop_scores = pop_scores.copy()
    
    n_elite = max(1, int(len(pop) * elite_frac))
    order = np.argsort(pop_scores)
    elites = [pop[i].copy() for i in order[:n_elite]]
    
    for _ in range(children_per_gen):
        if time.time() > deadline:
            break
        
        # Select parents
        i1 = order[rng.integers(len(order))]
        i2 = order[rng.integers(len(order))]
        p1 = pop[i1]
        p2 = pop[i2]
        
        # Crossover: random mask
        mask = rng.random(n) < 0.5
        child = np.where(mask, p1, p2)
        
        # Rebalance
        cnt = int(np.sum(child))
        target = n_one
        if cnt > target:
            ones_idx = np.where(child == 1)[0]
            flip = ones_idx[rng.integers(len(ones_idx), cnt - target)]
            child[flip] = 0
        elif cnt < target:
            zeros_idx = np.where(child == 0)[0]
            flip = zeros_idx[rng.integers(len(zeros_idx), target - cnt)]
            child[flip] = 1
        
        # Mutation: flip k pairs
        n_mut = rng.integers(1, max(2, n // 40))
        for _ in range(n_mut):
            oi = n_one if rng.random() < 0.5 else int(np.sum(child == 1))
            ones_idx = np.where(child == 1)[0]
            zeros_idx = np.where(child == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i_flip = ones_idx[rng.integers(len(ones_idx))]
            j_flip = zeros_idx[rng.integers(len(zeros_idx))]
            child[i_flip] = 0
            child[j_flip] = 1
        
        # Evaluate
        score = _raw_max(child)
        
        # Replace worst if better
        worst_idx = int(np.argmax(pop_scores))
        if score < pop_scores[worst_idx]:
            pop[worst_idx] = child
            pop_scores[worst_idx] = score
    
    # Update elites
    for e in elites:
        e_score = _raw_max(e)
        worst_idx = int(np.argmax(pop_scores))
        if e_score < pop_scores[worst_idx]:
            pop[worst_idx] = e.copy()
            pop_scores[worst_idx] = e_score
    
    return pop, pop_scores

def _raw_max(h):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw))

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
