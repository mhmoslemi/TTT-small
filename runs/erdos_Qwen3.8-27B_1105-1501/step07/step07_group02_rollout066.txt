def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on threshold
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _soft_weights(R, beta=20.0):
    """Compute softmax weights exp(-beta*(M - R_k)) / sum."""
    M = float(np.max(R))
    if M < 1e-15:
        return np.ones(len(R)) / len(R)
    # Shift for numerical stability
    vals = R - M  # all <= 0
    w = np.exp(-beta * vals)  # exp(0) = 1 for max, smaller for others
    s = float(np.sum(w))
    if s < 1e-30:
        # Fallback: uniform over top
        top_k = max(1, len(R) // 5)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _grad_jacobian(h):
    """Build the L x n Jacobian G where G[r, i] = dR[r]/dh[i]."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        r = k + n - 1
        if k >= 0:
            m = n - k
            # h[i] as first factor: i in [0, m-1], j = i+k
            G[r, :m] += (1.0 - h[k:k+m])  # (1-h[i+k]) for i in [0, m-1]
            # h[i] as second factor: i in [k, n-1], j = i-k
            G[r, k:] += (-h[:m])  # -h[i-k] for i in [k, n-1]
        else:

            m = n - p

            G[r, p:] += (1.0 - h[:m])  # (1-h[i-p]) for i in [p, n-1]
            # h[i] as second factor: i in [0, m-1], j = i+p
            G[r, :m] += (-h[p:])  # -h[i+p] for i in [0, m-1]
    return G
