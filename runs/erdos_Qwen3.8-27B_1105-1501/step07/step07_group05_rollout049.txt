    """Gradient of raw correlation at output index k."""
    n = len(h)
    L = k - (n - 1)  # signed lag

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, k):
    n = len(h)
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        if m > 0:
            g[:m] += (1.0 - h[L:L+m])
            g[L:] -= h[:m]
    else:
        s = -L
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:s+m])
            g[s:] -= h[:m]
    return g

def _compute_TQ(h, s):
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    return T, Q

def _line_search(h, s, dx, raw, max_alpha=None):
    n = len(h)
    T, Q = _compute_TQ(h, s)
    if max_alpha is None:
        max_alpha = 0.0
        pos = s > 1e-14
        neg = s < -1e-14
        if np.any(pos):
            max_alpha = min(max_alpha + 999, float(np.min((1.0 - h[pos]) / s[pos])) - 1e-10)
        if np.any(neg):
            max_alpha = min(max_alpha, float(np.min(h[neg] / (-s[neg]))) - 1e-10)
        max_alpha = max(0.0, min(max_alpha, 1.0))
    if max_alpha < 1e-14:
        return 0.0, float(np.max(raw) * dx)
    cands = np.linspace(0.0, max_alpha, 64)
    for k in range(len(raw)):
        if Q[k] > 1e-12:
            av = T[k] / (2.0 * Q[k])
            if 0.0 < av < max_alpha:
                cands = np.concatenate([cands, [av]])
    best_a = 0.0
    best_val = float(np.max(raw) * dx)
    for a in cands:
        if a < 0 or a > max_alpha:
            continue
        val = float(np.max(raw + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    if best_a > 1e-12:
        lo_r, hi_r = max(0.0, best_a - 0.05 * max_alpha), min(max_alpha, best_a + 0.05 * max_alpha)

def _lp_direction(h, top_k, radius, raw, use_norm=False):
    n = len(h)
    top_idx = np.argsort(-raw)[:top_k]
    if top_k < 2:
        return None
    A_rows = []
    for idx in top_idx:
        g = _lag_grad(h, int(idx))
        if use_norm:
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                g = g / nrm
        A_rows.append(g)
    K = len(A_rows)
    A_mat = np.array(A_rows)
    nvar = n + 1
    A_ub = np.zeros((K, nvar))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lo = -radius * h
    hi = radius * (1.0 - h)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))
    c_obj = np.zeros(nvar)
    c_obj[n] = -1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not res.success:
        return None
    s = res.x[:n]
    tau = res.x[n]
    if tau < 1e-10:
        return None
    if np.max(np.abs(s)) < 1e-12:
        return None
    return s

def _smooth_grad(h, beta, raw=None):
    n = len(h)
    dx = 2.0 / n
    if raw is None:
        raw = np.correlate(h, 1.0 - h, mode='full')
    sc = raw * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(raw)
        w[np.argmax(raw)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

def _to_bin(h, n):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _grad_dir(h, raw):
    top_idx = np.argsort(-raw)[:20]
    g = np.zeros(len(h))

    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-10:
        return g / nrm
    return None

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
