import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute gradient matrix G where G[r] = grad of corr[r] w.r.t. h."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _grad_for_lag(h, ell):
    """Compute gradient of the overlap at lag ell w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g = 1.0 - 2.0 * h
    elif ell > 0:
        m = n - ell
        g[:m] += (1.0 - h[ell:])
        g[ell:] -= h[:m]
    else:
        p = -ell
        m = n - p
        g[p:] += (1.0 - h[:m])
        g[:m] -= h[p:]
    return g

def _smooth_weights(S0, beta=20.0):
    """Compute softmax weights over correlation values."""
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    w = np.exp(beta * (S0 - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _projected_grad_step(h, beta, max_iter=200, deadline=None, momentum=0.0, seed=42):
    """
    Projected gradient method with exact line search.
    Moves all coordinates simultaneously along the negative projected gradient.
    """
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    # Momentum
    v = np.zeros(n)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        S0 = np.correlate(h, 1.0 - h, mode='full')
        w = _smooth_weights(S0, beta)
        
        # Compute gradient: G^T @ w
        G = _grad_matrix(h)
        grad = G.T @ w
        dx_val = 2.0 / n
        grad *= dx_val
        
        # Project gradient to satisfy sum constraint (make it sum to 0)
        grad -= np.mean(grad)
        
        # Momentum
        v = momentum * v - grad
        d = v  # direction
        
        # Make direction satisfy sum = 0
        d -= np.mean(d)
        
        if np.max(np.abs(d)) < 1e-14:
            break
        
        # Find max step size
        pos_mask = d > 1e-14
        neg_mask = d < -1e-14
        t_max = 1e10
        if np.any(pos_mask):
            t_max = min(t_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
        if np.any(neg_mask):
            t_max = min(t_max, np.min(h[neg_mask] / (-d[neg_mask])))
        t_max = min(t_max, 2.0)  # cap
        
        if t_max < 1e-14:
            break
        
        # Line search: minimize F(h + t*d) for t in [0, t_max]
        # Use parabolic interpolation
        t1 = 0.3 * t_max
        t2 = 0.7 * t_max
        f0 = cur_c5
        f1 = _exact_c5(np.clip(h + t1 * d, 0, 1))
        f2 = _exact_c5(np.clip(h + t2 * d, 0, 1))
        
        # Parabolic interpolation
        best_t = 0.0
        best_f = f0
        if f1 < best_f:
            best_t, best_f = t1, f1
        if f2 < best_f:
            best_t, best_f = t2, f2
        
        # Refine with golden section
        if best_f < f0 - 1e-14:
            lo = 0.0
            hi = t_max
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            # Use the already-evaluated points
            # Golden section refinement
            a = hi - gr * (hi - lo)
            b = lo + gr * (hi - lo)
            
            # We have f(0)=f0, f(t1)=f1, f(t2)=f2
            # Just do a few more evaluations
            for _ in range(10):
                if hi - lo < 1e-10 * t_max:
                    break
                fa = _exact_c5(np.clip(h + a * d, 0, 1))
                fb = _exact_c5(np.clip(h + b * d, 0, 1))
                if fa < fb:
                    hi = b
                    b = a
                    a = hi - gr * (hi - lo)
                else:
                    lo = a
                    a = b
                    b = lo + gr * (hi - lo)
            opt_t = (lo + hi) / 2.0
            opt_f = _exact_c5(np.clip(h + opt_t * d, 0, 1))
            
            if opt_f < best_f:
                best_t, best_f = opt_t, opt_f
        
        if best_f < cur_c5 - 1e-14:
            h = np.clip(h + best_t * d, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            beta *= 1.02  # slowly increase concentration
        else:
            # No improvement, try adaptive step
            momentum = max(0.0, momentum * 0.8)
            # Random perturbation to escape
            if rng.random() < 0.3:
                h = h + rng.normal(0, 0.01, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
    
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):
    """Pair-based optimization (from previous code, simplified)."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        n_sel = min(n_pairs, n // 4)
        sources = sort_idx[-n_sel:]
        sinks = sort_idx[:n_sel]
        
        one_mh = 1.0 - h
        best_c5_try = cur_c5
        best_step_h = None
        
        for ii in range(n_sel):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            
            # Line search
            ts = np.linspace(t_lo, t_hi, 30)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
            best_idx = int(np.argmin(vals))
            t_opt = float(ts[best_idx])
            
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step_h = h_try
        
        if best_step_h is not None:
            h = _project(best_step_h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1
            radius *= 0.85
            if radius < 0.01:
                radius = 0.10

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _weighted_grad(h, w):

def _smooth_weights(S0, beta=20.0):
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    w = np.exp(beta * (S0 - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _dense_grad_opt(h, deadline, beta_start=12.0, beta_end=60.0,
                    mom_start=0.0, mom_end=0.5, max_iter=400,

    best_c5 = cur_c5
    v = np.zeros(n)
    rng = np.random.default_rng(seed)

        w = _smooth_weights(S0, beta)
        grad = _weighted_grad(h, w) * dx
        grad -= np.mean(grad)

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0,
                   stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline:
        if stall > stall_limit:
            break
        w = _smooth_weights(S0, beta)
        grad = _weighted_grad(h, w) * (2.0 / n)
        sort_idx = np.argsort(grad)
        n_sel = min(n_pairs, n // 4)
        sources = sort_idx[-n_sel:]
        sinks = sort_idx[:n_sel]

        one_mh = 1.0 - h
        best_c5_try = cur_c5
        best_step_h = None

        for ii in range(n_sel):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d_vec = np.zeros(n)
            d_vec[sink] = 1.0
            d_vec[source] = -1.0
            T = np.correlate(d_vec, one_mh, mode='full') - np.correlate(h, d_vec, mode='full')
            C = np.correlate(d_vec, d_vec, mode='full')
            ts = np.linspace(t_lo, t_hi, 25)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
            best_idx = int(np.argmin(vals))
            t_opt = float(ts[best_idx])
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)

                best_c5_try = c5_try
                best_step_h = h_try

            cur_c5 = _exact_c5(h)

                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1
            radius *= 0.85
            if radius < 0.008:

            beta *= 1.05

    h3, c5_3 = _pair_optimize(h3, sub_dl, radius=0.18, n_pairs=60,
                              beta=20.0, stall_limit=100, seed=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    n_curr = len(best_h)

    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.08)
        h3b = _project(best_h.copy(), n_curr / 2.0)
        h3b, c5_3b = _pair_optimize(h3b, sub_dl2, radius=0.12, n_pairs=50,
                                    beta=30.0, stall_limit=80, seed=300)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Binary diversification (10% time)
    d4 = deadline - 5
    n_curr = len(best_h)
    if time.time() < d4:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=12000, seed=777)
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + budget_s * 0.02)
            h_bin, c5_bin = _sweep_pair_swap(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary
        if time.time() < d4:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl3, radius=0.08,
                                           n_pairs=40, beta=35.0,
                                           stall_limit=50, seed=555)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Final dense gradient polish (5% time)
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = deadline - 2
    h5, c5_5 = _dense_grad_opt(h5, sub_dl, beta_start=40, beta_end=80,
                               mom_start=0.0, mom_end=0.2, max_iter=200,
                               n_evals=3, seed=888)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final pair polish with small radius
    n_curr = len(best_h)
    h6 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = deadline - 1
    h6, c5_6 = _pair_optimize(h6, sub_dl, radius=0.05, n_pairs=30,
                              beta=50.0, stall_limit=40, seed=999)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
