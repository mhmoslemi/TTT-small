from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, radius, K=20):
    """Solve LP to find a direction that reduces the max overlap."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    
    # Select top K lags
    top_idx = np.argsort(-raw)[:K]
    
    # Build LP: minimize t
    # s.t. for each active lag r: sum_j x[j] * (1-h[j+d_r]) <= t
    #      sum x = target
    #      0 <= x <= 1
    #      h[j] - radius <= x[j] <= h[j] + radius
    
    # Variables: x[0..n-1], t
    # Total variables: n + 1
    
    # For each active lag r, the constraint is:
    # sum_j x[j] * (1 - h[j + d_r]) - t <= 0
    # where d_r = r - (n-1), and j+d_r must be in [0, n-1]
    
    # A_ub matrix: (K + n) x (n+1)
    # for lag constraints: row is [(1-h[j+d_r]) for valid j, -1]
    # for x <= 1: identity rows with 0 in t column
    # for x >= 0: handled by bounds
    
    A_ub = np.zeros((K + n, n + 1))
    b_ub = np.zeros(K + n)
    
    for idx_r, r in enumerate(top_idx):
        d = int(r) - (n - 1)
        valid_j = np.arange(max(0, -d), min(n, n - d))
        # x[j] coefficient: (1 - h[j+d]) for j in valid_j
        for j in valid_j:
            A_ub[idx_r, j] = 1.0 - h[j + d]
        A_ub[idx_r, n] = -1.0  # coefficient of t
        b_ub[idx_r] = 0.0
    
    # x <= 1 constraints
    for j in range(n):
        A_ub[K + j, j] = 1.0
        b_ub[K + j] = 1.0
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds
    bounds = [(max(0, h[j] - radius), min(1, h[j] + radius)) for j in range(n)]
    bounds.append((0, None))  # t is unbounded above
    
    # Equality constraint: sum x = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    x_lp = res.x[:n]
    t_lp = res.x[n]
    d = x_lp - h
    return d, t_lp

def _line_search(h, d, raw, dx, radius, n_top=16):
    """Exact line search in direction d from h."""
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):

    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    for aa in [(B + sd) / (2.0 * A), (B - sd) / (2.0 * A)]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    
    grid = np.linspace(0.0, a_hi, 40)
    candidates.extend(grid.tolist())
    
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    
    # Golden section refinement
    if best_alpha > 0:
        step = a_hi / 40
        lo = max(0.0, best_alpha - step)
        hi = min(a_hi, best_alpha + step)
        if hi - lo > 1e-11:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            for _ in range(50):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            opt = (lo + hi) / 2.0
            val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
            if val_opt < best_val:
                best_val = val_opt
                best_alpha = opt
    
    return best_alpha, best_val

def _lp_step(h, radius, K=20):
    """Solve LP to find direction reducing top-K lags."""
    n = len(h)
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:K]
    cur_max = float(np.max(raw))
    
    # Build LP using cvxpy
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h - radius,
        x <= h + radius,
    ]
    
    # Linearized lag constraints
    for r in top_idx:
        d = int(r) - (n - 1)
        valid_j = np.arange(max(0, -d), min(n, n - d))
        # Linearized: sum_j x[j] * (1 - h[j+d]) <= t
        coeffs = (1.0 - h[valid_j + d])
        constraints.append(cp.sum(x[valid_j] * coeffs) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.HIGHS)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return None
    
    if x.value is None:
        return None
    
    x_lp = np.clip(x.value, 0, 1)
    x_lp = _project(x_lp, target)
    d = x_lp - h
    if np.linalg.norm(d) < 1e-14:
        return None
    return d

def _gradient_step(h, beta=20.0, radius=0.25, K=20):
    """Weighted gradient step."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    w = np.exp(np.clip(beta * shifted, -80, 0))
    w = w / (float(np.sum(w)) + 1e-30)
    
    # Compute weighted gradient
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wr * (1.0 - h[gp:])
                g[gp:] -= wr * h[:m]
    
    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm < 1e-14:
        return None
    return -g / nm

def _smooth_perturb(h, amplitude, rng):

def _lp_direction(h, radius, K):
    n = len(h)
    target = n / 2.0
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:K]
    
    # Build A_ub for lag constraints
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for idx_r, r in enumerate(top_idx):
        r = int(r)
        d = r - (n - 1)
        j_start = max(0, -d)
        j_end = min(n, n - d)
        for j in range(j_start, j_end):
            A_ub[idx_r, j] = 1.0 - h[j + d]
        A_ub[idx_r, n] = -1.0  # -t
        b_ub[idx_r] = 0.0
    
    # Equality: sum x = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds
    lb = np.maximum(0.0, h - radius)
    ub = np.minimum(1.0, h + radius)
    bounds = list(zip(lb, ub)) + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    x_lp = res.x[:n]
    d = x_lp - h
    if np.linalg.norm(d) < 1e-12:
        return None

    bounds = list(zip(lb, ub))
