For the gradient of F_r(h) where r = n-1+k:

  - dF/dh[j] for j in [0, n-k-1]: -h[j+k]  (from the (1-h[i]) term where i=j)

- If k < 0, let u = -k > 0: F_r = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])

Wait, let me be more careful. np.correlate(a, v, mode='full')[n+a-1+k] = sum_i a[i+k] * v[i] for appropriate ranges.

Actually, from numpy docs:
`np.correlate(a, v)` = sum_n a[n+k] * conj(v[n])

So for `np.correlate(h, 1-h, mode='full')`:
- The output has length 2n-1
- output[n-1+k] = sum_{valid i} h[i+k] * (1-h)[i] for k >= 0
- output[n-1-k] = sum_{valid i} h[i] * (1-h)[i+k] for k >= 0

So for k > 0:
output[n-1+k] = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])

Gradient of this w.r.t. h[j]:
- If j is in the h[i+k] part: j = i+k, so i = j-k. Valid when i in [0, n-k-1], i.e., j in [k, n-1]. Contribution: (1-h[j-k])
- If j is in the (1-h[i]) part: j = i. Valid when i in [0, n-k-1], i.e., j in [0, n-k-1]. Contribution: -h[j+k]

For k < 0, let u = -k:
output[n-1-u] = sum_{i=0}^{n-u-1} h[i] * (1-h[i+u])

Gradient of this w.r.t. h[j]:
- If j is in the h[i] part: j = i. Valid when i in [0, n-u-1], i.e., j in [0, n-u-1]. Contribution: (1-h[j+u])
- If j is in the (1-h[i+u]) part: j = i+u, so i = j-u. Valid when i in [0, n-u-1], i.e., j in [u, n-1]. Contribution: -h[j-u]

For k = 0:
output[n-1] = sum_{i=0}^{n-1} h[i] * (1-h[i])
Gradient: 1 - 2*h[j] for all j

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_gradient(h, k, n):
    """Compute gradient of correlation at lag k (where output index is n-1+k)."""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # F = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
        # dF/dh[j] = (1-h[j-k]) for j in [k, n-1]
        # dF/dh[j] = -h[j+k] for j in [0, n-k-1]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        # F = sum_{i=0}^{n-u-1} h[i] * (1-h[i+u])
        # dF/dh[j] = (1-h[j+u]) for j in [0, n-u-1]
        # dF/dh[j] = -h[j-u] for j in [u, n-1]
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

def _lp_direction(h, corr, radius, n_active_max=80):
    """Find LP descent direction minimizing linearized max correlation."""
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    
    # Select active constraints (most correlated)
    sorted_idx = np.argsort(-corr)
    n_active = min(n_active_max, m)
    active_idx = sorted_idx[:n_active]
    
    # Build gradient matrix for active constraints
    G = np.zeros((n_active, n))
    F_vals = np.zeros(n_active)
    for i, r in enumerate(active_idx):
        k = int(r) - (n - 1)
        G[i] = _corr_gradient(h, k, n) * dx
        F_vals[i] = corr[r] * dx
    
    # Current max
    cur_max = float(np.max(corr * dx))
    
    # LP variables: d[0..n-1], t
    # Objective: minimize t + eps * sum(|d|)
    # Constraints: G[i] @ d <= t - F_vals[i] for each active i
    #              sum(d) = 0
    # Bounds: -min(radius, h[i]) <= d[i] <= min(radius, 1-h[i])
    #         t >= 0 (or t >= cur_max - some margin)
    
    eps = 1e-6
    c_obj = np.zeros(n + 1)
    c_obj[:n] = eps  # small penalty on |d|
    # Actually we can't penalize |d| directly in LP without extra vars
    # Let's just minimize t for now
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: G[i] @ d - t <= -F_vals[i]
    # i.e., [G[i], -1] @ [d, t] <= -F_vals[i]
    A_ineq = np.zeros((n_active, n + 1))
    A_ineq[:, :n] = G
    A_ineq[:, n] = -1.0
    b_ineq = -F_vals
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    d_lower = -np.minimum(radius, h)
    d_upper = np.minimum(radius, 1.0 - h)
    bounds = list(zip(d_lower, d_upper))
    bounds.append((cur_max - 1.0, cur_max + 5.0))  # t bounds
    
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    d = result.x[:n]
    # Check if it's a meaningful direction
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    
    return d

def _line_search(h, d, dx, radius):
    """Exact line search for h + alpha*d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible alpha range
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    # Check endpoints
    candidates = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_local_optimize(h, deadline, radius_init=0.08, max_active=80, n_active_start=40):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    n_active = n_active_start
    
    while time.time() < deadline:
        if stall > 120:
            break
        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _lp_direction(h, corr, radius, n_active_max=n_active)
        
        if d is None:
            stall += 5
            radius *= 0.7
            # Try with more active constraints
            n_active = min(n_active + 20, 160)
            continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
            n_active = max(30, n_active - 5)
        else:
            stall += 1
            radius *= 0.7
            n_active = min(n_active + 10, 160)
    
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 5.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.4, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx)

def _binary_local_search(h_bin, deadline, max_iters=100000, temp_init=0.003, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 3000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _binary_c5(cur)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9997)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            best_stall += 1
    
    return best_h, best_c5

def _targeted_binary_search(h_bin, deadline, max_iters=30000, seed=99):
    """Targeted binary search focusing on the maximum correlation lag."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find the lag with max correlation
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_lag = int(np.argmax(corr))
        k = max_lag - (n - 1)
        
        # Try moves that reduce this specific correlation
        best_delta = 0.0
        best_move = None
        n_try = min(150, len(ones) * len(zeros))
        
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_c5 = _binary_c5(cur)
            cur[oi] = 1.0
            cur[oz] = 0.0
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        u = -k
        g[:n - u] += (1.0 - h[u:])
        g[u:] -= h[:n - u]
    return g

def _lp_direction(h, corr, radius, max_active=120):
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    sorted_idx = np.argsort(-corr)
    n_act = min(max_active, m)
    active_idx = sorted_idx[:n_act]
    G = np.zeros((n_act, n))
    F_vals = np.zeros(n_act)
    for i, r in enumerate(active_idx):
        k = int(r) - (n - 1)
        G[i] = _corr_grad(h, k, n) * dx
        F_vals[i] = corr[r] * dx
    cur_max = float(np.max(corr * dx))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ineq = np.zeros((n_act, n + 1))
    A_ineq[:, :n] = G
    A_ineq[:, n] = -1.0
    b_ineq = -F_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    d_lower = -np.minimum(radius, h) + 1e-14
    d_upper = np.minimum(radius, 1.0 - h) - 1e-14
    bounds = list(zip(d_lower, d_upper))
    bounds.append((0.0, cur_max + 5.0))
    try:
        result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    if not result.success:
        return None
    d = result.x[:n]
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    candidates = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _subgrad_direction(h, corr):
    n = len(h)
    max_val = float(corr.max())
    active = np.where(corr >= max_val - 1e-10)[0]

        k = int(m_idx) - (n - 1)
        g += _corr_grad(h, k, n)
    g /= len(active)
    d = -g
    d = d - np.mean(d)
    return d

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.4, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.08, scale * 1.12)

    # Pair swap loop
    if time.time() < d4 - 15:
        n_best = len(best_h)
        h_ps = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4 - 10, time.time() + budget_s * 0.04)
        h_ps, c5_ps = _pair_swap_loop(h_ps, sub_dl, n_pairs=5000, radius=0.25, seed=666)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Final LP polish
    if time.time() < deadline - 5:
        n_fin = len(best_h)
        h_fin = _project(best_h.copy(), n_fin / 2.0)
        h_fin, c5_fin = _lp_local_optimize(h_fin, deadline - 3, radius_init=0.03, n_active_start=80)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final binary attempt
    if time.time() < deadline - 4:
        n_fin = len(best_h)
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(deadline - 2, time.time() + budget_s * 0.02)
        h_b, c5_b = _targeted_binary_search(h_b, sub_dl, max_iters=30000, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Ensure feasibility
    n_out = len(best_h)
    best_h = _project(best_h, n_out / 2.0)
    final_c5 = _exact_c5(best_h)

    # Safety: if something went wrong, return the initial
    if not np.all(np.isfinite(best_h)) or final_c5 > best_c5 + 0.01:
        best_h = _project(h_init, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
        n_out = n0

    return best_h, final_c5, n_out
