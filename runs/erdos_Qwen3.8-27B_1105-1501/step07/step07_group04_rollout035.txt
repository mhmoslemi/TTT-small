import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _init_corr(h):
    """Compute full correlation for binary h."""
    return np.correlate(h, 1.0 - h, mode='full')

def _update_corr_binary(corr, h, p, delta):
    """
    Update correlation after flipping h[p] by delta (+1 or -1).
    h is the OLD binary vector (before flip).
    corr is the OLD correlation vector.
    Returns new corr vector (modified in place or new).
    """
    n = len(h)
    center = n - 1
    new_corr = corr.copy()
    
    # For positive lag k (index center + k), k = 1..n-1:
    # delta_c = delta * [(1-h[p-k]) * 1_{k<=p} - h[p+k] * 1_{p+k<=n-1}]
    for k in range(1, n):
        dc = 0.0
        if k <= p:
            dc += (1.0 - h[p - k])
        if p + k <= n - 1:
            dc -= h[p + k]
        if dc != 0.0:
            new_corr[center + k] += delta * dc
        
        # For negative lag k (index center - k):
        # delta_c = delta * [(1-h[p+k]) * 1_{p+k<=n-1} - h[p-k] * 1_{k<=p<=n-1-k}]
        dc2 = 0.0
        if p + k <= n - 1:
            dc2 += (1.0 - h[p + k])
        if k <= p and p <= n - 1 - k:
            dc2 -= h[p - k]
        if dc2 != 0.0:
            new_corr[center - k] += delta * dc2
    
    return new_corr

    if target >= n: return np.ones(n)
