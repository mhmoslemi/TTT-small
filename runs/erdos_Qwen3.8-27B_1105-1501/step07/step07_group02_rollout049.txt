def _project(v, target):

    n = len(v)

        return np.zeros(n)

        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_of_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k + m])
            g[k:k + m] += (-h[0:m])
    else:
        p = -k
        m = n - p
        if m > 0:
            g[p:p + m] += (1.0 - h[0:m])
            g[0:m] += (-h[p:p + m])
    return g

def _weighted_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        k = r - (n - 1)
        g += w * _grad_of_lag(h, k)
    return g

def _softmax_weights(C, beta):
    Cs = C - float(np.max(C))
    e = np.exp(beta * Cs)
    s = float(np.sum(e))
    if s < 1e-300:
        return np.ones(len(C)) / len(C)
    return e / s

    v = np.zeros(n)
    stall = 0

    for it in range(max_iter):

            break
        frac = float(it) / max_iter
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.6)
        C_raw = np.correlate(h, 1.0 - h, mode='full')
        C = C_raw * dx
        weights = _softmax_weights(C, beta)
        g = _weighted_grad(h, weights) * dx
        g_feas = g - np.mean(g)
        v = momentum * v - alpha_init * g_feas
        h_new = _project(h + v, n / 2.0)
        f_new = _exact_c5(h_new)

def _epi_solve(h_init, deadline, n, maxiter=150, ftol=1e-8):
    from scipy.optimize import minimize
    h = _project(h_init.copy(), n / 2.0)
    cur_c5 = _exact_c5(h)
    t0 = cur_c5 + 0.01
    x0 = np.concatenate([h, [t0]])
    dx = 2.0 / n
    n_c = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_grad(x):
        j = np.zeros(n + 1)
        j[n] = 1.0
        return j

    def ineq(x):
        h_l = x[:n]
        h_l = np.clip(h_l, 0.0, 1.0)
        C = np.correlate(h_l, 1.0 - h_l, mode='full') * dx
        return x[n] - C

    def ineq_jac(x):
        h_l = np.clip(x[:n], 0.0, 1.0)
        J = np.zeros((n_c, n + 1))
        for r in range(n_c):
            k = r - (n - 1)
            J[r, :n] = dx * _grad_of_lag(h_l, k)
            J[r, n] = 1.0
        return J

    def eq_c(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_c, 'jac': eq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    result = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    if result.success or result.fun < cur_c5:
        h_res = _project(result.x[:n], n / 2.0)
        c5_res = _exact_c5(h_res)
        return h_res, c5_res
    return h_init, cur_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=60, beta=20.0, stall_limit=80, seed=42):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    best_c5 = cur_c5
    ones = sorted(set(np.where(cur > 0.5)[0].tolist()))
    zeros = sorted(set(np.where(cur <= 0.5)[0].tolist()))
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                return best_h, best_c5
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def _make_starts(h_parent, n, rng):
    starts = []
    starts.append(_project(h_parent.copy(), n / 2.0))
    # Random binary
    for _ in range(3):
        perm = rng.permutation(n)
        hb = np.zeros(n)
        hb[perm[:n // 2]] = 1.0
        starts.append(hb)
    # Smooth block
    for _ in range(2):
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.15, 0.35) * n
        x = np.arange(n)
        h_smooth = 1.0 / (1.0 + np.exp((np.abs(x - center) - width) / 5.0))
        starts.append(_project(h_smooth, n / 2.0))
    # Block pattern
    for _ in range(2):
        nblocks = int(rng.integers(3, 8))
        block_size = n // nblocks
        h_block = np.zeros(n)
        for b in range(nblocks):
            if rng.random() < 0.5:
                h_block[b * block_size:(b + 1) * block_size] = 1.0
        starts.append(_project(h_block, n / 2.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except:
        h0 = np.zeros(180)
        h0[:90] = 1.0

    n0 = len(h0)
    if n0 < 10:
        h0 = np.zeros(180)
        h0[:90] = 1.0
        n0 = 180

    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient descent from multiple starts at parent n (20% time)
    d1 = t0 + budget_s * 0.20
    starts = _make_starts(h_init, n0, rng)
    for si, s in enumerate(starts):
        sub_dl = min(d1, time.time() + budget_s * 0.04)
        h_s, c5_s = _smooth_descent(s, sub_dl, n0, max_iter=3000,
                                    beta_start=15.0, beta_end=300.0,
                                    alpha_init=0.04, momentum=0.85, seed=10 + si)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    # Continue from best with higher beta
    sub_dl = d1
    h_s, c5_s = _smooth_descent(best_h, sub_dl, n0, max_iter=5000,
                                beta_start=100.0, beta_end=500.0,
                                alpha_init=0.02, momentum=0.9, seed=99)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 2: SLSQP epigraph solve (20% time)
    d2 = t0 + budget_s * 0.40
    sub_dl = min(d2, time.time() + budget_s * 0.12)
    try:
        h_epi, c5_epi = _epi_solve(best_h, sub_dl, n0, maxiter=120, ftol=1e-8)
        if c5_epi < best_c5:
            best_c5 = c5_epi
            best_h = h_epi.copy()
    except Exception:
        pass
    # Try SLSQP from a different start
    if time.time() < d2:
        h_alt = np.zeros(n0)
        h_alt[:n0 // 2] = 1.0
        sub_dl2 = min(d2, time.time() + budget_s * 0.07)
        try:
            h_epi2, c5_epi2 = _epi_solve(h_alt, sub_dl2, n0, maxiter=100, ftol=1e-8)
            if c5_epi2 < best_c5:
                best_c5 = c5_epi2
                best_h = h_epi2.copy()
        except Exception:
            pass

    # Phase 3: Pair optimization (25% time)
    d3 = t0 + budget_s * 0.65
    sub_dl = min(d3, time.time() + budget_s * 0.10)
    h_p, c5_p = _pair_optimize(best_h, sub_dl, radius=0.15, n_pairs=60, beta=25.0,
                               stall_limit=100, seed=200)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()
    # Continue pair optimization
    sub_dl2 = min(d3, time.time() + budget_s * 0.08)
    h_p2, c5_p2 = _pair_optimize(best_h, sub_dl2, radius=0.10, n_pairs=50, beta=30.0,
                                 stall_limit=80, seed=300)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()

    # Phase 4: Multi-n exploration (15% time)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    for dn in [-8, 8, -16, 16, -4, 4, -24, 24, -32, 32]:
        if time.time() > d4:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl_r = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _smooth_descent(h_r, sub_dl_r, n_try, max_iter=2000,
                                    beta_start=50.0, beta_end=300.0,
                                    alpha_init=0.03, momentum=0.88, seed=400 + dn)
        sub_dl_r2 = min(d4, time.time() + budget_s * 0.010)
        h_r, c5_r = _pair_optimize(h_r, sub_dl_r2, radius=0.12, n_pairs=45, beta=25.0,
                                   stall_limit=60, seed=500 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Binary local search (10% time)
    d5 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl_b = min(d5, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl_b, max_iter=60000, seed=777)
        sub_dl_b2 = min(d5, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_sweep(h_bin, sub_dl_b2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish from binary
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl_p = min(d5, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl_p, radius=0.08, n_pairs=40,
                                           beta=30.0, stall_limit=50, seed=888)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Final intensive refinement (10% time)
    d6 = deadline - 1.0
    n_curr = len(best_h)
    h6 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl6 = min(d6, time.time() + budget_s * 0.04)
    h6, c5_6 = _pair_optimize(h6, sub_dl6, radius=0.08, n_pairs=50, beta=35.0,
                              stall_limit=60, seed=1000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # One more SLSQP attempt from best
    if time.time() < d6:
        sub_dl6b = min(d6, time.time() + budget_s * 0.03)
        try:
            h6b, c5_6b = _epi_solve(best_h, sub_dl6b, n_curr, maxiter=80, ftol=1e-9)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()
        except Exception:
            pass
    # Final smooth descent polish
    if time.time() < d6:
        n_curr = len(best_h)
        sub_dl6c = d6
        h6c, c5_6c = _smooth_descent(best_h, sub_dl6c, n_curr, max_iter=3000,
                                     beta_start=200.0, beta_end=600.0,
                                     alpha_init=0.01, momentum=0.9, seed=2000)
        if c5_6c < best_c5:
            best_c5 = c5_6c
            best_h = h6c.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
