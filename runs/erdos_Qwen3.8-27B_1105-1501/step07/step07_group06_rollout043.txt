```python
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _compute_gesture(h):
    n = len(h)
    T = np.arange(-(n - 1), n)
    gesture = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for t in range(-(n - 1), n):
            r = t + (n - 1)
            if t >= 0:
                m = n - t
                p = 0.0
                if i < m:
                    p += -h[i + t]
                if i >= t:
                    p += 1.0 - h[i - t]
            else:
                d = -t
                m = n - d
                p = 0.0
                if i < m:
                    p += 1.0 - h[i + d]
                if i >= d:
                    p += -h[i - d]
            gesture[i, r] = p
    return gesture

def _swap_delta(gesture, i, j, n):
    delta = -gesture[i] + gesture[j]
    len_d = len(delta)
    r1 = (n - 1) + (i - j)
    r2 = (n - 1) - (i - j)
    if 0 <= r1 < len_d:
        delta[r1] += 1.0
    if 0 <= r2 < len_d:
        delta[r2] += 1.0
    return delta

def _apply_swap_state(h, base_corr, i, j, n):
    h = h.copy()
    h[i] = 0.0
    h[j] = 1.0
    delta = _swap_delta(_compute_gesture(h_new_placeholder), i, j, n)
    return h

def _swap_state(h, base_corr, gesture, i, j, n):
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    delta = _swap_delta(gesture, i, j, n)
    new_corr = base_corr + delta
    new_gesture = _compute_gesture(h_new)
    return h_new, new_corr, new_gesture

def _grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    v = beta * (corr - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        g = _grad_single(h, int(np.argmax(corr)))
        return g
    w /= wsum
    g = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for r in active:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[:m] -= wr * (1.0 - h[tau:tau + m])
                g[tau:tau + m] += wr * (1.0 - h[:m])
        else:
            d = -tau
            m = n - d
            if m > 0:
                g[:m] += wr * (1.0 - h[d:d + m])
                g[d:] -= wr * h[:m]
    g *= dx
    g -= float(np.mean(g))
    return g

def _grad_single(h, r_idx):
    n = len(h)
    g = np.zeros(n)
    tau = r_idx - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[:m] -= (1.0 - h[tau:tau + m])
            g[tau:tau + m] += (1.0 - h[:m])
    else:
        d = -tau
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d + m])
            g[d:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    ng = 50
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-5:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _continuous_opt(h, deadline, beta_lo=5.0, beta_hi=50.0, radius_init=0.15, max_stall=40, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = float(total_iters) / float(total_iters + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 2.0)
        g = _grad_smooth(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            stall += 1
            continue
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            stall += 1
            continue
        d = d / dn
        alpha, val = _line_search(h, d, radius)
        if alpha != 0.0 and val < cur_c5 - 1e-14:
            h_new = _project(h + alpha * d, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
            else:
                stall += 1
                radius = max(0.03, radius * 0.8)
        else:
            # Try random directions
            improved = False
            for _ in range(5):
                dr = rng.normal(0, 1, n)
                dr -= np.mean(dr)
                drn = np.linalg.norm(dr)
                if drn < 1e-14:
                    continue
                dr = dr / drn
                alpha_r, val_r = _line_search(h, dr, radius * 0.5)
                if alpha_r != 0.0 and val_r < cur_c5 - 1e-14:
                    h_new = _project(h + alpha_r * dr, target)
                    nc = _exact_c5(h_new)
                    if nc < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = nc
                        if nc < best_c5:
                            best_c5 = nc
                            best_h = h.copy()
                        stall = 0
                        improved = True
                        break
            if not improved:
                stall += 1
                radius = max(0.03, radius * 0.85)
    return best_h, best_c5

def _binary_local_gradient(h, deadline, K=25, max_iters=100, seed=99):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    gesture = _compute_gesture(cur)
    base_corr = np.correlate(cur, 1.0 - cur, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        g = _grad_smooth(cur, 20.0)
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        g_ones = g[ones]
        top_i = ones[np.argsort(g_ones)[-K:]]
        g_zeros = g[zeros]
        top_j = zeros[np.argsort(g_zeros)[:K]]
        best_move = None
        best_val = cur_c5
        # Try gradient-guided pairs
        for i in top_i:
            for j in top_j:
                if i == j:
                    continue
                delta = _swap_delta(gesture, i, j, n)
                new_max = float(np.max(base_corr + delta)) * dx
                if new_max < best_val:
                    best_val = new_max
                    best_move = (i, j)
        # Try random pairs
        for _ in range(8):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            delta = _swap_delta(gesture, i, j, n)
            new_max = float(np.max(base_corr + delta)) * dx
            if new_max < best_val:
                best_val = new_max
                best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            cur, base_corr, gesture = _swap_state(cur, base_corr, gesture, i, j, n)
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
            stall = 0
        else:
            stall += 1
    return best, best_c5

def _binary_sa(h, deadline, T_init=0.004, max_iter=50000, seed=555):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    gesture = _compute_gesture(cur)
    base_corr = np.correlate(cur, 1.0 - cur, mode='full')
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400 or not ones or not zeros:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        delta = _swap_delta(gesture, i, j, n)
        new_c5 = float(np.max(base_corr + delta)) * (2.0 / n)
        d_c5 = new_c5 - cur_c5
        if d_c5 < 0 or (T > 1e-12 and rng.random() < np.exp(-d_c5 / T)):
            cur, base_corr, gesture = _swap_state(cur, base_corr, gesture, i, j, n)
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best, best_c5

def _pair_sweep_fast(h, deadline, seed=222):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    gesture = _compute_gesture(cur)
    base_corr = np.correlate(cur, 1.0 - cur, mode='full')
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for ii in range(len(ones)):
            if time.time() > deadline:
                break
            i = ones[ii]
            g_i = -gesture[i]
            for j in zeros:
                delta = g_i + gesture[j]
                r1 = (n - 1) + (i - j)
                r2 = (n - 1) - (i - j)
                dc = delta.copy()
                if 0 <= r1 < len(dc):
                    dc[r1] += 1.0
                if 0 <= r2 < len(dc):
                    dc[r2] += 1.0
                new_c5 = float(np.max(base_corr + dc)) * dx
                if new_c5 < cur_c5 - 1e-14:
                    cur, base_corr, gesture = _swap_state(cur, base_corr, gesture, i, j, n)
                    cur_c5 = new_c5
                    ones.remove(i)
                    zeros.remove(j)
                    zeros.append(i)
                    ones.append(j)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best = cur.copy()
                    improved = True
                    break
    return best, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 3.0
    pert += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _make_dirs_cont(h, g, beta, rng, n_max=6):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_max = int(np.argmax(corr))
    gs2 = _grad_single(h, r_max)
    gs2n = np.linalg.norm(gs2)
    if gs2n > 1e-12:
        d = gs2 / gs2n
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for blk in [4, 8, 16]:
        if n % blk != 0:
            continue
        nblk = n // blk
        gb = g.reshape(nblk, blk).mean(axis=1)
        d = np.tile(gb, (1, blk)).reshape(n).copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for freq in [1, 2, 3]:
        x = np.arange(n, dtype=float)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    return dirs[:n_max]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0

    if 'initial_h_values' in dir() or True:

        best_h = hb.copy()

    # Phase 3: Population binary SA + local search (35%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    # Create diverse binary population
    pop = [best_h.copy()]
    # Add shifted versions
    for s in [1, 2, 3, 5, 7, 13]:
        pop.append(np.roll(best_h, s).copy())
    # Add random perturbations of parent binary
    h_bin_p = _to_binary(best_h)
    for _ in range(8):
        h_p = h_bin_p.copy()
        ones = np.where(h_p > 0.5)[0]
        zeros = np.where(h_p <= 0.5)[0]
        n_flips = rng.integers(n_curr // 20, n_curr // 8)
        idx_o = rng.choice(ones, size=min(n_flips, len(ones)), replace=False)
        idx_z = rng.choice(zeros, size=min(n_flips, len(zeros)), replace=False)
        h_p[idx_o] = 0.0
        h_p[idx_z] = 1.0
        pop.append(h_p)

    # Evaluate and sort population
    pop_scores = [_exact_c5(p) for p in pop]
    order = np.argsort(pop_scores)
    pop = [pop[i] for i in order]
    pop_scores = [pop_scores[i] for i in order]
    if pop_scores[0] < best_c5:
        best_c5 = pop_scores[0]
        best_h = pop[0].copy()

    # Run SA + local on top members
    n_members = min(6, len(pop))
    for mi in range(n_members):
        sub_dl = min(d3, time.time() + budget_s * 0.04)
        if time.time() > d3:
            break
        h_sa, c5_sa = _binary_sa(pop[mi], sub_dl, T_init=0.003, max_iter=20000, seed=100 + mi)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        if time.time() < d3 - 10:
            sub_dl2 = min(d3, time.time() + budget_s * 0.02)

        if time.time() < d4 - 5:

    # Phase 5: Continuous optimization on best binary (20%)
    d5 = deadline
    h_cont = _project(best_h.copy(), len(best_h) / 2.0)

            best_c5 = c5f
            best_h = hf.copy()

    # Final: try binary polish once more
    if deadline - time.time() > 3:
        h_ff = _to_binary(best_h)
        h_ff2, c5_ff = _binary_local_gradient(h_ff, deadline, K=20, max_iters=30, seed=900)
        if c5_ff < best_c5:
            best_c5 = c5_ff
            best_h = h_ff2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_pts = len(best_h)
    return best_h, min(final_c5, best_c5 + 1e-12), n_pts
```
