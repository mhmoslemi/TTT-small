def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smoothed_grad(h, temperature):
    """Vectorized gradient of log-sum-exp smoothed max of correlate(h, 1-h, 'full')."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    S_shifted = S - np.max(S)
    exp_S = np.exp(temperature * S_shifted)
    weights = exp_S / np.sum(exp_S)
    
    grad = np.zeros(n)
    for t in range(len(S)):
        w = weights[t]
        if w < 1e-9:
            continue
        d = t - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:])
                grad[d:] += w * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
                grad[:m] += w * (-h[g:])
    return grad

               n_iter_max=50000, beta=0.85, seed=42):

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    lr = lr_init
    T = T_start
    rng = np.random.default_rng(seed)
    
    # Anneal T from T_start to T_end over the course
    total_iters = n_iter_max
    for it in range(total_iters):
        if time.time() > deadline:
            break
        # Anneal temperature
        frac = min(1.0, it / max(1, total_iters * 0.8))
        T = T_start + (T_end - T_start) * frac
        
        # Compute gradient
        grad = _smoothed_grad(h, T)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        
        # Try step
        h_trial = h - lr * momentum
        h_trial = _project(h_trial, target)
        c5_trial = _exact_c5(h_trial)
        
        if c5_trial < cur_c5 - 1e-14:
            h = h_trial
            cur_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-4:

                h = _project(h + 0.05 * rng.normal(0, 1, n), target)
                momentum = np.zeros(n)
                lr = 0.1
                cur_c5 = _exact_c5(h)
            else:
                h = h_trial  # still move, even if not improving (helps escape)
                if _exact_c5(h) < best_c5:
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                cur_c5 = _exact_c5(h)
    
    return best_h, best_c5

1. **Pairwise optimization**: For each pair (i, j), find the optimal values of h[i] and h[j] (given all others fixed). This is a 2D optimization problem that can be solved exactly.

            else:
                h_trial = _project(h - lr * momentum, target)
                if _exact_c5(h_trial) < cur_c5 + 0.001:
                    h = h_trial
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                stall += 1
    return best_h, best_c5

def _stochastic_subgrad(h, deadline, n_subgrad=50000, lr_init=0.02, seed=99):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    beta = 0.7
    lr = lr_init
    rng = np.random.default_rng(seed)
    stall = 0
    for it in range(n_subgrad):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        S = np.correlate(h, 1.0 - h, mode='full')
        S_max = np.max(S)
        # Pick top-k active lags with probability proportional to closeness to max
        top_k = 8
        idx = np.argsort(S)[-top_k:]
        t_pick = int(rng.choice(idx))
        grad = _single_lag_grad(h, t_pick)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_trial = _project(h - lr * momentum, target)
        c5_trial = _exact_c5(h_trial)
        if c5_trial < cur_c5 - 1e-15:
            h = h_trial
            cur_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
            lr = min(lr * 1.005, 0.1)
            stall = 0
        else:
            lr *= 0.7
            if lr < 1e-6:
                h = _project(h + 0.05 * rng.normal(0, 1, n), target)
                momentum[:] = 0
                lr = lr_init
                cur_c5 = _exact_c5(h)
            stall += 1
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=3000, seed=123):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            m1 = n - j
            if m1 > 0:
                g[n-1:n-1+m1] = (1.0 -h[j:])
            if j > 0:
                g[n-1-j:n-1] = -h[:j]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            for _ in range(40):
                m1v = lo_d + (hi_d - lo_d) / 3.0
                m2v = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1v * g))
                f2 = float(np.max(S0 + m2v * g))
                if f1 < f2:
                    hi_d = m2v
                else:
                    lo_d = m1v
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * (2.0 / n)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _lp_refine(h, deadline, radius_init=0.12, stall_limit=80, top_k=12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        if radius < 1e-7:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(scores)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, :m] += (1.0 - h[d:])
                    G[ai, d:] += -h[:m]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, gg:] += (1.0 - h[:m])
                    G[ai, :m] += -h[gg:]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj])
            hi_b = min(radius, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            continue
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        dx = 2.0 / n
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, fe)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.35)
        else:
            stall += 1
            radius *= 0.72
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, deadline, n_restarts=30, perturb_scale=0.15, seed=777):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    scale = perturb_scale
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 15:
            break
        x = np.arange(n)
        freqs = rng.uniform(1.0, 8.0, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        amp = scale * rng.uniform(0.5, 1.5)
        pert = np.zeros(n)
        for f, p in zip(freqs, phases):
            pert += np.sin(2.0 * np.pi * f * x / n + p)
        pert /= np.max(np.abs(pert)) + 1e-10
        pert *= amp
        pert += scale * 0.3 * rng.normal(0, 1, n)
        h_pert = _project(best_h + pert, n / 2.0)
        sub_dl1 = time.time() + remaining * 0.6
        h_t, c5_t = _smooth_gd(h_pert, sub_dl1, T_start=30, T_end=200,
                               lr_init=0.06, max_iters=5000, seed=2000 + r)
        if time.time() < deadline and c5_t < best_c5:
            sub_dl2 = min(deadline, time.time() + remaining * 0.25)
            h_t2, c5_t2 = _coord_descent(h_t, sub_dl2, max_sweeps=1000, seed=3000 + r)
            if c5_t2 < c5_t:
                c5_t = c5_t2
                h_t = h_t2
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.02, scale * 0.85)
        else:
            scale = min(0.4, scale * 1.15)
    return best_h, best_c5

def _pair_swap(h, deadline, n_pairs=5000, seed=222):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        t_hi = min(1.0 - h[i], h[j])
        if t_hi < 1e-14:
            stall += 1
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
        for _ in range(30):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
        f0 = float(np.max(S0))
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C))
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, fe)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, fe)
        if best_v < cur_c5 / dx - 1e-12:
            h = h + best_t * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient descent (20% of budget)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _smooth_gd(h_init, d1, T_start=10.0, T_end=250.0,
                          lr_init=0.1, beta=0.8, max_iters=20000, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (25%)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.15),
                              max_sweeps=3000, seed=123)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    h2b, c5_2b = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.10),
                                max_sweeps=2000, seed=456)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: LP refinement (10%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_refine(best_h, d3, radius_init=0.10, stall_limit=60, top_k=14)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Stochastic subgradient (10%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _stochastic_subgrad(best_h, d4, n_subgrad=40000, lr_init=0.015, seed=200)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-n exploration (12%)
    d5 = t0 + budget_s * 0.77
    n_curr = len(best_h)
    tried_ns = set()
    for dn in [-8, -4, 0, 4, 8, 12, -12]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _smooth_gd(h_r, sub_dl, T_start=20, T_end=200,
                               lr_init=0.08, max_iters=5000, seed=500 + dn)
        if time.time() < d5 - 1:
            sub_dl2 = min(d5, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _coord_descent(h_r, sub_dl2, max_sweeps=800, seed=600 + dn)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Basin hopping (10%)
    d6 = t0 + budget_s * 0.87
    h6, c5_6 = _basin_hopping(best_h, d6, n_restarts=25, perturb_scale=0.12, seed=777)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final intensive polish (13%)
    d7 = deadline
    sub_dl = min(d7, time.time() + budget_s * 0.03)
    h7, c5_7 = _coord_descent(best_h, sub_dl, max_sweeps=4000, seed=999)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    sub_dl = min(d7, time.time() + budget_s * 0.025)
    h7b, c5_7b = _lp_refine(best_h, sub_dl, radius_init=0.06, stall_limit=50, top_k=18)
    if c5_7b < best_c5:
        best_c5 = c5_7b
        best_h = h7b.copy()

    if time.time() < d7 - 5:
        sub_dl = min(d7, time.time() + budget_s * 0.02)
        h7c, c5_7c = _pair_swap(best_h, sub_dl, n_pairs=3000, seed=555)
        if c5_7c < best_c5:
            best_c5 = c5_7c
            best_h = h7c.copy()

    if time.time() < d7 - 3:
        sub_dl = min(d7, time.time() + budget_s * 0.015)
        h7d, c5_7d = _smooth_gd(best_h, sub_dl, T_start=50, T_end=400,
                                lr_init=0.04, beta=0.85, max_iters=10000, seed=1500)
        if c5_7d < best_c5:
            best_c5 = c5_7d
            best_h = h7d.copy()

    # Final
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
