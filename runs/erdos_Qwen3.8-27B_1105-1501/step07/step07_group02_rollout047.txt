def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    # grad shape: (L, n)

    # dcorr_r/dh[j] = (1 - h[j + lag_r]) - h[j - lag_r] (with boundary handling)
    grad = np.zeros((L, n))
    for r in range(L):
        lag = r - (n - 1)
        if lag == 0:
            # corr_r = sum h[i]*(1-h[i])
            # d/dh[j] = 1 - 2*h[j]
            grad[r, :] = 1.0 - 2.0 * h
        elif lag > 0:

            # d/dh[j] = (1-h[0]) if j+lag valid as first index contribution
            # First index: j is in [0, n-1-lag], contributes (1 - h[j+lag])
            # Second index: j-lag is in [0, n-1-lag], i.e. j in [lag, n-1], contributes -h[j-lag]
            m = n - lag
            # First part: j in [0, m-1]
            grad[r, :m] += (1.0 - h[lag:lag+m])

            grad[r, lag:] -= h[:m]
        else:

            m = n - p

    return corr, grad * dx  # multiply by dx for the actual objective gradient

                        max_iter=200, step_init=0.1, seed=42):

    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    step = step_init
    beta = beta_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        L = 2 * n - 1
        
        # Compute smoothing weights
        max_corr = np.max(corr)
        w = np.exp(beta * (corr - max_corr) / max(1e-10, max_corr))
        w /= np.sum(w)
        
        # Compute weighted gradient
        # grad_r shapes (L, n), we need w @ grad
        grad = np.zeros(n)
        for r in range(L):
            lag = r - (n - 1)
            if w[r] < 1e-10:
                continue
            if lag == 0:
                grad += w[r] * (1.0 - 2.0 * h)
            elif lag > 0:
                m = n - lag
                grad[:m] += w[r] * (1.0 - h[lag:lag+m])
                grad[lag:] -= w[r] * h[:m]
            else:
                p = -lag
                m = n - p
                grad[:m] -= w[r] * h[p:p+m]
                grad[p:] += w[r] * (1.0 - h[:m])
        grad *= 2.0 / n  # dx factor
        
        # Line search with backtracking
        h_try = h.copy()
        alpha = step
        best_alpha = None
        best_tried = cur_c5
        
        for ls in range(20):
            h_try = _project(h - alpha * grad, target)
            c_try = _exact_c5(h_try)
            if c_try < cur_c5 - 1e-12:
                best_alpha = alpha
                best_tried = c_try
                break
            alpha *= 0.5
        
        if best_alpha is not None:
            h = h_try
            cur_c5 = best_tried
            if cur_c5 < best_c5 - 1e-14:

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_smoothed(h, beta):
    """Compute smoothed gradient of max_k corr_k(h)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = 2 * n - 1
    M = np.max(corr)
    if M < 1e-15:
        return np.zeros(n)
    w = np.exp(beta * (corr - M) / M)
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros(L)
        active = np.where(corr > M - 1e-10)[0]
        w[active] = 1.0 / len(active)
        s = 1.0
    else:
        w /= s
    
    grad = np.zeros(n)
    for r in range(L):
        if w[r] < 1e-12:
            continue
        cr = w[r]
        ell = r - (n - 1)
        if ell == 0:
            grad += cr * (1.0 - 2.0 * h)
        elif ell > 0:
            m = n - ell
            grad[:m] += cr * (1.0 - h[ell:ell+m])
            grad[ell:] -= cr * h[:m]
        else:
            p = -ell
            m = n - p
            grad[:m] -= cr * h[p:p+m]
            grad[p:] += cr * (1.0 - h[:m])
    return grad * dx

def _proj_grad(h, deadline, beta_list=None, max_iter=5000, step_init=0.3, seed=42):
    """Projected gradient with annealing and restarts."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    if beta_list is None:
        beta_list = [3.0, 5.0, 8.0, 15.0, 30.0, 60.0, 100.0]
    
    step = step_init
    stall = 0
    
    for beta in beta_list:
        if time.time() > deadline:
            break
        for it in range(max_iter):
            if time.time() > deadline:
                break
            if stall > 200:
                break
            
            grad = _grad_smoothed(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            
            # Normalize step
            alpha = step / gnorm
            h_try = _project(h - alpha * grad, target)
            c_try = _exact_c5(h_try)
            
            if c_try < cur_c5 - 1e-12:
                h = h_try
                cur_c5 = c_try
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step = min(step * 1.05, 2.0)
            else:
                # Try smaller step
                alpha2 = step / gnorm
                improved = False
                for _ in range(10):
                    alpha2 *= 0.5
                    h_try = _project(h - alpha2 * grad, target)
                    c_try = _exact_c5(h_try)
                    if c_try < cur_c5 - 1e-12:
                        h = h_try
                        cur_c5 = c_try
                        if cur_c5 < best_c5 - 1e-14:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        step = max(step * 0.9, 0.01)
                        improved = True
                        break
                if not improved:
                    stall += 1
                    step *= 0.5
                    if step < 0.001:
                        # Restart with perturbation
                        pert = rng.random(n) * 0.02 - 0.01
                        h = _project(h + pert, target)
                        cur_c5 = _exact_c5(h)
                        step = step_init * 0.5
                        stall = 0
    
    return best_h, best_c5
