```python
def _binary_local_search_fast(h_bin, deadline):
    n = len(h_bin)
    ar = np.arange(n)
    h = h_bin.astype(float).copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(raw))
    
    best_h = h.copy()
    best_c5 = cur_max * (2.0 / n)
    
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h <= 0.5)[0].tolist())
    
    while time.time() < deadline:
        found = False
        # For each i in ones, try to find a j in zeros that improves
        for i in list(ones):
            if time.time() > deadline:
                break
            # Compute delta_i (effect of h[i] going 1->0)
            di = np.zeros(2 * n - 1)
            di[(i - ar) + (n - 1)] += h - 1.0   # (h[i-k]-1) terms
            di[(ar - i) + (n - 1)] += h            # h[i+k] terms
            
            after_i = raw + di
            slack = cur_max - after_i  # need delta_j < slack element-wise
            
            # For each j in zeros, check if delta_j < slack
            for j in list(zeros):
                if time.time() > deadline:
                    break
                dj = np.zeros(2 * n - 1)
                dj[(j - ar) + (n - 1)] += 1.0 - h  # (1-h[j-k]) terms
                dj[(ar - j) + (n - 1)] += -h         # -h[j+k] terms
                
                if np.all(raw + di + dj <= cur_max - 1e-10):
                    # This swap improves (or at least doesn't worsen)
                    # Actually we need strict improvement: max(raw+di+dj) < cur_max
                    new_max = float(np.max(raw + di + dj))
                    if new_max < cur_max - 1e-12:
                        h[i] = 0.0
                        h[j] = 1.0
                        raw = np.correlate(h, 1.0 - h, mode='full')
                        cur_max = float(np.max(raw))
                        ones.discard(i)
                        ones.add(j)
                        zeros.discard(j)
                        zeros.add(i)
                        if cur_max * (2.0/n) < best_c5:
                            best_c5 = cur_max * (2.0/n)
                            best_h = h.copy()
                        found = True
                        break
        if not found:
            break
    
    return best_h, best_c5
```
