def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    hx = np.clip(h, 0.0, 1.0)
    return float(np.max(np.correlate(hx, 1.0 - hx, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] -= h[:m]
    return G

def _line_search(h, d, dx, radius):

def _active_lp(h, scores, radius, n_active=12, gap=0.004):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=20.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _binary_swap_delta(counts, ones, zeros, oi, zi, n):
    """Compute new counts after swapping one at ones[oi] with zero at zeros[zi]."""
    r = int(ones[oi])  # position currently 1 -> 0
    s = int(zeros[zi])  # position currently 0 -> 1
    Q = counts.copy()
    # Step A: r goes from 1 to 0
    # All pairs where r was the "1" and s was "0": those pairs disappear
    # Pairs (r, z) for all z in zeros: count[idx(r-z)] -= 1
    for z in zeros:
        d = r - int(z)
        Q[d + n - 1] -= 1
    # Pairs (o, r) for all o in ones (o != r): those pairs now appear
    for o in ones:
        if int(o) == r:
            continue
        d = int(o) - r
        Q[d + n - 1] += 1
    
    # Step B: s goes from 0 to 1
    # Now ones = ones - {r}, zeros = zeros + {r} - {s}
    # Pairs (o, s) for all o in new ones: count[idx(o-s)] -= 1
    for o in ones:
        if int(o) == r:
            continue
        d = int(o) - s
        Q[d + n - 1] -= 1
    # Pairs (s, z) for all z in new zeros (zeros + {r} - {s}): count[idx(s-z)] += 1
    for z in zeros:
        if int(z) == s:
            continue
        d = s - int(z)
        Q[d + n - 1] += 1
    d = s - r
    Q[d + n - 1] += 1
    
    return Q

def _binary_sa(b, deadline, T_init=0.004, max_iter=100000):
    """Binary SA with exact O(n) incremental updates."""
    n = len(b)
    half = n // 2
    b = b.copy()
    ones = np.array(np.where(b > 0.5)[0], dtype=int)
    zeros = np.array(np.where(b <= 0.5)[0], dtype=int)
    
    # Initial counts
    counts = np.rint(np.correlate(b.astype(int), 1 - b.astype(int), mode='full')).astype(int)
    cur_c5 = float(np.max(counts)) * (2.0 / n)
    best_c5 = cur_c5
    best_b = b.copy()
    
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        
        oi = int(rng.integers(0, half))
        zi = int(rng.integers(0, half))
        
        r = int(ones[oi])
        s = int(zeros[zi])
        
        # Compute new counts
        Q = _binary_swap_delta(counts, ones, zeros, oi, zi, n)
        new_max = int(np.max(Q))
        delta = (new_max - int(np.max(counts))) * dx
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # Accept
            counts = Q
            ones[oi] = s
            zeros[zi] = r
            b[r] = 0
            b[s] = 1
            cur_c5 = (int(np.max(counts))) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_b = b.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
            T = max(T * 0.9999, 1e-5)
    
    return best_b, best_c5

def _random_balanced(n, rng):
    half = n // 2
    perm = rng.permutation(n)
    b = np.zeros(n)
    b[perm[:half]] = 1.0
    return b

def _low_disc_seed(n, offset, step):
    """Generate a low-discrepancy balanced binary sequence."""
    half = n // 2
    b = np.zeros(n)
    positions = np.array([(i * step + offset) % n for i in range(half)], dtype=int)
    # Handle duplicates by replacing
    seen = set()
    unique_positions = []
    for p in positions:
        if p not in seen:
            seen.add(p)
            unique_positions.append(p)
    # Fill remaining from unused
    if len(unique_positions) < half:
        unused = [i for i in range(n) if i not in seen]
        rng_local = np.random.default_rng(offset * 31 + step)
        shuffle_idx = rng_local.permutation(len(unused))
        for i in range(half - len(unique_positions)):
            unique_positions.append(unused[shuffle_idx[i]])
    b[unique_positions[:half]] = 1.0
    return b

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 10.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _average_binaries(binaries, n, k_top):
    """Average top k binary solutions, project to sum=n/2."""
    if len(binaries) == 0:
        return None
    if len(binaries) == 1:
        return binaries[0].copy()
    avg = np.zeros(n)
    for b in binaries[:k_top]:
        avg += b
    avg /= min(len(binaries), k_top)
    return _project(avg, n / 2.0)

    return float(np.max(np.correlate(hx, 1.0 - hx, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] -= h[:m]
    return G
