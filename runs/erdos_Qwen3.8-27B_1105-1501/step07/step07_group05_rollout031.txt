def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgradient(h, dx):
    """Compute the subgradient of C5(h) = max_k corr[k]*dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = corr * dx
    mx = float(np.max(sc))
    # Active set: shifts achieving (or near) the max
    active = sc >= mx - 1e-10
    # Subgradient for each active shift
    g = np.zeros(n)
    for j in range(len(corr)):
        if not active[j]:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            # corr[j] = sum_{i=lag}^{n-1} h[i]*(1-h[i-lag])
            # d/dh[i] = (1-h[i-lag]) for i in [lag, n-1]
            # d/dh[i-lag] = -h[i] for i in [lag, n-1], i.e., d/dh[j'] = -h[j'+lag] for j' in [0, n-1-lag]
            m = n - lag
            if m > 0:
                g[lag:] += (1.0 - h[:m])
                g[:m] -= h[lag:]
        else:
            s = -lag
            m = n - s
            if m > 0:
                g[s:] += (1.0 - h[:m])
                g[:m] -= h[s:s+m]
    # Average over active set
    cnt = int(np.sum(active))
    if cnt > 0:
        g /= cnt
    g *= dx
    return g, mx

    h_new = h + v
    g_new, _ = _subgradient(h_new, 2.0/len(h))
    h_candidate = h_new - alpha * g_new
    h_candidate = _project(h_candidate, target)
    v_new = beta * v + (1 - beta) * (h_candidate - h)
    return h_candidate, v_new

    v = np.zeros(n)
    best_h = h.copy()
    best_c = _exact_c5(h)
    alpha = alpha0
    for t in range(100000):

            break
        g, mx = _subgradient(h, dx)
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        h_new = h + v
        h_new = _project(h_new, target)
        g_new, _ = _subgradient(h_new, dx)
        gnn = np.linalg.norm(g_new)
        if gnn > 1e-15:
            g_new = g_new / gnn
        
        alpha_t = 2.0 / (t + 2)
        h_candidate = h_new - alpha_t * g_new
        h_candidate = _project(h_candidate, target)
        
        beta_t = (t + 1) / (t + 3)
        v = beta_t * v + (1 - beta_t) * (h_candidate - h_new)
        h = h_candidate
        
        c = _exact_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

import time
import math

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_dx(h):
    """Returns correlation * dx for all shifts"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        else:

            x = x.copy()

            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    """Gradient of log-sum-exp approximation of max correlation"""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _line_search_full(h, d, dx, radius, max_iter=60):
    """Line search along direction d, respecting [0,1] constraints"""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    # Bisect based on the upper envelope of quadratics
    # f_k(a) = S0[k] + a*T[k] - a^2*Q[k]
    # We want to minimize max_k f_k(a)
    # Use ternary search on the max of quadratics
    lo, hi = a_lo, a_hi
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(max_iter):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    best_a = (lo + hi) / 2.0
    best_val = float(np.max(S0 + best_a * T - best_a * best_a * Q)) * dx
    return best_a, best_val

            g = _smooth_grad(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)
        # Add a random direction
        d = rng.normal(0, 1, n)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
        dirs = dirs[:nd]
        
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:

            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:

def _binary_SA(hb, deadline, max_iter=10000, T_init=0.003):

    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc

    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return bh, bc

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _shift_grad(h, j):
    n = len(h)
    lag = j - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        s = -lag
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s + m]
    return g

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:

        if l >= 0:
            m = n - l
            if m > 0:

            m = n - s
            if m > 0:

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    lo, hi = a_lo, a_hi
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    best_a = (lo + hi) / 2.0
    best_val = float(np.max(S0 + best_a * T - best_a * best_a * Q)) * dx
    return best_a, best_val

def _local_opt(h, deadline, r0=0.20, stall_lim=60, nd=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(9999)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = []
        for beta in [20.0, 50.0]:
            g = _smooth_grad(h, beta)
            nrm = np.linalg.norm(g)
            if nrm > 1e-10:
                dirs.append(g / nrm)
        act = _active_dirs(h, 5)
        dirs.extend(act[:min(3, nd - len(dirs))])
        d = rng.normal(0, 1, n)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
        dirs = dirs[:nd]
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
        else:
            stall += 1
            rad *= 0.80
            if rad < 0.008:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall > stall_lim:
                    break
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=30000, Ti=0.0025):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9993
        else:
            stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _gen_pattern(n, kind, rng):
    x = np.arange(n, dtype=float) / n
    if kind == 0:
        return 0.5 + 0.45 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif kind == 1:
        return 0.5 + 0.45 * np.cos(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif kind == 2:
        return np.where(x < 0.5, 1.0, 0.0)
    elif kind == 3:
        return np.where((x < 0.25) | (0.5 < x < 0.75), 1.0, 0.0)
    elif kind == 4:
        return np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x), 0, 1)
    elif kind == 5:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        return h
    else:
        return 0.5 + 0.4 * np.sin(2 * np.pi * (1 + kind) * x)

def _anneal_local(h, deadline, T0=0.004, r0=0.15):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(5555)
    T = T0
    stall = 0
    total_steps = max(1, int((deadline - time.time()) / 0.02))
    step_n = 0
    while time.time() < deadline:
        step_n += 1
        if stall > 800:
            break
        frac = step_n / max(total_steps, 1)
        T_cur = T0 * (0.01) ** frac
        if rng.random() < 0.75:
            g = _smooth_grad(h, 30.0)
            nrm = np.linalg.norm(g)
            d = g / nrm if nrm > 1e-10 else rng.normal(0, 1, n)
        else:
            d = rng.normal(0, 1, n)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm < 1e-10:
            continue
        d = d / nrm
        a, c_try = _line_search(h, d, dx, r0)
        if a == 0.0:
            stall += 1
            continue
        h_new = _project(h + a * d, target)
        c_new = _exact_c5(h_new)
        delta = c_new - cur
        if delta < 0 or rng.random() < math.exp(-delta / max(T_cur, 1e-12)):
            h = h_new
            cur = c_new
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
                stall = 0
        else:
            stall += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Multi-resolution coarse-to-fine with diverse starts (18%)
    d1 = t0 + budget_s * 0.18
    coarse_sizes = [16, 32, 64]
    for cs in coarse_sizes:
        if time.time() > d1 - 10:
            break
        for kind in range(6):
            if time.time() > d1 - 5:
                break
            p = _gen_pattern(cs, kind, rng)
            pc = _project(p.copy(), cs / 2.0)
            sd = min(d1, time.time() + budget_s * 0.015)
            pc, cc = _local_opt(pc, sd, r0=0.30, stall_lim=30, nd=4)
            # Upsample
            if time.time() > d1 - 3:
                continue
            up = _resample(pc, n0)
            sd2 = min(d1, time.time() + budget_s * 0.01)
            up, cu = _local_opt(up, sd2, r0=0.12, stall_lim=20, nd=3)
            if cu < best_c:
                best_c = cu
                best_h = up.copy()

    # Phase 2: Intensive local opt on best (20%)
    d2 = t0 + budget_s * 0.38
    for r_init in [0.18, 0.09, 0.04]:
        if time.time() > d2 - 5:
            break
        h2 = _project(best_h.copy(), n0 / 2.0)
        sd = min(d2, time.time() + budget_s * 0.06)
        h2, c2 = _local_opt(h2, sd, r0=r_init, stall_lim=70, nd=6)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Simulated annealing in continuous space (12%)
    d3 = t0 + budget_s * 0.50
    h3 = _project(best_h.copy(), n0 / 2.0)
    sd = min(d3, time.time() + budget_s * 0.10)
    h3, c3 = _anneal_local(h3, sd, T0=0.005, r0=0.18)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()
    # Polish after anneal
    if time.time() < d3 - 3:
        h3p = _project(best_h.copy(), n0 / 2.0)
        sd2 = min(d3, time.time() + budget_s * 0.02)
        h3p, c3p = _local_opt(h3p, sd2, r0=0.06, stall_lim=30, nd=4)
        if c3p < best_c:
            best_c = c3p
            best_h = h3p.copy()

    # Phase 4: Pair swaps (12%)
    d4 = t0 + budget_s * 0.62
    h4 = _project(best_h.copy(), n0 / 2.0)
    sd = min(d4, time.time() + budget_s * 0.10)
    h4, c4 = _pair_opt(h4, sd, max_it=60000)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()

    # Phase 5: Binary exploration (12%)
    d5 = t0 + budget_s * 0.74
    hbin = _to_bin(best_h, n0)
    sd = min(d5, time.time() + budget_s * 0.08)
    hbin, cb = _bin_sa(hbin, sd, mx=25000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d5 - 2:
        hp = _project(hbin.copy(), n0 / 2.0)
        sd2 = min(d5, time.time() + budget_s * 0.025)
        hp, cp = _local_opt(hp, sd2, r0=0.10, stall_lim=30, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    # Try a random binary start
    if time.time() < d5 - 5:
        hb_rand = _gen_pattern(n0, 5, rng)
        sd3 = min(d5, time.time() + budget_s * 0.02)
        hb_rand, cb2 = _bin_sa(hb_rand, sd3, mx=8000, Ti=0.002)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb_rand.copy()

    # Phase 6: Basin hopping with diverse perturbations (14%)
    d6 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d6 and n_re < 100:
        n_re += 1
        # Structured perturbation
        x_vec = np.arange(n0, dtype=float)
        pert = np.zeros(n0)
        nf = rng.integers(2, 5)
        for _ in range(nf):
            f = rng.uniform(0.3, 20.0)
            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 1.0)
            if rng.random() < 0.5:
                pert += amp * np.sin(2 * np.pi * f * x_vec / n0 + p)
            else:
                pert += amp * np.cos(2 * np.pi * f * x_vec / n0 + p)
        pert += 0.3 * scale * rng.normal(0, 1, n0)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, n0 / 2.0)
        # Quick local opt on perturbed
        sd = min(d6, time.time() + budget_s * 0.008)
        ht, ct = _local_opt(hp, sd, r0=0.06, stall_lim=20, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.07, scale * 1.15)

    # Phase 7: Final intensive polish (12%)
    d7 = deadline
    # Anneal + local
    if deadline - time.time() > 30:
        hf = _project(best_h.copy(), n0 / 2.0)
        sd = min(deadline, time.time() + budget_s * 0.05)
        hf, cf = _anneal_local(hf, sd, T0=0.002, r0=0.10)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    # Local opt
    if deadline - time.time() > 20:
        hf = _project(best_h.copy(), n0 / 2.0)
        sd = min(deadline, time.time() + budget_s * 0.04)
        hf, cf = _local_opt(hf, sd, r0=0.04, stall_lim=50, nd=5)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    # Pair swap
    if deadline - time.time() > 10:
        hf = _project(best_h.copy(), n0 / 2.0)
        sd = min(deadline, time.time() + budget_s * 0.02)
        hf, cf = _pair_opt(hf, sd, max_it=20000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Try nearby n values
    if deadline - time.time() > 15:
        for dn in [-4, 4, -8, 8, -12, 12]:
            if deadline - time.time() < 5:
                break
            n_try = n0 + dn
            if n_try < 50:
                continue
            hr = _resample(best_h, n_try)
            sd = min(deadline, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(hr, sd, r0=0.05, stall_lim=25, nd=3)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
