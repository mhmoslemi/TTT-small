def _project(v, target):

    n = len(v)

    if target >= n:
        return np.ones(n)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute dR[shift s] / dh as a vector of length n."""
    n = len(h)

        m = n - s
        if m > 0:

            g[s:] -= h[:m]
    else:

        m = n - b
        if m > 0:
            g[b:] += (1.0 - h[:m])
        if b > 0:
            g[:m] -= h[b:b+m]

    # Include alpha=0 and alpha=alpha_max

    best_a = 0.0

        val = float(np.max(R0 + a*A + a*a*B))

    f1 = float(np.max(R0 + c1*A + c1*c1*B))

    if opt_v < best_val:
