def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _smooth_max_and_grad(h, beta):
    """
    Compute smooth max of correlate(h, 1-h, 'full') and its gradient w.r.t. h.
    Returns (smooth_max_value, gradient_vector_of_length_n).
    """
    n = len(h)
    y = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    
    # Softmax weights
    y_max = float(np.max(y))
    y_shifted = y - y_max
    exp_y = np.exp(beta * y_shifted)
    Z = float(np.sum(exp_y))
    w = exp_y / Z  # length 2n-1
    
    smooth_max = y_max + np.log(Z) / beta
    
    # Compute gradient: grad[k] = sum_m w[m] * ∂y[m]/∂h[k]
    # y[m] corresponds to shift s = m - (n-1)
    # ∂y[m]/∂h[k] = (1-h[k-s]) * [0<=k-s<n] - h[k+s] * [0<=k+s<n]
    # where s = m - (n-1)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        s = m - (n - 1)
        wm = w[m]
        # Contribution from (1-h[k-s]) term: k-s in [0, n-1] => k in [s, s+n-1]
        #   For each valid k, add wm * (1-h[k-s])
        # Contribution from -h[k+s] term: k+s in [0, n-1] => k in [-s, n-1-s]
        #   For each valid k, add wm * (-h[k+s])
        
        # First term: k from max(0, s) to min(n-1, s+n-1)
        k_lo1 = max(0, s)
        k_hi1 = min(n - 1, s + n - 1)
        if k_lo1 <= k_hi1:
            idx = np.arange(k_lo1, k_hi1 + 1)
            grad[idx] += wm * (1.0 - h[idx - s])
        
        # Second term: k from max(0, -s) to min(n-1, n-1-s)
        k_lo2 = max(0, -s)
        k_hi2 = min(n - 1, n - 1 - s)
        if k_lo2 <= k_hi2:
            idx = np.arange(k_lo2, k_hi2 + 1)
            grad[idx] -= wm * h[idx + s]
    
    return smooth_max, grad

def _smooth_max_optimize(h, deadline, target, beta_start=5.0, beta_end=50.0, 
                          lr_start=0.01, momentum=0.9, max_iter=50000):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    v = np.zeros(n)  # momentum velocity
    lr = lr_start
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Anneal beta
        progress = it / max_iter
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute smooth max and gradient
        sm_val, grad = _smooth_max_and_grad(h, beta)
        
        # Project gradient to feasible directions (zero-sum space)
        grad_mean = float(np.mean(grad))
        grad_zero = grad - grad_mean
        
        # Update with momentum
        v = momentum * v + lr * grad_zero
        h_new = h + v
        h_new = _project(h_new, target)
        
        # Evaluate
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Accept/reject

    """Compute gradient of smooth max of correlate(h, 1-h, 'full')."""

    j = np.arange(n)[None, :]  # 1×n

    term1 = M1 @ (1.0 - h)
    term2 = M1.T @ h
    grad = term1 - term2

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    perturb = np.zeros(n)
    for f, p in zip(freqs, phases):
        perturb += amplitude * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth-max gradient descent (45% of time)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _smooth_max_descend(h_init, d1, n0 / 2.0,
                                   beta_start=2.0, beta_end=150.0,
                                   lr_start=0.003, momentum=0.93,
                                   max_iter=80000, seed=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent polish (20%)
    d2 = t0 + budget_s * 0.65
    h2, c5_2 = _coord_descent_fast(best_h, d2, max_sweeps=8000, seed=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Guided pair transfer (15%)
    d3 = t0 + budget_s * 0.80
    h3, c5_3 = _guided_pair_transfer(best_h, d3, len(best_h) / 2.0,
                                     beta=8.0, max_iters=30000, seed=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-n exploration (10%)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in [-10, -8, -6, -4, -2, 2, 4, 6, 8, 10, -14, 14]:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 50 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.006)
        h_r, c5_r = _smooth_max_descend(h_r, sub_dl, n_try / 2.0,
                                        beta_start=5.0, beta_end=80.0,
                                        lr_start=0.003, momentum=0.9,
                                        max_iter=5000, seed=500 + dn)
        sub_dl2 = min(d4, time.time() + budget_s * 0.004)
        h_r, c5_r2 = _coord_descent_fast(h_r, sub_dl2, max_sweeps=1000, seed=600 + dn)
        if c5_r2 < c5_r:
            c5_r = c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final intensive polish (10%)
    d5 = deadline - 2
    n_fin = len(best_h)
    h5 = best_h.copy()
    for attempt in range(5):
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h5, c5_5 = _coord_descent_fast(h5, sub_dl, max_sweeps=3000, seed=900 + attempt * 7)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() > d5:
            break
        sub_dl2 = min(d5, time.time() + budget_s * 0.008)
        h5, c5_5b = _smooth_max_descend(h5, sub_dl2, len(h5) / 2.0,
                                         beta_start=20.0, beta_end=300.0,
                                         lr_start=0.001, momentum=0.95,
                                         max_iter=10000, seed=1100 + attempt * 3)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5.copy()
        if time.time() > d5:
            break
        h_pert = _smooth_perturb(best_h, 0.015, rng)
        sub_dl3 = min(d5, time.time() + budget_s * 0.008)
        h5, c5_5c = _coord_descent_fast(h_pert, sub_dl3, max_sweeps=1500, seed=1300 + attempt * 11)
        if c5_5c < best_c5:
            best_c5 = c5_5c
            best_h = h5.copy()

    # Final projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
