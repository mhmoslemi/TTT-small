import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
            if not np.any(mask):
                mask = (x < 1.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
            if not np.any(mask):
                mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, active_indices):
    """Build gradient rows for active correlation constraints."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    rows = []
    for r in active_indices:
        k = int(r) - center
        g = np.zeros(n)
        if k > 0:
            # corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # grad[j] = (1-h[j-k]) for j in [k, n-1]
            #         - h[j+k] for j in [0, n-1-k]
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        elif k < 0:
            u = -k
            # corr[n-1-u] = sum_{i=0}^{n-1-u} h[i]*(1-h[i+u])
            # grad[j] = (1-h[j+u]) for j in [0, n-1-u]
            #         - h[j-u] for j in [u, n-1]
            g[:n-u] += (1.0 - h[u:])
            g[u:] -= h[:n-u]
        else:
            # corr[n-1] = sum h[i]*(1-h[i])
            g[:] = 1.0 - 2.0 * h
        rows.append(g * dx)
    return np.array(rows)

def _line_search(h, d, dx, radius, n_grid=40):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    def eval_at(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([eval_at(t) for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = eval_at(opt_t)
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = eval_at(c1)
        f2 = eval_at(c2)
        for _ in range(50):
            if hi_b - lo_b < 1e-11:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = eval_at(c1)
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = eval_at(c2)
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = eval_at(opt_t)
    
    candidates = [(opt_t, opt_val), (0.0, eval_at(0.0)), (t_hi, eval_at(t_hi))]
    best_c = min(candidates, key=lambda x: x[1])
    return best_c[0], best_c[1]

def _lp_direction(h, corr_scaled, radius, n_active=50):
    """Solve LP to get descent direction considering multiple active constraints."""
    n = len(h)
    center = n - 1
    m = len(corr_scaled)
    
    # Select active constraints: top n_active by value
    sorted_idx = np.argsort(-corr_scaled)
    active = sorted_idx[:min(n_active, m)]
    
    # Also include constraints close to max
    max_val = corr_scaled[active[0]]
    threshold = max_val - max(0.005, 0.05 * max_val)
    close = np.where(corr_scaled >= threshold)[0]
    active = np.unique(np.concatenate([active, close]))
    active = active[:min(120, len(active))]
    
    # Build gradient matrix
    G = _grad_rows(h, active)
    # G shape: (len(active), n)
    vals = corr_scaled[active]
    
    # LP: min t + eps * sum(u)
    # s.t. G @ d - t <= -vals  (i.e., G@d + vals <= t)
    #      sum(d) = 0
    #      -cap_neg <= d <= cap_pos
    #      d_i - u_i <= 0, -d_i - u_i <= 0, u >= 0
    
    n_act = len(active)
    n_vars = n + 1 + n  # d, t, u
    d_idx = np.arange(n)
    t_idx = n
    u_idx = np.arange(n + 1, 2 * n + 1)
    
    # Objective: [0...0, 1, eps...eps]
    c_obj = np.zeros(n_vars)
    c_obj[t_idx] = 1.0
    c_obj[u_idx] = 1e-4
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint r: G[r] @ d - t <= -vals[r]
    # => G[r] @ d + 0*t + 0*u <= -vals[r] - ... wait
    # We want: G[r] @ d <= t - vals[r]
    # => G[r] @ d - t <= -vals[r]
    
    A_ub = np.zeros((n_act + n + n, n_vars))
    b_ub = np.zeros(n_act + n + n)
    
    for i in range(n_act):
        A_ub[i, d_idx] = G[i]
        A_ub[i, t_idx] = -1.0
        b_ub[i] = -vals[i]
    
    # d_i - u_i <= 0
    row_off = n_act
    for i in range(n):
        A_ub[row_off + i, d_idx[i]] = 1.0
        A_ub[row_off + i, u_idx[i]] = -1.0
        b_ub[row_off + i] = 0.0
    
    # -d_i - u_i <= 0
    row_off2 = n_act + n
    for i in range(n):
        A_ub[row_off2 + i, d_idx[i]] = -1.0
        A_ub[row_off2 + i, u_idx[i]] = -1.0
        b_ub[row_off2 + i] = 0.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, d_idx] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        neg_cap = min(radius, h[i]) + 1e-12
        pos_cap = min(radius, 1.0 - h[i]) + 1e-12
        bounds.append((-neg_cap, pos_cap))
    bounds.append((0.0, max_val + 1.0))  # t >= 0, t <= max_val + 1
    for i in range(n):
        bounds.append((0.0, 2.0 * radius))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[d_idx].copy()
            # Clean up
            d[np.abs(d) < 1e-10] = 0.0
            if float(np.max(np.abs(d))) < 1e-10:
                return None
            return d
        else:
            return None
    except Exception:
        return None

def _lp_local_optimize(h, deadline, radius_init=0.08, max_iter=2000, n_active=60, stall_limit=50):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break
        
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr_scaled = corr_raw * dx
        
        d = _lp_direction(h, corr_scaled, radius, n_active=n_active)
        if d is None:
            # Fallback: subgradient direction
            max_val = float(corr_raw.max())
            active = np.where(corr_raw >= max_val - 1e-10)[0]
            g = np.zeros(n)
            for m in active:
                k = int(m) - (n - 1)
                if k >= 0:
                    g[k:] += (1.0 - h[:n-k])
                    g[:n-k] -= h[k:]
                else:
                    ka = -k
                    g[:n-ka] += (1.0 - h[ka:])
                    g[ka:] -= h[:n-ka]
            g /= len(active)
            d = -g
            d = d - np.mean(d)
            d = np.clip(d, -radius, radius)
            d = d - np.mean(d)
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _to_binary(h, n):

def _binary_guided_search(h_bin, deadline, max_iters=100000, seed=42):
    """Guided binary search using gradient information."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 10000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute correlation to find active constraints
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_val = float(corr.max())
        active = np.where(corr >= max_val - 1e-10)[0]
        
        # Score each potential move by its effect on active constraints
        # For efficiency, sample a subset
        n_try = min(300, max(50, len(ones) * len(zeros) // 50))
        best_delta = 0.0
        best_move = None
        
        S0 = corr
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            
            # Compute delta efficiently
            # Changing cur[oi] from 1 to 0 and cur[oz] from 0 to 1
            # d[oi] = -1, d[oz] = +1
            # new_corr = S0 + d.T - d.S0 (for bilinear)
            # Actually: new_corr = S0 + T - C where T = corr(d, 1-cur) - corr(cur, d), C = corr(d,d)
            # For binary swap: d has -1 at oi, +1 at oz
            # T[r] = (1-cur)[r-shift]*d stuff...
            # Simpler: just compute the change
            # corr(h, 1-h) where h changes at two positions
            # Let's just do the full computation for the trial
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_corr = np.correlate(trial, 1.0 - trial, mode='full')
            new_c5 = float(np.max(new_corr)) * dx
            delta = new_c5 - cur_c5
            
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1
            # Add some randomness to escape
            if stall % 50 == 0 and stall < 5000:
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                oi = ones[i]
                oz = zeros[j]
                cur[oi] = 0.0
                cur[oz] = 1.0
                cur_c5 = _exact_c5(cur)
                ones[i] = oz
                zeros[j] = oi
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iters=100000, temp_init=0.005, seed=42):
    """Simulated annealing for binary optimization."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5 = _exact_c5(trial)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            best_stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 4
    freqs = rng.uniform(0.3, 6.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amps = rng.uniform(0.5, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _greedy_construction(n):
    h = np.zeros(n)
    ones_needed = n // 2
    for step in range(ones_needed):
        best_pos = -1
        best_c5 = float('inf')
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = 1.0
                c5 = _exact_c5(h)
                h[pos] = 0.0
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h

def _block_swap_search(h_bin, deadline, max_iters=15000, seed=77):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(6, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        block1 = cur[i:i+length].copy()
        block2 = cur[j:j+length].copy()
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i+length] = block2
        trial[j:j+length] = block1
        new_c5 = _exact_c5(trial)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1

def _grad_row(h, k):
    n = len(h)

    elif k < 0:

        g[:n-u] += (1.0 - h[u:])

    else:
