import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smooth_weights(R, beta):
    M = float(np.max(R))
    if beta * (M - M) > 700:  # always false, just for overflow safety
        return np.ones(len(R)) / len(R)
    shifted = R - M
    w = np.exp(beta * shifted)
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, int(len(R) * 0.05))
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _build_grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _line_search(R0, A, B, t_lo, t_hi):
    """Minimize max_k(R0[k] + t*A[k] - t^2*B[k]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R0))
    n_grid = 60
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        t_best = float(ts[best_idx])
        return t_best, float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))

    f_hi = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smooth_weights(R, beta):
    M = float(np.max(R))
    shifted = R - M
    w = np.exp(beta * shifted)
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, int(len(R) * 0.05))
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k

    return w

                    A_r = np.correlate(d_r, one_mh, mode='full') - np.correlate(h, d_r, mode='full')
                    B_r = np.correlate(d_r, d_r, mode='full')
                    t_opt_r, _ = _line_search(R, A_r, B_r, t_lo_r, t_hi_r)
                    h_new2 = _project(h + t_opt_r * d_r, target)
                    c5_new2 = float(np.max(_exact_corr(h_new2)) * dx)
                    if c5_new2 < cur_c5 - 1e-12:
                        h = h_new2
                        R = _exact_corr(h)
                        cur_c5 = c5_new2
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        continue
            stall += 1
            if stall % 15 == 0:
                beta = max(beta_start, beta * 0.8)
                step_cap = min(0.8, step_cap * 1.1)

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=25.0, max_stall=80, seed=42):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h.copy(), target)
    R = _exact_corr(h)
    cur_c5 = float(np.max(R) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    rad = float(radius)

    while time.time() < deadline:
        if stall > max_stall or rad < 1e-7:
            break
        w = _smooth_weights(R, beta)
        G = _build_grad_matrix(h)
        cg = G.T @ w
        sort_idx = np.argsort(cg)
        sinks = sort_idx[:min(n_pairs, n // 3)]
        sources = sort_idx[-min(n_pairs, n // 3):]
        pairs = list(zip(sinks.tolist(), sources.tolist()))
        for _ in range(min(15, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_c5_try = cur_c5
        best_h_try = None
        one_mh = 1.0 - h
        for (si, so) in pairs:
            if time.time() > deadline:
                break
            if si == so:
                continue
            d = np.zeros(n)
            d[si] = 1.0
            d[so] = -1.0
            t_lo_p = max(-h[si], h[so] - 1.0, -rad)
            t_hi_p = min(1.0 - h[si], h[so], rad)
            if t_hi_p - t_lo_p < 1e-13:
                continue
            A_p = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            B_p = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search(R, A_p, B_p, t_lo_p, t_hi_p)
            h_try = h.copy()
            h_try[si] += t_opt
            h_try[so] -= t_opt
            c5_try = float(np.max(_exact_corr(h_try)) * dx)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_h_try = h_try
        if best_h_try is not None:
            h = _project(best_h_try, target)
            R = _exact_corr(h)
            cur_c5 = float(np.max(R) * dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.3)
        else:
            stall += 1
            rad *= 0.8
            if stall % 30 == 0 and rad < 0.015:
                rad = 0.12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_pattern(n, mode, rng):
    h = np.zeros(n)
    if mode == 0:
        idx = rng.permutation(n)[:n // 2]
        h[idx] = 1.0
    elif mode == 1:
        h[:n // 2] = 1.0
    elif mode == 2:
        h[::2][:n // 2] = 1.0
    elif mode == 3:
        period = n // 3
        for i in range(n):
            if (i % (period * 2)) < period:
                h[i] = 1.0
        h = _project(h, n / 2.0)
        h = (h > 0.5).astype(float)
    else:
        h[np.arange(0, n, 2)] = 1.0
        if n % 2 == 0:
            h[-1] = 0.0
        h = _project(h, n / 2.0)
        h = (h > 0.5).astype(float)
    return h

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.002, seed=999):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = float(np.max(_exact_corr(cur)) * dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones_list = np.where(cur > 0.5)[0]
    zeros_list = np.where(cur <= 0.5)[0]

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = float(np.max(_exact_corr(h_new)) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense descent from parent (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _dense_descent(h_init, d1, beta_start=8.0, beta_max=80.0,
                              max_stall=100, step_cap=0.5, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary diversification (15% time)
    d2 = t0 + budget_s * 0.45
    if time.time() < d2:
        n_curr = len(best_h)
        best_bin_c5 = best_c5
        best_bin_h = best_h.copy()
        for mode in range(5):
            if time.time() > d2:
                break
            sub_dl = min(d2, time.time() + budget_s * 0.02)
            h_bin = _binary_pattern(n_curr, mode, rng)
            h_bin_sa, c5_sa = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.002, seed=100 + mode)
            if c5_sa < best_bin_c5:
                best_bin_c5 = c5_sa
                best_bin_h = h_bin_sa.copy()
        # LP polish on best binary
        if time.time() < d2:
            h_pol = _project(best_bin_h.copy(), n_curr / 2.0)
            sub_dl2 = min(d2, time.time() + budget_s * 0.025)
            h_pol, c5_pol = _dense_descent(h_pol, sub_dl2, beta_start=15.0,
                                           beta_max=60.0, max_stall=50, step_cap=0.3, seed=200)
            if c5_pol < best_bin_c5:
                best_bin_c5 = c5_pol
                best_bin_h = h_pol.copy()
        if best_bin_c5 < best_c5:
            best_c5 = best_bin_c5
            best_h = best_bin_h.copy()

    # Phase 3: Multi-resolution exploration (30% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -28, 28, -32, 32]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 280 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _dense_descent(h_r, sub_dl, beta_start=10.0, beta_max=50.0,
                                   max_stall=40, step_cap=0.4, seed=300 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Intensive dense + pair refinement (18% time)
    d4 = t0 + budget_s * 0.93
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4, c5_4 = _dense_descent(h4, sub_dl, beta_start=12.0, beta_max=100.0,
                              max_stall=120, step_cap=0.35, seed=500)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Pair polish
    if time.time() < d4:
        n_curr = len(best_h)
        h4b = _project(best_h.copy(), n_curr / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.05)
        h4b, c5_4b = _pair_optimize(h4b, sub_dl2, radius=0.12, n_pairs=50,
                                     beta=30.0, max_stall=60, seed=600)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Final polish
    d5 = deadline - 3
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d5, deadline)
    h5, c5_5 = _dense_descent(h5, sub_dl, beta_start=20.0, beta_max=120.0,
                              max_stall=60, step_cap=0.2, seed=700)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final projection and verification
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n_final
