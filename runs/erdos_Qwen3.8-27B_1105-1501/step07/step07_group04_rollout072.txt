import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    """Compute gradient of correlation at lag k (positive k = forward shift)."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # h[m] = h[n+k] contributes (1-h[m-k]) for m in [k, n-1]
        g[k:] += (1.0 - h[:n-k])
        # h[m] in (1-h[n]) contributes -h[m+k] for m in [0, n-1-k]
        g[:n-k] -= h[k:n]
    else:
        u = -k
        # h[m] = h[n-u] contributes (1-h[m+u]) for m in [0, n-1-u]
        g[:n-u] += (1.0 - h[u:])
        # h[m] in (1-h[n]) contributes -h[m-u] for m in [u, n-1]
        g[u:] -= h[:n-u]
    return g

def _active_indices(corr, n_top=60):
    """Get indices of top active correlation peaks."""
    m = len(corr)
    idx = np.argsort(-corr)[:min(n_top, m)]
    return sorted(idx.tolist())

def _lp_direction(h, corr, active, radius, cur_c5):
    """Find LP descent direction."""
    n = len(h)
    dx = 2.0 / n
    m_act = len(active)
    
    # Build constraint matrix
    # Variables: d[0..n-1], t
    # n_var = n + 1
    n_var = n + 1
    tau_idx = n  # index of t variable
    
    A_ub = np.zeros((m_act, n_var))
    b_ub = np.zeros(m_act)
    
    for i, r in enumerate(active):
        k = r - (n - 1)
        g = _grad_row(h, k) * dx
        A_ub[i, :n] = g
        A_ub[i, tau_idx] = -1.0
        b_ub[i] = -(corr[r] * dx)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_b = -min(radius, h[i]) + 1e-15
        hi_b = min(radius, 1.0 - h[i]) + 1e-15
        if lo_b > hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((cur_c5 - 0.5, cur_c5 + 1e-10))
    
    # Objective: minimize t + eps * sum|d_i| approximated by eps * sum d_i^2
    # Actually just minimize t with tiny tiebreak
    c_obj = np.zeros(n_var)
    c_obj[tau_idx] = 1.0
    c_obj[:n] = 1e-8  # tiny tiebreak to prefer smaller moves
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            # Zero out tiny values
            d[np.abs(d) < 1e-14] = 0.0
            if np.max(np.abs(d)) > 1e-14:
                return d
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    """Exact quadratic line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C2 = np.correlate(d, d, mode='full')
    
    # Feasible upper bound on alpha
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search then golden section
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T1 - t * t * C2)) * dx
    best_idx = int(np.argmin(grid_f))
    
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T1 - c1 * c1 * C2)) * dx
        f2 = float(np.max(S0 + c2 * T1 - c2 * c2 * C2)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T1 - c1 * c1 * C2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T1 - c2 * c2 * C2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    
    opt_val = float(np.max(S0 + opt_t * T1 - opt_t * opt_t * C2)) * dx
    # Also try 0 and t_hi
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T1 - t_hi * t_hi * C2)) * dx
    candidates = [(0.0, f0), (opt_t, opt_val), (t_hi, fhi)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_optimize(h, deadline, radius_init=0.08, max_active=60, stall_limit=50):
    """Main LP-based optimization loop."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    active_count = max_active
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = _active_indices(corr, active_count)
        
        d = _lp_direction(h, corr, active, radius, cur)
        if d is None:
            # Fallback: subgradient direction
            max_val = float(corr.max())
            act = np.where(corr >= max_val - 1e-10)[0]
            g = np.zeros(n)
            for m in act:
                k = int(m - (n - 1))
                g += _grad_row(h, k)
            g /= len(act)
            d = -g
            d -= np.mean(d)
        
        # Clip to bounds
        for i in range(n):
            if d[i] > 0:
                d[i] = min(d[i], min(radius, 1.0 - h[i]))
            elif d[i] < 0:
                d[i] = max(d[i], -min(radius, h[i]))
        d -= np.mean(d)
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0:
                active_count = min(active_count + 10, 2 * n - 1)
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_search(h_bin, deadline, max_iters=50000, seed=42):

    n = len(h_bin)
    dx = 2.0 / n

            break
        if stall > 2000:
            break
        if not ones or not zeros:
            break

        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        u = -k
        g[:n - u] += (1.0 - h[u:])
        g[u:] -= h[:n - u]
    return g

def _grad_matrix(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    mat = np.zeros((len(active_idx), n))
    for i, r in enumerate(active_idx):
        k = r - (n - 1)
        mat[i] = _grad_row(h, k) * dx
    return mat

def _lp_direction(h, corr, radius, cur_c5, n_active, gap_frac=0.1):
    n = len(h)
    dx = 2.0 / n
    m = len(corr)
    max_c = float(corr.max())
    thresh = max_c - max(1e-10, gap_frac * max_c)
    candidates = np.where(corr >= thresh)[0]
    if len(candidates) > n_active:
        order = np.argsort(-corr[candidates])[:n_active]
        candidates = candidates[order]
    active = sorted(candidates.tolist())
    m_act = len(active)
    if m_act == 0:
        return None
    grad_mat = _grad_matrix(h, active)
    n_var = n + 1
    tau_idx = n
    A_ub = np.zeros((m_act, n_var))
    A_ub[:, :n] = grad_mat
    A_ub[:, tau_idx] = -1.0
    b_ub = -(corr[active] * dx)
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i]) + 1e-14
        hi_b = min(radius, 1.0 - h[i]) + 1e-14
        if lo_b > hi_b - 1e-13:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((max_c * dx - 0.3, cur_c5 + 1e-9))
    c_obj = np.zeros(n_var)
    c_obj[tau_idx] = 1.0
    c_obj[:n] = 1e-7
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            d = res.x[:n].copy()
            d[np.abs(d) < 1e-13] = 0.0
            if float(np.max(np.abs(d))) > 1e-13:
                return d
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C2 = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_grid = 35
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T1 - t * t * C2)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T1 - c1 * c1 * C2)) * dx
        f2 = float(np.max(S0 + c2 * T1 - c2 * c2 * C2)) * dx
        for _ in range(45):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T1 - c1 * c1 * C2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T1 - c2 * c2 * C2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T1 - opt_t * opt_t * C2)) * dx
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T1 - t_hi * t_hi * C2)) * dx
    best = min([(0.0, f0), (opt_t, opt_val), (t_hi, fhi)], key=lambda x: x[1])
    return best[0], best[1]

def _lp_optimize(h, deadline, radius_init=0.08, n_active=80, gap_frac=0.1, stall_limit=35):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit or radius < 5e-6:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _lp_direction(h, corr, radius, cur, n_active, gap_frac)
        if d is None:
            max_val = float(corr.max())
            act = np.where(corr >= max_val - 1e-10)[0]
            g = np.zeros(n)
            for m_idx in act:
                k = int(m_idx - (n - 1))
                g += _grad_row(h, k)
            g /= len(act)
            d = -g - np.mean(-g)
        d = np.clip(d, -radius, radius)
        for i in range(n):
            if d[i] > 0:
                d[i] = min(d[i], 1.0 - h[i])
            elif d[i] < 0:
                d[i] = max(d[i], -h[i])
        d -= np.mean(d)
        if float(np.max(np.abs(d))) < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            cur_actual = _exact_c5(h_new)
            if cur_actual < cur - 1e-14:
                h = h_new
                cur = cur_actual
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.30)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.7
            if stall % 8 == 0:
                n_active = min(n_active + 20, 2 * n - 1)
                gap_frac = min(gap_frac * 1.3, 0.5)
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_targeted(h_bin, deadline, max_iters=60000, seed=42):

    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)

    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 1500 or not ones or not zeros:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        n_try = min(80, len(ones) * len(zeros))
        best_delta = 0.0
        best_move = None
        for _t in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5 = float(np.max(np.correlate(trial, 1.0 - trial, mode='full')) * dx)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 = float(np.max(np.correlate(cur, 1.0 - cur, mode='full')) * dx)
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.3, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += rng.normal(0, 0.4, n)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)
