```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_dx = 0.0
_g_tau = 1.0
_g_target = 0.0

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[mid + d, :m] += (1.0 - h[d:d+m])
        G[mid + d, d:d+m] += -h[:m]
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:]
    return G

def _lse_objective(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    scores = _raw_scores(x) * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    exp_u = np.exp(u)
    zsum = float(np.sum(exp_u))
    if zsum < 1e-300:
        # All underflowed, return max
        return mval
    return mval + tau * np.log(zsum)

def _lse_gradient(x):
    n = _g_n
    dx = _g_dx
    tau = _g_tau
    scores = _raw_scores(x) * dx
    mval = float(np.max(scores))
    u = (scores - mval) / tau
    exp_u = np.exp(u)
    zsum = float(np.sum(exp_u))
    if zsum < 1e-300:
        # Fallback: uniform over max scores
        mask = scores >= mval - 1e-10
        p = np.zeros(len(scores))
        p[mask] = 1.0 / np.sum(mask)
    else:
        p = exp_u / zsum
    G = _build_G(x)
    return dx * (G.T @ p)

def _eq_constraint(x):
    return _g_target - float(np.sum(x))

def _eq_jacobian(x):
    return np.ones(_g_n)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_anneal(h0, deadline, n_tau=8, tau_start=None, tau_end=None, maxiter=100):
    """Run LSE temperature annealing starting from h0."""
    global _g_n, _g_dx, _g_tau, _g_target
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    _g_n = n
    _g_dx = dx
    _g_target = target
    
    if tau_start is None:
        cur_c5 = _exact_c5(h0)
        tau_start = max(0.05, 0.5 * cur_c5)
    if tau_end is None:
        tau_end = max(1e-5, tau_start / (2 ** (n_tau - 1)))
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [
        {'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian}
    ]
    
    for i in range(n_tau):
        frac = i / max(1, n_tau - 1)
        _g_tau = tau_start * (tau_end / tau_start) ** frac

            break

                fun=_lse_objective,
                x0=h,
                jac=_lse_gradient,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-10}

                h_new = _project_sum(res.x, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                # Always move to new point for warm-starting (even if not best)
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def _line_search(h1, h2, deadline, n_grid=30):
    """Exact line search between h1 and h2."""
    target = len(h1) / 2.0
    c5_1 = _exact_c5(h1)
    t_grid = np.linspace(0.0, 1.0, n_grid)
    best_t = 0.0
    best_c5 = c5_1
    for t in t_grid:
        h_try = h1 + t * (h2 - h1)
        h_try = _project_sum(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_t = t
    # Golden section refinement
    if 0 < best_t < 1 and time.time() < deadline:
        t_idx = int(best_t * (n_grid - 1))
        lo = t_grid[max(0, t_idx - 1)]
        hi = t_grid[min(n_grid - 1, t_idx + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
```

```python
            h_new[j] -= delta
            c5_try = _exact_c5(h_new)
            if c5_try < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 15.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _greedy_balance(h, deadline, n_rounds=50):
    """Try to balance top scores by coordinate adjustments."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        scores = _raw_scores(h)
        max_s = float(np.max(scores))
        # Find lags with scores close to max
        top_lags = np.where(scores >= max_s - 0.01 * max_s)[0]
        if len(top_lags) < 2:
            break
        G = _build_G(h)
        # Greedy: for each coordinate, find direction that decreases top scores
        best_imp = 1e-14
        best_move = None
        stride = max(1, n // 40)
        for i in range(0, n, stride):
            for j in range(i + 1, n, stride):
                if 1.0 - h[i] < 1e-8 or h[j] < 1e-8:
                    continue
                # Try moving mass from j to i
                deltas = [0.1 * h[j], 0.5 * h[j], 0.9 * h[j]]
                for dl in deltas:
                    h_t = h.copy()
                    h_t[i] += dl
                    h_t[j] -= dl
                    s_t = _raw_scores(h_t)
                    new_max = float(np.max(s_t))
                    if new_max < max_s - best_imp:
                        best_imp = max_s - new_max
                        best_move = (i, j, dl)
        if best_move is not None:
            i, j, dl = best_move
            h[i] += dl
            h[j] -= dl
            h = _project_sum(h, target)
            c5_new = _exact_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            cur_c5 = c5_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 - 1)
        n0 = len(h0)
    h0 = _project_sum(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Generate diverse seeds
    seeds = [h0.copy()]
    h_rev = _project_sum(h0[::-1].copy(), n0 / 2.0)
    seeds.append(h_rev)
    h_comp = _project_sum(1.0 - h0, n0 / 2.0)
    seeds.append(h_comp)
    h_rc = _project_sum(1.0 - h0[::-1], n0 / 2.0)
    seeds.append(h_rc)
    h_avg = _project_sum(0.5 * (h0 + h0[::-1]), n0 / 2.0)
    seeds.append(h_avg)
    # Add perturbed seeds
    for _ in range(3):
        hp = _perturb(h0, rng, 0.15)
        seeds.append(hp)

    # Phase 1: LSE annealing on seeds (40% time)
    d1 = t0 + budget_s * 0.40
    time_per_seed = (d1 - t0) / len(seeds)
    for i, sh in enumerate(seeds):
        if time.time() > d1 - 2:
            break
        sd = min(d1, time.time() + time_per_seed)
        h_seg, c5_seg = _lse_anneal(sh, sd, n_tau=10, maxiter=80)
        if c5_seg < best_c5:
            best_c5 = c5_seg
            best_h = h_seg.copy()
        # Quick line search back to current best
        if c5_seg > best_c5 - 0.001 and time.time() < d1 - 1:
            h_ls, c5_ls = _line_search(best_h, h_seg, time.time() + 3)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 2: Pair polish + greedy balance (15%)
    d2 = t0 + budget_s * 0.55
    h_pp, c5_pp = _pair_polish(best_h, min(d2, time.time() + budget_s * 0.06),
                               n_iter=15000, seed=100)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()
    if time.time() < d2 - 2:
        h_gb, c5_gb = _greedy_balance(best_h, min(d2, time.time() + budget_s * 0.05),
                                      n_rounds=40)
        if c5_gb < best_c5:
            best_c5 = c5_gb
            best_h = h_gb.copy()
    # Another LSE round on polished result
    if time.time() < d2 - 1:
        h_lp = _project_sum(best_h, len(best_h) / 2.0)
        h_seg, c5_seg = _lse_anneal(h_lp, d2, n_tau=8,
                                     tau_start=0.02, tau_end=0.001, maxiter=60)
        if c5_seg < best_c5:
            best_c5 = c5_seg
            best_h = h_seg.copy()

    # Phase 3: Intensive local search (20%)
    d3 = t0 + budget_s * 0.75
    # Multiple LSE rounds with decreasing tau
    for rnd in range(4):
        if time.time() > d3 - 3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.04)
        ts = 0.015 * (0.6 ** rnd)
        te = ts * 0.1
        h_seg, c5_seg = _lse_anneal(best_h, sub_dl, n_tau=6,
                                     tau_start=ts, tau_end=te, maxiter=70)
        if c5_seg < best_c5:
            best_c5 = c5_seg
            best_h = h_seg.copy()
        # Pair polish after each round
        if time.time() < d3 - 2:
            h_pp, c5_pp = _pair_polish(best_h, min(d3, time.time() + budget_s * 0.015),
                                       n_iter=4000, seed=300 + rnd)
            if c5_pp < best_c5:
                best_c5 = c5_pp
                best_h = h_pp.copy()

    # Phase 4: Binary + basin hopping (15%)
    d4 = t0 + budget_s * 0.90
    if time.time() < d4 - 2:
        h_bin = _to_binary(best_h)
        h_bp, c5_bp = _pair_polish(h_bin, min(d4, time.time() + budget_s * 0.04),
                                   n_iter=8000, seed=555)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
        # LSE polish on binary result
        if time.time() < d4 - 1:
            h_seg, c5_seg = _lse_anneal(_project_sum(best_h, len(best_h) / 2.0),
                                        min(d4, time.time() + budget_s * 0.02),
                                        n_tau=5, tau_start=0.01, tau_end=0.002, maxiter=40)
            if c5_seg < best_c5:
                best_c5 = c5_seg
                best_h = h_seg.copy()
    # Basin hopping
    if time.time() < d4 - 2:
        scale = 0.05
        for _ in range(30):
            if time.time() > d4 - 1:
                break
            h_pert = _perturb(best_h, rng, scale)
            sub_dl = min(d4, time.time() + budget_s * 0.005)
            h_seg, c5_seg = _lse_anneal(h_pert, sub_dl, n_tau=4,
                                         tau_start=0.01, tau_end=0.002, maxiter=30)
            if c5_seg < best_c5:
                best_c5 = c5_seg
                best_h = h_seg.copy()
                scale = max(0.01, scale * 0.8)
            else:
                scale = min(0.15, scale * 1.1)

    # Phase 5: Multi-n exploration (5%)
    d5 = t0 + budget_s * 0.95
    if time.time() < d5 - 2:
        n_best = len(best_h)
        for dn in [-4, 4, -8, 8, -12, 12]:
            if time.time() > d5 - 1:
                break
            n_try = n_best + dn
            if n_try < 80 or n_try > 260 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(d5, time.time() + budget_s * 0.01)
            h_seg, c5_seg = _lse_anneal(h_r, sub_dl, n_tau=5, maxiter=40)
            if time.time() < d5 - 1:
                h_pp, c5_pp = _pair_polish(h_seg, min(d5, time.time() + 2),
                                           n_iter=2000, seed=n_try)
                if c5_pp < c5_seg:
                    h_seg, c5_seg = h_pp, c5_pp
            if c5_seg < best_c5:
                best_c5 = c5_seg
                best_h = h_seg.copy()

    # Phase 6: Final polish (5%)
    if time.time() < deadline - 1:
        h_pp, c5_pp = _pair_polish(best_h, deadline, n_iter=20000, seed=999)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp.copy()
    if time.time() < deadline - 1:
        h_seg, c5_seg = _lse_anneal(_project_sum(best_h, len(best_h) / 2.0),
                                     deadline, n_tau=4,
                                     tau_start=0.005, tau_end=0.0005, maxiter=30)
        if c5_seg < best_c5:
            best_c5 = c5_seg
            best_h = h_seg.copy()

    # Final projection and evaluation
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
