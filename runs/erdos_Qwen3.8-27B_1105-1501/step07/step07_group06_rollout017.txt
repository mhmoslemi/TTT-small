def _grad_full(h, beta):

    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')

    z = np.clip(z, -50.0, 0.0)

        w = np.zeros(len(raw))
        w[idx] = 1.0

        raw = np.correlate(h, 1.0 - h, mode='full')
        r_max = int(np.argmax(raw))
        g1 = _grad_single(h, r_max)
        g1n = np.linalg.norm(g1)
        d1 = g1 / g1n if g1n > 1e-15 else d_s
        # Random pair direction
        pi = int(rng.integers(0, n))
        pj = int(rng.integers(0, n))
        if pi == pj:
            pj = (pj + 1) % n
        dp = np.zeros(n)
        dp[pi] = 1.0
        dp[pj] = -1.0
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for dd in [d_s, d1, dp]:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, dd, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, dd
        if best_d is not None and abs(best_alpha) > 1e-15:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_search(h_bin, deadline, max_iter=80000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600 or not ones or not zeros:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize_n(h_start, deadline):
    """Full pipeline for a single n value."""
    n = len(h_start)
    if n % 2 != 0:
        return h_start, _exact_c5(h_start)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    d_lbfgs = min(deadline, time.time() + (deadline - time.time()) * 0.40)
    m = n // 2
    x0 = _full_to_anti(h_start)
    for beta in [12.0, 25.0, 50.0, 80.0]:
        if time.time() > d_lbfgs - 5:
            break
        x_cur, h_ant = _lbfgs_anti(x0, beta, 150, d_lbfgs)
        c5_ant = _exact_c5(h_ant)
        if c5_ant < best_c5:
            best_c5 = c5_ant
            best_h = h_ant.copy()
        x0 = _full_to_anti(best_h)
    d_full = min(deadline, time.time() + (deadline - time.time()) * 0.30)
    h_f, c5_f = _full_descend(best_h, d_full, beta=35.0, radius=0.12, max_stall=40)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    d_bin = min(deadline, time.time() + (deadline - time.time()) * 0.30)
    h_b = _to_binary(best_h)
    h_b, c5_b = _binary_search(h_b, d_bin, max_iter=50000, T_init=0.003)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 + 1)
        n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Phase 1: Primary optimization at parent n
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _optimize_n(h_init, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: Multi-n exploration
    d2 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    n_list = []
    for dn in range(-20, 21, 4):
        nt = n_curr + dn
        if 60 <= nt <= 260 and nt % 2 == 0 and nt != n_curr:
            n_list.append(nt)
    # Also try some shorter values
    for nt in [80, 100, 120, 140]:
        if nt != n_curr and nt not in n_list:
            n_list.append(nt)
    n_list = sorted(set(n_list))
    sub_time = max(10.0, (d2 - time.time()) / max(len(n_list), 1))
    for nt in n_list:
        if time.time() > d2 - 5:
            break
        h_r = _resample(best_h, nt)
        sub_dl = min(d2, time.time() + sub_time)
        h_r, c5_r = _optimize_n(h_r, sub_dl)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Phase 3: Intensive polish at best n
    d3 = deadline
    n_best = len(best_h)
    # L-BFGS-B with very high beta
    if time.time() < d3 - 20:
        m = n_best // 2
        x0 = _full_to_anti(best_h)
        for beta in [80.0, 120.0, 200.0]:
            if time.time() > d3 - 15:
                break
            x_cur, h_ant = _lbfgs_anti(x0, beta, 100, d3)
            c5_ant = _exact_c5(h_ant)
            if c5_ant < best_c5:
                best_c5 = c5_ant
                best_h = h_ant.copy()
            x0 = _full_to_anti(best_h)
    if time.time() < d3 - 10:
        h_f, c5_f = _full_descend(best_h, d3 - 5, beta=60.0, radius=0.08, max_stall=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # Final binary refinement
    if time.time() < d3 - 5:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_search(h_b, d3 - 2, max_iter=40000, T_init=0.002)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
