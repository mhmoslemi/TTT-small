def _lp_direction(h, raw, n_active=30, radius=0.3):
    """Solve LP to find a direction that reduces the max overlap.
    Returns direction d (zero-sum) or None if LP fails."""
    n = len(h)
    target = n / 2.0
    
    # Select active lags
    top_k = np.argsort(-raw)[:n_active]
    
    # Compute gradients for active lags
    u = 1.0 - h
    grads = np.zeros((n_active, n))
    vals = np.zeros(n_active)
    for idx, k in enumerate(top_k):
        d = k - (n - 1)
        g = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += u[:m]
                g[:m] -= h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += u[gp:]
                g[gp:] -= h[:m]
        grads[idx] = g
        vals[idx] = raw[k]
    
    # LP: minimize t
    # subject to: grads[i] @ x + (vals[i] - grads[i] @ h) <= t for each i
    #             sum(x) = target
    #             max(0, h[i]-radius) <= x[i] <= min(1, h[i]+radius)
    
    # Variables: x[0..n-1], t (index n)
    # Objective: minimize t → c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: grads[i] @ x - t <= vals[i] - grads[i] @ h
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for i in range(n_active):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = vals[i] - float(np.dot(grads[i], h))
    
    # Equality: sum(x) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    lb = np.maximum(0.0, h - radius)
    ub = np.minimum(1.0, h + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        x_lp = res.x[:n]
        d_dir = x_lp - h
        # Ensure zero sum (should be by constraint, but fix numerical issues)
        d_dir -= np.mean(d_dir)
        if np.linalg.norm(d_dir) < 1e-10:
            return None
        return d_dir
    return None

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_quad(h, d, raw, dx, a_max, n_top=20):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    a_hi = a_max
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-14)
    best_val = float(np.max(raw)) * dx
    best_alpha = 0.0
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)

                    candidates.append(aa)

        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx

            best_alpha = alpha

def _gradient_direction(h, raw, beta=20.0):
    n = len(h)
    u = 1.0 - h
    raw_shifted = raw - float(np.max(raw))
    vp = np.exp(np.clip(beta * raw_shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * u[:m]
                g[:m] -= w[r] * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w[r] * u[gp:]
                g[gp:] -= w[r] * h[:m]
    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return None
    return -g / nm

        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search_quad(h, d, raw, dx, radius, n_top=8)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _kkt_check(h, n_active=50):
    """Check how close to KKT conditions we are.
    Returns the spread (max - min) of active lag values."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(raw)[-n_active:]
    return float(top[-1] - top[0])

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: LP-based optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _lp_optimize(h_init, d1, beta=20.0, radius_init=0.25, stall_limit=80, n_active=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary refinement + polish (15%)
    d2 = t0 + budget_s * 0.60
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003)
    sub_dl = min(d2, time.time() + budget_s * 0.03)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_p = _lp_optimize(h_pol, sub_dl, beta=15.0, radius_init=0.15, stall_limit=50, n_active=25)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()
    
    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in range(-12, 13, 2):
        if time.time() > d3 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, beta=18.0, radius_init=0.20, stall_limit=40, n_active=25)
        if time.time() < d3 - 3 and c5_r < best_c5 + 0.003:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.002 and time.time() < d3 - 2:
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_r, c5_r = _lp_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, beta=14.0, radius_init=0.10, stall_limit=30, n_active=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 4: Intensive final optimization (15%)
    d4 = deadline
    n_curr = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, beta=30.0, radius_init=0.12, stall_limit=120, n_active=40)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_swap_cont(h4, sub_dl, n_pairs=6000, radius=0.20, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=25000, T_init=0.0015)
        sub_dl2 = min(d4, time.time() + budget_s * 0.015)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Basin hopping with LP
    scale = 0.025
    n_restarts = 0
    while time.time() < deadline - 5 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(deadline - 3, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, beta=25.0, radius_init=0.10, stall_limit=30, n_active=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)
    
    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline, beta=40.0, radius_init=0.05, stall_limit=60, n_active=50)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=2000, radius=0.10, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
