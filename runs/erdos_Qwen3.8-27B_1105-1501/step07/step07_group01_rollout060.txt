import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj(h, T):
    Y = _get_corr(h)
    Y_max = float(np.max(Y))
    e = np.exp((Y - Y_max) / T)
    return Y_max + T * float(np.log(np.sum(e)))

def _smooth_grad(h, T):
    n = len(h)
    Y = _get_corr(h)
    Y_max = float(np.max(Y))
    e = np.exp((Y - Y_max) / T)

    grad = np.zeros(n)

        if s == 0:
            grad += wp * (1.0 - 2.0 * h)
        else:

            grad[:n - s] -= wp * h[s:]
        if s > 0:
            wn = w[n - 1 - s]
            grad[:n - s] += wn * (1.0 - h[s:])
            grad[s:] -= wn * h[:n - s]
    return grad

    n = len(h)
    target = n / 2.0

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)

        if time.time() > deadline:
            break
        # Temperature for this epoch
        T = T_start * (T_end / T_start) ** (epoch / max(n_epochs - 1, 1))
        
        for step in range(steps_per_epoch):
            if time.time() > deadline:
                break
            g = _smooth_grad(h, T)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-12:

            f_cur = _smooth_obj(h, T)

                alpha *= 0.5
            if not found:

        # Evaluate actual objective
        cur_c5 = _exact_c5(h)

def _project(v, target):

        return np.zeros(n)

        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj(h, T):
    Y = _get_corr(h)
    Y_max = float(np.max(Y))
    e = np.exp((Y - Y_max) / T)
    return Y_max + T * float(np.log(np.sum(e)))

def _smooth_grad(h, T):
    n = len(h)
    Y = _get_corr(h)
    Y_max = float(np.max(Y))
    e = np.exp((Y - Y_max) / T)
    w = e / float(np.sum(e))
    grad = np.zeros(n)
    for s in range(n):
        wp = w[n - 1 + s]
        if s == 0:
            grad += wp * (1.0 - 2.0 * h)
        else:
            grad[s:] += wp * (1.0 - h[:n - s])
            grad[:n - s] -= wp * h[s:]
        if s > 0:
            wn = w[n - 1 - s]
            grad[:n - s] += wn * (1.0 - h[s:])
            grad[s:] -= wn * h[:n - s]
    return grad

        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            if old <= 1e-14 or old >= 1.0 - 1e-14:
                continue
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d, hi_d = -old, 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_d - gr * (hi_d - lo_d)
            c2 = lo_d + gr * (hi_d - lo_d)
            f1 = float(np.max(S0 + c1 * g))
            f2 = float(np.max(S0 + c2 * g))
            for _ in range(40):
                if hi_d - lo_d < 1e-13:
                    break
                if f1 < f2:
                    hi_d, c2, f2 = c2, c1, f1
                    c1 = hi_d - gr * (hi_d - lo_d)
                    f1 = float(np.max(S0 + c1 * g))
                else:
                    lo_d, c1, f1 = c1, c2, f2
                    c2 = lo_d + gr * (hi_d - lo_d)
                    f2 = float(np.max(S0 + c2 * g))
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            new_c5 = float(np.max(S0 + opt_d * g)) * dx
            if new_c5 < cur_c5 - 1e-13:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                cur_c5 = new_c5
                improved = True
        if improved:
            h = _project(h, target)
            S0 = _correlate_full(h)
            cur_c5_check = float(np.max(S0)) * dx
            if cur_c5_check < best_c5:
                best_c5 = cur_c5_check
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    h = rng.uniform(0, 1, n)
    return _project(h, n / 2.0)

def _init_binary_random(n, seed=0):
    rng = np.random.default_rng(seed)
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _init_structured(n):
    h = np.zeros(n)
    block = max(2, n // 10)
    for j in range(0, n - block, block * 2):
        h[j:j + block] = 1.0
    return _project(h, n / 2.0)

def _interleave_optimize(h, deadline, n_smooth=30, n_pairs=50, seed=100):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(20):
        if time.time() > deadline:
            break
        # Smooth gradient step
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h, c5s = _smooth_gcd(h, sub_dl, T_start=5.0, T_end=0.5, n_temps=n_smooth, steps_per_T=8, seed=seed)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = h.copy()
        if time.time() > deadline:
            break
        # Pair sweep
        sub_dl2 = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h, c5p = _pair_sweep(h, sub_dl2, max_sweeps=n_pairs, quiet_frac=0.8)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = h.copy()
        if time.time() > deadline:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Multi-scale warm start (10% time)
    d1 = t0 + budget_s * 0.10
    n_coarse = 60
    h_coarse = _resample(h_init, n_coarse)
    h_coarse, c5_coarse = _smooth_gcd(h_coarse, min(d1, time.time() + budget_s * 0.04),

    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
    # Also try structured init at coarse scale
    h_struct = _init_structured(n_coarse)
    h_struct, c5_st = _smooth_gcd(h_struct, min(d1, time.time() + budget_s * 0.02),
                                   T_start=8.0, T_end=1.0, n_temps=15, steps_per_T=8, seed=seed + 2)
    h_st_up = _resample(h_struct, n0)
    if c5_st * (n_coarse / float(n0)) < best_c5 * 0.99:
        h_st_up, c5_st2 = _smooth_gcd(h_st_up, min(d1, time.time() + budget_s * 0.02),
                                       T_start=3.0, T_end=0.5, n_temps=10, steps_per_T=6, seed=seed + 3)
        if c5_st2 < best_c5:
            best_c5 = c5_st2
            best_h = h_st_up.copy()

        best_h = h2.copy()
