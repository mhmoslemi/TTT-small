def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        wt = w[r]
        if wt < 1e-14:

        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wt * u[:m]
                g[:m] -= wt * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wt * u[gp:]
                g[gp:] -= wt * h[:m]
    return g

def _fw_direction(h, raw, n_top_lags=3):
    """Frank-Wolfe style: find binary vertex minimizing linearized top lags."""
    n = len(h)
    top_lags = np.argsort(-raw)[:n_top_lags]

            radius *= 0.75
            beta_cur = max(beta_cur * 0.90, 3.0)
            if stall % 40 == 0 and radius < 0.06:
                radius = 0.25
            if stall % 20 == 0:
                focus_top = max(1, focus_top - 1)
    
    return best_h, best_c5

def _binary_local_search(h_bin, deadline, max_iter=80000):
    """Aggressive binary pair-swap local search."""
    n = len(h_bin)
    cur = h_bin.copy()
    raw = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(raw)) * (2.0 / n)
    best_h = cur.copy()
    best_c5 = cur_c5
    dx = 2.0 / n
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(777)
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        # Try a random pair swap
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        
        # Compute change in raw exactly
        # Flipping h[oi] from 1 to 0 and h[oz] from 0 to 1
        # Change in raw[r] for each lag r
        # raw[r] = sum h[i]*(1-h[i+d])
        # This is complex, so just compute directly
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        raw_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_c5 = float(np.max(raw_new)) * dx
        
        if new_c5 < cur_c5 - 1e-15:
            cur = h_new
            raw = raw_new
            cur_c5 = new_c5
            ones.remove(oi)
            zeros.remove(oz)
            ones.append(oz)
            zeros.append(oi)
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
            # Occasionally try a different strategy: swap two ones or two zeros
            # (not useful for sum constraint, skip)
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, T_init=0.005, cooling=0.9995):
    """Simulated annealing on binary space."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(424242)
    T = T_init
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 800:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * dx
        delta = new_c5 - cur_c5
        if delta < -1e-15 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur = h_new
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
            T *= cooling
            if delta < -1e-15:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _periodic_template(n, period):
    """Create a periodic 0/1 template with given period."""
    h = np.zeros(n)
    half_p = period // 2
    for i in range(n):
        if (i % period) < half_p:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    
    # Get initial
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Quick gradient from parent (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _gradient_optimize(h_init, d1, beta=20.0, radius=0.3, stall_limit=60, focus_top=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary SA + local search (12%)
    d2 = t0 + budget_s * 0.20
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, T_init=0.008, cooling=0.9996)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_local_search(h_bin, sub_dl, max_iter=100000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    # Continuous polish of binary
    if time.time() < d2 - 3:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_pol, c5_p = _gradient_optimize(h_pol, sub_dl, beta=30.0, radius=0.15, stall_limit=50, focus_top=1)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()
    
    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.40
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    # Try various n values
    offset_list = list(range(-20, 21, 2))
    # Sort by proximity to current
    offset_list.sort(key=lambda x: abs(x))
    for dn in offset_list:
        if time.time() > d3 - 10:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _gradient_optimize(h_r, sub_dl, beta=18.0, radius=0.25, stall_limit=40, focus_top=1)
        # Binary refinement
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.004)
            h_b, c5_b2 = _binary_local_search(h_b, sub_dl2, max_iter=30000)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 4: Intensive gradient optimization (20%)
    d4 = t0 + budget_s * 0.60
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    h4, c5_4 = _gradient_optimize(h4, d4, beta=35.0, radius=0.20, stall_limit=200, focus_top=1)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Periodic template exploration (10%)
    d5 = t0 + budget_s * 0.70
    periods = [3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 15, 16, 18, 20, 24, 28, 30, 36, 40]
    for p in periods:
        if time.time() > d5 - 5:
            break
        h_p = _periodic_template(n_curr, p)
        # Perturb the periodic template
        h_p = _smooth_perturb(h_p, 0.3, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.003)
        h_p, c5_p = _gradient_optimize(h_p, sub_dl, beta=15.0, radius=0.3, stall_limit=25, focus_top=1)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    
    # Phase 6: Basin hopping (10%)
    d6 = t0 + budget_s * 0.80
    scale = 0.04
    n_restarts = 0
    while time.time() < d6 - 5 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 - 3, time.time() + budget_s * 0.004)
        h_t, c5_t = _gradient_optimize(h_pert, sub_dl, beta=25.0, radius=0.15, stall_limit=35, focus_top=2)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.10)
    
    # Phase 7: Final intensive polish (12%)
    d7 = deadline
    h7 = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d7, time.time() + budget_s * 0.05)
    h7, c5_7 = _gradient_optimize(h7, sub_dl, beta=50.0, radius=0.10, stall_limit=150, focus_top=1)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Binary final push
    if time.time() < d7 - 5:
        h_b = _to_binary(best_h)
        sub_dl = min(d7, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iter=150000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Very final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _gradient_optimize(h6, deadline, beta=80.0, radius=0.05, stall_limit=100, focus_top=1)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
