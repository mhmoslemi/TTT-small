```python
import numpy as np
from scipy.optimize import minimize
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lse_value(h, tau):
    s = _scores(h)
    m = np.max(s)
    u = (s - m) / tau
    return m + tau * np.log(np.sum(np.exp(u)))

def _lse_gradient(h, tau):
    n = len(h)
    s = _scores(h)
    m = np.max(s)
    u = (s - m) / tau
    exp_u = np.exp(u)
    p = exp_u / np.sum(exp_u)
    
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        w = p[r]
        if w < 1e-15:
            continue
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                grad[d:] += w * (1.0 - h[:m2])
                grad[:m2] += w * (-h[d:])
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                grad[:m2] += w * (1.0 - h[g:])
                grad[g:] += w * (-h[:m2])
    return grad

def _project_sum(v, target):
    # Project v onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = _scores(h)
    return float(np.max(corr) * dx)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _lse_val(x, tau):
    s = _scores(x)
    m = float(np.max(s))
    u = (s - m) / tau
    return m + tau * float(np.log(np.sum(np.exp(u))))

def _lse_grad(x, tau):
    n = len(x)
    s = _scores(x)
    m = float(np.max(s))
    u = (s - m) / tau
    eu = np.exp(u)
    p = eu / float(np.sum(eu))
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w = p[n-1+d]
        if w < 1e-14:
            continue
        if d >= 0:
            m2 = n - d
            grad[d:] += w * (1.0 - x[:m2])
            grad[:m2] += w * (-x[d:])
        else:
            g = -d
            m2 = n - g
            grad[:m2] += w * (1.0 - x[g:])
            grad[g:] += w * (-x[:m2])
    return grad

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = (x < 1.0) if res > 0 else (x > 0.0)
            cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _sum_eq(x, target):
    return float(np.sum(x)) - target

def _sum_eq_jac(x, target):
    return np.ones(len(x))

def _lse_slsqp(h, tau_seq, deadline, max_iter=300):
    n = len(h)
    target = n / 2.0
    h = _project_sum(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (target,)}]
    
    for tau in tau_seq:
        if time.time() > deadline:
            break
        res = minimize(_lse_val, h, args=(tau,), jac=_lse_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.fun < _lse_val(h, tau) + 1e-10:
            h = _project_sum(res.x, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        # If SLSQP didn't help, try again from best
        if time.time() > deadline:
            break
        h2 = _project_sum(res.x, target)
        if _exact_c5(h2) < best_c5:
            best_c5 = _exact_c5(h2)
            best_h = h2.copy()
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=100000, T_init=0.005, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 1000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T = max(T * 0.9999, 1e-6)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=50):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _grad_guided_pairs(h, deadline, n_iter=5000, seed=99):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    tau = 2.0
    stall = 0
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        # Get gradient
        grad = _lse_grad(cur, tau)
        # Find best pair: increase where grad is most negative, decrease where most positive
        # But we need to respect bounds
        candidates_up = np.where((grad < -1e-6) & (cur < 0.99))[0]
        candidates_down = np.where((grad > 1e-6) & (cur > 0.01))[0]
        if len(candidates_up) == 0 or len(candidates_down) == 0:
            stall += 1
            tau *= 0.9
            continue
        # Use random selection from top candidates
        k_up = min(5, len(candidates_up))
        k_down = min(5, len(candidates_down))
        top_up = candidates_up[np.argsort(grad[candidates_up])[:k_up]]
        top_down = candidates_down[np.argsort(-grad[candidates_down])[:k_down]]
        i = top_up[int(rng.integers(0, k_up))]
        j = top_down[int(rng.integers(0, k_down))]
        # Try the move
        cur[i] = min(1.0, cur[i] + 0.5)
        cur[j] = max(0.0, cur[j] - 0.5)
        cur = _project_sum(cur, target)
        new_c5 = _exact_c5(cur)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
            stall = 0
        else:
            # Revert - need to recompute from best
            cur = best_h.copy()
            cur_c5 = best_c5
            stall += 1
            tau = max(0.5, tau * 0.95)
    
    return best_h, best_c5
```
