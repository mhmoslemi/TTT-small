def _exact_c5(h):

def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < (1.0 - 1e-10)
        else:

        cnt = int(np.sum(mask))

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    """Compute gradient rows for given lags (signed offsets from 0)."""
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:

def _lp_direction(h, scores, radius, n_active=20, gap=0.003):
    """Find direction using active-set LP."""
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:

    if len(active) > n_active:

        active = active[order][-n_active:]

    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))

    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):

        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))

                      bounds=bounds, method='highs', options={'presolve': True})

            d = res.x[:n].copy()
            d -= np.mean(d)  # ensure zero sum

def _gradient_direction(h, scores, beta=30.0, radius=0.1):
    """Subgradient direction with softmax weighting."""
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -100.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Clamp to feasible direction
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g -= float(np.mean(g))
    if np.max(np.abs(g)) < 1e-14:
        return None
    return g

def _local_opt(h, deadline, radius_init=0.15, max_iters=5000):
    """Active-set local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 20
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_direction(h, scores, beta=30.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.4
                n_active = min(n_active + 3, 40)
                if radius < 1e-6:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 12)
        else:
            stall += 1
            radius *= 0.75
            gap *= 1.2
            n_active = min(n_active + 2, 35)
            if stall % 30 == 0 and radius < 0.03:
                radius = 0.1
    
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=300):
    """Random pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.4)[0])
    zeros = list(np.where(h < 0.6)[0])
    rng = np.random.default_rng(999)
    stall = 0
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        n_attempts = max(1, len(ones) // 2)
        for _ in range(n_attempts):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.5)
            if c_try < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = list(np.where(h > 0.4)[0])
                zeros = list(np.where(h < 0.6)[0])
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 5
    freqs = rng.uniform(0.5, 10.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amplitudes = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _milp_small_n(n, deadline):
    """Try to find optimal binary sequence of length n using MILP."""
    import cvxpy as cp
    if n % 2 != 0 or n < 20 or n > 60:
        return None
    
    x = cp.Variable(n, boolean=True)
    # s_{j,d} for each pair at distance d
    # We'll use a dictionary approach
    s_vars = {}
    for d in range(1, n):
        for j in range(n - d):
            s_vars[(j, d)] = cp.Variable(boolean=True)
    
    M = cp.Variable(nonneg=True)
    
    constraints = [cp.sum(x) == n // 2]
    
    # Absolute difference constraints
    for d in range(1, n):
        for j in range(n - d):
            idx = (j, d)
            constraints.append(s_vars[idx] >= x[j + d] - x[j])
            constraints.append(s_vars[idx] >= x[j] - x[j + d])
    
    # Overlap constraints
    for d in range(1, n):
        E_d = cp.sum([s_vars[(j, d)] for j in range(n - d)])
        B_d = cp.sum([x[j + d] - x[j] for j in range(n - d)])
        constraints.append(E_d + B_d <= 2 * M)
        constraints.append(E_d - B_d <= 2 * M)
    
    obj = cp.Minimize(M)
    prob = cp.Problem(obj, constraints)
    
    try:
        # Try different solvers
        for solver in [cp.CBC, cp.SCIPOPT, cp.CPLEX, cp.GUROBI]:
            try:
                prob.solve(solver=solver, time_limit=max(5, int(deadline - time.time())))
                break
            except Exception:
                continue
        else:
            # Fallback
            remaining = deadline - time.time()
            if remaining < 5:
                return None
            prob.solve()
    except Exception:
        return None
    
    if x.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    x_val = np.round(x.value).astype(float)
    if abs(np.sum(x_val) - n // 2) > 1e-6:
        # Fix balance
        idx = np.argsort(x_val)[-n//2:]
        x_val = np.zeros(n)
        x_val[idx] = 1.0
    
    c5 = _exact_c5(x_val)
    return x_val, c5
