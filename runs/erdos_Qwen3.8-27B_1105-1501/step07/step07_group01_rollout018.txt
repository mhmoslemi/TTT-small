    w = np.exp(scaled)
    s = float(w.sum())
    if s < 1e-30:
        w = np.zeros(len(R))
        w[int(np.argmax(R))] = 1.0
    else:
        w /= s
    return w

def _multibeta_grad_descent(h, deadline, betas, max_iter=80000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    R = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(R)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 800:
            break
        improved = False
        R = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(R)) * dx
        for beta in betas:
            if time.time() > deadline:
                break
            w = _softmax_weights(R, beta)
            g = _wgrad(h, w)
            g = g - float(np.mean(g))
            d = -g
            dsum = float(np.sum(d))
            if abs(dsum) < 1e-18:
                continue
            T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            pos = d > 1e-15
            neg = d < -1e-15
            t_max = 2.0
            if np.any(pos):
                t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
            if np.any(neg):
                t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
            t_max *= (1.0 - 1e-10)
            if t_max < 1e-14:
                continue
            best_t, best_val = _line_search(R, T, C, t_max)
            if best_val < cur_c5 - 1e-13:
                h_new = h + best_t * d
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = _project(h_new, target)
                new_c5 = _exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-13:
                    h = h_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
    return best_h, best_c5
