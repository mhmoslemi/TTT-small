def _block_perturb(hb, rng, n_flips=8):
    n = len(hb)
    half = n // 2
    ones = np.where(hb > 0.5)[0]
    zeros = np.where(hb <= 0.5)[0]
    hp = hb.copy()
    k = min(n_flips, len(ones), len(zeros))
    if k == 0:
        return hp
    sel_o = rng.choice(len(ones), size=k, replace=False)
    sel_z = rng.choice(len(zeros), size=k, replace=False)
    for i in range(k):
        hp[ones[sel_o[i]]] = 0.0
        hp[zeros[sel_z[i]]] = 1.0

def _block_shift_perturb(hb, rng, max_block=15, max_shift=5):
    n = len(hb)
    hp = hb.copy()
    ones = np.where(hb > 0.5)[0]
    if len(ones) == 0:
        return hp
    # Find a contiguous block of 1s
    block_sizes = []
    i = 0
    while i < n:
        if hp[i] > 0.5:
            j = i
            while j < n and hp[j] > 0.5:
                j += 1
            bs = j - i
            if bs <= max_block:
                block_sizes.append((i, j))
            i = j
        else:
            i += 1
    if not block_sizes:
        return _block_perturb(hb, rng, 6)
    start, end = block_sizes[rng.integers(len(block_sizes))]
    length = end - start
    shift = rng.integers(-max_shift, max_shift + 1)
    ns, ne = start + shift, end + shift

    # Move the block

    hp[start:end] = 0.0
    hp[ns:ne] = tmp

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _line_search_c5(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    width = a_hi - a_lo
    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo_r, hi_r = a_lo, a_hi
    c1 = hi_r - gr * width
    c2 = lo_r + gr * width
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if hi_r - lo_r < 1e-13:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr * (hi_r - lo_r)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    mid = (lo_r + hi_r) / 2.0
    fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
    best_a, best_val = mid, fmid
    if f1 < best_val:
        best_a, best_val = c1, f1
    if f2 < best_val:
        best_a, best_val = c2, f2
    if a_lo > best_a - 1e-15 and float(np.max(S0)) * dx < best_val:
        best_a, best_val = a_lo, float(np.max(S0)) * dx
    return best_a, best_val

                d += rng.uniform(0.5, 1.0) * np.sin(2 * np.pi * f * x / n + p)
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-12:
                d /= nrm
                a, c_try = _line_search_c5(h, d, dx, rad)
                if c_try < best_new - 1e-14:
                    best_new = c_try
                    best_a = a
                    best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = _project(h + best_a * best_d, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.005:
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.10
                elif stall > stall_lim:
                    break
    return best_h, best_c

def _cubic_perturb(hb, rng, n_domain):
    n = len(hb)
    hp = hb.copy()
    f = rng.uniform(0.5, 6.0)
    p = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.1, 0.4)
    x = np.arange(n, dtype=float)
    pert = amp * np.sin(2 * np.pi * f * x / n + p)
    hp = _project(hp + pert, n / 2.0)
    # Threshold back to binary
    top = np.argsort(hp)[-n // 2:]
    hb_new = np.zeros(n)
    hb_new[top] = 1.0
    return hb_new

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)
    
    # Start from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)
    
    # Phase 1: Quick continuous optimization (8%)
    d1 = t0 + budget_s * 0.08
    h1, c1 = _continuous_opt(h_init, d1, r0=0.12, stall_lim=30, ndirs=3)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    
    # Phase 2: Intensive binary optimization (35%)
    d2 = t0 + budget_s * 0.43
    nc = len(best_h)
    hb = _to_bin(best_h, nc)
    
    # Multiple rounds of binary local search with perturbations
    best_bin = hb.copy()
    best_bin_c = _exact_c5(hb)
    rng_bin = np.random.default_rng(777)
    
    for round_num in range(30):
        if time.time() > d2:
            break
        # Local search
        sd = min(d2, time.time() + 8.0)
        hb_ls, c_ls = _bin_local_search(best_bin, sd, rng_bin, max_iter=80000, non_improve_limit=1500)
        if c_ls < best_bin_c:
            best_bin_c = c_ls
            best_bin = hb_ls.copy()
            if c_ls * (2.0 / nc) < best_c:
                best_c = c_ls * (2.0 / nc)
                best_h = best_bin.copy()
        # Perturb and retry
        if time.time() > d2 - 5:
            break
        if round_num % 3 == 0:
            hp = _block_perturb(best_bin, rng_bin, n_flips=10)
        elif round_num % 3 == 1:
            hp = _block_shift_perturb(best_bin, rng_bin)
        else:
            hp = _cubic_perturb(best_bin, rng_bin, nc)
        sd = min(d2, time.time() + 5.0)
        hb_ls2, c_ls2 = _bin_local_search(hp, sd, rng_bin, max_iter=50000, non_improve_limit=1000)
        if c_ls2 < best_bin_c:
            best_bin_c = c_ls2
            best_bin = hb_ls2.copy()
            if c_ls2 * (2.0 / nc) < best_c:
                best_c = c_ls2 * (2.0 / nc)
                best_h = best_bin.copy()
    
    # Phase 3: Multi-resolution search (20%)
    d3 = t0 + budget_s * 0.63
    base_n = len(best_h)
    for dn in [-20, -10, 10, 20, -30, 30, -40, 40]:
        if time.time() > d3:
            break
        n_try = base_n + dn
        if n_try < 80 or n_try > 300:
            continue
        hr = _resample(best_h, n_try)
        # Quick continuous opt
        sd = min(d3, time.time() + 4.0)
        hr, cr = _continuous_opt(hr, sd, r0=0.10, stall_lim=20, ndirs=3)
        if cr < best_c + 0.005 and time.time() < d3 - 3:
            # Binary refinement at this resolution
            hb_t = _to_bin(hr, n_try)
            rng_t = np.random.default_rng(999 + n_try)
            sd2 = min(d3, time.time() + 10.0)
            for rr in range(5):
                if time.time() > d3 - 2:
                    break
                hb_t, ct = _bin_local_search(hb_t, time.time() + 2.0, rng_t, max_iter=30000, non_improve_limit=800)
                if ct * (2.0 / n_try) < best_c:
                    best_c = ct * (2.0 / n_try)
                    best_h = hb_t.copy()
                hp = _block_perturb(hb_t, rng_t, 8)
                hb_t, ct2 = _bin_local_search(hp, time.time() + 2.0, rng_t, max_iter=20000, non_improve_limit=600)
                if ct2 * (2.0 / n_try) < best_c:
                    best_c = ct2 * (2.0 / n_try)
                    best_h = hb_t.copy()
        # Also try continuous result directly
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
    
    # Phase 4: Continuous refinement of best binary (20%)
    d4 = t0 + budget_s * 0.83
    nc = len(best_h)
    if nc % 2 != 0:
        # Re-center
        best_h = _to_bin(best_h, nc)
    
    for beta_start in [5.0, 10.0, 20.0]:
        if time.time() > d4 - 5:
            break
        # Add noise proportional to beta
        noise = rng.normal(0, 1.0 / beta_start, nc)
        noise -= np.mean(noise)
        hc = _project(best_h + noise, nc / 2.0)
        sd = min(d4, time.time() + budget_s * 0.05)
        hc, cc = _continuous_opt(hc, sd, r0=0.08, stall_lim=40, ndirs=4)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()
    
    # Phase 5: Final intensive binary + continuous (10%)
    d5 = deadline
    nc = len(best_h)
    
    # Last binary push
    hb = _to_bin(best_h, nc)
    rng_f = np.random.default_rng(55555)
    for _ in range(20):
        if time.time() > d5 - 5:
            break
        sd = min(d5 - 5, time.time() + 3.0)
        hb_f, cf = _bin_local_search(hb, sd, rng_f, max_iter=40000, non_improve_limit=1000, p2swap=0.2)
        cfx = cf * (2.0 / nc)
        if cfx < best_c:
            best_c = cfx
            best_h = hb_f.copy()
        if time.time() < d5 - 8:
            hp = _block_perturb(hb_f, rng_f, 6)
            sd2 = min(d5 - 5, time.time() + 2.0)
            hb_f2, cf2 = _bin_local_search(hp, sd2, rng_f, max_iter=25000, non_improve_limit=800)
            cfx2 = cf2 * (2.0 / nc)
            if cfx2 < best_c:
                best_c = cfx2
                best_h = hb_f2.copy()
  
    # Final continuous polish
    if deadline - time.time() > 5:
        nc = len(best_h)
        hf = _project(best_h.copy(), nc / 2.0)
        hf, cf = _continuous_opt(hf, deadline, r0=0.03, stall_lim=25, ndirs=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
