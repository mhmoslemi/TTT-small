  Changes happen when i is i or j (affecting h[i]) or when i+d is i or j (i.e., i = i-d or i = j-d, affecting (1-h[i+d])).
  
  So at most 4 terms change per lag. For each lag d:
  - Term at i: was h[i]*(1-h[i+d]), now (1-h[i])*(1-h[i+d]) → change = -h[i]*(1-h[i+d]) + (1-h[i])*(1-h[i+d]) = (1-h[i+d]) - h[i]*(1-h[i+d]) = (1-h[i+d])*(1-h[i]+h[i]-1)... 

  Hmm, let me just be concrete. Before swap: h[i]=1, h[j]=0. After: h[i]=0, h[j]=1.
  
  For lag d, the score changes by:
  Δscore_d = [new h[i]*(1-new h[i+d]) - old h[i]*(1-old h[i+d])] (if i is valid source for lag d)
           + [new h[i]*(1-new h[i+d]) - ... for i being the target]
  
  This is getting complicated. Let me just use the full correlation computation. For n=180, it's fast enough.

  Actually, for n=180, np.correlate(h, 1-h, mode='full') takes about 5-10 microseconds (it's a small array). So evaluating C₅ for a candidate swap is ~10μs. A full sweep of 8100 swaps would take ~81ms. We can do thousands of sweeps in the available time.

  But we don't need a full sweep each time. We can do a "best improvement" or "first improvement" strategy:
  - First improvement: try swaps in some order, accept the first one that improves.
  - Best improvement: try all swaps, accept the best one.
  - Random: try a random swap, accept if it improves (or with SA acceptance).

  For speed, I'll use a hybrid: try a subset of random swaps per iteration, and do a full sweep only when stuck.

Let me now write the code:

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[::-1][:k]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _build_grad(h):
    """Build gradient matrix G where G[r, :] = d(score_r)/d h."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # Center (lag 0): d/dh_i [h_i * (1-h_i)] = 1 - 2*h_i
    G[n-1, :] = 1.0 - 2.0 * h
    # Positive lags
    for d in range(1, n):
        r = n - 1 + d
        m = n - d
        G[r, :m] += (1.0 - h[d:d+m])
        G[r, d:d+m] += -h[:m]
    # Negative lags
    for g in range(1, n):
        r = n - 1 - g
        m = n - g
        G[r, g:g+m] += (1.0 - h[:m])
        G[r, :m] += -h[g:g+m]
    return G

def _line_search_exact(h, p, dx, p_raw):
    """Exact line search along h + t*p using parabolic formula.
    p_raw = np.correlate(p, p, mode='full') precomputed."""
    n = len(h)
    S0 = _raw_scores(h)
    T = _build_grad(h) @ p  # G @ p
    # Safe upper bound for t
    t_hi = 1.0
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])) * 0.99)
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))) * 0.99)
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search
    n_grid = 50
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t*T - t*t*p_raw)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    if hi_b - lo_b < 1e-12:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1*T - c1*c1*p_raw)) * dx
        f2 = float(np.max(S0 + c2*T - c2*c2*p_raw)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1*T - c1*c1*p_raw)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2*T - c2*c2*p_raw)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    
    return opt_t, float(np.max(S0 + opt_t*T - opt_t*opt_t*p_raw)) * dx

def _lp_direction_full(h, scores, G, radius, active_margin):
    """Full epigraph LP: minimize max score using active constraints."""
    n = len(h)
    max_score = float(np.max(scores))
    active = np.where(scores >= max_score - active_margin)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    # Cap active set
    if len(active) > 60:
        # Keep top 60
        active_scores = scores[active]
        top_in_active = np.argsort(active_scores)[::-1][:60]
        active = active[top_in_active]
    
    k = len(active)
    # Variables: p (n) + eta (1) = n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize eta
    
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # eta free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _continuous_local(h, deadline, radius_init=0.10, active_margin=3.0, max_iter=200):
    """Continuous local optimization with full epigraph LP."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        
        scores = _raw_scores(h)
        G = _build_grad(h)
        p = _lp_direction_full(h, scores, G, radius, active_margin)
        
        if p is None or np.max(np.abs(p)) < 1e-12:
            stall += 1
            radius *= 0.7
            active_margin *= 0.85
            if radius < 1e-5:
                break
            continue
        
        p_raw = np.correlate(p, p, mode='full')
        alpha, pred_c5 = _line_search_exact(h, p, dx, p_raw)
        
        if alpha > 1e-12:
            new_h = h + alpha * p
            new_h = _project_sum(new_h, target)
            new_c5 = _exact_c5(new_h)
            
            if new_c5 < cur_c5 - 1e-13:
                h = new_h
                cur_c5 = new_c5
                stall = 0
                radius = min(radius * 1.05, 0.2)
                active_margin = min(active_margin * 1.02, 8.0)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
                radius *= 0.8
                active_margin *= 0.9
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5
