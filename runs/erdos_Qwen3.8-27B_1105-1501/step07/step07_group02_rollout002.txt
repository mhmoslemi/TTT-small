def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _row_grad(h, k):
    """Compute gradient of correlation row with lag k w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    p = abs(k)
    m = n - p
    if p == 0:
        # k=0: S = sum h[i]*(1-h[i]), grad = 1-2h
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # S = sum_{i=0}^{m-1} h[i+p]*(1-h[i])
        # d/dh[i] for i in [p, n-1]: += (1-h[i-p])
        # d/dh[i] for i in [0, m-1]: -= h[i+p]
        g[p:] += (1.0 - h[:m])
        g[:m] -= h[p:]
    else:
        # k < 0, p = -k
        # S = sum_{i=p}^{n-1} h[i-p]*(1-h[i])
        # d/dh[i] for i in [0, m-1]: += (1-h[i+p])
        # d/dh[i] for i in [p, n-1]: -= h[i-p]
        g[:m] += (1.0 - h[p:])
        g[p:] -= h[:m]
    return g

def _weighted_direction(h, S, beta, k_top):
    """Compute weighted gradient direction from top-k rows."""
    n = len(h)
    L = 2 * n - 1
    # Select top-k rows
    top_idx = np.argsort(S)[-k_top:]
    top_S = S[top_idx]
    # Softmax weights
    w = np.exp(beta * (top_S - top_S.max()))
    w /= w.sum()
    # Weighted gradient
    g = np.zeros(n)

        g += w[i] * _row_grad(h, k)

    g -= np.mean(g)
    # Normalize
    maxabs = np.max(np.abs(g))
    if maxabs < 1e-15:

    g /= maxabs
    return g
