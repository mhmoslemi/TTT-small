    a_max = min(h[i], 1.0 - h[j])
    if a_max < 1e-14:
        return 0.0, float(np.max(B))
    omh = 1.0 - h
    diff = omh[::-1] - h
    T = np.zeros(2 * n - 1)
    T[n-1-i : 2*n-1-i] -= diff
    T[n-1-j : 2*n-1-j] += diff
    Q = np.zeros(2 * n - 1)
    Q[n-1] = 2.0
    m1 = j - i + (n - 1)
    if 0 <= m1 < 2*n-1:
        Q[m1] += -1.0
    m2 = i - j + (n - 1)
    if 0 <= m2 < 2*n-1:
        Q[m2] += -1.0
    best_a = 0.0
    best_v = float(np.max(B))
    alphas = np.linspace(0.0, a_max, 16)
    vals = np.empty(16)
    for idx in range(16):
        a = alphas[idx]
        vals[idx] = float(np.max(B + a * T - a * a * Q))
        if vals[idx] < best_v:
            best_v = vals[idx]
            best_a = a
    if best_a > 0.0 and best_a < a_max:
        bidx = int(np.argmin(vals))
        lo_b = alphas[max(0, bidx - 1)]
        hi_b = alphas[min(15, bidx + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(B + c1 * T - c1 * c1 * Q))
        f2 = float(np.max(B + c2 * T - c2 * c2 * Q))
        for _ in range(35):
            if hi_b - lo_b < 1e-10:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(B + c1 * T - c1 * c1 * Q))
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(B + c2 * T - c2 * c2 * Q))
        mid = (lo_b + hi_b) / 2.0
        fm = float(np.max(B + mid * T - mid * mid * Q))
        if fm < best_v:
            best_v = fm
            best_a = mid
    return best_a, best_v

def _pair_pivot_search(h, deadline, max_sweeps=200, top_k=25, tau_init=0.5):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    B = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(B)) * dx
    best_h = h.copy()
    best_c5 = cur
    tau = tau_init
    stall = 0
    total_stall = 0
    rng = np.random.default_rng(12345)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if total_stall > 400:
            break
        corr = B
        g = _weighted_grad(h, corr, tau)
        g_centered = g - np.mean(g)
        pos_mask = (g_centered > 0) & (h > 1e-6)
        neg_mask = (g_centered < 0) & (h < 1.0 - 1e-6)
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        pos_sorted = pos_idx[np.argsort(g_centered[pos_idx])[::-1]][:top_k]
        neg_sorted = neg_idx[np.argsort(g_centered[neg_idx])[:top_k]]
        best_improve = 0.0
        best_pair = None
        n_pairs = len(pos_sorted) * len(neg_sorted)
        for pi in pos_sorted:
            for nj in neg_sorted:
                a, v = _pair_line_search(B, h, int(pi), int(nj))
                c5_try = v * dx
                if c5_try < cur - 1e-12:
                    if cur - c5_try > best_improve:
                        best_improve = cur - c5_try
                        best_pair = (int(pi), int(nj), a)
        if best_pair is not None:
            pi, nj, a = best_pair
            h[pi] -= a
            h[nj] += a
            h = np.clip(h, 0.0, 1.0)
            B = np.correlate(h, 1.0 - h, mode='full')
            cur = float(np.max(B)) * dx
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            total_stall = 0
            tau = max(0.01, tau * 0.95)
        else:
            stall += 1
            total_stall += 1
            tau *= 1.3
            if tau > 50.0:
                tau = 50.0
            if stall > 30:
                # Random restart: try random pairs
                found = False
                for _ in range(2000):
                    pi = int(rng.integers(0, n))
                    nj = int(rng.integers(0, n))
                    if h[pi] < 1e-6 or h[nj] > 1.0 - 1e-6:
                        continue
                    a, v = _pair_line_search(B, h, pi, nj)
                    c5_try = v * dx
                    if c5_try < cur - 1e-12:
                        h[pi] -= a
                        h[nj] += a
                        h = np.clip(h, 0.0, 1.0)
                        B = np.correlate(h, 1.0 - h, mode='full')
                        cur = float(np.max(B)) * dx
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    stall = 0
                    tau = max(0.05, tau * 0.5)
                else:
                    break
    return best_h, best_c5

def _binary_swap_search(h_bin, deadline, max_iters=200000, top_k=30):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    B = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(B)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        g = _weighted_grad(cur, B, 1.0)
        g_c = g - np.mean(g)
        os_arr = np.array(ones)
        zs_arr = np.array(zeros)
        o_sorted = os_arr[np.argsort(g_c[os_arr])[::-1]][:top_k]
        z_sorted = zs_arr[np.argsort(g_c[zs_arr])[:top_k]]
        found = False
        for oi in o_sorted:
            for zj in z_sorted:
                cur[int(oi)] = 0.0
                cur[int(zj)] = 1.0
                B_new = np.correlate(cur, 1.0 - cur, mode='full')
                c5_new = float(np.max(B_new)) * dx
                cur[int(oi)] = 1.0
                cur[int(zj)] = 0.0
                if c5_new < cur_c5 - 1e-14:
                    cur[int(oi)] = 0.0
                    cur[int(zj)] = 1.0
                    B = B_new
                    cur_c5 = c5_new
                    oi_idx = ones.index(int(oi))
                    zj_idx = zeros.index(int(zj))
                    ones[oi_idx] = int(zj)
                    zeros[zj_idx] = int(oi)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                    stall = 0
                    found = True
                    break
            if found:
                break
        if not found:
            # Try random swaps
            improved = False
            for _ in range(500):
                oi = int(rng.integers(0, len(ones)))
                zj = int(rng.integers(0, len(zeros)))
                cur[ones[oi]] = 0.0
                cur[zeros[zj]] = 1.0
                B_new = np.correlate(cur, 1.0 - cur, mode='full')
                c5_new = float(np.max(B_new)) * dx
                if c5_new < cur_c5 - 1e-14:
                    B = B_new
                    cur_c5 = c5_new
                    ones[oi], zeros[zj] = zeros[zj], ones[oi]
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()
                    stall = 0
                    improved = True
                    break
                else:
                    cur[ones[oi]] = 1.0
                    cur[zeros[zj]] = 0.0
            if not improved:
                stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    p = np.zeros(n)
    for f, ph in zip(freqs, phases):
        p += np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amplitude / 2.0
    p += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Continuous pair-pivot search (50% of budget)
    d1 = t0 + budget_s * 0.50
    h1, c5_1 = _pair_pivot_search(h_init, d1, max_sweeps=500, top_k=20, tau_init=0.3)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary search on best (20%)
    d2 = t0 + budget_s * 0.70
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    h_bin2, c5_bin2 = _binary_swap_search(h_bin, d2, max_iters=300000, top_k=35)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.90
    n_best = len(best_h)
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10]:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _pair_pivot_search(h_r, min(d3, time.time() + budget_s * 0.015), max_sweeps=80, top_k=15, tau_init=0.3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
        # Try binary on this
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r, n_try)
            h_b, c5_b = _binary_swap_search(h_b, min(d3, time.time() + budget_s * 0.01), max_iters=50000, top_k=25)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                n_best = n_try

    # Phase 4: Final intensive refinement (10%)
    d4 = deadline - 5
    n_fin = len(best_h)

    # More pair-pivot on the best
    h4 = _pair_pivot_search(best_h.copy(), min(d4, time.time() + budget_s * 0.03), max_sweeps=200, top_k=30, tau_init=0.1)
    if h4[1] < best_c5:
        best_c5 = h4[1]
        best_h = h4[0].copy()
        n_fin = len(best_h)

    # Basin hopping
    if time.time() < d4 - 10:
        scale = 0.03
        h_bash = best_h.copy()
        for _ in range(60):
            if time.time() > d4 - 5:
                break
            h_p = _perturb(h_bash, scale, rng)
            h_t, c5_t = _pair_pivot_search(h_p, min(d4, time.time() + budget_s * 0.005), max_sweeps=40, top_k=15, tau_init=0.2)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                h_bash = h_t.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.1)

    # Final binary polish
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h, n_fin)
        h_b, c5_b = _binary_swap_search(h_b, min(d4, time.time() + budget_s * 0.02), max_iters=100000, top_k=40)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            n_fin = len(best_h)

    # Final project and evaluate
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
