def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x > 0.0) & (x < 1.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
        else:
            # All at bounds, distribute to non-bound
            if res > 0:
                mask = x < 1.0
            else:
                mask = x > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                x = x.copy()
                x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    M = float(np.max(S0))
    if M < 1e-15:

    s = float(np.sum(w))
    if s < 1e-30:

        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:

    return w

def _grad_matrix(h):

    L = 2 * n - 1

        row = k + n - 1
        if k >= 0:

            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

        return t_lo, float(np.max(S0))

    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        top_k = max(1, len(S0) // 8)
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    t2 = ts[:, None] * ts[:, None]
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - t2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _max_step(h, d, margin=0.999):
    t_max = 1e10
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_pos = (1.0 - h[pos]) / d[pos]
        t_max = min(t_max, float(t_pos.min()))
    if np.any(neg):
        t_neg = (-h[neg]) / d[neg]
        t_max = min(t_max, float(t_neg.min()))
    if t_max < 0 or not np.isfinite(t_max):
        return 0.0
    return t_max * margin

def _lp_direction(h, S0, G, deadline):
    n = len(h)
    try:
        import cvxpy as cp
        delta = cp.Variable(n)
        t_var = cp.Variable()
        lo = -h + 1e-10
        hi = 1.0 - h - 1e-10
        constraints = [
            delta >= lo,
            delta <= hi,
            cp.sum(delta) == 0,
            G @ delta + S0 <= t_var
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.GLOP, cp.PDLP, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved or delta.value is None:
            return None
        d = np.array(delta.value, dtype=float).copy()
        d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
        return d
    except Exception:
        return None

def _gradient_direction(h, S0, beta):
    w = _smooth_weights(S0, beta)
    G = _grad_matrix(h)
    grad = G.T @ w
    grad = grad - np.mean(grad)
    return -grad, G

def _lp_ls_optimize(h, deadline, max_iter=5000, beta_init=8.0, beta_max=60.0, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    beta = beta_init
    stall = 0
    rng = np.random.default_rng(seed)
    one_mh = 1.0 - h
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        G = _grad_matrix(h)
        d = _lp_direction(h, S0, G, deadline)
        
        if d is None or float(np.linalg.norm(d)) < 1e-12:
            # Fallback to gradient direction
            d, G = _gradient_direction(h, S0, beta)
        
        d = d - np.mean(d)
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-10:
            stall += 1
            continue
        
        # Scale direction for reasonable step size
        scale = 0.05 / (d_norm + 1e-12)
        d = d * scale
        d = d - np.mean(d)
        
        t_max = _max_step(h, d)
        if t_max < 1e-12:
            stall += 1
            continue
        t_cap = min(t_max, 0.3)
        
        one_mh = 1.0 - h
        T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        t_opt, val_opt = _line_search(S0, T, C, 0.0, t_cap)
        
        h_try = h + t_opt * d
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        
        if c5_try < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            beta = min(beta * 1.03, beta_max)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.125, 0.0625]:
                t_try = t_opt * frac
                if t_try < 1e-12:
                    break
                h_try = h + t_try * d
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-14:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    S0 = np.correlate(h, 1.0 - h, mode='full')
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                beta = max(beta * 0.92, beta_init)
                if stall % 30 == 0:
                    pert = rng.standard_normal(n) * 0.02
                    pert -= np.mean(pert)
                    h_pert = _project(h + pert, target)
                    c5_pert = _exact_c5(h_pert)
                    if c5_pert < cur_c5 + 0.003:
                        h = h_pert
                        S0 = np.correlate(h, 1.0 - h, mode='full')
    
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    radius = float(radius)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        w = _smooth_weights(S0, beta)
        G = _grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        n_sel = min(n_pairs, n // 2)
        sources = sort_idx[-n_sel:]
        sinks = sort_idx[:n_sel]
        pairs = []
        for ii in range(n_sel):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(10, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        
        best_c5_try = cur_c5
        best_step = None
        one_mh = 1.0 - h
        
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if stall % 50 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones[:]:
            if time.time() > deadline:
                break
            for oz in zeros[:]:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    
    if h0 is not None and len(h0) > 0 and np.all(np.isfinite(h0)):
        n0 = len(h0)
        h_init = _project(h0.copy(), n0 / 2.0)
    else:
        n0 = 180
        h_init = _to_binary(np.full(n0, 0.5))
    
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: LP + line search at parent n (35% time)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_ls_optimize(h_init, d1, max_iter=5000, beta_init=8.0, beta_max=60.0, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-n exploration (25% time)
    d2 = t0 + budget_s * 0.60
    n_base = len(best_h)
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, 20, -20, 24, -24]
    tried = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n_base + dn
        if n_try < 60 or n_try > 250 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_r, c5_r = _lp_ls_optimize(h_r, sub_dl, max_iter=1500, beta_init=10.0, beta_max=50.0, seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_base = n_try
    
    # Phase 3: Pair optimization refinement (15% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    if time.time() < d3:
        h3 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.08)
        h3, c5_3 = _pair_optimize(h3, sub_dl, radius=0.12, n_pairs=50, beta=25.0, stall_limit=100, seed=555)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    
    # Phase 4: Binary diversification (12% time)
    d4 = t0 + budget_s * 0.87
    n_curr = len(best_h)
    if time.time() < d4:
        h_bin = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.004, seed=777)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary result
        if time.time() < d4:
            n_curr = len(best_h)
            h_pol = _project(best_h.copy(), n_curr / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _lp_ls_optimize(h_pol, sub_dl3, max_iter=1000, beta_init=15.0, beta_max=50.0, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 5: Final LP polish (7% time)
    d5 = deadline - 3
    n_curr = len(best_h)
    if time.time() < d5:
        h5 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d5, deadline)
        h5, c5_5 = _lp_ls_optimize(h5, sub_dl, max_iter=3000, beta_init=20.0, beta_max=80.0, seed=888)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Final validation
    n_final = len(best_h)
    best_h = _project(best_h.copy(), n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_check = float(np.sum(best_h))
    if abs(sum_check - n_final / 2.0) > 0.01:
        best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h.copy(), final_c5, n_final
