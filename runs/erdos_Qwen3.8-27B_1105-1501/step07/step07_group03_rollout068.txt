import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project_sum(v, target):
    """Project v to [0,1]^n with sum = target using bisection on threshold"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new

def _binary_greedy(h, deadline, max_sweeps=50):
    """Steepest descent binary local search"""
    n = len(h)
    cur = (h > 0.5).astype(float)
    cur_m = _corr_max(cur)
    best = cur.copy()
    best_m = cur_m
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        if not ones or not zeros:
            break
        best_improve = 0.0
        best_swap = None
        for oi in ones:
            for oz in zeros:
                if cur[oi] != 1.0 or cur[oz] != 0.0:
                    continue
                cur[oi] = 0.0
                cur[oz] = 1.0
                m = _corr_max(cur)
                improve = cur_m - m
                cur[oi] = 1.0
                cur[oz] = 0.0
                if improve > best_improve + 1e-15:
                    best_improve = improve
                    best_swap = (oi, oz)
                    cur[oi] = 0.0
                    cur[oz] = 1.0
        if best_swap is not None:
            # Apply the best swap
            oi, oz = best_swap
            # cur already has the swap applied from the loop
            cur_m = _corr_max(cur)
            if cur_m < best_m:
                best_m = cur_m
                best = cur.copy()
        else:
            break
    return best, best_m * (2.0 / n)

def _binary_sa(h, deadline, max_iter=100000, T_init=0.5, T_min=0.01):
    """Simulated annealing for binary optimization"""
    n = len(h)
    cur = (h > 0.5).astype(float)
    cur_m = _corr_max(cur)
    best = cur.copy()
    best_m = cur_m
    rng = np.random.default_rng(12345)
    T = T_init
    alpha = (T_min / T_init) ** (1.0 / max_iter)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T *= alpha
        if not ones or not zeros:
            break
        i = ones[int(rng.integers(0, len(ones)))]
        j = zeros[int(rng.integers(0, len(zeros)))]
        cur[i] = 0.0
        cur[j] = 1.0
        new_m = _corr_max(cur)
        delta = new_m - cur_m
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_m = new_m
            # Update ones/zeros lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_m < best_m - 1e-15:
                best_m = cur_m
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_m * (2.0 / n)

def _frac_polish(h_bin, deadline, max_iter=2000):
    """Polish a binary solution by allowing small fractional values"""
    n = len(h_bin)
    target = n / 2.0
    h = h_bin.astype(float).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(67890)
    # Try perturbing pairs of positions
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a 1-position and a 0-position
        ones = np.where(h > 0.99)[0]
        zeros = np.where(h < 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # Try any pair
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
        else:
            i = ones[int(rng.integers(0, len(ones)))]
            j = zeros[int(rng.integers(0, len(zeros)))]
        if i == j:
            continue
        # Try moving mass from i to j
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
            h_try = h.copy()
            amt = min(frac * h[i], (1.0 - h[j]) * frac)
            if amt < 1e-6:
                continue
            h_try[i] -= amt
            h_try[j] += amt
            h_try = _project_sum(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
    return best_h, best_c5

def _grad_optimize(h, deadline, beta=20.0, max_iter=500):
    """Projected gradient with softmax weighting"""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.astype(float).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.1
    stall = 0
    beta_cur = beta
    for it in range(max_iter):
        if time.time() > deadline:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta_cur * shifted, -50, 0))
        w = vp / (vp.sum() + 1e-30)
        # Compute gradient of weighted sum of correlations
        g = _compute_grad(h, w)
        g -= g.mean()  # Preserve sum
        # Line search
        best_alpha = 0.0
        best_val = cur_c5
        for a in [step, step*0.5, step*0.25, step*0.1, step*2]:
            h_try = _project_sum(h - a * g, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = a
        if best_alpha > 0:
            h = _project_sum(h - best_alpha * g, target)
            cur_c5 = _exact_c5(h)
            step *= 1.1
            beta_cur = min(beta_cur * 1.03, 80.0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            step *= 0.7
            beta_cur = max(beta_cur * 0.9, 5.0)
            if stall > 50:
                break
    return best_h, best_c5

def _compute_grad(h, w):
    """Compute gradient of sum_k w[k] * correlate(h, 1-h)[k]"""
    n = len(h)
    g = np.zeros(2*n - 1)  # This isn't what we want

    # d/dh[i] of correlate(h, 1-h)[r] = ?
    # correlate(h, 1-h)[r] = sum_t h[t] * (1-h[t+r-(n-1)])
    # d/dh[i] = (1-h[i+r-(n-1)]) if i+r-(n-1) valid, as the h[t] term
    #           + h[i-(r-(n-1))] * (-1) if i-(r-(n-1)) valid, as the (1-h) term

    # Let f_r = sum_t h[t] * (1-h[t+s_r]) where s_r = r - (n-1)
    # d f_r / d h[i] = (1-h[i+s_r]) * [indicator that i+s_r is valid] 
    #                  + h[i] * (-1) * [indicator that i = t+s_r for some valid t]
    #                    wait no: d/dh[i] of h[t]*(1-h[t+s_r])
    #                    = (1-h[i+s_r]) if t=i (the h[t] factor)
    #                    + h[t]*(-1) if t+s_r = i, i.e., t = i-s_r (the (1-h) factor)
    # So: d f_r / d h[i] = (1-h[i+s_r])_valid - h[i-s_r]_valid
    # where _valid means the index is in [0, n-1]
    
    # The gradient of sum_r w[r] * f_r with respect to h[i] is:
    # sum_r w[r] * [(1-h[i+s_r])_valid - h[i-s_r]_valid]

    # sum_r w[r] * (1-h[i+s_r]) = sum_r w[r] * (1-h[i+r-(n-1)])
    # = sum_s w[s+(n-1)] * (1-h[i+s])  (reindexing with s = r-(n-1))
    # = (correlate_with_1)(i) where weights are w shifted

    n2 = 2*n - 1

        if -s >= 0:

            grad[s:] -= w[r] * h[:n-s]
        else:
            # i ranges from 0 to n-1+s
            grad[:n+s] -= w[r] * h[-s:]
    return grad
