        return np.ones(n)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_move_eval(h, i, j, corr):
    """Compute A, B vectors for pair move h[i]+=t, h[j]-=t."""
    n = len(h)
    m = 2 * n - 1
    s = np.arange(-(n-1), n)
    
    A = np.zeros(m)
    js_minus = j - s
    js_plus = j + s
    is_minus = i - s
    is_plus = i + s
    
    m1 = (js_minus >= 0) & (js_minus < n)
    A[m1] += h[js_minus[m1]]
    m2 = (js_plus >= 0) & (js_plus < n)
    A[m2] += h[js_plus[m2]]
    m3 = (is_minus >= 0) & (is_minus < n)
    A[m3] -= h[is_minus[m3]]
    m4 = (is_plus >= 0) & (is_plus < n)
    A[m4] -= h[is_plus[m4]]
    
    B = np.zeros(m)
    B[n-1] = -2.0
    d = j - i
    if d != 0 and abs(d) < n:
        B[n-1+d] = 1.0
        B[n-1-d] = 1.0
    
    return A, B

def _optimal_t(corr, A, B, t_lo, t_hi):
    """Find t in [t_lo, t_hi] minimizing max(corr + t*A + t^2*B)."""
    # Build candidate set
    candidates = [t_lo, t_hi, 0.0, (t_lo + t_hi) / 2.0]
    
    # Vertices of near-max quadratics
    thresh = corr.max() - 0.15 * max(1.0, corr.max())
    active = np.where(corr > thresh)[0]
    for idx in active[:30]:
        b = B[idx]
        a = A[idx]
        if abs(b) > 1e-12:
            tv = -a / (2.0 * b)
            if t_lo <= tv <= t_hi:
                candidates.append(tv)
    
    # Quantile-based candidates
    if t_hi - t_lo > 1e-8:
        for q in [0.1, 0.3, 0.5, 0.7, 0.9]:
            candidates.append(t_lo + q * (t_hi - t_lo))
    
    best_t = 0.0
    best_val = float(np.max(corr))
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        val = float(np.max(corr + t * A + t * t * B))
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def _pairwise_sa(h, corr, deadline, rng, n_iter=100000, T_init=0.3, T_min=0.01, decay=0.9998):
    """Simulated annealing with pairwise mass transfers."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = corr.copy()
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init

            break
        if T < T_min:
            break
            
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:

        A, B = _pair_move_eval(h, i, j, corr)
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-14:
            continue
        
        t_opt, val_opt = _optimal_t(corr, A, B, t_lo, t_hi)
        
        if val_opt < cur_c5 / dx - 1e-14:
            # Improving move, always accept
            h[i] += t_opt
            h[j] -= t_opt
            corr = corr + t_opt * A + t_opt * t_opt * B
            cur_c5 = val_opt * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Check if we should accept (annealing)
            delta = (val_opt - cur_c5) * dx
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h[i] += t_opt
                h[j] -= t_opt
                corr = corr + t_opt * A + t_opt * t_opt * B
                cur_c5 = val_opt * dx
