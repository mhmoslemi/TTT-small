Hmm, this momentum approach might be too simple. Let me think of a better approach.

Actually, the key issue with gradient descent on the smooth max is that:
1. The smooth max is not strongly convex (it's a max of linear/quadratic functions).
2. The constraint (sum = n/2) creates a coupling between all variables.
3. The objective is nonsmooth (we're approximating it with the smooth max).

A better approach might be to use **proximal gradient** or **accelerated proximal gradient** (FISTA) on the smooth part, with the projection onto the feasible set as the proximal operator.

But the projection onto {h: 0≤h≤1, sum(h)=n/2} is exactly what `_project` does! So FISTA would be:
1. y = x + (t_{k-1}-1)/(t_k+1) * (x - x_prev)
2. x_new = project(y - step * gradient(y))
3. t_{k+1} = (1 + sqrt(1+4*t_k²))/2

This should converge at rate O(1/k²) for the smooth part.

Let me implement FISTA:

```python
def _fista(h, deadline, tau=10.0, step_init=0.01, max_iter=10000, seed=42):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    x_prev = h.copy()
    t = 1.0
    step = step_init
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = x + (t - 1) / (t + 1) * (x - x_prev)
        y = np.clip(y, 0, 1)  # Keep in box for gradient eval
        val, grad = _smooth_obj_and_grad(y, tau)
        gc = grad - np.mean(grad)
        x_new = _project(y - step * gc, target)
        
        # Check improvement
        new_c5 = _exact_c5(x_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = x_new.copy()
        
        # Adaptive step size
        if new_c5 > _exact_c5(x) + 1e-10:
            step *= 0.7
            x_new = x  # Reject step
        else:
            step = min(step * 1.1, 0.1)
        
        x_prev = x
        x = x_new
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Adaptively increase tau
        if it % 500 == 0:
            tau = min(tau * 1.5, 100.0)
    
    return best_h, best_c5
```

For the active set LP, I'll use the formulation:
- Minimize t
- Subject to: ∇C_k · d - t ≤ C_k(h₀) - C_0 for all k ∈ L (active set)
  where C_0 = max_k C_k(h₀)
  (This ensures C_k(h₀+d) ≈ C_k(h₀) + ∇C_k·d ≤ C_0 + (C_k(h₀) - C_0) = C_k(h₀)... hmm that's not quite right)

Actually, let me reformulate. I want to find d that reduces the maximum correlation:
- Minimize t
- Subject to: C_k(h₀) + ∇C_k · d ≤ t for all k in active set L
- sum(d) = 0
- |d[j]| ≤ radius for all j (to keep the step small)
- h₀[j] + d[j] ∈ [0, 1] for all j

The objective t represents the new maximum correlation after the step. We want it to be as small as possible.

Variables: d (n) + t (1) = n+1
Constraints:
- |L| inequalities: ∇C_k · d - t ≤ -C_k(h₀) for k ∈ L
- 1 equality: sum(d) = 0
- n box constraints: -h₀[j] ≤ d[j] ≤ 1-h₀[j]
- Plus the radius constraint: |d[j]| ≤ radius (this can be combined with the box)

```python
def _active_set_lp(h, deadline, top_k=15, radius=0.1, max_iters=200, stall_limit=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = float(np.max(corr))
        
        # Get top-k active lags
        idx = np.argsort(corr)[-top_k:]
        
        # Compute gradients for active lags
        G = np.zeros((len(idx), n))
        for ai, ki in enumerate(idx):
            k = int(ki - (n - 1))
            if k >= 0:
                m = n - k
                G[ai, :m] += (1.0 - h[k:])
                G[ai, k:] += -h[:m]
            else:
                kk = -k
                m = n - kk
                G[ai, kk:] += (1.0 - h[:m])
                G[ai, :m] += -h[kk:]
        
        # LP: minimize t
        # s.t. G[i]·d - t ≤ -corr[idx[i]] for i in range(top_k)
        #      sum(d) = 0
        #      -radius ≤ d[j] ≤ radius
        #      h[j]-1 ≤ d[j] ≤ 1-h[j] (box)
        
        # Variables: [d_0, ..., d_{n-1}, t]
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr[idx]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi = min(radius, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((None, None))  # t is unbounded
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
        except:
            stall += 1
            radius *= 0.7
            continue
        
        d = res.x[:n]
        if np.max(np.abs(d)) < 1e-14:
            stall += 1
            radius *= 0.8
            continue
        
        # Take the step
        h_new = _project(h + d, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
    
    return best_h, best_c5
```

```python
def _binary_local_search(h_bin, deadline, max_sweeps=200, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline or stall > 30:
            break
        improved = False
        # Random order
        order = rng.permutation(len(ones))
        for ii in order:
            if time.time() > deadline:
                break
            oi = ones[ii]
            # Try swapping with each zero (or a random subset)
            # For efficiency, try a random subset of zeros
            j_candidates = rng.choice(len(zeros), size=min(len(zeros), 50), replace=False)
            for jj in j_candidates:
                oz = zeros[jj]
                if oi == oz:
                    continue
                # Compute the change in correlation
                # Moving a 1 from oi to oz
                # corr change at lag k:
                #   - (1-h[oi+k]) for the removal at oi (contributes to lags where oi is the "1" position)
                #   + (1-h[oz+k]) for the addition at oz
                # Actually, let me just compute it directly
                h[oi] = 0
                h[oz] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(new_corr)) * dx
                if new_c5 < cur_c5 - 1e-15:
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    # Update lists
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    improved = True
                    break
                else:
                    h[oi] = 1
                    h[oz] = 0
            
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5
```
