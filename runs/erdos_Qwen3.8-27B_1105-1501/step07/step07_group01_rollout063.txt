def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    # Project v onto {x: 0 ≤ x ≤ 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)

    w = exp_c / np.sum(exp_c)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

            step *= 1.0 / step_norm
        h_new = _project(h + step, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h = h_new
            stall += 1
            if stall > max_stall:
                break
    return best_h, best_c5

def _coord_opt(h, deadline, max_sweeps=2000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline or stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(-j, 0):
                g[n - 1 + d] = -h[j + d]
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            for _ in range(40):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = float(np.max(S0 + m1 * g))
                f2 = float(np.max(S0 + m2 * g))
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                improved = True
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap(h, deadline, n_iters=5000, seed=99):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_iters):
        if time.time() > deadline or stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = 1.0
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            continue
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        candidates = [(0.0, float(np.max(S0)) * dx),
                      (opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx),
                      (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
        best_t = min(candidates, key=lambda x: x[1])[0]
        best_v = min(c[1] for c in candidates)
        if best_v < cur_c5 - 1e-14:
            h = _project(h + best_t * d, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
    return best_h, best_c5

def _lp_refine(h, deadline, radius_init=0.08, stall_limit=60, top_k=12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(scores)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[jj]), min(radius, 1.0 - h[jj])) for jj in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
        for _ in range(40):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
        opt_t = (lo_t + hi_t) / 2.0
        f0 = float(np.max(S0))
        fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C))
        fo = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
        best_t = min([(0.0, f0), (opt_t, fo), (t_hi, fe)], key=lambda x: x[1])[0]
        if best_t > 0:
            h = _project(h + best_t * p, target)
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 - 1e-13:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.3)
            else:
                stall += 1
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.5, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_opt(h_bin, deadline, max_iter=30000, seed=777):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.005
    stall = 0
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        oi = list(ones)[int(rng.integers(0, len(ones)))]
        oz = list(zeros)[int(rng.integers(0, len(zeros)))]
        g = np.zeros(2 * n - 1)
        for dd in range(0, n - oi):
            g[n - 1 + dd] -= (1.0 - cur[oi + dd])
        for dd in range(-oi, 0):
            g[n - 1 + dd] += cur[oi + dd]
        for dd in range(0, n - oz):
            g[n - 1 + dd] += (1.0 - cur[oz + dd])
        for dd in range(-oz, 0):
            g[n - 1 + dd] += -cur[oz + dd]
        new_S0 = S0 + g
        new_c5 = float(np.max(new_S0)) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            S0 = new_S0
            cur_c5 = new_c5
            ones.discard(oi); ones.add(oz)
            zeros.discard(oz); zeros.add(oi)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient descent (adaptive T)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _adam_smooth(best_h, d1, T_start=2.0, T_end=15.0, lr=0.4, seed=10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _coord_opt(best_h, min(d2, time.time() + budget_s * 0.10), max_sweeps=2000, seed=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_refine(best_h, d3, radius_init=0.06, stall_limit=50, top_k=12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair swap
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _pair_swap(best_h, d4, n_iters=4000, seed=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Second smooth gradient pass with different params
    d5 = t0 + budget_s * 0.70
    h5, c5_5 = _adam_smooth(best_h, d5, T_start=1.0, T_end=10.0, lr=0.2, seed=400)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-n exploration
    d6 = t0 + budget_s * 0.82
    n_curr = len(best_h)
    for dn in [-8, -4, 4, 8, -12, 12]:
        if time.time() > d6:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _adam_smooth(h_r, sub_dl, T_start=2.0, T_end=12.0, lr=0.3, seed=500+dn)
        sub_dl2 = min(d6, time.time() + budget_s * 0.006)
        h_r, c5_r2 = _coord_opt(h_r, sub_dl2, max_sweeps=500, seed=600+dn)
        if c5_r2 < c5_r:
            c5_r = c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Binary exploration
    d7 = t0 + budget_s * 0.90
    n_bin = len(best_h)
    h_bin_init = np.zeros(n_bin)
    h_bin_init[np.argsort(best_h)[-n_bin // 2:]] = 1.0
    h_bin, c5_bin = _binary_opt(h_bin_init, min(d7, time.time() + budget_s * 0.04), seed=700)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d7 - 1:
        h_pol = _project(h_bin.copy(), n_bin / 2.0)
        h_pol, c5_pol = _lp_refine(h_pol, min(d7, time.time() + budget_s * 0.02), radius_init=0.04, stall_limit=30, top_k=10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Final intensive optimization
    d8 = deadline
    h8, c5_8 = _coord_opt(best_h, min(d8, time.time() + budget_s * 0.015), max_sweeps=2000, seed=800)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8 - 2:
        h8, c5_8 = _lp_refine(best_h, min(d8, time.time() + budget_s * 0.01), radius_init=0.03, stall_limit=40, top_k=15)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    if time.time() < d8 - 2:
        h8, c5_8 = _pair_swap(best_h, min(d8, time.time() + budget_s * 0.01), n_iters=3000, seed=900)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    # Basin hopping with gradient
    scale = 0.015
    n_restarts = 0
    while time.time() < d8 - 1 and n_restarts < 40:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d8 - 1, time.time() + budget_s * 0.003)
        h_t, c5_t = _coord_opt(h_pert, sub_dl, max_sweeps=200, seed=1000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.1)

    # Final check
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
