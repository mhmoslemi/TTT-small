import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_dx = 0.0
_g_target = 0.0
_g_tau = 0.05

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[mid + d, d:] += 1.0 - h[:m]
        G[mid + d, :m] -= h[d:]
        G[mid - d, :m] += 1.0 - h[d:]
        G[mid - d, d:] -= h[:m]
    return G

def _smooth_obj(x):
    s = _raw_scores(x) * _g_dx
    mval = float(np.max(s))
    u = (s - mval) / _g_tau
    u = np.clip(u, -500.0, 0.0)
    z = np.exp(u)
    return mval + _g_tau * float(np.log(np.sum(z)))

def _smooth_jac(x):
    s = _raw_scores(x) * _g_dx
    mval = float(np.max(s))
    u = (s - mval) / _g_tau
    u = np.clip(u, -500.0, 0.0)
    z = np.exp(u)
    w = z / float(np.sum(z))
    G = _build_grad_matrix(x)
    return (G.T @ w) * _g_dx

def _smooth_eq(x):
    return float(np.sum(x)) - _g_target

def _smooth_eq_jac(x):
    return np.ones(len(x))

def _slsqp_smooth(h, deadline, tau, maxiter=80):

    _g_target = _g_n / 2.0
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    cons = {'type': 'eq', 'fun': _smooth_eq, 'jac': _smooth_eq_jac}
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', jac=_smooth_jac,
                       bounds=bounds, constraints=cons,

            x = np.clip(res.x, 0.0, 1.0)

    # More pair polish
    sub_d3 = min(d4, time.time() + budget_s * 0.03)
    h4, c5_4 = _pair_polish(best_h, sub_d3, n_pairs=8000, seed=444)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.85
    scale = 0.02
    for it in range(150):
        if time.time() > d5 - 5:
            break
        h_p = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.007)
        h_t, c5_t = _annealed_smooth(h_p, sub_dl, n_stages=3, tau_start=max(0.02, best_c5 * 0.2), tau_end=0.003)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.88)
        else:
            scale = min(0.08, scale * 1.04)

    # Phase 6: Dimension transfer (10%)
    d6 = deadline
    n_cur = len(best_h)
    offsets = [-4, 4, -8, 8, -6, 6, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d6 - 5:
            break
        n_try = n_cur + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.01)
        h_r, c5_r = _annealed_smooth(h_r, sub_dl, n_stages=4, tau_start=0.04, tau_end=0.003)
        sub_dl2 = min(d6, time.time() + budget_s * 0.008)
        h_r, c5_r = _pair_polish(h_r, sub_dl2, n_pairs=5000, seed=n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Final polish
    if time.time() < deadline - 2:
        h_f, c5_f = _gradient_pair_polish(best_h, deadline, n_pairs=5000, seed=888)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection and verification
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
