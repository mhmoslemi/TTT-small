def fast_swap_eval(h, C, p, q, n):
    """Evaluate change in max(C) after swapping h[p]=1->0, h[q]=0->1.
    Returns (new_max, C_change_at_max_lag, affected_count)"""
    u = 1.0 - h
    # For each affected lag, compute the change
    # p contributes at lags where p-L or p+L is in range
    # q contributes at lags where q-L or q+L is in range
    
    # p-L in [0,n-1] means L in [p-(n-1), p], i.e., r in [2p-n+2, 2p-n+... ]
    # This is getting complicated. Let me just iterate over the affected lags.
    # p-L valid: L from max(0, p-n+1) to min(n-1, p), so r from n-1+max(0,p-n+1) to n-1+p
    # p+L valid: L from max(0, n-1-p)... wait p+L < n means L < n-p, and L >= 0 (or negative)
    # Actually L can be negative too.
    # p+L in [0,n-1]: L in [-p, n-1-p]
    # p-L in [0,n-1]: L in [p-(n-1), p]
    
    # Combined: L in [min(-p, p-n+1), max(n-1-p, p)] = [p-n+1, p] ∪ [-p, n-1-p]
    # Union: L in [p-n+1, max(p, n-1-p)]
    
    # Similarly for q.
    # So affected lags are in [min(p,q)-n+1, max(p,q)) roughly.
    # More precisely, the union of affected L values.
    
    # For simplicity and correctness, let's just compute the change for all lags
    # that could be affected. The range of affected L is:
    # From p: L in [p-(n-1), p] ∪ [-p, n-1-p]  → [p-n+1, max(p, n-1-p)]
    # From q: L in [q-(n-1), q] ∪ [-q, n-1-q]  → [q-n+1, max(q, n-1-q)]
    # Union: [min(p-n+1, q-n+1), max(max(p,n-1-p), max(q,n-1-q))]
    
    L_lo = min(p, q) - (n - 1)
    L_hi = max(max(p, n - 1 - p), max(q, n - 1 - q))
    
    # Clip to valid range [-n+1, n-1]
    L_lo = max(L_lo, -(n - 1))
    L_hi = min(L_hi, n - 1)
    
    new_max = 0.0
    for L in range(int(L_lo), int(L_hi) + 1):
        r = n - 1 + L
        delta = 0.0
        # -u[p-L] if 0 <= p-L < n
        if 0 <= (p - L) < n:
            delta -= u[p - L]
        # +u[q-L] if 0 <= q-L < n
        if 0 <= (q - L) < n:
            delta += u[q - L]
        # +h[p+L] if 0 <= p+L < n
        if 0 <= (p + L) < n:
            delta += h[p + L]
        # -h[q+L] if 0 <= q+L < n
        if 0 <= (q + L) < n:
            delta -= h[q + L]
        # Cross term (only at L=0)
        if L == 0:
            delta -= 2.0  # (-1)(+1) + (+1)(-1) = -2
        C[r] += delta
    
    new_max = float(np.max(C))
    return new_max

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

        else:
