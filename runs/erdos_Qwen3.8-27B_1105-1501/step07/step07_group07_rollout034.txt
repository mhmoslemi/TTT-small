def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _full_lp_direction(h, scores, G, radius, active_margin):
    n = len(h)
    max_s = float(np.max(scores))
    active = np.where(scores > max_s - active_margin)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    
    A_act = G[active]
    s_act = scores[active]
    k = len(active)
    
    # min eta s.t. A_act @ p - eta <= -s_act, sum(p) = 0, bounds on p
    # Variables: [p_0, ..., p_{n-1}, eta]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize eta
    
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = A_act
    A_ub[:, n] = -1.0
    b_ub = -s_act
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-12:
                return p
    except Exception:
        pass
    return None

                if radius < 1e-6:

    return best_h, best_c5

def _to_binary(h):

    target = n // 2
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0

def _binary_local(h_bin, deadline, max_rounds=50):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:

                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, scores, G, radius, n_active=15):
    n = len(h)
    max_s = float(np.max(scores))
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:n_active]
    A_act = G[active]
    s_act = scores[active]
    k = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = A_act
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-12:
                return p
    except Exception:
        pass
    return None

def _line_search(h, p, t_max_cap=2.0):
    n = len(h)
    dx = 2.0 / n
    R = _raw_scores(h)
    G = _grad_matrix(h)
    T = G @ p
    P = np.correlate(p, p, mode='full')
    t_hi = t_max_cap
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14:
        return 0.0, float(np.max(R) * dx)
    n_grid = 50
    grid_t = np.linspace(0.0, t_hi, n_grid)
    best_idx = 0
    best_val = float('inf')
    for idx in range(n_grid):
        t = grid_t[idx]
        val = float(np.max(R + t * T - t * t * P)) * dx
        if val < best_val:
            best_val = val
            best_idx = idx
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-12:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(R + c1 * T - c1 * c1 * P)) * dx
        f2 = float(np.max(R + c2 * T - c2 * c2 * P)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(R + c1 * T - c1 * c1 * P)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(R + c2 * T - c2 * c2 * P)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(R + opt_t * T - opt_t * opt_t * P)) * dx
    else:
        opt_t = grid_t[best_idx]
        opt_val = best_val
    candidates = [(opt_t, opt_val),
                  (0.0, float(np.max(R) * dx)),
                  (t_hi, float(np.max(R + t_hi * T - t_hi * t_hi * P)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _continuous_opt(h, deadline, radius_init=0.10, stall_limit=50, max_radius=0.25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    n_active = 12
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _raw_scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            n_active = max(4, n_active - 3)
            radius *= 0.85
            if radius < 1e-6:
                break
            p = _lp_direction(h, scores, G, radius, n_active)
            if p is not None and float(np.max(np.abs(p))) > 1e-13:
                stall = max(0, stall - 1)
            else:
                if stall > 20:
                    n_active = min(30, n_active + 4)
                    radius = min(radius * 1.3, max_radius)
                continue
        alpha, pred_c5 = _line_search(h, p, t_max_cap=radius * 2.5)
        if alpha > 1e-10 and pred_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = _project_sum(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.02, max_radius)
                n_active = max(6, n_active - 1)
            else:
                stall += 1
                radius *= 0.9
        else:
            stall += 1
            radius *= 0.88
            if radius < 1e-6:
                break
    return best_h, best_c5

def _pair_descent(h, deadline, n_trials=5000, radius=0.2, seed=99):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        p = np.zeros(n)
        p[i] = 1.0
        p[j] = -1.0
        alpha, pred_c5 = _line_search(h, p, t_max_cap=radius)
        if alpha > 1e-10 and pred_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = _project_sum(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _binary_local(h_bin, deadline, max_rounds=40):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2.0 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    if not np.isfinite(final_c5) or final_c5 > 0.5:
        best_h = _project_sum(h_init, n0 / 2.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
