import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 0
_g_target = 0
_g_dx = 0
_g_tau = 1.0

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_value_and_grad(h, tau):
    """Compute LSE value and gradient for the raw correlation scores."""
    raw = _raw_scores(h)
    m = float(np.max(raw))
    u = (raw - m) / tau
    # Clip for numerical stability
    u = np.clip(u, -50.0, 50.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        # All underflowed, use uniform over max rows
        lse = m
        p = np.zeros_like(raw)
        max_mask = raw >= m - 1e-10
        p[max_mask] = 1.0 / max(1, int(np.sum(max_mask)))
    else:
        lse = m + tau * np.log(zsum)
        p = z / zsum
    
    # Compute gradient: grad[i] = sum_r p[r] * d(raw[r])/d(h[i])
    n = len(h)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if p[r] < 1e-15:
            continue
        lag = r - (n - 1)
        pr = p[r]
        # d(raw[r])/d(h[i]) = (1-h[i-lag])*[0<=i-lag<n] - h[i+lag]*[0<=i+lag<n]
        if lag >= 0:
            mvalid = n - lag
            # First term: (1-h[i-lag]) for i from lag to n-1
            grad[lag:] += pr * (1.0 - h[:mvalid])
            # Second term: -h[i+lag] for i from 0 to n-1-lag
            grad[:mvalid] -= pr * h[lag:]
        else:
            g = -lag
            mvalid = n - g
            # First term: (1-h[i-lag]) = (1-h[i+g]) for i from 0 to n-1-g
            grad[:mvalid] += pr * (1.0 - h[g:])
            # Second term: -h[i+lag] = -h[i-g] for i from g to n-1
            grad[g:] -= pr * h[:mvalid]
    
    return float(lse), grad

def _smooth_obj(x):
    val, _ = _lse_value_and_grad(x, _g_tau)
    return val

def _smooth_jac(x):
    _, grad = _lse_value_and_grad(x, _g_tau)
    return grad

def _eq_con(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _bounds():
    return [(0.0, 1.0)] * _g_n

def _lse_anneal(h, deadline, tau_list=None, maxiter_per_tau=80):
    """Run LSE annealing on h."""
    global _g_n, _g_target, _g_dx, _g_tau
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    
    # Determine tau schedule
    raw = _raw_scores(h)
    raw_max = float(np.max(raw))
    raw_range = raw_max - float(np.min(raw))
    if tau_list is None:
        tau0 = max(raw_range * 0.5, raw_max * 0.3, 5.0)
        tau_list = []
        t = tau0
        while t > 0.05:
            tau_list.append(t)
            t *= 0.5
        tau_list.append(0.05)
    
    cur_h = _project_sum(h.copy(), n / 2.0)
    cur_c5 = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    
    upper_cons = [
        {'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}
    ]
    
    for tau in tau_list:
        if time.time() > deadline:
            break
        _g_tau = tau
        try:
            res = minimize(
                fun=_smooth_obj,
                x0=cur_h,
                jac=_smooth_jac,
                method='SLSQP',
                bounds=_bounds(),
                constraints=upper_cons,
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-10, 'disp': False}
            )
            if res.x is not None:
                new_h = _project_sum(res.x, n / 2.0)
                new_c5 = _exact_c5(new_h)
                if new_c5 < cur_c5 + 0.01:  # Accept if not too much worse
                    cur_h = new_h
                    cur_c5 = _exact_c5(cur_h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
                # else: keep cur_h, try next tau from current position
        except Exception:
            pass
    
    return best_h, best_c5

def _line_search_exact(h1, h2, deadline, n_grid=30):
    """Exact line search between h1 and h2."""
    best_t = 0.0
    best_val = _exact_c5(h1)
    
    # Check feasibility of the segment
    # Both h1 and h2 should be feasible (sum = target, 0<=h<=1)
    # Linear interpolation preserves sum if both have same sum
    
    if time.time() > deadline:
        return best_t, best_val
    
    ts = np.linspace(0, 1, n_grid)
    vals = np.zeros(n_grid)
    for i in range(n_grid):
        t = ts[i]
        h_mid = h1 + t * (h2 - h1)
        # Check bounds
        if np.min(h_mid) < -1e-10 or np.max(h_mid) > 1 + 1e-10:
            vals[i] = 1e10
            continue
        h_mid = np.clip(h_mid, 0, 1)
        h_mid = _project_sum(h_mid, _g_target)
        vals[i] = _exact_c5(h_mid)
    
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Golden section around best
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project_sum(h1 + c1 * (h2 - h1), _g_target))
        f2 = _exact_c5(_project_sum(h1 + c2 * (h2 - h1), _g_target))
        for _ in range(30):
            if hi - lo < 1e-10 or time.time() > deadline:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project_sum(h1 + c1 * (h2 - h1), _g_target))
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project_sum(h1 + c2 * (h2 - h1), _g_target))
        mid = (lo + hi) / 2.0
        fmid = _exact_c5(_project_sum(h1 + mid * (h2 - h1), _g_target))
        if fmid < best_val:
            best_val = fmid
            best_t = mid
    
    return best_t, best_val

def _pair_polish(h, deadline, n_trials=5000):

    global _g_n, _g_target
    n = len(h)

    _g_target = n / 2.0

            break
        if stall > 300:
            break

        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Direction: +1 at i, -1 at j (or vice versa)
        for sign in [1, -1]:
            d = np.zeros(n)
            d[i] = sign
            d[j] = -sign
            # Check feasibility
            h_trial = cur_h + d
            if np.min(h_trial) < -0.5 or np.max(h_trial) > 1.5:
                continue
            h_trial = _project_sum(h_trial, n / 2.0)
            c_try = _exact_c5(h_trial)
            if c_try < cur_c5 - 1e-12:
                cur_h = h_trial
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur_h.copy()
                stall = 0
                break
        else:
            # Try small perturbation
            for amp in [0.1, 0.05, 0.25]:
                d = np.zeros(n)
                d[i] = amp
                d[j] = -amp
                h_trial = cur_h + d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amp):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(1.0, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amp
    return _project_sum(h + pert, n / 2.0)
