from scipy.optimize import minimize

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    m = len(x)
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h):
    n = len(h)
    m = n // 2
    x = 0.5 * h[:m] + 0.5 * (1.0 - h[n-1:m-1:-1])
    return np.clip(x, 0.0, 1.0)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _anti_smooth_and_grad(x, beta):
    m = len(x)
    n = 2 * m
    h = _anti_to_full(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(corr))
    z = beta * (corr - M)
    z = np.clip(z, -50.0, 0.0)
    ep = np.exp(z)
    ep_sum = float(np.sum(ep))
    obj = (M + np.log(ep_sum) / beta) / n
    w = ep / ep_sum
    G = np.zeros(n)

            best_c5 = c5_trial
            best_x = x_cur.copy()
    return best_x, best_c5

def _resample_anti(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return _full_to_anti(h)
    if n_new % 2 != 0:
        n_new += 1
    m_new = n_new // 2
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return np.clip(_full_to_anti(h_new), 0.0, 1.0)

def _full_smooth_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(corr))
    z = beta * (corr - M)
    z = np.clip(z, -50.0, 0.0)
    ep = np.exp(z)
    ep_sum = float(np.sum(ep))
    obj = (M + np.log(ep_sum) / beta) * (2.0 / n)
    w = ep / ep_sum
    G = np.zeros(n)
    active = w > 1e-8
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            mm = n - tau
            if mm > 0:
                G[tau:] += wr * (1.0 - h[:mm])
                G[:mm] -= wr * h[tau:tau + mm]
        else:
            d = -tau
            mm = n - d
            if mm > 0:
                G[:mm] += wr * (1.0 - h[d:])
                G[d:] -= wr * h[:mm]
    G -= float(np.mean(G))
    return obj, G

def _optimize_full_lbfgs(h_init, deadline, target, beta_start=10.0, beta_end=80.0, n_steps=6):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    h_cur = h_init.copy()
    for bs in range(n_steps):
        if time.time() > deadline:
            break
        frac = bs / max(n_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (n_steps - bs) * 1.3)
        
        def fun_and_grad(hv):
            hv_c = np.clip(hv, 0.0, 1.0)
            o, g = _full_smooth_and_grad(hv_c, beta)
            return o, g
        
        try:
            res = minimize(fun_and_grad, h_cur, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxfun': 1500, 'disp': False})
            h_cur = _project(np.clip(res.x, 0.0, 1.0), target)
        except Exception:
            h_cur = _project(h_cur, target)
        c5_trial = _exact_c5(h_cur)
        if c5_trial < best_c5 - 1e-12:
            best_c5 = c5_trial
            best_h = h_cur.copy()
    return best_h, best_c5

def _basin_hopping_anti(x_best, c5_best, deadline, rng, n_restarts_max=60, amp_init=0.04):
    m = len(x_best)
    amp = amp_init
    for _ in range(n_restarts_max):
        if time.time() > deadline:
            break
        pert = amp * rng.normal(0, 1, m)
        x_pert = np.clip(x_best + pert, 0.0, 1.0)
        sub_dl = min(deadline - 2, time.time() + 8.0)
        x_trial, c5_trial = _optimize_anti_lbfgs(x_pert, sub_dl, beta_start=15.0, beta_end=80.0, max_beta_steps=4)
        if c5_trial < c5_best - 1e-12:
            c5_best = c5_trial
            x_best = x_trial.copy()
            amp = max(0.005, amp * 0.85)
        else:
            amp = min(0.08, amp * 1.1)
    return x_best, c5_best

def _pair_sweep_full(h, deadline, max_rounds=5):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        i = int(rng_pair.integers(0, n))
        j = int(rng_pair.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a_lo = -1e20
        a_hi = 1e20
        if d[i] > 0:
            a_lo = max(a_lo, -h_cur[i])
            a_hi = min(a_hi, 1.0 - h_cur[i])
        if d[j] < 0:
            a_lo = max(a_lo, h_cur[j] - 1.0)
            a_hi = min(a_hi, -h_cur[j])
        if a_hi <= a_lo + 1e-14:
            continue
        u = 1.0 - h_cur
        S0 = np.correlate(h_cur, u, mode='full')
        T = np.correlate(d, u, mode='full') - np.correlate(h_cur, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        dx = 2.0 / n
        def fa(a):
            return float(np.max(S0 + a * T - a * a * Q)) * dx
        gt = np.linspace(a_lo, a_hi, 40)
        gf = np.array([fa(t) for t in gt])
        bi = int(np.argmin(gf))
        lo = gt[max(0, bi - 1)]
        hi = gt[min(len(gt) - 1, bi + 1)]
        gr_fn = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr_fn * (hi - lo)
        c2 = lo + gr_fn * (hi - lo)
        f1, f2 = fa(c1), fa(c2)
        for _ in range(30):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr_fn * (hi - lo)
                f1 = fa(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr_fn * (hi - lo)
                f2 = fa(c2)
        opt_t = (lo + hi) / 2.0
        if opt_t != 0.0:
            h_new = h_cur + opt_t * d
            h_new = _project(h_new, target)
            c_new = _exact_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
                improved = True
        if not improved and time.time() > deadline:
            break
    return best_h, best_c5

rng_pair = np.random.default_rng(12345)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)
    
    # Phase 1: Anti-palindromic optimization at parent n (30%)
    d1 = t0 + budget_s * 0.30
    if n0 % 2 != 0:
        h0p = _project(h0, n0 // 2 * 2 / (n0 // 2 * 2) * (n0 // 2))
    x0 = _full_to_anti(best_h)
    x1, c5_1 = _optimize_anti_lbfgs(x0, d1, beta_start=6.0, beta_end=50.0, max_beta_steps=10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = _anti_to_full(x1).copy()
    
    # Phase 2: Multi-resolution anti-space search (40%)
    d2 = t0 + budget_s * 0.70
    n_list = []
    for dn in range(-40, 61, 8):
        nt = n0 + dn
        if nt >= 60 and nt <= 280 and nt % 2 == 0 and nt not in n_list:
            n_list.append(nt)
    n_list.sort(key=lambda n: abs(n - n0))
    
    for nt in n_list:
        if time.time() > d2:
            break
        h_resampled = np.interp((np.arange(nt) + 0.5) / nt,
                                 (np.arange(len(best_h)) + 0.5) / len(best_h), best_h)
        x_init = np.clip(_full_to_anti(h_resampled), 0.0, 1.0)
        sub_dl = min(d2 - 2, time.time() + budget_s * 0.04)
        x_opt, c5_opt = _optimize_anti_lbfgs(x_init, sub_dl, beta_start=10.0, beta_end=60.0, max_beta_steps=5)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = _anti_to_full(x_opt).copy()
    
    # Phase 3: Basin hopping in anti-space (20%)
    d3 = t0 + budget_s * 0.88
    x_best = _full_to_anti(best_h)
    x_best, c5_bh = _basin_hopping_anti(x_best, best_c5, d3, rng, n_restarts_max=80, amp_init=0.05)
    if c5_bh < best_c5:
        best_c5 = c5_bh
        best_h = _anti_to_full(x_best).copy()
    
    # Phase 4: Full-space polish (12%)
    d4 = deadline
    n_cur = len(best_h)
    h_full = _project(best_h.copy(), n_cur / 2.0)
    h_full, c5_full = _optimize_full_lbfgs(h_full, min(d4 - 10, time.time() + budget_s * 0.06),
                                           n_cur / 2.0, beta_start=15.0, beta_end=100.0, n_steps=4)
    if c5_full < best_c5:
        best_c5 = c5_full
        best_h = h_full.copy()
    
    if time.time() < d4 - 5:
        h_full, c5_ps = _pair_sweep_full(best_h, d4 - 5, max_rounds=200)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_full.copy()
    
    # Final: try a few neighbor n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > d4 - 8:
            break
        nt = n_final + dn
        if nt < 60 or nt > 300 or nt % 2 != 0:
            continue
        h_r = np.interp((np.arange(nt) + 0.5) / nt,
                         (np.arange(n_final) + 0.5) / n_final, best_h)
        x_r = np.clip(_full_to_anti(h_r), 0.0, 1.0)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.02)
        x_r, c5_r = _optimize_anti_lbfgs(x_r, sub_dl, beta_start=20.0, beta_end=80.0, max_beta_steps=4)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = _anti_to_full(x_r).copy()
    
    # Final validation and repair
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 and final_c5 - best_c5 < 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
