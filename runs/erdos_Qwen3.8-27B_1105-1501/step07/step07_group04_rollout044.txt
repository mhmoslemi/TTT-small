def _swap_delta(h, i0, j0):
    """Compute change in correlation vector when swapping h[i0]:1->0, h[j0]:0->1."""
    n = len(h)
    delta = np.zeros(2*n-1)
    
    # Effect of h[i0] going 1->0
    # As first factor (h[i+k]): contributes -(1-h[i0-k]) to corr[n-1+k]
    #   valid for k where 0 <= i0-k < n, i.e., i0-n+1 <= k <= i0
    #   corr index = n-1+k
    for k in range(max(0, i0-n+1), min(n, i0+1)):
        # corr index n-1+k, the term is h[i0]*(1-h[i0-k]) which was (1-h[i0-k]), now 0
        idx = i0 - k
        val = 1.0 - h[idx]
        if idx == j0:
            val = 0.0  # new h[j0] = 1, so 1-h[j0] = 0
        delta[n-1+k] -= val
    
    # As second factor (1-h[i]): contributes +h[i0+k] to corr[n-1+k]
    #   valid for k where 0 <= i0+k < n, i.e., 0 <= k < n-i0
    for k in range(0, n - i0):
        val = h[i0+k]
        if i0+k == j0:
            val = 1.0  # new h[j0] = 1
        delta[n-1+k] += val
    
    # Effect of h[j0] going 0->1
    # As first factor (h[i+k]): contributes +(1-h[j0-k]) to corr[n-1+k]
    for k in range(max(0, j0-n+1), min(n, j0+1)):
        idx = j0 - k
        val = 1.0 - h[idx]
        if idx == i0:
            val = 1.0  # new h[i0] = 0, so 1-h[i0] = 1
        delta[n-1+k] += val
    
    # As second factor (1-h[i]): contributes -h[j0+k] to corr[n-1+k]
    for k in range(0, n - j0):
        val = h[j0+k]
        if j0+k == i0:
            val = 0.0  # new h[i0] = 0
        delta[n-1+k] -= val
    
    return delta
