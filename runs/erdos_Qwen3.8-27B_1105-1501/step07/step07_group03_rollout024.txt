import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, d):
    """Gradient of correlation at lag d (actual shift) for h of length n."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] += -h[d:]
    else:
        d2 = -d
        g[:n - d2] += (1.0 - h[d2:])
        g[d2:] += -h[:n - d2]
    return g

def _lp_step(h, top_k=15, delta=0.3):
    """Solve LP to minimize max correlation over top_k lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Select top lags
    top_idx = np.argsort(-corr)[:top_k]
    # Build constraint matrix
    # For each top lag r: sum_j g_r[j] * x_j - t <= -corr[r]
    # where g_r is gradient at h for lag r
    m = len(top_idx)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, r in enumerate(top_idx):
        d = int(r) - (n - 1)
        g = _lag_grad(h, d)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[r]
    
    # Equality: sum x = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h-delta) <= x_j <= min(1, h+delta), and t free (or t >= 0)
    lb = np.maximum(0.0, h - delta)
    ub = np.minimum(1.0, h + delta)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))  # t >= 0
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        x_lp = res.x[:n]
        return x_lp, top_idx
    return None, top_idx

def _line_search_lp(h, x_lp, dx, max_calls=50):
    """Line search on direction from h to x_lp."""
    n = len(h)
    d = x_lp - h
    dm = np.max(np.abs(d))
    if dm < 1e-15:
        return h, _exact_c5(h)
    
    # Find max alpha keeping in [0,1] and sum = n/2
    # sum(d) should be 0 (both h and x_lp satisfy sum = n/2)
    # alpha_max from bounds
    alpha_max = 1.0  # since x_lp is already in [0,1] and we go from h to x_lp
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    alpha_max = max(alpha_max, 0.0)
    
    # Golden section search
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    lo, hi = 0.0, alpha_max
    if hi < 1e-12:
        return best_h, best_c5
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * d, n / 2.0))
    
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d, n / 2.0))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d, n / 2.0))
    
    opt = (lo + hi) / 2.0
    best_alpha = opt
    h_cand = _project(h + opt * d, n / 2.0)
    c5_cand = _exact_c5(h_cand)
    if c5_cand < best_c5:
        best_c5 = c5_cand
        best_h = h_cand
    # Also check the endpoint
    h_end = _project(h + alpha_max * d, n / 2.0)
    c5_end = _exact_c5(h_end)
    if c5_end < best_c5:
        best_c5 = c5_end
        best_h = h_end
    
    return best_h, best_c5

def _gradient_step(h, beta=20.0, radius=0.2):
    """Softmax-weighted gradient step."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    shifted = corr - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    
    # Compute weighted gradient
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        lg = _lag_grad(h, d)
        g += w[r] * lg
    g -= np.mean(g)
    
    # Scale to radius
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return h, _exact_c5(h)
    d_dir = -g / nm
    return d_dir

def _gradient_line_search(h, d_dir, radius=0.2):
    """Line search along gradient direction."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    raw = corr
    
    # Quadratic model
    T = np.zeros(2 * n - 1)
    Q = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        lg = _lag_grad(h, d)
        T[r] = np.dot(d_dir, lg)
        # Second order: T' and change in gradient
        # For simplicity, just do golden section
        Q[r] = np.dot(d_dir, d_dir)  # rough
    
    # Just do golden section
    alpha_max = radius
    pos = d_dir > 1e-12
    neg = d_dir < -1e-12
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d_dir[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d_dir[neg]))))
    alpha_max = max(alpha_max, 0.0)
    
    if alpha_max < 1e-12:
        return h, _exact_c5(h)
    
    lo, hi = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * d_dir, n / 2.0))
    f2 = _exact_c5(_project(h + c2 * d_dir, n / 2.0))
    
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * d_dir, n / 2.0))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * d_dir, n / 2.0))
    
    opt = (lo + hi) / 2.0
    h_new = _project(h + opt * d_dir, n / 2.0)
    c5_new = _exact_c5(h_new)
    
    if c5_new < _exact_c5(h) - 1e-14:
        return h_new, c5_new
    return h, _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for k in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur = h_new
            cur_c5 = new_c5
            ones = [k for k in range(n) if cur[k] > 0.5]
            zeros = [k for k in range(n) if cur[k] <= 0.5]
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:

import numpy as np

from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] += -h[d:]
    else:
        d2 = -d
        g[:n - d2] += (1.0 - h[d2:])
        g[d2:] += -h[:n - d2]
    return g

def _lp_solve(h, top_k, delta):

        A_ub[i, :n] = g
        A_ub[i, n] = -1.0

def _softmax_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    shifted = corr - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0.0))

        d = r - (n - 1)

    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm < 1e-12:

    return -g / nm

                neg = d_dir < -1e-12
                if np.any(pos):
                    alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d_dir[pos])))
                if np.any(neg):
                    alpha_max = min(alpha_max, float(np.min(h[neg] / (-d_dir[neg]))))
                if alpha_max > 1e-12:
                    h_new, c5_new = _line_search(h, h + alpha_max * d_dir, max_iter=40)
                    if c5_new < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = c5_new
                        if cur_c5 < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
                        stall = 0
                        improved = True
        if not improved:
            stall += 1
            n_lags = min(n_lags + 1, 25)
            if stall % 50 == 0:
                n_lags = 12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for k in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.004):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 350:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9996
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_improve(h_bin, deadline, max_sweeps=5):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-15:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    improved = True
                    zeros = list(np.where(cur <= 0.5)[0])
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _opt_lp(h_init, d1, n_lags=15, dllta=0.25, stall_limit=120, use_soft=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary refinement (15%)
    d2 = t0 + budget_s * 0.35
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.06)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_bin, c5_b = _pair_improve(h_bin, sub_dl, max_sweeps=3)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()
    if time.time() < d2 - 1:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_pol, c5_p = _opt_lp(h_pol, sub_dl, n_lags=12, stall_limit=40, use_soft=True)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in list(range(-14, 15, 2)):
        if time.time() > d3 - 4:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 220 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _opt_lp(h_r, sub_dl, n_lags=12, stall_limit=35, use_soft=True)
        if c5_r < best_c5 + 0.001 and time.time() < d3 - 2:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=10000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    # Tightening LP passes
    for delta in [0.15, 0.08, 0.04]:
        if time.time() > d4 - 4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4 = _project(best_h.copy(), n_curr / 2.0)
        h4, c5_4 = _opt_lp(h4, sub_dl, n_lags=20, stall_limit=80, use_soft=True)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary round
    if time.time() < d4 - 4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=30000, T_init=0.0015)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        if time.time() < d4 - 2:
            sub_dl = min(d4, time.time() + budget_s * 0.02)
            h_b, c5_b = _pair_improve(h_b, sub_dl, max_sweeps=2)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
    # Final LP polish
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _opt_lp(h4, sub_dl, n_lags=25, stall_limit=60, use_soft=True)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 50:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.010)
        h_t, c5_t = _opt_lp(h_pert, sub_dl, n_lags=15, stall_limit=25, use_soft=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.10, scale * 1.12)

    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _opt_lp(h6, deadline, n_lags=25, stall_limit=50, use_soft=False)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
