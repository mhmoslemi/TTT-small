def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    err = target - float(np.sum(x))
    if abs(err) > 1e-12:
        if err > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += err / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[mid + d, d:] += (1.0 - h[:m])
        G[mid + d, :m] += -h[d:d + m]
        G[mid - d, :m] += (1.0 - h[d:d + m])
        G[mid - d, d:] += -h[:m]
    return G

def _lse_value_grad(h, tau):
    scores = _raw_scores(h)
    max_s = float(np.max(scores))
    u = (scores - max_s) / tau
    exp_u = np.exp(u)
    s_exp = float(np.sum(exp_u))
    if s_exp < 1e-300:
        p = np.zeros(len(scores))
        cnt = int(np.sum(scores == max_s))
        p[scores == max_s] = 1.0 / cnt
    else:
        p = exp_u / s_exp
    lse_val = max_s + tau * np.log(s_exp) if s_exp > 0 else max_s
    G = _grad_matrix(h)
    grad = G.T @ p
    return lse_val, grad, p

def _thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        c = i
        p = 0
        while c > 0:
            p ^= 1
            c >>= 1
        tm[i] = float(p)
    return tm

            best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d2:
        h_pol = _project_sum(h_bin.copy(), len(h_bin) / 2.0)
        h_pol, c5_pol = _focused_lp(h_pol, min(d2, time.time() + budget_s * 0.04), radius=0.05, top_k=4, stall_limit=50)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (30% budget)
    d3 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    n_candidates = []
    for dn in range(-30, 31, 2):
        n_try = n_cur + dn
        if 80 <= n_try <= 250 and n_try % 2 == 0:
            n_candidates.append(n_try)
    # Sort by priority: closer to current n first, then prefer even numbers that give good dx
    n_candidates.sort(key=lambda x: abs(x - n_cur))
    tried = set()
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick LP
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _focused_lp(h_r, sub_dl, radius=0.08, top_k=4, max_top_k=10, stall_limit=40)
        # Quick binary
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=10)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
            if time.time() < d3 - 2:
                sub_dl3 = min(d3, time.time() + budget_s * 0.008)
                h_b, c5_b = _binary_sa(h_b, sub_dl3, T_init=0.002, max_iter=15000, seed=n_try)
                if c5_b < c5_r:
                    h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Intensive refinement (15% budget)
    d4 = t0 + budget_s * 0.90
    n_fin = len(best_h)
    # Focused LP with tight params
    h4 = _project_sum(best_h.copy(), n_fin / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.05)
    h4, c5_4 = _focused_lp(h4, sub_dl, radius=0.03, top_k=3, max_top_k=8, stall_limit=150)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Binary polish
    if time.time() < d4:
        h_b = _to_binary(best_h)
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        if time.time() < d4:
            sub_dl3 = min(d4, time.time() + budget_s * 0.03)
            h_b, c5_b = _binary_sa(h_b, sub_dl3, T_init=0.001, cool=0.9997, max_iter=30000, seed=999)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            if time.time() < d4:
                h_b, c5_b = _pair_sweep(h_b, min(d4, time.time() + budget_s * 0.01), max_rounds=15)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
    # LSE final
    if time.time() < d4:
        tau_list_fin = [0.5, 0.25, 0.1, 0.05, 0.02]
        h_lse2, c5_lse2 = _lse_anneal(best_h, tau_list_fin, min(d4, time.time() + budget_s * 0.04))
        if c5_lse2 < best_c5:
            best_c5 = c5_lse2
            best_h = h_lse2.copy()

    # Phase 5: Final (10% budget)
    d5 = deadline
    # Try a few more n values
    n_best = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > d5 - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = _focused_lp(h_r, sub_dl, radius=0.04, top_k=3, max_top_k=7, stall_limit=30)
        if time.time() < d5 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d5, time.time() + budget_s * 0.006)
            h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=10)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final LP
    h_fin = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_fin, c5_fin = _focused_lp(h_fin, d5, radius=0.02, top_k=3, max_top_k=6, stall_limit=60)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    # Final pair sweep
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, d5, max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
