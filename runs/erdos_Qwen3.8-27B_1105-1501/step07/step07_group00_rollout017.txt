import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):

def _full_grad_matrix(h):
    """Compute gradient rows for ALL lags."""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):

        if r >= 0:

            if m > 0:
                # h[i]*(1-h[i+r]) for i=0..m-1
                # d/dh[m]: (1-h[m+r]) if m in [0,m-1] + (-h[m-r]) if m in [r, n-1]
                G[k, 0:m] += (1.0 - h[r:n])  # (1-h[i+r]) for i=0..m-1
                G[k, r:n] += -h[0:m]          # -h[i] for i=0..m-1, position i+r
        else:

                G[k, 0:m] += -h[g:n]          # -h[i+g] for i=0..m-1, position i
    return G

def _full_lp_direction(h, scores, radius):

    n_lags = len(scores)
    G = _full_grad_matrix(h)
    # min tau

    #      sum(d) = 0
    #      bounds on d

    c[-1] = 1.0

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []

        bounds.append((lo, hi))

def _line_search(h, d, dx):

    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = 1.0  # will be limited by box constraints
    pos = d > 1e-15
    neg = d < -1e-15

    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Coarse grid
    grid_t = np.linspace(t_lo, t_hi, 80)
    # Vectorized: S0 + t*T - t^2*C for all t
    # Shape: (80, 2n-1)
    vals = S0[None, :] + grid_t[:, None] * T[None, :] - (grid_t * grid_t)[:, None] * C[None, :]
    grid_f = np.max(vals, axis=1) * dx
    
    best_idx = int(np.argmin(grid_f))
    # Golden section refinement
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(79, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f_t(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], float(grid_f[best_idx])
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = f_t(c1)
        f2 = f_t(c2)
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1

                f1 = f_t(c1)
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = f_t(c2)
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = f_t(opt_t)
    
    # Check endpoints and grid best
    cands = [(opt_t, opt_val),
             (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx),
             (grid_t[best_idx], float(grid_f[best_idx]))]
    best = min(cands, key=lambda x: x[1])

def _binary_hill_climb(h_bin, deadline, max_iter=100000):
    """Hill climbing on binary vectors with swap moves."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try multiple random swaps, pick the best
        n_trials = min(20, len(ones) * len(zeros))
        best_delta = 0
        best_oi, best_oz = -1, -1
        
        for _ in range(n_trials):
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            h_try = cur.copy()
            h_try[zi] = 1.0
            h_try[oj] = 0.0
            c_try = _exact_c5(h_try)
            delta = c_try - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_oi, best_oz = oj, zi
        
        if best_delta < 0 and best_oi >= 0:
            cur[best_oz] = 1.0
            cur[best_oi] = 0.0
            ones.remove(best_oi)
            zeros.remove(best_oz)
            ones.append(best_oz)
            zeros.append(best_oi)
            cur_c5 += best_delta
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(1.0, 10.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.5 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _generate_seeds(n, rng):
    """Generate diverse binary seeds."""
    seeds = []
    # Pattern 1: irrational rotation
    alpha = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    for phase in np.linspace(0, 1, 7, endpoint=False):
        x = (np.arange(n) * alpha + phase) % 1.0
        idx = np.argsort(x)[-n // 2:]
        b = np.zeros(n)
        b[idx] = 1.0
        seeds.append(b)
    
    # Pattern 2: block patterns
    for block_size in [max(2, n // 10), max(2, n // 8), max(2, n // 6)]:
        b = np.zeros(n)
        for start in range(0, n, 2 * block_size):
            b[start:start + block_size] = 1.0
        # Balance
        ones_count = int(np.sum(b > 0.5))
        if ones_count > n // 2:
            # Remove excess
            excess = ones_count - n // 2
            one_idx = np.where(b > 0.5)[0]
            to_remove = rng.choice(one_idx, excess, replace=False)
            b[to_remove] = 0.0
        elif ones_count < n // 2:
            deficit = n // 2 - ones_count
            zero_idx = np.where(b <= 0.5)[0]
            to_add = rng.choice(zero_idx, deficit, replace=False)
            b[to_add] = 1.0
        seeds.append(b)
    
    # Pattern 3: Fourier-based
    for freq in [3, 5, 7, 11]:
        x = np.arange(n, dtype=float)
        val = np.sin(2 * np.pi * freq * x / n) + 0.5 * np.sin(2 * np.pi * (freq + 1) * x / n)
        idx = np.argsort(val)[-n // 2:]
        b = np.zeros(n)
        b[idx] = 1.0
        seeds.append(b)
    
    return seeds
