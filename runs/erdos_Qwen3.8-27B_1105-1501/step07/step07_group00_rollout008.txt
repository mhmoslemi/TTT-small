def _binary_swap_delta_vec(h, r, s, n):

    delta = np.zeros(2*n - 1)

    # Case A: 0 <= r-k < n => k in [r-(n-1), r]

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x[mask] += res / cnt

def _bin_init_counts(b, n):
    """Compute all correlation counts for binary b."""
    return np.correlate(b.astype(float), 1.0 - b.astype(float), mode='full').astype(int)

def _bin_swap_delta(b, r, s, n):
    """Vectorized O(n) delta for swapping b[r]:1->0, b[s]:0->1."""
    k = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1, dtype=int)
    
    # r: 1 -> 0
    mask_a = (r - k >= 0) & (r - k < n)
    delta[mask_a] += -(1 - b[r - k[mask_a]])
    mask_b = (r + k >= 0) & (r + k < n)
    delta[mask_b] += b[r + k[mask_b]]
    
    # s: 0 -> 1
    mask_c = (s - k >= 0) & (s - k < n)
    delta[mask_c] += (1 - b[s - k[mask_c]])
    mask_d = (s + k >= 0) & (s + k < n)
    delta[mask_d] -= b[s + k[mask_d]]
    
    return delta

def _binary_sa(b_init, n, deadline, T_init=0.02, cool=0.9998, max_iters=500000):
    """Binary SA with exact incremental updates."""
    half = n // 2
    b = b_init.copy()
    ones = list(np.where(b > 0.5)[0])
    zeros = list(np.where(b <= 0.5)[0])
    counts = _bin_init_counts(b, n)
    cur_max = int(counts.max())
    best_b = b.copy()
    best_max = cur_max
    
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    iter_count = 0
    
    while time.time() < deadline and iter_count < max_iters:
        if stall > 5000:
            break
        iter_count += 1
        if iter_count % 2048 == 0 and time.time() > deadline:
            break
        
        oi = int(rng.integers(half))
        zi = int(rng.integers(half))
        r = ones[oi]
        s = zeros[zi]
        
        # Compute delta
        d = _bin_swap_delta(b, r, s, n)
        new_counts = counts + d
        new_max = int(new_counts.max())
        delta = new_max - cur_max
        
        if delta < 0 or (delta == 0 and rng.random() < 0.3) or rng.random() < np.exp(-delta * 5.0 / max(T, 1e-10)):
            # Accept

def _grad_rows(h, lags, n):
    """Compute gradient rows for given lags."""
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):

    n = len(h)
    u = 1.0 - h

        return opt_t, opt_val
