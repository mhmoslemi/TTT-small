def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G

def _soft_weights(R, beta):
    M = float(np.max(R))
    shift = -beta * (M - R)
    # Avoid overflow
    shift = shift - np.max(shift)
    w = np.exp(shift)
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(R) // 100)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _line_search(R0, A, B, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(R0))
    n_grid = 60
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None]) ** 2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2 * c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(R0 + opt_t * A - opt_t * opt_t * B))
    grid_val = float(vals[best_idx])
    lo_val = float(np.max(R0 + t_lo * A - t_lo * t_lo * B))
    hi_val = float(np.max(R0 + t_hi * A - t_hi * t_hi * B))
    best_val = min(opt_val, grid_val, lo_val, hi_val)
    if best_val == opt_val:
        return opt_t, best_val
    if best_val == grid_val:
        return float(ts[best_idx]), best_val
    if best_val == lo_val:
        return t_lo, best_val
    return t_hi, best_val

def _sweep_violations(h, d):
    """Check if h + t*d stays in [0,1] for t in [0, t_max]. Return t_max."""
    t_max = float('inf')
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):
        t_max = min(t_max, float(np.min(-h[neg] / d[neg])))
    if not np.isfinite(t_max):
        t_max = 1.0
    return t_max

    sinks = sort_idx[:min(n_pairs, n // 3)]     # low g → increase h
    
    best_c5 = cur_c5
    best_h = h.copy()
    one_mh = 1.0 - h
    
    for ii in range(min(n_pairs, len(sinks), len(sources))):
        sink = int(sinks[ii])
        source = int(sources[ii])
        if sink == source:
            continue
        t_max_fwd = min(1.0 - h[sink], h[source], 0.1)
        if t_max_fwd < 1e-12:
            continue
        d = np.zeros(n)
        d[sink] = 1.0
        d[source] = -1.0
        A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        B = np.correlate(d, d, mode='full')
        t_opt, _ = _line_search(R, A, B, 0.0, t_max_fwd)
        if t_opt < 1e-12:
            continue
        h_try = h.copy()
        h_try[sink] += t_opt
        h_try[source] -= t_opt
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5
