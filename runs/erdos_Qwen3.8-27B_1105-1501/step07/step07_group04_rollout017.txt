    # Select active constraints
    max_raw = float(corr.max())
    # Active: within 5% of max in raw units
    threshold = max_raw * (1.0 - active_fracs[1])
    active = np.where(corr >= threshold)[0]
    
    # Also always include the top min_active
    top_idx = np.argsort(-corr)[:min_active]
    active = np.unique(np.concatenate([active, top_idx]))
    
    na = len(active)
    
    # Build gradient rows for active constraints
    # For each active m, compute grad = _grad_lag(h, m - (n-1))
    # G[active_idx, i] = grad_i
    G = np.zeros((na, n))
    for idx, m in enumerate(active):
        k = m - (n - 1)
        G[idx] = _grad_lag(h, k) * dx  # scale by dx since objective is scaled
    
    # LP: minimize t + eps * sum(u)
    # Variables: d[0..n-1], t, u[0..n-1]
    # Total: 2n+1 variables
    nv = 2 * n + 1
    t_idx = n
    u_start = n + 1
    
    # Objective: minimize t + eps * sum(u_i)
    c_obj = np.zeros(nv)
    c_obj[t_idx] = 1.0
    eps = 1e-5
    c_obj[u_start:] = eps
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint m:
    #   corr[m]*dx + G[idx,:] @ d <= t
    #   => G[idx,:] @ d - t <= -corr[m]*dx
    # i.e., [G[idx,:], -1, 0...] @ [d, t, u] <= -corr[m]*dx
    A_ub = np.zeros((na, nv))
    b_ub = np.zeros(na)
    for idx in range(na):
        A_ub[idx, :n] = G[idx]
        A_ub[idx, t_idx] = -1.0
        b_ub[idx] = -corr[active[idx]] * dx
    
    # u_i - d_i <= 0 (u_i >= d_i)
    # -u_i - d_i <= 0 (u_i >= -d_i)
    A_box_u = np.zeros((2*n, nv))
    for i in range(n):
        A_box_u[i, i] = 1.0
        A_box_u[i, u_start + i] = -1.0  # d_i - u_i <= 0
        A_box_u[i+n, i] = -1.0
        A_box_u[i+n, u_start + i] = -1.0  # -d_i - u_i <= 0
    b_box_u = np.zeros(2*n)
    
    A_ub_full = np.vstack([A_ub, A_box_u])
    b_ub_full = np.concatenate([b_ub, b_box_u])
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(radius, h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((cur_max - 1.0, None))  # t (can be much less than current)
    for i in range(n):
        bounds.append((0.0, 2.0 * radius))  # u_i
