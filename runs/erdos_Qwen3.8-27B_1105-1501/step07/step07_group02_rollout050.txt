        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _binary_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = sorted(set(np.where(cur > 0.5)[0].tolist()))
    zeros = sorted(set(np.where(cur <= 0.5)[0].tolist()))
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    ones.append(oz)
                    ones.sort()
                    zeros.remove(oz)
                    zeros.append(oi)
                    zeros.sort()
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def _smooth_start(n, mode=0, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if mode == 0:
        h = np.where(x < 0.5, 1.0, 0.0)
    elif mode == 1:
        h = np.sin(np.pi * x) ** 2
        h = h / h.sum() * (n / 2.0)
    elif mode == 2:
        h = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 2 * np.pi))
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    else:
        h = np.where(rng.uniform(0, 1, n) < 0.5, 1.0, 0.0)
    return _project(h, n / 2.0)

def _triple_move(h, deadline, n_moves=30, radius=0.12, seed=55):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    w = _smooth_weights(S0, 20.0)
    G = _gradMatrix(h)
    coord_grad = G.T @ w
    stall = 0
    max_iter = 500
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        found = False
        for _ in range(n_moves):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            c = int(rng.integers(0, n))
            if len(set([a, b, c])) < 3:
                continue
            # Move: +t at c, -t/2 at a, -t/2 at b
            t_lo = max(-h[c], h[a]/2.0 - 1.0, h[b]/2.0 - 1.0, -radius)
            t_hi = min(1.0 - h[c], h[a]/2.0, h[b]/2.0, radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[a] = -0.5
            d[b] = -0.5
            d[c] = 1.0
            one_mh = 1.0 - h
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[a] -= 0.5 * t_opt
            h_try[b] -= 0.5 * t_opt
            h_try[c] += t_opt
            h_try = _project(h_try, n / 2.0)
            c_try = _exact_c5(h_try)
            if c_try < cur_c5 - 1e-12:
                h = h_try
                cur_c5 = c_try
                S0 = np.correlate(h, 1.0 - h, mode='full')
                found = True
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h.copy()
                break
        if not found:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.zeros(180)
        h0[::2] = 1.0
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: FW + pair on parent (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _fw_optimize(h_init, min(d1, t0 + budget_s * 0.15), beta=15.0, max_iter=4000, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    h1b, c5_1b = _pair_optimize(best_h, d1, radius=0.18, n_pairs=50, beta=20.0, stall_limit=100, seed=2)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b.copy()

    # Phase 2: Multi-start exploration (20% time)
    d2 = t0 + budget_s * 0.45
    starts = []
    starts.append(_smooth_start(n0, mode=0, seed=10))
    starts.append(_smooth_start(n0, mode=1, seed=11))
    starts.append(_smooth_start(n0, mode=2, seed=12))
    starts.append(_smooth_start(n0, mode=4, seed=13))
    per_start = max(5, (d2 - time.time()) / len(starts))
    for si, s in enumerate(starts):
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + per_start)
        h_s, c5_s = _fw_optimize(s, min(sub_dl, time.time() + per_start * 0.5), beta=12.0, max_iter=2000, seed=50 + si)
        h_s, c5_s = _pair_optimize(h_s, sub_dl, radius=0.15, n_pairs=40, beta=20.0, stall_limit=40, seed=60 + si)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Intensive pair + triple on best (25% time)
    d3 = t0 + budget_s * 0.70
    h3, c5_3 = _pair_optimize(best_h, min(d3, t0 + budget_s * 0.50), radius=0.15, n_pairs=60, beta=22.0, stall_limit=120, seed=100)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _triple_move(best_h, d3, n_moves=25, radius=0.10, seed=200)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-n exploration (15% time)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    n_list = sorted(set([n_curr + dn for dn in [-8, -4, 4, 8, -12, 12, -16, 16, -24, 24, 140, 160, 200]
                         if 64 <= (n_curr + dn if abs(dn) < 50 else dn) <= 240]))
    # Filter for even
    n_list = [x for x in n_list if x % 2 == 0]
    per_n = max(5, (d4 - time.time()) / max(1, len(n_list)))
    for ni, n_try in enumerate(n_list):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + per_n)
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _fw_optimize(h_r, min(sub_dl, time.time() + per_n * 0.5), beta=15.0, max_iter=1500, seed=300 + ni)
        h_r, c5_r = _pair_optimize(h_r, sub_dl, radius=0.12, n_pairs=40, beta=22.0, stall_limit=40, seed=400 + ni)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Binary optimization (8% time)
    d5 = t0 + budget_s * 0.93
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d5, time.time() + budget_s * 0.035)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=25000, T_init=0.003, seed=777)
        sub_dl2 = min(d5, time.time() + budget_s * 0.025)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish binary result with pair moves
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d5, time.time() + budget_s * 0.015)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl3, radius=0.08, n_pairs=40, beta=25.0, stall_limit=50, seed=555)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Final polish (7% time)
    d6 = deadline - 2
    n_curr = len(best_h)
    h6 = _project(best_h.copy(), n_curr / 2.0)
    h6, c5_6 = _pair_optimize(h6, d6, radius=0.06, n_pairs=60, beta=30.0, stall_limit=100, seed=888)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
