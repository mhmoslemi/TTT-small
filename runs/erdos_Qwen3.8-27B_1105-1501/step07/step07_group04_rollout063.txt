    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_all_grads(h, n):
    m = 2 * n - 1
    center = n - 1
    G = np.zeros((m, n))
    G[center] = 1.0 - 2.0 * h
    for k in range(1, n):
        G[center + k, k:] = 1.0 - h[:n - k]
        G[center + k, :n - k] -= h[k:n]
        G[center - k, :n - k] = 1.0 - h[k:n]
        G[center - k, k:] -= h[:n - k]
    return G

def _lp_direction(h, corr_raw, radius, max_rows=None):
    n = len(h)
    m = 2 * n - 1
    center = n - 1
    if max_rows is not None and max_rows < m:
        idx = np.argsort(-corr_raw)[:max_rows]
    else:
        idx = np.arange(m)
    n_rows = len(idx)
    G = np.zeros((n_rows, n))
    for ii, r in enumerate(idx):
        k = int(r - center)
        if k == 0:
            G[ii] = 1.0 - 2.0 * h
        elif k > 0:
            G[ii, k:] = 1.0 - h[:n - k]
            G[ii, :n - k] -= h[k:n]
        else:
            u = -k
            G[ii, :n - u] = 1.0 - h[u:n]
            G[ii, u:] -= h[:n - u]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([G, -np.ones((n_rows, 1))])
    b_ub = -float(corr_raw[idx])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(-1e6, 1e6)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    if result.status != 0 or result.x is None:
        return None
    d = result.x[:n].copy()
    t_new = float(result.x[n])
    cur_max = float(np.max(corr_raw))
    if t_new >= cur_max - 1e-9:
        return None
    d = d - np.mean(d)
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d

def _subgradient_dir(h, corr_raw, radius):
    n = len(h)
    max_val = float(corr_raw.max())
    active = np.where(corr_raw >= max_val - 1e-8)[0]
    G = _build_all_grads(h, n)
    g = G[active].mean(axis=0)
    d = -g
    d = d - np.mean(d)
    d = np.clip(d, -radius, radius)
    d = d - np.mean(d)
    return d

def _line_search(h, d, t_max, c5_current):
    n = len(h)
    dx = 2.0 / n
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    n_grid = 40
    if t_max < 1e-12:
        return 0.0, c5_current
    grid_t = np.linspace(0.0, t_max, n_grid)
    best_val = c5_current
    best_t = 0.0
    for idx in range(n_grid):
        t = grid_t[idx]
        vals = C0 + t * L + t * t * Q
        cv = float(np.max(vals)) * dx
        if cv < best_val:
            best_val = cv
            best_t = t
    lo_b = grid_t[max(0, int(np.argmin([float(np.max(C0 + grid_t[i]*L + grid_t[i]**2*Q)) for i in range(n_grid)])) - 2)]
    hi_b = grid_t[min(n_grid - 1, int(np.argmin([float(np.max(C0 + grid_t[i]*L + grid_t[i]**2*Q)) for i in range(n_grid)])) + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(C0 + c1 * L + c1 * c1 * Q)) * dx
    f2 = float(np.max(C0 + c2 * L + c2 * c2 * Q)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-10:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(C0 + c1 * L + c1 * c1 * Q)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(C0 + c2 * L + c2 * c2 * Q)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(C0 + opt_t * L + opt_t * opt_t * Q)) * dx
    candidates = [(0.0, c5_current), (opt_t, opt_val), (t_max, float(np.max(C0 + t_max * L + t_max * t_max * Q)) * dx)]
    best_c = min(candidates, key=lambda x: x[1])
    return best_c[0], best_c[1]

def _lp_loop(h, deadline, radius_init=0.10, max_stall=80, use_lp=True):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > max_stall:
            break
        if radius < 1e-7:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(np.max(corr_raw))
        if use_lp:
            d = _lp_direction(h, corr_raw, radius)
            if d is None:
                d = _subgradient_dir(h, corr_raw, radius)
        else:
            d = _subgradient_dir(h, corr_raw, radius)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            stall += 1
            radius *= 0.8
            continue
        pos = d > 1e-14
        neg = d < -1e-14
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        t_hi = max(t_hi, 0.0)
        alpha, new_c5 = _line_search(h, d, t_hi, cur)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_search(h_bin, deadline, max_iters=100000, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = 0.004
    stall = 0
    best_stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 4000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5 = _exact_c5(trial)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9998)
        else:
            stall += 1
            best_stall += 1
    return best_h, best_c5

def _targeted_binary(h_bin, deadline, max_iters=50000, seed=99):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_idx = int(np.argmax(corr))
        k = abs(max_idx - (n - 1))
        best_delta = 0.0
        best_move = None
        n_try = min(150, len(ones) * len(zeros))
        for _t in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            if k > 0 and abs(oi - oz) == k:
                trial = cur.copy()
                trial[oi] = 0.0
                trial[oz] = 1.0
                nc5 = _exact_c5(trial)
                if nc5 - cur_c5 < best_delta:
                    best_delta = nc5 - cur_c5
                    best_move = (oi, oz, i, j)
        if best_move is None:
            for _t in range(n_try // 3):
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                oi = ones[i]
                oz = zeros[j]
                trial = cur.copy()
                trial[oi] = 0.0
                trial[oz] = 1.0
                nc5 = _exact_c5(trial)
                if nc5 - cur_c5 < best_delta:
                    best_delta = nc5 - cur_c5
                    best_move = (oi, oz, i, j)
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    perturb = np.zeros(n)
    for f, p in zip(freqs, phases):
        perturb += np.sin(2.0 * np.pi * f * x / n + p)
    perturb = perturb / max(1.0, float(np.max(np.abs(perturb))))
    perturb *= amp
    perturb += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization from start (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.10, max_stall=60, use_lp=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Basin hopping with LP (25%)
    d2 = t0 + budget_s * 0.50
    amp = 0.03
    for it in range(200):
        if time.time() > d2:
            break
        if time.time() > d2 - 5:
            break
        h_pert = _perturb(best_h, amp, rng)
        sub_dl = min(d2 - 2, time.time() + budget_s * 0.02)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08, max_stall=30, use_lp=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            amp = max(0.005, amp * 0.85)
        else:
            amp = min(0.08, amp * 1.1)

    # Phase 3: Binary search (20%)
    d3 = t0 + budget_s * 0.70
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=50000, seed=101)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sub_dl = min(d3, time.time() + budget_s * 0.07)
    h_bin, c5_bin = _binary_search(h_bin, sub_dl, max_iters=80000, seed=202)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=40000, seed=303)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.90
    n_best = len(best_h)
    n_candidates = []
    for dn in range(-20, 21, 2):
        nt = n0 + dn
        if 80 <= nt <= 260 and nt % 2 == 0:
            n_candidates.append(nt)
    for nt in [100, 120, 130, 140, 160, 200, 220, 240]:
        if nt not in n_candidates:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))
    for nt in n_candidates:
        if time.time() > d4 - 5:
            break
        h_r = _resample(best_h, nt)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.08, max_stall=20, use_lp=True)
        if c5_r < best_c5 + 0.003 and time.time() < d4 - 10:
            h_b = _to_binary(h_r, nt)
            sub_dl2 = min(d4 - 5, time.time() + budget_s * 0.008)
            h_b, c5_b = _targeted_binary(h_b, sub_dl2, max_iters=20000, seed=1000 + nt)
            sub_dl3 = min(d4 - 3, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_search(h_b, sub_dl3, max_iters=30000, seed=2000 + nt)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = nt

    # Phase 5: Final refinement (remaining)
    d5 = deadline
    n_fin = len(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h5 = _project(best_h.copy(), n_fin / 2.0)
    h5, c5_5 = _lp_loop(h5, sub_dl, radius_init=0.05, max_stall=50, use_lp=True)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < d5 - 15:
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.04)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, max_iters=50000, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d5 - 10:
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_search(h_b, sub_dl, max_iters=60000, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Try nearby n values
    for dn in [-4, -2, 2, 4]:
        if time.time() > d5 - 3:
            break
        nt = n_fin + dn
        if nt < 60 or nt % 2 != 0:
            continue
        h_r = _resample(best_h, nt)
        sub_dl = min(d5 - 1, time.time() + 4)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.04, max_stall=15, use_lp=True)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
