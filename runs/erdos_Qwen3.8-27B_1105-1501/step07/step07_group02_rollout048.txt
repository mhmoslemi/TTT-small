def _project(v, target):

    n = len(v)

        return np.zeros(n)

            hi = mid
    theta = (lo + hi) / 2.0

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Decay step
        step *= 0.5

def _fw_line_search(h, b, beta, dx, max_iter=30):
    """Golden section search for optimal t in [0,1] for h_new = (1-t)*h + t*b."""
    n = len(h)
    def _f(t):
        h_t = (1.0 - t) * h + t * b
        val, _ = _lse_val_grad(h_t, beta, dx)
        return val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = _f(c), _f(d)
    for _ in range(max_iter):

        if fc < fd:

            c = hi - gr * (hi - lo)
            fc = _f(c)

            d = lo + gr * (hi - lo)
            fd = _f(d)

    if f_lo < f_mid and f_lo < f_hi:
        t_opt = 0.0
    elif f_hi < f_mid and f_hi < f_lo:
        t_opt = 1.0

                best_h = h.copy()

def _slsqp_lse(h0, beta, deadline, maxiter=200):
    from scipy.optimize import minimize
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h0.copy(), target)
    x0 = h.copy()
    
    def obj(x):
        return _lse_val_grad(x, beta, dx)[0]
    
    def obj_grad(x):
        return _lse_val_grad(x, beta, dx)[1]
    
    def eq_fun(x):
        return x.sum() - target
    
    def eq_grad(x):
        return np.ones(n)
    
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}]
    
    res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    
    if res.x is not None:
        h_res = _project(res.x, target)
        c5 = _exact_c5(h_res)
        return h_res, c5
    return h, _exact_c5(h)
