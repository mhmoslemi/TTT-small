        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_x = best_local.copy()
        x0 = best_local.copy()
    
    return _anti_make(best_x, m), best_c5

def _grad_full_smooth(h, beta):
    n = len(h)
    raw = _raw_corr(h)
    M = float(np.max(raw))
    z = np.clip(beta * (raw - M), -50.0, 0.0)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws < 1e-30:
        idx = int(np.argmax(raw))
        return _grad_full_single(h, idx)
    w /= ws
    G = np.zeros(n)
    for r in range(2*n - 1):
        if w[r] < 1e-10: continue
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m_t = n - tau
            G[tau:tau+m_t] += wr * (1.0 - h[:m_t])
            G[:m_t] -= wr * h[tau:tau+m_t]
        else:
            d = -tau
            m_t = n - d
            G[:m_t] += wr * (1.0 - h[d:d+m_t])
            G[d:d+m_t] -= wr * h[:m_t]
    G -= float(np.mean(G))
    return G

def _grad_full_single(h, r):
    n = len(h)
    G = np.zeros(n)
    tau = r - (n-1)
    if tau >= 0:
        m_t = n - tau
        if m_t > 0:
            G[tau:tau+m_t] += (1.0 - h[:m_t])
            G[:m_t] -= h[tau:tau+m_t]
    else:
        d = -tau
        m_t = n - d
        if m_t > 0:
            G[:m_t] += (1.0 - h[d:d+m_t])
            G[d:d+m_t] -= h[:m_t]
    G -= float(np.mean(G))
    return G

def _line_search_exact(h, d, radius=0.2):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    def f(a):
        h_new = h + a * d
        return _exact_c5(h_new)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_a = (lo + hi) / 2.0
    opt_v = f(opt_a)
    # Check endpoints and 0
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_a, opt_v = t, ft
    return opt_a, opt_v

def _smooth_perturb(h, amp, rng):
    n = len(h)
    pert = rng.normal(0, 1, n) * amp
    for _ in range(2):
        freq = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0, 2*np.pi)
        pert += amp * 0.5 * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
    return _project(h + pert, n/2.0)

def _binary_sa_anti(h_init, deadline, max_iter=80000, T_init=0.003):
    """Binary SA respecting anti-palindromic structure."""
    n = len(h_init)
    m = n // 2
    # Convert to binary anti-palindromic
    x = _anti_extract(h_init, m)
    x_bin = (x >= 0.5).astype(float)
    cur = _anti_make(x_bin, m)
    cur_c5 = _exact_c5(cur)
    best_x = x_bin.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(888)
    T = T_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = int(rng.integers(0, m))
        x_bin[i] = 1.0 - x_bin[i]
        new_h = _anti_make(x_bin, m)
        new_c5 = _exact_c5(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_x = x_bin.copy()
                stall = 0
            T = max(T * 0.9997, 0.0001)
        else:
            x_bin[i] = 1.0 - x_bin[i]
            stall += 1
    
    return _anti_make(best_x, m), best_c5

def _pair_sweep_anti(h_init, deadline):
    """Pair swap in anti-palindromic binary space."""
    n = len(h_init)
    m = n // 2
    x = _anti_extract(h_init, m)
    x_bin = (x >= 0.5).astype(float)
    cur = _anti_make(x_bin, m)
    cur_c5 = _exact_c5(cur)
    best_x = x_bin.copy()
    best_c5 = cur_c5
    
    ones = np.where(x_bin > 0.5)[0]
    zeros = np.where(x_bin <= 0.5)[0]
    
    # Try flipping individual bits (anti-palindromic move)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(m):
            if time.time() > deadline:
                break
            x_bin[i] = 1.0 - x_bin[i]
            new_c5 = _exact_c5(_anti_make(x_bin, m))
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = x_bin.copy()
                improved = True
            else:
                x_bin[i] = 1.0 - x_bin[i]
    
    # Try pair swaps
    ones = np.where(x_bin > 0.5)[0]
    zeros = np.where(x_bin <= 0.5)[0]
    for oi in ones:
        if time.time() > deadline:
            break
        for oz in zeros:
            x_bin[oi] = 0.0
            x_bin[oz] = 1.0
            new_c5 = _exact_c5(_anti_make(x_bin, m))
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_x = x_bin.copy()
                ones = np.where(x_bin > 0.5)[0]
                zeros = np.where(x_bin <= 0.5)[0]
                break
            else:
                x_bin[oi] = 1.0
                x_bin[oz] = 0.0
    
    return _anti_make(best_x, m), best_c5

def _full_smooth_descent(h_init, deadline, beta_lo=5.0, beta_hi=60.0, max_stall=80):
    """Full-space smooth-max gradient descent."""
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    stall = 0
    radius = 0.15
    total = 0
    
    while time.time() < deadline and stall < max_stall:
        total += 1
        frac = min(1.0, total / 100.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        G = _grad_full_smooth(h, beta)
        gn = np.linalg.norm(G)
        if gn < 1e-12:
            break
        d = G / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        d /= dn
        
        alpha, val = _line_search_exact(h, d, radius)
        if val < cur_c5 - 1e-14 and alpha != 0.0:
            h_new = np.clip(h + alpha * d, 0, 1)
            h_new = _project(h_new, n/2.0)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.4)
            else:
                stall += 1
                radius = max(0.03, radius * 0.8)
        else:
            # Try a random direction
            d_rand = rng.normal(0, 1, n)
            d_rand -= np.mean(d_rand)
            d_rand /= np.linalg.norm(d_rand)
            alpha, val = _line_search_exact(h, d_rand, radius * 0.5)
            if val < cur_c5 - 1e-14 and alpha != 0.0:
                h_new = np.clip(h + alpha * d_rand, 0, 1)
                h_new = _project(h_new, n/2.0)
                nc = _exact_c5(h_new)
                if nc < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = nc
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                    radius = max(0.03, radius * 0.8)
            else:
                stall += 1
                radius = max(0.03, radius * 0.8)
    
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _anti_make(x, m):
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[m - 1::-1]
    return h

def _anti_extract(h, m):
    n = 2 * m
    x = 0.5 * h[:m] + 0.5 * (1.0 - h[n - 1:m - 1:-1])
    return np.clip(x, 0.0, 1.0)

def _anti_obj_grad(x, m, beta):
    h = _anti_make(x, m)
    n = 2 * m
    raw = _raw_corr(h)
    M = float(np.max(raw))
    z = np.clip(beta * (raw - M), -50.0, 0.0)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws < 1e-30:
        return M / n, np.zeros(m)
    w /= ws
    S = (M + np.log(ws) / beta) / n
    G = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m_t = n - tau
            G[tau:tau + m_t] += wr * (1.0 - h[:m_t])
            G[:m_t] -= wr * h[tau:tau + m_t]
        else:
            d = -tau
            m_t = n - d
            G[:m_t] += wr * (1.0 - h[d:d + m_t])
            G[d:d + m_t] -= wr * h[:m_t]
    dx = (G[:m] - G[n - 1:m - 1:-1]) / n
    return S, dx

class _AntiOpt:
    def __init__(self, m, beta):
        self.m = m
        self.beta = beta
    def value(self, x):
        return _anti_obj_grad(x, self.m, self.beta)[0]
    def gradient(self, x):
        return _anti_obj_grad(x, self.m, self.beta)[1]

def _grad_full_smooth(h, beta):
    n = len(h)
    raw = _raw_corr(h)
    M = float(np.max(raw))
    z = np.clip(beta * (raw - M), -50.0, 0.0)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws < 1e-30:
        idx = int(np.argmax(raw))
        tau = idx - (n - 1)
        G = np.zeros(n)
        if tau >= 0:
            m_t = n - tau
            if m_t > 0:
                G[tau:tau + m_t] += (1.0 - h[:m_t])
                G[:m_t] -= h[tau:tau + m_t]
        else:
            d = -tau
            m_t = n - d
            if m_t > 0:
                G[:m_t] += (1.0 - h[d:d + m_t])
                G[d:d + m_t] -= h[:m_t]
        G -= float(np.mean(G))
        return G
    w /= ws
    G = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m_t = n - tau
            G[tau:tau + m_t] += wr * (1.0 - h[:m_t])
            G[:m_t] -= wr * h[tau:tau + m_t]
        else:
            d = -tau
            m_t = n - d
            G[:m_t] += wr * (1.0 - h[d:d + m_t])
            G[d:d + m_t] -= wr * h[:m_t]
    G -= float(np.mean(G))
    return G

class _FullOpt:
    def __init__(self, beta):
        self.beta = beta
    def value(self, h):
        return _exact_c5(h)
    def gradient(self, h):
        return _grad_full_smooth(h, self.beta)

def _line_search_exact(h, d, radius=0.2):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)

    def f(a):
        return _exact_c5(h + a * d)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_a = (lo + hi) / 2.0
    opt_v = f(opt_a)
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_a, opt_v = t, ft
    return opt_a, opt_v

def _lbfgs_anti(h_init, deadline, betas=None):
    n = len(h_init)
    m = n // 2
    if betas is None:
        betas = [8.0, 15.0, 30.0, 60.0, 100.0]
    x0 = _anti_extract(h_init, m).copy()
    best_x = x0.copy()
    best_c5 = _exact_c5(_anti_make(x0, m))
    for beta in betas:
        if time.time() > deadline:
            break
        opt = _AntiOpt(m, beta)
        for attempt in range(4):
            if time.time() > deadline:
                break
            res = minimize(opt.value, x0, jac=opt.gradient, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * m,
                          options={'maxiter': 400, 'gtol': 1e-10, 'iprint': 0})
            c5_try = _exact_c5(_anti_make(res.x, m))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_x = res.x.copy()
            if attempt < 3:
                rng_a = np.random.default_rng(hash((beta, attempt)) % 100000)
                x0 = np.clip(res.x + 0.02 * rng_a.normal(0, 1, m), 0, 1)
            else:
                x0 = res.x.copy()
    return _anti_make(best_x, m), best_c5

def _full_smooth_descent(h_init, deadline, beta_lo=10.0, beta_hi=80.0, max_stall=70):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    stall = 0
    radius = 0.18
    total = 0
    while time.time() < deadline and stall < max_stall:
        total += 1
        frac = min(1.0, total / 80.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        G = _grad_full_smooth(h, beta)
        gn = np.linalg.norm(G)
        if gn < 1e-12:
            break
        d = G / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        d /= dn
        alpha, val = _line_search_exact(h, d, radius)
        accepted = False
        if val < cur_c5 - 1e-14 and alpha != 0.0:
            h_new = np.clip(h + alpha * d, 0, 1)
            h_new = _project(h_new, n / 2.0)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.06, 0.45)
                accepted = True
        if not accepted:
            if stall % 5 == 4:
                d_r = rng.normal(0, 1, n)
                d_r -= np.mean(d_r)
                d_r /= np.linalg.norm(d_r)
                alpha2, val2 = _line_search_exact(h, d_r, radius * 0.4)
                if val2 < cur_c5 - 1e-14 and alpha2 != 0.0:
                    h_new = np.clip(h + alpha2 * d_r, 0, 1)
                    h_new = _project(h_new, n / 2.0)
                    nc2 = _exact_c5(h_new)
                    if nc2 < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = nc2
                        if nc2 < best_c5:
                            best_c5 = nc2
                            best_h = h.copy()
                        stall = 0
                        continue
            stall += 1
            radius = max(0.03, radius * 0.8)
    return best_h, best_c5

def _binary_sa_anti(h_init, deadline, max_iter=80000, T_init=0.004):
    n = len(h_init)
    m = n // 2
    x = _anti_extract(h_init, m)
    x_bin = (x >= 0.5).astype(float)
    cur = _anti_make(x_bin, m)
    cur_c5 = _exact_c5(cur)
    best_x = x_bin.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        i = int(rng.integers(0, m))
        x_bin[i] = 1.0 - x_bin[i]
        new_c5 = _exact_c5(_anti_make(x_bin, m))
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_x = x_bin.copy()
                stall = 0
            T = max(T * 0.9996, 0.0001)
        else:
            x_bin[i] = 1.0 - x_bin[i]
            stall += 1
    return _anti_make(best_x, m), best_c5

def _pair_sweep_anti(h_init, deadline):
    n = len(h_init)
    m = n // 2
    x = _anti_extract(h_init, m)
    x_bin = (x >= 0.5).astype(float)
    cur_c5 = _exact_c5(_anti_make(x_bin, m))
    best_x = x_bin.copy()
    best_c5 = cur_c5
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(m):
            if time.time() > deadline:
                break
            x_bin[i] = 1.0 - x_bin[i]
            nc = _exact_c5(_anti_make(x_bin, m))
            if nc < cur_c5 - 1e-14:
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_x = x_bin.copy()
                improved = True
            else:
                x_bin[i] = 1.0 - x_bin[i]
    ones = np.where(x_bin > 0.5)[0]
    zeros = np.where(x_bin <= 0.5)[0]
    for oi in ones:
        if time.time() > deadline:
            break
        for oz in zeros:
            x_bin[oi] = 0.0
            x_bin[oz] = 1.0
            nc = _exact_c5(_anti_make(x_bin, m))
            if nc < cur_c5 - 1e-14:
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_x = x_bin.copy()
                ones = np.where(x_bin > 0.5)[0]
                zeros = np.where(x_bin <= 0.5)[0]
                break
            else:
                x_bin[oi] = 1.0
                x_bin[oz] = 0.0
    return _anti_make(best_x, m), best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    pert = rng.normal(0, 1, n) * amp
    for _ in range(2):
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=3000, radius=0.15, seed=333):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search_exact(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = np.clip(h + alpha * d, 0, 1)
            h_new = _project(h_new, n / 2.0)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5
