def _exact_c5(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        else:

        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _lp_direction(h, raw, radius, frac=0.95, max_kags=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    mx = float(np.max(raw))
    # Select lags: those with raw > frac * mx
    thresh = frac * mx
    active = np.where(raw > thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(raw))])
    if len(active) > max_kags:
        idx = np.argsort(-raw[active])[:max_kags]
        active = active[idx]
    
    # Build coefficient matrix
    # For each active lag r, the linearized constraint is:
    # sum_i coeff[r, i] * x[i] <= t - raw[r]
    # which is: sum_i coeff[r, i] * x[i] + raw[r] <= t
    
    K = len(active)
    # Build coeff matrix
    A = np.zeros((K, n))
    for ki, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            # x[j+d] * (1-h[j]) - h[j+d] * x[j] for j in [0, n-1-d]
            # For variable i:
            #   from first term: (1-h[i-d]) if d <= i <= n-1
            #   from second term: -h[i+d] if 0 <= i <= n-1-d
            for i in range(n):
                c = 0.0
                if d <= i <= n - 1:
                    c += (1.0 - h[i - d])
                if 0 <= i <= n - 1 - d:
                    c -= h[i + d]
                A[ki, i] = c
        else:
            # d < 0, let dd = -d
            dd = -d
            # x[j+d] * (1-h[j]) - h[j+d] * x[j] for j in [dd, n-1]
            # For variable i:
            #   from x[j+d] term (i = j+d, j = i-d = i+dd): (1-h[j]) = (1-h[i+dd]) if 0 <= i <= n-1-dd
            #   from (1-h[j]) term (j = i): -h[j+d] = -h[i-d] = -h[i-dd] if dd <= i <= n-1
            for i in range(n):
                c = 0.0
                if 0 <= i <= n - 1 - dd:
                    c += (1.0 - h[i + dd])
                if dd <= i <= n - 1:
                    c -= h[i - dd]
                A[ki, i] = c
    
    # LP: min t
    # s.t. A @ x + raw[active] <= t  (K constraints)
    #      sum(x) = target
    #      0 <= x <= 1
    #      |x - h| <= radius
    
    x_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(x_var) == target,
        x_var >= 0,
        x_var <= 1,
        x_var >= h - radius,
        x_var <= h + radius,
    ]
    
    for ki in range(K):
        constraint = A[ki] @ x_var + raw[active[ki]] <= t_var
        constraints.append(constraint)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    # Try different solvers
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                return None
    
    if x_var.value is None:
        return None
    
    x_lp = np.array(x_var.value).flatten()
    # Clean up
    x_lp = np.clip(x_lp, 0.0, 1.0)
    x_lp = _project(x_lp, target)
    
    d = x_lp - h
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-10:
        return None
    
    return d
