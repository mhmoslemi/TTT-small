def _push_relabel(cap, s, t):
    n = cap.shape[0]
    res = cap.astype(np.float64).copy()
    excess = np.zeros(n)

        if v == s or v == t:
            continue

            res[v, s] += f

    inQ = np.zeros(n, dtype=bool)

        for v in range(n):
            if res[u, v] > 0:
                min_h = min(min_h, height[v])
