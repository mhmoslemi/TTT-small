        return np.zeros(n)

        return np.ones(n)

        else:

        # row = mid - d
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:]
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project_sum(v, target):

    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(v.min()) - 1.0

        if s > target:

        else:
            hi = mid

def _grad_matrix(h):
    n = len(h)

        G[mid + d, :m] += (1.0 - h[d:d+m])
        G[mid + d, d:] += -h[:m]
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += -h[d:]
    return G
