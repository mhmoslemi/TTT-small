Let me design a cleaner, more focused implementation:

1. **Core mechanism**: At each iteration, select K active (top) correlation rows, compute their gradients, and solve a least-norm problem to find a zero-sum direction d that reduces those rows. Then do an exact line search along d.

2. **Gradient computation**: For correlation row r (lag ell = r - (n-1)):
   R_r = sum_j h[j+ell]*(1-h[j]) for valid j
   gradient: dR_r/dh[p] = (1-h[p-ell]) if valid - h[p+ell] if valid

3. **Least-norm solution**: 
   - Build A matrix: K rows (gradients of selected correlation rows) + 1 row of all-ones (sum constraint)
   - Want: minimize ||d||² (or weighted) subject to A_d = b
   - Solution: d = A^T (A A^T)^{-1} b
   - With ridge regularization for stability

4. **Target selection**: b_i = target - R_ri where target is some value below the current max

5. **Line search**: Given direction d, compute U and B vectors, find t that minimizes max(S + t*U - t²*B) subject to h + t*d ∈ [0,1]

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_gradient(h, r):
    """Compute gradient of correlation row r with respect to h."""
    n = len(h)
    ell = r - (n - 1)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        m = n - ell
        for p in range(m):
            g[p + ell] += (1.0 - h[p])
            g[p] -= h[p + ell]
    else:
        p = -ell
        m = n - p
        for q in range(m):
            g[q + p] += (1.0 - h[q])
            g[q] -= h[q + p]
    return g

def _row_gradient_fast(h, r):
    """Vectorized version of row gradient."""
    n = len(h)
    ell = r - (n - 1)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        m = n - ell
        g[ell:] += (1.0 - h[:m])
        g[:m] -= h[ell:]
    else:
        p = -ell
        m = n - p
        g[p:] += (1.0 - h[:m])
        g[:m] -= h[p:]
    return g

def _line_search_dense(h, S, d, move_cap=0.5):
    """Exact line search along direction d. Returns (best_t, best_val)."""
    n = len(h)
    one_mh = 1.0 - h
    U = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -move_cap
    t_hi = move_cap
    nz = np.abs(d) > 1e-14
    if np.any(nz):
        for i in np.where(nz)[0]:
            if d[i] > 0:
                t_lo = max(t_lo, -h[i] / d[i] - 1e-12)
                t_hi = min(t_hi, (1.0 - h[i]) / d[i] - 1e-12)
            else:
                t_lo = max(t_lo, (1.0 - h[i]) / d[i] + 1e-12)
                t_hi = min(t_hi, -h[i] / d[i] + 1e-12)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S))
    
    # Grid search
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Predicted values
    vals = np.array([float(np.max(S + t * U - t * t * B)) for t in ts])
    best_idx = int(np.argmin(vals))
    
    # Golden section refinement
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S + opt_t * U - opt_t * opt_t * B))
    
    # Check endpoints and grid best too
    candidates = [(opt_t, opt_val), (t_lo, float(np.max(S + t_lo*U - t_lo*t_lo*B))),
                  (t_hi, float(np.max(S + t_hi*U - t_hi*t_hi*B))),
                  (float(ts[best_idx]), float(vals[best_idx])), (0.0, float(np.max(S)))]
    best_t, best_v = min(candidates, key=lambda x: x[1])
    return best_t, best_v

def _solve_dense_direction(h, S, K, target, ridge=1e-6):
    """Solve for dense direction reducing top K active rows."""
    n = len(h)
    L = 2 * n - 1
    # Select top K rows
    idx = np.argsort(S)[-K:]
    idx = np.sort(idx)
    
    # Build A matrix (K+1) x n: K gradient rows + all-ones
    A = np.zeros((K + 1, n))
    for i, r in enumerate(idx):
        A[i] = _row_gradient_fast(h, r)
    A[K] = 1.0  # sum constraint
    
    # Target: reduce rows toward 'target' value
    b = np.zeros(K + 1)
    for i, r in enumerate(idx):
        b[i] = max(0.0, target - S[r])  # positive if row is above target
    b[K] = 0.0  # sum = 0
    
    # Least norm: d = A^T (A A^T + ridge*I)^{-1} b
    # But we want to REDUCE the rows, so direction should be -gradient direction
    # If S[r] > target, we want d to decrease S[r], i.e., d = -grad * (S[r] - target)
    # So b should be (S[r] - target) and we negate
    
    for i, r in enumerate(idx):
        b[i] = max(0.0, S[r] - target)
    
    AA = A @ A.T + ridge * np.eye(K + 1)
    try:
        lam = np.linalg.solve(AA, b)
    except np.linalg.LinAlgError:
        lam = np.linalg.lstsq(AA, b, rcond=None)[0]
    
    d = A.T @ lam
    
    # Ensure zero sum (the constraint row should enforce this, but numerically)
    d -= np.mean(d)
    
    # Check if direction actually reduces the target rows
    # (validate by dot product)
    improvement = 0
    for i, r in enumerate(idx):
        if b[i] > 0:
            improvement -= A[i] @ d  # should be negative (reducing)
    
    if not np.all(np.isfinite(d)):
        return None
    
    dmax = np.max(np.abs(d))
    if dmax < 1e-14:
        return None
    
    return d / dmax  # normalize to unit max-abs

def _dense_refine(h, deadline, K_list=None, target_frac=0.95, move_cap_init=0.4,
                  stall_limit=80, rng=None):
    """Dense direction refinement loop."""
    if K_list is None:
        K_list = [3, 5, 8, 10, 6, 12, 8, 5]
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(h)
    h = _project(h.copy(), n / 2.0)
    S = np.correlate(h, 1.0 - h, mode='full')
    cur_score = float(np.max(S))
    best_h = h.copy()
    best_score = cur_score
    move_cap = move_cap_init
    stall = 0
    k_idx = 0
    n_iter = 0
    ridge = 1e-6
    
    while time.time() < deadline:
        if stall > stall_limit or move_cap < 1e-5:
            break
        n_iter += 1
        
        K = K_list[k_idx % len(K_list)]
        k_idx += 1
        target = target_frac * cur_score + 0.02 * float(np.min(S))
        target = min(target, cur_score - 1e-6)
        
        # Try dense direction
        d = _solve_dense_direction(h, S, K, target, ridge)
        found_improvement = False
        
        if d is not None:
            t_opt, val_opt = _line_search_dense(h, S, d, move_cap)
            if abs(t_opt) > 1e-12 and val_opt < cur_score - 1e-12:
                h_try = h + t_opt * d
                h_try = _project(h_try, n / 2.0)
                new_S = np.correlate(h_try, 1.0 - h_try, mode='full')
                new_score = float(np.max(new_S))
                if new_score < best_score:
                    best_h = h_try.copy()
                    best_score = new_score
                if new_score < cur_score - 1e-12:
                    h = h_try
                    S = new_S
                    cur_score = new_score
                    found_improvement = True
        
        # If dense didn't work, try pairwise
        if not found_improvement:
            h_try, c5_try = _pairwise_move(h, S, radius=move_cap * 0.5, n_pairs=20,
                                           beta=20.0, rng=rng)
            if c5_try < cur_score - 1e-12:
                h = h_try
                S = np.correlate(h, 1.0 - h, mode='full')
                cur_score = float(np.max(S))
                if cur_score < best_score:
                    best_h = h.copy()
                    best_score = cur_score
                found_improvement = True
        
        if found_improvement:
            stall = 0
            move_cap = min(move_cap * 1.01, 0.5)
            ridge *= 0.99
        else:
            stall += 1
            move_cap *= 0.85
            ridge *= 1.1
            # Occasionally try random zero-sum move
            if stall % 20 == 0:
                d_rand = rng.standard_normal(n)
                d_rand -= np.mean(d_rand)
                dmax = np.max(np.abs(d_rand))
                if dmax > 1e-14:
                    d_rand = d_rand / dmax * (move_cap * 0.3)
                    t_opt, val_opt = _line_search_dense(h, S, d_rand, move_cap * 0.3)
                    if abs(t_opt) > 1e-12:
                        h_try = h + t_opt * d_rand
                        h_try = _project(h_try, n / 2.0)
                        new_score = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))

                            h = h_try
                            S = np.correlate(h, 1.0 - h, mode='full')
                            cur_score = new_score
                            if cur_score < best_score:
                                best_h = h.copy()
                                best_score = cur_score
                            stall = 0
                            move_cap = min(move_cap * 1.1, 0.5)
    
    # Final projection
    best_h = _project(best_h, n / 2.0)
    best_score = _exact_c5(best_h)
    return best_h, best_score

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    # Add tiny noise to break ties
    h_noisy = h + np.random.default_rng(123).random(n) * 1e-10
    idx = np.argsort(h_noisy)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi_idx, oi in enumerate(ones[::]):
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones[oi_idx] = oz
                    zeros[zeros.index(oz)] = oi
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

            beta = max(beta * 0.95, 5.0)
            ridge *= 1.2
            if stall % 15 == 0:
                d_rand = rng.standard_normal(n)
                d_rand -= np.mean(d_rand)
                dm = np.max(np.abs(d_rand))
                if dm > 1e-14:
                    d_rand = d_rand / dm * (move_cap * 0.4 + 0.05)
                    t_opt, val_opt = _line_search(h, S, d_rand, move_cap * 0.4 + 0.05)
                    if abs(t_opt) > 1e-12:
                        h_try = h + t_opt * d_rand
                        h_try = _project(h_try, n / 2.0)
                        new_score = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
                        if new_score < cur_score - 1e-12:
                            h = h_try
                            S = np.correlate(h, 1.0 - h, mode='full')
                            cur_score = new_score
                            if cur_score < best_score:
                                best_h = h.copy()
                                best_score = cur_score
                            stall = 0
                            move_cap = min(move_cap * 1.2, 0.5)

    best_h = _project(best_h, n / 2.0)
    best_score = _exact_c5(best_h)
    return best_h, best_score

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n, seed=123):
    rng = np.random.default_rng(seed)
    h_noisy = h + rng.random(n) * 1e-10
    idx = np.argsort(h_noisy)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 350:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi_i in range(len(ones)):
            if time.time() > deadline:
                break
            oi = ones[oi_i]
            for oz_i in range(len(zeros)):
                oz = zeros[oz_i]
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones[oi_i] = oz
                    zeros[oz_i] = oi
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5
