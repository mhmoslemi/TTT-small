def _project_sum(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        h5b, c5_5b = _gradient_descent_lse(best_h, sub_dl2, n_steps=2000, tau=0.01)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()
    if time.time() < d5:
        sub_dl3 = min(d5, time.time() + budget_s * 0.03)
        h5c, c5_5c = _pair_polish(_project_sum(best_h, n_cur / 2.0), sub_dl3, n_pairs=15000, seed=222)
        if c5_5c < best_c5:
            best_c5 = c5_5c
            best_h = h5c.copy()

    # Phase 6: Dimension transfer (10%)
    d6 = t0 + budget_s * 0.90
    n_best = len(best_h)
    for dn in [4, -4, 8, -8, 12, -12]:
        if time.time() > d6 - 2:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 300 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_anneal(h_r, sub_dl, n_levels=4, tau_mult=0.5)
        if time.time() < d6 - 2 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d6, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _pair_polish(_project_sum(h_r, n_try / 2.0), sub_dl2, n_pairs=8000, seed=n_try)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 7: Basin hopping (7%)
    d7 = t0 + budget_s * 0.97
    scale = 0.04
    n_restarts = 0
    while time.time() < d7 and n_restarts < 100:
        n_restarts += 1
        h_p = _perturb(best_h, rng, scale)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        hp, cp = _lse_anneal(h_p, sub_dl, n_levels=3, tau_mult=0.5)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 8: Final polish (3%)
    n_final = len(best_h)
    h_best = _project_sum(best_h, n_final / 2.0)
    if deadline - time.time() > 5:
        h_p, c5_p = _pair_polish(h_best, deadline, n_pairs=10000, seed=333)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            h_best = _project_sum(best_h, n_final / 2.0)
    if deadline - time.time() > 3:
        h_g, c5_g = _gradient_descent_lse(h_best, deadline, n_steps=500, tau=0.008)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h_best = _project_sum(best_h, n_final / 2.0)

    h_best = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(h_best)
    return h_best, final_c5, len(h_best)
