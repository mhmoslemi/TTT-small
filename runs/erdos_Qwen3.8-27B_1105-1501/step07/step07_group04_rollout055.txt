def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_gradient(h, k):
    """Gradient of correlation at lag k (offset from center)."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # c[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[i] for i in [k, n-1]: +(1-h[i-k])
        # d/dh[i] for i in [0, n-1-k]: -h[i+k]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        # c[-k] = sum_{i=0}^{n-1-u} h[i]*(1-h[i+u])
        # d/dh[i] for i in [0, n-1-u]: +(1-h[i+u])
        # d/dh[i] for i in [u, n-1]: -h[i-u]
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

def _line_search(h, d, dx, radius):
    """Exact quadratic line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-12:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section search on [0, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b, hi_b = 0.0, t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(50):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    # Also check endpoints

    best = min(candidates, key=lambda x: x[1])

def _lp_direction(h, corr, dx, active_indices, radius):
    """Solve LP to find minimax descent direction."""
    n = len(h)
    target = n / 2.0
    m = len(active_indices)
    
    # Variables: d_0..d_{n-1}, t
    # n+1 variables total
    # Objective: minimize t + eps * sum(d_i^2) approximated by eps * sum(|d_i|)
    # For linprog, use: minimize [0,...,0, 1, eps,...,eps]  (t first or last?)
    # Let's put d_0..d_{n-1} first, then t last.
    # c = [eps, ..., eps, 1]
    
    eps = 1e-6
    c = np.concatenate([np.full(n, eps), [1.0]])
    
    # Inequalities: for each active constraint r:
    #   grad_r . d + F_r <= t
    #   => grad_r . d - t <= 0
    # But we want: F_r(h) + grad_r . d <= t (linearized value <= t)
    #   => grad_r . d - t <= -F_r(h)
    
    # A_ub: m x (n+1) matrix
    # b_ub: m vector
    A_rows = []
    b_vec = []
    
    center = n - 1
    for r in active_indices:
        k = r - center
        g = _corr_gradient(h, k) * dx  # scaled gradient
        row = np.concatenate([g, [-1.0]])
        A_rows.append(row)
        b_vec.append(-corr[r] * dx)  # -(scaled correlation value)
    
    if len(A_rows) > 0:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vec)
    else:
        A_ub = None
        b_ub = None
    
    # Equality: sum(d_i) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -radius <= d_i <= radius, but also -h_i <= d_i <= 1-h_i
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((-1.0, 1.0))  # t unbounded (within reason)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            # Remove mean to ensure mass conservation (numerical safety)
            d = d - np.mean(d)
            return d
    except Exception:
        pass
    return None

def _corr_row_gradient(h, k):

    else:

            new_corr = np.correlate(cur, 1.0 - cur, mode='full')
            new_c5 = float(np.max(new_corr) * dx)
            cur[oi] = 1.0
            cur[oz] = 0.0
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
                best_new_corr = new_corr
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = best_new_corr
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
        else:
            best_stall += 1
    return best_h, best_c5

def _binary_directed_search(h_bin, deadline, max_iters=200000, seed=77, n_try=80):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _exact_c5_corr(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = [int(x) for x in np.where(cur > 0.5)[0]]
    zeros = [int(x) for x in np.where(cur <= 0.5)[0]]
    best_stall = 0
    center = n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 6000:
            break
        if not ones or not zeros:
            break
        max_idx = int(np.argmax(cur_corr))
        k_act = max_idx - center
        # Build directed candidate set: ones that contribute to active lag
        if k_act > 0:
            contrib_ones = [p for p in ones if p - k_act >= 0 and p - k_act < n and cur[p - k_act] == 0.0]
        elif k_act < 0:
            u = -k_act
            contrib_ones = [p for p in ones if p + u < n and cur[p + u] == 0.0]
        else:
            contrib_ones = ones[:]
        if not contrib_ones:
            contrib_ones = ones[:]
        best_delta = 0.0
        best_move = None
        best_new_corr = None
        for _ in range(n_try):
            # Mix directed and random
            if rng.random() < 0.7 and len(contrib_ones) > 0:
                oi = contrib_ones[int(rng.integers(0, len(contrib_ones)))]
            else:
                oi = ones[int(rng.integers(0, len(ones)))]
            oz = zeros[int(rng.integers(0, len(zeros)))]
            if oi == oz:
                continue
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_corr = np.correlate(cur, 1.0 - cur, mode='full')
            new_c5 = float(np.max(new_corr) * dx)
            cur[oi] = 1.0
            cur[oz] = 0.0
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz)
                best_new_corr = new_corr
        if best_move is not None and best_delta < -1e-14:
            oi, oz = best_move
            i_one = ones.index(oi)
            i_zero = zeros.index(oz)
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = best_new_corr
            ones[i_one] = oz
            zeros[i_zero] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
        else:
            best_stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 10.0, size=8)
    phases = rng.uniform(0, 2 * np.pi, size=8)
    amps = rng.uniform(0.3, 1.0, size=8)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amplitude
    p += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def _pair_swap_search(h, deadline, n_pairs=20000, radius=0.2, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur - 1e-13:
            h = _project(h + alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5
