        if c2b < best_c:
            best_c = c2b
            best_h = h2b.copy()

    # Phase 3: Binary SA + re-optimize (15%)
    d3 = t0 + budget_s * 0.41
    nb = len(best_h)
    hbin = _to_bin(best_h)
    hbin, cb = _bin_sa(hbin, min(d3, time.time() + budget_s * 0.08), mx=12000, Ti=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 3:
        hp = _project(hbin.copy(), nb / 2.0)
        hp, cp = _local_opt(hp, d3, r0=0.10, stall_lim=35, use_lp=True, lp_k=15, lp_every=3)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.61
    nc = len(best_h)
    offsets = [-8, -4, 4, 8, -12, 12, -16, 16, -6, 6, -10, 10, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d4 - 5:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, use_lp=True, lp_k=15, lp_every=3)
        if cr < best_c + 0.003 and time.time() < d4 - 5:
            hb = _to_bin(hr)
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d4 - 3:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, use_lp=True, lp_k=10, lp_every=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Intensive re-optimization (15%)
    d5 = t0 + budget_s * 0.76
    nc = len(best_h)
    for r_init in [0.12, 0.06, 0.03]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.05)
        h5, c5r = _local_opt(h5, sd, r0=r_init, stall_lim=50, use_lp=True, lp_k=20, lp_every=2)
        if c5r < best_c:
            best_c = c5r
            best_h = h5.copy()
        if time.time() < d5 - 5:
            h5b = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.025)
            h5b, c5b = _pair_opt(h5b, sd, max_it=12000)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b.copy()

    # Phase 6: Basin hopping (12%)
    d6 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d6 and n_re < 50:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.3) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=20, use_lp=False, lp_k=10, lp_every=5)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline - 2, r0=0.02, stall_lim=25, use_lp=True, lp_k=15, lp_every=2)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    if final_c > best_c + 1e-9:
        final_c = best_c
    return best_h, final_c, len(best_h)
