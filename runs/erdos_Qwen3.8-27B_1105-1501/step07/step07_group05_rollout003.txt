def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

    n = len(h)
    dirs = []
    g = _softmax_grad(h, raw, dx, 30.0)
    nrm = np.linalg.norm(g)
    if nrm > 1e-10:
        dirs.append(g / nrm)
    g2 = _softmax_grad(h, raw, dx, 80.0)
    nrm2 = np.linalg.norm(g2)
    if nrm2 > 1e-10:
        dirs.append(g2 / nrm2)
    if use_lp:
        max_k = 4 if lp_count is None else lp_count
        for k in [3, 8, 20]:
            if k > 2 * n - 2:
                break
            s = _lp_direction(h, raw, n, k, 0.03)
            if s is not None:
                dirs.append(s)
            if len(dirs) >= nd:
                break
    if len(dirs) < nd:
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(3):
            f = rng.uniform(1.0, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    if len(dirs) < nd:
        d = rng.normal(0, 1, n)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

def _local_refine(h, deadline, r0=0.15, stall_lim=60, nd=6, use_lp=True, lp_every=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(raw) * dx)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    it_count = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.2:
            break
        it_count += 1
        do_lp = use_lp and (it_count % lp_every == 1)
        dirs = _gen_dirs(h, raw, dx, rng, nd, use_lp=do_lp)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            raw = np.correlate(h, 1.0 - h, mode='full')
            cur = float(np.max(raw) * dx)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
        else:
            stall += 1
            rad *= 0.78
            if rad < 0.006:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                raw_p = np.correlate(hp, 1.0 - hp, mode='full')
                cp = float(np.max(raw_p) * dx)
                if cp < cur - 1e-12:
                    h = hp
                    raw = raw_p
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 20 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(raw) * dx)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            raw_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            c5a = float(np.max(raw_new) * dx)
            if c5a < cur - 1e-14:
                h = h_new
                raw = raw_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_sa(hb, deadline, mx=20000, Ti=0.002):
    n = len(hb)
    dx = 2.0 / n
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return bh, bc

def _construct_spectral(n, rng):
    x = np.arange(n, dtype=float)
    h = np.zeros(n)
    for k in range(1, 6):
        amp = rng.uniform(0.3, 1.0) / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * freq * x / n + phase)
    h += 0.5
    return _project(h, n / 2.0)

def _construct_block(n, rng):
    block_size = int(rng.integers(8, 20))
    nb = n // block_size
    h = np.zeros(n)
    for b in range(nb):
        val = rng.uniform(0.2, 0.8)
        h[b * block_size:(b + 1) * block_size] = val
    if nb * block_size < n:
        h[nb * block_size:] = rng.uniform(0.2, 0.8)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Local refine with LP (20%)
    d1 = t0 + budget_s * 0.20
    h1, c1 = _local_refine(h_init, d1, r0=0.15, stall_lim=50, nd=6, use_lp=True, lp_every=3)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair optimization (12%)
    d2 = t0 + budget_s * 0.32
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=30000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA + local (15%)
    d3 = t0 + budget_s * 0.47
    nb = len(best_h)
    hbin = _to_binary(best_h, nb)
    hbin, cb = _binary_sa(hbin, min(d3, t0 + budget_s * 0.40), mx=15000, Ti=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 2:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.03)
        hp, cp = _local_refine(hp, sd, r0=0.08, stall_lim=30, nd=4, use_lp=False)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Resolution exploration (25%)
    d4 = t0 + budget_s * 0.72
    nc = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.015)
        hr, cr = _local_refine(hr, sd, r0=0.12, stall_lim=35, nd=5, use_lp=True, lp_every=4)
        if cr < best_c + 0.003 and time.time() < d4 - 3:
            hb = _to_binary(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hb, cb2 = _binary_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.002 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr2 = _local_refine(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, nd=4, use_lp=False)
            if cr2 < cr:
                cr = cr2
                hr = hr
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Alternate starting points (10%)
    d5 = t0 + budget_s * 0.82
    if time.time() < d5:
        for h_alt in [_construct_spectral(nc, rng), _construct_block(nc, rng)]:
            if time.time() > d5 - 3:
                break
            sd = min(d5, time.time() + budget_s * 0.02)
            ha, ca = _local_refine(h_alt, sd, r0=0.15, stall_lim=40, nd=6, use_lp=True, lp_every=3)
            if ca < best_c:
                best_c = ca
                best_h = ha.copy()

    # Phase 6: Intensive local + pair (8%)
    d6 = t0 + budget_s * 0.90
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.02)
        h6, c6 = _local_refine(h6, sd, r0=r_init, stall_lim=40, nd=5, use_lp=True, lp_every=3)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
    if time.time() < d6:
        h6 = _project(best_h.copy(), nc / 2.0)
        h6, c6 = _pair_opt(h6, d6, max_it=15000)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Basin hopping (5%)
    d7 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d7 and n_re < 40:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.5, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_refine(hp, sd, r0=0.05, stall_lim=15, nd=4, use_lp=False)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_refine(hf, deadline, r0=0.02, stall_lim=25, nd=4, use_lp=False)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 1.5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
