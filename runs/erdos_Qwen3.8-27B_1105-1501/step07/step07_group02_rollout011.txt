import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _softmax_grad(h, beta):
    """Compute gradient of smoothed max of overlap integrals."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    # Compute all g_k values
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx  # these are the actual overlap values
    
    # Softmax weights
    g_shifted = g - np.max(g)
    w = np.exp(beta * g_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        # degenerate: uniform over max
        max_idx = np.where(g == np.max(g))[0]
        w = np.zeros(L)
        w[max_idx] = 1.0 / len(max_idx)
    else:
        w = w / w_sum
    
    # Compute gradient: sum_k w_k * grad g_k
    # grad g_k (with respect to h_i) computed element-wise
    # For lag ell = r - (n-1):
    # grad_k[h_i] = dx * [(1 - h[i+ell]) * 1_{valid} - h[i-ell] * 1_{valid}]
    
    grad = np.zeros(n)
    # Vectorized computation
    for r in range(L):
        if w[r] < 1e-15:
            continue
        ell = r - (n - 1)
        # h_i * (1 - h_{i+ell}): derivative wrt h_i is (1-h_{i+ell}), wrt h_{i+ell} is -h_i
        if ell >= 0:
            # i ranges 0 to n-1-ell
            m = n - ell
            valid_i = np.arange(m)
            # derivative at h_i (first factor): (1 - h[i+ell])
            grad[valid_i] += w[r] * dx * (1.0 - h[valid_i + ell])
            # derivative at h_{i+ell} (second factor, with negative sign): -h_i
            grad[valid_i + ell] += w[r] * dx * (-h[valid_i])
        else:
            p = -ell
            m = n - p
            valid_i = np.arange(p, n)
            # i ranges p to n-1, i+ell = i-p ranges 0 to n-1-p
            # derivative at h_i: (1 - h[i-p])
            grad[valid_i] += w[r] * dx * (1.0 - h[valid_i - p])
            # derivative at h_{i-p}: -h_i
            grad[valid_i - p] += w[r] * dx * (-h[valid_i])
    
    return grad, float(np.max(g)), w

def _grad_descent_smooth(h, deadline, beta=50.0, max_iter=2000, seed=42):
    """Gradient descent with softmax-smoothed max."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    
    step_size = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, cur_c5, w = _softmax_grad(h, beta)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Projected gradient step
        # We want to minimize, so step in -grad direction
        # But we need to maintain sum(h) = target
        # Remove the mean component (which corresponds to the equality constraint)
        grad_tangent = grad - np.mean(grad)
        
        # Backtracking line search
        best_step = None
        best_val = cur_c5
        for alpha in [step_size, step_size*0.5, step_size*0.25, step_size*0.1, step_size*0.01]:
            h_try = _project(h - alpha * grad_tangent, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_step = h_try
                step_size = alpha  # record successful step
                break
            else:
                step_size = alpha * 0.5
        
        if best_step is None:
            # No improvement with current step size
            step_size *= 0.5
            if step_size < 1e-8:
                break
            continue
        
        h = best_step
        beta = min(beta * 0.98, 200.0)  # slowly increase smoothness
    
    return best_h, best_c5

But wait, the target is sum(h) = n/2, and for binary h with n/2 ones, the lag-0 term is 0. So the max is determined by non-zero lags.

For a binary sequence with n/2 ones and n/2 zeros, the overlap at lag k is:
C_k = dx * |{i : h_i=1, h_{i+k}=0}| (for positive k, with boundary)

And C_{-k} = dx * |{i : h_i=0, h_{i-k}=1}| = dx * |{i : h_i=1, h_{i-k}=0}|

So C_k + C_{-k} = dx * |{i : h_i ≠ h_{i+k}}| (counting boundary effects properly)

This is the "boundary length" at scale k. For a random binary sequence, this is approximately n (for each position, probability of mismatch is about 1/2, so expected mismatches ≈ n/2... wait no, for a random sequence with half 1s, P(h_i ≠ h_j) ≈ 1/2 for i≠j, so expected mismatches at any lag ≈ (n-|k|)/2).

So for random, C_k ≈ dx * (n-|k|)/2 = (2/n) * (n-|k|)/2 = (n-|k|)/n ≈ 1 for small k.

But we want C_k ≈ 0.38, which is much less than 1. So we need a very structured sequence.

The optimal structure is likely related to "low-discrepancy" sequences or "balanced" sequences that minimize the maximum autocorrelation.

One known good construction: the "Reiman" construction or "Golay complementary sequences" adapted to this problem.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _softmax_weights(S0, beta):
    M = float(np.max(S0))
    g = S0 - M
    w = np.exp(beta * g)
    s = float(np.sum(w))
    if s < 1e-30:
        k = max(1, len(S0) // 8)
        idx = np.argsort(S0)[-k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / k
    else:
        w /= s
    return w

def _grad_matrix(h, S0, beta):
    n = len(h)
    L = 2 * n - 1
    w = _softmax_weights(S0, beta)
    G = np.zeros((L, n))
    for r in range(L):
        if w[r] < 1e-15:
            continue
        ell = r - (n - 1)
        if ell >= 0:
            m = n - ell
            G[r, :m] += (1.0 - h[ell:ell+m])
            G[r, ell:ell+m] -= h[:m]
        else:
            p = -ell
            m = n - p
            G[r, p:] += (1.0 - h[:m])
            G[r, :m] -= h[p:]
    return (G.T @ w)

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, min(opt_val, float(vals[best_idx]))

def _group_move(h, S0, beta, radius, n_groups, deadline, seed=42):
    """Move mass from groups of sources to groups of sinks simultaneously."""
    n = len(h)
    target = n / 2.0
    target_n = max(4, min(n // 4, n_groups))
    
    grad = _grad_matrix(h, S0, beta)
    sort_idx = np.argsort(grad)
    sinks = sort_idx[:target_n]
    sources = sort_idx[-target_n:]
    
    d = np.zeros(n)
    for i in sinks:
        d[i] = 1.0
    for i in sources:
        d[i] = -1.0
    
    one_mh = 1.0 - h
    T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find valid range for t
    t_lo = 0.0
    for i in sinks:
        t_lo = max(t_lo, -h[i])
    for i in sources:
        t_lo = max(t_lo, h[i] - 1.0)
    t_lo = max(t_lo, -radius)
    
    t_hi = radius
    for i in sinks:
        t_hi = min(t_hi, 1.0 - h[i])
    for i in sources:
        t_hi = min(t_hi, h[i])
    t_hi = min(t_hi, radius)
    
    if t_hi - t_lo < 1e-13:
        return None
    
    t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
    h_try = _project(h + t_opt * d, target)
    return h_try

def _hybrid_optimize(h, deadline, radius=0.12, n_pairs=40, n_groups=8, beta=25.0, stall_limit=80, seed=42):
    """Combined pair and group optimization."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = _corr_values(h)
    rng = np.random.default_rng(seed)
    stall = 0
    radius = float(radius)
    beta_cur = float(beta)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        
        improved = False
        one_mh = 1.0 - h
        
        # Group moves
        for gi in range(min(3, n_groups)):
            if time.time() > deadline:
                break
            h_try = _group_move(h, S0, beta_cur, radius, n_groups, deadline, seed + gi)
            if h_try is not None:
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-12:
                    h = h_try
                    cur_c5 = c5_try
                    S0 = _corr_values(h)
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
        
        # Pair moves
        if not improved:
            grad = _grad_matrix(h, S0, beta_cur)
            sort_idx = np.argsort(grad)
            n_p = min(n_pairs, n // 2)
            sinks = sort_idx[:n_p]
            sources = sort_idx[-n_p:]
            
            best_step = None
            best_c5_try = cur_c5
            
            for ii in range(min(n_pairs, len(sinks), len(sources))):
                if time.time() > deadline:
                    break
                sink = int(sinks[ii])
                source = int(sources[ii])
                if sink == source:
                    continue
                t_lo = max(-h[sink], h[source] - 1.0, -radius)
                t_hi = min(1.0 - h[sink], h[source], radius)
                if t_hi - t_lo < 1e-13:
                    continue
                d = np.zeros(n)
                d[sink] = 1.0
                d[source] = -1.0
                T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
                C = np.correlate(d, d, mode='full')
                t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
                h_try = h.copy()
                h_try[sink] += t_opt
                h_try[source] -= t_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5_try - 1e-14:
                    best_c5_try = c5_try
                    best_step = h_try
            
            # Random pairs
            for _ in range(min(10, n_pairs // 4)):
                if time.time() > deadline:
                    break
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
                t_lo = max(-h[a], h[b] - 1.0, -radius)
                t_hi = min(1.0 - h[a], h[b], radius)
                if t_hi - t_lo < 1e-13:
                    continue
                d = np.zeros(n)
                d[a] = 1.0
                d[b] = -1.0
                T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
                C = np.correlate(d, d, mode='full')
                t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
                h_try = h.copy()
                h_try[a] += t_opt
                h_try[b] -= t_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5_try - 1e-14:
                    best_c5_try = c5_try
                    best_step = h_try

                h = _project(best_step, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                S0 = _corr_values(h)
                stall = 0
                radius = min(radius * 1.03, 0.25)
                beta_cur = min(beta_cur * 1.02, 80.0)
            else:
                stall += 1
                radius *= 0.80
                beta_cur = max(beta_cur * 0.95, 10.0)
                if stall > 30 and radius < 0.03:
                    radius = 0.12
                    beta_cur = 20.0
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_search(h_bin, deadline, seed=42):
    """Local binary search via pair swaps."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = 0.004
    stall = 0
    max_iter = 50000
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    
    # Greedy swap pass
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones_set = set(np.where(cur > 0.5)[0].tolist())
        zeros_set = set(np.where(cur <= 0.5)[0].tolist())
        for oi in list(ones_set):
            if time.time() > deadline:
                break
            for oz in list(zeros_set):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones_set.discard(oi)
                    zeros_set.discard(oz)
                    ones_set.add(oz)
                    zeros_set.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
    return best_h, best_c5

def _gradient_descent(h, deadline, beta=40.0, max_iter=500, seed=42):
    """Softmax-smoothed gradient descent."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    S0 = _corr_values(h)
    step_size = 0.3
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad = _grad_matrix(h, S0, beta)
        grad_tan = grad - np.mean(grad)
        
        best_step = None
        best_val = _exact_c5(h)
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = step_size * alpha_mult
            h_try = _project(h - alpha * grad_tan, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val - 1e-12:
                best_val = c5_try
                best_step = h_try
                step_size = alpha
                break
            step_size = alpha * 0.8
        
        if best_step is None:
            step_size *= 0.5
            if step_size < 1e-10:
                break
            continue
        
        h = best_step
        S0 = _corr_values(h)
        if h is not None and _exact_c5(h) < best_c5:
            best_c5 = _exact_c5(h)
            best_h = h.copy()
        beta = min(beta * 1.01, 150.0)
    
    return best_h, best_c5

def _make_smooth_init(n, seed=0):
    """Create a smooth initial profile."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * (1.5 * x + rng.random())) + 0.2 * np.sin(2 * np.pi * (3.0 * x + rng.random()))
    h = np.clip(h, 0, 1)
    return _project(h, n / 2.0)

def _make_block_init(n, seed=0):
    """Create a block-structured initial profile."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    remaining = n // 2
    pos = 0
    while remaining > 0 and pos < n:
        block_len = max(1, int(rng.integers(2, max(3, n // 8))))
        end = min(pos + block_len, n)
        h[pos:end] = 1.0
        remaining -= end - pos
        pos = end + max(1, int(rng.integers(1, max(2, n // 10))))
    return _project(h, n / 2.0)
