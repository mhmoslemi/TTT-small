2. **Gradient computation**: For row r (corresponding to lag ℓ = r - (n-1)):

   - ∂R_r/∂h_p = (1-h_{p-ℓ}) if valid - h_{p+ℓ} if valid

3. **Direction solving**: Given selected active rows with weights, find d that:
   - Minimizes ||d||² (least norm)
   - Subject to G_I @ d ≈ -w * (S_I - target) (reduce active rows toward target)
   - And sum(d) = 0

4. **Line search**: Along h + t*d, the objective is max_r [S_r + t*U_r - t²*B_r] where U and B come from correlations with d. Find t that minimizes this subject to box constraints.

def _line_search_dense(S, d, h, move_cap=1.0):
    """Find t that minimizes max(S + t*U - t²*B) subject to h+t*d in [0,1] and |t| <= move_cap."""
    U = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    t_lo, t_hi = _feasible_interval(h, d)
    t_lo = max(t_lo, -move_cap)
    t_hi = min(t_hi, move_cap)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S))
    
    # Grid search
    n_grid = 80
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Compute max over rows for each t
    # S (L,), U (L,), B (L,)
    # For each t: max(S + t*U - t²*B)
    # Shape: (n_grid, L)
    vals = np.max(S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine around best
    lo_idx = max(0, best_idx - 2)
    hi_idx = min(n_grid - 1, best_idx + 2)
    t_lo2 = ts[lo_idx]
    t_hi2 = ts[hi_idx]
    if t_hi2 - t_lo2 < 1e-12:
        best_t = ts[best_idx]
    else:
        n_fine = 50
        ts_fine = np.linspace(t_lo2, t_hi2, n_fine)
        vals_fine = np.max(S[None, :] + ts_fine[:, None] * U[None, :] - (ts_fine[:, None])**2 * B[None, :], axis=1)
        best_fine_idx = int(np.argmin(vals_fine))
        best_t = ts_fine[best_fine_idx]
    
    # Also check endpoints and 0
    candidates = [t_lo, t_hi, 0.0, best_t]
    best_val = float('inf')
    for tc in candidates:
        if t_lo <= tc <= t_hi:
            v = float(np.max(S + tc * U - tc * tc * B))
            if v < best_val:
                best_val = v
                best_t = tc
    
    return best_t, best_val

def _solve_dense_direction(h, S, K_active, target_mode, rng):
    """Solve for dense zero-sum direction from active rows."""
    n = len(h)
    L = 2 * n - 1
    
    # Select active rows
    sorted_idx = np.argsort(S)[::-1]
    
    if target_mode == 0:
        # Use top K
        K = min(K_active, n)
        active_rows = sorted_idx[:K]
    elif target_mode == 1:
        # Use top K but skip some
        K = min(K_active, n)
        active_rows = sorted_idx[:K]
    elif target_mode == 2:
        # Use top K with random from top 2K
        K = min(K_active, n)
        top_2K = sorted_idx[:2*K]
        active_rows = top_2K[rng.integers(0, len(top_2K), size=K)]
    else:
        K = min(K_active, n)
        active_rows = sorted_idx[:K]
    
    active_rows = np.sort(active_rows).astype(int)
    
    # Build gradient matrix
    A = np.zeros((K + 1, n))
    for i, r in enumerate(active_rows):
        lag = r - (n - 1)
        A[i] = _row_gradient(h, lag)
    A[K] = 1.0  # sum constraint
    
    # Target values
    S_active = S[active_rows]
    if target_mode == 0:
        t_target = float(np.min(S_active))  # pull down toward min
    elif target_mode == 1:
        t_target = float(np.median(S_active))
    elif target_mode == 2:
        t_target = float(np.mean(S_active) - 0.1 * (S_active.max() - S_active.min()))
    else:
        t_target = float(S_active.max() - 0.15 * (S_active.max() - S_active.min()))
    
    b = np.zeros(K + 1)
    b[:K] = t_target - S_active  # negative for rows above target
    b[K] = 0.0  # sum = 0
    
    # Solve (AA') μ = b
    M = A @ A.T
    M += 1e-8 * np.eye(K + 1)  # ridge
    try:
        mu = np.linalg.solve(M, b)
    except np.linalg.LinAlgError:
        mu = np.linalg.lstsq(M, b, rcond=None)[0]
    
    d = A.T @ mu
    
    # Ensure zero sum (should already be close)
    d -= np.mean(d)
    
    # Check if direction is useful
    if np.max(np.abs(d)) < 1e-10:
        return None
    
    # Normalize
    d_max = np.max(np.abs(d))
    d /= d_max
    
    return d

import numpy as np
import time

def _c5_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 0.1, float(v.max()) + 0.1
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    th = (lo + hi) * 0.5
    x = np.clip(v - th, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-14:
        m = (x < 1) if r > 0 else (x > 0)
        c = m.sum()
        if c > 0:
            x = x.copy()
            x[m] += r / c
    return np.clip(x, 0, 1)

def _row_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0: return g
        g[lag:] += 1.0 - h[:m]
        g[:m] -= h[lag:]
    else:
        p = -lag
        m = n - p
        if m <= 0: return g
        g[:m] += 1.0 - h[p:]
        g[p:] -= h[:m]
    return g

def _feasible_t(h, d):
    tl, th = -1e18, 1e18
    pos = d > 1e-14
    neg = d < -1e-14
    if pos.any():
        tl = max(tl, (-h[pos] / d[pos]).max())
        th = min(th, ((1 - h[pos]) / d[pos]).min())
    if neg.any():
        tl = max(tl, ((1 - h[neg]) / d[neg]).max())
        th = min(th, (-h[neg] / d[neg]).max())
    return tl, th

def _line_search(S, U, B, tl, th):
    if th - tl < 1e-14:
        return 0.0
    n1 = 80
    ts = np.linspace(tl, th, n1)
    v = (S + ts[:, None] * U - (ts[:, None])**2 * B).max(axis=1)
    bi = v.argmin()
    lo = max(0, bi - 2)
    hi = min(n1 - 1, bi + 2)
    if hi > lo:
        ts2 = np.linspace(ts[lo], ts[hi], 50)
        v2 = (S + ts2[:, None] * U - (ts2[:, None])**2 * B).max(axis=1)
        bi2 = v2.argmin()
        bt = ts2[bi2]
    else:
        bt = ts[bi]
    # Check candidates
    cands = [tl, 0.0, th, bt]
    best_v = np.inf
    for c in cands:
        if tl <= c <= th:
            val = (S + c * U - c * c * B).max()
            if val < best_v:
                best_v = val
                bt = c
    return bt

def _solve_dir(h, S, K, mode, rng):
    n = len(h)
    L = 2 * n - 1
    idx = np.argsort(S)[::-1]
    K = min(K, n - 1)
    if K < 2:
        return None
    if mode == 0:
        ar = idx[:K]
    elif mode == 1:
        ar = idx[:K]
    elif mode == 2:
        pool = idx[:2 * K]
        ar = pool[rng.integers(0, len(pool), size=K)]
    else:
        ar = idx[:K]
    ar = np.unique(ar).astype(int)
    K = len(ar)
    A = np.zeros((K + 1, n))
    for i, r in enumerate(ar):
        A[i] = _row_grad(h, int(r) - (n - 1))
    A[K] = 1.0
    Sa = S[ar]
    if mode == 3:
        tgt = Sa.min()
    elif mode == 1:
        tgt = Sa.max() - 0.12 * (Sa.max() - Sa.min() + 1e-10)
    else:
        tgt = Sa.max() - 0.2 * (Sa.max() - Sa.min() + 1e-10)
    b = np.zeros(K + 1)
    b[:K] = tgt - Sa
    M = A @ A.T + 1e-7 * np.eye(K + 1)
    try:
        mu = np.linalg.solve(M, b)
    except:
        mu = np.linalg.lstsq(M, b, rcond=None)[0]
    d = A.T @ mu
    d -= d.mean()
    mx = np.abs(d).max()
    if mx < 1e-12:
        return None
    d /= mx
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb
