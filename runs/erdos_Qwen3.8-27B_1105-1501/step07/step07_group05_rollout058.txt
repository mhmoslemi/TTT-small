import numpy as np
import time
import math
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, k):
    """Gradient of raw correlation at output index k w.r.t. h."""
    n = len(h)
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        L = n - lag
        g[:L] += 1.0 - h[lag:lag + L]
        g[lag:lag + L] -= h[:L]
    else:
        s = -lag
        L = n - s
        g[s:s + L] += 1.0 - h[:L]
        g[:L] -= h[s:s + L]
    return g

def _line_search(h, s, dx):
    """Exact line search along h + alpha*s for alpha in [0, 1].
    Returns (best_alpha, best_c5)."""
    n = len(h)
    raw0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    
    # Find feasible alpha range [0, 1] (since s already satisfies box constraints)
    a_lo, a_hi = 0.0, 1.0
    
    # Grid search + golden section
    def f(a):
        return float(np.max(raw0 + a * T - a * a * Q)) * dx
    
    # Sample on grid
    best_a = 0.0
    best_f = f(0.0)
    grid = np.linspace(0, 1, 50)
    for a in grid:
        val = f(a)
        if val < best_f - 1e-14:
            best_f = val
            best_a = a
    
    # Golden section refinement
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo_r, hi_r = 0.0, 1.0
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi_r - lo_r < 1e-10:
            break
        if f1 < f2:
            hi_r, c2, f2 = c2, c1, f1
            c1 = hi_r - gr * (hi_r - lo_r)
            f1 = f(c1)
        else:
            lo_r, c1, f1 = c1, c2, f2
            c2 = lo_r + gr * (hi_r - lo_r)
            f2 = f(c2)
    mid = (lo_r + hi_r) / 2.0
    fm = f(mid)
    if fm < best_f:
        best_f = fm
        best_a = mid
    
    return best_a, best_f

def _lp_direction(h, active_k, radius, max_time=0.5):
    """Solve LP to find direction s that decreases all active lags.
    Returns s (feasible step) or None."""
    n = len(h)
    t_start = time.time()
    
    # Build gradients for active lags
    grads = []
    for k in active_k:
        g = _lag_grad(h, k)
        gn = np.linalg.norm(g)
        if gn > 1e-10:
            grads.append(g / gn)
        else:
            grads.append(g)
    
    if not grads:
        return None
    
    K = len(grads)
    A = np.array(grads)  # K x n
    
    # Variables: s (n) + tau (1) = n+1 variables
    # Maximize tau
    # min -tau
    # s.t.: A @ s + tau <= 0  (K constraints)
    #       sum(s) = 0       (1 equality)
    #       -radius*h <= s <= radius*(1-h)  (2n constraints)
    #       tau >= 0          (1 constraint)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -tau
    
    # Inequality constraints: A @ s + tau <= 0
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.zeros((K + n + 1, n + 1))
    b_ub = np.zeros(K + n + 1)
    
    # A @ s + tau <= 0
    for i in range(K):
        A_ub[i, :n] = A[i]
        A_ub[i, n] = 1.0
        b_ub[i] = 0.0
    
    # s_i <= radius*(1-h_i)
    for i in range(n):
        A_ub[K + i, i] = 1.0
        b_ub[K + i] = radius * (1.0 - h[i])
    
    # -s_i <= radius*h_i  i.e. s_i >= -radius*h_i
    for i in range(n):
        A_ub[K + n + i, i] = -1.0
        b_ub[K + n + i] = radius * h[i]
    
    # -tau <= 0 i.e. tau >= 0
    A_ub[K + 2*n, n] = -1.0
    b_ub[K + 2*n] = 0.0
    
    # Equality: sum(s) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * n + [(0, None)],
                      method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    s = res.x[:n]
    tau = res.x[n]
    
    # Check: tau should be positive for a useful direction
    if tau < 1e-10:
        return None
    
    # Verify constraints
    if np.any(s < -radius * h - 1e-8) or np.any(s > radius * (1 - h) + 1e-8):
        s = np.clip(s, -radius * h, radius * (1.0 - h))
    s -= np.mean(s)
    
    return s

def _softmax_grad(h, beta):

    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:

            m = n - s
            if m > 0:

        b_ub[K + i] = radius * (1.0 - h[i])
    for i in range(n):
        A_ub[K + n + i, i] = -1.0
        b_ub[K + n + i] = radius * h[i]
    A_ub[K + 2 * n, n] = -1.0
    b_ub[K + 2 * n] = 0.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * n + [(0, None)], method='highs')
    except Exception:
        return None
    if not res.success or res.x[n] < 1e-8:
        return None
    s = res.x[:n].copy()
    s = np.clip(s, -radius * h, radius * (1.0 - h))
    s -= np.mean(s)
    if np.linalg.norm(s) < 1e-10:
        return None
    return s

def _softmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _scale_feasible(h, d, radius):
    pos = d > 1e-12
    neg = d < -1e-12
    max_a = radius
    if np.any(pos):
        max_a = min(max_a, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        max_a = min(max_a, float(np.min(h[neg] / (-d[neg]))))
    max_a = max(max_a, 0.0)
    return d * max_a, max_a

def _make_feasible_dir(h, raw_dir, radius):
    raw_dir = raw_dir - np.mean(raw_dir)
    nrm = np.linalg.norm(raw_dir)
    if nrm < 1e-12:
        return None
    raw_dir = raw_dir / nrm
    s, ma = _scale_feasible(h, raw_dir, radius)
    if ma < 1e-10:
        return None
    s -= np.mean(s)
    s = np.clip(s, -h + 1e-12, 1.0 - h - 1e-12)
    s -= np.mean(s)
    if np.linalg.norm(s) < 1e-10:
        return None
    return s

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _local_opt(h, deadline, r0=0.12, stall_lim=40, use_lp=True, lp_every=2):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(54321)
    rad = r0
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        it += 1
        dirs = []
        if use_lp and it % lp_every == 1:
            raw = np.correlate(h, 1.0 - h, mode='full')
            mx = float(np.max(raw))
            thresh = mx - 0.03 * mx
            act = list(np.where(raw >= thresh)[0])
            if len(act) > 16:
                act = list(np.argsort(-raw)[:16])
            if act:
                s = _lp_direction(h, act, rad)
                if s is not None:
                    dirs.append(s)
        for beta in [20.0, 40.0]:
            g = _softmax_grad(h, beta)
            s = _make_feasible_dir(h, -g, rad)
            if s is not None:
                dirs.append(s)
        if rng.random() < 0.25:
            pert = rng.normal(0, 1, n)
            s = _make_feasible_dir(h, pert, rad * 0.5)
            if s is not None:
                dirs.append(s)
        best_new = cur
        best_s = None
        best_a = 0.0
        for s in dirs:
            s = s - np.mean(s)
            a, c_try = _line_search(h, s, dx)
            if a < 1e-10:
                continue
            h_try = h + a * s
            h_try = _project(h_try, target)
            c_act = _exact_c5(h_try)
            if c_act < best_new - 1e-14:
                best_new = c_act
                best_s = s
                best_a = a
        if best_s is not None:
            h = h + best_a * best_s
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.28)
        else:
            stall += 1
            rad *= 0.70
            if rad < 0.008:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.10
                elif stall % 25 == 0:
                    rad = 0.05
    return best_h, best_c

def _pair_opt(h, deadline, max_it=20000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(999)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        if h[i] < h[j]:
            d[i] = 1.0
            d[j] = -1.0
            ma = min(h[j] - h[i], 0.15)
        else:
            d[i] = -1.0
            d[j] = 1.0
            ma = min(h[i] - h[j], 0.15)
        s = d * ma
        a, c_try = _line_search(h, s, dx)
        if a < 1e-10:
            stall += 1
            continue
        h_try = h + a * s
        h_try = _project(h_try, target)
        c_act = _exact_c5(h_try)
        if c_act < cur - 1e-14:
            h = h_try
            cur = c_act
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c

def _bin_sa(hb, deadline, mx=10000, Ti=0.0015):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    if not ones or not zeros:
        return hb.copy(), _exact_c5(hb)
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0.copy(), n0 / 2.0)
    best_c = _exact_c5(best_h)

    # Phase 1: LP-guided local opt on parent (18%)
    d1 = t0 + budget_s * 0.18
    h1, c1 = _local_opt(best_h, d1, r0=0.10, stall_lim=35, use_lp=True, lp_every=2)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair polish (10%)
    d2 = t0 + budget_s * 0.28
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=12000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (22%)
    d3 = t0 + budget_s * 0.50
    nc = len(best_h)
    offsets = [-8, -4, 4, 8, -12, 12, -16, 16, -6, 6]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 240 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        cr = _exact_c5(hr)
        sd = min(d3, time.time() + budget_s * 0.020)
        hr, cr = _local_opt(hr, sd, r0=0.08, stall_lim=25, use_lp=True, lp_every=3)
        if cr < best_c + 0.003 and time.time() < d3 - 5:
            sd2 = min(d3, time.time() + budget_s * 0.010)
            hr, cr = _pair_opt(_project(hr.copy(), n_try / 2.0), sd2, max_it=3000)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 4: Binary + structural seeds (15%)
    d4 = t0 + budget_s * 0.65
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, min(d4, time.time() + budget_s * 0.025), mx=8000)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4:
        hf = _project(best_h.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.015)
        hf, cf = _local_opt(hf, sd, r0=0.06, stall_lim=20, use_lp=True, lp_every=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    # Low-discrepancy binary seed
    if time.time() < d4 - 10:
        hr2 = np.zeros(nb)
        phi = 0.61803398874989
        for i in range(nb):
            if (i * phi) % 1.0 < 0.5:
                hr2[i] = 1.0
        n_ones = int(np.sum(hr2 > 0.5))
        if n_ones > nb // 2:
            ones_idx = np.where(hr2 > 0.5)[0]
            hr2[ones_idx[nb // 2:]] = 0.0
        elif n_ones < nb // 2:
            zeros_idx = np.where(hr2 <= 0.5)[0]
            hr2[zeros_idx[:nb // 2 - n_ones]] = 1.0
        cr2 = _exact_c5(hr2)
        if cr2 < best_c + 0.008:
            sd2 = min(d4, time.time() + budget_s * 0.02)
            hr2, cr2 = _local_opt(hr2, sd2, r0=0.12, stall_lim=30, use_lp=True, lp_every=2)
            if cr2 < best_c:
                best_c = cr2
                best_h = hr2.copy()

    # Phase 5: Intensive local opt (20%)
    d5 = t0 + budget_s * 0.85
    h5 = _project(best_h.copy(), len(best_h) / 2.0)
    h5, c5 = _local_opt(h5, d5, r0=0.06, stall_lim=50, use_lp=True, lp_every=2)
    if c5 < best_c:
        best_c = c5
        best_h = h5.copy()

    # Phase 6: Final polish (10%)
    d6 = deadline
    hf = _project(best_h.copy(), len(best_h) / 2.0)
    hf, cf = _pair_opt(hf, d6, max_it=8000)
    if cf < best_c:
        best_c = cf
        best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
