        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    if hi_b - lo_b > 1e-10:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=30.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    top_k = min(len(scores), 50)
    top_idx = np.argsort(w)[-top_k:]
    G_top = _grad_rows(h, lags[top_idx])
    g = G_top.T @ w[top_idx]
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 12
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        d = _active_lp(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_direction(h, scores, beta=30.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.72
                gap *= 1.4
                n_active = min(n_active + 2, 28)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.75
            gap *= 1.3
            n_active = min(n_active + 2, 28)
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _binary_sa(b, deadline, seed=123, T_init=None, max_iter=50000):
    n = len(b)
    dx = 2.0 / n
    half = n // 2
    if T_init is None:
        T_init = 1.5 * dx
    cur = b.astype(float).copy()
    counts = np.correlate(cur.astype(np.int64), 1 - cur.astype(np.int64), mode='full')
    cur_max = int(np.max(counts))
    best_max = cur_max
    best_h = cur.copy()
    rng = np.random.default_rng(seed)
    T = T_init
    ones_list = list(np.where(cur > 0.5)[0])
    zeros_list = list(np.where(cur <= 0.5)[0])
    stall = 0
    cooling = 0.99975
    scale = n / (2.0 * dx)  # = n^2/4, for acceptance probability scaling
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        oi = int(rng.integers(0, half))
        zi = int(rng.integers(0, half))
        r = ones_list[oi]
        s = zeros_list[zi]
        
        Q = counts.copy()
        # r: 1->0
        lost_A = (r - np.array(zeros_list)) + (n - 1)
        other_ones = np.array(ones_list)
        mask_nr = other_ones != r
        gained_A = (other_ones[mask_nr] - r) + (n - 1)
        Q[lost_A] -= 1
        Q[gained_A] += 1
        
        # s: 0->1
        cur_zeros_arr = np.array(zeros_list + [r])
        cur_ones_arr = other_ones[mask_nr]
        lost_B = (cur_ones_arr - s) + (n - 1)
        gained_B = (s - cur_zeros_arr) + (n - 1)
        Q[lost_B] -= 1
        Q[gained_B] += 1
        
        new_max = int(np.max(Q))
        delta = new_max - cur_max
        
        if delta <= 0 or (T > 1e-10 and rng.random() < np.exp(-float(delta) * scale / max(T, 1e-10))):
            counts = Q
            cur_max = new_max
            cur[r] = 0.0
            cur[s] = 1.0
            ones_list[oi] = s
            zeros_list[zi] = r
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur.copy()
                stall = 0
            T *= cooling
        else:
            stall += 1
            T = max(T * 0.99995, 1e-12)
    
    return best_h, best_max * dx

def _to_binary(h, rng=None):
    n = len(h)
    half = n // 2
    idx = np.argsort(h)[-half:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_sweep(h, deadline, max_rounds=100):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    rng = np.random.default_rng(777)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(max(1, len(ones) // 4)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.3)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Quick continuous optimization (8% time)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_opt(h_init, d1, radius_init=0.12, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Binary SA from diverse seeds (35% time)
    d2 = t0 + budget_s * 0.43
    n_bin = n0 if n0 % 2 == 0 else n0 + 1
    if n_bin != n0:
        h_bin_init = _resample(best_h, n_bin)
    else:
        h_bin_init = best_h.copy()
    
    # Generate diverse binary starting points
    bin_seeds = []
    # Seed 1: threshold of current best
    b1 = _to_binary(h_bin_init)
    bin_seeds.append(b1)
    # Seed 2: perturbed threshold
    b2 = _to_binary(_smooth_perturb(h_bin_init, 0.15, rng))
    bin_seeds.append(b2)
    # Seed 3: low-discrepancy arithmetic progression
    half = n_bin // 2
    step = 7  # coprime with many n
    pos3 = sorted((np.arange(half) * step) % n_bin)
    b3 = np.zeros(n_bin)
    b3[pos3] = 1.0
    bin_seeds.append(b3)
    # Seed 4: different step
    step4 = 13
    pos4 = sorted((np.arange(half) * step4) % n_bin)
    b4 = np.zeros(n_bin)
    b4[pos4] = 1.0
    bin_seeds.append(b4)
    # Seed 5: random balanced
    perm5 = rng.permutation(n_bin)
    b5 = np.zeros(n_bin)
    b5[perm5[:half]] = 1.0
    bin_seeds.append(b5)
    
    bin_results = []
    per_seed_time = (d2 - time.time()) / len(bin_seeds)
    for i, b_seed in enumerate(bin_seeds):
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + per_seed_time * (1.5 if i < 2 else 1.0))
        b_res, c5_bin = _binary_sa(b_seed, sub_dl, seed=100 + i, T_init=1.5 * (2.0 / n_bin))
        bin_results.append((b_res, c5_bin))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            # Convert to continuous and quick polish
            h_try = _project(b_res, n_bin / 2.0)
            h_try, c5_try = _local_opt(h_try, min(deadline - 1, time.time() + 5), radius_init=0.06, max_iters=500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 3: Convex combination of best binary solutions (15% time)
    d3 = t0 + budget_s * 0.58
    if len(bin_results) >= 2 and time.time() < d3:
        # Sort by c5 and take top k
        bin_results.sort(key=lambda x: x[1])
        k = min(4, len(bin_results))
        # Average top k
        combined = np.mean([b_r[0] for b_r in bin_results[:k]], axis=0)
        combined = _project(combined, n_bin / 2.0)
        h_comb, c5_comb = _local_opt(combined, min(d3, time.time() + budget_s * 0.06), radius_init=0.08, max_iters=1000)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
        # Also try pairwise combinations
        for i in range(min(3, k)):
            for j in range(i + 1, min(3, k)):
                if time.time() > d3:
                    break
                pair_avg = 0.5 * bin_results[i][0] + 0.5 * bin_results[j][0]
                pair_avg = _project(pair_avg, n_bin / 2.0)
                h_p, c5_p = _local_opt(pair_avg, min(d3, time.time() + budget_s * 0.02), radius_init=0.06, max_iters=500)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
    
    # Phase 4: Multi-resolution exploration (15% time)
    d4 = t0 + budget_s * 0.73
    n_curr = len(best_h)
    tried_ns = set()
    offsets = list(range(-16, 17, 4))
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 230 or n_try in tried_ns:
            continue
        if n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick binary from this
        b_r = _to_binary(h_r)
        sub_dl1 = min(d4, time.time() + budget_s * 0.012)
        b_res, c5_br = _binary_sa(b_r, sub_dl1, seed=200 + n_try, T_init=1.5 * (2.0 / n_try))
        # Polish
        h_pol = _project(b_res, n_try / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.012)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl2, radius_init=0.08, max_iters=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 5: Basin hopping (15% time)
    d5 = t0 + budget_s * 0.88
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.06, max_iters=400)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.06)
    
    # Phase 6: Final polish (12% time)
    d6 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    sub_dl1 = min(d6, time.time() + budget_s * 0.05)
    h_f, c5_f = _local_opt(h_f, sub_dl1, radius_init=0.04, max_iters=3000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Final pair sweep
    if deadline - time.time() > 5:
        h_ps = _project(best_h.copy(), len(best_h) / 2.0)
        h_ps, c5_ps = _pair_sweep(h_ps, deadline, max_rounds=80)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
