def _raw_corr(h):

def _project_to_sum(h, target):
    """Project h to have exact sum = target while staying in [0,1]."""
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0, 1)
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def _project_sum(h, target):
    h = np.clip(h.copy(), 0, 1)
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0, 1)

            best_h = _project_sum(best_h, len(best_h) / 2.0)
    # Gradient descent polish
    if time.time() < d1:
        x_g = _anti_from_h(best_h)
        x_g, c5_g = _grad_descent_anti(x_g, d1, beta=50.0, max_iter=200, radius_init=0.12)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = _anti_map(x_g)
            best_h = _project_sum(best_h, len(best_h) / 2.0)

    # Phase 2: Multi-n exploration (20%)
    d2 = t0 + budget_s * 0.40
    n_curr = len(best_h)
    n_list = sorted(set([max(100, n_curr + dn) for dn in range(-36, 37, 4)
                         if 100 <= n_curr + dn <= 248 and (n_curr + dn) % 2 == 0]))
    rng_n = np.random.default_rng(seed + 1)
    rng_n.shuffle(n_list)
    n_list = [n_curr] + n_list
    for n_try in n_list:
        if time.time() > d2:
            break
        if n_try == n_curr:
            continue
        h_r = _resample_to_n(best_h, n_try)
        x_r = _anti_from_h(h_r)
        sub_dl = min(d2, time.time() + budget_s * 0.01)
        x_b, c5_b, _ = _bfgs_anti(x_r, 30.0, sub_dl, max_outer=2, max_inner=25)
        if time.time() < d2 - 1:
            x_g, c5_g = _grad_descent_anti(x_b, min(d2, time.time() + budget_s * 0.008),
                                            beta=40.0, max_iter=60, radius_init=0.1)
            if c5_g < c5_b:
                c5_b = c5_g
                x_b = x_g
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = _anti_map(x_b)
            best_h = _project_sum(best_h, len(best_h) / 2.0)
            n_curr = n_try

    # Phase 3: Binary optimization (25%)
    d3 = t0 + budget_s * 0.65
    n_b = len(best_h)
    h_b = np.zeros(n_b)
    top = np.argsort(best_h)[-n_b // 2:]
    h_b[top] = 1.0
    sub_dl = min(d3, time.time() + budget_s * 0.10)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=50000, T_init=0.005)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.05)
        h_p, c5_p = _pair_sweep(best_h, sub_dl2)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: Continuous refinement with high beta (20%)
    d4 = t0 + budget_s * 0.85
    n_c = len(best_h)
    x_c = _anti_from_h(best_h)
    for beta in [80.0, 160.0, 320.0]:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.035)
        x_b, c5_b, x_c = _bfgs_anti(x_c, beta, sub_dl, max_outer=3, max_inner=30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = _anti_map(x_b)
            best_h = _project_sum(best_h, len(best_h) / 2.0)
    if time.time() < d4:
        x_c = _anti_from_h(best_h)
        x_g, c5_g = _grad_descent_anti(x_c, min(d4, time.time() + budget_s * 0.04),
                                        beta=100.0, max_iter=250, radius_init=0.08)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = _anti_map(x_g)
            best_h = _project_sum(best_h, len(best_h) / 2.0)

    # Phase 5: Basin hopping (15%)
    d5 = deadline
    x_bh = _anti_from_h(best_h)
    x_b, c5_b = _basin_hopping(x_bh, d5, rng, scale_init=0.04, time_budget=budget_s)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = _anti_map(x_b)
        best_h = _project_sum(best_h, len(best_h) / 2.0)

    # Final: nearby n values
    if deadline - time.time() > 15:
        n_f = len(best_h)
        for dn in [-4, 4, -8, 8]:
            if time.time() > deadline - 8:
                break
            n_try = n_f + dn
            if n_try < 100 or n_try > 260 or n_try % 2 != 0:
                continue
            h_r = _resample_to_n(best_h, n_try)
            x_r = _anti_from_h(h_r)
            sub_dl = min(deadline - 5, time.time() + budget_s * 0.012)
            x_b, c5_b, _ = _bfgs_anti(x_r, 60.0, sub_dl, max_outer=2, max_inner=25)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = _anti_map(x_b)
                best_h = _project_sum(best_h, len(best_h) / 2.0)

    # Final polish
    if deadline - time.time() > 5:
        x_f = _anti_from_h(best_h)
        x_g, c5_g = _grad_descent_anti(x_f, deadline, beta=200.0, max_iter=150, radius_init=0.05)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = _anti_map(x_g)
            best_h = _project_sum(best_h, len(best_h) / 2.0)

    best_h = np.clip(best_h, 0, 1)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
