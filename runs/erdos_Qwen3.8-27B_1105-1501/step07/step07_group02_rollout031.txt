    h = x
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # Soft max
    shifted = beta * corr
    max_val = np.max(shifted)
    return (1.0 / beta) * (max_val + np.log(np.sum(np.exp(shifted - max_val))))

def _soft_max_grad(x, n, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    shifted = beta * corr
    max_val = np.max(shifted)
    exp_shifted = np.exp(shifted - max_val)
    w = exp_shifted / np.sum(exp_shifted)

4. The actual direction is d = h_proj - h
5. Line search along d: minimize F(h + t*d) for t ∈ [0, 1]

The effective direction d is a weighted combination of all coordinates moving in their gradient directions, but adjusted by the projection to maintain feasibility. This is what makes it different from coordinate or pair descent.

For the line search, since F(h + t*d) is not convex, I'll use a simple backtracking or golden section search.

def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = beta * corr
    max_val = np.max(shifted)
    return float(max_val / beta + np.log(np.sum(np.exp(shifted - max_val))) / beta)

def _soft_max_gradient(h, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    max_val = np.max(shifted)
    exp_shifted = np.exp(shifted - max_val)
    w = exp_shifted / np.sum(exp_shifted)

        grad[i_lo:i_hi] += wk * dx * (1.0 - h[i_lo + ell:i_hi + ell])
        # Second factor: grad[i+ell] += wk * dx * (-h[i])
        grad[i_lo + ell:i_hi + ell] += wk * dx * (-h[i_lo:i_hi])

def _fista(h_init, deadline, beta=20.0, seed=42):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    y = h.copy()
    h_prev = h.copy()
    t_k = 1.0
    L = 5.0  # initial Lipschitz estimate
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    stall = 0
    
    for k in range(50000):
        if time.time() > deadline:
            break
        if stall > 3000:
            # Restart with perturbation
            y = _project(y + rng.normal(0, 0.03, n), target)
            h = y.copy()
            h_prev = y.copy()
            t_k = 1.0
            L = 5.0
            stall = 0
            continue
        
        # Compute gradient at y
        f_y, g_y = _sm_val_grad(y, beta)
        
        # FISTA step
        step = 1.0 / L
        tentative = y - step * g_y
        h_new = _project(tentative, target)
        
        # Backtracking on L
        f_new = _sm_val_grad(h_new, beta)[0]
        if f_new > f_y + np.dot(g_y, h_new - y) + 0.5 * L * np.dot(h_new - y, h_new - y):
            L *= 2.0
            tentative = y - (1.0 / L) * g_y
            h_new = _project(tentative, target)
            f_new = _sm_val_grad(h_new, beta)[0]
        
        # Update
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new
        h_prev = h
        t_k = t_new
    
    return best_h, best_c5
