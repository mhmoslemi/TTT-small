def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_qp_hessian(weights, n):
    """Build the Hessian matrix for the weighted overlap QP.
    weights[k] corresponds to lag k (1 to n-1).
    The objective is sum_k weights[k] * overlap_k(x)
    where overlap_k = sum_{i=0}^{n-1-k} x[i] - x[i]*x[i+k]
    """
    H = np.zeros((n, n))
    for k in range(1, n):
        wk = weights[k]
        if abs(wk) < 1e-15:
            continue
        for i in range(n - k):
            H[i, i+k] -= wk
            H[i+k, i] -= wk
    return H

def _qp_solve(h_current, weights, n, target, deadline, timeout=5.0):
    """Solve the QP relaxation to minimize weighted sum of overlaps."""
    x = cp.Variable(n)
    
    # Build the objective
    # overlap_k(x) = sum_{i=0}^{n-1-k} x[i] - x[i]*x[i+k]
    # Total objective = sum_k w_k * overlap_k(x)
    
    # Quadratic terms: -sum_k w_k * sum_i x[i]*x[i+k]
    # We need to build this efficiently
    
    # For efficiency, let's build the quadratic form matrix
    Q = np.zeros((n, n))
    for k in range(1, n):
        wk = weights[k]
        if abs(wk) < 1e-15:
            continue
        for i in range(n - k):
            Q[i, i+k] -= 0.5 * wk  # factor of 2 in x^T Q x
            Q[i+k, i] -= 0.5 * wk
    
    # Linear terms: sum_k w_k * sum_{i=0}^{n-1-k} x[i]
    lin = np.zeros(n)
    for k in range(1, n):
        wk = weights[k]
        if abs(wk) < 1e-15:
            continue
        for i in range(n - k):
            lin[i] += wk
    
    objective = cp.Minimize(cp.quad_form(x, cp.psd_wrap(Q)) + lin @ x)
    constraints = [x >= 0, x <= 1, cp.sum(x) == target]
    
    prob = cp.Problem(objective, constraints)
    
    t_start = time.time()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, eps=1e-6)
    except:

        except:

            except:
                return h_current, None

        return h_current, None
    
    if x.value is None:
        return h_current, None
    
    return np.clip(x.value, 0, 1), prob.value

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(48, max(24, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    return best_a, best_val

def _bin_local_opt(hb, deadline, max_iter=30000):

    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(12345)
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        nc = _exact_c5(cur)
        if nc < cc:
            ones[i] = oz
            zeros[j] = oi
            cc = nc
            if cc < bc:
                bc = cc
                bh = cur.copy()
            stall = 0
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return bh, bc

def _qp_iteration(h, deadline, n_critical=30, timeout=4.0):

    S = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(S))
    
    # Identify critical lags
    idx = np.argsort(-S)
    critical = idx[:n_critical]
    
    # Build weights: emphasize lags near the maximum
    weights = np.zeros(2 * n - 1)
    for ci in critical:
        lag = int(ci) - (n - 1)
        k = abs(lag)
        if k >= 1 and k < n:
            # Weight proportional to how close to max
            w = (S[int(ci)] - 0.5 * mx) ** 2
            weights[k] += w
