import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {0 <= x <= 1, sum(x) = target}
    # Using bisection on the threshold
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = v.copy()
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding: distribute residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
            if not mask.any():
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0)
            if not mask.any():
                mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smooth_weights(S0, beta):
    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    shifted = beta * (S0 - M)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    s = float(w.sum())
    if s < 1e-30:
        # Fallback: uniform over top entries
        k = max(1, len(S0) // 10)
        idx = np.argsort(S0)[-k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / k
    else:
        w /= s
    return w

def _grad_rows(h):
    """Compute the L×n Jacobian of the correlations w.r.t. h."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    omh = 1.0 - h
    for ell in range(-(n-1), n):
        row = ell + n - 1

def _line_search(S0, T, C, t_lo, t_hi):
    """Minimize max_k(S0[k] + t*T[k] - t^2*C[k]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    # Coarse grid
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    # Vectorized: shape (n_grid, L)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C))
    # Check endpoints and grid
    candidates = [(opt_t, opt_val), (float(ts[best_idx]), float(vals[best_idx])),
                  (t_lo, float(np.max(S0 + t_lo*T - t_lo*t_lo*C))),
                  (t_hi, float(np.max(S0 + t_hi*T - t_hi*t_hi*C)))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _dense_optimize(h, deadline, max_iter=10000, beta_start=5.0, beta_end=100.0, 
                    radius_start=0.3, radius_end=0.05, seed=42, stall_limit=200):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    radius = radius_start
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        S0 = np.correlate(h, 1.0 - h, mode='full')
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.8))
        w = _smooth_weights(S0, beta)
        
        # Compute gradient
        G = _grad_rows(h)
        grad = G.T @ w  # shape (n,)
        
        # Make sum-zero
        grad -= np.mean(grad)
        
        # Handle bounds: zero out gradient where it would push out of bounds
        d = -grad
        # Where h is near 0 and d < 0, or h is near 1 and d > 0, zero out
        d[(h < 1e-6) & (d < 0)] = 0.0
        d[(h > 1 - 1e-6) & (d > 0)] = 0.0
        
        # Re-make sum-zero after zeroing
        d_sum = d.sum()
        if abs(d_sum) > 1e-12:
            # Redistribute
            d -= d_sum / n
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            break
        d = d / d_norm  # unit direction
        
        # Feasible range
        pos_mask = d > 1e-10
        neg_mask = d < -1e-10
        t_max = radius
        if pos_mask.any():
            t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
        t_min = -radius
        if neg_mask.any():
            t_min = max(t_min, float(np.min(h[neg_mask] / (-d[neg_mask]))))
        
        if t_max - t_min < 1e-12:
            stall += 1
            radius *= 0.9
            continue
        
        # Compute T and C
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        # Line search
        t_opt, t_val = _line_search(S0, T, C, t_min, t_max)
        
        # Apply step
        h_try = h + t_opt * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _project(h_try, target)
        
        c5_try = _exact_c5(h_try)
        
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, radius_start)
        else:
            # Try smaller step
            t_half = t_opt * 0.5
            h_half = h + t_half * d
            h_half = np.clip(h_half, 0.0, 1.0)
            h_half = _project(h_half, target)
            c5_half = _exact_c5(h_half)
            if c5_half < cur_c5 - 1e-12:
                h = h_half
                cur_c5 = c5_half
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                radius *= 0.85
                if stall % 50 == 0:
                    # Restart with perturbation
                    pert = rng.normal(0, 0.01, n)
                    h = _project(h + pert, target)
                    cur_c5 = _exact_c5(h)
    
    return best_h, best_c5
