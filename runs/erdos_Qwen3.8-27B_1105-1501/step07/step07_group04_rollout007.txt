       """Compute change in corr when h[a] goes 1->0 and h[b] goes 0->1."""
       center = n - 1
       d = np.zeros(2*n-1)
       
       # Positive lags k=1..n-1, index center+k
       for k in range(1, n):
           idx = center + k
           val = 0.0
           if a >= k:
               val -= (1.0 - h[a-k])
           if a + k < n:
               val += h[a+k]
           if b >= k:
               val += (1.0 - h[b-k])
           if b + k < n:
               val -= h[b+k]
           d[idx] = val
       
       # Negative lags u=1..n-1, index center-u
       for u in range(1, n):
           idx = center - u
           val = 0.0
           if a + u < n:
               val -= (1.0 - h[a+u])
           if a >= u:
               val += h[a-u]
           if b + u < n:
               val += (1.0 - h[b+u])
           if b >= u:
               val -= h[b-u]
           d[idx] = val

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
