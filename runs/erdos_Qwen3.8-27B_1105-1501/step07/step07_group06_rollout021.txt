import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _raw_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr

def _block_cd_binary(h_bin, deadline, block_size=10, max_rounds=5000, rng=None):
    n = len(h_bin)
    cur = h_bin.copy()
    dx = 2.0 / n
    cur_max_raw, cur_raw = _raw_max(cur)
    best_h = cur.copy()
    best_max_raw = cur_max_raw
    if rng is None:
        rng = np.random.default_rng(777)
    n_blocks = n // block_size
    if n_blocks == 0:
        return best_h, best_max_raw * dx
    combos_cache = {}
    for b in range(n_blocks):
        k = int(np.sum(cur[b * block_size:(b + 1) * block_size]))
        key = (block_size, k)
        if key not in combos_cache:
            combos_cache[key] = list(combinations(range(block_size), k))
    stall = 0
    rnd = 0
    while time.time() < deadline and rnd < max_rounds:
        rnd += 1
        improved = False
        for b in range(n_blocks):
            if time.time() > deadline:
                break
            start = b * block_size
            end = start + block_size
            k = int(np.sum(cur[start:end]))
            key = (block_size, k)
            if key not in combos_cache:
                combos_cache[key] = list(combinations(range(block_size), k))
            best_block = cur[start:end].copy()
            best_block_max = cur_max_raw + 1.0
            for combo in combos_cache[key]:
                new_block = np.zeros(block_size)
                for p in combo:
                    new_block[p] = 1.0
                if new_block[0] == best_block[0] and new_block[-1] == best_block[-1]:
                    if np.array_equal(new_block, cur[start:end]):
                        continue
                cur[start:end] = new_block
                nm, _ = _raw_max(cur)
                if nm < best_block_max:
                    best_block_max = nm
                    best_block = new_block.copy()
            cur[start:end] = best_block
            if best_block_max < cur_max_raw - 1e-12:
                cur_max_raw = best_block_max
                improved = True
                if cur_max_raw < best_max_raw:
                    best_max_raw = cur_max_raw
                    best_h = cur.copy()
        if not improved:
            stall += 1
            if stall > 30:
                n_kick = int(rng.integers(2, min(6, n // 10) + 1))
                ones = np.where(cur == 1.0)[0]
                zeros = np.where(cur == 0.0)[0]
                for _ in range(n_kick):
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    cur[i] = 0.0
                    cur[j] = 1.0
                cur_max_raw, _ = _raw_max(cur)
                stall = 0
        else:
            stall = 0
    return best_h, best_max_raw * dx

            break

        n_try = min(100, n_ones * n_zeros)
        tried_ones = set()
        tried_zeros = set()
        for _ in range(n_try):
            oi = ones[rng.integers(n_ones)]
            oz = zeros[rng.integers(n_zeros)]
            if oi in tried_ones or oz in tried_zeros:
                continue
            tried_ones.add(oi)
            tried_zeros.add(oz)
            cur[oi] = 0.0
            cur[oz] = 1.0
            nm, _ = _raw_max(cur)
            cur[oi] = 1.0
            cur[oz] = 0.0
            if nm < best_new_max:
                best_new_max = nm
                best_pair = (oi, oz)
        if best_pair is not None and best_new_max < cur_max_raw:
            cur[best_pair[0]] = 0.0
            cur[best_pair[1]] = 1.0
            cur_max_raw = best_new_max
            stall = 0
            if cur_max_raw < best_max_raw:
                best_max_raw = cur_max_raw
                best_h = cur.copy()
        else:
            stall += 1
