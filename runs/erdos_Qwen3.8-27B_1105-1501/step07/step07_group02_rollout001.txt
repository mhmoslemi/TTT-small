    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S + opt_t * U - opt_t * opt_t * B))
    candidates = [opt_t, float(ts[best_idx]), t_lo, t_hi]
    candidate_vals = [opt_val, float(vals[best_idx]),
                      float(np.max(S + t_lo * U - t_lo * t_lo * B)),
                      float(np.max(S + t_hi * U - t_hi * t_hi * B))]
    bi = int(np.argmin(candidate_vals))
    return candidates[bi], candidate_vals[bi]

def _refine_dense(h, deadline, move_cap_init=0.4, seed=42, max_iters=20000):
    n = len(h)
    target_sum = n / 2.0
    cur = _project(h.copy(), target_sum)
    cur_S = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = float(np.max(cur_S))
    best = cur.copy()
    best_max = cur_max
    rng = np.random.default_rng(seed)
    move_cap = move_cap_init
    stall = 0
    k_schedule = [4, 6, 8, 10, 6, 12, 8, 5]
    k_idx = 0
    wmode_cycle = [0, 1, 0, 2, 0, 1]
    w_idx = 0
    iter_count = 0
    while time.time() < deadline and iter_count < max_iters:
        iter_count += 1
        if stall > 150:
            break
        S = np.correlate(cur, 1.0 - cur, mode='full')
        nL = len(S)
        active_K = k_schedule[k_idx % len(k_schedule)]
        k_idx += 1
        sorted_idx = np.argsort(S)[::-1]
        Extra = min(active_K + int(active_K * 0.5), nL - 1)
        top_rows = sorted_idx[:Extra]
        top_vals = S[top_rows]
        if np.std(top_vals) < 1e-6:
            target = float(np.max(top_vals)) * 0.98
        else:
            target = 0.5 * float(np.min(top_vals)) + 0.3 * float(np.mean(top_vals)) + 0.2 * (float(np.max(top_vals)) - 0.05 * float(np.std(top_vals)))
        wmode = wmode_cycle[w_idx % len(wmode_cycle)]
        w_idx += 1
        active_lags = []
        for ri in top_rows[:active_K]:
            lag = int(ri) - (n - 1)
            active_lags.append(lag)
        d = _dense_direction(cur, S, active_lags, target, wmode)
        if d is None or not np.all(np.isfinite(d)):
            stall += 1
            continue
        d_max = float(np.max(np.abs(d)))
        if d_max < 1e-12:
            stall += 1
            continue
        d = d / d_max
        t_opt, pred_val = _line_search(cur, S, d, move_cap)
        if abs(t_opt) < 1e-12:
            stall += 1
            continue
        trial = cur + t_opt * d
        if np.any(trial < -1e-9) or np.any(trial > 1.0 + 1e-9):
            trial = _project(trial, target_sum)
        trial_S = np.correlate(trial, 1.0 - trial, mode='full')
        trial_max = float(np.max(trial_S))
        if trial_max < best_max - 1e-12:
            best = trial.copy()
            best_max = trial_max
            cur = trial.copy()
            stall = 0
            move_cap = min(move_cap * 1.01, 0.5)
        else:
            stall += 1
            if stall % 20 == 0:
                move_cap = max(move_cap * 0.85, 0.05)
            if stall > 50:
                active_K_new = min(active_K + 3, 20)
                k_schedule[k_idx % len(k_schedule)] = active_K_new
    return best, float(best_max * 2.0 / n)

def _refine_pair(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=100, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(S0))
    w = np.exp(beta * (S0 - M))
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.ones(len(S0)) / len(S0)
    else:
        w /= ws
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.ones(len(S0)) / len(S0)
        else:
            w /= ws
        one_mh = 1.0 - h
        coord_grad = np.zeros(n)
        for k in range(-(n - 1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                grad_k = np.zeros(n)
                grad_k[:m] += (1.0 - h[k:])
                grad_k[k:] -= h[:m]
            else:
                p = -k
                m = n - p
                grad_k = np.zeros(n)
                grad_k[p:] += (1.0 - h[:m])
                grad_k[:m] -= h[p:]
            coord_grad += w[row] * grad_k
        sort_idx = np.argsort(coord_grad)
        npairs = min(n_pairs, n // 2)
        sources = sort_idx[-npairs:]
        sinks = sort_idx[:npairs]
        one_mh = 1.0 - h
        best_step = None
        best_c5_try = cur_c5
        for ii in range(npairs):
            if time.time() > deadline:
                break
            sink = int(sinks[ii])
            source = int(sources[ii])
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T_vec = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C_vec = np.correlate(d, d, mode='full')
            n_grid = 40
            ts = np.linspace(t_lo, t_hi, n_grid)
            vals = np.max(S0[None, :] + ts[:, None] * T_vec[None, :] - (ts[:, None]) ** 2 * C_vec[None, :], axis=1)
            bi = int(np.argmin(vals))
            t_opt = float(ts[bi])
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.02, 0.28)
        else:
            stall += 1
            radius *= 0.78
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_swap_sweep(h_bin, deadline, max_rounds=5):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in ones[:]:
            if time.time() > deadline:
                break
            for oz in zeros[:]:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, AttributeError):
        h0 = None
    if h0 is None or not np.all(np.isfinite(h0)) or len(h0) < 10:
        n0 = 160
        h0 = np.full(n0, 0.5)
        h0[:n0 // 2] = 1.0
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Phase 1: Dense direction optimization (50% time)
    d1 = t0 + budget_s * 0.50
    h1, c5_1 = _refine_dense(h_init, d1, move_cap_init=0.4, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: Pairwise refinement (20% time)
    d2 = t0 + budget_s * 0.70
    h2, c5_2 = _refine_pair(best_h, d2, radius=0.12, n_pairs=50, beta=22.0, stall_limit=120, seed=seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Phase 3: Multi-n exploration (15% time)
    d3 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _refine_dense(h_r, sub_dl, move_cap_init=0.3, seed=seed + 100 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    # Phase 4: Binary sweep (8% time)
    d4 = t0 + budget_s * 0.93
    n_curr = len(best_h)
    if time.time() < d4:
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_swap_sweep(h_bin, sub_dl, max_rounds=3)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    # Phase 5: Final polish (7% time)
    d5 = t0 + budget_s * 0.99
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d5, deadline - 1.0)
    h5, c5_5 = _refine_dense(h5, sub_dl, move_cap_init=0.2, seed=seed + 500)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
