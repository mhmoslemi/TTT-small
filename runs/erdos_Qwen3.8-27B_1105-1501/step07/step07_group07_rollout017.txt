def _push_relabel(n, s, t, edges):
    """
    n: number of nodes
    s: source, t: sink
    edges: list of (u, v, capacity)
    Returns (max_flow, min_cut_set) where min_cut_set is the set of nodes reachable from s in residual graph
    """
    INF = float('inf')
    
    # Build adjacency with residual capacities
    # adj[u] = list of [v, capacity, reverse_index]
    adj = [[] for _ in range(n)]
    for i, (u, v, cap) in enumerate(edges):
        adj[u].append([v, cap, len(adj[v])])
        adj[v].append([u, 0.0, len(adj[u]) - 1])
    
    height = [0] * n
    excess = [0.0] * n
    height[s] = n
    
    # Saturate all edges from source
    for e in adj[s]:
        v, cap, rev = e
        if cap > 0:
            excess[v] += cap
            excess[s] -= cap
            e[1] = 0
            adj[v][rev][1] += cap
    
    # Active nodes (nodes with excess > 0, excluding s and t)
    active = set()
    for v in range(n):
        if v != s and v != t and excess[v] > 0:
            active.add(v)
    
    # Discharge
    count_height = [0] * (2 * n + 1)
    count_height[0] = n - 2
    count_height[n] = 1  # source
    
    def relabel(u):
        min_h = INF
        for v, cap, rev in adj[u]:
            if cap > 0:
                min_h = min(min_h, height[v])
        if min_h < INF:
            height[u] = min_h + 1
    
    def push(u, e_idx):
        v, cap, rev = adj[u][e_idx]
        amt = min(excess[u], cap)
        adj[u][e_idx][1] -= amt
        adj[v][rev][1] += amt
        excess[u] -= amt
        excess[v] += amt
        if v != s and v != t and excess[v] > 0 and height[v] > 0:
            active.add(v)
    
    while active:
        # Pick highest height active node
        u = max(active, key=lambda x: height[x])
        active.remove(u)
        
        # Try to push
        pushed = False
        for i in range(len(adj[u])):
            if excess[u] <= 0:
                break
            v, cap, rev = adj[u][i]
            if cap > 0 and height[u] == height[v] + 1:
                push(u, i)
                pushed = True
                if excess[u] > 0 and v != s and v != t:
                    active.add(v)
        
        if not pushed and excess[u] > 0:
            # Relabel
            old_h = height[u]
            relabel(u)
            if height[u] != old_h:
                count_height[old_h] -= 1
                count_height[height[u]] += 1
                active.add(u)
    
    # Find min cut: BFS from source in residual graph
    visited = [False] * n
    visited[s] = True
    queue = [s]
    while queue:
        u = queue.pop(0)
        for v, cap, rev in adj[u]:
            if cap > 0 and not visited[v]:
                visited[v] = True
                queue.append(v)
    
    max_flow = excess[t]
    return max_flow, visited

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

def _maxflow(n_nodes, s, t, edges):
    """Push-relabel max flow. Returns (flow, reachable_set_from_s)."""
    adj = [[] for _ in range(n_nodes)]
    for u, v, c in edges:
        if c > 0:
            adj[u].append([v, c, len(adj[v])])
            adj[v].append([u, 0.0, len(adj[u]) - 1])
    
    hgt = [0] * n_nodes
    exc = [0.0] * n_nodes
    hgt[s] = n_nodes
    
    for e in adj[s]:
        v, c, rev = e
        if c > 0:
            exc[v] += c
            exc[s] -= c
            e[1] = 0.0
            adj[v][rev][1] += c
    
    # Use a list as stack for active nodes (highest label first would be better but this works)
    active = []
    in_act = [False] * n_nodes
    for v in range(n_nodes):
        if v != s and v != t and exc[v] > 0:
            active.append(v)
            in_act[v] = True
    
    while active:
        u = active.pop()
        in_act[u] = False
        eu = exc[u]
        
        # Discharge
        i = 0
        while i < len(adj[u]) and exc[u] > 0:

                adj[u][i][1] -= amt
                adj[v][rev][1] += amt
                exc[u] -= amt
                exc[v] += amt
                if v != s and v != t and not in_act[v] and exc[v] > 0:
                    active.append(v)
                    in_act[v] = True

                i += 1

                    min_h = min(min_h, hgt[v])

                hgt[u] = min_h + 1
            active.append(u)
            in_act[u] = True

            if not reached[i + 1]:
                x[i] = 1.0
