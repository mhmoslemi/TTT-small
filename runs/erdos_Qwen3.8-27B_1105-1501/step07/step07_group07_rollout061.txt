def _lse_computation(scores, tau):
    """Compute LSE value and weights."""
    s = scores  # already scaled by dx
    mval = float(np.max(s))
    u = (s - mval) / tau
    # Clip for numerical stability
    u = np.clip(u, -500, 500)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        # All underflow, use uniform on max
        w = np.zeros_like(s)
        w[s == mval] = 1.0 / np.sum(s == mval)
        return mval, w
    w = z / zsum
    lse = mval + tau * np.log(zsum)
    return float(lse), w

def _exact_c5(h):

def _project_sum(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        G[mid + k, k:] += (1.0 - h[:m])
        G[mid + k, :m] += -h[k:]
    for m in range(1, n):
        G[mid - m, :n - m] += (1.0 - h[m:])
        G[mid - m, m:] += -h[:n - m]
    return G

def _lse_val_weights(s, tau):
    s = np.array(s, dtype=float)
    mval = float(np.max(s))
    u = (s - mval) / tau
    u = np.clip(u, -400.0, 400.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        w = np.zeros_like(s)
        idx = np.where(np.isclose(s, mval, atol=1e-12))[0]
        if len(idx) > 0:
            w[idx] = 1.0 / len(idx)
        return mval, w
    w = z / zsum
    val = mval + tau * np.log(zsum)
    return float(val), w

def _power_mean_val_weights(s, p):
    s = np.array(s, dtype=float)
    s_clip = np.clip(s, 0.0, None)
    if float(np.max(s_clip)) < 1e-30:
        w = np.full(len(s), 1.0 / len(s))
        return 0.0, w
    sc = s_clip / float(np.max(s_clip))
    sp = sc ** p
    spsum = float(np.sum(sp))
    if spsum < 1e-300:
        w = np.zeros_like(s)
        idx = np.where(s_clip == float(np.max(s_clip)))[0]
        w[idx] = 1.0 / len(idx)
        return float(np.max(s_clip)), w
    w = sp / spsum
    val = float(np.max(s_clip)) * (spsum / len(s)) ** (1.0 / p)
    return val, w

def _slsqp_stage(h, tau, deadline, maxiter=80):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

        h_p = _perturb(best_h, rng, scale)
        sub_dl = min(d6, time.time() + budget_s * 0.003)
        h_t, c5_t = _lse_anneal(h_p, sub_dl, n_stages=3, maxiter=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.08)

    # Phase 7: Final polish (5%)
    d7 = deadline
    n_f = len(best_h)
    for dn in [-4, 4, -8, 8, -2, 2]:
        if time.time() > d7 - 4:
            break
        n_try = n_f + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = _lse_anneal(h_r, sub_dl, n_stages=3, maxiter=40)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_f = n_try

    # Final LSE polish
    if deadline - time.time() > 5:
        h_f = _project_sum(best_h.copy(), len(best_h) // 2.0)
        h_f, c5_f = _lse_anneal(h_f, deadline, tau_start=0.015, tau_end=0.001, n_stages=4, maxiter=80)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final pair sweep if time
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, deadline, max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
