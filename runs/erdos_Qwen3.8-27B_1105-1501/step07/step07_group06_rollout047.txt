import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, tau):
    n = len(h)
    g = np.zeros(n)
    if tau == 0:
        g[:] = 1.0 - 2.0 * h
    elif tau > 0:
        m = n - tau
        g[:m] -= h[tau:tau+m]
        g[tau:tau+m] += 1.0 - h[:m]
    else:
        d = -tau
        m = n - d
        g[:m] += 1.0 - h[d:d+m]
        g[d:d+m] -= h[:m]
    return g

def _grad_smooth(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        return _lag_grad(h, idx - (n - 1))
    w /= wsum
    active = w > 1e-8
    if not np.any(active):
        idx = int(np.argmax(raw))
        return _lag_grad(h, idx - (n - 1))
    g = np.zeros(n)
    for r in np.where(active)[0]:
        g += w[r] * _lag_grad(h, r - (n - 1))
    g -= float(np.mean(g))
    return g

def _grad_topk(h, raw, k=3):
    n = len(h)
    top_idx = np.argsort(raw)[::-1][:k]
    g = np.zeros(n)
    for r in top_idx:
        g += _lag_grad(h, r - (n - 1))
    g /= k
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-6:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo - 1e-12 <= av <= a_hi + 1e-12:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _opt_continuous(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.15, max_stall=60, use_topk=False, topk=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    rng = np.random.default_rng(12345 + n)
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        it += 1
        frac = float(it) / float(it + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 1.5)
        
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        
        if use_topk and it % 3 == 0:
            g = _grad_topk(h, raw, topk)
        else:
            g = _grad_smooth(h, raw, beta)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        d = -g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        
        # Also try a few alternative directions
        dirs = [d]
        if use_topk:
            g2 = _grad_topk(h, raw, max(2, topk - 1))
            g2n = np.linalg.norm(g2)
            if g2n > 1e-14:
                d2 = -g2 / g2n
                d2 -= np.mean(d2)
                d2n = np.linalg.norm(d2)
                if d2n > 1e-14:
                    dirs.append(d2 / d2n)
        
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for dd in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, dd, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, dd
        
        if best_d is not None and abs(best_alpha) > 1e-15:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.03, radius * 0.70)
        else:
            stall += 1
            radius = max(0.03, radius * 0.75)
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_block=3):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    stall = 0
    
    while time.time() < deadline and stall < 50:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            best_oz = None
            best_val = cur_c5
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c = _exact_c5(cur)
                cur[oi] = 1.0
                cur[oz] = 0.0
                if c < best_val - 1e-14:
                    best_val = c
                    best_oz = oz
            if best_oz is not None:
                cur[oi] = 0.0
                cur[best_oz] = 1.0
                cur_c5 = best_val
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                improved = True
                break
        if not improved:
            # Try block swaps
            block_improved = False
            for L in range(2, max_block + 1):
                if time.time() > deadline:
                    break
                for i in range(0, n - L + 1):
                    if time.time() > deadline:
                        break
                    if cur[i:i+L].sum() != L:

                    for j in range(0, n - L + 1):
                        if abs(j - i) < L:
                            continue
                        if cur[j:j+L].sum() != 0:
                            continue
                        cur[i:i+L] = 0.0
                        cur[j:j+L] = 1.0
                        c = _exact_c5(cur)
                        if c < cur_c5 - 1e-14:
                            cur_c5 = c
                            if c < best_c5:
                                best_c5 = c
                                best = cur.copy()
                            ones = np.where(cur > 0.5)[0].tolist()
                            zeros = np.where(cur <= 0.5)[0].tolist()
                            block_improved = True
                            break
                        else:
                            cur[i:i+L] = 1.0
                            cur[j:j+L] = 0.0
                    if block_improved:
                        break
                if block_improved:
                    break
            if not improved and not block_improved:
                stall += 1
    
    return best, best_c5

def _binary_sa(h_bin, deadline, T_init=0.003, T_decay=0.9998, max_stall=300):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    while time.time() < deadline:
        if stall > max_stall or len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best = cur.copy()
                stall = 0
            T *= T_decay
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, pattern_type, rng):
    """Generate structured initial patterns."""
    h = np.zeros(n)
    if pattern_type == 0:
        # Alternating blocks of size p
        p = max(2, n // 10)
        for i in range(n):
            block = (i // p) % 2
            h[i] = 1.0 if block == 0 else 0.0
    elif pattern_type == 1:
        # Single period
        period = max(2, n // 4)
        duty = period // 2
        for i in range(n):
            h[i] = 1.0 if (i % period) < duty else 0.0
    elif pattern_type == 2:
        # Two-scale pattern
        p1 = max(2, n // 8)
        p2 = max(1, n // 32)
        for i in range(n):
            if (i // p1) % 2 == 0:
                h[i] = 1.0 if (i % p2) < p2 // 2 else 0.0
            else:
                h[i] = 0.5
    elif pattern_type == 3:
        # Random
        idx = rng.permutation(n)[:n // 2]
        h[idx] = 1.0
    else:
        # Gradient-like
        h[:n // 2] = 1.0
        h[n // 2:] = 0.0
    
    return _project(h, n / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for k in range(4):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

    n_final = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > deadline - 10:
            break
        n_try = n_final + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.012)
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _opt_continuous(h_r, sub_dl, beta_lo=15.0, beta_hi=70.0, radius_init=0.06, max_stall=30, use_topk=True, topk=3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_continuous(h_f, deadline, beta_lo=25.0, beta_hi=100.0, radius_init=0.04, max_stall=30, use_topk=True, topk=4)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5  # Use the verified value
    
    return best_h, best_c5, len(best_h)
