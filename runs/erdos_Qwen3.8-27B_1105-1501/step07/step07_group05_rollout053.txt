    a_hi = radius

    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)

    cands = [a_lo, a_hi, 0.0, (a_lo + a_hi) / 2.0]
    width = a_hi - a_lo

    for k in range(len(S0)):

            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:

        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
        for _ in range(20):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)

        mid = (lo_r + hi_r) / 2.0

        if fmid < best_val:
            best_val = fmid
            best_a = mid

def _lp_direction(h, top_k, r, deadline):
    """Find direction using LP to minimize max of top active constraints."""
    import cvxpy as cp
    
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top K active lags
    top_idx = np.argsort(-raw)[:top_k]
    
    # Build gradient matrix for these lags
    # g[t, j] = d(raw_t)/dh[j]
    G = np.zeros((top_k, n))
    for idx, t in enumerate(top_idx):
        lag = t - (n - 1)
        if lag >= 0:
            L = n - lag
            if L > 0:
                G[idx, :L] += (1.0 - h[lag:lag+L])
                G[idx, lag:lag+L] -= h[:L]
        else:
            s = -lag
            L = n - s
            if L > 0:
                G[idx, s:s+L] += (1.0 - h[:L])
                G[idx, :L] -= h[s:s+L]
    
    # Normalize rows
    norms = np.linalg.norm(G, axis=1, keepdims=True)
    norms[norms < 1e-12] = 1.0
    G = G / norms
    
    # LP: maximize tau
    # s.t. G @ s + tau <= 0 (for each row)
    #      sum(s) = 0
    #      -r*h_i <= s_i <= r*(1-h_i)
    #      tau >= 0
    
    s = cp.Variable(n)
    tau = cp.Variable()
    
    lo_bound = -r * h
    hi_bound = r * (1.0 - h)
    
    constraints = [
        cp.sum(s) == 0,
        s >= lo_bound,
        s <= hi_bound,
        tau >= 0,
        G @ s + tau <= 0
    ]
    
    prob = cp.Problem(cp.Maximize(tau), constraints)
    
    try:
        prob.solve(solver=cp.CBC, timeout=5.0)
    except:
        try:
            prob.solve(solver=cp.CVXOPT)
        except:
            return None
    
    if s.value is None:
        return None
    
    s_val = np.array(s.value).flatten()
    if not np.all(np.isfinite(s_val)):
        return None
    
    s_val -= np.mean(s_val)  # Ensure exactly zero mean
    s_val = np.clip(s_val, lo_bound, hi_bound)
    s_val -= np.mean(s_val)
    
    nrm = np.linalg.norm(s_val)
    if nrm < 1e-12:
        return None
    
    return s_val / nrm

    try:
        import cvxpy as cp
    except:
        return None

    top_idx = np.argsort(-raw)[:top_k]

        if lag >= 0:
            L = n - lag
            if L > 0:
                G[idx, :L] += (1.0 - h[lag:lag+L])
                G[idx, lag:lag+L] -= h[:L]
        else:

            L = n - s
            if L > 0:
                G[idx, s:s+L] += (1.0 - h[:L])
                G[idx, :L] -= h[s:s+L]

    norms[norms < 1e-12] = 1.0
    G = G / norms
    
    s = cp.Variable(n)
    tau = cp.Variable()
    lo_bound = -r * h
    hi_bound = r * (1.0 - h)
    
    constraints = [
        cp.sum(s) == 0,
        s >= lo_bound,
        s <= hi_bound,
        tau >= -1e-10,
        G @ s + tau <= 0
    ]
    
    prob = cp.Problem(cp.Maximize(tau), constraints)
    try:
        prob.solve(solver=cp.CBC, timeout=3.0)
    except:
        try:
            prob.solve(solver=cp.CVXOPT)
        except:
            return None
    
    if s.value is None or tau.value is None:
        return None
    if tau.value < 1e-8:
        return None
    
    s_val = np.array(s.value).flatten()
    if not np.all(np.isfinite(s_val)):
        return None
    s_val -= np.mean(s_val)
    nrm = np.linalg.norm(s_val)
    if nrm < 1e-12:
        return None
    return s_val / nrm
