```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def _max_flow(n, cap, src, sink):
    """
    Push-relabel max flow.
    n: number of nodes
    cap: (n, n) numpy array of capacities (residual capacities)
    src, sink: source and sink node indices
    Returns: (max_flow, reachable) where reachable is boolean array for min-cut
    """
    height = np.zeros(n, dtype=np.int32)
    height[src] = n
    excess = np.zeros(n)
    
    # Initialize preflow
    for j in range(n):
        if cap[src, j] > 0:
            f = cap[src, j]
            excess[j] += f
            excess[src] -= f  # not really needed
            cap[src, j] = 0
            cap[j, src] += f
    
    # Active nodes (height > 0, excess > 0, not src or sink)
    # Use a simple list as queue
    active = []
    for j in range(n):
        if j != src and j != sink and excess[j] > 0:
            active.append(j)
    
    while active:
        u = active.pop()
        old_height = height[u]
        # Try to push from u
        # Find neighbors v where cap[u,v] > 0 and height[u] == height[v] + 1
        neighbors = np.where(cap[u, :] > 0)[0]
        for v in neighbors:
            if excess[u] <= 0:
                break
            if height[u] == height[v] + 1:
                f = min(excess[u], cap[u, v])
                cap[u, v] -= f
                cap[v, u] += f
                excess[u] -= f
                excess[v] += f
                if v != src and v != sink and excess[v] == f:  # just became active
                    active.append(v)
        
        if excess[u] > 0 and height[u] == old_height:
            # Relabel
            min_h = np.where(cap[u, :] > 0)[0]
            if len(min_h) > 0:
                height[u] = int(np.min(height[min_h])) + 1
                active.append(u)
    
    max_flow = excess[sink]
    
    # Find reachable nodes from src in residual graph
    reachable = np.zeros(n, dtype=bool)
    reachable[src] = True
    changed = True
    while changed:
        changed = False
        for i in range(n):
            if reachable[i]:
                nbrs = np.where(cap[i, :] > 0)[0]
                for j in nbrs:
                    if not reachable[j]:
                        reachable[j] = True
                        changed = True
    
    return max_flow, reachable
```

For a graph where node i only connects to nodes in {i-L, ..., i+L}, the min-cut can be found by processing nodes from left to right, keeping track of the last L nodes' states.

State at position i: (x_{i-L+1}, ..., x_i) - the states of the last L nodes.
Transition: add x_{i+1}, remove x_{i-L+1}.

The cost of a state depends on all edges between the current window and previous nodes.

This is a standard "dp on a path decomposition" and runs in O(n * 2^L * L) time.
