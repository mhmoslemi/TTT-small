def _dense_grad_step(h, k_active, beta, step_scale=0.01, deadline=None):
    """Dense gradient step: move all coordinates in direction that reduces top-k active correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * dx
    L = 2 * n - 1
    
    # Find top-k active lags
    top_k = np.argsort(corr)[-k_active:]
    
    # Compute weighted gradient
    # Weights: exponential of (corr - max) * beta, normalized
    c_max = float(np.max(corr))
    if c_max < 1e-15:
        return h, c5
    w = np.exp(beta * (corr[top_k] - c_max))
    w /= float(np.sum(w))
    
    # Compute gradient for each active lag and sum with weights
    G = np.zeros(n)
    for idx, m in enumerate(top_k):
        lag = m - (n - 1)
        g = np.zeros(n)

            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
        else:
            p = -lag
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
        G += w[idx] * g
    
    # Direction: negative gradient (we want to decrease the weighted sum)
    d = -G
    # Project direction to be orthogonal to all-ones (to preserve mass)
    d = d - np.mean(d)

    best_h = h.copy()
    best_c5 = c5
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-15:

            best_h = h_try
        if deadline is not None and time.time() > deadline:

    L = 2 * n - 1
    
    # softmax weights
    scaled = beta * corr * dx
    max_scaled = float(np.max(scaled))
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / float(np.sum(exp_vals))  # shape (L,)
    
    # smooth objective
    smooth_obj = (max_scaled + float(np.log(np.sum(exp_vals)))) / beta
    
    # gradient: dx * sum_m w[m] * dc[m]/dh
    G = np.zeros(n)
    for m in range(L):
        if w[m] < 1e-8:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            G[lag:] += w[m] * (1.0 - h[:n-lag])
            G[:n-lag] -= w[m] * h[lag:]
        else:
            p = -lag
            G[:n-p] += w[m] * (1.0 - h[p:])
            G[p:] -= w[m] * h[:n-p]
    G *= dx
    
    return smooth_obj, G

def _dense_gradient_descent(h, n_iters, beta_start, beta_end, step_start, step_end, deadline, stall_limit=20):
    """Dense gradient descent on the smooth-max approximation."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        
        # Adaptive beta and step size
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        step = step_start + (step_end - step_start) * frac
        
        _, G = _smooth_max_grad(h, beta)
        
        # Project gradient to tangent space (orthogonal to 1)
        G_proj = G - np.mean(G)
        norm_g = np.linalg.norm(G_proj)
        if norm_g < 1e-12:
            stall += 20
            continue
        
        # Multiple step size attempts
        best_h_try = h.copy()
        best_c5_try = cur_c5
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = step * scale
            h_try = _project(h - alpha * G_proj, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_h_try = h_try
        
        if best_c5_try < best_c5 - 1e-14:
            h = best_h_try
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            h = best_h_try  # still move in the direction, even if slightly worse
            cur_c5 = _exact_c5(h)
            stall += 1
            step *= 0.7
    
    return best_h, best_c5

Let me think about whether this approach will actually work. The issue is that the smooth max is a convex function (it's a composition of the convex log-sum-exp with the bilinear c[m], which is NOT convex). So the gradient of the smooth max is not a guaranteed descent direction for the original max.

Actually, c[m] = sum_i h[i+lag]*(1-h[i]) is bilinear (not convex or concave in general). So the smooth max of bilinear functions is non-convex. The gradient will give a local descent direction but we might get stuck.

The key advantage over pair swaps: the dense gradient moves ALL coordinates at once, which can help when multiple coordinates need to adjust simultaneously. Pair swaps are limited to 2-coordinate moves.

import numpy as np

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_c5_corr(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _get_grad_at_lag(h, lag):
    """Gradient of correlation at a specific lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        p = -lag
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def _weighted_grad(h, corr, beta=50.0, k_top=30):
    """Compute weighted gradient using top-k active correlations with softmax weights."""
    n = len(h)
    L = 2 * n - 1
    top_idx = np.argsort(corr)[-k_top:]
    c_max = float(np.max(corr))
    w = np.exp(beta * (corr[top_idx] - c_max))
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum
    G = np.zeros(n)
    dx = 2.0 / n
    for i, m in enumerate(top_idx):
        lag = m - (n - 1)
        G += w[i] * _get_grad_at_lag(h, lag)
    return G * dx

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        p = -lag
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def _active_grad(h, corr, beta, k_top=50):
    """Weighted gradient of top-k active correlations."""
    n = len(h)
    L = 2 * n - 1
    top = np.argsort(corr)[-min(k_top, L):]
    cmax = float(np.max(corr))
    if cmax < 1e-15:
        return np.zeros(n)
    w = np.exp(beta * (corr[top] - cmax))
    s = float(np.sum(w))
    if s < 1e-30:
        w = np.ones(len(top)) / len(top)
    else:
        w /= s
    G = np.zeros(n)
    for i, m in enumerate(top):
        if w[i] < 1e-10:
            continue
        lag = m - (n - 1)
        G += w[i] * _grad_lag(h, lag)
    return G

def _pair_line_search(h, S0, sink, source, radius, target):
    """Line search for a pair swap: move t mass from source to sink."""
    n = len(h)
    t_lo = max(-h[sink], h[source] - 1.0, -radius)
    t_hi = min(1.0 - h[sink], h[source], radius)
    if t_hi - t_lo < 1e-13:
        return None, None
    d = np.zeros(n)
    d[sink] = 1.0
    d[source] = -1.0
    one_mh = 1.0 - h
    T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    # Minimize max_k (S0[k] + t*T[k] - t^2*C[k]) over t in [t_lo, t_hi]
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None,:] + ts[:,None]*T[None,:] - (ts[:,None])**2*C[None,:], axis=1)
    best_idx = int(np.argmin(vals))
    # Golden section refinement
    lo = ts[max(0, best_idx-1)]
    hi = ts[min(n_grid-1, best_idx+1)]
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))
    f2 = float(np.max(S0 + c2*T - c2*c2*C))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))
    t_opt = (lo+hi)/2.0
    val_opt = float(np.max(S0 + t_opt*T - t_opt*t_opt*C))
    # Check endpoints and grid best
    best_val = min(val_opt, float(vals[best_idx]),
                   float(np.max(S0+t_lo*T-t_lo*t_lo*C)),
                   float(np.max(S0+t_hi*T-t_hi*t_hi*C)))
    return t_opt, best_val

def _pair_refine(h, deadline, radius=0.12, n_pairs=40, beta=25.0, stall_limit=80, seed=42):
    """Pair-based refinement (similar to previous code but simplified)."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5, S0 = _exact_c5_corr(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        # Identify good pair moves using gradient
        w = np.exp(beta * (S0 - float(np.max(S0))))
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.ones(len(S0)) / len(S0)
        else:
            w /= ws
        # Gradient for coordinate selection
        dx = 2.0 / n
        G = _active_grad(h, S0, beta=beta, k_top=30)
        sort_idx = np.argsort(G)
        sources = sort_idx[-min(n_pairs, n//3):]
        sinks = sort_idx[:min(n_pairs, n//3)]
        pairs = []
        for ii in range(min(n_pairs, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        # Add random pairs
        for _ in range(min(10, n_pairs//4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        
        best_step = None
        best_c5_try = cur_c5
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_opt, _ = _pair_line_search(h, S0, sink, source, radius, target)
            if t_opt is not None:
                h_try = h.copy()
                h_try[sink] += t_opt
                h_try[source] -= t_opt
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5_try - 1e-14:
                    best_c5_try = c5_try
                    best_step = h_try
        if best_step is not None:
            h = best_step
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0-h, mode='full')
            stall = 0
            radius = min(radius * 1.01, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _dense_grad_run(h, deadline, n_iters=200, beta_start=15.0, beta_end=80.0,
                     step_start=0.02, step_end=0.001, k_top=40):
    """Dense gradient descent on smooth max."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    n_eff = min(n_iters, 500)
    
    for it in range(n_eff):
        if time.time() > deadline:
            break
        if stall > 30:
            # Reheat
            stall = 0
            beta_start = 10.0
            continue
        frac = it / max(1, n_eff - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        step = step_start + (step_end - step_start) * frac
        
        c5_curr, corr = _exact_c5_corr(h)
        G = _active_grad(h, corr, beta=beta, k_top=k_top)
        # Project to mass-preserving direction
        G = G - np.mean(G)
        norm_g = np.linalg.norm(G)
        if norm_g < 1e-14:
            stall += 10
            continue
        
        best_h_try = h.copy()
        best_c5_try = cur_c5
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = _project(h - step * scale * G / max(norm_g, 1e-10), target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_h_try = h_try
        
        if best_c5_try < cur_c5 - 1e-15:
            h = best_h_try
            cur_c5 = best_c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n//2:]

def _binary_local_search(h_bin, deadline, max_iter=15000, seed=999):
    """Binary local search: swap a 1 with a 0 if it improves C5."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5, _ = _exact_c5_corr(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        # Try a random swap
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        i, j = ones[oi], zeros[oz]
        h_try = cur.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        c_try = _exact_c5(h_try)
        if c_try < cur_c5 - 1e-15:
            cur = h_try
            cur_c5 = c_try
            ones[oi] = j  # i becomes 0, j becomes 1
            zeros[oz] = i
            if c_try < best_c5:
                best_c5 = c_try
                best_h = cur.copy()
    
    return best_h, best_c5
