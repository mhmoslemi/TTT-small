def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

            x[mask] += res / cnt

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, T=10.0, top_k=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_corr = corr[top_idx]
    max_tc = float(np.max(top_corr))
    exps = np.exp(np.clip(T * (top_corr - max_tc), -500, 500))
    sw = float(np.sum(exps))
    if sw < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = exps / sw
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        s = int(ki) - (n - 1)
        if s > 0:
            grad[:n - s] += wi * (1.0 - h[s:n])
            grad[s:] -= wi * h[:n - s]
        elif s == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            t = -s
            grad[t:] += wi * (1.0 - h[:n - t])
            grad[:n - t] -= wi * h[t:]
    return grad, float(np.max(corr))

def _subgrad_active(h, top_k=3):

    return grad, float(np.max(corr))

def _grad_descent(h, deadline, T_init=8.0, T_decay=0.9997, lr_init=0.004,
                  top_k=20, max_iters=80000, switch_to_subgrad=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    T = T_init
    stall = 0

        if time.time() > deadline:
            break
        if stall > 150:
            break
        use_sub = (it > switch_to_subgrad)

            grad, _ = _subgrad_active(h, top_k=max(3, top_k // 5))
        else:
            grad, _ = _smooth_grad(h, T=T, top_k=top_k)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        d = grad / gnorm
        best_h_new = h.copy()
        best_c5_new = cur_c5

def _fix_binary_count(h_bin, n):
    h_bin = h_bin.copy()
    target = n // 2
    cur_ones = int(np.sum(h_bin > 0.5))
    if cur_ones == target:
        return h_bin
    if cur_ones > target:
        extra = cur_ones - target
        one_idx = np.where(h_bin > 0.5)[0]
        to_remove = one_idx[-extra:]
        h_bin[to_remove] = 0.0
    else:
        need = target - cur_ones
        zero_idx = np.where(h_bin <= 0.5)[0]
        to_add = zero_idx[:need]
        h_bin[to_add] = 1.0
    return h_bin

def _binary_correlation_delta(h_bin, i, j):
    n = len(h_bin)
    g = np.zeros(2 * n - 1)
    oi = i
    oz = j
    for dd in range(0, n - oi):
        g[n - 1 + dd] = (1.0 - h_bin[oi + dd]) * (-1.0)
    for dd in range(-oi, 0):
        g[n - 1 + dd] = -h_bin[oi + dd] * (-1.0)
    for dd in range(0, n - oz):
        g[n - 1 + dd] += (1.0 - h_bin[oz + dd])
    for dd in range(-oz, 0):
        g[n - 1 + dd] += -h_bin[oz + dd]
    return g

            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(rng.integers(0, len(ones)))
        oz = int(rng.integers(0, len(zeros)))
        i = int(ones[oi])
        j = int(zeros[oz])

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    f1 = rng.uniform(0.5, 5.0)
    p1 = rng.uniform(0, 2 * np.pi)
    f2 = rng.uniform(1.0, 8.0)
    p2 = rng.uniform(0, 2 * np.pi)
    perturb = amplitude * (0.6 * np.sin(2.0 * np.pi * f1 * x / n + p1) +
                           0.4 * np.cos(2.0 * np.pi * f2 * x / n + p2))
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

        sub_dl = min(d5, time.time() + 8)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iters=15000, stall_limit=80)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Quick continuous
        h_c = _project(h_b.copy(), n_try / 2.0)
        sub_dl2 = min(d5, time.time() + 5)
        h_c, c5_c = _grad_descent(h_c, sub_dl2, T_init=6.0, lr_init=0.003, top_k=15, switch_to_subgrad=50)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 6: Intensive final optimization (25%)
    d6 = deadline
    n_final = len(best_h)
    # LP
    sub_dl = min(d6, time.time() + budget_s * 0.06)
    h6, c5_6 = _lp_linearize(best_h, sub_dl, top_k=15, radius=0.05, stall_limit=60)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Gradient descent
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6, c5_6 = _grad_descent(best_h, sub_dl, T_init=3.0, lr_init=0.002, top_k=30, switch_to_subgrad=100)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Pair swap
    if time.time() < d6 - 15:
        sub_dl = min(d6, time.time() + budget_s * 0.03)
        h6, c5_6 = _pair_swap(best_h, sub_dl, n_pairs=5000, radius=0.15, seed=314)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    # Basin hopping
    scale = 0.03
    n_restarts = 0
    while time.time() < d6 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub_dl = min(d6 - 5, time.time() + 3)
        h_t, c5_t = _grad_descent(h_pert, sub_dl, T_init=5.0, lr_init=0.003, top_k=15, switch_to_subgrad=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)
    # Final binary try
    if time.time() < d6 - 3:
        h_b = (best_h > 0.5).astype(float)
        h_b = _fix_binary_count(h_b, n_final)
        sub_dl = min(d6 - 1, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iters=20000, stall_limit=100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
