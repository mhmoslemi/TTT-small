```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def _pair_flip_delta(scores, h, i, j, n):
    """Compute delta in scores after flipping h[i] 1->0 and h[j] 0->1.
    Returns the new scores vector (copy)."""
    new_scores = scores.copy()
    for d in range(1, n):
        r_pos = n - 1 + d  # positive lag d
        r_neg = n - 1 - d  # negative lag d (i.e., lag -d)
        
        # Positive lag d: score = sum h[k]*(1-h[k+d]) for k=0..n-d-1
        delta = 0.0
        if i < n - d:  # i appears as first index
            delta += -(1.0 - h[i + d])
        if j < n - d:  # j appears as first index
            delta += (1.0 - h[j + d])
        new_scores[r_pos] += delta
        
        # Negative lag d: score = sum h[k+d]*(1-h[k]) for k=0..n-d-1

        delta = 0.0
        if i >= d:  # i appears as second index
            delta += -(1.0 - h[i - d])
        if j >= d:  # j appears as second index
            delta += (1.0 - h[j - d])
        new_scores[r_neg] += delta
    
    return new_scores
```

```python
    # For negative lag g: delta = -(1-h[pos-g]) if g<=pos, + h[pos+g] if pos+g<n
    pass
```

This is getting complex. Let me just use a simpler approach: for each pair (i,j), compute the new scores by doing the flip and calling np.correlate. This is O(n²) per pair but in C (fast). For n=180, np.correlate on 180-element arrays is very fast (microseconds).

The real bottleneck in the pair sweep is the O(n²) number of pairs. For n=180, that's 8100 pairs per round. Each evaluation takes maybe 10-50 μs, so one round takes about 0.1-0.4 seconds. With 20 rounds, that's 2-8 seconds. Very feasible.

So the previous approach of just doing np.correlate for each pair is actually fine. The issue is more about algorithmic efficiency (trying the right pairs) rather than computational speed.

Let me focus on a different angle: what if I use a smarter local search that doesn't just do random pair swaps, but targets the most "promising" pairs?

Idea: For each lag that is at the maximum, find pairs (i,j) that would reduce that lag's score the most. Then try those pairs first.

Actually, let me think about this more carefully. The problem is that the max is achieved at possibly multiple lags. A good move should reduce the max without increasing any other lag above the new max.

Here's a refined idea:
1. Identify the set S of "active" lags (those with score = max or within ε of max)
2. For each active lag d, compute the gradient of score_d with respect to h
3. Find a direction that decreases all active scores simultaneously
4. Take a step in that direction (projected)

This is essentially what the LSE approach does! The LSE gradient is a weighted average of the gradients of all lag scores, with weights proportional to softmax(scores/τ). When τ is small, the weights concentrate on the active lags.

So the LSE approach should work well here. Let me make sure my implementation is correct and efficient.

One issue: my `_lse_projected_gradient` does a backtracking line search which requires multiple LSE evaluations. Each LSE evaluation requires one np.correlate call. For n=180, this is fast. But I'm doing up to 10 backtracking steps, each with a new h, so that's 10 correlations per gradient step. With 40 steps per anneal, that's 400 correlations per anneal. With 8 anneals, that's 3200 correlations. Each correlation is O(n²) = 32,400 operations. Total: about 100 million operations. In C (numpy), this should take less than a second.

OK so the LSE approach should be efficient enough. Let me focus on making the overall algorithm work well.

Key improvements over the previous code:
1. LSE-smoothed objective instead of LP-based linearization (handles multiple active lags more naturally)
2. More diverse starting points
3. Better annealing schedule
4. Incremental score updates for pair sweep (optional speedup)
5. "Reheating" in SA to escape local optima

Let me also add a key technique: **multi-start with different n values**. The optimal n might not be 180. By trying n = 150, 160, 170, 180, 190, 200, ..., we might find a better solution.

Actually, I realize there's another important consideration. The problem says "Smaller sequences with less than 2k samples are preferred." This suggests that the evaluation might prefer smaller n. But the C₅ value depends on n through dx = 2/n. A solution with smaller n has larger dx, which means the same raw score gives a larger C₅. So smaller n is actually harder to optimize for (you need the raw max score to be smaller).

Wait, but the constraint Σh = n/2 means the "density" is always 1/2 regardless of n. So the structure of the problem is scale-invariant in some sense. The question is whether the discrete approximation is better for some n values.

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_value_and_grad(h, tau):
    """Compute LSE value and gradient."""
    n = len(h)
    scores = _scores(h)
    m = float(np.max(scores))
    u = (scores - m) / tau
    se = np.exp(u)
    L = float(np.sum(se))
    val = m + tau * float(np.log(L))
    p = se / L
    
    # Gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                grad[:m_] += p[r] * (1.0 - h[d:d+m_])
                grad[d:d+m_] += -p[r] * h[:m_]
        else:
            g = -d
            m_ = n - g
            if m_ > 0:
                grad[g:] += p[r] * (1.0 - h[:m_])
                grad[:m_] += -p[r] * h[g:]
    return val, grad

def _lse_pgd(h, tau, n_steps=40, deadline=None, lr=None):
    """Projected gradient descent on LSE objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    if lr is None:
        lr = 2.0 / (n + 1.0)  # step size
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        val, g = _lse_value_and_grad(h, tau)
        
        # Gradient step with backtracking
        alpha = lr
        improved = False
        for bt in range(8):
            h_try = _project_sum(h - alpha * g, target)
            val_try, _ = _lse_value_and_grad(h_try, tau)
            if val_try < val - 1e-15:
                h = h_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            break
        
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _lse_anneal(h, deadline, tau_seq=None, n_steps_per=35):
    """Anneal LSE temperature."""
    if tau_seq is None:
        tau_seq = [4.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15]
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for tau in tau_seq:
        if time.time() > deadline:
            break
        remaining = max(1, len(tau_seq) - tau_seq.index(tau))
        dl = min(deadline, time.time() + (deadline - time.time()) / (remaining + 1))
        h, c5 = _lse_pgd(h, tau, n_steps=n_steps_per, deadline=dl)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_sweep(h, deadline, max_rounds=25):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if cur[i] == cur[j]:
            continue
        cur[i] = 1.0 - cur[i]
        cur[j] = 1.0 - cur[j]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[i] = 1.0 - cur[i]
            cur[j] = 1.0 - cur[j]
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)
