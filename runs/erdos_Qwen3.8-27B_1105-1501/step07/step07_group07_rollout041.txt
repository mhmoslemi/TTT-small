    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        else: hi = mid

        else: mask = x > 0.0

_g_tau = 0.1
_g_dx = 0.0
_g_target = 0.0

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _lse_obj(x):
    s = np.correlate(x, 1.0 - x, mode='full') * _g_dx
    mval = float(np.max(s))
    u = (s - mval) / _g_tau
    u = np.clip(u, -500.0, 500.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        return mval
    return mval + _g_tau * np.log(zsum)

def _lse_grad(x):
    s = np.correlate(x, 1.0 - x, mode='full') * _g_dx
    mval = float(np.max(s))
    u = (s - mval) / _g_tau
    u = np.clip(u, -500.0, 500.0)
    z = np.exp(u)
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        return np.zeros(len(x))
    probs = z / zsum
    n = _g_n
    mid = n - 1
    grad = np.zeros(n)
    for g in range(n):
        if g == 0:
            w = probs[mid] * _g_dx
        else:
            w = (probs[mid + g] + probs[mid - g]) * _g_dx

            continue
        if g == 0:
            grad += w * (1.0 - 2.0 * x)
        else:
            grad[g:] += w * (1.0 - x[:n - g])
            grad[:n - g] -= w * x[g:]
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_tau = tau

    try:

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _quick_optimize(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = 0.05
    while time.time() < deadline and tau > 0.01:
        h = _slsqp_run(h, tau, maxiter=80)
        h = _project_sum(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        tau *= 0.6
    if time.time() < deadline:
        h = _project_sum(best_h, target)
        h, c5 = _pair_sweep(h, min(deadline, time.time() + 30), max_rounds=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, n0 + 1)
        n0 = len(h0)
    h_init = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    if time.time() > deadline:
        return best_h, best_c5, len(best_h)

    # Phase 1: LSE annealing from multiple starts (40% of budget)
    d_phase1 = t0 + budget_s * 0.40
    seeds_list = [h_init.copy()]
    seeds_list.append(_project_sum(h_init[::-1].copy(), n0 / 2.0))
    h_comp = 1.0 - h_init
    if abs(float(np.sum(h_comp)) - n0 / 2.0) < 1.0:
        seeds_list.append(_project_sum(h_comp, n0 / 2.0))

    for si, h_s in enumerate(seeds_list):
        if time.time() > d_phase1:
            break
        sub_dl = min(d_phase1, time.time() + budget_s * 0.12)
        tau_start = 0.10 if si == 0 else 0.07
        h_r, c5_r = _lse_anneal(h_s, sub_dl, tau_start=tau_start, ratio=0.6, tau_min=0.006)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Quick pair sweep after LSE
    if time.time() < d_phase1:
        h_ps = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d_phase1, time.time() + budget_s * 0.04)
        h_ps, c5_ps = _pair_sweep(h_ps, sub_dl, max_rounds=40)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 2: Binary refinement (15%)
    d_phase2 = t0 + budget_s * 0.55
    if time.time() < d_phase2:
        h_bin = _to_binary(best_h)
        sub_dl = min(d_phase2, time.time() + budget_s * 0.06)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=777)
        sub_dl2 = min(d_phase2, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_pair_sweep(h_bin, sub_dl2, max_rounds=50)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Multi-n exploration (20%)
    d_phase3 = t0 + budget_s * 0.75
    n_best = len(best_h)
    n_candidates = []
    for dn in range(-16, 17, 2):
        n_try = n_best + dn
        if 80 <= n_try <= 240 and n_try % 2 == 0:
            n_candidates.append(n_try)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    for n_try in n_candidates[:8]:
        if time.time() > d_phase3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d_phase3, time.time() + budget_s * 0.02)
        h_r, c5_r = _quick_optimize(h_r, sub_dl)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Intensive local optimization (15%)
    d_phase4 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    sub_dl = min(d_phase4, time.time() + budget_s * 0.05)
    h4 = _project_sum(best_h.copy(), n_cur / 2.0)
    h4, c5_4 = _lse_anneal(h4, sub_dl, tau_start=0.04, ratio=0.55, tau_min=0.005)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    if time.time() < d_phase4:
        h4 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d_phase4, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_sweep(h4, sub_dl, max_rounds=60)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    if time.time() < d_phase4:
        h4 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d_phase4, time.time() + budget_s * 0.03)
        h4, c5_4 = _random_pair_sweep(h4, sub_dl, max_iter=15000, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    d_phase5 = deadline
    scale = 0.02
    n_attempts = 0
    while time.time() < d_phase5 and n_attempts < 200:
        n_attempts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d_phase5, time.time() + budget_s * 0.008)
        h_t, c5_t = _quick_optimize(h_pert, sub_dl)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.05)

    # Final polish
    n_final = len(best_h)
    best_h = _project_sum(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)

    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_pair_sweep(h_b, deadline, max_rounds=20)
        if c5_b < final_c5:
            best_h = h_b
            final_c5 = c5_b

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
