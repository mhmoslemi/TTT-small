def _block_direction(h, raw, block_size):
    """Direction that shifts mass between blocks."""

        return None

    for b in range(nblocks):
        sl = slice(b * block_size, (b + 1) * block_size)
        block_means[b] = np.mean(g[sl])

    d = np.zeros(n)

        d[sl] = -block_means[b]
    d -= np.mean(d)
    nm = np.linalg.norm(d)

        return None
