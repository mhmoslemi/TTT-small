    # Δ(d) = [r+d∈R and d≠0] + [s-d∈Z and d≠0] + [d=s-r]

    #                   s+d for d in R (from -[s+d∈R] term)  
    #                   s-j for j in Z (from +[s-j=d] term, i.e., d=s-j)
    #                   i-r for i in R (from +[i-r=d] term, i.e., d=i-r)
    #                   d=s-r and d=r-s

    # The affected lags are:
    # From +[r+d∈R]: d such that r+d ∈ R\{r}, i.e., d = i-r for i∈R\{r}
    # From +[s-d∈Z]: d such that s-d ∈ Z\{s}, i.e., d = s-j for j∈Z\{s}
    # From +[d=s-r]: d = s-r

    # From -[d=r-s]: d = r-s

    delta = np.zeros(2*n-1, dtype=np.int32)
    # Using index = d + n - 1
    
    # +1 at (i-r) for i in ones, i≠r
    d_idx = (ones[ones != r] - r + n - 1)

    # +1 at (s-j) for j in zeros, j≠s

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_delta(ones, zeros, r, s, n):
    delta = np.zeros(2 * n - 1, dtype=np.int32)
    off = n - 1
    others_ones = ones[ones != r]
    others_zeros = zeros[zeros != s]
    delta[others_ones - r + off] += 1
    delta[s - others_zeros + off] += 1
    delta[s - r + off] += 1
    delta[others_ones - s + off] -= 1
    delta[r - others_zeros + off] -= 1
    delta[r - s + off] -= 1
    return delta

def _binary_counts_from_state(ones, zeros, n):

    for r in ones:

def _resample_h(h, n_new):
    n = len(h)

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _active_lp(h, scores, radius, n_active=16, gap=0.005):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])

        order = np.argsort(scores[active])
        active = active[order][-n_active:]

    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_dir(h, scores, beta=25.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:

        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

        d = _active_lp(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_dir(h, scores, beta=25.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.4
                n_active = min(n_active + 3, 30)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.3
            n_active = min(n_active + 2, 28)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=150):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.35)[0]
    zeros = np.where(h < 0.65)[0]
    rng = np.random.default_rng(12345)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(1, len(ones) // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(len(zeros)))
            j = int(rng.integers(len(ones)))
            zi = int(zeros[i])
            oj = int(ones[j])
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.2)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.35)[0]
                zeros = np.where(h < 0.65)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick continuous polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _continuous_opt(h_init, d1, radius_init=0.08, max_iters=1500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    if time.time() < d1:
        h1b, c5_1b = _pair_sweep(_project(best_h.copy(), n0 / 2.0), d1, max_rounds=40)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Binary SA (45%)
    d2 = t0 + budget_s * 0.53
    n_bin = n0 if n0 % 2 == 0 else n0 + 1
    h_bin_start = _resample_h(best_h, n_bin) if n_bin != n0 else best_h.copy()
    best_bin_max = 9999
    best_bin_ones = None
    best_bin_zeros = None

    seeds_list = []
    b, ones_s, zeros_s = _to_binary(h_bin_start)
    counts_s = _binary_counts_from_state(ones_s, zeros_s, n_bin)
    seeds_list.append((ones_s, zeros_s, counts_s))

    for _ in range(3):
        ones_r, zeros_r, counts_r = _random_balanced(n_bin, rng)
        seeds_list.append((ones_r, zeros_r, counts_r))
    for _ in range(3):
        ones_p, zeros_p, counts_p = _periodic_seed(n_bin, rng)
        seeds_list.append((ones_p, zeros_p, counts_p))

    n_seeds = len(seeds_list)
    time_per_seed = (d2 - time.time()) / n_seeds
    for si in range(n_seeds):
        if time.time() >= d2:
            break
        sub_dl = min(d2, time.time() + max(1.0, time_per_seed * 1.1))
        ones_si, zeros_si, counts_si = seeds_list[si]
        cur_mx = int(np.max(counts_si))
        if cur_mx < best_bin_max:
            best_bin_max = cur_mx
            best_bin_ones = ones_si.copy()
            best_bin_zeros = zeros_si.copy()
        T_init = 0.015 if si == 0 else 0.01
        ones_o, zeros_o, mx_o = _binary_sa(ones_si.copy(), zeros_si.copy(), counts_si.copy(),
                                           cur_mx, n_bin, sub_dl, rng, T_init=T_init)
        if mx_o < best_bin_max:
            best_bin_max = mx_o
            best_bin_ones = ones_o
            best_bin_zeros = zeros_o

    if best_bin_ones is not None:
        b_best = np.zeros(n_bin)
        b_best[best_bin_ones] = 1.0
        b_best = _project(b_best, n_bin / 2.0)
        c5_bin = _exact_c5(b_best)
        # Continuous polish from binary
        d3 = t0 + budget_s * 0.73
        h_pol, c5_pol = _continuous_opt(b_best, d3, radius_init=0.06, max_iters=2000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if time.time() < d3:
            h_ps, c5_ps = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), d3, max_rounds=60)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()

    # Phase 3: Multi-resolution (17%)
    d3b = t0 + budget_s * 0.73
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    tried = set([n_curr])
    for dn in [-12, -8, -4, 4, 8, 12, -20, 20]:
        if time.time() >= d4:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample_h(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.01:
            sub_dl = min(d4, time.time() + budget_s * 0.015)
            h_r, c5_r = _continuous_opt(h_r, sub_dl, radius_init=0.08, max_iters=800)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_curr = n_try

    # Phase 4: Second binary round with best continuous (10%)
    if time.time() < d4:
        n_b2 = len(best_h) if len(best_h) % 2 == 0 else len(best_h) + 1
        h_b2 = _resample_h(best_h, n_b2) if n_b2 != len(best_h) else best_h.copy()
        b2, ones2, zeros2 = _to_binary(h_b2)
        counts2 = _binary_counts_from_state(ones2, zeros2, n_b2)
        cur2 = int(np.max(counts2))
        sub_dl = min(d4, time.time() + budget_s * 0.08)
        ones2o, zeros2o, mx2 = _binary_sa(ones2.copy(), zeros2.copy(), counts2.copy(),
                                          cur2, n_b2, sub_dl, rng, T_init=0.008)
        if mx2 < best_bin_max:
            best_bin_max = mx2
            best_bin_ones = ones2o
            best_bin_zeros = zeros2o
            b_best2 = np.zeros(n_b2)
            b_best2[best_bin_ones] = 1.0
            b_best2 = _project(b_best2, n_b2 / 2.0)
            if time.time() < d4:
                h_p2, c5_p2 = _continuous_opt(b_best2, min(d4, time.time() + budget_s * 0.03),
                                              radius_init=0.05, max_iters=600)
                if c5_p2 < best_c5:
                    best_c5 = c5_p2
                    best_h = h_p2.copy()

    # Phase 5: Final polish (10%)
    d5 = deadline - 1
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _continuous_opt(h_f, d5, radius_init=0.04, max_iters=2000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if deadline - time.time() > 5:
        h_f2, c5_f2 = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), deadline, max_rounds=60)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
