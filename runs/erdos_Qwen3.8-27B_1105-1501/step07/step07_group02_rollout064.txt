def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection on clipping threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_grad_matrix(h):
    """Build the L x n gradient matrix for all correlation rows."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    one_minus_h = 1.0 - h
    for ell in range(-(n-1), n):
        r = ell + n - 1
        if ell >= 0:
            # h[j+ell] * (1-h[j]) for j=0..n-1-ell
            m = n - ell
            G[r, ell:] += one_minus_h[:m]
            G[r, :m] -= h[ell:]
        else:
            p = -ell
            m = n - p
            G[r, p:] += one_minus_h[:m]
            G[r, :m] -= h[p:]
    return G

def _dense_direction(h, S, G, K, target_mode='mean'):
    """Find a dense zero-sum direction that decreases the top-K active rows."""
    n = len(h)
    # Find top-K rows
    sorted_idx = np.argsort(S)[::-1]
    top_k = sorted_idx[:K]
    
    # Get gradients for these rows
    grad_k = G[top_k]  # K x n
    
    # Weight: give more weight to higher rows
    vals = S[top_k]
    max_val = vals[0]
    weights = 1.0 / (1.0 + (max_val - vals) * 50.0)  # exponential decay
    weights = weights / weights.sum()
    
    # Direction: negative weighted sum of gradients
    d = -(grad_k.T @ weights)  # n
    d -= d.mean()  # zero-sum
    
    # Normalize
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-12:
        return None
    d = d / max_abs
    
    return d

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    f2 = predict(c2)

            c2 = lo + gr * (hi - lo)
            f2 = predict(c2)

            t_lo, t_hi = _feasible_interval(h, d)
            if t_hi > t_lo + 1e-12:
                U = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
                B = np.correlate(d, d, mode='full')
                t_opt, _ = _line_search(S, U, B, t_lo, t_hi)
                h_try = _project(h + t_opt * d, target)
                c_try = _exact_c5(h_try)
                if c_try < best_trial:
                    best_trial = c_try
                    best_t = t_opt
                    best_h_try = h_try
        # Try negative direction too
        if d is not None:
            t_lo2, t_hi2 = _feasible_interval(h, -d)
            if t_hi2 > t_lo2 + 1e-12:
                U2 = np.correlate(-d, omh, mode='full') - np.correlate(h, -d, mode='full')
                B2 = np.correlate(-d, -d, mode='full')
                t_opt2, _ = _line_search(S, U2, B2, t_lo2, t_hi2)
                h_try2 = _project(h + t_opt2 * (-d), target)
                c_try2 = _exact_c5(h_try2)
                if c_try2 < best_trial:
                    best_trial = c_try2
                    best_h_try = h_try2
        # Also try pairwise
        pd, pv = _pair_dir(h, S, G, rng, n_pairs=15)
        if pd is not None and pv < best_trial:
            best_trial = pv
            t_lo3, t_hi3 = _feasible_interval(h, pd)
            if t_hi3 > t_lo3 + 1e-12:
                U3 = np.correlate(pd, omh, mode='full') - np.correlate(h, pd, mode='full')
                B3 = np.correlate(pd, pd, mode='full')
                t_opt3, _ = _line_search(S, U3, B3, t_lo3, t_hi3)
                h_try3 = _project(h + t_opt3 * pd, target)
                c_try3 = _exact_c5(h_try3)
                if c_try3 < best_trial:
                    best_trial = c_try3
                    best_h_try = h_try3
        if best_trial < cur_c5 - 1e-13:
            h = best_h_try
            cur_c5 = best_trial
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S = np.correlate(h, 1.0 - h, mode='full')
            G = _build_grad_matrix(h)
            omh = 1.0 - h
            stall = 0
        else:
            stall += 1
            # Random perturbation when stalled
            if stall % 10 == 0:
                rdir = rng.standard_normal(n)
                rdir -= rdir.mean()
                ma = float(np.max(np.abs(rdir)))
                if ma > 1e-12:
                    rdir /= ma
                    rdir *= 0.05
                    t_lo4, t_hi4 = _feasible_interval(h, rdir)
                    if t_hi4 > t_lo4 + 1e-12:
                        h_try4 = _project(h + 0.05 * rdir, target)
                        c_try4 = _exact_c5(h_try4)
                        if c_try4 < cur_c5:
                            h = h_try4
                            cur_c5 = c_try4
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            S = np.correlate(h, 1.0 - h, mode='full')
                            G = _build_grad_matrix(h)
                            omh = 1.0 - h
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h, deadline, seed=999):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    cur = np.zeros(n)
    cur[idx] = 1.0
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.002
    ones_list = np.where(cur > 0.5)[0].tolist()
    zeros_list = np.where(cur <= 0.5)[0].tolist()
    stall = 0
    while time.time() < deadline - 1.0:
        if stall > 500:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones_list[i], ones_list[j] = ones_list[j], ones_list[i]
            zeros_list[i], zeros_list[j] = zeros_list[j], zeros_list[i]
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9999
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense direction refinement at parent n (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _refine_dense(h_init, d1, seed=1, max_stall=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (30% time)
    d2 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -16, 16, -12, 12]
    tried_ns = set([n0])
    for dn in offsets:
        if time.time() > d2 - 5:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 240 or n_try in tried_ns:
            continue
        if n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2 - 2, time.time() + budget_s * 0.022)
        h_r, c5_r = _refine_dense(h_r, sub_dl, seed=200 + dn, max_stall=50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Binary + SA (15% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    if time.time() < d3 - 5:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d3 - 5, time.time() + budget_s * 0.05)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, seed=777)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish
        if time.time() < d3 - 3:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _refine_dense(h_pol, sub_dl2, seed=333, max_stall=60)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Intensive refinement (15% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d4 - 1, time.time() + budget_s * 0.12)
    h4, c5_4 = _refine_dense(h4, sub_dl, seed=555, max_stall=120)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final polish (5% time)
    d5 = deadline - 2.0
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d5, deadline - 0.5)
    h5, c5_5 = _refine_dense(h5, sub_dl, seed=888, max_stall=40)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
