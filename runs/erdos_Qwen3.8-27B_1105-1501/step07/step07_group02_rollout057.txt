def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x: 0≤x≤1, sum(x)=target}
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt

def _dense_gradient(h, beta=20.0, top_k=15):
    """Compute smoothed max gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    if M < 1e-15:
        return np.zeros(n)
    w = np.exp(beta * (corr - M))
    s = np.sum(w)
    if s < 1e-30:
        # Fallback: uniform weight on top few
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / top_k
    else:
        w /= s

    top_k_actual = min(top_k, n)

        if w_r < 1e-6:
            continue
        lag = (n - 1) - r

            grad[p:] += w_r * (1 - h[:m])

    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    momentum = np.zeros(n)
    beta_cur = beta

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            kick = rng.random(n) - 0.5
            kick *= kick_size
            kick -= np.mean(kick)
            h = _project(h + kick, target)
            cur_c5 = _exact_c5(h)
            momentum *= 0.3
            beta_cur = beta
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue

        g = _dense_gradient(h, beta=beta_cur, top_k=top_k)
        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-12:
            stall += 2
            continue

        d = -g / g_norm
        d -= np.mean(d)
        momentum = mom_decay * momentum + (1.0 - mom_decay) * d
        momentum -= np.mean(momentum)
        m_norm = float(np.max(np.abs(momentum)))
        if m_norm > 1e-14:
            d = -momentum / m_norm
            d -= np.mean(d)

        d_abs_max = float(np.max(np.abs(d)))
        if d_abs_max < 1e-14:
            stall += 1
            continue

        t_max = 0.12
        pos_mask = d > 1e-13
        neg_mask = d < -1e-13
        if np.any(pos_mask):
            tm = (1.0 - h[pos_mask]).min() / d[pos_mask].max()
            t_max = min(t_max, max(tm, 0.0))
        if np.any(neg_mask):
            tm = h[neg_mask].min() / (-d[neg_mask].max())
            t_max = min(t_max, max(tm, 0.0))
        t_max = max(t_max, 1e-9)

        t_opt = _line_search_dense(h, d, t_max)
        if t_opt > 1e-10:
            h_new = np.clip(h + t_opt * d, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                    beta_cur = min(beta_cur * 1.01, 150.0)
            else:
                stall += 1
                beta_cur = max(beta * 0.6, beta_cur * 0.92)
                momentum *= 0.8
        else:
            stall += 1
            beta_cur = max(beta * 0.7, beta_cur * 0.9)

    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline:
        if stall > stall_limit or radius < 5e-5:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        s = float(np.sum(w))
        if s < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0 / top_k
        else:
            w /= s

        coord_grad = np.zeros(n)
        for r in range(len(S0)):
            if w[r] < 1e-7:
                continue
            lag = (n - 1) - r
            if lag >= 0:
                m = n - lag
                coord_grad[:m] += w[r] * (1.0 - h[lag:])
                coord_grad[lag:] -= w[r] * h[:m]
            else:
                p = -lag
                m = n - p
                coord_grad[p:] += w[r] * (1.0 - h[:m])
                coord_grad[:m] -= w[r] * h[p:]

        sort_idx = np.argsort(coord_grad)
        half_n = min(n_pairs, n // 2)
        sources = sort_idx[-half_n:]
        sinks = sort_idx[:half_n]
        pairs = []
        for ii in range(half_n):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(10, half_n // 3)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))

        best_c5_try = cur_c5
        best_step = None
        one_mh = 1.0 - h

        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            # Line search: minimize max(S0 + t*T - t^2*C)
            if t_hi - t_lo < 1e-14:
                continue
            n_g = 25
            ts = np.linspace(t_lo, t_hi, n_g)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
            bi = int(np.argmin(vals))
            lo_g = float(ts[max(0, bi - 1)])
            hi_g = float(ts[min(n_g - 1, bi + 1)])
            if hi_g - lo_g < 1e-11:
                t_opt = float(ts[bi])
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi_g - gr * (hi_g - lo_g)
                c2 = lo_g + gr * (hi_g - lo_g)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
                for _ in range(25):
                    if hi_g - lo_g < 1e-11:
                        break
                    if f1 < f2:
                        hi_g, c2, f2 = c2, c1, f1
                        c1 = hi_g - gr * (hi_g - lo_g)
                        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
                    else:
                        lo_g, c1, f1 = c1, c2, f2
                        c2 = lo_g + gr * (hi_g - lo_g)
                        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
                t_opt = (lo_g + hi_g) / 2.0
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try

        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.015, 0.25)
        else:
            stall += 1
            radius *= 0.8
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.08

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=15000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _binary_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, UnboundLocalError):
        n0 = 180
        h0 = np.zeros(n0)
        idx = np.arange(0, n0, 2)
        h0[idx] = 1.0

    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Dense optimizer from parent (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _dense_optimizer(h_init.copy(), d1, max_iter=2000, beta=20.0,
                                 top_k=12, stall_limit=50, kick_size=0.03, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair optimizer refinement (20% time)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _pair_optimize(best_h.copy(), d2, radius=0.12, n_pairs=45,
                               beta=25.0, stall_limit=70, seed=11)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (15% time)
    d3 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    offsets = [-6, 6, -10, 10, -14, 14, -4, 4]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _dense_optimizer(h_r, sub_dl, max_iter=500, beta=20.0,
                                      top_k=10, stall_limit=30, kick_size=0.02, seed=200+dn)
        sub_dl2 = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.10, n_pairs=30,
                                    beta=22.0, stall_limit=30, seed=300+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Combined dense+pair with diverse restarts (20% time)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    # Try a few restarts with different seeds
    for rs in range(3):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4 = best_h.copy()
        # Add small perturbation
        pert = rng.random(n_curr) - 0.5
        pert *= 0.05
        pert -= np.mean(pert)
        h4 = _project(h4 + pert, n_curr / 2.0)
        h4, c5_4 = _dense_optimizer(h4, sub_dl, max_iter=800, beta=15.0,
                                     top_k=10, stall_limit=40, kick_size=0.03, seed=500+rs*7)
        sub_dl2 = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _pair_optimize(h4, sub_dl2, radius=0.08, n_pairs=35,
                                   beta=25.0, stall_limit=40, seed=600+rs*7)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Binary diversification (10% time)
    d5 = t0 + budget_s * 0.95
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.003, seed=777)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _binary_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish from binary
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d5, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _dense_optimizer(h_pol, sub_dl3, max_iter=400, beta=30.0,
                                              top_k=10, stall_limit=30, seed=888)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Final intensive refinement (5% time)
    d6 = deadline - 2
    n_curr = len(best_h)
    if time.time() < d6:
        h6 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d6, deadline)
        h6, c5_6 = _dense_optimizer(h6, sub_dl, max_iter=1500, beta=30.0,
                                     top_k=12, stall_limit=100, kick_size=0.01, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        # Final pair polish
        if time.time() < d6:
            h6b, c5_6b = _pair_optimize(best_h.copy(), d6, radius=0.05, n_pairs=30,
                                         beta=30.0, stall_limit=80, seed=1111)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Final validation
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    # Ensure validity
    if not np.all(best_h >= -1e-7) or not np.all(best_h <= 1.0 + 1e-7):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project(best_h, n_final / 2.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, n_final
