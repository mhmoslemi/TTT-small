def _project(v, target):
    # Bisection to find threshold for projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            opt_a = (lo + hi) / 2.0
            val_opt = float(np.max(raw + opt_a * T - opt_a * opt_a * Q)) * dx
            if val_opt < best_val:
                best_val = val_opt
                best_a = opt_a
        h_new = _project(h + best_a * d, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _multi_lp_optimize(h, deadline, top_k_list=None, radius_list=None):
    """Try LP with different top_k and radius settings."""
    if top_k_list is None:
        top_k_list = [10, 20, 30, 40, 50]
    if radius_list is None:
        radius_list = [0.08, 0.12, 0.18, 0.25]
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rk in top_k_list:
        if time.time() > deadline:
            break
        for rad in radius_list:
            if time.time() > deadline:
                break
            sub_dl = min(deadline, time.time() + (deadline - time.time()) / 8.0)
            h_try, c5_try = _lp_optimize(h, sub_dl, top_k=rk, radius_init=rad, max_rounds=80)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5
