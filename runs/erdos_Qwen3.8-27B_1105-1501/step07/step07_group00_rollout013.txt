def _full_lp(h, scores, radius):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, d:] += u[:m]
                G[k, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, :m] += u[g:]
                G[k, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None
