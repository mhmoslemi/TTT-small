def _project(v, target):

        return np.ones(n)

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx):

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=25.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -60.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    if np.any(pos):
        r = (1.0 - h[pos]) / radius
        g[pos] = np.minimum(g[pos], r)
    if np.any(neg):
        r = -h[neg] / radius
        g[neg] = np.maximum(g[neg], r)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _resample(h, n_new):

    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.25 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _build_balanced(n):
    if n % 2 != 0:
        n = n - 1
    h = np.zeros(n)
    step = (n // 2)
    for i in range(n // 2):
        pos = (i * step + i) % n
        h[pos] = 1.0
    cnt = int(np.sum(h))
    if cnt > n // 2:
        h[np.where(h == 1.0)[0][n // 2:]] = 0.0
    elif cnt < n // 2:
        free = np.where(h == 0.0)[0]
        h[free[:n // 2 - cnt]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = 1.0

    raw = globals().get('initial_h_values', None)
    if raw is not None and len(raw) > 0:
        raw = np.array(raw, dtype=float)
        n0 = len(raw)
        h0 = _project(raw.copy(), n0 / 2.0)
        c5_0 = _exact_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
        n_init = n0
    else:
        n_init = 120
        h0 = _build_balanced(n_init)
        best_h = h0.copy()
        best_c5 = _exact_c5(h0)

    # Phase 1: MILP exploration (18%)
    d1 = t0 + budget_s * 0.18
    milp_sizes = [40, 52, 64, 76, 88, 100]
    for n_try in milp_sizes:
        if time.time() > d1 - 15:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.028)
        result = _try_milp_binary(n_try, budget_s * 0.025, sub_dl)
        if result is not None:
            h_m, c5_m, M_val = result
            if c5_m < best_c5:
                best_c5 = c5_m
                best_h = h_m.copy()
            # Also try the complement-flipped version
            h_mf = 1.0 - h_m
            if abs(float(np.sum(h_mf)) - n_try / 2.0) < 1e-10:
                c5_mf = _exact_c5(h_mf)
                if c5_mf < best_c5:
                    best_c5 = c5_mf
                    best_h = h_mf.copy()

    # Phase 2: Local optimization on best (15%)
    d2 = t0 + budget_s * 0.33
    h_l = _project(best_h.copy(), len(best_h) / 2.0)
    h_l, c5_l = _local_opt(h_l, d2, radius_init=0.12, max_iters=4000)
    if c5_l < best_c5:
        best_c5 = c5_l
        best_h = h_l.copy()

    # Phase 3: Pair sweep (8%)
    d3 = t0 + budget_s * 0.41
    h_p = _project(best_h.copy(), len(best_h) / 2.0)
    h_p, c5_p = _pair_sweep(h_p, d3, max_rounds=200)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 4: Multi-resolution (22%)
    d4 = t0 + budget_s * 0.63
    n_curr = len(best_h)
    tried_ns = set()
    for dn in range(-10, 15, 2):
        if time.time() > d4 - 5:
            break
        n_try = n_curr + dn
        if n_try < 36 or n_try > 220 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.014)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.10, max_iters=1200)
        if c5_r < best_c5 + 0.003 and time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.005)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2, max_rounds=50)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Binary optimization (15%)
    d5 = t0 + budget_s * 0.78
    n_b = len(best_h)
    h_bin = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=60000, T_init=0.004)
    if time.time() < d5 - 5:
        sub_dl2 = min(d5, time.time() + budget_s * 0.025)
        h_bin2, c5_bin2 = _pair_sweep(h_bin, sub_dl2, max_rounds=100)
        if c5_bin2 < c5_bin:
            h_bin, c5_bin = h_bin2, c5_bin2
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on best binary
    if time.time() < d5 - 3:
        h_pol = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl3 = min(d5, time.time() + budget_s * 0.02)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl3, radius_init=0.06, max_iters=800)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 6: Basin hopping (12%)
    d6 = t0 + budget_s * 0.90
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6, time.time() + budget_s * 0.004)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.07, max_iters=500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.07)

    # Phase 7: Final polish (10%)
    d7 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, d7 - 5, radius_init=0.05, max_iters=3000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if deadline - time.time() > 5:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_f2, c5_f2 = _pair_sweep(h_f2, deadline, max_rounds=80)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
