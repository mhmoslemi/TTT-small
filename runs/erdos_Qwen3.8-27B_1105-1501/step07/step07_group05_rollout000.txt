        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair optimization (12%)
    d2 = t0 + budget_s * 0.37
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt_v2(h2, d2, max_it=40000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA + reconverge (15%)
    d3 = t0 + budget_s * 0.52
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa_v2(hbin, d3, max_it=25000, T_init=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.02)
        hp, cp = _descend(hp, sd, r0=0.10, max_stall=30, n_dirs=6)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-resolution search (20%)
    d4 = t0 + budget_s * 0.72
    nc = len(best_h)
    offsets = [-10, -6, -4, -2, 2, 4, 6, 8, 10, -14, 14, -8, 8, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.015)
        hr, cr = _descend(hr, sd, r0=0.12, max_stall=30, n_dirs=6)
        if cr < best_c + 0.003 and time.time() < d4 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa_v2(hb, sd2, max_it=5000, T_init=0.0015)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _descend(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, max_stall=20, n_dirs=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Deep local optimization (18%)
    d5 = t0 + budget_s * 0.90
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.05)
        h5, c5v = _descend(h5, sd, r0=r_init, max_stall=50, n_dirs=8)
        if c5v < best_c:
            best_c = c5v
            best_h = h5.copy()
        if time.time() < d5 - 3:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.025)
            h5, c5v = _pair_opt_v2(h5, sd, max_it=20000)
            if c5v < best_c:
                best_c = c5v
                best_h = h5.copy()

    # Phase 6: Basin hopping (8%)
    d6 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.3, 12.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.25 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _descend(hp, sd, r0=0.05, max_stall=15, n_dirs=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _descend(hf, deadline, r0=0.015, max_stall=25, n_dirs=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt_v2(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    if final_c > best_c + 1e-12:
        final_c = best_c
    return best_h, final_c, len(best_h)
