```python
def _smooth_subgradient(h, beta, alpha_init=0.01, n_iter=5000):
    """Minimize log-sum-exp approximation of C5 using projected subgradient."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    
    for it in range(n_iter):
        # Compute current lag values
        f = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (L,)
        
        # Compute softmax weights
        max_f = float(np.max(f))
        exp_f = np.exp(beta * (f - max_f))
        p = exp_f / float(np.sum(exp_f))
        
        # Compute gradient of log-sum-exp
        # grad[j] = sum_k p[k] * df_k/dh[j]
        grad = np.zeros(n)
        for k in range(L):
            if p[k] < 1e-10:
                continue
            ell = k - (n - 1)
            if ell == 0:
                grad += p[k] * dx * (1.0 - 2.0 * h)
            elif ell > 0:
                m = n - ell
                grad[:m] += p[k] * dx * (1.0 - h[ell:])
                grad[ell:] -= p[k] * dx * h[:m]
            else:
                pe = -ell
                m = n - pe
                grad[pe:] += p[k] * dx * (1.0 - h[:m])
                grad[:m] -= p[k] * dx * h[pe:]
        
        # Projected gradient step
        h_new = _project(h - alpha * grad, target)
        
        # Exact evaluation
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Backtrack
            alpha *= 0.9
            h = _project(h - alpha * grad, target)
            c5_check = _exact_c5(h)
            if c5_check < _exact_c5(best_h):
                best_c5 = c5_check
                best_h = h.copy()
        
        # Slowly increase beta
        beta *= 1.005
        
        if alpha < 1e-6:
            break
    
    return best_h, best_c5
```

```python
        f = np.correlate(h, 1.0 - h, mode='full') * dx
        max_f = float(np.max(f))
        log_sum = max_f + np.log(float(np.sum(np.exp(beta * (f - max_f)))))
```

```python
        G = _compute_grad_matrix(h)
        grad = dx * (p @ G)
        vel = mu * vel - alpha * grad
        h_new = _project(h + vel, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            alpha = min(alpha * 1.01, alpha_init)
        else:
            alpha *= 0.93
            vel *= 0.7
            h_try = _project(h - alpha * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < _exact_c5(h):
                h = h_try
            stall += 1
            if stall > 80:
                alpha = alpha_init
                vel = np.zeros(n)
                mu = min(mu + 0.02, 0.95)
                stall = 0
        beta *= beta_rate
        mu = min(mu + 0.0005, 0.97)
    return best_h, best_c5

def _fista_smooth(h_init, deadline, beta=40.0, n_iter=8000):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h_init.copy(), target)
    y = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    t = 1.0
    L_est = 4.0 * dx
    alpha = 1.0 / L_est
    for it in range(n_iter):
        if time.time() > deadline:
            break
        f_y = np.correlate(y, 1.0 - y, mode='full') * dx
        max_fy = float(np.max(f_y))
        exp_fy = np.exp(beta * (f_y - max_fy))
        p_y = exp_fy / float(np.sum(exp_fy))
        G_y = _compute_grad_matrix(y)
        grad_y = dx * (p_y @ G_y)
        x_new = _project(y - alpha * grad_y, target)
        t_old = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t_old - 1.0) / (t + 1.0)) * (x_new - h)
        h = x_new
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _slsqp_smooth(h_init, deadline, beta=50.0, maxiter=150):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h_init_p = _project(h_init.copy(), target)

    def _obj(z):
        hz = z[:n]
        f = np.correlate(hz, 1.0 - hz, mode='full') * dx
        mf = float(np.max(f))
        return (mf + float(np.log(np.sum(np.exp(beta * (f - mf)))))) / beta

    def _obj_grad(z):
        hz = z[:n]
        f = np.correlate(hz, 1.0 - hz, mode='full') * dx
        mf = float(np.max(f))
        ef = np.exp(beta * (f - mf))
        p = ef / float(np.sum(ef))
        G = _compute_grad_matrix(hz)
        g = dx * (p @ G)
        return np.concatenate([g, [0.0]])

    def _con_eq(z):
        return float(np.sum(z[:n]) - target)

    def _con_eq_grad(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    c0 = _exact_c5(h_init_p)
    x0 = np.concatenate([h_init_p, [c0 + 0.005]])
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_grad}]
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        h_sol = _project(res.x[:n], target)
        c5 = _exact_c5(h_sol)
        return h_sol, c5
    except Exception:
        return None, None

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    ts = np.linspace(t_lo, t_hi, 50)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(49, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    best_val = min(opt_val, float(vals[best_idx]))
    if best_val == opt_val:
        return opt_t, best_val
    return float(ts[best_idx]), best_val

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    one_mh = 1.0 - h
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-7:
            break
        M = float(np.max(S0))
        if M < 1e-15:
            break
        w = np.exp(beta * (S0 - M))
        s = float(np.sum(w))
        if s < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0 / top_k
        else:
            w /= s
        G = _compute_grad_matrix(h)
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        ns = min(n_pairs, n // 2)
        sources = sort_idx[-ns:]
        sinks = sort_idx[:ns]
        pairs = []
        for ii in range(min(ns, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, max(5, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_c5_try = cur_c5
        best_step = None
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, val_opt = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            one_mh = 1.0 - h
            stall = 0
            radius = min(radius * 1.02, 0.28)
        else:
            stall += 1
            radius *= 0.78
            if stall % 40 == 0 and radius < 0.02:
                radius = 0.10
    return best_h, best_c5

def _block_optimize(h, deadline, block_size=6, n_blocks=None, beta=20.0, stall_limit=60, seed=100):
    """Optimize blocks of coordinates simultaneously using SLSQP on sub-problem."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    if n_blocks is None:
        n_blocks = n // block_size
    stall = 0

    while time.time() < deadline and stall < stall_limit:
        S0 = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(S0))
        if M < 1e-15:
            break
        w = np.exp(beta * (S0 - M))
        s = float(np.sum(w))
        if s < 1e-30:
            w = np.ones(len(S0)) / len(S0)
        else:
            w /= s
        G = _compute_grad_matrix(h)
        coord_grad = G.T @ w
        # Pick blocks with highest gradient magnitude
        block_grads = []
        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            block_grads.append((start, end, float(np.sum(np.abs(coord_grad[start:end])))))
        block_grads.sort(key=lambda x: -x[2])
        improved = False
        for (start, end, _) in block_grads[:min(12, n_blocks)]:
            if time.time() > deadline:
                break
            bs = end - start
            h_fix = h.copy()
            h_block = h[start:end].copy()
            target_block = float(np.sum(h_block))

            def _block_obj(z):
                hz = h_fix.copy()
                hz[start:end] = z
                f = np.correlate(hz, 1.0 - hz, mode='full') * dx
                mf = float(np.max(f))
                by = 30.0
                return (mf + float(np.log(np.sum(np.exp(by * (f - mf)))))) / by

            def _block_obj_grad(z):
                hz = h_fix.copy()
                hz[start:end] = z
                f = np.correlate(hz, 1.0 - hz, mode='full') * dx
                mf = float(np.max(f))
                ef = np.exp(30.0 * (f - mf))
                p = ef / float(np.sum(ef))
                Gb = _compute_grad_matrix(hz)
                return dx * (p @ Gb)[start:end]

            def _block_con(z):
                return float(np.sum(z) - target_block)

            def _block_con_grad(z):
                return np.ones(bs)

            bounds_b = [(0.0, 1.0)] * bs
            cons_b = [{'type': 'eq', 'fun': _block_con, 'jac': _block_con_grad}]
            try:
                res = minimize(_block_obj, h_block.copy(), method='SLSQP',
                               jac=_block_obj_grad, bounds=bounds_b,
                               constraints=cons_b, options={'maxiter': 40, 'ftol': 1e-8})
                h_new_block = _project(res.x, target_block)
                h_try = h_fix.copy()
                h_try[start:end] = h_new_block
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
            except Exception:
                pass
        if not improved:
            stall += 1
        else:
            stall = 0
            beta = min(beta * 1.05, 60.0)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h.copy(), n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_seeds(h0, n, rng, n_seeds=5):
    """Generate diverse starting points."""
    seeds = []
    seeds.append(h0.copy())
    for i in range(n_seeds - 1):
        scale = 0.05 + 0.03 * i
        pert = rng.standard_normal(n) * scale
        seeds.append(_project(h0 + pert, n / 2.0))
    return seeds

def _binary_sa(h_bin, deadline, max_iter=15000, T_init=0.002, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth subgradient with momentum (25% time)
    d1 = t0 + budget_s * 0.25
    seeds = _make_seeds(h_init, n0, rng, n_seeds=4)
    for si, s in enumerate(seeds):
        sub_dl = min(d1, time.time() + budget_s * 0.06)
        h_s, c5_s = _soft_subgradient(s, sub_dl, beta_start=8.0 + 4.0 * si,
                                       beta_rate=1.003, alpha_init=0.3,
                                       momentum_init=0.85 + 0.02 * si, max_iter=15000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: FISTA (10% time)
    d2 = t0 + budget_s * 0.35
    for bi in range(3):
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.03)
        h_f, c5_f = _fista_smooth(best_h, sub_dl, beta=20.0 + 20.0 * bi, n_iter=6000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: SLSQP on smooth approximation (20% time)
    d3 = t0 + budget_s * 0.55
    slsqp_starts = [best_h, h_init]
    pert = rng.standard_normal(n0) * 0.08
    slsqp_starts.append(_project(h_init + pert, n0 / 2.0))
    pert2 = rng.standard_normal(n0) * 0.12
    slsqp_starts.append(_project(h_init + pert2, n0 / 2.0))
    for si, s in enumerate(slsqp_starts):
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.045)
        h_sl, c5_sl = _slsqp_smooth(s, sub_dl, beta=40.0 + 15.0 * si, maxiter=120)
        if h_sl is not None and c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 4: Block optimization (15% time)
    d4 = t0 + budget_s * 0.70
    if time.time() < d4:
        for bs in [4, 6]:
            sub_dl = min(d4, time.time() + budget_s * 0.05)
            h_b, c5_b = _block_optimize(best_h, sub_dl, block_size=bs,
                                         beta=20.0, stall_limit=40, seed=200 + bs)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 5: Pair optimization refinement (20% time)
    d5 = t0 + budget_s * 0.90
    h_p = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.08)
    h_p, c5_p = _pair_optimize(h_p, sub_dl, radius=0.15, n_pairs=55,
                                beta=22.0, stall_limit=120, seed=555)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Try nearby n values
    n_curr = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl2 = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.12, n_pairs=40,
                                    beta=22.0, stall_limit=50, seed=444 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Binary diversification (5% time)
    d6 = t0 + budget_s * 0.95
    if time.time() < d6:
        n_curr = len(best_h)
        h_bin = _to_binary(best_h, n_curr)
        sub_dl = min(d6, time.time() + budget_s * 0.025)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.002, seed=777)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # LP polish on binary
        if time.time() < d6:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d6, time.time() + budget_s * 0.02)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl2, radius=0.10, n_pairs=40,
                                            beta=25.0, stall_limit=50, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final polish
    d7 = deadline - 2
    n_curr = len(best_h)
    h5 = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d7, deadline - 0.5)
    h5, c5_5 = _pair_optimize(h5, sub_dl, radius=0.08, n_pairs=45,
                               beta=30.0, stall_limit=60, seed=888)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
```
