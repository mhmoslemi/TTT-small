def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    # Project onto {x in [0,1]^n : sum(x) = target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad_weighted(h, w):

    n = len(h)

        if d >= 0:
            m = n - d
            if m > 0:

        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wr * u[gp:]
                g[gp:] -= wr * h[:m]
    return g

def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_weighted(h, w):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wr * u[:m]
                g[:m] -= wr * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wr * u[gp:]
                g[gp:] -= wr * h[:m]
    return g

def _line_search(h, d, raw, dx, a_max):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_hi = a_max
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    if a_hi < 1e-12:
        return 0.0, float(np.max(raw)) * dx
    best_a = 0.0
    best_val = float(np.max(raw)) * dx
    n_grid = 50
    grid = np.linspace(0.0, a_hi, n_grid)
    cand = list(grid)
    top_k = 20
    top_idx = np.argsort(-raw)[:top_k]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 0 < av < a_hi:
                cand.append(av)
    tl = min(top_k, len(top_idx))
    for i in range(tl):
        for j in range(i + 1, tl):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)
                    a2 = (B - sd) / (2.0 * A)
                    if 0 < a1 < a_hi:
                        cand.append(a1)
                    if 0 < a2 < a_hi:
                        cand.append(a2)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 0 < aa < a_hi:
                    cand.append(aa)
    cand = [max(0.0, min(a, a_hi)) for a in cand]

        val = float(np.max(raw + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    span = a_hi / n_grid

        c1 = hi - gr * (hi - lo)

        val = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = opt
    return best_a, best_val

def _make_dirs(h, raw, n, rng):
    directions = []
    mx = float(np.max(raw))
    for beta in [8.0, 25.0, 60.0, 150.0]:

        h4, c5_4 = _coord_local(h4, sub, n_passes=2, seed=222)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # 4d: More gradient
    if time.time() < d4 - 3:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _optimize(h4, sub, radius=0.06, stall_limit=80, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping + final polish (15%)
    d5 = deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d5 - 12 and n_restarts < 100:
        n_restarts += 1
        h_pert = _perturb(best_h, scale, rng)
        sub = min(d5 - 8, time.time() + budget_s * 0.004)
        h_t, c5_t = _optimize(h_pert, sub, radius=0.08, stall_limit=20, seed=n_restarts + 1000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.78)
        else:
            scale = min(0.05, scale * 1.06)

    # Final micro-polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _optimize(h6, deadline, radius=0.03, stall_limit=40, seed=7777)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _coord_local(h6, deadline, n_passes=1, seed=8888)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
