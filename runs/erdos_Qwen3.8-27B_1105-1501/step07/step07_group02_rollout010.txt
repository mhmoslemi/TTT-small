import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float, copy=True)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-14:
        return v
    if s > target:
        if s < 1e-15:
            return np.zeros(n)
        v *= (target / s)
    else:
        remaining = target - s
        capacity = float(np.sum(1.0 - v))
        if capacity < 1e-15:
            return np.ones(n)
        alpha = remaining / capacity
        v = v + alpha * (1.0 - v)
    # Fix tiny residual
    res = target - float(np.sum(v))
    if abs(res) > 1e-14:
        v = v.copy()
        if res > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += res / cnt
    return np.clip(v, 0.0, 1.0)

def _row_gradient(h, ell, n):
    """Gradient of correlation row with lag ell w.r.t. h."""
    g = np.zeros(n)
    j_min = max(0, -ell)
    j_max = min(n - 1, n - 1 - ell)
    if j_min > j_max:
        return g
    j = np.arange(j_min, j_max + 1)
    g[j + ell] += (1.0 - h[j])
    g[j] -= h[j + ell]
    return g

def _smooth_gradient(h, S, beta=20.0):
    """Weighted sum of row gradients with softmax weights."""
    n = len(h)
    M = float(np.max(S))
    w = np.exp(beta * (S - M))
    sw = float(np.sum(w))
    if sw < 1e-30:
        # Fallback: top-K uniform
        K = max(1, (2*n-1) // 10)
        idx = np.argsort(S)[-K:]
        w = np.zeros(2*n-1)
        w[idx] = 1.0 / K
    else:
        w /= sw
    g = np.zeros(n)
    for r in range(2*n - 1):
        if w[r] > 1e-12:
            ell = r - (n - 1)
            g += w[r] * _row_gradient(h, ell, n)
    g -= np.mean(g)
    return g

def _feasible_interval(h, d):
    """Compute the feasible t interval for h + t*d in [0,1]."""
    t_lo = -np.inf
    t_hi = np.inf
    nz = np.abs(d) > 1e-14
    if not np.any(nz):
        return 0.0, 0.0
    idx = np.where(nz)[0]
    for i in idx:
        if d[i] > 0:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    if t_lo >= t_hi:
        return 0.0, 0.0
    return t_lo, t_hi

def _line_search(S, U, B, t_lo, t_hi, grid_n=40, golden_n=30):
    """Minimize max_r(S[r] + t*U[r] - t^2*B[r]) over t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S))
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, grid_n)
    # Vectorized: each row r gives S[r] + t*U[r] - t^2*B[r]
    # Shape: (grid_n, 2n-1)
    vals = S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :]
    row_max = np.max(vals, axis=1)
    best_idx = int(np.argmin(row_max))
    
    # Narrow to bracket
    lo_idx = max(0, best_idx - 2)
    hi_idx = min(grid_n - 1, best_idx + 2)
    lo = ts[lo_idx]
    hi = ts[hi_idx]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo > 1e-13:
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(S + c1 * U - c1*c1 * B))
        f2 = float(np.max(S + c2 * U - c2*c2 * B))
        for _ in range(golden_n):
            if hi - lo < 1e-13:
                break
            if f1 < f2:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float, copy=True)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-14:
        return v
    if s > target:
        if s < 1e-15:
            return np.zeros(n)
        v *= (target / s)
    else:
        remaining = target - s
        capacity = float(np.sum(1.0 - v))
        if capacity < 1e-15:
            return np.ones(n)
        alpha = remaining / capacity
        v = v + alpha * (1.0 - v)
    res = target - float(np.sum(v))
    if abs(res) > 1e-14:
        v = v.copy()
        if res > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += res / cnt
    return np.clip(v, 0.0, 1.0)

def _row_gradient(h, ell, n):
    g = np.zeros(n)
    j_min = max(0, -ell)
    j_max = min(n - 1, n - 1 - ell)
    if j_min > j_max:
        return g
    j = np.arange(j_min, j_max + 1)
    g[j + ell] += (1.0 - h[j])
    g[j] -= h[j + ell]
    return g

def _feasible_interval(h, d):
    t_lo = -np.inf
    t_hi = np.inf
    nz = np.abs(d) > 1e-14
    idx = np.where(nz)[0]
    for i in idx:
        di = d[i]
        hi = h[i]
        if di > 0:
            t_lo = max(t_lo, -hi / di)
            t_hi = min(t_hi, (1.0 - hi) / di)
        else:
            t_lo = max(t_lo, (1.0 - hi) / di)
            t_hi = min(t_hi, -hi / di)
    if t_lo >= t_hi:
        return 0.0, 0.0
    return t_lo, t_hi

def _line_search(S, U, B, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S))
    grid_n = 50
    ts = np.linspace(t_lo, t_hi, grid_n)
    vals = S[None, :] + ts[:, None] * U[None, :] - (ts[:, None])**2 * B[None, :]
    row_max = np.max(vals, axis=1)
    best_idx = int(np.argmin(row_max))
    lo_idx = max(0, best_idx - 2)
    hi_idx = min(grid_n - 1, best_idx + 2)
    lo = ts[lo_idx]
    hi = ts[hi_idx]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo > 1e-13:
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(S + c1 * U - c1*c1 * B))
        f2 = float(np.max(S + c2 * U - c2*c2 * B))
        for _ in range(25):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(S + c1 * U - c1*c1 * B))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(S + c2 * U - c2*c2 * B))
    t_opt = (lo + hi) / 2.0
    candidates = [t_lo, t_hi, t_opt, float(ts[best_idx]), 0.5*(t_lo+t_hi)]
    best_t = t_opt
    best_val = float('inf')
    for tc in candidates:
        if tc < t_lo - 1e-12 or tc > t_hi + 1e-12:
            continue
        vc = float(np.max(S + tc * U - tc*tc * B))
        if vc < best_val:
            best_val = vc
            best_t = tc
    return best_t, best_val

def _constrained_direction(h, S, top_k, target_mode, ridge=1e-6):
    """Solve constrained least-norm system for a dense direction."""
    n = len(h)
    L = 2 * n - 1
    # Select top-K rows
    row_vals = S  # unscaled
    # For target_mode: 0=min, 1=mean, 2=median
    if target_mode == 0:
        # top K by value
        top_idx = np.argsort(row_vals)[-top_k:]
    else:
        top_idx = np.argsort(row_vals)[-top_k:]
    
    selected = row_vals[top_idx]
    if target_mode == 0:
        t = float(np.min(selected))
    elif target_mode == 1:
        t = float(np.mean(selected))
    else:
        t = float(np.median(selected))
    
    # Build gradient matrix (K x n)
    K = len(top_idx)
    G = np.zeros((K, n))
    for i, r in enumerate(top_idx):
        ell = r - (n - 1)
        G[i] = _row_gradient(h, ell, n)
    
    # Compute Gram matrix and sums
    GtG = G @ G.T  # K x K
    s = G.sum(axis=1)  # K
    b = t - selected  # K (should be negative for rows we want to reduce)
    
    # Build (K+1) x (K+1) system
    M = np.zeros((K+1, K+1))
    M[:K, :K] = GtG
    M[:K, K] = s
    M[K, :K] = s
    M[K, K] = float(n)
    
    rhs = np.zeros(K+1)
    rhs[:K] = -2.0 * b
    rhs[K] = 0.0
    
    # Add ridge for stability
    M[:K, :K] += ridge * np.eye(K)
    
    try:
        sol = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(M, rhs, rcond=None)[0]
        except:
            return None
    
    lam = sol[:K]
    mu = sol[K]
    
    # Compute direction
    d = -(0.5 * (mu * np.ones(n) + G.T @ lam))
    
    # Ensure zero-sum (should already be, but fix tiny drift)
    d -= np.mean(d)
    
    # Check validity
    if not np.all(np.isfinite(d)):
        return None
    dmax = float(np.max(np.abs(d)))
    if dmax < 1e-12:
        return None
    
    return d / dmax  # normalized to max abs = 1

def _smooth_direction(h, S, beta=20.0):
    """Compute smooth gradient direction (weighted sum of row gradients)."""
    n = len(h)
    L = 2 * n - 1
    M = float(np.max(S))
    w = np.exp(beta * (S - M))
    sw = float(np.sum(w))
    if sw < 1e-30:
        K = max(1, L // 10)
        idx = np.argsort(S)[-K:]
        w = np.zeros(L)
        w[idx] = 1.0 / K
    else:
        w /= sw
    
    g = np.zeros(n)
    for r in range(L):
        if w[r] > 1e-10:
            ell = r - (n - 1)
            g += w[r] * _row_gradient(h, ell, n)
    
    g -= np.mean(g)
    dmax = float(np.max(np.abs(g)))
    if dmax < 1e-14:
        return None
    return g / dmax

def _make_step(h, d, S, target, move_cap):
    """Try a step along direction d with line search. Returns (h_new, c5_new, accepted)."""
    n = len(h)
    d = d * move_cap
    t_lo, t_hi = _feasible_interval(h, d)
    if t_hi - t_lo < 1e-13:
        return None, None, False
    
    one_minus_h = 1.0 - h
    U = np.correlate(d, one_minus_h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    
    t_opt, _ = _line_search(S, U, B, t_lo, t_hi)
    
    best_h = None
    best_c5 = float('inf')
    cur_c5 = float(np.max(S)) * (2.0 / n)
    
    for tc in [t_opt, t_lo * 0.999, t_hi * 0.999, 0.5 * (t_lo + t_hi), 0.3 * t_opt, 0.7 * t_opt]:
        if tc < t_lo - 1e-12 or tc > t_hi + 1e-12:
            continue
        h_try = h + tc * d
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    if best_h is not None and best_c5 < cur_c5 - 1e-12:
        return best_h, best_c5, True
    return None, None, False

def _refine_d
