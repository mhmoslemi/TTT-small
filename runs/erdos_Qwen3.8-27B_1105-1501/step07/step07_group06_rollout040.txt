```python
def _compute_base(h):
    """
    Compute base[p][k] for all positions p and correlation indices k.
    base[p] represents the "interaction" of position p with all lags.
    For a swap (a:1->0, b:0->1): new_corr = corr + base[b] - base[a]
    """
    n = len(h)
    R = 2 * n - 1
    base = np.zeros((n, R))
    
    for tau in range(1, n):
        # Positive lag: corr[n-1+tau] = sum_i h[i]*(1-h[i+tau])
        # base[p, n-1+tau] = (1-h[p+tau])*[p<n-tau] + h[p-tau]*[p>=tau]
        idx = n - 1 + tau
        vals = np.zeros(n)
        # (1-h[p+tau]) for p = 0,...,n-tau-1
        vals[:n-tau] = 1.0 - h[tau:tau+(n-tau)]
        # h[p-tau] for p = tau,...,n-1
        vals[tau:] += h[:n-tau]
        base[:, idx] = vals
        
        # Negative lag: corr[n-1-tau] = sum_j h[j+tau]*(1-h[j])
        # base[p, n-1-tau] = (1-h[p-tau])*[p>=tau] + h[p+tau]*[p<n-tau]
        idx2 = n - 1 - tau
        vals2 = np.zeros(n)
        vals2[tau:] = 1.0 - h[:n-tau]  # (1-h[p-tau]) for p=tau,...,n-1
        vals2[:n-tau] += h[tau:tau+(n-tau)]  # h[p+tau] for p=0,...,n-tau-1
        base[:, idx2] = vals2
    
    return base
```

        nc = corr[None, None, :] + base[zeros][:, None, :] - base[ones][None, :, :]
        scores = nc.max(axis=2) + eps * np.var(nc, axis=2)
        flat_idx = int(np.argmin(scores))
        bi = flat_idx // len(ones)
        ai = flat_idx % len(ones)
        best_score = scores[bi, ai]
        if best_score < (M + eps * np.var(corr)) - 1e-10:
            a = ones[ai]
            b = zeros[bi]
            h[a] = 0.0
            h[b] = 1.0
            corr = _corr_vals(h)
            base = _compute_base(h)
            M = float(np.max(corr))
            if M < best_M:
                best_M = M
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, _exact_c5(best_h)

def _sa_binary(h, deadline, T_init=0.3, T_min=0.005, cool=0.9995, beta_lse=8.0):
    n = len(h)
    h = h.copy()
    corr = _corr_vals(h)
    base = _compute_base(h)
    M = float(np.max(corr))
    dx = 2.0 / n
    lse_curr = float(np.max(corr) - np.log(np.sum(np.exp(beta_lse * (corr - M)))) / beta_lse)
    best_h = h.copy()
    best_c5 = M * dx
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    iters = 0
    while time.time() < deadline:
        iters += 1
        if iters % 2000 == 0 and stall > 300:
            break
        if stall > 500:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[int(rng.integers(len(ones)))]
        b = zeros[int(rng.integers(len(zeros)))]
        new_corr = corr + base[b] - base[a]
        new_M = float(np.max(new_corr))
        new_lse = float(new_M - np.log(np.sum(np.exp(beta_lse * (new_corr - new_M)))) / beta_lse)
        delta = (new_lse - lse_curr) * dx
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            h[a] = 0.0
            h[b] = 1.0
            corr = _corr_vals(h)
            base = _compute_base(h)
            M = new_M
            lse_curr = new_lse
            c5 = M * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            T = max(T * cool, T_min)
        else:
            stall += 1
            T = max(T * cool, T_min)
    return best_h, best_c5

def _double_swap_search(h, deadline, max_iters=5000):
    n = len(h)
    h = h.copy()
    corr = _corr_vals(h)
    base = _compute_base(h)
    M = float(np.max(corr))
    best_h = h.copy()
    best_M = M
    dx = 2.0 / n
    rng = np.random.default_rng(4242)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        a1 = ones[int(rng.integers(len(ones)))]
        a2 = ones[int(rng.integers(len(ones)))]
        if a1 == a2:
            continue
        b1 = zeros[int(rng.integers(len(zeros)))]
        b2 = zeros[int(rng.integers(len(zeros)))]
        if b1 == b2:
            continue
        new_corr = corr - base[a1] - base[a2] + base[b1] + base[b2]
        new_M = float(np.max(new_corr))
        if new_M < M - 1e-10:
            h[a1] = 0.0
            h[a2] = 0.0
            h[b1] = 1.0
            h[b2] = 1.0
            corr = _corr_vals(h)
            base = _compute_base(h)
            M = new_M
            if M < best_M:
                best_M = M
                best_h = h.copy()
    return best_h, _exact_c5(best_h)

def _smooth_gradient(h):
    n = len(h)
    corr = _corr_vals(h)
    beta = 15.0
    M = float(np.max(corr))
    v = beta * (corr - M)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    for r in np.where(w > 1e-8)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[tau:tau + m]
        else:
            d = -tau
            m = n - d
            if m > 0:
                g[:m] += wr * (1.0 - h[d:])
                g[d:] -= wr * h[:m]
    g -= float(np.mean(g))
    return g

def _line_search_c5(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-14
    neg = d < -1e-14
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    ng = 50
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = float(np.max(S0 + gt[i] * T - gt[i] * gt[i] * Q)) * dx
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo + hi) / 2.0
    opt_v = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    for t in [a_lo, a_hi, 0.0]:
        ft = float(np.max(S0 + t * T - t * t * Q)) * dx
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _continuous_refine(h, deadline, radius=0.12, max_stall=60):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    while time.time() < deadline and stall < max_stall:
        g = _smooth_gradient(h)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        d = g / gn
        d -= np.mean(d)
        dn = float(np.linalg.norm(d))
        if dn < 1e-14:
            break
        d /= dn
        alpha, val = _line_search_c5(h, d, radius_cur)
        if alpha != 0.0 and val < cur_c5 - 1e-14:
            h_new = _project(h + alpha * d, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
                radius_cur = min(radius_cur * 1.05, 0.3)
            else:
                stall += 1
                radius_cur = max(0.03, radius_cur * 0.7)
        else:
            stall += 1
            radius_cur = max(0.03, radius_cur * 0.75)
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_binary(h, rng, n_flips=20):
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for _ in range(min(n_flips, len(ones), len(zeros))):
        a = ones[int(rng.integers(len(ones)))]
        b = zeros[int(rng.integers(len(zeros)))]
        h[a] = 0.0
        h[b] = 1.0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick greedy binary search (10% budget)
    d1 = t0 + budget_s * 0.10
    h_bin = _to_binary(h_init)
    h_bin, c5_bin = _greedy_binary(h_bin, d1)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Binary SA (35% budget)
    d2 = t0 + budget_s * 0.45
    n_curr = len(best_h)
    h_sa = _to_binary(best_h)
    h_sa, c5_sa = _sa_binary(h_sa, d2, T_init=0.4, T_min=0.005, cool=0.9994)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Double swap escape (10% budget)
    d3 = t0 + budget_s * 0.55
    h_ds = _to_binary(best_h)
    h_ds, c5_ds = _double_swap_search(h_ds, d3, max_iters=3000)
    if c5_ds < best_c5:
        best_c5 = c5_ds
        best_h = h_ds.copy()
    if time.time() < d3:
        h_ds2 = _to_binary(best_h)
        h_g2, c5_g2 = _greedy_binary(h_ds2, min(d3, time.time() + budget_s * 0.03), eps=0.02)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 4: Multi-n exploration (20% budget)
    d4 = t0 + budget_s * 0.75
    n_base = len(best_h)
    n_options = []
    for dn in [-30, -20, -10, 10, 20, 30, -40, 40]:
        nt = n_base + dn
        if 60 <= nt <= 260 and nt % 2 == 0 and nt != n_base:
            n_options.append(nt)
    for nt in n_options:
        if time.time() > d4:
            break
        if time.time() > d4 - 5:
            break
        h_r = _resample(best_h, nt)
        h_rb = _to_binary(h_r)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_rb, c5_rb = _greedy_binary(h_rb, sub_dl)
        if time.time() < d4 - 3:
            sub_dl2 = min(d4 - 2, time.time() + budget_s * 0.015)
            h_rb, c5_rb2 = _sa_binary(h_rb, sub_dl2, T_init=0.3, cool=0.9995)
            if c5_rb2 < c5_rb:
                c5_rb = c5_rb2
                h_rb = h_rb.copy()
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()
            n_base = nt

    # Phase 5: Continuous refinement (15% budget)
    d5 = t0 + budget_s * 0.90
    h_cont = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.06)
    h_cont, c5_cont = _continuous_refine(h_cont, sub_dl, radius=0.10, max_stall=50)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()
    if time.time() < d5:
        h_cont2 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.04)
        h_cont2, c5_cont2 = _continuous_refine(h_cont2, sub_dl2, radius=0.06, max_stall=40)
        if c5_cont2 < best_c5:
            best_c5 = c5_cont2
            best_h = h_cont2.copy()

    # Phase 6: Basin hopping with binary perturbations (15% budget)
    d6 = deadline
    n_bp = len(best_h)
    scale = 15
    while time.time() < d6 - 15 and scale >= 3:
        h_p = _perturb_binary(_to_binary(best_h), rng, n_flips=scale)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.008)
        h_p, c5_p = _greedy_binary(h_p, sub_dl)
        if time.time() < d6 - 5:
            sub_dl2 = min(d6 - 5, time.time() + budget_s * 0.005)
            h_p, c5_p2 = _sa_binary(h_p, sub_dl2, T_init=0.2, cool=0.9996)
            if c5_p2 < c5_p:
                c5_p = c5_p2
                h_p = h_p.copy()
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            scale = max(3, int(scale * 0.8))
        else:
            scale = min(30, int(scale * 1.15) + 1)

    # Final polish
    if deadline - time.time() > 8:
        h_f = _to_binary(best_h)
        h_f, c5_f = _greedy_binary(h_f, deadline - 2, eps=0.01)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        hf_cont = _project(best_h.copy(), len(best_h) / 2.0)
        hf_cont, c5_fc = _continuous_refine(hf_cont, deadline - 1, radius=0.05, max_stall=30)
        if c5_fc < best_c5:
            best_c5 = c5_fc
            best_h = hf_cont.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
