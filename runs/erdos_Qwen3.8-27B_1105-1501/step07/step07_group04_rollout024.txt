def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_detail(B, T, Q, dx, t_max, n_grid=24):
    if t_max < 1e-14:
        return 0.0, float(np.max(B) * dx)
    ts = np.linspace(0.0, t_max, n_grid)
    vals = np.max(B[None, :] + ts[:, None] * T[None, :] - (ts * ts)[:, None] * Q[None, :], axis=1) * dx
    bi = int(np.argmin(vals))
    best_t = ts[bi]
    best_val = float(vals[bi])
    lo_b = ts[max(0, bi - 2)]
    hi_b = ts[min(n_grid - 1, bi + 2)]
    if hi_b - lo_b > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(B + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx

def _pair_search_one_pass(h, dx, B, n_grid=24, max_pairs=None):

    for i in range(n):
        for j in range(i + 1, n):
            if pairs_tried >= total_pairs:
                break
            pairs_tried += 1
            # Direction: move from j to i (h[i] increases, h[j] decreases)
            t_max1 = min(1.0 - h[i], h[j])

            # Try move from j to i: d = e_i - e_j
            if t_max1 > 1e-12:

                    Q[rn] = -1.0

                cur_max = float(np.max(B) * dx)
                if cur_max - bv > best_improve:
                    best_improve = cur_max - bv
                    best_pair = (i, j, 1)  # move from j to i
                    best_t = bt
            if t_max2 > 1e-12:
                # Move from i to j: d = e_j - e_i
                T = np.zeros(2 * n - 1)
                lo1 = max(0, nm1 - j)
                hi1 = min(2 * n - 2, 2 * n - 2 - j)

def _pair_search_loop(h_init, deadline, n_grid=24, stall_limit=50):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_init.copy(), target)

    best_h = h.copy()

    stall = 0
    B = np.correlate(h, 1.0 - h, mode='full')
    while time.time() < deadline:
        best_pair, best_t, best_improve = _pair_search_one_pass(h, dx, B, n_grid)
        if best_pair is not None and best_improve > 1e-12:
            i, j, direction = best_pair
            if direction == 1:
                h[i] += best_t
                h[j] -= best_t
            else:
                h[i] -= best_t
                h[j] += best_t
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            cur = _exact_c5(h)
            B = np.correlate(h, 1.0 - h, mode='full')
            if cur < best_c5:

        else:

def _random_direction_search(h_init, deadline, n_grid=20, max_iter=5000, seed=42):

            break

            stall += 1

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

        h2c, c5_2c = _pair_search_loop(h_pert, d2b, n_grid=16, stall_limit=30)
        if c5_2c < best_c5:
            best_c5 = c5_2c
            best_h = h2c.copy()

    # Phase 3: Multi-n exploration (200s)
    d3 = t0 + budget_s * 0.60
    n_best = len(best_h)
    tried_ns = set([n_best])
    for dn in [-12, -8, -4, 4, 8, 12, -20, 20, -16, 16, -6, 6]:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _pair_search_loop(h_r, sub_dl, n_grid=16, stall_limit=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Binary refinement (150s)
    d4 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    h_bin = _to_binary(best_h, n_cur)
    c5_bin = _exact_c5(h_bin)
    # Targeted binary search
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_targeted_search(h_bin, sub_dl, seed=seed+200)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Random binary search
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_pair_search(best_h if not np.allclose(best_h, np.round(best_h)) else h_bin, sub_dl, max_stall=300, seed=seed+300)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Block swap
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h_bs, c5_bs = _block_swap(_to_binary(best_h, len(best_h)), sub_dl, seed=seed+400)
    if c5_bs < best_c5:
        best_c5 = c5_bs
        best_h = h_bs.copy()

    # Phase 5: Final intensive refinement (budget_s * 0.20)
    d5 = deadline - 10
    n_fin = len(best_h)
    # Continuous pair search
    sub_dl = min(d5, time.time() + budget_s * 0.06)
    h5, c5_5 = _pair_search_loop(best_h, sub_dl, n_grid=24, stall_limit=50)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Random direction polish
    if time.time() < d5 - 20:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h5b, c5_5b = _random_direction_search(best_h, sub_dl, n_grid=16, max_iter=2000, seed=seed+500)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Basin hopping
    if time.time() < d5 - 15:
        scale = 0.03
        for _ in range(30):
            if time.time() > d5 - 5:
                break
            h_pert = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d5, time.time() + budget_s * 0.006)
            h_t, c5_t = _pair_search_loop(h_pert, sub_dl, n_grid=16, stall_limit=15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.1)

    # Try a few more n values near the best
    n_final = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > deadline - 15:
            break
        n_try = n_final + dn
        if n_try < 80:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline - 10, time.time() + 8)
        h_r, c5_r = _pair_search_loop(h_r, sub_dl, n_grid=16, stall_limit=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_final = n_try

    # Final projection and verification
    n_out = len(best_h)
    best_h = _project(best_h.copy(), n_out / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Something went wrong, recompute
        best_c5 = final_c5

    return best_h, best_c5, n_out
