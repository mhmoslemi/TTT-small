import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _project_sum(v, target):
    # Project v onto {x: sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_weights(scores, tau):
    """Compute softmax weights for LSE gradient."""
    m = np.max(scores)
    u = (scores - m) / tau
    exp_u = np.exp(u)
    return exp_u / np.sum(exp_u)

def _lse_gradient(h, tau):
    """Compute gradient of LSE objective."""
    scores = _scores(h)
    p = _lse_weights(scores, tau)
    G = _grad_matrix(h)
    return G.T @ p

    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _project_sum(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 48
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 4)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 4)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    return min(cands, key=lambda x: x[1])

def _lse_grad(h, tau):
    scores = _scores(h)
    m = np.max(scores)
    u = (scores - m) / tau
    eu = np.exp(u)
    p = eu / np.sum(eu)
    G = _grad_matrix(h)
    return G.T @ p, p

def _top_k_grad(h, k):
    scores = _scores(h)
    idx = np.argsort(scores)[::-1][:k]
    G = _grad_matrix(h)
    return -G[idx].sum(axis=0) / float(k)

def _make_direction(h, tau, k_fallback=2):
    """Get a descent direction: LSE gradient, fallback to top-k."""
    grad, p = _lse_grad(h, tau)
    grad = grad - np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        d = _top_k_grad(h, k_fallback)
        d = d - np.mean(d)
        gnorm = np.linalg.norm(d)
        if gnorm < 1e-12:
            return None
        return -d / gnorm
    return -grad / gnorm

def _lse_anneal(h, deadline, tau_init=2.0, tau_final=0.02, n_temps=40,
                radius=0.15, max_inner=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    taus = np.geomspace(tau_init, tau_final, n_temps)
    total_stall = 0
    for tau in taus:
        if time.time() > deadline:
            break
        stall = 0
        for _ in range(max_inner):
            if time.time() > deadline or stall > 35:
                break
            d = _make_direction(h, tau)
            if d is None:
                break
            alpha, new_c5 = _line_search(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project_sum(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                total_stall = 0
            else:
                stall += 1
                total_stall += 1
        if total_stall > 300:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _pair_sweep(h, deadline, max_rounds=30):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(ones[rng.integers(len(ones))])
        oz = int(zeros[rng.integers(len(zeros))])
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=6000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _slsqp_polish(h, deadline, tau=0.1, maxiter=200):
    """Polish using SLSQP on LSE objective."""
    from scipy.optimize import minimize
    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    if time.time() > deadline:
        return h0, _exact_c5(h0)
    def obj(x):
        s = _scores(x)
        m = np.max(s)
        u = (s - m) / tau
        return m + tau * np.log(np.sum(np.exp(u)))
    def grad(x):
        g, _ = _lse_grad(x, tau)
        g = g - np.mean(g)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.success and res.fun < 1e6:
        h_new = _project_sum(res.x, target)
        return h_new, _exact_c5(h_new)
    return h0, _exact_c5(h0)

    rng = np.random.default_rng(seed)

    # Final LSE with very low tau

    return best_h, final_c5, len(best_h)
