            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _pair_optimize(h, deadline, max_iter=30000, seed=999):
    """Optimize pairs of coordinates simultaneously with exact quadratic line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        # Feasible range
        t_hi = 1.0
        t_hi = min(t_hi, h[i] if d[i] < 0 else (1.0 - h[i])/d[i])  # wait, need to be careful
        # d[i] = 1, so h[i] + t*d[i] <= 1 means t <= 1-h[i]
        # d[j] = -1, so h[j] + t*d[j] >= 0 means t <= h[j]
        t_hi = min(1.0, 1.0 - h[i], h[j])
        if t_hi < 1e-14:
            stall += 1
            continue
        # Also try the reverse direction
        d2 = -d
        t_hi2 = min(1.0, 1.0 - h[j], h[i])
        if t_hi2 < 1e-14:
            t_hi2 = 0
        # Line search in both directions
        best_t, best_v = 0.0, cur_c5
        if t_hi > 1e-14:
            t1, v1 = _line_search(S0, T, C, t_hi, dx)
            if v1 < best_v:
                best_t, best_v = t1, v1
        if t_hi2 > 1e-14:
            T2 = -T
            t2, v2 = _line_search(S0, T2, C, t_hi2, dx)
            if v2 < best_v:
                best_t, best_v = -t2, v2
        if best_v < cur_c5 - 1e-13:
            h = h + best_t * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _line_search(S0, T, C, t_hi, dx):
    """Exact line search for max(S0 + t*T - t^2*C)*dx on [0, t_hi]."""
    if t_hi < 1e-15:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_t, hi_t = 0.0, t_hi
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    f0 = float(np.max(S0)) * dx
    for _ in range(50):
        if hi_t - lo_t < 1e-12:
            break
        if f1 < f2:
            hi_t = c2
            c2 = c1
            f2 = f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_t = c1
            c1 = c2
            f1 = f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_t + hi_t) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    # Also check endpoints
    fe = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
    candidates = [(0.0, f0), (opt_t, opt_val), (t_hi, fe)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]
