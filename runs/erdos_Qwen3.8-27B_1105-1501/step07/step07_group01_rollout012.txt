def _compute_grad(h, j):
    """Compute gradient vector g for coordinate j."""
    n = len(h)
    g = np.zeros(2 * n - 1)
    for d in range(0, n - j):
        g[n - 1 + d] = (1.0 - h[j + d])
    for d in range(-j, 0):
        g[n - 1 + d] = -h[j + d]
    return g

def _coord_descent(h, deadline, max_sweeps=5000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = _compute_grad(h, j)
            lo_d = -old
            hi_d = 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue

            lo_t, hi_t = lo_d, hi_d
            for _ in range(30):

                if g[k_star] < 0:
                    lo_t = mid
                else:
                    hi_t = mid
            opt_d = (lo_t + hi_t) / 2.0
            # Also check endpoints
            f_mid = float(np.max(S0 + opt_d * g))
            f_lo = float(np.max(S0 + lo_d * g))
            f_hi = float(np.max(S0 + hi_d * g))
            best_f = min(f_mid, f_lo, f_hi)
            if best_f == f_lo:
                opt_d = lo_d
            elif best_f == f_hi:
                opt_d = hi_d
            new_val = old + opt_d
            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                h[j] = new_val
                S0 = S0 + delta * g
                improved = True
        
        h = _project(h, target)
        S0 = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S0)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5
