    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _make_anti(x, m):

    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

    m = n // 2
    x = 0.5 * (h[:m] + (1.0 - h[n-1::-1]))
    return np.clip(x, 0.0, 1.0)

def _full_grad_smooth(h, beta):
    """Compute smooth max value and gradient w.r.t. h (before projection)."""
    n = len(h)
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')
    mx = float(np.max(corr))
    z = beta * (corr - mx)
    z = np.clip(z, -500.0, 0.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    smax = mx
    if wsum > 1e-30:
        w /= wsum
        smax = mx + np.log(wsum) / beta

        g -= np.mean(g)  # project to sum-zero subspace

        a_hi = radius

        if np.any(neg):
            a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))

        if a_hi <= a_lo + 1e-14:
            stall += 1
            continue
        
        # Golden section search
        a_lo2 = a_lo
        a_hi2 = a_hi
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = a_hi2 - gr * (a_hi2 - a_lo2)
        c2 = a_lo2 + gr * (a_hi2 - a_lo2)
        h1 = _project(h + c1 * d, target)
        h2 = _project(h + c2 * d, target)

def _pair_move(h, deadline, max_improves=100):
    """Try pair moves (swap values between two positions)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n_improve = 0
    rng = np.random.default_rng(456)
    
    for _ in range(max_improves * 10):
        if time.time() > deadline or n_improve >= max_improves:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Try to move mass from i to j (or vice versa)
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        
        # Line search
        a_lo = -min(h[i], 1.0 - h[j])
        a_hi = min(1.0 - h[i], h[j])
        if a_hi <= a_lo + 1e-14:
            continue
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = a_lo, a_hi
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h1 = h + c1 * d
        h2 = h + c2 * d
        f1 = _c5_exact(h1)
        f2 = _c5_exact(h2)
        for _ in range(30):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                h1 = h + c1 * d
                f1 = _c5_exact(h1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                h2 = h + c2 * d
                f2 = _c5_exact(h2)
        
        opt_a = (lo + hi) / 2.0
        h_new = h + opt_a * d
        new_c5 = _c5_exact(h_new)
        
        if new_c5 < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            n_improve += 1
    
    return best_h, best_c5

def _binary_sa(h, deadline, max_iter=20000, T_init=0.003):
    """Simulated annealing on binary (0/1) values."""
    n = len(h)
    target = n // 2
    h = np.round(h)
    h = np.clip(h, 0, 1)
    # Ensure exactly target ones
    ones = np.where(h > 0.5)[0]
    if len(ones) > target:
        # Remove some
        extra = ones[len(ones)-target:]
        h[extra] = 0
    elif len(ones) < target:
        zeros = np.where(h <= 0.5)[0]
        need = target - len(ones)
        h[zeros[:need]] = 1
    
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(789)
    T = T_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(ones[rng.integers(len(ones))])
        oz = int(zeros[rng.integers(len(zeros))])
        h[oi] = 0
        h[oz] = 1
        new_c5 = _c5_exact(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            T *= 0.999
        else:
            h[oi] = 1
            h[oz] = 0
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
