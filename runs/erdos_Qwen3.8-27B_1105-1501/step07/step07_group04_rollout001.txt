        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_exact(h, d, dx, t_max=None):
    """Exact line search along direction d (with sum(d)=0) using the quadratic structure."""
    n = len(h)
    B = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible range
    lo = 0.0
    hi = t_max if t_max is not None else 1.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        hi = min(hi, float(np.min(h[neg] / (-d[neg]))))
    if hi < 1e-14:
        return 0.0, float(np.max(B) * dx)
    
    # Grid search + golden section
    n_grid = 40
    grid_t = np.linspace(0.0, hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(B + t * T - t * t * Q)) * dx
    best_idx = int(np.argmin(grid_f))
    
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(B + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(B + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(B + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(B + opt_t * T - opt_t * opt_t * Q)) * dx
    
    candidates = [(0.0, float(np.max(B) * dx)), (opt_t, opt_val), (hi, float(np.max(B + hi * T - hi * hi * Q)) * dx)]
    return min(candidates, key=lambda x: x[1])

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
