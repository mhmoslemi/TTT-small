def _project(v, target):

    v = np.asarray(v, dtype=float)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _lag_gradient(h, t):
    """Gradient of the t-th correlation output w.r.t. h"""
    n = len(h)
    G = np.zeros(n)
    lag = t - (n - 1)
    if lag >= 0:
        L = n - lag
        if L > 0:
            G[:L] += 1.0 - h[lag:lag + L]
            G[lag:lag + L] -= h[:L]
    else:
        s = -lag
        L = n - s
        if L > 0:
            G[s:s + L] += 1.0 - h[:L]
            G[:L] -= h[s:s + L]
    return G

def _smooth_grad(h, beta):
    """Softmax-weighted gradient of max overlap"""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]

            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _project(v, target):

    v = np.asarray(v, dtype=float)

        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _lag_gradient_direct(h, t):
    n = len(h)
    G = np.zeros(n)
    lag = t - (n - 1)
    if lag >= 0:
        L = n - lag
        if L > 0:
            G[:L] += 1.0 - h[lag:lag + L]
            G[lag:lag + L] -= h[:L]
    else:
        s = -lag
        L = n - s
        if L > 0:
            G[s:s + L] += 1.0 - h[:L]
            G[:L] -= h[s:s + L]
    return G

def _lp_direction(h, top_k, r):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-S0)[:top_k]
    A = np.zeros((len(top_idx), n))
    for i, t in enumerate(top_idx):
        A[i] = _lag_gradient_direct(h, t)
    for i in range(len(A)):
        nrm = np.linalg.norm(A[i])
        if nrm > 1e-14:
            A[i] /= nrm
    try:
        import cvxpy as cp
        s = cp.Variable(n)
        tau = cp.Variable()
        obj = cp.Maximize(tau)
        constraints = [cp.sum(s) == 0, s >= -r, s <= r, tau >= 0]
        for i in range(len(A)):
            constraints.append(A[i] @ s + tau <= 0)
        prob = cp.Problem(obj, constraints)
        solved = False
        for solver in [cp.SCS, cp.ECOS, cp.CVXOPT, cp.CLARABEL]:
            try:
                t_start = time.time()
                prob.solve(solver=solver, verbose=False)
                if time.time() - t_start > 30:
                    continue
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved or s.value is None:
            return None
        return np.array(s.value, dtype=float)
    except Exception:
        return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _triplet_opt(h, deadline, max_it=20000):
    """Triplet exchange: move mass from one position to two others (or vice versa)"""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31415)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if i == j or i == k or j == k:
            continue
        # Try: decrease at i, increase at j and k (or reverse)
        for sign in [1, -1]:
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign * 0.5
            d[k] = sign * 0.5
            a, c5t = _line_search(h, d, dx, -1.0, 1.0)
            if c5t < cur - 1e-14:
                h_new = h + a * d
                h_new = _project(h_new, target)
                c5a = _exact_c5(h_new)
                if c5a < cur - 1e-14:
                    h = h_new
                    cur = c5a
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    stall = 0
                    break
        else:
            stall += 1
    return best_h, best_c
