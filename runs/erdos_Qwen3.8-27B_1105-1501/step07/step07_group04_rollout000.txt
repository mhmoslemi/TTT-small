import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v to [0,1]^n with sum(v) = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_and_grad(h, active_indices=None):
    """
    Compute correlation values and gradients for active constraints.
    Returns: corr_values (array of length 2n-1), 
             grad_matrix (len(active_indices) x n)
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1
    
    if active_indices is None:
        active_indices = np.arange(2*n - 1)
    
    m = len(active_indices)
    G = np.zeros((m, n))
    
    for idx, r in enumerate(active_indices):
        k = r - center
        if k == 0:
            G[idx, :] = (1.0 - 2.0 * h) * dx
        elif k > 0:
            # F_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # grad w.r.t. h[j]:
            #   j in [k, n-1]: (1 - h[j-k])
            #   j in [0, n-1-k]: -h[j+k]
            G[idx, k:] += (1.0 - h[:n-k])
            G[idx, :n-k] -= h[k:]
            G[idx, :] *= dx
        else:
            u = -k
            # F_{-u} = sum_{i=0}^{n-1-u} h[i]*(1-h[i+u])
            # grad w.r.t. h[j]:
            #   j in [0, n-1-u]: (1 - h[j+u])
            #   j in [u, n-1]: -h[j-u]
            G[idx, :n-u] += (1.0 - h[u:])
            G[idx, u:] -= h[:n-u]
            G[idx, :] *= dx
    
    return corr, G

def _select_active(corr, n_active, prev_active=None):
    """Select active constraint indices for LP."""
    m = len(corr)
    max_val = float(corr.max())
    # Include all constraints within a tolerance of the max
    threshold = max_val - 1e-10
    active = np.where(corr >= threshold)[0]
    
    # Also include top n_active by value
    sorted_idx = np.argsort(-corr)
    top = sorted_idx[:n_active]
    
    # Union
    active_set = set(active.tolist()) | set(top.tolist())
    if prev_active is not None:
        active_set |= set(prev_active)
    
    result = np.array(sorted(active_set))
    return result

def _lp_direction(h, corr, active, radius):
    """
    Solve LP to find direction d that reduces max active constraint.
    Variables: d[0..n-1], t
    Minimize: t + eps * sum(|d[i]|) (approximated with auxiliary vars)
    Subject to:
      G @ d - t <= -corr[active]  (for each active constraint)
      sum(d) = 0
      -min(h[i], radius) <= d[i] <= min(1-h[i], radius)
    """
    n = len(h)
    m = len(active)
    dx = 2.0 / n
    
    # Build gradient matrix
    G = np.zeros((m, n))
    center = n - 1
    for idx, r in enumerate(active):
        k = r - center
        if k == 0:
            G[idx, :] = 1.0 - 2.0 * h
        elif k > 0:
            G[idx, k:] += (1.0 - h[:n-k])
            G[idx, :n-k] -= h[k:]
        else:
            u = -k
            G[idx, :n-u] += (1.0 - h[u:])
            G[idx, u:] -= h[:n-u]
        G[idx, :] *= dx
    
    # LP: minimize t + eps * sum(s_i)
    # vars: d[0..n-1], t, s[0..n-1]
    # s[i] >= d[i], s[i] >= -d[i], s[i] >= 0
    # total vars: n + 1 + n = 2n + 1
    eps = 1e-6
    
    # Objective
    c = np.zeros(2*n + 1)
    c[n] = 1.0  # t
    c[n+1:] = eps  # s
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint: G[i,:] @ d - t <= -corr[active][i]
    # d vars are [0..n-1], t is at index n, s vars are [n+1..2n]
    A_ub = np.zeros((m, 2*n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr[active]
    
    # s[i] >= d[i] => -d[i] + s[i] >= 0 => d[i] - s[i] <= 0
    # s[i] >= -d[i] => d[i] + s[i] >= 0 => -d[i] - s[i] <= 0
    # s[i] >= 0
    # These are:
    # d[i] - s[i] <= 0
    # -d[i] - s[i] <= 0
    
    # Actually, for the L1 penalty, we need:
    # s[i] >= d[i] and s[i] >= -d[i]
    # => d[i] - s[i] <= 0
    # => -d[i] - s[i] <= 0
    
    # Let's add these as additional inequality constraints
    # Total inequality constraints: m + 2n
    A_full = np.zeros((m + 2*n, 2*n + 1))
    A_full[:m, :n] = G
    A_full[:m, n] = -1.0
    b_full = np.concatenate([b_ub, np.zeros(2*n)])
    
    for i in range(n):
        # d[i] - s[i] <= 0
        A_full[m + i, i] = 1.0
        A_full[m + i, n + 1 + i] = -1.0
        # -d[i] - s[i] <= 0
        A_full[m + n + i, i] = -1.0
        A_full[m + n + i, n + 1 + i] = -1.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, 2*n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -min(h[i], radius)
        hi = min(1.0 - h[i], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t
    for i in range(n):
        bounds.append((0.0, None))  # s
    
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            # Remove mean to ensure sum(d) = 0 (numerical)
            d = d - np.mean(d)
            return d
    except Exception:
        pass
    return None

def _line_search_quad(h, d, dx, t_hi, n_grid=50):
    """
    Exact line search using quadratic expansion.
    h(t) = h + t*d
    corr(t) = C0 + t*L1 - t²*C2
    """
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C2 = np.correlate(d, d, mode='full')
    
    best_t = 0.0
    best_val = float(np.max(C0) * dx)
    
    # Grid search
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(C0 + t*L1 - t*t*C2)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement
    lo_b = grid_t[max(0, best_idx - 1)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    if hi_b - lo_b > 1e-12:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(C0 + c1*L1 - c1*c1*C2)) * dx
        f2 = float(np.max(C0 + c2*L1 - c2*c2*C2)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(C0 + c1*L1 - c1*c1*C2)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(C0 + c2*L1 - c2*c2*C2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(C0 + opt_t*L1 - opt_t*opt_t*C2)) * dx
        if opt_val < best_val:
            best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

def _lp_optimize(h, deadline, radius_init=0.10, max_iter=2000, n_active_max=80):
    """Main LP-based local optimizer."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    prev_active = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if radius < 1e-7:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        active = _select_active(corr, n_active_max, prev_active)
        d = _lp_direction(h, corr / dx, active, radius)  # pass raw corr for grad computation
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            # Fallback: subgradient
            max_val = float(corr.max())
            active2 = np.where(corr >= max_val - 1e-10)[0]
            g = np.zeros(n)
            center = n - 1
            for r in active2:
                k = r - center
                if k == 0:
                    g += (1.0 - 2.0 * h) * dx
                elif k > 0:
                    g[k:] += (1.0 - h[:n-k]) * dx
                    g[:n-k] -= h[k:] * dx
                else:
                    u = -k
                    g[:n-u] += (1.0 - h[u:]) * dx
                    g[u:] -= h[:n-u] * dx
            g /= len(active2)
            d = -g - np.mean(-g)
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            radius *= 0.7
            continue
        
        # Compute max feasible step
        t_hi = radius
        pos = d > 1e-15
        neg = d < -1e-15
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        
        if t_hi < 1e-12:
            stall += 1
            radius *= 0.5
            continue
        
        alpha, new_c5 = _line_search_quad(h, d, dx, t_hi)
        
        if new_c5 < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                prev_active = active
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.65
    
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_flip_search(h_bin, deadline, max_iters=50000, seed=42):
    """Local binary search: flip one 1 to 0 and one 0 to 1."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    T = 0.003
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        if not ones or not zeros:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or (delta < 0.05 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            else:
                stall += 1
            T = max(0.0005, T * 0.999)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
