    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

    """
    Dinic's algorithm for max flow.

        adj[v].append([u, 0, len(adj[u]) - 1])

                    if ret > 0:
                        adj[u][it[u]][1] -= ret
                        adj[v][rev][1] += ret
                        return ret
                it[u] += 1
            return 0

            pushed = dfs(s, float('inf'))
            if pushed == 0:
                break
            flow += pushed

    reachable = set()
    queue = [s]
    reachable.add(s)
    for qi in range(len(queue)):
        u = queue[qi]
        for v, cap, _ in adj[u]:
            if cap > 0 and v not in reachable:
                reachable.add(v)
                queue.append(v)
    
    return flow, reachable
