import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return x

def _feasible_alpha(h, d):
    """Compute maximum alpha such that h + alpha*d is in [0,1]."""
    pos = d > 1e-14
    neg = d < -1e-14
    alpha = 1.0  # default
    if np.any(pos):
        alpha = min(alpha, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha = min(alpha, float(np.min(h[neg] / (-d[neg]))))
    return max(alpha, 0.0)

def _line_search(h, d, dx):
    """Exact line search along feasible direction d. Returns (best_alpha, best_c5)."""
    alpha_max = _feasible_alpha(h, d)
    if alpha_max < 1e-14:
        return 0.0, _exact_c5(h)
    
    S0 = _raw_corr(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find candidates
    cands = [0.0, alpha_max]
    
    # Vertices of individual quadratics
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if 0.0 <= av <= alpha_max:
                cands.append(av)
    
    # Grid
    ng = min(48, max(24, int(alpha_max * 100)))
    cands.extend(np.linspace(0.0, alpha_max, ng).tolist())
    
    # Pairs of top-quadratic intersection points
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            # Solve S0[ki] + a*T[ki] - a^2*Q[ki] = S0[kj] + a*T[kj] - a^2*Q[kj]
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if 0.0 <= ar <= alpha_max:
                        cands.append(ar)
            else:
                disc = B*B - 4.0*A*C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B+sq)/(2.0*A), (-B-sq)/(2.0*A)]:
                        if 0.0 <= ar <= alpha_max:
                            cands.append(ar)
    
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(0.0, min(alpha_max, a))
        val = float(np.max(S0 + a*T - a*a*Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    # Find the interval around best_a
    lo_r = max(0.0, best_a - 0.05 * alpha_max)
    hi_r = min(alpha_max, best_a + 0.05 * alpha_max)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
        f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-11:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
        fmid = float(np.max(S0 + ((lo_r+hi_r)/2.0)*T - ((lo_r+hi_r)/2.0)**2*Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = (lo_r + hi_r) / 2.0
    
    return best_a, best_val

            m = n - s

    #             -r <= d_i <= r

        c = np.zeros(n+1)
        c[-1] = 1.0  # minimize tau
        
        A_ub = np.zeros((top_k, n+1))
        for i, g in enumerate(grads):
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0  # g^T d - tau <= 0
        b_ub = np.zeros(top_k)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return x

def _feasible_alpha(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    alpha = 10.0
    if np.any(pos):
        alpha = min(alpha, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha = min(alpha, float(np.min(h[neg] / (-d[neg]))))
    return max(alpha, 0.0)

def _line_search(h, d, dx):
    alpha_max = _feasible_alpha(h, d)
    if alpha_max < 1e-14:
        return 0.0, _exact_c5(h)
    S0 = _raw_corr(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    cands = [0.0, alpha_max]
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if 0.0 <= av <= alpha_max:
                cands.append(av)
    ng = min(48, max(24, int(alpha_max * 80) + 1))
    cands.extend(np.linspace(0.0, alpha_max, ng).tolist())
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if 0.0 <= ar <= alpha_max:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if 0.0 <= ar <= alpha_max:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(0.0, min(alpha_max, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(0.0, best_a - 0.06 * alpha_max)
    hi_r = min(alpha_max, best_a + 0.06 * alpha_max)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(35):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        fmid = float(np.max(S0 + ((lo_r + hi_r) / 2.0) * T - ((lo_r + hi_r) / 2.0) ** 2 * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = (lo_r + hi_r) / 2.0
    return best_a, best_val

def _lag_gradient(h, p):
    n = len(h)
    l = p - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:l + m])
            g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s + m]
    return g

def _lp_direction(h, top_k, r):
    n = len(h)
    raw = _raw_corr(h)
    idx = np.argsort(-raw)[:top_k]
    grads = []
    for p in idx:
        g = _lag_gradient(h, int(p))
        grads.append(g)
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        for i in range(top_k):
            A_ub[i, :n] = grads[i]
            A_ub[i, n] = -1.0
        b_ub = np.zeros(top_k)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-r, r)] * n + [(float('-inf'), 3.0 * r)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n].copy()
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-12:
                return d / nrm
        return None
    except Exception:
        return None

def _single_grad_dir(h):
    raw = _raw_corr(h)
    p = int(np.argmax(raw))
    g = _lag_gradient(h, p)
    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-12:
        return -g / nrm
    return None

def _weighted_grad_dir(h, beta):
    n = len(h)
    raw = _raw_corr(h)
    sc = raw * (2.0 / n)
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -60.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(raw)
        w[np.argmax(raw)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l_idx in range(-(n - 1), n):
        wl = w[n - 1 + l_idx]
        if wl < 1e-15:
            continue
        if l_idx >= 0:
            m = n - l_idx
            if m > 0:
                g[:m] += wl * (1.0 - h[l_idx:l_idx + m])
                g[l_idx:] -= wl * h[:m]
        else:
            s = -l_idx
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= h[s:s + m]
    g -= np.mean(g)
    nrm = np.linalg.norm(g)
    if nrm > 1e-12:
        return -g / nrm
    return None

def _gen_dirs(h, rng, r):
    dirs = []
    for k in [3, 5, 8]:
        d = _lp_direction(h, k, r)
        if d is not None:
            dirs.append(d)
    d = _single_grad_dir(h)
    if d is not None:
        dirs.append(d)
    d = _weighted_grad_dir(h, 30.0)
    if d is not None:
        dirs.append(d)
    d = _weighted_grad_dir(h, 60.0)
    if d is not None:
        dirs.append(d)
    if len(dirs) < 6:
        i = int(rng.integers(0, len(h)))
        j = int(rng.integers(0, len(h)))
        if i != j:
            d = np.zeros(len(h))
            d[i] = 1.0
            d[j] = -1.0
            dirs.append(d)
    return dirs[:8]

def _local_opt_lp(h, deadline, r0=0.15, stall_lim=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        it += 1
        dirs = _gen_dirs(h, rng, rad)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm * rad
            a, c_try = _line_search(h, d, dx)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.03, 0.30)
        else:
            stall += 1
            rad *= 0.80
            if rad < 0.01:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 40 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_opt(h, deadline, max_it=50000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=15000, Ti=0.003):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        hn = cur.copy()
        hn[ones[i]] = 0.0
        hn[zeros[j]] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _top_k_pair_dir(h, rng, k=10):
    """Generate pair directions targeting top lags."""
    n = len(h)
    raw = _raw_corr(h)
    top_p = np.argsort(-raw)[:k]
    dirs = []
    for p in top_p[:4]:
        g = _lag_gradient(h, int(p))
        pos_idx = np.argsort(-g)[:5]
        neg_idx = np.argsort(g)[:5]
        for pi in pos_idx[:3]:
            for ni in neg_idx[:3]:
                if pi != ni:
                    d = np.zeros(n)
                    d[pi] = 1.0
                    d[ni] = -1.0
                    dirs.append(d)
    if not dirs:
        return []
    if len(dirs) > 6:
        idx = rng.choice(len(dirs), size=6, replace=False)
        dirs = [dirs[i] for i in idx]
    return dirs

def _focused_opt(h, deadline, r0=0.10):
    """Optimization that alternates between LP directions and top-k pair directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(55555)
    rad = r0
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > 100:
            break
        if deadline - time.time() < 0.5:
            break
        it += 1
        dirs = []
        if it % 3 == 0:
            for k in [3, 6]:
                d = _lp_direction(h, k, rad)
                if d is not None:
                    dirs.append(d)
        else:
            d = _lp_direction(h, 5, rad)
            if d is not None:
                dirs.append(d)
        dirs.extend(_top_k_pair_dir(h, rng, k=8))
        d = _weighted_grad_dir(h, 40.0)
        if d is not None:
            dirs.append(d)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm * rad
            a, c_try = _line_search(h, d, dx)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(np.clip(h, 0.0, 1.0), target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.02, 0.25)
        else:
            stall += 1
            rad *= 0.82
            if rad < 0.008:
                pert = rng.normal(0, 0.03, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.10
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization (35%)
    d1 = t0 + budget_s * 0.35
    h1, c1 = _local_opt_lp(h_init, d1, r0=0.15, stall_lim=80)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Focused optimization with pair directions (20%)
    d2 = t0 + budget_s * 0.55
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _focused_opt(h2, d2, r0=0.10)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Pair swaps (10%)
    d3 = t0 + budget_s * 0.65
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c3 = _pair_opt(h3, d3, max_it=40000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: Binary exploration (10%)
    d4 = t0 + budget_s * 0.75
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d4, mx=12000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4 - 1:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.02)
        hp, cp = _local_opt_lp(hp, sd, r0=0.08, stall_lim=40)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Multi-resolution (10%)
    d5 = t0 + budget_s * 0.85
    nc = len(best_h)
    offsets = [-6, -4, 4, 6, -8, 8, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 250 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.012)
        hr, cr = _local_opt_lp(hr, sd, r0=0.12, stall_lim=40)
        if cr < best_c + 0.003 and time.time() < d5 - 2:
            sd2 = min(d5, time.time() + budget_s * 0.008)
            hr, cr = _focused_opt(hr, sd2, r0=0.06)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Final intensive polish (10%)
    d6 = deadline
    for r_init in [0.08, 0.04, 0.02]:
        if time.time() > d6 - 2:
            break
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        sd = min(d6, time.time() + budget_s * 0.025)
        h6, c6 = _local_opt_lp(h6, sd, r0=r_init, stall_lim=50)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=10000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    if deadline - time.time() > 1:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt_lp(hf, deadline, r0=0.015, stall_lim=30)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
