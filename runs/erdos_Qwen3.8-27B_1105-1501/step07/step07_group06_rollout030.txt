```python
    best_h = h_bin.copy()
    best_c5 = cur_c5

    T = T_init
    stall = 0
    ones = np.where(h_bin > 0.5)[0].tolist()
    zeros = np.where(h_bin <= 0.5)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500 or len(ones) == 0 or len(zeros) == 0:
            break
        
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]

        # For oi: contributes +h[oi+k]_v and -(1-h[oi-k])_v
```

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _swap_delta_vec(h, i, j):
    """Compute delta vector for binary swap h[i]:1->0, h[j]:0->1.
    Returns the change in corr = np.correlate(h, 1-h, 'full')."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # h[i+k] term: valid for 0 <= i+k < n, i.e., -i <= k < n-i
    lo = max(-(n-1), -i)
    hi = min(n-1, n-1-i)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        delta[ks + (n-1)] += h[i + ks]
    
    # -h[j+k] term: valid for 0 <= j+k < n
    lo = max(-(n-1), -j)
    hi = min(n-1, n-1-j)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        delta[ks + (n-1)] -= h[j + ks]
    
    # -(1-h[i-k]) term: valid for 0 <= i-k < n, i.e., i-n+1 <= k <= i
    lo = max(-(n-1), i - (n-1))
    hi = min(n-1, i)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        delta[ks + (n-1)] -= (1.0 - h[i - ks])
    
    # +(1-h[j-k]) term: valid for 0 <= j-k < n
    lo = max(-(n-1), j - (n-1))
    hi = min(n-1, j)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        delta[ks + (n-1)] += (1.0 - h[j - ks])
    
    # Cross/self terms: -2 at k=0, +1 at k=j-i, +1 at k=i-j
    if 0 >= -(n-1) and 0 <= n-1:
        delta[n-1] -= 2.0
    kj = j - i
    if -(n-1) <= kj <= (n-1):
        delta[kj + (n-1)] += 1.0
    ki = i - j
    if -(n-1) <= ki <= (n-1):
        delta[ki + (n-1)] += 1.0
    
    return delta
```

```python
    n = len(h_bin)
    dx = 2.0 / n
    cur_c5 = float(np.max(_exact_corr(h_bin))) * dx

    T = 0.005

    ones = set(np.where(h_bin > 0.5)[0].tolist())
    zeros = set(np.where(h_bin <= 0.5)[0].tolist())
    
    it = 0
    while time.time() < deadline and stall < max_stall:
        it += 1
        # Choose swap: 30% directed, 70% random
        if rng.random() < 0.3 and len(zeros) > 0:
            # Directed: target the max lag
            r_max = int(np.argmax(corr))
            k_max = r_max - (n - 1)
            # Find positions where h[n+k]=1 and h[n]=0 at this lag
            candidates = []
            if k_max >= 0:
                for nn in range(n - k_max):
                    if h_bin[nn + k_max] > 0.5 and h_bin[nn] <= 0.5:
                        candidates.append(nn)
            else:

                    if h_bin[nn] > 0.5 and h_bin[nn + k_max] <= 0.5:
                        candidates.append(nn)
            if candidates:
                nn = candidates[int(rng.integers(0, len(candidates)))]
                # Swap h[nn] (0) with a random 1
                oi = int(rng.choice(list(ones)))
                oz = nn
            else:
                oi = int(rng.choice(list(ones)))
                oz = int(rng.choice(list(zeros)))
        else:
            oi = int(rng.choice(list(ones)))
            oz = int(rng.choice(list(zeros)))

            continue
        
        delta = _swap_delta_vec(h_bin, oi, oz)
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        new_c5 = new_max * dx
        
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            delta_c5 = new_c5 - cur_c5
            if delta_c5 < 0.05:  # limit for exp overflow
                if rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
                    accept = True
        
        if accept:
            h_bin[oi] = 0.0
            h_bin[oz] = 1.0
            corr = new_corr
            cur_c5 = new_c5
            ones.discard(oi)
            zeros.discard(oz)
            ones.add(oz)

            stall = 0
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            T = max(0.0001, T * 0.9997)
        else:
            stall += 1
```
