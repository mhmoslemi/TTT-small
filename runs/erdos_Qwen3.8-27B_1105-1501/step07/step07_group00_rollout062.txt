This is like a 1D "Chebyshev approximation" problem. The optimal δ* can be found by:
- The max of lines |A_k + B_k δ| is minimized where the two most extreme lines cross.

More precisely, we want to find δ that minimizes max_k (A_k + B_k δ).
Let M(δ) = max_k (A_k + B_k δ).
M is convex. M'(δ) = B_k* where k* is an active constraint.
The minimum is where M'(δ) changes sign, i.e., where the maximum of the B_k for active constraints changes from positive to negative.

This can be solved in O(n log n) by sorting the B_k values, or in O(n) with a linear-time selection algorithm.

import numpy as np

import time

# --- Core functions ---

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-13:
        mask = x < 0.999 if r > 0 else x > 1e-9
        if mask.any():
            x = x.copy()
            x[mask] += r / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, d):
    n = len(h)
    g = np.zeros(n)
    if d > 0:
        m = n - d
        g[:m] -= h[d:]
        g[d:] += 1.0 - h[:m]
    elif d < 0:
        gg = -d
        m = n - gg
        g[:m] += 1.0 - h[gg:]
        g[gg:] -= h[:m]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

# --- Optimizer ---

def _lp_direction(h, scores, radius, max_active=200):
    n = len(h)
    n_lags = 2 * n - 1
    mx = scores.max()
    # Use all lags, or cap at max_active if too many
    if n_lags <= max_active:
        active = np.arange(n_lags)
    else:
        # Use all lags above a threshold
        thresh = mx - 0.5 * (mx - scores.min())
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            idx = np.argsort(scores)[-min(max_active, n_lags):]
            active = idx
        elif len(active) > max_active:
            idx = np.argsort(scores[active])[-max_active:]
            active = active[idx]
    
    m = len(active)
    G = np.zeros((m, n))
    lags = active - (n - 1)
    for i, d in enumerate(lags):
        G[i] = _grad_row(h, d)
    
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            y = res.x[:n]
            if y.max() - y.min() > 1e-14:
                y -= y.mean()
                return y
    except Exception:
        pass
    return None

    if t_max < 1e-14:
        return 0.0, _exact_c5(h)
    best_a, best_c5 = 0.0, _exact_c5(h)

            best_a = a
