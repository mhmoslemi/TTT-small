def _subgrad_active(h, row_indices, weights):
    n=len(h)

        if d >= 0:

            s[:ml] -= wd*h[d:]
        else:
            b=-d
            ml=n-b
            s[:ml] += wd*(1.0-h[b:])

    return s

def _build_G(h):
    n=len(h)
    m=2*n-1
    G=np.zeros((m,n), dtype=np.float64)

        row = n-1+d

    for b in range(1,n):

        ml = n-b

import numpy as np

def _project(v, target):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=np.float64)
    if target >= float(n):
        return np.ones(n, dtype=np.float64)

    vmin = float(np.min(v))
    vmax = float(np.max(v))
    lo = min(vmin, 0.0) - 1.0
    hi = max(vmax, 1.0) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0.0:
            mask = x < (1.0 - 1e-12)
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / float(cnt)
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw)) * dx


def _fallback_h(n=180, seed=1):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / float(n)

def _resample(h, n_new):
    n = len(h)
    if n_new <= 0:
        return h
    if n_new == n:
        return _project(h, n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _to_binary(h, target):
    n = len(h)
    z = np.zeros(n, dtype=np.float64)
    k = int(round(float(target)))
    if k <= 0:
        return z
    if k >= n:
        z.fill(1.0)
        return z
    idx = np.argsort(h)[-k:]
    z[idx] = 1.0
    return z


def _build_G(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=np.float64)

    for d in range(n):
        row = n - 1 + d
        ml = n - d
        G[row, d:n] += 1.0 - h[:ml]
        G[row, :ml] -= h[d:]

    for b in range(1, n):
        row = n - 1 - b
        ml = n - b
        G[row, :ml] += 1.0 - h[b:]
        G[row, b:] -= h[:ml]

    return G


def _subgrad_active(h, row_indices, weights):
    n = len(h)
    s = np.zeros(n, dtype=np.float64)
    center = n - 1

    for t in range(len(row_indices)):
        wd = float(weights[t])
        if wd == 0.0:
            continue
        d = int(row_indices[t]) - center
        if d >= 0:
            ml = n - d
            s[d:n] += wd * (1.0 - h[:ml])
            s[:ml] -= wd * h[d:]
        else:
            b = -d
            ml = n - b
            s[:ml] += wd * (1.0 - h[b:])
            s[b:] -= wd * h[:ml]

    return s


def _make_weights(raw, K, rng, mode, rmax):
    m = len(raw)
    K = int(max(1, min(K, m)))
    noise_scale = max(1e-9, abs(rmax) * 1e-5)
    score = raw + rng.uniform(0.0, noise_scale, m)
    order = np.argsort(score)

    if mode == 0:
        idx = order[-K:]
        w = np.full(max(1, len(idx)), 1.0 / float(len(idx)), dtype=np.float64)
        return idx, w

    if mode == 1:
        cand = int(min(m, max(K, 15)))
        idx = order[-cand:]
        vals = raw[idx] - float(rmax)
        span = float(np.max(vals)) - float(np.min(vals))
        scale = max(1e-9, 0.25 * span + 0.02 * abs(rmax))
        ww = np.exp(vals / scale)
        sw = float(ww.sum())
        if sw > 0.0:
            ww = ww / sw
        else:
            ww = np.full(len(idx), 1.0 / float(len(idx)), dtype=np.float64)
        return idx, ww

    cand = int(min(m, max(K, 20)))

    sub = np.argsort(raw[idx])
    vals = np.linspace(0.2, 1.0, cand)
    vals = vals / float(vals.sum())
    w = np.empty(cand, dtype=np.float64)

    return idx, w

def _seg_eval(S0, T, C, a):

def _golden_eval(S0, T, C, lo, hi, maxit=28):

            c = d
            fc = fd

            fd = _seg_eval(S0, T, C, d)

def _line_search(S0, T, C, a0, b0, grid=121, max_local=4):

                else:
                    Cp[zero] = 0.0

                a_best, f_best = _line_search(S0, T, Cp, 0.0, ahi, grid=49, max_local=2)
                if f_best < cur_raw - tol:
                    orig_i = float(h[i])
                    orig_j = float(h[j])
                    h[i] += a_best
                    h[j] -= a_best
                    if h[i] < 0.0:
                        h[i] = 0.0
                    if h[i] > 1.0:
                        h[i] = 1.0
                    if h[j] < 0.0:
                        h[j] = 0.0
                    if h[j] > 1.0:
                        h[j] = 1.0

                    S0_new = np.correlate(h, 1.0 - h, mode="full")
                    try_raw = float(np.max(S0_new))
                    if try_raw < cur_raw - 1e-12:
                        S0 = S0_new
                        cur_raw = try_raw
                        if cur_raw < best_raw - 1e-14:
                            best_raw = cur_raw
                            best_h = h.copy()
                        random_acc = True
                        stall = 0
                        global_stall = 0
                        break

        if random_acc:
            continue

        stall += 1
        global_stall += 1
        if stall > 25:
            kidx += 1
            stall = 0

        if global_stall > 70:
            h = _project(best_h + scale * rng.standard_normal(n), target)
            S0 = np.correlate(h, 1.0 - h, mode="full")
            cur_raw = float(np.max(S0))
            scale = min(0.08, max(0.002, scale * 1.2))
            global_stall = 0

        if (it % 120) == 0:
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode="full")
            cur_raw = float(np.max(S0))

    best_h = _project(best_h, target)
    S0b = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_raw = float(np.max(S0b))
    return best_h, best_raw


def _binary_local(h0, deadline, seed, L=50, H=50):
    n = len(h0)
    if n % 2 != 0:
        return None

    target = n // 2
    h = _to_binary(h0, target)
    S0 = np.correlate(h, 1.0 - h, mode="full")
    cur_raw = float(np.max(S0))
    best_h = h.copy()
    best_raw = cur_raw

    rng = np.random.default_rng(seed)
    Klist = [5, 10, 20, 35, 60]
    kidx = 0
    it = 0
    stall = 0
    global_stall = 0
    zero = n - 1
    tol = 1e-10

    while time.time() < deadline - 1.0:
        it += 1
        if global_stall > 300:
            break

        G = _build_G(h)
        rmax = cur_raw
        K = Klist[kidx % len(Klist)]
        mode = (it // 5) % 3
        idx, w = _make_weights(S0, K, rng, mode, rmax)
        s = w @ G[idx]

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        noise_z = rng.uniform(0.0, 1e-8, len(zeros))
        noise_o = rng.uniform(0.0, 1e-8, len(ones))

        jl = min(L, len(zeros))
        jh = min(H, len(ones))
        lows = zeros[np.argsort(s[zeros] + noise_z)[:jl]]
        highs = ones[np.argsort(-(s[ones] + noise_o))[:jh]]

        best_f = cur_raw + 1.0
        best_pair = None

        for jj in range(len(lows)):
            j = int(lows[jj])
            kh = len(highs)
            if kh == 0:
                break
            raws = S0[None, :] + G[j][None, :] - G[:, highs].T
            raws[:, zero] -= 2.0
            for kk in range(kh):
                d = int(highs[kk]) - j
                raws[kk, zero + d] += 1.0
                raws[kk, zero - d] += 1.0

            f = np.max(raws, axis=1)
            bk = int(np.argmin(f))
            if f[bk] < best_f:
                best_f = float(f[bk])
                best_pair = (int(highs[bk]), j)

        if best_pair is not None and best_f < cur_raw - tol:
            i, j = best_pair
            S0_new = S0 + G[j] - G[i]
            S0_new = S0_new.copy()
            S0_new[zero] -= 2.0
            d = j - i
            S0_new[zero + d] += 1.0
            S0_new[zero - d] += 1.0
            trial_raw = float(np.max(S0_new))

            if trial_raw < cur_raw - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                S0 = np.correlate(h, 1.0 - h, mode="full")
                cur_raw = float(np.max(S0))
                if cur_raw < best_raw - 1e-14:
                    best_raw = cur_raw
                    best_h = h.copy()
                stall = 0
                global_stall = 0
                continue

        found = False
        for _ in range(120):
            i = int(ones[int(rng.integers(0, len(ones)))])
            j = int(zeros[int(rng.integers(0, len(zeros)))])
            new = S0 + G[j] - G[i]
            new = new.copy()
            new[zero] -= 2.0
            d = j - i
            new[zero + d] += 1.0
            new[zero - d] += 1.0
            mf = float(np.max(new))
            if mf < best_f:
                best_f = mf
                best_pair = (i, j)
                found = True
                if mf < cur_raw - tol:
                    break

        if found and best_pair is not None and best_f < cur_raw - tol:
            i, j = best_pair
            S0_new = S0 + G[j] - G[i]
            S0_new = S0_new.copy()
            S0_new[zero] -= 2.0
            d = j - i
            S0_new[zero + d] += 1.0
            S0_new[zero - d] += 1.0
            trial_raw = float(np.max(S0_new))
            if trial_raw < cur_raw - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                S0 = np.correlate(h, 1.0 - h, mode="full")
                cur_raw = float(np.max(S0))
                if cur_raw < best_raw - 1e-14:
                    best_raw = cur_raw
                    best_h = h.copy()
                stall = 0
                global_stall = 0
                continue

        stall += 1
        global_stall += 1
        if stall > 25:
            kidx += 1
            stall = 0

        if global_stall > 80:
            start = _to_binary(best_h, target)
            if rng.random() < 0.3:
                start = _to_binary(rng.standard_normal(n), target)
            h = _to_binary(start + 0.2 * rng.standard_normal(n), target)
            S0 = np.correlate(h, 1.0 - h, mode="full")
            cur_raw = float(np.max(S0))
            stall = 0
            global_stall = 0
            kidx = 0

    best_h = _project(best_h, target)
    S0b = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_raw = float(np.max(S0b))
    return best_h, best_raw


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(max(10.0, budget_s))

    try:
        hv = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        hv = _fallback_h(180, seed)

    if hv.size == 0 or not np.all(np.isfinite(hv)):
        hv = _fallback_h(180, seed)

    n_init = len(hv)
    h_init = _project(hv, n_init / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(best_h)
    best_n = n_init

    def _maybe_update(h, raw):
        nonlocal best_h, best_c5, best_n
        if h is None:
            return
        nh = len(h)
        if nh == 0:
            return
        c = raw * (2.0 / float(nh))
        if c < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c
            best_n = nh

    try:
        d1 = min(deadline, t0 + budget_s * 0.30)
        h1, raw1 = _fw_optimize(best_h, d1, seed + 1)
        _maybe_update(h1, raw1)

        d2 = min(deadline, t0 + budget_s * 0.48)
        h2, raw2 = _pair_optimize(best_h, d2, seed + 2)
        _maybe_update(h2, raw2)

        if best_n % 2 == 0:
            d3 = min(deadline, t0 + budget_s * 0.60)
            h3, raw3 = _binary_local(best_h, d3, seed + 3)
            _maybe_update(h3, raw3)

        d4 = min(deadline, t0 + budget_s * 0.66)
        h4, raw4 = _pair_optimize(best_h, d4, seed + 4)
        _maybe_update(h4, raw4)

        d5 = min(deadline, t0 + budget_s * 0.84)
        ns = []
        for dn in range(-24, 65, 4):
            nt = n_init + dn
            if 60 <= nt <= 320 and nt % 2 == 0:
                ns.append(nt)
        for extra in (80, 100, 120, 140, 160, 200, 240):
            if 60 <= extra <= 320 and extra % 2 == 0:
                ns.append(extra)

        uniq = []
        for nt in ns:
            if nt not in uniq:
                uniq.append(nt)

        pairs = []
        for nt in uniq:
            if nt != best_n:
                pairs.append((abs(nt - best_n), nt))
        pairs.sort()

        for cnt, pair in enumerate(pairs[:22]):
            if time.time() >= d5:
                break
            nt = pair[1]
            per = max(1.2, min(5.0, (d5 - time.time()) / float(max(1, len(pairs[:22]) - cnt))))
            sub = min(d5, time.time() + per)
            hr = _resample(best_h, nt)
            hr, rawr = _fw_optimize(hr, sub, seed + 100 + cnt)
            _maybe_update(hr, rawr)

            if time.time() < d5 - 1.5:
                sub2 = min(d5 - 0.5, time.time() + max(0.6, (d5 - time.time()) * 0.18))
                hp, rawp = _pair_optimize(hr, sub2, seed + 200 + cnt)
                _maybe_update(hp, rawp)

        final_budget = max(2.0, (deadline - time.time()) * 0.35)
        hf, rawf = _fw_optimize(best_h, min(deadline, time.time() + final_budget), seed + 500)
        _maybe_update(hf, rawf)

        if best_n % 2 == 0 and time.time() < deadline - 2.0:
            hb, rawb = _binary_local(best_h, min(deadline, time.time() + max(1.0, budget_s * 0.04)), seed + 600)
            _maybe_update(hb, rawb)

        if time.time() < deadline - 1.0:
            hp, rawp = _pair_optimize(
                best_h,
                min(deadline - 0.5, time.time() + max(1.0, (deadline - time.time()) * 0.45)),
                seed + 700,
            )
            _maybe_update(hp, rawp)

    except Exception:
        pass

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
