def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt

    for t in grid_t:
        f = float(np.max(S0 + t * T - t * t * C)) * dx
        if f < best_f:
            best_f = f
            best_t = t
    return best_t, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick transforms (5s)
    d1 = t0 + 5
    transforms = [h_init.copy(), h_init[::-1].copy(), 1.0 - h_init]
    for shift in range(1, min(20, n0 // 5) + 1):
        transforms.append(np.roll(h_init, shift).copy())
        transforms.append(np.roll(h_init, -shift).copy())
    for t in transforms:
        t = _project(t.copy(), n0 / 2.0)
        c = _exact_c5(t)
        if c < best_c5:
            best_c5 = c
            best_h = t.copy()
        if time.time() > d1:
            break

    # Phase 2: SLSQP with tau annealing (200s)
    d2 = t0 + budget_s * 0.18
    h_sl = h_init.copy()
    h_sl, c5_sl = _slsqp_anneal(h_sl, d2, tau_start=0.04, tau_end=0.003, n_solves=10)
    if c5_sl < best_c5:
        best_c5 = c5_sl
        best_h = h_sl.copy()

    # Multi-start SLSQP
    for start_i in range(6):
        if time.time() > d2 - 10:
            break
        h_pert = _perturb(best_h, 0.03 + 0.01 * start_i, rng)
        h_s, c5_s = _slsqp_anneal(h_pert, min(d2 - 5, time.time() + 15),
                                   tau_start=0.02, tau_end=0.003, n_solves=4)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Intensive binary search (45% of budget)
    d3 = t0 + budget_s * 0.62
    h_bin = _to_binary(best_h, n0)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Greedy binary descent (most important)
    sub_dl = min(d3, time.time() + budget_s * 0.15)
    h_bin, c5_bin = _greedy_binary(h_bin, sub_dl)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # More greedy rounds
    sub_dl = min(d3, time.time() + budget_s * 0.12)
    h_bin, c5_bin = _greedy_binary(h_bin, sub_dl)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Targeted search
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _targeted_binary(h_bin, sub_dl, max_iters=80000, seed=200)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # SA
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _sa_binary(h_bin, sub_dl, temp_init=0.003, seed=300)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Block swaps
    sub_dl = min(d3, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _block_swap(h_bin, sub_dl, max_iters=20000, seed=400)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Another greedy pass
    if time.time() < d3 - 10:
        sub_dl = min(d3, time.time() + budget_s * 0.08)
        h_bin, c5_bin = _greedy_binary(h_bin, sub_dl)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.82
    n_curr = len(best_h)
    tried_ns = set()
    for dn in range(-24, 25, 4):
        if time.time() > d4 - 10:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        h_r = _slsqp_solve(h_r, 0.01, n_try, maxiter=60, ftol=1e-10)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.002:
            h_b = _to_binary(h_r, n_try)
            sub_dl = min(d4, time.time() + 10)
            h_b, c5_b = _greedy_binary(h_b, sub_dl)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
            sub_dl = min(d4, time.time() + 8)
            h_b, c5_b = _sa_binary(h_b, sub_dl, temp_init=0.002, seed=500 + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 5: Final refinement (18%)
    d5 = deadline
    n_best = len(best_h)

    # Re-greedy on best binary
    if time.time() < d5 - 20:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d5, time.time() + budget_s * 0.06)
        h_b, c5_b = _greedy_binary(h_b, sub_dl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # SLSQP polish on continuous version
    if time.time() < d5 - 15:
        h_c = _project(best_h.copy(), n_best / 2.0)
        h_c = _slsqp_solve(h_c, 0.003, n_best, maxiter=80, ftol=1e-11)
        c5_c = _exact_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # More SA with lower temp
    if time.time() < d5 - 15:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_b, c5_b = _sa_binary(h_b, sub_dl, temp_init=0.001, seed=900)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final greedy
    if time.time() < d5 - 5:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d5 - 2, time.time() + 8)
        h_b, c5_b = _greedy_binary(h_b, sub_dl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = _project(best_h, len(best_h) / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
