def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):

    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14

        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _block_dir(h, g, bs, rng):
    n = len(h)
    d = np.zeros(n)
    nb = n // bs
    if nb < 2:
        return d
    signs = np.ones(nb)
    for _ in range(rng.integers(1, nb // 2)):
        idx = rng.integers(nb)
        signs[idx] = -1.0
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = signs[b] * g[b * bs:(b + 1) * bs].mean()
    if nb * bs < n:
        d[nb * bs:] = signs[-1] * g[nb * bs:].mean()
    d -= np.mean(d)
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

            hb[si:si+bs] = 1.0
            stall += 1
            T = min(0.01, T * 1.03)
    
    return best, best_c

def _sa_binary(hb, deadline, mx=30000, Ti=0.003):
    n = len(hb)
    dx = 2.0 / n
    cur = hb.copy()
    cc = _exact_c5(cur) * n / 2.0  # raw max corr
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(314159)
    T = Ti
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        cur[i] = 0.0
        cur[j] = 1.0
        nc = float(np.max(np.correlate(cur, 1.0 - cur, mode='full')))
        dl = nc - cc
        if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
    return bh, bc * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)
    
    # Phase 1: Binary optimization (30% time)
    d1 = t0 + budget_s * 0.30
    tried_ns = set()
    n_candidates = [n0, n0-2, n0+2, n0-4, n0+4, n0-6, n0+6]
    for n_try in n_candidates:
        if time.time() > d1:
            break
        if n_try < 80 or n_try > 250 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        hr = _resample(h_init, n_try)
        hb = _to_bin(hr, n_try)
        
        # Quick binary local search
        sd1 = min(d1, time.time() + budget_s * 0.015)
        hb1, c1 = _binary_local_search(hb, sd1, max_iter=50000)
        if c1 < best_c:
            best_c = c1
            best_h = hb1.copy()
        
        # Block swap refinement
        if time.time() < d1 - 2:
            sd2 = min(d1, time.time() + budget_s * 0.008)
            hb2, c2 = _binary_block_swap(hb1, sd2, max_iter=20000)
            if c2 < best_c:
                best_c = c2
                best_h = hb2.copy()
    
    # Also try SA on the best found
    if best_c > 0.381 and time.time() < d1:
        n_b = len(best_h)
        hb = _to_bin(best_h, n_b)
        sd3 = min(d1, time.time() + budget_s * 0.02)
        hb3, c3 = _sa_binary(hb, sd3, mx=30000, Ti=0.002)
        if c3 < best_c:
            best_c = c3
            best_h = hb3.copy()
    
    # Phase 2: Continuous refinement from best binary (15%)
    d2 = t0 + budget_s * 0.45
    nc = len(best_h)
    hc = _project(best_h.copy(), nc / 2.0)
    hc, cc = _local_opt(hc, d2, r0=0.15, stall_lim=50, nd=5)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()
    
    # Phase 3: Pair optimization (10%)
    d3 = t0 + budget_s * 0.55
    hc = _project(best_h.copy(), len(best_h) / 2.0)
    hc, cc = _pair_opt(hc, d3, max_it=25000)
    if cc < best_c:
        best_c = cc
        best_h = hc.copy()
    
    # Phase 4: Multi-resolution exploration (15%)
    d4 = t0 + budget_s * 0.70
    nc = len(best_h)
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -6, 6, -10, 10]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.012)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, nd=4)
        if cr < best_c + 0.003 and time.time() < d4 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.008)
            hb, cb2 = _binary_local_search(hb, sd2, max_iter=20000)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
    
    # Phase 5: Intensive continuous optimization (15%)
    d5 = t0 + budget_s * 0.85
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.04)
        h5, c5 = _local_opt(h5, sd, r0=r_init, stall_lim=50, nd=6)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d5 - 5:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.02)
            h5, c5 = _pair_opt(h5, sd, max_it=10000)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()
    
    # Phase 6: Basin hopping (7%)
    d6 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.2 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)
    
    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=25, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
