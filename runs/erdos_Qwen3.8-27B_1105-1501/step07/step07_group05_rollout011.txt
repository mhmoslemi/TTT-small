import numpy as np
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, dx, radius):
    """Line search along direction d from point h."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Collect candidate points
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    
    # Quadratic optima for each k
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Uniform grid
    ng = min(50, max(20, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    
    # Pairwise intersections of top quadratics
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    
    # Evaluate candidates
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-12:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-13:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

def _soft_grad(h, beta):
    """Compute smoothed gradient of the max-of-correlations objective."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -70.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _smooth_dir(g, n_keep):
    """Smooth gradient by keeping only first n_keep frequency components."""
    n = len(g)
    G = np.fft.rfft(g)
    if n_keep < len(G):
        G[n_keep:] = 0
    gs = np.fft.irfft(G, n=n)
    gs -= np.mean(gs)
    nrm = np.linalg.norm(gs)
    if nrm > 1e-12:
        gs /= nrm
    return gs

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-6
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(40, max(16, int(width * 80)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    M = min(10, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a

            else:

                c2 = lo_r + gr * (hi_r - lo_r)

def _grad_soft(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -70.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

    sc = S * dx

def _fft_dir(g, n_keep):
    n = len(g)
    G = np.fft.rfft(g)
    n_keep = min(n_keep, len(G))
    G2 = G.copy()
    G2[n_keep:] = 0
    gs = np.fft.irfft(G2, n=n)
    gs -= np.mean(gs)
    nrm = np.linalg.norm(gs)
    if nrm > 1e-12:
        gs /= nrm
    return gs

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
