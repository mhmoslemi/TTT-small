  corr(h+αp, (1-h)-αp) = corr(h, 1-h) + α[corr(p, 1-h) - corr(h, p)] - α² corr(p, p)

So we can evaluate the objective at any α exactly with just 4 correlate calls (precomputed).

This is the key efficiency! Once we have p = v - h, we need:
- S0 = corr(h, 1-h) (already have)
- T = corr(p, 1-h) - corr(h, p) (2 correlate calls)
- C = corr(p, p) (1 correlate call)

Then for any α: objective(α) = dx * max(S0 + α*T - α²*C)

The minimum over α ∈ [0,1] (or [0, α_max] for box constraints) of max of quadratic functions can be found by:
- Evaluating at α=0, α=1, and at the "crossing" points where the max switches
- Or just doing a golden section search (which is clean and reliable)

Golden section with maybe 30-40 iterations will give us a very accurate minimum.

Each evaluation is O(n) (just a max of a vector of length 2n-1), so 40 evaluations = O(n) total. Very fast!

    best_c5 = cur_c5

            break
        if stall > 200:
            break

        R = S0  # correlation values (raw, before dx)
        # Get top-k lags
        k = min(top_k, 2*n-1)
        active_idx = np.argpartition(R, -k)[-k:]
        active_R = R[active_idx]
        Rmax = R[active_idx[0]]  # max
        
        # Weight active lags: emphasize those near the max
        # Use exponential weighting
        weights = np.exp(-(active_R - active_R.max()) / max(1.0, (active_R.max() - active_R.min()) * 0.1))
        weights /= weights.sum()

        for idx in active_idx:
            b = abs(int(idx - (n-1)))
            grad += weights[int(idx - active_idx[0])] * _grad_lag(h, b)
        
        # Actually, let me just use the max lag's gradient for simplicity
        # (or a simple sum of top few)
        grad = np.zeros(n)
        for idx in active_idx[:5]:  # top 5
            b = abs(int(idx - (n-1)))
            grad += _grad_lag(h, b)

        order = np.argsort(grad)
        v = np.zeros(n)
        v[order[:int(target)]] = 1.0
        frac = target - int(target)
        if frac > 0:
            v[order[int(target)]] = frac

        p = v - h
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            continue
        
        # Precompute for line search
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')

        pos = p > 1e-10
        neg = p < -1e-10
        alpha_hi = 1.0
        if np.any(pos):

        if np.any(neg):
            alpha_hi = min(alpha_hi, float(np.min(h[neg] / (-p[neg]))))
        alpha_hi = max(1e-10, alpha_hi)

        lo_a, hi_a = 0.0, alpha_hi
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
        for _ in range(35):
            if hi_a - lo_a < 1e-12 * max(1.0, alpha_hi):
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1; f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
            else:
                lo_a = c1
                c1 = c2; f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        
        best_alpha = (lo_a + hi_a) / 2.0
        best_val = float(np.max(S0 + best_alpha*T - best_alpha*best_alpha*C)) * dx
        
        # Also check alpha = 0 and alpha = alpha_hi
        f0 = cur_c5
        fe = float(np.max(S0 + alpha_hi*T - alpha_hi*alpha_hi*C)) * dx
        candidates = [(0.0, f0), (best_alpha, best_val), (alpha_hi, fe)]
        best_alpha = min(candidates, key=lambda x: x[1])[0]

            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')

                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

    f0 = float(np.max(S0)) * dx
    fe = float(np.max(S0 + alpha_hi * T - alpha_hi * alpha_hi * C)) * dx
    best_a = min([(0.0, f0), (opt_a, opt_v), (alpha_hi, fe)], key=lambda x: x[1])
    return best_a[0], best_a[1]

def _fw_optimize(h, deadline, top_k=10, seed=42):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    perturb_scale = 0.1

    while time.time() < deadline:
        if stall > 400:
            break

        grad, R, active_idx = _combined_grad(h, top_k)
        v = _fw_vertex(grad, n, target)
        p = v - h
        max_p = float(np.max(np.abs(p)))
        if max_p < 1e-12:
            stall += 1
            h = _project(h + rng.normal(0, perturb_scale, n), target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            perturb_scale = min(0.3, perturb_scale * 1.2)
            continue

        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')

        pos = p > 1e-10
        neg = p < -1e-10
        alpha_hi = 1.0
        if np.any(pos):
            alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            alpha_hi = min(alpha_hi, float(np.min(h[neg] / (-p[neg]))))
        alpha_hi = max(1e-10, min(alpha_hi, 1.0))

        opt_a, opt_v = _line_search(S0, T, C, alpha_hi, dx)

        if opt_v < cur_c5 - 1e-13:
            h = h + opt_a * p
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                perturb_scale = max(0.02, perturb_scale * 0.9)
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                h = _project(h + rng.normal(0, perturb_scale, n), target)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                cur_c5 = float(np.max(S0)) * dx
                perturb_scale = min(0.3, perturb_scale * 1.15)
                stall = 0

    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=5000, seed=123):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old = h[j]
            g = np.zeros(2 * n - 1)
            for d in range(0, n - j):
                g[n - 1 + d] = (1.0 - h[j + d])
            for d in range(1, j + 1):
                g[n - 1 - d] = -h[j - d]
            lo_d, hi_d = -old, 1.0 - old
            if hi_d - lo_d < 1e-12:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            m1 = lo_d + gr * (hi_d - lo_d)
            m2 = hi_d - gr * (hi_d - lo_d)
            f1 = float(np.max(S0 + m1 * g))
            f2 = float(np.max(S0 + m2 * g))
            for _ in range(25):
                if hi_d - lo_d < 1e-11:
                    break
                if f1 < f2:
                    hi_d, m2, f2 = m2, m1, f1
                    m1 = lo_d + gr * (hi_d - lo_d)
                    f1 = float(np.max(S0 + m1 * g))
                else:
                    lo_d, m1, f1 = m1, m2, f2
                    m2 = hi_d - gr * (hi_d - lo_d)
                    f2 = float(np.max(S0 + m2 * g))
            opt_d = (lo_d + hi_d) / 2.0
            new_val = old + opt_d
            f_new = float(np.max(S0 + opt_d * g)) * dx
            if f_new < cur_c5 - 1e-14:
                delta = new_val - old
                h[j] = new_val
                S0 += delta * g
                cur_c5 = f_new
                improved = True
        if improved:
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap_opt(h, deadline, top_k=10, seed=77):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0

    while time.time() < deadline:
        if stall > 300:
            break

        grad, R, _ = _combined_grad(h, top_k)
        found = False
        for _ in range(100):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or h[i] < 0.02 or h[j] > 0.98:
                continue
            if grad[j] - grad[i] >= 0:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            alpha_hi = min(h[i], 1.0 - h[j])
            opt_a, opt_v = _line_search(S0, T, C, alpha_hi, dx)
            if opt_v < cur_c5 - 1e-13:
                h[i] -= opt_a
                h[j] += opt_a
                h = _project(h, target)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                cur_c5 = float(np.max(S0)) * dx
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall % 50 == 0:
                h = _project(h + rng.normal(0, 0.05, n), target)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                cur_c5 = float(np.max(S0)) * dx
                stall = 0
    return best_h, best_c5

def _sturmian_init(n, seed=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    count = n // 2
    positions = sorted(set(int(k * n / phi) % n for k in range(count)))
    while len(positions) < count:
        positions.append(int((len(positions) * n / phi) % n))
    for p in positions[:count]:
        h[p] = 1.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, seed=99):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    S0 = np.correlate(cur, 1.0 - cur, mode='full')
    cur_c5 = float(np.max(S0)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    T = 0.005

    for _ in range(200000):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        d = np.zeros(n)
        d[oi] = -1.0
        d[oz] = 1.0
        T2 = np.correlate(d, 1.0 - cur, mode='full') - np.correlate(cur, d, mode='full')
        C2 = np.correlate(d, d, mode='full')
        new_S0 = S0 + T2 - C2
        new_c5 = float(np.max(new_S0)) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            S0 = new_S0
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                T = max(0.001, T * 0.999)
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
        else:
            stall += 1
            T = max(0.001, T * 0.9995)
    return best_h, best_c5

def _basin_hopping(h, deadline, seed=555):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    scale = 0.05
    count = 0

    while time.time() < deadline - 5 and count < 200:
        count += 1
        x = np.arange(n, dtype=float)
        freqs = rng.uniform(0.5, 8.0, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        pert = np.zeros(n)
        for f, p in zip(freqs, phases):
            pert += np.sin(2 * np.pi * f * x / n + p)
        pert *= scale / 3.0
        pert += scale * 0.3 * rng.normal(0, 1, n)
        h_pert = _project(best_h + pert, target)

        # Quick local optimize
        sub_dl = min(deadline - 5, time.time() + 8)
        h_t, c5_t = _fw_optimize(h_pert, sub_dl, top_k=8, seed=2000 + count)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.9)
        else:
            scale = min(0.15, scale * 1.1)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: FW from parent (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _fw_optimize(h_init, d1, top_k=10, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (20%)
    d2 = t0 + budget_s * 0.28
    n_list = [120, 140, 160, 180, 200, 220, 240, 100, 130, 170, 190, 210]
    n_list = [x for x in n_list if 80 <= x <= 280]
    for n_try in n_list:
        if time.time() > d2 - 5:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2 - 5, time.time() + 4)
        h_r, c5_r = _fw_optimize(h_r, sub_dl, top_k=8, seed=100 + n_try)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Also try Sturmian initialization
    if time.time() < d2:
        for n_s in [n0, n0 - 20, n0 + 20]:
            if time.time() > d2 - 3:
                break
            h_s = _sturmian_init(n_s, seed=seed + n_s)
            sub_dl = min(d2 - 3, time.time() + 4)
            h_s, c5_s = _fw_optimize(h_s, sub_dl, top_k=8, seed=500 + n_s)
            if c5_s < best_c5 - 1e-13:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 3: Intensive FW (20%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _fw_optimize(best_h, d3, top_k=12, seed=777)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Coordinate descent (15%)
    d4 = t0 + budget_s * 0.63
    h4, c5_4 = _coord_descent(best_h, d4, max_sweeps=5000, seed=314)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair swap (10%)
    d5 = t0 + budget_s * 0.73
    h5, c5_5 = _pair_swap_opt(best_h, d5, top_k=10, seed=271)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary optimization (10%)
    d6 = t0 + budget_s * 0.83
    n_curr = len(best_h)
    h_bin = _to_binary(best_h, n_curr)
    h_bin, c5_bin = _binary_local_search(h_bin, min(d6, time.time() + budget_s * 0.06), seed=888)
    if c5_bin < best_c5 - 1e-13:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Polish binary with continuous
    if time.time() < d6 - 3:
        h_pol = _project(h_bin.copy(), n_curr / 2.0)
        h_pol, c5_pol = _fw_optimize(h_pol, min(d6, time.time() + budget_s * 0.03), top_k=10, seed=444)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Basin hopping (10%)
    d7 = t0 + budget_s * 0.93
    h7, c5_7 = _basin_hopping(best_h, d7, seed=999)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final polish (7%)
    d8 = deadline - 3
    h8, c5_8 = _fw_optimize(best_h, min(d8, time.time() + budget_s * 0.03), top_k=15, seed=1234)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8 - 2:
        h8, c5_8 = _coord_descent(best_h, min(d8, time.time() + budget_s * 0.02), max_sweeps=2000, seed=5678)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
