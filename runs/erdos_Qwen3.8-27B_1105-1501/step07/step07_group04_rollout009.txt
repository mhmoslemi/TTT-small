dC_k^+/dh[i] = -h[i+k]*1_{0<=i<=n-1-k} + (1-h[i-k])*1_{k<=i<=n-1}

For negative lag k: C_k^- = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])

dC_k^-/dh[i]:
- If i = j for some j in [0, n-1-k]: term is h[i]*(1-h[i+k]), d/dh[i] = (1-h[i+k])
- If i = j+k for some j in [0, n-1-k], i.e., i in [k, n-1]: term is h[i-k]*(1-h[i]), d/dh[i] = -h[i-k]

So:
dC_k^-/dh[i] = (1-h[i+k])*1_{0<=i<=n-1-k} + (-h[i-k])*1_{k<=i<=n-1}

Combined for lag k with weights wp = w[n-1+k] and wn = w[n-1-k]:

For i in [0, n-1-k] (the "left" region, i < n-k):
  contribution = wp*(-h[i+k]) + wn*(1-h[i+k]) = wn - (wp+wn)*h[i+k]

For i in [k, n-1] (the "right" region, i >= k):
  contribution = wp*(1-h[i-k]) + wn*(-h[i-k]) = wp - (wp+wn)*h[i-k]

For i in [k, n-1-k] (overlap region, both apply):
  contribution = wn - (wp+wn)*h[i+k] + wp - (wp+wn)*h[i-k]

So the gradient is:
grad = w[n-1]*(1-2h) 
     + sum_{k=1}^{n-1} [ (wn_k - (wp_k+wn_k)*h[shifted]) for left part + (wp_k - (wp_k+wn_k)*h[shifted]) for right part ]

Let me implement this:

```python
def smooth_grad(h, w):
    n = len(h)
    grad = w[n-1] * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        wp = w[n-1+k]
        wn = w[n-1-k]
        W = wp + wn
        if W < 1e-15:
            continue
        
        # Left region: i = 0 to n-1-k
        # contribution = wn - W * h[i+k]
        grad[0:n-k] += wn - W * h[k:n]
        
        # Right region: i = k to n-1
        # contribution = wp - W * h[i-k]
        grad[k:n] += wp - W * h[0:n-k]
    
    return grad
```

from scipy.optimize import minimize

def exact_c5(h):

def project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_objective(h_flat, tau, n, dx):
    """Compute smooth max of correlation terms and its gradient."""
    h = h_flat.reshape(-1)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(corr.max())
    # For numerical stability
    diff = (corr - m) / tau
    diff = np.clip(diff, -50.0, 0.0)
    exps = np.exp(diff)
    S = float(np.sum(exps))
    value = m + tau * np.log(S)
    
    # Softmax weights
    w = exps / S
    
    # Gradient
    grad = np.zeros(n)
    # Center term (index n-1)
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        wp = w[n-1+k]
        wn = w[n-1-k]
        W = wp + wn
        if W < 1e-15:
            continue
        # Left region: i = 0 to n-1-k
        grad[0:n-k] += wn - W * h[k:n]
        # Right region: i = k to n-1
        grad[k:n] += wp - W * h[0:n-k]
    
    return value, grad

def smooth_obj_wrapper(x, tau, n, dx):
    return smooth_objective(x, tau, n, dx)

def smooth_grad_wrapper(x, tau, n, dx):
    _, g = smooth_objective(x, tau, n, dx)
    return g

def run_slsqp(h_start, tau, n, dx, maxiter=200, time_limit=30):
    """Run SLSQP with smooth objective."""
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}
    
    try:
        res = minimize(
            fun=smooth_obj_wrapper,
            x0=h_start.copy(),
            jac=smooth_grad_wrapper,
            args=(tau, n, dx),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
    except Exception:
        return h_start.copy(), float('inf')
    
    if time.time() - t0 > time_limit:
        return h_start.copy(), float('inf')
    
    x = res.x
    x = np.clip(x, 0.0, 1.0)
    x = project(x, n / 2.0)
    c5 = exact_c5(x)
    return x, c5
