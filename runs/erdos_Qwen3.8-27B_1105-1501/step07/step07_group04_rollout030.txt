def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    exp_vals = np.exp(tau * (corr - max_c))
    w_sum = float(exp_vals.sum())
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / w_sum
    
    grad = np.zeros(n)
    center = n - 1
    for p in range(-(n-1), n):
        w = weights[center + p] * dx
        if w < 1e-15:

        if p >= 0:

            if p > 0:

                grad[idx1] += w * (1.0 - h[idx1 - p])
            else:
                idx1 = np.arange(0, n)
                grad[idx1] += w * (1.0 - h[idx1])
            # -h[j+p]*[0<=j<n-p]

- Increase h[i+k] for positions where h[i+k] is small (reduces the "second" factor)

    n = len(h)

    m = np.zeros(n)  # Adam moment

        g = lse_gradient(h, tau)

        if g_norm < 1e-12:
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Learning rate decay
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        
        direction = -m_hat / (np.sqrt(v_hat) + eps)
        # Remove mean to preserve sum constraint approximately
        direction = direction - np.mean(direction)
        
        h_new = h + lr_t * direction
        h_new = project(h_new, target)
        
        new_c5 = exact_c5(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h = h  # reject
            # Decay tau to sharpen
            t += 10  # burn time
