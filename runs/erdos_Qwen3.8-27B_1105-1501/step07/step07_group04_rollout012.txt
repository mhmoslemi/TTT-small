```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

def _focused_binary_descent(h_bin, max_iters=10000):
    """Focused binary descent targeting active lags"""
    n = len(h_bin)
    dx = 2.0 / n

    curr_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5

    zeros = set(np.where(h <= 0.5)[0].tolist())

        max_val = corr.max()
        # Find active lags (within tolerance of max)
        tol = 0.01  # in units of corr values
        active = np.where(corr >= max_val - tol)[0]
        
        best_delta = 0
        best_swap = None

                # Compute change in active lags

                if delta < best_delta:

                    best_swap = (a, b)

            zeros.add(a)

            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
