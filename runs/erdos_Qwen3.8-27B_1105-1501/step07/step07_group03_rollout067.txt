def _resample(h, n_new):

    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _project(v, target):

        return np.zeros(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, weights):

        else:

            m = n - gp

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for k in range(5):
        f = rng.uniform(0.3, 7.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.2, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.08 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=4000, radius=0.2, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, _ = _line_search(h, d, raw, dx, radius, n_top=6)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _multi_lp(h, deadline, lag_list=(8, 15, 30, 50)):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    per_time = (deadline - time.time()) / max(1, len(lag_list))
    for nl in lag_list:
        if time.time() > deadline - 2:
            break
        sub_dl = min(deadline, time.time() + per_time)
        h_try, c5_try = _lp_optimize(h.copy(), sub_dl, n_lags=nl, trust_init=0.25, stall_limit=40)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization from start (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _lp_optimize(h_init, d1, n_lags=20, trust_init=0.35, stall_limit=80)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (22%)
    d2 = t0 + budget_s * 0.40
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in [-16, -12, -8, -4, 4, 8, 12, 16, -20, 20, -24, 24]:
        if time.time() > d2 - 15:
            break
        n_try = n_curr + dn
        if n_try < 70 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_lags=15, trust_init=0.3, stall_limit=45)
        if c5_r < best_c5 + 0.004 and time.time() < d2 - 8:
            h_r, c5_r = _gradient_optimize(h_r.copy(), min(d2, time.time() + budget_s * 0.01), beta=20.0, radius=0.12, stall_limit=25)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Binary refinement (12%)
    d3 = t0 + budget_s * 0.52
    h_bin = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.045)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=30000, T_init=0.002)
    sub_dl = min(d3, time.time() + budget_s * 0.025)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d3:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_pol, c5_p = _lp_optimize(h_pol, sub_dl, n_lags=15, trust_init=0.12, stall_limit=30)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 4: Intensive multi-method (20%)
    d4 = t0 + budget_s * 0.72
    n_curr = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4, c5_4 = _multi_lp(best_h.copy(), sub_dl, lag_list=(8, 18, 35))
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_cont(h4, sub_dl, n_pairs=5000, radius=0.2, seed=222)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=20000, T_init=0.001)
        sub_dl2 = min(d4, time.time() + budget_s * 0.015)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: Basin hopping with LP (20%)
    d5 = deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 200:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        rem = d5 - 8 - time.time()
        if rem < 1.0:
            break
        sub_dl = min(d5 - 5, time.time() + max(1.5, rem * 0.012))
        if n_restarts % 3 == 0:
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, n_lags=15, trust_init=0.15, stall_limit=25)
        elif n_restarts % 3 == 1:
            h_t, c5_t = _gradient_optimize(h_pert, sub_dl, beta=25.0, radius=0.1, stall_limit=25)
        else:
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, n_lags=30, trust_init=0.12, stall_limit=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.82)
        else:
            scale = min(0.05, scale * 1.07)

    # Final polish
    if deadline - time.time() > 6:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, min(deadline - 3, time.time() + budget_s * 0.03), n_lags=30, trust_init=0.08, stall_limit=50)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _gradient_optimize(h6, deadline, beta=40.0, radius=0.05, stall_limit=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
