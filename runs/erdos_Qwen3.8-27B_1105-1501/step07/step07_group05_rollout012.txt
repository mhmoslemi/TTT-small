def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

def _lag_gradient(h, t, n):
    """Compute gradient of raw[t] w.r.t. h, where raw = correlate(h, 1-h, 'full')"""
    l = t - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        g[:m] += (1.0 - h[l:l+m])
        g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:s+m]
    return g

def _line_search(h, d, dx, radius):

def _lp_direction(h, n, top_k, r_param, timeout_ms=200):
    """Solve LP to find direction that decreases all top-K active lags simultaneously.
    Returns direction vector s (zero-mean) or None if LP fails."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    # Get top K indices
    top_idx = np.argsort(-corr)[:top_k]
    # Filter to those close to max
    threshold = mx - 0.05 * mx
    top_idx = top_idx[corr[top_idx] >= threshold]
    if len(top_idx) < 2:
        top_idx = np.argsort(-corr)[:min(top_k, len(corr))]
    
    K = len(top_idx)
    # Build gradient matrix
    G = np.zeros((K, n))
    for i, t in enumerate(top_idx):
        G[i] = _lag_gradient(h, t, n)
    
    # LP: maximize tau
    # s.t. G @ s <= -tau  (i.e., G @ s + tau <= 0)
    #      sum(s) = 0
    #      -r*min(h, 1-h) <= s <= r*min(h, 1-h)  -- actually use -h <= s <= 1-h for α in [0,1]
    #      tau >= 0
    # Scale: to make it a proper LP, let s = r * u where |u| <= capacity
    # Actually let's just use box constraints -h_i <= s_i <= 1-h_i
    
    s_var = cp.Variable(n)
    tau_var = cp.Variable()
    
    obj = cp.Maximize(tau_var)
    constraints = [
        sum(s_var) == 0,
        tau_var >= 0,
        s_var >= -h,
        s_var <= 1.0 - h
    ]
    # For each active lag: G[i] @ s <= -tau
    for i in range(K):
        constraints.append(G[i] @ s_var <= -tau_var)
    
    prob = cp.Problem(obj, constraints)
    
    try:
        # Try solvers in order
        for solver in [cp.ECOS, cp.SCS, cp.CVXOPT, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_time=timeout_ms / 1000.0)
                if s_var.value is not None:
                    sv = np.asarray(s_var.value).flatten()
                    if np.all(np.isfinite(sv)):
                        sv -= np.mean(sv)  # ensure zero mean
                        nrm = np.linalg.norm(sv)
                        if nrm > 1e-8:
                            return sv
                        else:
                            return None
            except Exception:
                continue
    except Exception:
        pass
    
    return None

def _smooth_grad(h, beta):
    """Softmax-weighted gradient of the max."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * (2.0 / n)
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            g[:m] += wl * (1.0 - h[l:l + m])
            g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            g[s:] += wl * (1.0 - h[:m])
            g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _bin_sa(hb, deadline, mx=15000, Ti=0.002):

    bc = cc

        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)

        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, t, n):
    l = t - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        g[:m] += (1.0 - h[l:l+m])
        g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:s+m]
    return g

def _line_search(h, d, dx, radius):

def _lp_direction(h, n, top_k, timeout_s=0.3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    if mx < 1e-12:
        return None
    top_idx = np.argsort(-corr)[:top_k]
    threshold = mx - 0.08 * mx
    top_idx = top_idx[corr[top_idx] >= threshold]
    if len(top_idx) < 2:
        top_idx = np.argsort(-corr)[:min(top_k, len(corr))]
    K = len(top_idx)
    G = np.zeros((K, n))
    for i, t in enumerate(top_idx):
        G[i] = _lag_gradient(h, int(t), n)
    s_var = cp.Variable(n)
    tau_var = cp.Variable()
    obj = cp.Maximize(tau_var)
    constraints = [
        sum(s_var) == 0,
        tau_var >= 0,
        s_var >= -h,
        s_var <= 1.0 - h,
    ]
    for i in range(K):
        constraints.append(G[i] @ s_var <= -tau_var)
    prob = cp.Problem(obj, constraints)
    t_solv = time.time()
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CVXOPT]:
        if time.time() - t_solv > timeout_s:
            break
        try:
            prob.solve(solver=solver)
            if s_var.value is not None:
                sv = np.asarray(s_var.value).flatten()
                if np.all(np.isfinite(sv)):
                    sv -= np.mean(sv)
                    nrm = np.linalg.norm(sv)
                    if nrm > 1e-8:
                        return sv
                break
        except Exception:
            continue
    return None

            g[s:] += wl * (1.0 - h[:m])

        best_c = cb
        best_h = hbin.copy()
    if time.time() < d4 - 2:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d4, time.time() + budget_s * 0.02)
        hp, cp_ = _local_opt(hp, sd, r0=0.10, stall_lim=25, use_lp=True)
        if cp_ < best_c:
            best_c = cp_
            best_h = hp.copy()

    # Phase 5: Multi-resolution (18%)
    d5 = t0 + budget_s * 0.65
    nc = len(best_h)
    offsets = [-4, -6, -2, 2, 4, 6, -8, 8, -10, 10, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d5, time.time() + budget_s * 0.014)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=28, use_lp=True)
        if cr < best_c + 0.003 and time.time() < d5 - 4:
            hb = _to_bin(hr, n_try)
            sd2 = min(d5, time.time() + budget_s * 0.007)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c and time.time() < d5 - 2:
            sd3 = min(d5, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, use_lp=True)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 6: Intensive (18%)
    d6 = t0 + budget_s * 0.83
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d6 - 3:
            break
        h6 = _project(best_h.copy(), nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.05)
        h6, c6 = _local_opt(h6, sd, r0=r_init, stall_lim=45, use_lp=True)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
        if time.time() < d6 - 4:
            h6 = _project(best_h.copy(), nc / 2.0)
            sd = min(d6, time.time() + budget_s * 0.02)
            h6, c6 = _pair_opt(h6, sd, max_it=8000)
            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Basin hopping + block directions (10%)
    d7 = deadline
    scale = 0.022
    n_re = 0
    while time.time() < d7 and n_re < 45:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.5, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        # Add block-averaged gradient direction
        g_blk = _block_dir(_smooth_grad(best_h, 30.0), 12)
        pert += 0.5 * scale * g_blk / (np.max(np.abs(g_blk)) + 1e-10)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d7, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, use_lp=True)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=20, use_lp=True)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
