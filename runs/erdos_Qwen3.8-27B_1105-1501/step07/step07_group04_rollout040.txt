def _project(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:

    """Compute gradient of correlation[k] w.r.t. h, unscaled."""
    g = np.zeros(n)
    center = n - 1
    if k == center:
        g = 1.0 - 2.0 * h
    elif k < center:
        lag = center - k  # lag > 0
        # grad_m = [m <= n-lag-1](1-h[m+lag]) - [m >= lag]h[m-lag]
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        u = k - center  # u > 0
        # grad_m = [m >= u](1-h[m-u]) - [m <= n-u-1]h[m+u]
        g[u:] += (1.0 - h[:n-u])
        g[:n-u] -= h[u:]
    return g

def _lp_direction(h, z, active_idx, radius, n, dx):
    """Find LP direction to minimize linearized max correlation.
    h: current point, shape (n,)
    z: current correlation values, shape (2n-1,), already scaled by dx
    active_idx: indices of active/near-active constraints
    radius: max step size
    Returns direction d (shape n,) or None if LP fails."""
    m = len(active_idx)
    if m == 0:
        return None
    
    center = n - 1
    # Build constraint matrix A_row * d <= (t - z[r])
    # We want: z[r] + G[r,:] @ d <= t  =>  G[r,:] @ d - t <= -z[r]
    # So: A = [G, -1] (each row), b = -z[active_idx]
    
    # d variables: 0 to n-1
    # t variable: index n
    # Total variables: n+1
    
    # Build G for active constraints
    G = np.zeros((m, n))
    for i, r in enumerate(active_idx):
        G[i, :] = _grad_row(h, r, n) * dx
    
    # Constraint: G @ d - t <= -z_active
    # A_eq: sum(d) = 0
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    # Small regularization on d to avoid degenerate solutions
    c[:n] = 1e-7  # tiny penalty
    
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = -z[active_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(d) = 0
    b_eq = np.array([0.0])
    
    # Bounds on d:
    lo = np.maximum(-radius, -h * 0.999)  # d_i >= -radius and d_i >= -h_i (so h+d >= 0)
    hi = np.minimum(radius, (1 - h) * 0.999)  # d_i <= radius and d_i <= 1-h_i (so h+d <= 1)
    
    bounds = list(zip(lo, hi))
    # t is unbounded below (we minimize it), no upper bound needed
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    # Check if d is meaningful
    if np.max(np.abs(d)) < 1e-10:
        return None
    return d
