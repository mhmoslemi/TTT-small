import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _logsumexp_obj(h, beta):
    """Compute f_beta(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0.0, 1.0)
    scores = _scores(h)
    m = float(np.max(scores))
    z = (scores - m) / beta
    w = np.exp(z)
    sum_w = float(np.sum(w))
    if sum_w < 1e-300:
        value = m
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        value = m + beta * np.log(sum_w)
        w = w / sum_w
    # Compute gradient
    g = np.zeros(n)
    for ell in range(len(scores)):
        if w[ell] < 1e-15:
            continue
        d = ell - (n - 1)
        wt = w[ell] * dx
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                g[d:] += wt * (1.0 - h[:m2])
                g[:m2] -= wt * h[d:]
        else:
            g2 = -d
            m2 = n - g2
            if m2 > 0:
                g[:m2] += wt * (1.0 - h[g2:])
                g[g2:] -= wt * h[:m2]
    return value, g

def _grad_for_lags(h, lags):
    """Compute gradient rows for specific lags."""
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G * dx

def _line_search_poly(h, d, dx, t_max):
    """Exact polynomial line search along direction d."""
    n = len(h)
    u = 1.0 - h
    A = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    if t_max < 1e-14:
        return 0.0, float(np.max(A)) * dx
    # Grid + golden section
    grid = np.linspace(0.0, t_max, 40)
    vals = np.array([float(np.max(A + t * T - t * t * Q)) * dx for t in grid])
    best_i = int(np.argmin(vals))
    lo = grid[max(0, best_i - 2)]
    hi = grid[min(39, best_i + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(A + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(A + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(A + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(A + c2 * T - c2 * c2 * Q)) * dx
    t_opt = (lo + hi) / 2.0
    f_opt = float(np.max(A + t_opt * T - t_opt * t_opt * Q)) * dx
    # Also check endpoints
    candidates = [(t_opt, f_opt), (0.0, float(np.max(A)) * dx),
                  (t_max, float(np.max(A + t_max * T - t_max * t_max * Q)) * dx)]
    return min(candidates, key=lambda x: x[1])

def _smooth_descent(h, beta_start, beta_end, deadline, max_iters=3000):
    """Smooth minimax descent with adaptive beta."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        # Adaptive beta: geometric interpolation
        frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        _, grad = _logsumexp_obj(h, beta)
        # Make feasible direction
        grad -= float(np.mean(grad))
        # Clip to box
        pos = h < 1.0 - 1e-12
        neg = h > 1e-12
        grad[h >= 1.0 - 1e-12] = 0.0
        grad[h <= 1e-12] = 0.0
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            stall += 1
            continue
        d = -grad / gnorm
        d -= float(np.mean(d))
        # Line search
        alpha, new_c5 = _line_search_poly(h, d, dx, 0.1)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            # Try smaller step
            alpha2 = alpha * 0.5
            h2 = h + alpha2 * d
            h2 = _project(h2, target)
            c5_2 = _exact_c5(h2)
            if c5_2 < cur_c5 - 1e-13:
                h = h2
                cur_c5 = c5_2
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
    return best_h, best_c5

def _lp_polish(h, deadline, radius=0.08, max_iters=1000):
    """Active-set LP polish."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    gap = 0.004
    n_active = 15
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        mx = float(np.max(scores))
        active = np.where(scores >= mx - gap)[0]
        if len(active) > n_active:
            order = np.argsort(scores[active])
            active = active[order][-n_active:]
        lags = active - (n - 1)
        G = _grad_for_lags(h, lags)
        m = len(active)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[active]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for j in range(n):
            lo = max(-radius, -h[j])
            hi2 = min(radius, 1.0 - h[j])
            bounds.append((lo, hi2))
        bounds.append((None, None))
        d = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
        except Exception:
            pass
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            # Fallback: gradient direction
            _, grad = _logsumexp_obj(h, 0.01)
            d = -grad
            d -= float(np.mean(d))
            dnorm = float(np.linalg.norm(d))
            if dnorm < 1e-12:
                stall += 1
                gap *= 1.3
                n_active = min(n_active + 2, 25)
                if stall % 20 == 0:
                    radius *= 1.5
                continue
            d = d / dnorm * radius
        alpha, new_c5 = _line_search_poly(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.2)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 10)
        else:
            stall += 1
            radius *= 0.75
            gap *= 1.2
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=150):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.4)[0]
    zeros = np.where(h < 0.6)[0]
    rng = np.random.default_rng(12345)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(3, len(ones) // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search_poly(h, d, dx, 0.3)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.4)[0]
                zeros = np.where(h < 0.6)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freq = 3
    perturb = np.zeros(n)
    for _ in range(n_freq):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + amplitude * perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
