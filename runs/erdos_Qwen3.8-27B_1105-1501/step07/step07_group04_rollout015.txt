def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def lag_gradient(h, k):
    """Exact gradient of F_k = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])"""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        # h[j] appears as h[i+k] when j = i+k, i.e., j >= k
        # contributes (1 - h[j-k])
        # h[j] appears as h[i] when j = i, i.e., j <= n-1-k
        # contributes -h[j+k]
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        g[idx2] -= h[idx2 + k]
    return g

    if target >= n:
        return np.ones(n)

    g = np.zeros(n)
    g[k:] += (1.0 - h[:n-k])
    g[:n-k] -= h[k:]

def pair_line_search(h, i, j):
    """Exact line search for d = e_i - e_j. Returns (best_alpha, best_c5)."""
    n = len(h)
    dx = 2.0 / n
    # Feasible interval for alpha in h + alpha*(e_i - e_j)
    # h[i] + alpha in [0,1] → alpha in [-h[i], 1-h[i]]
    # h[j] - alpha in [0,1] → alpha in [h[j]-1, h[j]]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi <= lo + 1e-15:
        return 0.0, exact_c5(h)
    
    S0 = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Dense grid
    n_grid = 80
    alphas = np.linspace(lo, hi, n_grid)
    vals = np.empty(n_grid)
    for idx in range(n_grid):
        a = alphas[idx]
        vals[idx] = float(np.max(S0 + a * T - a * a * Q))
    
    best_i = int(np.argmin(vals))
    
    # Golden section refinement
    lo_r = alphas[max(0, best_i - 2)]
    hi_r = alphas[min(n_grid - 1, best_i + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        if hi_r - lo_r < 1e-15:
            break
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * Q))
        f2 = float(np.max(S0 + c2 * T - c2*c2 * Q))
        if f1 < f2:
            hi_r = c2
        else:
            lo_r = c1
    
    opt_a = 0.5 * (lo_r + hi_r)
    opt_c5 = float(np.max(S0 + opt_a * T - opt_a*opt_a * Q)) * dx
    
    # Also check endpoints
    c5_lo = float(np.max(S0 + lo * T - lo*lo * Q)) * dx
    c5_hi = float(np.max(S0 + hi * T - hi*hi * Q)) * dx
    c5_0 = float(np.max(S0)) * dx
    
    best = min(opt_c5, c5_lo, c5_hi, c5_0)
    if best == c5_0:
        return 0.0, c5_0
    if best == opt_c5:

    if best == c5_lo:

    return hi, c5_hi

1. Subgradient + line search (generic direction)

What if I focus more on:
1. **Exact pair-coordinate search** with smart pair selection (this is the most targeted)

- n = 180, so corr has 359 entries

Wait, I'm making this too complicated. Let me simplify the pair selection:

For the worst lag k:
- F_k = sum_{i=0}^{n-k-1} h[i+k] * (1 - h[i])
- The pairs (i, i+k) are the natural ones to try
- For each such pair, the direction that reduces F_k is: if h[i+k] > h[i], move mass from i+k to i (d[i]=1, d[i+k]=-1)

But we should also consider: maybe moving mass from i+k to some other position j (not i) would help more. So let's also try random pairs.
