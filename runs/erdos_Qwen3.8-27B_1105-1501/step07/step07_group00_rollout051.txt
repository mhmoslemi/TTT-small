import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Golden section search on [0, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    # Also check endpoints
    f_lo = float(np.max(S0)) * dx
    f_hi = float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx
    if f_lo < opt_val:
        opt_t, opt_val = 0.0, f_lo
    if f_hi < opt_val:
        opt_t, opt_val = t_hi, f_hi
    return opt_t, opt_val

def _active_lp(h, scores, radius, n_active=12, gap=0.004):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    # Build gradient matrix
    m = len(active)
    G = np.zeros((m, n))
    for idx, d in enumerate(lags):
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[idx, d:] += (1.0 - h[:mm])
                G[idx, :mm] += -h[d:]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[idx, :mm] += (1.0 - h[g:])
                G[idx, g:] += -h[:mm]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_dir(h, scores, beta=30.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = beta * (scores - mx)
    shifted = np.clip(shifted, -100.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        idx = int(np.argmax(scores))
        w = np.zeros(len(scores))
        w[idx] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    g = np.zeros(n)
    for i in range(len(lags)):
        if w[i] < 1e-12:
            continue
        d = lags[i]
        wi = w[i]
        if d >= 0:
            mm = n - d
            if mm > 0:
                g[d:] += wi * (1.0 - h[:mm])
                g[:mm] += wi * (-h[d:])
        else:
            gg = -d
            mm = n - gg
            if mm > 0:
                g[:mm] += wi * (1.0 - h[gg:])
                g[gg:] += wi * (-h[:mm])
    g -= float(np.mean(g))
    # Clip to stay within box
    pos = g > 0
    neg = g < 0
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / max(radius, 1e-10))
    g[neg] = np.maximum(g[neg], -h[neg] / max(radius, 1e-10))
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g -= float(np.mean(g))
    return g

                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)

            radius *= 0.75
            gap *= 1.3

            if stall % 30 == 0 and radius < 0.02:

    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=300):

    target = n / 2.0

    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_swaps = max(1, len(ones) // 4)
        for _ in range(n_swaps):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.25)
            if c_try < cur_c5 - 1e-12:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual = _exact_c5(h_new)
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    nfreq = 5
    freqs = rng.uniform(0.5, 10.0, size=nfreq)
    phases = rng.uniform(0, 2 * np.pi, size=nfreq)
    amps = rng.uniform(0.3, 1.0, size=nfreq)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
