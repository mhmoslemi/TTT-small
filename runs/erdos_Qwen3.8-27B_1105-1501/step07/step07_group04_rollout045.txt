Let me think about the LP formulation more carefully:

For the LP subproblem at point h:
- Variables: d_0, ..., d_{n-1} (direction), u_0, ..., u_{n-1} (L1 penalty slack), t (epigraph)
- Constraints:
  - For each active correlation r: F_r(h) + ∇F_r(h)·d - t ≤ 0
  - sum(d_i) = 0 (mass conservation)
  - -radius ≤ d_i ≤ radius (trust region)
  - d_i - u_i ≤ 0, -d_i - u_i ≤ 0 (L1 penalty)
  - t ≤ cur_score + epsilon (don't increase)

Objective: minimize t + eps * sum(u_i)

This should give a direction that simultaneously reduces multiple active correlation peaks.

Let me also think about what makes this problem hard. The key insight is that we need h to be "spread out" so that for every shift k, the overlap between h and (1-h shifted by k) is minimized. 

For a step function with n points, we have n/2 ones and n/2 zeros. The binary case is a combinatorial optimization problem.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    # Project v onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_junction(h, k, n):
    """Compute gradient of correlation at lag k (signed) with respect to h."""
    # k = 0: center, grad_i = 1 - 2*h_i
    # k > 0: F = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
    #   dF/dh[j] for j in [k, n-1]: += (1 - h[j-k])
    #   dF/dh[j] for j in [0, n-k-1]: -= h[j+k]
    # k < 0: u = -k, F = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])
    #   dF/dh[j] for j in [0, n-u-1]: += (1 - h[j+u])
    #   dF/dh[j] for j in [u, n-1]: -= h[j-u]
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

def _build_lp(h, z, active_idx, radius, cur_score, eps=1e-5):
    """Build and solve LP for descent direction.
    
    h: current point (n,)
    z: correlation values (2n-1,) scaled by dx
    active_idx: indices of active constraints
    radius: trust region radius
    cur_score: current C5 value
    eps: L1 penalty weight
    """
    n = len(h)
    m = len(active_idx)
    dx = 2.0 / n
    
    # Variables: d_0...d_{n-1}, u_0...u_{n-1}, t
    # Total: 2n + 1
    n_vars = 2 * n + 1
    tau_idx = 2 * n
    
    # Objective: t + eps * sum(u_i)
    c_obj = np.zeros(n_vars)
    c_obj[tau_idx] = 1.0
    c_obj[n:2*n] = eps  # u_i penalty
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint r: grad_r · d - t <= -z[r]
    # Rearranged: grad_r · d - t <= cur_score - z[r]  ... wait
    # Actually: z[r] + grad_r · d - t <= 0, i.e., grad_r · d - t <= -z[r]
    # But we want t >= z[r] + grad_r · d for all r
    # So: -grad_r · d + t >= z[r], i.e., grad_r · d - t <= -z[r]
    
    # In linprog form: A_ub @ x <= b_ub
    # Row for constraint r: [grad_r, 0, -1] @ x <= -z[r]
    
    A_rows = []
    bvals = []
    
    for r in active_idx:
        g = _grad_junction(h, int(r - (n-1)), n) * dx  # scaled gradient
        row = np.zeros(n_vars)
        row[:n] = g  # grad_r · d
        row[tau_idx] = -1.0  # -t
        A_rows.append(row)
        bvals.append(-z[r])
    
    # Trust region: -radius <= d_i <= radius
    # d_i <= radius: [e_i, 0, 0] @ x <= radius
    # -d_i <= radius: [-e_i, 0, 0] @ x <= radius
    for i in range(n):
        row_pos = np.zeros(n_vars)
        row_pos[i] = 1.0
        row_pos[i] = min(radius, 1.0 - h[i] + 1e-14)
        # Actually let's use the tighter bounds
        row_neg = np.zeros(n_vars)
        row_neg[i] = -1.0
        A_rows.append(row_pos)
        bvals.append(min(radius, 1.0 - h[i] + 1e-14))
        A_rows.append(row_neg)
        bvals.append(min(radius, h[i] + 1e-14))
    
    # L1 penalty: d_i - u_i <= 0, -d_i - u_i <= 0
    for i in range(n):
        row1 = np.zeros(n_vars)
        row1[i] = 1.0
        row1[n+i] = -1.0
        A_rows.append(row1)
        bvals.append(0.0)
        
        row2 = np.zeros(n_vars)
        row2[i] = -1.0
        row2[n+i] = -1.0
        A_rows.append(row2)
        bvals.append(0.0)
    
    # t <= cur_score + small (don't increase)
    row_t = np.zeros(n_vars)
    row_t[tau_idx] = 1.0
    A_rows.append(row_t)
    bvals.append(cur_score + 1e-9)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(bvals)
    
    # Equality: sum(d_i) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_d = max(-radius, -(1.0 - h[i]) - 1e-14)  # h[i] + d[i] >= 0
        hi_d = min(radius, 1.0 - h[i] + 1e-14)  # h[i] + d[i] <= 1
        # Wait, h[i] + d[i] >= 0 means d[i] >= -h[i]
        # h[i] + d[i] <= 1 means d[i] <= 1 - h[i]
        lo_d = max(-radius, -h[i] - 1e-14)
        hi_d = min(radius, 1.0 - h[i] + 1e-14)
        bounds.append((lo_d, hi_d))
    for i in range(n):
        bounds.append((0.0, None))  # u_i >= 0
    bounds.append((None, None))  # t unconstrained
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        d = res.x[:n].copy()
        # Clean up tiny values
        d[np.abs(d) < 1e-12] = 0.0
        return d
    return None

def _line_search(h, d, dx, radius):
    """Exact line search for h + alpha*d along feasible direction."""
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find max feasible alpha
    alpha_max = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    
    if alpha_max < 1e-12:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, alpha_max, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        vals = C0 + t * T - t * t * Q
        grid_f[idx] = float(np.max(vals)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-10:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(C0 + c1*T - c1*c1*Q)) * dx
        f2 = float(np.max(C0 + c2*T - c2*c2*Q)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(C0 + c1*T - c1*c1*Q)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(C0 + c2*T - c2*c2*Q)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    else:
        opt_t = grid_t[best_idx]
    opt_val = float(np.max(C0 + opt_t*T - opt_t*opt_t*Q)) * dx
    
    # Compare with endpoints
    candidates = [(0.0, float(np.max(C0)*dx)),
                  (opt_t, opt_val),
                  (alpha_max, float(np.max(C0 + alpha_max*T - alpha_max*alpha_max*Q))*dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_optimize(h, deadline, radius_init=0.06, max_iter=500):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        if radius < 1e-5:
            break
            
        z = _corr_values(h)
        cur_max = float(np.max(z))
        
        # Select active constraints
        # Top K + those within 1% of max
        tol = max(1e-8, 0.01 * cur_max)
        active = np.where(z >= cur_max - tol)[0]
        # Limit to top 80 for speed
        if len(active) > 80:
            top_vals = z[active]
            order = np.argsort(-top_vals)[:80]
            active = active[order]
        
        d = _build_lp(h, z, active, radius, cur_max)
        
        if d is None or float(np.max(np.abs(d))) < 1e-13:
            stall += 1
            radius *= 0.7
            continue
        
        # Ensure zero sum (mass conservation)
        d = d - np.mean(d)
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.15)
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c5

def _binary_eval(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr

def _to_binary(h, n):
    """Convert to binary by taking top n/2 values."""
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iters=200000, temp_init=0.003, seed=42):
    """Simulated annealing on binary configurations."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_eval(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5, _ = _binary_eval(cur)
        
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            best_stall += 1
    
    return best_h, best_c5

def _targeted_binary(h_bin, deadline, max_iters=50000, seed=99):
    """Targeted binary search focusing on active constraints."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_eval(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Focus on positions related to maximum correlation
        max_idx = int(np.argmax(cur_corr))
        k = max_idx - (n - 1)
        
        best_delta = 0.0
        best_move = None
        n_try = min(150, len(ones) * len(zeros))
        
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            
            # Check if this move affects the active constraint
            if k > 0:
                affects = (abs(oi - k) < n and oi >= k) or (oz >= k and oz < n - k) or (oi < n - k and oi + k < n) or (oz < n - k and oz + k < n)
            elif k < 0:
                u = -k
                affects = (oz >= u) or (oi >= u)
            else:
                affects = True
            
            if not affects:
                continue
                
            cur[oi] = 0.0
            cur[oz] = 1.0
            new_c5, _ = _binary_eval(cur)
            delta = new_c5 - cur_c5
            cur[oi] = 1.0
            cur[oz] = 0.0
            
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full') * dx
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
