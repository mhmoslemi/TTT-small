        else:

def _thue_morse(n, k):
    """Generate first k bits of Thue-Morse sequence, then take n balanced."""
    # Thue-Morse: t(i) = popcount(i) mod 2
    tm = np.array([(i.bit_count() & 1) for i in range(k)], dtype=float)
    # We need exactly n/2 ones. Count how many ones we have.
    n_ones = int(np.sum(tm))
    n_needed = n // 2
    if n_ones == n_needed:
        return tm[:n]  # wait, tm has length k, not n
    # Just use first n elements and balance
    h = tm[:n].copy()
    n_ones = int(np.sum(h))
    n_zeros = n - n_ones
    if n_ones > n_needed:
        # Need to flip some 1s to 0s
        excess = n_ones - n_needed
        ones_idx = np.where(h == 1)[0]
        # Flip the last 'excess' ones
        flip = ones_idx[-excess:]
        h[flip] = 0
    elif n_ones < n_needed:
        deficit = n_needed - n_ones
        zeros_idx = np.where(h == 0)[0]
        flip = zeros_idx[-deficit:]
        h[flip] = 1
    return h
