def _line_search(h, s, dx):
    """Find best alpha in [0, alpha_max] along h + alpha*s"""
    n = len(h)
    # Compute alpha_max from box constraints
    pos = s > 1e-14
    neg = s < -1e-14
    alpha_max = 1.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / s[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-s[neg]))))
    
    if alpha_max < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Precompute T and Q
    S0 = _raw_corr(h)
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    
    # Model: raw_alpha = S0 + alpha*T - alpha^2*Q
    # Evaluate at candidate alphas
    cands = [0.0, alpha_max]
    cands.extend(np.linspace(0, alpha_max, 33).tolist())
    
    # Add min points of each quadratic
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if 0 < av < alpha_max:
                cands.append(av)
    
    # Also add pairwise intersection points of top constraints
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            # S0[ki] + a*T[ki] - a^2*Q[ki] = S0[kj] + a*T[kj] - a^2*Q[kj]
            # (Q[ki]-Q[kj])*a^2 - (T[ki]-T[kj])*a + (S0[ki]-S0[kj]) = 0
            A = Q[ki] - Q[kj]
            B = -(T[ki] - T[kj])

                if abs(B) > 1e-14:

                    if 0 <= ar <= alpha_max:
                        cands.append(ar)
            else:

                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B+sq)/(2*A), (-B-sq)/(2*A)]:
                        if 0 <= ar <= alpha_max:
                            cands.append(ar)

    for a in cands:
        a = max(0.0, min(a_max, a))
        val = float(np.max(S0 + a*T - a*a*Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement around best_a
    width = alpha_max * 0.1
    lo_r = max(0.0, best_a - width)
    hi_r = min(alpha_max, best_a + width)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx

    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx)
        if c5t < cur - 1e-14:
            h_new = _project(h + a * d, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=20000, Ti=0.003):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _cont_sa(h, deadline, mx=15000, Ti=0.02):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur = h.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(2024)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = rng.uniform(-0.15, 0.15)
        cur[i] += delta
        cur[j] -= delta
        cur[i] = max(0.0, min(1.0, cur[i]))
        cur[j] = max(0.0, min(1.0, cur[j]))
        if abs(float(np.sum(cur)) - target) > 0.01:
            cur = _project(cur, target)
        nc = _exact_c5(cur)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cc = nc
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Core local optimization with LP directions (30%)
    d1 = t0 + budget_s * 0.30
    h1, c1 = _local_opt(h_init, d1, r0=0.18)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair optimization (10%)
    d2 = t0 + budget_s * 0.40
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=25000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary SA + continuous SA (12%)
    d3 = t0 + budget_s * 0.52
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, t0 + budget_s * 0.46, mx=12000, Ti=0.003)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3:
        hp = _project(best_h.copy(), nb / 2.0)
        hp, cp = _cont_sa(hp, d3, mx=8000, Ti=0.02)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
        if time.time() < d3:
            hp2 = _project(hp.copy(), nb / 2.0)
            hp2, cp2 = _local_opt(hp2, d3, r0=0.10)
            if cp2 < best_c:
                best_c = cp2
                best_h = hp2.copy()

    # Phase 4: Multi-resolution exploration (18%)
    d4 = t0 + budget_s * 0.70
    nc = len(best_h)
    offsets = [-6, -4, -8, 4, 6, 8, -12, 12, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.14)
        if cr < best_c + 0.003 and time.time() < d4 - 2:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.007)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.002)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d4 - 1:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.06)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Intensive polishing (20%)
    d5 = t0 + budget_s * 0.90
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d5 - 2:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.05)
        h5, c5 = _local_opt(h5, sd, r0=r_init)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d5 - 4:
            h5b = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.025)
            h5b, c5b = _pair_opt(h5b, sd, max_it=12000)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b.copy()

    # Phase 6: Basin hopping (5%)
    d6 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.4 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sd, r0=0.07)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    if final_c > best_c + 1e-10:
        best_c = final_c
    return best_h, final_c, len(best_h)
