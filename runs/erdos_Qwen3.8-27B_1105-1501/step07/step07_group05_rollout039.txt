1. j = i: contributes h[i]*(1-h[i+p]) if 0≤i+p<n
2. j+p = i, i.e., j = i-p: contributes h[i-p]*(1-h[i]) if 0≤i-p<n

So:
corr[m] = [terms not involving h[i]] + h[i]*(1-h[i+p])*[0≤i+p<n] + h[i-p]*(1-h[i])*[0≤i-p<n]
= [terms not involving h[i]] + h[i]*(1-h[i+p])*[0≤i+p<n] + h[i-p] - h[i]*h[i-p]*[0≤i-p<n]
= [[terms not involving h[i]] + h[i-p]*[0≤i-p<n]] + h[i]*[(1-h[i+p])*[0≤i+p<n] - h[i-p]*[0≤i-p<n]]

So:
b_m = corr[m] - c_m * h[i] where
c_m = (1-h[i+p])*[0≤i+p<n] - h[i-p]*[0≤i-p<n]

And b_m = corr[m] - c_m * h[i] (evaluated at current values).

This is what I had before. Good.

Now, the binary search for the optimal t:
We want to find t* = argmin_{t∈[0,1]} max_m (c_m * t + b_m).

The binary search on the value v works as follows:
- v is feasible if ∃ t ∈ [0,1] such that c_m * t + b_m ≤ v for all m.
- This means: for all m with c_m > 0: t ≤ (v - b_m)/c_m
  For all m with c_m < 0: t ≥ (v - b_m)/c_m
  For all m with c_m = 0: b_m ≤ v
- So the feasible interval for t is [max(0, max_{c<0} (v-b_m)/c_m), min(1, min_{c>0} (v-b_m)/c_m)]
- If this interval is non-empty, v is feasible.

The minimum feasible v is the optimum. Binary search finds it to arbitrary precision.

Once we find v*, we find t* in the feasible interval (which will be a single point or small interval at the boundary).

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _coord_optimize(h, i):
    n = len(h)
    corr = _full_corr(h)
    p = np.arange(2 * n - 1) - (n - 1)
    ip = i + p
    im = i - p
    valid_ip = (ip >= 0) & (ip < n)
    valid_im = (im >= 0) & (im < n)
    h_ip = np.zeros(2 * n - 1)
    h_im = np.zeros(2 * n - 1)
    h_ip[valid_ip] = h[ip[valid_ip]]
    h_im[valid_im] = h[im[valid_im]]
    c = (1.0 - h_ip) * valid_ip.astype(float) - h_im * valid_im.astype(float)
    b = corr - c * h[i]
    f0 = float(np.max(b))
    f1 = float(np.max(c + b))
    hi_v = max(f0, f1)
    lo_v = min(f0, f1)
    for _ in range(50):
        mid_v = (lo_v + hi_v) / 2.0
        pos_m = c > 1e-14
        neg_m = c < -1e-14
        zero_m = ~pos_m & ~neg_m
        if np.any(zero_m) and float(np.max(b[zero_m])) > mid_v:
            lo_v = mid_v
            continue
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg_m):
            t_lo = max(t_lo, float(np.max((mid_v - b[neg_m]) / c[neg_m])))
        if np.any(pos_m):
            t_hi = min(t_hi, float(np.min((mid_v - b[pos_m]) / c[pos_m])))
        if t_lo <= t_hi:
            hi_v = mid_v
        else:
            lo_v = mid_v
    opt_v = (lo_v + hi_v) / 2.0
    pos_m = c > 1e-14
    neg_m = c < -1e-14
    t_lo = 0.0
    t_hi = 1.0
    if np.any(neg_m):
        t_lo = max(t_lo, float(np.max((opt_v - b[neg_m]) / c[neg_m])))
    if np.any(pos_m):
        t_hi = min(t_hi, float(np.min((opt_v - b[pos_m]) / c[pos_m])))
    t_opt = (t_lo + t_hi) / 2.0
    t_opt = max(0.0, min(1.0, t_opt))

def _coord_sweep(h, max_sweeps=20, deadline=None):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _exact_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:

        c = _exact_c5(h)

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    best_val = float('inf')
    best_a = 0.0
    cands = [a_lo, a_hi, 0.0]
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    width = a_hi - a_lo
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

def _gen_dirs(h, rng, nd=5):
    n = len(h)
    dirs = []
    for beta in [15.0, 30.0, 60.0]:
        g = _smooth_grad(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:

        d -= np.mean(d)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_local_opt(hb, deadline, max_it=30000):
    n = len(hb)
    ones = np.where(hb > 0.5)[0].tolist()
    zeros = np.where(hb <= 0.5)[0].tolist()
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        if nc < cc - 1e-14:
            cur = hn
            cc = nc
            ones[ i] = oz
            zeros[j] = oi
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
            stall = 0
        else:
            stall += 1
    return bh, bc

    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
