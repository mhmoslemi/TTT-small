    # sum(sp) - sum(sn) = 0

    c = np.zeros(n_vars)

    # sum(sp) + sum(sn) = 1

def _lp_direction(h, peak_count=8):
    """Find LP direction to reduce peaks. Returns direction u with sum(u)=0, ||u||_1=1."""
    n = len(h)
    L = 2 * n - 1
    R = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(R)[-peak_count:]
    
    G = _grad_matrix(h)
    
    # Variables: sp(0..n-1), sn(n..2n-1), beta(2n)
    n_vars = 2 * n + 1
    beta_idx = 2 * n
    
    c = np.zeros(n_vars)
    c[beta_idx] = -1.0
    
    A_eq = np.zeros((2, n_vars))
    b_eq = np.zeros(2)
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq[1] = 1.0
    
    A_ub = np.zeros((len(top_idx), n_vars))
    for j, r in enumerate(top_idx):
        A_ub[j, :n] = G[r, :]
        A_ub[j, n:2*n] = -G[r, :]
        A_ub[j, beta_idx] = 1.0
    b_ub = np.zeros(len(top_idx))
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(-100.0, 100.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n] - res.x[n:2*n]
            if abs(np.sum(u)) > 1e-8:
                u = u - np.sum(u) / n
            if np.max(np.abs(u)) < 1e-10:
                return None
            return u
    except Exception:
        pass
    return None

    c = np.zeros(n_vars)
    c[beta_idx] = -1.0
    A_eq = np.zeros((2, n_vars))
    b_eq = np.array([0.0, 1.0])
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    k = len(active_idx)
    A_ub = np.zeros((k, n_vars))
    for j, r in enumerate(active_idx):
        A_ub[j, :n] = G[r, :]
        A_ub[j, n:2*n] = -G[r, :]
        A_ub[j, beta_idx] = 1.0
    b_ub = np.zeros(k)
    bounds = [(0.0, 1.0)] * (2 * n) + [(-50.0, 50.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            u = res.x[:n] - res.x[n:2*n]
            u = u - np.sum(u) / n
            if np.max(np.abs(u)) < 1e-12:
                return None
            return u
    except Exception:
        pass
    return None

def _try_dir(h, d, t_cap=2.0):
    n = len(h)
    d = d - np.sum(d) / n
    if np.max(np.abs(d)) < 1e-14:
        return None, None
    iv = _feasible_interval(h, d, t_cap)
    if iv is None:
        return None, None
    t_lo, t_hi = iv
    R0 = np.correlate(h, 1.0 - h, mode='full')
    A = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')
    t_opt, _ = _line_search_quad(R0, A, B, t_lo, t_hi)
    h_new = h + t_opt * d
    h_new = np.clip(h_new, 0.0, 1.0)
    if abs(np.sum(h_new) - n / 2.0) > 1e-10:
        h_new = _project(h_new, n / 2.0)
    c5_new = _exact_c5(h_new)
    return h_new, c5_new

def _select_active(R, max_count=20, gap=0.005):
    M = float(np.max(R))
    idx = np.where(R > M - gap)[0]
    if len(idx) > max_count:
        idx = np.argsort(R)[-max_count:]
    elif len(idx) < 3:
        idx = np.argsort(R)[-5:]
    return idx

def _lp_descent(h, deadline, peak_cfg=None, t_cap=2.0, seed=42):
    n = len(h)
    if peak_cfg is None:
        peak_cfg = [(6, 0.002), (10, 0.004), (15, 0.006), (25, 0.01)]
    h = _project(h.copy(), n / 2.0)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    while time.time() < deadline and stall < 250:
        it += 1
        if it > 3:
            R = np.correlate(h, 1.0 - h, mode='full')
            active = _select_active(R)
            cfg = peak_cfg[it % len(peak_cfg)]
            d = _lp_direction(h, active)
            if d is not None:
                h_try, c5_try = _try_dir(h, d, t_cap)
                if h_try is not None and c5_try < cur_c5 - 1e-13:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    continue
        # Fallback: weighted gradient step
        R = np.correlate(h, 1.0 - h, mode='full')
        w = np.exp(30.0 * (R - np.max(R)))
        w = w / np.sum(w)
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.sum(grad) / n
        h_try, c5_try = _try_dir(h, grad, t_cap * 0.3)
        if h_try is not None and c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            continue
        # Pair moves
        improved = False
        for _ in range(12):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d2 = np.zeros(n)
            if h[i] <= h[j]:
                d2[i] = 1.0
                d2[j] = -1.0
            else:
                d2[i] = -1.0
                d2[j] = 1.0
            h_try, c5_try = _try_dir(h, d2, t_cap * 0.4)
            if h_try is not None and c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
            if stall % 80 == 0:
                pert = rng.normal(0, 0.02, n)
                pert -= np.sum(pert) / n
                h_try, c5_try = _try_dir(h, pert, 0.8)
                if h_try is not None and c5_try < best_c5:
                    h = h_try
                    cur_c5 = c5_try
                    best_h = h_try.copy()
                    best_c5 = cur_c5
                    stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=80000, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        c_try = _exact_c5(cur)
        if c_try < cur_c5:
            cur_c5 = c_try
            ones[ones.index(oi)] = oz
            zeros[zeros.index(oz)] = oi
            stall = 0
            if c_try < best_c5:
                best_c5 = c_try
                best_h = cur.copy()
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(160) * 0.5
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP descent at parent (30% time)
    d1 = t0 + budget_s * 0.30
    pc1 = [(6, 0.002), (10, 0.004), (15, 0.006)]
    h1, c5_1 = _lp_descent(h, d1, peak_cfg=pc1, t_cap=2.0, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (30% time)
    d2 = t0 + budget_s * 0.60
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 300 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.022)
        pc2 = [(8, 0.003), (12, 0.005)]
        h_r, c5_r = _lp_descent(h_r, sub_dl, peak_cfg=pc2, t_cap=1.5, seed=200 + abs(dn))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        sub_dl = min(d3, time.time() + budget_s * 0.07)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=120000, seed=777)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.05)
            pc3 = [(6, 0.002), (10, 0.004)]
            h_pol, c5_pol = _lp_descent(h_bin.copy(), sub_dl2, peak_cfg=pc3, t_cap=1.0, seed=333)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Final intensive polish (20% time)
    d4 = deadline - 5.0
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    pc4 = [(4, 0.001), (6, 0.002), (8, 0.003), (12, 0.005)]
    sub_dl = min(d4, deadline - 2.0)
    h4, c5_4 = _lp_descent(h4, sub_dl, peak_cfg=pc4, t_cap=0.8, seed=555)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Try a few nearby n values if time remains
    if time.time() < deadline - 10:
        n_curr = len(best_h)
        for dn in [-4, 4, -8, 8]:
            if time.time() > deadline - 8:
                break
            n_try = n_curr + dn
            if n_try < 64 or n_try > 300 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(deadline - 5, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_descent(h_r, sub_dl, peak_cfg=pc4, t_cap=0.7, seed=600 + abs(dn))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
