    """Find a direction u (sum=0, L1=1) that reduces the top n_peaks correlation values."""
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    R = np.correlate(h, o, mode='full')
    
    # Select active peaks
    top_idx = np.argsort(R)[-n_peaks:]
    G = _grad_matrix(h)
    G_active = G[top_idx, :]  # shape (n_peaks, n)
    
    # LP: minimize -alpha
    # Variables: sp_0..sp_{n-1}, sn_0..sn_{n-1}, alpha
    # Total: 2n + 1
    n_var = 2 * n + 1
    alpha_idx = 2 * n
    
    # Objective: minimize -alpha
    c = np.zeros(n_var)
    c[alpha_idx] = -1.0
    
    # Equality constraints:
    # sum(sp) - sum(sn) = 0
    # sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, n_var))
    A_eq[0, :n] = 1.0       # sum(sp)
    A_eq[0, n:2*n] = -1.0   # -sum(sn)
    A_eq[1, :n] = 1.0       # sum(sp)
    A_eq[1, n:2*n] = 1.0    # sum(sn)
    b_eq = np.array([0.0, 1.0])
    
    # Inequality constraints: G_active @ (sp - sn) + alpha <= 0
    # i.e., G_active @ sp - G_active @ sn + alpha <= 0
    A_ub = np.zeros((n_peaks, n_var))
    A_ub[:, :n] = G_active      # coefficient of sp
    A_ub[:, n:2*n] = -G_active  # coefficient of sn
    A_ub[:, alpha_idx] = 1.0    # coefficient of alpha
    b_ub = np.zeros(n_peaks)
    
    # Bounds: 0 <= sp <= 1, 0 <= sn <= 1, alpha free
    bounds = [(0, 1)] * (2 * n) + [(-1e6, 1e6)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success and result.fun < -1e-10:
        # alpha = -result.fun > 0, direction is improving
        sp = result.x[:n]
        sn = result.x[n:2*n]
        u = sp - sn
        return u, float(-result.fun)
    return None, 0.0

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    if abs(v.sum() - target) < 1e-14:
        return v
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-11:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

            x[mask] += res / cnt

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        m = k + (n - 1)
        if k >= 0:
            G[m, k:n] += o[:n - k]
            G[m, 0:n - k] -= h[k:n]
        else:
            p = -k
            G[m, 0:n - p] += o[p:n]
            G[m, p:n] -= h[:n - p]
    return G

        if n_try < 80 or n_try > 280 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.025)
        h_r, c5_r = _local_opt(h_r, sub_dl, n_peaks_init=6, stall_limit=60, seed=200 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    if time.time() < d3:
        h_bin = _to_binary(best_h)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d3, time.time() + budget_s * 0.08)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=40000, seed=333)
        sub_dl2 = min(d3, time.time() + budget_s * 0.035)
        h_bin, c5_bin = _binary_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Continuous polish from binary
        if time.time() < d3:
            h_pol = _project(best_h.copy(), n_curr / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.04)
            h_pol, c5_pol = _local_opt(h_pol, sub_dl3, n_peaks_init=8, stall_limit=50, seed=444)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 4: Intensive refinement (10% time)
    d4 = deadline - 2
    n_curr = len(best_h)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _local_opt(h4, d4, n_peaks_init=10, stall_limit=120, seed=555)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Final: try a couple nearby resolutions
    n_curr = len(best_h)
    for dn in [-2, 2]:
        remaining = deadline - time.time()
        if remaining < 10:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline - 2, time.time() + min(remaining * 0.4, 30))
        h_r, c5_r = _local_opt(h_r, sub_dl, n_peaks_init=6, stall_limit=40, seed=666 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
