```python
import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 0
_g_target = 0
_g_dx = 0
_g_tau = 0.1

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * (2.0 / n))

def _build_G(h):
    """Build gradient matrix G of shape (2n-1, n)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    # Zero lag
    G[mid, :] = 1.0 - 2.0 * h
    # Positive lags
    for d in range(1, n):
        m = n - d
        # Row mid + d: score = sum_{j=0}^{m-1} h[j]*(1-h[j+d])
        G[mid + d, :m] += (1.0 - h[d:d+m])
        G[mid + d, d:] += (-h[:m])
        # Row mid - d: score = sum_{j=d}^{n-1} h[j]*(1-h[j-d])
        G[mid - d, d:] += (1.0 - h[:m])
        G[mid - d, :m] += (-h[d:d+m])
    return G

def _lse_value(x):
    """Compute LSE of scaled scores"""
    s = _scores(x) * _g_dx
    mval = s.max()
    u = (s - mval) / _g_tau
    return float(mval + _g_tau * np.log(np.sum(np.exp(u))))

def _lse_grad(x):
    """Compute gradient of LSE"""
    s = _scores(x) * _g_dx
    mval = s.max()
    u = (s - mval) / _g_tau
    z = np.exp(u)
    p = z / z.sum()
    G = _build_G(x)
```
