def _lp_direction(h, raw, dx, n, m_lags=20, radius=0.15):
    """
    Solve LP to find direction d that minimizes max linearized overlap.
    Returns (d, t_new) or (None, None) if LP fails.
    """
    n = len(h)
    # Select top m lags
    top_idx = np.argsort(-raw)[:m_lags]
    
    # Compute gradients for selected lags
    # In correlation vector, position r corresponds to shift d = r - (n-1)
    grads = []
    for r in top_idx:
        shift = int(r) - (n - 1)
        g = _lag_gradient_vec(h, shift)
        grads.append(g)
    grads = np.array(grads)  # shape (m, n)
    
    # LP: minimize t
    # subject to: grads[k] @ d <= (raw[r_k] - max_raw) + t  for each k
    #             sum(d) = 0
    #             -h[i] <= d[i] <= 1-h[i]
    #             |d[i]| <= radius (optional trust region)
    
    max_raw = float(np.max(raw))
    # We want: f_r(h + d) <= max_raw + t (linearized)
    # f_r(h + d) ≈ raw[r] + grads[k] @ d
    # So: grads[k] @ d <= max_raw + t - raw[r_k]
    #   => grads[k] @ d - t <= max_raw - raw[r_k]
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -h,
        d_var <= 1 - h,
        d_var >= -radius,
        d_var <= radius,
    ]
    for k in range(len(top_idx)):
        r_k = int(top_idx[k])
        constraints.append(grads[k] @ d_var - t_var <= max_raw - raw[r_k])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver='CLARABEL')  # or 'ECOS', 'SCS'
    except:
        try:
            prob.solve(solver='ECOS')
        except:
            prob.solve(solver='SCS')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    d_opt = d_var.value
    if d_opt is None:
        return None
    d_opt = np.real(d_opt)
    return d_opt

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        i_start, i_end = shift, n
    else:
        i_start, i_end = 0, n + shift
    if i_end <= i_start:
        return grad
    idx1 = np.arange(i_start, i_end)
    idx2 = idx1 - shift
    grad[idx1] += (1.0 - h[idx2])
    grad[idx2] -= h[idx1]
    return grad

def _subgradient(h, weights):
    n = len(h)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        g += w * _lag_gradient(h, d)
    return g

def _fw_direction(h, g, n):

    s = np.zeros(n)
    s[idx] = 1.0
    d = s - h
    nm = np.linalg.norm(d)
    if nm < 1e-14:
        return None
    return d / nm

    u = 1.0 - h

    a_hi = max(a_hi, 1e-12)

    best_alpha = 0.0
    top_idx = np.argsort(-raw)[:n_top]
    candidates = [0.0, a_hi]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):

            best_alpha = alpha

def _lp_direction(h, raw, n, m_lags=25, radius=0.15):
    import cvxpy as cp
    top_idx = np.argsort(-raw)[:m_lags]
    grads = []
    for r in top_idx:
        shift = int(r) - (n - 1)
        grads.append(_lag_gradient(h, shift))
    grads = np.array(grads)
    max_raw = float(np.max(raw))
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -h,
        d_var <= 1.0 - h,
        d_var >= -radius,
        d_var <= radius,
    ]
    for k in range(len(top_idx)):
        r_k = int(top_idx[k])
        constraints.append(grads[k] @ d_var - t_var <= max_raw - raw[r_k])
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver='ECOS')
    except Exception:
        try:
            prob.solve(solver='CVXOPT')
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    d_opt = d_var.value
    if d_opt is None:
        return None
    d_opt = np.real(d_opt)
    nm = np.linalg.norm(d_opt)
    if nm < 1e-14:
        return None
    return d_opt

def _make_directions(h, g, raw, n, rng, radius, use_lp=False, time_left=10.0):

    d0 = -g.copy()
    d0 -= np.mean(d0)
    nm = np.linalg.norm(d0)
    if nm > 1e-12:
        directions.append(d0 / nm)
    fw = _fw_direction(h, g, n)
    if fw is not None:
        directions.append(fw)
    for bl in [4, 8, 16]:
        nb = n // bl
        if nb < 2:
            continue
        blk = np.zeros(n)
        for b in range(nb):
            sl = slice(b * bl, min((b + 1) * bl, n))
            blk[sl] = np.mean(g[sl])
        db = -blk.copy()
        db -= np.mean(db)
        nm = np.linalg.norm(db)
        if nm > 1e-12:
            directions.append(db / nm)
    idx_pos = np.argsort(g)[-3:]
    idx_neg = np.argsort(g)[:3]
    for ip in idx_pos:
        for ino in idx_neg:
            if ip == ino:
                continue
            dp = np.zeros(n)
            dp[ip] = -1.0
            dp[ino] = 1.0
            dp -= np.mean(dp)
            nm = np.linalg.norm(dp)
            if nm > 1e-12:
                directions.append(dp / nm)
    for _ in range(2):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            dr = np.zeros(n)
            dr[i] = 1.0
            dr[j] = -1.0
            dr -= np.mean(dr)
            nm = np.linalg.norm(dr)
            if nm > 1e-12:
                directions.append(dr / nm)
    if use_lp and time_left > 2.0:
        dl = _lp_direction(h, raw, n, m_lags=20, radius=radius)
        if dl is not None:

def _binary_sa(h_bin, deadline, max_iter=60000, T_init=0.003):

            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))

                best_c5 = cur_c5
                best_h = cur.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += rng.normal(0, 1, n)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.3, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search_quad(h, d, raw, dx, radius, n_top=10)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _fast_binary_local(h_bin, deadline, n_rounds=200):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    ones_set = set(np.where(cur > 0.5)[0].tolist())
    zeros_set = set(np.where(cur <= 0.5)[0].tolist())
    stall = 0
    for _ in range(n_rounds):
        if time.time() > deadline or stall > 100:
            break
        oi = int(rng.choice(list(ones_set)))
        oz = int(rng.choice(list(zeros_set)))
        h_try = cur.copy()
        h_try[oi] = 0.0
        h_try[oz] = 1.0
        c_try = _exact_c5(h_try)
        if c_try < cur_c5 - 1e-15:
            cur = h_try
            cur_c5 = c_try
            ones_set.discard(oi)
            ones_set.add(oz)
            zeros_set.discard(oz)
            zeros_set.add(oi)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Gradient + FW optimization (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _main_optimize(h_init, d1, beta=20.0, radius=0.25, stall_limit=100, use_lp=True, lp_interval=4)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary refinement (15%)
    d2 = t0 + budget_s * 0.35
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003)
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_p = _main_optimize(h_pol, sub_dl, beta=18.0, radius=0.15, stall_limit=60, use_lp=True, lp_interval=3)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = list(range(-20, 21, 2))
    for dn in offsets:
        if time.time() > d3 - 8:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _main_optimize(h_r, sub_dl, beta=16.0, radius=0.22, stall_limit=50, use_lp=False)
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.003 and time.time() < d3 - 3:
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_r, c5_r = _main_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, beta=20.0, radius=0.12, stall_limit=40, use_lp=True, lp_interval=3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _main_optimize(h4, sub_dl, beta=30.0, radius=0.12, stall_limit=200, use_lp=True, lp_interval=3)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 3:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_cont(h4, sub_dl, n_pairs=10000, radius=0.25, seed=333)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.0015)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _main_optimize(h4, sub_dl, beta=40.0, radius=0.06, stall_limit=100, use_lp=True, lp_interval=2)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 150:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 3, time.time() + budget_s * 0.005)
        h_t, c5_t = _main_optimize(h_pert, sub_dl, beta=25.0, radius=0.10, stall_limit=40, use_lp=False)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.06)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline, time.time() + budget_s * 0.02)
        h6, c5_6 = _main_optimize(h6, sub_dl, beta=50.0, radius=0.04, stall_limit=80, use_lp=True, lp_interval=2)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline, time.time() + budget_s * 0.01)
        h6, c5_6 = _pair_swap_cont(h6, sub_dl, n_pairs=2000, radius=0.10, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h_b = _to_binary(best_h)
        sub_dl = min(deadline, time.time() + budget_s * 0.005)
        h_b, c5_b = _fast_binary_local(h_b, sub_dl, n_rounds=300)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
