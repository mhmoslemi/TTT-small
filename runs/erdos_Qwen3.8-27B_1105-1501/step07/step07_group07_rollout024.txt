                height[u] = new_h
                count_h[new_h] += 1
            if excess[u] > 1e-12:
                active.append(u)
                in_q[u] = True

    # BFS for min cut
    visited = np.zeros(N, dtype=bool)
    visited[s] = True
    q = deque([s])
    while q:
        u = q.popleft()
        for v in range(N):
            if not visited[v] and cap[u, v] > 0:
                visited[v] = True
                q.append(v)

    return excess[t], visited


def _min_cut_solve(n, w_plus, w_minus, lam):
    """Solve min-cut for binary correlation. Returns (x, flow)."""
    s = n
    t = n + 1
    N = n + 2
    cap = np.zeros((N, N), dtype=np.float64)

    # Pairwise edges: for lag d, edge i->(i+d) with cap w_plus[d], edge (i+d)->i with cap w_minus[d]
    for d in range(1, n):
        wp = w_plus[d]
        wm = w_minus[d]
        if wp > 0 or wm > 0:
            for i in range(n - d):
                j = i + d
                if wp > 0:
                    cap[i, j] += wp
                if wm > 0:
                    cap[j, i] += wm

    # Unary edges
    if lam > 0:
        cap[:, t] += lam  # edge i->t for all i
        cap[s, t] = 0  # don't add s->t
        cap[:, t] = np.where(np.arange(N) < n, lam, 0.0)
    elif lam < 0:
        cap[s, :] = 0.0
        cap[s, :n] = -lam

    flow, visited = _push_relabel_cap(cap, s, t)
    x = np.zeros(n)
    x[visited[:n]] = 1.0
    return x, flow


def _min_cut_card(n, w_plus, w_minus, target):
    """Find binary solution with approximately target ones via bisection on lambda."""
    lo, hi = -10.0, 10.0
    # Find bounds
    x_test, _ = _min_cut_solve(n, w_plus, w_minus, lo)
    s_lo = np.sum(x_test)
    x_test, _ = _min_cut_solve(n, w_plus, w_minus, hi)
    s_hi = np.sum(x_test)

    if s_lo <= target <= s_hi:
        pass
    elif s_lo > target:
        for _ in range(30):
            lo /= 2.0
            x_test, _ = _min_cut_solve(n, w_plus, w_minus, lo)
            if np.sum(x_test) <= target:
                break
    else:
        for _ in range(30):
            hi *= 2.0
            x_test, _ = _min_cut_solve(n, w_plus, w_minus, hi)
            if np.sum(x_test) >= target:
                break

    # Bisection
    x_lo, _ = _min_cut_solve(n, w_plus, w_minus, lo)
    x_hi, _ = _min_cut_solve(n, w_plus, w_minus, hi)
    for _ in range(30):
        mid = (lo + hi) / 2.0
        x_mid, _ = _min_cut_solve(n, w_plus, w_minus, mid)
        s_mid = np.sum(x_mid)
        if s_mid > target:
            lo = mid
            x_lo = x_mid
        else:
            hi = mid
            x_hi = x_mid

    # Pick closest to target
    if abs(np.sum(x_lo) - target) <= abs(np.sum(x_hi) - target):
        return x_lo
    else:
        return x_hi


def _mc_refine(h_bin, deadline, n_iter=200, seed=42):
    """Iterative min-cut refinement with adaptive weights."""
    n = len(h_bin)
    target = n / 2.0
    x = h_bin.copy()
    cur_c5 = _exact_c5(x)
    best_x = x.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)

    for it in range(n_iter):
        if time.time() > deadline:
            break
        scores = _scores(x)
        mid = n - 1
        # Extract R_d for positive and negative lags
        w_plus = np.zeros(n)
        w_minus = np.zeros(n)
        for d in range(1, n):
            w_plus[d] = scores[mid + d]
            w_minus[d] = scores[mid - d]

        # Adaptive weight boost on worst lags
        max_p = w_plus.max() if n > 1 else 1.0
        max_m = w_minus.max() if n > 1 else 1.0
        max_s = max(max_p, max_m, 1e-10)
        boost = 3.0
        w_plus[w_plus > 0.5 * max_s] *= boost
        w_minus[w_minus > 0.5 * max_s] *= boost
        # Add small uniform to avoid degeneracy
        w_plus[1:] += 0.01
        w_minus[1:] += 0.01

        x_new = _min_cut_card(n, w_plus, w_minus, target)
        new_c5 = _exact_c5(x_new)
        if new_c5 < cur_c5 - 1e-12:
            x = x_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_x = x.copy()
        else:
            # Try with different weight scheme: focus on top-k worst lags
            k = 5
            top_p = np.argsort(w_plus)[::-1][:k]
            top_m = np.argsort(w_minus)[::-1][:k]
            w2p = np.ones(n) * 0.1
            w2m = np.ones(n) * 0.1
            for d in top_p:
                w2p[d] = 5.0
            for d in top_m:
                w2m[d] = 5.0
            x_new2 = _min_cut_card(n, w2p, w2m, target)
            new_c5_2 = _exact_c5(x_new2)
            if new_c5_2 < cur_c5 - 1e-12:
                x = x_new2
                cur_c5 = new_c5_2
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_x = x.copy()
            else:
                # Random restart with perturbation
                if it % 20 == 19:
                    x_pert = x.copy()
                    for _ in range(10):
                        i, j = rng.choice(n, 2, replace=False)
                        x_pert[i], x_pert[j] = x_pert[j], x_pert[i]
                    c5_pert = _exact_c5(x_pert)
                    if c5_pert < cur_c5:
                        x = x_pert
                        cur_c5 = c5_pert

    return best_x, best_c5


def _pair_swap_binary(h_bin, deadline, n_swaps=20000, seed=123):
    """Local search by pair swaps on binary vectors."""
    n = len(h_bin)
    x = h_bin.copy()
    cur_c5 = _exact_c5(x)
    best_x = x.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(x > 0.5)[0])
    zeros = list(np.where(x <= 0.5)[0])
    stall = 0

    for _ in range(n_swaps):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        x[i], x[j] = x[j], x[i]
        new_c5 = _exact_c5(x)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_x = x.copy()
            stall = 0
        else:
            x[i], x[j] = x[j], x[i]
            stall += 1
    return best_x, best_c5


def _greedy_construct(n, seed=0):
    """Greedy construction: place ones to minimize max R_d."""
    x = np.zeros(n)
    rng = np.random.default_rng(seed)
    candidates = list(range(n))
    for _ in range(n // 2):
        best_pos = -1
        best_max = float('inf')
        for pos in candidates:
            x[pos] = 1.0
            scores = _scores(x)
            m = float(np.max(scores))
            x[pos] = 0.0
            if m < best_max:
                best_max = m
                best_pos = pos
        x[best_pos] = 1.0
        candidates.remove(best_pos)
    return x


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _to_binary(h_new)


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G


def _line_search_c5(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m])
                C[r] = np.sum(d[:m] * d[lag:lag + m])
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m])
                C[r] = np.sum(d[g:] * d[:m])
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 30
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.array([float(np.max(S0 + t * T - t * t * C)) * dx for t in grid_t])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx))]
    return min(cands, key=lambda z: z[1])


def _lp_direction(h, scores, G, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    top_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores - 1e-6
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _focused_lp_loop(h, deadline, radius_init=0.1, top_k_init=4, max_top_k=10, stall_limit=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k)
        if p is None or np.max(np.abs(p)) < 1e-12:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search_c5(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = _project_sum(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.03, 0.2)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.8
    return best_h, best_c5


def _perturb_binary(h_bin, rng, n_flips=6):
    x = h_bin.copy()
    n = len(x)
    ones = np.where(x > 0.5)[0]
    zeros = np.where(x <= 0.5)[0]
    for _ in range(n_flips):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        x[i], x[j] = x[j], x[i]
    return x


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.zeros(180)
        h0[:90] = 1.0

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = _resample(h0, (n0 // 2) * 2)
        n0 = len(h0)

    h_bin = _to_binary(h0)
    best_h = h_bin.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: Min-cut global optimization (35% of budget)
    d1 = t0 + budget_s * 0.35
    # Multiple weight strategies
    strategies = [
        ("uniform", 0.0),
        ("boost_worst", 1.0),
        ("top_k", 2.0),
    ]
    for name, _ in strategies:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.08)
        h_mc, c5_mc = _mc_refine(best_h, sub_dl, n_iter=150, seed=seed + 1)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()

    # Phase 2: Local search refinement (20%)
    d2 = t0 + budget_s * 0.55
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    h_ls, c5_ls = _pair_swap_binary(best_h, sub_dl, n_swaps=15000, seed=seed + 10)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.75
    n_best = len(best_h)
    tried = set()
    for dn in range(-20, 21, 4):
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 240 or n_try in tried or n_try % 2 != 0:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h_mc2, c5_mc2 = _mc_refine(h_r, sub_dl, n_iter=80, seed=n_try)
        if time.time() < d3 - budget_s * 0.02:
            sub_dl2 = min(d3, time.time() + budget_s * 0.015)
            h_ls2, c5_ls2 = _pair_swap_binary(h_mc2, sub_dl2, n_swaps=5000, seed=n_try + 500)
            if c5_ls2 < c5_mc2:
                c5_mc2 = c5_ls2
                h_mc2 = h_ls2
        if c5_mc2 < best_c5:
            best_c5 = c5_mc2
            best_h = h_mc2.copy()
            n_best = n_try

    # Phase 4: Continuous polish (15%)
    d4 = t0 + budget_s * 0.90
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_cont = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h_lp, c5_lp = _focused_lp_loop(h_cont, sub_dl, radius_init=0.05, top_k_init=3, max_top_k=8, stall_limit=40)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    if time.time() < d4:
        h_bin2 = _to_binary(best_h)
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_mc3, c5_mc3 = _mc_refine(h_bin2, sub_dl2, n_iter=60, seed=seed + 999)
        if c5_mc3 < best_c5:
            best_c5 = c5_mc3
            best_h = h_mc3.copy()

    # Phase 5: Basin hopping + final refinement (10%)
    d5 = deadline
    scale = 4
    n_restarts = 0
    while time.time() < d5 - 10 and n_restarts < 100:
        n_restarts += 1
        h_pert = _perturb_binary(best_h, rng, n_flips=scale)
        sub_dl = min(d5 - 10, time.time() + budget_s * 0.008)
        h_t, c5_t = _mc_refine(h_pert, sub_dl, n_iter=40, seed=seed + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(2, int(scale * 0.8))
        else:
            scale = min(15, scale + 1)

    # Final local search
    if deadline - time.time() > 5:
        h_final, c5_final = _pair_swap_binary(best_h, deadline, n_swaps=5000, seed=seed + 7777)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ensure validity
    best_h = _to_binary(best_h)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
