def _project(v, target):

def _lp_direction(h, c, active_indices, radius):

    n = len(h)
    dx = 2.0 / n
    m = len(active_indices)
    
    # Build gradient rows for active constraints
    # grad[r, i] = partial F_r / partial h[i] * dx
    grad = np.zeros((m, n))
    for idx, r in enumerate(active_indices):
        k = r - (n - 1)
        if k > 0:
            grad[idx, k:] += (1.0 - h[:n-k])
            grad[idx, :n-k] -= h[k:n]
        elif k < 0:

            grad[idx, :n-u] += (1.0 - h[u:n])
            grad[idx, u:] -= h[:n-u]
        else:
            grad[idx, :] = 1.0 - 2.0 * h
        grad[idx] *= dx

    #   For each active r: c[r]*dx + grad[r]·d ≤ t  (i.e., grad[r]·d - t ≤ -c[r]*dx)
    #   d_i ≤ min(radius, 1-h_i)

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb


def _compute_AB(h):
    n = len(h)
    A = np.zeros((n, n - 1))
    B = np.zeros((n, n - 1))
    for p in range(n):
        if p > 0:
            A[p, :p] += 1.0 - h[p - 1::-1]
            B[p, :p] -= h[p - 1::-1]
        if p < n - 1:
            A[p, :n - 1 - p] -= h[p + 1:n]
            B[p, :n - 1 - p] += 1.0 - h[p + 1:n]
    return A, B


def _exhaustive_binary_search(h_bin, deadline, max_sweeps=200, perturb=False, rng=None):
    n = len(h_bin)
    dx = 2.0 / n
    h_bin = h_bin.copy()
    if perturb and rng is not None:
        for _ in range(int(n * 0.05)):
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = int(rng.integers(0, len(ones)))
                j = int(rng.integers(0, len(zeros)))
                h_bin[ones[i]] = 0.0
                h_bin[zeros[j]] = 1.0

    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    best_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A, B = _compute_AB(h_bin)
        c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        c_pos = c[n:]
        c_neg = c[:n - 1]

        oi_grid, oz_grid = np.meshgrid(ones, zeros, indexing='ij')
        delta_pos = A[oz_grid] - A[oi_grid]
        delta_neg = B[oz_grid] - B[oi_grid]

        new_c_pos = c_pos[None, None, :] + delta_pos
        new_c_neg = c_neg[None, None, :] + delta_neg
        new_max = np.maximum(new_c_pos, new_c_neg).max(axis=2) * dx

        min_flat = int(np.argmin(new_max))
        min_val = float(new_max.flat[min_flat])

        if min_val < best_c5 - 1e-12:
            ii, jj = np.unravel_index(min_flat, new_max.shape)
            oi = int(ones[ii])
            oz = int(zeros[jj])
            h_bin[oi] = 0.0
            h_bin[oz] = 1.0
            best_c5 = min_val
            best_h = h_bin.copy()
        else:
            break

    return best_h, best_c5


def _line_search(h, d, dx, radius):
    n = len(h)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')

    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(0.0, t_hi)

    if t_hi < 1e-14:
        return 0.0, float(np.max(c0) * dx)

    def eval_f(alpha):
        return float(np.max(c0 + alpha * L1 - alpha * alpha * Q2) * dx)

    grid = np.linspace(0.0, t_hi, 15)
    grid_f = np.array([eval_f(t) for t in grid])
    bi = int(np.argmin(grid_f))
    lo = grid[max(0, bi - 1)]
    hi = grid[min(len(grid) - 1, bi + 1)]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = eval_f(c1)
    f2 = eval_f(c2)
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_f(c2)

    opt_t = (lo + hi) / 2.0
    opt_val = eval_f(opt_t)
    f0 = eval_f(0.0)
    f_hi = eval_f(t_hi)
    best_t, best_v = min([(0.0, f0), (opt_t, opt_val), (t_hi, f_hi)], key=lambda x: x[1])
    return best_t, best_v


def _grad_matrix(h, r_idx, dx):
    n = len(h)
    g = np.zeros(n)
    k = r_idx - (n - 1)
    if k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:n]
    elif k < 0:
        u = -k
        g[:n - u] += (1.0 - h[u:n])
        g[u:] -= h[:n - u]
    else:
        g[:] = 1.0 - 2.0 * h
    g *= dx
    return g


def _lp_optimize(h, deadline, radius_init=0.08, max_iter=2000, active_k=30, seed=42):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(seed)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 40:
            break

        c = np.correlate(h, 1.0 - h, mode='full')
        c_s = c * dx
        max_val = float(np.max(c_s))

        tol = max(1e-10, max_val * 0.05)
        active = np.where(c_s >= max_val - tol)[0]
        if len(active) > active_k:
            order = np.argsort(-c_s[active])[:active_k]
            active = active[order]

        m = len(active)
        grad = np.array([_grad_matrix(h, int(r), dx) for r in active])

        d_lo = -np.minimum(radius, h)
        d_hi = np.minimum(radius, 1.0 - h)
        t_hi = max_val + 1e-6

        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        A_ub[:, :n] = grad
        A_ub[:, n] = -1.0
        b_ub = -c_s[active]

        bounds = list(zip(d_lo, d_hi)) + [(-2.0, t_hi)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        d = None
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and np.max(np.abs(res.x[:n])) > 1e

14:
                d = res.x[:n].copy()
                d -= np.mean(d)
        except Exception:
            pass

        if d is None:
            g_avg = np.mean(grad, axis=0)
            d = -g_avg
            d -= np.mean(d)
            d = np.clip(d, -radius, radius)
            d -= np.mean(d)
            if np.max(np.abs(d)) < 1e-14:
                radius *= 0.5
                if radius < 1e-6:
                    break
                stall += 1
                continue

        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.25)
        else:
            stall += 1
            radius *= 0.7

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert *= amp
    pert += amp * 0.3 * rng.standard_normal(n)
    return _project(h + pert, n / 2.0)

        h_ps, c5_ps = _pair_swap_continue(h_ps, sub_dl, max_iters=8000, radius=0.12, seed=3000)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
            n_curr = len(best_h)

    # More basin hopping
    scale = 0.04
    for _ in range(30):
        if time.time() > d3:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        h_pb = _to_binary(h_pert)
        sub_dl = min(d3, time.time() + 10)
        h_e, c5_e = _exhaustive_binary_search(h_pb, sub_dl, max_sweeps=300)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e.copy()
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 4: Final intense search
    d4 = deadline - 5
    n_fin = len(best_h)

    # LP with smaller radius
    h_f = _project(best_h.copy(), n_fin / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h_f, c5_f = _lp_optimize(h_f, sub_dl, radius_init=0.03, max_iter=1500, active_k=40,


                             seed=4000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final binary exhaustive
    h_fb = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h_e, c5_e = _exhaustive_binary_search(h_fb, sub_dl, max_sweeps=1000)
    if c5_e < best_c5:
        best_c5 = c5_e
        best_h = h_e.copy()

    # Final pair swaps
    if time.time() < d4 - 5:
        h_fs = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d4 - 2, time.time() + budget_s * 0.03)
        h_fs, c5_fs = _pair_swap_continue(h_fs, sub_dl, max_iters=6000, radius=0.08, seed=5000)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Final LP polish
    if time.time() < d4 - 3:
        h_fp = _project(best_h.copy(), len(best_h) / 2.0)
        h_fp, c5_fp = _lp_optimize(h_fp, d4 - 1, radius_init=0.02, max_iter=800, active_k=50,
                                   seed=6000)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final binary from latest
    h_fb2 = _to_binary(best_h)
    if time.time() < deadline - 2:
        h_e2, c5_e2 = _exhaustive_binary_search(h_fb2, deadline - 1, max_sweeps=500)
        if c5_e2 < best_c5:
            best_c5 = c5_e2
            best_h = h_e2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
