def _line_search(h, d, dx, radius):
    """Polynomial line search for the quadratic objective along direction d."""
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    # score(t) = max_k (S0[k] + t*T[k] - t^2*C[k])
    
    # Find valid range for t
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_lo = 0.0
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    def f_of_t(t):
        return float(np.max(S0 + t * T - t * t * C)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = t_lo
    hi_b = t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    for _ in range(60):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, f2 = c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = f_of_t(c1)
        else:
            lo_b, f1 = c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = f_of_t(c2)
    
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = f_of_t(opt_t)
    
    # Check boundaries
    cands = [(opt_t, opt_val), (t_lo, f_of_t(t_lo)), (t_hi, f_of_t(t_hi))]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _full_lp(h, scores, radius):
    """Full-lag LP: minimize tau subject to all score constraints."""
    n = len(h)
    all_lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, all_lags)
    m = len(scores)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores  # G@dv - tau <= -scores[k] for all k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', 
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgradient_step(h, scores, step_size):
    """Compute subgradient direction for the max function."""
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - 1e-10)[0]
    
    # Average gradient over active set
    g = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        u = 1.0 - h
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:n] += u[:m]
                g[:m] += -h[d:d+m]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += u[gg:gg+m]
                g[gg:n] += -h[:m]
    g /= len(active)
    
    # Remove mean to preserve sum constraint
    g -= np.mean(g)
    
    # Scale
    g *= step_size
    
    # Enforce bounds
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / max(step_size, 1e-6))
    g[neg] = np.maximum(g[neg], -h[neg] / max(step_size, 1e-6))
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g -= np.mean(g)
    
    return g

def _pair_sweep(h, deadline, max_rounds=200):

    n = len(h)

    dx = 2.0 / n

    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_tries = min(len(ones) + 5, 50)
        for _ in range(n_tries):
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))

            d[zeros[i]] = 1.0

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.004):

            ones_idx[i] = oz

                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amplitudes = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _direct_binary_descent(h_bin, deadline, max_iters=10000):
    """Direct binary descent: try all promising swaps greedily."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99)
    
    ones_idx = np.where(cur > 0.5)[0]
    zeros_idx = np.where(cur <= 0.5)[0]
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Try a few random swaps, keep the best
        best_swap = None
        best_swap_c5 = cur_c5
        n_tries = min(30, len(ones_idx) * len(zeros_idx))
        
        for _ in range(n_tries):
            i = int(rng.integers(0, len(ones_idx)))
            j = int(rng.integers(0, len(zeros_idx)))
            h_new = cur.copy()
            h_new[ones_idx[i]] = 0.0
            h_new[zeros_idx[j]] = 1.0
            c_try = _exact_c5(h_new)
            if c_try < best_swap_c5:
                best_swap_c5 = c_try
                best_swap = (ones_idx[i], zeros_idx[j], h_new)
        
        if best_swap is not None:
            oi, oz, h_new = best_swap
            cur = h_new
            cur_c5 = best_swap_c5
            # Update indices

            ones_idx = np.append(ones_idx, oz)
            ones_idx.sort()
            zeros_idx = zeros_idx[zeros_idx != oz]
            zeros_idx = np.append(zeros_idx, oi)
            zeros_idx.sort()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            break
    
    return best_h, best_c5
