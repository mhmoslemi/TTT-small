import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h, active_idx):
    """Compute gradient rows for active correlation lags.
    active_idx: array of indices into the full correlation (length 2n-1)
    Returns: array of shape (len(active_idx), n) with dx-scaled gradients
    """
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    m = len(active_idx)
    G = np.zeros((m, n))
    for row in range(m):
        r = int(active_idx[row])
        k = r - center  # shift (can be negative, zero, or positive)
        if k == 0:
            G[row] = (1.0 - 2.0 * h) * dx
        elif k > 0:
            # F_k = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
            # grad[i] = (1-h[i-k]) for i in [k, n-1] minus h[i+k] for i in [0, n-k-1]
            G[row, k:] += (1.0 - h[:n-k])
            G[row, :n-k] -= h[k:]
            G[row] *= dx
        else:
            u = -k  # u > 0
            # F_{-u} = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])
            # grad[i] = (1-h[i+u]) for i in [0, n-u-1] minus h[i-u] for i in [u, n-1]
            G[row, :n-u] += (1.0 - h[u:])
            G[row, u:] -= h[:n-u]
            G[row] *= dx
    return G

def _line_search(h, d, dx, alpha_max):
    """Exact line search for h + alpha*d. Returns (best_alpha, best_c5)."""
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    def c5_at(alpha):
        vals = C0 + alpha * L1 - alpha * alpha * Q2
        return float(np.max(vals) * dx)
    
    # Grid search
    n_grid = 40
    grid = np.linspace(0.0, alpha_max, n_grid)
    grid_vals = np.array([c5_at(t) for t in grid])
    best_local = int(np.argmin(grid_vals))
    
    # Golden section refinement
    lo = grid[max(0, best_local - 2)]
    hi = grid[min(n_grid - 1, best_local + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-12:
        opt_alpha = (lo + hi) / 2.0
    else:
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = c5_at(c1)
        f2 = c5_at(c2)
        for _ in range(50):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = c5_at(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = c5_at(c2)
        opt_alpha = (lo + hi) / 2.0
    
    opt_val = c5_at(opt_alpha)
    # Also check 0 and alpha_max
    v0 = c5_at(0.0)
    v_max = c5_at(alpha_max)
    candidates = [(0.0, v0), (opt_alpha, opt_val), (alpha_max, v_max)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

    m = len(active_lags)

    corr = np.correlate(h, 1.0 - h, mode='full')
    z = dx * corr  # scaled values

    G = _grad_matrix(h, active_lags)  # shape (m, n)

    # Variables: d[0..n-1], u[0..n-1], t
    # Total: 2n + 1 variables
    n_var = 2 * n + 1
    t_idx = 2 * n  # index of t
    
    # Objective: [0...0, eps...eps, 1]
    eps = 1e-4
    c_obj = np.zeros(n_var)
    c_obj[n:2*n] = eps  # regularization on |d| via u
    c_obj[t_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag r: sum_i G[r,i] * d_i <= t - z[r]
    # i.e., sum_i G[r,i] * d_i - t <= -z[r]
    
    # Also: u_i >= |d_i| means d_i - u_i <= 0 and -d_i - u_i <= 0
    
    # Build A_ub
    # Rows: m (lag constraints) + 2n (bounds on u)
    n_rows = m + 2 * n
    A_ub = np.zeros((n_rows, n_var))
    b_ub = np.zeros(n_rows)

def _get_active_lags(h, tol=0.005, max_count=60):
    """Get indices of active (near-maximum) correlation lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = dx * corr
    z_max = float(np.max(z))
    # Active: within tol of max
    active = np.where(z >= z_max - tol)[0]
    if len(active) > max_count:
        # Take the top max_count
        order = np.argsort(-z[active])
        active = active[order[:max_count]]
    return active, z, z_max
