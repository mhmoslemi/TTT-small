    lo = float(v.min()) - 1.0

def _lse_obj_grad(h, beta):
    """Compute LSE objective value and gradient at h with smoothing beta."""
    n = len(h)
    dx = 2.0 / n
    h_eval = np.clip(h, 0.0, 1.0)
    scores = np.correlate(h_eval, 1.0 - h_eval, mode='full') * dx
    m = float(np.max(scores))
    z = (scores - m) / beta
    # Clamp z to avoid overflow
    z = np.clip(z, -500.0, 500.0)
    exp_z = np.exp(z)
    sum_exp = float(np.sum(exp_z))
    if sum_exp < 1e-300:
        value = m
        weights = np.zeros(len(scores))
        weights[int(np.argmax(scores))] = 1.0
    else:
        value = m + beta * np.log(sum_exp)
        weights = exp_z / sum_exp
    # Compute weighted gradient

                break

                if f_try < f_val - 1e-4 * alpha * np.dot(f_grad, h_try - h):

            f_grad = g_try
