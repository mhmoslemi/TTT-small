def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):

    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)

def _perturb_smooth(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.standard_normal(n)
    return _project(h + perturb, n / 2.0)

def _perturb_jumpy(h, adjacent_swaps, rng):
    """Flip a contiguous block of binary-like values."""
    n = len(h)
    h_new = h.copy()

        # Swap values at i and i+1
        h_new[i], h_new[i + 1] = h_new[i + 1], h_new[i]
    return _project(h_new, n / 2.0)

            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.001 and time.time() < d3 - 2:
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_r, c5_r = _optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, beta=18.0, radius=0.10, stall_limit=30, seed=400 + n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive iterative refinement (25%)
    d4 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    h4, c5_4 = _iterative_refine(best_h, d4, n_cycles=4)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair swap continuous (10%)
    d5 = t0 + budget_s * 0.80
    h5 = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5, c5_5 = _pair_swap_cont(h5, sub_dl, n_pairs=10000, radius=0.25, seed=555)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # More gradient after pair swaps
    if time.time() < d5 - 2:
        h5 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h5, c5_5 = _optimize(h5, sub_dl, beta=30.0, radius=0.10, stall_limit=60, seed=666)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Basin hopping (15%)
    d6 = deadline
    scale = 0.04
    n_restarts = 0
    while time.time() < d6 - 5 and n_restarts < 150:
        n_restarts += 1
        if n_restarts % 5 == 0:
            h_pert = _perturb_jumpy(_to_binary(best_h), rng.integers(1, 5), rng)
        else:
            h_pert = _perturb_smooth(best_h, scale, rng)
        sub_dl = min(d6 - 3, time.time() + budget_s * 0.007)
        h_t, c5_t = _optimize(h_pert, sub_dl, beta=22.0, radius=0.14, stall_limit=35, seed=10000 + n_restarts)
        if time.time() < d6 - 2:
            h_tb = _to_binary(h_t)
            sub_dl2 = min(d6 - 2, time.time() + budget_s * 0.003)
            h_tb, c5_tb = _pair_sweep(h_tb, sub_dl2)
            if c5_tb < c5_t:
                c5_t = c5_tb
                h_t = h_tb
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.10, scale * 1.06)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _optimize(h6, deadline, beta=35.0, radius=0.06, stall_limit=60, seed=7777)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=2000, radius=0.10, seed=8888)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
