    max_val = float(corr.max())
    threshold = max_val - max(0.03 * max_val, 2e-4)
    active = np.where(corr >= threshold)[0]
    if len(active) > max_active:
        active = np.argsort(-corr[:2*n-1])[
            np.argsort(-corr)[:max_active]]
        active = np.where(corr >= threshold)[0][:max_active]
    m = len(active)
    if m == 0:
        return None

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, r in enumerate(active):
        k = int(r) - (n - 1)
        g = np.zeros(n)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            u = -k
            g[:n - u] += (1.0 - h[u:])
            g[u:] -= h[:n - u]
        g *= dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -float(corr[r]) * dx

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        lo_i = -min(radius, h[i] + 1e-12)
        hi_i = min(radius, 1.0 - h[i] + 1e-12)
        bounds.append((lo_i, hi_i))
    bounds.append((-100.0, 100.0))

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    c_obj[:n] = 1e-9

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            d -= np.mean(d)
            if float(np.max(np.abs(d))) > 1e-10:
                return d
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.08, max_active=50, stall_limit=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(5000):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _lp_direction(h, corr, radius, max_active)
        if d is None:
            g = _subgradient(h, corr)
            d = -g - np.mean(-g)
        if float(np.max(np.abs(d))) < 1e-12:
            stall += 2
            radius *= 0.7
            continue
        t_hi = _max_feasible_alpha(h, d)
        if t_hi < 1e-10:
            stall += 1
            radius *= 0.7
            continue
        alpha, new_c5 = _line_search(h, d, t_hi)
        if new_c5 < cur - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur - 1e-13:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.25)
            else:
                stall += 1
                radius *= 0.75
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _subgrad_optimize(h, deadline, radius_init=0.10, stall_limit=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(5000):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = _subgradient(h, corr)
        d = -g
        d -= np.mean(d)
        if float(np.max(np.abs(d))) < 1e-12:
            break
        t_hi = _max_feasible_alpha(h, d)
        if t_hi < 1e-10:
            break
        alpha, new_c5 = _line_search(h, d, t_hi)
        if new_c5 < cur - 1e-12:

            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur - 1e-13:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.03, 0.20)
            else:
                stall += 1
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _binary_c5(hb):
    n = len(hb)
    dx = 2.0 / n
    corr = np.correlate(hb, 1.0 - hb, mode='full')
    return float(np.max(corr)) * dx, corr

def _binary_search(hb, deadline, seed=42, max_iters=200000, temp_init=0.004):
    n = len(hb)
    cur = hb.copy()
    cur_c5, cur_corr = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5, _ = _binary_c5(trial)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0003, T * 0.9995)
        else:
            stall += 1
            best_stall += 1
    return best_h, best_c5

def _targeted_binary(hb, deadline, seed=99, max_iters=60000):
    n = len(hb)
    cur = hb.copy()
    cur_c5, cur_corr = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 4000:
            break
        max_idx = int(np.argmax(cur_corr))
        best_delta = 0.0
        best_move = None
        n_try = min(150, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5, _ = _binary_c5(trial)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (oi, oz, i, j)
        if best_move is not None and best_delta < -1e-14:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            _, cur_corr = _binary_c5(cur)
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swap(h, deadline, n_tries=5000, radius=0.20, seed=111):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        t_hi = _max_feasible_alpha(h, d)
        if t_hi < 1e-10:
            stall += 1
            continue
        t_hi = min(t_hi, radius)
        alpha, new_c5 = _line_search(h, d, t_hi)
        if new_c5 < cur - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur - 1e-13:
                h = h_new
                cur = actual
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _block_swap(hb, deadline, seed=77, max_iters=10000):
    n = len(hb)
    cur = hb.copy()
    cur_c5 = _binary_c5(cur)[0]
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        length = int(rng.integers(2, min(6, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        b1 = cur[i:i+length].copy()
        b2 = cur[j:j+length].copy()
        if np.sum(b1) != np.sum(b2):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i+length] = b2
        trial[j:j+length] = b1
        new_c5 = _binary_c5(trial)[0]
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 4.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    perturb = np.zeros(n)
    for f, p in zip(freqs, phases):
        perturb += np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 180
        h0 = np.full(n0, 0.5)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization (20% budget)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lp_optimize(h_init, d1, radius_init=0.08, max_active=50, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Subgradient polish
    if time.time() < d1 + 10:
        d1b = min(t0 + budget_s * 0.25, deadline)
        h1b, c5_1b = _subgrad_optimize(best_h, d1b, radius_init=0.05, stall_limit=60)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Binary search (35% budget)
    d2 = t0 + budget_s * 0.55
    hb = _to_binary(best_h, n0)
    c5_bin = _binary_c5(hb)[0]
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = hb.copy()

    # Targeted binary
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    hb2, c5_2 = _targeted_binary(hb, sub_dl, seed=101, max_iters=80000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = hb2.copy()

    # Simulated annealing
    sub_dl = min(d2, time.time() + budget_s * 0.10)
    hb2, c5_2 = _binary_search(hb2, sub_dl, seed=202, max_iters=150000, temp_init=0.003)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = hb2.copy()

    # Block swaps
    sub_dl = min(d2, time.time() + budget_s * 0.06)
    hb2, c5_2 = _block_swap(hb2, sub_dl, seed=303, max_iters=12000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = hb2.copy()

    # More targeted
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    hb2, c5_2 = _targeted_binary(hb2, sub_dl, seed=404, max_iters=50000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = hb2.copy()

    # LP polish on best
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    hpol = _project(best_h.copy(), n0 / 2.0)
    hpol, c5_pol = _lp_optimize(hpol, sub_dl, radius_init=0.05, max_active=50, stall_limit=40)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = hpol.copy()

    # Phase 3: Multi-n exploration (20% budget)
    d3 = t0 + budget_s * 0.75
    n_best = len(best_h)
    tried = set()
    n_offsets = sorted(range(-30, 31, 3), key=lambda x: abs(x))
    for dn in n_offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.06, max_active=40, stall_limit=25)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.005)
            h_b, c5_b = _targeted_binary(h_b, sub_dl2, seed=1000 + n_try, max_iters=20000)
            sub_dl3 = min(d3, time.time() + budget_s * 0.004)
            h_b, c5_b = _binary_search(h_b, sub_dl3, seed=2000 + n_try, max_iters=30000, temp_init=0.003)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Final refinement (25% budget)
    d4 = deadline
    n_fin = len(best_h)

    # LP polish
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = _project(best_h.copy(), n_fin / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, radius_init=0.04, max_active=60, stall_limit=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Pair swap
    if time.time() < d4 - 15:
        h4 = _project(best_h.copy(), n_fin / 2.0)
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_swap(h4, sub_dl, n_tries=6000, radius=0.20, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Binary targeted on best
    if time.time() < d4 - 20:
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.05)
        h_b, c5_b = _targeted_binary(h_b, sub_dl, seed=777, max_iters=80000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Binary SA
    if time.time() < d4 - 15:
        h_b = _to_binary(best_h, n_fin)
        sub_dl = min(d4 - 3, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_search(h_b, sub_dl, seed=888, max_iters=80000, temp_init=0.002)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Basin hopping
    if time.time() < d4 - 10:
        scale = 0.015
        for _ in range(40):
            if time.time() > d4 - 4:
                break
            h_p = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4 - 2, time.time() + budget_s * 0.003)
            h_t, c5_t = _lp_optimize(h_p, sub_dl, radius_init=0.04, max_active=30, stall_limit=15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.05, scale * 1.1)

    # Final polish
    if time.time() < d4 - 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, d4 - 1, radius_init=0.02, max_active=40, stall_limit=20)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final project and evaluate
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
