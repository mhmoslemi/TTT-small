    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_active(h, raw, idx, weights):
    n = len(h)
    n1 = n - 1
    grad = np.zeros(n)
    for ai in range(len(idx)):
        k = int(idx[ai])
        d = k - n1
        w = weights[ai]
        if w < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:])
                grad[d:] += -w * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += w * (1.0 - h[:m])
                grad[:m] += -w * h[g:]
    return grad

def _line_search(raw, T, Q, dx, a_max):
    if a_max < 1e-15:
        return 0.0, float(np.max(raw)) * dx
    n_g = 32
    alphas = np.linspace(0.0, a_max, n_g)
    best_a, best_v = 0.0, float(np.max(raw)) * dx
    for a in alphas:
        v = float(np.max(raw + a * T - a * a * Q)) * dx
        if v < best_v:
            best_v, best_a = v, a
    if best_a > 0 and best_a < a_max:
        lo = max(0.0, best_a * 0.5)
        hi = min(a_max, best_a * 1.5)
        if hi - lo > 1e-14:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            for _ in range(35):
                if hi - lo < 1e-12 * max(1.0, a_max):
                    break
                if f1 < f2:
                    hi, c2, f2 = c1, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            opt = (lo + hi) / 2.0
            f_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
            if f_opt < best_v:
                best_a, best_v = opt, f_opt
    v1 = float(np.max(raw + a_max * T - a_max * a_max * Q)) * dx
    if v1 < best_v:
        best_a, best_v = a_max, v1
    return best_a, best_v

def _try_dir(h, raw, p, dx):
    n = len(h)
    pos = p > 1e-15
    neg = p < -1e-15
    a_max = 1.0
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-p[neg]))))
    if a_max < 1e-14:
        return h, _c5_exact(h)
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    Q = np.correlate(p, p, mode='full')
    a, val = _line_search(raw, T, Q, dx, a_max)
    cur_c5 = float(np.max(raw)) * dx
    if val < cur_c5 - 1e-14 and a > 1e-14:
        h_new = h + a * p
        h_new = _project(h_new, n / 2.0)
        c5_new = _c5_exact(h_new)
        if c5_new < cur_c5 - 1e-14:
            return h_new, c5_new
    return h, cur_c5

def _optimize_grad(h, deadline, seed=42):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(raw)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    tau = 0.8
    top_k = 10
    stall = 0
    max_stall = 50

    while time.time() < deadline:
        raw = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(raw)) * dx

        if stall > 25:
            top_k = int(rng.integers(4, min(30, 2 * n - 1)))

        idx = np.argsort(raw)[-top_k:]
        vals = raw[idx]
        vmax = float(vals.max())
        exps = np.exp((vals - vmax) / tau)
        weights = exps / exps.sum()

        grad = _grad_active(h, raw, idx, weights)
        p = float(grad.mean()) - grad

        h_new, c5_new = _try_dir(h, raw, p, dx)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue

        # Try a few pair directions
        n1 = n - 1
        k_max = int(np.argmax(raw))
        d_max = k_max - n1
        if d_max >= 0:
            m = n - d_max
            press = np.zeros(n)
            press[:m] += (1.0 - h[d_max:])
            press[d_max:] += -h[:m]
        else:
            g = -d_max
            m = n - g
            press = np.zeros(n)
            press[g:] += (1.0 - h[:m])
            press[:m] += -h[g:]

        order = np.argsort(press)
        n_test = max(6, n // 15)
        lows = order[:n_test]
        highs = order[-n_test:]
        improved = False
        for _ in range(8):
            if time.time() > deadline:
                break
            pi = int(lows[int(rng.integers(0, len(lows)))])
            pj = int(highs[int(rng.integers(0, len(highs)))])
            if pi == pj:
                continue
            p_pair = np.zeros(n)
            if h[pi] >= h[pj]:
                p_pair[pi] = -1.0
                p_pair[pj] = 1.0
            else:
                p_pair[pi] = 1.0
                p_pair[pj] = -1.0
            h_new, c5_new = _try_dir(h, raw, p_pair, dx)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break

        if improved:
            stall = 0
            continue

        # Vertex direction
        h_bin_idx = np.argsort(h)[-n // 2:]
        h_bin = np.zeros(n)
        h_bin[h_bin_idx] = 1.0
        p_v = h_bin - h
        if float(np.max(np.abs(p_v))) > 1e-8:
            h_new, c5_new = _try_dir(h, raw, p_v, dx)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue

        # Anti-gradient (sometimes helps escape flat regions)
        p_neg = grad - float(grad.mean())
        h_new, c5_new = _try_dir(h, raw, p_neg, dx)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue

        stall += 1
        if stall % 12 == 0:
            tau = tau * 1.4 if tau < 15.0 else tau * 0.6
            tau = max(0.1, min(tau, 40.0))

        if stall > max_stall:
            pert = rng.normal(0, 0.015, n)
            h = _project(h + pert, n / 2.0)
            stall = 0
            tau = 0.8

    return best_h, best_c5

def _optimize_pairs(h, deadline, seed=100):
    """Focused pair-swap optimization."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(raw)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0

    while time.time() < deadline:
        if stall > 150:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(raw)) * dx
        n1 = n - 1

        # Focus on top lags for direction
        top_idx = np.argsort(raw)[-6:]
        combined_press = np.zeros(n)
        for k in top_idx:
            d = int(k) - n1
            w = raw[k] - raw[top_idx[0]] + 1.0
            if d >= 0:
                m = n - d
                if m > 0:
                    combined_press[:m] += w * (1.0 - h[d:])
                    combined_press[d:] += -w * h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    combined_press[g:] += w * (1.0 - h[:m])
                    combined_press[:m] += -w * h[g:]

        order = np.argsort(combined_press)
        n_test = max(8, n // 12)
        lows = order[:n_test]
        highs = order[-n_test:]

        improved = False
        for _ in range(12):
            if time.time() > deadline:
                break
            pi = int(lows[int(rng.integers(0, len(lows)))])
            pj = int(highs[int(rng.integers(0, len(highs)))])
            if pi == pj:
                continue
            p_pair = np.zeros(n)
            if h[pi] >= h[pj]:
                p_pair[pi] = -1.0
                p_pair[pj] = 1.0
            else:
                p_pair[pi] = 1.0
                p_pair[pj] = -1.0
            h_new, c5_new = _try_dir(h, raw, p_pair, dx)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_step(h, deadline):
    """Try SLSQP with smooth-max surrogate."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    def obj_func(x, tau):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cm = float(np.max(corr))
        s = float(np.sum(np.exp((corr - cm) / tau)))
        return cm + tau * np.log(s)

    def obj_jac(x, tau):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cm = float(np.max(corr))
        exps = np.exp((corr - cm) / tau)
        w = exps / exps.sum()
        idx = np.where(exps > 1e-10)[0]
        weights = w[idx]
        return _grad_active(x, corr, idx, weights) * dx + np.max(_grad_active(x, corr, idx, weights) * dx)

    def cons_fun(x, target):
        return np.sum(x) - target

    for tau in [2.0, 5.0, 10.0]:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                lambda x: obj_func(x, tau),
                x0=h,
                method='SLSQP',
                jac=lambda x: obj_jac(x, tau),
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: cons_fun(x, target)}],
                options={'maxiter': 30, 'ftol': 1e-12}
            )
            if res.success:
                h_new = _project(res.x, target)
                c5_new = _c5_exact(h_new)
                if c5_new < _c5_exact(h) - 1e-10:
                    return h_new
        except Exception:
            pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _c5_exact(h_init)

    # Phase 1: Gradient optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _optimize_grad(h_init, d1, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair optimization (10%)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _optimize_pairs(best_h, d2, seed=seed + 50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP attempts (8%)
    d3 = t0 + budget_s * 0.63
    h3 = _slsqp_step(best_h, d3)
    if h3 is not None:
        c5_3 = _c5_exact(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-n exploration (20%)
    d4 = t0 + budget_s * 0.83
    n_curr = len(best_h)
    n_candidates = []
    for dn in [-30, -20, -12, -8, -4, 2, 6, 10, 20, 32, 48]:
        nt = n0 + dn
        if 50 <= nt <= 320 and nt != n_curr:
            n_candidates.append(nt)
    # Add some fixed values
    for fixed_n in [100, 132, 144, 160, 176, 184, 200, 216, 240]:
        if 50 <= fixed_n <= 320 and fixed_n not in n_candidates and fixed_n != n_curr:
            n_candidates.append(fixed_n)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.time() > d4 - 5:
            break
        h_r = _resample(best_h, n_try)
        c5_r = _c5_exact(h_r)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _optimize_grad(h_r, sub_dl, seed=seed + n_try)
        if time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.005)
            h_r, c5_r2 = _optimize_pairs(h_r, sub_dl2, seed=seed + n_try + 200)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Final intensive optimization (17%)
    d5 = deadline
    h5, c5_5 = _optimize_grad(best_h, d5, seed=seed + 777)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Last SLSQP polish
    if time.time() < deadline - 10:
        h_s = _slsqp_step(best_h, deadline - 5)
        if h_s is not None:
            c5_s = _c5_exact(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    return best_h, final_c5, len(best_h)
