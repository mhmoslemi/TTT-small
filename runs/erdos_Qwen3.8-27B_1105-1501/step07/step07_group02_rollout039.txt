```python
import numpy as np

from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlate_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad(h):
    """Compute gradient of each correlation row with respect to h.
    Returns array of shape (L, n) where L = 2n-1.
    Row r corresponds to lag ell = (n-1) - r.
    For lag ell: c_ell = sum_i h_i * (1 - h_{i+ell})
    dc_ell/dh_j = (1 - h_{j+ell}) * [j+ell in range] - h_{j-ell} * [j-ell in range]
    """
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n-1), n):
        row = k + n - 1
        if k >= 0:
            # For valid i: 0 <= i < n-k
            # dh[i]/dh[i] contributes (1-h[i+k]) to G[row, i] for i in [0, n-k)
            # dh[i+k]/dh[i] contributes -h[i] to G[row, i+k] for i in [0, n-k)
            m = n - k
            G[row, :m] += (1.0 - h[k:])  # first part: (1-h[i+k]) for i in [0,m)
            G[row, k:] -= h[:m]           # second part: -h[i] added to position i+k
        else:
            p = -k
            m = n - p
            # For valid i: p <= i < n
            # h[i] * (1-h[i-p])
            # dh[i]/dh[i] contributes (1-h[i-p]) to G[row, i] for i in [p, n)
            # dh[i-p] contributes -h[i] to G[row, i-p] for i in [p, n)
            G[row, p:] += (1.0 - h[:m])   # (1-h[i-p]) for i in [p, n), i-p in [0, m)
            G[row, :m] -= h[p:]           # -h[i] added to position i-p
    return G

def _project_onto_box_sum(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target} using bisection."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)
```

```python
def _gradient_descent(h_start, n, deadline, eps_start=0.05, eps_end=0.01, 
                      lr_start=0.01, max_iter=5000, seed=42):
    h = _project_onto_box_sum(h_start.copy(), n/2.0)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    eps = eps_start
    lr = lr_start
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        # Compute correlations and gradient
        corr = _correlate_full(h)
        L = len(corr)
        # Softmax weights
        M = np.max(corr)
        w = np.exp((corr - M) / eps)
        w /= np.sum(w)
        
        # Compute gradient
        G = _corr_grad(h)
        g = G.T @ w  # shape (n,)
        
        # Line search
        # Direction: -g (projected)
        # Try step: h_new = project(h - lr * g)
        best_step_val = cur_c5
        best_h_try = h.copy()
        
        for factor in [1.0, 0.7, 0.5, 0.3, 0.1]:
            step = lr * factor
            h_try = h - step * g
            h_try = _project_onto_box_sum(h_try, n/2.0)
            c5_try = _exact_c5(h_try)
            if c5_try < best_step_val:
                best_step_val = c5_try
                best_h_try = h_try
        
        if best_step_val < cur_c5 - 1e-12:
            h = best_h_try
            cur_c5 = best_step_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.05)
        else:
            stall += 1
            lr *= 0.8
            if stall % 50 == 0:
                eps *= 0.8  # make the approximation sharper
                if eps < eps_end:
                    eps = eps_end
        
        # Periodically sharpen epsilon
        if it % 100 == 0:
            eps = max(eps * 0.9, eps_end)
    
    return best_h, best_c5
```

```python
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        ell = k - (n - 1)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        ell = k - (n - 1)
        if ell >= 0:
            m = n - ell
            G[k, :m] += (1.0 - h[ell:])
            G[k, ell:] -= h[:m]
        else:
            p = -ell
            m = n - p
            G[k, p:] += (1.0 - h[:m])
            G[k, :m] -= h[p:]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_descent(h_start, deadline, eps_init=0.08, eps_min=0.01,
                  lr_init=0.015, lr_max=0.04, momentum=0.3,
                  max_iter=100000, seed=42):
    """Smooth gradient descent on log-sum-exp approximation of max correlation."""
    n = len(h_start)
    target = n / 2.0
    h = _project(h_start.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    eps = eps_init
    lr = lr_init
    vel = np.zeros(n)
    stall = 0
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            # Restart
            h = _project(h + 0.05 * (rng.random(n) - 0.5), target)
            cur_c5 = _exact_c5(h)
            eps = max(eps * 2, 0.05)
            lr = lr_init
            vel = np.zeros(n)
            stall = 0
            continue
        
        # Compute correlation and gradient
        corr = _corr_full(h)
        M = float(np.max(corr))
        # Softmax weights with temperature eps
        w = np.exp((corr - M) / eps)
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.zeros(len(corr))
            idx = np.argmax(corr)
            w[idx] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        G = _corr_grad(h)
        g = G.T @ w  # shape (n,)
        
        # Momentum update
        vel = momentum * vel - lr * g
        h_try = h + vel
        h_try = _project(h_try, target)
        
        # Line search: try several step sizes
        best_val = cur_c5
        best_h_try = h.copy()
        for factor in [1.0, 0.7, 0.5, 0.3, 0.1]:
            ht = h + factor * vel
            ht = _project(ht, target)
            c5t = _exact_c5(ht)
            if c5t < best_val:
                best_val = c5t
                best_h_try = ht
        
        if best_val < cur_c5 - 1e-13:
            h = best_h_try
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adapt lr and eps
            lr = min(lr * 1.03, lr_max)
            if it % 50 == 0:
                eps = max(eps * 0.97, eps_min)
        else:
            stall += 1
            lr *= 0.85
            if stall % 30 == 0:
                eps = max(eps * 0.9, eps_min)
                momentum = max(momentum * 0.9, 0.1)
    
    return best_h, best_c5
```

```python
def _slsqp_solve(h_start, deadline, K=30, max_solves=5, seed=42):
    """SLSQP with active set of top K lags."""
    n = len(h_start)
    target = n / 2.0
    h = _project(h_start.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for solve_iter in range(max_solves):
        if time.time() > deadline:
            break
        
        corr = _corr_full(h)
        active = np.argsort(corr)[-K:]
        
        # Build initial guess
        x0 = np.empty(n + 1)
        x0[:n] = h
        x0[n] = cur_c5 + 0.005
        
        # Objective
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        # Inequality constraints: t - c_r(h) >= 0 for r in active
        def ineq_fun(x):
            h_val = np.clip(x[:n], 0, 1)
            t_val = x[n]
            c = _corr_full(h_val)
            return t_val - c[active]
        
        def ineq_jac(x):
            h_val = np.clip(x[:n], 0, 1)
            G = _corr_grad(h_val)
            J = G[active].T  # shape (n, K)
            # d(t - c_r)/dh_j = -dc_r/dh_j = -G[r, j]
            # d(t - c_r)/dt = 1
            full_J = np.zeros((K, n + 1))
            full_J[:, :n] = -J.T  # shape (K, n)
            full_J[:, n] = 1.0
            return full_J
        
        # Equality constraint: sum(h) - n/2 = 0
        def eq_fun(x):
            return np.sum(x[:n]) - target
        
        def eq_jac(x):
            j = np.zeros(n + 1)
            j[:n] = 1.0
            return j
        
        constraints = [
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        ]
        
        bounds = [(0, 1)] * n + [(0, 2.0)]
        
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
        except Exception:
            continue
        
        if res.success or res.fun < best_c5:
            h_sol = _project(res.x[:n].copy(), target)
            c5_sol = _exact_c5(h_sol)
            if c5_sol < best_c5 - 1e-12:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                h = h_sol.copy()
            # Update active set and try again
            corr = _corr_full(h)
            if float(np.max(corr)) > res.x[n] + 0.01:
                # Some non-active constraint is violated, re-solve
                continue
            else:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        ell = k - (n - 1)
        if ell >= 0:
            m = n - ell
            G[k, :m] += (1.0 - h[ell:])
            G[k, ell:] -= h[:m]
        else:
            p = -ell
            m = n - p
            G[k, p:] += (1.0 - h[:m])
            G[k, :m] -= h[p:]
    return G * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _slsqp_solve(h_start, deadline, K=25, max_rounds=4, seed=42):
    n = len(h_start)
    target = n / 2.0
    h = _project(h_start.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = _corr_full(h)
        active = np.argsort(corr)[-K:]
        x0 = np.empty(n + 1)
        x0[:n] = h
        x0[n] = cur_c5 + 0.01
        def _obj(x):
            return x[n]
        def _obj_g(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def _ineq_fun(x):
            hv = np.clip(x[:n], 0.0, 1.0)
            cv = _corr_full(hv)
            return x[n] - cv[active]
        def _ineq_jac(x):
            hv = np.clip(x[:n], 0.0, 1.0)
            Gr = _corr_grad(hv)[active]
            J = np.zeros((K, n + 1))
            J[:, :n] = -Gr
            J[:, n] = 1.0
            return J
        def _eq_fun(x):
            return np.sum(x[:n]) - target
        def _eq_jac(x):
            j = np.zeros(n + 1)
            j[:n] = 1.0
            return j
        constraints = [
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
        ]
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        try:
            res = _smin(_obj, x0, method='SLSQP', jac=_obj_g,
                        bounds=bounds, constraints=constraints,
                        options={'maxiter': 150, 'ftol': 1e-10})
        except Exception:
            break
        if res.x is not None and len(res.x) == n + 1:
            h_sol = _project(np.clip(res.x[:n], 0, 1), target)
            c5_sol = _exact_c5(h_sol)
            if c5_sol < best_c5 - 1e-12:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                h = h_sol.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_start(h_base, n, rng, mode=0):
    if mode == 0:
        return _project(h_base.copy(), n / 2.0)
    elif mode == 1:
        noise = rng.normal(0, 0.05, n)
        return _project(h_base + noise, n / 2.0)
    elif mode == 2:
        shifted = np.roll(h_base, int(rng.integers(-5, 6)))
        noise = rng.normal(0, 0.03, n)
        return _project(shifted + noise, n / 2.0)
    elif mode == 3:
        smooth = h_base.copy()
        for _ in range(3):
            smooth = 0.5 * (smooth + np.roll(smooth, 1)) + 0.5 * (smooth + np.roll(smooth, -1))
            smooth = 0.75 * smooth
        return _project(h_base * 0.5 + smooth * 0.5 + rng.normal(0, 0.03, n), n / 2.0)
    else:
        half = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        half[idx] = 1.0
        return _project(half, n / 2.0)
```
