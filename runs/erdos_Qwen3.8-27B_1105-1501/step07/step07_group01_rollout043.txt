def _smooth_value_and_weights(scores, beta):
    m = float(np.max(scores))
    if beta < 1e-12: return m, None
    s = scores - m
    # s <=0
    ex = np.exp(beta * s)
    z = ex.sum()
    if z == 0: ex = np.zeros_like(scores); ex[np.argmax(scores)] = 1.0; z=1.0
    w = ex / z
    val = m + np.log(z) / beta
    return val, w

def _project(v, target):
    n = len(v)
    if target <= 0.0: return np.zeros(n)
    if target >= float(n): return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
       mid=(lo+hi)/2
       s=float(np.sum(np.clip(v-mid,0.0,1.0)))
       if s > target: lo=mid
       else: hi=mid
    theta=(lo+hi)/2
    x=np.clip(v-theta,0.0,1.0)
    res=target - float(np.sum(x))
    if abs(res) > 1e-12:
       if res > 0: mask=x<1.0
       else: mask=x>0.0
       cnt=int(np.sum(mask))
       if cnt>0:
          x = x.copy()
          x[mask] += res/cnt
          x = np.clip(x,0.0,1.0)
    return x

def _weighted_gradient(h, weights):

        return 0.0, cur

    hi = float(ts[min(len(ts)-1, best_i + 2)])

import numpy as np
import time
from scipy.optimize import linprog


def _project(v, target):
    n = len(v)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0.0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / float(cnt)
            x = np.clip(x, 0.0, 1.0)
    return x


def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / float(n)
    return float(np.max(_raw_scores(h)) * dx)


def _select_active(scores, tol, cap):
    max_s = float(np.max(scores))
    idx = np.flatnonzero(scores >= max_s - tol)
    if len(idx) == 0:
        return np.array([int(np.argmax(scores))], dtype=int)
    if len(idx) > cap:
        order = np.argsort(scores[idx])
        idx = idx[order[-cap:]]
    return idx


def _fill_gradient_full(h, k, g):
    n = len(h)
    d = int(k) - (n - 1)
    g.fill(0.0)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[d:] += 1.0 - h[:m]
        g[:m] -= h[d:d + m]
    else:
        e = -d
        m = n - e
        g[:m] += 1.0 - h[e:e + m]
        g[e:] -= h[:m]


def _weighted_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    center = n - 1
    for k in range(2 * n - 1):
        w = float(weights[k])
        if w <= 1e-13:
            continue
        d = int(k) - center
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[d:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[d:d + m]
        else:
            e = -d
            m = n - e
            grad[:m] += w * (1.0 - h[e:e + m])
            grad[e:] -= w * h[:m]
    return grad


def _smooth_weights(scores, beta):
    if beta <= 1e-12:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
        return w
    m = float(np.max(scores))
    ex = np.exp(beta * (scores - m))
    z = float(np.sum(ex))
    if not np.isfinite(z) or z <= 0.0:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
        return w
    return ex / z


def _smooth_value(scores, beta):
    if beta <= 1e-12:
        return float(np.max(scores))
    m = float(np.max(scores))
    ex = np.exp(beta * (scores - m))
    z = float(np.sum(ex))
    if not np.isfinite(z) or z <= 0.0:
        return m
    return m + float(np.log(z)) / beta


def _scale_direction(d, h, target_mag):
    max_abs = float(np.max(np.abs(d)))
    if max_abs <= 1e-14 or target_mag <= 1e-14:
        return None
    s_max = np.inf
    pos = d > 1e-14
    if np.any(pos):
        vals = (1.0 - h[pos]) / d[pos]
        if vals.size:
            tm = float(np.min(vals))
            if tm < s_max:
                s_max = tm
    neg = d < -1e-14
    if np.any(neg):
        vals = h[neg] / (-d[neg])
        if vals.size:
            tm = float(np.min(vals))
            if tm < s_max:
                s_max = tm
    if not np.isfinite(s_max) or s_max <= 1e-12:
        return None
    scale = target_mag / max_abs
    if scale > s_max:
        scale = s_max
    return d * scale


def _line_search(h, d, S, dx):
    cur = float(np.max(S)) * dx
    t_max = np.inf
    pos = d > 1e-14
    if np.any(pos):
        vals = (1.0 - h[pos]) / d[pos]
        tm = float(np.min(vals))
        if tm < t_max:
            t_max = tm
    neg = d < -1e-14
    if np.any(neg):
        vals = h[neg] / (-d[neg])
        tm = float(np.min(vals))
        if tm < t_max:
            t_max = tm
    if not np.isfinite(t_max) or t_max <= 1e-12:
        return 0.0, cur
    if t_max > 1.0:
        t_max = 1.0

    T = np.correlate(d, 1.0 - h, mode="full")
    T = T - np.correlate(h, d, mode="full")
    C = np.correlate(d, d, mode="full")

    g1 = np.linspace(0.0, 1.0, 64)
    g2 = np.linspace(0.0, 0.05, 24)
    ts = np.unique(np.concatenate((g1, g2))) * t_max
    ts2 = ts * ts
    vals = np.max(S[None, :] + ts[:, None] * T[None, :] - ts2[:, None] * C[None, :], axis=1) * dx

    best_t = float(ts[best_i])
    best_v = float(vals[best_i])
    if best_t <= 1e-12 or best_v >= cur:

    lo = float(ts[max(0, best_i - 2)])

    best_v = float(vals_r[i2])

    return best_t, best_v


def _smooth_descent_direction(h, S, beta, target_mag, anti_flag=False):
    w = _smooth_weights(S, beta)
    g = _weighted_gradient(h, w)
    d = -g
    d -= float(np.mean(d))
    if anti_flag and len(h) % 2 == 0:
        m = len(h) // 2
        mm = 0.5 * (d[:m] - d[m:])
        d[:m] = mm
        d[m:] = -mm
    return _scale_direction(d, h, target_mag)


def _make_anti(h):
    n = len(h)
    if n % 2 != 0:
        return _project(h, n / 2.0)
    m = n // 2
    a = 0.5 * (h[:m] + 1.0 - h[m:])
    out = np.empty(n, dtype=float)
    out[:m] = a
    out[m:] = 1.0 - a
    return _project(out, n / 2.0)


def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n, dtype=float)
    for _ in range(3):
        f = float(rng.uniform(0.5, 8.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(rng.uniform(0.5, 1.0))
        pert += a * np.sin(2.0 * np.pi * f * x / float(n) + ph)
    pert += 0.25 * amp * rng.normal(0.0, 1.0, size=n)
    pert *= amp
    return _project(h + pert, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n, dtype=float) + 0.5) / float(n)
    x_new = (np.arange(n_new, dtype=float) + 0.5) / float(n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        for i in range(var):
            hi0 = float(h[i])
            lo = max(-float(delta), -hi0)
            hi = min(float(delta), 1.0 - hi0)
            if lo > hi:
                lo = hi = 0.0
            bounds.append((lo, hi))

    c = np.zeros(var + 1, dtype=float)
    c[var] = -1.0
    A_ub = np.zeros((m_idx, var + 1), dtype=float)
    A_ub[:, :var] = G
    A_ub[:, var] = 1.0
    b_ub = max_s - S[idx]
    b_ub = np.maximum(b_ub, 0.0)
    bounds.append((None, 100.0))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None, -1.0

    if res is None or getattr(res, "x", None) is None:
        return None, -1.0
    if not np.all(np.isfinite(res.x)):
        return None, -1.0

    y = res.x[:var]
    gamma = float(res.x[var])
    if gamma <= 1e-8 or float(np.max(np.abs(y))) <= 1e-13:
        return None, gamma

    if anti_flag:
        d = np.zeros(n, dtype=float)
        d[:var] = y
        d[var:] = -y
    else:
        d = y.copy()
    d[np.abs(d) < 1e-14] = 0.0
    return d, gamma


def _local_polish(h, deadline, anti_flag=False, delta_init=0.18, tol_init=0.12, seed=1):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / float(n)

    if anti_flag:
        h = _make_anti(np.asarray(h, dtype=float).copy())
    else:
        h = _project(np.asarray(h, dtype=float).copy(), target)

    S = _raw_scores(h)
    cur = float(np.max(S)) * dx
    best_h = h.copy()
    best_c = cur

    rng = np.random.default_rng(1000003 + int(seed))
    delta = float(delta_init)
    tol = float(tol_init)
    no_improve = 0
    caps = (40, 22, 12, 5, 2, 1)

    while time.time() < deadline:
        max_s = float(np.max(S))
        improved = False

        for cap in caps:
            if time.time() >= deadline:
                break
            idx = _select_active(S, tol, cap)
            d, _ = _lp_solve_direction(h, S, idx, max_s, delta, anti_flag)
            if d is None:
                continue
            ds = _scale_direction(d, h, delta)
            if ds is None:
                continue
            t, val = _line_search(h, ds, S, dx)
            if t > 1e-10 and val < cur - 1e-12:
                h_new = h + t * ds
                if anti_flag:
                    h_new = _make_anti(h_new)
                else:
                    h_new = _project(h_new, target)
                S_new = _raw_scores(h_new)
                val_new = float(np.max(S_new)) * dx
                if val_new < cur - 1e-13:
                    h = h_new
                    S = S_new
                    cur = val_new
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    improved = True
                    no_improve = 0
                    delta = min(delta * 1.04, 0.35)
                    tol = max(tol * 0.98, 0.02)
                    break

        if improved:
            continue

        for beta in (30.0, 10.0):
            if time.time() >= deadline:
                break
            d = _smooth_descent_direction(h, S, beta, min(delta, 0.12), anti_flag)
            if d is None:
                continue
            t, val = _line_search(h, d, S, dx)
            if t > 1e-10 and val < cur - 1e-12:
                h_new = h + t * d
                if anti_flag:
                    h_new = _make_anti(h_new)
                else:
                    h_new = _project(h_new, target)
                S_new = _raw_scores(h_new)
                val_new = float(np.max(S_new)) * dx
                if val_new < cur - 1e-13:
                    h = h_new
                    S = S_new
                    cur = val_new
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    improved = True
                    no_improve = 0
                    break

        if improved:
            continue
        if no_improve > 0:
            for _ in range(8):
                if time.time() >= deadline:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                d = np.zeros(n, dtype=float)
                d[i] = 1.0
                d[j] = -1.0
                if anti_flag and n % 2 == 0:
                    m = n // 2
                    mm = 0.5 * (d[:m] - d[m:])
                    d[:m] = mm
                    d[m:] = -mm
                target_mag = min(0.25, max(0.08, delta))
                ds = _scale_direction(d, h, target_mag)
                if ds is None:
                    continue
                t, val = _line_search(h, ds, S, dx)
                if t > 1e-10 and val < cur - 1e-12:
                    h_new = h + t * ds
                    if anti_flag:
                        h_new = _make_anti(h_new)
                    else:
                        h_new = _project(h_new, target)
                    S_new = _raw_scores(h_new)
                    val_new = float(np.max(S_new)) * dx
                    if val_new < cur - 1e-13:
                        h = h_new
                        S = S_new
                        cur = val_new
                        if cur < best_c:
                            best_c = cur
                            best_h = h.copy()
                        improved = True
                        no_improve = 0
                        break
        if improved:
            continue

        no_improve += 1
        tol = min(tol * 1.4, 2.0)
        delta = delta * 0.86

        if no_improve % 25 == 0:
            amp = 0.01 + 0.001 * min(no_improve, 50)
            h_try = _smooth_perturb(best_h, amp, rng)
            if anti_flag:
                h_try = _make_anti(h_try)
            S_try = _raw_scores(h_try)
            c_try = float(np.max(S_try)) * dx
            if c_try < best_c + 0.01:
                h = h_try
                S = S_try
                cur = c_try
            else:
                h = best_h.copy()
                S = _raw_scores(h)
                cur = best_c
            no_improve = 0
            delta = max(0.08, min(delta, 0.18))
            tol = 0.12

        if delta < 0.025:
            break

    return best_h, best_c


def _smooth_polish(h, deadline, seed=7):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / float(n)
    h = _project(np.asarray(h, dtype=float).copy(), target)
    S = _raw_scores(h)
    cur = float(np.max(S)) * dx
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(2000003 + int(seed))

    for beta in (5.0, 15.0, 40.0, 90.0):
        stall = 0
        while time.time() < deadline:
            d = _smooth_descent_direction(h, S, beta, 0.08, False)
            if d is None:
                break
            t, val = _line_search(h, d, S, dx)
            if t > 1e-10 and val < cur - 1e-12:
                h_new = _project(h + t * d, target)
                S_new = _raw_scores(h_new)
                val_new = float(np.max(S_new)) * dx
                if val_new < cur - 1e-13:
                    h = h_new
                    S = S_new
                    cur = val_new
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    stall = 0
                    continue
            stall += 1
            if stall > 40:
                break
    return best_h, best_c


def _fallback_initial(n, seed):
    rng = np.random.default_rng(seed)
    if n % 2 == 0:
        m = n // 2
        x = np.arange(m, dtype=float)
        a = 0.5 + 0.45 * np.sin(2.0 * np.pi * (x / float(m) + float(rng.uniform(0.0, 1.0))))
        a += 0.08 * rng.normal(0.0, 1.0, size=m)
        a = np.clip(a, 0.0, 1.0)
        h = np.empty(n, dtype=float)
        h[:m] = a
        h[m:] = 1.0 - a
        return _project(h, n / 2.0)
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * x / float(n) + float(rng.uniform(0.0, 1.0)))
    h += 0.1 * rng.normal(0.0, 1.0, size=n)
    return _project(np.clip(h, 0.0, 1.0), n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.default_rng(seed)

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or len(h0) < 10 or not np.all(np.isfinite(h0)):
        h0 = _fallback_initial(180, seed)

    n = len(h0)
    h_init = _project(h0, n / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    d1 = t0 + 0.42 * budget_s
    if d1 > deadline:
        d1 = deadline
    h1, c1 = _local_polish(best_h, d1, anti_flag=False, delta_init=0.18, tol_init=0.12, seed=101)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    if n % 2 == 0 and time.time() < deadline:
        d2 = t0 + 0.58 * budget_s
        if d2 > deadline:
            d2 = deadline
        h_anti = _make_anti(best_h)
        ca0 = _exact_c5(h_anti)
        if ca0 < best_c:
            best_c = ca0
            best_h = h_anti.copy()
        h2, c2 = _local_polish(h_anti, d2, anti_flag=True, delta_init=0.18, tol_init=0.12, seed=202)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    d3 = t0 + 0.68 * budget_s
    if d3 > deadline:
        d3 = deadline
    if time.time() < d3:
        h3, c3 = _smooth_polish(best_h, d3, seed=303)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    d4 = t0 + 0.80 * budget_s
    if d4 > deadline:
        d4 = deadline
    if time.time() < d4:
        base_n = len(best_h)
        tried = set()
        for dn in (2, 4, -2, 6, -4, 8, -8):
            if time.time() >= d4 - 8:
                break
            n_try = base_n + dn
            if n_try < 120 or n_try > 260 or n_try in tried:
                continue
            tried.add(n_try)
            h_r = _resample(best_h, n_try)
            sub = min(d4, time.time() + 12.0)
            h_r, c_r = _local_polish(h_r, sub, anti_flag=(n_try % 2 == 0),
                                     delta_init=0.15, tol_init=0.10, seed=404 + n_try)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
                base_n = n_try

    d5 = deadline - 15.0
    if d5 < time.time() + 5:
        d5 = deadline - 2.0

    amp = 0.035
    restarts = 0
    while time.time() < d5:
        restarts += 1
        remain = d5 - time.time()
        if remain < 6.0:
            break
        tp = max(4.0, min(26.0, remain * 0.08))
        sub = min(d5, time.time() + tp)

        h_p = _smooth_perturb(best_h, amp, rng)
        h_r, c_r = _local_polish(h_p, sub, anti_flag=False,
                                 delta_init=0.14, tol_init=0.10, seed=500 + 7 * restarts)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            amp = max(0.004, amp * 0.85)
        else:
            if c_r < best_c + 0.004:
                amp = max(0.001, amp * 0.95)
            else:
                amp = min(0.14, amp * 1.25)

        if len(best_h) % 2 == 0 and restarts % 2 == 0 and time.time() < d5 - 5.0:
            h_pa = _make_anti(_smooth_perturb(best_h, amp, rng))
            sub_a = min(d5, time.time() + tp * 0.8)
            h_a, c_a = _local_polish(h_pa, sub_a, anti_flag=True,
                                     delta_init=0.14, tol_init=0.10, seed=600 + 11 * restarts)
            if c_a < best_c:
                best_c = c_a
                best_h = h_a.copy()
                amp = max(0.004, amp * 0.85)
            else:
                if c_a < best_c + 0.004:
                    amp = max(0.001, amp * 0.95)
                else:
                    amp = min(0.14, amp * 1.15)

        if restarts > 90:
            break

    d_final = deadline - 1.5
    if d_final > time.time() + 2.0:
        h_f, c_f = _local_polish(best_h, d_final, anti_flag=False,
                                 delta_init=0.08, tol_init=0.06, seed=999)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    best_h = _project(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, float(final_c), int(len(best_h))
