import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_lp(h, n_top_lags, radius):
    """Build and solve LP to find direction reducing top lags."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    # Select top lags
    top_idx = np.argsort(-raw)[:n_top_lags]
    
    # Build linearized coefficients for each selected lag
    # For lag r (0-indexed in raw, actual shift = r - (n-1)):
    # If shift >= 0: raw[r] = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
    # Linearized: sum_i d[i]*(1-h[i+shift]) - sum_i h[i]*d[i+shift]
    # For shift < 0: let s = -shift, raw[r] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
    # Linearized: sum_i d[i+s]*(1-h[i]) - sum_i h[i+s]*d[i]
    
    A_rows = []
    b_const = []
    for idx in top_idx:
        shift = idx - (n - 1)
        if shift >= 0:
            m = n - shift
            # coeff for d[i] is (1-h[i+shift]) for i in [0, m)
            # coeff for d[j] is -h[j-shift] for j in [shift, n)
            coeff = np.zeros(n)
            coeff[:m] += (1.0 - h[shift:shift+m])
            coeff[shift:] -= h[:m]
            A_rows.append(coeff)
            b_const.append(float(raw[idx]))
        else:
            s = -shift
            m = n - s
            # raw = sum_{i=0}^{m-1} h[i+s]*(1-h[i])
            # Linearized: sum_i d[i+s]*(1-h[i]) + sum_i h[i+s]*(-d[i])
            # coeff for d[j] where j = i+s: (1-h[i]) = (1-h[j-s])
            # coeff for d[i]: -h[i+s]
            coeff = np.zeros(n)
            coeff[s:] += (1.0 - h[:m])  # d[j] for j in [s, n), coeff = 1-h[j-s]
            coeff[:m] -= h[s:s+m]       # d[i] for i in [0, m), coeff = -h[i+s]
            A_rows.append(coeff)
            b_const.append(float(raw[idx]))
    
    A = np.array(A_rows)
    b_const = np.array(b_const)
    
    # Variable: d (direction)
    d = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        A @ d + b_const <= t,
        cp.sum(d) == 0,
        d >= -radius,
        d <= radius
    ]
    # Also ensure 0 <= h + d <= 1
    lo_bound = np.maximum(-radius, -h)
    hi_bound = np.minimum(radius, 1.0 - h)
    constraints.append(d >= lo_bound)
    constraints.append(d <= hi_bound)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # Try GLOP first, fallback to CBC
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return None
    
    if d.value is None:
        return None
    
    d_val = d.value
    if not np.all(np.isfinite(d_val)):
        return None
    
    return d_val

def _line_search(h, d, raw, dx, a_max, n_grid=50):
    """Exact 1D line search along direction d from h."""
    n = len(h)
    # Boundary: h + a*d must be in [0,1]
    pos = d > 1e-12
    neg = d < -1e-12
    a_hi = a_max
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    
    # Precompute for quadratic approximation
    # raw(a) = raw(h + a*d) = correlate(h+ad, 1-h-ad)
    # = correlate(h, 1-h) + a*correlate(d, 1-h) + a*correlate(h, -d) - a^2*correlate(d, d)
    # = raw + a*(correlate(d,1-h) - correlate(h,d)) - a^2*correlate(d,d)
    
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Max of raw + a*T - a^2*Q over a in [0, a_hi]
    # Each component is a concave quadratic in a
    # The max of concave quadratics is not concave, but we can find its max
    
    # Strategy: find critical points
    candidates = [0.0, a_hi]
    
    # For each active index, the vertex of the parabola is at T[i]/(2*Q[i])
    top_idx = np.argsort(-raw)[:20]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 0.0 < av < a_hi:
                candidates.append(av)
    
    # Grid
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    
    # Refine with golden section
    if best_alpha > 0:
        step = a_hi / n_grid
        lo = max(0.0, best_alpha - step)
        hi = min(a_hi, best_alpha + step)
        if hi - lo > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            for _ in range(40):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            opt = (lo + hi) / 2.0
            val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
            if val_opt < best_val:
                best_val = val_opt
                best_alpha = opt
    
    return best_alpha, best_val

def _lp_optimize(h, deadline, n_top=15, radius=0.3, stall_limit=50):
    """Main LP-based optimization loop."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()

    stall = 0
    radius_cur = radius

    while time.time() < deadline:

            break

        d_dir = _build_lp(h, n_top_cur, radius_cur)
        
        if d_dir is not None:

            # Re-project to ensure feasibility
            h_try = h + d_dir

            d_dir = h_try - h  # Adjusted direction after projection

                        cur_c5 = actual

                            best_h = h.copy()

        u = 1.0 - h

            else:

                m = n - gp
                if m > 0:

        n_top_cur = max(5, n_top_cur - 1)

            radius_cur = 0.2

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        if s > target:

        else:

    res = target - float(np.sum(x))

def _build_lp(h, n_top, radius):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:n_top]
    A_rows = []
    b_const = []
    for idx in top_idx:
        s = idx - (n - 1)
        coeff = _linearized_coeff(h, n, s)
        A_rows.append(coeff)
        b_const.append(float(raw[idx]))
    A = np.array(A_rows)
    b_const = np.array(b_const)
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ d + b_const <= t,
        cp.sum(d) == 0,
        d >= -radius,
        d <= radius
    ]
    lo_bound = np.maximum(-radius, -h)
    hi_bound = np.minimum(radius, 1.0 - h)
    constraints.append(d >= lo_bound)
    constraints.append(d <= hi_bound)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK)
            except Exception:
                return None
    if d.value is None:
        return None
    d_val = np.array(d.value, dtype=float)
    if not np.all(np.isfinite(d_val)):
        return None
    return d_val

        for j in range(i + 1, min(len(top_idx), 8)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A2 = Q[ri] - Q[rj]
            B2 = T[ri] - T[rj]
            C2 = raw[ri] - raw[rj]
            if abs(A2) > 1e-14:
                disc = B2 * B2 + 4.0 * A2 * C2
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B2 + sd) / (2.0 * A2)
                    a2 = (B2 - sd) / (2.0 * A2)
                    for aa in [a1, a2]:
                        if 0.0 < aa < a_hi:
                            candidates.append(aa)
            elif abs(B2) > 1e-14:
                aa = -C2 / B2
                if 0.0 < aa < a_hi:
                    candidates.append(aa)
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    if best_alpha > 0.0:
        step = a_hi / n_grid
        lo = max(0.0, best_alpha - step)
        hi = min(a_hi, best_alpha + step)
        if hi - lo > 1e-13:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            for _ in range(50):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
            opt = (lo + hi) / 2.0
            val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
            if val_opt < best_val:
                best_val = val_opt
                best_alpha = opt
    return best_alpha, best_val

def _compute_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        wp = w[r]
        if wp < 1e-14:
            continue
        s = r - (n - 1)
        if s >= 0:
            m = n - s
            if m > 0:
                g[s:] += wp * u[:m]
                g[:m] -= wp * h[s:]
        else:
            p = -s
            m = n - p
            if m > 0:
                g[:m] += wp * u[p:]
                g[p:] -= wp * h[:m]
    return g

def _lp_optimize(h, deadline, n_top=15, radius=0.25, stall_limit=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    n_top_cur = n_top
    beta = 20.0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        accepted = False
        d_dir = _build_lp(h, n_top_cur, radius_cur)
        if d_dir is not None:
            d_dir -= np.mean(d_dir)
            h_try = h + d_dir
            h_try = _project(h_try, target)
            d_adj = h_try - h
            if np.max(np.abs(d_adj)) > 1e-10:
                raw = np.correlate(h, 1.0 - h, mode='full')
                alpha, new_c5 = _line_search(h, d_adj, raw, dx, radius_cur)
                if alpha > 1e-10:
                    h_new = h + alpha * d_adj
                    h_new = _project(h_new, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius_cur = min(radius_cur * 1.01, 0.45)
                        n_top_cur = min(n_top_cur + 1, 30)
                        accepted = True
        if not accepted:
            raw = np.correlate(h, 1.0 - h, mode='full')
            mx = float(np.max(raw))
            shifted = raw - mx
            vp = np.exp(np.clip(beta * shifted, -80, 0))
            w = vp / (float(np.sum(vp)) + 1e-30)
            g = _compute_grad(h, w)
            g -= np.mean(g)
            nm = np.linalg.norm(g)
            if nm > 1e-12:
                d_grad = -g / nm
                alpha, new_c5 = _line_search(h, d_grad, raw, dx, radius_cur)
                if alpha > 1e-10:
                    h_new = h + alpha * d_grad
                    h_new = _project(h_new, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        accepted = True
        if not accepted:
            stall += 1
            radius_cur *= 0.82
            n_top_cur = max(8, n_top_cur - 1)
            beta *= 1.05
            if beta > 50.0:
                beta = 50.0
            if stall % 25 == 0 and radius_cur < 0.04:
                radius_cur = 0.15
                n_top_cur = min(n_top_cur + 5, 30)
    return best_h, best_c5

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.25, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_pairs):
        if time.time() > deadline or stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = np.correlate(h, 1.0 - h, mode='full')
        alpha, c5_try = _line_search(h, d, raw, dx, radius, n_grid=40)
        if alpha > 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9994
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    improved = True
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.3, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _multi_dir_lp(h, deadline, n_top=20, radius=0.2, stall_limit=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    raw = np.correlate(h, 1.0 - h, mode='full')
    while time.time() < deadline:
        if stall > stall_limit:
            break
        top_idx = np.argsort(-raw)[:n_top]
        selected = top_idx[np.random.default_rng(int(time.time()*1000)%10000).choice(len(top_idx), size=min(10, len(top_idx)), replace=False)]
        A_rows = []
        b_const = []
        for idx in selected:
            s = idx - (n - 1)
            coeff = _linearized_coeff(h, n, s)
            A_rows.append(coeff)
            b_const.append(float(raw[idx]))
        A = np.array(A_rows)
        b_const = np.array(b_const)
        d = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            A @ d + b_const <= t,
            cp.sum(d) == 0,
            d >= -radius_cur,
            d <= radius_cur
        ]
        lo_bound = np.maximum(-radius_cur, -h)
        hi_bound = np.minimum(radius_cur, 1.0 - h)
        constraints.append(d >= lo_bound)
        constraints.append(d <= hi_bound)
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                prob.solve(solver=cp.GLPK)
        accepted = False
        if d.value is not None and np.all(np.isfinite(d.value)):
            d_dir = np.array(d.value, dtype=float)
            d_dir -= np.mean(d_dir)
            h_try = _project(h + d_dir, target)
            d_adj = h_try - h
            if np.max(np.abs(d_adj)) > 1e-10:
                alpha, new_c5 = _line_search(h, d_adj, raw, dx, radius_cur)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d_adj, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = actual
                        raw = np.correlate(h, 1.0 - h, mode='full')
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        accepted = True
        if not accepted:
            stall += 1
            radius_cur *= 0.85
            if radius_cur < 0.03:
                radius_cur = 0.12
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP optimization from initial (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _lp_optimize(h_init, d1, n_top=15, radius=0.25, stall_limit=50)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary refinement (12%)
    d2 = t0 + budget_s * 0.30
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=35000, T_init=0.003)
    sub_dl = min(d2, time.time() + budget_s * 0.03)
    h_bin, c5_b = _pair_sweep(h_bin, sub_dl)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2:
        h_pol = _project(h_bin.copy(), len(h_bin) // 2 * 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_pol, c5_p = _lp_optimize(h_pol, sub_dl, n_top=12, radius=0.15, stall_limit=30)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = list(range(-20, 21, 2))
    for dn in offsets:
        if time.time() > d3 - 8:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, n_top=12, radius=0.20, stall_limit=30)
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.005)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=8000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if time.time() < d3 - 3:
            sub_dl3 = min(d3, time.time() + budget_s * 0.006)
            h_r, c5_r = _lp_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3, n_top=14, radius=0.12, stall_limit=25)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.07)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _lp_optimize(h4, sub_dl, n_top=20, radius=0.20, stall_limit=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4 - 5:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_cont(h4, sub_dl, n_pairs=6000, radius=0.20, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=25000, T_init=0.0015)
        sub_dl2 = min(d4, time.time() + budget_s * 0.015)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4 - 3:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _multi_dir_lp(h4, sub_dl, n_top=20, radius=0.10, stall_limit=40)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (12%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 8 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 6, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_optimize(h_pert, sub_dl, n_top=15, radius=0.15, stall_limit=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, deadline - 2, n_top=25, radius=0.06, stall_limit=60)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline - 1, n_pairs=2000, radius=0.08, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
