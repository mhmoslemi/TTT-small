Let me be more careful. The score at lag d (row n-1+d) is:
S_{n-1+d} = sum_{i=0}^{n-1-d-1} h_i * (1 - h_{i+d})

When we change h_i from 1 to 0:
- For lag d > 0: if i < n-d, the term h_i*(1-h_{i+d}) was 1*(1-h_{i+d}). Now it's 0. Change: -(1-h_{i+d}) = -1+h_{i+d}
  Also, for lag d' < 0 (i.e., the term where i appears as the second index): if i-d' >= 0, the term h_{i-d'}*(1-h_i) was h_{i-d'}*1. Now it's 0. Change: -h_{i-d'}
  
When we change h_j from 0 to 1:
- For lag d > 0: if j < n-d, the term h_j*(1-h_{j+d}) was 0. Now it's 1*(1-h_{j+d}). Change: +(1-h_{j+d})
  Also for lag d' < 0: if j-d' >= 0, the term h_{j-d'}*(1-h_j) was h_{j-d'}*1. Now it's 0. Change: -h_{j-d'}

So the total change for swapping i (1→0) and j (0→1):
ΔS_{n-1+d} = -(1-h_{i+d})[i < n-d] + (1-h_{j+d})[j < n-d] + ... (similar for negative lags)

This is O(n) per pair to compute the full effect. With ~90 ones and ~90 zeros, there are ~8100 pairs to try. At O(n) per pair, that's about 8100*180 ≈ 1.5M operations per full sweep. That's fast.

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n-1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m = n - d
        G[n-1+d, :m] = 1.0 - h[d:]
        G[n-1+d, d:m+d] -= h[:m]
        G[n-1-d, d:m+d] = 1.0 - h[:m]
        G[n-1-d, :m] -= h[d:]
    return G

def _lp_full(h, S, G, radius):
    n = len(h)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = -S.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, S0, G, radius):
    n = len(h)
    Q = np.correlate(p, p, mode='full')
    T = G @ p
    
    t_hi = radius
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
    
    if t_hi < 1e-12:
        return 0.0, S0.copy()
    
    # Grid search
    n_grid = 40
    grid = np.linspace(0.0, t_hi, n_grid)
    best_val = float(np.max(S0))
    best_t = 0.0
    for i in range(n_grid):
        t = grid[i]
        val = float(np.max(S0 + t * T - t * t * Q))
        if val < best_val:
            best_val = val
            best_t = t
    
    # Golden section refinement
    bi = int(np.argmin([float(np.max(S0 + grid[i]*T - grid[i]**2*Q)) for i in range(n_grid)]))
    lo_g = grid[max(0, bi - 3)]
    hi_g = grid[min(n_grid - 1, bi + 3)]
    if hi_g - lo_g > 1e-13:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_g - gr * (hi_g - lo_g)
        c2 = lo_g + gr * (hi_g - lo_g)
        f1 = float(np.max(S0 + c1*T - c1**2*Q))
        f2 = float(np.max(S0 + c2*T - c2**2*Q))
        for _ in range(45):
            if hi_g - lo_g < 1e-14:
                break
            if f1 < f2:
                hi_g, c2, f2 = c1, c1, f1
                c1 = hi_g - gr * (hi_g - lo_g)
                f1 = float(np.max(S0 + c1*T - c1**2*Q))
            else:
                lo_g, c1, f1 = c2, c2, f2
                c2 = lo_g + gr * (hi_g - lo_g)
                f2 = float(np.max(S0 + c2*T - c2**2*Q))
        dt = (lo_g + hi_g) / 2.0
        dv = float(np.max(S0 + dt*T - dt**2*Q))
        if dv < best_val:
            best_val, best_t = dv, dt
    
    return best_t, best_val

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        S = _raw_scores(h)
        G = _grad_matrix(h)
        p = _lp_full(h, S, G, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        
        alpha, pred_raw = _line_search(h, p, S, G, radius)
        
        if alpha > 1e-12:
            h_new = _project_sum(h + alpha * p, target)

            if c_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c_new
                stall = 0

                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[-k:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0

def _binary_local(h_b, deadline, max_rounds=100):
    n = len(h_b)
    cur = h_b.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()

    for rnd in range(max_rounds):

            break
        improved = False

                    zeros.append(oi)
                    ones.append(oz)
                    break
                else:

                    cur[oz] = 0.0
            if improved_inner:

def _binary_sa(h_b, deadline, max_iter=30000, T_init=0.005, seed=123):
    n = len(h_b)
    cur = h_b.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(1.0, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    return _project_sum(h + pert, n / 2.0)
