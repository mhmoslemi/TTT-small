import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Water-filling projection to [0,1] with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h, all_lags=True):
    """Build the gradient matrix for all (or active) lags."""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        r = k - (n - 1)
        if r >= 0:
            m = n - r
            if m > 0:
                # S_r = sum_{j=0}^{m-1} h[j+r]*(1-h[j])
                # dS_r/dh[j+r] = (1-h[j]), dS_r/dh[j] = -h[j+r]
                G[k, r:r+m] += (1.0 - h[:m])
                G[k, :m] += -h[r:r+m]
        else:
            g = -r
            m = n - g
            if m > 0:
                # S_{-g} = sum_{j=0}^{m-1} h[j]*(1-h[j+g])
                # dS_{-g}/dh[j] = (1-h[j+g]), dS_{-g}/dh[j+g] = -h[j]
                G[k, :m] += (1.0 - h[g:g+m])
                G[k, g:g+m] += -h[:m]
    return G

def _full_lp(h, scores, radius):
    """Full-lag LP to find a descent direction."""
    n = len(h)
    n_lags = 2 * n - 1
    G = _grad_matrix(h)
    # min tau
    # s.t. G_k @ d - tau <= -scores[k] for all k
    #      sum(d) = 0
    #      -radius <= d_i <= radius
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, t_max):
    """Exact line search for h + t*d."""
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find max t such that h + t*d stays in [0,1]
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = t_max
    if np.any(pos):
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_hi = min(t_hi, np.min(h[neg] / (-d[neg])))
    t_hi = max(t_hi, 0.0)
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Grid + golden section
    grid = np.linspace(0, t_hi, 40)
    fvals = np.array([float(np.max(S0 + t*T - t*t*C)) * dx for t in grid])
    best_idx = np.argmin(fvals)
    
    lo = grid[max(0, best_idx-2)]
    hi = grid[min(39, best_idx+2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-10:
        opt_t, opt_val = lo, float(np.max(S0 + lo*T - lo*lo*C)) * dx
    else:
        c1 = hi - gr*(hi-lo)
        c2 = lo + gr*(hi-lo)
        f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        for _ in range(40):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr*(hi-lo)
                f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr*(hi-lo)
                f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
        opt_t = (lo+hi)/2.0
        opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    
    # Check boundaries
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx
    best = min([(opt_t, opt_val), (0.0, f0), (t_hi, fhi)], key=lambda x: x[1])
    return best[0], best[1]

def _continuous_opt(h, deadline, radius_init=0.15, max_iters=5000):
    """Full-lag LP continuous optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = _all_scores(h)
        d = _full_lp(h, scores, radius)
        
        if d is None or np.max(np.abs(d)) < 1e-13:
            # Fallback: negative gradient of max score
            mx = np.max(scores)
            g = np.zeros_like(h)
            # Subgradient of max
            active = np.where(np.abs(scores - mx) < 1e-6)[0]
            if len(active) > 0:
                G = _grad_matrix(h)
                g = G[active].mean(axis=0)
            g -= np.mean(g)
            d = g
            if np.max(np.abs(d)) < 1e-13:
                stall += 1
                radius *= 0.8
                if radius < 1e-6:
                    break
                continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.1  # Reset to escape
    
    return best_h, best_c5

def _binary_swap_opt(h_bin, deadline, max_iters=50000):

        mx = np.max(scores)

        b[pj] = 1.0

            b[pi] = 1.0
            b[pj] = 0.0
            stall += 1
    
    return best_h, best_c5
