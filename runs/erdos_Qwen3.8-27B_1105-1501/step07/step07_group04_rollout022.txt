def _project(v, target):

def _build_gradient_row(h, k):

def _line_search(h, d, dx, radius):

    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    # Find alpha_max for feasibility
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 0.0)
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(C0 + t * L1 - t * t * Q2)) * dx

    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 4.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _subgradient_fallback(h, radius):
    """Simple subgradient direction as fallback."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(corr.max())
    active = np.where(corr >= max_val - 0.01 * max_val)[0]
    g = np.zeros(n)
    center = n - 1
    for r in active:
        k = int(r - center)
        grad = _build_gradient_row(h, k)
        g += grad
    g /= len(active)
    d = -g * dx
    d = d - np.mean(d)
    norm = float(np.linalg.norm(d))
    if norm > 1e-12:
        d = d / norm * radius * 0.5
    return d

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_rows(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    center = n - 1
    for r in range(m):
        k = r - center
        if k == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif k > 0:
            G[r, k:] += (1.0 - h[:n - k])
            G[r, :n - k] -= h[k:]
        else:
            u = -k
            G[r, :n - u] += (1.0 - h[u:])
            G[r, u:] -= h[:n - u]
    return G

def _lp_direction(h, corr_scaled, G, radius):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    A[:, :n] = G * dx
    A[:, n] = -1.0
    b[:] = -corr_scaled
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    cor_max = float(np.max(corr_scaled))
    bounds_list = []
    for i in range(n):
        lo_b = -min(radius, h[i]) + 1e-14
        hi_b = min(radius, 1.0 - h[i]) - 1e-14
        if lo_b > hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds_list.append((lo_b, hi_b))
    bounds_list.append((cor_max - 2.0, cor_max + 1.0))
    try:
        result = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_list, method='highs')
    except Exception:
        return None, None
    if result.status != 0:
        return None, None
    d = result.x[:n].copy()
    t_val = float(result.x[n])
    d = d - np.mean(d)
    if float(np.max(np.abs(d))) < 1e-13:
        return None, None
    return d, t_val

def _line_search(h, d, dx, radius):
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(C0 + t * L1 - t * t * Q2)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b > 1e-10:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2)) * dx
        f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-12:

            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(C0 + opt_t * L1 - opt_t * opt_t * Q2)) * dx
    else:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    f0 = float(np.max(C0) * dx)
    f_end = float(np.max(C0 + t_hi * L1 - t_hi * t_hi * Q2)) * dx
    candidates = [(0.0, f0), (opt_t, opt_val), (t_hi, f_end)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_local_optimize(h, deadline, radius_init=0.08, stall_limit=50, max_iter=3000):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if radius < 1e-5:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr_scaled = corr_raw * dx
        G = _build_grad_rows(h)
        d, t_pred = _lp_direction(h, corr_scaled, G, radius)
        if d is None:
            max_idx = int(np.argmax(corr_scaled))
            k = max_idx - (n - 1)
            grad = G[max_idx - (n - 1)]
            d = -grad * dx
            d = d - np.mean(d)
            norm = float(np.max(np.abs(d)))
            if norm > 1e-12:
                d = d / norm * radius * 0.1
            else:
                stall += 1
                radius *= 0.5
                continue
        d = np.clip(d, -radius, radius)
        d = d - np.mean(d)
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.25)
        else:
            stall += 1
            radius *= 0.7

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 4.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        b1 = cur[i:i + length].copy()
        b2 = cur[j:j + length].copy()
        if np.sum(b1) != np.sum(b2):

            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5

            stall = 0
        else:
            cur[i:i + length] = b1

    # LP from perturbation
    if time.time() < d1 - 3:
        h_p = _smooth_perturb(best_h, 0.04, rng)
        h_p, c5_p = _lp_local_optimize(h_p, d1, radius_init=0.08, stall_limit=30)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 2: Binary optimization (25%)
    d2 = t0 + budget_s * 0.60
    n_cur = len(best_h)
    h_bin = _to_binary(best_h, n_cur)
    c5_b0 = _exact_c5(h_bin)

    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_b1, c5_b1 = _binary_targeted(h_bin, sub_dl, max_iters=70000, seed=101)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = h_b1.copy()

    if time.time() < d2 - 5:
        h_b2 = _to_binary(best_h, len(best_h))
        sub_dl = min(d2, time.time() + budget_s * 0.06)
        h_b2, c5_b2 = _binary_sa(h_b2, sub_dl, max_iters=80000, temp_init=0.003, seed=202)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    if time.time() < d2 - 3:
        h_b3 = _to_binary(best_h, len(best_h))
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_b3, c5_b3 = _block_swap(h_b3, sub_dl, max_iters=15000, seed=303)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # LP polish on best
    if time.time() < d2 - 2:
        n_pol = len(best_h)
        h_pol = _project(best_h.copy(), n_pol / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _lp_local_optimize(h_pol, sub_dl, radius_init=0.05, stall_limit=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.85
    n_best = len(best_h)
    tried = set()
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, 28, 32, 36, 40]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 320 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r0 = _exact_c5(h_r)
        if c5_r0 > best_c5 + 0.02:
            continue
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_local_optimize(h_r, sub_dl, radius_init=0.08, stall_limit=25)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Basin hopping + final (remaining)
    d4 = soft_dl
    scale = 0.015
    for _ in range(100):
        if time.time() > d4 - 4:
            break
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.005)
        h_t, c5_t = _lp_local_optimize(h_pert, sub_dl, radius_init=0.04, stall_limit=20)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Final concentrated LP polish
    if time.time() < soft_dl - 2:
        n_fin = len(best_h)
        h_fin = _project(best_h.copy(), n_fin / 2.0)
        h_fin, c5_fin = _lp_local_optimize(h_fin, soft_dl - 1, radius_init=0.03, stall_limit=30)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Check a few nearby n values at the end
    n_chk = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > soft_dl - 1:
            break
        n_try = n_chk + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_local_optimize(h_r, soft_dl - 1, radius_init=0.02, stall_limit=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n_final
