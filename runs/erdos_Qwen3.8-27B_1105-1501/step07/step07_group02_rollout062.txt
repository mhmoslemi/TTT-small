def _grad_descent(h, deadline, beta_start=15.0, beta_end=200.0,
                  n_iters_max=8000, seed=42):

    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    stall = 0
    iters = 0
    lags = np.arange(2 * n - 1) - (n - 1)

        progress = min(1.0, iters / max(1, int(n_iters_max * 0.6)))
        beta = max(5.0, beta_start + (beta_end - beta_start) * progress)

        grad_mean = float(np.sum(grad)) / n
        d = -grad + grad_mean  # sum-zero direction

        if d_norm < 1e-13:
            stall += 5

            t_lo = 0.0
            t_hi = 1e8
            pos = d > 1e-14
            neg = d < -1e-14
            if np.any(pos):
                t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
            if np.any(neg):

            t_lo = max(t_lo, 0.0) + 1e-10

                dx = 2.0 / n

                h_new = _project(h + t_opt * d, target)
                c5_new = _exact_c5(h_new)

                    cur_c5 = c5_new
                    stall = 0

    stall = 0

    radius = float(radius)

        if stall > stall_limit or radius < 1e-7:
            break

        grad = np.zeros(n)

                    grad[p:n] += -wr * h[0:m]

            source = int(sources[ii])
            if sink == source:
                continue

            d[source] = -1.0

                else:
                    a, c, fc = c, dd, fd
                    dd = a + gr * (b - a)
                    fd = float(np.max(S0 + dd * T - dd * dd * C) * dx)
            t_opt = (a + b) / 2.0
            
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt

            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            dv = np.zeros(n)
            dv[sink] = 1.0
            dv[source] = -1.0
            T = np.correlate(dv, one_mh, mode='full') - np.correlate(h, dv, mode='full')
            C = np.correlate(dv, dv, mode='full')
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = t_lo, t_hi
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            fc = float(np.max(S0 + c1 * T - c1 * c1 * C) * dx)
            fd = float(np.max(S0 + c2 * T - c2 * c2 * C) * dx)
            for _ in range(28):
                if b - a < 1e-12:
                    break
                if fc < fd:
                    b, c2, fd = c2, c1, fc
                    c1 = b - gr * (b - a)
                    fc = float(np.max(S0 + c1 * T - c1 * c1 * C) * dx)
                else:
                    a, c1, fc = c1, c2, fd
                    c2 = a + gr * (b - a)
                    fd = float(np.max(S0 + c2 * T - c2 * c2 * C) * dx)
            t_opt = (a + b) / 2.0
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try

        if best_step is not None:
            h = best_step
            cur_c5 = best_c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = _compute_corr(h)
            stall = 0
            radius = min(radius * 1.02, 0.22)
        else:
            stall += 1
            radius *= 0.82
            if stall % 25 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _multi_start_grad(h, deadline, n_starts=3, seed=42):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    per_start = (deadline - time.time()) / n_starts
    for s in range(n_starts):
        sub_dl = time.time() + per_start
        if s == 0:
            start_h = h.copy()
        else:
            noise = rng.normal(0, 0.03, n)
            noise -= float(np.sum(noise)) / n
            start_h = _project(h + noise, target)
        bh, bc = _grad_descent(start_h, sub_dl, beta_start=10.0 + s * 10.0,
                               beta_end=100.0 + s * 50.0, n_iters_max=3000, seed=seed + 10 * s)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Gradient descent with multi-start (35% time)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _multi_start_grad(best_h, d1, n_starts=3, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair move optimization (20% time)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _pair_optimize(best_h, d2, radius=0.12, n_pairs=45, beta=20.0, stall_limit=50, seed=seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-n exploration with gradient (20% time)
    d3 = t0 + budget_s * 0.75
    for dn in [-4, 4, -8, 8, -12, 12, 16, -16]:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.018)
        h_r, c5_r = _grad_descent(h_r, sub_dl, beta_start=15.0, beta_end=120.0, n_iters_max=2500, seed=seed + 50 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Alternating gradient + pair (15% time)
    d4 = t0 + budget_s * 0.90
    sub4a = min(d4, time.time() + budget_s * 0.07)
    h4, c5_4 = _grad_descent(best_h, sub4a, beta_start=25.0, beta_end=250.0, n_iters_max=4000, seed=seed + 200)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    sub4b = min(d4, time.time() + budget_s * 0.07)
    h4b, c5_4b = _pair_optimize(best_h, sub4b, radius=0.08, n_pairs=35, beta=25.0, stall_limit=40, seed=seed + 400)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()

    # Phase 5: Final polish (5% time)
    d5 = deadline - 3
    sub5a = min(d5, time.time() + budget_s * 0.025)
    h5, c5_5 = _grad_descent(best_h, sub5a, beta_start=50.0, beta_end=400.0, n_iters_max=3000, seed=seed + 500)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < deadline - 5:
        sub5b = min(d5, time.time() + budget_s * 0.02)
        h5b, c5_5b = _pair_optimize(best_h, sub5b, radius=0.05, n_pairs=25, beta=35.0, stall_limit=25, seed=seed + 600)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
