def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

        best_c5, best_h = c5_bp, h_bp.copy()

    # Phase 6: Final continuous polish (7%)
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, safety, max_iters=3000, radius_init=0.04)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
