def _compute_gradient(h, w):
    """
    Compute the weighted gradient of the correlation objective.
    h: array of length n
    w: array of length 2n-1 (weights for each lag)
    Returns: gradient array of length n
    """
    n = len(h)
    g = np.zeros(n)
    # For each lag k (correlation index r = n-1+k), compute contribution
    for r in range(2*n-1):
        k = r - (n-1)
        if w[r] < 1e-15:
            continue
        # dR_k/dh[i] = (1-h[i+k]) * 1_{0<=i+k<n} - h[i-k] * 1_{0<=i-k<n}
