import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        ka = -k
        g[:n-ka] += (1.0 - h[ka:])
        g[ka:] -= h[:n-ka]
    return g

def _lp_direction(h, corr_scaled, active_idx, radius):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    
    # Variables: [d_0, ..., d_{n-1}, t]
    # c = [0,...,0, 1] minimize t
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    
    # Constraints: dx * grad_r · d - t <= -S_r(h)
    A_ub = np.zeros((m, nv))
    b_ub = np.zeros(m)
    for i, r in enumerate(active_idx):
        k = int(r) - (n - 1)
        g = _grad_row(h, k) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr_scaled[r]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_b = -min(radius, h[i])
        hi_b = min(radius, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, 10.0))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or not hasattr(res, 'x'):
        return None
    d = res.x[:n].copy()
    # Clean up tiny values
    d[np.abs(d) < 1e-12] = 0.0
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    # Ensure zero sum
    d -= np.mean(d)
    return d

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Compute max feasible alpha
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    
    # Evaluate candidates
    candidates = [(opt_t, float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx)]
    candidates.append((0.0, float(np.max(S0)) * dx))
    candidates.append((t_hi, float(np.max(S0 + t_hi*T - t_hi*t_hi*C)) * dx))
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_optimize(h_start, deadline, radius_init=0.08, stall_limit=100):
    n = len(h_start)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h_start.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        if radius < 1e-6:
            break
            
        corr_scaled = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = float(np.max(corr_scaled))
        
        # Select active constraints: those within tolerance of max
        tol = max(0.01, 0.1 * max_val)
        active_mask = corr_scaled >= (max_val - tol)
        active_idx = np.where(active_mask)[0]
        
        # Limit number of active constraints for LP efficiency
        if len(active_idx) > 80:
            # Take top 80
            sorted_idx = np.argsort(-corr_scaled)[:80]
            active_idx = sorted_idx
        
        if len(active_idx) == 0:
            break
            
        d = _lp_direction(h, corr_scaled, active_idx, radius)
        
        if d is None:
            stall += 1
            radius *= 0.7
            continue
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr

def _binary_swap_delta(h, i, j):
    """Compute change in C5 when swapping h[i] (1→0) and h[j] (0→1)."""
    n = len(h)
    dx = 2.0 / n
    # Computing delta efficiently:
    # For each shift k, the correlation changes based on neighbors of i and j
    # But for simplicity with n~180, just recompute
    trial = h.copy()
    trial[i] = 0.0
    trial[j] = 1.0
    new_c5 = _exact_c5(trial)
    return new_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_targeted(h_bin, deadline, max_iters=100000, n_try_per_step=100, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 3000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find the worst correlation and target swaps that might fix it
        max_idx = int(np.argmax(cur_corr))
        
        best_delta = 0.0
        best_move = None
        
        for _ in range(n_try_per_step):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0
            trial[oz] = 1.0
            new_c5 = _exact_c5(trial)
            delta = new_c5 - cur_c5
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_move = (oi, oz, i, j)
        
        if best_move is not None:
            oi, oz, i, j = best_move
            cur[oi] = 0.0
            cur[oz] = 1.0
            cur_c5 += best_delta
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iters=200000, temp_init=0.003, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, _ = _binary_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 5000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5 = _exact_c5(trial)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0005, T * 0.9998)
        else:
            stall += 1
            best_stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 4.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _block_swap(h_bin, deadline, max_iters=20000, seed=77):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        
        length = int(rng.integers(2, min(6, n // 6)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        
        block1 = cur[i:i+length]
        block2 = cur[j:j+length]
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue
        
        trial = cur.copy()
        trial[i:i+length] = block2
        trial[j:j+length] = block1
        new_c5 = _exact_c5(trial)
        
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _pair_swap(h, deadline, n_pairs=10000, radius=0.15, seed=111):
    """Continuous pair swap: move mass from i to j."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5 - 1e-14:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

    h0 = np.array(initial_h_values, dtype=float)
