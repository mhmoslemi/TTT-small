import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x: sum(x) = target, 0 ≤ x ≤ 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _softmax_obj(h, beta):
    """Compute soft-max approximation of max correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    if m < -1e15:
        return m, np.ones(len(corr)) / len(corr)
    shifted = corr - m
    weights = np.exp(beta * shifted)
    s = float(np.sum(weights))
    if s < 1e-300:
        weights = np.ones(len(corr))
        weights[np.argmax(corr)] = len(corr)
        s = float(np.sum(weights))
    weights /= s
    obj = m + np.log(s) / beta
    return obj, weights

def _grad_softmax(h, weights, beta):
    """Gradient of soft-max objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grad = np.zeros(n)
    for k in range(L):
        wk = weights[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[lag:]
        else:
            p = -lag
            m = n - p
            grad[p:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[p:]
    return grad * dx

def _softmax_descend(h, deadline, beta_start=5.0, beta_end=80.0, n_steps=300, seed=42):
    """Projected gradient descent on soft-max approximation"""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    
    # Beta schedule: geometric from beta_start to beta_end
    if n_steps > 1:
        ratio = (beta_end / beta_start) ** (1.0 / n_steps)
    else:
        ratio = 1.0
    
    cur_obj, cur_weights = _softmax_obj(h, beta_start)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        beta = beta_start * (ratio ** step)
        
        # Compute objective and weights
        cur_obj, cur_weights = _softmax_obj(h, beta)
        
        # Compute gradient
        g = _grad_softmax(h, cur_weights, beta)
        
        # Projected gradient step with backtracking
        # First step: try a few step sizes
        g_norm = float(np.dot(g, g))
        if g_norm < 1e-20:
            break
        
        alpha_init = 2.0 / (beta * np.sqrt(g_norm) + 1e-10)
        alpha = min(alpha_init, 0.5)
        
        improved = False
        for _bt in range(20):
            h_try = _project(h - alpha * g, target)
            obj_try, _ = _softmax_obj(h_try, beta)
            if obj_try < cur_obj - 1e-12:
                h = h_try
                cur_obj = obj_try
                improved = True
                break
            alpha *= 0.6
        if not improved:
            # Try smaller
            alpha = 1e-4
            for _bt in range(5):
                h_try = _project(h - alpha * g, target)
                obj_try, _ = _softmax_obj(h_try, beta)
                if obj_try < cur_obj:
                    h = h_try
                    cur_obj = obj_try
                    improved = True
                    break
                alpha *= 0.5

        c5_now = _exact_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _softmax_obj_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = corr - m
    if beta * np.max(shifted) > 500:
        weights = np.zeros(len(corr))
        top = np.where(np.isclose(corr, m, atol=m + 1e-12))[0]
        weights[top] = 1.0
        weights /= len(top)
        return m, weights
    weights = np.exp(beta * shifted)
    s = float(np.sum(weights))
    if s < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
        return m, weights
    weights /= s
    obj = m + np.log(s) / beta
    return obj, weights

def _grad_full(h, weights):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grad = np.zeros(n)
    for k in range(L):
        wk = weights[k]
        if wk < 1e-14:
            continue
        ell = k - (n - 1)
        if ell >= 0:
            m = n - ell
            grad[ell:] += wk * (1.0 - h[:m])

        else:
            p = -ell
            m = n - p
            grad[:m] += wk * (1.0 - h[p:])

def _softmax_descend(h, deadline, beta_start=5.0, beta_end=100.0, n_steps=200, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if n_steps > 1:
        ratio = (beta_end / beta_start) ** (1.0 / n_steps)
    else:
        ratio = 1.0
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        beta = beta_start * (ratio ** step)
        obj, weights = _softmax_obj_weights(h, beta)
        g = _grad_full(h, weights)
        g_norm = float(np.dot(g, g))
        if g_norm < 1e-22:
            break
        alpha = min(2.0 / (beta * np.sqrt(g_norm) + 1e-10), 0.8)
        improved = False
        for _bt in range(25):
            h_try = _project(h - alpha * g, target)
            if np.array_equal(h_try, h):
                alpha *= 0.5
                continue
            c5_try = _exact_c5(h_try)
            obj_try, _ = _softmax_obj_weights(h_try, beta)
            if c5_try < best_c5 - 1e-14 or obj_try < obj - 0.5 * alpha * g_norm:
                h = h_try
                if c5_try < best_c5:

                    best_h = h.copy()
                improved = True
                break

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    n_grid = 50
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    return opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C))

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=42):

    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        ws = float(np.sum(w))
        if ws < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0 / top_k
        else:
            w /= ws
        L = len(S0)
        coord_grad = np.zeros(n)
        for k in range(L):
            wk = w[k]
            if wk < 1e-12:
                continue
            ell = k - (n - 1)
            if ell >= 0:
                m = n - ell
                coord_grad[ell:] += wk * (1.0 - h[:m])
                coord_grad[:m] -= wk * h[ell:]
            else:
                p = -ell
                m = n - p
                coord_grad[:m] += wk * (1.0 - h[p:])
                coord_grad[p:] -= wk * h[:m]
        sort_idx = np.argsort(coord_grad)
        ns = min(n_pairs, n // 2)
        sources = sort_idx[-ns:]
        sinks = sort_idx[:ns]
        pairs = []
        for ii in range(ns):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, max(3, n_pairs // 4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        one_mh = 1.0 - h
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search(S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.75
            if stall % 50 == 0 and radius < 0.02:
                radius = 0.12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):

    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=15000, T_init=0.004, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _pair_swap_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                h_try = cur.copy()
                h_try[oi] = 0.0
                h_try[oz] = 1.0
                c_try = _exact_c5(h_try)
                if c_try < cur_c5 - 1e-14:
                    cur = h_try
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                    improved = True
                    break
    return best_h, best_c5

def _multi_beta_gradient(h, deadline, betas, n_steps_each=40, seed=123):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        for step in range(n_steps_each):
            if time.time() > deadline:
                break
            obj, weights = _softmax_obj_weights(h, beta)
            g = _grad_full(h, weights)
            g_norm = float(np.dot(g, g))
            if g_norm < 1e-22:
                break
            alpha = min(3.0 / (beta * np.sqrt(g_norm) + 1e-10), 1.0)
            improved = False
            for _bt in range(20):
                h_try = _project(h - alpha * g, target)
                if np.array_equal(h_try, h):
                    alpha *= 0.5
                    continue
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
                obj_try, _ = _softmax_obj_weights(h_try, beta)
                if obj_try < obj - 0.3 * alpha * g_norm:
                    h = h_try
                    improved = True
                    c5_try = _exact_c5(h)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                    break
                alpha *= 0.5
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
    else:
        n0 = 180
        x = (np.arange(n0) + 0.5) / n0
        h0 = np.where(x < 0.5, 1.0, 0.0)

    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Soft-max projected gradient at n0 (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _softmax_descend(h_init, d1, beta_start=5.0, beta_end=80.0, n_steps=150, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-beta gradient refinement (15% time)
    d2 = t0 + budget_s * 0.35
    betas_list = [10, 20, 40, 60, 80, 100]
    h2, c5_2 = _multi_beta_gradient(best_h, d2, betas_list, n_steps_each=30, seed=77)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair optimization at n0 (20% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _pair_optimize(best_h, d3, radius=0.18, n_pairs=55, beta=22.0, stall_limit=120, seed=3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-n exploration (20% time)
    d4 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _softmax_descend(h_r, sub_dl, beta_start=8.0, beta_end=60.0, n_steps=60, seed=200+dn)
        sub_dl2 = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.12, n_pairs=40, beta=20.0, stall_limit=50, seed=300+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Binary diversification (10% time)
    d5 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=20000, T_init=0.003, seed=555)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_bin, c5_bin = _pair_swap_sweep(h_bin, sub_dl2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d5, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _pair_optimize(h_pol, sub_dl3, radius=0.10, n_pairs=40, beta=25.0, stall_limit=50, seed=444)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Final intensive refinement (10% time)
    d6 = deadline - 2.0
    n_curr = len(best_h)
    if time.time() < d6:
        # Gradient refinement
        sub_dl = min(d6, time.time() + budget_s * 0.03)
        h6a, c5_6a = _multi_beta_gradient(best_h, sub_dl, [30, 50, 80], n_steps_each=20, seed=666)
        if c5_6a < best_c5:
            best_c5 = c5_6a
            best_h = h6a.copy()
        # Pair optimization
        sub_dl2 = min(d6, time.time() + budget_s * 0.05)
        h6b, c5_6b = _pair_optimize(best_h, sub_dl2, radius=0.08, n_pairs=60, beta=30.0, stall_limit=100, seed=777)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Final projection and verification
    n_final = len(best_h)
    best_h = _project(best_h.copy(), n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, final_c5, n_final
