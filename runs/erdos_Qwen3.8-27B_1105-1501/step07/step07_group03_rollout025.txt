import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lp_direction(h0, raw, n_active, radius, target):
    """Find LP direction that minimizes max of top-K linearized lag values."""
    n = len(h0)
    # Select top-K lags
    top_idx = np.argsort(-raw)[:n_active]
    
    # Build A matrix for linearized constraints
    # Each row corresponds to one active lag
    # A[r, j] = coefficient of x[j] in the linearized constraint for lag r
    
    A_rows = []
    b_offsets = []  # f(h0) - grad·h0 for each row
    
    for r in top_idx:
        d = r - (n - 1)
        if d >= 0:
            i_max = n - d
        else:
            i_max = n + d  # which is n - |d|
        
        i_vals = np.arange(i_max)
        j1 = i_vals  # indices for x[i]
        j2 = i_vals + d  # indices for x[i+d]
        
        # grad contribution from x[i]: (1 - h0[i+d])
        # grad contribution from x[i+d]: -h0[i]
        grad = np.zeros(n)
        grad[j1] += (1.0 - h0[j2])
        grad[j2] += -h0[j1]
        
        A_rows.append(grad.copy())
        b_offsets.append(raw[r] - float(grad @ h0))
    
    A_mat = np.array(A_rows)  # shape (K, n)
    b_off = np.array(b_offsets)
    
    # LP: minimize t
    # subject to: A_mat @ x - t <= -b_off  (i.e., A_mat @ x + t <= t_old... wait)
    # Actually: f(h0) + grad·(x-h0) <= t
    # grad·x - grad·h0 <= t - f(h0)
    # grad·x - t <= grad·h0 - f(h0) = -b_off... 
    # wait: b_off = f(h0) - grad·h0
    # So grad·h0 - f(h0) = -b_off
    # Constraint: grad·x - t <= -b_off
    # i.e., [grad, -1] @ [x, t] <= -b_off
    
    # Variables: [x_0, ..., x_{n-1}, t]  (n+1 vars)
    # Objective: minimize t => c = [0, ..., 0, 1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # For each active lag k: A_mat[k] @ x - t <= -b_off[k]
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, -1] = -1.0
    b_ub = -b_off.copy()
    
    # Trust region: |x_i - h0_i| <= radius
    # x_i - h0_i <= radius   =>  e_i @ x <= h0_i + radius
    # -(x_i - h0_i) <= radius  =>  -e_i @ x <= -(h0_i - radius) = -h0_i + radius
    
    # Equality: sum x = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= x_i <= 1, t >= 0 (or t >= some lower bound)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    # Add trust region constraints to A_ub
    # Actually, to keep it simpler, just use bounds for x and add trust as extra constraints
    # Let me add trust region as additional inequality constraints
    # (x_i - h0_i) <= radius for each i: e_i @ x <= h0_i + radius
    # This is 2n more constraints. For n=180, that's 360 extra rows.
    # Total: n_active + 360 ≈ 400 rows, 181 cols. Should be fine.
    
    # Actually, let me just use bounds for the trust region too:
    # max(0, h0_i - radius) <= x_i <= min(1, h0_i + radius)
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h0[i] - radius)
        hi_i = min(1.0, h0[i] + radius)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            x_lp = res.x[:n]
            # Ensure feasibility
            x_lp = np.clip(x_lp, 0.0, 1.0)
            x_lp = _project(x_lp, target)
            return x_lp, res.x[-1]
        else:
            return None, None
    except:
        return None, None

def _line_search_exact(h, d, raw, dx, a_max):
    """Exact 1D line search along direction d, using the true objective."""
    n = len(h)
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    # Compute T and Q arrays for the quadratic approximation
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # a_max is the maximum step before hitting bounds
    candidates = [0.0, a_max]
    
    # Add golden section candidates
    n_grid = 40
    grid = np.linspace(0.0, a_max, n_grid)
    candidates.extend(grid.tolist())
    
    # Also add quadratic-optimal points for top lags
    top_idx = np.argsort(-raw)[:15]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 0 < av < a_max:
                candidates.append(av)
    
    for alpha in candidates:
        alpha = max(0.0, min(alpha, a_max))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    
    # Golden section refinement
    step = a_max / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_max, best_alpha + step)
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(40):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    
    return best_alpha, best_val

def _max_step(h, d, eps=1e-10):
    """Find maximum step alpha such that 0 <= h + alpha*d <= 1."""
    a_max = 1e10
    pos = d > eps
    neg = d < -eps
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
    return min(a_max, 1.0)

def _gradient_direction(h, raw, n_lags, radius, target):
    """Fallback gradient-based direction using soft max of top lags."""
    n = len(h)
    top = np.argsort(-raw)[:n_lags]
    mx = raw[top[0]]
    beta = 20.0
    vp = np.exp(beta * (raw[top] - mx))
    w = vp / float(np.sum(vp))
    
    g = np.zeros(n)
    for k in range(len(top)):
        r = int(top[k])
        d = r - (n - 1)
        m = min(n, n - abs(d))
        if d >= 0:
            g[:m] += w[k] * (1.0 - h[d:d+m])
            g[d:d+m] -= w[k] * h[:m]
        else:
            gp = -d
            g[gp:] += w[k] * (1.0 - h[:m])
            g[:m] -= w[k] * h[gp:]
    
    g -= np.mean(g)  # preserve sum
    nm = np.linalg.norm(g)
    if nm < 1e-15:
        return None
    return g / nm

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _local_optimize(h, deadline, max_iter=200, n_active=12, radius_init=0.2, 
                    use_lp=True, seed=42):
    """Local optimizer combining LP directions and gradient directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    raw = _raw_corr(h)
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(seed)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        # Update raw
        raw = _raw_corr(h)
        
        # Try LP direction first
        direction = None
        if use_lp and radius > 0.02:
            x_lp, t_val = _lp_direction(h, raw, n_active, radius, target)
            if x_lp is not None:
                d = x_lp - h
                # Also zero-mean to preserve sum (in case projection shifted it)
                d = d - np.mean(d)
                nm = np.linalg.norm(d)
                if nm > 1e-12:
                    direction = d / nm
                    # Scale so that max step is within radius
                    a_max_lp = radius / nm * 1.0  # rough
                    a_max_lp = min(a_max_lp, _max_step(h, direction))
                    direction_data = (direction, a_max_lp, 'lp')
        
        if direction is not None:
            d_dir, a_max_dir, src = direction_data
        else:
            # Gradient direction
            gdir = _gradient_direction(h, raw, n_active, radius, target)
            if gdir is None:
                stall += 1
                continue
            a_max_dir = min(radius, _max_step(h, gdir))
            d_dir = gdir
            direction_data = (gdir, a_max_dir, 'grad')
        
        # Line search
        a_max_final = min(a_max_dir, _max_step(h, d_dir))
        if a_max_final < 1e-10:
            stall += 1
            continue
        
        alpha, c5_try = _line_search_exact(h, d_dir, raw, dx, a_max_final)
        
        if alpha < 1e-10:
            stall += 1
            continue
            
        h_new = h + alpha * d_dir
        h_new = _project(h_new, target)
        actual_c5 = _exact_c5(h_new)
        
        if actual_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
            # Occasionally shrink n_active to focus more
            n_active = max(6, n_active - (1 if it % 10 == 0 else 0))
        else:
            stall += 1
            radius *= 0.8
            n_active = min(20, n_active + 1)
        
    return best_h, best_c5

def _pair_swap(h, deadline, n_trials=2000, seed=111):
    """Random pair swap with line search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for _ in range(n_trials):
        if time.time() > deadline or stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = _raw_corr(h)
        a_max = min(1.0, _max_step(h, d))
        alpha, c5_try = _line_search_exact(h, d, raw, dx, a_max)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    beta = 25.0

        # Fall back to gradient
        if direction is None:
            gdir = _gradient_direction(h, raw, n_active)
            if gdir is not None:
                direction = gdir
                a_max_dir = min(radius, _max_step(h, direction))
        if direction is None or a_max_dir < 1e-10:
            stall += 1
            radius *= 0.85
            continue
        alpha, c5_try = _line_search_exact(h, direction, raw, dx, a_max_dir)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.85
            continue
        h_new = h + alpha * direction
        h_new = _project(h_new, target)
        actual_c5 = _exact_c5(h_new)
        if actual_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.82
    return best_h, best_c5

def _pair_swap(h, deadline, n_trials=2000, seed=111):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline or stall > 120:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        raw = _raw_corr(h)
        a_max = min(1.0, _max_step(h, d))
        alpha, c5_try = _line_search_exact(h, d, raw, dx, a_max)
        if alpha < 1e-10:
            stall += 1
            continue
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        actual = _exact_c5(h_new)
        if actual < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 350:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP-based optimization (22%)
    d1 = t0 + budget_s * 0.22
    h1, c5_1 = _local_optimize(h_init, d1, max_iter=200, n_active=10, radius_init=0.25, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary + polish (13%)
    d2 = t0 + budget_s * 0.35
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.05)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.002)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_p = _local_optimize(h_pol, sub_dl, max_iter=100, n_active=12, radius_init=0.15, seed=seed + 1)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offsets = list(range(-14, 15, 2))
    for dn in offsets:
        if time.time() > d3 - 4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 220 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _local_optimize(h_r, sub_dl, max_iter=60, n_active=10, radius_init=0.20, seed=seed + dn)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=12000, T_init=0.0015)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
        if c5_r < best_c5 + 0.001 and time.time() < d3 - 2:
            sub_dl3 = min(d3, time.time() + budget_s * 0.007)
            h_r, c5_r = _local_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3,
                                         max_iter=50, n_active=12, radius_init=0.10, seed=seed + dn + 100)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive optimization (25%)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    for attempt in range(5):
        if time.time() > d4 - 3:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4 = _project(best_h.copy(), n_curr / 2.0)
        na = 10 + attempt * 2
        rad = max(0.05, 0.18 - attempt * 0.025)
        h4, c5_4 = _local_optimize(h4, sub_dl, max_iter=100, n_active=na, radius_init=rad, seed=seed + 200 + attempt)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.035)
        h4, c5_4 = _pair_swap(h4, sub_dl, n_trials=4000, seed=seed + 500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary round in phase 4
    if time.time() < d4 - 3:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=25000, T_init=0.001)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        if c5_b < best_c5 + 0.0005 and time.time() < d4 - 2:
            h_pol = _project(h_b.copy(), len(h_b) / 2.0)
            sub_dl2 = min(d4, time.time() + budget_s * 0.02)
            h_pol, c5_p = _local_optimize(h_pol, sub_dl2, max_iter=60, n_active=14,
                                           radius_init=0.08, seed=seed + 600)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()

    # Phase 5: Basin hopping (15%)
    d5 = deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d5 - 4 and n_restarts < 100:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 2, time.time() + budget_s * 0.007)
        h_t, c5_t = _local_optimize(h_pert, sub_dl, max_iter=50, n_active=12,
                                     radius_init=0.10, seed=seed + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.80)
        else:
            scale = min(0.05, scale * 1.06)

    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _local_optimize(h6, deadline, max_iter=60, n_active=16,
                                    radius_init=0.04, seed=seed + 999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap(h6, deadline, n_trials=2000, seed=seed + 888)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
