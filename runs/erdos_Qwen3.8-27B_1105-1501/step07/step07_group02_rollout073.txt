    M = float(R.max())

    sw = float(np.sum(exp_R))
    if sw < 1e-300:
        top = np.argmax(R)
        w = np.zeros(2*n-1)
        w[top] = 1.0
    else:
        w = exp_R / sw
    g = _grad_from_weights(h, w)
    return g, R, w

def _active_grad(h, top_k=15):
    n = len(h)
    R = _all_R(h)
    L = 2*n - 1
    idx = np.argsort(R)[-top_k:]
    w = np.zeros(L)
    peak = float(R[idx].max())
    spread = float(R[idx].max() - R[idx].min())
    if spread < 1e-15:
        w[idx] = 1.0 / len(idx)
    else:
        tw = (R[idx] - float(R[idx].min())) / spread
        tw = tw / float(np.sum(tw))
        w[idx] = tw
    g = _grad_from_weights(h, w)
    return g, R, w

def _gradient_descent(h_init, deadline, beta_init=25.0, alpha_init=0.4,
                      momentum=0.6, max_iter=15000, stall_limit=400, seed=42):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = alpha_init
    v = np.zeros(n)
    stall = 0
    use_active = 0
    rng = np.random.default_rng(seed)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        if it % 3 == 0:
            g, R, w = _active_grad(h, top_k=12)
        else:
            g, R, w = _smooth_grad(h, beta_init)
        g_proj = g - float(np.mean(g))
        v = momentum * v + (1.0 - momentum) * g_proj
        best_step = None
        best_c5_try = cur_c5
        for _ in range(10):
            step = -alpha * v
            h_try = _project(h + step, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_step = h_try
                alpha = min(alpha * 1.15, 3.0)
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        if best_step is not None:
            h = best_step
            cur_c5 = best_c5_try
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            if stall % 30 == 29:
                pert = rng.normal(0, 0.01, n)
                pert -= pert.mean()
                h = _project(h + pert, target)
                cur_c5 = _exact_c5(h)
        else:
            stall += 1
            alpha = min(alpha * 1.3, 2.0)
            if stall % 50 == 0:
                pert = rng.normal(0, 0.03, n)
                pert -= pert.mean()
                h = _project(h + pert, target)
                cur_c5 = _exact_c5(h)
                v *= 0.3
    return best_h, best_c5

def _line_search_pair(h, sink, source, S0, T, C, t_lo, t_hi):
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None])**2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = float(ts[max(0, best_idx-2)])
    hi = float(ts[min(n_grid-1, best_idx+2)])
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))
    f2 = float(np.max(S0 + c2*T - c2*c2*C))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C))
    return opt_t, min(opt_val, float(vals[best_idx]))

def _pair_optimize(h, deadline, radius=0.15, n_pairs=50, beta=20.0,
                   stall_limit=150, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = _all_R(h)
    radius = float(radius)
    stall = 0
    rng = np.random.default_rng(seed)
    one_mh = 1.0 - h
    
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        sw = float(np.sum(w))
        if sw < 1e-30:
            top_k = max(1, len(S0)//10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0/top_k
        else:
            w /= sw
        G = np.zeros((2*n-1, n))
        for k in range(-(n-1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                G[row, k:] += (1.0 - h[:m])
                G[row, :m] -= h[k:]
            else:
                p = -k
                m = n - p
                G[row, p:] += (1.0 - h[:m])
                G[row, :m] -= h[p:]
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        sources = sort_idx[-min(n_pairs, n//2):]
        sinks = sort_idx[:min(n_pairs, n//2)]
        pairs = []
        for ii in range(min(n_pairs, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(15, max(3, n_pairs//4))):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source]-1.0, -radius)
            t_hi = min(1.0-h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            t_opt, _ = _line_search_pair(h, sink, source, S0, T, C, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = _all_R(h)
            one_mh = 1.0 - h
            stall = 0
            radius = min(radius * 1.02, 0.3)
        else:
            stall += 1
            radius *= 0.78
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.10
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=15000, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.002
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_h, best_c5

def _block_optimize(h, deadline, block_size=3, n_blocks=30, seed=123):
    """Try moving small blocks of adjacent coordinates."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    S0 = _all_R(h)
    
    for _ in range(2000):
        if time.time() > deadline:
            break
        if cur_c5 < best_c5 - 0.01:
            break
        # Pick a source block and sink block
        s_start = int(rng.integers(0, n - block_size + 1))
        t_start = int(rng.integers(0, n - block_size + 1))
        if abs(s_start - t_start) < block_size:
            continue
        # Compute direction
        d = np.zeros(n)
        d[t_start:t_start+block_size] = 1.0
        d[s_start:s_start+block_size] = -1.0
        d_norm = float(np.sum(d**2))
        if d_norm < 1e-10:
            continue
        d = d / math.sqrt(d_norm)
        # Determine max step from box constraints
        max_step = 0.2
        for i in range(t_start, t_start+block_size):
            max_step = min(max_step, 1.0 - h[i])
        for i in range(s_start, s_start+block_size):
            max_step = min(max_step, h[i])
        if max_step < 1e-10:
            continue
        one_mh = 1.0 - h
        T_dir = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        C_dir = np.correlate(d, d, mode='full')
        t_lo = -max_step * 0.5
        t_hi = max_step
        t_opt, _ = _line_search_pair(h, -1, -1, S0, T_dir, C_dir, t_lo, t_hi)
        h_try = _project(h + t_opt * d, target)
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            S0 = _all_R(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    
    # Get parent
    h0_np = np.array(initial_h_values, dtype=float)
    n0 = len(h0_np)
    h_init = _project(h0_np.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Gradient descent from parent (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _gradient_descent(h_init, d1, beta_init=25.0, alpha_init=0.4,
                                 momentum=0.6, max_iter=20000, stall_limit=400, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Pair-based refinement (25% time)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _pair_optimize(best_h, d2, radius=0.18, n_pairs=60, beta=20.0,
                              stall_limit=200, seed=100)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Multi-n exploration with gradient (25% time)
    d3 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, 24, -24]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _gradient_descent(h_r, sub_dl, beta_init=20.0, alpha_init=0.3,
                                      momentum=0.5, max_iter=5000, stall_limit=150, seed=200+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
        # Quick pair refinement
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.012)
            h_r2, c5_r2 = _pair_optimize(best_h, sub_dl2, radius=0.12, n_pairs=40,
                                         beta=22.0, stall_limit=60, seed=300+dn)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
    
    # Phase 4: Block optimization + gradient (10% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    if time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _block_optimize(best_h, sub_dl, block_size=3, n_blocks=40, seed=500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + budget_s * 0.03)
            h4b, c5_4b = _gradient_descent(best_h, sub_dl2, beta_init=30.0,
                                           alpha_init=0.25, momentum=0.7,
                                           max_iter=8000, stall_limit=200, seed=600)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()
    
    # Phase 5: Binary diversification (5% time)
    d5 = t0 + budget_s * 0.95
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _to_binary(best_h, n_curr)
        c5_bin = _exact_c5(h_bin)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=20000, seed=888)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish binary with gradient
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d5, time.time() + budget_s * 0.015)
            h_pol, c5_pol = _gradient_descent(h_pol, sub_dl2, beta_init=20.0,
                                              alpha_init=0.2, momentum=0.5,
                                              max_iter=3000, stall_limit=80, seed=777)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 6: Final intensive refinement (5% time)
    n_curr = len(best_h)
    sub_dl = deadline - 1.0
    if sub_dl > time.time() + 5:
        h6, c5_6 = _pair_optimize(best_h, sub_dl, radius=0.10, n_pairs=50,
                                   beta=25.0, stall_limit=100, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        h6b, c5_6b = _gradient_descent(best_h, deadline - 0.5, beta_init=35.0,
                                       alpha_init=0.15, momentum=0.8,
                                       max_iter=5000, stall_limit=80, seed=1000)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
    
    # Final projection and verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Something went wrong with projection, use previous
        pass
    best_c5 = final_c5
    return best_h, best_c5, len(best_h)
