import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] -= h[g:]
    return G

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 0.999999
        else: mask = x > 1e-7
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, G, t_cap=1.0):
    S0 = _scores(h)
    T = G @ d
    P = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    t_max = t_cap
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    t_max = max(0.0, min(t_max, t_cap))
    if t_max < 1e-12:
        return 0.0, float(np.max(S0))
    n_grid = 60
    ts = np.linspace(0.0, t_max, n_grid)
    vals = np.array([float(np.max(S0 + t*T - t*t*P)) for t in ts])
    best_i = int(np.argmin(vals))
    lo_g = ts[max(0, best_i - 2)]
    hi_g = ts[min(n_grid - 1, best_i + 2)]
    if hi_g - lo_g < 1e-10:
        t_opt = ts[best_i]
        v_opt = vals[best_i]
    else:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_g - gr * (hi_g - lo_g)
        c2 = lo_g + gr * (hi_g - lo_g)
        f1 = float(np.max(S0 + c1*T - c1*c1*P))
        f2 = float(np.max(S0 + c2*T - c2*c2*P))
        for _ in range(50):
            if hi_g - lo_g < 1e-13: break
            if f1 < f2:
                hi_g = c2; c2 = c1; f2 = f1
                c1 = hi_g - gr * (hi_g - lo_g)
                f1 = float(np.max(S0 + c1*T - c1*c1*P))
            else:
                lo_g = c1; c1 = c2; f1 = f2
                c2 = lo_g + gr * (hi_g - lo_g)
                f2 = float(np.max(S0 + c2*T - c2*c2*P))
        t_opt = (lo_g + hi_g) / 2.0
        v_opt = float(np.max(S0 + t_opt*T - t_opt*t_opt*P))
    best_t, best_v = 0.0, float(np.max(S0))
    if v_opt < best_v: best_t, best_v = t_opt, v_opt
    v_end = float(np.max(S0 + t_max*T - t_max*t_max*P))
    if v_end < best_v: best_t, best_v = t_max, v_end
    return best_t, best_v
