def smooth_max_obj(x, n, beta):
    # x is the anti-palindromic half, length m = n/2
    m = n // 2
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[m-1::-1]
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')
    # stable log-sum-exp
    mx = corr.max()
    z = beta * (corr - mx)
    z = np.clip(z, -50, 0)
    lse = np.log(np.sum(np.exp(z)))
    return (mx + lse / beta)  # smooth max of corr

def smooth_max_grad(x, n, beta):
    m = n // 2
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[m-1::-1]
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')
    mx = corr.max()
    z = beta * (corr - mx)
    z = np.clip(z, -50, 0)
    w = np.exp(z)
    wsum = w.sum()
    w /= wsum
    
    # Compute gradient of weighted sum of corr[r] w.r.t. h
    G = np.zeros(n)
    for r in range(len(corr)):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)  # lag
        if k >= 0:
            # c[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[i]: from h[i+k] term: (1-h[i-k]) for k <= i <= n-1
            #          from (1-h[i]) term: -h[i+k] for 0 <= i <= n-1-k
            for i in range(n):
                contrib = 0.0
                if k <= i <= n-1:  # h[i+k] = h[i], i.e., n+k=i means n=i-k

                    # valid if 0 <= i-k <= n-1-k

                    if 0 <= ny <= n-1-k:

                if 0 <= i <= n-1-k:
                    contrib -= h[i+k]
                G[i] += w[r] * contrib
        else:

            # c[k] = sum_{i=d}^{n-1} h[i-d]*(1-h[i])
            for i in range(n):
                contrib = 0.0
                # h[i-d] appears when "i" in sum is i+d
                if 0 <= i <= n-1-d:

                    if d <= ny <= n-1:
                        contrib += (1.0 - h[ny])

                    contrib -= h[i-d]
                G[i] += w[r] * contrib

    # So h[j] = x[j] for j < m
    # h[m+j] = 1 - x[m-1-j] for j = 0..m-1
    # d/dx[i]: contributes to h[i] (for i < m) and h[m + (m-1-i)] = h[2m-1-i] = h[n-1-i]
    # For h[n-1-i], the derivative of (1-x[m-1-i]) w.r.t. x[i] is -1
    gx = np.zeros(m)
    for i in range(m):
        # h[i] depends on x[i] with coefficient +1
        gx[i] = G[i]
        # h[n-1-i] depends on x[i] with coefficient -1 (since h[n-1-i] = 1-x[m-1-i] = 1-x[i])

        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search_full(h, d, radius)
        if c_try < cur_c5 - 1e-14 and abs(alpha) > 1e-15:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Anti-palindromic L-BFGS-B (20% time)
    d1 = t0 + budget_s * 0.20
    x_anti = _to_anti(h_init)
    h_ant, c5_ant = _lbfgs_anti(x_anti, n0, d1, max_stages=8)
    if c5_ant < best_c5:
        best_c5 = c5_ant
        best_h = h_ant.copy()

    # Phase 2: Full-space continuous optimization (25%)
    d2 = t0 + budget_s * 0.45
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    dl_a = min(d2, time.time() + budget_s * 0.08)
    h2, c5_2 = _opt_full(h2, dl_a, beta_lo=6.0, beta_hi=50.0, radius_init=0.18, max_stall=50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    if time.time() < d2:
        dl_b = min(d2, time.time() + budget_s * 0.08)
        h2b, c5_2b = _pair_continuous(best_h, dl_b, n_iter=5000, radius=0.15, seed=99)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()
    if time.time() < d2:
        dl_c = min(d2, time.time() + budget_s * 0.06)
        h2c, c5_2c = _opt_full(best_h, dl_c, beta_lo=15.0, beta_hi=100.0, radius_init=0.08, max_stall=40)
        if c5_2c < best_c5:
            best_c5 = c5_2c
            best_h = h2c.copy()

    # Phase 3: Multi-n exploration (15%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    for dn in range(-12, 13, 3):
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 40 or n_try > 260:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        # Try anti-palindromic optimization
        x_r = _to_anti(h_r)
        h_ar, c5_ar = _lbfgs_anti(x_r, n_try, sub_dl, max_stages=5)
        if c5_ar < best_c5:
            best_c5 = c5_ar
            best_h = h_ar.copy()
            n_curr = n_try
        # Also try full-space
        if time.time() < d3 - 2:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_fr, c5_fr = _opt_full(h_ar, sub_dl2, beta_lo=10.0, beta_hi=60.0, radius_init=0.10, max_stall=30)
            if c5_fr < best_c5:
                best_c5 = c5_fr
                best_h = h_fr.copy()
                n_curr = n_try

    # Phase 4: Binary optimization (20%)
    d4 = t0 + budget_s * 0.80
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    c5_b_init = _exact_c5(h_b)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=60000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d4:
        sub_dl2 = min(d4, time.time() + budget_s * 0.05)
        h_b2, c5_b2 = _binary_swap_opt(best_h, sub_dl2, max_rounds=30)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
    if time.time() < d4:
        sub_dl3 = min(d4, time.time() + budget_s * 0.05)
        h_b3, c5_b3 = _binary_sa(best_h, sub_dl3, max_iter=40000, T_init=0.002)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
    # Polish binary result with continuous
    if time.time() < d4:
        n_bf = len(best_h)
        h_pol = _project(best_h.copy(), n_bf / 2.0)
        sub_dl4 = min(d4, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _opt_full(h_pol, sub_dl4, beta_lo=20.0, beta_hi=100.0, radius_init=0.06, max_stall=30)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Basin hopping + final (20%)
    d5 = deadline
    scale = 0.015
    n_restarts = 0
    while time.time() < d5 - 10 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.012)
        h_t, c5_t = _opt_full(h_p, sub_dl, beta_lo=10.0, beta_hi=60.0, radius_init=0.08, max_stall=25)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.1)

    # Final polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_full(h_f, deadline, beta_lo=20.0, beta_hi=100.0, radius_init=0.05, max_stall=35)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if deadline - time.time() > 5:
        x_f = _to_anti(best_h)
        h_af, c5_af = _lbfgs_anti(x_f, len(best_h), deadline, max_stages=4)
        if c5_af < best_c5:
            best_c5 = c5_af
            best_h = h_af.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
