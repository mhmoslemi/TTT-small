To reduce S_m*, we want to increase h where g < 0 and decrease h where g > 0.
So a good pair is (i, j) where g[i] < 0 and g[j] > 0 (increase i, decrease j) or vice versa.

Actually, for the pair direction d = e_i - e_j (increase h_i, decrease h_j), the change in S_m* is t * (g_i[m*] - g_j[m*]). To decrease S_m*, we need g_i[m*] - g_j[m*] < 0, i.e., g_i[m*] < g_j[m*].

So: find i with small g_i[m*] and j with large g_j[m*], and try the pair (i, j).

def _project(v, target):
    # Same as before - project v to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # Upper index: k+s = i, k = i-s, valid for i in [s, n-1]
        # Contribution: (1 - h[i-s])
        g[s:] += (1.0 - h[:n-s])

        # Lower index: k = i, k+s = i-d, valid for i in [d, n-1]
        g[d:] += -h[:n-d]
