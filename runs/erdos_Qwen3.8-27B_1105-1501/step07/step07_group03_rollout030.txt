Now let me think about the line search more carefully. Given a direction d (with sum(d)=0), we want to find the best alpha:

new_h = h + alpha*d, projected to [0,1] and sum = target.

But projecting changes the direction. A simpler approach: find the max alpha such that h + alpha*d stays in [0,1], then do the line search on the exact objective.

The exact objective along the line:
f(alpha) = max_r [raw[r] + alpha*L_r(d) - alpha^2*Q_r(d)]

where Q_r(d) = sum of product terms > 0 always.

So f(alpha) is the max of concave quadratic functions, which is concave. We can use ternary search or golden section search.

Wait, actually max of concave functions is not necessarily concave. Let me reconsider.

Each individual g_r(alpha) = raw[r] + alpha*L_r(d) - alpha^2*Q_r(d) is a concave quadratic (opening downward since Q_r(d) >= 0). The max of concave functions is not in general concave. However, in practice it often is, and we can use a robust line search.

Actually, for the line search, the simplest robust approach is:
1. Find alpha_max (where h + alpha*d first hits a boundary)
2. Sample f at several points in [0, alpha_max]
3. Use golden section or similar to refine

Since f is the max of quadratics, it's piecewise quadratic, so we can find the exact minimum by checking all breakpoints. But that's O(n^2) which is too slow. A practical approach is to use the sampling + refinement.

import numpy as np
import time
import cvxpy as cp

def _raw_correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_raw_correlate(h)) * dx)

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_coefficients(h, n, r):
    """Compute c_j such that L_r(d) = raw[r] + sum_j c_j * d_j"""
    d_offset = r - (n - 1)
    c = np.zeros(n)
    if d_offset >= 0:
        m = n - d_offset
        if m <= 0:
            return c
        # First sum: d[i]*(1-h[i+d_offset]) for i=0..m-1
        c[:m] += (1.0 - h[d_offset:d_offset+m])
        # Second sum: h[i]*d[i+d_offset] for i=0..m-1, contributes to c[j] for j=d_offset..n-1
        c[d_offset:] -= h[:m]
    else:
        g = -d_offset
        m = n - g
        if m <= 0:
            return c
        # First sum: d[i]*(1-h[i-g]) for i=g..n-1
        c[g:] += (1.0 - h[:m])
        # Second sum: h[i]*d[i-g] for i=g..n-1, j=i-g, contributes to c[j] for j=0..m-1
        c[:m] -= h[g:]
    return c

def _line_search(h, d, raw, dx, alpha_max=None, n_samples=50):
    """Line search along h + alpha*d, return best alpha and value"""
    n = len(h)
    if alpha_max is None:
        alpha_max = 1.0
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
        alpha_max = max(alpha_max, 1e-12)
    
    # Precompute L and Q coefficients for efficiency
    # f(alpha) = max_r [raw[r] + alpha*L_r - alpha^2*Q_r]
    # where L_r = sum c_j * d_j, Q_r = sum of the quadratic coefficient
    
    # For efficiency, precompute L for all r
    n_corr = 2 * n - 1
    L = np.zeros(n_corr)
    Q = np.zeros(n_corr)
    for r in range(n_corr):
        c = _compute_coefficients(h, n, r)
        L[r] = float(np.dot(c, d))
        # Q_r = sum of d[i]*d[i+d_offset] terms (the quadratic coefficient, positive)
        d_offset = r - (n - 1)
        if d_offset >= 0:
            m = n - d_offset
            if m > 0:
                Q[r] = float(np.dot(d[:m], d[d_offset:d_offset+m]))
        else:
            g = -d_offset
            m = n - g
            if m > 0:
                Q[r] = float(np.dot(d[g:], d[:m]))
    
    # f(alpha) = max_r(raw + alpha*L - alpha^2*Q)
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    
    # Sample
    alphas = np.linspace(0, alpha_max, n_samples)
    vals = np.zeros(len(alphas))
    for i, a in enumerate(alphas):
        vals[i] = float(np.max(raw + a * L - a * a * Q)) * dx
    best_idx = np.argmin(vals)
    best_alpha = alphas[best_idx]
    best_val = vals[best_idx]
    
    # Refine with golden section
    lo = alphas[max(0, best_idx - 1)]
    hi = alphas[min(len(alphas)-1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(raw + c1 * L - c1 * c1 * Q)) * dx
    f2 = float(np.max(raw + c2 * L - c2 * c2 * Q)) * dx
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(raw + c1 * L - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(raw + c2 * L - c2 * c2 * Q)) * dx
    opt = (lo + hi) / 2.0
    val_opt = float(np.max(raw + opt * L - opt * opt * Q)) * dx
    if val_opt < best_val:
        best_val = val_opt
        best_alpha = opt
    
    return best_alpha, best_val

def _lp_direction(h, raw, n_top_lags=15, max_norm=0.5):
    """Use LP to find a direction that lowers the top lags simultaneously."""
    n = len(h)
    n_corr = 2 * n - 1
    
    # Select top lags
    top_idx = np.argsort(-raw)[:n_top_lags]
    
    # Compute coefficients for selected lags
    C = np.zeros((n_top_lags, n))
    raw_sel = np.zeros(n_top_lags)
    for i, r in enumerate(top_idx):
        C[i] = _compute_coefficients(h, n, int(r))
        raw_sel[i] = raw[r]
    
    # LP: minimize t
    # s.t. raw_sel[i] + C[i] @ d <= t  for all i
    #      sum(d) = 0
    #      -h <= d <= 1-h
    #      ||d|| <= max_norm (approximate with box or just rely on box)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -h,
        d_var <= 1.0 - h,
        cp.norm(d_var, 2) <= max_norm,
    ]
    for i in range(n_top_lags):
        constraints.append(raw_sel[i] + C[i] @ d_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except:
            return None
    
    if d_var.value is None:
        return None
    
    d_out = np.array(d_var.value).flatten()
    d_out -= np.mean(d_out)  # Ensure zero sum (in case of numerical issues)
    return d_out

def _gradient_direction(h, raw, beta=20.0):
    """Softmax-weighted gradient direction."""
    n = len(h)
    mx = float(np.max(raw))
    shifted = raw - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    
    # Compute gradient of max_r raw[r] (approximated by weighted sum)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-12:
            continue
        d_offset = r - (n - 1)
        if d_offset >= 0:
            m = n - d_offset
            if m > 0:
                g[:m] += wr * (1.0 - h[d_offset:d_offset+m])
                g[d_offset:] -= wr * h[:m]
        else:
            gp = -d_offset
            m = n - gp
            if m > 0:
                g[gp:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[gp:]
    g -= np.mean(g)
    return -g / (np.linalg.norm(g) + 1e-30)

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    """Simulated annealing on binary sequences."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep_binary(h_bin, deadline, max_rounds=10):
    """Exhaustive pair swap for binary."""
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                cur[oi] = 1.0
                cur[oz] = 0.0
                if c_try < cur_c5 - 1e-15:
                    cur[oi] = 0.0
                    cur[oz] = 1.0
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    improved = True
                    ones[oi] = -1  # mark as removed
                    break
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _local_optimize(h, deadline, use_lp=True, beta=20.0, max_iter=200, lp_interval=3):
    """Local optimization combining LP and gradient directions."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    max_norm = 0.3
    
    for iteration in range(max_iter):
        if time.time() > deadline or stall > 80:
            break
        
        raw = _raw_correlate(h)
        improved = False
        
        # LP direction (every lp_interval steps or when stalled)
        if use_lp and (stall >= 5 or iteration % lp_interval == 0):
            d_lp = _lp_direction(h, raw, n_top_lags=15, max_norm=max_norm)
            if d_lp is not None and np.linalg.norm(d_lp) > 1e-8:
                alpha, val = _line_search(h, d_lp, raw, dx)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d_lp, target)

        d_var <= 1.0 - h - 1e-10,
        cp.norm(d_var, 2) <= max_norm,
    ]
    for i in range(k):
        constraints.append(raw_sel[i] + C[i] @ d_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=20000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000)
        except Exception:
            return None
    if d_var.value is None:
        return None
    d_out = np.array(d_var.value).flatten().astype(float)
    d_out[d_out > 1e-12] = np.minimum(d_out[d_out > 1e-12], 1e-12 * 0 + 1.0)
    d_out -= np.mean(d_out)
    if np.linalg.norm(d_out) < 1e-10:
        return None
    return d_out

def _gradient_direction(h, raw, beta=20.0):
    n = len(h)
    mx = float(np.max(raw))
    shifted = raw - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-12:
            continue
        d_offset = r - (n - 1)
        if d_offset >= 0:
            m = n - d_offset
            if m > 0:
                g[:m] += wr * (1.0 - h[d_offset:d_offset + m])
                g[d_offset:] -= wr * h[:m]
        else:
            gp = -d_offset
            m = n - gp
            if m > 0:
                g[gp:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[gp:]
    g -= np.mean(g)
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return np.zeros(n)
    return -g / nm

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9994
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep_binary(h_bin, deadline, max_rounds=12):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                cur[oi] = 1.0
                cur[oz] = 0.0
                if c_try < cur_c5 - 1e-15:
                    cur[oi] = 0.0
                    cur[oz] = 1.0
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    improved = True
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones = list(np.where(cur > 0.5)[0])
                    zeros = list(np.where(cur <= 0.5)[0])
                    break
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.12 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _local_optimize(h, deadline, beta=20.0, lp_interval=3, max_norm_init=0.30,
                    max_iter=250, use_lp=True, lp_frac=0.92):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    max_norm = max_norm_init
    rng_local = np.random.default_rng(12345)
    for iteration in range(max_iter):
        if time.time() > deadline or stall > 100:
            break
        raw = _raw_correlate(h)
        improved = False
        use_lp_this = use_lp and (stall >= 4 or iteration % lp_interval == 0)
        if use_lp_this:
            d_lp = _lp_direction(h, raw, n_top=22, max_norm=max_norm, fraction=lp_frac)
            if d_lp is not None:
                alpha, val = _line_search(h, d_lp, raw, dx)
                if alpha > 1e-10:
                    h_new = _project(h + alpha * d_lp, target)
                    actual = _exact_c5(h_new)
                    if actual < cur_c5 - 1e-13:
                        h = h_new
                        cur_c5 = actual
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        max_norm = min(max_norm * 1.06, 0.50)
                        improved = True
        if not improved:
            d_grad = _gradient_direction(h, raw, beta=beta)
            alpha, val = _line_search(h, d_grad, raw, dx, alpha_max=0.30)
            if alpha > 1e-10:
                h_new = _project(h + alpha * d_grad, target)
                actual = _exact_c5(h_new)
                if actual < cur_c5 - 1e-13:
                    h = h_new
                    cur_c5 = actual
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
        if not improved:
            if iteration % 7 == 3:
                i = int(rng_local.integers(0, n))
                j = int(rng_local.integers(0, n))
                if i != j:
                    d_pair = np.zeros(n)
                    d_pair[i] = 1.0
                    d_pair[j] = -1.0
                    alpha, val = _line_search(h, d_pair, raw, dx, alpha_max=0.25)
                    if alpha > 1e-10:
                        h_new = _project(h + alpha * d_pair, target)
                        actual = _exact_c5(h_new)
                        if actual < cur_c5 - 1e-13:
                            h = h_new
                            cur_c5 = actual
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            stall = 0
                            improved = True
        if not improved:
            stall += 1
            max_norm = max(0.12, max_norm * 0.92)
            if stall % 25 == 0 and max_norm < 0.22:
                max_norm = 0.35
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    # Phase 1: LP+gradient local optimization (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _local_optimize(h_init, d1, beta=20.0, lp_interval=3,
                               max_norm_init=0.30, max_iter=300, use_lp=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: Binary refinement (14%)
    d2 = t0 + budget_s * 0.32
    h_bin = _to_binary(best_h)
    sub_dl = min(d2, time.time() + budget_s * 0.06)
    h_bin, c5_b = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.003)
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_b = _pair_sweep_binary(h_bin, sub_dl, max_rounds=14)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_bin.copy()
    if time.time() < d2 - 2:
        h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_pol, c5_p = _local_optimize(h_pol, sub_dl, beta=18.0, lp_interval=2,
                                      max_norm_init=0.25, max_iter=120, use_lp=True)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()
    # Phase 3: Multi-n exploration (18%)
    d3 = t0 + budget_s * 0.50
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in range(-24, 25, 2):
        if time.time() > d3 - 4:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 280 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.007)
        h_r, c5_r = _local_optimize(h_r, sub_dl, beta=18.0, lp_interval=4,
                                    max_norm_init=0.28, max_iter=80, use_lp=True)
        if c5_r < best_c5 + 0.002 and time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.004)
            h_b, c5_b2 = _binary_sa(h_b, sub_dl2, max_iter=12000, T_init=0.002)
            if c5_b2 < c5_r:
                c5_r = c5_b2
                h_r = h_b
            if c5_r < best_c5 + 0.002 and time.time() < d3 - 2:
                sub_dl3 = min(d3, time.time() + budget_s * 0.004)
                h_r, c5_r = _local_optimize(_project(h_r.copy(), n_try / 2.0), sub_dl3,
                                            beta=22.0, lp_interval=2, max_norm_init=0.20,
                                            max_iter=60, use_lp=True)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    # Phase 4: Intensive LP optimization (22%)
    d4 = t0 + budget_s * 0.72
    n_curr = len(best_h)
    for rnd in range(4):
        if time.time() > d4 - 3:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.045)
        h4 = _project(best_h.copy(), n_curr / 2.0)
        beta_r = 15.0 + rnd * 8.0
        frac_r = 0.90 + rnd * 0.015
        h4, c5_4 = _local_optimize(h4, sub_dl, beta=beta_r, lp_interval=2,
                                   max_norm_init=0.25 - rnd * 0.03, max_iter=200,
                                   use_lp=True, lp_frac=frac_r)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4 - 4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4 - 2, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=35000, T_init=0.0015)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep_binary(h_b, sub_dl2, max_rounds=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4 - 2:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.025)
        h4, c5_4 = _local_optimize(h4, sub_dl, beta=30.0, lp_interval=1,
                                   max_norm_init=0.15, max_iter=120, use_lp=True, lp_frac=0.88)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Phase 5: Basin hopping (12%)
    d5 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 - 4 and n_restarts < 200:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 2, time.time() + budget_s * 0.0035)
        h_t, c5_t = _local_optimize(h_pert, sub_dl, beta=22.0, lp_interval=4,
                                    max_norm_init=0.18, max_iter=50, use_lp=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.78)
        else:
            scale = min(0.10, scale * 1.12)
    # Final polish
    if deadline - time.time() > 4:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _local_optimize(h6, deadline, beta=35.0, lp_interval=1,
                                   max_norm_init=0.10, max_iter=80, use_lp=True, lp_frac=0.85)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if deadline - time.time() > 2:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _local_optimize(h6, deadline, beta=40.0, lp_interval=2,
                                   max_norm_init=0.08, max_iter=40, use_lp=False)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
