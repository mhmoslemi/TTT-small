def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _thue_morse(n):
    """Generate Thue-Morse sequence of length n (first n bits)."""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        c = 0
        x = i
        while x:
            c ^= 1
            x >>= 1
        tm[i] = float(c)
    # Balance: if not exactly n/2 ones, adjust

def _sa_binary(h, deadline, max_iter=500000, T_init=0.001, T_min=0.00001):
    """Simulated annealing on binary sequences with multiple move types."""
    n = len(h)
    cur = h.copy()
    cur_max = _max_overlap(cur)
    best = cur.copy()
    best_max = cur_max
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    
    ones = list(np.where(cur == 1.0)[0])
    zeros = list(np.where(cur == 0.0)[0])
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        if T < T_min:
            break
        
        # Choose move type
        r = rng.random()
        if r < 0.4:
            # Single swap
            if not ones or not zeros:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            h_new = cur.copy()
            h_new[zi] = 1.0
            h_new[oj] = 0.0
        elif r < 0.7:
            # Block reversal
            if n < 4:
                break
            length = int(rng.integers(2, min(n, 20)))
            start = int(rng.integers(0, n - length))
            h_new = cur.copy()
            h_new[start:start+length] = cur[start:start+length][::-1]
        elif r < 0.85:
            # Block move
            if n < 6:
                break
            length = int(rng.integers(2, min(n // 3, 15)))
            start = int(rng.integers(0, n - length))
            # Move block to another position
            # Choose destination not overlapping
            remaining = n - length
            if remaining <= 0:
                continue
            dest = int(rng.integers(0, remaining))
            if dest == start:
                continue
            h_new = cur.copy()
            block = cur[start:start+length].copy()
            # Remove block
            h_new2 = np.delete(cur, range(start, start+length))
            # Insert at dest
            h_new = np.insert(h_new2, dest, block)
        else:
            # Double swap (two 0s with two 1s)
            if len(ones) < 2 or len(zeros) < 2:
                break
            i1 = int(rng.integers(0, len(zeros)))
            i2 = int(rng.integers(0, len(zeros) - 1))
            i2 = (i2 + (1 if i2 >= i1 else 0)) % (len(zeros) - 1) if i2 == i1 else i2
            j1 = int(rng.integers(0, len(ones)))
            j2 = int(rng.integers(0, len(ones) - 1))
            j2 = (j2 + (1 if j2 >= j1 else 0)) % (len(ones) - 1) if j2 == j1 else j2
            zi1, zi2 = zeros[i1], zeros[i2]
            oj1, oj2 = ones[j1], ones[j2]
            h_new = cur.copy()
            h_new[zi1] = 1.0
            h_new[zi2] = 1.0
            h_new[oj1] = 0.0
            h_new[oj2] = 0.0
        
        new_max = _max_overlap(h_new)
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur = h_new
            cur_max = new_max
            # Update ones/zeros lists

            zeros = list(np.where(cur == 0.0)[0])

                best_max = cur_max
                best = cur.copy()
                stall = 0
            T = max(T * 0.9998, T_min)
        else:
            stall += 1
        
    return best, best_max

    """Create a fractal-like binary pattern."""
