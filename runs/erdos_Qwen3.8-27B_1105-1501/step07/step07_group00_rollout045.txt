def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

def _grad_rows(h, lags):
    """Compute gradient rows for given lags.
    For lag d >= 0: R = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
    dR/dh[j+d] += 1-h[j], dR/dh[j] -= h[j+d]
    For lag d < 0 (d = -g, g>0): R = sum_{j=0}^{n-1-g} h[j]*(1-h[j+g])
    dR/dh[j] += 1-h[j+g], dR/dh[j+g] -= h[j]
    """
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:d+m] += 1.0 - h[:m]
                G[idx, :m] -= h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += 1.0 - h[g:g+m]
                G[idx, g:g+m] -= h[:m]

def _line_search_exact(h, d, dx, t_max):
    """Exact line search along h + t*d.
    The score along the line is quadratic: S(t) = S0 + t*T - t^2*C
    We want to minimize max_k S_k(t) * dx.
    """
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Determine valid range
    t_lo = 0.0
    t_hi = t_max
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section search on max(S0 + t*T - t^2*C)

    lo = t_lo
    hi = t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(50):

            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    
    # Also check endpoints
    f0 = float(np.max(S0)) * dx
    fhi = float(np.max(S0 + t_hi * T - t_hi*t_hi * C)) * dx

    if f0 < best_v:

    if fhi < best_v:

def _fw_step(h, scores, beta=20.0):
    """Frank-Wolfo direction: v - h where v is the minimizing vertex."""
    n = len(h)
    target = n // 2
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    
    # FW: find vertex v that minimizes g^T v subject to 0<=v<=1, sum(v)=n/2
    # v[i] = 1 for the n/2 indices with smallest g[i]
    idx = np.argsort(g)[:target]
    v = np.zeros(n)
    v[idx] = 1.0
    return v - h  # direction

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    """Active-set LP direction."""
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=20.0, radius=0.1):
    """Weighted gradient direction (fallback)."""
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    
    # Scale to fit in radius
    max_abs = float(np.max(np.abs(g)))
    if max_abs > 0:
        scale = min(1.0, radius / max_abs)
        g *= scale
    return g

def _local_opt(h, deadline, radius_init=0.12, max_iters=5000, use_fw=False):
    """Local optimization with active-set LP and optional FW steps."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 12
    fw_count = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = _scores(h)
        
        # Try active-set LP first
        d = _active_lp(h, scores, radius, n_active=n_active, gap=gap)
        
        # If LP fails or gives tiny direction, try FW or gradient
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            if use_fw and fw_count < 20:
                d = _fw_step(h, scores, beta=15.0)
                # Scale FW direction to fit radius
                max_abs = float(np.max(np.abs(d)))
                if max_abs > radius:
                    d *= radius / max_abs
                fw_count += 1
            else:
                d = _gradient_direction(h, scores, beta=20.0, radius=radius)
                if d is None or float(np.max(np.abs(d))) < 1e-12:
                    stall += 1
                    radius *= 0.7
                    gap *= 1.4
                    n_active = min(n_active + 3, 30)
                    if radius < 1e-6:
                        break
                    continue
        
        alpha, new_c5_pred = _line_search_exact(h, d, dx, radius)
        
        if new_c5_pred < cur_c5 - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.3)
                gap = max(gap * 0.92, 0.0005)
                n_active = max(n_active - 1, 8)
            else:
                stall += 1
                radius *= 0.75
                gap *= 1.3
                n_active = min(n_active + 2, 25)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.3
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=150):
    """Random pair swap optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        improved = False
        for _ in range(max(1, n // 4)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c_try = _line_search_exact(h, d, dx, 0.5)
            if c_try < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                new_c5 = _exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amplitudes = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _fw_polish(h, deadline, max_iters=200):
    """Frank-Wolfo polynomial for escaping local optima."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = 15.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        d = _fw_step(h, scores, beta=beta)
        
        if float(np.max(np.abs(d))) < 1e-8:
            break
        
        alpha, c_pred = _line_search_exact(h, d, dx, 1.0)
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            beta = max(beta * 0.95, 5.0)
        else:
            beta *= 1.1
            if beta > 50.0:
                break
    
    return best_h, best_c5
