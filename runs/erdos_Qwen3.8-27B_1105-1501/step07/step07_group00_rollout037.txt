def _full_lp_direction(h, scores, radius):
    """Solve LP with ALL lag constraints to get a descent direction."""
    n = len(h)
    G, _ = _full_grad_matrix(h)
    m = len(scores)
    # minimize tau
    # s.t. G_k^T d - tau <= -scores[k] for all k  (model: scores[k] + G_k^T d <= tau)
    #      sum(d) = 0
    #      -radius <= d[i] <= radius (and respecting bounds)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores  # G_k^T d - tau <= -scores[k] => scores[k] + G_k^T d <= tau
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True,
                      'time_limit': 5})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _softmax_gradient(h, scores, beta):
    """Compute gradient of soft-max approximation."""
    shifted = beta * (scores - float(np.max(scores)))
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w = w / sw
    G, _ = _full_grad_matrix(h)
    g = G.T @ w
    return g

def _line_search(h, d, dx, radius):

    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Coarse grid
    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)

        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 3)]

            g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)

            g -= float(np.mean(g))
            d = g

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        else:
            stall += 1
            radius *= 0.75
            beta *= 1.2

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 3
    freqs = rng.uniform(0.5, 6.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amplitudes = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _gen_binary_seed(n, rng, mode=0):
    """Generate a diverse binary seed."""
    if mode == 0:
        # Random balanced
        idx = rng.permutation(n)[:n // 2]
        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 1:
        # Alternating with noise
        h = np.zeros(n)
        for i in range(0, n, 2):
            h[i] = 1.0
        # Randomly flip some
        n_flips = n // 10
        flip_idx = rng.choice(n, size=n_flips, replace=False)
        for i in flip_idx:
            h[i] = 1.0 - h[i]
        # Rebalance
        ones = int(np.sum(h))
        target = n // 2
        if ones > target:
            excess = ones - target
            one_positions = np.where(h > 0.5)[0]
            to_flip = rng.choice(one_positions, size=excess, replace=False)
            h[to_flip] = 0.0
        elif ones < target:
            deficit = target - ones
            zero_positions = np.where(h <= 0.5)[0]
            to_flip = rng.choice(zero_positions, size=deficit, replace=False)
            h[to_flip] = 1.0
    elif mode == 2:
        # Block-based
        block = max(2, n // 20)
        h = np.zeros(n)
        for i in range(0, n, block):
            if (i // block) % 2 == 0:
                h[i:i+block] = 1.0
        # Noise
        for i in range(n):
            if rng.random() < 0.1:
                h[i] = 1.0 - h[i]
        # Rebalance
        ones = int(np.sum(h))
        target = n // 2
        if ones > target:
            excess = ones - target
            one_positions = np.where(h > 0.5)[0]
            to_flip = rng.choice(one_positions, size=excess, replace=False)
            h[to_flip] = 0.0
        elif ones < target:
            deficit = target - ones
            zero_positions = np.where(h <= 0.5)[0]
            to_flip = rng.choice(zero_positions, size=deficit, replace=False)
            h[to_flip] = 1.0
    else:
        # Threshold-based with sinusoidal
        x = np.linspace(0, 2, n)
        val = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2*np.pi))
        val += rng.normal(0, 0.1, n)
        idx = np.argsort(val)[-n // 2:]
        h = np.zeros(n)
        h[idx] = 1.0
    return h

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

            m = n - g

def _lp_direction(h, scores, radius):
    n = len(h)
    G, _ = _grad_matrix(h)
    m = len(scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            if np.max(np.abs(d)) > 1e-14:
                return d
    except Exception:
        pass
    return None

def _softmax_grad(h, scores, beta):
    n = len(h)
    G, _ = _grad_matrix(h)
    mx = float(np.max(scores))
    shifted = beta * (scores - mx)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:

        w[int(np.argmax(scores))] = 1.0

    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    cands = [(opt_t, opt_val), (t_lo, float(np.max(S0)) * dx),
             (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx),
             (grid_t[best_idx], grid_f[best_idx])]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _full_local_opt(h, deadline, radius_init=0.08, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    beta = 25.0
    prev_d = None
    momentum = 0.3

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        d = _lp_direction(h, scores, radius)
        if d is None:
            g = _softmax_grad(h, scores, beta)
            d = _clip_dir(g, h, radius)
            if float(np.max(np.abs(d))) < 1e-13:
                if prev_d is not None and float(np.max(np.abs(prev_d))) > 1e-13:
                    d = prev_d * 0.5
                else:
                    stall += 1
                    radius *= 0.7
                    beta *= 1.4
                    if radius < 1e-7:
                        break
                    continue
        else:
            if prev_d is not None and float(np.max(np.abs(prev_d))) > 1e-13:
                norm_prev = np.linalg.norm(prev_d)
                norm_new = np.linalg.norm(d)
                if norm_new > 1e-14 and norm_prev > 1e-14:
                    d = (1.0 - momentum) * d + momentum * prev_d * (norm_new / norm_prev)
            prev_d = d.copy()
            d = _clip_dir(d, h, radius)
            if float(np.max(np.abs(d))) < 1e-13:
                stall += 1
                radius *= 0.75
                continue

        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.3)
            beta = max(beta * 0.95, 8.0)
        else:
            stall += 1
            radius *= 0.72
            beta *= 1.3
            momentum = max(0.1, momentum * 0.85)
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.06
                momentum = 0.3

    return best_h, best_c5

def _binary_descent(h_bin, deadline, max_rounds=3000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99)
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]

    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(150, max(20, len(ones) * len(zeros) // 8))
        for _ in range(n_tries):
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            zi, oj = zeros[i], ones[j]
            h_new = cur.copy()
            h_new[zi] = 1.0
            h_new[oj] = 0.0
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                cur = h_new
                cur_c5 = new_c5
                ones = np.where(cur > 0.5)[0]
                zeros = np.where(cur <= 0.5)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 60 and len(zeros) >= 2 and len(ones) >= 2:
                i1 = int(rng.integers(0, len(zeros)))
                i2 = int(rng.integers(0, len(zeros)))
                j1 = int(rng.integers(0, len(ones)))
                j2 = int(rng.integers(0, len(ones)))
                if i1 != i2 and j1 != j2:
                    h_new = cur.copy()
                    h_new[zeros[i1]] = 1.0
                    h_new[zeros[i2]] = 1.0
                    h_new[ones[j1]] = 0.0
                    h_new[ones[j2]] = 0.0
                    new_c5 = _exact_c5(h_new)
                    if new_c5 < cur_c5 - 1e-14:
                        cur = h_new
                        cur_c5 = new_c5
                        ones = np.where(cur > 0.5)[0]
                        zeros = np.where(cur <= 0.5)[0]
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = cur.copy()
                        stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 4:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 4
    freqs = rng.uniform(0.3, 7.0, size=n_freqs)
    phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
    amps = rng.uniform(0.3, 1.0, size=n_freqs)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _gen_binary_seed(n, rng, mode=0):
    if mode == 0:
        idx = rng.permutation(n)[:n // 2]
        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        for i in range(0, n, 2):
            h[i] = 1.0
        n_flips = max(1, n // 8)
        flip_idx = rng.choice(n, size=n_flips, replace=False)
        for i in flip_idx:
            h[i] = 1.0 - h[i]
        _rebalance_bin(h, n, rng)
    elif mode == 2:
        block = max(2, n // 15)
        h = np.zeros(n)
        for i in range(0, n, block):
            if (i // block) % 2 == 0:
                h[i:i + block] = 1.0
        for i in range(n):
            if rng.random() < 0.12:
                h[i] = 1.0 - h[i]
        _rebalance_bin(h, n, rng)
    elif mode == 3:
        x = np.linspace(0, 2, n)
        val = 0.5 + 0.35 * np.sin(2 * np.pi * rng.uniform(1.0, 4.0) * x + rng.uniform(0, 2 * np.pi))
        val += 0.2 * np.sin(2 * np.pi * rng.uniform(5.0, 10.0) * x + rng.uniform(0, 2 * np.pi))
        val += rng.normal(0, 0.08, n)
        idx = np.argsort(val)[-n // 2:]
        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 4:
        x = np.linspace(0, 2, n)
        period = n // int(rng.uniform(2, 5))
        val = (np.sin(2 * np.pi * x / (2.0 * period / n)) > 0).astype(float)
        for i in range(n):
            if rng.random() < 0.1:
                val[i] = 1.0 - val[i]
        _rebalance_bin(val, n, rng)
    else:
        h = np.zeros(n)
        skip = int(rng.uniform(2, 4))
        for i in range(0, n, skip):
            h[i:i + skip // 2] = 1.0
        _rebalance_bin(h, n, rng)
    return h

def _rebalance_bin(h, n, rng):
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        excess = ones - target
        one_pos = np.where(h > 0.5)[0]
        to_flip = rng.choice(one_pos, size=excess, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        deficit = target - ones
        zero_pos = np.where(h <= 0.5)[0]
        to_flip = rng.choice(zero_pos, size=deficit, replace=False)
        h[to_flip] = 1.0

def _binary_sa(h_bin, deadline, max_iter=20000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Full-lag LP local opt from parent (18% time)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _full_local_opt(h_init, d1, radius_init=0.08, max_iters=4000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary exploration (15%)
    d2 = t0 + budget_s * 0.33
    n_b = len(best_h)
    best_bin_c5 = _exact_c5(_to_binary(best_h))
    best_bin_h = _to_binary(best_h)

    h_b, c5_b = _binary_descent(_to_binary(best_h), min(d2, time.time() + budget_s * 0.025), max_rounds=800)
    if c5_b < best_bin_c5:
        best_bin_c5 = c5_b
        best_bin_h = h_b.copy()

    for mode in range(6):
        if time.time() > d2 - 5:
            break
        h_seed = _gen_binary_seed(n_b, rng, mode=mode)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_s, c5_s = _binary_descent(h_seed, sub_dl, max_rounds=500)
        if c5_s < best_bin_c5:
            best_bin_c5 = c5_s
            best_bin_h = h_s.copy()

    # Polish best binary with continuous LP
    if time.time() < d2 - 2:
        h_pol = _project(best_bin_h.copy(), n_b / 2.0)
        sub_dl = min(d2, time.time() + budget_s * 0.02)
        h_pol, c5_pol = _full_local_opt(h_pol, sub_dl, radius_init=0.05, max_iters=400)
        if c5_pol < best_bin_c5:
            best_bin_c5 = c5_pol
            best_bin_h = h_pol.copy()

    if best_bin_c5 < best_c5:
        best_c5 = best_bin_c5
        best_h = best_bin_h.copy()

    # Phase 3: Multi-resolution (12%)
    d3 = t0 + budget_s * 0.45
    n_curr = len(best_h)
    tried_ns = set()
    for dn in list(range(-12, 13, 3)):
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 220 or n_try in tried_ns or n_try % 2 != 0:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.008)
        h_r, c5_r = _full_local_opt(h_r, sub_dl, radius_init=0.07, max_iters=500)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Deep full LP optimization (22%)
    d4 = t0 + budget_s * 0.67
    h4, c5_4 = _full_local_opt(best_h.copy(), d4, radius_init=0.05, max_iters=8000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary SA + basin hopping (20%)
    d5 = t0 + budget_s * 0.87
    # Binary SA on current best
    n_sa = len(best_h)
    h_sa = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h_sa, c5_sa = _binary_sa(h_sa, sub_dl, max_iter=30000, T_init=0.002)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Polish SA result
    if c5_sa < best_c5 + 0.001 and time.time() < d5 - 10:
        h_sp = _project(h_sa.copy(), n_sa / 2.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_sp, c5_sp = _full_local_opt(h_sp, sub_dl2, radius_init=0.04, max_iters=300)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()

    # Basin hopping
    scale = 0.012
    n_restarts = 0
    while time.time() < d5 and n_restarts < 120:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _full_local_opt(h_pert, sub_dl, radius_init=0.04, max_iters=250)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.83)
        else:
            scale = min(0.06, scale * 1.07)

    # Phase 6: Final deep polish (11%)
    d6 = t0 + budget_s * 0.99
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    h_f, c5_f = _full_local_opt(h_f, d6, radius_init=0.03, max_iters=5000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # One more tiny binary check
    if deadline - time.time() > 15:
        h_bf = _to_binary(best_h)
        h_bf, c5_bf = _binary_descent(h_bf, deadline - 2, max_rounds=500)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    final_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(final_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return final_h, best_c5, len(final_h)
