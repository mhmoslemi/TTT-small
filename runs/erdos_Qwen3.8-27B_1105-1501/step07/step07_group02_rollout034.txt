    shifted = beta * (O - O_max)
    s_max = float(np.max(shifted))
    exp_shifted = np.exp(shifted - s_max)
    log_sum = s_max + np.log(np.sum(exp_shifted))
    F = O_max + log_sum / beta
    weights = exp_shifted / np.sum(exp_shifted)
    s_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    M = weights[j_idx - s_idx + (n - 1)]
    omh = 1.0 - h
    T1 = M @ omh
    T2 = M.T @ h
    grad = dx * (T1 - T2)
    return F, grad

def _smooth_descent(h_init, deadline, beta_schedule=None, momentum=0.9, max_iter=300):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    vel = np.zeros(n)
    if beta_schedule is None:
        beta_schedule = [8, 15, 30, 60, 120, 250, 500]
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        lr = min(0.01, 2.0 / (beta * 0.5 + 1.0))
        it_stall = 0
        for it in range(max_iter):
            if time.time() > deadline:
                break
            if it_stall > 60:
                break
            F, grad = _softmax_grad(h, beta)
            grad -= np.mean(grad)
            vel = momentum * vel - lr * grad
            h_try = _project(h + vel, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                it_stall = 0
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.7
                vel *= 0.5
                it_stall += 1
    h_final = _project(h, target)
    c5_final = _exact_c5(h_final)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    return best_h, best_c5

def _dense_coord_search(h, deadline, max_rounds=50, radius=0.1):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            for delta in [radius, -radius, radius*0.5, -radius*0.5]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                h_try = h.copy()
                h_try[i] += delta
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-14:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            radius *= 0.7
            if radius < 0.01:
                break
    return best_h, best_c5

def _pair_optimize(h, deadline, radius=0.15, n_pairs=40, beta=20.0, stall_limit=80, seed=42):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(seed)
    one_mh = 1.0 - h
    while time.time() < deadline:
        if stall > stall_limit or radius < 1e-6:
            break
        M = float(np.max(S0))
        w = np.exp(beta * (S0 - M))
        ws = float(np.sum(w))
        if ws < 1e-30:
            top_k = max(1, len(S0) // 10)
            idx = np.argsort(S0)[-top_k:]
            w = np.zeros(len(S0))
            w[idx] = 1.0 / top_k
        else:
            w /= ws
        G = np.zeros((2 * n - 1, n))
        for k in range(-(n - 1), n):
            row = k + n - 1
            if k >= 0:
                m = n - k
                G[row, k:] += (1.0 - h[:m])
                G[row, :m] -= h[k:]
            else:
                p = -k
                m = n - p
                G[row, p:] += (1.0 - h[:m])
                G[row, :m] -= h[p:]
        coord_grad = G.T @ w
        sort_idx = np.argsort(coord_grad)
        sources = sort_idx[-min(n_pairs, n // 2):]
        sinks = sort_idx[:min(n_pairs, n // 2)]
        pairs = []
        for ii in range(min(n_pairs, len(sinks), len(sources))):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(min(10, n_pairs // 4)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_step = None
        best_c5_try = cur_c5
        for (sink, source) in pairs:
            if time.time() > deadline:
                break
            if sink == source:
                continue
            t_lo = max(-h[sink], h[source] - 1.0, -radius)
            t_hi = min(1.0 - h[sink], h[source], radius)
            if t_hi - t_lo < 1e-13:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            T = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
            C = np.correlate(d, d, mode='full')
            ts = np.linspace(t_lo, t_hi, 30)
            vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
            best_idx = int(np.argmin(vals))
            h_try = h.copy()
            h_try[sink] += ts[best_idx]
            h_try[source] -= ts[best_idx]
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            S0 = np.correlate(h, 1.0 - h, mode='full')
            one_mh = 1.0 - h
            stall = 0
            radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.78
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=15000, T_init=0.002, seed=999):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        h_new = cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _structured_start(n, rng, kind=0):
    h = np.zeros(n)
    target = n // 2
    if kind == 0:
        period = max(4, n // 15)
        for i in range(n):
            if (i % period) < period // 2:
                h[i] = 1.0
        h = _project(h + rng.normal(0, 0.3, n), target)
    elif kind == 1:
        block = max(2, n // 20)
        for i in range(n):
            if (i // block) % 2 == 0:
                h[i] = 1.0
        h = _project(h + rng.normal(0, 0.2, n), target)
    elif kind == 2:
        h = rng.random(n)
        h = _project(h, target)
    else:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x * 7 / 2)
        h = _project(h + rng.normal(0, 0.1, n), target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Soft-max gradient descent on parent (30% time)
    d1 = t0 + budget_s * 0.30
    h_smooth, c5_smooth = _smooth_descent(
        h_init, d1,
        beta_schedule=[10, 20, 40, 80, 160, 320, 600],
        momentum=0.85, max_iter=200)
    if c5_smooth < best_c5:
        best_c5 = c5_smooth
        best_h = h_smooth.copy()

    # Phase 2: Pair optimization refinement (20% time)
    d2 = t0 + budget_s * 0.50
    h_pair, c5_pair = _pair_optimize(
        best_h.copy(), d2, radius=0.15, n_pairs
        =50, beta=25.0, stall_limit=100, seed=1)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair.copy()

    # Phase 3: Multi-resolution exploration (20% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, 20, -20]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 80 or n_try > 220 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _smooth_descent(h_r, sub_dl, beta_schedule=[15, 40, 100, 250],
                                    momentum=0.85, max_iter=80)
        sub_dl2 = min(d3, time.time() + budget_s * 0.010)
        h_r, c5_r = _pair_optimize(h_r, sub_dl2, radius=0.12, n_pairs=35,
                                   beta=20.0, stall_limit=40, seed=200+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Dense coordinate search (10% time)
    d4 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    if time.time() < d4:
        h_dense, c5_dense = _dense_coord_search(best_h.copy(), d4, max_rounds=30, radius=0.08)
        if c5_dense < best_c5:
            best_c5 = c5_dense
            best_h = h_dense.copy()

    # Phase 5: Binary diversification + recovery (10% time)
    d5 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    if time.time() < d5:
        h_bin = _make_binary(best_h, n_curr)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_bin, c5_bin = _binary_local(h_bin, sub_dl, max_iter=20000, T_init=0.002, seed=555)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time.time() < d5:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl2 = min(d5, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _smooth_descent(h_pol, sub_dl2, beta_schedule=[30, 80, 200],
                                            momentum=0.85, max_iter=60)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Final intensive refinement (10% time)
    d6 = deadline - 3
    n_curr = len(best_h)
    h_final = _project(best_h.copy(), n_curr / 2.0)
    sub_dl = min(d6, deadline)
    h_final, c5_final = _pair_optimize(
        h_final, sub_dl, radius=0.08, n_pairs=50, beta=30.0, stall_limit=80, seed=777)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    # Try a couple nearby n values at the end
    n_curr = len(best_h)
    for dn in [-4, 4, 8, -8]:
        if time.time() > deadline - 5:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 220 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        if time.time() > deadline - 5:
            break
        h_r, c5_r = _pair_optimize(h_r, min(deadline - 3, time.time() + budget_s * 0.012),
                                   radius=0.06, n_pairs=35, beta=25.0, stall_limit=30, seed=900+dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
