    best_c5 = cur_c5

    """Compute delta in C[n-1+d] when h[p] changes.
    If p_to_zero=True (1→0), else (0→1).
    d is the lag (can be positive, negative, or zero).
    """
    if d == 0:
        # C[n-1] = sum h[i]*(1-h[i]) = 0 for binary, stays 0
        return 0.0
    if d > 0:
        # C[n-1+d] = sum_{i=0}^{n-1-d} (1-h[i])*h[i+d]
        if p_to_zero:  # h[p]: 1→0
            delta = 0.0

                delta -= (1.0 - h[p - d])  # h[p] was 1, now 0: removes (1-h[p-d])
            return delta

            if p <= n - 1 - d:

            if p >= d:
                delta += (1.0 - h[p - d])  # h[p] was 0, now 1: adds (1-h[p-d])
            return delta

        if p_to_zero:  # h[p]: 1→0

def _binary_local_search(h_bin, deadline, top_k=5, max_passes=2000):

    zeros = np.where(h <= 0.5)[0].tolist()

        if time.time() > deadline:
            break
        C = np.correlate(h, 1.0 - h, mode='full')
        cur_max_raw = float(np.max(C))
        # top_k lags
        top_idx = np.argsort(C)[-top_k:]
        top_lags = top_idx - (n - 1)
        cur_top_vals = C[top_idx].copy()

        best_new_max = cur_max_raw
        
        for p in ones:
            # Precompute delta_p for each top lag
            dp = np.array([_delta_lag(h, p, d, n, True) for d in top_lags])
            for q in zeros:
                if q == p:
                    continue

                new_top = cur_top_vals + dp + dq
                new_max = float(np.max(new_top))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (p, q)
        
        if best_swap is not None and best_new_max < cur_max_raw - 1e-10:
            p, q = best_swap
            h[p] = 0.0
            h[q] = 1.0
            ones.remove(p)
            ones.append(q)
            zeros.remove(q)
            zeros.append(p)
            # Verify
            new_C = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_C))
            if new_max < cur_max_raw - 1e-10:
                cur_max_raw = new_max

                    best_h = h.copy()

                h[p] = 1.0
                h[q] = 0.0
                ones.remove(q)
                ones.append(p)
                zeros.remove(p)
                zeros.append(q)
                break
        else:
            break
    
    return best_h, best_c5

            h[p] = 0.0

            C[r_max] += delta_max

                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
            else:
                h[p] = 1.0
                h[q] = 0.0
                ones.remove(q)
                ones.append(p)
                zeros.remove(p)
                zeros.append(q)
                break
        else:
            break
    return best_h, best_c5

def _binary_sa_fast(h_bin, deadline, T_init=0.4, max_iter=300000, seed=7777):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    C = _exact_corr(h)
    cur_max_raw = float(np.max(C))
    best_h = h.copy()
    best_c5 = cur_max_raw * dx
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500 or len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[int(rng.integers(0, len(ones)))]
        q = zeros[int(rng.integers(0, len(zeros)))]
        if p == q:
            continue
        r_max = int(np.argmax(C))
        d_max = r_max - (n - 1)
        dp = _delta_lag(h, p, d_max, n, True)
        dq = _delta_lag(h, q, d_max, n, False)
        delta = dp + dq
        if delta < -1e-10:
            accept = True
        elif rng.random() < np.exp(-max(delta, 0.0) / max(T, 0.05)):
            accept = True
        else:
            accept = False
        if accept:
            h[p] = 0.0
            h[q] = 1.0
            if it % 20 == 0:
                C = _exact_corr(h)
                cur_max_raw = float(np.max(C))
            else:
                C[r_max] += delta
                cur_max_raw = float(np.max(C))
            pi = ones.index(p)
            qi = zeros.index(q)
            ones[pi] = q
            zeros[qi] = p
            if cur_max_raw * dx < best_c5 - 1e-14:
                best_c5 = cur_max_raw * dx
                best_h = h.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
            T *= 0.9999
    return best_h, best_c5

def _grad_smooth(h, beta):
    n = len(h)
    u = 1.0 - h
    raw = _exact_corr(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        r_max = int(np.argmax(raw))
        g = np.zeros(n)
        d = r_max - (n - 1)
        if d >= 0:
            m = n - d
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:d + m]
        else:
            dd = -d
            m = n - dd
            g[:m] += (1.0 - h[dd:])
            g[dd:] -= h[:m]
        g -= float(np.mean(g))
        return g
    w /= wsum
    g = np.zeros(n)
    for r in range(len(raw)):
        if w[r] < 1e-8:
            continue
        d = r - (n - 1)
        wr = w[r]
        if d >= 0:
            m = n - d
            g[d:] += wr * (1.0 - h[:m])
            g[:m] -= wr * h[d:d + m]
        else:
            dd = -d
            m = n - dd
            g[:m] += wr * (1.0 - h[dd:])
            g[dd:] -= wr * h[:m]
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg]))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = float(np.max(S0 + gt[i] * T - gt[i] * gt[i] * Q)) * dx
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (lo + hi) / 2.0
    opt_v = float(np.max(S0 + opt_t * T - opt_t * opt_t * Q)) * dx
    top_idx = np.argsort(S0)[-5:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = float(np.max(S0 + av * T - av * av * Q)) * dx
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = float(np.max(S0 + t * T - t * t * Q)) * dx
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _make_dirs(h, g, raw, rng, n_max=6):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    r_max = int(np.argmax(raw))
    d = np.zeros(n)
    lag = r_max - (n - 1)
    if lag >= 0:
        m = n - lag
        d[lag:] += (1.0 - h[:m])
        d[:m] -= h[lag:lag + m]
    else:
        dd = -lag
        m = n - dd
        d[:m] += (1.0 - h[dd:])
        d[dd:] -= h[:m]
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn > 1e-12:
        dirs.append(d / dn)
    for blk in [4, 8, 16]:
        if n % blk != 0:
            continue
        nblk = n // blk
        gb = g.reshape(nblk, blk).mean(axis=1)
        d = np.tile(gb, (1, blk)).reshape(n).copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for freq in [1, 2, 3]:
        x = np.arange(n, dtype=float)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    return dirs[:n_max]

def _opt_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.15, max_stall=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(555)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = float(total_iters) / float(total_iters + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 2.5)
        raw = _exact_corr(h)
        g = _grad_smooth(h, beta)
        dirs = _make_dirs(h, g, raw, rng, 6)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.03, radius * 0.7)
        else:
            stall += 1
            radius = max(0.03, radius * 0.75)
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=3000, radius=0.15, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _make_quadratic_pattern(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    target_count = n // 2
    k = 0
    while len(positions) < target_count and k < n * 10:
        pos = int((k * golden) % n)
        if pos not in positions:
            positions.add(pos)
        k += 1
    for p in positions:
        h[p % n] = 1.0
    return h

def _make_periodic_pattern(n, period, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    half = period // 2
    positions_in_period = set()
    for i in range(half):
        positions_in_period.add(i)
    for i in range(n):
        h[i] = 1.0 if (i % period) in positions_in_period else 0.0
    idx = np.where(h > 0.5)[0]
    if len(idx) != n // 2:
        need = n // 2 - len(idx)
        zero_idx = np.where(h <= 0.5)[0]
        if need > 0:
            for i in range(need):
                h[zero_idx[i % len(zero_idx)]] = 1.0
        else:
            for i in range(-need):
                h[idx[i % len(idx)]] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Fast binary greedy from parent (20%)
    d1 = t0 + budget_s * 0.20
    h_b = _to_binary(h_init)
    h_b, c5_b = _binary_greedy(h_b, d1, top_k=5, max_passes=3000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 2: Binary SA + more greedy (20%)
    d2 = t0 + budget_s * 0.40
    h_b2 = _to_binary(best_h)
    h_b2, c5_b2 = _binary_sa_fast(h_b2, min(d2, t0 + budget_s * 0.10), T_init=0.3, max_iter=100000, seed=111)
    if c5_b2 < best_c5:
        best_c5 = c5_b2
        best_h = h_b2.copy()
    if time.time() < d2:
        h_b3 = _to_binary(best_h)
        h_b3, c5_b3 = _binary_greedy(h_b3, d2, top_k=7, max_passes=3000)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # Phase 3: Multi-n exploration (20%)
    d3 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    for dn in range(-20, 21, 4):
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 260:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_br = _to_binary(h_r)
        h_br, c5_br = _binary_greedy(h_br, sub_dl, top_k=5, max_passes=1500)
        if time.time() < d3 - 2:
            sub_dl2 = min(d3, time.time() + budget_s * 0.010)
            h_br, c5_br = _binary_sa_fast(h_br, sub_dl2, T_init=0.25, max_iter=50000, seed=n_try)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            n_curr = n_try

    # Phase 4: Continuous optimization (25%)
    d4 = t0 + budget_s * 0.85
    n_c = len(best_h)
    h4 = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4, c5_4 = _opt_smooth(h4, sub_dl, beta_lo=10.0, beta_hi=60.0, radius_init=0.12, max_stall=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _pair_continuous(h4, sub_dl2, n_iter=4000, radius=0.15, seed=444)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _opt_smooth(h4, sub_dl3, beta_lo=20.0, beta_hi=100.0, radius_init=0.06, max_stall=60)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping + final polish (15%)
    d5 = deadline
    scale = 0.015
    n_restarts = 0
    while time.time() < d5 - 15 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 10, time.time() + budget_s * 0.007)
        h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=15.0, beta_hi=70.0, radius_init=0.07, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.1)

    # Try nearby n values at the end
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8, -12, 12]:
        if time.time() > d5 - 20:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_br = _to_binary(h_r)
        sub_dl = min(d5 - 10, time.time() + budget_s * 0.012)
        h_br, c5_r = _binary_greedy(h_br, sub_dl, top_k=5, max_passes=1000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_br.copy()

    # Final continuous polish
    if deadline - time.time() > 8:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_smooth(h_f, deadline, beta_lo=25.0, beta_hi=100.0, radius_init=0.04, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, min(final_c5, best_c5), len(best_h)
