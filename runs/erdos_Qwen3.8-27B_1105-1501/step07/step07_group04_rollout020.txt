import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    center = n - 1
    M = 2 * n - 1
    delta = np.zeros(M)
    
    # Effect of p (1->0)
    # corr[center+k] for k=1..n-1:
    #   -(1-h[p-k]) if k <= min(p, n-1-p)
    #   +h[p+k] if k <= n-1-p (i.e., p+k < n)

For the SCA approach:
At each iteration, linearize the correlation constraints around the current point h_0:
F_k(h) ≈ F_k(h_0) + ∇F_k(h_0)^T (h - h_0)

Then solve:
minimize t
subject to: t >= F_k(h_0) + ∇F_k(h_0)^T (h - h_0) for all k
sum(h) = n/2
0 <= h <= 1

This is an LP! We can solve it with linprog. The linearization is valid in a neighborhood of h_0, so we take a step in the solution direction and re-linearize.

This is essentially the "linear programming approximation" method for non-convex problems. It's different from the subgradient method used before.
