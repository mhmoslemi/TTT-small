import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _anti_to_full(x, m):
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def _full_to_anti(h):
    m = len(h) // 2
    return 0.5 * (h[:m] + 1.0 - h[2*m-1:m-1:-1])

def _smoothed_c5_and_grad(x, n, beta, raw=None):
    m = n // 2
    if raw is None:
        h = _anti_to_full(x, m)
        raw = _raw_corr(h)
    h = _anti_to_full(x, m)
    M = float(np.max(raw))
    z = np.clip(beta * (raw - M), -50.0, 0.0)
    w = np.exp(z)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        sm = M * (2.0 / n)
        w = np.zeros(2*n-1)
        w[idx] = 1.0
    else:
        sm = (M + np.log(wsum) / beta) * (2.0 / n)
        w /= wsum
    
    # Compute gradient
    G = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    u = 1.0 - h
    for r in active:
        tau = r - (n - 1)
        wr = w[r]
        if tau >= 0:
            m_eff = n - tau
            if m_eff > 0:
                G[tau:tau+m_eff] += wr * u[:m_eff]
                G[:m_eff] -= wr * h[tau:tau+m_eff]
        else:

            m_eff = n - d
            if m_eff > 0:
                G[:m_eff] += wr * u[d:d+m_eff]
                G[d:d+m_eff] -= wr * h[:m_eff]

    def func(x):
        val, _ = _smoothed_c5_and_grad(x, n, beta)
        return val
    
    def jac(x):
        _, g = _smoothed_c5_and_grad(x, n, beta)
        return g
    
    bounds = [(0.0, 1.0)] * m
    result = minimize(func, x0, method='L-BFGS-B', jac=jac, bounds=bounds,
                      options={'maxiter': max_iter, 'maxfun': max_iter * 3,

    x_opt = np.clip(result.x, 0.0, 1.0)
    h_opt = _anti_to_full(x_opt, m)
    c5 = _exact_c5(h_opt)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _single_grad(h, r):
    """Gradient of corr[r] w.r.t. h."""
    n = len(h)
    G = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:

def _line_search_exact(h, d, radius):
    """Exact line search for C5 along direction d (sum d = 0)."""
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-14
    neg = d < -1e-14
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    
    def f(a):
        h_a = h + a * d
        return _exact_c5(h_a)
    
    # Coarse grid
    ng = 60
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.array([f(t) for t in gt])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng-1, bi + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(40):
        if hi - lo < 1e-12:

        if f1 < f2:

            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    # Check endpoints and midpoint
    for t in [a_lo, a_hi, 0.0, a_lo + (a_hi-a_lo)*0.25, a_lo + (a_hi-a_lo)*0.75]:
        ft = f(t)
        if ft < opt_v:

    return opt_t, opt_v

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.004):

    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline or stall > 300:
            break
        if not ones or not zeros:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5

            zeros = np.where(cur <= 0.5)[0].tolist()

                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

The breakpoints occur where c[tau1](x[j]) = c[tau2](x[j]), i.e., where the two linear functions cross. For each pair of active tau values, there's at most one crossing point. The minimum of the max is at one of these crossing points or at a boundary (x[j]=0 or x[j]=1).

The number of breakpoints is O(n^2) in the worst case (for 2n-1 linear functions), but in practice, only a few tau values are "active" (near the maximum), so it's much less.

Algorithm for 1D minimax:
1. For current x[j], compute all c[tau] values.
2. Find the active set (tau values where c[tau] is within epsilon of the max).
3. For each pair of active tau values, compute the crossing point x_cross.
4. Evaluate max c[tau] at each crossing point (clipped to [0,1]) and at boundaries.
5. Choose the x[j] that minimizes the max.

This is O(n * |active|^2) per coordinate, which is fast.

def _raw_corr(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

        else:

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(2024)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        u = 1.0 - h
        raw = _raw_corr(h)
        M = float(np.max(raw))
        z = np.clip(beta * (raw - M), -50.0, 0.0)
        w = np.exp(z)
        wsum = float(np.sum(w))
        if wsum > 1e-30:
            w /= wsum
        else:
            w = np.zeros(2*n-1)
            w[int(np.argmax(raw))] = 1.0
        G = np.zeros(n)
        for r in range(2*n-1):
            if w[r] < 1e-8:
                continue
            tau = r - (n-1)
            if tau >= 0:
                me = n - tau
                if me > 0:
                    G[tau:tau+me] += w[r] * u[:me]
                    G[:me] -= w[r] * h[tau:tau+me]
            else:
                d = -tau
                me = n - d
                if me > 0:
                    G[:me] += w[r] * u[d:d+me]
                    G[d:d+me] -= w[r] * h[:me]
        G -= np.mean(G)
        gn = np.linalg.norm(G)
        dirs = []
        if gn > 1e-12:
            dirs.append(G / gn)
        if n % 4 == 0:
            gb = G.reshape(n//4, 4).mean(axis=1)
            db = np.tile(gb, (1,4)).reshape(n).copy()
            db -= db.mean()
            nrm = np.linalg.norm(db)
            if nrm > 1e-12:
                dirs.append(db / nrm)
        rd = rng.standard_normal(n)
        rd -= rd.mean()
        nrm = np.linalg.norm(rd)
        if nrm > 1e-12:
            dirs.append(rd / nrm)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            a, val = _line_search_exact(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, a, d
        if best_d is not None:
            h_new = _project(h + best_alpha * best_d, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=40000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(7)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline or stall > 350:
            break
        if not ones or not zeros:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _worst_shift_target(h, n_refine=5):
    """Try to specifically reduce the worst shifts."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555)
    for _ in range(n_refine * 10):
        if time.time() > 0:
            pass
        raw = _raw_corr(h)
        worst = np.argsort(raw)[-3:]
        G = np.zeros(n)
        for r in worst:
            tau = r - (n-1)
            u = 1.0 - h
            if tau >= 0:
                me = n - tau
                if me > 0:
                    G[tau:tau+me] += u[:me]
                    G[:me] -= h[tau:tau+me]
            else:
                d = -tau
                me = n - d
                if me > 0:
                    G[:me] += u[d:d+me]
                    G[d:d+me] -= h[:me]
        G -= G.mean()
        gn = np.linalg.norm(G)
        if gn < 1e-12:
            break
        d_dir = G / gn
        d_dir += 0.3 * rng.standard_normal(n)
        d_dir -= d_dir.mean()
        d_dir /= np.linalg.norm(d_dir)
        a, val = _line_search_exact(h, d_dir, 0.1)
        if a != 0:
            h_new = _project(h + a * d_dir, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    if n0 % 2 != 0:
        n0 += 1
        h0 = np.zeros(n0)
        h0[:len(initial_h_values)] = initial_h_values
        h0 = _project(h0, n0 // 2)
    m0 = n0 // 2
    x0 = _full_to_anti(h0)
    x0 = np.clip(x0, 0.0, 1.0)
    h_init = _anti_to_full(x0, m0)

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Anti-palindromic L-BFGS-B with beta annealing (30%)
    d1 = t0 + budget_s * 0.30
    betas = [15, 30, 50, 80, 120, 200]
    x_cur = x0.copy()
    for beta in betas:
        if time.time() > d1 - 5:
            break
        remaining = d1 - time.time()
        max_it = int(min(400, remaining * 3))
        if max_it < 20:
            break
        x_opt, h_opt, c5_opt = _anti_lbfgsb(x_cur, n0, beta, max_it, d1)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            x_cur = x_opt.copy()

    # Phase 2: Anti-palindromic coordinate descent (20%)
    d2 = t0 + budget_s * 0.50
    m_cur = len(best_h) // 2
    x_cur = _full_to_anti(_project(best_h.copy(), len(best_h)/2.0))
    x_cur = np.clip(x_cur, 0.0, 1.0)
    x_cd, h_cd, c5_cd = _anti_coord_descent(x_cur, len(best_h), d2, max_sweeps=150)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: Multi-resolution search (15%)
    d3 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    for dn in range(-24, 25, 4):
        if time.time() > d3 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        m_r = n_try // 2
        x_r = _full_to_anti(h_r)
        x_r = np.clip(x_r, 0.0, 1.0)
        sub_dl = min(d3 - 2, time.time() + budget_s * 0.02)
        x_opt, h_opt, c5_opt = _anti_lbfgsb(x_r, n_try, 50, 200, sub_dl)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n_curr = n_try
        else:
            sub_dl2 = min(d3 - 2, time.time() + budget_s * 0.01)
            x_cd2, h_cd2, c5_cd2 = _anti_coord_descent(x_r, n_try, sub_dl2, max_sweeps=50)
            if c5_cd2 < best_c5 - 1e-12:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                n_curr = n_try

    # Phase 4: Full-space gradient descent (15%)
    d4 = t0 + budget_s * 0.80
    h_f = _project(best_h.copy(), len(best_h) / 2.0)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_f, c5_f = _full_gd(h_f, sub_dl, n_iter=250, beta=50.0, radius=0.10)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d4 - 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_f, c5_f = _worst_shift_target(h_f, n_refine=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Binary + refine (10%)
    d5 = t0 + budget_s * 0.90
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=
30000, T_init=0.003)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d5 - 3:
        n_bh = len(best_h)
        m_bh = n_bh // 2
        x_bh = _full_to_anti(_project(best_h.copy(), n_bh/2.0))
        x_bh = np.clip(x_bh, 0.0, 1.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        x_opt, h_opt, c5_opt = _anti_lbfgsb(x_bh, n_bh, 80, 150, sub_dl2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Basin hopping (10%)
    d6 = deadline
    scale = 0.025
    n_restarts = 0
    while time.time() < d6 - 15 and n_restarts < 120:
        n_restarts += 1
        n_bh = len(best_h)
        m_bh = n_bh // 2
        x_bh = _full_to_anti(_project(best_h.copy(), n_bh/2.0))
        x_bh = np.clip(x_bh, 0.0, 1.0)
        x_pert = x_bh + scale * rng.standard_normal(m_bh)
        x_pert = np.clip(x_pert, 0.0, 1.0)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.005)
        remaining = d6 - time.time()
        max_it = int(min(80, remaining * 2))
        if max_it < 10:
            break
        x_opt, h_opt, c5_opt = _anti_lbfgsb(x_pert, n_bh, 60, max_it, d6)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)

    # Final: nearby n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > deadline - 10:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        m_r = n_try // 2
        x_r = _full_to_anti(h_r)
        x_r = np.clip(x_r, 0.0, 1.0)
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.012)
        max_it = int(min(100, (deadline - time.time()) * 2))
        if max_it < 10:
            break
        x_opt, h_opt, c5_opt = _anti_lbfgsb(x_r, n_try, 100, max_it, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final coordinate descent polish
    if deadline - time.time() > 15:
        n_fp = len(best_h)
        m_fp = n_fp // 2
        x_fp = _full_to_anti(_project(best_h.copy(), n_fp/2.0))
        x_fp = np.clip(x_fp, 0.0, 1.0)
        x_opt, h_opt, c5_opt = _anti_coord_descent(x_fp, n_fp, deadline, max_sweeps=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    s = float(np.sum(best_h))
    target = len(best_h) / 2.0
    if abs(s - target) > 1e-9:
        best_h = _project(best_h, target)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
