                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.06 * width)
    hi_r = min(a_hi, best_a + 0.06 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _gen_dirs(h, rng, nd=4):
    n = len(h)
    dirs = []
    for beta in [15.0, 35.0]:
        g = _smooth_grad(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(1.0, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-10 and len(dirs) < nd:
        dirs.append(d / nrm)
    g20 = _smooth_grad(h, 20.0)
    bs = max(4, n // 15)
    d2 = np.zeros(n)
    for b in range(n // bs + 1):
        sl = slice(b * bs, min(n, (b + 1) * bs))
        d2[sl] = np.mean(g20[sl])
    d2 -= np.mean(d2)
    nrm = np.linalg.norm(d2)
    if nrm > 1e-10 and len(dirs) < nd:
        dirs.append(d2 / nrm)
    return dirs[:nd]

def _continuous_opt(h, deadline, r0=0.15, stall_lim=60, nd=4):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(55555)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = _gen_dirs(h, rng, nd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.35)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.005:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                elif stall % 40 == 0:
                    rad = 0.06
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _focused_binary_search(hb, deadline, max_stall=60):
    """Focused search: only consider pairs that reduce the top lags."""
    n = len(hb)
    hb = hb.astype(float).copy()
    overlap = _overlap_array(hb)
    cur_max = float(np.max(overlap))
    best_hb = hb.copy()
    best_max = cur_max
    D = _compute_D(hb, n)
    ones = np.where(hb > 0.5)[0].tolist()
    zeros = np.where(hb <= 0.5)[0].tolist()
    stall = 0
    rng = np.random.default_rng(999)
    max_iter = 50000
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        if not ones or not zeros:
            break
        top_k = int(np.argsort(overlap)[-10:])
        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        best_new_max = cur_max + 1.0
        best_i = -1
        best_j = -1
        for i in ones_arr:
            tmp = overlap + D[i]
            diffs = tmp[None, :] - D[zeros_arr]
            new_maxes = diffs.max(axis=1)
            min_val = float(new_maxes.min())
            if min_val < best_new_max:
                best_new_max = min_val
                best_i = int(i)
                best_j = int(zeros_arr[int(np.argmin(new_maxes))])
        if best_i >= 0 and best_new_max < cur_max - 1e-12:
            overlap = overlap + D[best_i] - D[best_j]
            hb[best_i] = 0.0
            hb[best_j] = 1.0
            cur_max = float(np.max(overlap))
            D = _compute_D(hb, n)
            ones.remove(best_i)
            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_hb = hb.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 20:
                if len(ones) >= 1 and len(zeros) >= 1:
                    oi = int(rng.integers(len(ones)))
                    oz = int(rng.integers(len(zeros)))
                    o = ones[oi]
                    z = zeros[oz]
                    hb[o] = 0.0
                    hb[z] = 1.0
                    ones[oi] = z
                    zeros[oz] = o
                    overlap = _overlap_array(hb)
                    cur_max = float(np.max(overlap))
                    D = _compute_D(hb, n)
                    stall = 0
    return best_hb, best_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Quick continuous optimization to get a good binary seed
    d1 = t0 + budget_s * 0.08
    h1, c1 = _continuous_opt(h_init, d1, r0=0.12, stall_lim=30, nd=4)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Binary local search (main workhorse) - multiple n values
    d2 = t0 + budget_s * 0.55
    n_list = [n0, n0 - 10, n0 + 10, n0 - 20, n0 + 20, 150, 200, 160, 220]
    n_list = sorted(set(x for x in n_list if 80 <= x <= 280))
    # Deduplicate
    seen = set()
    n_list2 = []
    for nn in n_list:
        if nn not in seen:
            seen.add(nn)
            n_list2.append(nn)
    n_list = n_list2

    for ni, n_try in enumerate(n_list):
        if time.time() > d2:
            break
        frac = max(0.04, 0.5 * (1.0 - ni / len(n_list)))
        sub_deadline = min(d2, time.time() + budget_s * frac)
        if sub_deadline - time.time() < 3:
            continue
        # Get a binary pattern at this n
        if n_try == n0:
            hb = _to_bin(best_h, n_try)
        else:
            hr = _resample(best_h, n_try)
            hb = _to_bin(hr, n_try)
        # Binary search
        hb, cb = _binary_search(hb, sub_deadline, max_stall=100, max_iter=200000, rng=rng)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Greedy constructions
    d3 = t0 + budget_s * 0.70
    if time.time() < d3:
        for n_g in [n0, n0 - 10, n0 + 10, 160, 200]:
            if time.time() > d3 - 2:
                break
            if n_g < 80 or n_g > 260:
                continue
            try:
                hb_g, cg = _greedy_construct(n_g, n_trials=3, rng=rng)
                if hb_g is not None and cg < best_c + 0.03:
                    sub_dl = min(d3, time.time() + budget_s * 0.02)
                    hb_g, cb_g = _binary_search(hb_g, sub_dl, max_stall=80, max_iter=50000, rng=rng)
                    if cb_g < best_c:
                        best_c = cb_g
                        best_h = hb_g.copy()
            except Exception:
                pass

    # Phase 4: More binary search with different perturbations
    d4 = t0 + budget_s * 0.82
    nc = len(best_h)
    for restart in range(5):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        if sub_dl - time.time() < 1:
            break
        if restart == 0:
            hb = _to_bin(best_h, nc)
        else:
            # Perturb the best
            hb_p = best_h.copy()
            k = max(3, nc // 30)
            for _ in range(k):
                i = int(rng.integers(nc))
                hb_p[i] = 1.0 - hb_p[i]
            hb = _to_bin(hb_p, nc)
        hb, cb = _binary_search(hb, sub_dl, max_stall=60, max_iter=30000, rng=rng)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: Continuous polish on best binary
    d5 = t0 + budget_s * 0.92
    if time.time() < d5:
        nc = len(best_h)
        h_cont = best_h.astype(float)
        h_cont = _project(h_cont, nc / 2.0)
        # Add small offset to enable continuous optimization
        h_work = h_cont + 0.01 * (rng.normal(0, 1, nc))
        h_work = _project(h_work, nc / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h_poll, c_poll = _continuous_opt(h_work, sub_dl, r0=0.08, stall_lim=50, nd=5)
        if c_poll < best_c:
            best_c = c_poll
            best_h = h_poll.copy()

    # Phase 6: Final pair swaps in continuous space
    if deadline - time.time() > 3:
        nc = len(best_h)
        dx = 2.0 / nc
        h_final = _project(best_h.copy(), nc / 2.0)
        cur = _exact_c5(h_final)
        rng2 = np.random.default_rng(88888)
        stall = 0
        while time.time() < deadline - 1 and stall < 50:
            i = int(rng2.integers(nc))
            j = int(rng2.integers(nc))
            if i == j:
                stall += 1
                continue
            d = np.zeros(nc)
            d[i] = 1.0
            d[j] = -1.0
            a, c_try = _line_search(h_final, d, dx, 0.5)
            if c_try < cur - 1e-14:
                h_new = h_final + a * d
                h_new = _project(h_new, nc / 2.0)
                c_act = _exact_c5(h_new)
                if c_act < cur - 1e-14:
                    h_final = h_new
                    cur = c_act
                    if cur < best_c:
                        best_c = cur
                        best_h = h_final.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
