Let me also think about the structure of good solutions. For the Erdős problem, we want a set S ⊂ {0,...,n-1} of size n/2 such that for every shift k, the number of x in S with x+k not in S is minimized. This is equivalent to: for every k, |S ∩ (S-k)^c| is minimized, where (S-k) = {x-k : x ∈ S}.

|S ∩ (S-k)^c| = |S| - |S ∩ (S-k)| = n/2 - |S ∩ (S-k)|

So minimizing max_k |S ∩ (S-k)^c| is equivalent to maximizing min_k |S ∩ (S-k)|.

|S ∩ (S-k)| is the autocorrelation of the indicator function of S at lag k.

So we want to maximize the minimum autocorrelation (over all lags). This is related to the theory of "difference sets" and "Sidon sets" in combinatorics!

A (v, k, λ) difference set is a subset D of a group of order v such that every nonzero group element can be expressed as a difference d1-d2 in exactly λ ways. For such a set, the autocorrelation at every nonzero lag is exactly λ.

If we could find a difference set with appropriate parameters, it would give an optimal solution!

For our problem: v = n (the group Z_n), k = n/2 (size of S), and we want λ to be as large as possible (to minimize the complement).

But difference sets have specific parameter requirements. A (v, k, λ) difference set in Z_v requires:
- k(k-1) = λ(v-1)
- So λ = k(k-1)/(v-1) = (n/2)(n/2-1)/(n-1) = n(n-2)/(2(n-1)) ≈ n/2 - 1

                best_j = int(np.argmin(new_maxes))
                best_i = ii
                best_improve = cur_max - min_new

            i = ones[best_i]
            j = zeros[best_j]
            h[i] = 0.0
            h[j] = 1.0
            out = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(out))
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_bin(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _compute_binary_deltas(h):
    n = len(h)
    m = 2 * n - 1
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    remove_d = np.zeros((n_ones, m))
    add_d = np.zeros((n_zeros, m))
    for ii in range(n_ones):
        i = int(ones[ii])
        k_lo = n - 1 - i
        k_hi = 2 * n - 2 - i
        pos = np.arange(k_lo, k_hi + 1) + (i - n + 1)
        remove_d[ii, k_lo:k_hi + 1] = h[pos] - 1.0
        k_lo2 = i
        k_hi2 = n - 1 + i
        pos2 = (n - 1 + i) - np.arange(k_lo2, k_hi2 + 1)
        remove_d[ii, k_lo2:k_hi2 + 1] += h[pos2]
    for jj in range(n_zeros):
        j = int(zeros[jj])
        k_lo = n - 1 - j
        k_hi = 2 * n - 2 - j
        pos = np.arange(k_lo, k_hi + 1) + (j - n + 1)
        add_d[jj, k_lo:k_hi + 1] = 1.0 - h[pos]
        k_lo2 = j
        k_hi2 = n - 1 + j
        pos2 = (n - 1 + j) - np.arange(k_lo2, k_hi2 + 1)
        add_d[jj, k_lo2:k_hi2 + 1] -= h[pos2]
    return remove_d, add_d, ones, zeros

    best_c = cur_max * dx
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        remove_d, add_d, _, _ = _compute_binary_deltas(h)
        new_outs = out[None, None, :] + add_d[None, :, :] - remove_d[:, None, :]
        new_maxes = np.max(new_outs, axis=2)
        min_val = float(np.min(new_maxes))

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(S))
    v = np.clip(beta * (S - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    lo_r = a_lo
    hi_r = a_hi
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_r - gr * (hi_r - lo_r)
    c2 = lo_r + gr * (hi_r - lo_r)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx

    d6 = deadline
    hb = _to_bin(best_h)
    sd6a = min(d6, time.time() + budget_s * 0.04)
    hb, cb = _binary_greedy(hb, sd6a, max_sweeps=5000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < d6 - 2:
        hc = _project(best_h.copy(), len(best_h) / 2.0)
        hc, cc = _local_cont(hc, d6, r0=0.04, stall_lim=30)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
