```python
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

            x = np.clip(x, 0.0, 1.0)

    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')

    C = np.correlate(d, d, mode='full')

    pos = d > 1e-15

    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx

        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_val = grid_t[best_idx], grid_f[best_idx]
    else:

        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx

            if f1 < f2:

        opt_t = (lo_b + hi_b) / 2.0

    best = min(cands, key=lambda x: x[1])
```

```python
def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h, rng=None):
    n = len(h)
    if rng is not None:
        scores = h + 0.005 * rng.standard_normal(n)
    else:
        scores = h
    idx = np.argsort(scores)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_max = float(np.min((1.0 - h[pos]) / d[pos]))
        t_hi = min(t_hi, t_max)
    if np.any(neg):
        t_max = float(np.min(h[neg] / (-d[neg])))
        t_hi = min(t_hi, t_max)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = t_lo
    hi_b = t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    # Check endpoints
    f_lo = float(np.max(S0)) * dx
    f_hi = float(np.max(S0 + t_hi * T - t_hi*t_hi * C)) * dx
    if f_lo < opt_val:
        opt_t, opt_val = 0.0, f_lo
    if f_hi < opt_val:
        opt_t, opt_val = t_hi, f_hi
    return opt_t, opt_val
```

```python
def _lp_direction(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active_mask = scores >= mx - gap
    active = np.where(active_mask)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
```

```python
def _local_opt(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        
        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active=n_active, gap=gap)
        if d is not None and float(np.max(np.abs(d))) > 1e-12:
            alpha, new_c5 = _line_search(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual_c5 = _exact_c5(h_new)
                if actual_c5 < best_c5 - 1e-14:
                    h = h_new
                    cur_c5 = actual_c5
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.3)
                    n_active = max(n_active - 1, 8)
                    gap = max(gap * 0.85, 0.001)
                else:
                    # Actual didn't improve but predicted did - projection issue
                    stall += 1
            else:
                stall += 1
        else:
            # Fallback: gradient-based direction
            scores_RAW = _scores_raw(h)
            mx = float(np.max(scores_RAW))
            active = np.where(scores_RAW >= mx - gap * n / 2.0)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(scores_RAW))])
            if len(active) > n_active:
                order = np.argsort(scores_RAW[active])
                active = active[order][-n_active:]
            lags = active - (n - 1)
            G = _grad_rows(h, lags)
            g = G.sum(axis=0)
            g -= float(np.mean(g))
            # Clip to feasible
            pos = (g > 0) & (h < 1.0)
            neg = (g < 0) & (h > 0.0)
            g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
            g[neg] = np.maximum(g[neg], -h[neg] / radius)
            g[(h >= 1.0 - 1e-12)] = 0.0
            g[(h <= 1e-12)] = 0.0
            g -= float(np.mean(g))
            d = g
            if float(np.max(np.abs(d))) > 1e-12:
                alpha, new_c5 = _line_search(h, d, dx, radius)
                if new_c5 < cur_c5 - 1e-13:
                    h_new = h + alpha * d
                    h_new = _project(h_new, target)
                    actual_c5 = _exact_c5(h_new)
                    if actual_c5 < best_c5:
                        h = h_new
                        cur_c5 = actual_c5
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
            stall += 1
            radius *= 0.75
            gap *= 1.3
            n_active = min(n_active + 2, 30)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.1
                gap = 0.003
                n_active = 15
    
    return best_h, best_c5
```

```python
def _binary_pair_sweep(h_bin, deadline, max_rounds=500):
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try multiple random pair swaps
        n_tries = len(ones) // 2 + 1
        found = False
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            # Swap: set oi to 0, oz to 1
            h_new = h.copy()
            h_new[oi] = 0.0
            h_new[oz] = 1.0
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                # Update one/zero lists
                ones.remove(oi)
                zeros.remove(oz)
                ones.append(oz)
                zeros.append(oi)
                found = True
                stall = 0
                break
        if not found:
            stall += 1
    
    return best_h, best_c5
```

```python
        constraints.append(D_d - B_d <= 2 * M)
        
        # Add s to x for retrieval (we'll reconstruct)

    # This approach has too many variables. Let me use a different formulation.
```

```python
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    f_lo = float(np.max(S0)) * dx
    f_hi = float(np.max(S0 + t_hi * T - t_hi*t_hi * C)) * dx
    if f_lo < opt_val:
        opt_t, opt_val = 0.0, f_lo
    if f_hi < opt_val:
        opt_t, opt_val = t_hi, f_hi
    return opt_t, opt_val

def _lp_direction(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active_mask = scores >= mx - gap
    active = np.where(active_mask)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_fallback(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    g = G.sum(axis=0)
    g -= float(np.mean(g))
    pos = (g > 0) & (h < 1.0 - 1e-12)
    neg = (g < 0) & (h > 1e-12)
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = 0.0
    g[h <= 1e-12] = 0.0
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break

        scores = _scores(h)
        d = _lp_direction(h, scores, radius, n_active=n_active, gap=gap)

        if d is not None and float(np.max(np.abs(d))) > 1e-12:
            alpha, new_c5 = _line_search(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual_c5 = _exact_c5(h_new)
                if actual_c5 < best_c5 - 1e-14:
                    h = h_new
                    cur_c5 = actual_c5
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.04, 0.25)
                    n_active = max(n_active - 1, 8)
                    gap = max(gap * 0.88, 0.001)
                else:
                    stall += 1
            else:
                stall += 1
        else:
            d = _gradient_fallback(h, scores, radius, n_active=n_active, gap=gap)
            if float(np.max(np.abs(d))) > 1e-12:
                alpha, new_c5 = _line_search(h, d, dx, radius)
                if new_c5 < cur_c5 - 1e-13:
                    h_new = h + alpha * d
                    h_new = _project(h_new, target)
                    actual_c5 = _exact_c5(h_new)
                    if actual_c5 < best_c5:
                        h = h_new
                        cur_c5 = actual_c5
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.04, 0.25)
            stall += 1

        if stall > 5 and stall % 10 == 0:
            radius *= 0.7
            gap *= 1.3
            n_active = min(n_active + 2, 30)
            if radius < 0.015:
                radius = 0.08
                gap = 0.003
                n_active = 12

    return best_h, best_c5

def _binary_hill_climb(h_bin, deadline, max_rounds=2000, n_samples=800):
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    stall = 0

    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_oi = -1
        best_oz = -1
        n_tries = min(n_samples, len(ones) * len(zeros))

        for t in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi, oz = ones[i], zeros[j]
            h[oi] = 0.0
            h[oz] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_oi = oi
                best_oz = oz
            h[oi] = 1.0
            h[oz] = 0.0

        if best_oi >= 0:
            h[best_oi] = 0.0
            h[best_oz] = 1.0
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def _binary_sa(h_bin, deadline, T_init=0.004, max_iter=50000):
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            T *= 0.9997
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            stall += 1

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 10.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.standard_normal(n)
    return _project(h + perturb, n / 2.0)

def _milp_small(n, deadline):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    if n % 2 != 0 or n < 30 or n > 70:
        return None
    remaining = deadline - time.time()
    if remaining < 10:
        return None
    try:
        x = cp.Variable(n, boolean=True)
        M = cp.Variable(nonneg=True)
        constraints = [cp.sum(x) == n // 2]
        t_limit = min(30, int(remaining) - 3)
        if t_limit < 5:
            return None
        for d in range(1, n):
            m = n - d
            s = cp.Variable(m, boolean=True)
            constraints.append(s >= x[d:d+m] - x[0:m])
            constraints.append(s >= x[0:m] - x[d:d+m])
            D_d = cp.sum(s)
            B_d = cp.sum(x[d:d+m]) - cp.sum(x[0:m])
            constraints.append(D_d + B_d <= 2 * M)
            constraints.append(D_d - B_d <= 2 * M)
        prob = cp.Problem(cp.Minimize(M), constraints)
        solved = False
        for solver in [cp.CBC, cp.SCIP, cp.GLPK_MI]:
            try:
                prob.solve(solver=solver, time_limit=t_limit)
                solved = True
                break
            except Exception:
                continue
        if not solved or x.value is None:
            return None
        x_val = np.round(np.array(x.value).flatten())
        x_val = np.clip(x_val, 0, 1)
        if abs(x_val.sum() - n/2) > 0.1:
            idx = np.argsort(x_val)[-n//2:]
            x_val = np.zeros(n)
            x_val[idx] = 1.0
        return x_val.astype(float)
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Quick LP polish on parent (8% time)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_opt(best_h, d1, radius_init=0.10, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: MILP for small n (12% time)
    d2 = t0 + budget_s * 0.20
    milp_ns = [40, 50, 60, 62, 64, 66, 68, 70]
    for nm in milp_ns:
        if time.time() > d2:
            break
        h_m = _milp_small(nm, d2)
        if h_m is not None:
            c5_m = _exact_c5(h_m)
            # Polish it
            sub_dl = min(d2, time.time() + budget_s * 0.01)
            h_mp, c5_mp = _local_opt(h_m, sub_dl, radius_init=0.08, max_iters=500)
            if c5_mp < c5_m:
                c5_m = c5_mp
                h_m = h_mp.copy()
            if c5_m < best_c5:
                best_c5 = c5_m
                best_h = h_m.copy()
        # Binary hill climb on resampled parent at this n
        if time.time() < d2 - 5:
            h_r = _resample(best_h, nm)
            h_rb = _to_binary(h_r, rng)
            sub_dl2 = min(d2, time.time() + budget_s * 0.012)
            h_rb, c5_rb = _binary_hill_climb(h_rb, sub_dl2, max_rounds=500, n_samples=400)
            # Fractional polish
            h_rf = _project(h_rb.copy(), nm / 2.0)
            sub_dl3 = min(d2, time.time() + budget_s * 0.008)
            h_rf, c5_rf = _local_opt(h_rf, sub_dl3, radius_init=0.06, max_iters=400)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()

    # Phase 3: Multi-n binary exploration (30% time)
    d3 = t0 + budget_s * 0.50
    target_ns = [100, 110, 120, 130, 140, 150, 160, 170, 180, 190, 200]
    # Sort by proximity to current best n
    n_curr = len(best_h)
    target_ns.sort(key=lambda x: abs(x - n_curr))
    for nm in target_ns:
        if time.time() > d3:
            break
        if nm == n_curr:
            continue
        h_r = _resample(best_h, nm)
        h_rb = _to_binary(h_r, rng)
        # Binary hill climb
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_rb, c5_rb = _binary_hill_climb(h_rb, sub_dl, max_rounds=1000, n_samples=600)
        # Binary SA
        if time.time() < d3 - budget_s * 0.01:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_sa, c5_sa = _binary_sa(h_rb, sub_dl2, T_init=0.003, max_iter=20000)
            if c5_sa < c5_rb:
                h_rb = h_sa
                c5_rb = c5_sa
        # Fractional polish
        h_rf = _project(h_rb.copy(), nm / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.012)
        h_rf, c5_rf = _local_opt(h_rf, sub_dl3, radius_init=0.08, max_iters=600)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
            n_curr = nm

    # Phase 4: Extended LP optimization on best (15% time)
    d4 = t0 + budget_s * 0.65
    n_bst = len(best_h)
    h4 = _project(best_h.copy(), n_bst / 2.0)
    h4, c5_4 = _local_opt(h4, d4, radius_init=0.10, max_iters=8000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Basin hopping (15% time)
    d5 = t0 + budget_s * 0.80
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.005)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.06, max_iters=500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: Binary refinement on best (10% time)
    d6 = t0 + budget_s * 0.90
    n_b = len(best_h)
    h_bin = _to_binary(best_h, rng)
    h_bin, c5_bin = _binary_hill_climb(h_bin, min(d6, time.time() + budget_s * 0.04),
                                        max_rounds=2000, n_samples=1000)
    if time.time() < d6 - budget_s * 0.01:
        h_bin2, c5_bin2 = _binary_sa(h_bin, time.time() + budget_s * 0.02,
                                      T_init=0.002, max_iter=30000)
        if c5_bin2 < c5_bin:
            h_bin = h_bin2
            c5_bin = c5_bin2
    h_pol = _project(h_bin.copy(), n_b / 2.0)
    h_pol, c5_pol = _local_opt(h_pol, min(d6, time.time() + budget_s * 0.02),
                                radius_init=0.05, max_iters=800)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 7: Final polish (10% time)
    d7 = deadline
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, d7, radius_init=0.04, max_iters=5000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
```
