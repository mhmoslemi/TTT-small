import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, k):
    """Compute gradient of correlation at lag k (offset from center)."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # F_k = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
        # dF/dh[j] = (1-h[j-k]) for j>=k, -h[j+k] for j<n-k
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        # F_{-u} = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])
        # dF/dh[j] = (1-h[j+u]) for j<n-u, -h[j-u] for j>=u
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

def _lp_direction(h, corr_vals, active_idx, radius, dx):
    """Solve LP to find direction reducing max of active correlations."""
    n = len(h)
    m = len(active_idx)
    if m == 0:
        return None
    
    # Build constraint matrix
    # Variables: d[0..n-1], t, u[0..n-1]
    # Total: 2*n + 1
    var_count = 2 * n + 1
    t_idx = n
    u_start = n + 1
    
    # Objective: minimize t + eps * sum(u)
    eps = 1e-5
    c_obj = np.zeros(var_count)
    c_obj[t_idx] = 1.0
    c_obj[u_start:u_start + n] = eps
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint r: sum_j grad_r[j] * d[j] - t <= -F_r(h)
    # i.e., sum_j grad_r[j] * d[j] - t + F_r(h) <= 0
    A_rows = []
    b_rows = []
    for r in active_idx:
        k = r - (n - 1)
        grad = _corr_grad(h, k) * dx
        row = np.zeros(var_count)
        row[:n] = grad
        row[t_idx] = -1.0
        A_rows.append(row)
        b_rows.append(-corr_vals[r])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, var_count))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i] + 1e-14)
        hi_i = min(radius, 1.0 - h[i] - 1e-14)
        bounds.append((lo_i, hi_i))
    bounds.append((corr_vals.max() - 1.0, corr_vals.max() + 1.0))  # t
    for i in range(n):
        bounds.append((0.0, 2.0 * radius))  # u
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        d = res.x[:n].copy()
        # Center d
        d -= np.mean(d)
        if np.max(np.abs(d)) < 1e-14:
            return None
        return d
    except Exception:
        return None

def _line_search(h, d, dx, radius):
    """Exact line search using quadratic expansion."""
    n = len(h)
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    
    # Find max feasible alpha
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    
    # Grid search
    n_grid = 40
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        a = grid_t[idx]
        grid_f[idx] = float(np.max(C0 + a * L1 - a * a * Q2)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo = grid_t[max(0, best_idx - 2)]
    hi = grid_t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi - lo < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2)) * dx
        f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2)) * dx
        for _ in range(60):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2)) * dx
        opt_t = (lo + hi) / 2.0
        opt_val = float(np.max(C0 + opt_t * L1 - opt_t * opt_t * Q2)) * dx
    
    candidates = [(opt_t, opt_val), (0.0, float(np.max(C0) * dx)),
                  (t_hi, float(np.max(C0 + t_hi * L1 - t_hi * t_hi * Q2)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _lp_local_opt(h, deadline, radius_init=0.06, active_count=50, max_iter=5000):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if radius < 1e-7:
            break
        
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr_vals = corr_raw * dx
        max_val = corr_vals.max()
        
        # Select active constraints
        threshold = max_val - max(0.005, 0.02 * max_val)
        active_mask = corr_vals >= threshold
        active_count_actual = int(np.sum(active_mask))
        
        if active_count_actual > active_count:
            # Take top active_count
            sorted_idx = np.argsort(-corr_vals)
            active_idx = sorted_idx[:active_count]
        else:
            active_idx = np.where(active_mask)[0]
        
        d = _lp_direction(h, corr_vals, active_idx, radius, dx)
        if d is None:
            # Fallback to subgradient
            max_idx = int(np.argmax(corr_vals))
            k = max_idx - (n - 1)
            g = _corr_grad(h, k)
            d = -g - np.mean(-g)
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            if alpha > 0.5 * radius:
                radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _binary_targeted_swap(h_bin, deadline, max_iters=50000, seed=42, n_trials_per_iter=50):
    """Targeted binary swap: try flips that reduce the most active lags."""
    n = len(h_bin)

def _binary_sa(h_bin, deadline, max_iters=100000, temp_init=0.003, seed=42):
    """Simulated annealing for binary solutions."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    best_stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 8000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                best_stall = 0
            else:
                best_stall += 1
            T = max(0.0003, T * 0.9998)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            best_stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 4.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
