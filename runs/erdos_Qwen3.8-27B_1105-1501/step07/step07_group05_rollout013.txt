import numpy as np

import math

def _exact_c5(h):

    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, 0.0]
    width = a_hi - a_lo
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    ng = min(64, max(32, int(width * 150)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    M = min(10, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.04 * width)
    hi_r = min(a_hi, best_a + 0.04 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(25):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _block_dir(g, bs):
    n = len(g)
    nb = n // bs
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        d[b * bs:(b + 1) * bs] = np.mean(g[b * bs:(b + 1) * bs])
    if nb * bs < n:
        d[nb * bs:] = np.mean(g[nb * bs:])
    d -= np.mean(d)
    return d

def _gen_dirs(h, rng, nd=5):
    n = len(h)
    dirs = []
    for beta in [15.0, 25.0, 35.0]:
        g = _smooth_grad(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    g20 = _smooth_grad(h, 20.0)
    d = _block_dir(g20, max(4, n // 16))
    nrm = np.linalg.norm(d)
    if nrm > 1e-10 and len(dirs) < nd:
        dirs.append(d / nrm)
    d = _block_dir(g20, max(8, n // 8))
    nrm = np.linalg.norm(d)
    if nrm > 1e-10 and len(dirs) < nd:
        dirs.append(d / nrm)
    if len(dirs) < nd:
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.4, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

        best_a = 0.0
        best_d = None

            nrm = np.linalg.norm(d)

                continue

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):

def _beatty_bin(n, alpha):

def _periodic_bin(n, period, on_frac):
    pat = np.zeros(period)
    k = int(round(period * on_frac))
    pat[:k] = 1.0
    nb = n // period
    hb = np.tile(pat, nb)
    if nb * period < n:
        hb = np.concatenate([hb, pat[:n - nb * period]])
    return hb

            sd = min(d4, time.time() + budget_s * 0.02)
            h5, c5 = _targeted_pair_opt(h5, sd, max_it=8000, top_k=6)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()

    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d5 and n_re < 50:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.2 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.006)
        ht, ct = _local_opt(hp, sd, r0=0.06, stall_lim=20, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.7)
        else:
            scale = min(0.06, scale * 1.15)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline - 2, r0=0.02, stall_lim=40, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
