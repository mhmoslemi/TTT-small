def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_gradients(h, active_lags):
    """Compute gradient of raw[r] w.r.t. h for each active lag."""
    n = len(h)
    u = 1.0 - h
    grads = np.zeros((len(active_lags), n))
    for idx, r in enumerate(active_lags):

        g = np.zeros(n)

def _lp_optimize(h, deadline, max_iters=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        
        # Active set: lags within some fraction of max
        # Start wide, narrow over iterations
        frac = max(0.5, 0.9 - 0.01 * it)
        threshold = mx * frac
        active = np.where(raw >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(-raw)[:5]
        
        # Compute gradients
        grads = _compute_gradients(h, active)
        
        # Compute c_r = grad_r^T h - raw[r]
        c_vals = grads @ h - raw[active]
        
        # Solve LP
        h_lp = _solve_one_lp(h, active, grads, c_vals, raw, n)
        
        if h_lp is None:
            stall += 1
            continue
        
        # Line search
        d = h_lp - h
        best_alpha = 0.0
        best_val = cur_c5
        for alpha in [0.25, 0.5, 0.75, 1.0, 0.1, 0.3, 0.6, 0.9]:
            h_try = h + alpha * d
            h_try = _project(h_try, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Golden section refinement
        if best_alpha > 0:
            lo = max(0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1*d, target))
            f2 = _exact_c5(_project(h + c2*d, target))
            for _ in range(30):
                if hi - lo < 1e-5:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(_project(h + c1*d, target))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _exact_c5(_project(h + c2*d, target))
            opt = (lo + hi) / 2
            val_opt = _exact_c5(_project(h + opt*d, target))
            if val_opt < best_val:
                best_val = val_opt
                best_alpha = opt
        
        if best_alpha > 1e-6 and best_val < cur_c5 - 1e-12:

            if cur_c5 < best_c5:

                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    
    return best_h, best_c5

```python
def _solve_one_lp(h_cur, active, grads, c_vals, raw, n):
    target = n / 2.0
    n_cons = len(active)
    # Variables: [h_0, ..., h_{n-1}, t], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # A_ub: n_cons rows, each of length n+1
    A_ub = np.zeros((n_cons, n + 1))
    b_ub = np.zeros(n_cons)
    for i in range(n_cons):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = c_vals[i]
    
    # A_eq: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        h_sol = _project(res.x[:n], target)
        return h_sol
    return None

import numpy as np
import time
import scipy.optimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _alt_lp(h_cur, active, n, target=None):
    """Solve alternating LP: fix u=1-h_cur, find h minimizing max <h, w_r>."""
    if target is None:
        target = n / 2.0
    u = 1.0 - h_cur
    n_act = len(active)
    if n_act == 0:
        return h_cur.copy()
    
    coeff = np.zeros((n_act, n))
    for i, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                coeff[i, :m] = u[d:]
        else:
            m = n + d
            if m > 0:
                coeff[i, -d:] = u[:m]
    
    # Check if all coefficients are zero (degenerate)
    if np.max(np.abs(coeff)) < 1e-15:
        return h_cur.copy()
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = coeff
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_act)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds, method='highs',
                                     options={'presolve': True, 'time_limit': 5.0})
        if res.success and res.x is not None:
            return _project(res.x[:n], target)
    except Exception:
        pass
    return None

def _lp_iterate(h, deadline, max_iters=200, top_k_list=None):
    """Main LP iteration loop."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    if top_k_list is None:
        top_k_list = [5, 8, 12, 20, 30, 50, 80, 120, 200]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        
        if mx < 1e-10:
            break
        
        # Try different active set sizes
        improved = False
        for top_k in top_k_list:
            if time.time() > deadline:
                break
            k = min(top_k, 2 * n - 1)
            active = np.argsort(-raw)[:k]
            
            h_cand = _alt_lp(h, active, n, target)
            if h_cand is None:
                continue
            
            if np.allclose(h_cand, h, atol=1e-8):
                continue
            
            # Line search
            d = h_cand - h
            best_alpha = 0.0
            best_val = cur_c5
            for alpha in [0.3, 0.5, 0.7, 1.0, 0.2, 0.4, 0.6, 0.8, 0.9]:
                h_try = _project(h + alpha * d, target)
                val = _exact_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            
            if best_alpha > 1e-6 and best_val < cur_c5 - 1e-12:
                # Golden section refinement
                lo = max(0.0, best_alpha - 0.1)
                hi = min(1.0, best_alpha + 0.1)
                if hi - lo > 1e-4:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = _exact_c5(_project(h + c1 * d, target))
                    f2 = _exact_c5(_project(h + c2 * d, target))
                    for _ in range(25):
                        if hi - lo < 1e-5:
                            break
                        if f1 < f2:
                            hi, c2, f2 = c2, c1, f1
                            c1 = hi - gr * (hi - lo)
                            f1 = _exact_c5(_project(h + c1 * d, target))
                        else:
                            lo, c1, f1 = c1, c2, f2
                            c2 = lo + gr * (hi - lo)
                            f2 = _exact_c5(_project(h + c2 * d, target))
                    opt = (lo + hi) / 2.0
                    val_opt = _exact_c5(_project(h + opt * d, target))
                    if val_opt < best_val:
                        best_val = val_opt
                        best_alpha = opt
                
                h = _project(h + best_alpha * d, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 30:
                break
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):

    cur = h_bin.copy()

    best_c5 = cur_c5

    zeros = list(np.where(cur <= 0.5)[0])

                    improved = True

                    cur[oz] = 0.0

def _gradient_step(h, beta, n, target):
    """Simple gradient step using softmax-weighted correlations."""
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    
    # Compute weighted gradient
    n_raw = len(raw)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(n_raw):
        wi = w[r]
        if wi < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += wi * u[d:]  # (1-h[j+d]) contribution
                g[d:] -= wi * h[:m]  # -h[j-d] contribution... 
        else:
            nd = -d
            m = n - nd
            if m > 0:
                g[nd:] += wi * u[:m]
                g[:m] -= wi * h[nd:]
    g -= np.mean(g)
    return g

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

Hmm, this is getting complicated. Let me simplify: for the coordinate optimization, I'll just do a dense grid search over [0,1] for each coordinate, computing the exact C5 each time. It's O(n * 100 * n^2) = O(100 * n^3) per pass, which for n=180 is about 100 * 5.8M = 580M operations per pass. That might be too slow.

Let me instead use a smarter approach: for each coordinate, compute the raw values at x=0 and x=1, then do a piecewise linear interpolation (ignoring the d=0 quadratic for simplicity). The breakpoints are where different raw[r] values cross zero relative to the max. I can find the optimal x by checking all relevant breakpoints.

Actually, for the piecewise linear case (ignoring d=0), the function to minimize is:
f(x) = max_r (base[r] + slope[r] * x) for x in [0,1]

This is a convex piecewise linear function. The minimum is at a point where two (or more) lines are equal and above all others. I can find this by:
1. For each pair (r1, r2), solve base[r1] + slope[r1]*x = base[r2] + slope[r2]*x → x = (base[r2]-base[r1])/(slope[r1]-slope[r2])
2. Check all such x values in [0,1] plus the endpoints 0 and 1
3. Pick the one that minimizes f(x)

This is O(n^2) per coordinate, which for n=180 is about 32000 per coordinate, and 180 coordinates, so about 5.8M per pass. That's fast!

But I need to handle the d=0 case. For r=n-1, the function is base[n-1] + x*(1-x) instead of base[n-1] + slope[n-1]*x. This is a concave parabola, so the max of {piecewise linear, concave parabola} is more complex. Let me just add x*(1-x) to the evaluation:

f(x) = max(max_{r≠n-1} (base[r] + slope[r]*x), base[n-1] + x*(1-x))

I'll just evaluate this at all candidate points (endpoints + crossings + vertex of parabola at x=0.5).

def _coord_descent(h, deadline, n_passes=10):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    r0 = n - 1  # index of d=0 lag
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            h[i] = 0.0
            u = 1.0 - h
            raw0 = np.correlate(h, u, mode='full')
            h[i] = 1.0
            u = 1.0 - h
            raw1 = np.correlate(h, u, mode='full')
            h[i] = hi  # restore
            base = raw0
            slope = raw1 - raw0
            
            # Candidate x values
            candidates = [0.0, 1.0, 0.5, hi]
            # Crossings
            for a in range(2*n-1):
                for b in range(a+1, 2*n-1):
                    if a == r0 or b == r0:
                        continue
                    ds = slope[a] - slope[b]
                    if abs(ds) > 1e-12:
                        xc = (base[b] - base[a]) / ds
                        if 0.0 <= xc <= 1.0:
                            candidates.append(xc)
            
            # Evaluate
            best_x = hi
            best_f = cur_c5 / dx  # work in raw units
            
            for x in candidates:
                x = max(0.0, min(1.0, x))
                raw_x = base + x * slope
                # Override r0 with true value
                raw_x[r0] = base[r0] + x * (1.0 - x)
                f_val = float(np.max(raw_x))
                if f_val < best_f - 1e-12:
                    best_f = f_val
                    best_x = x
            
            if best_x != hi:
                # Need to maintain sum constraint
                # Set h[i] = best_x, then adjust another coordinate
                # For simplicity, just set h[i] and re-project
                h[i] = best_x
                h = _project(h, target)
                new_c5 = _exact_c5(h)
                if new_c5 < cur_c5 - 1e-12:
                    cur_c5 = new_c5
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                else:
                    h[i] = hi  # revert
        if not improved:
            break
    
    return best_h, best_c5

def _gradient_descent(h, deadline, beta=20.0, radius=0.2):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_raw = 2 * n - 1
    
    while time.time() < deadline and stall < 50:
        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        
        # Compute gradient
        u = 1.0 - h
        g = np.zeros(n)
        for r in range(n_raw):
            wi = w[r]
            if wi < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[:m] += wi * u[d:]
                if d > 0:
                    g[d:] -= wi * h[:m]
                else:
                    g[:] -= wi * h[:]
            else:
                nd = -d
                m = n - nd
                if m > 0:
                    g[nd:] += wi * u[:m]
                if nd > 0:
                    g[:m] -= wi * h[nd:]
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g /= gn
        
        # Line search
        best_alpha = 0.0
        best_val = cur_c5
        for alpha in np.linspace(0, radius, 20):
            h_try = _project(h + alpha * g, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-6 and best_val < cur_c5 - 1e-13:
            h = _project(h + best_alpha * g, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.8
            if radius < 0.01:
                break
    
    return best_h, best_c5
```
