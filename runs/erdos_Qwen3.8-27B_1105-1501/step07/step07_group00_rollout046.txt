    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_fallback(h, scores, beta, radius):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    # Bound the direction
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.004
    n_active = 12
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        d = _active_lp_direction(h, scores, radius, n_active, gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_fallback(h, scores, beta=20.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.5
                n_active = min(n_active + 3, 30)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search_exact(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_new.copy()
            h = h_new
            cur_c5 = actual_c5
            stall = 0
            radius = min(radius * 1.05, 0.3)
            gap = max(gap * 0.85, 0.001)
            n_active = max(n_active - 1, 6)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.3
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(1, len(ones) // 4)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, len(zeros)))
            j = int(rng.integers(0, len(ones)))
            d = np.zeros(n)
            d[zeros[i]] = 1.0
            d[ones[j]] = -1.0
            alpha, c_try = _line_search_exact(h, d, dx, 0.25)
            if c_try < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                c_new = _exact_c5(h_new)
                if c_new < cur_c5:
                    h = h_new
                    cur_c5 = c_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _block_perturb(h, amplitude, rng):
    """Invert a random contiguous block."""
    n = len(h)
    target = n / 2.0
    length = int(rng.integers(max(2, n // 10), max(3, n // 3)))
    start = int(rng.integers(0, n - length))
    perturb = np.zeros(n)
    perturb[start:start+length] = amplitude
    # Flip the block (invert)
    h_pert = h.copy()
    h_pert[start:start+length] = 1.0 - h_pert[start:start+length]
    # Blend with the amplitude
    h_pert = (1-amplitude) * h + amplitude * h_pert
    return _project(h_pert, target)

def _binary_local_search(h_bin, deadline, max_iters=5000):
    """Local search on binary sequences using pair swaps."""
    n = len(h_bin)
    h = h_bin.astype(float)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(1234)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(zeros)))
        j = int(rng.integers(0, len(ones)))
        h_new = h.copy()
        h_new[zeros[i]] = 1.0
        h_new[ones[j]] = 0.0
        c_new = _exact_c5(h_new)
        if c_new < cur_c5:
            h = h_new
            cur_c5 = c_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _multi_pair_swap(h, deadline, max_rounds=300):
    """Try multiple simultaneous pair swaps for larger moves."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(5555)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Try 2-3 simultaneous swaps
        n_swaps = int(rng.integers(1, 4))
        n_swaps = min(n_swaps, len(ones), len(zeros))
        d = np.zeros(n)
        si = rng.choice(len(zeros), size=n_swaps, replace=False)
        sj = rng.choice(len(ones), size=n_swaps, replace=False)
        for k in range(n_swaps):
            d[zeros[si[k]]] = 1.0
            d[ones[sj[k]]] = -1.0
        d *= 0.5  # Scale down for multi-swap
        alpha, c_try = _line_search_exact(h, d, dx, 0.3)
        if c_try < cur_c5 - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c_new = _exact_c5(h_new)
            if c_new < cur_c5:
                h = h_new
                cur_c5 = c_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                continue
        stall += 1
    return best_h, best_c5
