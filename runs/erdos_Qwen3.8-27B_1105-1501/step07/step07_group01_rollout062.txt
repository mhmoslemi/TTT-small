def _project(v, target):

    n = len(v)

    if target >= n: return np.ones(n)

        else: hi = mid

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_grad(h, corr, beta):
    n = len(h)
    max_c = float(np.max(corr))
    exps = np.exp(beta * (corr - max_c))
    total = float(np.sum(exps))
    if total < 1e-300:
        return np.zeros(n)
    w = exps / total
    grad = np.zeros(n)
    for j in range(n):
        t1 = np.dot(w[n-1:n-1+n-j], 1.0 - h[j:n])
        t2 = np.dot(w[n-1:n+j][::-1], h[0:j+1])

        else:
            t4 = 0.0
        grad[j] = t1 - t2 + t3 - t4
    return grad

    target = n / 2.0

    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    
    n_steps = min(n_steps_target, int((deadline - time.time()) / 0.005))
    n_steps = max(n_steps, 100)
    
    step = 0
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        t_frac = step / max(n_steps, 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        beta = max(beta, beta_start)
        
        corr = _corr(h)
        grad = _smooth_grad(h, corr, beta)
        d = -(grad - grad.mean())
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-14:
            stall += 1
            continue
        
        # Max step size
        pos = d > 1e-14
        neg = d < -1e-14

def _pair_swap(h, deadline, n_iters=20000, seed=555):

            break
        if stall > 300:
            break

        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue

        # Change in corr: corr + alpha * (g_i - g_j) - alpha^2 * (quadratic)

        for s in range(1, n - i):
            gi[n - 1 - s] -= h[i + s]
        
        gj = np.zeros(2 * n - 1)
        for s in range(0, n - j):
            gj[n - 1 + s] += (1.0 - h[j + s])
        for s in range(0, j + 1):
            gj[n - 1 + s] -= h[j - s]
        for s in range(1, j + 1):
            gj[n - 1 - s] += (1.0 - h[j - s])
        for s in range(1, n - j):
            gj[n - 1 - s] -= h[j + s]
        
        d = gi - gj
        # Quadratic: at lag 0, -2*alpha^2; at lag |i-j|, alpha^2 (small)
        # For line search, use the quadratic at lag 0
        lo_a, hi_a = -h[i], 1.0 - h[j]
        if hi_a - lo_a < 1e-12:
            stall += 1
            continue
        # Cap step size
        hi_a = min(hi_a, 0.5)
        lo_a = max(lo_a, -0.5)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        
        def _fp(delta):
            nc = corr + delta * d
            nc[n-1] -= 2.0 * delta * delta
            return float(np.max(nc)) * dx
        
        f1 = _fp(c1)
        f2 = _fp(c2)
        for _ in range(28):
            if hi_a - lo_a < 1e-11:
                break
            if f1 < f2:
                hi_a = c2
                c2 = c1
                f2 = f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = _fp(c1)
            else:
                lo_a = c1
                c1 = c2
                f1 = f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = _fp(c2)
        opt_a = (lo_a + hi_a) / 2.0
        f_opt = _fp(opt_a)
        
        if f_opt < cur_c5 - 1e-13:
            h[i] += opt_a
            h[j] -= opt_a
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            corr = _corr(h)
            cur_c5 = float(np.max(corr)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.002, seed=31337):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    corr = _corr(cur)
    cur_c5 = float(np.max(corr)) * dx
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = int(rng.integers(0, len(ones)))
        j_idx = int(rng.integers(0, len(zeros)))
        oi = ones[i_idx]
        oz = zeros[j_idx]
        
        # Compute effect: remove oi, add oz
        # Change in corr: -g_oi + g_oz (where g is the gradient pattern)
        dS = np.zeros(2 * n - 1)
        # Effect of h[oi] going from 1 to 0: delta = -1 at position oi
        for s in range(0, n - oi):
            dS[n - 1 + s] -= (1.0 - cur[oi + s])
        for s in range(0, oi + 1):
            dS[n - 1 + s] += cur[oi - s]
        for s in range(1, oi + 1):
            dS[n - 1 - s] -= (1.0 - cur[oi - s])
        for s in range(1, n - oi):
            dS[n - 1 - s] += cur[oi + s]
        dS[n-1] += 1.0  # quadratic correction for removing a 1
        
        # Effect of h[oz] going from 0 to 1: delta = +1 at position oz
        for s in range(0, n - oz):
            dS[n - 1 + s] += (1.0 - cur[oz + s])
        for s in range(0, oz + 1):
            dS[n - 1 + s] -= cur[oz - s]
        for s in range(1, oz + 1):
            dS[n - 1 - s] += (1.0 - cur[oz - s])
        for s in range(1, n - oz):
            dS[n - 1 - s] -= cur[oz + s]
        dS[n-1] += 1.0  # quadratic correction for adding a 1
        
        new_c5 = float(np.max(corr + dS)) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur[oi] = 0.0
            cur[oz] = 1.0
            corr += dS
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_refine(h, deadline, radius_init=0.08, stall_limit=50, top_k=15):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _corr(h)
        idx = np.argsort(scores)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj])
            hi_b = min(radius, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        # Line search along p
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        
        corr = _corr(h)
        T_vec = np.zeros(2 * n - 1)
        for ai in range(top_k):
            pass  # We'll just evaluate directly
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        
        def _ft(t):
            h_t = h + t * p
            h_t = np.clip(h_t, 0.0, 1.0)
            return float(np.max(_corr(h_t))) * dx
        
        f1 = _ft(c1)
        f2 = _ft(c2)
        for _ in range(25):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = _ft(c1)
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = _ft(c2)
        opt_t = (lo_t + hi_t) / 2.0
        f_opt = _ft(opt_t)
        f0 = _ft(0.0)
        f_end = _ft(t_hi)
        
        candidates = [(0.0, f0), (opt_t, f_opt), (t_hi, f_end)]
        best_t = min(candidates, key=lambda x: x[1])[0]
        best_v = min(f0, f_opt, f_end)
        
        if best_v < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def _diversify(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.1 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    # Phase 1: Smooth-max gradient optimization (25% of budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _smooth_max_optimize(best_h, min(d1, time.time() + budget_s * 0.12),
                                     beta_start=3.0, beta_end=50.0, n_steps_target=5000, seed=77)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Coordinate descent (25%)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.12),
                               max_sweeps=3000, seed=123)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # LP refinement
    h2b, c5_2b = _lp_refine(best_h, min(d2, time.time() + budget_s * 0.06),
                             radius_init=0.06, stall_limit=40, top_k=15)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()
    
    # Another coord descent round
    h2c, c5_2c = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.07),
                                 max_sweeps=2000, seed=456)
    if c5_2c < best_c5:
        best_c5 = c5_2c
        best_h = h2c.copy()
    
    # Phase 3: Smooth-max with different parameters (15%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _smooth_max_optimize(best_h, min(d3, time.time() + budget_s * 0.08),
                                     beta_start=5.0, beta_end=60.0, n_steps_target=4000, seed=177)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Pair swaps
    h3b, c5_3b = _pair_swap(best_h, min(d3, time.time() + budget_s * 0.07),
                             n_iters=30000, seed=555)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()
    
    # Phase 4: Diversified search (20%)
    d4 = t0 + budget_s * 0.85
    n_restarts = 0
    scale = 0.03
    while time.time() < d4 and n_restarts < 40:
        n_restarts += 1
        h_pert = _diversify(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        # Quick smooth-max then coord descent
        h_t, c5_t = _smooth_max_optimize(h_pert, sub_dl, beta_start=5.0,
                                          beta_end=30.0, n_steps_target=800, seed=3000 + n_restarts)
        sub_dl2 = min(d4, time.time() + budget_s * 0.008)
        h_t2, c5_t2 = _coord_descent(h_t, sub_dl2, max_sweeps=500, seed=4000 + n_restarts)
        if c5_t2 < c5_t:
            c5_t = c5_t2
            h_t = h_t2
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)
    
    # Phase 5: Binary exploration (8%)
    d5 = t0 + budget_s * 0.93
    h_bin = _to_binary(best_h)
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=50000, T_init=0.002, seed=999)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP on binary
    sub_dl2 = min(d5, time.time() + budget_s * 0.02)
    h_pol = _project(h_bin.copy(), n0 / 2.0)
    h_pol, c5_pol = _lp_refine(h_pol, sub_dl2, radius_init=0.05, stall_limit=30, top_k=12)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Phase 6: Multi-n exploration (7%)
    d6 = deadline
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]:
        if time.time() > d6 - 5:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 300 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.008)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=400, seed=7000 + n_try)
        if time.time() < d6 - 6:
            sub_dl2 = min(d6 - 5, time.time() + budget_s * 0.005)
            h_r, c5_r = _smooth_max_optimize(h_r, sub_dl2, beta_start=5.0,
                                              beta_end=30.0, n_steps_target=500, seed=8000 + n_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Final polish
    if time.time() < deadline - 3:
        n_final = len(best_h)
        sub_dl = min(deadline - 2, time.time() + budget_s * 0.02)
        h_f, c5_f = _coord_descent(best_h, sub_dl, max_sweeps=2000, seed=9999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < deadline - 2:
            sub_dl2 = min(deadline - 1, time.time() + budget_s * 0.01)
            h_f, c5_f = _pair_swap(best_h, sub_dl2, n_iters=10000, seed=7777)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
    
    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
