def _binary_best_improvement(h, deadline, seed):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(corr.max() * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h <= 0.5)[0])
    no_improve = 0
    
    while time.time() < deadline:
        if no_improve > 500:
            break
        best_delta = 0  # We want to minimize max, so best improvement is most negative change in max
        best_a = None
        best_b = None
        
        # Full sweep: try all (a, b) pairs
        improved = False
        for a in ones:
            for b in zeros:
                delta = _swap_delta(h, a, b, n, corr)
                new_max = float(np.max(corr + delta))
                if new_max < cur_c5 - 1e-12:
                    # This swap reduces the max
                    # But we want the BEST reduction
                    # Actually, we want to minimize the new max
                    if new_max < best_delta or best_a is None:
                        best_delta = new_max
                        best_a = a
                        best_b = b
                        improved = True
        
        if best_a is not None:
            a, b = best_a, best_b
            h[a] = 0.0
            h[b] = 1.0
            ones.remove(a)
            zeros.remove(b)
            ones.add(b)
            zeros.add(a)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(corr.max() * dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            break
    
    return best_h, best_c5
