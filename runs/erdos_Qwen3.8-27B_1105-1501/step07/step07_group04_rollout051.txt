import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k, n):
    """Gradient of correlation at lag k with respect to h."""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        # F = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
        # dF/dh[j] for j in [k, n-1]: (1 - h[j-k])
        # dF/dh[j] for j in [0, n-k-1]: -h[j+k]
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        u = -k
        # F = sum_{i=0}^{n-u-1} h[i]*(1-h[i+u])
        # dF/dh[j] for j in [0, n-u-1]: (1 - h[j+u])
        # dF/dh[j] for j in [u, n-1]: -h[j-u]
        g[:n-u] += (1.0 - h[u:])
        g[u:] -= h[:n-u]
    return g

def _lp_direction(h, radius, n_active=40, eps=1e-5):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    
    # Select active constraints: top n_active by value
    active_idx = np.argsort(scaled)[::-1][:n_active]
    
    m = len(active_idx)
    n_vars = n + m + 1  # d[0..n-1], |d| slacks, t
    
    # Actually let's use a simpler formulation:
    # Variables: d[0..n-1] (free, bounded by [-radius, radius]), t (free)
    # Constraints: grad_r · d + F_r <= t for each active r
    #               sum d = 0
    # Objective: minimize t + eps * sum(|d_i|)
    
    # For |d_i| we need extra variables u_i >= d_i, u_i >= -d_i
    # Total vars: n (d) + n (u) + 1 (t) = 2n+1
    
    n_vars = 2 * n + 1
    t_idx = 2 * n
    u_start = n  # u variables are d[n..2n-1]
    
    # Objective: minimize t + eps * sum(u_i)
    c_obj = np.zeros(n_vars)
    c_obj[t_idx] = 1.0
    c_obj[u_start:2*n] = eps
    
    # Inequality constraints A_ub @ x <= b_ub
    # For each active constraint r:
    #   grad_r · d - t <= -F_r
    #   i.e., [grad_r, 0] · [d; u; t] <= -F_r
    
    A_rows = []
    b_rows = []
    
    for r in active_idx:
        k = int(r) - center
        g = _grad_row(h, k, n) * dx
        row = np.zeros(n_vars)
        row[0:n] = g
        row[t_idx] = -1.0
        A_rows.append(row)
        b_rows.append(-float(scaled[int(r)]))
    
    # Mass constraint: sum d = 0
    # This is an equality, so we add two inequalities or handle separately
    # Let's use A_eq, b_eq
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 0:n] = 1.0
    b_eq = [0.0]
    
    # u_i >= d_i:  d_i - u_i <= 0
    # u_i >= -d_i: -d_i - u_i <= 0
    for i in range(n):
        row1 = np.zeros(n_vars)
        row1[i] = 1.0
        row1[n + i] = -1.0
        A_rows.append(row1)
        b_rows.append(0.0)
        
        row2 = np.zeros(n_vars)
        row2[i] = -1.0
        row2[n + i] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds: d_i in [-radius, radius], u_i in [0, 2*radius], t in [cur-0.5, cur+1]
    cur_c5 = float(np.max(scaled))
    bounds = []
    for i in range(n):
        bounds.append((-radius, radius))
    for i in range(n):
        bounds.append((0.0, 2.0 * radius))
    bounds.append((cur_c5 - 0.5, cur_c5 + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[0:n]
            t_val = res.x[t_idx]
            # Check mass constraint residual
            if abs(float(np.sum(d))) > 1e-6:
                d = d - np.mean(d)
            return d, t_val
        else:
            return None, None
    except Exception:
        return None, None

def _line_search(h, d, dx, radius):
    """Exact line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    grid_t = np.linspace(0.0, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(45):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    candidates = [(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best_c = min(candidates, key=lambda x: x[1])
    return best_c[0], best_c[1]

def _lp_optimize(h, deadline, radius_init=0.08, n_active=40, stall_limit=50):
    """Main LP-based optimization loop."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < stall_limit:
        if radius < 1e-6:
            break
        d, t_pred = _lp_direction(h, radius, n_active=n_active)
        if d is None:
            # Fallback: subgradient
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_val = float(corr.max())
            active = np.where(corr >= max_val * dx - 1e-10 / dx)[0]
            g = np.zeros(n)
            for m_idx in active:
                k = int(m_idx) - (n - 1)
                g += _grad_row(h, k, n)
            g /= max(1, len(active))
            d = -g - np.mean(-g)
        
        d = d - np.mean(d)  # ensure zero sum
        
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.7
            n_active = min(n_active + 8, 80)
    
    return best_h, best_c5

            # The main affected lag is diff (and -diff)

            idx2 = n - 1 - diff

def _binary_sa(h_bin, deadline, max_iters=100000, temp_init=0.004, seed=42):

    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 5000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))

            T = max(0.0003, T * 0.9998)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_restart(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _multi_pair_swap(h, deadline, n_pairs=5000, seed=111):

def _project(v, target):

        else:
            hi = mid

    res = target - float(np.sum(x))

def _grad_row(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        u = -k
        g[:n - u] += (1.0 - h[u:])
        g[u:] -= h[:n - u]

    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = _project(best_h.copy(), n_best / 2.0)
    h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=4000, radius=0.15, seed=666)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # More targeted binary
    if time.time() < d4 - 25:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_targeted(h_b, sub_dl, max_iters=100000, seed=777, n_try=50)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Basin hopping
    if time.time() < d4 - 20:
        scale = 0.012
        for _ in range(40):
            if time.time() > d4 - 5:
                break
            h_pert = _perturb_restart(best_h, scale, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.004)
            h_t, c5_t = _lp_optimize(h_pert, sub_dl, radius_init=0.04, n_active=55, stall_limit=18)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.002, scale * 0.8)
            else:
                scale = min(0.04, scale * 1.12)
    
    # Final LP polish
    if time.time() < d4 - 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_optimize(h6, d4 - 1, radius_init=0.03, n_active=80, stall_limit=25)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Try nearby n values one more time
    n_fin = len(best_h)
    for dn_f in [-2, 2, -4, 4, -1, 1, -3, 3]:
        if time.time() > d4 - 1:
            break
        n_try = n_fin + dn_f
        if n_try < 80:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 0.5, time.time() + 2.0)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, radius_init=0.03, n_active=50, stall_limit=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
