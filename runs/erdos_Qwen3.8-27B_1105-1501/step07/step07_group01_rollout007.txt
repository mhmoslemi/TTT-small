import numpy as np
from scipy.optimize import linprog
import time
import math

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0 - 1e-12)
        else:
            mask = (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smax_grad(h, beta):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    Ss = S - np.max(S)
    w = np.exp(beta * Ss)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    W = np.zeros(n)
    W[0] = w[n - 1]
    if n > 1:
        W[1:] = w[n:] + w[n-2::-1]
    grad = W[0] * (1.0 - 2.0 * h)
    for d in range(1, n):
        if W[d] < 1e-16:
            continue
        m = n - d
        grad[:m] += W[d] * (1.0 - h[d:])
        grad[d:] -= W[d] * h[:m]
    return grad

def _smax_optimize(h0, t_start, time_frac, beta_init, beta_final, seed=42):
    t_start = float(t_start)
    elapsed = lambda: time.time() - t_start
    total_time = time_frac if time_frac is not None else float('inf')
    deadline = t_start + (time_frac if time_frac is not None else 1e9)
    n = len(h0)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h = _project(h0.copy(), target)
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    L = 1.0 / max(2.0 * n * n, 1.0)
    L = max(L, 0.02 / (n * n))
    n_iter = 0
    stall = 0
    stall_limit = 1500
    while elapsed() < max(0.5, total_time) and n_iter < 60000:
        progress = (time_frac is not None) and (n_iter / float(max(n_iter,1)) < 1.0)
        frac = min(1.0, elapsed() / max(total_time, 0.1))
        beta = beta_init + (beta_final - beta_init) * frac
        g = _smax_grad(h, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        gnorm = g / gn
        best_f = cur_c5
        best_t = 0.0
        t_backtrack = 1.0 / (L * gn)
        improved = False
        for _ in range(45):
            t_try = t_backtrack
            if t_try < 1e-12:
                break
            h_new = _project(h + t_try * g, target)
            f_new = _c5(h_new)
            if f_new < best_f - 1e-15:
                best_f = f_new
                best_t = t_try
                improved = True
                break
            t_backtrack *= 0.5
        if improved:
            h = _project(h + best_t * g, target)
            cur_c5 = best_f
            n_iter += 1
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > stall_limit:
                h = _smooth_perturb(best_h, 0.03, rng)
                cur_c5 = _c5(h)
                stall = 0
                t_backtrack = 1.0 / (L * gn)
        else:
            stall += 1
            t_backtrack *= 0.5
            if stall > stall_limit:
                h = _smooth_perturb(best_h, 0.04, rng)
                cur_c5 = _c5(h)
                stall = 0
                t_backtrack = 1.0 / (L * gn)
                n_iter += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2.0 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap(h, deadline, n_trials=6000, radius=0.30, seed=555):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S0)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        pos = d > 1e-15
        neg = d < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            continue
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * (2.0 / n)
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * (2.0 / n)
        for _ in range(35):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * (2.0 / n)
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * (2.0 / n)
        opt_t = (lo_t + hi_t) / 2.0
        vals = [f1, f2, float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * (2.0 / n)]
        opts = [c1, c2, opt_t]
        bvi = int(np.argmin(vals))
        if vals[bvi] < cur_c5 - 1e-14:
            h = h + opts[bvi] * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * (2.0 / n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_refine(h, deadline, top_k=12, radius_init=0.08, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(scores)[-top_k:]
        G = np.zeros((top_k, n))
        for ai, d_idx in enumerate(idx):
            d = int(d_idx - (n - 1))
            if d >= 0:
                m = n - d
                if m > 0:
                    G[ai, d:] += (1.0 - h[:m])
                    G[ai, :m] += -h[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    G[ai, :m] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:m]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            bounds.append((max(-radius, -h[jj]), min(radius, 1.0 - h[jj])))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-8:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
        C = np.correlate(p, p, mode='full')
        pos = p > 1e-15
        neg = p < -1e-15
        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-p[neg]))))
        if t_hi < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * (2.0 / n)
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * (2.0 / n)
        for _ in range(40):
            if hi_t - lo_t < 1e-12:
                break
            if f1 < f2:
                hi_t = c2; c2 = c1; f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * (2.0 / n)
            else:
                lo_t = c1; c1 = c2; f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * (2.0 / n)
        opt_t = (lo_t + hi_t) / 2.0
        candidates = [(0.0, float(np.max(S0)) * (2.0 / n)),
                      (opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * (2.0 / n)),
                      (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * (2.0 / n))]
        bt, bv = min(candidates, key=lambda x: x[1])
        if bv < cur_c5 - 1e-12:
            h = h + bt * p
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.7
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    h_init = _project(h0.copy(), n0 / 2.0)

    smax_time = budget_s * 0.45

    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement (12%)
    lp_time = budget_s * 0.12
    d2 = t0 + budget_s * 0.57
    h2, c5_2 = _lp_refine(best_h, d2, top_k=14, radius_init=0.06, stall_limit=50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair swap local refinement (10%)
    d3 = t0 + budget_s * 0.67
    h3, c5_3 = _pair_swap(best_h, d3, n_trials=5000, radius=0.25, seed=777)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Second smooth-max pass with different params (10%)
    d4 = t0 + budget_s * 0.77
    smax_time4 = min(budget_s * 0.10, d4 - time.time())
    if smax_time4 > 5:
        h4, c5_4 = _smax_optimize(best_h, time.time(), smax_time4, beta_init=8.0, beta_final=120.0, seed=200)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.92
    scale = 0.025
    n_restarts = 0
    while time.time() < d5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 1, time.time() + budget_s * 0.006)
        h_t, c5_t = _smax_optimize(h_pert, time.time(), max(2.0, sub_dl - time.time()),
                                   beta_init=5.0, beta_final=50.0, seed=3000 + n_restarts)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.90)
        else:
            scale = min(0.08, scale * 1.10)

    # Phase 6: Final polish (8%)
    d6 = deadline
    if time.time() < d6 - 3:
        sub_time = min(budget_s * 0.04, d6 - time.time() - 2)

    # Final LP polish

        h6, c5_6 = _lp_refine(best_h, d6 - 1, top_k=16, radius_init=0.03, stall_limit=30)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, len(best_h)
