def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    k = n // 2
    idx = np.argsort(h)[-k:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

    m = len(edge_list)
    f = cp.Variable(m)
    constraints = [f >= 0, f <= cap_list]
    
    # Flow conservation: for each node v != s, t
    # sum of inflow - sum of outflow = 0
    # Build incidence matrix
    from scipy.sparse import coo_matrix
    rows = [], cols = [], vals = []
    for e, (u, v) in enumerate(edge_list):
        # Outflow from u: +f_e
        rows.append(u); cols.append(e); vals.append(1.0)
        # Inflow to v: -f_e
        rows.append(v); cols.append(e); vals.append(-1.0)
    
    B = coo_matrix((vals, (rows, cols)), shape=(n_nodes, m)).toarray()
    
    # For nodes other than s and t: B[node] @ f = 0
    cons = []

def _max_flow(n, s, t, edges, caps):
    """
    n: number of nodes
    s, t: source and sink
    edges: list of (u, v) - directed edges
    caps: list of capacities (float)
    Returns: (flow_value, source_side_set)
    """
    m = len(edges)
    # Residual capacities: 2*m entries
    # Edge i (0..m-1): forward, residual cap initially = caps[i]
    # Edge m+i: reverse of edge i, residual cap initially = 0
    rcap = caps[:] + [0.0] * m
    
    # Adjacency: for each node, list of edge indices (both forward and reverse)
    adj = [[] for _ in range(n)]
    for i in range(m):
        u, v = edges[i]
        adj[u].append(i)       # forward edge i
        adj[v].append(m + i)   # reverse edge m+i
    
    def neighbor(ei, u):
        """Given edge index ei and the node u it's outgoing from, return the other node."""
        if ei < m:
            # forward edge
            return edges[ei][1]
        else:
            # reverse edge, corresponds to forward edge ei-m
            return edges[ei - m][0]
    
    # Push-relabel with highest label selection
    excess = [0.0] * n
    height = [0] * n
    height[s] = n
    # count[h] = number of vertices at height h
    count = [0] * (2 * n + 1)
    count[0] = n - 1
    count[n] = 1
    
    # Active vertices (those with excess > 0 and not s or t)
    from collections import deque
    active = [deque() for _ in range(2 * n)]  # active[h] = deque of vertices at height h with excess
    max_h = 0
    active_count = 0
    
    def activate(v):
        nonlocal active_count, max_h
        if v != s and v != t and excess[v] > 0 and height[v] < 2 * n:
            if not [x for x in active[height[v]]]:  # this is O(1) check... not quite
                active_count += 1
            active[height[v]].append(v)
            max_h = max(max_h, height[v])
    
    # Initialize: push from s
    for ei in adj[s]:
        v = neighbor(ei, s)
        if rcap[ei] > 0:
            push = rcap[ei]
            rcap[ei] -= push
            rcap[ei ^ m] += push  # XOR with m gives reverse
            excess[v] += push
            activate(v)
    
    # Discharge
    def relabel(v):
        old_h = height[v]
        min_h = 2 * n
        for ei in adj[v]:
            if rcap[ei] > 0:
                w = neighbor(ei, v)
                min_h = min(min_h, height[w])
        if min_h < 2 * n:
            new_h = min_h + 1
        else:
            new_h = 2 * n  # cut off
        height[v] = new_h
        count[old_h] -= 1
        count[new_h] = 1
        # Gap heuristic
        if count[old_h] == 0:
            for u in range(n):
                if u != s and u != t and height[u] > old_h and height[u] < 2 * n:
                    count[height[u]] -= 1
                    height[u] = 2 * n + 1  # or some large value
                    count[height[u]] += 1
    
    def push(v, ei):
        w = neighbor(ei, v)
        amt = min(excess[v], rcap[ei])
        rcap[ei] -= amt
        rcap[ei ^ m] += amt
        excess[v] -= amt
        excess[w] += amt
        activate(w)
    
    # Main loop
    while active_count > 0:
        # Find highest active vertex
        while max_h >= 0 and not active[max_h]:
            max_h -= 1
        if max_h < 0:
            break
        v = active[max_h].popleft()
        if excess[v] <= 0 or height[v] != max_h:
            continue
        active_count -= 1
        
        # Discharge v
        done = False
        while excess[v] > 0 and not done:
            # Try to push along an admissible edge
            pushed = False
            for ei in adj[v]:
                w = neighbor(ei, v)
                if rcap[ei] > 0 and height[v] == height[w] + 1:
                    push(v, ei)
                    pushed = True
                    break
            if not pushed:
                relabel(v)
                if height[v] >= 2 * n:
                    done = True
        
        if excess[v] > 0 and height[v] < 2 * n:
            activate(v)
    
    # Flow value
    flow = excess[t]
    
    # Find source side: BFS from s in residual graph
    visited = [False] * n
    visited[s] = True
    q = deque([s])

        for ei in adj[u]:
            if rcap[ei] > 0:
                w = neighbor(ei, u)
                if not visited[w]:
                    visited[w] = True
                    q.append(w)
    
    return flow, visited
