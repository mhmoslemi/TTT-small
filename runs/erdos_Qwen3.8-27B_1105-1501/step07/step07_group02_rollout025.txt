    ts = np.linspace(t_min, t_max, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))

    hi = ts[min(n_grid - 1, best_idx + 2)]

    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    f2 = float(np.max(R0 + c2 * A - c2**2 * B))
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(R0 + c1 * A - c1**2 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2**2 * B))
    t_opt = (lo + hi) / 2.0
    best_val = min(f1, f2, float(np.max(R0 + t_opt * A - t_opt**2 * B)),
                   float(vals[best_idx]),
                   float(np.max(R0 + t_min * A - t_min**2 * B)),

            if vc < best_val:
                best_val = vc
                best_t = tc

def _lp_solve(G, peaks, n):
    """Solve LP for direction. Returns direction vector or None."""
    K = len(peaks)
    # Variables: sp (n), sn (n), beta (1) → 2n+1 total
    nv = 2 * n + 1
    # Objective: minimize -beta → c[nv-1] = -1
    c = np.zeros(nv)
    c[-1] = -1.0
    
    # Inequality: G[m,:] @ (sp - sn) + beta <= 0 for each peak m
    A_ub = np.zeros((K, nv))
    b_ub = np.zeros(K)
    for i, m in enumerate(peaks):
        A_ub[i, :n] = G[m, :]
        A_ub[i, n:2*n] = -G[m, :]
        A_ub[i, nv-1] = 1.0
    
    # Equality: sum(sp) - sum(sn) = 0, sum(sp) + sum(sn) = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(-1e6, 1e6)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            sp = res.x[:n]
            sn = res.x[n:2*n]
            u = sp - sn
            l1 = np.sum(np.abs(u))
            if l1 > 1e-10:
                u /= l1  # Normalize to L1 = 1
            return u, float(res.x[-1])
    except Exception:
        pass
    
    # Fallback: try without presolve
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            sp = res.x[:n]
            sn = res.x[n:2*n]
            u = sp - sn
            l1 = np.sum(np.abs(u))
            if l1 > 1e-10:
                u /= l1
            return u, float(res.x[-1])
    except Exception:
        pass
    
    return None, None

def _resample(h, n_new):

        lp_res = _lp_direction(G, peaks, n)
        if lp_res[0] is not None:
            d, beta = lp_res
            if beta < -1e-8:
                t_opt = _line_search(h, d, R0, omh, t_cap=0.7)
                if abs(t_opt) > 1e-9:
                    h_new = _project(h + t_opt * d, n / 2.0)
                    c_new = _exact_c5(h_new)
                    if c_new < cur_c5 - 1e-12:
                        h = h_new
                        omh = 1.0 - h
                        cur_c5 = c_new
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
        if not improved:
            M = float(np.max(R0))
            w = np.exp(12.0 * (R0 - M))
            ws = float(np.sum(w))
            if ws < 1e-30:
                w = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                w /= ws
            cg = G.T @ w
            n_try = min(25, n // 2)
            sources = np.argsort(cg)[-n_try:]
            sinks = np.argsort(cg)[:n_try]
            best_pc5 = cur_c5
            best_ph = None
            for _ in range(n_try):
                si = int(sinks[rng.integers(n_try)])
                so = int(sources[rng.integers(n_try)])
                if si == so:
                    continue
                d2 = np.zeros(n)
                d2[si] = 1.0
                d2[so] = -1.0
                t2 = _line_search(h, d2, R0, omh, t_cap=0.5)
                if abs(t2) > 1e-9:
                    h_try = _project(h + t2 * d2, n / 2.0)
                    c_try = _exact_c5(h_try)
                    if c_try < best_pc5 - 1e-13:
                        best_pc5 = c_try
                        best_ph = h_try
            if best_ph is not None:
                h = best_ph
                omh = 1.0 - h
                cur_c5 = best_pc5
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
                if stall % 4 == 0:
                    cycle += 1
                if stall > 25 and rng.random() < 0.4:
                    dr = rng.standard_normal(n)
                    dr -= dr.mean()
                    dr /= np.sum(np.abs(dr))
                    tr = _line_search(h, dr, R0, omh, t_cap=0.4)
                    if abs(tr) > 1e-8:
                        hr = _project(h + tr * dr, n / 2.0)
                        cr = _exact_c5(hr)
                        if cr < cur_c5 - 1e-12:
                            h = hr
                            omh = 1.0 - h
                            cur_c5 = cr
                            stall = 0
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                if stall > 80:
                    break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_sa(hb, deadline, seed=999):
    n = len(hb)
    cur = hb.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.004
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    for _ in range(40000):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        cur[i], cur[j] = cur[j], cur[i]
        nc5 = _exact_c5(cur)
        delta = nc5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = nc5
            ones[np.where(ones == i)[0]] = j
            ones[np.where(ones == j)] = i
            zeros[np.where(zeros == i)[0]] = s if (s := j) else i
            zeros[np.where(zeros == j)] = i
            # Fix: swap labels properly
            ones_list = list(ones)
            zeros_list = list(zeros)
            if i in ones_list:
                ones_list[ones_list.index(i)] = j
            if j in zeros_list:
                zeros_list[zeros_list.index(j)] = i
            ones = np.array(ones_list)
            zeros = np.array(zeros_list)
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = cur.copy()
            T *= 0.9997
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    # Phase 1: LP optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c51 = _local_optimize(h, d1, seed=1, peak_counts=[8, 12, 18, 28, 40])
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()
    # Phase 2: Multi-n (20%)
    d2 = t0 + budget_s * 0.65
    offsets = [0, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 16, -16]
    tried = set()
    for dn in offsets:
        if time.time() > d2:
            break
        nt = n0 + dn
        if nt < 80 or nt > 240 or nt % 2 != 0 or nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        dl = min(d2, time.time() + budget_s * 0.014)
        hr, cr = _local_optimize(hr, dl, seed=300 + dn, peak_counts=[8, 14, 22])
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
    # Phase 3: Binary diversification (15%)
    d3 = t0 + budget_s * 0.80
    nc = len(best_h)
    if time.time() < d3:
        hb = _to_binary(best_h, nc)
        dl = min(d3, time.time() + budget_s * 0.07)
        hb, cb = _binary_sa(hb, dl, seed=777)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        if time.time() < d3:
            dl2 = min(d3, time.time() + budget_s * 0.05)
            hp, cp = _local_optimize(hb.copy(), dl2, seed=444, peak_counts=[8, 14, 24])
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()
    # Phase 4: Final refinement (20%)
    d4 = deadline - 3.0
    nc = len(best_h)
    h4 = _project(best_h.copy(), nc / 2.0)
    h4, c54 = _local_optimize(h4, d4, seed=888, peak_counts=[10, 16, 24, 32, 44])
    if c54 < best_c5:
        best_c5 = c54
        best_h = h4.copy()
    # Final projection
    best_h = _project(best_h, len(best_h) / 2.0)
    fc5 = _exact_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5
    return best_h, best_c5, len(best_h)
