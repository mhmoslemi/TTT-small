    n = len(v)
    if target <= 0: return np.zeros(n)

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _soft_weights(R, beta):
    M = float(np.max(R))
    if M < 1e-15:
        return np.ones(len(R)) / len(R)
    w = np.exp(beta * (R - M))
    s = float(np.sum(w))
    if s < 1e-30:
        top_k = max(1, len(R) // 10)
        idx = np.argsort(R)[-top_k:]
        w = np.zeros(len(R))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _dense_gradient(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if abs(wi) < 1e-15:
            continue
        delta = k - (n - 1)
        # First term: i from max(0,delta) to min(n, n+delta), add wi*(1-h[i-delta])
        i_lo = max(0, delta)
        i_hi = min(n, n + delta)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wi * (1.0 - h[i_lo - delta:i_hi - delta])
        # Second term: i from max(0,-delta) to min(n, n-delta), subtract wi*h[i+delta]
        i_lo2 = max(0, -delta)
        i_hi2 = min(n, n - delta)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= wi * h[i_lo2 + delta:i_hi2 + delta]
    return grad

def _line_search(R0, A, B, t_max, n_grid=80):

        return 0.0, float(np.max(R0))
    ts = np.linspace(0.0, t_max, n_grid)
    vals = np.max(R0[None, :] + ts[:, None] * A[None, :] - (ts[:, None])**2 * B[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 3)]
    hi = ts[min(n_grid - 1, best_idx + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(R0 + c1 * A - c1*c1 * B))
    f2 = float(np.max(R0 + c2 * A - c2*c2 * B))

            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(R0 + c2 * A - c2*c2 * B))
    t_opt = (lo + hi) / 2.0

    best = min(candidates, key=lambda x: x[0])

def _dense_descent(h, deadline, beta_start=3.0, beta_end=60.0, max_iter=2000, seed=42):

    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_start
    stall = 0
    rng = np.random.default_rng(seed)

            continue

        one_mh = 1.0 - h
        A = np.correlate(d, one_mh, mode='full') - np.correlate(h, d, mode='full')
        B = np.correlate(d, d, mode='full')
        
        # Line search
        t_opt, val_opt = _line_search(R, A, B, t_max)

            cur_c5 = c5_new

                    stall = 0
