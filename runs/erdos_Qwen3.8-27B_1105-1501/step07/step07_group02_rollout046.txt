def _slsqp_grad(z, n):

def _slsqp_constraints(z, n):
    # Inequality: t - dx * corr_r >= 0 for each r
    # Equality: sum(h) - n/2 = 0
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n

    L = len(corr)
    
    fun = np.zeros(L + 1)

        jac[r, -1] = 1.0

    fun[-1] = np.sum(h) - n / 2.0
    jac[-1, :n] = 1.0

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * dx * corr
    max_s = float(np.max(scaled))
    exp_s = np.exp(scaled - max_s)
    sum_e = float(np.sum(exp_s))
    if sum_e < 1e-300:
        return np.zeros(n)
    w = exp_s / sum_e
    grad = np.zeros(n)
    for r in range(len(corr)):
        if w[r] < 1e-14:
            continue
        ell = r - (n - 1)
        ww = w[r] * dx
        js1 = max(0, ell)
        je1 = min(n, ell + n)
        if je1 > js1:
            grad[js1:je1] += ww * (1.0 - h[js1 - ell:je1 - ell])
        js2 = max(0, -ell)
        je2 = min(n, n - ell)
        if je2 > js2:
            grad[js2:je2] -= ww * h[js2 + ell:je2 + ell]
    return grad

def _slsqp_solve(h_init, deadline, maxiter=80, seed=42):
    n = len(h_init)
    target = n / 2.0
    h0 = _project(h_init.copy(), target)
    t0_val = _exact_c5(h0) + 0.005
    x0 = np.concatenate([h0, [t0_val]])
    dx = 2.0 / n
    L = 2 * n - 1

    def _obj(z):
        return z[-1]

    def _obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def _con_fun(z):
        h = z[:n]
        t = z[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.concatenate([t - dx * corr, [np.sum(h) - target]])

    def _con_jac(z):
        h = z[:n]
        J = np.zeros((L + 1, n + 1))
        for r in range(L):
            ell = r - (n - 1)
            js1 = max(0, ell)
            je1 = min(n, ell + n)
            if je1 > js1:
                J[r, js1:je1] = -dx * (1.0 - h[js1 - ell:je1 - ell])
            js2 = max(0, -ell)
            je2 = min(n, n - ell)
            if je2 > js2:
                J[r, js2:je2] += dx * h[js2 + ell:je2 + ell]
            J[r, -1] = 1.0
        J[-1, :n] = 1.0
        return J

    cons = []
    for r in range(L):
        cons.append({'type': 'ineq', 'fun': lambda z, r=r: z[-1] - dx * float(np.correlate(z[:n], 1.0 - z[:n], mode='full')[r]),
                     'jac': lambda z, r=r: _con_jac(z)[r]})
    cons.append({'type': 'eq', 'fun': lambda z: np.sum(z[:n]) - target,
                 'jac': lambda z: _con_jac(z)[-1]})

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    start_time = time.time()
    try:
        res = minimize(_obj, x0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

    except Exception:

        c5_sol = _exact_c5(h_sol)
        return h_sol, c5_sol
    return h0, t0_val

def _line_search(S0, T, C, t_lo, t_hi):
    if t_hi - t_lo < 1e-13:
        return t_lo, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (lo + hi) / 2.0
    return opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C))

                                         beta_init=20.0, beta_end=400.0, seed=101)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    # Phase 2: SLSQP at multiple resolutions (25% time)
    d2 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    for n_try in [n0, n0 - 20, n0 + 20, n0 - 40, 120, 100]:
        if time.time() > d2:
            break
        if n_try < 60 or n_try > 220:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _slsqp_solve(h_r, min(d2, time.time() + 0.15 * budget_s), maxiter=60)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = _resample(h_r, n_curr)

    # Phase 3: Block moves (15% time)
    d3 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    h_b, c5_b = _block_move_optimize(best_h, min(d3, time.time() + 0.12 * budget_s),
                                      n_blocks=10, block_size=4, seed=202)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 4: Pair-based polish (20% time)
    d4 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    h_p, c5_p = _pair_optimize(best_h, min(d4, time.time() + 0.18 * budget_s),
                                radius=0.15, n_pairs=50, beta=20.0, stall_limit=100, seed=303)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 5: Additional gradient + nearby n (10% time)
    d5 = deadline - 3
    n_curr = len(best_h)
    h_g2, c5_g2 = _gradient_step_optimize(best_h, min(d5, time.time() + 0.05 * budget_s),
                                           beta_init=100.0, beta_end=800.0, seed=404)
    if c5_g2 < best_c5:
        best_c5 = c5_g2
        best_h = h_g2.copy()

    for dn in [-4, 4, -8, 8]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 64 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _gradient_step_optimize(h_r, min(d5, time.time() + 0.02 * budget_s),
                                             beta_init=50.0, beta_end=400.0, seed=500 + dn)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Final polish
    if time.time() < deadline - 5:
        h_final, c5_final = _pair_optimize(best_h, deadline - 2,
                                            radius=0.08, n_pairs=30, beta=30.0,
                                            stall_limit=40, seed=999)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final projection and verification
    n_out = len(best_h)
    best_h = _project(best_h, n_out / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        best_h = _project(h_init.copy(), n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
