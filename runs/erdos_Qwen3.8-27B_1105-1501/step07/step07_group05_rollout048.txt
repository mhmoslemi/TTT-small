import numpy as np
import time
import math
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_grad(h, t):
    """Gradient of raw correlation at output index t with respect to h."""
    n = len(h)
    lag = t - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += 1.0 - h[lag:lag+L]
            g[lag:lag+L] -= h[:L]
    else:
        s = -lag
        L = n - s
        if L > 0:
            g[s:s+L] += 1.0 - h[:L]
            g[:L] -= h[s:s+L]
    return g

def _line_search_exact(h, d, dx):
    """Exact line search along h + a*d, where d has sum 0 and is bounded."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find bounds on alpha
    pos = d > 1e-14
    neg = d < -1e-14
    a_hi = 1.0
    a_lo = -1.0
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Candidate alphas
    cands = [0.0, a_lo, a_hi]
    
    # Quadratic optima for each output
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                cands.append(av)
    
    # Top M constraints' pairwise intersections
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    
    # Uniform grid
    width = a_hi - a_lo
    ng = min(40, max(20, int(width * 100)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    lo_r = max(a_lo, best_a - 0.03 * width)
    hi_r = min(a_hi, best_a + 0.03 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(20):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    
    return best_a, best_val

def _grad_direction(h, beta, top_k=0):
    """Generate a gradient-based direction."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    
    if top_k > 0:
        # Only use top_k lags with equal weight
        idx = np.argsort(-sc)[:top_k]
        w = np.zeros_like(sc)
        w[idx] = 1.0 / len(idx)
    else:
        # Softmax weighting
        v = np.clip(beta * (sc - mx), -80.0, 0.0)
        w = np.exp(v)
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.zeros_like(S)
            w[np.argmax(S)] = 1.0
        else:
            w /= ws
    
    g = np.zeros(n)
    for t in range(len(S)):
        wt = w[t]
        if wt < 1e-15:
            continue
        lag = t - (n - 1)
        if lag >= 0:
            L = n - lag
            if L > 0:
                g[:L] += wt * (1.0 - h[lag:lag+L])
                g[lag:lag+L] -= wt * h[:L]
        else:
            s = -lag
            L = n - s
            if L > 0:
                g[s:s+L] += wt * (1.0 - h[:L])
                g[:L] -= wt * h[s:s+L]
    g -= np.mean(g)
    return -g  # negative gradient

def _lp_direction(h, top_k, radius):
    """LP-based minimax direction finding."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    
    # Select active lags
    active = np.argsort(-sc)[:top_k]
    
    # Build gradient matrix
    Grads = np.zeros((len(active), n))
    for i, t in enumerate(active):
        Grads[i] = _lag_grad(h, int(t))
    
    # Normalize
    norms = np.linalg.norm(Grads, axis=1, keepdims=True)
    norms[norms < 1e-15] = 1.0
    Grads /= norms
    
    # LP: maximize tau s.t. Grads @ d <= -tau, sum(d)=0, -r*h <= d <= r*(1-h)
    # Variables: [d (n vars), tau (1 var)]
    # Objective: minimize -tau
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: Grads @ d + tau <= 0
    A_ub = np.hstack([Grads, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    
    # Bounds
    lo_b = -radius * h
    hi_b = radius * (1.0 - h)
    tau_lo = -1.0
    tau_hi = 1.0
    bounds = list(zip(lo_b, hi_b)) + [(tau_lo, tau_hi)]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    try:
        sol = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if sol.success:
            d = sol.x[:n]
            tau = sol.x[-1]
            if tau > 1e-6:
                d -= np.mean(d)
                nrm = np.linalg.norm(d)
                if nrm > 1e-12:
                    return d / nrm
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _pair_opt_v2(h, deadline, max_it=20000):

    for _ in range(max_it):

        else:
            stall += 1
    return best_h, best_c

def _project(v, target):

def _lag_grad(h, t):

    else:
        s = -lag
        L = n - s

            w[np.argmax(S)] = 1.0

            w /= ws

    for t in range(len(S)):
        wt = w[t]
        if wt < 1e-15:
            continue
        lag = t - (n - 1)
        if lag >= 0:
            L = n - lag
            if L > 0:
                g[:L] += wt * (1.0 - h[lag:lag+L])

            L = n - s
            if L > 0:
                g[s:s+L] += wt * (1.0 - h[:L])
                g[:L] -= wt * h[s:s+L]

def _lp_dir(h, top_k, radius):

    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    active = np.argsort(-sc)[:top_k]

    for i, t in enumerate(active):
        Grads[i] = _lag_grad(h, int(t))
    norms = np.linalg.norm(Grads, axis=1, keepdims=True)
    norms[norms < 1e-15] = 1.0
    Grads /= norms
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.hstack([Grads, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    lo_b = -radius * h
    hi_b = radius * (1.0 - h)
    bounds = list(zip(lo_b, hi_b)) + [(-1.0, 1.0)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    try:
        sol = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if sol.success and sol.x[-1] > 1e-6:
            d = sol.x[:n].copy()
            d -= np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm > 1e-12:

        pass
    return None

            pert += 0.3 * scale * rng.normal(0, 1, nc2)
            mxp = np.max(np.abs(pert))
            if mxp > 1e-10:
                pert *= scale / mxp
            hp = _project(best_h + pert, nc2 / 2.0)
            sd = min(deadline - 3, time.time() + budget_s * 0.004)
            ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15)
            if ct < best_c:
                best_c = ct
                best_h = ht.copy()
                scale = max(0.003, scale * 0.7)
            else:
                scale = min(0.08, scale * 1.15)

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
