def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    log_sum = np.log(np.sum(np.exp(tau * (R - R_max))))
    return (1.0 / tau) * log_sum + R_max

def _smooth_max_grad(h, tau):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    log_w = tau * (R - R_max)
    w = np.exp(log_w)

            if 0 <= m1 < n:

            m2 = i + s
            if 0 <= m2 < n:
                val += wi * (-h[m2])

    C = -np.correlate(d, d, mode='full')
