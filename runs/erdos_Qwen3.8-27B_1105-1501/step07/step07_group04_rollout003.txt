    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Get current correlation to identify active lags
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_val = corr.max()
        # Focus on lags near the maximum
        active = np.where(corr >= max_val - 0.01 * max_val)[0]
        
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5 = _exact_c5(trial)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _targeted_binary(h_bin, deadline, max_iters=30000, seed=99):
    """Targeted binary search: try all swaps involving positions in active lags."""
    n = len(h_bin)
    dx = 2.0 / n
    c = n - 1
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    n_ones = n // 2
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_val = corr.max()
        active_lags = np.where(corr >= max_val - 0.02)[0]
        
        best_delta = 0.0
        best_move = None
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        
        # Try targeted swaps
        n_try = min(100, len(ones) * len(zeros))
        for _ in range(n_try):
            # Focus on positions involved in active lags
            if len(active_lags) > 0:
                lag = active_lags[int(rng.integers(0, len(active_lags)))]
                k = lag - c
                if k >= 0:
                    # corr at lag k: h[i+k](1-h[i]) for i in [0, n-k-1]
                    # Try to reduce by swapping
                    valid_i = np.arange(0, n - k)
                    # Positions where h[i]=0 and h[i+k]=1 (these contribute)
                    contrib = np.where((cur[valid_i] <= 0.5) & (cur[valid_i + k] > 0.5))[0]
                    if len(contrib) > 0:
                        i_idx = contrib[int(rng.integers(0, len(contrib)))]
                        oi = valid_i[i_idx]  # position with 0
                        oz = valid_i[i_idx] + k  # position with 1
                        # Try: move 1 from oz to oi
                        trial = cur.copy()
                        trial[oz] = 0.0
                        trial[oi] = 1.0
                        new_c5 = _exact_c5(trial)
                        delta = new_c5 - cur_c5
                        if delta < best_delta:
                            best_delta = delta
                            best_move = (oi, oz)
                else:
                    ka = -k
                    valid_i = np.arange(0, n - ka)
                    contrib = np.where((cur[valid_i] > 0.5) & (cur[valid_i + ka] <= 0.5))[0]
                    if len(contrib) > 0:
                        i_idx = contrib[int(rng.integers(0, len(contrib)))]
                        oi = valid_i[i_idx] + ka  # position with 0
                        oz = valid_i[i_idx]  # position with 1
                        trial = cur.copy()
                        trial[oz] = 0.0
                        trial[oi] = 1.0
                        new_c5 = _exact_c5(trial)
                        delta = new_c5 - cur_c5
                        if delta < best_delta:
                            best_delta = delta
                            best_move = (oi, oz)
            else:
                oi = ones[int(rng.integers(0, len(ones)))]
                oz = zeros[int(rng.integers(0, len(zeros)))]
                trial = cur.copy()
                trial[oi] = 0.0
                trial[oz] = 1.0
                new_c5 = _exact_c5(trial)
                delta = new_c5 - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (oi, oz)
        
        if best_move is not None and best_delta < -1e-14:
            oi, oz = best_move
            cur[oi] = 1.0
            cur[oz] = 0.0
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_n_explore(h_best, c5_best, deadline, n_list):
    """Explore different n values."""
    best_h = h_best.copy()
    best_c5 = c5_best
    for n_try in n_list:
        if time.time() > deadline:
            break
        if n_try < 60 or n_try > 300:
            continue
        h_r = _resample(best_h, n_try)
        # Quick SLSQP
        h_r, c5_r = _slsqp_solve(h_r, 0.01, maxiter=50, tol=1e-9)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            # Continue with smaller tau
            h_r, c5_r2 = _slsqp_solve(h_r, 0.003, maxiter=80, tol=1e-10)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r.copy()
        # Try binary
        h_b = _to_binary(best_h if c5_r < c5_best else h_r, n_try)
        h_b, c5_b = _targeted_binary(h_b, min(deadline, time.time() + 3), max_iters=5000, seed=n_try)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    return best_h, best_c5

def _smooth_perturb_optimize(h, deadline, seed=777):
    """Basin hopping with smooth perturbations + SLSQP."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    amplitude = 0.03
    
    for _ in range(100):
        if time.time() > deadline - 3:
            break
        freqs = rng.uniform(0.5, 5.0, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        pert = np.zeros(n)
        x = np.arange(n, dtype=float)
        for f, p in zip(freqs, phases):
            pert += np.sin(2.0 * np.pi * f * x / n + p)
        pert *= amplitude / 3.0
        h_pert = _project(h + pert, n / 2.0)
        h_try, c5_try = _slsqp_solve(h_pert, 0.005, maxiter=60, tol=1e-9)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            amplitude = max(0.005, amplitude * 0.85)
        else:
            amplitude = min(0.08, amplitude * 1.1)
    
    return best_h, best_c5
