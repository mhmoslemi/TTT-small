def _compute_gradient(h, weights):
    """Compute gradient of weighted sum of correlations.
    h: array of length n
    weights: array of length L=2n-1, should sum to 1
    Returns: gradient array of length n (the gradient of dx * sum_k weights[k] * corr_k)
    """
    n = len(h)
    L = 2*n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for k in range(-(n-1), n):
        row = (n-1) + k
        if k >= 0:
            m = n - k
            # for j = k..n-1: h[j]*(1-h[j-k])
            # d/dh[m]: (1-h[m-k]) if m>=k, -h[m+k] if m < n-k
            G[row, k:n] += (1.0 - h[:m])  # m ranges from k to n-1, 1-h[m-k] where m-k ranges from 0 to m-1
            G[row, :m] -= h[k:n]  # m ranges from 0 to n-k-1, h[m+k] where m+k ranges from k to n-1

def _grad_matrix(h):
    n = len(h)
    L = 2*n - 1
    G = np.zeros((L, n))
    for k in range(-(n-1), n):
        row = (n-1) + k
        if k >= 0:
            m = n - k
            G[row, k:n] += (1.0 - h[:m])
            G[row, :m] += -h[k:n]
        else:
            p = -k
            m = n - p
            G[row, :m] += (1.0 - h[p:n])
            G[row, p:n] += -h[:m]
    return G

        G = _build_G(h)
        grad = dx * (G.T @ w)
        gn = float(np.sqrt(np.sum(grad * grad)))
        if gn < 1e-12:
            break
        step_a = alpha / gn
        h_try = _project(h - step_a * grad, target)
        c5_try = _exact_c5(h_try)
        if c5_try < c5_cur - 1e-15:
            h = h_try
            alpha = min(alpha * 1.05, 5.0)
            stall = 0
        else:
            alpha *= 0.7
            stall += 1
            if stall > 40:
                break
    return best_h, best_c5

def _pair_step(h, deadline, n_pairs=30, beta=20.0, max_rounds=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    G = _build_G(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    exp_c = np.exp(beta * (corr - mc))
    w = exp_c / float(np.sum(exp_c))
    coord_grad = G.T @ w
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        sg = np.argsort(coord_grad)
        sinks = sg[:n_pairs]
        sources = sg[-n_pairs:]
        improved = False
        for i in range(n_pairs):
            if time.time() > deadline:
                break
            s = int(sinks[i])
            src = int(sources[i])
            if s == src:
                continue
            t_lo = max(-h[s], h[src] - 1.0)
            t_hi = min(1.0 - h[s], h[src])
            if t_hi - t_lo < 1e-14:
                continue
            t_opt = (t_lo + t_hi) / 2.0
            best_t = t_opt
            best_val = _exact_c5(h)
            h_try = h.copy()
            for _ in range(20):
                h_try[s] = h[s] + t_opt
                h_try[src] = h[src] - t_opt
                v = _exact_c5(h_try)
                if v < best_val:
                    best_val = v
                    best_t = t_opt
                mid = (t_lo + t_hi) / 2.0
                if t_opt > mid:
                    t_lo = t_opt
                else:
                    t_hi = t_opt
                t_opt = (t_lo + t_hi) / 2.0
                if t_hi - t_lo < 1e-10:
                    break
            if best_val < best_c5 - 1e-14:
                h = h.copy()
                h[s] += best_t
                h[src] -= best_t
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                improved = True
                corr = np.correlate(h, 1.0 - h, mode='full')
                mc2 = float(np.max(corr))
                exp_c2 = np.exp(beta * (corr - mc2))
                w = exp_c2 / float(np.sum(exp_c2))
                coord_grad = G.T @ w
        if not improved:
            break
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_rounds=200):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in ones[:]:
            if time.time() > deadline:
                break
            for oz in zeros[:]:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.remove(oi)
                    zeros.remove(oz)
                    ones.append(oz)
                    zeros.append(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient descent at parent n (35% time)
    d1 = t0 + budget_s * 0.35
    betas_p1 = [10, 25, 50, 100, 200, 500]
    h1, c5_1 = _smooth_descent(h_init, d1, betas_p1, max_iter=120, alpha_init=1.0, reheat=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n smooth descent (25% time)
    d2 = t0 + budget_s * 0.60
    n_curr = len(best_h)
    n_list = [n0 - 8, n0 - 4, n0 + 4, n0 + 8, n0 - 12, n0 + 12, n0 - 16, n0 + 16]
    for n_try in n_list:
        if time.time() > d2:
            break
        if n_try < 80 or n_try > 220 or n_try == n_curr:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.025)
        h_r, c5_r = _smooth_descent(h_r, sub_dl, [30, 80, 150], max_iter=80, alpha_init=0.8, reheat=False)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Active-set subgradient refinement (15% time)
    d3 = t0 + budget_s * 0.75
    h3, c5_3 = _active_subgrad(best_h, d3, n_iter=300, eps=1e-3, alpha_init=1.0)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair-based refinement (10% time)
    d4 = t0 + budget_s * 0.85
    h4, c5_4 = _pair_step(best_h, d4, n_pairs=40, beta=25.0, max_rounds=60)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary local search (5% time)
    d5 = t0 + budget_s * 0.90
    n_final_try = len(best_h)
    h_bin = _to_binary(best_h, n_final_try)
    h_bin, c5_bin = _binary_local_search(h_bin, d5, max_rounds=100)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 6: Final active-set polish (5% time)
    d6 = deadline - 2
    h6, c5_6 = _active_subgrad(best_h, d6, n_iter=200, eps=1e-4, alpha_init=0.5)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final projection
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n_final
