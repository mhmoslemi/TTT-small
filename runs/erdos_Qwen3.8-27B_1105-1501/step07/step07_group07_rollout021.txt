import numpy as np
from scipy.optimize import minimize
import time

_g_tau = 0.05
_g_N = 180

def _exact_c5(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def _project_sum(v, target):
    N = len(v)
    if target <= 0:
        return np.zeros(N)
    if target >= N:
        return np.ones(N)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-12:
        m = (x < 1.0) if r > 0 else (x > 0.0)
        c = int(np.sum(m))
        if c > 0:
            x = x.copy()
            x[m] += r / c
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    N = len(h)
    G = np.zeros((2 * N - 1, N))
    G[N - 1] = 1.0 - 2.0 * h
    for k in range(1, N):
        m = N - k
        G[N - 1 + k, k:] += 1.0 - h[:m]
        G[N - 1 + k, :m] -= h[k:]
        G[N - 1 - k, :m] += 1.0 - h[k:]
        G[N - 1 - k, k:] -= h[:m]
    return G

def _lse_val(x):
    N = len(x)
    h = np.clip(x, 0.0, 1.0)
    raw = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    m = raw.max()
    u = (raw - m) / _g_tau
    z = np.exp(u)
    s = z.sum()
    if s < 1e-300:
        return float(m)
    return float(m + _g_tau * np.log(s))

def _lse_jac(x):
    N = len(x)
    h = np.clip(x, 0.0, 1.0)
    raw = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    m = raw.max()
    u = (raw - m) / _g_tau
    z = np.exp(u)
    s = z.sum()
    if s < 1e-300:
        p = np.zeros(2 * N - 1)
        p[np.argmax(raw)] = 1.0
    else:
        p = z / s
    return (2.0 / N) * (_build_G(h).T @ p)

def _eq_f(x):
    return np.sum(x) - _g_N / 2.0

def _eq_j(x):
    return np.ones(len(x))

def _lsq_solve(h, tau, maxit=60):
    global _g_tau, _g_N
    N = len(h)
    _g_tau = tau
    _g_N = N
    bds = [(0.0, 1.0)] * N
    try:
        r = minimize(_lse_val, h.copy(), jac=_lse_jac, method='SLSQP',
                     bounds=bds,
                     constraints=[{'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}],
                     options={'maxiter': maxit, 'ftol': 1e-12})
        x = _project_sum(np.clip(r.x, 0.0, 1.0), N / 2.0)
        return x
    except Exception:
        return h.copy()

def _lse_anneal(h, dl, t0v=0.08, t1v=0.002, ns=7, mi=60):
    bh = h.copy()
    bc = _exact_c5(h)
    taus = np.geomspace(t0v, t1v, ns)
    for tau in taus:
        if time.time() > dl - 1.0:
            break
        xn = _lsq_solve(bh, float(tau), mi)
        cn = _exact_c5(xn)
        if cn < bc - 1e-14:
            bc = cn
            bh = xn
    return bh, bc

def _coord_descent(h, dl, max_rounds=40, n_pairs=3000, seed_val=999):
    N = len(h)
    bh = h.copy()
    bc = _exact_c5(h)
    rng = np.random.default_rng(seed_val)
    for rnd in range(max_rounds):
        if time.time() > dl - 2.0:
            break
        improved = False
        pairs = rng.integers(0, N, size=(n_pairs, 2))
        for i, j in pairs:
            if i == j:
                continue
            t_lo = max(-bh[i], bh[j] - 1.0)
            t_hi = min(1.0 - bh[i], bh[j])
            if t_hi - t_lo < 1e-14:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = t_lo, t_hi
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            bh[i] += c; bh[j] -= c
            fc = _exact_c5(bh)
            bh[i] = bh[i] - c + d; bh[j] = bh[j] + c - d
            fd = _exact_c5(bh)
            bh[i] -= d; bh[j] += d
            for _ in range(35):
                if b - a < 1e-13:
                    break
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a)
                    bh[i] += c; bh[j] -= c
                    fc = _exact_c5(bh)
                    bh[i] -= c; bh[j] += c
                else:
                    a = c; c = d; fc = fd
                    d = a + gr * (b - a)
                    bh[i] += d; bh[j] -= d
                    fd = _exact_c5(bh)
                    bh[i] -= d; bh[j] += d
            t_opt = (a + b) / 2.0
            bh[i] += t_opt; bh[j] -= t_opt
            bh = _project_sum(bh, N / 2.0)
            f_opt = _exact_c5(bh)
            if f_opt < bc - 1e-13:
                bc = f_opt
                improved = True
        if not improved:
            break
    return bh, bc

def _to_binary(h):
    N = len(h)
    idx = np.argsort(h)[-N // 2:]
    hb = np.zeros(N)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    N = len(h)
    if n_new == N:
        return h.copy()
    xo = (np.arange(N) + 0.5) / N
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project_sum(np.interp(xn, xo, h), n_new / 2.0)

def _perturb(h, rng, amp):
    N = len(h)
    x = np.arange(N, dtype=float)
    pert = np.zeros(N)
    for _ in range(3):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0.0, 2.0 * np.pi)
        a = rng.uniform(0.3, 1.0)
        pert += a * np.sin(2.0 * np.pi * f * x / N + p)
    pert += 0.1 * amp * rng.normal(0.0, 1.0, N)
    pert *= amp
    return _project_sum(h + pert, N / 2.0)

def _binary_sa(hb, dl, max_it=30000, Ti=0.003, sv=555):
    N = len(hb)
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(sv)
    T = Ti
    stall = 0
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    for _ in range(max_it):
        if time.time() > dl - 1.0:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[int(rng.integers(0, len(ones)))])
        j = int(zeros[int(rng.integers(0, len(zeros)))])
        cur[i] = 0.0
        cur[j] = 1.0
        nc = _exact_c5(cur)
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cc = nc
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if nc < bc - 1e-14:
                bc = nc
                bh = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            stall += 1
    return bh, bc
