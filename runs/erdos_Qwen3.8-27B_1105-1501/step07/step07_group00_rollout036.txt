def _line_search_exact(h, d, dx, radius):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    n = len(h)
    idx = np.argsort(h)[-n // 2:]

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 10.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amplitudes = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _exact_c5(h):

def _scores(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G


def _line_search_exact(h, d, dx, radius):

def _active_lp(h, radius, n_active=15, gap=0.003):
    n = len(h)
    scores = _scores(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]

        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

        idx = np.sort(idx)
    else:
        idx = np.arange(ns)
    lags = idx - (n - 1)
    G = _grad_rows(h, lags)
    m = len(idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            d = res.x[:n].copy()
            d -= float(np.mean(d))
            return d
    except Exception:
        pass
    return None


def _softmax_direction(h, beta):
    n = len(h)
    dx = 2.0 / n
    R_raw = np.correlate(h, 1.0 - h, mode='full')
    shifted = beta * (R_raw - np.max(R_raw))
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(R_raw))
        w[int(np.argmax(R_raw))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(R_raw)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g *= dx
    g -= float(np.mean(g))
    return g


def _local_opt(h, deadline, radius_init=0.1, max_iters=3000, mode='active'):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    top_k = 60
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        if mode == 'active':
            d = _active_lp(h, radius, n_active=n_active, gap=gap)
        elif mode == 'full':
            d = _full_constraints_lp(h, radius, top_k=top_k)
        else:
            if stall % 4 == 0:
                d = _full_constraints_lp(h, radius, top_k=50 + stall // 4)
            else:
                d = _active_lp(h, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _softmax_direction(h, 25.0)
            if float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.5
                n_active = min(n_active + 3, 30)
                top_k = min(top_k + 10, 120)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search_exact(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.3)
            gap = max(gap * 0.88, 0.001)
            n_active = max(n_active - 1, 8)
            top_k = max(30, top_k - 3)
        else:
            stall += 1
            radius *= 0.7
            gap *= 1.35
            n_active = min(n_active + 3, 30)
            top_k = min(120, top_k + 10)
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.08
                gap = 0.005
                n_active = 12
                top_k = 60
    return best_h, best_c5


def _pair_sweep(h, deadline, max_rounds=200, deterministic=False):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777 + (1 if deterministic else 0))
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        ones = np.where(h > 0.35)[0]
        zeros = np.where(h < 0.65)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = max(5, min(len(ones) // 3, 25))
        if deterministic:
            step = max(1, len(ones) // 15)
            for oi in range(0, len(ones), step):
                if time.time() > deadline:
                    break
                step_z = max(1, len(zeros) // 15)
                for zi in range(0, len(zeros), step_z):
                    if time.time() > deadline:
                        break
                    d = np.zeros(n)
                    d[zeros[zi]] = 1.0
                    d[ones[oi]] = -1.0
                    alpha, c_try = _line_search_exact(h, d, dx, 0.3)
                    if c_try < cur_c5 - 1e-13:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
        else:
            for _ in range(n_tries):
                if time.time() > deadline:
                    break
                i = int(rng.integers(0, len(zeros)))
                j = int(rng.integers(0, len(ones)))
                d = np.zeros(n)
                d[zeros[i]] = 1.0
                d[ones[j]] = -1.0
                alpha, c_try = _line_search_exact(h, d, dx, 0.3)
                if c_try < cur_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
    return best_h, best_c5


def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        h_new = cur.copy()
        h_new[ones[i]] = 0.0
        h_new[zeros[j]] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin


def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 12.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amplitudes = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.25 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)


def _block_perturb(h, rng):
    n = len(h)
    h_new = h.copy()
    n_blocks = int(rng.integers(2, 5))
    for _ in range(n_blocks):
        width = int(rng.integers(5, max(6, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        flip = rng.random() < 0.5
        if flip:
            h_new[start:start + width] += 0.3
        else:
            h_new[start:start + width] -= 0.3
    return _project(h_new, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Active-set LP warmup (12%)
    d1 = t0 + budget_s * 0.12
    h1, c5_1 = _local_opt(h_init.copy(), d1, radius_init=0.1, max_iters=2000, mode='active')
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Full-constraints LP (18%) - key difference from parent
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _local_opt(_project(best_h.copy(), len(best_h) / 2.0), d2,
                          radius_init=0.08, max_iters=2000, mode='full')
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Mixed mode LP (15%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _local_opt(_project(best_h.copy(), len(best_h) / 2.0), d3,
                          radius_init=0.07, max_iters=2500, mode='mixed')
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Deterministic pair sweep (10%)
    d4 = t0 + budget_s * 0.55
    h4, c5_4 = _pair_sweep(_project(best_h.copy(), len(best_h) / 2.0), d4,
                           max_rounds=400, deterministic=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution exploration (15%)
    d5 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    tried_ns = set()
    offsets = list(range(-24, 29, 3))
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.009)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.08, max_iters=500, mode='mixed')
        if c5_r < best_c5 + 0.003 and time.time() < d5 - 3:
            sub_dl2 = min(d5, time.time() + budget_s * 0.004)
            h_r2, c5_r2 = _pair_sweep(_project(h_r.copy(), n_try / 2.0), sub_dl2,
                                       max_rounds=40, deterministic=False)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Binary optimization (13%)
    d6 = t0 + budget_s * 0.83
    n_b = len(best_h)
    h_bin = _to_binary(best_h)
    sub_dl = min(d6, time.time() + budget_s * 0.05)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.002)
    if time.time() < d6 - 3:
        sub_dl2 = min(d6, time.time() + budget_s * 0.02)
        h_bin2, c5_bin2 = _pair_sweep(h_bin, sub_dl2, max_rounds=50, deterministic=False)
        if c5_bin2 < c5_bin:
            h_bin, c5_bin = h_bin2, c5_bin2
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    if time.time() < d6 - 2:
        h_pol = _project(h_bin.copy(), n_b / 2.0)
        sub_dl3 = min(d6, time.time() + budget_s * 0.015)
        h_pol, c5_pol = _local_opt(h_pol, sub_dl3, radius_init=0.05, max_iters=400, mode='active')
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Basin hopping with diverse perturbations (8%)
    d7 = t0 + budget_s * 0.91
    scale = 0.015
    n_restarts = 0
    while time.time() < d7 and n_restarts < 120:
        n_restarts += 1
        if n_restarts % 3 == 0:
            h_pert = _block_perturb(best_h, rng)
        else:
            h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.05, max_iters=300, mode='mixed')
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.12)

    # Phase 8: Final polish (9%)
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, deadline - 2, radius_init=0.03, max_iters=1500, mode='active')
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if deadline - time.time() > 4:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_f2, c5_f2 = _pair_sweep(h_f2, deadline, max_rounds=50, deterministic=True)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
