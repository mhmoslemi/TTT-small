def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wr * u[:m]
                g[:m] -= wr * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += wr * u[gp:]
                g[gp:] -= wr * h[:m]
    return g

def _smooth_max_weights(raw, beta):
    mx = float(np.max(raw))

    vp = np.exp(np.clip(beta * shifted, -120, 0))
    s = float(np.sum(vp))
    if s < 1e-30:
        return np.full(len(raw), 1.0 / len(raw))
    return vp / s

def _line_search(h, d, raw, dx, radius, n_top=20):

    pos = d > 1e-12

    a_hi = max(a_hi, 1e-12)

    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:

    npairs = min(n_top, 14)
    for i in range(npairs):

            B = T[ri] - T[rj]

def _make_directions(h, g, n, rng, it_count):

    d0 = -g.copy()
    d0 -= np.mean(d0)
    nm = np.linalg.norm(d0)
    if nm > 1e-12:
        directions.append(d0 / nm)
    for bl in [3, 5, 7, 11]:
        if bl >= n:
            continue
        nb = n // bl
        if nb < 2:
            continue
        blk = np.zeros(n)
        for b in range(nb):
            sl = slice(b * bl, min((b + 1) * bl, n))
            blk[sl] = np.mean(g[sl])
        db = -blk.copy()
        db -= np.mean(db)
        nm = np.linalgnorm(db) if hasattr(np, 'linalgorm') else np.linalg.norm(db)
        if nm > 1e-12:
            directions.append(db / nm)
    idx_pos = np.argsort(g)[-4:]
    idx_neg = np.argsort(g)[:4]
    for ip in idx_pos:
        for ino in idx_neg:
            if ip == ino:
                continue
            dp = np.zeros(n)
            dp[ip] = -1.0
            dp[ino] = 1.0
            dp -= np.mean(dp)
            nm = np.linalg.norm(dp)
            if nm > 1e-12:
                directions.append(dp / nm)
    for _ in range(2):
        fr = rng.uniform(1.0, 7.0)
        ph = rng.uniform(0, 2 * np.pi)
        dr = np.sin(2.0 * np.pi * fr * np.arange(n) / n + ph)
        dr -= np.mean(dr)
        nm = np.linalg.norm(dr)
        if nm > 1e-12:

    if it_count % 7 == 0:
        dr = rng.normal(0, 1, n)
        dr -= np.mean(dr)
        nm = np.linalg.norm(dr)
        if nm > 1e-12:
            directions.append(dr / nm)

def _resample(h, n_new):
