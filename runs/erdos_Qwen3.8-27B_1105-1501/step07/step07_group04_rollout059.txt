import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0 - 1e-14) if res > 0 else (x > 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _build_grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        u = -k
        g[u:] += (1.0 - h[:n - u])
        g[:n - u] -= h[u:]
    return g * dx


def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    C0 = np.correlate(h, 1.0 - h, mode='full')
    L1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q2 = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(C0) * dx)
    grid_t = np.linspace(0.0, t_hi, 40)
    grid_f = np.array([float(np.max(C0 + grid_t[i] * L1 - grid_t[i] ** 2 * Q2) * dx)
                       for i in range(40)])
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    invphi = 0.6180339887498949
    if hi_b - lo_b < 1e-12:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi_b - invphi * (hi_b - lo_b)
        c2 = lo_b + invphi * (hi_b - lo_b)
        f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2) * dx)
        f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2) * dx)
        for _ in range(50):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - invphi * (hi_b - lo_b)
                f1 = float(np.max(C0 + c1 * L1 - c1 * c1 * Q2) * dx)
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + invphi * (hi_b - lo_b)
                f2 = float(np.max(C0 + c2 * L1 - c2 * c2 * Q2) * dx)
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(C0 + opt_t * L1 - opt_t * opt_t * Q2) * dx)
    cand = [(0.0, float(np.max(C0) * dx)),
            (opt_t, opt_val),
            (t_hi, float(np.max(C0 + t_hi * L1 - t_hi * t_hi * Q2) * dx))]
    best = min(cand, key=lambda x: x[1])
    return best[0], best[1]


def _lp_solve(h, active, radius):
    n = len(h)
    m = len(active)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, r in enumerate(active):
        k = int(r - (n - 1))
        A_ub[i, :n] = _build_grad_row(h, k)
        A_ub[i, n] = -1.0
    corr_scaled = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    for i, r in enumerate(active):
        b_ub[i] = 0.0
    max_c = float(np.max(corr_scaled))
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i]) + 1e-12
        hi = min(radius, 1.0 - h[i]) - 1e-12
        if lo < -1e-12:
            lo = -1e-12
        if hi > 1e-12:
            hi = 1e-12
        bounds.append((lo, hi))
    bounds.append((max_c - 0.05, max_c + 1.0))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10, 'presolve': True})
    except Exception:
        return None, None
    if res is None or res.status != 0:
        return None, None
    d = res.x[:n].copy()
    t_opt = float(res.x[n])
    if np.max(np.abs(d)) < 1e-12:
        return None, None
    return d, t_opt


def _subgradient_dir(h, active, corr_scaled):
    n = len(h)
    max_val = float(np.max(corr_scaled))
    tol = max(1e-10, 0.002 * max_val)
    active_idx = np.where(corr_scaled >= max_val - tol)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(corr_scaled))])
    g = np.zeros(n)
    for r in active_idx:
        k = int(r - (n - 1))
        g += _build_grad_row(h, k)
    g /= len(active_idx)
    d = -g
    d -= np.mean(d)
    return d


def _lp_optimize(h, deadline, radius_init=0.10, max_iter=5000):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    active_size = 36
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        corr_scaled = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_val = float(np.max(corr_scaled))
        thr = max_val - max(0.003 * max_val, 0.002)
        active = np.where(corr_scaled >= thr)[0]
        if len(active) > active_size:
            order = np.argsort(-corr_scaled[active])
            active = active[order[:active_size]]
        d, t_opt = _lp_solve(h, active, radius)
        if d is None:
            d = _subgradient_dir(h, active, corr_scaled)
        d = d - np.mean(d)
        norm_d = float(np.max(np.abs(d)))
        if norm_d < 1e-12:
            stall += 1
            radius *= 0.7
            active_size = min(active_size + 8, 100)
            if radius < 1e-5:
                break
            continue
        d = d / norm_d * min(1.0, radius / norm_d)
        alpha, new_c5 = _line_search(h, d, radius)
        if alpha > 1e-10 and new_c5 < cur - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            cur_new = _exact_c5(h_new)
            if cur_new < cur - 1e-14:
                h = h_new
                cur = cur_new
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.25)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0:
                active_size = min(active_size + 8, 120)
            if radius < 1e-5:
                break
    return best_h, best_c5


def _binary_sa(h_bin, deadline, max_iters=200000, temp_init=0.004, seed=42):

        new_c5, _ = _exact_c5_and_corr(cur)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.3, 1.0, size=5)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.2 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)
