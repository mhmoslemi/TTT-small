        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(35):
            if hi_r - lo_r < 1e-13:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _weighted_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    if top_k is not None and top_k < len(S):
        threshold = np.sort(sc)[-top_k][0]
        mask = sc >= threshold
        sc2 = sc.copy()
        sc2[~mask] = -np.inf
        v = np.clip(beta * (sc2 - mx), -80.0, 0.0)
    else:
        v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-18:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return g

def _single_lag_grad(h, lag_idx):
    n = len(h)
    l = lag_idx - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:l + m])
            g[l:] -= h[:m]
    else:
        s = -l
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] -= h[s:s + m]
    g -= np.mean(g)
    return g

def _gen_dirs(h, rng, nd=6):
    n = len(h)
    dirs = []
    g1 = _weighted_grad(h, 30.0)
    nrm = np.linalg.norm(g1)
    if nrm > 1e-10:
        dirs.append(g1 / nrm)
    g2 = _weighted_grad(h, 80.0)
    nrm = np.linalg.norm(g2)
    if nrm > 1e-10:
        dirs.append(g2 / nrm)
    g3 = _weighted_grad(h, 20.0, top_k=6)
    nrm = np.linalg.norm(g3)
    if nrm > 1e-10:
        dirs.append(g3 / nrm)
    S = np.correlate(h, 1.0 - h, mode='full')
    top_lag = int(np.argmax(S))
    gs = _single_lag_grad(h, top_lag)
    nrm = np.linalg.norm(gs)
    if nrm > 1e-10:
        dirs.append(gs / nrm)
    g4 = _weighted_grad(h, 20.0, top_k=4)
    nrm = np.linalg.norm(g4)
    if nrm > 1e-10 and len(dirs) < nd:
        dirs.append(g4 / nrm)
    if len(dirs) < nd:
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

def _local_opt(h, deadline, r0=0.18, stall_lim=60, nd=6, momentum=0.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(7777)
    rad = r0
    stall = 0
    mom_vec = np.zeros(n)
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        dirs = _gen_dirs(h, rng, nd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            d = d / nrm
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            if momentum > 0.0:
                mom_vec = momentum * mom_vec + (1.0 - momentum) * best_a * best_d
                h = h + mom_vec
            else:
                h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.40)
        else:
            stall += 1
            rad *= 0.72
            if rad < 0.006:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
                    mom_vec = np.zeros(n)
                elif stall % 40 == 0:
                    rad = 0.06
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _block_opt(h, deadline, max_it=20000, bsize=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(5555)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n - bsize + 1))
        j = int(rng.integers(0, n - bsize + 1))
        if abs(i - j) < bsize:
            stall += 1
            continue
        d = np.zeros(n)
        d[i:i+bsize] = 1.0
        d[j:j+bsize] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _bin_sa(hb, deadline, mx=20000, Ti=0.002):
    n = len(hb)
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.998
        else:
            stall += 1
    return bh, bc

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _smooth_anneal(h, deadline, steps=30):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _exact_c5(h)
    cur = h.copy()
    sigma = 0.3
    rng = np.random.default_rng(1234)
    x = np.arange(n, dtype=float)
    for s in range(steps):
        if time.time() > deadline:
            break
        cur = _project(cur + sigma * (0.5 * rng.normal(0, 1, n) + 0.3 * np.sin(2*np.pi*rng.uniform(1,5)*x/n)), target)
        sd = min(deadline, time.time() + 8.0)
        cur, cc = _local_opt(cur, sd, r0=0.08, stall_lim=25, nd=4)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = cur.copy()
            sigma *= 0.8
        else:
            sigma *= 0.9
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4.0
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Quick local opt (12%)
    d1 = t0 + budget_s * 0.12
    h1, c1 = _local_opt(h_init, d1, r0=0.16, stall_lim=45, nd=6)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair + block opt (13%)
    d2 = t0 + budget_s * 0.25
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, min(d2, time.time() + budget_s * 0.07), max_it=30000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    if time.time() < d2 - 2:
        h2b = _project(best_h.copy(), len(best_h) / 2.0)
        h2b, c2b = _block_opt(h2b, d2, max_it=12000, bsize=3)
        if c2b < best_c:
            best_c = c2b
            best_h = h2b.copy()

    # Phase 3: Binary SA + refine (15%)
    d3 = t0 + budget_s * 0.40
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, min(d3, time.time() + budget_s * 0.08), mx=15000, Ti=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3 - 3:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.06)
        hp, cp = _local_opt(hp, sd, r0=0.10, stall_lim=40, nd=5)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-resolution (18%)
    d4 = t0 + budget_s * 0.58
    nc = len(best_h)
    offsets = [-6, 6, -10, 10, -14, 14, -4, 4, -18, 18, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 280 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.016)
        hr, cr = _local_opt(hr, sd, r0=0.14, stall_lim=35, nd=5)
        if cr < best_c + 0.003 and time.time() < d4 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.006)
            hb, cb2 = _bin_sa(hb, sd2, mx=5000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.001 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.008)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=25, nd=4)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Intensive local opt with momentum (17%)
    d5 = t0 + budget_s * 0.75
    nc = len(best_h)
    for r_init in [0.10, 0.05, 0.025]:
        if time.time() > d5 - 4:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.05)
        h5, c5 = _local_opt(h5, sd, r0=r_init, stall_lim=55, nd=6)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d5 - 5:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.025)
            h5, c5 = _pair_opt(h5, sd, max_it=12000)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()

    # Phase 6: Basin hopping (15%)
    d6 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 14.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.3) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.4 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.005)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=20, nd=4)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.09, scale * 1.15)

    # Final polish
    if deadline - time.time() > 4:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.015, stall_lim=40, nd=5)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=8000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
