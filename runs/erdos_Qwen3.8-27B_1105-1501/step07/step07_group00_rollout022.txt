```python
def _weighted_grad(h, scores, weights):
    """Compute sum_ell weights[ell] * grad(scores[ell])"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for ell in range(2*n-1):
        w = weights[ell]
        if w < 1e-15:
            continue
        d = ell - (n-1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            grad[d:] += w * dx * (1 - h[:m])
            grad[:m] += -w * dx * h[d:d+m]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            grad[:m] += w * dx * (1 - h[g:])
            grad[g:] += -w * dx * h[:m]
    return grad
```

```python
    n = len(v)
    v = np.clip(v, -10, 10)  # handle infinities
    lo = float(v.min()) - 5.0
    hi = float(v.max()) + 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
```

```python
    amps = rng.uniform(0.5, 1.0, size=nf)
    perturb = np.zeros(n)
    for i in range(nf):
        perturb += amps[i] * np.sin(2.0 * np.pi * freqs[i] * x / n + phases[i])
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.standard_normal(n)
    return _project(h + perturb, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_best = _project(h0.copy(), n0 / 2.0)
    c5_best = _exact_c5(h_best)

    # Phase 1: Deep smooth optimization (30% time)
    d1 = t0 + budget_s * 0.30
    betas_deep = [0.20, 0.10, 0.05, 0.02, 0.008, 0.003, 0.001]
    h_t, c5_t = _smooth_opt(h_best, d1, betas_deep, maxiter=200)
    if c5_t < c5_best:
        c5_best = c5_t
        h_best = h_t.copy()

    # Phase 2: Gradient line search refinement (10%)
    d2 = t0 + budget_s * 0.40
    h_t, c5_t = _grad_line_search(h_best, 0.01, d2, max_steps=300)
    if c5_t < c5_best:
        c5_best = c5_t
        h_best = h_t.copy()

    # Phase 3: LP polish (15%)
    d3 = t0 + budget_s * 0.55
    h_t, c5_t = _lp_polish(h_best, d3, radius_init=0.08, max_iters=1000, gap_init=0.008)
    if c5_t < c5_best:
        c5_best = c5_t
        h_best = h_t.copy()

    # Phase 4: Diverse restarts with smooth optimization (20%)
    d4 = t0 + budget_s * 0.75
    n_restarts = 0
    scale = 0.025
    while time.time() < d4 and n_restarts < 30:
        n_restarts += 1
        h_pert = _smooth_perturb(h_best, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.010)
        betas_shallow = [0.05, 0.02, 0.006]
        h_t, c5_t = _smooth_opt(h_pert, sub_dl, betas_shallow, maxiter=120)
        if c5_t < c5_best:
            c5_best = c5_t
            h_best = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 5: Resampling to different n (10%)
    d5 = t0 + budget_s * 0.85
    n_curr = len(h_best)
    for dn in [-10, -5, -2, 2, 5, 10]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280:
            continue
        h_r = _resample(h_best, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.010)
        h_r, c5_r = _smooth_opt(h_r, sub_dl, [0.04, 0.015, 0.005], maxiter=100)
        if c5_r < c5_best:
            sub_dl2 = min(d5, time.time() + budget_s * 0.005)
            h_r, c5_r = _lp_polish(h_r, sub_dl2, radius_init=0.05, max_iters=300)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r.copy()
                n_curr = n_try

    # Phase 6: Binary optimization attempt (5%)
    d6 = t0 + budget_s * 0.90
    n_b = len(h_best)
    h_bin = _to_binary(h_best)
    c5_bin = _exact_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    else:
        # Interpolate between binary and continuous
        for eps in [0.05, 0.1, 0.2]:
            h_mix = _project((1 - eps) * h_bin + eps * h_best, n_b / 2.0)
            c5_mix = _exact_c5(h_mix)
            if c5_mix < c5_best:
                c5_best = c5_mix
                h_best = h_mix.copy()

    # Phase 7: Final LP polish (5%)
    h_final, c5_final = _lp_polish(h_best, deadline, radius_init=0.04, max_iters=500, gap_init=0.003)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h_final.copy()

    # Ensure final feasibility
    h_final = _project(h_best, len(h_best) / 2.0)
    c5_check = _exact_c5(h_final)
    if c5_check < c5_best:
        c5_best = c5_check
    h_best = np.clip(h_best, 0.0, 1.0)
    return h_best, c5_best, len(h_best)
```
