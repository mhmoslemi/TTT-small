import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b = t_lo
    hi_b = t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gradient_direction(h, scores, beta=25.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -50.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    G = _grad_rows(h, lags)
    g = G.T @ w
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.12, max_iters=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 15
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        d = _active_lp(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_direction(h, scores, beta=25.0, radius=radius)
            if d is None or float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.5
                n_active = min(n_active + 3, 35)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
            gap = max(gap * 0.88, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.4
            n_active = min(n_active + 2, 30)
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _binary_local_search(h, deadline, max_iters=50000):
    """Local search on binary sequences using pair swaps."""
    n = len(h)
    target = n // 2
    h = h.copy()
    cur = h.astype(float)
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    rng = np.random.default_rng(123)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Pick a random 1 and a random 0
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # Swap: set i to 0, j to 1
        cur[i] = 0
        cur[j] = 1
        new_c5 = _exact_c5(cur)
        if new_c5 <= cur_c5 + 1e-15:
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_c5

def _scores(h):
    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):

            m = n - d
            if m > 0:

                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    return G

    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_b, hi_b = t_lo, t_hi
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(50):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    return opt_t, float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    G = _grad_rows(h, lags)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
