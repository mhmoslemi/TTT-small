def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, r, n):
    """Compute gradient of raw[r] w.r.t. h at point h."""
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:

def _lp_direction(h, raw, n, k_lags, radius):
    """Solve LP to find direction from h."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Select top k_lags
    top_idx = np.argsort(-raw)[:k_lags]
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h - radius,
        x <= h + radius,
    ]
    
    for r in top_idx:
        d = r - (n - 1)
        g = _grad_for_lag(h, d, n)
        # Linearized: raw[r] + g · (x - h) <= t
        constraints.append(raw[r] + g @ (x - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Try solvers in order of preference
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if x.value is not None:
                    x_val = np.array(x.value).flatten()
                    x_val = np.clip(x_val, 0.0, 1.0)
                    x_val = _project(x_val, target)
                    return x_val
            except:
                continue
    except:
        pass
    return None

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag_vec(h, d, n):
    """Gradient of raw[n-1+|d|... actually raw at shift d. Vectorized."""
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
    else:
        dp = -d
        g[:n-dp] += (1.0 - h[dp:])
        g[dp:] -= h[:n-dp]
    return g

def _line_search_quad(h, d, raw, dx, t_max=1.0, n_top=20):
    """Line search: minimize max(raw + t*T + t^2*Q)*dx for t in [0, t_max]."""
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Find t_max such that h + t*d stays in [0,1]
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    t_max = max(t_max, 0.0)
    
    if t_max < 1e-14:
        return 0.0, float(np.max(raw)) * dx
    
    best_t = 0.0
    best_val = float(np.max(raw)) * dx
    
    # Grid search
    n_grid = 40
    grid = np.linspace(0, t_max, n_grid)
    for t in grid:
        val = float(np.max(raw + t * T + t * t * Q)) * dx
        if val < best_val:
            best_val = val
            best_t = t
    
    # Golden section on [best_t - step, best_t + step]
    step = t_max / n_grid
    lo = max(0.0, best_t - step)
    hi = min(t_max, best_t + step)
    
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T + c1 * c1 * Q)) * dx

        for _ in range(60):

                c1 = hi - gr * (hi - lo)

            else:

        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T + opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_t = opt
    
    return best_t, best_val

def _lp_direction(h, raw, n, top_k, radius):
    """Solve trust-region LP to get a direction from h."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Select top lags
    top_idx = np.argsort(-raw)[:top_k]
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
        x >= h - radius,
        x <= h + radius,
    ]
    
    for r in top_idx:
        d = int(r) - (n - 1)
        g = _grad_lag_vec(h, d, n)
        constraints.append(cp.multiply(g, x).sum() <= t - (raw[r] - g @ h))
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = [cp.ECOS, cp.SCS]
    if hasattr(cp, 'CLARABEL'):
        solvers.insert(0, cp.CLARABEL)
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x.value is not None:
                x_val = np.array(x.value).flatten().copy()
                x_val = np.clip(x_val, 0.0, 1.0)
                x_val = _project(x_val, target)
                return x_val
        except Exception:
            continue
    return None
