                cands.append(av)
    for r in range(n_corr):
        if Q[r] > 1e-18 and r not in top_k:
            av = T[r] / (2.0 * Q[r])
            if a_lo + 1e-14 <= av <= a_hi - 1e-14:
                cands.append(av)
    top20 = np.argsort(S0)[-20:]
    for i in range(len(top20)):
        for j in range(i + 1, len(top20)):
            r1, r2 = top20[i], top20[j]
            A = Q[r1] - Q[r2]
            B = T[r2] - T[r1]
            C = S0[r2] - S0[r1]
            if abs(A) < 1e-20:
                if abs(B) > 1e-20:
                    av = -C / B
                    if a_lo + 1e-14 <= av <= a_hi - 1e-14:
                        cands.append(av)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for av in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo + 1e-14 <= av <= a_hi - 1e-14:
                            cands.append(av)
    best_val = float('inf')
    best_a = 0.0
    for av in cands:
        fv = f_a(av)
        if fv < best_val:
            best_val = fv
            best_a = av
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    step_w = (a_hi - a_lo) / 80.0
    lo = max(a_lo, best_a - 6 * step_w)
    hi = min(a_hi, best_a + 6 * step_w)
    if hi - lo < 1e-14:
        lo, hi = a_lo, a_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f_a(c1), f_a(c2)
    for _ in range(55):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f_a(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f_a(c2)
    opt_a = (lo + hi) / 2.0
    opt_v = f_a(opt_a)
    if opt_v < best_val:
        best_val = opt_v
        best_a = opt_a
    return best_a, best_val

def _make_dirs(h, g, raw, n, rng, n_dir=6):
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    top3 = np.argsort(raw)[-3:]
    for r in top3:
        gs = _lag_grad(h, r - (n - 1), n)
        gsn = np.linalg.norm(gs)
        if gsn > 1e-12:
            d = gs / gsn
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                dirs.append(d / dn)
    x = np.arange(n, dtype=float)
    for freq in [1, 2, 3]:
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    idx_s = np.argsort(g)
    for s in idx_s[:2]:
        for k in idx_s[-2:]:
            if s != k:
                d = np.zeros(n)
                d[s] = 1.0
                d[k] = -1.0
                d -= np.mean(d)
                dirs.append(d)
                break
    return dirs[:n_dir]

def _optimize_smooth(h, deadline, beta_lo=5.0, beta_hi=80.0, radius_init=0.15, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(777)
    vel = np.zeros(n)
    momentum = 0.3
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = min(1.0, float(total_iters) / 300.0)
        beta = beta_lo + (beta_hi - beta_lo) * frac
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        g = _smooth_grad(h, raw, beta)
        vel = momentum * vel + (1.0 - momentum) * g
        dirs = _make_dirs(h, vel, raw, n, rng, 6)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.40)
            else:
                stall += 1
                radius = max(0.03, radius * 0.72)
                momentum = max(0.1, momentum * 0.95)
        else:
            stall += 1
            radius = max(0.03, radius * 0.78)
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_rounds=200):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    improved = True
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    stall = 0
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
                    stall += 1
        if not improved:
            stall += 50
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.005):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.999
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        freq = rng.uniform(0.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 4.0
    pert += amp * 0.2 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_cont(h, deadline, radius=0.20, n_iter=5000, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth gradient descent with Nesterov (28%)
    d1 = t0 + budget_s * 0.28
    h1, c5_1 = _optimize_smooth(h_init, d1, beta_lo=4.0, beta_hi=60.0, radius_init=0.18, max_stall=70)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (20%)
    d2 = t0 + budget_s * 0.48
    n_curr = len(best_h)
    n_options = []
    for dn in range(-20, 21, 4):
        n_try = n_curr + dn
        if 80 <= n_try <= 260 and n_try % 2 == 0 and n_try != n_curr:
            n_options.append(n_try)
    rng.shuffle(n_options)
    for n_try in n_options:
        if time.time() > d2:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _optimize_smooth(h_r, sub_dl, beta_lo=6.0, beta_hi=50.0, radius_init=0.12, max_stall=35)
        if time.time() < d2 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d2, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=8000, T_init=0.003)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Binary optimization (20%)
    d3 = t0 + budget_s * 0.68
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.10)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=50000, T_init=0.005)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_local(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d3:
        h_pol = _project(h_b.copy(), n_b / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _optimize_smooth(h_pol, sub_dl3, beta_lo=10.0, beta_hi=60.0, radius_init=0.10, max_stall=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.88
    scale = 0.025
    n_restarts = 0
    while time.time() < d4 - 10 and n_restarts < 60:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.009)
        h_t, c5_t = _optimize_smooth(h_p, sub_dl, beta_lo=8.0, beta_hi=60.0, radius_init=0.08, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 5: Intensive continuous refinement (12%)
    d5 = deadline
    n_c = len(best_h)
    h4 = _project(best_h.copy(), n_c / 2.0)
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h4, c5_4 = _optimize_smooth(h4, sub_dl, beta_lo=20.0, beta_hi=120.0, radius_init=0.08, max_stall=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d5:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d5, time.time() + budget_s * 0.04)
        h4, c5_4 = _pair_cont(h4, sub_dl2, radius=0.15, n_iter=4000, seed=555)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d5:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d5, time.time() + budget_s * 0.03)
        h4, c5_4 = _optimize_smooth(h4, sub_dl3, beta_lo=40.0, beta_hi=150.0, radius_init=0.05, max_stall=40)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Final: try nearby n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5 - 15:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.012)
        h_r, c5_r = _optimize_smooth(h_r, sub_dl, beta_lo=15.0, beta_hi=80.0, radius_init=0.08, max_stall=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _optimize_smooth(h_f, deadline, beta_lo=30.0, beta_hi=150.0, radius_init=0.05, max_stall=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
