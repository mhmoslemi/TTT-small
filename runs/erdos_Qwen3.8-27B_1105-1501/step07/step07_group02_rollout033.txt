import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    s = float(np.sum(v))
    if abs(s - target) < 1e-13:
        return v.copy()
    if s > target:
        lo, hi = 0.0, 2.0
    else:
        lo, hi = -2.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(x)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        ell = k - (n - 1)
        if ell >= 0:
            m = n - ell
            if m > 0:
                G[k, ell:] += (1.0 - h[:m])
                G[k, :m] -= h[ell:]
        else:
            p2 = -ell
            m = n - p2
            if m > 0:
                G[k, :m] += (1.0 - h[p2:])
                G[k, p2:] -= h[:m]
    return G

def _feas_t_range(h, d, cap=1.0):
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = -cap
    t_hi = cap
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.max(h[neg] / (-d[neg]))))
    return t_lo, t_hi

def _line_search(h, S, d, t_lo, t_hi):

    omh = 1.0 - h
    U = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
    B = np.correlate(d, d, mode='full')

    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([np.max(S + t * U - t * t * B) for t in ts])
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_grid - 1, best_idx + 1)]
    if hi - lo < 1e-15:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S + c1 * U - c1 * c1 * B))
    f2 = float(np.max(S + c2 * U - c2 * c2 * B))
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S + c1 * U - c1 * c1 * B))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)

    t_opt = (lo + hi) / 2.0
    v_opt = float(np.max(S + t_opt * U - t_opt * t_opt * B))
    best_t, best_v = float(ts[best_idx]), float(vals[best_idx])

    vl = float(np.max(S + t_lo * U - t_lo * t_lo * B))
    vh = float(np.max(S + t_hi * U - t_hi * t_hi * B))
    if vl < best_v:

    if vh < best_v:

                h_try = _project(h_try, target)
            cv = _exact_c5(h_try)
            if cv < best_cv - 1e-14:
                best_cv = cv
                best_trial = h_try.copy()
        if best_trial is not None:
            h = best_trial
            cur_c5 = best_cv
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            S = np.correlate(h, 1.0 - h, mode='full')
            beta_cur = min(beta_cur * 1.01, beta_end)
        else:
            if it % 20 == 0:
                beta_cur = max(beta_cur * 0.9, beta_start * 0.5)
            if stall > 30:
                d_rand = rng.standard_normal(n)
                d_rand -= d_rand.mean()
                dm = float(np.max(np.abs(d_rand)))
                if dm > 1e-14:
                    d_rand /= dm
                    t_lo2, t_hi2 = _feas_t_range(h, d_rand, cap=0.3)
                    if t_hi2 - t_lo2 > 1e-12:
                        t_r = (t_lo2 + t_hi2) / 2.0
                        h_try = np.clip(h + t_r * d_rand, 0.0, 1.0)
                        h_try = _project(h_try, target)
                        cv = _exact_c5(h_try)
                        if cv < cur_c5 - 1e-12:
                            h = h_try
                            cur_c5 = cv
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            S = np.correlate(h, 1.0 - h, mode='full')
                            stall = 0
    return best_h, best_c5

def _pair_optimize(h, deadline, seed=42, radius=0.15, n_pairs=40, beta=20.0, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    stall = 0
    rad = radius
    while time.time() < deadline:
        if stall > stall_limit or rad < 1e-7:
            break
        Smax = float(np.max(S))
        w = np.exp(beta * (S - Smax))
        wsum = float(np.sum(w))
        if wsum < 1e-30:
            topk = max(3, n // 5)
            idx = np.argsort(S)[-topk:]
            w = np.zeros(len(S))
            w[idx] = 1.0 / topk
        else:
            w /= wsum
        G = _grad_matrix(h)
        cg = G.T @ w
        sort_idx = np.argsort(cg)
        nsink = min(n_pairs, n // 3)
        nsour = min(n_pairs, n // 3)
        sinks = sort_idx[:nsink]
        sources = sort_idx[-nsour:]
        pairs = []
        for ii in range(min(nsink, nsour)):
            pairs.append((int(sinks[ii]), int(sources[ii])))
        for _ in range(max(5, n_pairs // 5)):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))
        best_c5_try = cur_c5
        best_step = None
        omh = 1.0 - h
        for (sink, source) in pairs:
            if sink == source:
                continue
            d = np.zeros(n)
            d[sink] = 1.0
            d[source] = -1.0
            U = np.correlate(d, omh, mode='null') - np.correlate(h, d, mode='null')
            # Use full correlate for correctness
            U = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
            B = np.correlate(d, d, mode='full')
            t_lo = max(-h[sink], h[source] - 1.0, -rad)
            t_hi = min(1.0 - h[sink], h[source], rad)
            if t_hi - t_lo < 1e-13:
                continue
            t_opt, _ = _line_search(h, S, d, t_lo, t_hi)
            h_try = h.copy()
            h_try[sink] += t_opt
            h_try[source] -= t_opt
            cv = _exact_c5(h_try)
            if cv < best_c5_try - 1e-14:
                best_c5_try = cv
                best_step = h_try
        if best_step is not None:
            h = _project(best_step, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
            S = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            rad = min(rad * 1.015, 0.25)
        else:
            stall += 1
            rad *= 0.8
            if stall % 30 == 0 and rad < 0.02:
                rad = 0.1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, seed=999, T_init=0.004, max_iter=50000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            stall += 1
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[i], cur[j] = cur[j], cur[i]
            stall += 1
    return best_h, best_c5

def _binary_local(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = set(np.where(cur > 0.5)[0].tolist())
    zeros = set(np.where(cur <= 0.5)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in list(ones):
            if time.time() > deadline:
                break
            for oz in list(zeros):
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    ones.discard(oi)
                    zeros.discard(oz)
                    ones.add(oz)
                    zeros.add(oi)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_best = _project(h0.copy(), n0 / 2.0)
    c5_best = _exact_c5(h_best)

    # Phase 1: Dense gradient optimization at n0 (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _dense_optimize(h_best, d1, seed=1, beta_start=15.0, beta_end=40.0)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1.copy()

    # Phase 1b: Pair optimize to polish (10% time)
    d1b = t0 + budget_s * 0.40
    h1b, c5_1b = _pair_optimize(h_best, d1b, seed=5, radius=0.12, n_pairs=50, beta=25.0, stall_limit=60)
    if c5_1b < c5_best:
        c5_best = c5_1b
        h_best = h1b.copy()

    # Phase 2: Multi-n exploration (25% time)
    d2 = t0 + budget_s * 0.65
    n_base = len(h_best)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32]
    tried = set()
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = n_base + dn
        if n_try < 80 or n_try > 260 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(h_best, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_r, c5_r = _dense_optimize(h_r, sub_dl, seed=200 + dn, beta_start=12.0, beta_end=35.0)
        sub_dl2 = min(d2, time.time() + budget_s * 0.008)
        h_r, c5_r2 = _pair_optimize(h_r, sub_dl2, seed=300 + dn, radius=0.10, n_pairs=35, beta=25.0, stall_limit=40)
        if c5_r2 < c5_r:
            c5_r = c5_r2
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()
            n_base = n_try

    # Phase 3: Binary diversification (15% time)
    d3 = t0 + budget_s * 0.80
    n_curr = len(h_best)
    if time.time() < d3:
        h_bin = _to_binary(h_best, n_curr)
        sub_dl = min(d3, time.time() + budget_s * 0.07)
        h_bin, c5_bin = _binary_sa(h_bin, sub_dl, seed=777, T_init=0.004, max_iter=40000)
        sub_dl2 = min(d3, time.time() + budget_s * 0.04)
        h_bin, c5_bin2 = _binary_local(h_bin, sub_dl2)
        if c5_bin2 < c5_bin:
            c5_bin = c5_bin2
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()
        # LP polish on binary
        if time.time() < d3:
            h_pol = _project(h_bin.copy(), n_curr / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _dense_optimize(h_pol, sub_dl3, seed=555, beta_start=20.0, beta_end=50.0)
            if c5_pol < c5_best:
                c5_best = c5_pol
                h_best = h_pol.copy()
        # Try binary at a different n
        n_alt = n_curr + (4 if n_curr < 200 else -4)
        if n_alt >= 80 and n_alt <= 260 and n_alt % 2 == 0 and time.time() < d3:
            h_r = _resample(h_best, n_alt)
            h_bin2 = _to_binary(h_r, n_alt)
            sub_dl4 = min(d3, time.time() + budget_s * 0.03)
            h_bin2, c5_b2 = _binary_sa(h_bin2, sub_dl4, seed=888, T_init=0.003, max_iter=20000)
            if c5_b2 < c5_best:
                c5_best = c5_b2
                h_best = h_bin2.copy()

    # Phase 4: Final intensive refinement (10% time)
    d4 = safe_deadline
    n_curr = len(h_best)
    h4 = _project(h_best.copy(), n_curr / 2.0)
    h4, c5_4 = _dense_optimize(h4, d4, seed=1000, beta_start=25.0, beta_end=60.0)
    if c5_4 < c5_best:
        c5_best = c5_4
        h_best = h4.copy()

    # Final: try one more pair polish
    if time.time() < safe_deadline - 2:
        sub_dl = min(safe_deadline, time.time() + budget_s * 0.02)
        h5 = _project(h_best.copy(), len(h_best) / 2.0)
        h5, c5_5 = _pair_optimize(h5, sub_dl, seed=2000, radius=0.06, n_pairs=30, beta=35.0, stall_limit=40)
        if c5_5 < c5_best:
            c5_best = c5_5
            h_best = h5.copy()

    # Final projection and verification
    n_final = len(h_best)
    h_best = _project(h_best, n_final / 2.0)
    final_c5 = _exact_c5(h_best)
    if final_c5 < c5_best:
        c5_best = final_c5
    return h_best, c5_best, n_final
