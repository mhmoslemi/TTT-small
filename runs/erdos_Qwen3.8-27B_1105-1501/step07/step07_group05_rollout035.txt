def _lp_direction(h, k_top=10, radius=0.1):
    """
    Find direction s via LP that maximally reduces top-k active outputs.
    Returns s (a feasible edge step) or None if LP fails.
    """
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Select top-k outputs
    top_k = min(k_top, len(S0))
    top_idx = np.argsort(-S0)[:top_k]
    
    # Compute gradients for each top output
    grads = []
    norms = []
    for t in top_idx:
        xi = t - (n - 1)
        G = np.zeros(n)
        if xi >= 0:
            L = n - xi
            if L > 0:
                G[:L] += (1.0 - h[xi:xi+L])
                G[xi:xi+L] -= h[:L]
        else:
            s = -xi
            L = n - s
            if L > 0:
                G[s:s+L] += (1.0 - h[:L])
                G[:L] -= h[s:s+L]
        nm = np.linalg.norm(G)
        if nm > 1e-10:
            grads.append(G / nm)
            norms.append(nm)
    
    if len(grads) == 0:
        return None
    
    A_mat = np.array(grads)  # (m, n)
    
    # LP: maximize tau
    # s.t. A_mat @ s <= -tau  (for each active row)
    #      sum(s) = 0
    #      -min(r, h_i) <= s_i <= min(r, 1-h_i)
    #      tau >= 0
    
    try:
        s_var = cp.Variable(n)
        tau_var = cp.Variable()
        
        lo_bound = -np.minimum(radius, h)
        hi_bound = np.minimum(radius, 1.0 - h)
        
        constraints = [
            cp.sum(s_var) == 0,
            s_var >= lo_bound,
            s_var <= hi_bound,
            tau_var >= 0
        ]
        for i in range(len(grads)):
            constraints.append(A_mat[i] @ s_var <= -tau_var)
        
        prob = cp.Problem(cp.Maximize(tau_var), constraints)
        
        # Try solvers in order of preference
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                    if s_var.value is not None and tau_var.value > 1e-6:
                        s_val = np.array(s_var.value).flatten()
                        s_val -= np.mean(s_val)  # ensure exact zero mean
                        return s_val
                break
            except Exception:
                continue
    except Exception:
        pass
    
    return None

def _aggregate_gradient(h, k_top=10):
    """Aggregate gradient from top-k outputs (fallback if LP fails)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    top_k = min(k_top, len(S0))
    top_idx = np.argsort(-S0)[:top_k]
    
    G = np.zeros(n)
    for t in top_idx:
        xi = t - (n - 1)
        if xi >= 0:
            L = n - xi
            if L > 0:
                G[:L] += (1.0 - h[xi:xi+L])
                G[xi:xi+L] -= h[:L]
        else:
            s = -xi
            L = n - s
            if L > 0:
                G[s:s+L] += (1.0 - h[:L])
                G[:L] -= h[s:s+L]
    
    nm = np.linalg.norm(G)
    if nm < 1e-10:
        return np.zeros(n)
    return -G / nm  # negative gradient (descent direction)

def _make_feasible_edge(h, d, target):
    """
    Convert direction d into a feasible edge step s.
    s = alpha * d projected to be feasible: sum(s)=0, 0<=h_i+s_i<=1.
    """
    n = len(h)
    d = d - np.mean(d)
    
    pos = d > 1e-14
    neg = d < -1e-14
    max_pos = 1e10
    max_neg = 1e10
    if np.any(pos):
        max_pos = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        max_neg = float(np.min(h[neg] / (-d[neg])))
    
    alpha_max = min(max_pos, max_neg)
    if alpha_max < 1e-14:
        return None
    
    # Take 80% of max step for safety
    alpha = 0.8 * alpha_max
    s = alpha * d
    s -= np.mean(s)
    
    # Verify feasibility
    if np.any(h + s < -1e-10) or np.any(h + s > 1 + 1e-10):
        # Re-project
        alpha *= 0.5
        s = alpha * d
        s -= np.mean(s)
        if np.any(h + s < -1e-10) or np.any(h + s > 1 + 1e-10):
            return None
    
    return s

def _pair_step(h, i, j, delta):
    """Create a pair exchange step: increase h[i] by delta, decrease h[j] by delta."""
    n = len(h)
    # Check feasibility
    if h[i] + delta > 1 + 1e-10 or h[j] - delta < -1e-10:
        return None
    s = np.zeros(n)
    s[i] = delta
    s[j] = -delta
    return s

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, s):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(s, 1.0 - h, mode='full') - np.correlate(h, s, mode='full')
    Q = np.correlate(s, s, mode='full')
    a_lo, a_hi = 0.0, 1.0
    pos = s > 1e-14
    neg = s < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / s[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-s[neg]))))
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    cands = [a_lo, a_hi, max(a_lo, min(a_hi, 0.5))]
    M = min(8, len(S0))
    top_idx = np.argsort(-S0)[:M]
    for ii in range(M):
        ki = int(top_idx[ii])
        if Q[ki] > 1e-14:
            av = T[ki] / (2.0 * Q[ki])
            if a_lo <= av <= a_hi:
                cands.append(av)
    width = a_hi - a_lo
    ng = min(24, max(10, int(width * 60)))
    cands.extend(np.linspace(a_lo, a_hi, ng).tolist())
    for ii in range(min(M, 5)):
        for jj in range(ii + 1, min(M, 5)):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    span = 0.04 * width
    lo_r = max(a_lo, best_a - span)
    hi_r = min(a_hi, best_a + span)
    if hi_r - lo_r > 1e-10:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(28):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _lp_direction(h, k_top, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    top_k = min(k_top, len(S0))
    top_idx = np.argsort(-S0)[:top_k]
    grads = []
    for t in top_idx:
        xi = t - (n - 1)
        G = np.zeros(n)
        if xi >= 0:
            L = n - xi
            if L > 0:
                G[:L] += (1.0 - h[xi:xi + L])
                G[xi:xi + L] -= h[:L]
        else:
            s = -xi
            L = n - s
            if L > 0:
                G[s:s + L] += (1.0 - h[:L])
                G[:L] -= h[s:s + L]
        nm = np.linalg.norm(G)
        if nm > 1e-10:
            grads.append(G / nm)
    if len(grads) == 0:
        return None
    A_mat = np.array(grads)
    m = len(A_mat)
    lo_b = -np.minimum(radius, h)
    hi_b = np.minimum(radius, 1.0 - h)
    try:
        s_var = cp.Variable(n)
        tau_var = cp.Variable()
        constraints = [
            cp.sum(s_var) == 0,
            s_var >= lo_b,
            s_var <= hi_b,
            tau_var >= 0,
        ]
        for i in range(m):
            constraints.append(A_mat[i] @ s_var <= -tau_var)
        prob = cp.Problem(cp.Maximize(tau_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.CBC]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    if s_var.value is not None and tau_var.value > 1e-8:
                        sv = np.array(s_var.value).flatten().astype(float)
                        sv -= np.mean(sv)
                        return sv
                    solved = True
                    break
            except Exception:
                continue
    except Exception:
        pass
    return None

def _agg_grad(h, k_top):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    top_k = min(k_top, len(S0))
    top_idx = np.argsort(-S0)[:top_k]
    G = np.zeros(n)
    for t in top_idx:
        xi = t - (n - 1)
        if xi >= 0:
            L = n - xi
            if L > 0:
                G[:L] += (1.0 - h[xi:xi + L])
                G[xi:xi + L] -= h[:L]
        else:
            s = -xi
            L = n - s
            if L > 0:
                G[s:s + L] += (1.0 - h[:L])
                G[:L] -= h[s:s + L]
    nm = np.linalg.norm(G)
    if nm < 1e-10:
        return np.zeros(n)
    return -G / nm

def _local_opt(h, deadline, k_top=8, stall_lim=50, use_lp=True):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(12345)
    stall = 0
    radius = 0.12
    iter_count = 0
    while time.time() < deadline - 0.3:
        stall += 1 if stall > 0 else 0
        if stall > stall_lim:
            break
        directions = []
        if use_lp and time.time() < deadline - 3:
            slp = _lp_direction(h, k_top, radius)
            if slp is not None:
                directions.append(slp)
        g = _agg_grad(h, k_top)
        if np.linalg.norm(g) > 1e-10:
            directions.append(g)
        n_pairs = 3
        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and abs(h[i] - h[j]) > 0.02:
                s = np.zeros(n)
                delta = 0.4 * (h[j] - h[i])
                if h[i] + delta <= 1 - 1e-8 and h[j] - delta >= 1e-8:
                    s[i] = delta
                    s[j] = -delta
                    directions.append(s)
        best_new = cur
        best_d = None
        best_a = 0.0
        for d in directions:
            if time.time() > deadline - 0.3:
                break
            d = d - np.mean(d)
            if np.linalg.norm(d) < 1e-12:
                continue
            a, c_try = _line_search(h, d)
            if c_try < best_new - 1e-13:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None:
            h_new = h + best_a * best_d
            h_new = _project(h_new, target)
            c_new = _exact_c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            cur = c_new
            stall = 0
            radius = min(radius * 1.03, 0.28)
            iter_count += 1
        else:
            radius *= 0.78
            if radius < 0.008:
                pert = rng.normal(0, 0.04, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp2 = _exact_c5(hp)
                if cp2 < cur - 1e-12:
                    h = hp
                    cur = cp2
                    radius = 0.10
                    stall = 0
                elif stall > stall_lim // 2:
                    rad_increments = [0.02, 0.05, 0.08]
                    for ri in rad_increments:
                        pert2 = rng.normal(0, ri, n)
                        pert2 -= np.mean(pert2)
                        hp2 = _project(h + pert2, target)
                        cp3 = _exact_c5(hp2)
                        if cp3 < cur - 1e-12:
                            h = hp2
                            cur = cp3
                            radius = 0.08
                            stall = 0
                            break
                    else:
                        break
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    interp = np.interp(xn, xo, h)
    return _project(interp, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_refine(hb, deadline, max_iter=30000):
    n = len(hb)
    cur = hb.copy()
    cc = _exact_c5(cur)
    best_h = cur.copy()
    best_c = cc
    rng = np.random.default_rng(555)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        nc = _exact_c5(cur)
        if nc <= cc + 1e-14:
            ones[i] = oz
            zeros[j] = oi
            cc = nc
            stall = 0
            if nc < best_c - 1e-14:
                best_c = nc
                best_h = cur.copy()
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c

def _weyl_start(n, alpha, offset=0.0):
    x = (np.arange(n, dtype=float) * alpha + offset) % 1.0
    h = np.where(x < 0.5, 1.0, 0.0)
    cnt = int(np.sum(h))
    if cnt != n // 2:
        if cnt > n // 2:
            idx = np.where(h > 0.5)[0]
            order = np.argsort(x[idx])
            for i in range(cnt - n // 2):
                h[idx[order[i]]] = 0.0
        else:
            idx = np.where(h <= 0.5)[0]
            order = np.argsort(-x[idx])
            for i in range(n // 2 - cnt):
                h[idx[order[i]]] = 1.0
    return h

def _block_start(n, blocks):
    h = np.zeros(n)
    pos = 0
    for bLen, val in blocks:
        end = min(pos + bLen, n)
        h[pos:end] = val
        pos = end
    return _project(h, n // 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            n0 = 180
            h0 = _weyl_start(n0, (5.0 - math.sqrt(5.0)) / 2.0)
        n0 = len(h0)
    else:
        n0 = 180
        h0 = _weyl_start(n0, (5.0 - math.sqrt(5.0)) / 2.0)

    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: LP-based local optimization on initial (20%)
    d1 = t0 + budget_s * 0.20
    h1, c1 = _local_opt(h_init, d1, k_top=8, stall_lim=50, use_lp=True)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Binary refinement (10%)
    d2 = t0 + budget_s * 0.30
    hb = _to_binary(best_h, len(best_h))
    hb, cb = _local_opt(_project(hb.copy(), len(hb) / 2.0), min(d2, t0 + budget_s * 0.12), k_top=10, stall_lim=30, use_lp=True)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    if time.time() < d2:
        hb2 = _to_binary(best_h, len(best_h))
        hb2, cb2 = _binary_refine(hb2, d2)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
        if time.time() < d2 - 2:
            hf = _project(best_h.copy(), len(best_h) / 2.0)
            hf, cf = _local_opt(hf, d2, k_top=6, stall_lim=30, use_lp=True)
            if cf < best_c:
                best_c = cf
                best_h = hf.copy()

    d3 = t0 + budget_s * 0.55
    ncur = len(best_h)
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -6, 6, -10, 10]

    for dn in offsets:

            break
        n_try = ncur + dn

            elif cs < best_c + 0.002 and time.time() < d4 - 3:
