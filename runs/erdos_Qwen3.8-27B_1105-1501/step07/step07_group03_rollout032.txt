2. **Key insight**: The overlap integral at lag k is:

   For a direction d, the change in raw[k] is approximately:
   `d_raw[k] = sum_i (d[i] * (1-h[i+k]) + h[i] * (-d[i+k]))` (linearized)
   
   Actually more precisely: `raw[k] = sum_i h[i](1-h[i+k])` so
   `d(raw[k]) = sum_i (d[i](1-h[i+k]) - h[i]d[i+k])`

   So if I want to minimize `max_k raw[k]`, I can set up:
   - minimize t
   - subject to: for each active lag k, raw[k] + d_raw[k] <= t
   - sum(d[i]) = 0 (preserving the constraint)
   - ||d||_inf <= radius (trust region)
   - 0 <= h[i] + d[i] <= 1

Let me think about the quadratic line search more carefully.

For direction d, h(α) = h + α*d. Then:
raw_k(α) = sum_i (h[i]+α*d[i]) * (1 - h[i+k] - α*d[i+k])
= sum_i h[i](1-h[i+k]) + α*sum_i (d[i](1-h[i+k]) - h[i]d[i+k]) - α²*sum_i d[i]*d[i+k]

So raw_k(α) = raw[k] + α*T_k - α²*Q_k

where T_k = sum_i (d[i](1-h[i+k]) - h[i]d[i+k]) and Q_k = sum_i d[i]*d[i+k]

Note: T_k can be computed as:
T_k = correlate(d, 1-h, mode='full')[k] - correlate(h, d, mode='full')[k]
Q_k = correlate(d, d, mode='full')[k]

The maximum over k of (raw[k] + α*T_k - α²*Q_k) is what we want to minimize over α.

This is a concave maximization of a set of concave quadratics (in α for α > 0), so the max is convex in α. We can use the bisection/golden section approach.

```python
import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        m = n - shift
        if m > 0:
            g[:m] += (1.0 - h[shift:shift + m])
            g[shift:] -= h[:m]
    else:
        gp = -shift
        m = n - gp
        if m > 0:
            g[gp:] += (1.0 - h[:m])
            g[:m] -= h[gp:]
    return g

def _all_lag_gradients(h):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw

def _line_search(h, d, raw, dx, a_max):
    """Exact line search: find alpha that minimizes max_k (raw[k] + alpha*T_k - alpha^2*Q_k)"""
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Clip a_max to feasible range
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
    a_max = max(a_max, 1e-10)
    
    # Golden section search on [0, a_max]
    def f(alpha):
        return float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, a_max
    c = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc, fd = f(c), f(d_)
    
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d_
            d_ = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = f(d_)
    
    alpha = (lo + hi) / 2.0
    return alpha, f(alpha)

def _lp_minimax(h, top_count, radius, deadline):
    """Solve LP to find direction that equalizes top lags."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-raw)[:top_count]
    
    # Compute gradients for selected lags
    grads = np.array([_lag_gradient(h, int(k)) for k in top_idx])
    raw_vals = raw[top_idx].astype(float)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    cons = [
        t_var >= raw_vals + grads @ d_var,
        cp.sum(d_var) == 0,
        d_var >= -h + 1e-9,
        d_var <= 1.0 - h - 1e-9,
        d_var >= -radius,
        d_var <= radius
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=2000, abstol=1e-8)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    
    if d_var.value is None:
        return None, raw
    
    d_val = np.array(d_var.value).flatten().astype(float)
    return d_val, raw

def _lp_optimize(h, deadline, max_iter=200, top_count=15, radius_init=0.3):
    """Main LP-based optimization loop."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        d, raw = _lp_minimax(h, top_count, radius, deadline)
        if d is None:
            stall += 5
            if stall > 50:
                break
            radius *= 0.8
            continue
        
        # Exact line search
        dx = 2.0 / n
        alpha, new_c5 = _line_search(h, d, raw, dx, radius)
        
        if alpha < 1e-10:
            stall += 1
            radius *= 0.9
            if stall > 80:
                break
            continue
        
        h_new = h + alpha * d
        h_new = _project(h_new, n / 2.0)
        actual = _exact_c5(h_new)
        
        if actual < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.85
            if stall > 60:
                break
    
    return best_h, best_c5

def _gradient_direction(h, raw, beta=20.0):
    """Compute smoothed gradient direction (like before)."""
    n = len(h)
    shifted = raw - float(np.max(raw))
    vp = np.exp(np.clip(beta * shifted, -80, 0))
    w = vp / (float(np.sum(vp)) + 1e-30)
    
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * u[:m]
                g[:m] -= w[r] * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w[r] * u[gp:]
                g[gp:] -= w[r] * h[:m]
    g -= np.mean(g)
    return g

def _gradient_optimize(h, deadline, beta=20.0, radius=0.2, stall_limit=60):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):

def _resample(h, n_new):
    n = len(h)

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
```

```python
import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        m = n - shift
        if m > 0:
            g[:m] += (1.0 - h[shift:shift + m])
            g[shift:] -= h[:m]
    else:
        gp = -shift
        m = n - gp
        if m > 0:
            g[gp:] += (1.0 - h[:m])
            g[:m] -= h[gp:]
    return g

def _line_search(h, d, raw, dx, a_max):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_max = min(a_max, float(np.min(h[neg] / (-d[neg]))))
    a_max = max(a_max, 1e-10)

    def f(alpha):
        return float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, a_max
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc, fd = f(c), f(dd)
    for _ in range(55):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = dd
            dd = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = dd
            fc = fd
            dd = lo + gr * (hi - lo)
            fd = f(dd)
    alpha = (lo + hi) / 2.0
    return alpha, f(alpha)

def _lp_solve(h, active_idx, radius):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    grads = np.array([_lag_gradient(h, int(k)) for k in active_idx])
    raw_vals = raw[active_idx].astype(float)
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    lb = np.maximum(-h, -radius)
    ub = np.minimum(1.0 - h, radius)
    cons = [
        t_var >= raw_vals + grads @ d_var,
        cp.sum(d_var) == 0,
        d_var >= lb,
        d_var <= ub
    ]
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iter=3000, abstol=1e-8, reltol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=3000)
        except Exception:
            return None
    if d_var.value is None:
        return None
    return np.array(d_var.value).flatten().astype(float)

def _lp_optimize(h, deadline, max_iter=150, top_count=14, radius_init=0.3):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    tc = top_count

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 70:
            break

        raw = np.correlate(h, 1.0 - h, mode='full')
        max_raw = float(np.max(raw))
        threshold = max_raw - 0.015 / dx
        active_mask = raw >= threshold
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = np.argsort(-raw)[:tc]
        else:
            active_idx = active_idx[:min(len(active_idx), tc + 5)]

        d_val = _lp_solve(h, active_idx, radius)

        if d_val is None or np.linalg.norm(d_val) < 1e-10:
            shifted = raw - max_raw
            vp = np.exp(np.clip(20.0 * shifted, -80, 0))
            w = vp / (float(np.sum(vp)) + 1e-30)
            g = np.zeros(n)
            u = 1.0 - h
            for r in range(2 * n - 1):
                if w[r] < 1e-12:
                    continue
                sh = r - (n - 1)
                if sh >= 0:
                    m = n - sh
                    if m > 0:
                        g[sh:] += w[r] * u[:m]
                        g[:m] -= w[r] * h[sh:]
                else:
                    gp = -sh
                    m = n - gp
                    if m > 0:
                        g[:m] += w[r] * u[gp:]
                        g[gp:] -= w[r] * h[:m]
            g -= np.mean(g)
            nm = np.linalg.norm(g)
            if nm < 1e-12:
                stall += 5
                radius *= 0.8
                if stall > 20:
                    break
                continue
            d_val = -g / nm

        alpha, new_c5 = _line_search(h, d_val, raw, dx, radius)
        if alpha < 1e-10:
            stall += 1
            radius *= 0.85
            tc = max(5, tc - 1)
            continue

        h_new = h + alpha * d_val
        h_new = _project(h_new, n / 2.0)
        actual = _exact_c5(h_new)

        if actual < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = actual
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.5)
            tc = min(top_count + 4, 25)
        else:
            stall += 1
            radius *= 0.85
            tc = max(5, tc - 1)

def _gradient_optimize(h, deadline, beta=22.0, radius=0.15, stall_limit=60):

            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        max_raw = float(np.max(raw))
        shifted = raw - max_raw
        vp = np.exp(np.clip(beta_cur * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        g = np.zeros(n)
        u = 1.0 - h
        for r in range(2 * n - 1):
            if w[r] < 1e-12:
                continue
            sh = r - (n - 1)
            if sh >= 0:
                m = n - sh
                if m > 0:
                    g[sh:] += w[r] * u[:m]
                    g[:m] -= w[r] * h[sh:]
            else:
                gp = -sh
                m = n - gp
                if m > 0:
                    g[:m] += w[r] * u[gp:]
                    g[gp:] -= w[r] * h[:m]
        g -= np.mean(g)
        nm = np.linalg.norm(g)
        if nm < 1e-12:
            break
        d = -g / nm

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003, seed=4242):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()

    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()

    for _ in range(max_iter):
        if time.time() > deadline or stall > 350:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))

        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9993
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
        ones = np.where(cur > 0.5)[0].tolist()
        zeros = np.where(cur <= 0.5)[0].tolist()

    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    improved = True

    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-15:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = np.where(cur > 0.5)[0].tolist()
                    zeros = np.where(cur <= 0.5)[0].tolist()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 6.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_cont(h, deadline, n_pairs=5000, radius=0.2, seed=111):

def _alt_start(n, rng):

    idx = rng.choice(n, size=half, replace=False)
```
