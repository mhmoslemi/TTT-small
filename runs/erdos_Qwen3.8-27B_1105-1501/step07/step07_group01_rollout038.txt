import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, p):
    """Compute gradient of smooth-max with weights p (length 2n-1)."""
    n = len(h)
    grad = np.zeros(n)
    g = 1.0 - h
    for d in range(n):
        if d == 0:
            grad += p[n-1] * (1.0 - 2.0 * h)
        else:
            grad[d:] += p[n-1+d] * g[:n-d]
            grad[:n-d] -= p[n-1+d] * h[d:]
            grad[:n-d] += p[n-1-d] * g[d:]
            grad[d:] -= p[n-1-d] * h[:n-d]
    return grad

def _smooth_max_optimize(h, deadline, beta_start=5.0, beta_end=60.0, 
                          step_start=0.01, max_iters=500, seed=42):
    """Optimize using smooth-max gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        # Compute S and softmax weights
        S = np.correlate(h, 1.0 - h, mode='full')
        beta = beta_start + (beta_end - beta_start) * (it / max(float(max_iters), 1))
        S_max = float(S.max())
        p = np.exp(beta * (S - S_max))
        p /= p.sum()
        
        # Compute gradient
        grad = _smooth_grad(h, p)
        # Project to sum=0
        grad -= grad.mean()
        
        # Line search (backtracking)
        step = step_start
        improved = False
        for _ in range(15):
            h_try = h - step * grad
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                break
            step *= 0.5
        if not improved:
            stall += 1
            # Add noise to escape
            if stall > 10:
                noise = rng.normal(0, 0.01 * stall, n)
                h = _project(h + noise, target)
                cur_c5 = _exact_c5(h)
    
    return best_h, best_c5

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x < 1.0)
        else:
            mask = (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, p):
    n = len(h)
    grad = np.zeros(n)
    g = 1.0 - h
    for d in range(n):
        if d == 0:
            grad += p[n - 1] * (1.0 - 2.0 * h)
        else:
            grad[d:] += p[n - 1 + d] * g[:n - d]
            grad[:n - d] -= p[n - 1 + d] * h[d:]
            grad[:n - d] += p[n - 1 - d] * g[d:]
            grad[d:] -= p[n - 1 - d] * h[:n - d]
    return grad

        grad = _smooth_grad(h, p)

def _pair_line_search(h, S, i, j, n):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sturmian_init(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        frac = (i * alpha) % 1.0
        if frac < 0.5:
            h[i] = 1.0
    h = _project(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth-max gradient descent (35% budget)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _smooth_max_opt(h_init, d1, beta_start=6.0, beta_end=45.0, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient-guided pair optimization (20%)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _gradient_guided_pair_opt(best_h, d2, n_iters=30000, seed=seed + 1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Random pair optimization (10%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _guided_pair_opt(best_h, d3, n_pairs=20000, seed=seed + 2)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Another round of smooth-max (10%)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _smooth_max_opt(best_h, d4, beta_start=20.0, beta_end=80.0, seed=seed + 3)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-n exploration (15%)
    d5 = t0 + budget_s * 0.90
    n_curr = len(best_h)
    for dn in [-16, -12, -8, -4, 4, 8, 12, 16, 20, -20]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 300:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _smooth_max_opt(h_r, sub_dl, beta_start=8.0, beta_end=50.0, seed=seed + 100 + dn)
        if time.time() < d5 - 1 and c5_r <= best_c5 + 0.001:
            sub_dl2 = min(d5, time.time() + budget_s * 0.01)
            h_r, c5_r2 = _gradient_guided_pair_opt(h_r, sub_dl2, n_iters=5000, seed=seed + 200 + dn)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2 if False else h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Alternative starting points (8%)
    d6 = t0 + budget_s * 0.98
    alts = []
    golden = (1.0 - np.sqrt(5.0)) / 2.0
    alts.append(_sturmian_init(n0, abs(golden)))
    alts.append(_sinusoidal_init(n0, 3, 0.3, 0.0, seed))
    alts.append(_sinusoidal_init(n0, 7, 0.25, np.pi / 4, seed))
    alts.append(_sinusoidal_init(n0, 11, 0.2, 1.0, seed))
    for ai, h_alt in enumerate(alts):
        if time.time() > d6:
            break
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_a, c5_a = _smooth_max_opt(h_alt, sub_dl, beta_start=10.0, beta_end=60.0, seed=seed + 500 + ai)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 7: Final intensive polish (2%)
    d7 = deadline
    if time.time() < d7 - 5:
        sub_dl = min(d7, time.time() + budget_s * 0.008)
        h7, c5_7 = _smooth_max_opt(best_h, sub_dl, beta_start=30.0, beta_end=100.0, seed=seed + 999)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    if time.time() < d7 - 3:
        sub_dl = min(d7, time.time() + budget_s * 0.008)
        h7, c5_7 = _gradient_guided_pair_opt(best_h, sub_dl, n_iters=10000, seed=seed + 888)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
