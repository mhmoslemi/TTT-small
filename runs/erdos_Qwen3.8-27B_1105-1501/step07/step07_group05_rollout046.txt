        a, c5t = _line_search(h, d, dx, 0.5)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=15000, Ti=0.0015):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return bh, bc

def _multi_dir_opt(h, deadline, r0=0.02):
    """Aggressive multi-direction local optimization with very small steps."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(55555)
    rad = r0
    stall = 0
    k_vals = [2, 3, 4, 5, 7, 10, 15, 25, 40, 80]
    ki = 0
    
    while time.time() < deadline:
        if stall > 60:
            break
        if deadline - time.time() < 0.2:
            break
        K = k_vals[ki % len(k_vals)]
        ki += 1
        d = _topk_dir(h, K)
        nn = np.linalg.norm(d)
        if nn < 1e-12:
            stall += 1
            continue
        d = d * rad
        a, cv = _line_search(h, d, dx, rad)
        if cv < cur - 1e-14:
            h = h + a * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.005, 0.15)
        else:
            stall += 1
            rad *= 0.7
            if rad < 0.002:
                pert = rng.normal(0, 0.02, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.03
                elif stall % 30 == 0:
                    rad = 0.03
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)
    
    # Phase 1: Focused local opt with hard top-K (20%)
    d1 = t0 + budget_s * 0.18
    h1, c1 = _local_opt(h_init, d1, r0=0.025, stall_lim=50, use_lp=True)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    
    # Phase 2: Very aggressive small-step optimization (15%)
    d2 = t0 + budget_s * 0.33
    h2, c2 = _multi_dir_opt(_project(best_h.copy(), n0/2.0), d2, r0=0.015)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()
    
    # Phase 3: Pair swaps (8%)
    d3 = t0 + budget_s * 0.41
    h3, c3 = _pair_opt(_project(best_h.copy(), n0/2.0), d3, max_it=40000)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()
    
    # Phase 4: More local opt (15%)
    d4 = t0 + budget_s * 0.56
    h4, c4 = _local_opt(_project(best_h.copy(), n0/2.0), d4, r0=0.02, stall_lim=60, use_lp=True)
    if c4 < best_c:
        best_c = c4
        best_h = h4.copy()
    if time.time() < d4 - 2:
        h4b, c4b = _multi_dir_opt(_project(best_h.copy(), n0/2.0), d4, r0=0.01)
        if c4b < best_c:
            best_c = c4b
            best_h = h4b.copy()
    
    # Phase 5: Binary exploration (10%)
    d5 = t0 + budget_s * 0.66
    hbin = _to_bin(best_h, n0)
    hbin, cb = _bin_sa(hbin, d5, mx=20000, Ti=0.001)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    # Re-optimize from binary
    if time.time() < d5 - 3:
        hf = _project(hbin.copy(), n0 / 2.0)
        hf, cf = _local_opt(hf, d5, r0=0.02, stall_lim=40, use_lp=True)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.78
    nc = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = nc + dn
        if n_try < 100 or n_try > 250 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d6, time.time() + budget_s * 0.012)
        hr, cr = _local_opt(hr, sd, r0=0.015, stall_lim=35, use_lp=True)
        if cr < best_c + 0.002 and time.time() < d6 - 5:
            sd2 = min(d6, time.time() + budget_s * 0.008)
            hr, cr = _multi_dir_opt(_project(hr, n_try/2.0), sd2, r0=0.01)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try
    
    # Phase 7: Intensive final optimization (10%)
    d7 = t0 + budget_s * 0.88
    for r_init in [0.02, 0.01, 0.005]:
        if time.time() > d7 - 2:
            break
        h7 = _project(best_h.copy(), len(best_h) / 2.0)
        sd = min(d7, time.time() + budget_s * 0.03)
        h7, c7 = _local_opt(h7, sd, r0=r_init, stall_lim=50, use_lp=True)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
        if time.time() < d7 - 3:
            sd = min(d7, time.time() + budget_s * 0.015)
            h7, c7 = _multi_dir_opt(_project(best_h.copy(), len(best_h)/2.0), sd, r0=0.008)
            if c7 < best_c:
                best_c = c7
                best_h = h7.copy()
        if time.time() < d7 - 2:
            sd = min(d7, time.time() + budget_s * 0.01)
            h7, c7 = _pair_opt(_project(best_h.copy(), len(best_h)/2.0), sd, max_it=10000)
            if c7 < best_c:
                best_c = c7
                best_h = h7.copy()
    
    # Phase 8: Basin hopping (6%)
    d8 = deadline
    scale = 0.02
    n_re = 0
    while time.time() < d8 and n_re < 40:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.5, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d8, time.time() + budget_s * 0.006)
        ht, ct = _local_opt(hp, sd, r0=0.01, stall_lim=25, use_lp=False)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)
    
    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _multi_dir_opt(hf, deadline - 1, r0=0.005)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
