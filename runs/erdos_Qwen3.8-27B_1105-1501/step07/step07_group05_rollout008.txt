    for ii in range(M):
        for jj in range(ii + 1, M):
            ki, kj = int(top_idx[ii]), int(top_idx[jj])
            A = Q[kj] - Q[ki]
            B = T[ki] - T[kj]
            C = S0[ki] - S0[kj]
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        cands.append(ar)
            else:
                disc = B * B - 4.0 * A * C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                        if a_lo <= ar <= a_hi:
                            cands.append(ar)
    best_val = float('inf')
    best_a = 0.0
    for a in cands:
        a = max(a_lo, min(a_hi, a))
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    lo_r = max(a_lo, best_a - 0.05 * width)
    hi_r = min(a_hi, best_a + 0.05 * width)
    if hi_r - lo_r > 1e-11:
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_r - gr * (hi_r - lo_r)
        c2 = lo_r + gr * (hi_r - lo_r)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if hi_r - lo_r < 1e-12:
                break
            if f1 < f2:
                hi_r, c2, f2 = c2, c1, f1
                c1 = hi_r - gr * (hi_r - lo_r)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_r, c1, f1 = c1, c2, f2
                c2 = lo_r + gr * (hi_r - lo_r)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
        mid = (lo_r + hi_r) / 2.0
        fmid = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
        if fmid < best_val:
            best_val = fmid
            best_a = mid
    return best_a, best_val

def _smooth_dir(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    sc = S * dx
    mx = float(np.max(sc))
    v = np.clip(beta * (sc - mx), -80.0, 0.0)
    w = np.exp(v)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[n - 1 + l]
        if wl < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                g[:m] += wl * (1.0 - h[l:l + m])
                g[l:] -= wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                g[s:] += wl * (1.0 - h[:m])
                g[:m] -= wl * h[s:s + m]
    g -= np.mean(g)
    return -g

def _gen_dirs(h, rng, nd=5):
    n = len(h)
    dirs = []
    for beta in [15.0, 35.0, 60.0]:
        g = _smooth_dir(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    g15 = _smooth_dir(h, 15.0)
    for bs in [8, 16, 32]:
        if len(dirs) >= nd:
            break
        d = np.zeros(n)
        nb = n // bs
        for b in range(nb):
            d[b * bs:(b + 1) * bs] = np.mean(g15[b * bs:(b + 1) * bs])
        if nb * bs < n:
            d[nb * bs:] = np.mean(g15[nb * bs:])
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    if len(dirs) < nd:
        x = np.arange(n, dtype=float)
        d = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(0.5, 10.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

def _local_opt(h, deadline, r0=0.20, stall_lim=80, nd=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _c5_of(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(12345)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.5:
            break
        dirs = _gen_dirs(h, rng, nd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _c5_of(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.05, 0.35)
        else:
            stall += 1
            rad *= 0.7
            if rad < 0.006:
                pert = rng.normal(0, 0.06, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _c5_of(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.12
    return best_h, best_c

def _pair_opt(h, deadline, max_it=40000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _c5_of(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(55555)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _c5_of(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _bin_sa_best_of_n(h_bin, deadline, max_it=80000, batch=8):
    """Binary SA with best-of-batch selection."""
    n = len(h_bin)
    h = h_bin.copy()
    cur_c = _c5_of(h)
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(888)
    T = 0.004
    stall = 0
    dx = 2.0 / n
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if not ones or not zeros:
            break
        # Generate batch of candidate swaps, pick best
        best_delta = float('inf')
        best_a = -1
        best_b = -1
        for _ in range(batch):
            ai = int(rng.integers(0, len(ones)))
            bi = int(rng.integers(0, len(zeros)))
            a = ones[ai]
            b = zeros[bi]
            h[a] = 0.0
            h[b] = 1.0
            nc = _c5_of(h)
            h[a] = 1.0
            h[b] = 0.0
            dl = nc - cur_c
            if dl < best_delta:
                best_delta = dl
                best_a = a
                best_b = b
        a = best_a
        b = best_b
        h[a] = 0.0
        h[b] = 1.0
        nc = _c5_of(h)
        dl = nc - cur_c
        if dl < 0 or (T > 1e-8 and rng.random() < math.exp(-dl / T)):
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            cur_c = nc
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                stall = 0
                T = max(T * 0.997, 0.0005)
            else:
                T = max(T * 0.999, 0.0005)
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
            T = min(T * 1.002, 0.01)
    return best_h, best_c

def _bin_sa_multi(h_bin, deadline, max_it=30000):
    """Binary SA with occasional 2-for-2 swaps."""
    n = len(h_bin)
    h = h_bin.copy()
    cur_c = _c5_of(h)
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(4242)
    T = 0.003
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 350:
            break
        if not ones or not zeros:
            break
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.2:
            # 2-for-2 swap
            a1 = int(rng.integers(0, len(ones)))
            a2 = int(rng.integers(0, len(ones)))
            while a2 == a1:
                a2 = int(rng.integers(0, len(ones)))
            b1 = int(rng.integers(0, len(zeros)))
            b2 = int(rng.integers(0, len(zeros)))
            while b2 == b1:
                b2 = int(rng.integers(0, len(zeros)))
            pa, pb = ones[a1], ones[a2]
            pc, pd = zeros[b1], zeros[b2]
            h[pa] = 0.0
            h[pb] = 0.0
            h[pc] = 1.0
            h[pd] = 1.0
            nc = _c5_of(h)
            dl = nc - cur_c
            if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
                ones = list(np.where(h > 0.5)[0])
                zeros = list(np.where(h <= 0.5)[0])
                cur_c = nc
                if cur_c < best_c - 1e-14:
                    best_c = cur_c
                    best_h = h.copy()
                    stall = 0
                T *= 0.998
            else:
                h[pa] = 1.0
                h[pb] = 1.0
                h[pc] = 0.0
                h[pd] = 0.0
                stall += 1
        else:
            ai = int(rng.integers(0, len(ones)))
            bi = int(rng.integers(0, len(zeros)))
            a = ones[ai]
            b = zeros[bi]
            h[a] = 0.0
            h[b] = 1.0
            nc = _c5_of(h)
            dl = nc - cur_c
            if dl < 0 or rng.random() < math.exp(-dl / max(T, 1e-12)):
                ones.remove(a)
                ones.append(b)
                zeros.remove(b)
                zeros.append(a)
                cur_c = nc
                if cur_c < best_c - 1e-14:
                    best_c = cur_c
                    best_h = h.copy()
                    stall = 0
                T *= 0.999
            else:
                h[a] = 1.0
                h[b] = 0.0
                stall += 1
                T = min(T * 1.001, 0.015)
    return best_h, best_c

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _alt_start(n, rng):
    """Generate diverse initial binary patterns."""
    h = np.zeros(n)
    method = rng.integers(0, 4)
    if method == 0:
        # Alternating with noise
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        # Random flips
        k = int(n * rng.uniform(0.05, 0.15))
        pos = rng.choice(n, size=k, replace=False)
        h[pos] = 1.0 - h[pos]
        # Fix count
        ones = int(np.sum(h > 0.5))
        target = n // 2
        if ones > target:
            excess = ones - target
            one_pos = np.where(h > 0.5)[0]
            flip = rng.choice(one_pos, size=excess, replace=False)
            h[flip] = 0.0
        elif ones < target:
            deficit = target - ones
            zero_pos = np.where(h <= 0.5)[0]
            flip = rng.choice(zero_pos, size=deficit, replace=False)
            h[flip] = 1.0
    elif method == 1:
        # Block pattern with varying block sizes
        sizes = []
        remaining = n
        while remaining > 0:
            s = int(rng.integers(3, 12))
            s = min(s, remaining)
            sizes.append(s)
            remaining -= s
        h = np.zeros(n)
        pos = 0
        for i, s in enumerate(sizes):
            if i % 2 == 0:
                h[pos:pos+s] = 1.0
            pos += s
        # Adjust to get exactly n/2 ones
        ones = int(np.sum(h > 0.5))
        target = n // 2
        if ones > target:
            excess = ones - target
            one_pos = np.where(h > 0.5)[0]
            flip = rng.choice(one_pos, size=excess, replace=False)
            h[flip] = 0.0
        elif ones < target:
            deficit = target - ones
            zero_pos = np.where(h <= 0.5)[0]
            flip = rng.choice(zero_pos, size=deficit, replace=False)
            h[flip] = 1.0
    elif method == 2:
        # Sine-wave threshold
        x = np.arange(n)
        wave = np.sin(2 * np.pi * rng.uniform(2, 8) * x / n + rng.uniform(0, 2*np.pi))
        thr = np.percentile(wave, 50.0)
        h[wave > thr] = 1.0
        ones = int(np.sum(h > 0.5))
        target = n // 2
        if ones > target:
            excess = ones - target
            one_pos = np.where(h > 0.5)[0]
            flip = rng.choice(one_pos, size=excess, replace=False)
            h[flip] = 0.0
        elif ones < target:
            deficit = target - ones
            zero_pos = np.where(h <= 0.5)[0]
            flip = rng.choice(zero_pos, size=deficit, replace=False)
            h[flip] = 1.0
    else:
        # Random
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Start from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _c5_of(h_init)

    # Phase 1: Quick local opt on initial (10%)
    d1 = t0 + budget_s * 0.10
    h1, c1 = _local_opt(h_init, d1, r0=0.15, stall_lim=40, nd=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Binary SA with best-of-batch (15%)
    d2 = t0 + budget_s * 0.25
    hb = _to_bin(best_h, n0)
    hb, cb = _bin_sa_best_of_n(hb, d2, max_it=40000, batch=6)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Quick fractional polish
    if time.time() < d2 - 2:
        hp = _project(best_h.copy(), n0 / 2.0)
        hp, cp = _local_opt(hp, d2, r0=0.08, stall_lim=25, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 3: Multi-resolution exploration (25%)
    d3 = t0 + budget_s * 0.50
    nc = len(best_h)
    offsets = [0, -10, 10, -20, 20, -40, 40, -60, 60]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = nc + dn
        if n_try < 80 or n_try > 300 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d3, time.time() + budget_s * 0.015)
        hr, cr = _local_opt(hr, sd, r0=0.12, stall_lim=30, nd=4)
        # Binary round
        if time.time() < d3 - 5:
            hbr = _to_bin(hr, n_try)
            sd2 = min(d3, time.time() + budget_s * 0.012)
            hbr, cbr = _bin_sa_best_of_n(hbr, sd2, max_it=15000, batch=8)
            if cbr < cr:
                cr = cbr
                hr = hbr
        # If promising, do more work
        if cr < best_c + 0.003 and time.time() < d3 - 3:
            sd3 = min(d3, time.time() + budget_s * 0.01)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.05, stall_lim=20, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 4: Intensive binary optimization on best (20%)
    d4 = t0 + budget_s * 0.70
    nc = len(best_h)
    hb = _to_bin(best_h, nc)
    # Round 1: best-of-batch
    sd4a = min(d4, time.time() + budget_s * 0.08)
    hb1, c1b = _bin_sa_best_of_n(hb, sd4a, max_it=30000, batch=10)
    if c1b < best_c:
        best_c = c1b
        best_h = hb1.copy()
    # Round 2: multi-swap
    if time.time() < d4 - 5:
        hb2 = _to_bin(best_h, nc)
        sd4b = min(d4, time.time() + budget_s * 0.07)
        hb2, c2b = _bin_sa_multi(hb2, sd4b, max_it=20000)
        if c2b < best_c:
            best_c = c2b
            best_h = hb2.copy()
    # Fractional polish
    if time.time() < d4 - 3:
        hp = _project(best_h.copy(), nc / 2.0)
        hp, cp = _local_opt(hp, d4, r0=0.06, stall_lim=30, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 5: Pair optimization and refinement (12%)
    d5 = t0 + budget_s * 0.82
    nc = len(best_h)
    if time.time() < d5:
        h5 = _project(best_h.copy(), nc / 2.0)
        h5, c5 = _pair_opt(h5, min(d5, time.time() + budget_s * 0.05), max_it=20000)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    if time.time() < d5 - 2:
        h5 = _project(best_h.copy(), nc / 2.0)
        h5, c5 = _local_opt(h5, d5, r0=0.04, stall_lim=40, nd=5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Phase 6: Basin hopping (13%)
    d6 = deadline
    scale = 0.03
    n_re = 0
    while time.time() < d6 and n_re < 80:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(4):
            f = rng.uniform(0.3, 15.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.4, 1.0) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.5 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.006)
        ht, ct = _local_opt(hp, sd, r0=0.05, stall_lim=15, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Final polish
    if deadline - time.time() > 5:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline - 3, r0=0.02, stall_lim=30, nd=4)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=3000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _c5_of(best_h)
    return best_h, final_c, len(best_h)
