import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h0, n, target, k_top=30, radius=0.3, solver='CBC'):
    """Solve minimax LP to get a direction."""
    dx = 2.0 / n
    raw = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Select top k_top lag indices
    top_idx = np.argsort(-raw)[:k_top]
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
    ]
    
    h0_vec = cp.Constant(h0)
    
    for k in top_idx:
        d = k - (n - 1)
        # c_lin[k] = sum_m h[m] * (1-h0[m-d]) where m-d in [0,n-1] and m in [0,n-1]
        m_min = max(0, d)
        m_max = min(n - 1, n - 1 + d)
        if m_max < m_min:
            continue
        ms = np.arange(m_min, m_max + 1)
        coeffs = 1.0 - h0[ms - d]
        # c_lin[k] = sum_m x[m] * coeffs
        # We can write this as dot product of x with a sparse vector
        cvec = np.zeros(n)
        cvec[ms] = coeffs
        constraints.append(cvec @ x <= t)
    
    # Trust region: L1
    constraints.append(cp.norm1(x - h0_vec) <= radius)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='GLPK')
        except Exception:
            return None, None
    
    if x.value is None:
        return None, None
    
    x_lp = np.array(x.value).flatten()
    x_lp = _project(x_lp, target)
    
    # Direction
    d = x_lp - h0
    nm = np.linalg.norm(d)
    if nm < 1e-12:
        return None, None
    
    return d, t.value if t.value is not None else None

def _line_search_exact(h, direction, raw, dx, a_max, n_candidates=50):
    """Line search: minimize max(raw + a*T - a^2*Q) over a in [0, a_max]"""
    n = len(h)
    # T[k] = change in raw[k] per unit a (first order)
    # Q[k] = second order term
    # raw[k] = sum h[i] * (1-h[i+d])
    # h_new = h + a*d
    # raw_new[k] = sum (h[i]+a*d[i]) * (1-h[i+d]-a*d[i+d])
    #           = sum h[i](1-h[i+d]) + a*sum d[i](1-h[i+d]) - a*sum h[i]*d[i+d] - a^2*sum d[i]*d[i+d]
    #           = raw[k] + a*T[k] - a^2*Q[k]
    
    # Compute T and Q efficiently
    # T[k] = sum_valid d[i] * (1-h[i+d]) - sum_valid h[i] * d[i+d]
    #      = correlate(d, 1-h) - correlate(h, d) at index k
    u = 1.0 - h
    T = np.correlate(direction, u, mode='full') - np.correlate(h, direction, mode='full')
    Q = np.correlate(direction, direction, mode='full')
    
    # Find maximum feasible a
    a_hi = a_max
    pos = direction > 1e-12
    neg = direction < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-direction[neg]))))
    a_hi = max(a_hi, 1e-14)
    
    # Golden section search on the max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = a_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(raw + c * T - c * c * Q)) * dx
    fd = float(np.max(raw + d * T - d * d * Q)) * dx
    
    best_a = 0.0
    best_val = float(np.max(raw)) * dx
    
    for _ in range(60):
        if hi - lo < 1e-14:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(raw + c * T - c * c * Q)) * dx
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(raw + d * T - d * d * Q)) * dx
        
        eval_c = float(np.max(raw + c * T - c * c * Q)) * dx
        eval_d = float(np.max(raw + d * T - d * d * Q)) * dx
        if eval_c < best_val:
            best_val = eval_c
            best_a = c
        if eval_d < best_val:
            best_val = eval_d
            best_a = d
    
    opt = (lo + hi) / 2.0
    val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
    if val_opt < best_val:
        best_val = val_opt
        best_a = opt
    
    return best_a, best_val

def _gradient_direction(h, beta=20.0):
    """Compute smoothed gradient direction."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    v = np.exp(np.clip(beta * shifted, -80, 0))
    w = v / (float(np.sum(v)) + 1e-30)
    
    # Gradient of max over weighted lags
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * u[:m]
                g[:m] -= w[r] * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w[r] * u[gp:]
                g[gp:] -= w[r] * h[:m]
    
    g -= np.mean(g)
    return g

def _main_optimize_loop(h, deadline, max_iter=200, lp_every=3, radius_init=0.3):
    """Main optimization loop combining LP and gradient directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    iter_count = 0
    
    while time.time() < deadline and stall < 100:
        raw = np.correlate(h, 1.0 - h, mode='full')
        
        # Try LP direction every lp_every iterations
        d_dir = None
        if iter_count % lp_every == 0:
            d_dir, _ = _lp_direction(h, n, target, k_top=30, radius=radius)
        
        if d_dir is None or np.linalg.norm(d_dir) < 1e-10:
            # Fall back to gradient
            g = _gradient_direction(h, beta=20.0)
            nm = np.linalg.norm(g)
            if nm < 1e-12:
                break
            d_dir = -g / nm
        
        # Line search
        a_max = radius
        alpha, new_c5_est = _line_search_exact(h, d_dir, raw, dx, a_max)
        
        if alpha < 1e-10:
            # Try smaller radius
            alpha, new_c5_est = _line_search_exact(h, d_dir, raw, dx, a_max * 0.3)
        
        if alpha < 1e-10:
            stall += 1
            radius *= 0.8
            continue
        
        h_new = h + alpha * d_dir
        h_new = _project(h_new, target)
        actual_c5 = _exact_c5(h_new)
        
        if actual_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = actual_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.02, 0.4)
        else:
            # The linearized direction didn't help; try gradient
            g = _gradient_direction(h, beta=20.0)
            nm = np.linalg.norm(g)
            if nm > 1e-12:
                g_dir = -g / nm
                alpha2, _ = _line_search_exact(h, g_dir, raw, dx, radius)
                if alpha2 > 1e-10:

                            best_h = h.copy()
                        continue
            stall += 1
            radius *= 0.85
        
        iter_count += 1
    
    return best_h, best_c5

Let m = i+d, so i = m-d: = sum_{m=d}^{n-1} h[m] * (1-h0[m-d])

For d < 0 (let d' = -d > 0): c_lin[k] = sum_{i=d'}^{n-1} h[i-d'] * (1-h0[i])
Let m = i-d', so i = m+d': = sum_{m=0}^{n-1-d'} h[m] * (1-h0[m+d'])

In both cases, c_lin[k] = sum_m h[m] * (1-h0[m-d]) where m-d is in [0,n-1] and m is in [0,n-1].

So the coefficient vector for h in c_lin[k] is:
coeff[m] = (1-h0[m-d]) if 0 <= m-d < n, else 0

This is correct.

import numpy as np
import time
import cvxpy as cp

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, direction, raw, dx, a_max):
    """Exact line search along h + a*direction."""
    n = len(h)
    u = 1.0 - h
    T = np.correlate(direction, u, mode='full') - np.correlate(h, direction, mode='full')
    Q = np.correlate(direction, direction, mode='full')
    
    # Feasibility bound
    a_hi = a_max
    pos = direction > 1e-12
    neg = direction < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-direction[neg]))))
    a_hi = max(a_hi, 1e-14)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = a_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = float(np.max(raw + c * T - c * c * Q)) * dx
    fd = float(np.max(raw + d * T - d * d * Q)) * dx
    
    best_a = 0.0
    best_val = float(np.max(raw)) * dx
    
    for _ in range(50):
        if hi - lo < 1e-15:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(raw + c * T - c * c * Q)) * dx
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = float(np.max(raw + d * T - d * d * Q)) * dx
        
        if fc < best_val:
            best_val = fc
            best_a = c
        if fd < best_val:
            best_val = fd
            best_a = d
    
    opt = (lo + hi) / 2.0
    val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
    if val_opt < best_val:
        best_val = val_opt
        best_a = opt
    
    return best_a, best_val

def _lp_solve(h0, n, target, k_top=40, radius=0.25):
    """Solve minimax LP for direction."""
    raw = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(-raw)[:k_top]
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(x) == target,
        x >= 0,
        x <= 1,
    ]
    
    h0_const = cp.Constant(h0)
    constraints.append(cp.norm1(x - h0_const) <= radius)
    
    for k in top_idx:
        d = int(k) - (n - 1)
        m_min = max(0, d)
        m_max = min(n - 1, n - 1 + d)
        if m_max < m_min:
            continue
        ms = np.arange(m_min, m_max + 1)
        coeffs = 1.0 - h0[ms - d]
        cvec = np.zeros(n)
        cvec[ms] = coeffs
        constraints.append(cvec @ x <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    x_lp = None
    for solver in ['CBC', 'GLPK', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=solver)
            if x.value is not None:
                x_lp = np.array(x.value).flatten()
                break
        except Exception:
            continue
    
    if x_lp is None:
        return None
    
    x_lp = _project(x_lp, target)
    d = x_lp - h0
    if np.linalg.norm(d) < 1e-10:
        return None
    return d

def _grad_direction(h, beta=20.0):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(raw))
    shifted = raw - mx
    v = np.exp(np.clip(beta * shifted, -80, 0))
    w = v / (float(np.sum(v)) + 1e-30)
    
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w[r] * u[:m]
                g[:m] -= w[r] * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w[r] * u[gp:]
                g[gp:] -= w[r] * h[:m]
    
    g -= np.mean(g)
    return g

def _optimize(h, deadline, radius_init=0.25, lp_every=2, stall_limit=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    
    while time.time() < deadline and stall < stall_limit:
        raw = np.correlate(h, 1.0 - h, mode='full')
        
        # Get direction
        d_dir = None
        if it % lp_every == 0:
            d_dir = _lp_solve(h, n, target, k_top=40, radius=radius)
        
        if d_dir is None:
            g = _grad_direction(h, beta=20.0)
            nm = np.linalg.norm(g)
            if nm < 1e-12:
                break
            d_dir = -g / nm
        
        # Line search
        alpha, _ = _line_search(h, d_dir, raw, dx, radius)
        if alpha < 1e-10:
            alpha, _ = _line_search(h, d_dir, raw, dx, radius * 0.5)
        
        if alpha > 1e-10:
            h_new = h + alpha * d_dir
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                radius = min(radius * 1.01, 0.35)
                it += 1
                continue
        
        # Try gradient direction
        g = _grad_direction(h, beta=25.0)
        nm = np.linalg.norm(g)
        if nm > 1e-12:
            d_g = -g / nm
            alpha2, _ = _line_search(h, d_g, raw, dx, radius)
            if alpha2 > 1e-10:
                h_new = h + alpha2 * d_g
                h_new = _project(h_new, target)
                actual = _exact_c5(h_new)
                if actual < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = actual
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    it += 1
                    continue
        
        stall += 1
        radius *= 0.85
        if radius < 0.05:
            radius = 0.2 # reset
    
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=30000, T_init=0.002):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(12345)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline or stall > 400:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, zeros.__len__()))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = [x for x in ones if x != oi]
            zeros = [x for x in zeros if x != oz]
            ones.append(oz)
            zeros.append(oi)
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    best_h = cur.copy()
                    best_c5 = c_try
                    ones = [x for x in ones if x != oi]
                    zeros = [x for x in zeros if x != oz]
                    ones.append(oz)
                    zeros.append(oi)
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
