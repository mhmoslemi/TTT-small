Actually, I just realized something important. The LSE objective f_tau(h) = tau * log(sum exp(s_r/tau)) is a smooth approximation to max_r s_r. But the max_r s_r is the thing I want to minimize. So minimizing f_tau is the right thing to do.

However, there's a subtlety: f_tau(h) >= max_r s_r(h) (since log(sum exp) >= log(exp(max)) = max). So f_tau is an UPPER bound on the max. Minimizing f_tau will also tend to minimize the max, but the relationship isn't exact.

As tau -> 0, f_tau -> max_r s_r, so this is asymptotically correct.

For the SLSQP calls, I should:
1. Start with a moderate tau (e.g., 0.1) where the problem is well-behaved
2. Gradually decrease tau
3. At each tau level, run SLSQP to convergence (or near convergence)
4. Use the result as initial guess for the next (smaller) tau

This is the standard "continuation" or "homotopy" approach for nonsmooth optimization.

from scipy.optimize import minimize
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_scores(h)) * dx)

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999
        else:
            mask = x > 0.001
        cnt = np.sum(mask)
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[r, :m] += (1.0 - h[d:d+m])
            G[r, d:] += -h[:m]
        else:
            g = -d
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] += -h[g:]
    return G

def _lse_obj(h, tau):
    s = _scores(h)
    m = float(np.max(s))
    u = (s - m) / tau
    e = np.exp(u)
    return m + tau * np.log(np.sum(e))

def _lse_grad(h, tau):
    s = _scores(h)
    m = float(np.max(s))
    u = (s - m) / tau
    e = np.exp(u)
    p = e / np.sum(e)
    G = _build_G(h)
    return G.T @ p

def _lse_obj_grad(h, tau):
    s = _scores(h)
    m = float(np.max(s))
    u = (s - m) / tau
    e = np.exp(u)
    total = np.sum(e)
    p = e / total
    obj = m + tau * np.log(total)
    G = _build_G(h)
    grad = G.T @ p
    return obj, grad

def _slsqp_lse(h, tau, deadline, maxiter=200):
    """Minimize LSE objective using SLSQP."""
    n = len(h)
    target = n / 2.0
    
    def fun(x):
        return _lse_obj(x, tau)
    
    def jac(x):
        return _lse_grad(x, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.fun < _lse_obj(h, tau):
            h_new = _project_sum(res.x, target)
            return h_new
    except Exception:
        pass
    return h

def _pg_lse(h, tau, n_iter=300, deadline=None, momentum=0.9):
    """Projected gradient descent with momentum on LSE objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = _exact_c5(h)

                      b_eq=np.zeros(1), bounds=bounds, method='highs')

def _resample(h, n_new):

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    best_h = _project_sum(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: LSE Annealing with SLSQP (30%)
    d1 = t0 + budget_s * 0.30
    taus = [0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.002, 0.001]
    h_cur = best_h.copy()
    for tau in taus:
        if time.time() > d1 - 3:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.025)
        h_cur = _slsqp_opt(h_cur, tau, sub_dl, maxiter=150)
        h_cur = _project_sum(h_cur, n0 / 2.0)
        c5 = _exact_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()

    # Phase 2: Projected gradient with focused LSE (12%)
    d2 = t0 + budget_s * 0.42
    for tau in [0.01, 0.005, 0.003, 0.0015, 0.001]:
        if time.time() > d2 - 3:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.018)
        h_pg, c5_pg = _pg_lse(best_h.copy(), tau, 800, sub_dl, momentum=0.85, top_k=8)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: LP-based local optimization (15%)
    d3 = t0 + budget_s * 0.57
    h_lp = _project_sum(best_h.copy(), n0 / 2.0)
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_lp, c5_lp = _lp_loop(h_lp, sub_dl, radius_init=0.08, top_k_init=4, max_top_k=10, stall_limit=60)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    # Another LP pass with different params
    if time.time() < d3 - 5:
        h_lp2 = _project_sum(best_h.copy(), n0 / 2.0)
        sub_dl2 = min(d3, time.time() + budget_s * 0.05)
        h_lp2, c5_lp2 = _lp_loop(h_lp2, sub_dl2, radius_init=0.04, top_k_init=3, max_top_k=7, stall_limit=80)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Phase 4: Binary SA + pair sweep (15%)
    d4 = t0 + budget_s * 0.72
    h_bin = _to_binary(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.07)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=60000, T_init=0.003, seed=99999)
    sub_dl2 = min(d4, time.time() + budget_s * 0.03)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2, max_rounds=30)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    if time.time() < d4 - 3:
        h_pol = _project_sum(h_bin.copy(), n0 / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.02)
        h_pol, c5_pol = _lp_loop(h_pol, sub_dl3, radius_init=0.05, top_k_init=3, max_top_k=8, stall_limit=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    # SLSQP polish on binary
    if time.time() < d4 - 3:
        h_sls = _project_sum(h_bin.copy(), n0 / 2.0)
        sub_dl4 = min(d4, time.time() + budget_s * 0.02)
        h_sls = _slsqp_opt(h_sls, 0.005, sub_dl4, maxiter=100)
        c5_sls = _exact_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()

    # Phase 5: Multi-n exploration (13%)
    d5 = t0 + budget_s * 0.85
    n_best = len(best_h)
    for dn in [-8, -4, 4, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d5 - 8:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 280 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        # LSE on this n
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r = _slsqp_opt(h_r, 0.1, sub_dl, maxiter=60)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r = _slsqp_opt(h_r, 0.02, sub_dl, maxiter=80)
        h_r = _project_sum(h_r, n_try / 2.0)
        c5_r = _exact_c5(h_r)
        # LP
        if time.time() < d5 - 5:
            sub_dl = min(d5, time.time() + budget_s * 0.01)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.06, top_k_init=3, max_top_k=8, stall_limit=30)
        # Binary
        if time.time() < d5 - 5 and c5_r < best_c5 + 0.004:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d5, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=15000, T_init=0.002, seed=n_try + 1)
            sub_dl3 = min(d5, time.time() + budget_s * 0.004)
            h_b, c5_b = _pair_sweep(h_b, sub_dl3, max_rounds=12)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 6: Basin hopping (7%)
    d6 = t0 + budget_s * 0.92
    scale = 0.025
    for _ in range(150):
        if time.time() > d6 - 5:
            break
        h_pert = _perturb(best_h, rng, scale)
        n_p = len(h_pert)
        sub_dl = min(d6 - 5, time.time() + budget_s * 0.004)
        h_t = _slsqp_opt(h_pert, 0.005, sub_dl, maxiter=50)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.06)

    # Phase 7: Final polish (8%)
    d7 = deadline
    # SLSQP with tiny tau
    if deadline - time.time() > 8:
        h_f = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline - 8, time.time() + budget_s * 0.03)
        h_f = _slsqp_opt(h_f, 0.002, sub_dl, maxiter=150)
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # LP final
    if deadline - time.time() > 8:
        h_f = _project_sum(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline - 8, time.time() + budget_s * 0.02)
        h_f, c5_f = _lp_loop(h_f, sub_dl, radius_init=0.03, top_k_init=3, max_top_k=6, stall_limit=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # PG final
    if deadline - time.time() > 5:
        h_f, c5_f = _pg_lse(best_h.copy(), 0.001, 400, deadline - 5, momentum=0.8, top_k=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # Binary pair sweep final
    if deadline - time.time() > 5:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, deadline - 2, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
