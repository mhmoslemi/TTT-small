        i = int(rng.integers(0, m))
        x[i] = 1.0 - x[i]
        new_c5 = _exact_c5(_anti_h(x))
        delta = new_c5 - cur_c5

                best_x = x.copy()

def _binary_pair_sa(x0, deadline, seed=99, max_iter=50000, T_init=0.002):

def _full_space_opt(h, deadline, beta=20.0, maxiter=100):
    """Fallback: full-space smoothed gradient with projection."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(maxiter):
        if time.time() > deadline:
            break
        # Compute smooth gradient
        raw = np.correlate(h, 1.0 - h, mode='full')
        R = len(raw)
        M = float(raw.max())
        z = np.clip(beta * (raw - M), -50.0, 0.0)
        ew = np.exp(z)
        wsum = float(ew.sum())
        if wsum < 1e-30:
            idx = int(np.argmax(raw))
            w = np.zeros(R)
            w[idx] = 1.0
        else:
            w = ew / wsum
        
        G = np.zeros(n)
        for r in range(R):
            wr = w[r]
            if wr < 5e-9:
                continue
            tau = r - (n - 1)
            if tau >= 0:
                mk = n - tau
                G[tau:] += wr * (1.0 - h[:mk])
                G[:mk] -= wr * h[tau:tau + mk]
            else:
                gg = -tau
                mk = n - gg
                G[:mk] += wr * (1.0 - h[gg:])
                G[gg:] -= wr * h[:mk]
        
        # Project gradient to sum-zero subspace
        G -= float(np.mean(G))
        
        # Line search
        gnorm = float(np.linalg.norm(G))
        if gnorm < 1e-10:
            break
        d = -G / gnorm
        
        # Find valid step range
        pos = d > 1e-12
        neg = d < -1e-12
        a_lo, a_hi = -1.0, 1.0
        if np.any(pos):
            a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
            a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
        
        if a_hi <= a_lo + 1e-14:
            break
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = a_lo, a_hi
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(np.clip(h + c1 * d, 0, 1))
        f2 = _exact_c5(np.clip(h + c2 * d, 0, 1))
        for _ in range(40):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(np.clip(h + c1 * d, 0, 1))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(np.clip(h + c2 * d, 0, 1))
        
        a_opt = (lo + hi) / 2.0
        h_new = np.clip(h + a_opt * d, 0.0, 1.0)
        # Project to sum constraint
        h_new = _project_sum(h_new, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _basin_hopping_anti(x_best, deadline, rng, max_restarts=80):
    m = len(x_best)
    best_c5 = _exact_c5(_anti_h(x_best))
    best_x = x_best.copy()
    scale = 0.04
    
    for _ in range(max_restarts):
        if time.time() > deadline:
            break
        x_p = x_best + scale * rng.normal(0, 1, m)
        x_p = np.clip(x_p, 0.0, 1.0)
        sub_dl = min(deadline - 2, time.time() + 12)
        x_t, c5_t = _opt_anti(x_p, sub_dl, beta_seq=(10, 30, 60), maxiter_base=100)
        if c5_t < best_c5 - 1e-10:
            best_c5 = c5_t
            best_x = x_t.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.12, scale * 1.08)
    
    return best_x, best_c5
