        if p == q:
            continue
        if h[p] <= 0.01 or h[q] <= 0.01:
            continue
        if h[p] >= 0.99 and h[q] >= 0.99:
            continue
        c = h[p] + h[q]
        if c < 1e-10 or c > 2.0 - 1e-10:
            continue
        ep = _coeff_for_pos(h, p)
        eq = _coeff_for_pos(h, q)
        e_diff = ep - eq
        u_lo = max(0.0, c - 1.0) - h[p]
        u_hi = min(1.0, c) - h[p]
        if u_hi - u_lo < 1e-12:
            continue
        grid = np.linspace(u_lo, u_hi, 40)
        best_val = 1e30
        best_u = 0.0
        for u in grid:
            val = float(np.max(scores + u * e_diff))
            if val < best_val:
                best_val = val
                best_u = u
        for _ in range(25):
            active = np.where(scores + best_u * e_diff >= best_val - 1e-12)[0]
            if len(active) == 0:
                break
            slopes = e_diff[active]
            if np.all(slopes > 1e-10):
                u_hi = best_u
                best_u = (u_lo + u_hi) / 2.0
            elif np.all(slopes < -1e-10):
                u_lo = best_u
                best_u = (u_lo + u_hi) / 2.0
            else:
                break
        h[p] = h[p] + best_u
        h[q] = c - h[p]
        new_scores = _scores(h) / dx
        new_max = float(np.max(new_scores))
        if new_max * dx < cur_c5 - 1e-13:
            scores = new_scores
            cur_c5 = new_max * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            h[p] = h[p] - best_u
            h[q] = c - h[p]
            stall += 1
    return best_h, best_c5

def _binary_sa(b, deadline, T_init=0.002, max_iter=500000):
    n = len(b)
    half = n // 2
    cur = b.copy().astype(float)
    counts = np.correlate(cur.astype(int), (1 - cur).astype(int), mode='full').astype(int)
    cur_max = int(np.max(counts))
    dx = 2.0 / n
    cur_c5 = cur_max * dx
    best_b = cur.copy()
    best_c5 = cur_c5
    best_counts = counts.copy()
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].copy()
    zeros = np.where(cur <= 0.5)[0].copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = int(rng.integers(0, len(ones)))
        zi = int(rng.integers(0, len(zeros)))
        r = ones[oi]
        s = zeros[zi]
        delta = np.zeros(2 * n - 1, dtype=int)
        for m in range(-n + 1, n):
            d = 0
            if 0 <= r + m < n:
                d -= (1 - int(cur[r + m]))
            if 0 <= r - m < n:
                d += int(cur[r - m])
            if 0 <= s + m < n:
                d += (1 - int(cur[s + m]))
            if 0 <= s - m < n:
                d -= int(cur[s - m])
            if d != 0:
                delta[m + n - 1] = d
        new_counts = counts + delta
        new_max = int(np.max(new_counts))
        new_c5 = new_max * dx
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta_c5 / T)):
            counts = new_counts
            cur[r] = 0.0
            cur[s] = 1.0
            ones[oi] = s
            zeros[zi] = r
            cur_max = new_max
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = cur.copy()
                best_counts = counts.copy()
                stall = 0
            T *= 0.9998
        else:
            stall += 1
    return best_b, best_c5

def _fast_binary_sa(b, deadline, iterations):
    n = len(b)
    half = n // 2
    cur = b.copy().astype(np.int8)
    dx = 2.0 / n
    counts = np.correlate(cur, (1 - cur), mode='full').astype(np.int32)
    cur_max = int(np.max(counts))
    cur_c5 = cur_max * dx
    best_b = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = 0.002
    stall = 0
    for it in range(iterations):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        r = int(rng.integers(0, n))
        while cur[r] == 0:
            r = int(rng.integers(0, n))
        s = int(rng.integers(0, n))
        while cur[s] == 1:
            s = int(rng.integers(0, n))
        if r == s:
            continue
        delta = np.zeros(2 * n - 1, dtype=np.int32)
        for m in range(-n + 1, n):
            d = 0
            if 0 <= r + m < n:
                d -= (1 - int(cur[r + m]))
            if 0 <= r - m < n:
                d += int(cur[r - m])
            if 0 <= s + m < n:
                d += (1 - int(cur[s + m]))
            if 0 <= s - m < n:
                d -= int(cur[s - m])
            if d != 0:
                delta[m + n - 1] = d
        new_counts = counts + delta
        new_max = int(np.max(new_counts))
        new_c5 = new_max * dx
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c5 / T)):
            counts = new_counts
            cur[r] = 0
            cur[s] = 1
            cur_max = new_max
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = cur.copy()
                stall = 0
            T *= 0.99993
        else:
            stall += 1
    return best_b.astype(float), best_c5

def _active_lp(h, scores, radius, n_active=15, gap=0.003):
    n = len(h)
    mx = float(np.max(scores))
    active = np.where(scores >= mx - gap)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > n_active:
        order = np.argsort(scores[active])
        active = active[order][-n_active:]
    lags = active - (n - 1)
    u = 1.0 - h
    G = np.zeros((len(active), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += u[:m]
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += u[g:]
                G[idx, g:] += -h[:m]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 2.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    grid_t = np.linspace(t_lo, t_hi, 40)
    grid_f = np.empty(40)
    for idx in range(40):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(39, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _gradient_direction(h, scores, beta=15.0, radius=0.1):
    n = len(h)
    mx = float(np.max(scores))
    shifted = np.clip(beta * (scores - mx), -40.0, 0.0)
    w = np.exp(shifted)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    lags = np.arange(len(scores)) - (n - 1)
    u = 1.0 - h
    g = np.zeros(n)
    for idx, d in enumerate(lags):
        wi = w[idx]
        if wi < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wi * u[:m]
                g[:m] += -wi * h[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += wi * u[gg:]
                g[gg:] += -wi * h[:m]
    g -= float(np.mean(g))
    pos = h > 1e-12
    neg = h < 1.0 - 1e-12
    g[pos] = np.minimum(g[pos], (1.0 - h[pos]) / radius)
    g[neg] = np.maximum(g[neg], -h[neg] / radius)
    g[h >= 1.0 - 1e-12] = np.minimum(g[h >= 1.0 - 1e-12], 0.0)
    g[h <= 1e-12] = np.maximum(g[h <= 1e-12], 0.0)
    g -= float(np.mean(g))
    return g

def _local_opt(h, deadline, radius_init=0.10, max_iters=3000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    gap = 0.003
    n_active = 12
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        scores = _scores(h)
        d = _active_lp(h, scores, radius, n_active=n_active, gap=gap)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            d = _gradient_direction(h, scores, beta=15.0, radius=radius)
            if float(np.max(np.abs(d))) < 1e-12:
                stall += 1
                radius *= 0.7
                gap *= 1.4
                n_active = min(n_active + 2, 25)
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
            gap = max(gap * 0.92, 0.001)
            n_active = max(n_active - 1, 8)
        else:
            stall += 1
            radius *= 0.72
            gap *= 1.3
            n_active = min(n_active + 2, 25)
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_balanced(n, rng):
    m = n // 2
    perm = rng.permutation(n)
    b = np.zeros(n)
    b[perm[:m]] = 1.0
    return b

def _low_disc_balanced(n, offset=0):
    m = n // 2
    b = np.zeros(n)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    for k in range(m):
        pos = int((k * golden + offset) % n)
        if b[pos] == 0:
            b[pos] = 1.0
        else:
            for alt in range(1, n):
                if (pos + alt) % n < n and b[(pos + alt) % n] == 0:
                    b[(pos + alt) % n] = 1.0
                    break
                if (pos - alt) % n >= 0 and b[(pos - alt) % n] == 0:
                    b[(pos - alt) % n] = 1.0
                    break
    if int(np.sum(b)) < m:
        for i in range(n):
            if b[i] == 0 and int(np.sum(b)) < m:
                b[i] = 1.0
    return b

def _arithmetic_balanced(n, step):
    m = n // 2
    g = np.gcd(step, n)
    if g > 1:
        step = step + 1
    b = np.zeros(n)
    for k in range(m):
        pos = (k * step) % n
        b[pos] = 1.0
    if int(np.sum(b)) != m:
        cnt = int(np.sum(b))
        if cnt < m:
            for i in range(n):
                if b[i] == 0 and cnt < m:
                    b[i] = 1.0
                    cnt += 1
    return b

def _pair_sweep(h, deadline, max_rounds=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h > 0.3)[0]
    zeros = np.where(h < 0.7)[0]
    rng = np.random.default_rng(777)
    stall = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(max(1, len(ones) // 4)):
            if time.time() > deadline:
                break
            zi = zeros[int(rng.integers(0, len(zeros)))]
            oj = ones[int(rng.integers(0, len(ones)))]
            d = np.zeros(n)
            d[zi] = 1.0
            d[oj] = -1.0
            alpha, c_try = _line_search(h, d, dx, 0.2)
            if c_try < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                ones = np.where(h > 0.3)[0]
                zeros = np.where(h < 0.7)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = _low_disc_balanced(180)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick LP polish (8% time)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_opt(h_init, d1, radius_init=0.08, max_iters=1500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary SA from diverse starts (45% time)
    d2 = t0 + budget_s * 0.53
    n_bin = n0 if n0 % 2 == 0 else n0 + 1
    if n_bin < 60 or n_bin > 300:
        n_bin = 180 if n0 % 2 == 0 else 180

    # Build diverse binary starts
    starts = []
    b_parent = _to_binary(best_h)
    if len(b_parent) == n_bin:
        starts.append(b_parent)
    else:
        b_p = _resample(best_h, n_bin)
        starts.append(_to_binary(b_p))

    starts.append(_random_balanced(n_bin, rng))
    starts.append(_low_disc_balanced(n_bin, offset=0.3))
    starts.append(_low_disc_balanced(n_bin, offset=1.7))
    starts.append(_arithmetic_balanced(n_bin, 7))
    starts.append(_arithmetic_balanced(n_bin, 13))
    starts.append(_arithmetic_balanced(n_bin, 19))
    b_rand2 = _random_balanced(n_bin, rng)
    starts.append(b_rand2)

    per_start_time = (d2 - time.time()) / max(1, len(starts))
    for si, b_start in enumerate(starts):
        sub_dl = min(d2, time.time() + per_start_time)
        b_best, c5_b = _fast_binary_sa(b_start, sub_dl, iterations=200000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b_best.copy()

    # Phase 3: Two-coordinate descent on best binary (15% time)
    d3 = t0 + budget_s * 0.68
    h_cd = _project(best_h.copy(), len(best_h) / 2.0)
    h_cd, c5_cd = _pair_cd2(h_cd, d3, max_iters=300000)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 4: LP local opt from binary and CD results (15% time)
    d4 = t0 + budget_s * 0.83
    candidates = []
    candidates.append(_project(best_h.copy(), len(best_h) / 2.0))
    b_cd = _to_binary(best_h)
    candidates.append(b_cd.astype(float))
    seen = []
    sub_t = (d4 - time.time()) / max(1, len(candidates))
    for ci, h_c in enumerate(candidates):
        if time.time() >= d4 - 1:
            break
        sub_dl = min(d4, time.time() + sub_t)
        h_lp, c5_lp = _local_opt(h_c, sub_dl, radius_init=0.06, max_iters=1500)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 5: Multi-resolution (8% time)
    d5 = t0 + budget_s * 0.91
    n_curr = len(best_h)
    tried = set([n_curr])
    for dn in [-12, -8, -4, 4, 8, 12, -16, 16]:
        if time.time() > d5:
            break
        n_try = n_curr + dn
        if n_try < 80 or n_try > 280 or n_try in tried or n_try % 2 != 0:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.08, max_iters=600)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 6: Pair sweep on best (5% time)
    d6 = t0 + budget_s * 0.96
    h_ps = _project(best_h.copy(), len(best_h) / 2.0)
    h_ps, c5_ps = _pair_sweep(h_ps, d6, max_rounds=150)
    if c5_ps < best_c5:
        best_c5 = c5_ps
        best_h = h_ps.copy()

    # Phase 7: Final polish
    n_f = len(best_h)
    h_f = _project(best_h.copy(), n_f / 2.0)
    h_f, c5_f = _local_opt(h_f, deadline - 0.5, radius_init=0.03, max_iters=500)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # CD final
    if deadline - time.time() > 5:
        h_cd2 = _project(best_h.copy(), len(best_h) / 2.0)
        h_cd2, c5_cd2 = _pair_cd2(h_cd2, deadline - 1, max_iters=50000)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
