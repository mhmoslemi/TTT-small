import numpy as np
import time


def compute_c_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    return float(np.max(compute_c_scaled(h)))


def make_feasible(v):
    n = len(v)
    target = n / 2.0
    v = np.clip(v.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        interior = np.where((result > 1e-8) & (result < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            adj = residual / len(interior)
            result[interior] += adj
            result = np.clip(result, 0.0, 1.0)
    return result


def feasible_interval(h, d):
    t_lo = -np.inf
    t_hi = np.inf
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((h[neg] - 1.0) / d[neg])))
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)
    if t_lo >= t_hi - 1e-12:
        return None
    return t_lo, t_hi


def line_coefficients(h, d):
    n = len(h)
    dx = 2.0 / n
    C = dx * np.correlate(h, 1.0 - h, mode='full')
    B = dx * (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full'))
    A = dx * (-np.correlate(d, d, mode='full'))
    return A, B, C


def exact_line_search(h, d, n_coarse=150, n_fine=100):
    interval = feasible_interval(h, d)
    if interval is None:
        return h, compute_c5(h)
    t_lo, t_hi = interval
    if t_hi - t_lo < 1e-10:
        return h, compute_c5(h)
    A, B, C = line_coefficients(h, d)
    t_vals = np.linspace(t_lo, t_hi, n_coarse)
    t2 = t_vals[:, None] ** 2
    t1 = t_vals[:, None]
    quad = t2 * A[None, :] + t1 * B[None, :] + C[None, :]
    phi = np.max(quad, axis=1)
    best_idx = int(np.argmin(phi))
    seg_lo = t_vals[max(0, best_idx - 1)]
    seg_hi = t_vals[min(n_coarse - 1, best_idx + 1)]
    if seg_hi - seg_lo < 1e-12:
        t_best = t_vals[best_idx]
    else:
        t_ref = np.linspace(seg_lo, seg_hi, n_fine)
        t2r = t_ref[:, None] ** 2
        t1r = t_ref[:, None]
        quad_r = t2r * A[None, :] + t1r * B[None, :] + C[None, :]
        phi_r = np.max(quad_r, axis=1)
        best_idx2 = int(np.argmin(phi_r))
        t_best = t_ref[best_idx2]
    h_new = h + t_best * d
    h_new = np.clip(h_new, 0.0, 1.0)

def active_gradient(h, top_r=10, tau=0.002):

    c = compute_c_scaled(h)

    weights = np.exp((c - c_max) / tau)

    weights[sorted_idx] = 1.0
    wsum = weights.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for j in range(2 * n - 1):

        if wj < 1e-12:

            stall += 1
            if stall > stall_limit:
                pert_size = 0.03 + 0.05 * rng.random()
                h = best_h + pert_size * rng.randn(n)
                h = make_feasible(h)
                stall = 0
        else:
            stall = max(0, stall - 2)

    return best_h, best_c5


def resample(h_src, n_target):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False) + 1.0 / n_target
    y = np.interp(x_tgt, x_src, h_src, left=0.0, right=0.0)
    y = np.clip(y, 0.0, 1.0)
    return make_feasible(y)


def coordinate_polish(h, time_budget, seed=42):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]

    while time.time() - t0 < time_budget:
        n_tries = 3000
        for _ in range(n_tries):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            delta = deltas[rng.randint(len(deltas))] * (1 if rng.random() < 0.5 else -1)
            h[i] += delta
            h[j] -= delta
            if h[i] < -1e-10 or h[i] > 1.0 + 1e-10 or h[j] < -1e-10 or h[j] > 1.0 + 1e-10:
                h[i] -= delta
                h[j] += delta
                continue
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
            else:
                h[i] -= delta
                h[j] += delta
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_init = None
    try:
        if 'initial_h_values' in globals() and len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = make_feasible(h_init)
        c5_init = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        best_n = n0

        alloc = min(remaining() * 0.55, 600.0)
        if alloc > 30:
            h_opt, c5_opt = optimize_from(h_init, alloc, seed)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = len(h_opt)

    if remaining() < 20:
        return best_h, best_c5, best_n

    n_options = [63, 47, 83, 99, 113, 127]
    if h_init is not None:
        if len(h_init) not in n_options:
            n_options.insert(0, len(h_init))

    rng_main = np.random.RandomState(seed + 7)

    for n in n_options:
        if remaining() < 30:
            break
        if n == best_n and best_h is not None:
            continue
        alloc = min(remaining() * 0.1, 100.0)
        if alloc < 15:
            break
        if h_init is not None and n != len(h_init):
            h_start = resample(h_init, n)
        else:
            h_start = np.zeros(n)
            h_start[:n // 2] = 1.0
            h_start = h_start[rng_main.permutation(n)]
        h_opt, c5_opt = optimize_from(h_start, alloc, seed + n * 31)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    if best_h is not None and remaining() > 30:
        alloc = min(remaining() * 0.3, 200.0)
        h_pol, c5_pol = coordinate_polish(best_h, alloc, seed + 9999)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
            best_n = len(h_pol)

    if best_h is not None and remaining() > 15:
        h_final = make_feasible(best_h)
        c5_final = compute_c5(h_final)
        if c5_final <= best_c5 + 1e-10:
            best_h = h_final
            best_c5 = c5_final
        else:
            best_h = make_feasible(best_h)
            best_c5 = compute_c5(best_h)

    if best_h is None:
        n_fallback = 63
        best_h = np.zeros(n_fallback)
        best_h[:n_fallback // 2] = 1.0
        best_h = best_h[np.random.RandomState(seed).permutation(n_fallback)]
        best_h = make_feasible(best_h)
        best_c5 = compute_c5(best_h)
        best_n = n_fallback

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
