= min_{t∈[0,1]} [max_k (const_k + slope_k * t) + λ*t]

This is convex in t (max of linear + linear). The derivative (where it exists) is:
d/dt = max_k slope_k (for the active k) + λ

Wait no, it's:
d/dt [max_k (const_k + slope_k * t) + λ*t]
= (active slope) + λ

where "active slope" is the slope of the line that achieves the max at the optimal t.

At the optimum, we need: active_slope + λ = 0 (or t at boundary).

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_C(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_Ck(h, k, n, dx):
    start = max(0, -k)
    end = min(n, n - k)
    return np.sum(h[start:end] * (1.0 - h[start + k:end + k])) * dx


def compute_grad_Ck(h, k, n, dx):
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if end <= start:
        return grad
    s1 = start
    e1 = end
    s2 = start + k
    e2 = end + k
    p_s = max(0, s2)
    p_e = min(n, e2)
    if p_s < p_e:
        grad[p_s:p_e] += dx * (1.0 - h[p_s - k:p_e - k])
    q_s = max(0, s1)
    q_e = min(n, e1)
    if q_s < q_e:
        grad[q_s + k:q_e + k] += dx * (-h[q_s:q_e])
    return grad


def compute_all_grads(h, n, dx):
    nags = 2 * n - 1
    G = np.zeros((nags, n))
    for k in range(-(n - 1), n):
        G[k + n - 1] = compute_grad_Ck(h, k, n, dx)
    return G


def project_feasible(z, n):
    z = np.asarray(z, dtype=float).ravel()
    target = n / 2.0
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    for _ in range(300):
        if abs(err) < 1e-14:
            break
        if err > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = err / len(idx)
        for i in idx:
            if abs(err) < 1e-15:
                break
            if err > 0:
                d = min(per, 1.0 - h[i])
            else:
                d = min(-per, h[i])
            h[i] += d
            err -= d
    return np.clip(h, 0.0, 1.0)


def smooth_obj(h, C, beta):
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax
    return cmax + (mk + np.log(ws)) / beta


def smooth_grad_total(h, G, C, beta):
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        w = np.ones(len(C)) / len(C)
    else:
        w = es / ws
    return G.T @ w


def smooth_solve(h0, n, beta, maxiter, budget_end):
    h0 = project_feasible(h0, n)
    nags = 2 * n - 1
    bounds = [(0.0, 1.0)] * n

    def obj(v):
        C = compute_all_C(v)
        return smooth_obj(v, C, beta)

    def grad(v):
        C = compute_all_C(v)
        G = compute_all_grads(v, n, 2.0 / n)
        return smooth_grad_total(v, G, C, beta)

    def eqf(v):
        return np.sum(v) - n / 2.0

    def eqj(v):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                       method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def active_set_subgrad_step(h, n, dx, alpha, C_vals):
    cmax = np.max(C_vals)
    active = np.where(C_vals >= cmax - 1e-12)[0]
    G = compute_all_grads(h, n, dx)
    g = G[active].mean(axis=0)
    h_new = project_feasible(h - alpha * g, n)
    return h_new


def coord_sweep(h, n, dx, lam):
    C_vals = compute_all_C(h)
    h_new = h.copy()
    for j in range(n):
        slopes = np.zeros(2 * n - 1)
        intercepts = np.zeros(2 * n - 1)
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)
            start = max(0, -k)
            end = min(n, n - k)
            s_k = 0.0
            c_k = 0.0
            if start <= j < end:
                if 0 <= j + k < n:
                    s_k += (1.0 - h_new[j + k])
                else:
                    s_k += 1.0
            i2 = j - k
            if start <= i2 < end and 0 <= j < n:
                s_k += (-h_new[i2])
                c_k += h_new[i2]
            slopes[k_idx] = s_k * dx
            intercepts[k_idx] = C_vals[k_idx] - s_k * dx * h_new[j]
        a, b = 0.0, 1.0
        for _ in range(50):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            f1 = np.max(slopes * m1 + intercepts) + lam * m1
            f2 = np.max(slopes * m2 + intercepts) + lam * m2
            if f1 < f2:
                b = m2
            else:
                a = m1
        t_new = (a + b) * 0.5
        h_new[j] = t_new
    return project_feasible(h_new, n)


def lagrangian_coord(h0, n, max_outer, max_sweeps, budget_end):
    lam = 0.0
    h = h0.copy()
    for outer in range(max_outer):
        if time.time() > budget_end:
            break
        for _ in range(max_sweeps):
            if time.time() > budget_end:
                break
            h = coord_sweep(h, n, 2.0 / n, lam)
        s = np.sum(h) - n / 2.0
        if abs(s) < 1e-6:
            break
        lam -= s * 0.5
        lam = np.clip(lam, -5.0, 5.0)
    return h


def binary_candidate(n, rng, mode=0):
    if n % 2 == 0:
        k = n // 2
    else:
        k = (n - 1) // 2
    h = np.zeros(n)
    if mode == 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
        rem = np.setdiff1d(np.arange(n), idx)
        if len(rem) > 0 and n % 2 == 1:
            h[rem[0]] = 1.0
    elif mode == 1:
        h[:k] = 1.0
    elif mode == 2:
        h[::2][:k] = 1.0
    else:
        blocks = n // 4
        for b in range(4):
            start_b = b * blocks
            h[start_b + b : start_b + b + blocks // 2] = 1.0
    return project_feasible(h, n)


def area_transfer(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return project_feasible(h_src.copy(), n_tgt)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    lo = np.concatenate([[0.0], (x_src[:-1] + x_src[1:]) / 2.0, [2.0]])
    hi = np.concatenate([(x_src[:-1] + x_src[1:]) / 2.0, [2.0]])
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    lo_t = np.concatenate([[0.0], (x_tgt[:-1] + x_tgt[1:]) / 2.0, [2.0]])
    hi_t = np.concatenate([(x_tgt[:-1] + x_tgt[1:]) / 2.0, [2.0]])
    result = np.zeros(n_tgt)
    for j in range(n_tgt):
        acc = 0.0
        for i in range(n_src):
            ol = max(lo_t[j], lo[i])
            oh = min(hi_t[j], hi[i])
            if oh > ol:
                acc += h_src[i] * (oh - ol)
        result[j] = acc / (2.0 / n_tgt)
    return project_feasible(result, n_tgt)


def try_update(h_cand, best_h, best_score):
    h_cand = project_feasible(h_cand, len(h_cand))
    if not np.all(np.isfinite(h_cand)):
        return h_cand, best_h, best_score
    sc = exact_score(h_cand)
    if np.isfinite(sc) and sc < best_score - 1e-14:
        return h_cand, h_cand.copy(), sc
    if np.isfinite(sc) and abs(sc - best_score) < 1e-14 and len(h_cand) < len(best_h):
        return h_cand, h_cand.copy(), sc
    return h_cand, best_h, best_score
