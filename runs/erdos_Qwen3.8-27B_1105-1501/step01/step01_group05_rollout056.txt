```python
def compute_gradient_lse(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr  # scaled correlations, length 2n-1
    
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / np.sum(exp_c)
    
    # Gradient
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # k values
    
    for r, k in enumerate(lags):
        if w[r] < 1e-15:
            continue
        # d(c_r)/dh[j] for each j
        # First term: (1-h[j-k]) for j such that 0<=j-k<n
        # Second term: -h[j+k] for j such that 0<=j+k<n
        # j-k >= 0 means j >= k; j-k < n means j < n+k
        # j+k >= 0 means j >= -k; j+k < n means j < n-k
        
        # First term: j in [max(0, k), min(n, n+k))
        j_start = max(0, k)
        j_end = min(n, n + k)
        if j_start < j_end:
            grad[j_start:j_end] += w[r] * dx * (1.0 - h[j_start - k:j_end - k])
        
        # Second term: j in [max(0, -k), min(n, n-k))
        j_start2 = max(0, -k)
        j_end2 = min(n, n - k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] += w[r] * dx * (-h[j_start2 + k:j_end2 + k])
    
    return grad
```

```python
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    return h

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    c_shifted = c - np.max(c)
    exp_c = np.exp(T * c_shifted)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for r in range(len(w)):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)
        # First term: (1-h[j-k]) for j in [max(0,k), min(n, n+k))
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += w[r] * dx * (1.0 - h[js - k:je - k])
        # Second term: -h[j+k] for j in [max(0,-k), min(n, n-k))
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] += w[r] * dx * (-h[js2 + k:je2 + k])
    return grad

def lsc_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(T * (c - c_max)))) / T

def lsc_local(h, T, max_iters=200, time_limit=30.0, seed=42):
    rng = np.random.default_rng(seed)
    h = project_to_target(h, len(h))
    n = len(h)
    score = official_score(h)
    obj = lsc_objective(h, T)
    t0 = time.perf_counter()
    
    lr = 0.5
    stale = 0
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        grad = lse_gradient(h, T)
        # Center the gradient (remove mean to preserve sum constraint approximately)
        grad -= np.mean(grad)
        
        # Direction
        direction = -grad
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        
        # Line search
        improved = False
        for alpha in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05, lr*0.01]:
            h_new = project_to_target(h + alpha * direction, n)
            s_new = official_score(h_new)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                obj = lsc_objective(h, T)
                stale = 0
                improved = True
                break
            elif s_new <= score + 1e-10:
                # Accept neutral moves occasionally
                if rng.random() < 0.3:
                    h = h_new
                    stale += 1
                    break
        
        if not improved:
            stale += 1
            lr *= 0.7
            # Try random perturbation if stuck
            if stale > 10:
                pert = rng.normal(0, 0.1, n)
                pert -= np.mean(pert)
                h_new = project_to_target(h + 0.1 * pert, n)
                s_new = official_score(h_new)
                if s_new < score + 0.001:
                    h = h_new
                    score = s_new
                stale = 0
                lr = 0.5
        
        if stale > 50:
            break
    
    return h, official_score(h)

def binary_swap_anneal(n, max_iters=5000, time_limit=20.0, seed=42):
    rng = np.random.default_rng(seed)
    ones = np.arange(n // 2)
    zeros = np.arange(n // 2, n)
    h = np.zeros(n)
    h[ones] = 1.0
    
    score = official_score(h)
    best_h = h.copy()
    best_score = score
    
    t0 = time.perf_counter()
    T_start = 0.1
    T_end = 0.001
    iters = min(max_iters, max(100, int(time_limit * 500)))
    
    one_set = set(ones.tolist())
    zero_set = set(zeros.tolist())
    
    for it in range(iters):
        if time.perf_counter() - t0 > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, iters - 1))
        
        # Pick random 1 position and random 0 position
        i = int(rng.integers(0, len(one_set)))
        j = int(rng.integers(0, len(zero_set)))
        # Convert to actual indices
        one_list = sorted(one_set)
        zero_list = sorted(zero_set)
        pos_i = one_list[i]
        pos_j = zero_list[j]
        
        # Swap
        h[pos_i] = 0.0
        h[pos_j] = 1.0
        one_set.remove(pos_i)
        one_set.add(pos_j)
        zero_set.remove(pos_j)
        zero_set.add(pos_i)
        
        s_new = official_score(h)
        delta = s_new - score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            score = s_new
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # Revert
            h[pos_i] = 1.0
            h[pos_j] = 0.0
            one_set.add(pos_i)
            one_set.remove(pos_j)
            zero_set.add(pos_j)
            zero_set.remove(pos_i)
    
    return best_h, best_score

def interpolate_to_n(parent, n_target, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n_target, 0.5), n_target)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    xt_sorted = np.sort(xt)
    order = np.argsort(xt)
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_target)

def slsqp_polish(h, time_limit=15.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        hx = x[:n]
        theta = x[n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        hx = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            # d(corr[ki])/dh[j]
            # First term: (1-h[j-k]) for j in [max(0,k), min(n, n+k))
            js = max(0, k)
            je = min(n, n + k)
            if js < je:
                j[ki, js:je] += dx * (1.0 - hx[js - k:je - k])
            # Second term: -h[j+k] for j in [max(0,-k), min(n, n-k))
            js2 = max(0, -k)
            je2 = min(n, n - k)
            if js2 < je2:
                j[ki, js2:je2] += dx * (-hx[js2 + k:je2 + k])
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    init_score = official_score(h)
    theta0 = init_score + 0.001
    x0 = np.concatenate([h, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.success or (result.fun is not None and np.isfinite(result.fun)):
            h_new = project_to_target(result.x[:n], n)
            s_new = official_score(h_new)
            if s_new < init_score - 1e-10:
                return h_new, s_new
    except Exception:
        pass
    return h, init_score

def targeted_local(h, max_iters=100, time_limit=20.0, seed=0):
    """Focus on reducing the most active lags."""
    rng = np.random.default_rng(seed)
    n = len(h)
    h = project_to_target(h, n)
    score = official_score(h)
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = corr * dx
        max_val = np.max(scaled)
        
        # Find active lags (within tolerance of max)
        tol = max(0.001, 0.02 * max_val)
        active = np.where(scaled >= max_val - tol)[0]
        
        if len(active) == 0:
            break
        
        # Compute combined gradient for active lags
        grad = np.zeros(n)
        for ki in active:
            k = ki - (n - 1)
            js = max(0, k)
            je = min(n, n + k)
            if js < je:
                grad[js:je] += dx * (1.0 - h[js - k:je - k])
            js2 = max(0, -k)
            je2 = min(n, n - k)
            if js2 < je2:
                grad[js2:je2] += dx * (-h[js2 + k:je2 + k])
        
        # Direction: reduce active lags
        direction = -grad
        direction -= np.mean(direction)
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        
        # Line search
        improved = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.5, 1.0]:
            h_new = project_to_target(h + alpha * direction, n)
            s_new = official_score(h_new)
            if s_new < score - 1e-12:
                h = h_new
                score = s_new
                improved = True
                break
        
        if not improved:
            # Try random pair swap
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for delta in [0.02, 0.05, 0.1, 0.2]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                    h_try[j] = np.clip(h_try[j] - delta, 0, 1)
                    h_new = project_to_target(h_try, n)
                    s_new = official_score(h_new)
                    if s_new < score - 1e-13:
                        h = h_new
                        score = s_new
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return h, official_score(h)
```
