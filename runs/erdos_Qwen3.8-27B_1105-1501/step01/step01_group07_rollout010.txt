def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                s = pair_scipy(cur, i, j, cur_c, n_pts=25)
                if s is not None:
                    cur_c = s
                    improved = True
            if not improved:
                for i in range(n):
                    if time.time() > final_end - 5:
                        break
                    s = coord_scipy(cur, i, cur_c, n_pts=20)
                    if s is not None:
                        cur_c = s
                        improved = True
            if not improved:
                break
        if cur_c < best_c:
            best_h, best_c = cur.copy(), cur_c

    # Final fine gradient
    if time.time() < final_end - 2:
        hf, cf = grad_descent_smooth(best_h, final_end - 2, 0.002, 0.00001, 0.998, 0.001, rng)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
