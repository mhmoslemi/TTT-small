4. Project gradient to sum-zero subspace: v = G - mean(G)
5. Line search along v

Let me also think about the exact line search. Given direction v (sum=0), the feasible interval for s is:
- For each i: if v_i > 0, s ≤ (1-h_i)/v_i; if v_i < 0, s ≥ -h_i/v_i

Within this interval [s_min, s_max], we want to minimize f(s) = max_k (F_k + L_k*s + Q_k*s²).

The minimum occurs at:
1. An endpoint of the interval, or
2. A point where two quadratics cross: F_a + L_a*s + Q_a*s² = F_b + L_b*s + Q_b*s²
   → (Q_a-Q_b)s² + (L_a-L_b)s + (F_a-F_b) = 0
3. A point where a single quadratic has its minimum (if Q_k > 0 and the vertex is in the interval)

Algorithm:
- Collect all candidate s values (endpoints, vertices of individual quadratics, pairwise crossings)
- Filter to those in [s_min, s_max]
- Evaluate f(s) at each candidate
- Return the best

This is exact (for the selected pair set). In practice, using all pairs of active constraints might be O(|A|²) which could be large. We can limit to top-K active constraints.

            elif jpk == i:
                vj_k = -1.0
            l_val += 1.0 * (1.0 - h[jpk]) - h[jpk] * vj_k
            q_val += -(1.0 * vj_k)
        L[idx] = l_val
        Q[idx] = q_val
    return L, Q

def find_best_s(F, L, Q, s_lo, s_hi, K_active=25):
    if s_hi - s_lo < 1e-13:
        return s_lo
    candidates = [s_lo, s_hi, 0.0]
    max_F = np.max(F)
    tol = 0.02 * max_F
    active = np.where(F > max_F - tol)[0]
    if len(active) > K_active:
        active = active[np.argsort(F[active])[-K_active:]]
    for idx in active:
        if Q[idx] > 1e-10:
            sv = -L[idx] / (2.0 * Q[idx])
            if s_lo < sv < s_hi:
                candidates.append(sv)
    for a in range(len(active)):
        for b in range(a + 1, len(active)):
            ia, ib = active[a], active[b]
            dQ = Q[ia] - Q[ib]
            dL = L[ia] - L[ib]
            dF = F[ia] - F[ib]
            if abs(dQ) < 1e-12:
                if abs(dL) > 1e-12:
                    sc = -dF / dL
                    if s_lo < sc < s_hi:
                        candidates.append(sc)
            else:
                disc = dL * dL - 4.0 * dQ * dF
                if disc >= 0:
                    sd = np.sqrt(disc)
                    s1 = (-dL + sd) / (2.0 * dQ)
                    s2 = (-dL - sd) / (2.0 * dQ)
                    if s_lo < s1 < s_hi:
                        candidates.append(s1)
                    if s_lo < s2 < s_hi:
                        candidates.append(s2)
    n_grid = 25
    for g in range(1, n_grid):
        candidates.append(s_lo + (s_hi - s_lo) * g / n_grid)
    best_s = 0.0
    best_val = np.inf
    for s in candidates:
        s = np.clip(s, s_lo, s_hi)
        val = np.max(F + L * s + Q * s * s)
        if val < best_val:
            best_val = val
            best_s = s
    return best_s, best_val

def try_pair_transfer(h, F, i, j):
    n = len(h)
    s_lo = max(0.0, h[i] - 1.0, -h[j])
    s_hi = min(h[i], 1.0 - h[j])
    if s_hi - s_lo < 1e-12:
        return None
    L, Q = pair_transfer_LQ(h, i, j)
    best_s, best_raw = find_best_s(F, L, Q, s_lo, s_hi)
    current_max_raw = np.max(F)
    if best_raw >= current_max_raw - 1e-10:
        return None
    h_new = h.copy()
    h_new[i] -= best_s
    h_new[j] += best_s
    h_new = make_feasible(h_new)
    return h_new

def subgradient_direction(h, F, G, top_k=15):
    max_F = np.max(F)
    tol = 0.015 * max_F + 0.001
    active = np.where(F > max_F - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(F)])
    w = F[active] - (max_F - tol)
    w = np.maximum(w, 0.01)
    w = w / w.sum()
    v = w @ G[active]
    v = v - np.mean(v)
    norm = np.linalg.norm(v)
    if norm < 1e-12:
        return None
    return v / norm

def line_search_general(h, v, F):
    n = len(h)
    s_lo = 0.0
    s_hi = 0.0
    for i in range(n):
        if v[i] > 1e-12:
            s_lo = max(s_lo, -h[i] / v[i])
            s_hi = min(s_hi, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-12:
            s_lo = max(s_lo, (1.0 - h[i]) / v[i])
            s_hi = min(s_hi, -h[i] / v[i])
    if s_hi - s_lo < 1e-12:
        return None
    best_h = None
    best_raw = np.inf
    n_points = 30
    for t in range(n_points + 1):
        s = s_lo + (s_hi - s_lo) * t / n_points
        h_s = h + s * v
        h_s = np.clip(h_s, 0.0, 1.0)
        h_s = make_feasible(h_s)
        raw = np.max(compute_all_F(h_s))
        if raw < best_raw:
            best_raw = raw
            best_h = h_s
    return best_h

def local_optimize(h, max_rounds=50, time_deadline=None):
    n = len(h)
    best_h = h.copy()
    best_score = official_score(best_h)
    F = compute_all_F(best_h)
    G = compute_gradient_matrix(best_h)
    current_time = time.time()
    for rnd in range(max_rounds):
        if time_deadline is not None and current_time > time_deadline:
            break
        improved = False
        max_F = np.max(F)
        tol = 0.015 * max_F + 0.001
        active = np.where(F > max_F - tol)[0]
        v_sub = subgradient_direction(best_h, F, G, top_k=15)
        if v_sub is not None:
            h_new = line_search_general(best_h, v_sub, F)
            if h_new is not None:
                sc = official_score(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new
                    best_score = sc
                    F = compute_all_F(best_h)
                    G = compute_gradient_matrix(best_h)
                    improved = True
        n_pairs_tried = 0
        max_pairs = min(n * (n - 1) // 2, 2000)
        active_idx = list(active)
        rng_local = np.random.default_rng(rnd * 1000 + n)
        tried_pairs = set()
        for _ in range(max_pairs):
            if time_deadline is not None and time.time() > time_deadline:
                break
            if n <= 70:
                i = rng_local.integers(0, n)
                j = rng_local.integers(0, n)
            else:
                i = int(rng_local.integers(0, n))
                j = int(rng_local.integers(0, n))
            if i == j or (i, j) in tried_pairs:
                continue
            tried_pairs.add((i, j))
            n_pairs_tried += 1
            h_new = try_pair_transfer(best_h, F, i, j)
            if h_new is not None:
                sc = official_score(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new
                    best_score = sc
                    F = compute_all_F(best_h)
                    G = compute_gradient_matrix(best_h)
                    improved = True
        if not improved and n_pairs_tried < 200:
            break
        if n_pairs_tried < 50 and not improved:
            break
    return best_h, best_score

def full_pair_pass(h, time_deadline=None):
    n = len(h)
    F = compute_all_F(h)
    best_h = h.copy()
    best_score = official_score(best_h)
    improved_count = 0
    order = np.random.default_rng(42).permutation(n * (n - 1) // 2)
    for idx in order:
        if time_deadline is not None and time.time() > time_deadline:
            break
        i = int(idx // (n - 1))
        j = int(idx % (n - 1))
        if j >= i:
            j += 1
        if i == j:
            continue
        h_new = try_pair_transfer(best_h, F, i, j)
        if h_new is not None:
            sc = official_score(h_new)
            if sc < best_score - 1e-12:
                best_h = h_new
                best_score = sc
                F = compute_all_F(best_h)
                improved_count += 1
    return best_h, best_score, improved_count

def interp_to_length(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

def gen_binary_pattern(n, rng):
    h = np.zeros(n)
    ones = n // 2
    if n % 2 == 1:
        h[rng.integers(0, n)] = 0.5
        ones -= 0
        positions = np.delete(np.arange(n), int(np.where(h > 0)[0][0]) if len(h > 0) else 0)
        selected = rng.choice(n, size=ones, replace=False)
        for pos in selected:
            if h[pos] == 0:
                h[pos] = 1.0
        s = np.sum(h)
        if abs(s - n / 2.0) < 1e-10:
            return make_feasible(h)
    positions = rng.permutation(n)
    h[positions[:ones]] = 1.0
    if n % 2 == 1:
        h[positions[ones]] = 1.0
        h[positions[ones + 1]] = 0.0
        h[positions[ones]] = 0.5
        h[positions[ones + 1]] = 0.5
    return make_feasible(h)

def gen_staircase(n, rng):
    h = np.zeros(n)
    mid = n // 2
    spread = max(2, n // 8)
    for i in range(n):
        if i < mid - spread:
            h[i] = 1.0
        elif i >= mid + spread:
            h[i] = 0.0
        else:
            t = (i - (mid - spread)) / max(1, 2 * spread)
            h[i] = 1.0 - t
    h += rng.normal(0, 0.1, n)
    return make_feasible(h)

def gen_cosine_pattern(n, phase, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.cos(2 * np.pi * x + phase)
    h2 = 0.5 + amp * 0.5 * np.cos(4 * np.pi * x + phase * 1.7)
    h = 0.7 * h + 0.3 * h2
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.04 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        parent = make_feasible(parent)
    else:
        parent = make_feasible(np.full(50, 0.5))

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(parent)

    def try_update(h_cand):
        nonlocal best_h, best_score, best_n
        if h_cand is None or not np.all(np.isfinite(h_cand)):
            return
        h_cand = make_feasible(h_cand)
        sc = official_score(h_cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc
            best_n = len(h_cand)

    length_list = []
    pn = len(parent)
    for ln in [pn, pn-4, pn+4, pn-8, pn+8, 36, 42, 48, 54, 61, 70, 78]:
        if 24 <= ln <= 96 and ln not in length_list:
            length_list.append(ln)
    length_list.sort()

    phase1_deadline = t_start + 0.20 * budget_s
    phase2_deadline = t_start + 0.55 * budget_s
    phase3_deadline = t_start + 0.85 * budget_s

    for n in length_list:
        if time.time() > phase1_deadline:
            break
        if n == len(parent):
            h_base = parent.copy()
        else:
            h_base = interp_to_length(parent, n)
        h_sym = make_feasible(1.0 - h_base[::-1])
        for h_init in [h_base, h_sym]:
            sc = official_score(h_init)
            if sc < best_score:
                best_h = h_init.copy()
                best_score = sc
                best_n = n
            d = deadline - (time.time() - t_start)
            tdl = min(phase1_deadline, time.time() + min(8.0, d * 0.3))
            h_opt, sc_opt = local_optimize(h_init, max_rounds=3, time_deadline=tdl)
            try_update(h_opt)

    # Phase 2: diverse starts + local optimization
    while time.time() < phase2_deadline:
        n_try = int(rng.choice(length_list[:8]))
        start_type = rng.integers(0, 5)
        if start_type == 0:
            h_init = interp_to_length(parent, n_try)
        elif start_type == 1:
            h_init = gen_binary_pattern(n_try, rng)
        elif start_type == 2:
            h_init = gen_staircase(n_try, rng)
        elif start_type == 3:
            h_init = gen_cosine_pattern(n_try, rng.uniform(0, 2 * np.pi), rng.uniform(0.2, 0.6))
        else:
            h_init = interp_to_length(best_h, n_try)
            h_init = make_feasible(h_init + rng.normal(0, 0.05, n_try))
        sc_init = official_score(h_init)
        if sc_init < best_score + 0.05:
            max_r = max(3, min(15, int((phase2_deadline - time.time()) * 0.3 / max(1, n_try / 50))))
            tdl = min(phase2_deadline - 1, time.time() + min(15.0, (phase2_deadline - time.time()) * 0.25))
            h_opt, sc_opt = local_optimize(h_init, max_rounds=max_r, time_deadline=tdl)
            try_update(h_opt)

    # Phase 3: intensive local optimization on best
    while time.time() < phase3_deadline:
        h_cur = best_h.copy()
        n_cur = len(h_cur)
        tdl = min(phase3_deadline - 1, time.time() + min(30.0, (phase3_deadline - time.time()) * 0.15))
        h_opt, sc_opt = local_optimize(h_cur, max_rounds=20, time_deadline=tdl)
        try_update(h_opt)
        # Also try full pair pass
        if time.time() < phase3_deadline - 5:
            tdl2 = min(phase3_deadline - 1, time.time() + min(20.0, (phase3_deadline - time.time()) * 0.2))
            h_fp, sc_fp, n_imp = full_pair_pass(best_h, time_deadline=tdl2)
            try_update(h_fp)
        # Try different length
        if time.time() < phase3_deadline - 10:
            for dl in [-4, 4, -8, 8]:
                alt_n = best_n + dl
                if 24 <= alt_n <= 96:
                    h_alt = interp_to_length(best_h, alt_n)
                    tdl3 = min(phase3_deadline - 2, time.time() + min(10.0, (phase3_deadline - time.time()) * 0.15))
                    h_opt2, _ = local_optimize(h_alt, max_rounds=5, time_deadline=tdl3)
                    try_update(h_opt2)

    # Final polish
    if time.time() < deadline - 5:
        tdl = min(deadline - 2, time.time() + min(15.0, (deadline - time.time()) * 0.5))
        h_opt, sc_opt = local_optimize(best_h, max_rounds=30, time_deadline=tdl)
        try_update(h_opt)

    if time.time() < deadline - 3:
        h_fp, _, _ = full_pair_pass(best_h, time_deadline=deadline - 2)
        try_update(h_fp)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score > best_score:
        best_h = make_feasible(best_h)
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
