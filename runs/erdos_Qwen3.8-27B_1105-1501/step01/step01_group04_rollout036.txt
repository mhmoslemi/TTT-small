            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


class _IneqCon:
    def __init__(self, k, N):
        self.k = k
        self.N = N

    def fun(self, x):
        h_x = x[1:]
        c_k = np.correlate(h_x, 1.0 - h_x, mode='full')[self.k]
        return x[0] - c_k * (2.0 / self.N)

    def jac(self, x):
        N = self.N
        dx = 2.0 / N
        l = self.k - (N - 1)
        h_x = x[1:]
        row = np.zeros(N + 1)
        row[0] = 1.0
        for m in range(N):
            a = m - l
            b = m + l
            v = 0.0
            if 0 <= a <= N - 1:
                v += 1.0 - h_x[a]
            if 0 <= b <= N - 1:
                v -= h_x[b]
            row[m + 1] = -dx * v
        return row


class _EqCon:
    def __init__(self, N):
        self.N = N

    def fun(self, x):
        return np.sum(x[1:]) - self.N / 2.0

    def jac(self, x):
        g = np.zeros(self.N + 1)
        g[1:] = 1.0
        return g


def _slsqp_obj(x):
    return x[0]


def _slsqp_obj_jac(x):
    g = np.zeros(len(x))
    g[0] = 1.0
    return g


def slsqp_optimize(h0, max_iter=500, ftol=1e-14, margin=3.0):
    N = len(h0)
    target = N / 2.0
    h = np.clip(h0, 0, 1).copy()
    h = project_box_sum(h, target)
    c0 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - margin)[0]

    constraints = []
    for k in active:
        c = _IneqCon(k, N)
        constraints.append({'type': 'ineq', 'fun': c.fun, 'jac': c.jac})

    eq_c = _EqCon(N)
    constraints.append({'type': 'eq', 'fun': eq_c.fun, 'jac': eq_c.jac})

    bounds = [(0.0, 1.0)] * (N + 1)
    x0 = np.zeros(N + 1)
    x0[0] = c0 + 0.005
    x0[1:] = h

    best_h = h.copy()
    best_c = c0

    try:
        result = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': ftol})
        if result.x is not None:
            h_new = np.clip(result.x[1:], 0, 1)
            h_new = project_box_sum(h_new, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    except Exception:
        pass

    return best_h, best_c
