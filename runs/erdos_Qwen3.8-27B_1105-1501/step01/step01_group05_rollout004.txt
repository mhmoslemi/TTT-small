def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = h[:n_target]
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-13:
            break

            idx = int(np.argmax(1.0 - h))
            h[idx] += min(residual, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            h[idx] -= min(-residual, h[idx])
    h = np.clip(h, 0.0, 1.0)
