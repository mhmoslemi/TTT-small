- If using projection: first find t that minimizes max_m(a_m*t + b_m) (ignoring sum constraint, just 0≤t≤1), then project

1. Evaluating f at t=0 and t=1
2. If f is decreasing at 0 and increasing at 1, the minimum is in between
3. Use bisection on the subgradient

1. Start with t = 0.5 (or some initial guess)
2. Find argmax set S
3. If min(a_m for m in S) ≤ 0 ≤ max(a_m for m in S), we're at the optimum
4. Otherwise, move t in the direction that reduces the max

This is essentially the same as the subgradient method but in 1D, so it converges in at most (number of kinks) steps.

        h = h_new

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    lo = -3.0
    hi = 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def tilted_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(beta * (corr - c_max), -500.0, 500.0)
    exp_s = np.exp(shifted)
    weights = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        d = m - (n - 1)
        if d >= 0:
            j1 = np.arange(0, n - d)
            grad[j1] += w * (1.0 - h[j1 + d])
            j2 = np.arange(d, n)
            grad[j2] -= w * h[j2 - d]
        else:
            D = -d
            j1 = np.arange(D, n)
            grad[j1] += w * (1.0 - h[j1 - D])
            j2 = np.arange(0, n - D)
            grad[j2] -= w * h[j2 + D]
    return grad * dx

    target = n / 2.0
    h = rng.random(n)
    return project_feasible(h, target)

def simulated_annealing(h, n, target_sum, n_iters, T0, T_end, rng):
    best_h = h.copy()
    best_c5 = eval_c5(h)
    T = T0
    for it in range(n_iters):
        T = T0 * (T_end / T0) ** (it / max(n_iters - 1, 1))
        h_pert = h.copy()
        n_pert = rng.integers(2, 6)
        idx = rng.choice(n, min(n_pert, n), replace=False)
        h_pert[idx] += rng.normal(0, 0.1, len(idx))
        h_pert = project_feasible(h_pert, target_sum)
        c5_new = eval_c5(h_pert)
        delta = c5_new - best_c5
        if c5_new < best_c5:
            h = h_pert
            best_c5 = c5_new
            best_h = h_pert.copy()
        elif rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_pert
    return best_h, best_c5

def local_polish(h, n, target_sum, rng, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    # Subgradient with different params
    for beta, lr0, decay, mom, niters in [
        (50, 0.02, 0.002, 0.5, 800),
        (100, 0.01, 0.002, 0.7, 800),
        (200, 0.008, 0.003, 0.8, 600),
        (500, 0.005, 0.003, 0.9, 600),
        (1000, 0.003, 0.004, 0.9, 500),
        (50, 0.015, 0.001, 0.3, 1000),
        (80, 0.012, 0.001, 0.5, 1000),
    ]:
        if time.time() - t_start > time_limit:
            break
        h_opt, c5_opt = subgrad_with_momentum(best_h, n, target_sum,
                                              niters, beta, lr0, decay, mom)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Coordinate descent polish
        if time.time() - t_start > time_limit:
            break
        h_opt2, c5_opt2 = coord_descent_1d(best_h, n, target_sum, 8)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
    # Simulated annealing
    if time.time() - t_start < time_limit * 0.7:
        h_sa, c5_sa = simulated_annealing(best_h, n, target_sum, 2000, 0.01, 0.0001, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish after SA
        if time.time() - t_start < time_limit * 0.9:
            h_opt3, c5_opt3 = subgrad_with_momentum(best_h, n, target_sum,
                                                    400, 200, 0.005, 0.003, 0.8)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining_frac(frac):
        return elapsed() < budget_s * frac

    # Get initial values
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_c5 = eval_c5(best_h)
    best_n = n_init

    # Phase 1: Optimize initial values aggressively
    # Allocate ~35% of budget
    t_phase1 = budget_s * 0.35
    h_opt, c5_opt = local_polish(best_h, n_init, target_init, rng, t_phase1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Try other n values with structured starts
    # Allocate ~35% of budget
    t_phase2 = budget_s * 0.35
    n_candidates = [64, 66, 68, 70, 72, 76, 80, 84, 96, 128]
    time_per_n = t_phase2 / len(n_candidates)

    for n in n_candidates:
        if elapsed() > t0 + budget_s * 0.70:
            break
        target = n / 2.0
        # Binary start
        h_try = gen_binary_start(n, rng)
        h_opt2, c5_2 = local_polish(h_try, n, target, rng, time_per_n * 0.6)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_opt2.copy()
            best_n = n
        # Random start if time permits
        if elapsed() < t0 + budget_s * 0.70:
            h_try3 = gen_random_start(n, rng)
            h_opt3, c5_3 = local_polish(h_try3, n, target, rng, time_per_n * 0.4)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h_opt3.copy()
                best_n = n

    # Phase 3: Perturbation-based global search on best
    # Allocate ~25% of budget
    t_phase3_budget = budget_s * 0.25
    pert_sizes = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]
    trial_count = 0
    max_trials = 30
    while elapsed() < t0 + budget_s * 0.95 and trial_count < max_trials:
        ps = pert_sizes[trial_count % len(pert_sizes)]
        trial_count += 1
        h_pert = best_h + rng.normal(0, ps, best_n)
        h_pert = project_feasible(h_pert, best_n / 2.0)
        c5_pert = eval_c5(h_pert)
        # Only polish if the perturbation didn't make things too bad
        if c5_pert < best_c5 + 0.02:
            t_left = budget_s * 0.95 - elapsed()
            if t_left > 5:
                h_opt4, c5_4 = local_polish(h_pert, best_n, best_n / 2.0,
                                            rng, min(t_left, 30))
                if c5_4 < best_c5:
                    best_c5 = c5_4
                    best_h = h_opt4.copy()

    # Phase 4: Final intensive polish on best
    if elapsed() < t0 + budget_s * 0.99:
        t_left = budget_s * 0.99 - elapsed()
        h_opt5, c5_5 = local_polish(best_h, best_n, best_n / 2.0, rng, t_left)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h_opt5.copy()

    # Final verification
    best_h = project_feasible(best_h, best_n / 2.0)
    # Ensure exact sum
    diff = best_n / 2.0 - float(np.sum(best_h))
    if abs(diff) > 1e-12:
        best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, best_n
