import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v to [0,1]^n with sum = n/2."""
    h = np.clip(v, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    # Bisection for shift
    lo, hi = -max(h.max(), 0.0) - 1.0, 1.0 - min(h.min(), 1.0) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact mass correction
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            if room[idx] <= 0:
                break
            h[idx] += min(residual, room[idx])
        else:
            idx = np.argmax(h)
            if h[idx] <= 0:
                break
            h[idx] -= min(-residual, h[idx])
    h = np.clip(h, 0.0, 1.0)
    return h

def corr_vector(h):
    """Return scaled correlation vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_row(h, k):
    """Gradient of correlation at offset k (signed, where k = index - (n-1))."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    p = np.arange(n)
    # h appears as first factor h_{j+k} when j+k is valid
    ok1 = (p + k >= 0) & (p + k < n)
    # For these j, h_{j+k} is the first factor, derivative wrt h_j contributes to (1-h_j) term
    # Wait, let me be more careful.
    # c_k = sum_j h_{j+k} * (1 - h_j) for valid j
    # d(c_k)/d(h_i) = (1-h_i) if i+k is valid (h_i appears as first factor at position i+k... no)
    # Actually: c_k = sum_{j: 0<=j<n, 0<=j+k<n} h_{j+k} (1-h_j)
    # d/dh_i: i appears in h_{j+k} when j+k=i, i.e., j=i-k (if valid)
    #         i appears in (1-h_j) when j=i (if valid and i+k < n)
    # So: d(c_k)/d(h_i) = (1-h_{i-k}) if 0<=i-k<n and 0<=i<n [first factor contribution]
    #                         -h_{i+k} if 0<=i<n and 0<=i+k<n [second factor contribution]
    
    # First factor: h_i appears as h_{j+k} where j = i-k
    ok_first = (p >= 0) & (p < n) & (p - k >= 0) & (p - k < n)
    # h_p = h_{j+k} => j = p-k, need j valid: 0<=p-k<n, and j+k=p which is valid by construction
    # Contribution: (1-h_j) = (1-h_{p-k})
    g[ok_first] += (1.0 - h[p[ok_first] - k])
    
    # Second factor: h_i appears in (1-h_j) where j=i
    ok_second = (p >= 0) & (p < n) & (p + k >= 0) & (p + k < n)
    # Contribution: -h_{j+k} = -h_{p+k}
    g[ok_second] += (-h[p[ok_second] + k])
    
    return g * dx

def lags_array(n):
    return np.arange(-(n-1), n)

def slsqp_active(h, n, top_k=20, maxiter=50, time_limit=30.0):
    """SLSQP with only top-k active constraints."""
    dx = 2.0 / n
    target_sum = n / 2.0
    laps = lags_array(n)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def make_ineq(idx_list):
        def ineq_fun(x):
            h_x = x[:n]
            theta = x[n]
            corr = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
            return theta - corr[idx_list]
        def ineq_jac(x):
            h_x = x[:n]
            L = len(idx_list)
            J = np.zeros((L, n + 1))
            J[:, n] = 1.0
            for ii, ki in enumerate(idx_list):
                k = laps[ki]
                J[ii, :n] = -grad_row(h_x, k)
            return J
        return {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    h = project_to_target(h, n)
    corr = corr_vector(h)
    score = float(np.max(corr))
    
    # Active set: top k correlations
    active_idx = np.argsort(corr)[-top_k:]
    
    theta0 = score + 0.001
    x0 = np.concatenate([h, [theta0]])
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        make_ineq(active_idx)
    ]
    
    best_h = h.copy()
    best_score = score
    start = time.perf_counter()
    
    # Iterate: solve, re-evaluate, update active set
    for it in range(3):
        if time.perf_counter() - start > time_limit:
            break
        cons = constraints[0]  # eq
        # Rebuild active set each iteration
        corr = corr_vector(h)
        score = float(np.max(corr))
        if score < best_score:
            best_score = score
            best_h = h.copy()
        active_idx = np.argsort(corr)[-top_k:]
        theta0 = score + 0.001
        x0 = np.concatenate([h, [theta0]])
        
        try:
            res = optimize.minimize(
                objective, x0, jac=objective_grad,
                method='SLSQP', bounds=bounds,
                constraints=[make_ineq(active_idx)],
                options={'maxiter': maxiter, 'ftol': 1e-9}
            )
            if res.x is not None and np.all(np.isfinite(res.x[:n])):
                h_new = project_to_target(res.x[:n], n)
                s = official_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_score

        if score < best_score:

            break
        direction /= norm

        improved = False

            h_try = project_to_target(h - alpha * direction, n)

def gen_interpolated(parent, n, shift=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + shift) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    val = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = val[np.argsort(order)]
    return project_to_target(h, n)

def gen_random(n, rng, mode='uniform'):
    if mode == 'uniform':
        h = rng.uniform(0.1, 0.9, n)
    elif mode == 'binary':
        target = n // 2
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:target]] = 1.0
        if n % 2 == 1:
            h[idx[target]] = 0.5
    elif mode == 'step':
        n_steps = rng.integers(3, n//3 + 1)
        boundaries = sorted(rng.choice(n-1, size=n_steps-1, replace=False))
        boundaries = [0] + list(boundaries) + [n]
        vals = rng.uniform(0, 1, n_steps)
        h = np.zeros(n)
        for i in range(n_steps):
            h[boundaries[i]:boundaries[i+1]] = vals[i]
    else:
        t = np.linspace(0, 1, n)
        freq = rng.uniform(1, 6)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.4 * np.sin(2*np.pi*freq*t + phase) + rng.normal(0, 0.1, n)
    return project_to_target(h, n)

def coordinate_descent(h, n, max_passes=10, time_limit=20.0, seed=42):
    """Optimize each coordinate in turn."""
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    start = time.perf_counter()
    dx = 2.0 / n
    target = n / 2.0
    
    for p in range(max_passes):
        if time.perf_counter() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            if time.perf_counter() - start > time_limit:
                break
            # Optimize h[i] given others fixed
            # Need to adjust for mass: if we change h[i], mass changes
            # Simple approach: try several values and pick best
            current = h[i]
            best_val = current
            best_s = score
            # Try a range of values
            for candidate in [0.0, 0.25, 0.5, 0.75, 1.0, current + 0.1, current - 0.1,
                             current + 0.05, current - 0.05]:
                candidate = max(0.0, min(1.0, candidate))
                # Adjust another index to maintain mass
                # Find index with most room/opposite
                h_try = h.copy()
                h_try[i] = candidate
                mass_diff = h_try.sum() - target
                if abs(mass_diff) > 1e-10:
                    # Find best place to put the adjustment
                    if mass_diff > 0:
                        # Need to reduce elsewhere
                        candidates_j = np.where(h_try > 0.05)[0]
                        if len(candidates_j) == 0:
                            continue
                        j = candidates_j[np.argmax(h_try[candidates_j])]
                        adj = min(mass_diff, h_try[j])
                        h_try[j] -= adj
                    else:
                        candidates_j = np.where(h_try < 0.95)[0]
                        if len(candidates_j) == 0:
                            continue
                        j = candidates_j[np.argmax(1.0 - h_try[candidates_j])]
                        adj = min(-mass_diff, 1.0 - h_try[j])
                        h_try[j] += adj
                
                if abs(h_try.sum() - target) > 1e-6:
                    h_try = project_to_target(h_try, n)
                
                s = official_score(h_try)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = candidate
                    best_h_try = h_try.copy()
            
            if best_val != current:
                h = best_h_try
                score = best_s
    
    return h, score

    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    h = np.clip(np.asarray(v, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    # Bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        r = target - h.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = np.argmax(h)
            h[idx] -= min(-r, h[idx])
    return np.clip(h, 0.0, 1.0)

    ok2 = (p + k >= 0) & (p + k < n)
    g[ok2] += (-h[p[ok2] + k])
    return g * dx
