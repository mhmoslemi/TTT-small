def project_onto_simplex_bounds(z, target_sum):

    n = len(z)
    z = np.asarray(z, dtype=float).copy()

            lo = mid
        else:

def slp_iteration(h0, target, n):
    """One SLP iteration"""
    from scipy.optimize import linprog
    dx = 2.0 / n
    
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        lag = k - (n-1)
        a = np.zeros(n)  # coefficient of h in linear part
        c = 0.0  # constant part
        
        if lag >= 0:
            for i in range(n - lag):
                a[i+lag] += (1.0 - h0[i])
                a[i] += -h0[i+lag]
                c += h0[i+lag] * h0[i]
        else:
            p = -lag
            for i in range(p, n):
                a[i] += (1.0 - h0[i-p])
                a[i-p] += -h0[i]
                c += h0[i] * h0[i-p]
        
        A_ub[k, :n] = dx * a
        A_ub[k, n] = -1.0
        b_ub[k] = -dx * c
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [target]
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return np.clip(h_new, 0.0, 1.0)
    return h0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, target):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    lo = -10.0 - np.max(z)
    hi = 10.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interp_resize(h_src, n_dst):

    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def slp_solve(h0, target, n):

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        a = np.zeros(n)
        c = 0.0
        if lag >= 0:
            for i in range(n - lag):
                a[i + lag] += (1.0 - h0[i])
                a[i] += -h0[i + lag]
                c += h0[i + lag] * h0[i]
        else:
            p = -lag
            for i in range(p, n):
                a[i] += (1.0 - h0[i - p])
                a[i - p] += -h0[i]
                c += h0[i] * h0[i - p]
        A_ub[k, :n] = dx * a
        A_ub[k, n] = -1.0
        b_ub[k] = -dx * c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def slp_descent(h0, target, n, max_iter=80, deadline=None):
    h_curr = h0.astype(float).copy()
    h_curr = project_feasible(h_curr, target)
    best_h = h_curr.copy()
    best_score = official_score(best_h)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h_lp = slp_solve(h_curr, target, n)
        h_lp = project_feasible(h_lp, target)
        improved = False
        for alpha in [1.0, 0.9, 0.75, 0.6, 0.4, 0.25, 0.1]:
            h_trial = project_feasible(h_curr + alpha * (h_lp - h_curr), target)
            s = official_score(h_trial)
            if s < best_score - 1e-13:
                h_curr = h_trial
                best_h = h_trial.copy()
                best_score = s
                improved = True
                break
        if not improved:
            # Try small random perturbations to escape
            for _ in range(3):
                if deadline is not None and time.time() > deadline:
                    break
                h_rand = project_feasible(h_curr + 0.005 * np.random.standard_normal(n), target)

                if s_rand < best_score - 1e-13:
                    h_curr = h_rand
                    best_h = h_rand.copy()
                    best_score = s_rand
                    improved = True
            if not improved:
                break
    return best_h, best_score

def smooth_obj(x, target, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_grad(x, target, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i + lag] += w * (1.0 - x[i])
                g[i] += w * (-x[i + lag])
        else:
            p = -lag
            for i in range(p, n):
                g[i] += w * (1.0 - x[i - p])
                g[i - p] += w * (-x[i])
    return g * dx

def slsqp_polish(h0, target, beta=500, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h0, args=(target, beta),
                       jac=lambda x: smooth_grad(x, target, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return project_feasible(res['x'], target)
    except Exception:
        pass
    return h0

                a[i + lag] += (1.0 - h0[i])
                a[i] += -h0[i + lag]

            p = -lag
            for i in range(p, n):
                a[i] += (1.0 - h0[i - p])
                a[i - p] += -h0[i]
                c += h0[i] * h0[i - p]
        A_ub[row_idx, :n] = dx * a
        A_ub[row_idx, n] = -1.0
        b_ub[row_idx] = -dx * c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(20.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(48) * 0.5

    parent = project_feasible(parent, len(parent) / 2.0)
    parent_n = len(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)

    rng = np.random.default_rng(seed)

    # Build list of n values to try (even preferred for clean target)
    n_candidates = set()
    for delta in range(-8, 9, 2):
        n = parent_n + delta
        if 24 <= n <= 80:
            n_candidates.add(n)
    # Add some specific values
    for n in [30, 36, 40, 44, 48, 52, 54, 56, 60, 64]:
        if 24 <= n <= 80:
            n_candidates.add(n)
    n_list = sorted(n_candidates)
    # Sort by proximity to parent_n first
    n_list = sorted(n_list, key=lambda x: abs(x - parent_n))

    for n in n_list:
        if time.time() > deadline - 80:
            break
        target = n / 2.0

        # Generate starting points
        starts = []
        h_base = project_feasible(interp_resize(parent, n), target)
        starts.append(h_base)
        starts.append(project_feasible(interp_resize(parent[::-1].copy(), n), target))
        starts.append(project_feasible(1.0 - h_base, target))
        # Smoothed versions
        kernel = np.array([0.25, 0.5, 0.25])
        h_sm = np.convolve(h_base, kernel, mode='same')
        starts.append(project_feasible(h_sm, target))
        # Noise
        for sig in [0.03, 0.08, 0.15]:
            starts.append(project_feasible(h_base + sig * rng.standard_normal(n), target))
        # Binary push
        starts.append(project_feasible(h_base ** 1.3, target))
        starts.append(project_feasible(h_base ** 2.0, target))
        starts.append(project_feasible(h_base ** 3.0, target))

        for h0 in starts:
            if time.time() > deadline - 40:
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)

            # SLP phase 1 (full constraints)
            if time.time() < deadline - 30:
                h1, s1 = slp_descent(cur_h, target, n, max_iter=40, deadline=deadline)
                if s1 < cur_score:
                    cur_h, cur_score = h1, s1

            # SLSQP polish
            if time.time() < deadline - 25:
                b = 100
                hp = slsqp_polish(cur_h, target, beta=b, maxiter=150)
                sp = official_score(hp)
                if sp < cur_score - 1e-13:
                    cur_h, cur_score = hp, sp

            # SLP phase 2 (active constraints)
            if time.time() < deadline - 20:
                h2, s2 = slp_descent(cur_h, target, n, max_iter=50, deadline=deadline)
                if s2 < cur_score:
                    cur_h, cur_score = h2, s2

            # More SLSQP with higher beta
            if time.time() < deadline - 15:
                hp2 = slsqp_polish(cur_h, target, beta=2000, maxiter=200)
                sp2 = official_score(hp2)
                if sp2 < cur_score - 1e-13:
                    cur_h, cur_score = hp2, sp2

            # Final SLP + SLSQP
            if time.time() < deadline - 10:
                h3, s3 = slp_descent(cur_h, target, n, max_iter=30, deadline=deadline)
                if s3 < cur_score:
                    cur_h, cur_score = h3, s3
                if time.time() < deadline - 8:
                    hp3 = slsqp_polish(cur_h, target, beta=5000, maxiter=150)
                    sp3 = official_score(hp3)
                    if sp3 < cur_score - 1e-13:
                        cur_h, cur_score = hp3, sp3

            if cur_score < best_score - 1e-13:
                best_h = cur_h.copy()
                best_score = cur_score

    # Intensive refinement around best
    remaining = deadline - time.time()
    if remaining > 25:
        bn = len(best_h)
        bt = bn / 2.0
        for trial in range(min(40, int(remaining / 10))):
            if time.time() > deadline - 5:
                break
            h_p = project_feasible(best_h + 0.01 * rng.standard_normal(bn), bt)
            # Quick SLP
            h_s, s_s = slp_descent(h_p, bt, bn, max_iter=25, deadline=deadline)
            if s_s < best_score - 1e-13:
                best_h, best_score = h_s, s_s
            # SLSQP
            if time.time() < deadline - 5:
                h_sl = slsqp_polish(best_h, bt, beta=3000, maxiter=100)
                s_sl = official_score(h_sl)
                if s_sl < best_score - 1e-13:
                    best_h, best_score = h_sl, s_sl

    # Final polish
    if time.time() < deadline - 3:
        fn = len(best_h)
        ft = fn / 2.0
        h_fin = slsqp_polish(best_h, ft, beta=10000, maxiter=200)
        s_fin = official_score(h_fin)
        if s_fin < best_score:
            best_h, best_score = h_fin, s_fin
        # One more SLP
        if time.time() < deadline - 2:
            h_f2, s_f2 = slp_descent(best_h, ft, fn, max_iter=20, deadline=deadline)
            if s_f2 < best_score:
                best_h, best_score = h_f2, s_f2

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
