def projected_subgradient(h0, f, grad_f, project, step_size_fn, n_iter, restart_every=50):
    h = h0.copy()
    h_avg = h0.copy()
    y = h0.copy()  # Nesterov momentum point
    best_h = h0.copy()
    best_val = f(h0)
    momentum = 0.9
    
    for i in range(n_iter):
        # Gradient at y (Nesterov)
        g = grad_f(y)
        
        # Step size (Polyak or decreasing)
        step = step_size_fn(i)
        
        # Update
        h_new = y - step * g
        h_new = project(h_new)
        
        # Nesterov update
        t_i = (i+1) / (i+0.5)  # momentum parameter
        y = h_new + (t_i - 1) / t_i * (h_new - h)
        
        h = h_new
        h_avg = h_avg + (h - h_avg) / (i+1)  # running average
        
        # Track best
        val = f(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        # Also check average
        val_avg = f(h_avg)
        if val_avg < best_val:
            best_val = val_avg
            best_h = h_avg.copy()
            
    return best_h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)

Phase 3: Coordinate descent

The coordinate descent is particularly interesting because it's exact: for each coordinate, we minimize a 1D function. The function C(h) as a function of h[i] (with all other h[j] fixed) is the maximum of a set of linear functions (since each c[k] is linear in h[i] when others are fixed). So we're minimizing the maximum of linear functions, which is a 1D LP - we can solve it exactly!

Wait, is that right? c[k] = sum_t h[t]*(1-h[t+s]). If we fix all h except h[i], then:
c[k] = (1-h[i+s])*h[i] + (-h[i-s])*h[i] + [other terms not involving h[i]]
     = [(1-h[i+s])*1[i+s valid] + (-h[i-s])*1[i-s valid]] * h[i] + constant_k

So c[k] is linear in h[i] for each k. And C = max_k c[k]*dx is the maximum of linear functions = a convex piecewise linear function of h[i]. We minimize this over h[i] ∈ [0,1] with the constraint that sum(h) = n/2 (which means h[i] = n/2 - sum_{j≠i} h[j]).

Wait, if we fix all other h[j], then h[i] is determined by the constraint! h[i] = n/2 - sum_{j≠i} h[j]. So there's no freedom to optimize!

Unless we relax the constraint temporarily... or optimize in a different way.

Actually, coordinate descent with the equality constraint sum(h) = n/2 is tricky. If we change h[i], we must compensate by changing some other h[j]. A "2-coordinate" descent would change h[i] and h[j] simultaneously while maintaining the constraint.

For a 2-coordinate update (h[i], h[j]), with delta = new_h[i] - old_h[i] = -(new_h[j] - old_h[j]):
c[k](delta) = c[k](0) + delta * (∂c[k]/∂h[i] - ∂c[k]/∂h[j])

So C(delta) = max_k [c[k](0) + delta * a_k] * dx where a_k = ∂c[k]/∂h[i] - ∂c[k]/∂h[j]

This is again a 1D convex piecewise linear function! We can find the minimum by checking the breakpoints.

The breakpoints are at delta values where two of the linear functions c[k](delta) cross. There are O(n^2) breakpoints, but for small n this is fine. Or we can just use a ternary search / golden section on the 1D function.

Actually, even simpler: since C(delta) is convex piecewise linear, the minimum is at a breakpoint or at a boundary. We can find it by:
1. Computing all c[k] and a_k values
2. The minimum of max_k (c[k] + a_k * delta) over delta ∈ [-min(h[i],1-h[j]), min(1-h[i], h[j])]
3. This can be solved by checking O(n) critical points

But for simplicity, let me just use a fine grid or golden section search on the 1D function.

This 2-coordinate descent is guaranteed to not increase the objective (if we find the exact minimum) and can escape some local minima of single-coordinate methods.

        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h = np.interp(x_dst, x_src, h_src)
    return h

def compute_lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w_k = w[k] * dx
        if w_k < 1e-15:
            continue
        s = k - (n - 1)
        i_lo1 = max(0, -s)
        i_hi1 = min(n - 1, n - 1 - s)
        for i in range(i_lo1, i_hi1 + 1):
            grad[i] += w_k * (1.0 - h[i + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n - 1, n - 1 + s)
        for i in range(i_lo2, i_hi2 + 1):
            grad[i] += w_k * (-h[i - s])
    return grad, max_c

def lse_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.exp(shifted).sum()
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def lse_slsqp(h0, beta, maxiter, deadline):
    n = len(h0)

    def obj(x):
        return lse_objective(x, beta)

    def grad(x):
        g, _ = compute_lse_grad(x, beta)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj, h0.copy(), jac=grad, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def nesterov_proj_grad(h0, n_iter, beta_start, beta_end, lr, deadline):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_val = official_score(h0)

    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad_y, _ = compute_lse_grad(y, beta)
        h_new = y - lr * grad_y
        h_new = make_feasible(h_new)
        if it == 0:
            theta = 0.0
        else:
            t_k = 2.0 / (it + 2)
            theta = (t_k - 1.0) / t_k
        y = h_new + theta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        h_avg = h_avg + (h - h_avg) / (it + 1)
        if it % 20 == 0:
            val = official_score(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            h_avg_p = make_feasible(h_avg)
            val_avg = official_score(h_avg_p)
            if val_avg < best_val:
                best_val = val_avg
                best_h = h_avg_p.copy()
    val = official_score(h)
    if val < best_val:
        best_h = h.copy()
    return best_h

def two_coord_descent(h0, max_rounds, deadline, rng):
    n = len(h0)
    h = h0.copy()
    dx = 2.0 / n
    current_score = official_score(h)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = np.max(corr)
        improved = False
        n_pairs = min(n, 40)
        idx_pool = rng.choice(n, size=2 * n_pairs, replace=False)

        for p in range(n_pairs):
            i = int(idx_pool[2 * p])
            j = int(idx_pool[2 * p + 1])
            if i == j:
                continue

            a = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s = k - (n - 1)
                dk_di = 0.0
                if 0 <= i + s < n:
                    dk_di += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    dk_di += (-h[i - s])
                dk_dj = 0.0
                if 0 <= j + s < n:
                    dk_dj += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    dk_dj += (-h[j - s])
                a[k] = dk_di - dk_dj

            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-10:
                continue

            best_d = 0.0
            best_c = cur_max
            grid_coarse = 25
            deltas = np.linspace(d_lo, d_hi, grid_coarse)
            for d in deltas:
                c = np.max(corr + d * a)
                if c < best_c:
                    best_c = c
                    best_d = d

            if best_d != 0.0:
                span = (d_hi - d_lo) / grid_coarse
                g_lo = max(d_lo, best_d - span)
                g_hi = min(d_hi, best_d + span)
                deltas_f = np.linspace(g_lo, g_hi, 25)
                for d in deltas_f:
                    c = np.max(corr + d * a)
                    if c < best_c:
                        best_c = c
                        best_d = d

            if best_d != 0.0 and best_c * dx < current_score - 1e-12:
                h[i] += best_d
                h[j] -= best_d
                h = make_feasible(h)
                current_score = official_score(h)
                improved = True
        if not improved:
            break
    return h

def simulated_annealing(h0, n_iter, t_start, t_end, step_scale, deadline, rng):
    n = len(h0)
    h = h0.copy()
    cur_score = official_score(h)
    best_h = h0.copy()
    best_score = cur_score
    dx = 2.0 / n

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter - 1))
        d_i = rng.standard_normal(n) * step_scale
        h_pert = make_feasible(h + d_i)
        new_score = official_score(h_pert)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_pert
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(12.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = make_feasible(parent)
    except Exception:
        parent = np.ones(44) * 0.5
        parent = make_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    parent_n = len(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Multi-size SLSQP with diverse starts
    size_options = sorted(set([
        max(26, parent_n - 10), max(26, parent_n - 6),
        max(26, parent_n - 4), max(26, parent_n - 2),
        parent_n, parent_n + 2, parent_n + 4, parent_n + 6,
    ]))
    size_options = [n for n in size_options if 26 <= n <= 68]

    for n in size_options:
        if time.time() > deadline - 80:
            break
        h_interp = make_feasible(interp_to_n(parent, n))
        h_anti = make_feasible(1.0 - h_interp[::-1].copy())

        starts = [h_interp, h_anti]
        for sigma in [0.04, 0.08, 0.14, 0.22]:
            starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
        for p in [1.2, 1.5, 2.0, 3.0]:
            starts.append(make_feasible(h_interp ** p))
        # Block-like starts
        for blk in [2, 3, 5]:
            h_blk = np.zeros(n)
            n_ones = n // 2
            pattern = np.zeros(n)
            for i in range(n_ones):
                pattern[i * n // n_ones if n_ones > 0 else 0] = 1.0
            h_blk = make_feasible(0.3 + 0.7 * pattern + 0.05 * rng.standard_normal(n))
            starts.append(h_blk)

        for h0 in starts:
            if time.time() > deadline - 20:
                break
            cur_h = h0.copy()
            cur_score = official_score(cur_h)

            for beta in [30, 100, 400, 1500, 5000]:
                if time.time() > deadline - 10:
                    break
                mi = 200 if n <= 50 else 120
                new_h = lse_slsqp(cur_h, beta, maxiter=mi, deadline=deadline)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-12:
                    cur_h = new_h
                    cur_score = new_score

            if cur_score < best_score - 1e-12:
                best_h = cur_h
                best_score = cur_score

    # Phase 2: Nesterov projected gradient from best
    remaining = deadline - time.time()
    if remaining > 40:
        n_iter = int(min(4000, remaining * 3))
        h_ng = nesterov_proj_grad(
            best_h, n_iter=n_iter,
            beta_start=30, beta_end=5000,
            lr=0.008, deadline=deadline - 15
        )
        ng_score = official_score(h_ng)
        if ng_score < best_score:
            best_h = h_ng
            best_score = ng_score

    # Phase 3: 2-coordinate descent
    remaining = deadline - time.time()
    if remaining > 25:
        max_r = int(min(80, remaining / 3))
        h_cd = two_coord_descent(best_h, max_rounds=max_r,
                                 deadline=deadline - 10, rng=rng)
        cd_score = official_score(h_cd)
        if cd_score < best_score:
            best_h = h_cd
            best_score = cd_score

    # Phase 4: Simulated annealing for escape
    remaining = deadline - time.time()
    if remaining > 50:
        n_sa = int(min(2000, remaining * 2))
        h_sa, sa_score = simulated_annealing(
            best_h, n_sa, t_start=0.003, t_end=0.0001,
            step_scale=0.05, deadline=deadline - 15, rng=rng
        )
        if sa_score < best_score:
            best_h = h_sa
            best_score = sa_score
        # Polish SA result
        if time.time() < deadline - 10:
            h_pol = lse_slsqp(best_h, 2000, maxiter=150, deadline=deadline)
            ps = official_score(h_pol)
            if ps < best_score:
                best_h = h_pol
                best_score = ps

    # Phase 5: Final polish rounds
    if time.time() < deadline - 8:
        for beta in [3000, 8000, 20000]:
            if time.time() > deadline - 5:
                break
            fh = lse_slsqp(best_h, beta, maxiter=200, deadline=deadline)
            fs = official_score(fh)
            if fs < best_score:
                best_h = fh
                best_score = fs

    if time.time() < deadline - 5:
        h_cd2 = two_coord_descent(best_h, max_rounds=40,
                                  deadline=deadline - 2, rng=rng)
        cd2_s = official_score(h_cd2)
        if cd2_s < best_score:
            best_h = h_cd2
            best_score = cd2_s

    # Final SLSQP touch
    if time.time() < deadline - 2:
        fh2 = lse_slsqp(best_h, 10000, maxiter=100, deadline=deadline)
        fs2 = official_score(fh2)
        if fs2 < best_score:
            best_h = fh2
            best_score = fs2

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)

    # Safety: if something went wrong, fall back to parent
    if not np.all(np.isfinite(best_h)) or len(best_h) < 4:
        best_h = parent
        final_score = official_score(best_h)

    return (best_h, final_score, len(best_h))
