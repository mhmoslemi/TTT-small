            min(90, n_base + 8),
            min(80, n_base + 12),
        ]))
        for n_opt in n_options:
            if time.time() >= deadline:
                break
            h_int = interpolate_to(best_h, n_opt)
            try_update(h_int)
            if time.time() < deadline:
                cur2 = h_int.copy()
                phase_dl = min(deadline, time.time() + 15.0)
                for _ in range(5):
                    if time.time() >= phase_dl:
                        break
                    new_s, n_i = pair_pass(cur2, rng, 15, 11, phase_dl)
                    if n_i == 0:
                        break
                    try_update(cur2)
                try_update(cur2)

    # Phase 4: Final intensive polishing on best solution
    if time.time() < deadline:
        cur_h = best_h.copy()
        # Increasingly fine grids
        grids = [(15, 11), (21, 15), (25, 19)]
        for (nc, nf) in grids:
            if time.time() >= deadline:
                break
            for _ in range(3):
                if time.time() >= deadline:
                    break
                new_s, n_i = pair_pass(cur_h, rng, nc, nf, deadline)
                try_update(cur_h)
                if n_i == 0:
                    break

    # Phase 5: Single-site refinement (adjust one site, spread to all others)
    if time.time() < deadline:
        cur_h = best_h.copy()
        n = len(cur_h)
        for target_idx in range(n):
            if time.time() >= deadline:
                break
            cur_s = score(cur_h)
            # Spread delta equally to all other sites
            for d in np.linspace(-0.1, 0.1, 21):
                new_val = cur_h[target_idx] + d
                if new_val < 0 or new_val > 1:
                    continue
                spread = -d / (n - 1)
                h_try = cur_h.copy()
                h_try[target_idx] = new_val
                h_try[np.arange(n) != target_idx] += spread
                if np.any(h_try < -1e-10) or np.any(h_try > 1 + 1e-10):
                    continue
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = repair(h_try)
                s_try = score(h_try)
                if s_try < cur_s - 1e-13:
                    cur_h = h_try
                    cur_s = s_try
                    try_update(cur_h)
        try_update(cur_h)

    # Phase 6: Final pass with very fine grid
    if time.time() < deadline:
        cur_h = best_h.copy()
        final_dl = min(deadline, time.time() + max(5, 0.03 * (deadline - t_start)))
        pair_pass(cur_h, rng, 25, 19, final_dl)
        try_update(cur_h)
        if time.time() < deadline:
            pair_pass(cur_h, rng, 31, 21, deadline)
            try_update(cur_h)

    # Final validation
    final_h = repair(best_h)
    final_score = score(final_h)
    final_n = len(final_h)

    # Ensure we return a valid bound
    final_bound = final_score + 1e-13

    return final_h, final_bound, final_n
