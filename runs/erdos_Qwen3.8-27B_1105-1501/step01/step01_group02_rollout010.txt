def pair_exchange_line_search(h, i, j, n_evals=20):
    """Try exchanging mass between positions i and j.
    h[i] += t, h[j] -= t for t in [lo, hi].
    Returns (best_h, best_c5, best_t) or (h, current_c5, 0) if no improvement.
    """
    n = len(h)
    hi = min(1.0 - h[i], h[j])
    lo = max(-h[i], h[j] - 1.0)

        return h, compute_c5(h), 0.0

    best_c5 = cur_c5
    best_t = 0.0
    orig_i = h[i]
    orig_j = h[j]

    t_vals = np.linspace(lo, hi, n_evals)
    for t in t_vals:
        h[i] = orig_i + t
        h[j] = orig_j - t
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # If best is at boundary, try to refine
    if abs(best_t - lo) < (hi-lo)/n_evals or abs(best_t - hi) < (hi-lo)/n_evals:

        pass
    elif best_t != 0.0:

        margin = (hi - lo) / n_evals
        a = max(lo, best_t - margin)
        b = min(hi, best_t + margin)
        for _ in range(10):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def corr_vector(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_line_search(h, i, j, n_evals=16):
    """Exchange mass between i and j, preserving sum. Find best t."""
    cur_c5 = compute_c5(h)
    hi_t = min(1.0 - h[i], h[j])
    lo_t = max(-h[i], h[j] - 1.0)
    if hi_t - lo_t < 1e-12:
        return h, cur_c5
    
    orig_i, orig_j = h[i], h[j]
    best_c5 = cur_c5
    best_t = 0.0
    
    # Coarse search
    for t in np.linspace(lo_t, hi_t, n_evals):
        h[i] = orig_i + t
        h[j] = orig_j - t
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # Fine search around best
    if best_t != 0.0:
        margin = (hi_t - lo_t) / n_evals
        a = max(lo_t, best_t - margin)
        b = min(hi_t, best_t + margin)
        for _ in range(15):
            m1 = a + 0.381966 * (b - a)
            m2 = b - 0.381966 * (b - a)
            h[i] = orig_i + m1
            h[j] = orig_j - m1
            c1 = compute_c5(h)
            h[i] = orig_i + m2
