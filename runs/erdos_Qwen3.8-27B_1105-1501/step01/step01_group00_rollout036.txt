def project_feasible(z, target_sum):
    """Project z onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(z)
    h = np.clip(z, 0.0, 1.0)
    # Bisection for the right offset
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

            lo = mid
        else:

    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)

    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:

            for idx in order:

                residual -= add

                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break

def official_score(h):

def compute_overlaps(h):

def project_feasible(z, target=None):
    n = len(z)
    if target is None:
        target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15: break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15: break
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, max_sweeps=2000, tol=1e-14):
    """Gauss-Seidel coordinate descent with exact 1D minimization."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    overlaps = compute_overlaps(h)
    best_score = official_score(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Compute A and B for this coordinate
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            for p in range(2 * n - 1):
                s = p - (n - 1)
                # A_s = dx * ((1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n])
                a_val = 0.0
                js = i - s
                if 0 <= js < n:
                    a_val += (1.0 - h[js])
                js2 = i + s
                if 0 <= js2 < n:
                    a_val -= h[js2]
                A[p] = dx * a_val
                B[p] = overlaps[p] - A[p] * old_val
            
            # Find optimal x
            new_val = find_optimal_x(A, B)
            
            if abs(new_val - old_val) > 1e-14:
                # Update overlaps incrementally
                delta = new_val - old_val
                for p in range(2 * n - 1):
                    overlaps[p] += delta * A[p]
                h[i] = new_val
                improved = True
        
        if not improved:
            break
        
        sc = float(overlaps.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_h, best_score

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()  # numerical stability
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient: grad[i] = dx * sum_p w[p] * ((1-h[i-s])*[valid] - h[i+s]*[valid])
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        s = p - (n - 1)
        for i in range(n):
            g = 0.0
            js = i - s
            if 0 <= js < n:
                g += (1.0 - x[js])
            js2 = i + s
            if 0 <= js2 < n:
                g -= x[js2]
            grad[i] += w[p] * g
    return dx * grad

def make_periodic(h, p):

    n = len(h)

    hp = np.zeros(p)
    counts = np.zeros(p)
    for i in range(n):
        hp[i % p] += h[i]
        counts[i % p] += 1

    # Expand back
