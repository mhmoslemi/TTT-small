import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_row(h, s):
    """Gradient of scaled correlation at lag s with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        # h[i] appears as first factor: i in [s, n-1], contributes (1-h[i-s])
        for i in range(s, n):
            g[i] += dx * (1.0 - h[i - s])
        # h[i] appears in (1-h[i]) as second factor: i in [0, n-1-s], contributes -h[i+s]
        for i in range(0, n - s):
            g[i] += dx * (-h[i + s])
    else:
        a = -s
        # h[i] appears as first factor: i in [0, n-a-1], contributes (1-h[i+a])
        for i in range(0, n - a):
            g[i] += dx * (1.0 - h[i + a])
        # h[i] appears in (1-h[i]) as second factor: i in [a, n-1], contributes -h[i-a]
        for i in range(a, n):
            g[i] += dx * (-h[i - a])
    return g

def grad_matrix(h):

    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        G[r] = grad_row(h, s)
    return G

        else:

        return project_to_target(np.full(n_target, 0.5), n_target)

def discrete_sa(h_init, n, time_limit=30.0, seed=42):
    """Simulated annealing on discrete (binary) sequences."""
    rng = np.random.default_rng(seed)
    # Start from binary approximation
    h = np.zeros(n)
    # Sort and set top n/2 to 1
    sorted_idx = np.argsort(h_init)[::-1]
    h[sorted_idx[:n//2]] = 1.0
    # For odd n, one value is 0.5

        h[sorted_idx[n//2]] = 0.5

    current_score = official_score(h)
    best_h = h.copy()
    best_score = current_score
    
    T = 0.1
    T_min = 1e-6
    cooling = 0.98
    start_time = time.perf_counter()
    
    while time.perf_counter() - start_time < time_limit and T > T_min:
        # Random swap: pick two positions, one with 1 and one with 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = official_score(h_new)
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        
        T *= cooling
    
    return best_h, best_score

def slsqp_polish(h, n, time_limit=20.0):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    initial_score = official_score(h)
    
    def objective(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def corr_func(h_x):
        return np.correlate(h_x, 1.0 - h_x, mode="full") * dx
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        c = corr_func(h_x)
        return theta - c
    
    def ineq_jac(x):
        h_x = x[:n]
        J = np.zeros((L, n + 1))
        G = grad_matrix(h_x)
        # ineq_fun[r] = theta - c_r
        # d(ineq)/dh_i = -G[r, i]
        # d(ineq)/dtheta = 1
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-5
    x0 = np.concatenate([h, [theta0]])
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(objective, x0, jac=obj_grad,
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 100, 'ftol': 1e-10})
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def official_score(h):

    if n == 0:
        return 0.0

    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def grad_row_vec(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[s:] += dx * (1.0 - h[:n - s])
        g[:n - s] += -dx * h[s:]
    else:
        a = -s
        g[:n - a] += dx * (1.0 - h[a:])
        g[a:] += -dx * h[:n - a]
    return g


def grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        G[r] = grad_row_vec(h, s)
    return G

def slsqp_reduce(h, n, time_limit=20.0, top_m=15):
    dx = 2.0 / n
    target = n / 2.0
    L = 2 * n - 1
    initial_score = official_score(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_m:]
    G = grad_matrix(h)

    def objective(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = project_to_target(x[:n], n)
        c = compute_corr(h_x)
        return x[n] - c[top_idx]

    def ineq_jac(x):
        h_x = project_to_target(x[:n], n)
        Gx = grad_matrix(h_x)
        J = np.zeros((top_m, n + 1))
        J[:, :n] = -Gx[top_idx]
        J[:, n] = 1.0
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [min(theta0, 1.0)]])

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    start = time.perf_counter()
    try:
        result = optimize.minimize(objective, x0, jac=obj_grad,
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,

    # Sort by score
    pool.sort(key=lambda x: (x[0], len(x[1])))
    if len(pool) > 0 and pool[0][0] < best_score:
        best_score = pool[0][0]
        best_h = pool[0][1].copy()
        best_n = len(best_h)

    # Phase 1: LP local search on top candidates
    top_k = min(10, len(pool))
    local_results = []
    for idx in range(top_k):
        if time.perf_counter() > deadline - 150:
            break
        h_c = pool[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_lim = min(30.0, max(4.0, remaining / max(1, top_k - idx) * 0.55))
        h_loc, s_loc = local_search_lp(h_c, n_c, max_iters=50, time_limit=t_lim, seed=seed + idx * 17)
        local_results.append((s_loc, h_loc.copy()))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

    local_results.sort(key=lambda x: x[0])

    # Phase 2: SLSQP polish
    to_refine = local_results[:min(6, len(local_results))]
    found_best = False
    for s_r, h_r in to_refine:
        if np.allclose(h_r, best_h, atol=1e-6):
            found_best = True
            break
    if not found_best and best_score < 1.0:
        to_refine.append((best_score, best_h.copy()))

    for idx, (s_c, h_c) in enumerate(to_refine):
        if time.perf_counter() > deadline - 100:
            break
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_lim = min(25.0, max(5.0, remaining / max(1, len(to_refine) - idx) * 0.5))
        h_ref, s_ref = slsqp_reduce(h_c, n_c, time_limit=t_lim, top_m=min(15, 2 * n_c - 1))
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = n_c

        # Follow-up local search after SLSQP
        if time.perf_counter() < deadline - 40:
            h_2, s_2 = local_search_lp(h_ref, n_c, max_iters=30, time_limit=10.0, seed=seed + 400 + idx)
            if s_2 < best_score:
                best_score = s_2
                best_h = h_2.copy()
                best_n = n_c

    # Phase 3: Discrete SA
    if time.perf_counter() < deadline - 80:
        even_ns = sorted(set([
            max(32, best_n - 6) if (best_n - 6) % 2 == 0 else max(32, best_n - 5),
            best_n if best_n % 2 == 0 else best_n + 1,
            best_n + 2 if (best_n + 2) % 2 == 0 else best_n + 3,
            best_n + 4 if (best_n + 4) % 2 == 0 else best_n + 5,
        ]))
        for n_d in even_ns:
            if time.perf_counter() > deadline - 40:
                break
            if n_d < 32 or n_d > 128:
                continue
            h_init = interpolate_samples(best_h, n_d)
            h_disc, s_disc = discrete_sa(h_init, n_d, time_limit=12.0, seed=seed + 600)
            if s_disc < best_score:
                best_score = s_disc
                best_h = h_disc.copy()
                best_n = n_d
                # Polish
                if time.perf_counter() < deadline - 20:
                    h_p, s_p = slsqp_reduce(h_disc, n_d, time_limit=10.0, top_m=12)
                    if s_p < best_score:
                        best_score = s_p
                        best_h = h_p.copy()
                        best_n = n_d

    # Phase 4: Binary push + local search on best
    if time.perf_counter() < deadline - 30:
        h_bp, s_bp = binary_push(best_h, best_n, time_limit=5.0, seed=seed + 800)
        if s_bp < best_score:
            best_score = s_bp
            best_h = h_bp.copy()
        if time.perf_counter() < deadline - 15:
            h_loc3, s_loc3 = local_search_lp(best_h, best_n, max_iters=30, time_limit=10.0, seed=seed + 900)
            if s_loc3 < best_score:
                best_score = s_loc3
                best_h = h_loc3.copy()
                best_n = len(h_loc3)

    # Final SLSQP
    if time.perf_counter() < deadline - 8:
        h_fin, s_fin = slsqp_reduce(best_h, best_n, time_limit=min(8.0, max(3.0, deadline - time.perf_counter() - 4)), top_m=12)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)
    c5_bound = float(max(0.0, final_score))

    return (best_h, c5_bound, int(len(best_h)))
