import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        order = np.argsort(-np.abs(h - 0.5))
        for i in order:
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def compute_lag_grad(h, s, n):
    grad = np.zeros(n)
    start = max(0, -s)
    end = min(n, n - s)
    if start < end:
        grad[start:end] += (1.0 - h[start + s:end + s])
    p_start = max(0, start + s)
    p_end = min(n, end + s)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - s:p_end - s])
    return grad


def stable_sigmoid(z):
    h = np.empty_like(z)
    pos = z >= 0
    h[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    exp_z = np.exp(z[~pos])
    h[~pos] = exp_z / (1.0 + exp_z)
    return h


def logit_to_h(y, n):
    target = n / 2.0
    lo = -60.0
    hi = 60.0
    while np.sum(stable_sigmoid(y + lo)) < target:
        lo -= 25.0
    while np.sum(stable_sigmoid(y + hi)) > target:
        hi += 25.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(stable_sigmoid(y + mid)) > target:
            lo = mid
        else:
            hi = mid
    return stable_sigmoid(y + (lo + hi) * 0.5)


def to_logit(h, eps=1e-12):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))


def logit_obj_grad(y, beta):
    n = len(y)
    h = logit_to_h(y, n)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    Cmax = np.max(C)
    shifted = beta * (C - Cmax)
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    total = np.sum(exps)
    if total < 1e-300:
        obj = Cmax
        weights = np.zeros(len(C))
        weights[np.argmax(C)] = 1.0
    else:
        obj = Cmax + (ms + np.log(total)) / beta
        weights = exps / total
    g = np.zeros(n)
    for s_idx in range(len(weights)):
        w = weights[s_idx]
        if w < 1e-10:
            continue
        s = s_idx - n + 1

    g *= dx
    w_h = h * (1.0 - h)
    W = np.sum(w_h)

        return obj, np.zeros(n)
    S = np.dot(g, w_h)
    grad_y = w_h * (g - S / W)
    return obj, grad_y

def slsqp_epigraph(h0, maxiter=120):
    n = len(h0)
    dx = 2.0 / n
    score0 = official_score(h0)
    theta0 = score0 + 0.01
    z0 = np.concatenate([h0, [theta0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def obj(z):
        return z[-1]

    def obj_jac(z):
        g = np.zeros(sz)
        g[-1] = 1.0
        return g

    sz = n + 1

    def ineq(z):
        h = z[:n]
        C = np.correlate(h, 1.0 - h, mode='full') * dx
        return z[-1] - C

    def ineq_jac(z):
        h = z[:n]
        J = np.zeros((2 * n - 1, sz))
        J[:, -1] = 1.0
        for k in range(-(n - 1), n):
            J[k + n - 1, :n] = -dx * compute_lag_grad(h, k, n)
        return J

    def eq(z):
        return np.sum(z[:n]) - n / 2.0

    def eq_jac(z):
        g = np.ones(sz)
        g[-1] = 0.0
        return g

    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, z0, jac=obj_jac, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return None

                    best_h = h_new.copy()
                    best_score = sc2

    # Phase 4: Active set descent (multiple configurations)
    if time_ok(120):
        for q in [4, 8, 16]:
            if not time_ok(40):
                break
            h_as, sc_as = active_descent(best_h, max_iter=300, q=q)
            if sc_as < best_score - 1e-14:
                best_h = h_as.copy()
                best_score = sc_as
        # Second pass with smaller q
        if time_ok(60):
            h_as, sc_as = active_descent(best_h, max_iter=400, q=3)
            if sc_as < best_score - 1e-14:
                best_h = h_as.copy()
                best_score = sc_as

    # Phase 5: SLSQP epigraph
    if time_ok(80):
        mi = max(50, min(200, int(15000 / max(len(best_h), 10))))
        h_new = slsqp_epigraph(best_h, maxiter=mi)
        if h_new is not None and np.all(np.isfinite(h_new)):
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc

    # Phase 6: Resolution transfer
    if time_ok(60):
        base_n = len(best_h)
        for delta in [-6, -3, 3, 6, -12, 12]:
            if not time_ok(15):
                break
            nn = base_n + delta
            if 20 <= nn <= 80:
                h_i = interp_h(best_h, nn)
                for beta in [500, 2000]:
                    if not time_ok(8):
                        break
                    h_o = logit_lbfgs(h_i, beta, maxiter=60)
                    if h_o is not None and np.all(np.isfinite(h_o)):
                        h_i = h_o
                sc = official_score(h_i)
                if sc < best_score - 1e-14 or (abs(sc - best_score) < 1e-13 and nn < len(best_h)):
                    best_h = h_i.copy()
                    best_score = sc

    # Phase 7: Final polishing - alternate active set and logit
    if time_ok(40):
        for _ in range(3):
            if not time_ok(15):
                break
            h_as, _ = active_descent(best_h, max_iter=200, q=6)
            if official_score(h_as) < best_score - 1e-14:
                best_h = h_as.copy()
                best_score = official_score(best_h)
            if not time_ok(10):
                break
            h_new = logit_lbfgs(best_h, 3000, maxiter=60)
            if h_new is not None and np.all(np.isfinite(h_new)):
                sc = official_score(h_new)
                if sc < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = sc

    # Phase 8: Final SLSQP if time remains
    if time_ok(20):
        h_new = slsqp_epigraph(best_h, maxiter=100)
        if h_new is not None and np.all(np.isfinite(h_new)):
            sc = official_score(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new.copy()
                best_score = sc

    # Final validation
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        best_h = parent.copy() if parent is not None else make_feasible(np.full(pn, 0.5))
        final_score = official_score(best_h)
    if final_score > parent_score + 1e-12:
        best_h = parent.copy() if parent is not None else make_feasible(np.full(pn, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
