def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    vals = np.max(c_vals[np.newaxis, :] + D[np.newaxis, :] * ts[:, np.newaxis], axis=1)
    bi = int(np.argmin(vals))
    bv = vals[bi]
    # Refine
    if bi == 0:
        lo_r, hi_r = t_lo, ts[1]
    elif bi == n_grid - 1:
        lo_r, hi_r = ts[-2], t_hi

    vals2 = np.max(c_vals[np.newaxis, :] + D[np.newaxis, :] * ts2[:, np.newaxis], axis=1)
    bi2 = int(np.argmin(vals2))
    return ts2[bi2], vals2[bi2]

def do_pair_sweep(h, dx, n_grid):
    n = len(h)
    A, B = precompute_AB(h)
    c_vals = compute_all_c(h)
    cur_max = np.max(c_vals)
    improved = 0
    
    for i in range(n):
        for j in range(i + 1, n):
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-10:
                continue
            D = dx * (A[i] - B[i] - A[j] + B[j])
            t_best, pred = find_best_t(c_vals, D, t_lo, t_hi, n_grid)
            if pred >= cur_max - 1e-10:
                continue
            # Apply
            h[i] += t_best
            h[j] -= t_best
            c_new = compute_all_c(h)
            nm = np.max(c_new)
            if nm < cur_max - 1e-12:
                c_vals = c_new
                cur_max = nm
                improved += 1
                A, B = precompute_AB(h)
            else:
                h[i] -= t_best
                h[j] += t_best
