```python
import numpy as np
import time

def compute_c5(h):
    """Compute the exact C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    """Compute the full correlation vector (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def project_binary(h, n):
    """Project h to binary: pick the n/2 largest values as 1s."""
    threshold_index = n // 2
    # Sort and find threshold
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:threshold_index]] = 1.0
    if n % 2 == 1:
        result[sorted_idx[threshold_index]] = 0.5  # or handle differently
    return result

def waterfill_project(z, target):
    """Project z to [0,1]^n with sum = target using water-filling."""
    n = len(z)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum exactly
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add diff; find positions with room
            slack = 1.0 - result
            for idx in np.argsort(slack)[::-1]:
                add = min(diff, slack[idx])
                result[idx] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            # Need to subtract |diff|
            for idx in np.argsort(result):
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
                if diff >= -1e-14:
                    break
    return result

def exact_line_search(h, i, j, sign, n_samples=50):
    """
    1D line search: h[i] += sign*delta, h[j] -= sign*delta
    Find the delta in [0, max_d] that minimizes C5.
    Returns (best_delta, best_c5, new_h) or (0, current_c5, h) if no improvement.
    """
    current_c5 = compute_c5(h)
    
    if sign > 0:
        max_d = min(1.0 - h[i], h[j])
    else:
        max_d = min(h[i], 1.0 - h[j])
    
    if max_d < 1e-10:
        return 0.0, current_c5, h.copy()
    
    # Dense sampling
    deltas = np.linspace(0, max_d, n_samples)
    best_delta = 0.0
    best_val = current_c5
    
    h_save = h.copy()
    for d in deltas:
        h[i] += sign * d
        h[j] -= sign * d
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_delta = d
    h[i] = h_save[i]
    h[j] = h_save[j]
    
    # Refine around best_delta
    if best_delta > 0 and best_delta < max_d:
        step = max_d / n_samples
        lo = max(0, best_delta - step)
        hi = min(max_d, best_delta + step)
        for _ in range(20):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            h[i] = h_save[i] + sign * m1
            h[j] = h_save[j] - sign * m1
            v1 = compute_c5(h)
            h[i] = h_save[i] + sign * m2
            h[j] = h_save[j] - sign * m2
            v2 = compute_c5(h)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            if v1 < best_val:
                best_val = v1
                best_delta = m1
            if v2 < best_val:
                best_val = v2
                best_delta = m2
    
    h[i] = h_save[i] + sign * best_delta
    h[j] = h_save[j] - sign * best_delta
```

```python
def resample(h_old, n_new):
    """Resample a step function from n_old to n_new points."""
    n_old = len(h_old)
    # h_old is a step function on [0,2] with n_old cells
    # Create a fine grid and interpolate
    # Cell centers in [0, 2]
    old_centers = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    new_centers = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # Piecewise constant interpolation
    h_new = np.zeros(n_new)
    for i in range(n_new):
        x = new_centers[i]
        # Find which old cell this falls in
        cell = int(x / (2.0 / n_old))
        cell = min(cell, n_old - 1)
        h_new[i] = h_old[cell]
    # Fix the sum
    target = n_new / 2.0
    diff = target - h_new.sum()
    # Distribute the difference
    if abs(diff) > 1e-12:
        # Add/subtract uniformly to non-saturated positions
        if diff > 0:

        else:
            slack = h_new

        if total_slack > 0:

    if abs(diff) > 1e-12:
```
