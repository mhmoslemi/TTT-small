import numpy as np
import time

def project_feasible(y, n=None):
    """L2 projection onto {h: 0<=h<=1, sum(h)=n/2} via binary search on threshold."""
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n * 0.5
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Exact correction
    diff = target - z.sum()
    if abs(diff) > 1e-14:
        # Spread correction among non-bound entries
        free = (z > 1e-12) & (z < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            z[free] += diff / nf
            z = np.clip(z, 0.0, 1.0)
            # If still off (due to clipping), fix at one point
            diff2 = target - z.sum()
            if abs(diff2) > 1e-14:
                if diff2 > 0:
                    idx = int(np.argmin(z))
                    z[idx] = min(1.0, z[idx] + diff2)
                else:
                    idx = int(np.argmax(z))
                    z[idx] = max(0.0, z[idx] + diff2)
        else:
            if diff > 0:
                idx = int(np.argmin(z))
                z[idx] = min(1.0, z[idx] + diff)
            else:
                idx = int(np.argmax(z))
                z[idx] = max(0.0, z[idx] + diff)
    return z

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def scores_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_grad(h, scores, T):
    """Gradient of smooth max of scores w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    base = scores.max()
    ex = np.exp((scores - base) / T)
    S = ex.sum()
    if S < 1e-300:
        w = np.zeros_like(scores)
        mask = scores >= base - 1e-10
        w[mask] = 1.0 / mask.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    g += w[n - 1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[n - 1 + d] * dx
        wn = w[n - 1 - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g, w

def grad_step(h, deadline, T, mu, max_iters):
    """Momentum gradient descent with exact projection."""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        sc = scores_all(cur)
        g, w = smooth_max_grad(cur, sc, T)
        # Project gradient onto sum=0 subspace
        g = g - g.mean()
        # Momentum
        v = mu * v - g
        # Line search
        for alpha in (0.01, 0.005, 0.002, 0.001, 0.0005):
            cand = project_feasible(cur + alpha * v, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        # Adapt temperature
        T *= 0.999
        if T < 0.001:
            T = 0.001
    return best, best_c

        else:
            if diff > 0:
                idx = int(np.argmin(z))
                z[idx] = min(1.0, z[idx] + diff)
            else:
                idx = int(np.argmax(z))
                z[idx] = max(0.0, z[idx] + diff)
    return z

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def scores_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_grad(h, scores, T):
    n = len(h)
    dx = 2.0 / n
    base = scores.max()
    ex = np.exp((scores - base) / T)
    S = ex.sum()
    if S < 1e-300:
        w = np.zeros_like(scores)
        mask = scores >= base - 1e-10
        w[mask] = 1.0 / mask.sum()
    else:
        w = ex / S
    g = np.zeros(n)
    g += w[n - 1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[n - 1 + d] * dx
        wn = w[n - 1 - d] * dx
        g[:m] += wn * (1.0 - h[d:]) - wp * h[d:]
        g[d:] += wp * (1.0 - h[:m]) - wn * h[:m]
    return g, w

def pair_optimize(h, i, j, cur_c, n_pts=30):
    n = len(h)
    oi = h[i]
    oj = h[j]
    lo = max(-oi, oj - 1.0)
    hi = min(1.0 - oi, oj)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = oi + a
        h[j] = oj - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = oi
    h[j] = oj
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = oi + a1
                h[j] = oj - a1
                s1 = score_exact(h)
                h[i] = oi + a2
                h[j] = oj - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = 0.5 * (lo2 + hi2)
            h[i] = oi + best_a
            h[j] = oj - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                h[i] = oi
                h[j] = oj
                return None
    return float(best_s) if (best_s < cur_c - 1e-14 and abs(best_a) > 1e-10) else None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 16:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def grad_phase(h, deadline, T_start, T_end, T_decay, step0, mu, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    v = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        sc = scores_all(cur)
        g, w = smooth_grad(cur, sc, T)
        g = g - g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-16:
            break
        v = mu * v - g / gn
        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.1, step * 0.05):
            cand = project_feasible(cur + s * v, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-13:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.8
            v *= 0.5
            stall += 1
            if stall > 30:
                cur = project_feasible(cur + rng.normal(0, max(T * 0.2, 0.003), n), n)
                cur_c = score_exact(cur)
                step = max(T * 0.5, 0.003)
                v = np.zeros(n)
                stall = 0
        else:
            step = min(step * 1.03, T * 1.5)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def targeted_pairs(h, cur_c, rng, max_tries=40):
    n = len(h)
    sc = scores_all(h)
    base = sc.max()
    active = np.where(sc >= base - 0.01)[0]
    active_lags = [abs(int(m) - (n - 1)) for m in active if abs(int(m) - (n - 1)) >= 1]
    best_c = cur_c
    improved_any = False
    for _ in range(max_tries):
        if not active_lags:
            break
        k = int(rng.choice(active_lags))
        if k < 1 or k >= n:
            continue
        i = int(rng.integers(0, n - k))
        j = i + k
        res = pair_optimize(h, i, j, best_c, n_pts=25)
        if res is not None:
            best_c = res
            improved_any = True
            # Recompute active lags
            sc = scores_all(h)
            base = sc.max()
            active = np.where(sc >= base - 0.01)[0]
            active_lags = [abs(int(m) - (n - 1)) for m in active if abs(int(m) - (n - 1)) >= 1]
    return best_c, improved_any

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        n_init = len(initial_h_values)
        best_h = project_feasible(np.array(initial_h_values, dtype=np.float64), n_init)
    except (NameError, Exception):
        best_h = project_feasible(np.full(70, 0.5), 70)
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Gradient descent from parent (12% time)
    p1_end = t0 + budget_s * 0.12
    if time.time() < p1_end:
        h1, c1 = grad_phase(best_h, p1_end, 0.08, 0.002, 0.9997, 0.02, 0.3, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Main optimization (48% time) - alternating gradient and targeted pairs
    p2_end = t0 + budget_s * 0.60
    cur = best_h.copy()
    cur_c = best_c
    T = 0.015
    step = 0.01
    v = np.zeros(n)
    consecutive_fail = 0
    while time.time() < p2_end:
        # Gradient step
        sc = scores_all(cur)
        g, w = smooth_grad(cur, sc, T)
        g = g - g.mean()
        gn = float(np.linalg.norm(g))
        if gn > 1e-16:
            v = 0.3 * v - g / gn
            improved = False
            for s in (step, step * 0.5, step * 0.25):
                cand = project_feasible(cur + s * v, n)
                c_new = score_exact(cand)
                if c_new < cur_c - 1e-13:
                    cur = cand
                    cur_c = c_new
                    improved = True
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                    break
            if not improved:
                step *= 0.85
                v *= 0.5
        # Targeted pair search
        c_t, imp_t = targeted_pairs(cur, cur_c, rng, max_tries=30)
        if imp_t:
            cur_c = c_t
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            if consecutive_fail > 50:
                cur = project_feasible(best_h + rng.normal(0, 0.008, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0
                step = 0.01
                v = np.zeros(n)
        T = max(0.003, T * 0.9995)

    # Phase 3: Try different resolutions (25% time)
    p3_end = t0 + budget_s * 0.85
    n_cur = len(best_h)
    n_candidates = sorted(set([n_cur - 8, n_cur - 4, n_cur - 2, n_cur + 2, n_cur + 4, n_cur + 8,
                                48, 56, 64, 72, 80, 88]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 120]
    time_per_n = max(5.0, (p3_end - time.time()) / max(len(n_candidates), 1))
    for nr in n_candidates:
        if time.time() > p3_end - 10:
            break
        if nr == n_cur:
            continue
        h_r = resample(best_h, nr)
        if h_r is None:
            continue
        c_r = score_exact(h_r)
        sub_dl = min(p3_end - 5, time.time() + time_per_n)
        if time.time() < sub_dl - 2:
            h_r, c_r = grad_phase(h_r, sub_dl, 0.01, 0.001, 0.999, 0.005, 0.2, rng)
        cur_r = h_r.copy()
        cur_rc = c_r
        # Quick pair search
        sub_dl2 = min(p3_end - 3, time.time() + time_per_n * 0.5)
        for _ in range(100):
            if time.time() > sub_dl2:
                break
            i = int(rng.integers(0, nr))
            j = int(rng.integers(0, nr))
            if i == j:
                continue
            res = pair_optimize(cur_r, i, j, cur_rc, n_pts=25)
            if res is not None:
                cur_rc = res
        if cur_rc < best_c:
            best_h, best_c = cur_r, cur_rc
            n_cur = nr

    # Phase 4: Final polish (15% of remaining time)
    p4_end = deadline
    n = len(best_h)
    if time.time() < p4_end - 5:
        h4, c4 = grad_phase(best_h, p4_end - 30, 0.005, 0.0003, 0.999, 0.003, 0.1, rng)
        if c4 < best_c:
            best_h, best_c = h4, c4
        # Final pair search
        cur = best_h.copy()
        cur_c = best_c
        sc = scores_all(cur)
        base = sc.max()
        active = np.where(sc >= base - 0.005)[0]
        active_lags = [abs(int(m) - (n - 1)) for m in active if abs(int(m) - (n - 1)) >= 1]
        while time.time() < p4_end - 5:
            improved = False
            for k in active_lags[:5]:
                if time.time() > p4_end - 5:
                    break
                if k < 1 or k >= n:
                    continue
                for _ in range(3):
                    i = int(rng.integers(0, n - k))
                    j = i + k
                    res = pair_optimize(cur, i, j, cur_c, n_pts=30)
                    if res is not None:
                        cur_c = res
                        improved = True
                        if cur_c < best_c:
                            best_h = cur.copy()
                            best_c = cur_c
            # Also try adjacent pairs
            for _ in range(5):
                if time.time() > p4_end - 5:
                    break
                i = int(rng.integers(0, n - 1))
                res = pair_optimize(cur, i, i + 1, cur_c, n_pts=30)
                if res is not None:
                    cur_c = res
                    improved = True
                    if cur_c < best_c:
                        best_h = cur.copy()
                        best_c = cur_c
            if not improved:
                # Small random pair
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    res = pair_optimize(cur, i, j, cur_c, n_pts=20)
                    if res is not None:
                        cur_c = res
                        if cur_c < best_c:
                            best_h = cur.copy()
                            best_c = cur_c

    # Ensure feasibility
    best_h = project_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
