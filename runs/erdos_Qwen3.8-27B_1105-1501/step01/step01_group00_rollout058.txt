def upper_envelope(a, b):
    """Compute upper envelope of lines y = a[i]*x + b[i].
    Returns list of (start_x, end_x, a_i, b_i) for each segment."""
    n = len(a)
    # Sort by slope, then by intercept (for ties)
    order = np.argsort(a, kind='stable')
    # For same slope, keep only the one with max intercept
    lines = []
    for idx in order:
        if lines and a[idx] == lines[-1][0]:
            if b[idx] > lines[-1][1]:
                lines[-1] = (a[idx], b[idx])
        else:
            lines.append((a[idx], b[idx]))
    
    # Build upper envelope using stack
    env = [lines[0]]  # list of (a, b, x_start)
    for i in range(1, len(lines)):
        ai, bi = lines[i]
        # Intersection of new line with last line in envelope
        Ax, _ = env[-1]
        # x intersection = (b_prev - b_new) / (a_new - a_prev)
        while len(env) > 0:
            Aa, Ab, Axs = env[-1]
            if ai == Aa:
                # Same slope, new line dominates if higher intercept
                if bi > Ab:
                    env.pop()
                    continue
                else:
                    break
            x_int = (Ab - bi) / (ai - Aa)
            if len(env) < 2 or x_int >= env[-1][2] - 1e-15:
                env.append((ai, bi, x_int))
                break
            else:
                env.pop()
        # If env is empty, add with x_start = -inf
        if not env:
            env.append((ai, bi, -1e18))
    
    # Close the last segment
    # Return segments: for each line, it's active from x_start to next x_start
    segments = []
    for i, (aa, ab, axs) in enumerate(env):
        if i + 1 < len(env):
            x_end = env[i+1][2]
        else:
            x_end = 1e18
        segments.append((axs, x_end, aa, ab))
    return segments

     - The envelope slope at x* is a[k*]
     - If a[k*] < 0, move right: new x* is crossing of k* with the line active to its right
     - If a[k*] > 0, move left
     - Converge in at most M steps

This is O(M^2) worst case but typically O(M) per coordinate.

def coordinate_descent(h, lam=0.0, max_sweeps=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(max_sweeps):
        h_old = h.copy()
        max_improve = 0.0
        
        for i in range(n):
            # Compute a_k and b_k for coordinate i
            a_k = np.zeros(2*n - 1)
            b_k = np.zeros(2*n - 1)
            
            for p in range(2*n - 1):
                s = p - (n - 1)
                if s >= 0:
                    # f_p = sum_{m=0}^{n-1-s} h[m+s] * (1 - h[m])
                    # coeff of h[i]:
                    # h[i] appears as h[m+s] if s <= i <= n-1: coeff = (1-h[i-s])
                    # h[i] appears in (1-h[m]) if 0 <= i <= n-1-s: coeff = -h[i+s]
                    coeff = 0.0
                    if s <= i:
                        coeff += (1.0 - h[i - s])
                    if i <= n - 1 - s:
                        coeff -= h[i + s]
                    a_k[p] = coeff * dx
                    
                    # b_p = f_p - coeff * h[i]
                    # f_p = sum h[m+s]*(1-h[m])
                    # = h[i]*(1-h[i-s]) [if s<=i<=n-1] + other terms not involving h[i]
                    #   + [-h[i]*h[i+s]] [if i <= n-1-s] + other terms
                    # So f_p = coeff * h[i] + (f_p - coeff * h[i])
                    # b_p = f_p - a_p * h[i]  (but we need to compute f_p first)
                else:
                    t = -s
                    # f_p = sum_{m=0}^{n-1-t} h[m] * (1 - h[m+t])
                    coeff = 0.0
                    if i <= n - 1 - t:
                        coeff += (1.0 - h[i + t])
                    if t <= i:
                        coeff -= h[i - t]
                    a_k[p] = coeff * dx
            
            # Compute b_k: need actual f_p values

            # Actually, let's compute b_k differently
            # b_k[p] = f_p - a_k[p] * h[i]
            b_k = f_vals - a_k * h[i]

            x_opt, _ = solve_1d_minimax(a_k + lam, b_k)
            h[i] = x_opt

            max_improve = sc_old - sc_new

    # Quick check: if all slopes are >= 0, min is at x=0
    # If all slopes are <= 0, min is at x=1

        return 0.0, b.max()
    if a.max() <= 1e-15:
        return 1.0, (a + b).max()

    # Find active line at x=0
    k = np.argmax(b)

    # The line k is active for x in [x_start, x_end)
    # x_end = min over j with a[j] > a[k] of (b[k] - b[j]) / (a[j] - a[k])

    active = k

    x = 0.0

            best_x = x

So df/dh[i] = (1-h[i-t]) * [t ≤ i ≤ n-1] - h[i+t] * [0 ≤ i ≤ n-1-t]

Let me code this up efficiently. For a given i, I need to compute this for all lags (p from 0 to 2n-2, or lag from -(n-1) to n-1).

def compute_a_k(h, i, dx):

        coeff = 0.0
        if lag >= 0:
            if i <= n - 1 - lag:
                coeff += (1.0 - h[i + lag])
            if lag <= i:  # and i <= n-1 always
                coeff -= h[i - lag]
        else:
            t = -lag
            if t <= i:  # and i <= n-1 always
                coeff += (1.0 - h[i - t])
            if i <= n - 1 - t:
                coeff -= h[i + t]
        a[p] = coeff * dx
    return a

def official_score(h):

    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -5.0, 5.0)
    lo, hi = -10.0, 10.0

        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break

                residual += sub

def solve_1d_minimax(a, b, n_iter=60):
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    x = (lo + hi) / 2.0
    val = np.max(a * x + b)
    return x, val

def coordinate_descent(h_init, max_sweeps=500, tol=1e-10, seed=0):
    n = len(h_init)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h_init.copy()

    rng = np.random.default_rng(seed)
    
    prev_score = f_vals.max()

        order = np.arange(n)
        np.random.default_rng(seed + sweep).shuffle(order)
        
        for i in order:
            a_k = compute_a_k(h, i, dx)
            b_k = f_vals - a_k * h[i]
            x_opt, _ = solve_1d_minimax(a_k, b_k, n_iter=50)
            delta = x_opt - h[i]
            if abs(delta) > 1e-15:
                h[i] = x_opt
                f_vals += a_k * delta

def try_smooth(h0, beta, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        logsum = np.log(np.sum(np.exp(shifted)))
        return cmax + logsum / beta
    def grad(x):
        vals = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = vals.max()
        shifted = beta * (vals - cmax)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= ws
        g = np.zeros(n)

            lag = p - (n - 1)
            if w[p] < 1e-10:
                continue
            if lag >= 0:
                L = n - lag
                g[:L] += w[p] * dx * (-x[lag:lag+L])
                g[lag:] += w[p] * dx * (1.0 - x[:L])
            else:
                t = -lag
                L = n - t
                g[:L] += w[p] * dx * (1.0 - x[t:])
                g[t:] += w[p] * dx * (-x[:L])
        return g

            return h_new

        pass
    return None
