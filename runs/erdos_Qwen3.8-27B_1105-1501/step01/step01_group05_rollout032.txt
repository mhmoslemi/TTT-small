def block_mutation(h, rng, block_size_range=(3, 10), step_range=(0.02, 0.15)):

    n = len(h)

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta < 1e-18:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta < 1e-18:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def eval_pair_move(h, i, j, x):
    S = h[i] + h[j]
    old_i = h[i]
    old_j = h[j]
    h[i] = x
    h[j] = S - x
    s = official_score(h)
    h[i] = old_i
    h[j] = old_j
    return s

def pair_line_minimize(h, i, j, scan_pts=20):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-12:
        return official_score(h)
    cur_score = official_score(h)
    xs = np.linspace(L, U, scan_pts)
    scores = np.empty(scan_pts)
    for k in range(scan_pts):
        scores[k] = eval_pair_move(h, i, j, float(xs[k]))
    min_idx = int(np.argmin(scores))
    best_s = float(scores[min_idx])
    best_x = float(xs[min_idx])
    if min_idx == 0:
        a, b = float(xs[0]), float(xs[1])
    elif min_idx == scan_pts - 1:
        a, b = float(xs[-2]), float(xs[-1])
    else:
        a, b = float(xs[min_idx - 1]), float(xs[min_idx + 1])
    gr = 0.6180339887498949
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_pair_move(h, i, j, c)
    fd = eval_pair_move(h, i, j, d)
    for _ in range(35):
        if abs(b - a) < 1e-13:

        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_pair_move(h, i, j, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_pair_move(h, i, j, d)
    x_opt = (a + b) / 2.0
    s_opt = eval_pair_move(h, i, j, x_opt)
    if s_opt < best_s:
        best_s = s_opt
        best_x = x_opt
    if best_s < cur_score - 1e-14:
        h[i] = best_x
        h[j] = S - best_x
        return best_s
    return cur_score

def build_pair_table(n):
    plist = []
    for i in range(n):
        for j in range(i + 1, n):
            plist.append((i, j))
    return plist

def pair_descent_pass(h, rng, deadline, max_pairs=None, scan_pts=20):
    n = len(h)
    cur_score = official_score(h)
    plist = build_pair_table(n)
    idx = list(range(len(plist)))
    rng.shuffle(idx)
    if max_pairs is not None:
        idx = idx[:min(max_pairs, len(idx))]
    for k in idx:
        if time.perf_counter() > deadline:
            break
        i, j = plist[k]
        S = h[i] + h[j]
        L = max(0.0, S - 1.0)
        U = min(1.0, S)
        if U - L < 1e-8:
            continue
        cur_score = pair_line_minimize(h, i, j, scan_pts=scan_pts)
    return h, cur_score

def block_mutation(h, rng):
    n = len(h)
    m = int(rng.integers(3, min(12, n)))
    idx = rng.choice(n, size=m, replace=False)
    d = rng.standard_normal(m)
    d -= d.mean()
    dn = np.max(np.abs(d))
    if dn < 1e-14:
        return h.copy()
    d /= dn
    max_step = 1e10
    for k in range(m):
        if d[k] > 1e-14:
            max_step = min(max_step, (1.0 - h[idx[k]]) / d[k])
        elif d[k] < -1e-14:
            max_step = min(max_step, h[idx[k]] / (-d[k]))
    if max_step < 1e-14:
        return h.copy()
    step = min(float(rng.uniform(0.03, 0.12)), max_step * 0.9)
    h_new = h.copy()
    h_new[idx] += step * d
    return repair_vector(h_new, n)

def interpolate_h(parent, n_target, offset=0):
    m = len(parent)
    if m == n_target:
        return repair_vector(parent.copy(), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) / n_target
    order = np.argsort(xt)
    h = np.interp(xt[order], xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n_target)

def try_local_descent(h, rng, deadline, max_passes=10, scan_pts=18):

    for _ in range(max_passes):

        h, new_s = pair_descent_pass(h.copy(), rng, deadline, scan_pts=scan_pts)
        if new_s >= cur_s + 1e-13:
            break
        h = h
        cur_s = new_s
    return h, cur_s
