def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    """Project h to {sum(h) = target, 0 <= h <= 1} via bisection on threshold."""
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_gradient_k(h, k):
    """Compute ∂c[k]/∂h for a specific lag k."""
    n = len(h)
    g = 1.0 - h
    tau = k - (n - 1)
    grad = np.zeros(n)
    # First term: g[m - tau] for valid m
    # g[m - tau] is valid when 0 <= m - tau <= n-1, i.e., tau <= m <= n-1+tau
    lo1 = max(0, tau)
    hi1 = min(n, n + tau)
    if hi1 > lo1:
        grad[lo1:hi1] += g[lo1 - tau : hi1 - tau]
    # Second term: -h[m + tau] for valid m
    # h[m + tau] is valid when 0 <= m + tau <= n-1, i.e., -tau <= m <= n-1-tau
    lo2 = max(0, -tau)
    hi2 = min(n, n - tau)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + tau : hi2 + tau]
    return grad

def active_set_lp(h, dx, active_k_indices, rho):
    """Solve the tangent LP for the active set.
    
    Minimize t subject to:
      t - dx * grad_k · delta >= dx * c_k for k in active set
      sum(delta) = 0
      -rho * h[i] <= delta[i] <= rho * (1-h[i])
    
    Returns delta (the step direction).
    """
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    A = active_k_indices
    m = len(A)
    
    # Variables: [t, delta[0], ..., delta[n-1]] (n+1 variables)
    # Objective: minimize t (coefficient 1 for t, 0 for delta)
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # t - dx * grad_k · delta >= dx * c_k
    # => -t + dx * grad_k · delta <= -dx * c_k
    A_ineq = np.zeros((m, n + 1))
    b_ineq = np.zeros(m)
    for i, k in enumerate(A):
        A_ineq[i, 0] = -1.0
        A_ineq[i, 1:] = dx * get_gradient_k(h, k)
        b_ineq[i] = -dx * c[k]
    
    # Box constraints on delta: -rho*h[i] <= delta[i] <= rho*(1-h[i])
    # lower_b
    lb = np.zeros(n + 1)
    ub = np.zeros(n + 1)
    lb[0] = 0  # t >= 0 (actually, t should be >= some value, but for the LP it's a relative thing)
    ub[0] = np.inf
    for i in range(n):
        lb[i+1] = -rho * h[i]
        ub[i+1] = rho * (1.0 - h[i])
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = [0.0]
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=list(zip(lb, ub)), method='highs')
    
    if result.success:
        return result.x[1:]
    return None

            rho *= 0.5

                rho = initial_rho
                stall = 0
    
    return best_h, best_c5

def project_to_constraint(h, target):
