import numpy as np
import time
from scipy.optimize import differential_evolution

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - z)
        else:
            order = np.argsort(z)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - z[idx])
                z[idx] += add
                diff -= add
            else:
                sub = min(-diff, z[idx])
                z[idx] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return np.clip(z, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_subgrad(h, scores, active_threshold=0.005):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    max_s = scores.max()
    active = np.where(scores >= max_s - active_threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    s_act = scores[active]
    T = max(0.005, (max_s - s_act.min()) * 0.5)
    w = np.exp((s_act - max_s) / T)
    w /= w.sum()
    g = np.zeros(n)
    for wi, lag in zip(w, active):
        k = int(lag) - center
        if k == 0:
            g += wi * dx * (1.0 - 2.0 * h)
        elif k > 0:
            m = n - k
            g[k:] += wi * dx * (1.0 - h[:m])
            g[:m] -= wi * dx * h[k:]
        else:
            kp = -k
            m = n - kp
            g[:m] += wi * dx * (1.0 - h[kp:])
            g[kp:] -= wi * dx * h[:m]
    return g, active

def pair_move(h, i, j, cur_c, n_scan=30):
    orig_i, orig_j = h[i], h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_d = 0.0
    best_s = cur_c
    for d in np.linspace(lo, hi, n_scan):
        h[i] = orig_i + d
        h[j] = orig_j - d
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_d = d
    h[i], h[j] = orig_i, orig_j
    if best_s < cur_c - 1e-14 and abs(best_d) > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        span = (hi - lo) / max(n_scan, 2)
        lo2 = max(lo, best_d - 1.5 * span)
        hi2 = min(hi, best_d + 1.5 * span)
        a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
        a2 = lo2 + gr * (hi2 - lo2)
        for _ in range(28):
            h[i] = orig_i + a1; h[j] = orig_j - a1
            s1 = score_exact(h)
            h[i] = orig_i + a2; h[j] = orig_j - a2
            s2 = score_exact(h)
            if s1 < s2:
                if s1 < best_s:
                    best_s, best_d = s1, a1
                hi2, a2 = a2, a1
                a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            else:
                if s2 < best_s:
                    best_s, best_d = s2, a2
                lo2, a1 = a1, a2
                a2 = lo2 + gr * (hi2 - lo2)
        best_d = (lo2 + hi2) * 0.5
        h[i] = orig_i + best_d
        h[j] = orig_j - best_d
        return float(score_exact(h))
    return None

def targeted_pair_move(h, i, active_lags, cur_c, n_scan=25):
    """Find best j to pair with i, based on active lags."""
    n = len(h)
    center = n - 1
    best_j = -1
    best_gain = 0.0
    for j in range(n):
        if j == i:
            continue
        gain = 0.0
        for lag in active_lags:
            k = int(lag) - center
            if k == 0:
                gain += 1.0
            elif k > 0:
                if 0 <= i < n - k:
                    gain += 1.0
                if k <= i < n:
                    gain += 1.0
                if 0 <= j < n - k:
                    gain += 1.0
                if k <= j < n:
                    gain += 1.0
            else:
                kp = -k
                if 0 <= i < n - kp:
                    gain += 1.0
                if kp <= i < n:
                    gain += 1.0
                if 0 <= j < n - kp:
                    gain += 1.0
                if kp <= j < n:
                    gain += 1.0
        if gain > best_gain:
            best_gain = gain
            best_j = j
    if best_j < 0:
        return None
    return pair_move(h, i, best_j, cur_c, n_scan)

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 15:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def fourier_to_h(params, n):
    K = len(params) // 2
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = params[2 * (k - 1)]
        b = params[2 * (k - 1) + 1]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, n)

def fourier_obj(params, n):
    h = fourier_to_h(params, n)
    return score_exact(h)

def poly_to_h(coefs, n):
    """h(x) = clip(sum coefs[i] * (2x/2 - 1)^i for i in range(len(coefs)))"""
    x = np.linspace(-1, 1, n)
    h = np.zeros(n)
    for i, c in enumerate(coefs):
        h += c * x ** i
    h = np.clip(h, 0.0, 1.0)
    if h.sum() * (2.0 / n) > 0:
        h = make_feasible(h, n)
    return h

def poly_obj(params, n):
    h = poly_to_h(params, n)
    return score_exact(h)
