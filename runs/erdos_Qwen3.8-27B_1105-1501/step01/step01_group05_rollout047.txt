import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(np.asarray(v, dtype=float).ravel()[:n], 0.0, 1.0)
    target = n / 2.0
    residual = target - np.sum(h)
    if abs(residual) < 1e-14:
        return h
    # Iterative adjustment
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx] + 1e-15)
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx] + 1e-15)
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradients(h):
    """Compute gradient matrix G of shape (2n-1, n) for scaled correlations."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for s in range(-(n-1), n):
        r = s + (n - 1)  # output index
        if s >= 0:
            # For m in [s, n-1]: G[r,m] += (1-h[m-s])
            if s < n:
                G[r, s:] += (1.0 - h[:n-s])
            # For m in [0, n-s-1]: G[r,m] += -h[m+s]
            if s < n:
                G[r, :n-s] += (-h[s:])
        else:
            a = -s
            # For m in [0, n-1-a]: G[r,m] += (1-h[m+a])
            G[r, :n-a] += (1.0 - h[a:])
            # For m in [a, n-1]: G[r,m] += -h[m-a]
            G[r, a:] += (-h[:n-a])
    G *= dx
    return G

def lp_minimax_direction(h, G, c_scaled, rho_target=None):
    """Find LP minimax direction. Returns (d, rho) or (None, None)."""
    n = len(h)
    L = 2 * n - 1
    c_max = np.max(c_scaled)
    margins = c_max - c_scaled  # how far below max each correlation is
    
    # LP: minimize rho
    # subject to: G_r @ d <= margins_r + rho for all r
    #             sum(d) = 0
    #             -1 <= d_i <= 1
    #
    # Variables: d_0,...,d_{n-1}, rho
    # Total: n+1 variables
    #
    # Inequality: G_r @ d - rho <= margins_r for all r
    # Equality: sum(d) = 0
    # Bounds: -1 <= d_i <= 1, rho >= rho_target (or -inf)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize rho
    
    # Inequality constraints: G_r @ d - 1*rho <= margins_r
    A_ineq = np.zeros((L, n + 1))
    A_ineq[:, :n] = G
    A_ineq[:, n] = -1.0
    b_ineq = margins.copy()
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * n + [(-10.0, 10.0)]  # rho unbounded-ish
    
    try:
        res = optimize.linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                               A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            rho = res.x[n]
            if np.all(np.isfinite(d)) and np.max(np.abs(d)) > 1e-10:
                return d, rho
    except Exception:
        pass
    return None, None

def line_search(h, d, current_score, n):
    """Find best step size along direction d."""
    best_alpha = 0.0
    best_score = current_score
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]

            if s < best_score - 1e-12:
                best_score = s
                best_alpha = alpha
    return best_alpha, best_score

def interpolate_to_n(parent, n_target, shift=0):
    """Interpolate parent to n_target points with optional shift."""
    m = len(parent)
    if m == 0:
        return np.full(n_target, 0.5)
    xs = (np.arange(m) + 0.5) / m * 2.0
    xt = ((np.arange(n_target) + 0.5 + shift) % n_target) / n_target * 2.0
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n_target)

def slsqp_polish(h, n, time_limit=15.0):
    """SLSQP minimax polish."""
    dx = 2.0 / n
    target = n / 2.0
    lags = np.arange(-(n-1), n)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq(x):
        h_x = x[:n]
        return x[n] - dx * np.correlate(h_x, 1.0 - h_x, mode="full")
    
    def ineq_jac(x):
        h_x = x[:n]
        G = corr_gradients(h_x)  # already scaled by dx
        j = np.zeros((2*n-1, n+1))
        j[:, :n] = -G
        j[:, n] = 1.0
        return j
    
    def eq(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    score = official_score(h)
    x0 = np.concatenate([h, [score + 1e-4]])
    bounds = [(0, 1)] * n + [(0, 2)]
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}]
    
    try:
        res = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 100, 'ftol': 1e-10})
        if res.success and np.all(np.isfinite(res.x[:n])):

            s = official_score(h_out)
            if s < score - 1e-10:

    except Exception:
        pass
    return h, score
