def pair_swap(h, n, i, j, delta):
    """Transfer delta mass from position i to position j."""
    h_try = h.copy()
    h_try[i] -= delta
    h_try[j] += delta
    # Clip and repair
    if h_try[i] < 0:
        delta = h[i]
        h_try[i] = 0
        h_try[j] = h[j] + delta
    if h_try[j] > 1:
        delta = 1 - h[j]
        h_try[i] = h[i] - delta
        h_try[j] = 1
    return h_try

But actually, for this problem, I think a simpler approach might work better: **direct coordinate/pair optimization**.

Let me think about this differently. The key observation is:

For a pair (i, j) and a transfer delta (moving mass from i to j), the change in f_k is:
Δf_k = delta * (∂f_k/∂h[j] - ∂f_k/∂h[i])

We want this to be negative (decrease f_k) for all active k. The most conservative choice is:
max_k Δf_k = delta * max_k (∂f_k/∂h[j] - ∂f_k/∂h[i])

We want this < 0, i.e., delta * max_k (...) < 0.

If delta > 0 (transferring from i to j), we need max_k (∂f_k/∂h[j] - ∂f_k/∂h[i]) < 0.
If delta < 0 (transferring from j to i), we need max_k (∂f_k/∂h[i] - ∂f_k/∂h[j]) < 0.

So the optimal pair is the one that maximizes the minimum reduction across active lags.

Computing this for all pairs is O(n^2 * |active|) which is fine for n=65.

Let me implement this more carefully:

```python
def gradient_for_lag(h, k, n, dx):
    """Compute gradient of f_k with respect to h."""
    grad = np.zeros(n)
    j = np.arange(n)
    # h[j] as first factor: valid if 0 <= j+k < n
    mask1 = (j + k >= 0) & (j + k < n)
    grad[mask1] += dx * (1.0 - h[j[mask1] + k])
    # h[j] as second factor: valid if 0 <= j-k < n
    mask2 = (j - k >= 0) & (j - k < n)
    grad[mask2] -= dx * h[j[mask2] - k]
    return grad
```

A weaker condition: we want the AVERAGE reduction to be good, or the MAX reduction to be bounded.

Let me use a different criterion: minimize the maximum of the active f_k's after the transfer.

For a small transfer delta from i to j:
f_k_new ≈ f_k + delta * (grad_k[j] - grad_k[i])

We want: max_k (f_k + delta * (grad_k[j] - grad_k[i])) < max_k f_k

This is a linear programming problem in delta (and the choice of i, j).

For a given pair (i, j), the optimal delta is:
- If max_k (grad_k[j] - grad_k[i]) < 0: use the maximum delta that keeps feasibility
- Otherwise: this pair doesn't help

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(1000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            # Distribute to positions with most room
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    # Final clip and fix
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def gen_quad_residue(p, n):
    """Generate a sequence based on quadratic residues mod p, resized to n."""
    S = set()
    for i in range(1, p):
        S.add((i*i) % p)
    # Create length-p sequence
    h_p = np.zeros(p)
    for i in range(p):
        if i in S:
            h_p[i] = 1.0
    # Resize to n
    if n == p:
        return repair_vector(h_p, n)
    else:
        return gen_interpolated(h_p, n)

import numpy as np
from scipy import optimize
import time

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    # Iteratively fix the sum
    for _ in range(2000):
        s = np.sum(h)
        residual = target - s
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    # Final tiny correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-16:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def gen_quad_residue(p, n, shift=0):
    S = set((i*i) % p for i in range(1, p))
    h_p = np.zeros(p)
    for i in range(p):
        if ((i + shift) % p) in S:
            h_p[i] = 1.0
    if n == p:
        return repair_vector(h_p, n)
    return gen_interpolated(h_p, n)

def gen_random_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:m_full]] = 1.0
    if frac > 1e-10:
        h[idx[m_full]] = frac
    return repair_vector(h, n)

def gen_mersenne(n, rng):
    """Generate using a Mersenne-like pattern."""
    h = np.zeros(n)
    block = int(rng.integers(2, 8))
    for i in range(0, n, block):
        h[i:min(i+block, n)] = rng.choice([0.0, 1.0])
    return repair_vector(h, n)

def local_search(h, n, time_limit=30.0, seed=42):
    """Balanced subgradient local search with simulated annealing."""
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n).copy()
    score = official_score(h)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    start = time.perf_counter()
    temp = 0.015
    cooling = 0.99995
    no_improve = 0
    
    for it in range(500000):  # max iterations as safety
        elapsed = time.perf_counter() - start
        if elapsed > time_limit:
            break
        if temp < 1e-8:
            break
        
        # Compute score and find active lags
        corr = np.correlate(h, 1.0 - h, mode="full")
        sv = corr * dx
        cur_max = float(np.max(sv))
        
        if cur_max < score - 1e-12:
            score = cur_max
            no_improve = 0
        else:
            no_improve += 1
        
        # Reset after stagnation with perturbation
        if no_improve > 300 and temp > 0.005:
            h = h + rng.normal(0, 0.03, n)
            h = repair_vector(h, n)
            score = official_score(h)
            no_improve = 0
            continue
        
        # Active lags
        threshold = cur_max - 0.006
        active = np.where(sv >= threshold)[0]
        n_act = len(active)
        if n_act == 0:
            active = np.array([int(np.argmax(sv))])
            n_act = 1
        
        # Subgradient: average gradient of active lags
        grad = np.zeros(n)
        for ki in active:
            k = int(lags[ki])
            j = np.arange(n)
            m1 = (j + k >= 0) & (j + k < n)
            grad[m1] += (1.0 - h[j[m1] + k])
            m2 = (j - k >= 0) & (j - k < n)
            grad[m2] -= h[j[m2] - k]
        grad /= n_act
        grad -= np.mean(grad)
        
        # Determine best pair transfer
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        
        improved = False
        # Try the gradient-suggested pair
        for step in [0.02, 0.04, 0.06, 0.08, 0.12, 0.18, 0.25]:
            amt = min(step, h[src], 1.0 - h[dst])
            if amt < 1e-6:
                break
            h_try = h.copy()
            h_try[src] -= amt
            h_try[dst] += amt
            s = float(np.max(np.correlate(h_try, 1.0 - h_try, mode="full") * dx))
            if s < cur_max - 1e-12:
                h = h_try
                score = min(score, s)
                improved = True
                break
        
        if not improved:
            # Try random pairs (up to 8)
            for _ in range(8):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                step = float(rng.uniform(0.01, 0.25))
                direction = 1 if rng.random() > 0.5 else -1
                if direction > 0:
                    amt = min(step, h[i], 1.0 - h[j])
                else:
                    amt = min(step, h[j], 1.0 - h[i])
                if amt < 1e-6:
                    continue
                h_try = h.copy()
                if direction > 0:
                    h_try[i] -= amt
                    h_try[j] += amt
                else:
                    h_try[j] -= amt
                    h_try[i] += amt
                s = float(np.max(np.correlate(h_try, 1.0 - h_try, mode="full") * dx))
                if s < cur_max - 1e-12:
                    h = h_try
                    score = min(score, s)
                    improved = True
                    break
        
        if not improved:
            # Simulated annealing acceptance
            accept_prob = min(1.0, temp * 5.0)
            if rng.random() < accept_prob:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and 0.05 < h[i] < 0.95:
                    step = float(rng.uniform(0.05, 0.3))
                    if rng.random() > 0.5:
                        amt = min(step, h[i], 1.0 - h[j])
                    else:
                        amt = min(step, h[j], 1.0 - h[i])
                    if amt > 1e-6:
                        h_try = h.copy()
                        if rng.random() > 0.5:
                            h_try[i] -= amt
                            h_try[j] += amt
                        else:
                            h_try[j] -= amt
                            h_try[i] += amt
                        h_try = np.clip(h_try, 0, 1)
                        h_try = repair_vector(h_try, n)
                        s = official_score(h_try)
                        if s < cur_max + temp:
                            h = h_try
        
        temp *= cooling
    
    h = repair_vector(h, n)
    return h, official_score(h)

def slsqp_refine(h, n, max_iter=60):
    """SLSQP refinement to minimize max overlap."""
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n-1), n)
    n_lags = 2 * n - 1
    initial_score = official_score(h)
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            jarr = np.arange(n)
            m1 = (jarr + k >= 0) & (jarr + k < n)
            j[ki, jarr[m1]] -= dx * (1.0 - h_x[jarr[m1] + k])
            m2 = (jarr - k >= 0) & (jarr - k < n)
            j[ki, jarr[m2]] += dx * h_x[jarr[m2] - k]
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    theta0 = initial_score + 1e-4
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-12:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

The Jacobian of (theta - dx * corr_k) with respect to h[j]:

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(2000):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-16:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_quad_residue(p, n, shift=0):
    S = set((i * i) % p for i in range(1, p))
    h_p = np.zeros(p)
    for i in range(p):
        if ((i + shift) % p) in S:
            h_p[i] = 1.0
    if n == p:
        return repair_vector(h_p, n)
    return gen_interpolated(h_p, n)
