    """Interpolate parent to target resolution with optional shift."""
    m = len(parent)
    if m == 0:
        return np.full(target_n, 0.5)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(target_n) + 0.5 + shift) % target_n) * 2.0 / target_n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, target_n)

    n = len(h)

def project_to_target(v, n):

    target = n / 2.0

        if s < target:
            lo = mid
        else:
            hi = mid

def weighted_gradient(h, tol_frac=0.05):
    """Compute weighted negative gradient for active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(corr)
    tol = max(0.001, tol_frac * max_val)
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)

        w = max(w, 0.5)

    return grad, max_val, len(active)

def interpolate_to(parent, target_n, shift=0):
    m = len(parent)

    target_sum = n / 2.0

        return x[n]
