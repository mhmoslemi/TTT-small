    reserve = max(15.0, 0.04 * budget_s)

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 4:
            raise ValueError("bad parent")
        parent = project_feasible(parent)
    except Exception:
        parent = np.arange(40) * 0.5
        parent = np.where(parent % 1.0 == 0.0, 1.0, 0.0)
        parent = project_feasible(parent)

    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)

    def try_improve(h0, n, label, dl):
        nonlocal best_h, best_score, best_n
        hs = project_feasible(h0)
        sc = official_score(hs)
        if sc < best_score - 1e-14:
            best_h = hs.copy()
            best_score = sc
            best_n = n
        if time.time() > dl:
            return
        betas = [30, 80, 200, 500, 1200]
        steps = 80
        h_new, sc_new = plsed(hs, betas, steps, dl)
        if sc_new < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc_new
            best_n = len(h_new)

    # Stage 1: Binary local search at even n values
    pn = len(parent)
    even_ns = sorted(set([n for n in [pn - 4, pn - 2, pn, pn + 2, pn + 4] if n % 2 == 0 and 24 <= n <= 70]))

    for n in even_ns:
        if time.time() > deadline - 60:
            break
        h_interp = project_feasible(interp_to_n(parent, n))
        bits = arr_to_bits(h_interp, n)
        
        # Try steepest descent local search
        bits2, sc2 = blocal(bits, n, 200, deadline - 30)
        if sc2 < bscore(bits, n) - 1e-15:
            bits = bits2
        
        bits_arr = bits_to_arr(bits, n)
        try_improve(bits_arr, n, "binary", deadline)

        # Also try reversed
        bits_rev = arr_to_bits(1.0 - h_interp[::-1], n)
        bits_rev2, _ = blocal(bits_rev, n, 100, deadline - 30)
        bits_arr_rev = bits_to_arr(bits_rev2, n)
        try_improve(bits_arr_rev, n, "binary_rev", deadline)

    # Stage 2: Continuous PLSED from parent and perturbations
    if time.time() < deadline - 30:
        try_improve(parent, pn, "parent", deadline)

    if time.time() < deadline - 30:
        h_anti = project_feasible(1.0 - parent[::-1])
        try_improve(h_anti, pn, "anti", deadline)

    # Stage 3: Random perturbations around best
    if time.time() < deadline - 20:
        ntries = min(20, int((deadline - time.time()) / 30))
        for k in range(ntries):
            if time.time() > deadline - 15:
                break
            sigma = 0.08 * (1.0 + 0.5 * k / max(1, ntries))
            h_pert = project_feasible(best_h + sigma * rng.standard_normal(best_n))
            try_improve(h_pert, best_n, "pert", deadline)

    # Stage 4: More binary search with random starts
    if time.time() < deadline - 40:
        for n in even_ns:
            if time.time() > deadline - 20:
                break
            for trial in range(5):
                if time.time() > deadline - 15:
                    break
                bits_rand = 0
                positions = list(range(n))
                rng.shuffle(positions)
                for i in positions[:n // 2]:
                    bits_rand |= (1 << i)
                bits_rand2, sc_rand2 = blocal(bits_rand, n, 150, deadline - 10)
                if sc_rand2 < best_score - 1e-4:
                    bits_arr_r = bits_to_arr(bits_rand2, n)
                    try_improve(bits_arr_r, n, "rand_bin", deadline)

    # Final polish
    if time.time() < deadline - 5:
        h_final, sc_final = plsed(best_h, [500, 1500, 3000], 60, deadline)
        if sc_final < best_score - 1e-14:
            best_h = h_final
            best_score = sc_final

    best_h = project_feasible(best_h)
    final_score = official_score(best_h)
    if final_score > best_score:
        final_score = best_score
    return (best_h, final_score, len(best_h))
