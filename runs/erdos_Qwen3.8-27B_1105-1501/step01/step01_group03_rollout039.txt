def make_feasible(z):

    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()

        if s < target:

        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.default_rng(seed)
    parent = get_parent()
    if parent is not None:
        best_h = parent.copy()
        best_n = len(parent)
    else:
        best_n = 61
        best_h = make_feasible(np.full(best_n, 0.5))
    best_score = official_score(best_h)
    phase1_end = t0 + 0.45 * (budget_s - 15)
    best_h = optimize_pairs(best_h, best_n, phase1_end, max_passes=200, K=30)
    sc = official_score(best_h)
    if sc < best_score:
        best_score, best_h = sc, best_h
    phase2_end = t0 + 0.72 * (budget_s - 15)
    n_offsets = [0, -4, 4, -8, 8, -12, 12, 16, -16, 20, -20]
    tried_n = set()
    for dn in n_offsets:
        if time.time() > phase2_end - 5:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 100 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_try = transfer_length(best_h, best_n, n_try)
        h_try = optimize_pairs(h_try, n_try, min(phase2_end - 2, t0 + budget_s * 0.12), max_passes=100, K=25)
        sc_try = official_score(h_try)
        if sc_try < best_score:
            best_h, best_score, best_n = h_try, sc_try, n_try
    if time.time() < phase2_end:
        h_rev = make_feasible(1.0 - best_h[::-1])
        h_rev = optimize_pairs(h_rev, best_n, min(phase2_end, t0 + budget_s * 0.1), max_passes=80, K=25)
        sc_rev = official_score(h_rev)
        if sc_rev < best_score:
            best_h, best_score = h_rev, sc_rev
    phase3_end = t0 + 0.88 * (budget_s - 15)
    best_h = multi_coord_search(best_h, best_n, phase3_end, rng, K=20, max_iters=300)
    sc = official_score(best_h)
    if sc < best_score:
        best_score, best_h = sc, best_h
    best_h = optimize_pairs(best_h, best_n, t0 + budget_s - 8, max_passes=200, K=30)
    sc = official_score(best_h)
    if sc < best_score:
        best_score, best_h = sc, best_h
    if time.time() < t0 + budget_s - 5:
        h_rev2 = make_feasible(1.0 - best_h[::-1])
        h_rev2 = optimize_pairs(h_rev2, best_n, t0 + budget_s - 3, max_passes=50, K=30)
        sc2 = official_score(h_rev2)
        if sc2 < best_score:
            best_h, best_score = h_rev2, sc2
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score > 0.5:
        best_h = make_feasible(np.full(best_n, 0.5))
        final_score = official_score(best_h)
    return best_h, final_score, len(best_h)
