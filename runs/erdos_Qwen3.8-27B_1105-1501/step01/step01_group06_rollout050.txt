import numpy as np
import time
from scipy.optimize import linprog

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_to_target(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Water-filling bisection
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        # Sequential fix
        if residual > 0:
            for i in range(n):
                if residual <= 0:
                    break
                room = 1.0 - x[i]
                if room > 1e-12:
                    amt = min(room, residual)
                    x[i] += amt
                    residual -= amt
        else:
            for i in range(n):
                if residual >= 0:
                    break
                room = x[i]
                if room > 1e-12:
                    amt = min(room, -residual)
                    x[i] -= amt
                    residual += amt
    return x

def compute_gradients(h):
    """Compute gradient of each lag score w.r.t. h.
    Returns grad array of shape (2N-1, N).
    grad[idx, j] = d(S_k)/d(h[j]) where k = idx - (N-1)
    """
    n = len(h)
    grad = np.zeros((2*n - 1, n))
    for idx in range(2*n - 1):
        k = idx - (n - 1)  # lag
        for j in range(n):
            g = 0.0
            # First term: (1-h[j-k]) if 0 <= j-k < n
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            # Second term: -h[j+k] if 0 <= j+k < n
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[idx, j] = g
    return grad

def lp_direction(h, radius, top_k=None, lambda_reg=5.0):
    """Solve LP to find descent direction.
    Returns (p, success, info)
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr  # shape (2n-1,)
    L = 2 * n - 1
    M = float(np.max(scores))
    
    # Select top-k lags if specified
    if top_k is not None and top_k < L:
        # Select lags with scores close to the max
        threshold = M - 0.1 * dx  # within 0.1*dx of max
        sel = np.where(scores >= M - 0.01)[0]  # within 0.01 of max
        if len(sel) < top_k:
            # Also add some more
            vals, idxs = np.sort(scores), np.arange(L)
            sel = idxs[-top_k:]
        sel = list(sel)
        # Always include the argmax
        argmax_idx = int(np.argmax(scores))
        if argmax_idx not in sel:
            sel.append(argmax_idx)
    else:
        sel = list(range(L))
    
    K = len(sel)
    
    # Compute gradients for selected lags only (more efficient)
    grad_sel = np.zeros((K, n))
    for i, idx in enumerate(sel):
        k = idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad_sel[i, j] = g
    
    # Variables: p (n), r (1), tau (1) → total n+2
    # Minimize: tau + lambda_reg * r
    c = np.zeros(n + 2)
    c[n] = lambda_reg  # r coefficient
    c[n + 1] = 1.0     # tau coefficient
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each selected lag i:
    #   grad_sel[i] dot p - tau <= -scores[sel[i]]
    #   → [grad_sel[i], 0, -1] dot [p, r, tau] <= -scores[sel[i]]
    
    # p_i - r <= 0 for all i
    # -p_i - r <= 0 for all i
    
    n_rows = K + 2 * n
    A_ub = np.zeros((n_rows, n + 2))
    b_ub = np.zeros(n_rows)
    
    row = 0
    for i in range(K):
        A_ub[row, :n] = grad_sel[i]
        A_ub[row, n + 1] = -1.0  # -tau
        b_ub[row] = -scores[sel[i]]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n] = -1.0  # -r
        b_ub[row] = 0.0
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n] = -1.0  # -r
        b_ub[row] = 0.0
        row += 1
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds:
    # p_i: [max(-radius, -h[i]), min(radius, 1-h[i])]
    # r: [0, radius]
    # tau: [-inf, inf]
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, radius))  # r
    bounds.append((None, None))   # tau
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None, False, "LP failure"
    
    if res.status != 0:
        return None, False, f"LP status {res.status}: {res.message}"
    
    p = res.x[:n]
    r_val = res.x[n]
    tau_val = res.x[n + 1]
    
    # Check if direction is meaningful
    if np.max(np.abs(p)) < 1e-10:
        return p, True, "zero direction"
    
    return p, True, f"r={r_val:.6f}, tau={tau_val:.6f}, predicted_c5={tau_val*dx:.6f}"

def line_search(h, p, current_c5, c5_corr):
    """Exact line search along h + alpha*p.
    Returns (best_alpha, best_c5, best_h)
    """
    n = len(h)
    dx = 2.0 / n
    
    # Find max step
    alpha_max = 1.0
    for i in range(n):
        if p[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-12:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Sample candidate alphas
    alphas = np.array([0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 
                       0.9, 1.0]) * alpha_max
    # Also add some finer values
    alphas = np.unique(np.sort(alphas))
    alphas = alphas[alphas > 0]
    
    best_c5 = current_c5
    best_alpha = 0.0
    best_h = h.copy()
    
    for a in alphas:
        h_try = h + a * p
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a
            best_h = h_try.copy()
    
    # Refine around best
    if best_alpha > 0:
        lo_a = max(0.0, best_alpha * 0.8)
        hi_a = min(alpha_max, best_alpha * 1.2)
        for a in np.linspace(lo_a, hi_a, 20):
            h_try = h + a * p
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_alpha = a
                best_h = h_try.copy()
    
    # Re-project to ensure exact feasibility
    best_h = project_to_target(best_h, n / 2.0)
    c5_final = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)
    if c5_final < current_c5:
        return best_alpha, c5_final, best_h
    return 0.0, current_c5, h

def subgradient_direction(h, radius):
    """Fallback: weighted subgradient of top active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr
    M = float(np.max(scores))
    L = len(scores)
    
    # Select active lags
    active = np.where(scores >= M - 0.005)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    
    # Weighted gradient (use uniform weights for active lags)
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += g
    
    grad /= len(active)
    
    # Descent direction: negative gradient, projected to sum=0
    p = -grad
    p -= np.mean(p)
    
    # Normalize to radius
    max_abs = np.max(np.abs(p))
    if max_abs > 0:
        p *= min(1.0, radius / max_abs)
    
    # Enforce bounds
    for i in range(n):
        if p[i] > 1.0 - h[i]:
            p[i] = 1.0 - h[i]
        if p[i] < -h[i]:
            p[i] = -h[i]
    
    return p

def lp_optimize(h, n, target_sum, max_iters, radius0, deadline):
    """Run LP-based local optimization."""
    best_h = h.copy()
    best_c5, _ = exact_c5(best_h)
    radius = radius0
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        p, success, info = lp_direction(best_h, radius, top_k=None, lambda_reg=3.0)
        
        if not success or p is None:
            # Try fallback
            p = subgradient_direction(best_h, radius)
        
        if p is None or np.max(np.abs(p)) < 1e-12:
            # No direction found, shrink radius
            radius *= 0.5
            stall_count += 1
            if stall_count > 10:
                break
            continue
        
        # Line search
        alpha, c5_new, h_new = line_search(best_h, p, best_c5, None)
        
        if alpha > 0 and c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            stall_count = 0
            radius = min(0.5, radius * 1.1)  # Slowly grow radius
        else:
            radius *= 0.7
            stall_count += 1
            if stall_count > 15:
                break
    
    return best_h, best_c5

def perturb_and_optimize(h, n, target_sum, rng, ps, deadline):
    """Perturb and optimize."""
    h_pert = h.copy()
    mask = rng.random(n) < 0.5
    if np.any(mask):
        h_pert[mask] += rng.normal(0, ps, np.sum(mask))
    h_pert = project_to_target(h_pert, target_sum)
    return h_pert

def resolution_transfer(h_old, n_new, rng, deadline):
    """Interpolate to new resolution, project, return."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_to_target(h_new, n_new / 2.0)
    return h_new

def complement_transform(h):
    """Apply h -> 1-h and reverse."""
    return 1.0 - h[::-1].copy()

def project_to_target(v, target):
    v = np.asarray(v, dtype=float)
    n = len(v)

        return np.zeros(n)

        return np.ones(n)

        else:

            best_h = h_opt.copy()

    # Phase 6: Final fine polish
    dl6 = t0 + budget_s * 0.995
    for radius in [0.003, 0.001, 0.0005]:
        if time.time() > dl6:
            break
        h_opt, c5_opt = lp_optimize(best_h, max_iters=20, radius0=radius,
                                     deadline=min(dl6, time.time() + 15),
                                     lambda_reg=2.0, top_k=None)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final: subgradient micro-polish
    if time.time() < dl6:
        h_sg = best_h.copy()
        c5_sg, _ = exact_c5(h_sg)
        r = 0.0005
        for it in range(100):
            if time.time() > dl6:
                break
            p = subgradient_direction(h_sg, r)
            if np.max(np.abs(p)) < 1e-14:
                break
            alpha, c5_new, h_new = line_search(h_sg, p, c5_sg)
            if alpha > 0 and c5_new < c5_sg - 1e-12:
                h_sg = h_new
                c5_sg = c5_new
            else:
                r *= 0.5
                if r < 1e-7:
                    break
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final verification
    best_h = project_to_target(best_h, best_n / 2.0)
    final_c5, _ = exact_c5(best_h)
    return best_h, final_c5, best_n
