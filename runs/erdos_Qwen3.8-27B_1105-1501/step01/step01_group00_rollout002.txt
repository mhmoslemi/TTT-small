def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

                if 0 <= j - t <= n - 1 - t:
                    grad[j] -= wi * x[j - t]
    return grad

def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_eq_jac(x):
    return np.ones(len(x))

def try_slsqp_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_sum(res.x)
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None

def make_wave(n, freq1=3, freq2=7, amp1=0.2, amp2=0.1, phase=0.0, seed_offset=0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.cos(2 * np.pi * freq2 * x + phase * 0.7)
    return repair_sum(h)

def anti_sym_transform(h):
    h_new = 1.0 - h[::-1].copy()
    return repair_sum(h_new)

def sym_transform(h):
    h_new = 0.5 * (h + 1.0 - h[::-1])
    return repair_sum(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(10.0, 0.06 * budget_s)
    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_score = float('inf')
    best_n = 0
    
    def is_deadline():
        return time.time() >= deadline
    
    def update(h, sc):
        nonlocal best_h, best_score, best_n
        if sc < best_score - 1e-14:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
        elif abs(sc - best_score) <= 1e-14 and len(h) < best_n:
            best_h = h.copy()
            best_score = sc
            best_n = len(h)
    
    # Load parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if parent.size >= 4 and np.all(np.isfinite(parent)):
            parent = repair_sum(parent)
            ps = official_score(parent)
            if np.isfinite(ps):
                best_h = parent.copy()
                best_score = ps
                best_n = len(parent)
    except (NameError, AttributeError, Exception):
        pass
    
    if best_h is None:
        best_h = make_wave(50, 3, 7, 0.25, 0.1)
        best_score = official_score(best_h)
        best_n = len(best_h)
    
    # Try anti-symmetric version
    h_asym = anti_sym_transform(best_h)
    sc_asym = official_score(h_asym)
    if sc_asym < best_score - 1e-14:
        update(h_asym, sc_asym)
    
    h_sym = sym_transform(best_h)
    sc_sym = official_score(h_sym)
    if sc_sym < best_score - 1e-14:
        update(h_sym, sc_sym)
    
    # Phase 1: Try multiple resolutions
    if not is_deadline():
        n_base = len(best_h)
        n_options = sorted(set([max(20, n_base + d) for d in [-10, -6, -4, -2, 0, 2, 4, 6, 8, 10, 14] if n_base + d >= 16]))
        for n_opt in n_options:
            if is_deadline():
                break
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            if sc_int < best_score - 1e-14:
                update(h_int, sc_int)
    
    # Phase 2: SLSQP smooth optimization at best resolution
    if not is_deadline():
        for beta in [30, 100, 400, 1500, 5000]:
            if is_deadline():
                break
            h_new, sc = try_slsqp_smooth(best_h, beta, maxiter=150)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
    
    # Phase 3: Active gradient descent
    if not is_deadline():
        h_new, sc = active_grad_descent(best_h, max_iters=80, top_k=12)
        if sc < best_score - 1e-14:
            update(h_new, sc)
            h_new2, sc2 = active_grad_descent(h_new, max_iters=80, top_k=8)
            if sc2 < best_score - 1e-14:
                update(h_new2, sc2)
    
    # Phase 4: Two-point local search (key new component)
    if not is_deadline():
        remaining = deadline - time.time()
        max_passes = max(5, int(remaining / 3))
        h_new, sc = two_point_local_search(best_h, rng, max_passes=max_passes)
        if sc < best_score - 1e-14:
            update(h_new, sc)
    
    # Phase 5: SLSQP again after two-point
    if not is_deadline():
        for beta in [200, 1000, 4000]:
            if is_deadline():
                break
            h_new, sc = try_slsqp_smooth(best_h, beta, maxiter=150)
            if h_new is not None and sc < best_score - 1e-14:
                update(h_new, sc)
    
    # Phase 6: Annealed two-point search for escaping local minima
    if not is_deadline():
        remaining = deadline - time.time()
        max_iters = max(2000, int(remaining / 2))
        h_new, sc = annealed_two_point(best_h, rng, t_start=0.08, t_end=0.003, max_iters=max_iters, patience=3000)
        if sc < best_score - 1e-14:
            update(h_new, sc)
        
        # Polish after annealing
        if not is_deadline():
            h_new2, sc2 = two_point_local_search(best_h, rng, max_passes=50)
            if sc2 < best_score - 1e-14:
                update(h_new2, sc2)
    
    # Phase 7: Resolution exploration with local search
    if not is_deadline():
        n_base = len(best_h)
        for n_opt in sorted(set([max(20, n_base + d) for d in [-8, -4, 4, 8, 12]])):
            if is_deadline():
                break
            h_int = interp_to(best_h, n_opt)
            sc_int = official_score(h_int)
            h_new, sc = two_point_local_search(h_int, rng, max_passes=30)
            if sc < best_score - 1e-14:
                update(h_new, sc)
            h_new2, sc2 = try_slsqp_smooth(h_new, 500, maxiter=100)
            if h_new2 is not None and sc2 < best_score - 1e-14:
                update(h_new2, sc2)
    
    # Phase 8: Random restarts with local search
    if not is_deadline():
        remaining = deadline - time.time()
        n_extra = max(3, min(15, int(remaining / 30)))
        n_best = len(best_h)
        for i in range(n_extra):
            if is_deadline():
                break
            # Generate diverse starting points
            if i % 3 == 0:
                sigma = 0.05 + 0.1 * rng.random()
                h_cand = repair_sum(best_h + rng.normal(0, sigma, size=n_best))
            elif i % 3 == 1:
                h_cand = make_wave(n_best, 
                                   freq1=int(rng.integers(2, 8)), 
                                   freq2=int(rng.integers(4, 12)),
                                   amp1=0.1 + 0.2 * rng.random(),
                                   amp2=0.05 + 0.1 * rng.random(),
                                   phase=rng.uniform(0, 2*np.pi))
            else:
                # Interpolated from best with perturbation
                n_try = n_best + int(rng.integers(-4, 5))
                if n_try < 16:
                    n_try = 16
                h_cand = interp_to(best_h, n_try)
                h_cand = repair_sum(h_cand + rng.normal(0, 0.03, size=len(h_cand)))
            
            # Quick local polish
            h_new, sc = two_point_local_search(h_cand, rng, max_passes=20)
            if sc < best_score - 1e-14:
                update(h_new, sc)
            
            if not is_deadline():
                h_new2, sc2 = try_slsqp_smooth(h_new, 300, maxiter=80)
                if h_new2 is not None and sc2 < best_score - 1e-14:
                    update(h_new2, sc2)
    
    # Phase 9: Final intensive polish
    if not is_deadline():
        h_new, sc = active_grad_descent(best_h, max_iters=60, top_k=15)
        if sc < best_score - 1e-14:
            update(h_new, sc)
        
        h_new, sc = two_point_local_search(best_h, rng, max_passes=80)
        if sc < best_score - 1e-14:
            update(h_new, sc)
        
        if not is_deadline():
            for beta in [300, 1000, 5000, 15000]:
                if is_deadline():
                    break
                h_new, sc = try_slsqp_smooth(best_h, beta, maxiter=120)
                if h_new is not None and sc < best_score - 1e-14:
                    update(h_new, sc)
        
        if not is_deadline():
            h_new, sc = two_point_local_search(best_h, rng, max_passes=60)
            if sc < best_score - 1e-14:
                update(h_new, sc)
        
        if not is_deadline():
            h_new, sc = active_grad_descent(best_h, max_iters=50, top_k=10)
            if sc < best_score - 1e-14:
                update(h_new, sc)
    
    # Phase 10: Last resort - annealed + polish
    if not is_deadline():
        remaining = deadline - time.time()
        if remaining > 30:
            h_new, sc = annealed_two_point(best_h, rng, t_start=0.03, t_end=0.002, 
                                            max_iters=int(remaining * 500), patience=1500)
            if sc < best_score - 1e-14:
                update(h_new, sc)
            if not is_deadline():
                h_new2, sc2 = two_point_local_search(best_h, rng, max_passes=40)
                if sc2 < best_score - 1e-14:
                    update(h_new2, sc2)
    
    # Final validation
    final_h = repair_sum(best_h)
    final_score = official_score(final_h)
    final_n = len(final_h)
    
    if not np.isfinite(final_score) or final_n < 4:
        # Fallback
        final_h = np.full(50, 0.5)
        final_n = 50
        final_score = official_score(final_h)
    
    final_bound = final_score + 1e-12
    return final_h, final_bound, final_n
