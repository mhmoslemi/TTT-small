def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - z
            order = np.argsort(-room)
            for i in order:
                add = min(abs(diff), room[i])
                z[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(-z)
            for i in order:
                sub = min(abs(diff), z[i])
                z[i] -= sub
                diff += sub

def score_exact(h):

def gradient_at_lag(h, k, n):

        if m <= 0:
            return g
        g[:m] += (1.0 - h[k:])

def compute_gradient(h, T=0.01, n_active=None):
    """
    Compute gradient of the objective.
    If n_active is None, use soft-max with temperature T.
    If n_active is int, use top n_active lags.
    """
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(overlaps))
    c = n - 1
    
    if n_active is not None:
        idx = np.argsort(-overlaps)[:n_active]
        w = np.zeros(len(overlaps))
        w[idx] = 1.0 / n_active
    else:
        ex = np.exp((overlaps - base) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            mask = overlaps >= base - 1e-10
            w = np.where(mask, 1.0, 0.0)
            w /= w.sum()
        else:
            w = ex / S
    
    g = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[c + k]
        if wk > 1e-15:
            g += wk * dx * gradient_at_lag(h, k, n)
    return g, base

def block_move(h, a, b, delta, n):
    """Add delta to h[a:b], compensate elsewhere to preserve sum."""
    m = b - a
    if m <= 0 or m >= n:
        return None
    rest = n - m
    # h[a:b] += delta, h[rest] += -delta*m/rest
    comp = -delta * m / rest
    if h[a] + delta < -1e-10 or h[b-1] + delta > 1 + 1e-10:
        return None
    # Check if compensation keeps rest in [0,1]
    if np.min(h[:a]) + comp < -1e-10 or np.max(h[:a]) + comp > 1 + 1e-10:
        return None
    if b < n:
        if np.min(h[b:]) + comp < -1e-10 or np.max(h[b:]) + comp > 1 + 1e-10:
            return None
    h_new = h.copy()
    h_new[a:b] += delta
    h_new[:a] += comp
    if b < n:
        h_new[b:] += comp
    return h_new

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)
