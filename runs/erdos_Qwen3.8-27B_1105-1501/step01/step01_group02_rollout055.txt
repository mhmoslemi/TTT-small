def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""

        result[n-1+s] = R[s]  # shift s
        result[n-1-s] = R[N-s]  # shift -s
    return result

    dx = 2.0 / len(h)
    corr = fft_corr(h)

def grad_c5_smooth(h, temperature=0.05):
    """Gradient of smooth max of (corr * dx) w.r.t. h using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    
    # grad[i] = sum_k weights[k] * dx * d(corr[k])/d(h[i])
    # d(corr[k])/d(h[i]) where k is the shift index (0 to 2n-2, representing shift k-(n-1))
    # For shift s = k-(n-1): d/dh[i] [sum_l h[l]*(1-h[l-s])]
    #   = 1 - h[i-s]*(i-s valid) - h[i+s]*(i+s valid)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # Contribution to grad[i] for all i:
        # dR_s/dh[i] = 1 - h[i-s]*1{i-s in [0,n-1]} - h[i+s]*1{i+s in [0,n-1]}
        grad += w  # the "1" part
        # subtract h[i-s] for valid i-s
        if s >= 0:
            grad[:n-s] -= w * h[s:n]  # for i in [0, n-s-1], i-s... wait
            # i-s valid means i >= s, so i ranges from s to n-1
            # h[i-s] for i in [s, n-1] is h[0, 1, ..., n-1-s]
            # So grad[i] -= w * h[i-s] for i in [s, n-1]
            # i.e., grad[s:] -= w * h[:n-s]
            grad[s:] -= w * h[:n-s]
        else:
            # i-s valid means i-s < n, i.e., i < n+s (but s<0 so n+s < n)
            # i-s >= 0 means i >= s (always true since s<0 and i>=0)
            # So i ranges from 0 to n+s-1, h[i-s] = h[i+|s|]
            # grad[i] -= w * h[i+|s|] for i in [0, n+|s|-1]
            abs_s = -s
            grad[:n-abs_s] -= w * h[abs_s:]
        
        if s >= 0:
            # i+s valid means i < n-s
            # grad[i] -= w * h[i+s] for i in [0, n-s-1]
            grad[:n-s] -= w * h[s:]
        else:
            abs_s = -s
            # i+s valid means i >= abs_s (since i+s >= 0 means i >= -s = abs_s)
            # also i+s < n means i < n - abs_s
            # grad[i] -= w * h[i+s] for i in [abs_s, n-1]
            grad[abs_s:] -= w * h[:n-abs_s]
    
    return grad

For the pair-line search with pair (i, j):
- h[i] += t, h[j] -= t
- Feasible: t_lo = max(-h[i], h[j]-1), t_hi = min(1-h[i], h[j])

  So t_lo = max(-h[i], h[j]-1), t_hi = min(1-h[i], h[j])

- For each shift s, R_s changes by t * (dR_s/dh[i] - dR_s/dh[j])
  dR_s/dh[i] = 1 - h[i-s]*1{i-s valid} - h[i+s]*1{i+s valid}
  dR_s/dh[j] = 1 - h[j-s]*1{j-s valid} - h[j+s]*1{j+s valid}

- New R_s(t) = R_s + t * g_s
- Objective: max_s (R_s + t*g_s) * dx
- This is a convex piecewise-linear function of t. Find min over [t_lo, t_hi].

To find the minimum of max of linear functions over an interval:
- The minimum is at a breakpoint or at an endpoint
- Breakpoints: for each pair (a,b), t* = (R_a - R_b) / (g_b - g_a) if g_b ≠ g_a
- Check all t* in [t_lo, t_hi] plus endpoints

def pair_line_search(h, i, j):
    """Find optimal t for moving mass from j to i. Returns (t_opt, new_c5)."""
    n = len(h)
    R = fft_corr(h)  # length 2n-1

        # dR_s/dh[i] = 1 - h[i-s]*1{valid} - h[i+s]*1{valid}

        if 0 <= i+s < n: di -= h[i+s]
        dj = 1.0
        if 0 <= j-s < n: dj -= h[j-s]

        return 0.0, np.max(R)

    # t* = (R[a] - R[b]) / (g[b] - g[a]) for g[b] != g[a]
    # Vectorized: for all pairs
    # Actually, let's just use a grid + refinement approach for simplicity
    # Or: since it's convex, use the fact that the minimum is where the
    # something crosses zero.
    
    # Simple approach: evaluate at breakpoints

    R_expanded = R[:, None]  # (S, 1)
    g_expanded = g[:, None]  # (S, 1)
    
    # This creates S x S matrix which is 139x139 = 19321 entries, fine
    R_diff = R[:, None] - R[None, :]  # (S, S): R[a] - R[b]
    g_diff = g[None, :] - g[:, None]  # (S, S): g[b] - g[a]
    
    valid = np.abs(g_diff) > 1e-12
    t_candidates = np.full_like(R_diff, np.nan)
    t_candidates[valid] = R_diff[valid] / g_diff[valid]
    
    # Keep only those in [t_lo, t_hi]
    t_candidates = t_candidates[valid]
    t_candidates = t_candidates[(t_candidates >= t_lo) & (t_candidates <= t_hi)]
    t_candidates = np.unique(t_candidates)

    t_candidates = np.concatenate([[t_lo, t_hi], t_candidates])
    
    if len(t_candidates) > 500:
        # Too many, sample
        idx = np.linspace(0, len(t_candidates)-1, 500).astype(int)
        t_candidates = t_candidates[idx]
        t_candidates = np.concatenate([[t_lo, t_hi], t_candidates])
    
    # Evaluate f(t) = max_s (R[s] + t*g[s]) at each candidate
    best_t = t_candidates[0]
    best_val = np.max(R + t_candidates[0] * g)
    for t in t_candidates[1:]:
        val = np.max(R + t * g)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fft_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    R = np.fft.irfft(H * np.conj(G), N)
    # R[0] = shift 0, R[1] = shift 1, ..., R[n-1] = shift n-1
    # R[N-1] = shift -1, ..., R[N-(n-1)] = shift -(n-1)
    result = np.zeros(2 * n - 1)
    result[n - 1] = R[0]
    for s in range(1, n):
        result[n - 1 + s] = R[s]
        result[n - 1 - s] = R[N - s]
    return result

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = fft_corr(h)
    return np.max(corr) * dx

def pair_line_search(h, i, j):
    """Find best t for h[i]+=t, h[j]-=t. Returns t_opt."""
    n = len(h)
    S = 2 * n - 1
    R = fft_corr(h)
    
    # Compute g_s
    g = np.zeros(S)
    for s_idx in range(S):
        s = s_idx - (n - 1)
        di = 1.0
        if 0 <= i - s < n:
            di -= h[i - s]
        if 0 <= i + s < n:
            di -= h[i + s]
        dj = 1.0
        if 0 <= j - s < n:
            dj -= h[j - s]
        if 0 <= j + s < n:
            dj -= h[j + s]
        g[s_idx] = di - dj
    
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    
    if t_hi - t_lo < 1e-12:
        return 0.0
    
    # Find breakpoints
    R_diff = R[:, None] - R[None, :]
    g_diff = g[None, :] - g[:, None]
    valid = np.abs(g_diff) > 1e-10
    t_cands = np.where(valid, R_diff / g_diff, np.nan)
    t_cands = t_cands[valid]
    t_cands = t_cands[(t_cands > t_lo) & (t_cands < t_hi)]
    t_cands = np.unique(t_cands)
    
    # Add a few points around the midpoint for robustness
    mid = (t_lo + t_hi) / 2
    t_cands = np.concatenate([t_cands, [t_lo, mid, t_hi]])
    
    if len(t_cands) > 300:
        idx = np.linspace(0, len(t_cands) - 1, 300).astype(int)
        t_cands = t_cands[idx]
    
    best_t = 0.0
    best_val = np.max(R)
    for t in t_cands:
        val = np.max(R + t * g)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

def gradient_step(h, lr, temperature=0.03):
    """One gradient step with smooth max using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[i] = sum_k weights[k]*dx * d(corr[k])/dh[i]
    # d(corr[k])/dh[i] where shift s = k-(n-1):
    #   = 1 - h[i-s]*1{valid} - h[i+s]*1{valid}
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        grad += w
        if s >= 0:
            grad[s:] -= w * h[:n - s]      # h[i-s] for i in [s, n-1]
            grad[:n - s] -= w * h[s:]       # h[i+s] for i in [0, n-s-1]
        else:
            a = -s
            grad[:n - a] -= w * h[a:]       # h[i-s] = h[i+a] for i in [0, n-a-1]
            grad[a:] -= w * h[:n - a]       # h[i+s] = h[i-a] for i in [a, n-1]
    
    return h - lr * grad

def make_structured_initial(n, rng):
    """Create a structured initial pattern."""
    h = np.zeros(n)
    # Use golden ratio for low-discrepancy placement
    phi = (1 + np.sqrt(5)) / 2
    for i in range(n):
        pos = int((i * phi) % n)
        h[pos] = 1.0
    # Set first n//2 to 1
    h = np.zeros(n)
    positions = np.sort(np.array([int((i * phi * n + rng.random()) % n) for i in range(n)]))
    positions = np.unique(positions)
    if len(positions) >= n // 2:
        h[positions[:n // 2]] = 1.0
    else:
        h[np.arange(0, n, 2)] = 1.0
    return project_onto_constraint(h, n / 2.0)
