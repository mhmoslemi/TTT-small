def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_target(v, n):
    """Project v to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(np.asarray(v, dtype=float).ravel()[:n], 0.0, 1.0)
    if len(h) < n:
        h = np.concatenate([h, np.zeros(n - len(h))])
    target = n / 2.0
    # Bisection for shift
    lo, hi = -1.0, 1.0
    # Ensure bracket

        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    # Bisection
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)

        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break

            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    return h

def compute_grad_matrix(h):
    """Compute gradient matrix G where G[r] = d(c_r * dx)/dh"""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for r in range(n_lags):
        s = r - (n - 1)
        if s >= 0:
            # c_s = sum_{j=0}^{n-s-1} h[j+s]*(1-h[j])

import numpy as np

def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.zeros(n)
    v = np.asarray(v, dtype=float).ravel()
    m = min(len(v), n)
    h[:m] = v[:m]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        residual = target - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + residual)
    return h


def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for s in range(n - 1):

        j = np.arange(n - s)
        G[r, j + s] += dx * (1.0 - h[j])
        G[r, j] -= dx * h[j + s]
        r2 = (n - 1) - s
        j2 = np.arange(s, n)
        G[r2, j2 - s] += dx * (1.0 - h[j2])
        G[r2, j2] -= dx * h[j2 - s]
    return G


def lp_direction(h):

    n_vars = 2 * n + 1

    b_ub = []

def gradient_direction(h, active_tol=0.005):

    n_lags = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

        active[np.argmax(corr)] = True
    d = np.zeros(n)
    for r in np.where(active)[0]:
        d -= G[r, :]
    d -= np.mean(d)
    norm = np.max(np.abs(d))
    if norm < 1e-12:
        return None
    return d / norm


def local_search(h, n, time_limit, seed, max_iters=200):
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    start_t = time.perf_counter()
    best_h, best_score = h.copy(), score
    no_improve = 0
    alphas = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0]
    for it in range(max_iters):
        if time.perf_counter() - start_t > time_limit:
            break
        d = lp_direction(h)
        if d is None:
            d = gradient_direction(h)
        if d is None:
            d = rng.normal(0, 1, n)
            d -= np.mean(d)
            norm = np.max(np.abs(d))
            if norm > 1e-12:
                d /= norm
        best_alpha = 0.0
        best_local = score
        for sign in [1, -1]:
            for a in alphas:
                alpha = sign * a
                h_try = project_to_target(h + alpha * d, n)
                s_try = official_score(h_try)
                if s_try < best_local - 1e-12:
                    best_local = s_try
                    best_alpha = alpha
        if best_alpha != 0.0:
            h = project_to_target(h + best_alpha * d, n)
            score = official_score(h)
            no_improve = 0
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                h_try = project_to_target(h + rng.normal(0, 0.05, n), n)
                s_try = official_score(h_try)
                if s_try < score - 1e-12:
                    h = h_try
                    score = s_try
                    no_improve = 0
                    if score < best_score:
                        best_score = score
                        best_h = h.copy()
            if no_improve > 50:
                break
    return best_h, best_score


def slsqp_minimax(h, n, time_limit=30.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    n_lags = 2 * n - 1
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return theta - corr

    def ineq_jac(x):
        h_x = x[:n]
        G = compute_grad_matrix(h_x)
        J = np.zeros((n_lags, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h, [theta0]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(objective, x0, jac=objective_grad,
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 100, 'ftol': 1e-10})
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    return h, initial_score


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return project_to_target(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n)


def gen_random(n, rng):
    h = rng.uniform(0, 1, n)
    return project_to_target(h, n)


def gen_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    if n % 2 == 1:
        h[idx[target]] = 0.5
    return project_to_target(h, n)


def gen_sine(n, rng):
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(2, 10)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.05, n)
    return project_to_target(h, n)


def discrete_search(h, n, time_limit, seed):
    rng = np.random.default_rng(seed)
    h = project_to_target(h, n)
    score = official_score(h)
    best_h, best_score = h.copy(), score
    start_t = time.perf_counter()
    threshold = 0.5
    b = (h >= threshold).astype(float)
    b = project_to_target(b, n)
    b_score = official_score(b)
    if b_score < score:
        h = b
        score = b_score
        best_h = b.copy()
        best_score = b_score
    no_improve = 0
    while time.perf_counter() - start_t < time_limit:
        if no_improve > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        h_try = h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_to_target(h_try, n)
        s_try = official_score(h_try)
        if s_try < score - 1e-12:
            h = h_try
            score = s_try
            no_improve = 0
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            no_improve += 1
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 5.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 10 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = project_to_target(parent, n_parent)
    else:
        n_parent = 64
        best_h = project_to_target(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)

    rng = np.random.default_rng(seed)

    candidates = [best_h.copy()]
    fl = best_h[::-1].copy()
    candidates.append(project_to_target(fl, best_n))
    inv = (1.0 - best_h).copy()
    candidates.append(project_to_target(inv, best_n))
    inv_flip = (1.0 - fl).copy()
    candidates.append(project_to_target(inv_flip, best_n))
    sym = (best_h + fl) / 2.0
    candidates.append(project_to_target(sym, best_n))

    n_list = sorted(set([
        best_n, max(32, best_n - 4), best_n + 2, best_n + 4,
        max(32, best_n - 8), best_n + 8, max(32, best_n + 10)
    ]))
    n_list = [n for n in n_list if 32 <= n <= 140]

    for n_try in n_list[:5]:
        if time.perf_counter() > deadline - 60:
            break
        candidates.append(gen_interpolated(best_h, n_try))
        candidates.append(gen_interpolated(best_h, n_try, offset=1))

    for n_try in n_list[:3]:
        if time.perf_counter() > deadline - 40:
            break
        for _ in range(3):
            candidates.append(gen_random(n_try, rng))
        for _ in range(2):
            candidates.append(gen_binary(n_try, rng))
        for _ in range(2):
            candidates.append(gen_sine(n_try, rng))

    scored = []
    for h_c in candidates:
        if not np.all(np.isfinite(h_c)):
            continue
        h_c = project_to_target(h_c, len(h_c))
        s = official_score(h_c)
        if np.isfinite(s) and s > 0:
            scored.append((s, h_c))

    scored.sort(key=lambda x: (x[0], len(x[1])))
    if len(scored) > 0 and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_h = scored[0][1].copy()
        best_n = len(best_h)

    n_local = min(6, len(scored))
    local_results = []
    for idx in range(n_local):
        if time.perf_counter() > deadline - 100:
            break
        h_c = scored[idx][1].copy()
        n_c = len(h_c)
        remaining = deadline - time.perf_counter()
        t_limit = min(60.0, max(8.0, remaining / max(1, n_local - idx) * 0.6))
        h_loc, s_loc = local_search(h_c, n_c, time_limit=t_limit, seed=seed + idx * 7)
        local_results.append((s_loc, h_loc))
        if s_loc < best_score:
            best_score = s_loc
            best_h = h_loc.copy()
            best_n = n_c

    local_results.sort(key=lambda x: x[0])

    top_k = min(4, len(local_results))
    for idx in range(top_k):
        if time.perf_counter() > deadline - 80:
            break
        s_c, h_c = local_results[idx]
        n_c = len(h_c)
        h_ref, s_ref = slsqp_minimax(h_c, n_c, time_limit=25.0)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref.copy()
            best_n = n_c
        if time.perf_counter() > deadline - 30:
            break
        h_2, s_2 = local_search(h_ref, n_c, time_limit=15.0, seed=seed + 200 + idx)
        if s_2 < best_score:
            best_score = s_2
            best_h = h_2.copy()
            best_n = n_c

    if time.perf_counter() < deadline - 50:
        for n_try in [best_n, best_n + 2, best_n + 4]:
            if time.perf_counter() > deadline - 40:
                break
            h_interp = gen_interpolated(best_h, n_try)
            h_loc, s_loc = local_search(h_interp, n_try, time_limit=20.0, seed=seed + 500)
            if s_loc < best_score:
                best_score = s_loc
                best_h = h_loc.copy()
                best_n = n_try
            h_ref, s_ref = slsqp_minimax(h_loc, n_try, time_limit=15.0)
            if s_ref < best_score:
                best_score = s_ref
                best_h = h_ref.copy()
                best_n = n_try

    if time.perf_counter() < deadline - 30:
        h_disc, s_disc = discrete_search(best_h, best_n, time_limit=10.0, seed=seed + 300)
        if s_disc < best_score:
            best_score = s_disc
            best_h = h_disc.copy()

    if time.perf_counter() < deadline - 20:
        h_fin, s_fin = slsqp_minimax(best_h, best_n, time_limit=12.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin.copy()

    if time.perf_counter() < deadline - 8:
        h_fin2, s_fin2 = local_search(best_h, best_n, time_limit=6.0, seed=seed + 999, max_iters=50)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2.copy()

    best_h = project_to_target(best_h, len(best_h))
    final_score = official_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
