                coeff[j] += (1.0 - h_star[i])
                const += h_star[j] * h_star[i]

        row = np.zeros(n + 1)
        row[:n] = coeff
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-const)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=method)
    
    if res.success:
        return res.x[:n]
    return None

def solve_slp_lp(h_star, n, active_indices):

    base = float(np.max(lags))
    w = np.exp((lags - base) / T)
    S = float(np.sum(w))
    if S < 1e-300:

        w /= max(w.sum(), 1e-300)
    else:
        w /= S
    if K is not None and K < len(lags):
        top_k = np.argsort(w)[-K:]
        w2 = np.zeros(len(lags))
        w2[top_k] = w[top_k]
        w = w2 / max(w2.sum(), 1e-300)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        wk = w[k]

        else:
            m = n + d
            grad[:m] += wk * (1.0 - h[(-d):])
            grad[(-d):] += wk * (-h[:m])

            if nr < 20 or nr > 300:
                continue
            h_r = resample_h(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p5_end - 10, time.time() + 30)
            if time.time() < sub_dl:
                h_r, c_r = slp_optimize(h_r, sub_dl, K=12, rng=rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
                n = len(best_h)

    # Final subgradient polish
    if time.time() < deadline - 2:
        h_f, c_f = subgradient_descent(best_h, deadline - 2, T_start=0.005, T_end=0.0001, step0=0.003, rng=rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final feasibility check
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
