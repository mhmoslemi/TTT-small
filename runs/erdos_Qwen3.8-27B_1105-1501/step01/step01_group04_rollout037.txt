    rng = np.random.RandomState(seed)
    while time.time() - t0 < time_budget:
        r = rng.random()
        if r < 0.4:
            mv = _smart_move(h, corr, n, rng)
        else:
            mv = _random_move(h, n, rng)
        if mv is None:
            continue
        a, b, d = mv
        ha = h[a] + d
        hb = h[b] - d
        if ha < 0 or ha > 1 or hb < 0 or hb > 1:
            continue
        h[a] = ha
        h[b] = hb
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr) * dx)
        if new_c5 < cur_c5:
            corr = new_corr
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = ha - d
            h[b] = hb + d
    return best_h, best_c5

def make_random_init(n, rng):
    h = rng.random(n)
    return fix_sum(h, n / 2.0)

def make_binary_init(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        ones = n // 2
        pos = rng.choice(n, size=ones, replace=False)
        h[pos] = 1.0
    else:
        ones = (n - 1) // 2
        pos = rng.choice(n, size=ones, replace=False)
        h[pos] = 1.0
        rem = np.setdiff1d(np.arange(n), pos)
        h[int(rng.choice(rem))] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass
    # Phase 1: Optimize initial
    if h_init is not None:
        n0 = len(h_init)
        alloc = min(remaining() * 0.5, 550.0)
        if alloc > 20:
            h, c5 = sa_optimize(h_init, alloc, seed)
            best_h, best_c5, best_n = h, c5, n0
    # Phase 2: Try other n values
    n_list = [51, 63, 81, 127, 33, 99]
    for n in n_list:
        if remaining() < 35:
            break
        if best_h is not None and n == best_n:
            continue
        alloc = min(remaining() * 0.1, 80.0)
        if alloc < 15:
            break
        rng = np.random.RandomState(seed + n * 7)
        for it in range(2):
            if remaining() < 12:
                break
            if it == 0:
                h_new = make_binary_init(n, rng)
            else:
                h_new = make_random_init(n, rng)
            sub = min(alloc / 2.0, remaining() - 8.0)
            if sub < 8:
                break
            h, c5 = sa_optimize(h_new, sub, seed + n * 11 + it)
            if c5 < best_c5:
                best_h, best_c5, best_n = h, c5, n
    # Phase 3: Polish best
    if best_h is not None and remaining() > 15:
        alloc = min(remaining() * 0.35, 150.0)
        h, c5 = greedy_polish(best_h, alloc, seed + 5555)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    # Phase 4: Another SA round on best
    if best_h is not None and remaining() > 15:
        alloc = min(remaining() * 0.3, 120.0)
        h, c5 = sa_optimize(best_h, alloc, seed + 7777)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    # Final polish
    if best_h is not None and remaining() > 5:
        h, c5 = greedy_polish(best_h, remaining() * 0.5, seed + 9999)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    # Ensure constraints
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = fix_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    else:
        n = 63
        rng = np.random.RandomState(seed)
        best_h = make_random_init(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n
    return best_h, best_c5, best_n
