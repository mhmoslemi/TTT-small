def project_onto_manifold(v, n):
    target = n / 2.0
    lo = v.min() - 3.0
    hi = v.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += residual / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n

def compute_lag_gradient(h, n, k):
    """Gradient of correlation at lag k with respect to h."""
    j = np.arange(n)

    ok2 = (j + k >= 0) & (j + k < n)
    if ok2.any():
        grad[ok2] += (-h[j[ok2] + k])
    return grad

    dx = 2.0 / n

    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp

        grad += w * compute_lag_gradient(h, n, k)

import time


def project_onto_manifold(v, n):
    """Project v onto {h in [0,1]^n : sum(h) = n/2} via bisection on lambda."""
    target = n / 2.0
    lo = v.min() - 3.0
    hi = v.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            mask = h < 1.0 - 1e-13

            mask = h > 1e-13

    return np.clip(h, 0.0, 1.0)

    ok1 = (j - k >= 0) & (j - k < n)
    if ok1.any():
        grad[ok1] += (1.0 - h[j[ok1] - k])
    ok2 = (j + k >= 0) & (j + k < n)

    best_score = score
    temp = t0

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        r = rng.random()
        h_try = h.copy()
        if r < 0.55:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.uniform(-0.15, 0.15)
                h_try[i] += delta
                h_try[j] -= delta
        elif r < 0.75:
            i = rng.integers(0, n)
            h_try[i] += rng.uniform(-0.25, 0.25)
        elif r < 0.9:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                d = rng.uniform(0.05, 0.3)
                h_try[i] += d
                h_try[j] -= d
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                swap_amt = 0.5 * (h_try[j] - h_try[i])
                h_try[i] += swap_amt
                h_try[j] -= swap_amt

        h_try = project_onto_manifold(h_try, n)
        s_try = score_fn(h_try)

        if s_try <= score or rng.random() < np.exp(-(s_try - score) / max(temp, 1e-9)):
            h = h_try
            score = s_try
            if score < best_score:
                best_score = score
                best_h = h.copy()
        temp *= 0.9995
        if temp < 1e-8:
            temp = 1e-8

    return best_h, best_score


def interpolate_to_n(parent, n):
    m = len(parent)
    if m == n:
        return parent.copy()
    if m == 1:
        return project_onto_manifold(np.full(n, parent[0]), n)
    xs = np.linspace(0.0, 1.0, m)
    xt = np.linspace(0.0, 1.0, n)
    h = np.interp(xt, xs, parent)
    return project_onto_manifold(h, n)


def gen_random_binary(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    if n % 2 == 1:
        h[idx[target]] = 0.5
    return project_onto_manifold(h, n)


def gen_quasiperiodic(n, rng):
    """Generate a quasiperiodic-like pattern."""
    alpha = rng.uniform(0.3, 0.7)
    phase = rng.uniform(0, 1)
    x = np.arange(n) / n
    h = (np.sin(2 * np.pi * (x / alpha + phase)) > 0).astype(float)
    return project_onto_manifold(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 3.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.full(65, 0.5)

    n_parent = len(parent)
    best_h = project_onto_manifold(parent, n_parent)
    best_score = score_fn(best_h)

    # Phase 1: LSE annealing on parent (moderate temp)
    t_rem = deadline - time.perf_counter()
    if t_rem > 50:
        tl = min(45, t_rem * 0.25)
        h1, s1 = anneal_optimize(best_h.copy(), n_parent, t_start=3.0, t_end=60.0,
                                  n_iter=500, time_limit=tl, seed=seed, use_lse=True)
        if s1 < best_score:
            best_score, best_h = s1, h1

    # Phase 2: SA refinement
    t_rem = deadline - time.perf_counter()
    if t_rem > 40:
        tl = min(30, t_rem * 0.2)
        h2, s2 = simulated_anneal(best_h.copy(), len(best_h), rng,
                                   max_iters=8000, time_limit=tl, t0=0.01)
        if s2 < best_score:
            best_score, best_h = s2, h2

    # Phase 3: Active-set optimization (high temp, focus on top lags)
    t_rem = deadline - time.perf_counter()
    if t_rem > 40:
        tl = min(40, t_rem * 0.25)
        n_now = len(best_h)
        h3, s3 = anneal_optimize(best_h.copy(), n_now, t_start=10.0, t_end=300.0,
                                  n_iter=500, time_limit=tl, seed=seed + 200,
                                  use_lse=True, noise_max=0.05)
        if s3 < best_score:
            best_score, best_h = s3, h3

    # Phase 4: Active-set (no LSE) for sharper max targeting
    t_rem = deadline - time.perf_counter()
    if t_rem > 30:
        tl = min(30, t_rem * 0.2)
        n_now = len(best_h)
        h4, s4 = anneal_optimize(best_h.copy(), n_now, t_start=5.0, t_end=50.0,
                                  n_iter=400, time_limit=tl, seed=seed + 400,
                                  use_lse=False, noise_max=0.04)
        if s4 < best_score:
            best_score, best_h = s4, h4

    # Phase 5: Try alternative n values
    n_candidates = sorted(set([len(best_h), len(best_h) - 5, len(best_h) - 10,
                                len(best_h) + 5, len(best_h) + 10, len(best_h) + 15,
                                50, 55, 60, 70, 75, 80]))
    n_candidates = [x for x in n_candidates if 30 <= x <= 130]

    for n_try in n_candidates:
        t_rem = deadline - time.perf_counter()
        if t_rem < 20:
            break
        tl = min(12, t_rem * 0.12)
        h_interp = interpolate_to_n(best_h, n_try)
        h_opt, s_opt = anneal_optimize(h_interp, n_try, t_start=5.0, t_end=80.0,
                                        n_iter=250, time_limit=tl, seed=seed + n_try,
                                        use_lse=True, noise_max=0.06)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt

    # Phase 6: Random restarts at current best n
    n_best = len(best_h)
    t_rem = deadline - time.perf_counter()
    n_restarts = min(6, max(1, int(t_rem / 25)))
    for ri in range(n_restarts):
        t_rem = deadline - time.perf_counter()
        if t_rem < 15:
            break
        # Generate diverse starting point
        if ri % 3 == 0:
            h_r = gen_random_binary(n_best, rng)
        elif ri % 3 == 1:
            h_r = gen_quasiperiodic(n_best, rng)
        else:
            h_r = interpolate_to_n(best_h, n_best)
            h_r = project_onto_manifold(h_r + 0.2 * rng.normal(0, 1, n_best), n_best)
        tl = min(15, t_rem * 0.25)
        h_opt, s_opt = anneal_optimize(h_r, n_best, t_start=5.0, t_end=100.0,
                                        n_iter=200, time_limit=tl, seed=seed + 1000 + ri,
                                        use_lse=True, noise_max=0.1)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt

    # Phase 7: SA polish on best found
    t_rem = deadline - time.perf_counter()
    if t_rem > 15:
        tl = min(20, t_rem * 0.3)
        h6, s6 = simulated_anneal(best_h.copy(), len(best_h), np.random.default_rng(seed + 900),
                                   max_iters=8000, time_limit=tl, t0=0.006)
        if s6 < best_score:
            best_score, best_h = s6, h6

    # Phase 8: Final high-temp LSE polish
    t_rem = deadline - time.perf_counter()
    if t_rem > 10:
        tl = min(15, t_rem * 0.4)
        n_fin = len(best_h)
        h7, s7 = anneal_optimize(best_h.copy(), n_fin, t_start=30.0, t_end=500.0,
                                  n_iter=400, time_limit=tl, seed=seed + 2000,
                                  use_lse=True, noise_max=0.03)
        if s7 < best_score:
            best_score, best_h = s7, h7

    # Phase 9: Tiny SA to squeeze out last bits
    t_rem = deadline - time.perf_counter()
    if t_rem > 4:
        h8, s8 = simulated_anneal(best_h.copy(), len(best_h), np.random.default_rng(seed + 3000),
                                   max_iters=3000, time_limit=max(1.0, t_rem - 1.0), t0=0.003)
        if s8 < best_score:
            best_score, best_h = s8, h8

    # Final validation
    best_h = project_onto_manifold(best_h, len(best_h))
    final_score = score_fn(best_h)

    return (best_h, float(final_score), int(len(best_h)))
