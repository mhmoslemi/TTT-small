    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):

        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_set(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(v)
    # Bisection on lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h < 1.0
            if np.any(mask):
                count = np.sum(mask)
                add = residual / count
                h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                count = np.sum(mask)
                sub = -residual / count
                h[mask] = np.maximum(h[mask] - sub, 0.0)
    return h

def compute_all_corr_scaled(h):
    """Compute all correlation values scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def lse_smoothedScore(h, tau):
    corr = compute_all_corr_scaled(h)
    m = np.max(corr)
    if m < 1e-12:
        return m, np.zeros_like(corr)
    shifted = tau * (corr - m)
    # Numerical stability
    max_shift = np.max(shifted)
    exp_vals = np.exp(shifted - max_shift)
    log_sum = max_shift + np.log(np.sum(exp_vals))
    lse = m + log_sum / tau
    weights = exp_vals / np.sum(exp_vals)
    return lse, weights

def lse_gradient(h, tau):
    """Compute gradient of LSE smoothed score."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_all_corr_scaled(h)
    m = np.max(corr)
    if m < 1e-12:
        return np.zeros(n)
    shifted = tau * (corr - m)
    max_shift = np.max(shifted)
    exp_vals = np.exp(shifted - max_shift)
    weights = exp_vals / np.sum(exp_vals)
    
    lags = np.arange(-(n-1), n)
    p = np.arange(n)
    grad = np.zeros(n)
    
    for ki in range(len(lags)):
        w = weights[ki]
        if w < 1e-12:
            continue
        k = lags[ki]
        ok1 = (p - k >= 0) & (p - k < n)
        grad[ok1] += w * dx * (1.0 - h[p[ok1] - k])
        ok2 = (p + k >= 0) & (p + k < n)
        grad[ok2] += w * dx * (-h[p[ok2] + k])
    
    return grad

def lse_optimize(h, n, tau_schedule, max_iters_per_tau=200, time_limit=30.0):
    """Optimize using LSE smoothed objective with increasing tau."""
    start_time = time.perf_counter()
    h = repair_vector(h, n)
    target_sum = n / 2.0
    best_score = official_score(h)
    best_h = h.copy()
    
    for tau in tau_schedule:
        if time.perf_counter() - start_time > time_limit:
            break
        
        lr = 1.0 / (tau * 2.0)  # initial learning rate
        h = repair_vector(h, n)
        
        for it in range(max_iters_per_tau):
            if time.perf_counter() - start_time > time_limit:
                break
            
            grad = lse_gradient(h, tau)
            # Center the gradient (since sum constraint)
            grad -= np.mean(grad)
            
            # Projected gradient step with backtracking
            # Direction: -grad (projected)
            direction = -grad
            # Project direction to satisfy sum constraint
            direction -= np.mean(direction)
            
            # Backtracking line search
            best_lr = 0
            best_fun = lse_smoothedScore(h, tau)[0]
            lr_try = lr
            for _ in range(10):
                h_new = h + lr_try * direction
                h_new = project_onto_set(h_new, target_sum)
                # Check box constraints
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = repair_vector(h_new, n)
                fun_new = lse_smoothedScore(h_new, tau)[0]
                if fun_new < best_fun - 1e-10:
                    best_fun = fun_new
                    best_lr = lr_try
                    h = h_new
                    break
                lr_try *= 0.5
            
            # Update best
            s = official_score(h)
            if s < best_score - 1e-12:
                best_score = s
                best_h = h.copy()
        
        # Decrease learning rate for next tau
        lr *= 0.5
    
    return best_h, best_score

def pair_descent(h, n, max_sweeps=5, time_limit=20.0):
    """Exact pair exchange descent on the true max objective."""
    start_time = time.perf_counter()
    h = repair_vector(h, n)
    score = official_score(h)
    target_sum = n / 2.0
    
    for sweep in range(max_sweeps):
        if time.perf_counter() - start_time > time_limit:
            break
        
        improved = False
        # Try pairs in random order
        rng = np.random.default_rng(sweep * 1000 + 42)
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        
        # Limit number of pairs per sweep for time management
        max_pairs = min(len(pairs), 500)
        
        for i, j in pairs[:max_pairs]:
            if time.perf_counter() - start_time > time_limit:
                break
            
            a, b = h[i], h[j]
            s = a + b
            
            # t ranges so that a+t ∈ [0,1] and b-t ∈ [0,1]
            t_lo = max(-a, b-1)
            t_hi = min(1-a, b)
            
            if t_lo >= t_hi:
                continue
            
            # Golden section search for best t
            # Minimize max_k c_k(t)
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            
            def obj_at_t(t):
                h[i] = a + t
                h[j] = b - t
                return official_score(h)
            
            # But we need to be careful: we're modifying h in place
            # Let's compute without modifying
            def score_at_t(t):
                h_try = h.copy()
                h_try[i] = a + t
                h_try[j] = b - t
                return official_score(h_try)
            
            # This is expensive (copy + full score). Let's limit evaluations.
            # Instead, do a simple 3-point check
            candidates_t = [0, (t_hi - t_lo) * 0.25, (t_hi - t_lo) * 0.5, (t_hi - t_lo) * 0.75]
            # Plus the current t=0
            best_t = 0
            best_s = score
            for t in candidates_t:
                if t < t_lo or t > t_hi:
                    continue
                s_try = score_at_t(t)
                if s_try < best_s - 1e-13:
                    best_s = s_try
                    best_t = t
            
            if best_t != 0 and best_s < score - 1e-13:
                h[i] = a + best_t
                h[j] = b - best_t
                h = repair_vector(h, n)
                score = best_s
                improved = True
        
        if not improved:
            break
    
    return h, official_score(h)

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def gen_binary_optimized(n, rng, max_iters=500):

def official_score(h):

    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
