import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    y = np.clip(y, 0.0, 1.0)
    if abs(y.sum() - target) < 1e-10:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-(1.0 - z))
            added = 0.0
            for idx in order:
                add = min(diff - added, 1.0 - z[idx])
                if add <= 0:
                    break
                z[idx] += add
                added += add
                if added >= diff - 1e-12:
                    break
        else:
            order = np.argsort(z)
            removed = 0.0
            for idx in order:
                rem = min(-diff - removed, z[idx])
                if rem <= 0:
                    break
                z[idx] -= rem
                removed += rem
                if removed >= -diff - 1e-12:
                    break
    return np.clip(z, 0.0, 1.0)

def fourier_eval(params, n):
    """Evaluate h from Fourier parameters.
    params = [a1, b1, a2, b2, ..., aK, bK, M]
    """
    K = (len(params) - 1) // 2
    M = params[-1]
    a = params[:K]
    b = params[K:2*K]
    i = np.arange(n)
    f = 0.5 * np.ones(n)
    for k in range(K):
        f += a[k] * np.cos(2.0 * np.pi * (k+1) * i / n) + b[k] * np.sin(2.0 * np.pi * (k+1) * i / n)
    h = 1.0 / (1.0 + np.exp(-M * (f - 0.5)))
    h = make_feasible(h, n)
    return h

    """CEM optimization in Fourier space."""
    n_param = 2 * K + 1
    lb = np.array([-0.3]*K + [-0.3]*K + [2.0])
    ub = np.array([0.3]*K + [0.3]*K + [25.0])
    
    # Initialize
    samples = np.zeros((50, n_param))
    for j in range(50):
        samples[j] = lb + (ub - lb) * rng.random(n_param)
    
    # Evaluate
    scores = np.zeros(50)
    for j in range(50):
        h = fourier_eval(samples[j], n)
        scores[j] = score_exact(h)
    
    best_score = scores.min()
    best_params = samples[scores.argmin()].copy()
    
    while time.time() < deadline:
        # Select top 10%
        n_sel = max(5, 50 // 10)
        idx = np.argsort(scores)[:n_sel]
        mu = samples[idx].mean(axis=0)
        sigma = samples[idx].std(axis=0)
        
        # Generate new samples
        samples = mu + sigma * rng.standard_normal((50, n_param))
        samples = np.clip(samples, lb, ub)
        
        # Evaluate
        for j in range(50):
            h = fourier_eval(samples[j], n)
            scores[j] = score_exact(h)
        
        if scores.min() < best_score:
            best_score = scores.min()
            best_params = samples[scores.argmin()].copy()
    
    h_best = fourier_eval(best_params, n)
    return h_best, score_exact(h_best)
