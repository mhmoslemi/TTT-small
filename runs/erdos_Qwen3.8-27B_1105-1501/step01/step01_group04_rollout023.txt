def project_feasible(h, target_sum):

def slsqp_polish(h0, time_budget, K=30, seed=42):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h = np.clip(h0, 0, 1).copy()
    # Ensure feasibility
    h = project_feasible(h, target_sum)
    c5_init = compute_c5_exact(h)
    
    best_h = h.copy()
    best_c5 = c5_init
    
    # Identify active constraints
    corr = compute_all_corrs_scaled(h)
    active_indices = np.argsort(corr)[-K:]
    
    t0 = time.time()
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def eq_fun(x):
        return np.sum(x[1:]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    for k_idx in active_indices:
        d = int(k_idx) - (n - 1)
        
        def ineq_fun(x, k_idx=k_idx, d=d):
            h_x = x[1:]
            c_k = np.correlate(h_x, 1.0 - h_x, mode='full')[k_idx] * dx
            return x[0] - c_k
        
        def ineq_jac(x, k_idx=k_idx, d=d):
            h_x = x[1:]
            g = np.zeros(n + 1)
            g[0] = 1.0
            # -dx * (1 - h[i-d] - h[i+d])
            for i in range(n):
                val = 1.0
                if 0 <= i - d < n:
                    val -= h_x[i - d]
                if 0 <= i + d < n:
                    val -= h_x[i + d]
                g[i + 1] = -dx * val
            return g
        
        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    bounds = [(0, None)] + [(0, 1)] * n
    
    x0 = np.zeros(n + 1)
    x0[0] = c5_init + 0.01  # Start slightly above current C5
    x0[1:] = h
    
    result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-12})
    
    if result.success and result.x[0] < best_c5:
        h_new = np.clip(result.x[1:], 0, 1)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5_exact(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

Given current h, and a pair (i, j) (decrease h[i], increase h[j] by eps):
- For each lag k: Δc_k = eps * (a_k - b_k) where a_k = h[i+d_k] - h[i-d_k], b_k = h[j+d_k] - h[j-d_k]

- New max = max_k (c_k + eps * (a_k - b_k))

To minimize the new max:
- If (a_k - b_k) < 0 for all k: the max decreases linearly. Optimal eps = min(h[i], 1-h[j]) (or limited by the point where some constraint becomes the new max)
- If (a_k - b_k) > 0 for some k: the max increases for those k. Optimal eps = 0 (don't make the move) or a small value where improving and worsening constraints balance.

More precisely, define f(eps) = max_k (c_k + eps * delta_k) where delta_k = a_k - b_k.
f is a piecewise linear convex function of eps. Its minimum is at:
- eps = 0 if all delta_k >= 0 (f is non-decreasing)
- Otherwise, the minimum is at the smallest eps where the constraint with the most negative delta_k would be "caught up" by a constraint with positive delta_k, or at the boundary.

Actually, f(eps) is the max of linear functions, so it's convex. The minimum of a convex piecewise linear function on [0, eps_max] is either at 0, at eps_max, or at a breakpoint.
