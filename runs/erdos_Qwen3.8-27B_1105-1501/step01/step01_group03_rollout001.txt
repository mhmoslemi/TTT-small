    n = len(h)

    A_ub = np.zeros((2*n-1, nvar))

        A_abs[2*i+1, i] = -1.0
        A_abs[2*i+1, n+i] = -1.0

    A_eq = np.zeros((1, nvar))
    A_eq[0, 0:n] = 1.0
    b_eq = np.zeros(1)
    # Bounds
    lo = np.zeros(nvar)
    up = np.zeros(nvar)
    for i in range(n):
        lo[i] = max(-rad, -h[i])
        up[i] = min(rad, 1.0 - h[i])
    for i in range(n):
        lo[n+i] = 0.0
        up[n+i] = rad + 1e-6
    lo[2*n] = 0.0
    up[2*n] = 2.0
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                     bounds=list(zip(lo, up)), method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[0:n].copy()
            # Center d to exactly satisfy sum=0
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None

def lp_descent(h, max_iters=50, rad_init=0.3, mu=0.05, time_limit=30.0):
    """Run LP-based local descent."""
    n = len(h)
    h = project_feasible(h.copy())
    cur_score = exact_score(h)
    rad = rad_init
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if rad < 1e-6:
            break
        
        C = compute_all_C(h)
        G = build_grad_matrix(h, n)
        
        d = lp_local_step(h, C, G, rad, mu)
        if d is None:
            rad *= 0.5
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_new = project_feasible(h + alpha * d)
            s_new = exact_score(h_new)
            if s_new < cur_score - 1e-14:
                h = h_new
                cur_score = s_new
                rad = min(rad * 1.1, 0.8)
                improved = True
                break
        
        if not improved:
            rad *= 0.4
    
    return h, cur_score

def binary_threshold(h, rng):
    """Create a binary candidate from h by thresholding."""
    n = len(h)
    target = n // 2
    if n % 2 == 0:
        # Pick top n/2 entries
        idx = np.argsort(h)[::-1][:target]
        b = np.zeros(n)
        b[idx] = 1.0
    else:
        # For odd: pick top (n-1)/2 as 1, one as 0.5, rest 0
        idx = np.argsort(h)[::-1]
        b = np.zeros(n)
        b[idx[:target]] = 1.0
        b[idx[target]] = 0.5
    return project_feasible(b)

def simulate_anneal_binary(h, n_swaps=5000, t_start=0.01, t_end=0.001, rng=np.random.default_rng(0)):
    """Simulated annealing on binary (or near-binary) values."""
    n = len(h)
    h = project_feasible(h.copy())
    # First, make it binary-ish
    h = binary_threshold(h, rng)
    cur_score = exact_score(h)
    best_h = h.copy()
    best_score = cur_score
    
    m = int(np.round(np.sum(h)))  # number of 1s
    
    T = t_start
    ratio = (t_end / t_start) ** (1.0 / max(n_swaps, 1))
    
    for it in range(n_swaps):
        T *= ratio
        # Pick two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Try to swap: if h[i]>0.5 and h[j]<0.5, swap them
        # Or if both are fractional, try moving mass
        if h[i] > 0.5 and h[j] < 0.5:
            h[i] -= 1.0
            h[j] += 1.0
            s = exact_score(h)
            if s < cur_score:
                cur_score = s
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
            else:
                delta = s - cur_score
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    cur_score = s
                else:
                    h[i] += 1.0
                    h[j] -= 1.0
        elif h[i] < 0.5 and h[j] > 0.5:
            # Same swap, different direction
            h[i] += 1.0
            h[j] -= 1.0
            s = exact_score(h)
            if s < cur_score:
                cur_score = s
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
            else:
                delta = s - cur_score
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    cur_score = s
                else:
                    h[i] -= 1.0
                    h[j] += 1.0
    
    return best_h, best_score

def interp_to_length(h_src, n_tgt):
    """Interpolate h from n_src to n_tgt preserving integral."""
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_tgt)

def symmetry_transforms(h):
    """Generate symmetry-related variants."""
    n = len(h)
    results = []
    # Complement + reverse: 1 - h[::-1]
    results.append(project_feasible(1.0 - h[::-1]))
    # Just reverse
    results.append(project_feasible(h[::-1]))
    # Complement
    results.append(project_feasible(1.0 - h))
    # Roll by various amounts
    for shift in [1, n//3, n//2, (n-1)//4]:
        results.append(project_feasible(np.roll(h, shift)))
    return results
