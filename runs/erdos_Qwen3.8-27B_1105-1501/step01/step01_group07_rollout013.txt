    n_scores = len(scores)
    s_max = float(np.max(scores))

    n_active = max(5, int(n_scores * active_frac))
    top_idx = np.argsort(scores)[-n_active:]
    
    # Compute gradients for active scores
    # s[p] = dx * sum_i h[i+k]*(1-h[i]) where k = p - n + 1

        constraints.append(linearized <= t_var)

        except:
            return h, score_exact(h)
    
    if h_var.value is None:
        return h, score_exact(h)
    
    h_new = make_feasible(h_var.value, n)
    c_new = score_exact(h_new)
    c_old = score_exact(h)
    
    if c_new < c_old:
        return h_new, c_new
    return h, c_old

                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        best_s = score_exact(h)
        if best_s >= cur_c:
            h[i] = orig_i
            h[j] = orig_j
            return None
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=20):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def soft_grad_step(h, T, deadline, max_iters=100):
    """Soft-max gradient descent with adaptive T."""
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    step = 0.02
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        base = float(np.max(scores))
        ex = np.exp((scores - base) / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            mask = scores >= base - 1e-10
            w = np.where(mask, 1.0, 0.0)
            w /= w.sum()
        else:
            w = ex / S
        
        # Compute gradient
        g = np.zeros(n)
        nfull = 2 * n - 1
        c = n - 1
        for d in range(0, nfull):
            wd = w[d] * dx
            if wd < 1e-15:
                continue
            k = d - n + 1
            for j in range(n):
                a = j - k
                if 0 <= a < n:
                    g[j] += wd * (1.0 - cur[a])
                b = j + k
                if 0 <= b < n:
                    g[j] -= wd * cur[b]
        
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        
        direction = -g / gn
        improved = False
        for s_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * s_mult
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        
        if improved:
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.7
            if step < 0.001:
                break
        
        T = max(0.001, T * 0.99)
    
    return best, best_c
