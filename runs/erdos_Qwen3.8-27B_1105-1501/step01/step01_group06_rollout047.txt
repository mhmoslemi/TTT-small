    # For lag d >= 0: score[d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
    # grad[j] for lag d: (1-h[j+d])*[j <= n-1-d] - h[j-d]*[j >= d]
    # For lag d < 0, g = -d: score[-g] = sum_{i=g}^{n-1} h[i]*(1-h[i-g])

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_to_target(v, target):
    """Project v to [0,1] with sum = target using bisection."""
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(150):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Repair exact sum
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-12:
        # Sequential repair
        if residual > 0:
            for i in range(n):
                if x[i] < 1.0:
                    add = min(1.0 - x[i], residual)
                    x[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            for i in range(n):
                if x[i] > 0.0:
                    rem = 0.0
                    rem = min(x[i], -residual)
                    x[i] -= rem
                    residual += rem
                    if abs(residual) < 1e-12:
                        break
    return x

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_to_target(v, target):
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(150):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-11:
        if resid > 0:
            for i in range(n):
                if x[i] < 1.0:
                    add = min(1.0 - x[i], resid)
                    x[i] += add
                    resid -= add
                    if abs(resid) < 1e-12:
                        break
        else:
            for i in range(n):
                if x[i] > 0.0:
                    rem = min(x[i], -resid)
                    x[i] -= rem
                    resid += rem
                    if abs(resid) < 1e-12:
                        break
    return x

def compute_grads(h, n):
    """All 2n-1 gradients, each of length n."""
    grads = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            j1 = np.arange(n - d)
            grads[r, j1] += 1.0 - h[j1 + d]
            j2 = np.arange(d, n)
            if len(j2) > 0:
                grads[r, j2] -= h[j2 - d]
        else:
            g = -d
            j1 = np.arange(g, n)
            if len(j1) > 0:
                grads[r, j1] += 1.0 - h[j1 - g]
            j2 = np.arange(n - g)
            if len(j2) > 0:
                grads[r, j2] -= h[j2 + g]
    return grads

def lp_direction(h, n, R):
    """Find LP trust-region direction."""
    scores = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_grads(h, n)
    # Include all lags
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i in range(m):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = -scores[i]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-R, -h[i])
        hi = min(R, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res is None or res.status != 0:
        return None
    p = res.x[:n]
    return p

def subgrad_direction(h, n, R, K=10):
    """Subgradient fallback direction."""
    scores = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(scores)[::-1][:K]
    weights = np.ones(K) / K
    g = np.zeros(n)
    grads = compute_grads(h, n)
    for i, lag in enumerate(idx):
        g += weights[i] * grads[lag]
    p = -g
    p -= np.mean(p)  # enforce sum = 0
    max_abs = np.max(np.abs(p))
    if max_abs > R:
        p *= R / max_abs
    elif max_abs < 1e-15:
        return None
    return p

def line_search(h, p, n):
    """Exact 1D line search along direction p."""
    # Max feasible alpha
    alpha_max = 1.0
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    if alpha_max <= 1e-14:
        return 0.0, None
    current_c5 = eval_c5(h)
    best_c5 = current_c5
    best_alpha = 0.0
    best_h = None
    # Grid search
    n_samp = 25
    alphas = np.concatenate([[0.01 * alpha_max],
                            np.linspace(0.05 * alpha_max, alpha_max, n_samp - 1)])
    for a in alphas:
        h_new = h + a * p
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_alpha = a
            best_h = h_new.copy()
    # Golden section refinement
    if best_alpha > 0.05 * alpha_max and best_alpha < 0.95 * alpha_max and best_h is not None:
        lo = best_alpha * 0.8
        hi = best_alpha * 1.2
        if lo < 0:
            lo = 0.001 * alpha_max
        if hi > alpha_max:
