            grad[:ell] += wk * dx * (1.0 - h[s:])
            grad[s:] -= wk * dx * h[:ell]
        else:
            t = -s
            ell = n - t
            if ell <= 0:
                continue
            grad[t:] += wk * dx * (1.0 - h[:ell])
            grad[:ell] -= wk * dx * h[t:]
    
    return phi, grad

def lse_descent(h0, betas, deadline, max_iter_per_beta=200):
    """Projected gradient descent on LSE objective with beta continuation."""
    h = h0.copy()
    best_h = h.copy()
    best_score = official_score(h)
    
    for beta in betas:
        lr = 0.5 / (beta + 1)
        prev_phi = compute_lse_grad(h, beta)[0]
        for it in range(max_iter_per_beta):
            if time.time() > deadline:
                break
            phi, grad = compute_lse_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Backtracking
            accepted = False
            for _ in range(20):
                h_cand = project_feasible(h - lr * grad)
                phi_cand = compute_lse_grad(h_cand, beta)[0]
                if phi_cand <= phi + 1e-14:
                    h = h_cand
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                break
            
            # Check exact score periodically
            if it % 10 == 0 or it == max_iter_per_beta - 1:
                sc = official_score(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
        
        # After each beta, check exact
        sc = official_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

    best_h = h.copy()
    best_score = official_score(h)
    n = len(h)
    dx = 2.0 / n
    lr = 0.01

        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)

        sg = np.zeros(n)
        if s >= 0:
            ell = n - s
            sg[:ell] += dx * (1.0 - h[s:])
            sg[s:] -= dx * h[:ell]
        else:
            t = -s
            ell = n - t
            sg[t:] += dx * (1.0 - h[:ell])
            sg[:ell] -= dx * h[t:]
        
        h_new = project_feasible(h - lr * sg)
        sc = official_score(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.05)
        else:
            h = h_new  # still move (might help escape)
            lr *= 0.95
        
        if lr < 1e-5:
            break
    
    return best_h, best_score

    n_src = len(h_src)

def official_score(h):

def project_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.where(np.isfinite(z), z, 0.5)

        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
            if abs(residual) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)
