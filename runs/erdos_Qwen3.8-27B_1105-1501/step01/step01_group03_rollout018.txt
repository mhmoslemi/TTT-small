import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    val = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    return float(val)

def project_feasible(z):
    """Project z to the feasible set [0,1]^n with sum = n/2."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection to find shift mu such that sum(clip(z+mu, 0, 1)) = target
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error evenly among non-saturated entries
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count > 0:
            h[mask] += err / count
    return np.clip(h, 0.0, 1.0)

def compute_C_k(h, k):
    """Compute C_k = dx * sum(h[i] * (1-h[i+k])) for given lag k."""
    n = len(h)
    dx = 2.0 / n
    start = max(0, -k)
    end = min(n, n - k)
    if start >= end:
        return 0.0
    return dx * np.sum(h[start:end] * (1.0 - h[start + k:end + k]))

def compute_all_C(h):
    """Compute all C_k for k = -(n-1) to n-1. Returns array of length 2n-1."""
    n = len(h)
    C = np.zeros(2 * n - 1)
    for idx, k in enumerate(range(-(n-1), n)):
        C[idx] = compute_C_k(h, k)
    return C

def compute_grad_C_k(h, k):
    """Compute gradient of C_k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    for i in range(start, end):
        j = i + k
        grad[i] += dx * (1.0 - h[j])
        grad[j] += dx * (-h[i])
    return grad

def compute_all_grads(h):
    """Compute gradients of all C_k. Returns (2n-1, n) array."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for idx, k in enumerate(range(-(n-1), n)):
        G[idx] = compute_grad_C_k(h, k)
    return G

def lp_step(h, radius, lam=0.01):
    """Solve the LP subproblem and return a step direction d."""
    n = len(h)
    C = compute_all_C(h)
    G = compute_all_grads(h)  # shape (2n-1, n)
    cmax = np.max(C)
    
    # Variables: d_0, ..., d_{n-1}, t  (n+1 variables)
    # Minimize: t + lam * sum(|d_i|) -- use auxiliary variables for |d_i|
    # Actually, let's use a simpler formulation: minimize t
    # with a trust region on d
    
    # For linprog: minimize c @ x subject to A_ub @ x <= b_ub, A_eq @ x == b_eq, bounds
    # x = [d_0, ..., d_{n-1}, t]
    
    # Objective: minimize t (c = [0, ..., 0, 1])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: C_k + G_k @ d <= t  =>  G_k @ d - t <= -C_k
    A_ub = np.zeros((2*n-1, n+1))
    for i in range(2*n-1):
        A_ub[i, :n] = G[i]
        A_ub[i, n] = -1.0
    b_ub = -C
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: d_i in [-h_i, 1-h_i] (also apply trust region)
    d_lo = np.maximum(-h, -radius)
    d_hi = np.minimum(1.0 - h, radius)
    bounds = list(zip(d_lo, d_hi))
    bounds.append((0.0, cmax + radius + 0.1))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            return d
    except Exception:
        pass
    
    # Fallback: try with relaxed bounds
    try:
        bounds2 = [(-radius, radius)] * n + [(0.0, 2.0)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds2, method='highs')
        if res.success:
            d = res.x[:n]
            # Clip to feasible
            d = np.clip(d, -h, 1.0 - h)
            return d
    except Exception:
        pass
    
    return None

def lp_refine(h, max_iters=50, radius_init=0.3, time_limit=None, t_start=None):

    h = project_feasible(h.copy())
    score = exact_score(h)
    radius = radius_init
    best_h = h.copy()
    best_score = score
    
    for it in range(max_iters):

                break
        
        d = lp_step(h, radius)
        if d is None or np.max(np.abs(d)) < 1e-10:
            radius *= 0.5

            continue
        
        # Line search

            s_trial = exact_score(h_trial)
            if s_trial < score - 1e-12:
                h = h_trial
                score = s_trial
                found = True
                break
        
        if found:

                best_score = score
                best_h = h.copy()
            radius = min(radius * 1.1, 0.5)
        else:
            radius *= 0.5
            if radius < 1e-6:
                break
    
    return best_h, best_score

    cool = 0.999

        if time.time() - t0 > time_limit_s:

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.array(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -np.max(z) - 1
    hi = 1.0 - np.min(z) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    err = target - np.sum(h)
    if abs(err) > 1e-13:
        if err > 0:
            free = h < (1 - 1e-12)
        else:
            free = h > 1e-12
        nf = free.sum()
        if nf > 0:
            h[free] += err / nf
    return np.clip(h, 0, 1)

def all_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_for_k(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    s = max(0, -k)
    e = min(n, n - k)
    for i in range(s, e):
        j = i + k
        g[i] += dx * (1 - h[j])
        g[j] += dx * (-h[i])
    return g

def all_grads(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for idx, k in enumerate(range(-(n-1), n)):
        G[idx] = grad_for_k(h, k)
    return G

def lp_solve(h, radius, time_check=None):
    n = len(h)
    C = all_C(h)
    G = all_grads(h)
    cmax = C.max()
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2*n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -C
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    d_lo = np.maximum(-h, -radius)
    d_hi = np.minimum(1 - h, radius)
    bounds = list(zip(d_lo, d_hi)) + [(0, cmax + radius + 0.5)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def lp_refine(h0, max_iters=100, r0=0.3, t_limit=None, t_start=None):
    h = project_feasible(h0.copy())
    s = exact_score(h)
    best_h, best_s = h.copy(), s
    r = r0
    stall = 0
    
    for _ in range(max_iters):
        if t_limit is not None and time.time() - t_start > t_limit:
            break
        d = lp_solve(h, r)
        if d is None or np.abs(d).max() < 1e-12:
            r *= 0.5
            stall += 1
            if stall > 10 or r < 1e-7:

            continue
        
        improved = False
        for alpha in (1.0, 0.6, 0.4, 0.25, 0.1, 0.04):
            ht = project_feasible(h + alpha * d)
            st = exact_score(ht)
            if st < s - 1e-13:

                improved = True
                break
        
        if improved:

            r = min(r * 1.15, 0.5)
            stall = 0
        else:

            stall += 1
            if stall > 15 or r < 1e-7:
                break
    
    return best_h, best_s

- Actually, the sum constraint is: new_sum = old_sum + (new_h[i] - h[i]) = n/2 + (new_h[i] - h[i]).
- For the sum to remain n/2, we need new_h[i] = h[i]. That's a problem!

The sum constraint couples all coordinates. So we can't do pure coordinate descent without also adjusting another coordinate.

One approach: "paired coordinate descent". For a pair (i, j):
- Fix all h[k] for k≠i,j.
- We have two degrees of freedom: h[i] and h[j], subject to h[i] + h[j] = const (from the sum constraint).
- So h[j] = const - h[i], and we optimize over h[i] ∈ [max(0, const-1), min(1, const)].
- The objective max_k C_k as a function of h[i] is the max of linear functions (since each C_k is linear in h[i] when h[j] is determined by the sum constraint... wait, is it?).

C_k(h) = dx * Σ_{a} h[a](1-h[a+k]). If we write h[j] = S - h[i] where S is a constant:
- Terms involving h[i]: h[i]*(1-h[i+k]) or h[i-j]*(1-h[i]) etc.
- Terms involving h[j]: (S-h[i])*(1-h[j+k]) or h[j-m]*(1-(S-h[i])) etc.

So C_k as a function of h[i] is: (linear in h[i]) + (linear in h[i]) = linear in h[i]. 

Wait, let me be more careful. C_k contains terms of the form h[a]*h[b] (with signs). If a = i and b = j (or vice versa), the term is h[i]*h[j] = h[i]*(S-h[i]) = S*h[i] - h[i]², which is NOT linear!

So paired coordinate descent doesn't give a convex subproblem in general. It's a quadratic. But a one-variable quadratic is easy to minimize (or maximize, in our case we want to minimize the max of quadratics).

This is getting complicated. Let me stick with the LP approach.

Actually, I realize I should also try a simple approach that might work: **gradient descent with barzilai-borwein step size**.

The "gradient" of f(h) = max_k C_k(h) at h is any vector in conv{∇C_k(h) : k ∈ A(h)}. The "steepest descent direction" is the point in this convex hull closest to the origin.

Finding the steepest descent direction is a QP:
minimize ||g||² subject to g = Σ_{k∈A} λ_k ∇C_k(h), Σλ_k = 1, λ_k ≥ 0

For |A| active constraints, this is a small QP. Then we do a line search.

This is the "minimum-norm subgradient" method, which can be more efficient than plain subgradient descent.

But implementing this correctly is more complex than the LP approach. Let me stick with LP.

One more thought: I can also try **Nesterov's accelerated gradient** or **FISTA** on a smoothed version of the max. But the previous code already tried this (with the softmax smoothing) and got 0.381021. The LP approach should do better because it doesn't introduce smoothing bias.

def exact_score(h):
    n = len(h)
    if n < 2:
        return 1.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.array(z, dtype=float).ravel()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, -1, 2)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5*(lo+hi), 0, 1)
    err = target - h.sum()
    if abs(err) > 1e-12:
        free = (h > 1e-10) & (h < 1 - 1e-10)
        nf = free.sum()
        if nf > 0:
            h[free] += err / nf
    return np.clip(h, 0, 1)

def all_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def all_grads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for idx, k in enumerate(range(-(n-1), n)):
        s = max(0, -k)
        e = min(n, n-k)
        i_arr = np.arange(s, e)
        j_arr = i_arr + k
        G[idx, i_arr] += dx * (1.0 - h[j_arr])
        G[idx, j_arr] += dx * (-h[i_arr])
    return G

def lp_step(h, radius):
    n = len(h)
    C = all_C(h)
    G = all_grads(h)
    cmax = C.max()
    
    c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -C
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    d_lo = np.maximum(-h, -radius)
    d_hi = np.minimum(1.0 - h, radius)
    bounds = list(zip(d_lo, d_hi)) + [(0.0, cmax + radius + 1.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n]
    return None

def lp_refine(h0, max_iters, r0, t_stop):
    h = project_feasible(np.array(h0, dtype=float))
    s = exact_score(h)
    bh, bs = h.copy(), s
    r = r0
    stall = 0
    for _ in range(max_iters):
        if time.time() > t_stop:
            break
        d = lp_step(h, r)
        if d is None or np.abs(d).max() < 1e-12:
            r *= 0.5
            stall += 1
            if stall > 12 or r < 1e-7:
                break
            continue
        ok = False

            r = min(r * 1.2, 0.5)
            stall = 0
        else:
            r *= 0.4
            stall += 1
            if stall > 15 or r < 1e-7:

    return bh, bs

def interp_to(h, n_t):
    n_s = len(h)
