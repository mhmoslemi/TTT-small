def overlap_values(h):
    """Returns array of 2n-1 overlap values (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_of_vals(vals, beta):
    """Smooth maximum of vals using log-sum-exp."""
    m = vals.max()
    shifted = beta * (vals - m)
    # For numerical stability
    max_s = shifted.max()
    return m + (max_s + np.log(np.exp(shifted - max_s).sum())) / beta

def smooth_max_grad_h(h, beta):
    """Gradient of smooth_max(overlap_values(h), beta) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    vals = overlap_values(h)
    m = vals.max()
    shifted = beta * (vals - m)
    max_s = shifted.max()
    exps = np.exp(shifted - max_s)
    w = exps / exps.sum()
    
    # Now compute grad = dx * sum_p w[p] * grad_v(p)
    # For each p, grad_v(p)[i] as computed above
    grad = np.zeros(n)
    
    for p in range(2 * n - 1):
        if w[p] < 1e-15:
            continue
        s = p - (n - 1)
        if s >= 0:
            # v = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
            # d/dh[i]: (1-h[i-s]) if s<=i<=n-1, minus h[i+s] if 0<=i<=n-1-s
            for i in range(n):
                g = 0.0
                if i >= s:
                    g += (1.0 - h[i - s])
                if i <= n - 1 - s:
                    g -= h[i + s]
                grad[i] += w[p] * g
        else:
            t = -s
            # v = sum_{j=0}^{n-1-t} h[j]*(1-h[j+t])
            # d/dh[i]: (1-h[i+t]) if 0<=i<=n-1-t, minus h[i-t] if t<=i<=n-1
            for i in range(n):
                g = 0.0
                if i <= n - 1 - t:
                    g += (1.0 - h[i + t])
                if i >= t:
                    g -= h[i - t]
                grad[i] += w[p] * g
    
    return dx * grad

    lo = -2.0
