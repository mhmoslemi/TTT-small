def binary_sa(n, n_ones, n_iterations, t0, t_end, rng):
    # Start with random balanced binary sequence
    h = np.zeros(n)
    ones_pos = rng.choice(n, n_ones, replace=False)
    h[ones_pos] = 1.0
    
    # Precompute correlation
    def compute_correlation(h):
        return np.correlate(h, 1.0 - h, mode='full')
    
    corr = compute_correlation(h)
    current_max = corr.max()
    
    best_h = h.copy()
    best_max = current_max
    
    for it in range(n_iterations):
        T = t0 * (t_end / t0) ** (it / n_iterations)
        
        # Choose a move: flip two bits (one 1→0, one 0→1) to maintain balance
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Compute change in correlation
        # h[i] goes from 1 to 0, h[j] goes from 0 to 1
        # Change in A(k) for each k:
        # A(k) = sum_valid_j h[j] * (1-h[j-k])

        #   now contributes h[i]*(1-h[i-k]) = 0*(1-h[i-k]) = 0
        #   change at i: -(1-h[i-k]) = h[i-k]-1 (for valid k)
        # At position j: was contributing h[j]*(1-h[j-k]) = 0
        #   now contributes h[j]*(1-h[j-k]) = 1*(1-h[j-k]) = 1-h[j-k]

        h[j] = 1.0
        new_corr = compute_correlation(h)
        new_max = new_corr.max()
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def binary_eval(h):
    """Compute all correlation values for a binary sequence."""
    return np.correlate(h, 1.0 - h, mode='full')

def binary_c5(h):
    n = len(h)
    corr = binary_eval(h)
    return float(np.max(corr) * 2.0 / n)

def random_balanced_binary(n, n_ones, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n_ones, replace=False)
    h[pos] = 1.0
    return h

def sa_binary(n, n_ones, n_iter, t0, t1, rng, h0=None):
    """Simulated annealing for binary sequences."""
    if h0 is None:
        h = random_balanced_binary(n, n_ones, rng)
    else:
        h = h0.copy()
    
    corr = binary_eval(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iter):
        T = t0 * (t1 / t0) ** (it / n_iter)
        
        # Pick random 1 and 0 to swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        
        # Recompute
        new_corr = binary_eval(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-max(delta, -50) / max(T, 1e-10)):
            current_max = new_max
            # Update ones/zeros tracking
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def thue_morse(n, k):
    """Generate Thue-Morse binary sequence of length n, with k ones (approximately n//2)."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    # Adjust to get exactly k ones
    ones = np.where(tm == 1)[0]
    zeros = np.where(tm == 0)[0]
    n_ones = len(ones)
    rng = np.random.default_rng(0)
    if n_ones > k:
        remove = rng.choice(ones, n_ones - k, replace=False)
        tm[remove] = 0
    elif n_ones < k:
        add = rng.choice(zeros, k - n_ones, replace=False)
        tm[add] = 1
    return tm.astype(float)

    if h_init is not None:
        h = h_init.astype(float).copy()
    else:
        h = np.zeros(n)
        pos = rng.choice(n, n_ones, replace=False)
        h[pos] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max

    one_pos = np.where(h == 1.0)[0].tolist()
    zero_pos = np.where(h == 0.0)[0].tolist()

                i2 = one_pos[rng.integers(len(one_pos))]
                if i1 == i2:
                    i2 = one_pos[rng.integers(len(one_pos))]
                j1 = zero_pos[rng.integers(len(zero_pos))]
                j2 = zero_pos[rng.integers(len(zero_pos))]
                if j1 == j2:
                    j2 = zero_pos[rng.integers(len(zero_pos))]
                # Only proceed if all four are distinct
                if len(set([i1, i2, j1, j2])) == 4:
                    h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(np.max(new_corr))
                    delta = new_max - current_max

                        current_max = new_max

                        zero_pos = np.where(h == 0.0)[0].tolist()

                best_h = h.copy()
        else:

            h[j] = 0.0
    
    return best_h, best_max

def smooth_max_obj(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_vals = corr * dx
    c_max = float(np.max(c_vals))
    shifted = np.clip(beta * (c_vals - c_max), -500.0, 500.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

`np.correlate(h, 1-h, mode='full')` has length 2n-1. The value at index p (0-indexed) is:

c[p] = sum_j h[j] * (1-h[j - (p - (n-1))])

where the sum is over valid j (0 ≤ j < n and 0 ≤ j - (p-(n-1)) < n).

Let me define s = p - (n-1). Then s ranges from -(n-1) to (n-1).
c[p] = sum_{j: 0≤j<n, 0≤j-s<n} h[j] * (1-h[j-s])

For the binary case, this is the count of j where h[j]=1 and h[j-s]=0 (within the valid range).

Now, the key insight: for s=0, c[n-1] = sum_j h[j]*(1-h[j]) = 0 for binary h. Good.

For s=1: c[n] = sum_{j=1}^{n-1} h[j]*(1-h[j-1]) = number of positions j where h[j]=1 and h[j-1]=0.
This is the number of "01" transitions (reading left to right) plus... wait, no.
If h[j]=1 and h[j-1]=0, that's a 0→1 transition. So c[n] = number of 0→1 transitions.

For a balanced binary sequence, the number of 0→1 transitions equals the number of 1→0 transitions (since the sequence starts and ends... well, not necessarily, since it's aperiodic).

Actually, for a binary sequence, the number of 0→1 transitions can range from 1 (alternating blocks) to n/2 (alternating).

For our problem, we want to MINIMIZE the maximum of c[p] over all p. So we want to minimize the maximum number of (1,0) pairs at any shift.

Let me think about the "alternating" sequence again: h = [1,0,1,0,...,1,0] (n even, starting with 1).
Number of 1s = n/2. ✓

c[n] (s=1): h[j]=1, h[j-1]=0 for j=1,3,5,...,n-1. Count = n/2.
c[n-2] (s=-1): h[j]=1, h[j+1]=0 for j=0,2,4,...,n-2. Count = n/2.

So the max is n/2, giving C5 = (n/2)*(2/n) = 1. Very bad.

def project_feasible(v, target_sum):
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def thue_morse_binary(n, n_ones, rng):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    ones = np.where(tm == 1.0)[0]
    zeros = np.where(tm == 0.0)[0]
    cur_ones = len(ones)
    if cur_ones > n_ones:
        remove = rng.choice(ones, cur_ones - n_ones, replace=False)
        tm[remove] = 0.0
    elif cur_ones < n_ones:

        tm[add] = 1.0

def smooth_max_obj(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    c_vals = corr * dx
    c_max = float(np.max(c_vals))
    shifted = np.clip(beta * (c_vals - c_max), -500.0, 500.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

def sum_constraint(x, target_sum):
    return float(np.sum(x)) - target_sum

def slsqp_refine(h, n, target_sum, maxiter, beta):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target_sum,)}]
    res = minimize(smooth_max_obj, h, method='SLSQP',
                   args=(beta, n), bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    return res.x
