    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def coord_descent(h, max_sweeps=150, grid_size=2001, tol=1e-12):
    n = len(h)
    h = h.astype(float).copy()
    improved_total = True
    sweep_count = 0
    while improved_total and sweep_count < max_sweeps:
        improved_total = False
        sweep_count += 1
        result = np.correlate(h, 1.0 - h, mode='full')
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            old_val = h[i]
            h[i] = 1.0
            r1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 0.0
            r0 = np.correlate(h, 1.0 - h, mode='full')
            a = r1 - r0
            b = result - old_val * a
            ts = np.linspace(0.0, 1.0, grid_size)
            vals = np.max(a[None, :] * ts[:, None] + b[None, :], axis=1)
            best_idx = np.argmin(vals)
            best_t = ts[best_idx]
            if 0 < best_t < 1:
                hw = 3.0 / grid_size
                lo_t = max(0.0, best_t - hw)
                hi_t = min(1.0, best_t + hw)
                ts_fine = np.linspace(lo_t, hi_t, 300)
                vals_fine = np.max(a[None, :] * ts_fine[:, None] + b[None, :], axis=1)
                best_t = ts_fine[np.argmin(vals_fine)]
            delta = best_t - old_val
            if abs(delta) > tol:
                h[i] = best_t
                result += delta * a
                improved_total = True
    return make_feasible(h)

def pair_move_descent(h, max_sweeps=30, grid_size=501, tol=1e-12):
    n = len(h)
    h = h.astype(float).copy()
    for sweep in range(max_sweeps):
        improved = False
        result = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            for j in range(i + 1, n):
                if abs(i - j) > n // 2:
                    continue
                h[i] = 1.0
                r1 = np.correlate(h, 1.0 - h, mode='full')
                h[i] = 0.0
                r0 = np.correlate(h, 1.0 - h, mode='full')
                a_i = r1 - r0
                h[j] = 1.0
                r1 = np.correlate(h, 1.0 - h, mode='full')
                h[j] = 0.0
                r0 = np.correlate(h, 1.0 - h, mode='full')
                a_j = r1 - r0
                b = result - h[i] * a_i - h[j] * a_j
                diff = a_i - a_j
                const = h[i] + h[j]
                h_lo = max(0.0, const - 1.0)
                h_hi = min(1.0, const)
                if h_lo >= h_hi:
                    continue
                ts = np.linspace(h_lo, h_hi, grid_size)
                vals = np.max(a_i[None, :] * ts[:, None] + a_j[None, :] * (const - ts)[:, None] + b[None, :], axis=1)
                best_t = ts[np.argmin(vals)]
                new_i = best_t
                new_j = const - best_t
                if abs(new_i - h[i]) > tol or abs(new_j - h[j]) > tol:
                    h[i] = new_i
                    h[j] = new_j
                    result = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
        if not improved:
            break
    return make_feasible(h)

def smooth_lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -400, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def smooth_lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -400, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        for p in range(j_lo, j_hi):
            grad[p] += w * (1.0 - x[p - shift])
        p_lo2 = max(0, j_lo - shift)
        p_hi2 = min(n, j_hi - shift)
        for p in range(p_lo2, p_hi2):
            grad[p] += w * (-x[p + shift])
    return grad

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def do_slsqp_lse(h0, beta, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            smooth_lse_obj, h0, args=(beta,),
            jac=smooth_lse_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(15.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve
    np.random.seed(seed)
    
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        parent = make_feasible(parent)
    except Exception:
        parent = make_feasible(np.ones(48) * 0.5)
    
    best_h = parent.copy()
    best_score = official_score(best_h)
    best_n = len(best_h)
    
    rng = np.random.default_rng(seed)
    parent_n = len(parent)
    
    n_options = sorted(set([
        parent_n - 6, parent_n - 4, parent_n - 2, parent_n,
        parent_n + 2, parent_n + 4, parent_n + 6
    ]))
    n_options = [n for n in n_options if 24 <= n <= 72]
    
    def score_candidate(h):
        nonlocal best_h, best_score, best_n
        s = official_score(h)
        if s < best_score - 1e-13:
            best_h = h.copy()
            best_score = s
            best_n = len(best_h)
        return s
    
    # Phase 1: Coordinate descent at multiple sizes
    for n in n_options:
        if time.time() > deadline - 90:
            break
        h_interp = make_feasible(interp_to_n(parent, n))
        starts = [h_interp.copy()]
        for p in [1.5, 3.0]:
            starts.append(make_feasible(h_interp ** p))
        for sigma in [0.05, 0.15]:
            starts.append(make_feasible(h_interp + sigma * rng.standard_normal(n)))
        starts.append(make_feasible(h_interp[::-1].copy()))
        
        for h0 in starts:
            if time.time() > deadline - 60:
                break
            h_opt = coord_descent(h0, max_sweeps=80, grid_size=2001)
            score_candidate(h_opt)
    
    # Phase 2: Pair moves on best
    if time.time() > deadline - 60:
        pass
    else:
        h_pm = pair_move_descent(best_h, max_sweeps=20, grid_size=401)
        sc = official_score(h_pm)
        if sc < best_score - 1e-13:
            best_h = h_pm
            best_score = sc
    
    # Phase 3: SLSQP LSE refinement
    if time.time() > deadline - 60:
        pass
    else:
        for beta in [50, 200, 500]:
            if time.time() > deadline - 30:
                break
            h_lse = do_slsqp_lse(best_h, beta, maxiter=150)
            sc = official_score(h_lse)
            if sc < best_score - 1e-13:
                best_h = h_lse
                best_score = sc
                # Follow up with coord descent
                h_cd = coord_descent(best_h, max_sweeps=50, grid_size=2001)
                sc2 = official_score(h_cd)
                if sc2 < best_score:
                    best_h = h_cd
                    best_score = sc2
    
    # Phase 4: Final high-resolution coordinate descent
    remaining = deadline - time.time()
    if remaining > 15:
        gs = 4001 if remaining > 40 else 2001
        ms = 100 if remaining > 40 else 50
        h_final = coord_descent(best_h, max_sweeps=ms, grid_size=gs)
        sc = official_score(h_final)
        if sc < best_score:
            best_h = h_final
            best_score = sc
    
    # Phase 5: One more pair move pass
    remaining = deadline - time.time()
    if remaining > 15:
        h_pm2 = pair_move_descent(best_h, max_sweeps=10, grid_size=301, tol=1e-10)
        sc = official_score(h_pm2)
        if sc < best_score:
            best_h = h_pm2
            best_score = sc
    
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    return (best_h, final_score, len(best_h))
