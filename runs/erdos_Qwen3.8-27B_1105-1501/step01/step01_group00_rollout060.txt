1. Compute all f_k values. Let M = max_k f_k.

5. This gives us |A| equations (f_k = M) plus the sum constraint, for a total of |A|+1 equations in n+1 unknowns (h[0],...,h[n-1], M).
6. If |A| < n, we have a underdetermined system. We can find the solution that's closest to the current h (minimum norm perturbation).

This is essentially a Newton-type method on the KKT conditions.

But solving this exactly is hard because f_k = M are nonlinear equations (quadratic in h).

However, we can LINEARIZE: at the current h, the Taylor expansion of f_k is:
f_k(h + δ) ≈ f_k(h) + ∇f_k(h)^T δ

And we want f_k(h + δ) = M for all k ∈ A.

So: ∇f_k(h)^T δ = M - f_k(h) for all k ∈ A

Plus: sum(δ) = 0 (to maintain the sum constraint)

This is a linear system! If |A| + 1 ≤ n, it's underdetermined, and we can find the minimum-norm solution.

Then we do a line search: h_new = h + α * δ for α ∈ (0, 1].

This is essentially a **Gauss-Newton method** applied to the active set!

from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, -1.0, 2.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        if r > 0:
            for idx in np.argsort(1.0 - h):
                a = min(r, 1.0 - h[idx])
                if a <= 0:
                    break
                h[idx] += a
                r -= a
                if abs(r) < 1e-14:
                    break
        else:
            for idx in np.argsort(h):
                a = min(-r, h[idx])
                if a <= 0:
                    break
                h[idx] -= a
                r += a
                if abs(r) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def compute_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def grad_lag(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            g[:L] -= dx * h[s:]
            g[s:] += dx * (1.0 - h[:L])
    else:
        t = -s
        L = n - t
        if L > 0:
            g[:L] += dx * (1.0 - h[t:])
            g[t:] -= dx * h[:L]
    return g

def all_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        G[p] = grad_lag(h, s)
    return G

def smooth_obj(h, beta):
    vals = compute_vals(h)
    cmax = vals.max()
    if beta < 5000:
        shifted = beta * (vals - cmax)
        max_sh = shifted.max()
        logsum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
        return cmax + logsum / beta
    return cmax

def smooth_grad(h, beta):
    vals = compute_vals(h)
    n = len(h)
    cmax = vals.max()
    if beta < 5000:
        shifted = beta * (vals - cmax)
        max_sh = shifted.max()
        w = np.exp(shifted - max_sh)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w /= ws
    else:
        w = np.zeros(len(vals))
        w[vals == cmax] = 1.0 / np.sum(vals == cmax)
    G = all_grads(h)
    return w @ G

def projected_grad_descend(h0, beta_start=30, beta_end=5000, max_iter=3000,
                           tol=1e-13, deadline=None):

    h = h0.copy()

    momentum = np.zeros(n)
    mu = 0.7
    alpha = 0.01
    for it in range(max_iter):

            break

        beta = beta_start * (beta_end / beta_start) ** frac

def active_set_polish(h0, max_iter=200, tol=1e-12, deadline=None):
    n = len(h0)

    for it in range(max_iter):

def anti_sym_transform(h):
    h_new = 1.0 - h[::-1].copy()
    return make_feasible(h_new)

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    src_c = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    tgt_c = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_int = np.interp(tgt_c, src_c, h_src)
    return make_feasible(h_int)

def slsqp_minimax(h0, maxiter=200, deadline=None):
    n = len(h0)
    vals0 = compute_vals(h0)
    M0 = vals0.max()
    
    def obj(z):
        return z[-1]
    
    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    def ineq_fun(z):
        hh = z[:-1]
        th = z[-1]
        v = compute_vals(hh)
        return th - v
    
    def ineq_jac(z):
        hh = z[:-1]
        G = all_grads(hh)
        J = -G
        J = np.hstack([J, np.ones((J.shape[0], 1))])
        return J
    
    def eq_fun(z):
        return np.sum(z[:-1]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(len(z))
        g[:-1] = 1.0
        return g
    
    z0 = np.concatenate([h0, [M0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'disp': False})
        if deadline is not None and time.time() > deadline:
            return None, None
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:-1])
            sc = official_score(h_new)
            if np.isfinite(sc):
                return h_new, sc
    except Exception:
        pass
    return None, None
