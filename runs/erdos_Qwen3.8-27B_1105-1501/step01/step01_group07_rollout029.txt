            # That's g[0:n-k] += wk * (1 - h[k:n])

def smooth_grad_direct(h, T):
    """Direct O(n²) gradient computation (same as above but cleaner)"""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    
    c = n - 1
    g = np.zeros(n)
    for ki in range(len(w)):
        wk = w[ki] * dx
        if wk < 1e-15:
            continue
        k = ki - c  # actual lag
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            kk = -k
            g[kk:] += wk * (1.0 - h[:n-kk])
            g[:n-kk] -= wk * h[kk:]
    
    return base, g

def grad_descent_v2(h, deadline, T_start, T_end, T_decay, step0, rng, top_k=0):
    """Improved gradient descent with adaptive top-k focusing."""
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    iters = 0
    
    while time.time() < deadline:
        base, g = smooth_grad_direct(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.5:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.7, step * 0.5, step * 0.25, step * 0.1):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 60:
                # Perturb and reset
                noise_scale = max(0.003, T * 0.5)
                cur = make_feasible(cur + rng.normal(0, noise_scale, n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.1, 0.5)
            stall = 0
        T = max(T_end, T * T_decay)
        iters += 1
    
    return best, best_c

def block_search(h, i, block_size, cur_c, n_sweeps=20):
    """Search over a block of consecutive coordinates."""
    n = len(h)
    j = i + block_size
    if j > n:
        j = n
        block_size = j - i
    if block_size < 2:
        return None
    
    orig = h[i:j].copy()
    best_block = orig.copy()
    best_s = cur_c
    
    # Try moving mass within the block (preserving sum)
    # and also try individual coordinate moves
    for sweep in range(n_sweeps):
        found = False
        for pos in range(i, j):
            orig_val = h[pos]
            # Try adjusting this one coordinate
            for delta in np.linspace(-orig_val * 0.5, (1.0 - orig_val) * 0.5, 15):
                if delta == 0:
                    continue
                h[pos] = orig_val + delta
                s = score_exact(h)
                if s < best_s - 1e-14:
                    best_s = s
                    best_block = h[i:j].copy()
                    found = True
        if not found:
            break
    
    h[i:j] = orig
    if best_s < cur_c - 1e-14:
        h[i:j] = best_block
        return float(best_s)
    return None

def pair_search_fast(h, i, j_init, cur_c, n_pts=30):
    """Fast pair search between coordinates i and j_init."""
    orig_i = h[i]
    orig_j = h[j_init]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j_init] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        # Golden section refinement
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(30):
                h[i] = orig_i + a1
                h[j_init] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j_init] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j_init] = orig_j - best_a
            best_s = score_exact(h)
    
    h[i] = orig_i
    h[j_init] = orig_j
    if best_s < cur_c - 1e-14:
        h[i] = orig_i + best_a
        h[j_init] = orig_j - best_a
        return float(best_s)
    return None

def adjacent_sweep(h, cur_c, n_pts=25):
    """Systematic sweep of all adjacent pairs."""
    n = len(h)
    improved = False
    for i in range(n - 1):
        res = pair_search_fast(h, i, i + 1, cur_c, n_pts=n_pts)
        if res is not None:
            cur_c = res
            improved = True
    return cur_c, improved

def structured_init(n, variant):
    """Create structured initial vectors."""
    x = np.arange(n)
    if variant == 0:
        # Uniform
        return np.full(n, 0.5)
    elif variant == 1:
        # Sine modulation
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x / n)

    elif variant == 2:

        return np.full(n, 0.5)

import numpy as np
import time


def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        if np.sum(z) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - z
            order = np.argsort(-slack)
        else:
            slack = z.copy()
            diff = -diff
            order = np.argsort(-slack)
        rem = diff
        for idx in order:
            if rem < 1e-14:
                break
            a = min(rem, slack[idx])
            z[idx] += a
            rem -= a
    return np.clip(z, 0.0, 1.0)


def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        mask = scores >= base - 1e-10
        w = np.where(mask, 1.0, 0.0)
        w /= w.sum()
    else:
        w = ex / S
    c = n - 1
    g = np.zeros(n)
    for ki in range(len(w)):
        wk = w[ki] * dx
        if wk < 1e-15:
            continue
        k = ki - c
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            kk = -k
            g[kk:] += wk * (1.0 - h[:n - kk])
            g[:n - kk] -= wk * h[kk:]
    return base, g


def grad_desc(h, deadline, T_start, T_end, T_decay, step0, rng):
    n = len(h)
    cur = h.copy()
    cur_c = score_exact(cur)
    best = cur.copy()
    best_c = cur_c
    T = T_start
    step = step0
    stall = 0
    while time.time() < deadline:
        base, g = smooth_grad(cur, T)
        if not np.all(np.isfinite(g)):
            T = max(T_end, T * 0.5)
            continue
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            T = max(T_end, T * 0.5)
            if T < T_end * 0.3:
                break
            continue
        direction = -g / gn
        improved = False
        for s in (step, step * 0.7, step * 0.5, step * 0.25):
            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = c_new
                break
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 50:
                ns = max(0.003, T * 0.4)
                cur = make_feasible(cur + rng.normal(0, ns, n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.003)
                stall = 0
        else:
            step = min(step * 1.08, 0.4)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c


def pair_search(h, i, j, cur_c, n_pts=25):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 2.0 * span)
        hi2 = min(hi, best_a + 2.0 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(35):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None


def coord_search(h, i, cur_c, n_pts=25):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return


def adjacent_sweep(h, cur_c, n_pts=20):
    n = len(h)
    improved = False
    for i in range(n - 1):
        res = pair_search(h, i, i + 1, cur_c, n_pts=n_pts)
        if res is not None:
            cur_c = res
            improved = True
    return cur_c, improved


def full_pair_sweep(h, cur_c, n_pts=15):
    n = len(h)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            res = pair_search(h, i, j, cur_c, n_pts=n_pts)
            if res is not None:
                cur_c = res
                improved = True
    return cur_c, improved


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)
