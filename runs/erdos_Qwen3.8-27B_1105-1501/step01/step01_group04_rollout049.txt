    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def optimize_nesterov(h0, time_limit, seed=42):
    """Smoothed-max gradient with Nesterov momentum and BB step size."""
    n = len(h0)
    target = n / 2.0
    h = np.clip(h0.astype(float).copy(), 0.0, 1.0)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    alpha = 10.0
    beta_nest = 0.0
    y = h.copy()
    h_prev = h.copy()
    grad_prev = None
    lr = 0.005
    stagnation = 0
    it = 0
    rng = np.random.RandomState(seed)

    while time.time() - t0 < time_limit:
        # Nesterov step
        t_iter = float(it + 1)
        beta = (-1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        y = h + beta * (h - h_prev)
        y = project_box_sum(y, target)

        # Compute gradient at y
        corr = get_corr(y)
        max_c = np.max(corr)
        exp_c = np.exp(alpha * (corr - max_c))
        p = exp_c / np.sum(exp_c)
        g = grad_weighted(y, p)

        # BB step size
        if grad_prev is not None:
            s = y - h_prev
            gy = g - grad_prev
            ss = np.dot(s, s)
            sy = np.dot(s, gy)
            if ss > 1e-14 and sy > 1e-14:
                lr_new = sy / ss
                lr = np.clip(lr_new, 0.0005, 0.03)

        # Gradient step
        h_new = project_box_sum(y - lr * g, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        h_prev = h
        grad_prev = g
        h = h_new
        it += 1

        # Adaptive schedule
        if it % 50 == 0:
            alpha = min(alpha * 1.08, 150.0)

        # Restart if stagnating
        if stagnation > 800:
            h = best_h + 0.03 * rng.randn(n)
            h = project_box_sum(h, target)
            h_prev = h.copy()
            y = h.copy()
            grad_prev = None
            lr = 0.01
            alpha = max(alpha * 0.5, 5.0)
            stagnation = 0

        # Aggressive restart
        if stagnation > 3000:
            h = best_h + 0.1 * rng.randn(n)
            h = project_box_sum(h, target)
            h_prev = h.copy()
            y = h.copy()
            grad_prev = None
            lr = 0.02
            alpha = 5.0
            stagnation = 0

    return best_h, best_c5


def optimize_penalty(h0, time_limit, seed=42):
    """Penalty-based: minimize sum_k max(0, c_k - tau)^2 with projection."""
    n = len(h0)
    target = n / 2.0
    h = np.clip(h0.astype(float).copy(), 0.0, 1.0)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    lr = 0.005

    while time.time() - t0 < time_limit:
        corr = get_corr(h)
        tau = np.max(corr) * 0.999  # target slightly below current max
        excess = np.maximum(0.0, corr - tau)
        if np.max(excess) < 1e-10:
            tau = np.max(corr) * 0.995
            excess = np.maximum(0.0, corr - tau)
        w = excess / (np.sum(excess) + 1e-15)
        g = grad_weighted(h, w)
        h_new = project_box_sum(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.1, 0.05)
        else:
            h = h_new
            lr *= 0.8
            if lr < 0.001:
                # restart
                h = best_h + 0.02 * rng.randn(n)
                h = project_box_sum(h, target)
                lr = 0.01
        h = h_new if c5_new >= best_c5 else h_new

    return best_h, best_c5


def local_polish(h0, time_limit, seed=42):
    """Random pair moves with acceptance."""
    n = len(h0)
    target = n / 2.0
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    while time.time() - t0 < time_limit:
        improved = False
        n_trials = 30000
        for _ in range(n_trials):
            if time.time() - t0 >= time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.01, 0.15) * rng.choice([-1, 1])
            hi = h[i] + d
            hj = h[j] - d
            if hi < -1e-10 or hi > 1.0 + 1e-10:
                continue
            if hj < -1e-10 or hj > 1.0 + 1e-10:
                continue
            h[i] = hi
            h[j] = hj
            c5_t = compute_c5(h)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved:
            # Try larger steps
            any_imp = False
            for _ in range(10000):
                if time.time() - t0 >= time_limit:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(0.1, 0.4) * rng.choice([-1, 1])
                hi = h[i] + d
                hj = h[j] - d
                if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                    continue
                h[i] = hi
                h[j] = hj
                c5_t = compute_c5(h)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    any_imp = True
                else:
                    h[i] -= d
                    h[j] += d
            if not any_imp:
                break
    return h, best_c5


def random_init(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    idx = rng.permutation(n)
    return h[idx]


def spread_init(n, rng):
    """Initialize with spread-out ones to reduce initial overlap."""
    h = np.zeros(n)
    ones_count = n // 2
    if n % 2 == 1:
        h[n // 2] = 0.5
    # Place ones at evenly spaced positions with random jitter
    base_pos = np.linspace(0, n - 1, ones_count, endpoint=False).astype(int)
    jitter = rng.randint(-2, 3, size=ones_count)
    pos = np.mod(base_pos + jitter, n)
    pos_unique = np.unique(pos)
    for p in pos_unique[:ones_count]:
        h[p] = 1.0
    # Fill remaining
    if n % 2 == 0:
        need = ones_count - int(np.sum(h))
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for idx in zeros[:need]:
            h[idx] = 1.0
    return h


def multi_start_optimize(n, time_limit, seed, n_restarts=3):
    """Try multiple initializations for given n."""
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time.time() % 1 == 0 and per_restart < 5:
            break
        if r == 0:
            h0 = spread_init(n, rng)
        elif r == 1:
            h0 = random_init(n, rng)
        else:
            h0 = random_init(n, rng)

        h_opt, c5_opt = optimize_nesterov(h0, per_restart, seed + r * 1000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Optimize from initial_h_values (n=99)
    h_init = None
    try:
        if len(initial_h_values) > 0:
            h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        pass

    if h_init is not None:
        n0 = len(h_init)
        alloc1 = min(remaining() * 0.30, 300.0)
        if alloc1 > 30:
            # Nesterov optimization
            h_opt, c5_opt = optimize_nesterov(h_init, alloc1, seed)
            if remaining() > 20:
                # Penalty polish
                alloc_p = min(remaining() * 0.05, 50.0)
                h_opt2, c5_opt2 = optimize_penalty(h_opt, alloc_p, seed + 500)
                if c5_opt2 < c5_opt:
                    h_opt, c5_opt = h_opt2, c5_opt2
            if remaining() > 20:
                # Local polish
                alloc_l = min(remaining() * 0.08, 80.0)
                h_opt3, c5_opt3 = local_polish(h_opt, alloc_l, seed + 700)
                if c5_opt3 < c5_opt:
                    h_opt, c5_opt = h_opt3, c5_opt2 if c5_opt2 < c5_opt else c5_opt3
                else:
                    c5_opt = min(c5_opt, c5_opt3)
                    if c5_opt3 < c5_opt:
                        h_opt, c5_opt = h_opt3, c5_opt3
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n0

    # Phase 2: Try n=99 with perturbations
    if h_init is not None:
        n0 = len(h_init)
        for pert in [0.05, 0.1, 0.2]:
            if remaining() < 40:
                break
            h_pert = h_init + pert * rng.randn(n0)
            h_pert = project_box_sum(h_pert, n0 / 2.0)
            alloc = min(remaining() * 0.08, 80.0)
            if alloc < 15:
                break
            h_opt, c5_opt = optimize_nesterov(h_pert, alloc, seed + int(pert * 1000))
            if remaining() > 15:
                h_opt2, c5_opt2 = local_polish(h_opt, min(remaining() * 0.04, 30.0), seed + 800)
                if c5_opt2 < c5_opt:
                    h_opt, c5_opt = h_opt2, c5_opt2
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n0

    # Phase 3: Try other n values
    n_candidates = [63, 127, 81, 161, 47]
    for n in n_candidates:
        if remaining() < 60:
            break
        alloc = min(remaining() * 0.12, 120.0)
        if alloc < 25:
            break
        h_best_n = None
        c5_best_n = float('inf')
        per_init = alloc / 3.0
        for init_r in range(3):
            if remaining() < 20:
                break
            if init_r == 0:
                h0 = spread_init(n, rng)
            else:
                h0 = random_init(n, rng)
            h_opt, c5_opt = optimize_nesterov(h0, per_init, seed + n * 100 + init_r)
            if c5_opt < c5_best_n:
                c5_best_n = c5_opt
                h_best_n = h_opt
        if h_best_n is not None:
            if remaining() > 15:
                h_best_n, c5_best_n = local_polish(h_best_n, min(remaining() * 0.05, 40.0), seed + n * 50)
            if c5_best_n < best_c5:
                best_c5 = c5_best_n
                best_h = h_best_n
                best_n = n

    # Phase 4: Final polish on best solution
    if best_h is not None and remaining() > 15:
        alloc_final = min(remaining() * 0.35, 200.0)
        h_final, c5_final = local_polish(best_h, alloc_final, seed + 9999)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Fallback
    if best_h is None:
        n = 99 if h_init is not None else 63
        best_h = random_init(n, rng)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Ensure constraints
    target = best_n / 2.0
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
