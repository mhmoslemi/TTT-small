def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()
    if len(h) > n_target:
        h = h[:n_target]
    elif len(h) < n_target:

            best_x = x

    max_c = np.max(cors)

    active_idx = np.where(active_mask)[0]
    lags = np.arange(-(n-1), n)
    
    # Composite gradient
    grad = np.zeros(n)
    for ki in active_idx:
        k = lags[ki]
        grad += compute_grad_lag(h, k)
    grad -= np.mean(grad)  # project onto sum=const
    
    pairs = select_pairs(h, n, grad, max_pairs)
    
    improved = False

        if s_new < score - 1e-13:
            h = h_new
            score = s_new
            improved = True
    return h, score, improved

        h = h[:n_target]
    elif len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(300):
        residual = target - np.sum(h)
        if abs(residual) < 1e-13:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            if room[idx] <= 1e-15:
                break
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            if h[idx] <= 1e-15:
                break
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h

def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)

def compute_grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    p = np.arange(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g

def pair_minimize(h, i, j, grid_n, golden_iters):
    S = h[i] + h[j]
    L = max(0.0, S - 1.0)
    U = min(1.0, S)
    if U - L < 1e-10:
        return h, official_score(h)
    cur_score = official_score(h)
    xs = np.linspace(L, U, grid_n)
    scores = np.empty(grid_n)
    for xi in range(grid_n):
        h[i] = xs[xi]
        h[j] = S - xs[xi]
        scores[xi] = official_score(h)
    best_idx = int(np.argmin(scores))
    best_s = scores[best_idx]
    best_x = xs[best_idx]
    lo_idx = max(0, best_idx - 1)
    hi_idx = min(grid_n - 1, best_idx + 1)
    if lo_idx == hi_idx:
        hi_idx = min(grid_n - 1, hi_idx + 1)
    a, b = xs[lo_idx], xs[hi_idx]
    if b - a < 1e-12:
        h[i] = best_x
        h[j] = S - best_x
        return h, best_s
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    h[i] = x1; h[j] = S - x1; f1 = official_score(h)
    h[i] = x2; h[j] = S - x2; f2 = official_score(h)
    for _ in range(golden_iters):
        if f1 < f2:
            b = x2; x2 = x1; f2 = f1
            x1 = b - gr * (b - a)
            h[i] = x1; h[j] = S - x1; f1 = official_score(h)
        else:
            a = x1; x1 = x2; f1 = f2
            x2 = a + gr * (b - a)
            h[i] = x2; h[j] = S - x2; f2 = official_score(h)
    x_opt = (a + b) * 0.5
    h[i] = x_opt; h[j] = S - x_opt
    f_opt = official_score(h)
    if f_opt < best_s:
        best_s = f_opt
        best_x = x_opt
    else:
        h[i] = best_x; h[j] = S - best_x
    if best_s < cur_score - 1e-13:
        return h, best_s
    h[i] = best_x; h[j] = S - best_x
    return h, best_s

def select_pairs(h, n, grad, max_pairs, rng):
    grad_abs = np.abs(grad)
    score = grad_abs * grad_abs * (1.0 - np.abs(h) * 2.0 * 0.0)
    idx_sorted = np.argsort(-grad_abs)
    pairs = []
    top = idx_sorted[:min(50, n)]
    for ai in range(len(top)):
        for bi in range(ai + 1, len(top)):
            i, j = top[ai], top[bi]
            diff = grad[i] - grad[j]
            S = h[i] + h[j]
            width = min(1.0, S) - max(0.0, S - 1.0)
            if width > 0.05 and abs(diff) > 0.01:
                pairs.append((i, j, abs(diff) * width))
    pairs.sort(key=lambda x: -x[2])
    result = []
    seen = set()
    for i, j, _ in pairs:
        key = (min(i, j), max(i, j))
        if key not in seen:
            seen.add(key)
            result.append(key)
        if len(result) >= max_pairs:
            break
    for _ in range(min(30, max_pairs - len(result))):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            key = (min(i, j), max(i, j))
            if key not in seen:
                seen.add(key)
                result.append(key)
    return result

def pair_pass(h, n, max_pairs, grid_n, golden_iters, rng):
    score = official_score(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(cors)
    active_mask = cors >= max_c - 0.004
    active_idx = np.where(active_mask)[0]
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki in active_idx:
        grad += compute_grad_lag(h, lags[ki])
    grad -= np.mean(grad)
    if np.max(np.abs(grad)) < 1e-10:
        return h, score, False
    pairs = select_pairs(h, n, grad, max_pairs, rng)
    improved = False
    for (i, j) in pairs:
        h, s_new = pair_minimize(h, i, j, grid_n, golden_iters)
        if s_new < score - 1e-13:
            score = s_new
            improved = True
    return h, score, improved

def smooth_direction_step(h, n, rng):
    score = official_score(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(cors)
    T = max(0.003 + 0.012 * max_c, 0.002)
    weights = np.exp((cors - max_c) / T)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        return h, score
    weights /= wsum
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki in range(len(cors)):
        if weights[ki] > 1e-5:
            grad += weights[ki] * compute_grad_lag(h, lags[ki])
    grad -= np.mean(grad)
    direction = -grad
    norm = np.max(np.abs(direction))
    if norm < 1e-12:
        return h, score
    direction /= norm
    noise = rng.normal(0, 0.15, n)
    noise -= np.mean(noise)
    direction = direction + noise
    best_h = h.copy()
    best_s = score
    for alpha in [0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.4, 0.7]:
        for sign in [1, -1]:
            h_try = repair_vector(h + sign * alpha * direction, n)
            s_try = official_score(h_try)
            if s_try < best_s - 1e-12:
                best_s = s_try
                best_h = h_try.copy()
    return best_h, best_s

def slsqp_refine(h, n, time_limit=20.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    cors = np.correlate(h, 1.0 - h, mode="full") * dx
    top_m = min(8, len(cors))
    top_indices = np.argsort(-cors)[:top_m]
    initial_score = official_score(h)

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        c = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return np.array([theta - c[i] for i in top_indices])

    def ineq_jac(x):
        h_x = x[:n]
        j = np.zeros((top_m, n + 1))
        for mi, ti in enumerate(top_indices):
            k = lags[ti]
            j[mi, :n] = -dx * compute_grad_lag(h_x, k)
            j[mi, n] = 1.0
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(initial_score, initial_score + 0.5)]
    x0 = np.concatenate([h, [initial_score + 1e-5]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 60, 'ftol': 1e-10})
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s_out = official_score(h_out)
            if s_out < initial_score - 1e-10:
                return h_out, s_out
    except Exception:
        pass
    return h, initial_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 8.0

    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = repair_vector(parent, n_parent)
    else:
        n_parent = 64
        best_h = repair_vector(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    candidates = []
    candidates.append((best_score, best_h.copy(), best_n))
    sym1 = repair_vector(best_h[::-1].copy(), best_n)
    candidates.append((official_score(sym1), sym1, best_n))
    sym2 = repair_vector((1.0 - best_h[::-1]).copy(), best_n)
    candidates.append((official_score(sym2), sym2, best_n))
    sym3 = repair_vector((1.0 - best_h).copy(), best_n)
    candidates.append((official_score(sym3), sym3, best_n))

    n_list = sorted(set([best_n, max(20, best_n - 5), best_n + 5,
                         max(20, best_n - 10), best_n + 10]))
    for n_try in n_list:
        for off in [0, 1, 2]:
            h_int = gen_interpolated(best_h, n_try, offset=off)
            s_int = official_score(h_int)
            candidates.append((s_int, h_int, n_try))

    candidates.sort(key=lambda x: x[0])

    top_k = min(6, len(candidates))
    time_budget = deadline - start_time

    for ci in range(top_k):
        if time.perf_counter() > deadline - 20:
            break
        s_c, h_c, n_c = candidates[ci]
        remaining = deadline - time.perf_counter()
        t_alloc = remaining * (0.4 if ci == 0 else 0.25 / (ci + 1))
        t_alloc = max(10.0, min(t_alloc, remaining * 0.6))

        h_work = h_c.copy()
        s_work = s_c
        n_work = n_c
        rng_local = np.random.default_rng(seed + ci * 7919)
        no_improve = 0
        max_stall = 4

        for p in range(30):
            if time.perf_counter() > start_time + t_alloc:
                break
            if p < 8:
                mp, gn, gi = 150, 15, 7
            elif p < 16:
                mp, gn, gi = 100, 12, 5
            else:
                mp, gn, gi = 60, 10, 4
            h_work, s_work, improved = pair_pass(h_work, n_work, mp, gn, gi, rng_local)
            if improved:
                no_improve = 0
                if s_work < best_score - 1e-14:
                    best_score = s_work
                    best_h = h_work.copy()
                    best_n = n_work
            else:
                no_improve += 1
                if no_improve >= 2:
                    h_work, s_work = smooth_direction_step(h_work, n_work, rng_local)
                    if s_work < best_score - 1e-14:
                        best_score = s_work
                        best_h = h_work.copy()
                        best_n = n_work
                    no_improve = 0
                    for p2 in range(4):
                        if time.perf_counter() > start_time + t_alloc:
                            break
                        h_work, s_work, imp2 = pair_pass(h_work, n_work, 80, 12, 5, rng_local)
                        if imp2 and s_work < best_score - 1e-14:
                            best_score = s_work
                            best_h = h_work.copy()
                            best_n = n_work
                    if no_improve >= max_stall:
                        break

    if time.perf_counter() < deadline - 30:
        h_ref, s_ref = slsqp_refine(best_h, best_n, time_limit=min(20.0, max(5.0, deadline - time.perf_counter() - 10)))
        if s_ref < best_score - 1e-12:
            best_score = s_ref
            best_h = h_ref
            best_n = len(h_ref)

    if time.perf_counter() < deadline - 10:
        rng_final = np.random.default_rng(seed + 555)
        for _ in range(8):
            if time.perf_counter() > deadline - 5:
                break
            best_h, best_score, _ = pair_pass(best_h, best_n, 80, 12, 5, rng_final)

    if time.perf_counter() < deadline - 5:
        h_ref2, s_ref2 = slsqp_refine(best_h, best_n, time_limit=min(8.0, max(3.0, deadline - time.perf_counter() - 3)))
        if s_ref2 < best_score - 1e-12:
            best_score = s_ref2
            best_h = h_ref2
            best_n = len(h_ref2)

    best_h = repair_vector(best_h, len(best_h))
    best_score = official_score(best_h)

    return (best_h, float(best_score), int(len(best_h)))
