```python
def manifold_smooth_obj_grad(a, layout, temperature):
    m = len(a)
    if layout == 'sc':
        h = build_sc(a)
    else:
        h = build_rc(a)
    n = 2 * m
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    exp_vals = np.exp((corr - mx) / temperature)
    z = np.sum(exp_vals)
    if z < 1e-300:
        return 1e10, np.zeros(m)
    obj = mx + temperature * np.log(z)
    weights = exp_vals / z
    
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n - 1)
        # First term: j in [max(0,s), min(n-1, n-1+s)]
        j1_lo = max(0, s)
        j1_hi = min(n - 1, n - 1 + s)
        if j1_lo <= j1_hi:
            grad_h[j1_lo:j1_hi+1] += w * (1.0 - h[j1_lo - s:j1_hi - s + 1])
        # Second term: j in [max(0,-s), min(n-1, n-1-s)]
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            grad_h[j2_lo:j2_hi+1] -= w * h[j2_lo + s:j2_hi + s + 1]
    
    # Map to manifold variables
    if layout == 'sc':
        grad_a = grad_h[:m] - grad_h[m:]
    else:
        grad_a = np.zeros(m)
        for p in range(m):
            grad_a[p] = grad_h[p] - grad_h[n - 1 - p]
    
    return obj, grad_a
```

    x = np.asarray(x, dtype=float).copy()
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in order:
            if residual > 0:
                add = min(1.0 - result[i], residual)
            else:
                add = max(result[i] - 1.0, residual)
            result[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    return result


def exact_c5(h):
    n = len(h)
    if n < 2:
        return 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def build_sc(a):
    return np.concatenate([a, 1.0 - a])


def build_rc(a):
    m = len(a)
    return np.concatenate([a, (1.0 - a)[::-1]])


def manifold_smooth_obj_grad(a, layout, temp):
    m = len(a)
    h = build_sc(a) if layout == 'sc' else build_rc(a)
    n = 2 * m
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    exp_vals = np.exp((corr - mx) / temp)
    z = np.sum(exp_vals)
    if z < 1e-300:
        return 1e10, np.zeros(m)
    obj = mx + temp * np.log(z)
    weights = exp_vals / z
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n - 1)
        j1_lo = max(0, s)
        j1_hi = min(n - 1, n - 1 + s)
        if j1_lo <= j1_hi:
            grad_h[j1_lo:j1_hi + 1] += w * (1.0 - h[j1_lo - s:j1_hi - s + 1])
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            grad_h[j2_lo:j2_hi + 1] -= w * h[j2_lo + s:j2_hi + s + 1]
    if layout == 'sc':
        grad_a = grad_h[:m] - grad_h[m:]
    else:
        grad_a = np.zeros(m)
        for p in range(m):
            grad_a[p] = grad_h[p] - grad_h[n - 1 - p]
    return obj, grad_a


def _wrap_obj(a, layout, temp):
    return manifold_smooth_obj_grad(a, layout, temp)[0]


def _wrap_grad(a, layout, temp):
    return manifold_smooth_obj_grad(a, layout, temp)[1]


def optimize_manifold_lbfgs(a0, layout, temps, time_limit):
    m = len(a0)
    bounds = [(0.0, 1.0) * m]
    best_a = a0.copy()
    h = build_sc(a0) if layout == 'sc' else build_rc(a0)
    best_c5 = exact_c5(h)
    t_start = time.time()
    current = a0.copy()
    for temp in temps:
        if time.time() - t_start > time_limit - 5:
            break
        res = minimize(
            _wrap_obj, current, args=(layout, temp),
            method='L-BFGS-B', jac=_wrap_grad,
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            candidate = np.clip(res.x, 0.0, 1.0)
            h_cand = build_sc(candidate) if layout == 'sc' else build_rc(candidate)
            c5_cand = exact_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_a = candidate
            current = candidate
            if time.time() - t_start > time_limit - 5:
                break
    return best_a, best_c5


def coord_descent_manifold(a, layout, time_limit, n_grid=60):
    m = len(a)
    t_start = time.time()
    h = build_sc(a) if layout == 'sc' else build_rc(a)
    best_c5 = exact_c5(h)
    improved_any = True
    passes = 0
    while improved_any and time.time() - t_start < time_limit:
        improved_any = False
        passes += 1
        for p in range(m):
            if time.time() - t_start > time_limit - 2:
                break
            old_vals = a.copy()
            grid = np.linspace(0.0, 1.0, n_grid)
            best_val = a[p]
            best_local = best_c5
            for v in grid:
                a[p] = v
                h_cand = build_sc(a) if layout == 'sc' else build_rc(a)
                c5_cand = exact_c5(h_cand)
                if c5_cand < best_local - 1e-14:
                    best_local = c5_cand
                    best_val = v
            a[p] = best_val
            if best_val != old_vals[p]:
                # Refine locally
                lo_r = max(0.0, best_val - 1.0 / n_grid)
                hi_r = min(1.0, best_val + 1.0 / n_grid)
                fine_grid = np.linspace(lo_r, hi_r, 20)
                for v in fine_grid:
                    a[p] = v
                    h_cand = build_sc(a) if layout == 'sc' else build_rc(a)
                    c5_cand = exact_c5(h_cand)
                    if c5_cand < best_local - 1e-14:
                        best_local = c5_cand
                        best_val = v
                a[p] = best_val
                if best_local < best_c5 - 1e-14:
                    improved_any = True
                    best_c5 = best_local
            a[:] = old_vals
            a[p] = best_val
            if best_local < best_c5 - 1e-14:
                best_c5 = best_local
    return a, best_c5


def pair_exchange_refine(h, time_limit, n_max_iter=200000):
    n = len(h)
    t_start = time.time()
    best_c5 = exact_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    cnt = 0
    # Precompute for speed
    while time.time() - t_start < time_limit and cnt < n_max_iter:
        cnt += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi = best_h[i]
        hj = best_h[j]
        # Transfer t from i to j: h[i]-t, h[j]+t
        lo_t = max(0.0 - hi, 0.0 - hj)  # 0
        hi_t = min(hi, 1.0 - hj)
        if hi_t - lo_t < 1e-12:
            continue
        # Try a few values of t
        ts = np.array([0.0, 0.25, 0.5, 0.75]) * (hi_t - lo_t) + lo_t
        ts = np.append(ts, hi_t)
        best_t = 0.0
        best_t_c5 = best_c5
        for t in ts:
            if t < -1e-12 or t > hi_t + 1e-12:
                continue
            t = np.clip(t, 0.0, hi_t)
            best_h[i] = hi - t
            best_h[j] = hj + t
            c5_try = exact_c5(best_h)
            if c5_try < best_t_c5 - 1e-13:
                best_t_c5 = c5_try
                best_t = t
        best_h[i] = hi - best_t
        best_h[j] = hj + best_t
        if best_t_c5 < best_c5 - 1e-13:
            best_c5 = best_t_c5
            if cnt % 5000 == 0:
                h_now = best_h.copy()
                best_h = h_now
    return best_h, best_c5


def make_fallback(n, seed):
    rng = np.random.RandomState(seed)
    m = n // 2
    a = np.clip(0.5 + 0.3 * np.sin(np.linspace(0, 2 * np.pi * 3, m) + seed * 0.1), 0, 1)
    h = build_sc(a)
    return project_box_sum(h, m)


def transfer_to_manifold(h, layout):
    n = len(h)
    m = n // 2
    if layout == 'sc':
        a = 0.5 * (h[:m] + (1.0 - h[m:]))
    else:
        a = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(a, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    reserve = max(10.0, 0.03 * budget_s)
    t_end = t0 + budget_s - reserve

    # Get initial values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        pass

    n_parent = 70
    if h_parent is None or len(h_parent) == 0:
        h_parent = make_fallback(n_parent, seed)
        n_parent = len(h_parent)

    n_parent = len(h_parent)
    h_parent = project_box_sum(h_parent, n_parent / 2.0)
    c5_parent = exact_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_parent

    # Determine which n values to try
    n_candidates = [n_parent]
    for nn in [68, 72, 80, 60, 50]:
        if nn % 2 == 0 and nn != n_parent and nn >= 40:
            n_candidates.append(nn)
    n_candidates = sorted(n_candidates)

    layouts = ['sc', 'rc']
    temps_sched = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]

    # Phase 1: Manifold optimization
    for n_try in n_candidates:
        if time.time() > t_end:
            break
        m = n_try // 2
        time_per_n = max(30.0, min(200.0, (t_end - time.time()) / max(1, len(n_candidates) / 2)))
        time_per_layout = time_per_n / len(layouts)

        for layout in layouts:
            if time.time() > t_end:
                break
            # Get starting point
            if n_try == n_parent:
                a0 = transfer_to_manifold(best_h, layout)
            else:
                # Resample parent to n_try
                if n_parent != n_try:
                    h_resample = np.interp(
                        np.linspace(0, n_parent - 1, n_try),
                        np.arange(n_parent),
                        h_parent
                    )
                    h_resample = project_box_sum(h_resample, n_try / 2.0)
                    a0 = transfer_to_manifold(h_resample, layout)
                else:
                    a0 = transfer_to_manifold(h_parent, layout)

            # Also try a random start
            rng = np.random.RandomState(seed + n_try + (1 if layout == 'rc' else 0))
            a_rand = np.clip(0.5 + 0.3 * rng.randn(m), 0.0, 1.0)

            # Optimize from structured start
            a_opt, c5_opt = optimize_manifold_lbfgs(
                a0, layout, temps_sched, time_per_layout * 0.6
            )
            if c5_opt < best_c5:
                h_new = build_sc(a_opt) if layout == 'sc' else build_rc(a_opt)
                h_new = project_box_sum(h_new, n_try / 2.0)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new

            # Coordinate descent from the optimized point
            if time.time() < t_end:
                a_cd, c5_cd = coord_descent_manifold(
                    a_opt, layout, min(60.0, time_per_layout * 0.3)
                )
                if c5_cd < best_c5:
                    h_new = build_sc(a_cd) if layout == 'sc' else build_rc(a_cd)
                    h_new = project_box_sum(h_new, n_try / 2.0)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new

            # Try random start if time allows
            if time.time() < t_end - 30:
                time_left = min(40.0, (t_end - time.time()) * 0.3)
                a_opt2, c5_opt2 = optimize_manifold_lbfgs(
                    a_rand, layout, temps_sched[:4], time_left
                )
                if c5_opt2 < best_c5:
                    h_new = build_sc(a_opt2) if layout == 'sc' else build_rc(a_opt2)
                    h_new = project_box_sum(h_new, n_try / 2.0)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new

    # Phase 2: Direct pair exchange on best_h
    if time.time() < t_end:
        time_for_pairs = min(400.0, t_end - time.time())
        best_h, best_c5 = pair_exchange_refine(best_h.copy(), time_for_pairs)

    # Phase 3: More coordinate descent on the best manifold representation
    if time.time() < t_end:
        n_best = len(best_h)
        if n_best % 2 == 0:
            for layout in layouts:
                if time.time() > t_end:
                    break
                m = n_best // 2
                a_cur = transfer_to_manifold(best_h, layout)
                # Quickly optimize
                a_opt, c5_opt = optimize_manifold_lbfgs(
                    a_cur, layout, [0.02, 0.01, 0.005], min(60.0, (t_end - time.time()) * 0.5)
                )
                if c5_opt < best_c5:
                    h_new = build_sc(a_opt) if layout == 'sc' else build_rc(a_opt)
                    h_new = project_box_sum(h_new, n_best / 2.0)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new

                a_cd, c5_cd = coord_descent_manifold(
                    a_opt, layout, min(40.0, max(10.0, (t_end - time.time()) * 0.3))
                )
                if c5_cd < best_c5:
                    h_new = build_sc(a_cd) if layout == 'sc' else build_rc(a_cd)
                    h_new = project_box_sum(h_new, n_best / 2.0)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new

    # Phase 4: Final pair exchange polish
    if time.time() < t_end:
        remaining = t_end - time.time()
        if remaining > 10:
            best_h, best_c5 = pair_exchange_refine(best_h.copy(), remaining * 0.8, n_max_iter=300000)

    # Final check
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, n_final
