        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(p)
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(p)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - p)
            p[idx] = min(1.0, p[idx] + residual)
        else:
            idx = np.argmax(p)
            p[idx] = max(0.0, p[idx] + residual)
    return p


def lse_score_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = dx * corr
    m = np.max(scaled)
    shifted = beta * (scaled - m)
    exps = np.exp(shifted)
    total = np.sum(exps)
    w = exps / total
    lse_val = m + np.log(total) / beta
    p = np.arange(n)
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ki in range(len(lags)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        k = lags[ki]
        ok1 = (p - k) >= 0
        ok1 &= (p - k) < n
        grad[ok1] += wk * dx * (1.0 - h[p[ok1] - k])
        ok2 = (p + k) >= 0
        ok2 &= (p + k) < n
        grad[ok2] += wk * dx * (-h[p[ok2] + k])
    return lse_val, grad


def lse_descent(h, n, beta_start=10, beta_end=100, iters=300, lr=0.05, time_limit=20.0):
    start = time.perf_counter()
    h = project_feasible(h.astype(float).copy(), n)
    best_h = h.copy()
    best_score = official_score(h)
    current_score = best_score

    for it in range(iters):
        if time.perf_counter() - start > time_limit:
            break
        t = it / max(1, iters - 1)
        beta = beta_start + (beta_end - beta_start) * t

        _, grad = lse_score_and_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = lr
        for _ in range(8):
            h_new = project_feasible(h - step * grad, n)
            s = official_score(h_new)
            if s < current_score:
                break
            step *= 0.5

        h_new = project_feasible(h - step * grad, n)
        s = official_score(h_new)
        if s < current_score - 1e-14:
            current_score = s
            h = h_new
            if s < best_score:
                best_score = s
                best_h = h.copy()
        else:
            # Try opposite direction
            h_new2 = project_feasible(h + step * 0.5 * grad, n)
            s2 = official_score(h_new2)
            if s2 < current_score:
                current_score = s2
                h = h_new2
                if s2 < best_score:
                    best_score = s2
                    best_h = h.copy()

    return best_h, best_score


def pair_sa(h, n, time_limit=15.0, seed=42):
    rng = np.random.default_rng(seed)
    h = project_feasible(h.astype(float).copy(), n)
    best_h = h.copy()
    best_score = official_score(h)
    temp = 0.008
    cooling = 0.97
    start = time.perf_counter()

    while time.perf_counter() - start < time_limit and temp > 0.0005:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(-0.15, 0.15)
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_new = project_feasible(h_try, n)
        s_new = official_score(h_new)
        s_old = official_score(h)
        if s_new < s_old or (s_new - s_old) < temp and rng.random() < np.exp(-(s_new - s_old) / temp):
            h = h_new
            if s_new < best_score:
                best_score = s_new
                best_h = h.copy()
        temp *= cooling

    return best_h, best_score


def interpolate_to(h_parent, n_target):
    m = len(h_parent)
    if m == n_target:
        return project_feasible(h_parent.copy(), n_target)
    xs = (np.arange(m) + 0.5) / m
    xt = (np.arange(n_target) + 0.5) / n_target
    h = np.interp(xt, xs, h_parent)
    return project_feasible(h, n_target)


def gen_structured(n, rng):
    method = rng.integers(0, 5)
    if method == 0:
        t = np.linspace(0, 1, n)
        freq = rng.uniform(1.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    elif method == 1:
        nblocks = rng.integers(2, 7)
        h = np.zeros(n)
        edges = sorted(rng.choice(range(1, n), size=min(nblocks - 1, n - 1), replace=False))
        bounds = [0] + list(edges) + [n]
        for b in range(len(bounds) - 1):
            h[bounds[b]:bounds[b + 1]] = rng.uniform(0.1, 0.9)
    elif method == 2:
        h = np.zeros(n)
        split = n // 2
        h[:split] = 1.0
        h[split:] = 0.0
        h += rng.normal(0, 0.25, n)
    elif method == 3:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(2, 8) * np.linspace(0, 1, n))
        h += rng.normal(0, 0.1, n)
    else:
        h = rng.uniform(0.2, 0.8, n)
    return project_feasible(h, n)


def active_balancing_descent(h, n, max_iters=200, time_limit=15.0, seed=42):
    """Min-norm direction balancing all near-active constraints."""
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    h = project_feasible(h.astype(float).copy(), n)
    best_h = h.copy()
    best_score = official_score(h)
    dx = 2.0 / n
    p = np.arange(n)
    lags = np.arange(-(n - 1), n)

    for it in range(max_iters):
        if time.perf_counter() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = dx * corr
        max_val = np.max(scaled)
        # Select constraints within tolerance of max
        tol = 0.003
        active = np.where(scaled >= max_val - tol)[0]

        if len(active) == 0:
            break

        # Sum gradients of active constraints (equal weight)
        grad = np.zeros(n)
        for ki in active:
            k = lags[ki]
            ok1 = (p - k) >= 0
            ok1 &= (p - k) < n
            grad[ok1] += dx * (1.0 - h[p[ok1] - k])
            ok2 = (p + k) >= 0
            ok2 &= (p + k) < n
            grad[ok2] += dx * (-h[p[ok2] + k])

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        direction = -grad / gnorm

        # Add small random perturbation for diversification
        rnd = rng.normal(0, 0.05, n)
        direction += rnd

        # Line search
        step = 0.05
        improved = False
        for _ in range(6):
            h_new = project_feasible(h + step * direction, n)
            s = official_score(h_new)
            if s < best_score - 1e-14:
                best_score = s
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try a few random pair moves
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = rng.choice([-0.1, -0.05, 0.05, 0.1])
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                h_new = project_feasible(h_try, n)
                s = official_score(h_new)
                if s < best_score - 1e-14:
                    best_score = s
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 4.0

    # Get parent construction
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        pass

    if parent is not None:
        n_parent = len(parent)
        best_h = project_feasible(parent.astype(float).copy(), n_parent)
    else:
        n_parent = 65
        best_h = project_feasible(np.full(n_parent, 0.5), n_parent)

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    # Phase 1: LSE descent on parent at native resolution
    remaining = deadline - time.perf_counter()
    t1 = min(30.0, remaining * 0.2)
    h_opt, s_opt = lse_descent(best_h, best_n, beta_start=10, beta_end=80, iters=200, lr=0.05, time_limit=t1)
    if s_opt < best_score:
        best_score = s_opt
        best_h = h_opt

    # Phase 2: Active balancing descent
    if time.perf_counter() < deadline - 30:
        remaining = deadline - time.perf_counter()
        t2 = min(20.0, remaining * 0.15)
        h_opt, s_opt = active_balancing_descent(best_h, best_n, max_iters=200, time_limit=t2, seed=seed)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt

    # Phase 3: LSE at different resolutions
    n_list = sorted(set([best_n - 3, best_n, best_n + 3, best_n + 7, best_n + 12]))
    n_list = [n for n in n_list if n >= 20]
    for n_try in n_list:
        if time.perf_counter() > deadline - 50:
            break
        h_init = interpolate_to(best_h, n_try)
        remaining = deadline - time.perf_counter()
        t_lim = min(18.0, max(5.0, remaining * 0.12))
        h_opt, s_opt = lse_descent(h_init, n_try, beta_start=8, beta_end=120, iters=250, lr=0.04, time_limit=t_lim)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt
            best_n = n_try
        # Follow with active balancing
        if time.perf_counter() < deadline - 30:
            h_opt2, s_opt2 = active_balancing_descent(h_opt, n_try, max_iters=100, time_limit=min(8.0, deadline - time.perf_counter() - 25), seed=seed + n_try)
            if s_opt2 < best_score:
                best_score = s_opt2
                best_h = h_opt2
                best_n = n_try

    # Phase 4: Simulated annealing
    if time.perf_counter() < deadline - 25:
        remaining = deadline - time.perf_counter()
        t3 = min(25.0, remaining * 0.2)
        h_sa, s_sa = pair_sa(best_h, best_n, time_limit=t3, seed=seed + 200)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa

    # Phase 5: Multiple structured initializations + LSE
    for n_try in [best_n, best_n + 5]:
        if time.perf_counter() > deadline - 60:
            break
        for trial in range(4):
            if time.perf_counter() > deadline - 20:
                break
            h_init = gen_structured(n_try, rng)
            remaining = deadline - time.perf_counter()
            t_lim = min(12.0, max(4.0, remaining * 0.1))
            h_opt, s_opt = lse_descent(h_init, n_try, beta_start=5, beta_end=150, iters=200, lr=0.06, time_limit=t_lim)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_opt
                best_n = n_try

    # Phase 6: High-beta LSE polish
    for _ in range(3):
        if time.perf_counter() > deadline - 20:
            break
        remaining = deadline - time.perf_counter()
        t_lim = min(12.0, max(4.0, remaining * 0.2))
        h_opt, s_opt = lse_descent(best_h, best_n, beta_start=30, beta_end=250, iters=300, lr=0.02, time_limit=t_lim)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt

    # Phase 7: Final active balancing polish
    if time.perf_counter() < deadline - 8:
        h_opt, s_opt = active_balancing_descent(best_h, best_n, max_iters=150, time_limit=min(10.0, deadline - time.perf_counter() - 5), seed=seed + 999)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt

    # Phase 8: One more SA pass
    if time.perf_counter() < deadline - 5:
        h_sa, s_sa = pair_sa(best_h, best_n, time_limit=min(5.0, deadline - time.perf_counter() - 3), seed=seed + 777)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_score = official_score(best_h)
    c5_bound = float(final_score)
    if c5_bound < 0:
        c5_bound = 0.0

    return (best_h, c5_bound, int(len(best_h)))
