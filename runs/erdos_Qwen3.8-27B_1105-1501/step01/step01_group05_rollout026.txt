    n = len(h)
    L = len(c_scaled)
    max_val = np.max(c_scaled)
    
    # Active set: correlations near the maximum
    active = c_scaled >= max_val - active_tol
    
    # LP: maximize eps
    # subject to: G[k,:] @ d <= -eps for active k
    #             G[k,:] @ d <= (max_val - c_scaled[k]) + margin for inactive k
    #             sum(d) = 0
    #             -1 <= d[j] <= 1
    #             eps <= 1 (bound on eps)

    # Objective: -eps (minimize)

    c_obj[-1] = -1.0  # minimize -eps = maximize eps
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For active k: G[k,:] @ d + eps <= 0  =>  G[k,:] @ d + [0..0, 1] @ x <= 0
    # For inactive k: G[k,:] @ d <= (max_val - c_scaled[k]) + margin

        if active[k]:
            A_ub[k, n] = 1.0  # eps term
            b_ub[k] = 0.0
        else:
            b_ub[k] = (max_val - c_scaled[k]) + margin

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

def binary_sa(h_init, n, time_limit, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.default_rng(seed)
    # Initialize binary sequence with exactly n//2 ones
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    
    # Initialize from h_init if given
    if h_init is not None and len(h_init) == n:
        sorted_idx = np.argsort(h_init)[::-1]
        h = np.zeros(n)
        h[sorted_idx[:n//2]] = 1.0
    
    best_h = h.copy()
    best_s = official_score(h)
    cur_s = best_s
    cur_h = h.copy()
    
    t0 = time.perf_counter()
    T = 0.1
    T_min = 0.001
    cooling = 0.995
    
    while time.perf_counter() - t0 < time_limit:
        if T < T_min:
            break
        # Pick a random position with 1 and a position with 0
        ones_pos = np.where(cur_h == 1.0)[0]
        zeros_pos = np.where(cur_h == 0.0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        i = int(rng.choice(ones_pos))
        j = int(rng.choice(zeros_pos))
        
        # Swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_s = official_score(cur_h)
        
        delta = new_s - cur_s
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_s

def slsqp_polish(h, time_limit=15.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1
    
    def objective(x):
        return x[n]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr
    
    def ineq_jac(x):
        h_x = x[:n]
        G = make_grad_matrix(h_x)  # already includes dx
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        j[:, :n] = -G
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5
    
    x0 = np.concatenate([h, [theta0]])
    
    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 60, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass
    
    return h, initial_score

def interpolate_samples(parent, n_target, offset=0):
    m = len(parent)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_target) + 0.5 + offset) % n_target) * 2.0 / n_target
    xt = np.sort(xt)
    h = np.interp(xt, xs, parent, left=parent[0], right=parent[-1])
    return project_to_target(h, n_target)

def gen_sine(n, rng):
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.integers(2, 8)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.03, n)
    return project_to_target(h, n)

def gen_step(n, rng, n_steps):
    if n_steps >= n:
        return project_to_target(rng.uniform(0, 1, n), n)
    bounds = sorted(rng.choice(n - 1, size=n_steps - 1, replace=False).tolist())
    bounds = [0] + bounds + [n]
    vals = rng.uniform(0, 1, n_steps)
    h = np.zeros(n)
    for i in range(n_steps):
        h[bounds[i]:bounds[i+1]] = vals[i]
    return project_to_target(h, n)

- The gradient condition: for any feasible direction d (i.e., sum(d) = 0, h* + εd ∈ [0,1]^n for small ε > 0):
  max_{k ∈ A} ∇f_k(h*) · d ≥ 0

This means: at the optimum, you can't find a direction that simultaneously decreases all active correlations.

The LP I'm solving is exactly checking this condition:
maximize t
subject to: ∇f_k(h*) · d ≤ -t for k ∈ A (active)
           ∇f_k(h*) · d ≤ f_max - f_k(h*) for k ∉ A (inactive, can't exceed max)
           sum(d) = 0
           |d_j| ≤ 1

If the optimal t > 0, then d is a descent direction (it decreases all active correlations by at least t).
If the optimal t = 0, we're at a (first-order) local optimum of the LP relaxation.

This is correct! The LP gives a certificate of local optimality when t = 0 (for the LP relaxation).

However, the original problem is non-convex (the f_k are bilinear in h), so the LP gives only a first-order condition. We might still improve with a proper line search.

def official_score(h):

def project_to_target(v, n=None):

    h = np.clip(np.asarray(v, dtype=float), 0.0, 1.0).copy()

    for _ in range(100):

        if s < target:
            lo = mid
        else:
            hi = mid

    for _ in range(500):

        if abs(residual) < 1e-15:
            break
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + residual)

            h[idx] = max(0.0, h[idx] + residual)
    return h

def make_grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for k in range(L):
        s = k - (n - 1)
        if s >= 0:
            G[k, s:] += (1.0 - h[:n - s])
            G[k, :n - s] += (-h[s:])
        else:
            a = -s
            G[k, :n - a] += (1.0 - h[a:])
            G[k, a:] += (-h[:n - a])
    G *= dx
    return G

def lp_direction(h, active_count=8):
    """Find descent direction using LP. Uses top active_count correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(corr)
    # Get indices of top correlations
    sorted_idx = np.argsort(corr)[::-1]
    active_idx = sorted_idx[:active_count]
    G = make_grad_matrix(h)
    L = len(corr)
    
    # LP: max t s.t. G[k,:]d + t <= 0 for active k
    #                      G[k,:]d <= max_val - corr[k] for inactive k
    #                      sum(d) = 0, |d_j| <= 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -t
    
    A_ub = np.zeros((L, n + 1))
    b_ub = np.zeros(L)
    active_set = set(active_idx.tolist())
    for k in range(L):
        A_ub[k, :n] = G[k, :]
        if k in active_set:
            A_ub[k, n] = 1.0
            b_ub[k] = 0.0
        else:
            b_ub[k] = max_val - corr[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * n + [(0.0, 0.05)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            d = res.x[:n].copy()
            mn = np.max(np.abs(d))
            if mn > 1e-12:
                d /= mn
            return d
    except Exception:
        pass
    return None
