def project_box_sum(x, target):
    """Project x onto {0<=x_i<=1, sum(x)=target}."""
    x = np.clip(np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0), 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - result)[::-1]
        else:
            order = np.argsort(result)[::-1]
        for i in order:
            if residual > 0:
                amt = min(1.0 - result[i], residual)
            else:
                amt = -min(result[i], -residual)
            result[i] = result[i] + amt
            residual -= amt
            if abs(residual) < 1e-14:
                break
    return result

    corr = np.correlate(h, 1.0 - h, mode="full")

    """Compute smooth max objective and gradient for manifold variable a."""
    n = 2 * m
    if layout == 'sc':
        h = build_sc(a, m)
    else:
        h = build_rc(a, m)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    M = np.max(F)
    shifted = (F - M) / temperature
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    sum_exp = np.sum(exp_s)
    if sum_exp < 1e-300:
        return M, np.zeros(m)
    obj = M + temperature * (max_s + np.log(sum_exp))
    weights = exp_s / sum_exp
    
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-16:
            continue
        s = k - (n - 1)
        # (1-h[i-s]) for valid i-s
        i_lo = max(0, s)
        i_hi = min(n - 1, s + n - 1)
        if i_lo <= i_hi:
            grad_h[i_lo:i_hi+1] += w * (1.0 - h[i_lo-s:i_hi-s+1])

    else:

                h_map = build_sc(a, m) if layout == 'sc' else build_rc(a, m)
                h_map = project_box_sum(h_map, m)
                best_c5 = compute_c5(h_map)
                best_h = h_map.copy()
                no_improve = 0
            elif cur_c5 < prev_c5 - 1e-11:
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 12:
                a = np.clip(a + rng.randn(m) * 0.05, 0.0, 1.0)
                no_improve = 0
            if no_improve > 40:
                break

    # Phase 3: Pair exchange on best_h (~15% budget)
    ph3_deadline = t0 + budget_s * 0.88
    if time.time() < ph3_deadline and best_h is not None:
        h_work = best_h.copy()
        remaining = ph3_deadline - time.time()
        max_evals = int(remaining * 4000)
        h_work, _ = pair_exchange_refine(h_work, rng, max_evals=max_evals, deadline=ph3_deadline)
        h_work = project_box_sum(h_work, len(h_work) / 2.0)
        c5_pe = compute_c5(h_work)
        if c5_pe < best_c5:
            best_c5 = c5_pe
            best_h = h_work.copy()

    # Phase 4: Try other n values
    if time.time() < deadline_hard:
        for n_try in [60, 80, 66, 74, 64, 76]:
            if time.time() > deadline_hard:
                break
            m_try = n_try // 2
            h_res = np.interp(np.linspace(0, n_parent - 1, n_try),
                              np.arange(n_parent), parent)
            h_res = project_box_sum(h_res, m_try)
            a_try = transfer_sc(h_res, m_try)
            a_try, c1 = coord_sweep_sc(a_try, m_try, n_grid=40)
            a_try, c2 = coord_sweep_sc(a_try, m_try, n_grid=80)
            a_try, c3 = coord_sweep_sc(a_try, m_try, n_grid=120)
            if c3 < best_c5:
                h_new = build_sc(a_try, m_try)
                h_new = project_box_sum(h_new, m_try)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Final pair exchange with remaining time
    remaining = deadline_hard - time.time()
    if remaining > 3 and best_h is not None:
        h_work = best_h.copy()
        max_evals = int(remaining * 3000)
        h_work, _ = pair_exchange_refine(h_work, rng, max_evals=max_evals,
                                         deadline=deadline_hard)
        h_work = project_box_sum(h_work, len(h_work) / 2.0)
        c5_fin = compute_c5(h_work)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_work.copy()

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
