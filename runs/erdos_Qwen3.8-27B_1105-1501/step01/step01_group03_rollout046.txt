To minimize F[0], we want to maximize sum h[i]^2, which means making h as "extreme" (close to 0 or 1) as possible. For binary h, F[0] = 0.

For small k (like k=1), F[1] = sum_{i=0}^{n-2} h[i](1-h[i+1]). For binary h with n/2 ones, this counts the number of "01" or "10" transitions... wait no. h[i](1-h[i+1]) for binary is 1 when h[i]=1 and h[i+1]=0. So F[1] counts the number of "10" transitions in the binary sequence.

For a binary sequence with n/2 ones and n/2 zeros, the number of "10" transitions equals the number of "01" transitions (they must be equal for a circular sequence, and for a linear sequence they differ by at most 1). The number of transitions is at most min(n, 2*(n/2)) = n, and at least... well, it depends on the arrangement.

If all 1s are together and all 0s are together: F[1] = 1 (one "10" transition).
If they alternate: F[1] = n/2 (for even n, exactly n/2 "10" transitions if starting with 1).

So for F[1], we want to GROUP the 1s together. But for F[0], we want them extreme (which they already are for binary).

For larger k, the structure becomes more complex. The optimal binary arrangement is a trade-off.

This analysis confirms that the optimal h should be close to binary (extreme values), which supports the binary SA approach.

import numpy as np
from scipy.optimize import minimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        for i in range(n):
            if abs(err) < 1e-14:
                break
            if err > 0 and h[i] < 1.0:
                d = min(err, 1.0 - h[i])
                h[i] += d
                err -= d
            elif err < 0 and h[i] > 0.0:
                d = min(-err, h[i])
                h[i] -= d
                err += d
    return np.clip(h, 0.0, 1.0)


def fast_feasible(z):
    n = len(z)
    target = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        if err > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                adj = min(err / cnt, 1.0 - h[mask])
                h[mask] = np.minimum(h[mask] + err / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + err / cnt, 0.0)
    err = target - np.sum(h)
    if abs(err) > 1e-10:
        free = (h > 0.01) & (h < 0.99)
        if np.sum(free) > 0:
            h[free] += err / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F


def grad_F_k(h, k, n):
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        grad[start:end] += (1.0 - h[start + k:end + k])
    p_start = max(0, start + k)
    p_end = min(n, end + k)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - k:p_end - k])
    return grad


def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta


def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = compute_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = es / ws
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if abs(w) < 1e-14:
            continue
        grad += w * grad_F_k(h, k, n)
    return grad * dx

def slsqp_polish(h0, n, beta, maxiter=300):
    h0 = make_feasible(h0.copy())
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return float(np.sum(x) - n / 2.0)

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(lse_obj, h0, args=(beta,), jac=lse_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0


def interpolate(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt)

            break
    return make_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = 15.0
    deadline = t_start + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if not np.all(np.isfinite(parent)) or len(parent) < 2:
            parent = None
    except (NameError, AttributeError, TypeError):
        parent = None

    if parent is not None:
        best_h = make_feasible(parent)
        best_score = official_score(best_h)
        base_n = len(parent)
    else:
        base_n = 61
        best_h = make_feasible(np.full(base_n, 0.5))
        best_score = official_score(best_h)

    def try_update(h_cand):
        nonlocal best_h, best_score
        if h_cand is None:
            return
        h_cand = make_feasible(h_cand)
        if not np.all(np.isfinite(h_cand)):
            return
        sc = official_score(h_cand)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_h = h_cand.copy()
            best_score = sc

    # Determine n values to try
    n_values = []
    for dl in [0, -4, 4, -8, 8, -12, 12, 16, 20, -16]:
        nl = base_n + dl
        if 30 <= nl <= 100 and nl not in n_values:
            n_values.append(nl)
    n_values.sort()

    # Phase 1: Binary SA at a few n values to find good structures
    for n in n_values[:5]:
        if time.time() > deadline - 100:
            break
        h_init = interpolate(best_h, n)
        for sa_seed in [seed, seed + 100, seed + 200]:
            if time.time() > deadline - 80:
                break
            bh, bs = binary_sa(h_init, n, t0=0.3, t1=0.002,
                               n_iter=20000, seed=sa_seed,
                               time_limit=deadline - 60)
            try_update(bh)
        # Also try from uniform
        bh, bs = binary_sa(np.full(n, 0.5) + rng.uniform(-0.2, 0.2, n),
                           n, t0=0.5, t1=0.003, n_iter=25000,
                           seed=seed + 50 + n, time_limit=deadline - 50)
        try_update(bh)

    # Phase 2: PG anneal on best candidates at multiple n
    for n in n_values[:6]:
        if time.time() > deadline - 60:
            break
        h_init = interpolate(best_h, n)
        # Also try from binary SA result
        bh_bin, _ = binary_sa(h_init, n, t0=0.3, t1=0.003,
                              n_iter=10000, seed=seed + n,
                              time_limit=deadline - 40)
        # Use whichever is better as start
        sc_init = official_score(h_init)
        sc_bin = official_score(bh_bin)
        if sc_bin < sc_init:
            h_init = bh_bin

        beta_sched = [10, 40, 150, 500, 1500, 4000]
        ips = 300
        ph, ps = pg_anneal(h_init, n, beta_sched, ips,
                           momentum=0.85, time_limit=deadline - 40)
        try_update(ph)

    # Phase 3: SLSQP polish on best solution
    if time.time() < deadline - 30:
        n_best = len(best_h)
        for beta in [500, 2000, 8000]:
            if time.time() > deadline - 15:
                break
            polished = slsqp_polish(best_h, n_best, beta, maxiter=200)
            try_update(polished)

    # Phase 4: Fine PG with high beta
    if time.time() < deadline - 20:
        n_best = len(best_h)
        beta_sched_fine = [2000, 5000, 12000]
        ph, ps = pg_anneal(best_h, n_best, beta_sched_fine,
                           iters_per_beta=200, momentum=0.9,
                           time_limit=deadline - 10)
        try_update(ph)

    # Phase 5: Try slightly different n values with full pipeline
    if time.time() < deadline - 40:
        n_best = len(best_h)
        for dn in [2, -2, 4, -4]:
            if time.time() > deadline - 25:
                break
            alt_n = n_best + dn
            if alt_n < 30 or alt_n > 100:
                continue
            alt_h = interpolate(best_h, alt_n)
            beta_sched = [50, 200, 1000, 3000]
            ph, ps = pg_anneal(alt_h, alt_n, beta_sched,
                               iters_per_beta=150, momentum=0.87,
                               time_limit=deadline - 15)
            try_update(ph)
            if official_score(ph) < best_score:
                polished = slsqp_polish(ph, alt_n, 5000, maxiter=150)
                try_update(polished)

    # Phase 6: Final SLSQP at very high beta
    if time.time() < deadline - 10:
        n_best = len(best_h)
        polished = slsqp_polish(best_h, n_best, 15000, maxiter=300)
        try_update(polished)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score > 0.75:
        if parent is not None:
            best_h = make_feasible(parent)
        else:
            best_h = make_feasible(np.full(base_n, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
