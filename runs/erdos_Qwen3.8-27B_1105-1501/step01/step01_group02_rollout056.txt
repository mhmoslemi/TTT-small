def binary_round(h, target_sum):
    """Round h to binary, maintaining sum = target_sum."""
    n = len(h)
    # Sort by value descending, take top n/2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target_sum]] = 1.0
    return h_bin

def binary_local_search(h_bin, target_sum, time_limit):
    """Local search on binary solutions using all-pairs swap."""
    n = len(h_bin)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    t_start = time.time()
    
    ones = [i for i in range(n) if h_bin[i] == 1]
    zeros = [i for i in range(n) if h_bin[i] == 0]
    
    while time.time() - t_start < time_limit:
        improved = False
        # Shuffle for diversity
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:]:
            if time.time() - t_start > time_limit:
                break
            for j in zeros[:]:
                h_bin[i] = 0
                h_bin[j] = 1
                c5 = compute_c5(h_bin)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_h = h_bin.copy()
                    ones.remove(i)
                    zeros.remove(j)
                    zeros.append(i)
                    ones.append(j)
                    improved = True
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_smooth_max(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))
    exp_vals = np.exp((corr - cmax) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-14:
            continue
        if s >= 0:
            lo_i = 0
            hi_i = n - 1 - s
        else:
            lo_i = -s
            hi_i = n - 1
        cnt = hi_i - lo_i + 1
        if cnt <= 0:
            continue
        idx = np.arange(lo_i, hi_i + 1)
        idx_s = idx + s
        grad[idx] += w * (1.0 - h[idx_s])
        grad[idx_s] -= w * h[idx]
    return grad
