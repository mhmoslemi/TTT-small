def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    return h


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def compute_corr_and_grads(h, n, dx, top_m=20):

    lags = np.arange(-(n - 1), n)
    
    # Find top M active lags
    sorted_idx = np.argsort(-corr_vals)

            rp = r + q
            grad[rp] += dx * (1.0 - h[r])
            grad[r] += -dx * h[rp]
    
    return corr_vals, max_val, grad

            rp = r + q
            rpi = (rp == i)
            rpj = (rp == j)
            ri = (r == i)
            rj = (r == j)
            
            if rpi and rj:
                # x*(1-(S-x)) = x(1-S+x)
                b += (1.0 - S)
                c += 1.0
            elif rpj and ri:
                # (S-x)*(1-x) = S - (S+1)x + x^2
                a += S
                b += -(S + 1.0)
                c += 1.0
            elif rpi and ri:
                # x*(1-x)
                b += 1.0
                c += -1.0
            elif rpj and rj:
                # (S-x)*(1-(S-x))
                a += S * (1.0 - S)
                b += (2.0 * S - 1.0)
                c += -1.0
            elif rpi:
                b += (1.0 - h[r])
            elif rpj:
                a += S * (1.0 - h[r])
                b += -(1.0 - h[r])
            elif ri:
                b += -h[rp]
            elif rj:
                a += h[rp] * (1.0 - S)
                b += h[rp]
            else:
                a += h[rp] * (1.0 - h[r])
        
        a_arr[idx] = dx * a
        b_arr[idx] = dx * b
        c_arr[idx] = dx * c
    
    # Compute non-affected max
    affected_set_lookup = set(affected_qs)
    all_lags = np.arange(-(n-1), n)
    h_test = h.copy()
    h_test[i] = L
    h_test[j] = S - L
    corr_full = np.correlate(h_test, 1.0 - h_test, mode="full") * dx
    non_aff_positions = [q + n - 1 for q in all_lags if q not in affected_set_lookup]
    if non_aff_positions:
        non_aff_max = np.max(corr_full[non_aff_positions])
    else:
        non_aff_max = 0.0
    
    # Find all candidate x values: endpoints, pairwise crossings, individual critical points
    candidates = [L, U]
    
    # Pairwise crossings
    if n_aff > 1:
        P, Q = np.triu_indices(n_aff, 1)
        ca = c_arr[P] - c_arr[Q]
        cb = b_arr[P] - b_arr[Q]
        cc = a_arr[P] - a_arr[Q]
        
        lin_mask = np.abs(ca) < 1e-14
        if np.any(lin_mask):
            cb_l = cb[lin_mask]
            cc_l = cc[lin_mask]
            nonzero = np.abs(cb_l) > 1e-14
            if np.any(nonzero):
                x_l = -cc_l[nonzero] / cb_l[nonzero]
                valid = (x_l >= L - 1e-10) & (x_l <= U + 1e-10)
                candidates.extend(x_l[valid].clip(L, U).tolist())
        
        quad_mask = ~lin_mask
        if np.any(quad_mask):
            ca_q = ca[quad_mask]
            cb_q = cb[quad_mask]
            cc_q = cc[quad_mask]
            disc = cb_q**2 - 4.0 * ca_q * cc_q
            valid_disc = disc >= -1e-14
            if np.any(valid_disc):
                disc_v = np.maximum(disc[valid_disc], 0.0)
                sqrt_d = np.sqrt(disc_v)
                ca_v = ca_q[valid_disc]
                cb_v = cb_q[valid_disc]
                x1 = (-cb_v + sqrt_d) / (2.0 * ca_v)
                x2 = (-cb_v - sqrt_d) / (2.0 * ca_v)
                for xa in [x1, x2]:
                    valid_x = (xa >= L - 1e-10) & (xa <= U + 1e-10) & np.isfinite(xa)
                    candidates.extend(xa[valid_x].clip(L, U).tolist())
    
    # Critical points of individual quadratics
    nonlin = np.abs(c_arr) > 1e-14
    if np.any(nonlin):
        x_crit = -b_arr[nonlin] / (2.0 * c_arr[nonlin])
        valid = (x_crit >= L - 1e-10) & (x_crit <= U + 1e-10) & np.isfinite(x_crit)
        candidates.extend(x_crit[valid].clip(L, U).tolist())
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= L - 1e-12) & (candidates <= U + 1e-12)]
    candidates = np.clip(candidates, L, U)
    if len(candidates) > 1:
        candidates = np.unique(np.round(candidates, 14))
    
    # Evaluate f(x) at all candidates
    if len(candidates) == 0:
        return None
    
    # Vectorized evaluation
    vals = a_arr[None, :] + b_arr[None, :] * candidates[:, None] + c_arr[None, :] * candidates[:, None] ** 2
    f_vals = np.maximum(np.max(vals, axis=1), non_aff_max)
    
    best_idx = np.argmin(f_vals)
    best_x = candidates[best_idx]
    best_f = f_vals[best_idx]
    
    # Check if this is an improvement over current
    cur_f = official_score(h)
    if best_f >= cur_f - 1e-14:
        return None
    
    new_h = h.copy()
    new_h[i] = best_x
    new_h[j] = S - best_x
    return (new_h, float(best_f))


def pair_optimize(h, n, deadline, rng, max_rounds=200, top_k=25):
    """Iterative pair coordinate descent with exact 1D solves."""
    dx = 2.0 / n
    h = h.copy()
    score = official_score(h)
    no_improve_count = 0
    max_no_improve = 12
    
    for rnd in range(max_rounds):
        if time.perf_counter() > deadline:
            break
        if no_improve_count >= max_no_improve:
            break
        
        # Compute gradient from active lags
        corr_vals, max_val, grad = compute_corr_and_grads(h, n, dx, top_m=15)
        # Center the gradient (remove mean for sum constraint)
        grad_centered = grad - np.mean(grad)
        
        # Score all pairs by |grad[i] - grad[j]|
        # For efficiency, only consider top pairs
        n_scores = n
        if n_scores > 120:
            # Subsample
            idx_pool = np.arange(n_scores)
            selected = rng.choice(idx_pool, size=120, replace=False)
        else:
            selected = np.arange(n_scores)
        
        # Compute pair scores
        pair_scores = np.abs(grad_centered[None, selected] - grad_centered[selected, None])
        np.fill_diagonal(pair_scores, -1)
        # Get top K pairs
        flat_scores = pair_scores.ravel()
        flat_idx = np.argpartition(flat_scores, -top_k)[-top_k:]
        
        best_improvement = None
        best_new_h = None
        best_new_score = score
        
        for fi in flat_idx:
            if time.perf_counter() > deadline:
                break
            pi = fi // n_scores
            pj = fi % n_scores
            if pi == pj:
                continue
            result = solve_pair_exact(h, pi, pj, n, dx)
            if result is not None:
                new_h, new_score = result
                if new_score < best_new_score - 1e-14:
                    best_new_score = new_score
                    best_new_h = new_h
        
        # Also try a few random pairs
        for _ in range(5):
            if time.perf_counter() > deadline:
                break
            pi = int(rng.integers(0, n))
            pj = int(rng.integers(0, n))
            if pi == pj:
                continue
            result = solve_pair_exact(h, pi, pj, n, dx)
            if result is not None:
                new_h, new_score = result
                if new_score < best_new_score - 1e-14:
                    best_new_score = new_score
                    best_new_h = new_h
        
        if best_new_h is not None:
            h = best_new_h
            score = best_new_score
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return h, official_score(h)


def smooth_descent_step(h, n, step=0.05, seed=0):
    """One step of smooth gradient descent with projection."""
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    corr_vals, max_val, grad = compute_corr_and_grads(h, n, dx, top_m=20)
    direction = -grad
    direction -= np.mean(direction)
    norm = np.linalg.norm(direction)
    if norm < 1e-12:
        return h
    direction /= norm
    h_new = repair_vector(h + step * direction, n)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 5.0
    
    # Get parent
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = repair_vector(pv, len(pv))
    except (NameError, Exception):
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        best_h = parent.copy()
    else:
        n_parent = 64
        best_h = repair_vector(np.full(n_parent, 0.5), n_parent)
    
    best_score = official_score(best_h)
    best_n = len(best_h)
    
    rng = np.random.default_rng(seed)
    
    # Generate initial candidates
    candidates = [(best_score, best_h.copy(), best_n)]
    
    # Symmetries
    for sym in [-1]:
        candidates.append((official_score(best_h[sym]), best_h[sym].copy(), best_n))
    comp = 1.0 - best_h
    s = official_score(comp)
    candidates.append((s, comp, best_n))
    comp_rev = 1.0 - best_h[::-1]
    s = official_score(comp_rev)
    candidates.append((s, comp_rev, best_n))
    
    # Interpolations to different sizes
    n_options = sorted(set([n_parent, n_parent - 5, n_parent + 5, n_parent + 10,
                             max(32, n_parent - 10), n_parent + 15]))
    n_options = [n for n in n_options if 32 <= n <= 120]
    
    for n_try in n_options:
        if time.perf_counter() > deadline - 60:
            break
        for off in [0, 1, 2]:
            h_interp = gen_interpolated(parent, n_try, offset=off) if parent is not None else repair_vector(np.full(n_try, 0.5), n_try)
            s = official_score(h_interp)
            candidates.append((s, h_interp, n_try))
    
    # Random candidates
    for n_try in n_options[:4]:
        h_rand = rng.uniform(0.2, 0.8, n_try)
        h_rand = repair_vector(h_rand, n_try)
        s = official_score(h_rand)
        candidates.append((s, h_rand, n_try))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    
    # Update global best if any candidate is better
    for s, h_c, n_c in candidates[:5]:
        if s < best_score - 1e-12:
            best_score = s
            best_h = h_c.copy()
            best_n = n_c
    
    # Main optimization loop: try pair optimization on top candidates
    processed = 0
    max_candidates_to_try = 25
    
    for idx in range(min(max_candidates_to_try, len(candidates))):
        if time.perf_counter() > deadline - 40:
            break
        
        s_c, h_c, n_c = candidates[idx]
        if s_c > best_score + 0.01:
            break  # Too far from best, skip
        
        processed += 1
        remaining = deadline - time.perf_counter()
        local_deadline = start_time + min(budget_s - 10, start_time + processed * (budget_s * 0.8 / max_candidates_to_try))
        local_deadline = min(local_deadline, deadline - 20)
        
        # Run pair optimization
        h_opt, s_opt = pair_optimize(h_c, n_c, local_deadline, rng, max_rounds=150, top_k=20)
        
        if s_opt < best_score - 1e-12:
            best_score = s_opt
            best_h = h_opt.copy()
            best_n = n_c
        
        # After improving, also try smooth descent
        if time.perf_counter() < deadline - 25:
            for step_sz in [0.03, 0.08, 0.15]:
                h_smooth = smooth_descent_step(h_opt, n_c, step=step_sz, seed=seed + idx * 10 + int(step_sz * 100))
                s_sm = official_score(h_smooth)
                if s_sm < best_score - 1e-12:
                    best_score = s_sm
                    best_h = h_smooth.copy()
                    best_n = n_c
                # Re-optimize the smoothed version
                if time.perf_counter() < deadline - 15:
                    h_opt2, s_opt2 = pair_optimize(h_smooth, n_c, deadline - 15, rng, max_rounds=50, top_k=15)
                    if s_opt2 < best_score - 1e-12:
                        best_score = s_opt2
                        best_h = h_opt2.copy()
                        best_n = n_c
    
    # Second round: interpolate best to new sizes and re-optimize
    if time.perf_counter() < deadline - 30:
        for n_try in [best_n, best_n + 5, best_n - 5]:
            if n_try < 32 or n_try > 120:
                continue
            if time.perf_counter() > deadline - 20:
                break
            h_interp = gen_interpolated(best_h, n_try)
            s_int = official_score(h_interp)
            if s_int < best_score + 0.02:
                h_opt, s_opt = pair_optimize(h_interp, n_try, deadline - 15, rng, max_rounds=80, top_k=20)
                if s_opt < best_score - 1e-12:
                    best_score = s_opt
                    best_h = h_opt.copy()
                    best_n = n_try
    
    # Third round: multiple random restarts from best
    if time.perf_counter() < deadline - 30:
        for restart in range(5):
            if time.perf_counter() > deadline - 20:
                break
            noise = rng.normal(0, 0.03, best_n)
            h_pert = repair_vector(best_h + noise, best_n)
            s_pert = official_score(h_pert)
            h_opt, s_opt = pair_optimize(h_pert, best_n, deadline - 18, rng, max_rounds=60, top_k=18)
            if s_opt < best_score - 1e-12:
                best_score = s_opt
                best_h = h_opt.copy()
                best_n = best_n
    
    # Final SLSQP polish
    if time.perf_counter() < deadline - 10:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        target_sum = n_fin / 2.0
        lags_fin = np.arange(-(n_fin - 1), n_fin)
        corr_init = np.correlate(best_h, 1.0 - best_h, mode="full") * dx_fin
        active_m = min(15, 2 * n_fin - 1)
        top_idx = np.argsort(-corr_init)[:active_m]
        top_lags = lags_fin[top_idx]
        
        def slsqp_obj(x):
            return x[n_fin]
        
        def slsqp_obj_grad(x):
            g = np.zeros(n_fin + 1)
            g[n_fin] = 1.0
            return g
        
        def slsqp_ineq(x):
            hx = x[:n_fin]
            return x[n_fin] - np.correlate(hx, 1.0 - hx, mode="full") * dx_fin
        
        def slsqp_ineq_jac(x):
            hx = x[:n_fin]
            n_lags = 2 * n_fin - 1
            j = np.zeros((n_lags, n_fin + 1))
            j[:, n_fin] = 1.0
            for li, lq in enumerate(lags_fin):
                r_s = max(0, -lq)
                r_e = min(n_fin, n_fin - lq)
                for r in range(r_s, r_e):
                    rp = r + lq
                    j[li, rp] += -dx_fin * (1.0 - hx[r])
                    j[li, r] += -dx_fin * hx[rp]
            return j
        
        def slsqp_eq(x):
            return np.sum(x[:n_fin]) - target_sum
        
        def slsqp_eq_jac(x):
            g = np.zeros(n_fin + 1)
            g[:n_fin] = 1.0
            return g
        
        theta0 = official_score(best_h) + 1e-4
        x0 = np.concatenate([best_h, [theta0]])
        bounds = [(0.0, 1.0)] * n_fin + [(0.0, 1.0)]
        cons = [
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac}
        ]
        
        try:
            result = optimize.minimize(
                slsqp_obj, x0, jac=slsqp_obj_grad,
                method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if result.x is not None and np.all(np.isfinite(result.x[:n_fin])):
                h_ref = repair_vector(result.x[:n_fin], n_fin)
                s_ref = official_score(h_ref)
                if s_ref < best_score - 1e-12:
                    best_score = s_ref
                    best_h = h_ref.copy()
                    best_n = n_fin
        except Exception:
            pass
    
    # Final pair polish
    if time.perf_counter() < deadline - 5:
        h_polish, s_polish = pair_optimize(best_h, best_n, deadline - 3, rng, max_rounds=30, top_k=15)
        if s_polish < best_score - 1e-12:
            best_score = s_polish
            best_h = h_polish.copy()
            best_n = best_n
    
    # Final validation and return
    best_h = repair_vector(best_h, len(best_h))
    final_score = official_score(best_h)
    
    return (best_h, float(final_score), int(len(best_h)))
