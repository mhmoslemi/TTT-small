def compute_derivatives(h):
    """Compute D[m, k] = d(corr[k])/d(h[m]) for all m, k."""
    n = len(h)
    m_total = 2 * n - 1
    s_vals = np.arange(-(n - 1), n)
    D = np.zeros((n, m_total))
    for m in range(n):

        idx_m = m - s_vals
        mask_p = (idx_p >= 0) & (idx_p < n)
        mask_m = (idx_m >= 0) & (idx_m < n)
        D[m, mask_p] = 1.0 - h[idx_p[mask_p]]
        D[m, mask_m] -= h[idx_m[mask_m]]
    return D


def find_optimal_delta(corr, g, delta_lo, delta_hi, n_bisect=35):
    """Find delta in [delta_lo, delta_hi] minimizing max(corr + delta*g)."""

        return 0.0

        sg_hi = np.max(g[active])

            mx_idx = np.argmax(corr)
            d_col = D[:, mx_idx]
            ki = np.argmax(d_col)
            kj = np.argmin(d_col)
            d_lo = max(-h[kj], h[ki] - 1.0)
            d_hi = min(1.0 - h[ki], h[kj])
            if d_lo < d_hi - 1e-14:
                g = D[ki] - D[kj]
                delta = find_optimal_delta(corr, g, d_lo, d_hi)
                new_c5 = np.max(corr + delta * g) * dx
                c5_now = compute_c5(h)
                if new_c5 < c5_now:
                    h[ki] = np.clip(h[ki] + delta, 0.0, 1.0)
                    h[kj] = np.clip(h[kj] - delta, 0.0, 1.0)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()

    # Phase 3: Fine tuning with more bisection precision (15% budget)
    ph3_end = t0 + budget_s - 3.0
    h = best_h.copy()
    stale = 0

    while time.time() < ph3_end:
        corr = compute_corr(h)
        cur_c5 = np.max(corr) * dx
        D = compute_derivatives(h)
        moved = False

        for _ in range(60):
            if time.time() >= ph3_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_lo = max(-h[j], h[i] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-14:
                continue
            g = D[i] - D[j]
            delta = find_optimal_delta(corr, g, d_lo, d_hi, n_bisect=50)
            new_c5 = np.max(corr + delta * g) * dx
            if new_c5 < cur_c5 - 1e-15:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h[j] = np.clip(h[j] - delta, 0.0, 1.0)
                cur_c5 = new_c5
                moved = True
                stale = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            else:
                stale += 1

        if not moved and stale > 30000:
            h = best_h + rng.randn(n) * 0.005
            h = project_onto_constraint(h, target_sum)
            stale = 0

    # Final feasibility check and return
    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
