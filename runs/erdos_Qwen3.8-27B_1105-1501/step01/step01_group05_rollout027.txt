def official_score(h):

    if n == 0:

def project_to_target(v, n):
    """Project v to [0,1]^n with sum = n/2."""
    h = np.clip(np.asarray(v, dtype=float).ravel()[:n], 0.0, 1.0)
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    target = n / 2.0
    # Bisection on uniform shift
    lo, hi = 1.0 - h.max() - 1, 2.0 - h.min() + 1
    # Actually: after shift λ, h_i + λ clipped to [0,1]
    # We want sum(clamp(h + λ, 0, 1)) = target
    # Try bisection
    for _ in range(60):
        mid = (lo + hi) / 2
        shifted = np.clip(h + mid, 0, 1)
        s = shifted.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fine correction
    for _ in range(200):
        residual = target - h.sum()
        if abs(residual) < 1e-14:

        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])

            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            if delta <= 0: break
            h[idx] -= delta
    return h

import numpy as np
from scipy import optimize
from scipy.optimize import linprog
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_target(v, n):
    h = np.clip(np.asarray(v, dtype=float).ravel()[:n], 0.0, 1.0)
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), constant_values=0.5)
    target = n / 2.0
    lo, hi = 1.0 - h.max() - 1.0, 2.0 - h.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(300):
        residual = target - h.sum()
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            delta = min(residual, room[idx])
            if delta <= 0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-residual, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
    return h


def compute_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")


def grad_row(h, s):
    n = len(h)
    g = np.zeros(n)
    idx = np.arange(n)
    j1 = idx - s
    m1 = (j1 >= 0) & (j1 < n)
    g[m1] += (1.0 - h[j1[m1]])
    j2 = idx + s
    m2 = (j2 >= 0) & (j2 < n)
    g[m2] += (-h[j2[m2]])
    return g


def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    H = np.zeros((L, n))
    for r in range(L):
        s = r - (n - 1)
        H[r] = dx * grad_row(h, s)
    return H


def lp_direction(h, tol=0.01, max_lags=50):
    n = len(h)
    sc = compute_scaled(h)
    theta = sc.max()
    margins = theta - sc
    active = margins <= tol
    n_act = active.sum()
    if n_act == 0:
        return None, 0.0
    if n_act > max_lags:
        order = np.argsort(margins)
        active = np.zeros(len(sc), dtype=bool)
        active[order[:max_lags]] = True
    H = grad_matrix(h)
    L = len(sc)
    c_obj = np.zeros(n + 1)
    c_obj[n] = -1.0
    A_rows = []
    b_rows = []
    eps = 1e-9
    for r in range(L):
        row = np.zeros(n + 1)
        row[:n] = H[r]
        if active[r]:
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(0.0)
        else:
            A_rows.append(row)
            b_rows.append(margins[r] + eps)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-1.0, 1.0)] * n + [(0.0, 5.0)]
    try:

            rho = res.x[n]

def line_search(h, direction, current_score):

    best_h = h.copy()

    if mx < 1e-14:

def slsqp_minimax(h, maxiter=80):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        return x[n] - dx * np.correlate(hx, 1.0 - hx, mode="full")

    def ineq_jac(x):
        hx = x[:n]
        j = np.zeros((L, n + 1))
        j[:, n] = 1.0
        for r in range(L):
            s = r - (n - 1)
            j[r, :n] = -dx * grad_row(hx, s)
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    init_score = official_score(h)
    x0 = np.concatenate([h, [init_score + 1e-5]])
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = optimize.minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = project_to_target(result.x[:n], n)
            s_out = official_score(h_out)
            if s_out < init_score - 1e-11:
                return h_out, s_out
    except Exception:
        pass
    return h, init_score


def interp_to_n(parent, n_t, shift=0):
    m = len(parent)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n_t) + 0.5 + shift) % n_t) * 2.0 / n_t
    order = np.argsort(xt)
    xt_s = xt[order]
    h = np.interp(xt_s, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return project_to_target(h, n_t)
