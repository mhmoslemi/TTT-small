def project_feasible(h, target):
    """Project h to [0,1]^n with sum = target using water-filling."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Water-filling bisection
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_active_gradient(h, n_active=None):
    """Compute weighted gradient of max correlation."""
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(C)
    c_max = np.max(C)
    
    if n_active is None:
        n_active = max(8, min(30, m // 8))
    
    # Get top correlations
    top_idx = np.argsort(C)[::-1][:n_active]
    
    # Compute weights: active ones get high weight, others get exponential decay
    tau = max(1e-4, 0.005 * c_max) if c_max > 0 else 1e-4
    weights = np.exp((C - c_max) / tau)
    weights = weights / weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for p in top_idx:
        L = p - (n - 1)
        w = weights[p] * dx
        if L > 0:
            # C_p = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
            # dC_p/dh[k] = (1-h[k-L]) for k in [L, n-1]
            #              - h[k+L] for k in [0, n-1-L]
            grad[L:] += w * (1.0 - h[:n-L])  # first part: (1-h[k-L]) for k=L..n-1
            grad[:n-L] -= w * h[L:]  # second part: -h[k+L] for k=0..n-1-L
        elif L < 0:
            M = -L
            # C_p = sum_{i=0}^{n-1-M} h[i]*(1-h[i+M])
            # dC_p/dh[k] = (1-h[k+M]) for k in [0, n-1-M]
            #              - h[k-M] for k in [M, n-1]
            grad[:n-M] += w * (1.0 - h[M:])  # (1-h[k+M]) for k=0..n-1-M
            grad[M:] -= w * h[:n-M]  # -h[k-M] for k=M..n-1
        else:
            # L = 0: C_p = sum h[i]*(1-h[i])
            grad += w * (1.0 - 2.0 * h)
    
    return grad, C, weights

def feasible_interval(h, d):
    """Find [t_lo, t_hi] such that 0 <= h + t*d <= 1 for all i."""
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.min(h[neg] / (-d[neg])))
    # Cap the interval
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)
    if t_lo >= t_hi:
        return None
    return t_lo, t_hi

def exact_line_search(h, d, t_lo, t_hi, n_top=15):
    """Exact 1D minimax line search along h + t*d for t in [t_lo, t_hi]."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    
    # Compute quadratic coefficients
    C = np.correlate(h, 1.0 - h, mode='full')  # C_p at t=0
    A = -np.correlate(d, d, mode='full')  # A_p
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')  # B_p
    
    # Scale: the actual C5 is max(C_p) * dx, so we minimize max(C_p)
    # (the dx factor is constant)
    
    # Candidate t values
    candidates = [t_lo, t_hi, 0.0]
    
    # Vertices of parabolas with A_p > 0
    pos_A = A > 1e-14
    if np.any(pos_A):
        verts = -B[pos_A] / (2.0 * A[pos_A])
        for t_v in verts:
            if t_lo < t_v < t_hi:
                candidates.append(float(t_v))
    
    # Pair intersections (among top correlations)
    C_sorted = np.argsort(C)[::-1]
    top_p = C_sorted[:n_top]
    
    for i in range(len(top_p)):
        for j in range(i+1, len(top_p)):
            p, q = top_p[i], top_p[j]
            da = A[p] - A[q]
            db = B[p] - B[q]
            dc = C[p] - C[q]
            if abs(da) < 1e-14:
                # Linear: db*t + dc = 0
                if abs(db) > 1e-14:
                    t_int = -dc / db
                    if t_lo < t_int < t_hi:
                        candidates.append(float(t_int))
            else:
                disc = db*db - 4.0*da*dc
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2.0 * da)
                    t2 = (-db - sqrt_disc) / (2.0 * da)
                    for t_int in [t1, t2]:
                        if t_lo < t_int < t_hi:
                            candidates.append(float(t_int))
    
    # Add some evenly spaced points for robustness
    n_dense = 20
    for i in range(n_dense + 1):
        t_dense = t_lo + (t_hi - t_lo) * i / n_dense
        candidates.append(float(t_dense))
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.max(C)  # value at t=0
    
    for t in candidates:
        if not (t_lo <= t <= t_hi):
            continue
        val = np.max(A * t*t + B * t + C)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val * dx

def make_direction(h, d):
    """Normalize direction and ensure sum is zero."""
    d = d - np.mean(d)  # Ensure sum = 0
    max_abs = np.max(np.abs(d))
    if max_abs < 1e-14:
        return None
    d = d / max_abs
    return d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-13:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_active_gradient(h):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(C))
    n_active = max(10, min(40, len(C) // 5))
    tau = max(2e-4, 0.002 * c_max) if c_max > 0 else 2e-4
    weights = np.exp((C - c_max) / tau)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    top_idx = np.argsort(C)[::-1][:n_active]
    for p in top_idx:
        p = int(p)
        L = p - (n - 1)
        w = weights[p] * dx
        if L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        elif L < 0:
            M = -L
            grad[:n - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n - M]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad, C


def make_direction(d):
    d = d - np.mean(d)
    max_abs = float(np.max(np.abs(d)))
    if max_abs < 1e-14:
        return None
    return d / max_abs


def feasible_interval(h, d):
    t_lo = -3.0
    t_hi = 3.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min(h[neg] / (-d[neg]))))
    if t_lo >= t_hi - 1e-12:
        return None
    return t_lo, t_hi


def exact_line_search(h, d, t_lo, t_hi, n_top=20):
    C = np.correlate(h, 1.0 - h, mode='full')
    A = -np.correlate(d, d, mode='full')
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    candidates = [t_lo, t_hi, 0.0]
    pos_A = A > 1e-12
    if np.any(pos_A):
        verts = -B[pos_A] / (2.0 * A[pos_A])
        for tv in verts:
            tv = float(tv)
            if t_lo < tv < t_hi:
                candidates.append(tv)
    top_p = np.argsort(C)[::-1][:n_top]
    for i in range(len(top_p)):
        for j in range(i + 1, len(top_p)):
            p, q = int(top_p[i]), int(top_p[j])
            da = A[p] - A[q]
            db = B[p] - B[q]
            dc = C[p] - C[q]
            if abs(da) < 1e-12:
                if abs(db) > 1e-12:
                    ti = -dc / db
                    if t_lo < ti < t_hi:
                        candidates.append(float(ti))
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= 0:
                    sd = np.sqrt(disc)
                    t1 = (-db + sd) / (2.0 * da)
                    t2 = (-db - sd) / (2.0 * da)
                    if t_lo < t1 < t_hi:
                        candidates.append(float(t1))
                    if t_lo < t2 < t_hi:
                        candidates.append(float(t2))
    for i in range(21):
        candidates.append(t_lo + (t_hi - t_lo) * i / 20.0)
    best_t = 0.0

        if val < best_val - 1e-14:
            best_val = val
            best_t = float(t)
    return best_t, best_val
