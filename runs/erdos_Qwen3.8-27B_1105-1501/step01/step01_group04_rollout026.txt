    return np.correlate(h, 1.0 - h, mode='full')

def lse_gradient(h, alpha):
    """Gradient of log-sum-exp of correlations (smooth max)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    # gradient of sum_j p[j] * c[j] * dx
    # = dx * (convolve(p, 1-h, 'full')[n-1:2n-1] - correlate(p, h, 'full')[n-1:2n-1][::-1])
    g1 = np.convolve(p, 1.0 - h, mode='full')[n-1:2*n-1]
    g2 = np.correlate(p, h, mode='full')[n-1:2*n-1][::-1]
    return dx * (g1 - g2)

def active_gradient(h, active_indices):
    """Gradient of max over active indices (approximate, equal weights)"""
    n = len(h)
    dx = 2.0 / n
    p = np.zeros(2*n-1)
    p[active_indices] = 1.0 / len(active_indices)
    g1 = np.convolve(p, 1.0 - h, mode='full')[n-1:2*n-1]
    g2 = np.correlate(p, h, mode='full')[n-1:2*n-1][::-1]
    return dx * (g1 - g2)

def project_box_sum(h, target):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target}"""

        else:

    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    
    while time.time() - t0 < time_budget:
        for _ in range(50):
            if time.time() - t0 >= time_budget:
                break
            t += 1
            grad = lse_gradient(h, alpha)
            grad -= np.mean(grad)  # remove mean (tangent to sum constraint)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            h = h_new
        
        if stall > 200:
            # Reset with perturbation
            rng = np.random.RandomState(seed + int(stall))
            h = best_h + 0.03 * rng.randn(n)
            h = project_box_sum(h, target)
            m = np.zeros(n)
            v = np.zeros(n)
            stall = 0
            lr = max(lr * 0.7, 0.005)
        
        alpha = min(alpha * 1.05, 100.0)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    g1 = np.convolve(p, 1.0 - h, mode='full')[n-1:2*n-1]
    g2 = np.correlate(p, h, mode='full')[n-1:2*n-1][::-1]
    return dx * (g1 - g2)

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    i_idx = np.arange(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    diff1 = i_idx[:, None] - lags[None, :]
    diff2 = i_idx[:, None] + lags[None, :]
    valid1 = (diff1 >= 0) & (diff1 < n)
    valid2 = (diff2 >= 0) & (diff2 < n)
    g1 = np.zeros((n, 2 * n - 1))
    g1[valid1] = b[diff1[valid1]]
    g2 = np.zeros((n, 2 * n - 1))
    g2[valid2] = h[diff2[valid2]]
    return dx * (g1 - g2).T

        g[0] = 1.0
        return g

        h = x[1:]

        return J

        return np.array([np.sum(x[1:]) - target])

        J[0, 1:] = 1.0
        return J

def lp_direction(h, tau=1e-4):
    """Find active-set descent direction via LP."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    C_max = np.max(C)
    active = np.where(C >= C_max - tau)[0]
    
    if len(active) == 0:
        return None
    
    G = compute_grad_matrix(h)  # (2n-1, n)
    G_active = G[active, :]  # (|A|, n)
    m_active = len(active)
    
    # LP: min eta
    # s.t. G_active @ delta <= 1 * eta  →  G_active @ delta - eta <= 0
    #      sum(delta) = 0
    #      -rho <= delta_i <= rho
    # Variables: [delta (n), eta (1)]
    rho = 0.1
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize eta
    
    # Inequality: G_active @ delta - eta <= 0
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.zeros((m_active, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_active)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-rho, rho)] * n + [(-1.0, 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[:n]
            eta = res.x[n]
            if eta < -1e-8:
                return delta, eta
    except Exception:
        pass
    
    return None
