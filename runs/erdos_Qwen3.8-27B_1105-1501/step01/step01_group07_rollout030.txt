        g[:] = (1.0 - 2.0 * h) * dx

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_grad(h, T):
    """Compute gradient of smooth maximum of lag values."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(np.max(scores))
    ex = np.exp((scores - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        # Use hard max: equal weight on all active lags
        active = np.where(scores >= base - 1e-10)[0]
        w = np.zeros(2*n-1)
        w[active] = 1.0 / len(active)
    else:
        w = ex / S
    # Compute gradient
    c = n - 1
    g = np.zeros(n)
    for d in range(n):
        idx = c + d
        if idx >= 2*n-1:
            break
        wi = w[idx] * dx
        if d == 0:
            g += wi * (1.0 - 2.0 * h)
        else:

            # Lag +d: f = sum_{i=0}^{m-1} h[i+d](1-h[i])*dx
            # df/dh[j] for j in [d, n-1]: dx*(1-h[j-d])
            # df/dh[j] for j in [0, m-1]: -dx*h[j+d]
            g[d:] += wi * (1.0 - h[:m])
            g[:m] += -wi * h[d:]
        idx2 = c - d
        if idx2 < 0:

        if d == 0:

        # Lag -d: f = sum_{i=d}^{n-1} h[i](1-h[i-d])*dx
        # df/dh[j] for j in [d, n-1]: dx*(1-h[j-d])
        # df/dh[j] for j in [0, m2-1]: -dx*h[j+d]
        g[d:] += wi2 * (1.0 - h[:m2])
        g[:m2] += -wi2 * h[d:]
