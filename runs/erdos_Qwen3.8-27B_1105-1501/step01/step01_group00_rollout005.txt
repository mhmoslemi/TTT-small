    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h):
    return float(exact_vals(h).max())

def grad_row(h, p):
    n = len(h)
    dx = 2.0 / n
    s = p - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        g[:L] -= h[s:]
        g[s:] += (1.0 - h[:L])
    else:
        t = -s
        L = n - t
        g[:L] += (1.0 - h[t:])
        g[t:] -= h[:L]
    return dx * g

    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
