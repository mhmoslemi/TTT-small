Actually, I realize I should also think about what makes the problem hard. The objective max_k C_k is non-smooth (it's a max of smooth functions). The log-sum-exp approximation smooths it, but at high beta, it's very close to the max and the gradient is concentrated on the few lags that achieve the max. This means:
- At high beta, the gradient is effectively the average of gradients of the top few lags
- This is similar to the active-set approach but with a continuous weighting

So the logit + high-beta approach should be very similar to active-set. The difference is:
- Logit + high-beta: smooth, can use L-BFGS-B, but the "active set" changes continuously
- Active-set: exactly targets the top Q lags, can use exact line search

I think combining both is best:
1. Use logit + medium beta (100-1000) for the main optimization (L-BFGS-B)
2. Follow up with active-set + exact line search for final polish

import numpy as np

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - np.max(z) - 1e-6
    hi = 1.0 - np.min(z) + 1e-6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - np.sum(h)
    for _ in range(200):
        if abs(err) < 1e-14:
            break
        if err > 0:
            caps = np.maximum(1.0 - h, 0.0)
            total = np.sum(caps)
            if total < 1e-15:
                break
            h += caps * (err / total)
        else:
            floors = np.maximum(h, 0.0)
            total = np.sum(floors)
            if total < 1e-15:
                break
            h -= floors * (-err / total)
    return np.clip(h, 0.0, 1.0)

def compute_all_F(h, n):
    F = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        start = max(0, -k)
        end = min(n, n - k)
        F[k + n - 1] = np.sum(h[start:end] * (1.0 - h[start + k:end + k]))
    return F

def lag_gradient(h, k, n):
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    if start < end:
        grad[start:end] += (1.0 - h[start + k:end + k])
    p_start = max(0, start + k)
    p_end = min(n, end + k)
    if p_start < p_end:
        grad[p_start:p_end] += (-h[p_start - k:p_end - k])
    return grad

def safe_sigmoid(x):
    out = np.zeros_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[~pos])
    out[~pos] = exp_x / (1.0 + exp_x)
    return out

def safe_logit(h, eps=1e-10):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def logit_to_h(y, n):
    target = n / 2.0
    lo, hi = -80.0, 80.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(safe_sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    c = (lo + hi) * 0.5
    h = safe_sigmoid(y + c)
    err = target - np.sum(h)
    for _ in range(100):
        if abs(err) < 1e-12:
            break
        if err > 0:
            caps = np.maximum(1.0 - h, 0.0)
            total = np.sum(caps)
            if total < 1e-15:
                break
            h += caps * (err / total)
        else:
            floors = np.maximum(h, 0.0)
            total = np.sum(floors)
            if total < 1e-15:
                break
            h -= floors * (-err / total)
    return np.clip(h, 0.0, 1.0)

def _logit_obj(y, n, beta):
    h = logit_to_h(y, n)
    dx = 2.0 / n
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return cmax + np.log(2 * n - 1) / beta
    return cmax + (mk + np.log(ws)) / beta

def _logit_grad(y, n, beta):
    h = logit_to_h(y, n)
    dx = 2.0 / n
    F = compute_all_F(h, n)
    C = F * dx
    cmax = np.max(C)
    shifted = beta * (C - cmax)
    mk = np.max(shifted)
    es = np.exp(shifted - mk)
    ws = np.sum(es)
    if ws < 1e-300:
        return np.zeros(n)
    weights = es / ws
    grad_h = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        grad_h += w * lag_gradient(h, k, n)
    grad_h *= dx
    w_vec = h * (1.0 - h)
    W = np.sum(w_vec)
    if W < 1e-300:
        return np.zeros(n)
    grad_hw = grad_h * w_vec
    sum_ghw = np.sum(grad_hw)
    grad_y = w_vec * (grad_hw - sum_ghw / W)
    return grad_y
